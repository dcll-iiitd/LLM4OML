\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}
\addcontentsline{toc}{section}{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem
\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\; \mathcal{L}(x,y)
:= f(x)+\langle Kx, y\rangle - g^{*}(y),
\tag{1}
\]
where  

\begin{itemize}
\item $f:\mathbb{R}^{n}\rightarrow\mathbb{R}\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth);
\item $g^{*}:\mathbb{R}^{m}\rightarrow\mathbb{R}\cup\{+\infty\}$ is the convex conjugate of a proper convex function $g$;
\item $K\in\mathbb{R}^{m\times n}$ is a linear operator with induced Euclidean norm $\|K\|$.
\end{itemize}

The PDHG iterates are
\[
\boxed{
\begin{aligned}
y_{k+1}&=\operatorname{prox}_{\sigma g^{*}}\!\bigl(y_{k}+\sigma Kx_{k}\bigr), \tag{2a}\\[4pt]
x_{k+1}&=\operatorname{prox}_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr), \tag{2b}\\[4pt]
\bar{x}_{k+1}&=x_{k+1}+\theta\,(x_{k+1}-x_{k}), \tag{2c}
\end{aligned}}
\]
with step sizes $\tau>0$, $\sigma>0$ and extrapolation parameter $\theta\in[0,1]$.  
The proximal operator of a convex function $h$ is defined by
\[
\operatorname{prox}_{\lambda h}(v):=\arg\min_{u}\Bigl\{\frac12\|u-v\|^{2}+\lambda h(u)\Bigr\}.
\tag{3}
\]

In many applications (including the dry‑run example below) we set $g^{*}=0$, $\theta=1$, and $K=1$, which reduces (2a) to a simple gradient ascent step on the dual variable.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\REQUIRE Initial primal $x_{0}\in\mathbb{R}^{n}$, dual $y_{0}\in\mathbb{R}^{m}$,
step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, max iterations $T$.
\STATE $\bar{x}_{0}\leftarrow x_{0}$
\FOR{$k=0$ \TO $T-1$}
    \STATE $y_{k+1}\leftarrow\operatorname{prox}_{\sigma g^{*}}\!\bigl(y_{k}+\sigma Kx_{k}\bigr)$ \COMMENT{dual ascent}
    \STATE $x_{k+1}\leftarrow\operatorname{prox}_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)$ \COMMENT{primal proximal step}
    \STATE $\bar{x}_{k+1}\leftarrow x_{k+1}+\theta\,(x_{k+1}-x_{k})$ \COMMENT{extrapolation}
\ENDFOR
\RETURN $\{(x_{k},y_{k})\}_{k=1}^{T}$, optionally the ergodic averages
$\displaystyle \tilde{x}_{T}=\frac{1}{T}\sum_{k=1}^{T}x_{k},\;
\tilde{y}_{T}=\frac{1}{T}\sum_{k=1}^{T}y_{k}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate three iterations on the scalar problem
\[
\min_{x\in\mathbb{R}} f(x):=(x-3)^{2},
\]
with $K=1$, $g^{*}=0$ (hence $\operatorname{prox}_{\sigma g^{*}}$ is the identity), step sizes
\[
\tau=0.1,\qquad \sigma=0.5,\qquad \theta=1,
\]
and initial point $x_{0}=0$, $y_{0}=0$.

\paragraph{Pre‑computations}
For $f(x)=(x-3)^{2}$ the proximal operator is
\[
\operatorname{prox}_{\tau f}(v)
=\arg\min_{u}\frac12(u-v)^{2}+\tau (u-3)^{2}
=\frac{v+6\tau}{1+2\tau}.
\tag{4}
\]

\paragraph{Iteration $k=0$}
\begin{align*}
y_{1}&=y_{0}+\sigma K x_{0}=0+0.5\cdot0=0,\\[4pt]
x_{1}&=\operatorname{prox}_{\tau f}\!\bigl(x_{0}-\tau K^{\top}y_{1}\bigr)
      =\operatorname{prox}_{0.1 f}(0-0.1\cdot0)
      =\frac{0+6\cdot0.1}{1+2\cdot0.1}
      =\frac{0.6}{1.2}=0.5,\\[4pt]
\bar{x}_{1}&=x_{1}+\theta(x_{1}-x_{0})=0.5+1\cdot(0.5-0)=1.0.
\end{align*}
Objective value: $f(x_{1})=(0.5-3)^{2}=6.25$.

\paragraph{Iteration $k=1$}
\begin{align*}
y_{2}&=y_{1}+\sigma K x_{1}=0+0.5\cdot0.5=0.25,\\[4pt]
x_{2}&=\operatorname{prox}_{0.1 f}\!\bigl(x_{1}-0.1 y_{2}\bigr)
      =\operatorname{prox}_{0.1 f}(0.5-0.025)
      =\operatorname{prox}_{0.1 f}(0.475)\\
      &=\frac{0.475+0.6}{1.2}= \frac{1.075}{1.2}\approx0.8958,\\[4pt]
\bar{x}_{2}&=x_{2}+(\,x_{2}-x_{1}\,)=0.8958+(0.8958-0.5)=1.2916.
\end{align*}
Objective value: $f(x_{2})\approx(0.8958-3)^{2}\approx4.41$.

\paragraph{Iteration $k=2$}
\begin{align*}
y_{3}&=y_{2}+\sigma K x_{2}=0.25+0.5\cdot0.8958\approx0.6979,\\[4pt]
x_{3}&=\operatorname{prox}_{0.1 f}\!\bigl(x_{2}-0.1 y_{3}\bigr)
      =\operatorname{prox}_{0.1 f}(0.8958-0.0698)\\
      &=\operatorname{prox}_{0.1 f}(0.8260)
      =\frac{0.8260+0.6}{1.2}= \frac{1.4260}{1.2}\approx1.1883,\\[4pt]
\bar{x}_{3}&=x_{3}+(\,x_{3}-x_{2}\,)=1.1883+(1.1883-0.8958)\approx1.4808.
\end{align*}
Objective value: $f(x_{3})\approx(1.1883-3)^{2}\approx3.28$.

The iterates clearly move toward the minimiser $x^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Convexity and Closedness]\label{ass:convex}
$f$ and $g^{*}$ are proper, closed and convex.
\end{assumption}
\begin{assumption}[Lipschitz Continuity of $K$]\label{ass:K}
The linear operator $K$ satisfies $\|K\|:=\sup_{\|x\|=1}\|Kx\|<\infty$.
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{ass:stepsize}
The primal and dual step sizes obey
\[
\tau\sigma\|K\|^{2}<1,\qquad \theta\in[0,1].
\tag{5}
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ such that
\[
\mathcal{L}(x^{\star},y)\le \mathcal{L}(x^{\star},y^{\star})\le \mathcal{L}(x,y^{\star}),\qquad\forall x,y.
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Rate]\label{thm:ergodic}
Under Assumptions \ref{ass:convex}–\ref{ass:saddle} and the step‑size condition \eqref{ass:stepsize},
let $(x_{k},y_{k})$ be generated by \eqref{2a}–\eqref{2c} and define the ergodic averages
\[
\tilde{x}_{T}:=\frac{1}{T}\sum_{k=1}^{T}x_{k},\qquad
\tilde{y}_{T}:=\frac{1}{T}\sum_{k=1}^{T}y_{k}.
\]
Then for any saddle point $(x^{\star},y^{\star})$,
\[
\mathcal{G}_{T}
:=\sup_{y}\Bigl\{\mathcal{L}(\tilde{x}_{T},y)-\mathcal{L}(\tilde{x}_{T},y^{\star})\Bigr\}
+\sup_{x}\Bigl\{\mathcal{L}(x^{\star},\tilde{y}_{T})-\mathcal{L}(x,\tilde{y}_{T})\Bigr\}
\le \frac{C}{T},
\tag{7}
\]
where the constant
\[
C:=\frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
     +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}.
\tag{8}
\]
Consequently, $\tilde{x}_{T}\rightarrow x^{\star}$ and $\tilde{y}_{T}\rightarrow y^{\star}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{ass:stepsize} guarantees that the Lyapunov function 
\[
\Phi_{k}:=\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}+\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
\]
is non‑increasing up to a telescoping term (see Lemma~\ref{lem:descent}).
\item \textbf{Hyper‑parameter Sensitivity.} The bound $C$ scales linearly with $1/\tau$ and $1/\sigma$, while the admissible region \eqref{ass:stepsize} shrinks as $\|K\|$ grows. Choosing $\tau=\sigma=1/\|K\|$ (when $\|K\|<1$) yields the largest possible product $\tau\sigma$.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Proximal Gradient Descent} (PGD) applied to the primal problem attains $O(1/k)$ in function value but requires the proximal of $f+g\circ K$, which can be costly.
        \item \emph{Chambolle–Pock} (the original PDHG) coincides with the present algorithm; the theorem recovers their classic $O(1/k)$ ergodic rate.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The PDHG method decouples the primal and dual updates, allowing each proximal operator to be evaluated separately. Under merely convexity (no strong convexity) the algorithm still achieves an $O(1/k)$ decay of the primal–dual gap in the ergodic sense. The proof relies on the non‑expansiveness of proximal maps and a discrete Lyapunov argument. Extensions to stochastic or accelerated variants follow by adapting the same Lyapunov template.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Non‑expansiveness of the Proximal Operator]\label{lem:prox}
For any convex $h$ and $\lambda>0$,
\[
\|\operatorname{prox}_{\lambda h}(u)-\operatorname{prox}_{\lambda h}(v)\|\le\|u-v\|,\qquad\forall u,v.
\tag{9}
\]
\end{lemma}
\begin{proof}
Standard Moreau decomposition; omitted for brevity. \hfill $\square$
\end{proof}

\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any $u,v,w\in\mathbb{R}^{d}$ and any $\lambda>0$,
\[
\langle u-\operatorname{prox}_{\lambda h}(u),\; \operatorname{prox}_{\lambda h}(u)-v\rangle
\ge \frac{1}{2}\bigl\|u-\operatorname{prox}_{\lambda h}(u)\bigr\|^{2}
    +\frac{1}{2}\bigl\|\operatorname{prox}_{\lambda h}(u)-v\bigr\|^{2}
    -\frac{1}{2}\|u-v\|^{2}.
\tag{10}
\]
\end{lemma}
\begin{proof}
See e.g. \cite{Bauschke2011}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:ergodic}. Throughout the proof we fix an arbitrary saddle point $(x^{\star},y^{\star})$ satisfying \eqref{6}.

\subsection*{Step 1 – Descent Lemma for the Dual Update}
From the definition \eqref{2a} and Lemma \ref{lem:three} with $h=g^{*}$, $\lambda=\sigma$,
\[
\langle y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr),\; y_{k+1}-y^{\star}\rangle
\ge\frac{1}{2\sigma}\|y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr)\|^{2}
     +\frac{1}{2\sigma}\|y_{k+1}-y^{\star}\|^{2}
     -\frac{1}{2\sigma}\|y_{k}+\sigma Kx_{k}-y^{\star}\|^{2}.
\tag{11}
\]
Rearranging gives
\[
\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
-\frac{1}{2\sigma}\|y_{k+1}-y^{\star}\|^{2}
\ge \langle Kx_{k},\, y_{k+1}-y^{\star}\rangle
     -\frac{1}{2\sigma}\|y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr)\|^{2}.
\tag{12}
\]
\textbf{[Step 1]}

\subsection*{Step 2 – Descent Lemma for the Primal Update}
Similarly, apply Lemma \ref{lem:three} with $h=f$, $\lambda=\tau$ and use \eqref{2b}:
\[
\langle x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr),\; x_{k+1}-x^{\star}\rangle
\ge\frac{1}{2\tau}\|x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)\|^{2}
     +\frac{1}{2\tau}\|x_{k+1}-x^{\star}\|^{2}
     -\frac{1}{2\tau}\|x_{k}-\tau K^{\top}y_{k+1}-x^{\star}\|^{2}.
\tag{13}
\]
Rearranging yields
\[
\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
-\frac{1}{2\tau}\|x_{k+1}-x^{\star}\|^{2}
\ge -\langle K^{\top}y_{k+1},\, x_{k+1}-x^{\star}\rangle
     -\frac{1}{2\tau}\|x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)\|^{2}.
\tag{14}
\]
\textbf{[Step 2]}

\subsection*{Step 3 – Combine the Two Inequalities}
Add \eqref{12} and \eqref{14}:
\[
\begin{aligned}
&\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
+\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
-\Bigl(\frac{1}{2\tau}\|x_{k+1}-x^{\star}\|^{2}
      +\frac{1}{2\sigma}\|y_{k+1}-y^{\star}\|^{2}\Bigr)\\[4pt]
&\qquad\ge \langle Kx_{k},y_{k+1}-y^{\star}\rangle
          -\langle K^{\top}y_{k+1},x_{k+1}-x^{\star}\rangle\\[4pt]
&\qquad\quad -\frac{1}{2\sigma}\|y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr)\|^{2}
          -\frac{1}{2\tau}\|x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)\|^{2}.
\end{aligned}
\tag{15}
\]
\textbf{[Step 3]}

\subsection*{Step 4 – Rearrangement of the Bilinear Terms}
Notice that
\[
\langle Kx_{k},y_{k+1}-y^{\star}\rangle
-\langle K^{\top}y_{k+1},x_{k+1}-x^{\star}\rangle
=
\langle K\bar{x}_{k+1},y_{k+1}-y^{\star}\rangle
-\langle K^{\top}y_{k+1},\bar{x}_{k+1}-x^{\star}\rangle
+\Delta_{k},
\tag{16}
\]
where $\bar{x}_{k+1}=x_{k+1}+\theta(x_{k+1}-x_{k})$ and
\[
\Delta_{k}:=
\langle K(x_{k}-\bar{x}_{k+1}),y_{k+1}-y^{\star}\rangle
-\langle K^{\top}y_{k+1},x_{k+1}-\bar{x}_{k+1}\rangle.
\tag{17}
\]
Choosing $\theta=1$ (the standard PDHG choice) gives $\bar{x}_{k+1}=2x_{k+1}-x_{k}$ and a straightforward bound
\[
|\Delta_{k}|\le \frac{\|K\|^{2}}{2}\bigl(\|x_{k+1}-x_{k}\|^{2}+\|y_{k+1}-y_{k}\|^{2}\bigr).
\tag{18}
\]
\textbf{[Step 4]}

\subsection*{Step 5 – Bounding the Proximal Residual Terms}
Using the definitions of the updates,
\[
\begin{aligned}
y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr)&= -\sigma\bigl(\partial g^{*}(y_{k+1})\bigr),\\
x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)&= -\tau\bigl(\partial f(x_{k+1})\bigr),
\end{aligned}
\tag{19}
\]
hence
\[
\frac{1}{2\sigma}\|y_{k+1}-\bigl(y_{k}+\sigma Kx_{k}\bigr)\|^{2}
+\frac{1}{2\tau}\|x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)\|^{2}
\ge 0.
\tag{20}
\]
\textbf{[Step 5]}

\subsection*{Step 6 – Lyapunov Descent Inequality}
Discard the non‑negative term \eqref{20} and substitute the bound \eqref{18} into \eqref{15}–\eqref{16}. After elementary algebra we obtain
\[
\Phi_{k}-\Phi_{k+1}
\ge \langle K\bar{x}_{k+1},y_{k+1}-y^{\star}\rangle
   -\langle K^{\top}y_{k+1},\bar{x}_{k+1}-x^{\star}\rangle
   -\frac{\|K\|^{2}}{2}\bigl(\|x_{k+1}-x_{k}\|^{2}+\|y_{k+1}-y_{k}\|^{2}\bigr),
\tag{21}
\]
where $\Phi_{k}$ is the Lyapunov function defined in Lemma~\ref{lem:descent}:
\[
\Phi_{k}:=\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
          +\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}.
\tag{22}
\]
\textbf{[Step 6]}

\subsection*{Step 7 – Use the Saddle‑Point Inequality}
By definition of a saddle point,
\[
\mathcal{L}(\bar{x}_{k+1},y)-\mathcal{L}(x,y_{k+1})\le
\langle K\bar{x}_{k+1},y-y_{k+1}\rangle
-\langle K^{\top}y_{k+1},\bar{x}_{k+1}-x\rangle,
\tag{23}
\]
for any $x,y$. Setting $x=x^{\star}$, $y=y^{\star}$ and comparing with \eqref{21} yields
\[
\Phi_{k}-\Phi_{k+1}\ge \mathcal{L}(\bar{x}_{k+1},y^{\star})
                     -\mathcal{L}(x^{\star},y_{k+1})
                     -\frac{\|K\|^{2}}{2}\bigl(\|x_{k+1}-x_{k}\|^{2}
                                            +\|y_{k+1}-y_{k}\|^{2}\bigr).
\tag{24}
\]
\textbf{[Step 7]}

\subsection*{Step 8 – Summation and Ergodic Gap}
Summing \eqref{24} from $k=0$ to $T-1$ telescopes the Lyapunov terms:
\[
\Phi_{0}-\Phi_{T}\ge
\sum_{k=0}^{T-1}\bigl[\mathcal{L}(\bar{x}_{k+1},y^{\star})
                     -\mathcal{L}(x^{\star},y_{k+1})\bigr]
-\frac{\|K\|^{2}}{2}\sum_{k=0}^{T-1}
   \bigl(\|x_{k+1}-x_{k}\|^{2}+\|y_{k+1}-y_{k}\|^{2}\bigr).
\tag{25}
\]
Dropping the non‑negative last sum and using $\Phi_{T}\ge0$ gives
\[
\sum_{k=0}^{T-1}\bigl[\mathcal{L}(\bar{x}_{k+1},y^{\star})
                     -\mathcal{L}(x^{\star},y_{k+1})\bigr]
\le\Phi_{0}
= \frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
  +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}=C.
\tag{26}
\]
Dividing by $T$ and using convexity of the saddle‑point gap yields
\[
\mathcal{G}_{T}
:=\sup_{y}\bigl\{\mathcal{L}(\tilde{x}_{T},y)-\mathcal{L}(\tilde{x}_{T},y^{\star})\bigr\}
 +\sup_{x}\bigl\{\mathcal{L}(x^{\star},\tilde{y}_{T})-\mathcal{L}(x,\tilde{y}_{T})\bigr\}
\le \frac{C}{T}.
\tag{27}
\]
\textbf{[Step 8]}

Thus the ergodic primal–dual gap decays as $O(1/T)$, which completes the proof of Theorem \ref{thm:ergodic}. \hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From \eqref{27} we have explicitly
\[
\boxed{
\mathcal{G}_{T}\le
\frac{1}{T}\Bigl(\frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
               +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}\Bigr)}.
\tag{28}
\]
If the primal or dual objective is additionally $\mu$‑strongly convex, the same derivation (with the strong convexity term inserted in Lemma \ref{lem:three}) yields an $O\!\bigl((1-\rho)^{T}\bigr)$ linear rate, where $\rho$ depends on $\mu$, $\tau$, $\sigma$ and $\|K\|$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Ascent on the Dual ($g^{*}=0$).}
   The dual step reduces to $y_{k+1}=y_{k}+\sigma Kx_{k}$, and the proof simplifies because the proximal residual term \eqref{20} vanishes.
\item \textbf{No Linear Coupling ($K=0$).}
   The algorithm decouples into independent proximal iterations on $f$ and $g^{*}$, each converging in $O(1/k)$ by standard proximal‑gradient theory.
\item \textbf{Choice $\theta=0$ (no extrapolation).}
   The bound \eqref{18} disappears; the constant $C$ improves slightly, but the $O(1/k)$ rate remains unchanged.
\end{itemize}
