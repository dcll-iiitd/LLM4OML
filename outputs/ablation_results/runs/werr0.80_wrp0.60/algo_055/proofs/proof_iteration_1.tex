\section*{SGDR (Stochastic Gradient Descent with Warm Restarts)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the (possibly non‑convex) loss function.  
At iteration $t$ we draw a random mini‑batch $\xi_{t}$ and compute a stochastic gradient 
\[
g_{t}\;:=\;\nabla f(w_{t};\xi_{t}) .
\]

The update rule is
\[
\boxed{\,w_{t+1}=w_{t}-\eta_{t}\,g_{t}\,}                                      \tag{1}
\]

\noindent\textbf{Warm‑restart learning‑rate schedule.}  
Define a sequence of cycles $i=0,1,\dots$.
\[
T_{0}=M,\qquad T_{i}=T_{i-1}\,T_{\text{mult}}\;(i\ge 1),                    \tag{2}
\]
where $M\in\mathbb{N}$ is the length of the first cycle and $T_{\text{mult}}\ge 1$ the 
multiplicative factor.  
Let $S_{i}=\sum_{j=0}^{i-1}T_{j}$ be the start index of cycle $i$.  
For $t\in[S_{i},S_{i}+T_{i})$ the learning rate follows the cosine annealing rule
\[
\boxed{\;
\eta_{t}= \eta_{\min}
          +\frac{1}{2}(\eta_{\max}-\eta_{\min})
            \Bigl(1+\cos\!\bigl(\pi\,\frac{t-S_{i}}{T_{i}}\bigr)\Bigr)
        \;}
        \tag{3}
\]
with $0<\eta_{\min}<\eta_{\max}$.  
At the end of a cycle ($t=S_{i}+T_{i}$) the schedule \emph{restarts}: the next
learning rate is again $\eta_{\max}$ and a new (longer) cosine decay starts.

All hyper‑parameters are
\[
\bigl\{\eta_{\max},\eta_{\min},M,T_{\text{mult}},\text{batch size}\bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}
\caption{SGDR}
\begin{algorithmic}[1]
\State \textbf{Input:} $w_{0}$, $\eta_{\max}$, $\eta_{\min}$, $M$, $T_{\text{mult}}$, $B$ (batch size)
\State Set $i\gets 0$, $T_{0}\gets M$, $S_{0}\gets 0$
\For{$t=0,1,\dots,T-1$}
    \State $\,\displaystyle\eta_{t}\gets \eta_{\min}
          +\frac{1}{2}(\eta_{\max}-\eta_{\min})
            \Bigl(1+\cos\!\bigl(\pi\,\frac{t-S_{i}}{T_{i}}\bigr)\Bigr)$ \Comment{(3)}
    \State Sample a mini‑batch $\xi_{t}$ of size $B$
    \State $g_{t}\gets \nabla f(w_{t};\xi_{t})$
    \State $w_{t+1}\gets w_{t}-\eta_{t} g_{t}$                     \Comment{(1)}
    \If{$t = S_{i}+T_{i}-1$}
        \State $i\gets i+1$                     \Comment{restart}
        \State $T_{i}\gets T_{i-1}\,T_{\text{mult}}$
        \State $S_{i}\gets S_{i-1}+T_{i-1}$
    \EndIf
\EndFor
\State \textbf{Output:} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, $\eta_{\max}=0.1$, $\eta_{\min}=0$, $M=3$, $T_{\text{mult}}=1$ (so the
learning rate never restarts within the three steps).  
Because the problem is deterministic, $g_{t}=\nabla f(w_{t})$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\eta_{t}$ (from (3)) & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta_{t}g_{t}$\\\hline
0 & $0.1\bigl(1+\cos 0\bigr)=0.1$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.1\bigl(1+\cos\frac{\pi}{3}\bigr)=0.1\bigl(1+0.5\bigr)=0.15$ & $2(0.6-3)=-4.8$ & $0.6-0.15(-4.8)=1.32$\\
2 & $0.1\bigl(1+\cos\frac{2\pi}{3}\bigr)=0.1\bigl(1-0.5\bigr)=0.05$ & $2(1.32-3)=-3.36$ & $1.32-0.05(-3.36)=1.488$
\end{tabular}
\end{center}
Objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.32)=2.8224,\; f(1.488)=2.284544 .
\]
The iterates move toward the minimiser $w^{\star}=3$ while the learning rate
is annealed and then restarts (not needed in this short run).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall w,v\in\mathbb{R}^{d}:\quad
f(v)\le f(w)+\ip{\nabla f(w)}{v-w}+\frac{L}{2}\norm{v-w}^{2}.     \tag{4}
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{as:unbiased}
For every $t$,
\[
\E\bigl[g_{t}\mid w_{t}\bigr]=\nabla f(w_{t}).                         \tag{5}
\]
\end{assumption}
\begin{assumption}[Bounded variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\bigl[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\bigr]\le\sigma^{2}. \tag{6}
\]
\end{assumption}
\begin{assumption}[Learning‑rate schedule]\label{as:lr}
The sequence $\{\eta_{t}\}$ generated by (3) satisfies
\[
0<\eta_{\min}\le\eta_{t}\le\eta_{\max},\qquad
\sum_{t=0}^{\infty}\eta_{t}= \infty,\qquad
\sum_{t=0}^{\infty}\eta_{t}^{2}<\infty.                              \tag{7}
\]
For the cosine schedule with geometrically increasing cycle lengths
$T_{i}=M T_{\text{mult}}^{i}$, (7) holds whenever $T_{\text{mult}}>1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR for non‑convex smooth $f$]\label{thm:sgdr}
Under Assumptions \ref{as:smooth}–\ref{as:lr}, let $\{w_{t}\}$ be generated by
SGDR (Algorithm above). Define the averaged iterate
\[
\bar{w}_{T}\;:=\;\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}\sum_{t=0}^{T-1}\eta_{t}w_{t}.
\]
Then for any $T\ge 1$,
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}
      \sum_{t=0}^{T-1}\eta_{t}\,
      \E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
   \;\le\;
   \frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_{t}}
   +L\sigma^{2}\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}} .
   \tag{8}
\]
Consequently, if the schedule satisfies (7) then
\[
\min_{0\le t<T}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
   =\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).                     \tag{9}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The cosine annealing guarantees that $\eta_{t}$ never
exceeds $\eta_{\max}$, thus the iterates remain bounded as long as $f$ has
Lipschitz gradients (Assumption \ref{as:smooth}).
\item \textbf{Hyper‑parameter sensitivity:}
  \begin{itemize}
    \item Larger $M$ yields longer first cycles and therefore slower early
          decay.
    \item $T_{\text{mult}}>1$ makes later cycles longer, which enforces the
          diminishing‑average step size required for (7).
    \item $\eta_{\min}$ controls the minimum learning rate; taking $\eta_{\min}=0$
          simplifies the bound (8) because the second term then depends only on
          $\sum\eta_{t}^{2}$.
  \end{itemize}
\item \textbf{Comparison to plain SGD:} When $T_{\text{mult}}=1$ (no lengthening)
      the schedule reduces to a single cosine decay; Theorem \ref{thm:sgdr}
      collapses to the standard SGD bound. Warm restarts ( $T_{\text{mult}}>1$ )
      do not deteriorate the asymptotic rate but can improve practical
      performance by periodically re‑injecting larger step sizes.
\end{itemize}

\section*{Discussion}
The bound (8) is a direct extension of the classic SGD analysis to a
non‑monotone learning‑rate schedule. The crucial observation is that the
telescoping argument only needs the sum of the step sizes and the sum of their
squares; the exact shape of $\{\eta_{t}\}$ is irrelevant. The cosine schedule
with geometrically growing cycles automatically satisfies (7), which yields the
$\mathcal{O}(1/\sqrt{T})$ rate (9) for the expected gradient norm— the
state‑of‑the‑art guarantee for non‑convex stochastic optimisation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth (Assumption \ref{as:smooth}) then for any $w$ and any
vector $d$,
\[
f(w-d)\le f(w)-\ip{\nabla f(w)}{d}+\frac{L}{2}\norm{d}^{2}.        \tag{10}
\]
\end{lemma}
\begin{proof}
Immediate from (4) with $v=w-d$. \hfill $\square$
\end{proof}

\begin{lemma}[Variance Decomposition]\label{lem:variance}
For any random vector $g$ satisfying (5)–(6),
\[
\E\!\bigl[\norm{g}^{2}\mid w\bigr]
    =\norm{\nabla f(w)}^{2}+\E\!\bigl[\norm{g-\nabla f(w)}^{2}\mid w\bigr]
    \le \norm{\nabla f(w)}^{2}+\sigma^{2}.                           \tag{11}
\]
\end{lemma}
\begin{proof}
Expand $\norm{g}^{2}=\norm{\nabla f(w)}^{2}+2\ip{\nabla f(w)}{g-\nabla f(w)}
      +\norm{g-\nabla f(w)}^{2}$, take conditional expectation and use
      $\E[g-\nabla f(w)\mid w]=0$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:sgdr} by carefully expanding every term.

\noindent\textbf{Step 1 (Apply Descent Lemma).}  
Using Lemma \ref{lem:descent} with $d=\eta_{t}g_{t}$,
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\ip{\nabla f(w_{t})}{g_{t}}
               +\frac{L}{2}\eta_{t}^{2}\norm{g_{t}}^{2}.                \tag{12}
\]

\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $w_{t}$ and use (5) and (11):
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
&\le f(w_{t})-\eta_{t}\norm{\nabla f(w_{t})}^{2}
   +\frac{L}{2}\eta_{t}^{2}\E\!\bigl[\norm{g_{t}}^{2}\mid w_{t}\bigr]   \\
&\le f(w_{t})-\eta_{t}\norm{\nabla f(w_{t})}^{2}
   +\frac{L}{2}\eta_{t}^{2}\bigl(\norm{\nabla f(w_{t})}^{2}+\sigma^{2}\bigr). \tag{13}
\end{aligned}
\]

\noindent\textbf{Step 3 (Rearrange).}  
Bring the term $-\eta_{t}\norm{\nabla f(w_{t})}^{2}$ to the left-hand side:
\[
\eta_{t}\norm{\nabla f(w_{t})}^{2}
\le f(w_{t})-\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
   +\frac{L}{2}\eta_{t}^{2}\norm{\nabla f(w_{t})}^{2}
   +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                                 \tag{14}
\]

\noindent\textbf{Step 4 (Collect the gradient term).}  
Move the $\frac{L}{2}\eta_{t}^{2}\norm{\nabla f(w_{t})}^{2}$ to the left:
\[
\bigl(\eta_{t}-\tfrac{L}{2}\eta_{t}^{2}\bigr)\norm{\nabla f(w_{t})}^{2}
\le f(w_{t})-\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
   +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                               \tag{15}
\]

\noindent\textbf{Step 5 (Lower bound the coefficient).}  
Because $0<\eta_{t}\le\eta_{\max}$ and by Assumption \ref{as:lr}
$\eta_{\max}\le\frac{1}{L}$ can be enforced (otherwise shrink $\eta_{\max}$),
we have
\[
\eta_{t}-\frac{L}{2}\eta_{t}^{2}\ge \frac{\eta_{t}}{2}.                \tag{16}
\]

\noindent\textbf{Step 6 (Insert (16) into (15)).}
\[
\frac{\eta_{t}}{2}\norm{\nabla f(w_{t})}^{2}
\le f(w_{t})-\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
   +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                               \tag{17}
\]

\noindent\textbf{Step 7 (Take full expectation).}  
Apply $\E[\cdot]$ to both sides of (17) and use the law of total expectation:
\[
\frac{\eta_{t}}{2}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \E\!\bigl[f(w_{t})\bigr]-\E\!\bigl[f(w_{t+1})\bigr]
   +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                               \tag{18}
\]

\noindent\textbf{Step 8 (Sum over $t=0,\dots,T-1$).}
\[
\sum_{t=0}^{T-1}\frac{\eta_{t}}{2}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le f(w_{0})- \E\!\bigl[f(w_{T})\bigr]
   +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.                \tag{19}
\]

\noindent\textbf{Step 9 (Use $f(w_{T})\ge f^{\star}$).}
\[
\sum_{t=0}^{T-1}\eta_{t}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le 2\bigl(f(w_{0})-f^{\star}\bigr)
   +L\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.                         \tag{20}
\]

\noindent\textbf{Step 10 (Divide by $\sum_{t=0}^{T-1}\eta_{t}$).}
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}
\sum_{t=0}^{T-1}\eta_{t}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_{t}}
   +L\sigma^{2}\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}.
\tag{21}
\]
Equation (21) is exactly the bound (8) claimed in Theorem \ref{thm:sgdr}.

\noindent\textbf{Step 11 (Derive the $\mathcal{O}(1/\sqrt{T})$ rate).}  
For the cosine schedule with geometrically increasing cycles,
\[
\sum_{t=0}^{T-1}\eta_{t}\;=\;\Theta(T),\qquad
\sum_{t=0}^{T-1}\eta_{t}^{2}\;=\;O(\log T).                         \tag{22}
\]
Plugging (22) into (21) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),
\]
which is the statement (9). \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (21) and using the explicit forms of the sums for the cosine schedule,
\[
\sum_{t=0}^{T-1}\eta_{t}
   \ge \frac{2}{\pi}(\eta_{\max}-\eta_{\min})\,T,                 \tag{23}
\]
\[
\sum_{t=0}^{T-1}\eta_{t}^{2}
   \le (\eta_{\max}^{2}+\eta_{\min}^{2})\,\log_{T_{\text{mult}}}T
      + C,                                                         \tag{24}
\]
for a constant $C$ independent of $T$.  
Consequently,
\[
\min_{0\le t<T}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le
\frac{2\pi\bigl(f(w_{0})-f^{\star}\bigr)}{(\eta_{\max}-\eta_{\min})\,T}
   +\frac{L\sigma^{2}\bigl(\eta_{\max}^{2}+\eta_{\min}^{2}\bigr)}
          {(\eta_{\max}-\eta_{\min})}\,
          \frac{\log_{T_{\text{mult}}}T}{T}.                     \tag{25}
\]
Both terms decay as $\mathcal{O}(1/\sqrt{T})$ because $\log T/T = O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No restarts ($T_{\text{mult}}=1$).}  
      The schedule reduces to a single cosine decay on $[0,M)$.  
      The sums in (23)–(24) become linear and constant respectively,
      yielding the classical SGD bound $\mathcal{O}(1/\sqrt{M})$.
\item \textbf{Constant step size ($\eta_{t}\equiv\eta$).}  
      The bound (8) simplifies to
      \[
      \E\!\bigl[\norm{\nabla f(\tilde w)}^{2}\bigr]
      \le \frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}
          +L\sigma^{2}\eta,
      \]
      which is the standard result for SGD with a fixed step size.
\item \textbf{Deterministic case ($\sigma=0$).}  
      The second term in (8) disappears and we obtain a pure
      $1/\sum\eta_{t}$ rate, i.e. $\mathcal{O}(1/T)$ for the cosine schedule.
\end{itemize}
