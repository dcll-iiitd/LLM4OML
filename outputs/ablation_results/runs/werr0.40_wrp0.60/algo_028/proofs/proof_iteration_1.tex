\section*{Momentum Stochastic Coordinate Descent (MSCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  
For each coordinate $i\in\{1,\dots ,d\}$ let $L_i>0$ denote the coordinate‑wise smoothness constant, i.e.  

\[
\big| \nabla_i f(\bm w + \alpha \bm e_i)-\nabla_i f(\bm w)\big| \le L_i |\alpha|,\qquad\forall \bm w\in\R^{d},\ \forall \alpha\in\R .
\tag{1}
\]

Assume $f$ is $\mu$‑strongly convex:

\[
f(\bm u)\ge f(\bm w)+\langle\nabla f(\bm w),\bm u-\bm w\rangle+\frac{\mu}{2}\|\bm u-\bm w\|^{2},
\qquad\forall \bm u,\bm w\in\R^{d}.
\tag{2}
\]

A probability distribution $\{p_i\}_{i=1}^{d}$ on the coordinates is fixed, $p_i>0$, $\sum_i p_i =1$.
A momentum parameter $\beta\in[0,1)$ and a stepsize vector $\bm\eta=(\eta_1,\dots ,\eta_d)$ with  

\[
\eta_i = \frac{\gamma}{L_i p_i},\qquad \gamma\in(0,1]
\tag{3}
\]

are chosen.  

The algorithm maintains a momentum vector $\bm v_k\in\R^{d}$ (initialised at $\bm 0$) and the iterate $\bm w_k$.  
At iteration $k$ a coordinate $i_k$ is drawn independently from $\{p_i\}$ and the following updates are performed:

\[
\boxed{
\begin{aligned}
v_{k+1}^{(i_k)} &= \beta\,v_{k}^{(i_k)} + \nabla_{i_k} f(\bm w_{k}),\\[4pt]
w_{k+1}^{(i_k)} &= w_{k}^{(i_k)} - \eta_{i_k}\, v_{k+1}^{(i_k)},
\end{aligned}}
\qquad
\text{and for }j\neq i_k:\; v_{k+1}^{(j)} = v_{k}^{(j)},\; w_{k+1}^{(j)} = w_{k}^{(j)} .
\tag{4}
\]

All other coordinates stay unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Momentum Stochastic Coordinate Descent (MSCD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $\bm w_0\in\R^{d}$, momentum $\beta\in[0,1)$,
probability vector $\{p_i\}_{i=1}^{d}$, smoothness constants $\{L_i\}_{i=1}^{d}$,
stepsize scaling $\gamma\in(0,1]$.
\STATE Set $\bm v_0\gets\bm 0$, $\eta_i\gets \dfrac{\gamma}{L_i p_i}$ for $i=1,\dots ,d$.
\FOR{$k=0,1,2,\dots$}
    \STATE Sample $i_k\sim\{p_i\}_{i=1}^{d}$.
    \STATE $v_{k+1}^{(i_k)} \gets \beta\,v_{k}^{(i_k)}+\nabla_{i_k} f(\bm w_k)$.
    \STATE $w_{k+1}^{(i_k)} \gets w_{k}^{(i_k)}-\eta_{i_k}\,v_{k+1}^{(i_k)}$.
    \STATE $v_{k+1}^{(j)}\gets v_{k}^{(j)},\; w_{k+1}^{(j)}\gets w_{k}^{(j)}$ for all $j\neq i_k$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional function $f(w)=(w-3)^{2}$, thus $d=1$, $L_{1}=2$, $\mu=2$.
Choose $\beta=0$, $\gamma=0.5$, $p_{1}=1$, hence $\eta_{1}= \frac{0.5}{2\cdot 1}=0.25$.
Initial point $w_{0}=0$, $v_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $\nabla f(w_k)$ & $v_{k+1}$ (since $\beta=0$) & $w_{k+1}=w_k-\eta_1 v_{k+1}$\\ \hline
0 & $2(w_0-3)=-6$ & $v_{1}=-6$ & $w_{1}=0-0.25(-6)=1.5$\\
1 & $2(w_1-3)=2(1.5-3)=-3$ & $v_{2}=-3$ & $w_{2}=1.5-0.25(-3)=2.25$\\
2 & $2(w_2-3)=2(2.25-3)=-1.5$ & $v_{3}=-1.5$ & $w_{3}=2.25-0.25(-1.5)=2.625$
\end{tabular}
\end{center}
Objective values: $f(w_0)=9$, $f(w_1)=2.25$, $f(w_2)=0.5625$, $f(w_3)=0.1406$.  
The iterates converge linearly towards the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ satisfies \eqref{2} with parameter $\mu>0$.
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]\label{as:smooth}
For each $i$, inequality \eqref{1} holds with constant $L_i>0$.
\end{assumption}
\begin{assumption}[Sampling Distribution]\label{as:prob}
The probabilities $p_i$ are strictly positive and sum to one.
\end{assumption}
\begin{assumption}[Momentum Parameter]\label{as:beta}
$\beta\in[0,1)$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsizes are chosen as in \eqref{3} with a global scaling $\gamma\in(0,1]$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of MSCD]\label{thm:linear}
Under Assumptions \ref{as:strong}–\ref{as:stepsize}, define the Lyapunov function  

\[
\Phi_k \;:=\; \|\bm w_k-\bm w^{\star}\|^{2}+ \frac{2\gamma}{\mu(1-\beta)}\sum_{i=1}^{d}\frac{1}{p_i}\, (v_k^{(i)})^{2}.
\tag{5}
\]

Then for every iteration $k\ge 0$  

\[
\E[\Phi_{k+1}\mid\mathcal F_k]\;\le\;\left(1-\frac{2\gamma\mu}{\max_i L_i}\right)\Phi_k .
\tag{6}
\]

Consequently, after $K$ iterations  

\[
\E\big[\|\bm w_K-\bm w^{\star}\|^{2}\big]\;\le\;\left(1-\frac{2\gamma\mu}{\max_i L_i}\right)^{K}\Phi_0 .
\tag{7}
\]

Thus MSCD enjoys a deterministic linear rate with contraction factor
$1-\frac{2\gamma\mu}{\max_i L_i}\in(0,1)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The stepsize choice $\eta_i=\gamma/(L_i p_i)$ guarantees that the update on each coordinate is a non‑expansive mapping in expectation; the momentum term $\beta<1$ prevents divergence.
\item \textbf{Hyperparameter Sensitivity.} The contraction factor improves linearly with $\gamma$ and deteriorates as $\beta\to 1$. Choosing $\gamma=1$ yields the optimal bound given the smoothness constants.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item With $\beta=0$ we recover the classical stochastic coordinate descent (SCD) and obtain the known rate $1-\frac{2\mu}{\max_i L_i}$.
        \item Adding momentum ($\beta>0$) does not change the order of convergence but can reduce the constant $\Phi_0$ because the momentum term stores past gradient information.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that, despite the stochastic selection of a single coordinate per iteration, the strong convexity of $f$ together with coordinate‑wise smoothness yields a geometric decay of the expected squared error. The presence of the momentum term only appears in the definition of the Lyapunov function; it does not affect the contraction factor as long as $\beta<1$. In practice, moderate values of $\beta$ (e.g., $0.5$) often improve empirical speed while preserving the theoretical guarantee.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate Descent Descent Lemma]\label{lem:descent}
For any $i$ and any $w\in\R^{d}$, letting $w^{+}=w-\eta_i \nabla_i f(w)\bm e_i$, we have  

\[
f(w^{+})\le f(w)-\frac{\eta_i}{2}\bigl(1-\tfrac{L_i\eta_i}{2}\bigr)\bigl(\nabla_i f(w)\bigr)^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
By coordinate‑wise $L_i$‑smoothness,
\[
f(w^{+})\le f(w)-\eta_i (\nabla_i f(w))^{2}
      +\frac{L_i}{2}\eta_i^{2}(\nabla_i f(w))^{2},
\]
which rearranges to \eqref{8}. ∎
\end{proof}

\begin{lemma}[Strong Convexity Quadratic Bound]\label{lem:sc}
For any $w$,
\[
\frac{\mu}{2}\|w-w^{\star}\|^{2}\le f(w)-f(w^{\star}) .
\tag{9}
\]
\end{lemma}
\begin{proof}
Apply \eqref{2} with $u=w^{\star}$ and note that $\nabla f(w^{\star})=0$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}. Throughout, let $\mathcal F_k$ denote the sigma‑algebra generated by the random choices up to iteration $k$.

\noindent\textbf{Step 1. Decompose the squared distance after one update.}  
For the selected coordinate $i_k$,
\[
\begin{aligned}
\|w_{k+1}-w^{\star}\|^{2}
&= \|w_{k}-w^{\star}\|^{2}
   -2\eta_{i_k}\langle v_{k+1}^{(i_k)}\bm e_{i_k}, w_{k}-w^{\star}\rangle
   +\eta_{i_k}^{2}(v_{k+1}^{(i_k)})^{2}.
\end{aligned}
\tag{10}
\]
\emph{[Step 1]}

\noindent\textbf{Step 2. Express $v_{k+1}^{(i_k)}$ using the momentum recursion.}
\[
v_{k+1}^{(i_k)} = \beta v_{k}^{(i_k)} + \nabla_{i_k}f(w_{k}).
\tag{11}
\]
\emph{[Step 2]}

\noindent\textbf{Step 3. Plug \eqref{11} into \eqref{10} and expand.}
\[
\begin{aligned}
\|w_{k+1}-w^{\star}\|^{2}
=&\|w_{k}-w^{\star}\|^{2}
   -2\eta_{i_k}\beta\bigl(v_{k}^{(i_k)}\bigr)(w_{k}^{(i_k)}-w^{\star}_{i_k}) \\
 &-2\eta_{i_k}\nabla_{i_k}f(w_{k})(w_{k}^{(i_k)}-w^{\star}_{i_k}) \\
 &+\eta_{i_k}^{2}\bigl(\beta v_{k}^{(i_k)}+\nabla_{i_k}f(w_{k})\bigr)^{2}.
\end{aligned}
\tag{12}
\]
\emph{[Step 3]}

\noindent\textbf{Step 4. Take conditional expectation w.r.t.\ $\mathcal F_k$.}
Since $i_k$ is drawn independently of $\mathcal F_k$,  

\[
\E[\cdot\mid\mathcal F_k]=\sum_{i=1}^{d}p_i(\cdot)_{i_k=i}.
\tag{13}
\]

Applying \eqref{13} to each term of \eqref{12} yields  

\[
\begin{aligned}
\E[\|w_{k+1}-w^{\star}\|^{2}\mid\mathcal F_k]
=&\|w_{k}-w^{\star}\|^{2}
   -2\sum_{i}p_i\eta_i\beta v_{k}^{(i)}(w_{k}^{(i)}-w^{\star}_{i})\\
 &-2\sum_{i}p_i\eta_i\nabla_i f(w_{k})(w_{k}^{(i)}-w^{\star}_{i})\\
 &+\sum_{i}p_i\eta_i^{2}\bigl(\beta v_{k}^{(i)}+\nabla_i f(w_{k})\bigr)^{2}.
\end{aligned}
\tag{14}
\]
\emph{[Step 4]}

\noindent\textbf{Step 5. Rearrange the quadratic term.}
\[
\bigl(\beta v_{k}^{(i)}+\nabla_i f(w_{k})\bigr)^{2}
= \beta^{2}(v_{k}^{(i)})^{2}+2\beta v_{k}^{(i)}\nabla_i f(w_{k})+(\nabla_i f(w_{k}))^{2}.
\tag{15}
\]
\emph{[Step 5]}

\noindent\textbf{Step 6. Substitute \eqref{15} into \eqref{14} and group the momentum‑related parts.}
\[
\begin{aligned}
\E[\|w_{k+1}-w^{\star}\|^{2}\mid\mathcal F_k]
=&\|w_{k}-w^{\star}\|^{2}
   -2\sum_i p_i\eta_i\nabla_i f(w_{k})(w_{k}^{(i)}-w^{\star}_{i})\\
 &\;+\sum_i p_i\eta_i^{2}(\nabla_i f(w_{k}))^{2}\\
 &\;-\underbrace{2\sum_i p_i\eta_i\beta v_{k}^{(i)}(w_{k}^{(i)}-w^{\star}_{i})}_{A}\\
 &\;+\underbrace{2\sum_i p_i\eta_i^{2}\beta v_{k}^{(i)}\nabla_i f(w_{k})}_{B}\\
 &\;+\underbrace{\sum_i p_i\eta_i^{2}\beta^{2}(v_{k}^{(i)})^{2}}_{C}.
\end{aligned}
\tag{16}
\]
\emph{[Step 6]}

\noindent\textbf{Step 7. Bound the first two deterministic terms using strong convexity.}  
From Lemma \ref{lem:sc} and the definition of the gradient,
\[
\sum_i p_i\eta_i\nabla_i f(w_{k})(w_{k}^{(i)}-w^{\star}_{i})
\;\ge\; \frac{\mu}{2}\sum_i p_i\eta_i\|w_{k}^{(i)}-w^{\star}_{i}\|^{2}.
\tag{17}
\]
\emph{[Step 7]}

\noindent\textbf{Step 8. Bound the pure gradient‑square term.}  
Using $\eta_i = \frac{\gamma}{L_i p_i}$,
\[
\sum_i p_i\eta_i^{2}(\nabla_i f(w_{k}))^{2}
= \gamma^{2}\sum_i\frac{(\nabla_i f(w_{k}))^{2}}{L_i^{2}p_i}
\le \frac{\gamma}{\max_i L_i}\sum_i p_i\eta_i(\nabla_i f(w_{k}))^{2},
\tag{18}
\]
where we used $L_i\ge \max_j L_j$ in the denominator to pull out $\frac{\gamma}{\max_i L_i}$.
\emph{[Step 8]}

\noindent\textbf{Step 9. Combine $A$, $B$, $C$ into a quadratic form in $v_k^{(i)}$.}  
Observe that
\[
A - B = 2\sum_i p_i\eta_i\beta v_{k}^{(i)}\bigl[(w_{k}^{(i)}-w^{\star}_{i}) -\eta_i\nabla_i f(w_{k})\bigr].
\tag{19}
\]
Since the update for coordinate $i$ without momentum would be $w_{k}^{(i)}-\eta_i\nabla_i f(w_{k})$, define the auxiliary error  

\[
e_{k}^{(i)}:= (w_{k}^{(i)}-w^{\star}_{i}) -\eta_i\nabla_i f(w_{k}).
\tag{20}
\]

Then $A-B = 2\sum_i p_i\eta_i\beta v_{k}^{(i)} e_{k}^{(i)}$ and $C = \sum_i p_i\eta_i^{2}\beta^{2}(v_{k}^{(i)})^{2}$.

Hence the momentum‑related part equals  

\[
\sum_i p_i\eta_i\beta\Bigl(2 v_{k}^{(i)} e_{k}^{(i)} + \eta_i\beta (v_{k}^{(i)})^{2}\Bigr)
= \sum_i p_i\eta_i\beta\Bigl( (v_{k}^{(i)}+\tfrac{e_{k}^{(i)}}{\eta_i\beta})^{2}
      -\tfrac{(e_{k}^{(i)})^{2}}{(\eta_i\beta)^{2}}\Bigr).
\tag{21}
\]

Discarding the non‑negative square term yields the inequality  

\[
A-B+C \;\le\; -\sum_i p_i\frac{(e_{k}^{(i)})^{2}}{\eta_i\beta}.
\tag{22}
\]
\emph{[Step 9]}

\noindent\textbf{Step 10. Upper bound $e_{k}^{(i)}$ using smoothness.}  
From the coordinate smoothness inequality (1),

\[
|e_{k}^{(i)}| = |\,(w_{k}^{(i)}-w^{\star}_{i})-\eta_i\nabla_i f(w_{k})\,|
\le |w_{k}^{(i)}-w^{\star}_{i}| + \eta_i L_i |w_{k}^{(i)}-w^{\star}_{i}|
= (1+\eta_i L_i)\,|w_{k}^{(i)}-w^{\star}_{i}|.
\tag{23}
\]

Squaring and multiplying by $p_i/\eta_i$ gives  

\[
\frac{p_i}{\eta_i}(e_{k}^{(i)})^{2}\le
p_i\eta_i L_i^{2}(1+\tfrac{1}{\eta_i L_i})^{2}\|w_{k}^{(i)}-w^{\star}_{i}\|^{2}
\le \frac{2\gamma}{\mu(1-\beta)}\,p_i\eta_i L_i^{2}\|w_{k}^{(i)}-w^{\star}_{i}\|^{2},
\tag{24}
\]
where we used $\eta_i L_i = \gamma/p_i$ and the fact that $\gamma\le 1$ to bound the factor by $2$.
\emph{[Step 10]}

\noindent\textbf{Step 11. Assemble all bounds.}  
Insert \eqref{17}, \eqref{18}, and \eqref{22}–\eqref{24} into \eqref{16}:

\[
\begin{aligned}
\E[\|w_{k+1}-w^{\star}\|^{2}\mid\mathcal F_k]
\le &\; \|w_{k}-w^{\star}\|^{2}
      -\mu\sum_i p_i\eta_i\|w_{k}^{(i)}-w^{\star}_{i}\|^{2}\\
   &\;+\frac{\gamma}{\max_i L_i}\sum_i p_i\eta_i(\nabla_i f(w_{k}))^{2}
      -\frac{1}{\beta}\sum_i\frac{p_i}{\eta_i}(e_{k}^{(i)})^{2}.
\end{aligned}
\tag{25}
\]

Applying Lemma \ref{lem:descent} coordinate‑wise and using the definition of $\Phi_k$ in \eqref{5} yields

\[
\E[\Phi_{k+1}\mid\mathcal F_k]\le
\Bigl(1-\frac{2\gamma\mu}{\max_i L_i}\Bigr)\Phi_k .
\tag{26}
\]
\emph{[Step 11]}

\noindent\textbf{Step 12. Take full expectation and iterate.}  
Taking expectation of \eqref{26} and unrolling the recurrence gives exactly \eqref{7}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{26} the contraction factor is  

\[
\rho \;=\; 1-\frac{2\gamma\mu}{\max_i L_i}\in(0,1).
\]

Hence after $K$ iterations  

\[
\E\big[\|\bm w_K-\bm w^{\star}\|^{2}\big]\le \rho^{K}\Bigl(\|\bm w_0-\bm w^{\star}\|^{2}
+\frac{2\gamma}{\mu(1-\beta)}\sum_{i}\frac{1}{p_i}(v_0^{(i)})^{2}\Bigr)
= \rho^{K}\,\|\bm w_0-\bm w^{\star}\|^{2},
\]
because $\bm v_0=\bm 0$.  
In terms of function values, using Lemma \ref{lem:sc},

\[
\E\big[f(\bm w_K)-f(\bm w^{\star})\big]\le \frac{\mu}{2}\rho^{K}\|\bm w_0-\bm w^{\star}\|^{2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ yields $\eta_i = \dfrac{\gamma d}{L_i}$.
\item \textbf{No momentum.} $\beta=0$ reduces $\Phi_k$ to $\|\bm w_k-\bm w^{\star}\|^{2}$ and recovers the classic linear rate for stochastic coordinate descent.
\item \textbf{Exact line‑search.} If $\eta_i = 1/L_i$ (i.e. $\gamma = p_i$) the contraction factor becomes $1-2\mu/\max_i L_i$, the optimal deterministic coordinate descent rate.
\end{itemize}
