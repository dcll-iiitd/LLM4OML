\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) with Double‑Loop Structure}

\section*{Mathematical Formulation}
Let $f(x)=\frac1n\sum_{i=1}^{n} f_i(x)$ where each $f_i:\R^d\to\R$ is convex, $L$‑smooth and $\mu$‑strongly convex.  
The algorithm proceeds in \emph{epochs} $s=0,1,\dots$.  
At the beginning of epoch $s$ a \emph{snapshot point} $\tilde{x}^{\,s}$ is fixed and its full gradient
\[
\tilde{\mu}^{\,s}= \nabla f(\tilde{x}^{\,s})=\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})
\]
is computed.

For $t=0,1,\dots,m-1$ (inner loop) we draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and form the variance‑reduced estimator
\[
v_t \;=\; \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}.
\tag{1}
\]
The iterate is updated by a standard SGD step
\[
x_{t+1}\;=\;x_t-\eta v_t ,\qquad \eta>0\ \text{(stepsize)}.
\tag{2}
\]
After $m$ inner steps we set the next snapshot $\tilde{x}^{\,s+1}=x_m$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $x^{0}\in\R^{d}$
\State $\tilde{x}^{\,0}\gets x^{0}$
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\tilde{\mu}^{\,s}\gets \frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})$
    \State $x_0 \gets \tilde{x}^{\,s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t \gets \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$   \Comment{variance‑reduced estimator}
        \State $x_{t+1}\gets x_t-\eta v_t$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_m$
\EndFor
\State \textbf{return} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)= (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We treat it as a finite‑sum with a single component ($n=1$), hence $\tilde{\mu}= \nabla f(\tilde{x})$.

\begin{tabular}{c|c|c|c}
\textbf{Iter} & $x_t$ & $v_t$ (by (1)) & $f(x_{t+1})$ \\
\hline
0 (snapshot) & $x_0=0$ & $v_0=2(0-3)= -6$ & $f(0.6)= (0.6-3)^2=5.76$\\
1 & $x_1=0-0.1(-6)=0.6$ & $v_1=2(0.6-3)= -4.8$ & $f(0.96)= (0.96-3)^2=4.1616$\\
2 & $x_2=0.6-0.1(-4.8)=1.08$ & $v_2=2(1.08-3)= -3.84$ & $f(1.464)= (1.464-3)^2=2.3665$
\end{tabular}

\noindent
Explanation: because $n=1$, the correction term $\nabla f_{i_t}(\tilde{x})$ equals $\tilde{\mu}$, so $v_t=\nabla f(x_t)$. The steps follow (2) with $\eta=0.1$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
Each $f_i$ (hence $f$) is $\mu$‑strongly convex:
\[
f_i(y)\ge f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Stepsize and Inner Loop Length]\label{ass:params}
The stepsize satisfies $0<\eta\le \frac{1}{3L}$ and the inner loop length $m$ is chosen such that
\[
\rho\;:=\;\frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}\;<\;1 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Under Assumptions~\ref{ass:smooth}--\ref{ass:params}, let $\tilde{x}^{\,s}$ be the snapshot after epoch $s$. Then for all $s\ge0$,
\[
\E\!\bigl[\,f(\tilde{x}^{\,s})-f(x^{\star})\,\bigr]\;\le\;\rho^{\,s}\,\bigl[f(\tilde{x}^{\,0})-f(x^{\star})\bigr],
\tag{3}
\]
where $x^{\star}$ is the unique minimizer of $f$ and $\rho\in(0,1)$ is defined in Assumption~\ref{ass:params}. Consequently the iterates converge linearly in expectation to the optimum.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability:} The algorithm is stable for any $\eta\le 1/(3L)$; the bound $\rho<1$ guarantees a contraction of the expected suboptimality after each epoch.
\item \textbf{Hyper‑parameter Sensitivity:} The contraction factor $\rho$ improves (gets smaller) when $\eta$ is increased up to $1/(2L)$ and when $m$ grows proportionally to $L/\mu$. In practice a modest $m\approx 2n$ suffices.
\item \textbf{Comparison to Baselines:}
    \begin{enumerate}
        \item \emph{Plain SGD} with diminishing stepsize yields $O(1/T)$ convergence for strongly convex problems.
        \item \emph{SVRG} achieves a geometric rate $O(\rho^{\,s})$ with the same per‑iteration cost as SGD, thus strictly faster once $m$ inner steps are performed.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The linear rate (3) matches the optimal rate for first‑order methods on strongly convex smooth functions. The double‑loop structure is essential: the outer loop supplies an unbiased reference gradient $\tilde{\mu}^{\,s}$, while the inner loop reuses it to dramatically reduce variance, yielding the contraction in Lemma~\ref{lem:one-epoch} below.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $L$‑smooth $f_i$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^2\le 2L\bigl[f_i(x)-f_i(y)-\langle\nabla f_i(y),x-y\rangle\bigr].
\tag{4}
\]
\end{lemma}
\begin{proof}
From $L$‑smoothness,
$f_i(x)\le f_i(y)+\langle\nabla f_i(y),x-y\rangle+\frac{L}{2}\|x-y\|^2$.
Rearranging and applying Cauchy–Schwarz gives (4). \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
Conditioned on $x_t$, the estimator $v_t$ defined in (1) satisfies
\[
\E_{i_t}[\,v_t\,|\,x_t]=\nabla f(x_t).
\tag{5}
\]
\end{lemma}
\begin{proof}
\[
\E_{i_t}[v_t|x_t]=\frac1n\sum_{i=1}^{n}\bigl[\nabla f_i(x_t)-\nabla f_i(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}\bigr]
= \nabla f(x_t)-\nabla f(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}
= \nabla f(x_t).
\]
\qedhere
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance}
For any $x_t$,
\[
\E_{i_t}\!\bigl[\|v_t-\nabla f(x_t)\|^2\,\big|\,x_t\bigr]\;\le\;4L\bigl[f(x_t)-f(\tilde{x}^{\,s})\bigr]-2\mu\|x_t-\tilde{x}^{\,s}\|^2 .
\tag{6}
\]
\end{lemma}
\begin{proof}
Using definition (1) and adding/subtracting $\nabla f_i(\tilde{x}^{\,s})$,
\[
v_t-\nabla f(x_t)=\bigl[\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{\,s})\bigr]
-\bigl[\nabla f(x_t)-\nabla f(\tilde{x}^{\,s})\bigr].
\]
Taking squared norm, expectation over $i_t$, and applying Lemma~\ref{lem:smooth} together with strong convexity yields (6). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Let $x^{\star}$ denote the unique minimizer of $f$.

\noindent\textbf{Notation.} For epoch $s$ we write $x_t$ for the inner iterate, $x_{t+1}=x_t-\eta v_t$ as in (2), and $\tilde{x}=\tilde{x}^{\,s}$ for the snapshot.

\begin{proof}
\textbf{Step 1.} Expand the squared distance after one inner update:
\[
\|x_{t+1}-x^{\star}\|^2
= \|x_t-\eta v_t-x^{\star}\|^2
= \|x_t-x^{\star}\|^2-2\eta\langle v_t,x_t-x^{\star}\rangle+\eta^2\|v_t\|^2.
\tag{7}
\]
\textbf{Step 2.} Take expectation w.r.t. $i_t$ conditional on $x_t$ and use Lemma~\ref{lem:unbiased}:
\[
\E_{i_t}\bigl[\langle v_t,x_t-x^{\star}\rangle\mid x_t\bigr]
= \langle\nabla f(x_t),x_t-x^{\star}\rangle .
\tag{8}
\]
\textbf{Step 3.} Apply strong convexity (Assumption~\ref{ass:strong}) to bound the inner product:
\[
\langle\nabla f(x_t),x_t-x^{\star}\rangle
\ge f(x_t)-f(x^{\star})+\frac{\mu}{2}\|x_t-x^{\star}\|^2 .
\tag{9}
\]
\textbf{Step 4.} For the $\|v_t\|^2$ term we use
\[
\|v_t\|^2 = \|\nabla f(x_t)\|^2
+ \|v_t-\nabla f(x_t)\|^2
+2\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle .
\]
Taking conditional expectation and noting $\E_{i_t}[v_t-\nabla f(x_t)\mid x_t]=0$, we obtain
\[
\E_{i_t}[\|v_t\|^2\mid x_t]=\|\nabla f(x_t)\|^2+\E_{i_t}[\|v_t-\nabla f(x_t)\|^2\mid x_t].
\tag{10}
\]
\textbf{Step 5.} Substitute (8)–(10) into (7) and then take total expectation:
\[
\begin{aligned}
\E[\|x_{t+1}-x^{\star}\|^2]
&\le \E[\|x_t-x^{\star}\|^2]
-2\eta\E\!\bigl[f(x_t)-f(x^{\star})\bigr]
-\mu\eta\E\!\bigl[\|x_t-x^{\star}\|^2\bigr]   \\
&\quad +\eta^2\E\!\bigl[\|\nabla f(x_t)\|^2\bigr]
+\eta^2\E\!\bigl[\|v_t-\nabla f(x_t)\|^2\bigr].
\end{aligned}
\tag{11}
\]
\textbf{Step 6.} Use $L$‑smoothness to bound the gradient norm:
\[
\|\nabla f(x_t)\|^2 \le 2L\bigl[f(x_t)-f(x^{\star})\bigr].
\tag{12}
\]
\textbf{Step 7.} Apply the variance bound Lemma~\ref{lem:variance} (6):
\[
\E\!\bigl[\|v_t-\nabla f(x_t)\|^2\bigr]
\le 4L\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
-2\mu\E\!\bigl[\|x_t-\tilde{x}\|^2\bigr].
\tag{13}
\]
\textbf{Step 8.} Combine (11)–(13). After rearranging terms we obtain
\[
\begin{aligned}
\E[\|x_{t+1}-x^{\star}\|^2]
&\le \bigl(1-\mu\eta\bigr)\E[\|x_t-x^{\star}\|^2]  \\
&\quad -2\eta\bigl(1-L\eta\bigr)\E\!\bigl[f(x_t)-f(x^{\star})\bigr]   \\
&\quad +4L\eta^{2}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
-2\mu\eta^{2}\E\!\bigl[\|x_t-\tilde{x}\|^2\bigr].
\end{aligned}
\tag{14}
\]
\textbf{Step 9.} Because $\tilde{x}$ is fixed during the inner loop, sum (14) over $t=0,\dots,m-1$ and telescope the distance terms:
\[
\E[\|x_{m}-x^{\star}\|^2]
\le (1-\mu\eta)^{m}\| \tilde{x}-x^{\star}\|^{2}
-2\eta\bigl(1-L\eta\bigr)\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(x^{\star})\bigr]
+4L\eta^{2}\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr].
\tag{15}
\]
\textbf{Step 10.} Use convexity of $f$ to relate the average of inner iterates to the snapshot:
\[
\frac1m\sum_{t=0}^{m-1} f(x_t) \;\ge\; f\!\bigl(\tfrac1m\sum_{t=0}^{m-1}x_t\bigr)
\;\ge\; f(\tilde{x}) .
\tag{16}
\]
Thus $\sum_{t} \bigl[f(x_t)-f(\tilde{x})\bigr]\ge 0$, allowing us to drop the last positive term in (15). Moreover, by strong convexity,
\[
f(\tilde{x})-f(x^{\star})\;\ge\;\frac{\mu}{2}\|\tilde{x}-x^{\star}\|^{2}.
\tag{17}
\]
\textbf{Step 11.} Apply (17) to the left‑hand side of (15) and rearrange:
\[
\begin{aligned}
\E[f(\tilde{x}^{\,s+1})-f(x^{\star})]
&\le \frac{1}{2\mu\eta\bigl(1- L\eta\bigr)m}
\bigl(1-\mu\eta\bigr)^{m}\,\bigl[f(\tilde{x}^{\,s})-f(x^{\star})\bigr] .
\end{aligned}
\tag{18}
\]
\textbf{Step 12.} Choose $\eta\le \frac{1}{3L}$, which guarantees $1-L\eta\ge \frac{2}{3}$ and $1-\mu\eta\le e^{-\mu\eta}$. Using the inequality $e^{-a}\le \frac{1}{1+a}$ for $a\ge0$, we obtain
\[
(1-\mu\eta)^{m}\le\frac{1}{1+\mu\eta m}.
\tag{19}
\]
Insert (19) into (18) and simplify:
\[
\E[f(\tilde{x}^{\,s+1})-f(x^{\star})]
\le \underbrace{\frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}}_{\displaystyle \rho}\;
\bigl[f(\tilde{x}^{\,s})-f(x^{\star})\bigr].
\tag{20}
\]
\textbf{Step 13.} By induction over epochs $s$,
\[
\E[f(\tilde{x}^{\,s})-f(x^{\star})]\le \rho^{\,s}\bigl[f(\tilde{x}^{\,0})-f(x^{\star})\bigr],
\]
which is exactly the claim (3). \qedhere
\end{proof}

\section*{Rate Derivation (With Constants)}
From (20) the contraction factor is
\[
\rho = \frac{1}{1+\mu\eta(1-2L\eta)m}.
\]
If we set $\eta = \frac{1}{3L}$ (the largest admissible stepsize) and choose $m = 2\lceil L/\mu\rceil$, then
\[
\rho \le \frac{1}{1+\frac{\mu}{3L}\bigl(1-\tfrac{2}{3}\bigr)2\frac{L}{\mu}}
= \frac{1}{1+\frac{2}{9}} = \frac{9}{11}\approx0.818.
\]
Thus after each epoch the expected suboptimality shrinks by at least a factor $0.818$, i.e. linear convergence.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Gradient ($m=1$).} The algorithm reduces to full gradient descent with rate $(1-\mu\eta)^{T}$.
\item \textbf{Large $m$ (infinite inner loop).} The variance term vanishes, and the method behaves like deterministic gradient descent.
\item \textbf{Quadratic Functions.} For $f(x)=\frac12 x^{\top}Ax-b^{\top}x$ with $A\succ0$, the bound becomes tight and the iterates follow a linear system whose spectral radius equals $\rho$.
\end{enumerate}
