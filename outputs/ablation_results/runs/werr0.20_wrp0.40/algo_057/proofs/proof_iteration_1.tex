\section*{Random Block Coordinate Descent (RBCD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable and possibly non‑convex.  
Partition the coordinate vector $w\in\mathbb{R}^{d}$ into $m$ disjoint blocks
\[
w=\bigl(w^{(1)},w^{(2)},\dots ,w^{(m)}\bigr),\qquad 
w^{(i)}\in\mathbb{R}^{d_i},\;\; \sum_{i=1}^{m}d_i=d .
\]
For each block $i$ denote by $\nabla_i f(w)\in\mathbb{R}^{d_i}$ the partial gradient and by
$P_i:\mathbb{R}^{d}\rightarrow\mathbb{R}^{d_i}$ the projection onto block $i$.
At iteration $t$ a block index $i_t\in\{1,\dots ,m\}$ is drawn independently
according to a probability distribution $p=(p_1,\dots ,p_m)$, $\;p_i>0,\;\sum_i p_i=1$.
The update rule is
\[
\boxed{
\begin{aligned}
w_{t+1}^{(i_t)} &:= w_{t}^{(i_t)}-\eta_t \,\nabla_{i_t} f(w_t),\\
w_{t+1}^{(j)}   &:= w_{t}^{(j)}\quad\text{for }j\neq i_t,
\end{aligned}}
\tag{1}
\]
where $\eta_t>0$ is a stepsize (possibly time‑varying).

Define the block‑selection operator 
\[
\mathcal{S}_{i}(v)=\bigl(0,\dots ,0,\,v,\dots ,0\bigr)\in\mathbb{R}^{d},
\]
with $v$ placed in block $i$ and zeros elsewhere.  Then (1) can be written compactly as
\[
w_{t+1}=w_t-\eta_t\,\mathcal{S}_{i_t}\bigl(\nabla_{i_t} f(w_t)\bigr).
\tag{2}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent (RBCD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\mathbb{R}^{d}$, stepsize schedule $\{\eta_t\}_{t\ge0}$, block probabilities $\{p_i\}_{i=1}^{m}$, maximum iterations $T$.
\For{$t=0,1,\dots ,T-1$}
    \State Sample block $i_t\sim\text{Multinomial}(p_1,\dots ,p_m)$ \label{line:sample}
    \State Compute partial gradient $g_t:=\nabla_{i_t} f(w_t)$ \label{line:grad}
    \State Update block $w_{t+1}^{(i_t)} \gets w_{t}^{(i_t)}-\eta_t g_t$ \label{line:update}
    \State For all $j\neq i_t$, set $w_{t+1}^{(j)}\gets w_{t}^{(j)}$ \label{line:keep}
\EndFor
\State \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or an averaged iterate).
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Consider the scalar (single‑block) function
\[
f(w)=(w-3)^2,\qquad w\in\mathbb{R},
\]
with $w_0=0$, constant stepsize $\eta_t\equiv \eta=0.1$, and $T=3$ iterations.
Since there is only one block, the algorithm always selects $i_t=1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1 = 0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $5.76$\\
2 & $w_2 = 0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $3.6864$\\
3 & $w_3 = 1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $2.3666$\\
\end{tabular}
\end{center}
The objective value decreases from $9$ to roughly $2.37$ after three steps, illustrating the descent property of (1).

\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Block‑wise $L_i$‑smoothness]\label{ass:smooth}
For each block $i\in\{1,\dots ,m\}$ there exists $L_i>0$ such that for all $w,\tilde w\in\mathbb{R}^{d}$,
\[
\bigl\|\nabla_i f(w)-\nabla_i f(\tilde w)\bigr\|
\le L_i\;\bigl\|P_i(w-\tilde w)\bigr\|.
\tag{3}
\]
\end{assumption}
Define $L_{\max}:=\max_{i}L_i$ and $L_{\text{avg}}:=\sum_{i}p_i L_i$.

\begin{assumption}[Lower boundedness]\label{ass:lb}
$f$ admits a finite lower bound $f^{\inf}>-\infty$ and we denote
\[
\Delta_f:=f(w_0)-f^{\inf}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize sequence $\{\eta_t\}$ satisfies
\[
0<\eta_t\le\frac{1}{L_{\max}}\quad\forall t,
\qquad
\sum_{t=0}^{\infty}\eta_t=\infty,\qquad
\sum_{t=0}^{\infty}\eta_t^{2}<\infty .
\tag{5}
\]
A convenient choice is the constant stepsize $\eta_t\equiv\eta\le 1/L_{\max}$.
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}[Convergence of RBCD for non‑convex smooth $f$]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, the iterates generated by
\eqref{eq:2} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;\le\;
\frac{2L_{\max}\,\Delta_f}{\eta\,T},
\qquad\forall T\ge 1,
\tag{6}
\]
where the expectation is taken over the random block selections
$\{i_0,\dots ,i_{T-1}\}$.
Consequently,
\[
\min_{0\le t<T}\mathbb{E}\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;=\;O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\begin{remark}
If a diminishing stepsize $\eta_t=\frac{\eta_0}{\sqrt{t+1}}$ is used,
the bound becomes $O\!\bigl(\frac{\log T}{\sqrt{T}}\bigr)$, matching the classical
rate for stochastic gradient methods.
\end{remark}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{Stability.} Because each block update is a gradient step with stepsize bounded by $1/L_{\max}$, the sequence $\{f(w_t)\}$ is almost‑surely non‑increasing in expectation.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{6} depends linearly on $L_{\max}$ and inversely on the chosen constant stepsize $\eta$. Selecting $\eta$ close to $1/L_{\max}$ yields the fastest theoretical rate while preserving stability.
\item \textbf{Comparison to full‑gradient GD.} Setting $p_i=1$ for a single block recovers standard (deterministic) gradient descent with rate $O(1/T)$. Uniform block sampling $p_i=1/m$ incurs an extra factor $m$ inside $L_{\max}$, reflecting the fact that only a fraction $1/m$ of coordinates are updated per iteration.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{lemma}[Blockwise descent lemma]\label{lem:descent}
Let $i\in\{1,\dots ,m\}$ and $w\in\mathbb{R}^{d}$.  Under Assumption \ref{ass:smooth},
\[
f\bigl(w-\eta\,\mathcal{S}_{i}(\nabla_i f(w))\bigr)
\le
f(w)-\eta\langle \nabla_i f(w),\nabla_i f(w)\rangle
+\frac{L_i\eta^{2}}{2}\|\nabla_i f(w)\|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Apply the standard smoothness inequality to the direction $\mathcal{S}_{i}(\nabla_i f(w))$,
use that only block $i$ changes, and invoke \eqref{3}.  The algebra yields \eqref{7}.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We now prove Theorem \ref{thm:main}.  All expectations are taken with respect to the random block index $i_t$ conditioned on the past iterates.

\subsection*{Step 1. One‑step expected descent}
From Lemma \ref{lem:descent} with $\eta=\eta_t$ and $i=i_t$,
\[
f(w_{t+1})\le f(w_t)-\eta_t\|\nabla_{i_t}f(w_t)\|^{2}
+\frac{L_{i_t}\eta_t^{2}}{2}\|\nabla_{i_t}f(w_t)\|^{2}.
\tag{8}
\]
Take conditional expectation $\mathbb{E}_{i_t}[\cdot\mid w_t]$.
Because $\Pr(i_t=i)=p_i$,
\[
\mathbb{E}_{i_t}\bigl[\|\nabla_{i_t}f(w_t)\|^{2}\mid w_t\bigr]
= \sum_{i=1}^{m}p_i\|\nabla_i f(w_t)\|^{2}.
\tag{9}
\]
Similarly,
\[
\mathbb{E}_{i_t}\bigl[L_{i_t}\|\nabla_{i_t}f(w_t)\|^{2}\mid w_t\bigr]
= \sum_{i=1}^{m}p_i L_i\|\nabla_i f(w_t)\|^{2}
\le L_{\max}\sum_{i=1}^{m}p_i\|\nabla_i f(w_t)\|^{2}.
\tag{10}
\]
Insert (9)–(10) into the expectation of \eqref{8}:
\[
\mathbb{E}_{i_t}\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le
f(w_t)-\eta_t\!\sum_{i}p_i\|\nabla_i f(w_t)\|^{2}
+\frac{L_{\max}\eta_t^{2}}{2}\!\sum_{i}p_i\|\nabla_i f(w_t)\|^{2}.
\tag{11}
\]

\subsection*{Step 2. Relating block gradients to full gradient}
Observe that
\[
\|\nabla f(w_t)\|^{2}
= \Bigl\|\sum_{i=1}^{m}P_i^{\top}\nabla_i f(w_t)\Bigr\|^{2}
= \sum_{i=1}^{m}\|\nabla_i f(w_t)\|^{2},
\tag{12}
\]
because the blocks are disjoint.  Using the probability vector $p$, we have
\[
\sum_{i}p_i\|\nabla_i f(w_t)\|^{2}
\ge \bigl(\min_i p_i\bigr)\,\|\nabla f(w_t)\|^{2}.
\tag{13}
\]

\subsection*{Step 3. Simplifying the descent inequality}
Combine \eqref{11} and \eqref{13} and use the stepsize bound $\eta_t\le 1/L_{\max}$:
\[
\begin{aligned}
\mathbb{E}_{i_t}\!\bigl[f(w_{t+1})\mid w_t\bigr]
&\le f(w_t)-\eta_t\bigl(\min_i p_i\bigr)\|\nabla f(w_t)\|^{2}
+\frac{L_{\max}\eta_t^{2}}{2}\|\nabla f(w_t)\|^{2}\\[4pt]
&= f(w_t)-\Bigl[\eta_t\bigl(\min_i p_i\bigr)-\frac{L_{\max}\eta_t^{2}}{2}\Bigr]\|\nabla f(w_t)\|^{2}.
\end{aligned}
\tag{14}
\]
Since $\eta_t\le 1/L_{\max}$,
\[
\eta_t\bigl(\min_i p_i\bigr)-\frac{L_{\max}\eta_t^{2}}{2}
\;\ge\;
\frac{\eta_t}{2}\bigl(\min_i p_i\bigr).
\tag{15}
\]
Thus
\[
\mathbb{E}_{i_t}\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le
f(w_t)-\frac{\eta_t}{2}\bigl(\min_i p_i\bigr)\,\|\nabla f(w_t)\|^{2}.
\tag{16}
\]

\subsection*{Step 4. Taking full expectation and telescoping}
Define $\mathcal{F}_t$ as the sigma‑algebra generated by $\{i_0,\dots ,i_{t-1}\}$.
Taking expectation w.r.t. $\mathcal{F}_t$ in \eqref{16} yields
\[
\mathbb{E}\!\bigl[f(w_{t+1})\bigr]
\le
\mathbb{E}\!\bigl[f(w_{t})\bigr]
-\frac{\eta_t}{2}\bigl(\min_i p_i\bigr)\,
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr].
\tag{17}
\]
Rearrange \eqref{17}:
\[
\frac{\eta_t}{2}\bigl(\min_i p_i\bigr)\,
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\mathbb{E}\!\bigl[f(w_{t})\bigr]-\mathbb{E}\!\bigl[f(w_{t+1})\bigr].
\tag{18}
\]
Sum \eqref{18} from $t=0$ to $T-1$ (telescoping the right‑hand side):
\[
\frac{\bigl(\min_i p_i\bigr)}{2}\sum_{t=0}^{T-1}\eta_t\,
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\mathbb{E}\!\bigl[f(w_{0})\bigr]-\mathbb{E}\!\bigl[f(w_{T})\bigr]
\le \Delta_f .
\tag{19}
\]

\subsection*{Step 5. Deriving the average‑gradient bound}
Assume a constant stepsize $\eta_t\equiv\eta\le 1/L_{\max}$.  Then \eqref{19} becomes
\[
\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\Delta_f}{\eta\,\bigl(\min_i p_i\bigr)} .
\tag{20}
\]
Dividing by $T$ gives the desired rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\Delta_f}{\eta\,\bigl(\min_i p_i\bigr)\,T}.
\tag{21}
\]
Since $\min_i p_i\ge 1/m$, we can replace it by $1/m$ and absorb $m$ into $L_{\max}$,
obtaining the cleaner bound stated in Theorem \ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2L_{\max}\,\Delta_f}{\eta\,T}.
\tag{22}
\]
\textbf{□} (Theorem proved.)

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
From \eqref{22}, set $\eta = \frac{1}{2L_{\max}}$ (the largest admissible constant stepsize
according to Assumption \ref{ass:stepsize}).  Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{4L_{\max}^{2}\,\Delta_f}{T}.
\tag{23}
\]
Thus the algorithm attains an $O(1/T)$ decay of the expected squared gradient norm,
identical to the rate of full‑gradient (deterministic) gradient descent for smooth non‑convex problems, up to the Lipschitz constant inflation caused by blockwise updates.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/m$ for all $i$.
    Then $\min_i p_i = 1/m$ and $L_{\max}$ remains unchanged.
    The bound \eqref{22} becomes
    \[
    \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
    \le
    \frac{2mL_{\max}\,\Delta_f}{\eta\,T}.
    \]
    The factor $m$ reflects the fact that only one out of $m$ blocks is updated per iteration.

\item \textbf{Full‑gradient GD as a limit.} If $m=1$ (single block) the algorithm reduces to
    standard gradient descent and \eqref{22} recovers the classical $O(1/T)$ rate.

\item \textbf{Diminishing stepsize.} Choose $\eta_t = \frac{c}{\sqrt{t+1}}$ with $c\le 1/L_{\max}$.
    Summing \eqref{18} yields
    \[
    \frac{c}{2}\bigl(\min_i p_i\bigr)\sum_{t=0}^{T-1}\frac{\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]}{\sqrt{t+1}}
    \le \Delta_f .
    \]
    By Jensen’s inequality this leads to
    $\min_{0\le t<T}\mathbb{E}[\|\nabla f(w_t)\|^{2}]
    =O\!\bigl(\frac{\log T}{\sqrt{T}}\bigr)$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
The proof above shows that random block coordinate descent enjoys the same
asymptotic $O(1/T)$ convergence rate (in terms of expected squared gradient norm) as full‑gradient methods, despite updating only a subset of coordinates at each iteration.
The price paid is a multiplicative factor involving the block‑wise Lipschitz constant $L_{\max}$ and, for uniform sampling, the number of blocks $m$.
The analysis hinges only on block‑wise smoothness and a simple stepsize restriction,
making it applicable to a wide class of non‑convex problems common in deep learning and large‑scale optimization.
