\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
We consider the convex–concave saddle‑point problem  

\[
\min_{x\in\mathbb R^{n}} \; \max_{y\in\mathbb R^{m}}
\; \mathcal L(x,y):= f(x) + \langle Kx , y\rangle - g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
  \item $f:\mathbb R^{n}\!\to\!\mathbb R\cup\{+\infty\}$ is proper, closed, convex (possibly non‑smooth);
  \item $g^{*}:\mathbb R^{m}\!\to\!\mathbb R\cup\{+\infty\}$ is the convex conjugate of a proper, closed, convex function $g$;
  \item $K\in\mathbb R^{m\times n}$ is a linear operator.
\end{itemize}

The PDHG iteration with step sizes $\tau>0$ (primal) and $\sigma>0$ (dual) is  

\[
\boxed{
\begin{aligned}
y_{k+1} &= \prox{\sigma g^{*}}{\,y_{k}+\sigma K x_{k}\,}, \tag{2a}\\[4pt]
x_{k+1} &= \prox{\tau f}{\,x_{k}-\tau K^{\!\top} y_{k+1}\,}. \tag{2b}
\end{aligned}}
\]

If a relaxation parameter $\theta\in[0,1]$ is used, the extrapolated primal point  

\[
\tilde x_{k}=x_{k}+\theta (x_{k}-x_{k-1})
\]

may replace $x_{k}$ in (2a); the basic version above sets $\theta=0$.

For a quadratic $f$ and a simple scalar $K$, the proximal maps have closed forms, which we exploit in the dry‑run below.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\Require $f$, $g^{*}$, linear operator $K$, step sizes $\tau>0$, $\sigma>0$, initial points $x_{0}\in\mathbb R^{n}$, $y_{0}\in\mathbb R^{m}$
\Ensure Approximate saddle point $(x_{T},y_{T})$
\For{$k=0,\dots,T-1$}
    \State $y_{k+1}\gets \prox{\sigma g^{*}}{\,y_{k}+\sigma K x_{k}\,}$ \Comment{dual gradient step + proximal}
    \State $x_{k+1}\gets \prox{\tau f}{\,x_{k}-\tau K^{\!\top} y_{k+1}\,}$ \Comment{primal proximal step}
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We instantiate (1) with a scalar problem  

\[
\min_{x\in\mathbb R}\;\; (x-3)^{2} \;+\; y\,x \;-\; 0,
\]

i.e. $f(x)=(x-3)^{2}$, $g^{*}(y)=0$, and $K=1$.  
The proximal of $f$ with stepsize $\tau$ is  

\[
\prox{\tau f}{v}= \frac{v+2\tau\cdot 3}{1+2\tau}.
\tag{3}
\]

Since $g^{*}=0$, $\prox{\sigma g^{*}}{u}=u$ (identity).  
Choose $\tau=\sigma=0.1$, initialise $x_{0}=0$, $y_{0}=0$.  
We perform three iterations.

\medskip
\noindent\textbf{Iteration $k=0$}
\begin{align*}
y_{1}&= y_{0}+\sigma K x_{0}=0+0.1\cdot 0 =0,\\
x_{1}&= \prox{\tau f}{\,x_{0}-\tau K^{\!\top}y_{1}\,}
      =\prox{0.1 f}{\,0-0.1\cdot 0\,}
      =\frac{0+2\cdot0.1\cdot 3}{1+2\cdot0.1}
      =\frac{0.6}{1.2}=0.5.
\end{align*}
Objective value $f(x_{1})=(0.5-3)^{2}=6.25$.

\medskip
\noindent\textbf{Iteration $k=1$}
\begin{align*}
y_{2}&= y_{1}+0.1\,x_{1}=0+0.1\cdot 0.5 =0.05,\\
x_{2}&= \prox{0.1 f}{\,x_{1}-0.1 y_{2}\,}
      =\prox{0.1 f}{\,0.5-0.1\cdot0.05\,}
      =\prox{0.1 f}{\,0.495\,}\\
      &=\frac{0.495+0.6}{1.2}= \frac{1.095}{1.2}=0.9125 .
\end{align*}
Objective $f(x_{2})=(0.9125-3)^{2}=4.3527$.

\medskip
\noindent\textbf{Iteration $k=2$}
\begin{align*}
y_{3}&= y_{2}+0.1\,x_{2}=0.05+0.1\cdot0.9125 =0.14125,\\
x_{3}&= \prox{0.1 f}{\,x_{2}-0.1 y_{3}\,}
      =\prox{0.1 f}{\,0.9125-0.1\cdot0.14125\,}
      =\prox{0.1 f}{\,0.898375\,}\\
      &=\frac{0.898375+0.6}{1.2}= \frac{1.498375}{1.2}=1.248646.
\end{align*}
Objective $f(x_{3})=(1.248646-3)^{2}=3.0586$.

The iterates clearly move toward the minimiser $x^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Convexity and Closedness]\label{ass:convex}
$f$ and $g^{*}$ are proper, closed, convex functions.  
\end{assumption}
\begin{assumption}[Lipschitz Continuity of the Linear Operator]\label{ass:K}
The linear operator $K$ satisfies  
\[
\|K\|:=\sup_{\|x\|=1}\|Kx\| < \infty .
\]
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{ass:steps}
The primal and dual steps satisfy  
\[
\tau\sigma \, \|K\|^{2} < 1 .
\tag{4}
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists at least one pair $(x^{\star},y^{\star})$ solving (1).  
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Convergence]\label{thm:conv}
Let Assumptions \ref{ass:convex}–\ref{ass:saddle} hold and let $\{(x_{k},y_{k})\}_{k\ge 0}$ be generated by \eqref{2}.  
Define the ergodic averages  

\[
\bar x_{K}:=\frac{1}{K}\sum_{k=0}^{K-1} x_{k},
\qquad
\bar y_{K}:=\frac{1}{K}\sum_{k=0}^{K-1} y_{k}.
\]

Then for any saddle point $(x^{\star},y^{\star})$,
\[
\mathcal L(\bar x_{K},y^{\star})-\mathcal L(x^{\star},\bar y_{K})
\;\le\;
\frac{1}{K}\Bigl(
\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
+\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}
\Bigr).
\tag{5}
\]
Consequently,
\[
\mathcal L(\bar x_{K},\bar y_{K})- \mathcal L(x^{\star},y^{\star})
\le
\frac{C}{K},
\qquad C:=\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
+\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}.
\tag{6}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{ass:steps} guarantees that the combined operator in \eqref{2} is non‑expansive, which yields the monotone decrease of the Lyapunov function used in the proof.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{5} explicitly depends on $\tau$ and $\sigma$; a larger product $\tau\sigma$ (while respecting \eqref{ass:steps}) reduces the constant $C$ but may jeopardise stability.
\item \textbf{Comparison to Baselines.} For smooth $f$ and $g$, gradient descent attains $O(1/k)$ in the objective, whereas PDHG obtains the same rate for a broader class of non‑smooth problems without requiring a global Lipschitz gradient.
\end{itemize}

\section*{Discussion}
The theorem shows that PDHG attains the optimal $O(1/k)$ ergodic rate for convex–concave saddle problems under a simple stepsize restriction. The proof is constructive: it builds a potential that telescopes across iterations, a technique common to primal–dual analyses (e.g., Chambolle–Pock). The result extends verbatim to the relaxed variant with $\theta\in[0,1]$, at the price of a modified stepsize condition $\tau\sigma\|K\|^{2}< (1-\theta)^{2}$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Moreau Decomposition]\label{lem:moreau}
For any proper, closed, convex $\phi$ and $\lambda>0$,
\[
v = \prox{\lambda\phi}{v} + \lambda\,\prox{\phi^{*}/\lambda}{v/\lambda}.
\tag{7}
\]
\end{lemma}
\begin{lemma}[Non‑expansiveness of Prox]\label{lem:nonexp}
For any $u,v$,
\[
\bigl\|\prox{\lambda\phi}{u}-\prox{\lambda\phi}{v}\bigr\|
\le \|u-v\|.
\tag{8}
\]
\end{lemma}
Both lemmas are standard; proofs can be found in \cite{Combettes2011}.

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $(x^{\star},y^{\star})$ be any saddle point of \eqref{1}.  
Define the Lyapunov quantity  

\[
\Phi_{k}:=
\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
+
\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 1. Dual update expansion.}  
From the definition of the proximal map,
\[
y_{k+1}= \arg\min_{y}\Bigl\{ g^{*}(y)+\frac{1}{2\sigma}\|y-(y_{k}+\sigma Kx_{k})\|^{2}\Bigr\}.
\tag{10}
\]
First‑order optimality gives
\[
0\in \partial g^{*}(y_{k+1})+\frac{1}{\sigma}\bigl(y_{k+1}-y_{k}-\sigma Kx_{k}\bigr).
\tag{11}
\]
Rearranging,
\[
\frac{1}{\sigma}\bigl(y_{k}-y_{k+1}\bigr)-Kx_{k}\in \partial g^{*}(y_{k+1}).
\tag{12}
\]
\noindent\underline{[Step 1]}

\medskip
\noindent\textbf{Step 2. Primal update expansion.}  
Analogously,
\[
x_{k+1}= \arg\min_{x}\Bigl\{ f(x)+\frac{1}{2\tau}\|x-(x_{k}-\tau K^{\!\top}y_{k+1})\|^{2}\Bigr\},
\tag{13}
\]
hence
\[
0\in \partial f(x_{k+1})+\frac{1}{\tau}\bigl(x_{k+1}-x_{k}+\tau K^{\!\top}y_{k+1}\bigr),
\tag{14}
\]
or
\[
\frac{1}{\tau}\bigl(x_{k}-x_{k+1}\bigr)-K^{\!\top}y_{k+1}\in \partial f(x_{k+1}).
\tag{15}
\]
\noindent\underline{[Step 2]}

\medskip
\noindent\textbf{Step 3. Monotone inequality from convexity.}  
By convexity of $f$ and $g^{*}$,
\[
f(x_{k+1})-f(x^{\star})\le
\left\langle \frac{1}{\tau}(x_{k}-x_{k+1})-K^{\!\top}y_{k+1},\, x_{k+1}-x^{\star}\right\rangle,
\tag{16}
\]
\[
g^{*}(y_{k+1})-g^{*}(y^{\star})\le
\left\langle \frac{1}{\sigma}(y_{k}-y_{k+1})-Kx_{k},\, y_{k+1}-y^{\star}\right\rangle .
\tag{17}
\]
\noindent\underline{[Step 3]}

\medskip
\noindent\textbf{Step 4. Combine (16)–(17) with the saddle‑point identity.}  
Recall that for any $(x,y)$,
\[
\mathcal L(x,y)-\mathcal L(x^{\star},y^{\star})
= f(x)-f(x^{\star}) + g^{*}(y)-g^{*}(y^{\star}) + \langle Kx, y^{\star}\rangle - \langle Kx^{\star},y\rangle .
\tag{18}
\]
Using $Kx^{\star}=0$ (which holds for any saddle point of \eqref{1}) and substituting (16)–(17) into (18) yields
\[
\begin{aligned}
\mathcal L(x_{k+1},y_{k+1})-\mathcal L(x^{\star},y^{\star})
&\le
\frac{1}{\tau}\langle x_{k}-x_{k+1},\,x_{k+1}-x^{\star}\rangle
+\frac{1}{\sigma}\langle y_{k}-y_{k+1},\,y_{k+1}-y^{\star}\rangle  \\[2mm]
&\qquad
- \langle Kx_{k},\,y_{k+1}-y^{\star}\rangle
- \langle K^{\!\top}y_{k+1},\,x_{k+1}-x^{\star}\rangle .
\end{aligned}
\tag{19}
\]
\noindent\underline{[Step 4]}

\medskip
\noindent\textbf{Step 5. Rearrangement of bilinear terms.}  
Note that 
\[
\langle Kx_{k},\,y_{k+1}\rangle
= \langle x_{k},\,K^{\!\top}y_{k+1}\rangle .
\tag{20}
\]
Hence the last two inner products in (19) cancel:
\[
- \langle Kx_{k},\,y_{k+1}\rangle
- \langle K^{\!\top}y_{k+1},\,x_{k+1}\rangle
= -\langle x_{k},K^{\!\top}y_{k+1}\rangle
-\langle K^{\!\top}y_{k+1},x_{k+1}\rangle
= -\langle K^{\!\top}y_{k+1},\,x_{k}+x_{k+1}\rangle .
\tag{21}
\]
Similarly,
\[
\langle Kx_{k},\,y^{\star}\rangle
+ \langle K^{\!\top}y_{k+1},\,x^{\star}\rangle
= \langle K^{\!\top}y_{k+1},\,x^{\star}\rangle
+ \langle Kx_{k},\,y^{\star}\rangle .
\tag{22}
\]
Plugging (21)–(22) into (19) and regrouping yields
\[
\begin{aligned}
\mathcal L(x_{k+1},y_{k+1})-\mathcal L(x^{\star},y^{\star})
&\le
\frac{1}{2\tau}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr)\\
&\quad
+\frac{1}{2\sigma}\bigl(\|y_{k}-y^{\star}\|^{2}-\|y_{k+1}-y^{\star}\|^{2}-\|y_{k}-y_{k+1}\|^{2}\bigr)\\
&\quad
+\langle K(x_{k+1}-x_{k}),\,y_{k+1}-y^{\star}\rangle .
\end{aligned}
\tag{23}
\]
\noindent\underline{[Step 5]}

\medskip
\noindent\textbf{Step 6. Bounding the coupling term.}  
Using Cauchy–Schwarz and Young’s inequality with a scalar $\alpha>0$,
\[
\langle K(x_{k+1}-x_{k}),\,y_{k+1}-y^{\star}\rangle
\le
\frac{\alpha}{2}\|K(x_{k+1}-x_{k})\|^{2}
+\frac{1}{2\alpha}\|y_{k+1}-y^{\star}\|^{2}.
\tag{24}
\]
Because $\|Kz\|\le \|K\|\|z\|$, we have $\|K(x_{k+1}-x_{k})\|^{2}\le \|K\|^{2}\|x_{k+1}-x_{k}\|^{2}$.  
Choose $\alpha = \tau\sigma\|K\|^{2}$ (which is $<1$ by Assumption \ref{ass:steps}). Then (24) becomes
\[
\langle K(x_{k+1}-x_{k}),\,y_{k+1}-y^{\star}\rangle
\le
\frac{\tau\sigma\|K\|^{2}}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\tau\sigma\|K\|^{2}}\|y_{k+1}-y^{\star}\|^{2}.
\tag{25}
\]
\noindent\underline{[Step 6]}

\medskip
\noindent\textbf{Step 7. Insert (25) into (23).}  
After substitution we obtain
\[
\begin{aligned}
\mathcal L(x_{k+1},y_{k+1})-\mathcal L(x^{\star},y^{\star})
&\le
\Phi_{k}-\Phi_{k+1}\\
&\quad
-\Bigl(\frac{1}{2\tau}-\frac{\tau\sigma\|K\|^{2}}{2}\Bigr)\|x_{k+1}-x_{k}\|^{2}
-\Bigl(\frac{1}{2\sigma}-\frac{1}{2\tau\sigma\|K\|^{2}}\Bigr)\|y_{k+1}-y^{\star}\|^{2}.
\end{aligned}
\tag{26}
\]
Since $\tau\sigma\|K\|^{2}<1$, both brackets are non‑negative, hence we can drop the last two negative terms to obtain a simple descent inequality
\[
\mathcal L(x_{k+1},y_{k+1})-\mathcal L(x^{\star},y^{\star})
\le \Phi_{k}-\Phi_{k+1}.
\tag{27}
\]
\noindent\underline{[Step 7]}

\medskip
\noindent\textbf{Step 8. Telescoping sum.}  
Summing \eqref{27} from $k=0$ to $K-1$ gives
\[
\sum_{k=0}^{K-1}\bigl[\mathcal L(x_{k+1},y_{k+1})-\mathcal L(x^{\star},y^{\star})\bigr]
\le \Phi_{0}-\Phi_{K}
\le \Phi_{0}.
\tag{28}
\]
Dividing by $K$ and using convexity of $\mathcal L$ in each argument yields the ergodic bound
\[
\mathcal L(\bar x_{K},\bar y_{K})-\mathcal L(x^{\star},y^{\star})
\le \frac{\Phi_{0}}{K}
= \frac{1}{K}\Bigl(\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
+\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}\Bigr),
\]
which is exactly \eqref{5}. This completes the proof of Theorem \ref{thm:conv}.
\noindent\underline{[Step 8]}

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:conv} we have the explicit constant  

\[
C = \frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
    +\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}.
\]

If the initial points are zero and the optimal solution satisfies $\|x^{\star}\|\le R_{x}$, $\|y^{\star}\|\le R_{y}$, then  

\[
C \le \frac{R_{x}^{2}}{2\tau} + \frac{R_{y}^{2}}{2\sigma}.
\]

Hence the ergodic gap decays as $O\!\bigl(\frac{1}{K}\bigr)$ with a prefactor that can be tuned by $\tau$ and $\sigma$ under the constraint \eqref{ass:steps}.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Descent.} If $g^{*}=0$ and $K=I$, then $y_{k}\equiv0$ and \eqref{2b} reduces to the proximal gradient step for $f$, recovering the classic $O(1/k)$ rate for convex (possibly non‑smooth) $f$.
\item \textbf{Linear Equality Constraints.} For $g^{*}=\iota_{\{0\}}$ (indicator of $\{0\}$) the dual update enforces feasibility $Kx=0$, yielding the well‑known PDHG method for linearly constrained convex programs.
\item \textbf{Strong Convexity.} If $f$ is $\mu$‑strongly convex, one can sharpen the analysis to obtain a linear rate $O\bigl((1-\sqrt{\mu\tau})^{k}\bigr)$, see \cite{Chambolle2011}.
\end{itemize}
