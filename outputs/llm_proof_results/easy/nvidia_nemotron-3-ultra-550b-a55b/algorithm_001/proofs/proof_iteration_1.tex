\section*{Algorithm Name}
Stochastic Gradient Descent (SGD)

\section*{Mathematical Formulation}
We consider the stochastic optimization problem
\begin{equation}
    \min_{w \in \mathbb{R}^d} \; f(w) \triangleq \mathbb{E}_{z \sim \mathcal{D}} \bigl[ f(w; z) \bigr],
\end{equation}
where each realization $f(\cdot; z)$ is differentiable. In the finite-sum setting with $n$ samples, this becomes
\begin{equation}
    \min_{w \in \mathbb{R}^d} \; f(w) = \frac{1}{n} \sum_{i=1}^n f_i(w).
\end{equation}
At iteration $t$, the algorithm samples a single index $i_t$ uniformly from $\{1,\dots,n\}$ (or draws $z_t \sim \mathcal{D}$) and computes a stochastic gradient
\begin{equation}
    g_t = \nabla f_{i_t}(w_t) \quad \bigl(\text{or } \nabla f(w_t; z_t)\bigr).
\end{equation}
The parameters are updated via
\begin{equation}
    w_{t+1} = w_t - \eta_t g_t,
\end{equation}
where $\eta_t > 0$ is the learning rate (step size). The learning rate may be constant ($\eta_t = \eta$) or follow a decreasing schedule (e.g., $\eta_t = c/\sqrt{t}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Gradient Descent}
\begin{algorithmic}
\STATE \textbf{Input:} initial point $w_0 \in \mathbb{R}^d$, learning rate schedule $\{\eta_t\}_{t=0}^{T-1}$, maximum iterations $T$
\STATE \textbf{Output:} final iterate $w_T$
\FOR{$t = 0, 1, \dots, T-1$}
    \STATE Sample $i_t$ uniformly from $\{1,\dots,n\}$ (or $z_t \sim \mathcal{D}$)
    \STATE Compute stochastic gradient $g_t = \nabla f_{i_t}(w_t)$
    \STATE Update $w_{t+1} = w_t - \eta_t g_t$
\ENDFOR
\STATE \textbf{return} $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the deterministic function $f(w) = (w-3)^2$ (a single data point, so the stochastic gradient equals the true gradient). Initialize $w_0 = 0$, constant step size $\eta = 0.1$, and run for $3$ iterations.

\begin{itemize}
    \item \textbf{Iteration 0:} $w_0 = 0$ \\
          Gradient: $g_0 = \nabla f(w_0) = 2(w_0-3) = -6$ \\
          Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1 \times (-6) = 0.6$ \\
          Objective: $f(w_1) = (0.6-3)^2 = 5.76$
    \item \textbf{Iteration 1:} $w_1 = 0.6$ \\
          Gradient: $g_1 = 2(w_1-3) = -4.8$ \\
          Update: $w_2 = 0.6 - 0.1 \times (-4.8) = 1.08$ \\
          Objective: $f(w_2) = (1.08-3)^2 = 3.6864$
    \item \textbf{Iteration 2:} $w_2 = 1.08$ \\
          Gradient: $g_2 = 2(w_2-3) = -3.84$ \\
          Update: $w_3 = 1.08 - 0.1 \times (-3.84) = 1.464$ \\
          Objective: $f(w_3) = (1.464-3)^2 \approx 2.3593$
\end{itemize}
The iterates move toward the minimizer $w^* = 3$ and the objective value decreases monotonically.

\section*{Assumptions}
We analyze SGD under the following standard assumptions for non-convex smooth objectives.

\begin{enumerate}
    \item \textbf{$L$-Smoothness:} The objective $f$ is differentiable and its gradient is $L$-Lipschitz continuous:
    \begin{equation}
        \|\nabla f(x) - \nabla f(y)\| \le L \|x - y\|, \quad \forall x,y \in \mathbb{R}^d.
    \end{equation}
    Equivalently, for all $x,y$,
    \begin{equation}
        f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2} \|y-x\|^2. \tag{A1}
    \end{equation}
    \item \textbf{Unbiased Stochastic Gradients:} The stochastic gradient $g_t$ satisfies
    \begin{equation}
        \mathbb{E}[g_t \mid w_t] = \nabla f(w_t). \tag{A2}
    \end{equation}
    \item \textbf{Bounded Variance:} The variance of the stochastic gradient is uniformly bounded by $\sigma^2$:
    \begin{equation}
        \mathbb{E}\bigl[ \|g_t - \nabla f(w_t)\|^2 \mid w_t \bigr] \le \sigma^2. \tag{A3}
    \end{equation}
    \item \textbf{Lower Bounded Objective:} There exists $f^* > -\infty$ such that $f(w) \ge f^*$ for all $w \in \mathbb{R}^d$. \tag{A4}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
Under Assumptions (A1)--(A4), let $\{w_t\}_{t=0}^{T-1}$ be the iterates produced by SGD with a constant step size $\eta \le 1/L$. Then
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}\bigl[ \|\nabla f(w_t)\|^2 \bigr]
    \le \frac{2\bigl(f(w_0) - f^*\bigr)}{\eta T} + L \eta \sigma^2.
\end{equation}
Choosing $\eta = \min\!\left(\frac{1}{L}, \frac{\sqrt{2(f(w_0)-f^*)}}{\sigma\sqrt{L T}}\right)$ yields the convergence rate
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}\bigl[ \|\nabla f(w_t)\|^2 \bigr]
    = \mathcal{O}\!\left( \frac{1}{\sqrt{T}} \right).
\end{equation}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stationarity Convergence:} SGD converges to a first-order stationary point in expectation; the average squared gradient norm decays as $O(1/\sqrt{T})$.
    \item \textbf{Step Size Sensitivity:} The bound reveals a trade-off: large $\eta$ accelerates initial progress but increases the asymptotic error floor $L\eta\sigma^2$; small $\eta$ reduces the floor but slows the $1/(\eta T)$ decay.
    \item \textbf{Variance Dependence:} The asymptotic error is proportional to the stochastic gradient variance $\sigma^2$. Variance reduction techniques (e.g., SVRG, SARAH) can improve this to $O(1/T)$.
    \item \textbf{Comparison to Gradient Descent:} Full-batch gradient descent achieves $O(1/T)$ for non-convex smooth functions. SGD's slower rate is the price of using single-sample gradients.
\end{itemize}

\section*{Discussion}
The analysis assumes i.i.d. sampling and uniform variance bound. In practice, step size schedules like $\eta_t \propto 1/\sqrt{t}$ or adaptive methods (Adam, RMSProp) are used. The $O(1/\sqrt{T})$ rate is tight for general non-convex stochastic optimization without further assumptions. For convex objectives, SGD achieves $O(1/\sqrt{T})$ in function value; for strongly convex objectives, $O(1/T)$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]
If $f$ is $L$-smooth, then for any $x,y \in \mathbb{R}^d$,
\begin{equation}
    f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2} \|y-x\|^2.
\end{equation}
\end{lemma}
\begin{lemma}[Variance Decomposition]
For any random vector $g$ with $\mathbb{E}[g] = \mu$,
\begin{equation}
    \mathbb{E}[\|g\|^2] = \|\mu\|^2 + \mathbb{E}[\|g - \mu\|^2].
\end{equation}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}
We now prove the Main Convergence Theorem. Let $\mathcal{F}_t$ denote the $\sigma$-algebra generated by $\{w_0, i_0, \dots, i_{t-1}\}$ (i.e., all randomness up to iteration $t$). Note that $w_t$ is $\mathcal{F}_t$-measurable.

\begin{proof}
\textbf{Step 1:} Apply the Descent Lemma (A1) with $x = w_t$, $y = w_{t+1} = w_t - \eta g_t$:
\begin{equation}
    f(w_{t+1}) \le f(w_t) + \langle \nabla f(w_t), -\eta g_t \rangle + \frac{L}{2} \| -\eta g_t \|^2. \tag{Step 1}
\end{equation}

\textbf{Step 2:} Simplify the right-hand side:
\begin{equation}
    f(w_{t+1}) \le f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L\eta^2}{2} \|g_t\|^2. \tag{Step 2}
\end{equation}

\textbf{Step 3:} Take conditional expectation $\mathbb{E}[\cdot \mid \mathcal{F}_t]$. Since $w_t$ and $\nabla f(w_t)$ are $\mathcal{F}_t$-measurable, and using unbiasedness (A2):
\begin{align}
    \mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t]
    &\le f(w_t) - \eta \langle \nabla f(w_t), \mathbb{E}[g_t \mid \mathcal{F}_t] \rangle + \frac{L\eta^2}{2} \mathbb{E}[\|g_t\|^2 \mid \mathcal{F}_t] \\
    &= f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L\eta^2}{2} \mathbb{E}[\|g_t\|^2 \mid \mathcal{F}_t]. \tag{Step 3}
\end{align}

\textbf{Step 4:} Decompose $\mathbb{E}[\|g_t\|^2 \mid \mathcal{F}_t]$ using the variance decomposition lemma and bounded variance (A3):
\begin{align}
    \mathbb{E}[\|g_t\|^2 \mid \mathcal{F}_t]
    &= \|\nabla f(w_t)\|^2 + \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 \mid \mathcal{F}_t] \\
    &\le \|\nabla f(w_t)\|^2 + \sigma^2. \tag{Step 4}
\end{align}

\textbf{Step 5:} Substitute the bound from Step 4 into the inequality of Step 3:
\begin{align}
    \mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t]
    &\le f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L\eta^2}{2} \bigl( \|\nabla f(w_t)\|^2 + \sigma^2 \bigr) \\
    &= f(w_t) - \eta \left(1 - \frac{L\eta}{2}\right) \|\nabla f(w_t)\|^2 + \frac{L\eta^2\sigma^2}{2}. \tag{Step 5}
\end{align}

\textbf{Step 6:} Since $\eta \le 1/L$, we have $1 - \frac{L\eta}{2} \ge \frac{1}{2}$. Thus,
\begin{equation}
    \mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t] \le f(w_t) - \frac{\eta}{2} \|\nabla f(w_t)\|^2 + \frac{L\eta^2\sigma^2}{2}. \tag{Step 6}
\end{equation}

\textbf{Step 7:} Rearrange to isolate the gradient norm term:
\begin{equation}
    \frac{\eta}{2} \|\nabla f(w_t)\|^2 \le f(w_t) - \mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t] + \frac{L\eta^2\sigma^2}{2}. \tag{Step 7}
\end{equation}

\textbf{Step 8:} Take full expectation (tower property) and sum over $t = 0$ to $T-1$:
\begin{align}
    \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2]
    &\le \sum_{t=0}^{T-1} \Bigl( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \Bigr) + \frac{T L\eta^2\sigma^2}{2} \\
    &= \mathbb{E}[f(w_0)] - \mathbb{E}[f(w_T)] + \frac{T L\eta^2\sigma^2}{2}. \tag{Step 8}
\end{align}

\textbf{Step 9:} Use the lower bound $f(w_T) \ge f^*$ (A4) and $\mathbb{E}[f(w_0)] = f(w_0)$ (deterministic initialization):
\begin{equation}
    \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \le f(w_0) - f^* + \frac{T L\eta^2\sigma^2}{2}. \tag{Step 9}
\end{equation}

\textbf{Step 10:} Divide both sides by $\eta T/2$ to obtain the average squared gradient norm bound:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \le \frac{2(f(w_0) - f^*)}{\eta T} + L\eta\sigma^2. \tag{Step 10}
\end{equation}
This completes the proof of the main inequality.
\end{proof}

\section*{Rate Derivation (With Constants)}
To obtain the optimal constant step size, minimize the right-hand side of the bound in Step 10 as a function of $\eta$ (subject to $\eta \le 1/L$). The unconstrained minimizer of $\frac{2\Delta}{\eta T} + L\eta\sigma^2$ (where $\Delta = f(w_0)-f^*$) is
\begin{equation}
    \eta^* = \sqrt{\frac{2\Delta}{L\sigma^2 T}}.
\end{equation}
If $\eta^* \le 1/L$, i.e., $T \ge \frac{2\Delta L}{\sigma^2}$, we set $\eta = \eta^*$ and obtain
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \le 2\sigma\sqrt{\frac{2L\Delta}{T}} = \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\end{equation}
If $T < \frac{2\Delta L}{\sigma^2}$, the constraint $\eta \le 1/L$ is active; we set $\eta = 1/L$ and get
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \le \frac{2L\Delta}{T} + \frac{\sigma^2}{L} \le \frac{2L\Delta}{T} + \frac{\sigma^2}{L},
\end{equation}
which is also $\mathcal{O}(1/\sqrt{T})$ for large $T$ (since $1/T \le 1/\sqrt{T}$). Thus, in all cases the rate is $\mathcal{O}(1/\sqrt{T})$.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Deterministic Gradient Descent ($\sigma=0$):} The bound becomes $\frac{2\Delta}{\eta T}$. Choosing $\eta = 1/L$ gives $\frac{2L\Delta}{T}$, recovering the standard $O(1/T)$ rate for non-convex smooth GD.
    \item \textbf{Decreasing Step Size $\eta_t = c/\sqrt{t+1}$:} A similar analysis yields $\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2] = \mathcal{O}(\log T / \sqrt{T})$.
    \item \textbf{Convex Objectives:} If $f$ is additionally convex, one can bound the expected suboptimality $\mathbb{E}[f(\bar{w}_T) - f^*]$ where $\bar{w}_T$ is a weighted average of iterates, achieving $O(1/\sqrt{T})$.
    \item \textbf{Strongly Convex Objectives:} With $\eta_t \propto 1/t$, SGD achieves $\mathbb{E}[f(w_T) - f^*] = \mathcal{O}(1/T)$.
\end{itemize}
