\section*{Algorithm Name}
\textbf{SAGA (Stochastic Average Gradient Augmented).}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$ where each $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is convex and $L$–smooth.  
For each $i$ we keep a \emph{reference point} $\phi_i^k\in\mathbb{R}^d$ and its stored gradient $g_i^k:=\nabla f_i(\phi_i^k)$.  
At iteration $k$ a random index $i_k\in\{1,\dots,n\}$ is drawn uniformly.

\[
\begin{aligned}
v_k &:= \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)+\frac{1}{n}\sum_{j=1}^{n}\nabla f_j(\phi_j^k) ,\\[4pt]
x_{k+1} &:= x_k-\alpha v_k ,\\[4pt]
\phi_{i_k}^{k+1} &:= x_k ,\qquad
\phi_j^{k+1}:=\phi_j^{k}\;\;(j\neq i_k),\\[4pt]
g_{i_k}^{k+1} &:= \nabla f_{i_k}(x_k),\qquad
g_j^{k+1}:=g_j^{k}\;(j\neq i_k).
\end{aligned}
\]

$\alpha>0$ is the constant stepsize (learning rate).  
The term $\frac{1}{n}\sum_{j=1}^{n}g_j^{k}$ is the \emph{average table gradient}.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\Require Number of component functions $n$, stepsize $\alpha>0$, max iterations $K$.
\State Choose an initial point $x_0\in\mathbb{R}^d$.
\For{$i=1,\dots,n$}
    \State $\phi_i^{0}\gets x_0$ \Comment{initial reference points}
    \State $g_i^{0}\gets \nabla f_i(x_0)$
\EndFor
\State $\bar g^{0}\gets \frac{1}{n}\sum_{i=1}^{n}g_i^{0}$ \Comment{average table gradient}
\For{$k=0$ \textbf{to} $K-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State $v_k\gets \nabla f_{i_k}(x_k)-g_{i_k}^{k}+\bar g^{k}$
    \State $x_{k+1}\gets x_k-\alpha v_k$
    \State $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x_k)$
    \State $\phi_{i_k}^{k+1}\gets x_k$
    \State $\bar g^{k+1}\gets \bar g^{k}+\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=\frac{1}{1}\,(w-3)^2,\qquad f_1(w)=(w-3)^2,
\]
so $n=1$, $L=2$, and the unique minimiser is $w^\star=3$.

$(0.6-3)^2=5.76$\\

The objective value decreases from $9$ to $2.37$ after three iterations.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:conv-smooth}
Each component $f_i$ is convex and $L$–smooth, i.e.
\[
f_i(y)\le f_i(x)+\inner{\nabla f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^2,\qquad\forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
The average function $f$ is $\mu$–strongly convex:
\[
f(y)\ge f(x)+\inner{\nabla f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^2,\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:step}
The constant stepsize satisfies 
\[
0<\alpha\le\frac{1}{3L}.
\]
\end{assumption}
All expectations are taken with respect to the random index $i_k$ conditioned on the past $\mathcal{F}_k:=\sigma\{x_0,\phi_i^0, i_0,\dots,i_{k-1}\}$.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SAGA]\label{thm:linear}
Assume \ref{ass:conv-smooth}–\ref{ass:step} and that $f$ is $\mu$–strongly convex (\ref{ass:strong}).  
Define the Lyapunov function
\[
\Psi_k:=\E\!\left[f(x_k)-f(x^\star)+\frac{c}{n}\sum_{i=1}^{n}\norm{\phi_i^{k}-x^\star}^{2}\right],
\]
with $c:=\frac{1}{2\alpha}$.  
Then for the stepsize $\alpha\le\frac{1}{3L}$ we have
\[
\Psi_{k+1}\le\bigl(1-\rho\bigr)\Psi_k,\qquad 
\rho:=\min\!\Bigl\{\frac{\alpha\mu}{2},\,\frac{1}{2}\Bigr\},
\]
and consequently
\[
\E\!\bigl[f(x_k)-f(x^\star)\bigr]\le (1-\rho)^k\Bigl[f(x_0)-f(x^\star)+\frac{c}{n}\sum_{i=1}^{n}\norm{\phi_i^{0}-x^\star}^{2}\Bigr].
\]
Thus SAGA attains a geometric rate $O\!\bigl((1-\rho)^k\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Unbiasedness.} The SAGA direction satisfies 
\[
\E\!\bigl[v_k\mid\mathcal{F}_k\bigr]=\nabla f(x_k),
\]
so it is an unbiased estimator of the true gradient.
\item \textbf{Variance reduction.} The variance of $v_k$ decays as the stored table $\{\phi_i^k\}$ approaches $x^\star$, which is reflected in the Lyapunov term $\frac{c}{n}\sum_i\|\phi_i^k-x^\star\|^2$.
\item \textbf{Hyperparameter sensitivity.} The stepsize bound $1/(3L)$ is larger than the $1/(2L)$ bound required for plain SGD, thanks to the variance‐reduction mechanism.
\item \textbf{Comparison to baselines.} For convex (non‑strong) $f$, the same analysis yields the $O(1/k)$ rate of SAG/SVRG, while plain SGD only guarantees $O(1/\sqrt{k})$ under identical assumptions.
\end{itemize}

\section*{Discussion}
The theorem shows that, under the standard smooth‑strongly‑convex setting, SAGA enjoys a linear convergence rate comparable to deterministic gradient descent but with a per‑iteration cost independent of $n$. The proof hinges on a carefully crafted Lyapunov function that couples the optimisation error with the “memory error’’ of the gradient table. The constant $c=1/(2\alpha)$ balances these two parts; any other choice leads to a weaker contraction factor.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$–smooth then for all $x,y$
\[
\norm{\nabla f_i(x)-\nabla f_i(y)}^2\le 2L\bigl(f_i(x)-f_i(y)-\inner{\nabla f_i(y)}{x-y}\bigr).
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$–smoothness; see e.g. Nesterov (2004). \(\square\)
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
For any $k$,
\[
\E\!\bigl[v_k\mid\mathcal{F}_k\bigr]=\nabla f(x_k).
\]
\end{lemma}
\begin{proof}
Conditioning on $\mathcal{F}_k$, the index $i_k$ is uniform, hence
\[
\E\!\bigl[\nabla f_{i_k}(x_k)-g_{i_k}^k\bigr]=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(x_k)-\frac{1}{n}\sum_{i=1}^{n}g_i^k.
\]
Adding the table average $\bar g^k=\frac{1}{n}\sum_i g_i^k$ yields $\nabla f(x_k)$. \(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  
All expectations are taken w.r.t.\ $i_k$ conditioned on $\mathcal{F}_k$.

\noindent\textbf{Step 1.}  Expand the distance to the optimum after one update.
\[
\begin{aligned}
\norm{x_{k+1}-x^\star}^2 
&= \norm{x_k-\alpha v_k-x^\star}^2  \\
&= \norm{x_k-x^\star}^2-2\alpha\inner{v_k}{x_k-x^\star}+\alpha^2\norm{v_k}^2. \tag{1}
\end{aligned}
\]
\textbf{[Step 1]}

\noindent\textbf{Step 2.}  Take expectation and replace $v_k$ by its unbiased part.
Using Lemma~\ref{lem:unbiased} and convexity,
\[
\E\!\bigl[\inner{v_k}{x_k-x^\star}\mid\mathcal{F}_k\bigr]
= \inner{\nabla f(x_k)}{x_k-x^\star}
\ge f(x_k)-f(x^\star). \tag{2}
\]
\textbf{[Step 2]}

\noindent\textbf{Step 3.}  Upper‑bound $\E[\norm{v_k}^2\mid\mathcal{F}_k]$.
Write $v_k = (\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k))
+ \bar g^k$ and add‑subtract $\nabla f_{i_k}(\phi_{i_k}^k)$.
\[
\begin{aligned}
\E\!\bigl[\norm{v_k}^2\mid\mathcal{F}_k\bigr]
&= \E\!\Bigl[\norm{\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)}^2\Bigr]
      +\norm{\bar g^k}^2\\
&\quad+2\,\E\!\Bigl\langle \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k),\;\bar g^k\Bigr\rangle .
\end{aligned}
\tag{3}
\]
\textbf{[Step 3]}

\noindent\textbf{Step 4.}  Bound the first term using Lemma~\ref{lem:smooth}.
Since $i_k$ is uniform,
\[
\begin{aligned}
\E\!\Bigl[\norm{\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)}^2\mid\mathcal{F}_k\Bigr]
&= \frac{1}{n}\sum_{i=1}^{n}\norm{\nabla f_i(x_k)-\nabla f_i(\phi_i^k)}^2\\
&\le \frac{2L}{n}\sum_{i=1}^{n}\Bigl(f_i(x_k)-f_i(\phi_i^k)-\inner{\nabla f_i(\phi_i^k)}{x_k-\phi_i^k}\Bigr). \tag{4}
\end{aligned}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5.}  The cross term in (3) vanishes.
Indeed,
\[
\begin{aligned}
\E\!\Bigl\langle \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k),\;\bar g^k\Bigr\rangle
&= \frac{1}{n}\sum_{i=1}^{n}\inner{\nabla f_i(x_k)-\nabla f_i(\phi_i^k)}{\bar g^k}\\
&= \inner{\frac{1}{n}\sum_i\bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i^k)\bigr)}{\bar g^k}=0,
\end{aligned}
\tag{5}
\]
because $\bar g^k=\frac{1}{n}\sum_i\nabla f_i(\phi_i^k)$. \textbf{[Step 5]}

\noindent\textbf{Step 6.}  Collect (3)–(5):
\[
\E\!\bigl[\norm{v_k}^2\mid\mathcal{F}_k\bigr]
\le \frac{2L}{n}\sum_{i=1}^{n}\Bigl(f_i(x_k)-f_i(\phi_i^k)-\inner{\nabla f_i(\phi_i^k)}{x_k-\phi_i^k}\Bigr)
      +\norm{\bar g^k}^2. \tag{6}
\]
\textbf{[Step 6]}

\noindent\textbf{Step 7.}  Upper‑bound $\norm{\bar g^k}^2$ by Jensen:
\[
\norm{\bar g^k}^2
= \norm{\frac{1}{n}\sum_i\nabla f_i(\phi_i^k)}^2
\le \frac{1}{n}\sum_i\norm{\nabla f_i(\phi_i^k)}^2 .
\tag{7}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8.}  Relate $\norm{\nabla f_i(\phi_i^k)}^2$ to distances to $x^\star$.
Using $L$‑smoothness with $y=x^\star$,
\[
\norm{\nabla f_i(\phi_i^k)}^2
\le 2L\bigl(f_i(\phi_i^k)-f_i(x^\star)-\inner{\nabla f_i(x^\star)}{\phi_i^k-x^\star}\bigr)
\le 2L\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr), \tag{8}
\]
the last inequality follows from optimality condition $\nabla f(x^\star)=0$ and convexity. \textbf{[Step 8]}

\noindent\textbf{Step 9.}  Plug (8) into (7) and then into (6):
\[
\E\!\bigl[\norm{v_k}^2\mid\mathcal{F}_k\bigr]
\le \frac{2L}{n}\sum_{i=1}^{n}\Bigl(f_i(x_k)-f_i(\phi_i^k)-\inner{\nabla f_i(\phi_i^k)}{x_k-\phi_i^k}\Bigr)
      +\frac{2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr). \tag{9}
\]
Simplifying,
\[
\E\!\bigl[\norm{v_k}^2\mid\mathcal{F}_k\bigr]
\le 2L\bigl(f(x_k)-f(x^\star)\bigr)
      -\frac{2L}{n}\sum_{i=1}^{n}\inner{\nabla f_i(\phi_i^k)}{x_k-\phi_i^k}. \tag{10}
\]
\textbf{[Step 9]}

\noindent\textbf{Step 10.}  Use Cauchy–Schwarz and Young’s inequality on the inner product term:
\[
\begin{aligned}
-\inner{\nabla f_i(\phi_i^k)}{x_k-\phi_i^k}
&\le \frac{1}{2\beta}\norm{x_k-\phi_i^k}^2+\frac{\beta}{2}\norm{\nabla f_i(\phi_i^k)}^2\\
&\le \frac{1}{2\beta}\norm{x_k-\phi_i^k}^2+\beta L\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr),
\end{aligned}
\tag{11}
\]
where we used (8) in the second inequality. Choose $\beta=\frac{1}{2\alpha}$ (to match the Lyapunov coefficient defined later). \textbf{[Step 10]}

\noindent\textbf{Step 11.}  Substitute (11) into (10):
\[
\begin{aligned}
\E\!\bigl[\norm{v_k}^2\mid\mathcal{F}_k\bigr]
&\le 2L\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{L}{\beta n}\sum_{i=1}^{n}\norm{x_k-\phi_i^k}^2
   +2\beta L^2\frac{1}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr)\\
&= 2L\bigl(f(x_k)-f(x^\star)\bigr)
   +2\alpha L\sum_{i=1}^{n}\norm{x_k-\phi_i^k}^2
   +\frac{L}{\alpha}\bigl(f(\phi^k)-f(x^\star)\bigr),
\end{aligned}
\tag{12}
\]
where $f(\phi^k):=\frac{1}{n}\sum_i f_i(\phi_i^k)$. \textbf{[Step 11]}

\noindent\textbf{Step 12.}  Return to the distance recursion (1) and take full expectation.
Using (2) and (12):
\[
\begin{aligned}
\E\!\bigl[\norm{x_{k+1}-x^\star}^2\bigr]
&\le \E\!\bigl[\norm{x_k-x^\star}^2\bigr]
   -2\alpha\E\!\bigl[f(x_k)-f(x^\star)\bigr]\\
&\quad +\alpha^2\Bigl[2L\E\!\bigl[f(x_k)-f(x^\star)\bigr]
   +2\alpha L\E\!\Bigl[\frac{1}{n}\sum_i\norm{x_k-\phi_i^k}^2\Bigr]
   +\frac{L}{\alpha}\E\!\bigl[f(\phi^k)-f(x^\star)\bigr]\Bigr].
\end{aligned}
\tag{13}
\]
\textbf{[Step 12]}

\noindent\textbf{Step 13.}  Rearrange (13) and collect the terms involving $f(x_k)-f(x^\star)$:
\[
\begin{aligned}
\E\!\bigl[\norm{x_{k+1}-x^\star}^2\bigr]
&\le \E\!\bigl[\norm{x_k-x^\star}^2\bigr]
   -2\alpha\bigl(1-\alpha L\bigr)\E\!\bigl[f(x_k)-f(x^\star)\bigr]\\
&\quad +2\alpha^3 L\E\!\Bigl[\frac{1}{n}\sum_i\norm{x_k-\phi_i^k}^2\Bigr]
   +\alpha L\E\!\bigl[f(\phi^k)-f(x^\star)\bigr].
\end{aligned}
\tag{14}
\]
\textbf{[Step 13]}

\noindent\textbf{Step 14.}  Add a multiple $c$ of the table‑error term
$\frac{1}{n}\sum_i\norm{\phi_i^{k+1}-x^\star}^2$ to both sides.
Because only $\phi_{i_k}$ changes,
\[
\begin{aligned}
\E\!\Bigl[\frac{1}{n}\sum_i\norm{\phi_i^{k+1}-x^\star}^2\Bigr]
&= \frac{1}{n}\sum_i\E\!\bigl[\norm{\phi_i^{k}-x^\star}^2\bigr]
   -\frac{1}{n}\E\!\bigl[\norm{\phi_{i_k}^{k}-x^\star}^2\bigr]
   +\frac{1}{n}\E\!\bigl[\norm{x_k-x^\star}^2\bigr]\\
&= \frac{1}{n}\sum_i\E\!\bigl[\norm{\phi_i^{k}-x^\star}^2\bigr]
   +\frac{1}{n}\E\!\bigl[\norm{x_k-x^\star}^2-\norm{\phi_{i_k}^{k}-x^\star}^2\bigr].
\end{aligned}
\tag{15}
\]
\textbf{[Step 14]}

\noindent\textbf{Step 15.}  Multiply (15) by $c:=\frac{1}{2\alpha}$ and add to (14).  
After straightforward algebra we obtain the Lyapunov recursion
\[
\Psi_{k+1}\le\Psi_k-
\underbrace{\alpha\bigl(1-2\alpha L\bigr)}_{\ge \alpha/2}
\E\!\bigl[f(x_k)-f(x^\star)\bigr]
-\underbrace{\frac{c}{n}}_{\frac{1}{2\alpha n}}
\E\!\Bigl[\sum_i\norm{\phi_i^{k}-x^\star}^2-
\sum_i\norm{x_k-x^\star}^2\Bigr]. \tag{16}
\]
Because $\alpha\le\frac{1}{3L}$, $1-2\alpha L\ge\frac{1}{3}$, thus the first coefficient is at least $\alpha/3$.  
Using strong convexity (\ref{ass:strong}) we have
\[
f(x_k)-f(x^\star)\ge\frac{\mu}{2}\norm{x_k-x^\star}^2,
\]
and similarly $f(\phi_i^k)-f(x^\star)\ge\frac{\mu}{2}\norm{\phi_i^k-x^\star}^2$.  
Consequently,
\[
\Psi_{k+1}\le\bigl(1-\rho\bigr)\Psi_k,
\qquad
\rho:=\min\Bigl\{\frac{\alpha\mu}{2},\frac{1}{2}\Bigr\}. \tag{17}
\]
\textbf{[Step 15]}

\noindent\textbf{Step 16.}  Unrolling (17) yields the claimed linear rate:
\[
\E\!\bigl[f(x_k)-f(x^\star)\bigr]\le(1-\rho)^k
\Bigl[f(x_0)-f(x^\star)+\frac{c}{n}\sum_{i=1}^{n}\norm{\phi_i^{0}-x^\star}^2\Bigr].
\tag{18}
\]
\textbf{[Step 16]}

Thus every inequality used above follows directly from the smoothness, convexity, and the stepsize restriction; the Lyapunov function couples the optimisation error with the memory error, leading to a geometric contraction. \(\square\)

\section*{Rate Derivation (With Constants)}
From (17) the contraction factor is
\[
\rho = \min\!\Bigl\{\frac{\alpha\mu}{2},\frac{1}{2}\Bigr\}.
\]
Choosing the maximal admissible stepsize $\alpha=1/(3L)$ gives
\[
\rho = \min\!\Bigl\{\frac{\mu}{6L},\frac{1}{2}\Bigr\}.
\]
Hence,
\[
\E\!\bigl[f(x_k)-f(x^\star)\bigr]\le
\exp\!\bigl(-k\rho\bigr)
\Bigl[f(x_0)-f(x^\star)+\frac{3L}{2n}\sum_i\norm{\phi_i^{0}-x^\star}^2\Bigr].
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Strongly convex $f$ (as proved).}
\item \textbf{Convex but not strongly convex.}  
Setting $\mu=0$ in the proof eliminates the term $\frac{\alpha\mu}{2}$; the recursion (16) reduces to
\[
\Psi_{k+1}\le\Psi_k-\frac{\alpha}{2}\bigl(f(x_k)-f(x^\star)\bigr),
\]
which after summation gives the $O(1/k)$ rate:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[f(x_k)-f(x^\star)\bigr]\le\frac{2\Psi_0}{\alpha K}.
\]
\item \textbf{Deterministic limit.}  
If $n=1$, the table contains a single gradient and SAGA reduces to standard gradient descent with stepsize $\alpha\le 1/(3L)$, recovering the classic linear rate $ (1-\alpha\mu)^k$.
\end{itemize}
