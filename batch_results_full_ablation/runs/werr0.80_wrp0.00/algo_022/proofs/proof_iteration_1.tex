\section*{Proximal Gradient Method (PGM) for Non‑Convex Smooth $f$}

\section*{Mathematical Formulation}
Let $f:\R^d\to\R$ be differentiable (not necessarily convex) and $g:\R^d\to\R\cup\{+\infty\}$ be proper, lower‑semicontinuous (possibly non‑convex).  
Given a stepsize $\eta>0$, the \emph{proximal gradient update} is
\[
x_{k+1}= \prox_{\eta g}\!\bigl(x_k-\eta\,\grad f(x_k)\bigr),\qquad k=0,1,2,\dots
\tag{1}
\]
where
\[
\prox_{\eta g}(z):=\arg\min_{u\in\R^d}\Big\{ g(u)+\frac{1}{2\eta}\|u-z\|^2\Big\}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^d$, stepsize $\eta>0$, maximal iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $v_k \gets x_k-\eta\,\grad f(x_k)$ \Comment{gradient step}
    \State $x_{k+1} \gets \prox_{\eta g}(v_k)$ \Comment{proximal step}
\EndFor
\State \Return $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^2$, $g\equiv0$, $\eta=0.1$, $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $\grad f(w_k)=2(w_k-3)$ & $v_k=w_k-\eta\grad f(w_k)$ & $w_{k+1}= \prox_{\eta g}(v_k)=v_k$ \\ \hline
0 & $-6$ & $0-0.1(-6)=0.6$ & $0.6$ \\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $1.08$ \\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $1.464$ 
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3530$,
showing monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]
$f$ is $L$‑smooth, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^d.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Bounded Below]
The composite objective $\Phi(x):=f(x)+g(x)$ is bounded below:
\[
\Phi_{\inf}:=\inf_{x\in\R^d}\Phi(x)>-\infty .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Stepsize]
The stepsize satisfies
\[
0<\eta<\frac{2}{L}.
\tag{A3}
\]
\end{assumption}
No convexity of $f$ or $g$ is required beyond the smoothness of $f$.

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity of Limit Points]\label{thm:main}
Under Assumptions (A1)–(A3), the iterates generated by (1) satisfy
\[
\sum_{k=0}^{\infty}\|x_{k+1}-x_k\|^2<\infty,
\]
and every accumulation point $\bar x$ of $\{x_k\}$ is a \emph{critical point} of $\Phi$, i.e.
\[
0\in \grad f(\bar x)+\partial g(\bar x).
\tag{3}
\]
Moreover, letting the \emph{gradient mapping}
\[
\mathcal{G}_\eta(x):=\frac{1}{\eta}\bigl(x-\prox_{\eta g}(x-\eta\grad f(x))\bigr),
\]
we have the rate
\[
\min_{0\le k\le K-1}\|\mathcal{G}_\eta(x_k)\|^2
\;\le\;
\frac{2\bigl(\Phi(x_0)-\Phi_{\inf}\bigr)}{\eta\,(2-L\eta)K}.
\tag{4}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Descent.}  For any $k$, $\Phi(x_{k+1})\le\Phi(x_k)-\frac{2-L\eta}{2\eta}\|x_{k+1}-x_k\|^2$.
\item \textbf{Stability w.r.t. $\eta$.}  The factor $2-L\eta$ in (4) shows that choosing $\eta$ close to $2/L$ yields the fastest theoretical decrease, while any $\eta\in(0,2/L)$ guarantees convergence.
\item \textbf{Comparison.}  When $g\equiv0$, $\mathcal{G}_\eta$ reduces to the ordinary gradient, and (4) recovers the classic $O(1/K)$ rate for gradient descent on non‑convex $L$‑smooth functions. With a non‑trivial $g$, the same $O(1/K)$ rate holds for the gradient mapping, matching the proximal‑gradient results for convex $f$.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the possible non‑convexity of $f$ and $g$, the proximal‑gradient scheme enjoys a **global** guarantee that the norm of the gradient mapping converges to zero at a sub‑linear $O(1/K)$ rate. The proof hinges only on the smoothness of $f$ and the **non‑expansiveness** of the proximal operator (Lemma~\ref{lem:nonexp}). No convexity of $g$ is required, making the result applicable to a wide class of regularizers (e.g., $\ell_0$‑type penalties) provided the proximal map can be evaluated.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Proximal Mapping]
For $\eta>0$ and proper lsc $g$, $\prox_{\eta g}$ is defined by (2). It is \emph{firmly non‑expansive}:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|^2\le \langle u-v,\prox_{\eta g}(u)-\prox_{\eta g}(v)\rangle,
\qquad\forall u,v\in\R^d.
\tag{5}
\]
\end{definition}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\R^d$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2.
\tag{6}
\]
\end{lemma}
\begin{lemma}[Non‑expansiveness of $\prox$]\label{lem:nonexp}
For any $u,v\in\R^d$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
\tag{7}
\]
\end{lemma}
\begin{proof}
Firm non‑expansiveness (5) implies (7) after discarding the inner‑product term.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we set $x_{k+1}$ according to (1) and write $v_k:=x_k-\eta\grad f(x_k)$ for brevity.

\bigskip
\noindent\textbf{[Step 1] (Optimality condition of the proximal step).}\\
By definition of $\prox_{\eta g}$,
\[
0\in \partial g(x_{k+1})+\frac{1}{\eta}(x_{k+1}-v_k).
\tag{8}
\]
Rearranging gives
\[
-\frac{1}{\eta}(x_{k+1}-x_k)+\grad f(x_k)\in \partial g(x_{k+1}).
\tag{9}
\]

\bigskip
\noindent\textbf{[Step 2] (Descent of $f$).}\\
Apply Lemma~\ref{lem:descent} with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^2.
\tag{10}
\]

\bigskip
\noindent\textbf{[Step 3] (Combine (9) and (10)).}\\
Add $g(x_{k+1})$ to both sides of (10) and use the subgradient inequality from (9):
\[
\begin{aligned}
\Phi(x_{k+1})
&=f(x_{k+1})+g(x_{k+1})\\
&\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^2
   +g(x_{k+1})\\
&\le f(x_k)+g(x_k)+\Big\langle\grad f(x_k),x_{k+1}-x_k\Big\rangle
   +\frac{L}{2}\|x_{k+1}-x_k\|^2\\
&\qquad+\Big\langle\frac{1}{\eta}(x_{k+1}-x_k),x_{k+1}-x_k\Big\rangle
   \quad\text{(by (9) with $u=x_{k+1}$)}\\[2mm]
&= \Phi(x_k)-\Big(\frac{1}{\eta}-\frac{L}{2}\Big)\|x_{k+1}-x_k\|^2.
\end{aligned}
\tag{11}
\]

\bigskip
\noindent\textbf{[Step 4] (Descent inequality).}\\
Since $\eta<2/L$ (Assumption A3), the coefficient $\frac{1}{\eta}-\frac{L}{2}>0$. Hence (11) yields
\[
\Phi(x_k)-\Phi(x_{k+1})\ge\Big(\frac{1}{\eta}-\frac{L}{2}\Big)\|x_{k+1}-x_k\|^2.
\tag{12}
\]

\bigskip
\noindent\textbf{[Step 5] (Summation).}\\
Summing (12) from $k=0$ to $K-1$ gives
\[
\sum_{k=0}^{K-1}\|x_{k+1}-x_k\|^2
\le\frac{\eta}{2-L\eta}\bigl(\Phi(x_0)-\Phi(x_K)\bigr)
\le\frac{\eta}{2-L\eta}\bigl(\Phi(x_0)-\Phi_{\inf}\bigr).
\tag{13}
\]
Thus $\sum_{k=0}^{\infty}\|x_{k+1}-x_k\|^2<\infty$, establishing the first claim of the theorem.

\bigskip
\noindent\textbf{[Step 6] (Gradient mapping bound).}\\
Recall the definition of $\mathcal{G}_\eta$:
\[
\mathcal{G}_\eta(x_k)=\frac{1}{\eta}\bigl(x_k-\prox_{\eta g}(x_k-\eta\grad f(x_k))\bigr)
           =\frac{1}{\eta}(x_k-x_{k+1}).
\tag{14}
\]
Hence $\|\mathcal{G}_\eta(x_k)\|^2=\frac{1}{\eta^2}\|x_{k+1}-x_k\|^2$. Using (12) we obtain for each $k$,
\[
\|\mathcal{G}_\eta(x_k)\|^2
\le\frac{2}{\eta(2-L\eta)}\bigl(\Phi(x_k)-\Phi(x_{k+1})\bigr).
\tag{15}
\]
Summing (15) over $k=0,\dots,K-1$ and dividing by $K$ yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\mathcal{G}_\eta(x_k)\|^2
\le\frac{2}{\eta(2-L\eta)K}\bigl(\Phi(x_0)-\Phi(x_K)\bigr)
\le\frac{2}{\eta(2-L\eta)K}\bigl(\Phi(x_0)-\Phi_{\inf}\bigr).
\tag{16}
\]
Consequently,
\[
\min_{0\le k\le K-1}\|\mathcal{G}_\eta(x_k)\|^2
\le\frac{2\bigl(\Phi(x_0)-\Phi_{\inf}\bigr)}{\eta(2-L\eta)K},
\]
which is exactly the rate (4).

\bigskip
\noindent\textbf{[Step 7] (Stationarity of accumulation points).}\\
From (13) we have $\|x_{k+1}-x_k\|\to0$ along any subsequence. Moreover, (9) implies
\[
-\mathcal{G}_\eta(x_k)+\grad f(x_k)\in\partial g(x_{k+1}).
\tag{17}
\]
Taking limits along a convergent subsequence $x_{k_j}\to\bar x$ and using the closedness of the limiting subdifferential gives
\[
0\in\grad f(\bar x)+\partial g(\bar x),
\]
i.e. $\bar x$ satisfies the criticality condition (3). ∎

\section*{Rate Derivation (With Constants)}
Define $\Delta_0:=\Phi(x_0)-\Phi_{\inf}>0$. From (4) we have the explicit bound
\[
\boxed{\;
\min_{0\le k\le K-1}\|\mathcal{G}_\eta(x_k)\|
\le \sqrt{\frac{2\Delta_0}{\eta(2-L\eta)K}}\;
}
\]
Thus the gradient‑mapping norm decays as $O(K^{-1/2})$, which is the standard sub‑linear rate for first‑order methods on non‑convex $L$‑smooth problems.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent ($g\equiv0$).}  $\prox_{\eta g}$ becomes the identity, $\mathcal{G}_\eta(x)=\grad f(x)$, and (4) reduces to the classic $O(1/K)$ bound on $\|\grad f(x_k)\|^2$.
\item \textbf{Convex $g$.}  The same proof holds; additional convexity can be used to strengthen (12) to a functional decrease bound without the factor $2-L\eta$.
\item \textbf{Exact line‑search stepsize.}  Choosing $\eta_k=1/L$ maximizes the denominator $2-L\eta$ and yields the simplest constant $1/(L K)$ in (4).
\end{itemize}
