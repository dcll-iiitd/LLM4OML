\section*{Algorithm Name}
\textbf{SGDR: Stochastic Gradient Descent with Warm Restarts (Cosine Annealing)}  

The method follows the usual stochastic gradient descent update but replaces a
single monotone learning‑rate schedule by a sequence of cosine‑annealed cycles.
After each cycle the learning rate is reset to its maximal value $\eta_{\max }$
and the length of the next cycle can be enlarged (warm restart).

---------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (possibly non‑convex) objective and let
$g_t$ be a stochastic gradient computed on a randomly drawn data point
$z_t$:
\[
g_t:=\nabla f(w_t;z_t),\qquad 
\EE[g_t\,|\,w_t]=\nabla f(w_t),\qquad 
\EE\big[\|g_t-\nabla f(w_t)\|^{2}\,|\,w_t\big]\le\sigma^{2}.
\]

\medskip
\noindent\textbf{Learning‑rate schedule.}
Denote by $T_{0}=M$ the length of the first cycle and by $\gamma\ge 1$ the
restart factor. After $k$ completed cycles the length of the current cycle is
\[
T_{k}= \gamma^{k}M .
\]
For an iteration $t\ge0$ let $k(t)$ be the unique integer such that
\[
\sum_{i=0}^{k(t)-1}T_{i}\le t < \sum_{i=0}^{k(t)}T_{i},
\qquad
s(t):=t-\sum_{i=0}^{k(t)-1}T_{i}\in\{0,\dots ,T_{k(t)}-1\}
\]
be respectively the index of the current cycle and the position inside that
cycle. The cosine‑annealed step size is
\[
\boxed{\;
\eta_t
:=\eta_{\min}
+\frac{1}{2}\bigl(\eta_{\max}-\eta_{\min}\bigr)
\Bigl(1+\cos\Bigl(\pi\frac{s(t)}{T_{k(t)}}\Bigr)\Bigr)
\; }                                             \tag{1}
\]
with $0<\eta_{\min}<\eta_{\max}$.

\medskip
\noindent\textbf{Update rule.}
\[
\boxed{\; w_{t+1}=w_{t}-\eta_{t}\,g_{t}\; }            \tag{2}
\]

---------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\Require Initial point $w_{0}$, maximal step size $\eta_{\max}$,
minimal step size $\eta_{\min}$, first cycle length $M$, restart factor $\gamma\ge1$.
\State Set $t\gets0$, $k\gets0$, $T_{k}\gets M$.
\While{not converged}
    \State Compute stochastic gradient $g_{t}=\nabla f(w_{t};z_{t})$.
    \State $s\gets t-\sum_{i=0}^{k-1}T_{i}$ \Comment{position inside current cycle}
    \State $\displaystyle \eta_{t}\gets 
        \eta_{\min}+\frac{1}{2}(\eta_{\max}-\eta_{\min})
        \bigl(1+\cos(\pi s/T_{k})\bigr)$          \Comment{Eq.~(1)}
    \State $w_{t+1}\gets w_{t}-\eta_{t}g_{t}$        \Comment{Eq.~(2)}
    \State $t\gets t+1$
    \If{$s = T_{k}-1$} \Comment{end of current cycle}
        \State $k\gets k+1$, $T_{k}\gets \gamma\,T_{k-1}$ \Comment{warm restart}
    \EndIf
\EndWhile
\State \textbf{return} $w_{t}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Choose $w_{0}=0$, $\eta_{\max}=0.1$, $\eta_{\min}=0$, $M=3$, $\gamma=1$ (no
length increase) and assume a \emph{deterministic} gradient (i.e. $g_{t}
=\nabla f(w_{t})$).  
For a cosine schedule with $M=3$, the step sizes are
\[
\eta_{0}=0.1,\qquad
\eta_{1}=0.1\Bigl(\frac{1+\cos(\pi/3)}{2}\Bigr)=0.1\cdot0.75=0.075,
\qquad
\eta_{2}=0.1\Bigl(\frac{1+\cos(2\pi/3)}{2}\Bigr)=0.1\cdot0.25=0.025.
\]

\medskip
\noindent\textbf{Iteration 0}
\[
g_{0}=2(0-3)=-6,\qquad
w_{1}=0-0.1(-6)=0.6,\qquad
f(w_{1})=(0.6-3)^{2}=5.76.
\]

\noindent\textbf{Iteration 1}
\[
g_{1}=2(0.6-3)=-4.8,\qquad
w_{2}=0.6-0.075(-4.8)=1.0,\qquad
f(w_{2})=(1-3)^{2}=4.
\]

\noindent\textbf{Iteration 2}
\[
g_{2}=2(1-3)=-4,\qquad
w_{3}=1.0-0.025(-4)=1.1,\qquad
f(w_{3})=(1.1-3)^{2}=3.61.
\]

The objective value decreases from $9$ to $3.61$ in three steps while the
learning rate follows the prescribed cosine annealing and is reset after the
third iteration (here the restart would set $\eta_{3}=0.1$ again).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall u,v\in\RR^{d}:\quad
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Stochastic Gradient]\label{as:stoch}
For every $t$,
\[
\EE[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad
\EE\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Step‑size bounds]\label{as:steps}
The schedule \eqref{1} satisfies
\[
0<\eta_{\min}\le\eta_{t}\le\eta_{\max}\le\frac{1}{L}\quad\forall t,
\tag{A3}
\]
and the sequence $\{\eta_{t}\}$ is \emph{non‑increasing} within each cycle.
\end{assumption}

---------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR]\label{thm:sgdr}
Under Assumptions \ref{as:smooth}–\ref{as:steps}, for any integer $T\ge1$
the iterates generated by \eqref{2} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_{0})-f^{*}\big)}{\eta_{\min}T}}_{\displaystyle
\text{optimization term}}
\;+\;
\underbrace{L\eta_{\max}\sigma^{2}}_{\displaystyle\text{variance term}}.
\tag{3}
\]
Consequently $\displaystyle\min_{0\le t<T}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
=O\!\big(\tfrac{1}{\sqrt{T}}\big)$.
\end{theorem}

---------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because each $\eta_{t}\le 1/L$, the factor
$1-\tfrac{L}{2}\eta_{t}\ge \tfrac12$ appears in the descent inequality,
guaranteeing that the expected function value never increases.
\item \textbf{Hyper‑parameter sensitivity.}
  The bound \eqref{3} depends linearly on $\eta_{\max}$ (through the variance
  term) and inversely on $\eta_{\min}$ (through the optimization term).  A
  larger $\eta_{\max}$ accelerates early descent but inflates the variance
  contribution; a larger $\eta_{\min}$ improves the denominator of the first
  term but may violate the stability condition $ \eta_{\min}\le 1/L $.
\item \textbf{Comparison to plain SGD.}
  For a constant step size $\eta$ satisfying $\eta\le 1/L$, the classic
  non‑convex SGD bound reads
  $\frac{1}{T}\sum_{t=0}^{T-1}\EE\|\nabla f(w_t)\|^{2}
   \le \frac{2(f(w_{0})-f^{*})}{\eta T}+L\eta\sigma^{2}$.
  SGDR replaces the constant $\eta$ by the interval
  $[\eta_{\min},\eta_{\max}]$ and the bound \eqref{3} is obtained by
  substituting the worst‑case values $\eta_{\min}$ and $\eta_{\max}$,
  showing that SGDR inherits the same $O(1/\sqrt{T})$ rate while allowing
  practical acceleration via periodic restarts.
\end{itemize}

---------------------------------------------------------------------

\section*{Discussion}
The theorem shows that the warm‑restart cosine schedule does not deteriorate
the classical non‑convex SGD guarantee; the only price is the appearance of
the extreme values $\eta_{\min}$ and $\eta_{\max}$ in the bound.  In practice,
the periodic reset to a large $\eta_{\max}$ helps the iterates escape shallow
local minima, whereas the gradual decay inside each cycle preserves the
descent property required for the proof.  The factor $\gamma$ that enlarges the
cycle length influences the constants only implicitly, because all
$\eta_{t}$ remain bounded by the same $\eta_{\max}$ and $\eta_{\min}$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $w$ and any stochastic gradient $g$,
\[
f(w-\eta g)\le f(w)-\eta\langle\nabla f(w),g\rangle
+\frac{L\eta^{2}}{2}\|g\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Apply Assumption \ref{as:smooth} with $u=w$ and $v=w-\eta g$ and rearrange.
\end{proof}

\begin{lemma}[Conditional expectation of the stochastic inner product]\label{lem:cond}
Under Assumption \ref{as:stoch},
\[
\EE\big[\langle\nabla f(w_{t}),g_{t}\rangle\mid w_{t}\big]
    =\|\nabla f(w_{t})\|^{2},
\qquad
\EE\big[\|g_{t}\|^{2}\mid w_{t}\big]
    =\|\nabla f(w_{t})\|^{2}+\EE\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]
    \le \|\nabla f(w_{t})\|^{2}+\sigma^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Immediate from unbiasedness and the variance bound.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:sgdr}.  All expectations are taken w.r.t. the
randomness up to iteration $t$ unless otherwise stated.

\begin{proof}
\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $w=w_{t}$,
$g=g_{t}$ and step size $\eta_{t}$:
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),g_{t}\rangle
            +\frac{L\eta_{t}^{2}}{2}\|g_{t}\|^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Take conditional expectation $\EE[\cdot\mid w_{t}]$.
Using Lemma \ref{lem:cond} (Eq.~\eqref{5}) we obtain
\[
\EE\!\big[f(w_{t+1})\mid w_{t}\big]
\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta_{t}^{2}}{2}\bigl(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\bigr).
\tag{7}
\]

\noindent\textbf{[Step 3]} Rearrange terms in \eqref{7}:
\[
\EE\!\big[f(w_{t+1})\mid w_{t}\big]
\le f(w_{t})
-\underbrace{\Bigl(\eta_{t}-\frac{L\eta_{t}^{2}}{2}\Bigr)}_{\displaystyle
\alpha_{t}}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}\sigma^{2}}{2}.
\tag{8}
\]

\noindent\textbf{[Step 4]} By Assumption \ref{as:steps}, $\eta_{t}\le 1/L$;
hence $\eta_{t}-\frac{L\eta_{t}^{2}}{2}\ge \frac{\eta_{t}}{2}$.  Define
$\displaystyle \beta_{t}:=\frac{\eta_{t}}{2}\ge\frac{\eta_{\min}}{2}$.
Thus \eqref{8} simplifies to
\[
\EE\!\big[f(w_{t+1})\mid w_{t}\big]
\le f(w_{t})-\beta_{t}\|\nabla f(w_{t})\|^{2}
    +\frac{L\eta_{t}^{2}\sigma^{2}}{2}.
\tag{9}
\]

\noindent\textbf{[Step 5]} Take full expectation and sum \eqref{9} over
$t=0,\dots ,T-1$:
\[
\EE\big[f(w_{T})\big]
\le f(w_{0})
-\sum_{t=0}^{T-1}\beta_{t}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 6]} Drop the non‑negative term $\EE[f(w_{T})]\ge f^{*}$,
use $\beta_{t}\ge\eta_{\min}/2$, and bound $\eta_{t}^{2}\le\eta_{\max}^{2}$:
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le f(w_{0})-f^{*}
    +\frac{L\sigma^{2}}{2}\,T\eta_{\max}^{2}.
\tag{11}
\]

\noindent\textbf{[Step 7]} Divide both sides of \eqref{11} by $T$ and
multiply by $2/\eta_{\min}$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\frac{2\big(f(w_{0})-f^{*}\big)}{\eta_{\min}T}
\;+\;
L\eta_{\max}\sigma^{2}.
\tag{12}
\]

Equation \eqref{12} is exactly the bound \eqref{3} claimed in the theorem.
\end{proof}

---------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From \eqref{12} we read off the two contributions:

\begin{itemize}
\item \emph{Optimization term} $\displaystyle
  \frac{2(f(w_{0})-f^{*})}{\eta_{\min}T}=O\!\big(\tfrac{1}{T}\big)$.
\item \emph{Variance term} $\displaystyle L\eta_{\max}\sigma^{2}=O(1)$.
\end{itemize}

Balancing both terms (i.e. choosing $\eta_{\min}\asymp\eta_{\max}
\asymp 1/\sqrt{T}$) yields the classical $O(1/\sqrt{T})$ rate for the
\emph{minimum} gradient norm:
\[
\min_{0\le t<T}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le O\!\big(\tfrac{1}{\sqrt{T}}\big).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{No restart} ($\gamma=1$, $M\to\infty$).  The schedule reduces to a
single cosine annealing from $\eta_{\max}$ to $\eta_{\min}$; the proof
remains unchanged because only the bounds $\eta_{\min},\eta_{\max}$ matter.
\item \textbf{Linear decay} (set $\eta_{\min}=0$ and replace the cosine by
$\eta_{t}= \eta_{\max}\bigl(1-\frac{s(t)}{T_{k(t)}}\bigr)$).  The same
analysis holds with $\eta_{\max}\le1/L$.
\item \textbf{Deterministic gradients} ($\sigma=0$).  The variance term vanishes,
and the bound becomes purely $O(1/T)$, matching the rate of gradient descent
on smooth non‑convex functions.
\end{enumerate}

---------------------------------------------------------------------

\section*{Conclusion}
We have presented a fully detailed description of SGDR, illustrated its
behaviour on a toy quadratic, and derived a rigorous non‑convex convergence
guarantee.  The proof follows directly from the standard smooth‑non‑convex SGD
analysis, with the only modifications being the explicit lower and upper
bounds on the cosine‑annealed learning rate.  Consequently, SGDR inherits the
optimal $O(1/\sqrt{T})$ rate while offering practical benefits through warm
restarts.
