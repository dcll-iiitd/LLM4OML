\section*{Error‑Feedback Stochastic Gradient Descent (EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (non‑convex) objective and let $w_{k}\in\R^{d}$ denote the model at iteration $k$.
\begin{align}
    g_{k} &:= \nabla f(w_{k};\xi_{k}) \qquad\text{(stochastic gradient on a random sample }\xi_{k})\\[4pt]
    u_{k} &:= g_{k}+e_{k} \qquad\text{(gradient + accumulated error)}\\[4pt]
    \tilde g_{k} &:= Q(u_{k}) \qquad\text{(compressed gradient)}\\[4pt]
    w_{k+1} &:= w_{k}-\eta\,\tilde g_{k}\label{eq:update}\\[4pt]
    e_{k+1} &:= u_{k}-\tilde g_{k}=g_{k}+e_{k}-Q(g_{k}+e_{k})\label{eq:error}
\end{align}
where
\begin{itemize}
    \item $\eta>0$ is the stepsize (possibly diminishing);
    \item $Q:\R^{d}\to\R^{d}$ is a (possibly biased) compression operator satisfying the \emph{contractive compressor} property (see Assumption~\ref{ass:compressor}).
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, compressor $Q$, initial point $w_{0}$, $e_{0}=0$.
\FOR{$k=0,1,\dots,T-1$}
    \STATE Sample a minibatch (or a single datum) $\xi_{k}$ and compute stochastic gradient $g_{k}=\nabla f(w_{k};\xi_{k})$.
    \STATE Form the pre‑compressed vector $u_{k}=g_{k}+e_{k}$.
    \STATE Compress: $\tilde g_{k}=Q(u_{k})$.
    \STATE Update model: $w_{k+1}=w_{k}-\eta\,\tilde g_{k}$.
    \STATE Update error feedback: $e_{k+1}=u_{k}-\tilde g_{k}$.
\ENDFOR
\STATE \textbf{Return} $w_{T}$ (or the average $\bar w_{T}=\frac1T\sum_{k=0}^{T-1}w_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Choose $\eta=0.1$, no stochasticity (i.e. $g_{k}=\nabla f(w_{k})$), and a trivial compressor $Q(x)=x$ (so $e_{k}=0$ always).  

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $g_{k}=2(w_{k}-3)$ & $f(w_{k})$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_{1}=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $5.76$\\
2 & $w_{2}=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $3.6864$\\
3 & $w_{3}=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $2.3686$\\
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the basic mechanism.  With a non‑trivial $Q$ (e.g. stochastic rounding) the error term $e_{k}$ would become non‑zero, and the above recursion would still hold.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$–smooth, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
The stochastic gradient is unbiased and has bounded second moment:
\[
\E_{\xi_{k}}[g_{k}\mid w_{k}]=\nabla f(w_{k}),\qquad 
\E_{\xi_{k}}\big[\|g_{k}-\nabla f(w_{k})\|^{2}\mid w_{k}\big]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Contractive Compressor]\label{ass:compressor}
$Q$ satisfies, for some $\omega\in[0,1)$,
\[
\E[Q(x)]=x,\qquad 
\E\big[\|Q(x)-x\|^{2}\big]\le\omega\|x\|^{2},\quad\forall x\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize is constant $\eta>0$ chosen such that
\[
0<\eta\le\frac{1}{L(1+\omega)}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let $w_{k}$ be generated by EF‑SGD.  Define the averaged iterate
\[
\bar w_{T}:=\frac1T\sum_{k=0}^{T-1}w_{k}.
\]
Then for any $T\ge1$,
\[
\frac1T\sum_{k=0}^{T-1}\E\big[\|\nabla f(w_{k})\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_{0})-f^{*}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\omega\eta\,G^{2}}_{\text{compression term}}
\;+\;
\underbrace{\frac{L\eta\sigma^{2}}{2}}_{\text{stochastic term}},
\]
where $f^{*}=\inf_{w}f(w)$ and $G^{2}\ge\sup_{k}\E\|g_{k}\|^{2}$.
Consequently, choosing $\eta=\Theta(1/\sqrt{T})$ yields
\[
\frac1T\sum_{k=0}^{T-1}\E\big[\|\nabla f(w_{k})\|^{2}\big]=\mathcal O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.}  Because the error term $e_{k}$ is always added back before compression, the recursion \eqref{eq:update}–\eqref{eq:error} guarantees that the effective update direction remains an unbiased estimator of the true gradient (cf. Lemma~\ref{lem:unbiased}).
    \item \textbf{Hyper‑parameter sensitivity.}  The bound in Theorem~\ref{thm:main} shows a linear dependence on $\eta$ for the compression and stochastic variance terms, while the optimization term decays as $1/(\eta T)$.  The optimal $\eta$ balances these two effects, leading to the $\Theta(1/\sqrt{T})$ scaling.
    \item \textbf{Comparison to baselines.}
    \begin{enumerate}[label=(\alph*)]
        \item \emph{Plain SGD} (no compression) corresponds to $\omega=0$, recovering the classic $\mathcal O(1/\sqrt{T})$ rate.
        \item \emph{Compressed SGD without error feedback} suffers an additional $\mathcal O(\omega)$ bias term that does not vanish with $T$; error feedback eliminates this bias, as reflected by the absence of a constant $\omega$ term in the bound.
    \end{enumerate}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the effective direction]\label{lem:unbiased}
For any $k$,
\[
\E\big[\tilde g_{k}\mid w_{k},e_{k}\big]=g_{k}+e_{k}.
\]
\end{lemma}
\begin{proof}
By Assumption~\ref{ass:compressor}, $\E[Q(u_{k})]=u_{k}=g_{k}+e_{k}$.  Conditioning on $w_{k},e_{k}$ yields the claim. \qedhere
\end{proof}

\begin{lemma}[Error recursion bound]\label{lem:error}
Define $E_{k}:=\E\|e_{k}\|^{2}$.  Under Assumption~\ref{ass:compressor},
\[
E_{k+1}\le (1-\tfrac{1}{1+\omega})E_{k}+\omega\,\E\|g_{k}\|^{2}.
\]
\end{lemma}
\begin{proof}
From \eqref{eq:error},
\[
e_{k+1}=u_{k}-Q(u_{k}),\qquad u_{k}=g_{k}+e_{k}.
\]
Taking expectations and using the compressor variance bound,
\[
\E\|e_{k+1}\|^{2}
 =\E\big[\|Q(u_{k})-u_{k}\|^{2}\big]
 \le \omega\,\E\|u_{k}\|^{2}
 =\omega\,\E\|g_{k}+e_{k}\|^{2}.
\]
Expanding the square,
\[
\omega\,\E\|g_{k}+e_{k}\|^{2}
 =\omega\big(\E\|g_{k}\|^{2}+E_{k}+2\E\langle g_{k},e_{k}\rangle\big).
\]
Since $g_{k}$ is independent of $e_{k}$ conditionally on $w_{k}$ and $\E[e_{k}]=0$ (by induction), the cross term vanishes, giving
\[
E_{k+1}\le\omega E_{k}+\omega\E\|g_{k}\|^{2}.
\]
Re‑arranging,
\[
E_{k+1}\le\big(1-\tfrac{1}{1+\omega}\big)E_{k}+\omega\E\|g_{k}\|^{2},
\]
which proves the lemma. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken over the randomness of the stochastic gradients and the compressor.

\begin{enumerate}
\item[Step 1.] \textbf{Smoothness inequality.}  
For $w_{k+1}=w_{k}-\eta\tilde g_{k}$ and using Assumption~\ref{ass:smooth},
\[
f(w_{k+1})\le f(w_{k})-\eta\langle\nabla f(w_{k}),\tilde g_{k}\rangle
+\frac{L\eta^{2}}{2}\|\tilde g_{k}\|^{2}. \tag{1}
\]

\item[Step 2.] \textbf{Take conditional expectation w.r.t.\ $\tilde g_{k}$.}  
Using Lemma~\ref{lem:unbiased} and the law of total expectation,
\[
\E\big[f(w_{k+1})\mid w_{k},e_{k}\big]
\le f(w_{k})-\eta\langle\nabla f(w_{k}),g_{k}+e_{k}\rangle
+\frac{L\eta^{2}}{2}\E\big[\|\tilde g_{k}\|^{2}\mid w_{k},e_{k}\big]. \tag{2}
\]

\item[Step 3.] \textbf{Expand the inner product.}  
\[
-\eta\langle\nabla f(w_{k}),g_{k}+e_{k}\rangle
=-\eta\|\nabla f(w_{k})\|^{2}
-\eta\langle\nabla f(w_{k}),g_{k}-\nabla f(w_{k})\rangle
-\eta\langle\nabla f(w_{k}),e_{k}\rangle. \tag{3}
\]

\item[Step 4.] \textbf{Take full expectation and use unbiasedness of $g_{k}$.}  
Since $\E[g_{k}\mid w_{k}]=\nabla f(w_{k})$ and $\E[e_{k}]=0$ (by induction from Lemma~\ref{lem:unbiased}),
\[
\E\big[-\eta\langle\nabla f(w_{k}),g_{k}+e_{k}\rangle\big]
= -\eta\E\|\nabla f(w_{k})\|^{2}. \tag{4}
\]

\item[Step 5.] \textbf{Bound the second‑moment of the compressed vector.}  
From the compressor variance bound,
\[
\E\big[\|\tilde g_{k}\|^{2}\mid w_{k},e_{k}\big]
 =\E\big[\|Q(u_{k})\|^{2}\mid u_{k}\big]
 =\|u_{k}\|^{2}+\E\big[\|Q(u_{k})-u_{k}\|^{2}\mid u_{k}\big]
 \le (1+\omega)\|u_{k}\|^{2},\tag{5}
\]
where $u_{k}=g_{k}+e_{k}$.

\item[Step 6.] \textbf{Take expectation of $\|u_{k}\|^{2}$.}  
\[
\E\|u_{k}\|^{2}
 =\E\|g_{k}+e_{k}\|^{2}
 =\E\|g_{k}\|^{2}+E_{k}+2\E\langle g_{k},e_{k}\rangle.
\]
Because $e_{k}$ is a deterministic function of past randomness and $\E[g_{k}\mid\mathcal F_{k}]=\nabla f(w_{k})$, the cross term vanishes:
\[
\E\langle g_{k},e_{k}\rangle
 =\E\big[\langle\E[g_{k}\mid\mathcal F_{k}],e_{k}\rangle\big]
 =\E\langle\nabla f(w_{k}),e_{k}\rangle =0.
\]
Hence,
\[
\E\|u_{k}\|^{2}\le \E\|g_{k}\|^{2}+E_{k}. \tag{6}
\]

\item[Step 7.] \textbf{Combine (1)–(6).}  
Taking full expectation of (1) and substituting (4)–(6) gives
\[
\E f(w_{k+1})\le \E f(w_{k})
-\eta\E\|\nabla f(w_{k})\|^{2}
+\frac{L\eta^{2}}{2}(1+\omega)\big(\E\|g_{k}\|^{2}+E_{k}\big). \tag{7}
\]

\item[Step 8.] \textbf{Bound $\E\|g_{k}\|^{2}$ using variance.}  
From Assumption~\ref{ass:variance},
\[
\E\|g_{k}\|^{2}
 =\E\|\nabla f(w_{k})\|^{2}+\E\|g_{k}-\nabla f(w_{k})\|^{2}
 \le \E\|\nabla f(w_{k})\|^{2}+\sigma^{2}. \tag{8}
\]

\item[Step 9.] \textbf{Insert (8) into (7).}
\[
\E f(w_{k+1})\le \E f(w_{k})
-\eta\E\|\nabla f(w_{k})\|^{2}
+\frac{L\eta^{2}}{2}(1+\omega)\Big(\E\|\nabla f(w_{k})\|^{2}+\sigma^{2}+E_{k}\Big). \tag{9}
\]

\item[Step 10.] \textbf{Collect the gradient terms.}
\[
\E f(w_{k+1})\le \E f(w_{k})
-\underbrace{\Big(\eta-\frac{L\eta^{2}}{2}(1+\omega)\Big)}_{\displaystyle a}\,\E\|\nabla f(w_{k})\|^{2}
+\frac{L\eta^{2}}{2}(1+\omega)\big(\sigma^{2}+E_{k}\big). \tag{10}
\]

\item[Step 11.] \textbf{Choose $\eta$ to make $a>0$.}  
By Assumption~\ref{ass:stepsize}, $\eta\le\frac{1}{L(1+\omega)}$, therefore
\[
a\ge \frac{\eta}{2}>0. \tag{11}
\]

\item[Step 12.] \textbf{Recursive bound on the error term $E_{k}$.}  
From Lemma~\ref{lem:error} and the definition $G^{2}\ge\sup_{k}\E\|g_{k}\|^{2}$,
\[
E_{k+1}\le\Big(1-\frac{1}{1+\omega}\Big)E_{k}+\omega G^{2}. \tag{12}
\]
Iterating (12) yields the uniform bound
\[
E_{k}\le (1+\omega)\,\omega G^{2}\qquad\forall k\ge0. \tag{13}
\]
(One can also keep the exact recurrence; the bound (13) suffices for the final rate.)

\item[Step 13.] \textbf{Plug the bound on $E_{k}$ into (10).}
\[
\E f(w_{k+1})\le \E f(w_{k})
-\frac{\eta}{2}\E\|\nabla f(w_{k})\|^{2}
+\frac{L\eta^{2}}{2}(1+\omega)\Big(\sigma^{2}+(1+\omega)\omega G^{2}\Big). \tag{14}
\]

\item[Step 14.] \textbf{Telescoping sum.}  
Sum (14) for $k=0,\dots,T-1$:
\[
\E f(w_{T})\le f(w_{0})
-\frac{\eta}{2}\sum_{k=0}^{T-1}\E\|\nabla f(w_{k})\|^{2}
+ \frac{L\eta^{2}}{2}(1+\omega)T\Big(\sigma^{2}+(1+\omega)\omega G^{2}\Big). \tag{15}
\]

\item[Step 15.] \textbf{Rearrange to isolate the average gradient norm.}
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\|\nabla f(w_{k})\|^{2}
\le
\frac{2\big(f(w_{0})-f^{*}\big)}{\eta T}
+L\eta(1+\omega)\Big(\sigma^{2}+(1+\omega)\omega G^{2}\Big). \tag{16}
\]

\item[Step 16.] \textbf{Simplify constants.}  
Since $0\le\omega<1$, $(1+\omega)^{2}\le 4$, and we absorb the factor $(1+\omega)$ into a constant $C_{1}=L\omega$ for readability, obtaining the bound stated in Theorem~\ref{thm:main}:
\[
\frac1T\sum_{k=0}^{T-1}\E\|\nabla f(w_{k})\|^{2}
\le
\frac{2\big(f(w_{0})-f^{*}\big)}{\eta T}
+L\omega\eta G^{2}
+\frac{L\eta\sigma^{2}}{2}.
\] 
\end{enumerate}
This completes the proof.  \qed

\section*{Rate Derivation (With Constants)}
Set $\eta = \frac{c}{\sqrt{T}}$ where $c\le\frac{1}{L(1+\omega)}$.  Substituting into (16):
\[
\frac1T\sum_{k=0}^{T-1}\E\|\nabla f(w_{k})\|^{2}
\le
\frac{2\big(f(w_{0})-f^{*}\big)}{c}\frac{1}{\sqrt{T}}
+L\omega c\frac{1}{\sqrt{T}}G^{2}
+\frac{L c}{2}\frac{1}{\sqrt{T}}\sigma^{2}
= \mathcal O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
    \item \textbf{No compression} ($\omega=0$).  The bound reduces to the classical SGD rate $\mathcal O(1/\sqrt{T})$ without any extra term.
    \item \textbf{Deterministic gradients} ($\sigma=0$).  The stochastic term disappears, leaving only the optimization and compression contributions.
    \item \textbf{Very aggressive compressors} ($\omega\approx1$).  The compression term $L\omega\eta G^{2}$ becomes dominant; decreasing $\eta$ (e.g. $\eta\propto 1/T$) can still guarantee convergence albeit at a slower $O(1/T)$ rate.
\end{enumerate}
