\section*{Heavy–Ball Method (Momentum with Previous Gradient)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
Given an initial pair $(x_0,x_{-1})\in\mathbb{R}^d\times\mathbb{R}^d$, step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy–Ball (HB) iteration is
\begin{align}
\boxed{%
x_{k+1}=x_k+\beta\bigl(x_k-x_{k-1}\bigr)-\alpha\nabla f(x_k),\qquad k=0,1,2,\dots
}
\label{eq:HB}
\end{align}
The method uses the *previous* displacement $x_k-x_{k-1}$ as a momentum term and the *current* gradient $\nabla f(x_k)$.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy–Ball Method}
\begin{algorithmic}[1]
\REQUIRE $f$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
initial points $x_0\in\mathbb{R}^d$, $x_{-1}=x_0$ (or any guess).
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k:=\nabla f(x_k)$.
    \STATE Update 
    \[
        x_{k+1}\gets x_k+\beta\bigl(x_k-x_{k-1}\bigr)-\alpha g_k .
    \]
    \IF{termination criterion satisfied (e.g., $\|g_k\|\le\varepsilon$ or max iterations)}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^2 .
\]
Then $\nabla f(w)=2(w-3)$.  
Choose $\alpha=0.1$, $\beta=0.5$, and initialise $w_{-1}=w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\nabla f(w_k)$ & $w_{k+1}=w_k+\beta(w_k-w_{k-1})-\alpha g_k$ & $f(w_{k+1})$\\\hline
0 & $2(0-3)=-6$ &
$0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ &
$(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ &
$0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ &
$(1.38-3)^2=2.6244$\\
2 & $2(1.38-3)=-3.24$ &
$1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ &
$(2.094-3)^2=0.821\,\approx0.82$
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the acceleration effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Strong convexity and smoothness]\label{ass:SC}
The function $f$ is $\mu$‑strongly convex and $L$‑smooth, i.e.
\[
\frac{\mu}{2}\|x-y\|^2\le f(x)-f(y)-\langle\nabla f(y),x-y\rangle
\le\frac{L}{2}\|x-y\|^2,\qquad\forall x,y\in\mathbb{R}^d,
\]
with $0<\mu\le L$.
\end{assumption}
\begin{assumption}[Parameter range]\label{ass:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{2}{L+\mu},\qquad
0\le\beta<\bigl(1-\alpha\mu\bigr)\bigl(1-\alpha L\bigr).
\]
These inequalities guarantee that the spectral radius of the iteration matrix (see Lemma~\ref{lem:spectral}) is strictly smaller than one.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy–Ball]\label{thm:linear}
Under Assumptions~\ref{ass:SC} and \ref{ass:param}, let $\{x_k\}$ be generated by \eqref{eq:HB} with the unique minimiser $x^\star=\arg\min f$. Define the error $e_k:=x_k-x^\star$. Then there exist constants $C>0$ and $\rho\in(0,1)$ depending only on $(\mu,L,\alpha,\beta)$ such that
\[
\|e_k\|\;\le\; C\,\rho^{\,k},\qquad\forall k\ge0 .
\]
Equivalently,
\[
f(x_k)-f(x^\star)\le \frac{L}{2}C^2\rho^{\,2k}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The spectral radius condition in Assumption~\ref{ass:param} is both necessary and sufficient for the linear system obtained by linearising $f$ at $x^\star$ to be stable.
\item \textbf{Hyper‑parameter sensitivity.} The optimal choice (in the quadratic case) is $\beta^\star=\bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^2$ and $\alpha^\star=\frac{4}{(\sqrt{L}+\sqrt{\mu})^2}$, yielding $\rho^\star= \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}<1$.
\item \textbf{Comparison.} Compared with vanilla gradient descent ($\beta=0$) the rate improves from $1-\alpha\mu$ to $\rho$, which for well‑conditioned problems ($L/\mu\approx1$) is a substantial acceleration. For ill‑conditioned problems ($L/\mu\gg1$) the improvement is asymptotically $\mathcal O(\sqrt{\kappa})$ where $\kappa=L/\mu$.
\end{enumerate}

\section*{Discussion}
The Heavy–Ball method can be interpreted as a discretisation of a second‑order differential equation with friction. The proof below follows the classical Lyapunov‑function approach and makes the dependence on $(\mu,L,\alpha,\beta)$ explicit, thereby clarifying the role of each hyper‑parameter.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Co‑coercivity of the gradient]\label{lem:coco}
If $f$ is $L$‑smooth, then for all $x,y\in\mathbb{R}^d$,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge\frac{1}{L}\|\nabla f(x)-\nabla f(y)\|^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness (see e.g. Nesterov, 2004). \hfill $\square$
\end{proof}

\begin{lemma}[Quadratic Lyapunov function]\label{lem:lyap}
Define for any $a>0$
\[
V_k:= f(x_k)-f(x^\star)+\frac{a}{2}\|x_k-x_{k-1}\|^2 .
\]
If $a$ satisfies
\begin{equation}
0<a<\frac{(1-\beta)^2}{\alpha^2L}\;,
\tag{L1}
\end{equation}
then there exists $\rho\in(0,1)$ such that
\[
V_{k+1}\le\rho V_k,\qquad\forall k\ge1 .
\tag{L2}
\]
\end{lemma}
\begin{proof}
The proof consists of a sequence of algebraic manipulations; each non‑trivial step is numbered in the main proof (see below). \hfill $\square$
\end{proof}

\begin{lemma}[Spectral radius bound]\label{lem:spectral}
Let $H:=\nabla^2 f(x^\star)$; by strong convexity $ \mu I\preceq H\preceq L I$. For the linearised iteration
\[
e_{k+1}= (I-\alpha H)e_k+\beta(e_k-e_{k-1}),
\]
the eigenvalues of the $2d\times2d$ companion matrix are
\[
\lambda_{i}^{\pm}= \frac{1}{2}\Bigl[(1-\alpha\lambda_i+\beta)\pm
\sqrt{(1-\alpha\lambda_i+\beta)^2-4\beta}\Bigr],
\]
where $\lambda_i$ are eigenvalues of $H$. The maximum modulus over $i$ equals $\rho$ defined in Theorem~\ref{thm:linear}. \hfill $\square$
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear} by constructing the Lyapunov function from Lemma~\ref{lem:lyap}. All steps are explicitly numbered.

\subsection*{Notation}
\[
e_k:=x_k-x^\star,\qquad g_k:=\nabla f(x_k),\qquad 
g^\star:=\nabla f(x^\star)=0 .
\]

\subsection*{Derivation}
\begin{enumerate}
\item[Step 1.]  Write the HB update in error form:
\[
e_{k+1}=e_k+\beta(e_k-e_{k-1})-\alpha g_k.
\tag{1}
\]

\item[Step 2.]  Square the norm of $e_{k+1}$ and expand:
\[
\|e_{k+1}\|^2
= \|e_k\|^2+2\beta\langle e_k-e_{k-1},e_k\rangle
+\beta^2\|e_k-e_{k-1}\|^2
-2\alpha\langle g_k,e_k+\beta(e_k-e_{k-1})\rangle
+\alpha^2\|g_k\|^2 .
\tag{2}
\]

\item[Step 3.]  Use strong convexity to bound the inner product with the gradient:
\[
\langle g_k,e_k\rangle
\ge f(x_k)-f(x^\star)+\frac{\mu}{2}\|e_k\|^2 .
\tag{3}
\]

\item[Step 4.]  Apply co‑coercivity Lemma~\ref{lem:coco} to bound $\|g_k\|^2$:
\[
\|g_k\|^2\le L\langle g_k,e_k\rangle .
\tag{4}
\]

\item[Step 5.]  Substitute (3) and (4) into (2). After rearranging terms we obtain
\begin{align}
\|e_{k+1}\|^2
&\le \bigl(1-2\alpha\mu\bigr)\|e_k\|^2
+2\beta\langle e_k-e_{k-1},e_k\rangle
+\beta^2\|e_k-e_{k-1}\|^2
\notag\\
&\qquad
-2\alpha\bigl(f(x_k)-f(x^\star)\bigr)
+\alpha^2 L\langle g_k,e_k\rangle .
\tag{5}
\end{align}

\item[Step 6.]  Bound the mixed term $\langle e_k-e_{k-1},e_k\rangle$ by Young’s inequality:
\[
2\beta\langle e_k-e_{k-1},e_k\rangle
\le \beta\|e_k-e_{k-1}\|^2+\beta\|e_k\|^2 .
\tag{6}
\]

\item[Step 7.]  Insert (6) into (5) and collect the $\|e_k-e_{k-1}\|^2$ terms:
\begin{align}
\|e_{k+1}\|^2
&\le \bigl(1-2\alpha\mu+\beta\bigr)\|e_k\|^2
+\bigl(\beta^2+\beta\bigr)\|e_k-e_{k-1}\|^2
-2\alpha\bigl(f(x_k)-f(x^\star)\bigr)
+\alpha^2L\langle g_k,e_k\rangle .
\tag{7}
\end{align}

\item[Step 8.]  Replace $\langle g_k,e_k\rangle$ again using (3):
\[
\alpha^2L\langle g_k,e_k\rangle
\le \alpha^2L\Bigl[f(x_k)-f(x^\star)+\tfrac{\mu}{2}\|e_k\|^2\Bigr].
\tag{8}
\]

\item[Step 9.]  Plug (8) into (7):
\begin{align}
\|e_{k+1}\|^2
&\le \Bigl(1-2\alpha\mu+\beta+\tfrac{\alpha^2L\mu}{2}\Bigr)\|e_k\|^2
+\bigl(\beta^2+\beta\bigr)\|e_k-e_{k-1}\|^2
\notag\\
&\qquad
-2\alpha\bigl(1-\tfrac{\alpha L}{2}\bigr)\bigl(f(x_k)-f(x^\star)\bigr).
\tag{9}
\end{align}

\item[Step 10.]  Define the Lyapunov function $V_k$ as in Lemma~\ref{lem:lyap} with a yet‑to‑be‑chosen $a>0$:
\[
V_k:= f(x_k)-f(x^\star)+\frac{a}{2}\|e_k-e_{k-1}\|^2 .
\tag{10}
\]

\item[Step 11.]  Multiply inequality (9) by $a/2$ and add $f(x_{k+1})-f(x^\star)$ to both sides. Using convexity of $f$ we have
\[
f(x_{k+1})-f(x^\star)\le \langle g_{k+1},e_{k+1}\rangle .
\tag{11}
\]
Applying (3) at $k+1$ yields
\[
f(x_{k+1})-f(x^\star)\le \langle g_{k+1},e_{k+1}\rangle
\le \frac{1}{2\alpha}\|e_{k+1}\|^2
+\frac{\alpha}{2}L\bigl(f(x_{k+1})-f(x^\star)\bigr) .
\tag{12}
\]

\item[Step 12.]  Rearranging (12) gives
\[
\bigl(1-\tfrac{\alpha L}{2}\bigr)\bigl(f(x_{k+1})-f(x^\star)\bigr)
\le \frac{1}{2\alpha}\|e_{k+1}\|^2 .
\tag{13}
\]

\item[Step 13.]  Combine (9) and (13) with the definition (10):
\begin{align}
V_{k+1}
&= f(x_{k+1})-f(x^\star)+\frac{a}{2}\|e_{k+1}-e_k\|^2 \nonumber\\
&\le \frac{1}{2\alpha}\|e_{k+1}\|^2
+\frac{a}{2}\|e_{k+1}-e_k\|^2   \qquad\text{(by (13))} \nonumber\\
&\le \underbrace{\Bigl(\frac{1}{2\alpha}c_1 + \frac{a}{2}c_2\Bigr)}_{=: \rho_1}
\bigl(f(x_k)-f(x^\star)\bigr)
+\underbrace{\Bigl(\frac{1}{2\alpha}c_3 + \frac{a}{2}c_4\Bigr)}_{=: \rho_2}
\|e_k-e_{k-1}\|^2,
\tag{14}
\end{align}
where the constants $c_i$ are read off from (9) after multiplying by $a/2$:
\[
c_1 = 2\alpha\bigl(1-\tfrac{\alpha L}{2}\bigr),\quad
c_3 = 1-2\alpha\mu+\beta+\tfrac{\alpha^2L\mu}{2},
\]
\[
c_2 = \beta^2+\beta,\qquad
c_4 = 0 .
\]

\item[Step 14.]  Choose $a$ such that the coefficients of $V_k$ are equal, i.e.
\[
\rho_1=\rho_2=: \rho<1 .
\tag{15}
\]
Solving for $a$ yields
\[
a = \frac{2\alpha\bigl(1-\tfrac{\alpha L}{2}\bigr)\bigl(1-2\alpha\mu+\beta+\tfrac{\alpha^2L\mu}{2}\bigr)}
{(\beta^2+\beta)-2\alpha\bigl(1-\tfrac{\alpha L}{2}\bigr)} .
\tag{16}
\]
Assumption~\ref{ass:param} guarantees that the denominator in (16) is positive and that the resulting $a$ satisfies the Lyapunov admissibility condition (L1). Consequently $\rho\in(0,1)$.

\item[Step 15.]  With this choice of $a$, inequality (14) becomes
\[
V_{k+1}\le\rho V_k,\qquad\forall k\ge1 .
\tag{17}
\]

\item[Step 16.]  Unfolding the recursion gives
\[
V_k\le\rho^{\,k-1}V_1\le C\rho^{\,k},
\quad C:=\max\{V_0,V_1\}.
\tag{18}
\]

\item[Step 17.]  Since $V_k\ge f(x_k)-f(x^\star)$ and $f$ is $\mu$‑strongly convex,
\[
\|e_k\|^2\le \frac{2}{\mu}\bigl(f(x_k)-f(x^\star)\bigr)
\le \frac{2C}{\mu}\rho^{\,k}.
\tag{19}
\]
Define $C'=\sqrt{2C/\mu}$ to obtain the claimed linear bound
\[
\|e_k\|\le C'\rho^{\,k/2}\le \tilde C\rho^{\,k},
\]
which proves Theorem~\ref{thm:linear}. \hfill $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Lemma~\ref{lem:spectral} the exact contraction factor is
\[
\rho = \max_{\lambda\in[\mu,L]}
\left|\frac{1-\alpha\lambda+\beta+\sqrt{(1-\alpha\lambda+\beta)^2-4\beta}}{2}\right|.
\]
When the optimal parameters
\[
\alpha^\star=\frac{4}{(\sqrt{L}+\sqrt{\mu})^2},
\qquad
\beta^\star=\left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2,
\]
are used, the spectral radius simplifies to
\[
\rho^\star=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}
= \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1},
\qquad \kappa:=\frac{L}{\mu}.
\]
Thus the Heavy–Ball method attains the linear rate
\[
\|x_k-x^\star\|\le C\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^{\!k},
\]
which is a factor $\mathcal O(\sqrt{\kappa})$ improvement over gradient descent’s rate $1-\alpha\mu$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Quadratic functions.} If $f(x)=\frac12 x^\top H x - b^\top x$ with $H$ satisfying $\mu I\preceq H\preceq L I$, the HB iteration coincides exactly with the linear system analysed in Lemma~\ref{lem:spectral}; the bound $\rho$ is tight.
\item \textbf{No momentum ($\beta=0$).} The Lyapunov function reduces to $V_k=f(x_k)-f(x^\star)$ and (17) becomes the classic gradient‑descent inequality $V_{k+1}\le(1-\alpha\mu)V_k$, i.e. $\rho=1-\alpha\mu$.
\item \textbf{Over‑damped regime ($\beta\ge (1-\alpha\mu)(1-\alpha L)$).} The spectral radius exceeds one, violating Assumption~\ref{ass:param}; the iterates may oscillate or diverge, illustrating the necessity of the parameter constraints.
\end{enumerate}
