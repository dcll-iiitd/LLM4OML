\section*{Proximal Gradient Method for Non‑Convex Smooth $f$}

\section*{Mathematical Formulation}
Let 
\[
\Phi(x)\;:=\;f(x)+g(x),\qquad x\in\mathbb{R}^{d},
\]
where 
\begin{itemize}[nosep]
  \item $f:\mathbb{R}^{d}\!\to\!\mathbb{R}$ is differentiable (possibly non‑convex) with $L$‑Lipschitz continuous gradient,
  \item $g:\mathbb{R}^{d}\!\to\!\mathbb{R}\cup\{+\infty\}$ is proper, closed and its proximal operator
  \[
  \operatorname{prox}_{\eta g}(v)
  \;:=\;
  \arg\min_{u\in\mathbb{R}^{d}}\Big\{g(u)+\tfrac{1}{2\eta}\|u-v\|^{2}\Big\}
  \]
  is assumed to be efficiently computable.
\end{itemize}
Given a stepsize $\eta>0$, the iterative scheme is
\begin{equation}
\boxed{\;
x_{k+1}
\;=\;
\operatorname{prox}_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),
\qquad k=0,1,2,\dots
\;}
\tag{1}
\end{equation}
Define the \emph{gradient mapping} (also called proximal gradient) as
\begin{equation}
G_{\eta}(x_{k})
\;:=\;
\frac{1}{\eta}\bigl(x_{k}-x_{k+1}\bigr)
\;=\;
\frac{1}{\eta}\Bigl(x_{k}
-\operatorname{prox}_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr)\Bigr).
\tag{2}
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[h]
\caption{Proximal Gradient Method (PGM)}\label{alg:pgm}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\mathbb{R}^{d}$, stepsize $\eta>0$, maximum iterations $T$.
\For{$k=0$ \textbf{to} $T-1$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$.
    \State Take a proximal step 
    \[
    x_{k+1}\gets\operatorname{prox}_{\eta g}\bigl(x_{k}-\eta g_{k}\bigr).
    \]
    \State (Optional) Check stopping criterion 
    $\|G_{\eta}(x_{k})\| \le \epsilon$.
\EndFor
\State \textbf{return} $x_{T}$ (or the last iterate satisfying the criterion).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so that $\operatorname{prox}_{\eta g}= \text{id}$.  
Take $w_{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{enumerate}[nosep]
\item $k=0$: $\nabla f(w_{0})=2(w_{0}-3)=-6$.
\[
w_{1}=w_{0}-\eta\nabla f(w_{0})=0-0.1(-6)=0.6,
\]
\[
f(w_{1})=(0.6-3)^{2}=5.76.
\]

\item $k=1$: $\nabla f(w_{1})=2(w_{1}-3)=2(0.6-3)=-4.8$.
\[
w_{2}=w_{1}-0.1(-4.8)=0.6+0.48=1.08,
\]
\[
f(w_{2})=(1.08-3)^{2}=3.6864.
\]

\item $k=2$: $\nabla f(w_{2})=2(1.08-3)=-3.84$.
\[
w_{3}=w_{2}-0.1(-3.84)=1.08+0.384=1.464,
\]
\[
f(w_{3})=(1.464-3)^{2}=2.3689.
\]
\end{enumerate}
The objective value decreases monotonically, illustrating the descent property of (1).

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{asm:smooth}
$f$ is differentiable and its gradient is $L$‑Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded Below]\label{asm:bounded}
The composite objective $\Phi$ is bounded from below:
\[
\Phi_{\inf}:=\inf_{x\in\mathbb{R}^{d}}\Phi(x)>-\infty .
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{asm:stepsize}
The stepsize satisfies
\[
0<\eta\le \frac{1}{L}.
\]
\end{assumption}
\begin{assumption}[Proximal Operator]\label{asm:prox}
For every $v\in\mathbb{R}^{d}$ the proximal map $\operatorname{prox}_{\eta g}(v)$ is well defined and \emph{non‑expansive}:
\[
\big\|\operatorname{prox}_{\eta g}(v)-\operatorname{prox}_{\eta g}(w)\big\|
\le\|v-w\|,\qquad\forall v,w\in\mathbb{R}^{d}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of the proximal gradient method]\label{thm:main}
Under Assumptions \ref{asm:smooth}–\ref{asm:prox} and with stepsize $\eta\le 1/L$, the iterates generated by \eqref{alg:pgm} satisfy for any $T\ge1$
\[
\frac{1}{T}\sum_{k=0}^{T-1}\big\|G_{\eta}(x_{k})\big\|^{2}
\;\le\;
\frac{2\big(\Phi(x_{0})-\Phi_{\inf}\big)}{\eta\,T}.
\tag{3}
\]
Consequently $\displaystyle\liminf_{k\to\infty}\|G_{\eta}(x_{k})\|=0$ and every limit point of $\{x_{k}\}$ is a stationary point of $\Phi$, i.e.
\[
0\in\nabla f(\bar x)+\partial g(\bar x).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[nosep]
\item \textbf{Descent Property.}  From the proof we obtain a stronger pointwise inequality
\[
\Phi(x_{k+1})\;\le\;\Phi(x_{k})-\frac{\eta}{2}\,\|G_{\eta}(x_{k})\|^{2},
\tag{4}
\]
which holds for every iteration.
\item \textbf{Hyper‑parameter Sensitivity.}  The bound \eqref{3} scales linearly with $1/\eta$.  Choosing $\eta$ close to $1/L$ yields the fastest guaranteed decrease.
\item \textbf{Comparison to Gradient Descent.}  When $g\equiv0$, $G_{\eta}(x_{k})=\nabla f(x_{k})$ and \eqref{3} reduces to the classic $O(1/T)$ bound for (non‑convex) gradient descent.
\item \textbf{Stability.}  The non‑expansiveness of $\operatorname{prox}_{\eta g}$ guarantees that the method cannot ``explode'' even if $g$ is nonsmooth.
\end{enumerate}

\section*{Discussion}
The theorem matches the best known rates for first‑order methods applied to non‑convex composite problems.  The analysis hinges only on smoothness of $f$ and the firm non‑expansiveness of the proximal map; convexity of $g$ is not required for the descent inequality, though it guarantees existence of the proximal point.  Extensions to stochastic gradients or adaptive stepsizes follow the same template by inserting unbiasedness and variance bounds.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $\nabla f$ is $L$‑Lipschitz, then for all $x,y\in\mathbb{R}^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient and the Cauchy–Schwarz inequality. ∎
\end{proof}
\begin{lemma}[Non‑expansiveness of the proximal operator]\label{lem:nonexp}
For any $v,w\in\mathbb{R}^{d}$,
\[
\big\|\operatorname{prox}_{\eta g}(v)-\operatorname{prox}_{\eta g}(w)\big\|
\le\|v-w\|.
\tag{6}
\]
\end{lemma}
\begin{proof}
Follows from the firm non‑expansiveness of proximal maps (see e.g. \cite{Parikh2014}). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we set $x_{k+1}$ as in \eqref{1} and denote $G_{k}=G_{\eta}(x_{k})$.

\noindent\textbf{[Step 1]}  \emph{Optimality condition of the proximal step}.  
By definition of $\operatorname{prox}_{\eta g}$,
\[
x_{k+1}
=\arg\min_{u}\Big\{g(u)+\frac{1}{2\eta}\|u-(x_{k}-\eta\nabla f(x_{k}))\|^{2}\Big\}.
\]
First‑order optimality gives
\[
0\in\nabla f(x_{k})+\frac{1}{\eta}(x_{k+1}-x_{k})+\partial g(x_{k+1}).
\tag{7}
\]
Rearranging,
\[
-\nabla f(x_{k})-\frac{1}{\eta}(x_{k+1}-x_{k})\in\partial g(x_{k+1}).
\tag{8}
\]

\noindent\textbf{[Step 2]}  \emph{Descent inequality for $\Phi$.}  
Using Lemma \ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{9}
\]
Add $g(x_{k+1})$ to both sides and employ the subgradient inequality for $g$ with the subgradient $s_{k+1}:=-\nabla f(x_{k})-\frac{1}{\eta}(x_{k+1}-x_{k})$ from \eqref{8}:
\[
g(x_{k+1})\le g(x_{k})+\langle s_{k+1},x_{k+1}-x_{k}\rangle.
\tag{10}
\]
Summing \eqref{9} and \eqref{10} yields
\[
\Phi(x_{k+1})
\le \Phi(x_{k})
+\langle\nabla f(x_{k})+s_{k+1},x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 3]}  \emph{Cancel the inner product}.  
From \eqref{8},
\[
\nabla f(x_{k})+s_{k+1}
= -\frac{1}{\eta}(x_{k+1}-x_{k}),
\]
hence
\[
\langle\nabla f(x_{k})+s_{k+1},x_{k+1}-x_{k}\rangle
= -\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 4]}  \emph{Insert \eqref{12} into \eqref{11}}:
\[
\Phi(x_{k+1})
\le \Phi(x_{k})-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{13}
\]
Because $\eta\le 1/L$ (Assumption \ref{asm:stepsize}), we have $L\eta\le1$ and therefore
\[
-\frac{1}{\eta}+\frac{L}{2}
\;=\;
-\frac{1}{2\eta}\bigl(2-L\eta\bigr)
\;\le\;
-\frac{1}{2\eta}.
\tag{14}
\]
Applying \eqref{14} to \eqref{13} gives the \emph{pointwise descent} inequality
\[
\Phi(x_{k+1})
\le \Phi(x_{k})-\frac{\eta}{2}\,\big\|G_{\eta}(x_{k})\big\|^{2},
\tag{15}
\]
which is exactly \eqref{4}.  This completes the derivation of the descent property.

\noindent\textbf{[Step 5]}  \emph{Summation and telescoping}.  
Sum \eqref{15} from $k=0$ to $T-1$:
\[
\Phi(x_{T})\le \Phi(x_{0})-\frac{\eta}{2}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}.
\tag{16}
\]
Since $\Phi(x_{T})\ge\Phi_{\inf}$ by Assumption \ref{asm:bounded},
\[
\frac{\eta}{2}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}
\le \Phi(x_{0})-\Phi_{\inf}.
\tag{17}
\]

\noindent\textbf{[Step 6]}  \emph{Average bound}.  
Dividing \eqref{17} by $\eta T$ yields exactly the rate \eqref{3}:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}
\le\frac{2\big(\Phi(x_{0})-\Phi_{\inf}\big)}{\eta\,T}.
\tag{18}
\]

\noindent\textbf{[Step 7]}  \emph{Stationarity of limit points}.  
From \eqref{18} the right‑hand side tends to zero as $T\to\infty$, implying $\liminf_{k\to\infty}\|G_{\eta}(x_{k})\|=0$.  Let $\{x_{k_{j}}\}$ be a subsequence with $G_{\eta}(x_{k_{j}})\to0$ and $x_{k_{j}}\to\bar x$.  Passing to the limit in the optimality condition \eqref{8} (using closedness of $\partial g$) gives
\[
0\in\nabla f(\bar x)+\partial g(\bar x),
\]
i.e. $\bar x$ is a stationary point of $\Phi$.

Thus every claim of Theorem \ref{thm:main} is proved.  ∎

\section*{Rate Derivation (With Constants)}
From \eqref{18} we can write the explicit constant $C$:
\[
C:=\frac{2\big(\Phi(x_{0})-\Phi_{\inf}\big)}{\eta}.
\]
Hence the bound reads
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}\le\frac{C}{T}=O\!\left(\frac{1}{T}\right).
\]
If additionally $\Phi$ satisfies the {\it Polyak‑Łojasiewicz} inequality
\[
\frac{1}{2}\|G_{\eta}(x)\|^{2}\ge\mu\big(\Phi(x)-\Phi_{\inf}\big),
\]
the descent \eqref{15} yields a linear convergence rate
\[
\Phi(x_{k})-\Phi_{\inf}\le (1-\eta\mu)^{k}\big(\Phi(x_{0})-\Phi_{\inf}\big).
\]

\section*{Special Cases}
\begin{enumerate}[nosep]
\item \textbf{Pure gradient descent ($g\equiv0$).}  $G_{\eta}(x_{k})=\nabla f(x_{k})$, and \eqref{3} becomes the classic $O(1/T)$ bound for non‑convex smooth optimization.
\item \textbf{Indicator function of a convex set $C$.}  $g=\iota_{C}$, then $\operatorname{prox}_{\eta g}$ reduces to the Euclidean projection onto $C$.  The algorithm coincides with the projected gradient method and the same convergence guarantee holds.
\item \textbf{Lasso regularizer $g(x)=\lambda\|x\|_{1}$.}  The proximal step is soft‑thresholding; the theorem guarantees that the average squared soft‑thresholding residual decays as $O(1/T)$.
\end{enumerate}
