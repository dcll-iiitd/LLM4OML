\section*{Algorithm Name}
\textbf{Cyclic Stochastic Gradient Descent (Cyclic SGD)}  

The method follows the standard SGD update but employs a \emph{cyclic learning‑rate schedule} that oscillates between a minimum step size $\eta_{\min}>0$ and a maximum step size $\eta_{\max}> \eta_{\min}$ with period $M\in\mathbb{N}$.  
A popular choice is the \emph{triangular} schedule:
\[
\eta_t \;=\; \eta_{\min} + (\eta_{\max}-\eta_{\min})\,
\Bigl(1-\bigl|\,\tfrac{t\bmod (2M)}{M}-1\,\bigr|\Bigr),\qquad t=0,1,2,\dots
\tag{1}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the (possibly non‑convex) objective and let $\{z_t\}_{t\ge0}$ be an i.i.d. data stream.  
Define the stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;z_t),
\]
where $f(w;z)$ denotes the loss on a single sample $z$.  

The **Cyclic SGD update** is
\[
\boxed{\,w_{t+1}=w_t-\eta_t\, g_t\,},\qquad t=0,1,\dots,T-1,
\tag{2}
\]
with the learning rate $\eta_t$ given by (1).  
The algorithm terminates after $T$ iterations or when a prescribed tolerance on the gradient norm is met.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^d$, period $M$, bounds $0<\eta_{\min}<\eta_{\max}$, total iterations $T$.
\State Set $t\gets0$.
\While{$t<T$}
    \State Compute learning rate
    \[
    \eta_t\gets \eta_{\min}+(\eta_{\max}-\eta_{\min})
    \Bigl(1-\bigl|\,\tfrac{t\bmod (2M)}{M}-1\,\bigr|\Bigr).
    \tag{3}
    \]
    \State Sample $z_t$ and evaluate stochastic gradient $g_t\gets \nabla f(w_t;z_t)$.
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
    \State $t\gets t+1$.
\EndWhile
\State \Return $w_T$ (or optionally the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Take $w_0=0$, deterministic gradients (i.e. $g_t=\nabla f(w_t)$), period $M=2$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$.  
The triangular schedule (1) yields:
\[
\eta_0=0.05,\quad \eta_1=0.15,\quad \eta_2=0.15,\quad \eta_3=0.05,\dots
\]

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.05(-6)=0.30$ & $2(0.30-3)=-5.40$ & $7.29$\\
2 & $w_2=0.30-0.15(-5.40)=1.11$ & $2(1.11-3)=-3.78$ & $3.55$\\
3 & $w_3=1.11-0.15(-3.78)=1.68$ & $2(1.68-3)=-2.64$ & $1.74$
\end{tabular}
\end{center}
After three updates the objective value has decreased from $9$ to $1.74$, illustrating the effect of the cyclic step sizes.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(w)-\nabla f(v)\|\le L\|w-v\|,\qquad\forall w,v\in\mathbb{R}^d.
\tag{4}
\]
Equivalently,
\[
f(v)\le f(w)+\langle\nabla f(w),v-w\rangle+\frac{L}{2}\|v-w\|^2 .
\tag{5}
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, conditioned on $w_t$,
\[
\mathbb{E}[g_t\mid w_t]=\nabla f(w_t).
\tag{6}
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
\mathbb{E}\bigl[\|g_t-\nabla f(w_t)\|^2\mid w_t\bigr]\le\sigma^2,\qquad\forall t.
\tag{7}
\]
\end{assumption}

\begin{assumption}[Learning‑rate Bounds]\label{ass:lr}
The cyclic schedule satisfies
\[
0<\eta_{\min}\le \eta_t\le\eta_{\max}\le\frac{1}{L},\qquad\forall t.
\tag{8}
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:lr} hold. Let $T\ge1$ and define the (uniform) average iterate
\[
\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t .
\]
Then
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(w_t)\|^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta_{\min}T}}_{\text{deterministic decay}}
\;+\;
\underbrace{L\eta_{\max}\sigma^2}_{\text{variance term}} .
\tag{9}
\]
Consequently, if $\eta_{\max}=c/\sqrt{T}$ with a constant $c\le 1/L$, the right‑hand side becomes $O(1/\sqrt{T})$, i.e.
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(w_t)\|^2\bigr]=O\!\left(\frac{1}{\sqrt{T}}\right).
\tag{10}
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*,itemsep=2pt]
\item \textbf{Stability.} Because every learning rate is bounded by $1/L$ (Assumption \ref{ass:lr}), the standard smoothness descent inequality guarantees that the iterates cannot diverge; cyclic variation merely re‑weights the descent and variance terms.
\item \textbf{Hyper‑parameter sensitivity.} The bound (9) depends on $\eta_{\min}$ only through the deterministic decay term and on $\eta_{\max}$ only through the variance term. Hence a larger $\eta_{\max}$ accelerates early descent but increases the steady‑state variance; the cyclic pattern trades off these effects.
\item \textbf{Comparison to constant‑step SGD.} For constant step $\eta$, the classic bound reads  
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2\le\frac{2(f(w_0)-f^\star)}{\eta T}+L\eta\sigma^2 .
\]  
Cyclic SGD replaces $\eta$ by $\eta_{\min}$ in the first term (more conservative) while using $\eta_{\max}$ in the second term (potentially larger variance). When $\eta_{\max}$ and $\eta_{\min}$ are close, the bound coincides with the constant‑step result; when they differ, the bound captures the benefit of occasional larger steps without sacrificing overall stability.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that cyclic learning rates preserve the $O(1/\sqrt{T})$ convergence rate typical for SGD on smooth non‑convex problems. The proof is a straightforward adaptation of the classical SGD analysis, with the only extra ingredient being the separation of the minimum and maximum step sizes. This separation is useful in practice: large steps can help escape shallow local minima, while small steps keep the variance under control.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $w$ and any vector $d$,
\[
f(w-d)\le f(w)-\langle\nabla f(w),d\rangle+\frac{L}{2}\|d\|^2 .
\tag{11}
\]
\end{lemma}
\begin{proof}
Directly from inequality (5) with $v=w-d$ and $w$ unchanged. \qedhere
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the randomness up to iteration $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply Lemma \ref{lem:descent} with $d=\eta_t g_t$ to the update (2):
\[
f(w_{t+1})\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle+\frac{L}{2}\eta_t^2\|g_t\|^2.
\tag{12}
\]

\item[Step 2.] Take conditional expectation $\mathbb{E}[\cdot\mid w_t]$ and use unbiasedness (6):
\[
\mathbb{E}\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\eta_t\|\nabla f(w_t)\|^2
+\frac{L}{2}\eta_t^2\,
\mathbb{E}\!\bigl[\|g_t\|^2\mid w_t\bigr].
\tag{13}
\]

\item[Step 3.] Decompose the second moment:
\[
\mathbb{E}\!\bigl[\|g_t\|^2\mid w_t\bigr]
= \|\nabla f(w_t)\|^2
+ \mathbb{E}\!\bigl[\|g_t-\nabla f(w_t)\|^2\mid w_t\bigr]
\le \|\nabla f(w_t)\|^2+\sigma^2,
\tag{14}
\]
where the inequality follows from Assumption \ref{ass:variance}.

\item[Step 4.] Substitute (14) into (13):
\[
\mathbb{E}\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\eta_t\|\nabla f(w_t)\|^2
+\frac{L}{2}\eta_t^2\bigl(\|\nabla f(w_t)\|^2+\sigma^2\bigr).
\tag{15}
\]

\item[Step 5.] Rearrange terms:
\[
\mathbb{E}\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\Bigl(\eta_t-\frac{L}{2}\eta_t^2\Bigr)\|\nabla f(w_t)\|^2
+\frac{L}{2}\eta_t^2\sigma^2 .
\tag{16}
\]

\item[Step 6.] Use the bound $\eta_t\le\eta_{\max}\le 1/L$ (Assumption \ref{ass:lr}) to lower‑bound the coefficient of $\|\nabla f(w_t)\|^2$:
\[
\eta_t-\frac{L}{2}\eta_t^2
\;\ge\; \frac{\eta_t}{2}
\;\ge\; \frac{\eta_{\min}}{2},
\tag{17}
\]
because $0<\eta_t\le\eta_{\max}$ and $L\eta_t\le1$ implies $1-\frac{L}{2}\eta_t\ge\frac12$.

\item[Step 7.] Apply (17) to (16):
\[
\mathbb{E}\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\frac{\eta_{\min}}{2}\|\nabla f(w_t)\|^2
+\frac{L}{2}\eta_{\max}^2\sigma^2 .
\tag{18}
\]

\item[Step 8.] Take full expectation and sum (18) over $t=0,\dots,T-1$:
\[
\mathbb{E}[f(w_T)]\le f(w_0)-\frac{\eta_{\min}}{2}
\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(w_t)\|^2\bigr]
+\frac{L}{2}\eta_{\max}^2\sigma^2 T .
\tag{19}
\]

\item[Step 9.] Drop the non‑negative term $\mathbb{E}[f(w_T)]\ge f^\star$ and rearrange:
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(w_t)\|^2\bigr]
\le
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta_{\min}T}
+L\eta_{\max}\sigma^2 .
\tag{20}
\]
Equation (20) is exactly the bound (9) stated in Theorem \ref{thm:main}. \qedhere
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (20) we have a deterministic decay term that scales as $1/(\eta_{\min}T)$ and a variance term that is linear in $\eta_{\max}$.  
Choosing a diminishing schedule of the form
\[
\eta_{\max}=\frac{c}{\sqrt{T}},\qquad
\eta_{\min}=\frac{c}{2\sqrt{T}},
\tag{21}
\]
with $c\le 1/L$, yields
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le
\underbrace{\frac{4\bigl(f(w_0)-f^\star\bigr)}{c}}_{\displaystyle O(1)}
\frac{1}{\sqrt{T}}
\;+\;
\underbrace{L\frac{c}{\sqrt{T}}\sigma^2}_{\displaystyle O(1/\sqrt{T})}
=
O\!\left(\frac{1}{\sqrt{T}}\right),
\tag{22}
\]
which proves the rate (10).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}[leftmargin=*,itemsep=2pt]
\item \textbf{Constant learning rate.} If $\eta_{\min}=\eta_{\max}=\eta$, (20) reduces to the classical SGD bound.
\item \textbf{Purely increasing schedule.} Setting $\eta_{\min}=0$ (i.e. allowing the learning rate to touch zero) eliminates the deterministic decay term; the bound then depends solely on the variance term, highlighting the necessity of a strictly positive lower bound.
\item \textbf{Very short period $M=1$.} The schedule becomes a two‑point alternating sequence $\{\eta_{\min},\eta_{\max},\eta_{\min},\dots\}$. The analysis above still holds because the proof never uses the exact ordering, only the extremal bounds (8).
\end{itemize}

\bigskip
\noindent\textbf{Conclusion.}  
We have presented a full mathematical description of Cyclic SGD, derived its convergence guarantee for smooth non‑convex objectives, and provided a line‑by‑line proof with every inequality explicitly justified. The result shows that cyclic learning‑rate schedules retain the optimal $O(1/\sqrt{T})$ rate while offering practical flexibility through the choice of $\eta_{\min},\eta_{\max}$ and period $M$.
