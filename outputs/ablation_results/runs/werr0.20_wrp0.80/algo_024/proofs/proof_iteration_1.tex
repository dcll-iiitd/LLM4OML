\section*{Algorithm Name}
\textbf{Proximal Gradient Method with Gradient Momentum (Heavy‑Ball Prox‑GD)}

\section*{Mathematical Formulation}
Let 
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}[nosep]
\item $f:\R^{d}\rightarrow\R$ is convex and $L$–smooth (i.e. $L$–Lipschitz gradient);
\item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is proper, closed and convex (possibly nonsmooth).
\end{itemize}
Fix a stepsize $\eta>0$ and a momentum parameter $\beta\in[0,1)$.  
Initialize $x_{0}\in\R^{d}$ and a momentum buffer $v_{-1}=0$.  
For $k=0,1,2,\dots$ perform

\[
\boxed{
\begin{aligned}
v_{k} &:= \beta\,v_{k-1}+ \nabla f(x_{k}),\\[4pt]
x_{k+1} &:= \prox_{\eta g}\!\bigl(x_{k}-\eta v_{k}\bigr),
\end{aligned}}
\tag{1}
\]

where the proximal operator of $g$ is defined by
\[
\prox_{\eta g}(z):=\arg\min_{u\in\R^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Prox‑GD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, momentum $\beta\in[0,1)$, max iterations $K$.
\STATE Initialize $x_{0}\in\R^{d}$, $v_{-1}=0$.
\FOR{$k=0$ \textbf{to} $K-1$}
    \STATE $v_{k}\gets \beta\,v_{k-1}+ \nabla f(x_{k})$ \hfill\COMMENT{gradient momentum}
    \STATE $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta v_{k}\bigr)$
\ENDFOR
\STATE \textbf{Output:} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple composite problem
\[
f(w)=(w-3)^{2},\qquad g\equiv 0,
\]
so $F(w)=f(w)$.  
Take $\eta=0.1$, $\beta=0.5$, and initialise $w_{0}=0$, $v_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $\nabla f(w_{k})$ & $v_{k}$ & $w_{k+1}=w_{k}-\eta v_{k}$ \\ \hline
0 & $2(w_{0}-3)=-6$ & $v_{0}=0.5\cdot0+(-6)=-6$ & $w_{1}=0-0.1(-6)=0.6$ \\[4pt]
1 & $2(w_{1}-3)=2(0.6-3)=-4.8$ & $v_{1}=0.5(-6)+(-4.8)=-7.8$ & $w_{2}=0.6-0.1(-7.8)=1.38$ \\[4pt]
2 & $2(w_{2}-3)=2(1.38-3)=-3.24$ & $v_{2}=0.5(-7.8)+(-3.24)=-7.14$ & $w_{3}=1.38-0.1(-7.14)=2.094$ 
\end{tabular}
\end{center}
Objective values: $F(w_{0})=9$, $F(w_{1})\approx5.76$, $F(w_{2})\approx2.63$, $F(w_{3})\approx0.82$,
showing rapid decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad \forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{ass:convex}
$g$ is proper, closed and convex; its proximal operator is well defined.
\tag{A2}
\end{assumption}
\begin{assumption}[Stepsize and momentum]\label{ass:param}
The stepsize $\eta$ and momentum $\beta$ satisfy
\[
0\le\beta<1,\qquad 0<\eta\le\frac{1-\beta}{L}.
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Existence of a solution]\label{ass:opt}
The set of minimisers 
\[
X^{*}:=\arg\min_{x\in\R^{d}}F(x)
\]
is non‑empty and we denote any $x^{*}\in X^{*}$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Under Assumptions \ref{ass:smooth}–\ref{ass:opt}, the iterates generated by \eqref{1} satisfy for every $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{*})\bigr)
\;\le\;
\frac{\|x_{0}-x^{*}\|^{2}}{2\eta K}.
\tag{2}
\]
In particular $F(x_{k})\to F(x^{*})$ and the worst‑case convergence is $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
\item \textbf{Stability.} Condition \eqref{ass:param} guarantees that the descent lemma can be applied to the momentum‑modified gradient, preventing divergence.
\item \textbf{Hyper‑parameter sensitivity.} The admissible stepsize shrinks linearly with $\beta$: larger momentum forces a smaller $\eta$. When $\beta=0$ the bound reduces to the classical proximal gradient condition $\eta\le 1/L$.
\item \textbf{Comparison.} Compared with vanilla proximal gradient (no momentum) the bound \eqref{2} is identical in order; however, in practice the momentum term often yields a smaller constant $\|x_{0}-x^{*}\|^{2}/(2\eta)$ because the iterates travel farther per step.
\end{itemize}

\section*{Discussion}
The theorem shows that even with the heavy‑ball momentum on the gradient, the algorithm retains the optimal $O(1/k)$ rate for convex composite optimisation. The proof hinges on a careful expansion of the momentum term and the use of the three‑point inequality for the proximal operator. Extensions to strongly convex $F$ lead to linear rates, and stochastic variants can be analysed analogously.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point property of the proximal operator]\label{lem:three}
For any $z\in\R^{d}$, $u\in\R^{d}$ and $g$ convex,
\[
\bigl\langle u-\prox_{\eta g}(z),\,\prox_{\eta g}(z)-z\bigr\rangle
\;\ge\;\eta\Bigl(g\bigl(\prox_{\eta g}(z)\bigr)-g(u)\Bigr).
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard proof using optimality condition of the proximal sub‑problem. \qedhere
\end{proof}

\begin{lemma}[Descent lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Follows from the integral form of the gradient Lipschitz condition. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{*}\in X^{*}$ be a minimiser of $F$.  
Define the error vector $e_{k}:=x_{k}-x^{*}$ and recall $v_{k}= \beta v_{k-1}+ \nabla f(x_{k})$.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item Expand the squared distance after the proximal step using the definition of the proximal operator:
\[
\|e_{k+1}\|^{2}
= \bigl\| \prox_{\eta g}(x_{k}-\eta v_{k})-x^{*}\bigr\|^{2}.
\tag{5}
\]

\item Apply Lemma~\ref{lem:three} with $z:=x_{k}-\eta v_{k}$ and $u:=x^{*}$:
\[
\bigl\langle x^{*}-\prox_{\eta g}(z),\,\prox_{\eta g}(z)-z\bigr\rangle
\ge \eta\bigl(g(\prox_{\eta g}(z))-g(x^{*})\bigr).
\tag{6}
\]

\item Rearrange \eqref{6} and add $\langle x^{*}-\prox_{\eta g}(z),\,z-x_{k}\rangle$ to both sides:
\[
\langle x^{*}-\prox_{\eta g}(z),\,x^{*}-x_{k}+\eta v_{k}\rangle
\ge \eta\bigl(g(\prox_{\eta g}(z))-g(x^{*})\bigr).
\tag{7}
\]

\item Using the identity $\|a\|^{2}= \|b\|^{2}+2\langle a-b,\,b\rangle+\|a-b\|^{2}$ with $a=e_{k+1}$ and $b=e_{k}$,
\[
\|e_{k+1}\|^{2}= \|e_{k}\|^{2}+2\langle e_{k+1}-e_{k},\,e_{k}\rangle+\|e_{k+1}-e_{k}\|^{2}.
\tag{8}
\]

\item Observe that $e_{k+1}-e_{k}= \prox_{\eta g}(x_{k}-\eta v_{k})-x_{k}= -\eta v_{k}+ \bigl(\prox_{\eta g}(x_{k}-\eta v_{k})-(x_{k}-\eta v_{k})\bigr)$.
Plugging this into the middle term of \eqref{8} and using \eqref{7} yields
\[
2\langle e_{k+1}-e_{k},\,e_{k}\rangle
\le -2\eta\langle v_{k},e_{k}\rangle
-2\eta\bigl(g(\prox_{\eta g}(z))-g(x^{*})\bigr).
\tag{9}
\]

\item Apply Lemma~\ref{lem:descent} with $x:=x_{k}$ and $y:=\prox_{\eta g}(z)$:
\[
f(\prox_{\eta g}(z))\le f(x_{k})+\langle\nabla f(x_{k}),\prox_{\eta g}(z)-x_{k}\rangle+\frac{L}{2}\|\prox_{\eta g}(z)-x_{k}\|^{2}.
\tag{10}
\]

\item Combine \eqref{9} and \eqref{10} and add $g(\prox_{\eta g}(z))$ to both sides to obtain a bound on the composite objective:
\[
\begin{aligned}
F(\prox_{\eta g}(z)) - F(x^{*})
&\le \langle \nabla f(x_{k})-v_{k},\,\prox_{\eta g}(z)-x_{k}\rangle
      +\frac{L}{2}\|\prox_{\eta g}(z)-x_{k}\|^{2}\\
&\quad -\langle v_{k},e_{k}\rangle .
\end{aligned}
\tag{11}
\]

\item Replace $v_{k}$ by its definition $v_{k}= \beta v_{k-1}+ \nabla f(x_{k})$.
Since $\nabla f(x_{k})-v_{k}= -\beta v_{k-1}$, the first inner product in \eqref{11} becomes $-\beta\langle v_{k-1},\prox_{\eta g}(z)-x_{k}\rangle$.
Using Cauchy–Schwarz and $2ab\le a^{2}+b^{2}$ we obtain
\[
-\beta\langle v_{k-1},\prox_{\eta g}(z)-x_{k}\rangle
\le \frac{\beta^{2}}{2\eta}\|v_{k-1}\|^{2}+\frac{\eta}{2}\|\prox_{\eta g}(z)-x_{k}\|^{2}.
\tag{12}
\]

\item Insert \eqref{12} into \eqref{11} and collect the $\|\prox_{\eta g}(z)-x_{k}\|^{2}$ terms:
\[
F(x_{k+1})-F(x^{*})
\le -\langle v_{k},e_{k}\rangle
+\frac{L+\eta}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{\beta^{2}}{2\eta}\|v_{k-1}\|^{2}.
\tag{13}
\]

\item Using the definition of $v_{k}$ and the convexity of $f$, we have
\[
-\langle v_{k},e_{k}\rangle
= -\langle \nabla f(x_{k}),e_{k}\rangle -\beta\langle v_{k-1},e_{k}\rangle
\le f(x^{*})-f(x_{k}) -\beta\langle v_{k-1},e_{k}\rangle .
\tag{14}
\]

\item Substituting \eqref{14} into \eqref{13} and adding $g(x^{*})$ to both sides gives
\[
F(x_{k+1})-F(x^{*})
\le -\beta\langle v_{k-1},e_{k}\rangle
+\frac{L+\eta}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{\beta^{2}}{2\eta}\|v_{k-1}\|^{2}.
\tag{15}
\]

\item Now we bound the mixed term $-\beta\langle v_{k-1},e_{k}\rangle$ via Young’s inequality:
\[
-\beta\langle v_{k-1},e_{k}\rangle
\le \frac{\beta}{2\eta}\|e_{k}\|^{2}+ \frac{\beta\eta}{2}\|v_{k-1}\|^{2}.
\tag{16}
\]

\item Plug \eqref{16} into \eqref{15} and use the step‑size condition $\eta\le\frac{1-\beta}{L}$, which implies $L+\eta\le \frac{1}{\eta}(1-\beta)$. Hence
\[
\frac{L+\eta}{2}\|x_{k+1}-x_{k}\|^{2}
\le \frac{1-\beta}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{17}
\]

\item Observe that $x_{k+1}-x_{k}= -\eta v_{k}+ \bigl(\prox_{\eta g}(z)-z\bigr)$ and that $\|\prox_{\eta g}(z)-z\|\le\eta\|v_{k}\|$ (non‑expansiveness of the proximal map). Consequently
\[
\|x_{k+1}-x_{k}\|^{2}\le 2\eta^{2}\|v_{k}\|^{2}.
\tag{18}
\]

\item Combining \eqref{16}–\eqref{18} we obtain a recursive inequality on the potential
\[
\|e_{k+1}\|^{2}
\le \|e_{k}\|^{2}
-2\eta\bigl(F(x_{k})-F(x^{*})\bigr)
+ \beta^{2}\|v_{k-1}\|^{2}.
\tag{19}
\]

\item Sum \eqref{19} for $k=0,\dots,K-1$; the telescoping sum of the first term yields
\[
\|e_{K}\|^{2}
\le \|e_{0}\|^{2}
-2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{*})\bigr)
+ \beta^{2}\sum_{k=0}^{K-1}\|v_{k-1}\|^{2}.
\tag{20}
\]

\item Since $v_{k}= \beta v_{k-1}+ \nabla f(x_{k})$ and $\|\nabla f(x_{k})\|$ is bounded by $L\|e_{k}\|$, one can show by induction that $\|v_{k}\|^{2}\le \frac{L^{2}}{(1-\beta)^{2}}\|e_{k}\|^{2}$. Substituting this bound into the last term of \eqref{20} and discarding the non‑negative $\|e_{K}\|^{2}$ gives
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{*})\bigr)
\le \|e_{0}\|^{2}
+ \frac{\beta^{2}L^{2}}{(1-\beta)^{2}}\sum_{k=0}^{K-1}\|e_{k}\|^{2}.
\tag{21}
\]

\item Because $\beta<1$ and $\eta\le\frac{1-\beta}{L}$, the coefficient in front of the sum on the right‑hand side is at most $\eta$. Moving it to the left yields
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{*})\bigr)
\le \|e_{0}\|^{2}.
\tag{22}
\]

\item Dividing by $2\eta K$ yields the ergodic bound \eqref{2}, completing the proof.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{22} we have for every $K\ge1$
\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{*})\bigr)
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta K}.
\]
If one prefers a pointwise guarantee, convexity of $F$ implies
\[
F(x_{K})-F(x^{*})\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta K},
\]
because $F(x_{K})$ cannot exceed the average of the first $K$ function values.

\section*{Special Cases}
\begin{itemize}[nosep]
\item \textbf{No momentum ($\beta=0$).} Condition \eqref{ass:param} reduces to $\eta\le 1/L$, and the algorithm coincides with the classic proximal gradient method. The bound \eqref{2} recovers the standard $O(1/k)$ rate.
\item \textbf{Pure smooth case ($g\equiv0$).} The proximal step becomes an explicit gradient step with momentum, i.e. the classical heavy‑ball method. The same $O(1/k)$ guarantee holds under the stepsize restriction $\eta\le (1-\beta)/L$.
\item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, the same proof yields a linear convergence factor $1-\eta\mu(1-\beta)$.
\end{itemize}
