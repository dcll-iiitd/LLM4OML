\section*{Randomized Coordinate Descent with Importance Sampling (RCD-IS)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable (not necessarily convex).  
For each coordinate $i\in\{1,\dots,d\}$ denote the partial derivative
\[
\nabla_{i}f(w)=\frac{\partial f(w)}{\partial w_{i}} .
\]
Choose a probability distribution $\{p_{i}\}_{i=1}^{d}$ on the coordinates such that
\[
p_{i}>0,\qquad \sum_{i=1}^{d}p_{i}=1 .
\]

At iteration $t=0,1,2,\dots$ a coordinate $i_{t}$ is sampled independently
according to $p$ and only that coordinate is updated:
\[
w_{t+1}=w_{t}-\eta_{t}\,\frac{1}{p_{i_{t}}}\,e_{i_{t}}\nabla_{i_{t}}f(w_{t}),
\tag{1}
\]
where $e_{i}$ is the $i$‑th canonical basis vector and $\eta_{t}>0$ is a stepsize.
The factor $1/p_{i_{t}}$ compensates the non‑uniform sampling and will appear
in the expectation analysis.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD-IS}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}_{t\ge0}$,
probability vector $p\in\Delta^{d}$.
\For{$t=0,1,2,\dots,T-1$}
    \State Sample $i_{t}\in\{1,\dots,d\}$ with $\Pr(i_{t}=i)=p_{i}$.
    \State Compute partial gradient $g_{i_{t}}^{t}\gets \nabla_{i_{t}}f(w_{t})$.
    \State Update only coordinate $i_{t}$:
    \[
    w_{t+1}\gets w_{t}-\eta_{t}\frac{1}{p_{i_{t}}}e_{i_{t}}g_{i_{t}}^{t}.
    \]
\EndFor
\State \textbf{Output:} $w_{T}$ (or the average $\bar w_{T}= \frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional function $f(w)=(w-3)^{2}$.
Here $d=1$, $p_{1}=1$, $e_{1}=1$, and $\nabla_{1}f(w)=2(w-3)$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})$ & $f(w_{t})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $w_{1}=0-0.1\cdot\frac{1}{1}(-6)=0.6$ &
$2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$\\
2 & $w_{2}=0.6-0.1(-4.8)=1.08$ &
$2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$\\
3 & $w_{3}=1.08-0.1(-3.84)=1.464$ &
$2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3689$
\end{tabular}
\end{center}
The algorithm moves towards the minimiser $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L_{i}$‑smoothness]\label{ass:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_{i}>0$ such that for all $w\in\R^{d}$ and
any scalar $\alpha$,
\[
\bigl| \nabla_{i}f(w+ \alpha e_{i})-\nabla_{i}f(w) \bigr|
\le L_{i}|\alpha|.
\tag{2}
\]
Equivalently,
\[
f(w+ \alpha e_{i})\le f(w)+\alpha\nabla_{i}f(w)+\frac{L_{i}}{2}\alpha^{2}.
\tag{2'}
\]
\end{assumption}

\begin{assumption}[Bounded variance of stochastic partial gradients]\label{ass:var}
The algorithm uses the exact partial gradient, hence
\[
\operatorname{Var}\!\bigl(\nabla_{i_{t}}f(w_{t})\mid w_{t}\bigr)=0 .
\]
(If a stochastic estimator is used, one can assume a uniform bound
$\E[\|\tilde g_{i_{t}}^{t}-\nabla_{i_{t}}f(w_{t})\|^{2}\mid w_{t}]\le\sigma^{2}$,
but this is not needed for the deterministic case.)
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes satisfy
\[
0<\eta_{t}\le \min_{i}\frac{p_{i}}{L_{i}} .
\tag{3}
\]
In particular a constant stepsize
$\eta_{t}\equiv\eta\in\bigl(0,\min_{i}\frac{p_{i}}{L_{i}}\bigr]$ is admissible.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected squared gradient norm]\label{thm:main}
Let $\{w_{t}\}_{t\ge0}$ be generated by \eqref{1} under Assumptions
\ref{ass:smooth}–\ref{ass:stepsize}. Define the weighted norm
\[
\| \nabla f(w)\|_{P^{-1}}^{2}\;:=\;\sum_{i=1}^{d}\frac{1}{p_{i}}\bigl(\nabla_{i}f(w)\bigr)^{2},
\qquad
P:=\operatorname{diag}(p_{1},\dots,p_{d}).
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\,T},
\tag{4}
\]
where $f^{\star}= \inf_{w}f(w)$.
Consequently,
\[
\min_{0\le t<T}\E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr]
=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Condition \eqref{3} guarantees that the quadratic term
in the smoothness inequality \eqref{2'} never dominates the descent term,
hence the iterates remain bounded.
\item \textbf{Importance sampling effect.} Coordinates with larger $L_{i}$
receive larger probability $p_{i}$, reducing the factor $1/p_{i}$ in the bound
\eqref{4}. The optimal choice $p_{i}\propto L_{i}^{1/2}$ minimises the constant
$\max_{i}L_{i}/p_{i}$ and yields the fastest theoretical rate.
\item \textbf{Comparison to uniform CD.} Setting $p_{i}=1/d$ recovers the
classical randomized coordinate descent rate
$O\bigl(d\,\bigl(f(w_{0})-f^{\star}\bigr)/(\eta T)\bigr)$.
\item \textbf{Non‑convexity.} No convexity is required; the guarantee is on the
expected norm of the gradient, a standard stationarity measure for non‑convex
optimization.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite non‑convexity, RCD‑IS drives the
expected weighted gradient norm to zero at a $1/T$ rate. The weighting by
$1/p_{i}$ is essential: it exactly compensates the bias introduced by the
non‑uniform sampling and appears naturally in the algebraic derivation
(see the proof). The result matches the optimal $O(1/T)$ rate for
first‑order methods applied to smooth (possibly non‑convex) functions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step descent under coordinate‑wise smoothness]\label{lem:descent}
For any $w\in\R^{d}$, any $i\in\{1,\dots,d\}$ and any stepsize $\eta$ satisfying
$\eta\le p_{i}/L_{i}$,
\[
f\!\bigl(w-\eta\frac{1}{p_{i}}e_{i}\nabla_{i}f(w)\bigr)
\le f(w)-\frac{\eta}{2p_{i}}\bigl(\nabla_{i}f(w)\bigr)^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply the smoothness inequality \eqref{2'} with $\alpha=-\eta\frac{1}{p_{i}}\nabla_{i}f(w)$:
\[
\begin{aligned}
f\!\bigl(w+\alpha e_{i}\bigr)
&\le f(w)+\alpha\nabla_{i}f(w)+\frac{L_{i}}{2}\alpha^{2}\\
&= f(w)-\eta\frac{1}{p_{i}}\bigl(\nabla_{i}f(w)\bigr)^{2}
       +\frac{L_{i}}{2}\eta^{2}\frac{1}{p_{i}^{2}}\bigl(\nabla_{i}f(w)\bigr)^{2}\\
&= f(w)-\frac{\eta}{p_{i}}\Bigl(1-\frac{L_{i}\eta}{2p_{i}}\Bigr)
       \bigl(\nabla_{i}f(w)\bigr)^{2}.
\end{aligned}
\]
Since $\eta\le p_{i}/L_{i}$, the bracket is at least $1/2$, yielding \eqref{5}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main} by taking expectations of the one‑step
descent Lemma~\ref{lem:descent}.

\noindent\textbf{[Step 1]} Take the conditional expectation w.r.t. the random
coordinate $i_{t}$ given $w_{t}$:
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
&= \sum_{i=1}^{d}p_{i}\,
   f\!\bigl(w_{t}-\eta_{t}\frac{1}{p_{i}}e_{i}\nabla_{i}f(w_{t})\bigr) .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 2]} Apply Lemma~\ref{lem:descent} to each term of the sum.
Because $\eta_{t}\le p_{i}/L_{i}$ for every $i$, we have
\[
f\!\bigl(w_{t}-\eta_{t}\frac{1}{p_{i}}e_{i}\nabla_{i}f(w_{t})\bigr)
\le f(w_{t})-\frac{\eta_{t}}{2p_{i}}\bigl(\nabla_{i}f(w_{t})\bigr)^{2}.
\tag{7}
\]

\noindent\textbf{[Step 3]} Substitute \eqref{7} into \eqref{6}:
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
&\le \sum_{i=1}^{d}p_{i}\Bigl(
      f(w_{t})-\frac{\eta_{t}}{2p_{i}}\bigl(\nabla_{i}f(w_{t})\bigr)^{2}
      \Bigr)\\[2mm]
&= f(w_{t})-\frac{\eta_{t}}{2}\sum_{i=1}^{d}\frac{1}{p_{i}}
      \bigl(\nabla_{i}f(w_{t})\bigr)^{2}\\[2mm]
&= f(w_{t})-\frac{\eta_{t}}{2}\,\|\nabla f(w_{t})\|_{P^{-1}}^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 4]} Take total expectation and telescope over $t=0,\dots,T-1$.
Using the law of total expectation on \eqref{8} gives
\[
\E\!\bigl[f(w_{t+1})\bigr]
\le \E\!\bigl[f(w_{t})\bigr]
      -\frac{\eta_{t}}{2}\E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr].
\tag{9}
\]

\noindent\textbf{[Step 5]} Sum \eqref{9} for $t=0$ to $T-1$:
\[
\E\!\bigl[f(w_{T})\bigr]
\le f(w_{0})
      -\frac{1}{2}\sum_{t=0}^{T-1}\eta_{t}
          \E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr].
\tag{10}
\]

\noindent\textbf{[Step 6]} Since $f(w_{T})\ge f^{\star}$,
\[
\sum_{t=0}^{T-1}\eta_{t}
      \E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr]
\le 2\bigl(f(w_{0})-f^{\star}\bigr).
\tag{11}
\]

\noindent\textbf{[Step 7]} For a constant stepsize $\eta_{t}\equiv\eta$,
divide both sides of \eqref{11} by $\eta T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}
   \E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr]
\le \frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T},
\]
which is exactly the bound \eqref{4}. $\square$

\section*{Rate Derivation (With Constants)}
If the stepsize is chosen as the maximal admissible
$\eta = \min_{i} \frac{p_{i}}{L_{i}}$, then
\[
\frac{1}{T}\sum_{t=0}^{T-1}
   \E\!\bigl[\|\nabla f(w_{t})\|_{P^{-1}}^{2}\bigr]
\le
\frac{2L_{\max}}{T}\,
\frac{f(w_{0})-f^{\star}}{\min_{i}p_{i}},
\qquad
L_{\max}:=\max_{i}L_{i}.
\]
Thus the convergence constant scales linearly with the
condition number $\kappa:=L_{\max}/\min_{i}p_{i}$.
Choosing the importance‑sampling distribution
$p_{i}=L_{i}^{1/2}/\sum_{j}L_{j}^{1/2}$ yields
\[
\min_{i}p_{i}\ge \frac{1}{d}\quad\text{and}\quad
\frac{L_{\max}}{\min_{i}p_{i}}
\le d\,L_{\max},
\]
which improves the uniform case by a factor of $d$ when the $L_{i}$'s are heterogeneous.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform sampling.} $p_{i}=1/d$ for all $i$.
   The bound \eqref{4} reduces to
   \[
   \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
   \le \frac{2d\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T},
   \]
   which coincides with the classical randomised CD rate.
\item \textbf{Single coordinate update ($d=1$).}
   The algorithm becomes ordinary gradient descent with stepsize
   $\eta\le 1/L_{1}$ and the bound simplifies to
   $\frac{2(f(w_{0})-f^{\star})}{\eta T}$.
\item \textbf{Block coordinate descent.}
   If coordinates are grouped into blocks $B_{k}$ with probabilities $p_{k}$,
   the same proof goes through with $L_{k}$ the block‑wise smoothness constant,
   yielding an identical $1/T$ rate for the block‑gradient norm.
\end{enumerate}
