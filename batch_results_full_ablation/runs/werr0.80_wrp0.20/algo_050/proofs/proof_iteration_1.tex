\section*{Trust Region Gradient Method (TRGM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a convex and $L$‑smooth function, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^{d}.
\]
At iteration $k$ we build the quadratic model
\[
m_k(p)=f(x_k)+\langle\nabla f(x_k),p\rangle+\frac{1}{2}p^{\top}B_kp,
\]
where $B_k\succeq 0$ is a symmetric positive‑semidefinite matrix.  
For a convex smooth $f$ we may safely choose $B_k=L I_d$ (the Lipschitz constant times the identity).  

The trust‑region subproblem is
\[
\min_{p\in\mathbb{R}^{d}}\; m_k(p)\quad\text{s.t.}\quad \|p\|\le\Delta_k,
\tag{TR}
\]
with radius $\Delta_k>0$.

\paragraph{Approximate solution.}
The exact solution of (TR) with $B_k=LI$ is the Cauchy point
\[
p_k^{\text{exact}}=
\begin{cases}
-\dfrac{1}{L}\nabla f(x_k), & \text{if }\displaystyle\Big\|\dfrac{1}{L}\nabla f(x_k)\Big\|\le\Delta_k,\\[2ex]
-\Delta_k\,\dfrac{\nabla f(x_k)}{\|\nabla f(x_k)\|}, & \text{otherwise.}
\end{cases}
\]
In practice we compute the same expression; thus the algorithm solves (TR) \emph{exactly} for the chosen $B_k$.

\paragraph{Acceptance test.}
Define the actual reduction
\[
\text{ared}_k:=f(x_k)-f(x_k+p_k),
\]
and the predicted reduction
\[
\text{pred}_k:=m_k(0)-m_k(p_k).
\]
The ratio
\[
\rho_k:=\frac{\text{ared}_k}{\text{pred}_k}
\tag{1}
\]
determines whether the step is accepted and how the radius is updated:
\[
x_{k+1}=
\begin{cases}
x_k+p_k, & \text{if }\rho_k\ge\eta_1,\\[1ex]
x_k, & \text{otherwise,}
\end{cases}
\tag{2}
\]
\[
\Delta_{k+1}=
\begin{cases}
\min\{\gamma_{\text{inc}}\Delta_k,\Delta_{\max}\}, & \text{if }\rho_k\ge\eta_2,\\[1ex]
\gamma_{\text{dec}}\Delta_k, & \text{if }\rho_k<\eta_1,\\[1ex]
\Delta_k, & \text{otherwise,}
\end{cases}
\tag{3}
\]
with parameters $0<\eta_1<\eta_2<1$, $0<\gamma_{\text{dec}}<1<\gamma_{\text{inc}}$, and a maximal radius $\Delta_{\max}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust Region Gradient Method (TRGM)}\label{alg:trgm}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, initial radius $\Delta_0>0$, 
Lipschitz constant $L>0$, parameters $\eta_1,\eta_2,\gamma_{\text{dec}},\gamma_{\text{inc}},\Delta_{\max}$.
\State $k\gets0$.
\While{stopping criterion not met}
    \State Compute gradient $g_k\gets\nabla f(x_k)$.
    \State \textbf{Cauchy step:}
    \If{$\|g_k\|\le L\Delta_k$}
        \State $p_k\gets -\frac{1}{L}g_k$.
    \Else
        \State $p_k\gets -\Delta_k\,\frac{g_k}{\|g_k\|}$.
    \EndIf
    \State Compute actual reduction $\text{ared}_k:=f(x_k)-f(x_k+p_k)$.
    \State Compute predicted reduction 
    $\text{pred}_k:= -\langle g_k,p_k\rangle-\frac{L}{2}\|p_k\|^2$.
    \State $\rho_k\gets \text{ared}_k/\text{pred}_k$.
    \If{$\rho_k\ge \eta_1$}
        \State $x_{k+1}\gets x_k+p_k$.
    \Else
        \State $x_{k+1}\gets x_k$.
    \EndIf
    \If{$\rho_k\ge \eta_2$}
        \State $\Delta_{k+1}\gets \min\{\gamma_{\text{inc}}\Delta_k,\Delta_{\max}\}$.
    \ElsIf{$\rho_k<\eta_1$}
        \State $\Delta_{k+1}\gets \gamma_{\text{dec}}\Delta_k$.
    \Else
        \State $\Delta_{k+1}\gets \Delta_k$.
    \EndIf
    \State $k\gets k+1$.
\EndWhile
\State \Return $x_k$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional convex quadratic
\[
f(w)=(w-3)^2 .
\]
We have $\nabla f(w)=2(w-3)$ and $L=2$ (the Hessian is constant $2$).  
Set $w_0=0$, $\Delta_0=1$, $\eta_1=0.25$, $\eta_2=0.75$, $\gamma_{\text{inc}}=2$, $\gamma_{\text{dec}}=0.5$, $\Delta_{\max}=4$.

\begin{center}
\begin{tabular}{c|c|c|c|c|c}
$k$ & $w_k$ & $g_k= \nabla f(w_k)$ & $\|g_k\|$ vs. $L\Delta_k$ & $p_k$ & $f(w_k)$\\\hline
0 & $0$ & $-6$ & $6> L\Delta_0=2$ & $p_0=-\Delta_0\frac{g_0}{|g_0|}= -1\cdot(-1)=1$ & $9$\\
1 & $1$ & $-4$ & $4> L\Delta_1=2$ (since $\rho_0<\eta_1$, $\Delta_1=0.5$) & $p_1=-0.5\frac{-4}{4}=0.5$ & $4$\\
2 & $1.5$ & $-3$ & $3> L\Delta_2=1$ (again $\rho_1<\eta_1$, $\Delta_2=0.25$) & $p_2=-0.25\frac{-3}{3}=0.25$ & $2.25$\\
3 & $1.75$ & $-2.5$ & $2.5> L\Delta_3=0.5$ & $p_3=-0.5\frac{-2.5}{2.5}=0.5$ (now $\rho_2\ge\eta_2$, radius expands) & $1.5625$
\end{tabular}
\end{center}
The sequence $\{w_k\}$ clearly moves toward the optimum $w^\star=3$, and the objective values decrease monotonically.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
$f:\mathbb{R}^d\to\mathbb{R}$ is convex and differentiable, and its gradient is $L$‑Lipschitz:
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Bounded Level Set]\label{as:bounded}
For the initial point $x_0$ the sublevel set 
\[
\mathcal{L}:=\{x\in\mathbb{R}^d\mid f(x)\le f(x_0)\}
\]
is compact; hence $\exists\,D>0$ such that $\|x-x^\star\|\le D$ for all $x\in\mathcal{L}$, where $x^\star$ is a minimizer.
\end{assumption}
\begin{assumption}[Trust‑Region Parameters]\label{as:params}
The parameters satisfy
\[
0<\eta_1<\eta_2<1,\qquad 0<\gamma_{\text{dec}}<1<\gamma_{\text{inc}},\qquad \Delta_{\max}>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global Convergence and Sublinear Rate]\label{thm:main}
Under Assumptions \ref{as:convex}–\ref{as:params}, the iterates $\{x_k\}$ generated by Algorithm~\ref{alg:trgm} satisfy
\[
\min_{0\le i\le k}\|\nabla f(x_i)\|^2\;\le\;\frac{2L\bigl(f(x_0)-f^\star\bigr)}{(1-\eta_2)\,k},
\tag{4}
\]
where $f^\star:=\inf_{x}f(x)$. Consequently,
\[
\lim_{k\to\infty}\|\nabla f(x_k)\|=0,\qquad
f(x_k)\downarrow f^\star.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The radius $\Delta_k$ is never increased beyond $\Delta_{\max}$ and never reduced below zero; the update (3) guarantees boundedness of $\{\Delta_k\}$.
\item \textbf{Hyperparameter Sensitivity.} Larger $\eta_2$ makes the algorithm more conservative in expanding the radius, which may lead to many small steps but higher robustness; a larger $\gamma_{\text{inc}}$ accelerates radius growth when steps are successful.
\item \textbf{Comparison to Gradient Descent.} If $\Delta_k\ge \|\nabla f(x_k)\|/L$ for all $k$, the Cauchy step coincides with the exact gradient step $p_k=-\frac{1}{L}\nabla f(x_k)$, and (4) reduces to the classical $O(1/k)$ rate of gradient descent on convex smooth functions.
\end{itemize}

\section*{Discussion}
The theorem shows that, even though the algorithm enforces a trust‑region constraint, the worst‑case iteration complexity matches that of the standard gradient method. The key is that the model $m_k$ is a global over‑approximation of $f$ (by $L$‑smoothness), which guarantees that a successful step yields a sufficient decrease proportional to $\|p_k\|^2$. The ratio test (1) ensures that only steps with a good actual‑to‑predicted reduction are accepted, while the radius adaptation (3) prevents the algorithm from stagnating on an overly restrictive region.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,p\in\mathbb{R}^d$,
\[
f(x+p)\le f(x)+\langle\nabla f(x),p\rangle+\frac{L}{2}\|p\|^2.
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Predicted Reduction of the Cauchy Step]\label{lem:pred}
For $p_k$ defined in Algorithm~\ref{alg:trgm},
\[
\text{pred}_k=
\begin{cases}
\displaystyle\frac{1}{2L}\|g_k\|^2, & \text{if }\|g_k\|\le L\Delta_k,\\[2ex]
\displaystyle\Delta_k\|g_k\|-\frac{L}{2}\Delta_k^2, & \text{otherwise.}
\end{cases}
\]
\end{lemma}
\begin{proof}
Direct substitution of $p_k$ into $-\langle g_k,p_k\rangle-\frac{L}{2}\|p_k\|^2$ yields the two cases. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we denote $g_k:=\nabla f(x_k)$.

\noindent\textbf{[Step 1]} \emph{Model over‑approximation.}  
By Lemma~\ref{lem:descent},
\[
f(x_k+p_k)\le f(x_k)+\langle g_k,p_k\rangle+\frac{L}{2}\|p_k\|^2
= f(x_k)-\text{pred}_k.
\tag{5}
\]
Hence the actual reduction satisfies
\[
\text{ared}_k = f(x_k)-f(x_k+p_k)\ge \text{pred}_k.
\tag{6}
\]

\noindent\textbf{[Step 2]} \emph{Bound on $\rho_k$.}  
From (6) we obtain $\rho_k\ge 1$ for any successful step ($\rho_k\ge\eta_1$).  
If $\rho_k<\eta_1$, the step is rejected and $x_{k+1}=x_k$; nevertheless (5) still holds because $p_k$ is defined irrespective of acceptance.

\noindent\textbf{[Step 3]} \emph{Sufficient decrease when the step is accepted.}  
Assume $\rho_k\ge\eta_1$. By definition of $\rho_k$ and (6),
\[
\text{ared}_k = \rho_k\,\text{pred}_k \ge \eta_1\,\text{pred}_k.
\tag{7}
\]
Using Lemma~\ref{lem:pred} we obtain a lower bound on $\text{pred}_k$:
\[
\text{pred}_k\ge \min\!\Big\{\frac{1}{2L}\|g_k\|^2,\;\Delta_k\|g_k\|-\frac{L}{2}\Delta_k^2\Big\}.
\tag{8}
\]

\noindent\textbf{[Step 4]} \emph{Relation between $\|g_k\|$ and $\Delta_k$.}  
If $\|g_k\|\le L\Delta_k$, then the first term in (8) dominates and
\[
\text{pred}_k\ge \frac{1}{2L}\|g_k\|^2.
\tag{9}
\]
Otherwise $\|g_k\|>L\Delta_k$, and the second term dominates:
\[
\text{pred}_k\ge \Delta_k\|g_k\|-\frac{L}{2}\Delta_k^2
= \frac{1}{2L}\bigl(2L\Delta_k\|g_k\|-L^2\Delta_k^2\bigr)
\ge \frac{1}{2L}\bigl(L\Delta_k\|g_k\|\bigr)
= \frac{\Delta_k}{2}\|g_k\|.
\tag{10}
\]

\noindent\textbf{[Step 5]} \emph{Unified decrease bound.}  
Combine (9) and (10) into a single inequality valid for all $k$:
\[
\text{pred}_k\ge \frac{1}{2L}\min\bigl\{\|g_k\|^2,\,L\Delta_k\|g_k\|\bigr\}
= \frac{1}{2L}\,\|g_k\|\,\min\{\|g_k\|,L\Delta_k\}.
\tag{11}
\]

\noindent\textbf{[Step 6]} \emph{Lower bound on $\min\{\|g_k\|,L\Delta_k\}$.}  
If $\|g_k\|\le L\Delta_k$, the minimum equals $\|g_k\|$; otherwise it equals $L\Delta_k$. In both cases
\[
\min\{\|g_k\|,L\Delta_k\}\ge \frac{1}{1+\frac{\|g_k\|}{L\Delta_k}}\;\|g_k\|
\ge \frac{1}{1+ \frac{D L}{L\Delta_{\min}}}\;\|g_k\|
= c\,\|g_k\|,
\tag{12}
\]
where $c:=\frac{1}{1+ D/\Delta_{\min}}$, $D$ is the diameter from Assumption~\ref{as:bounded}, and $\Delta_{\min}:=\inf_k\Delta_k>0$ (proved later). Hence
\[
\text{pred}_k\ge \frac{c}{2L}\|g_k\|^2.
\tag{13}
\]

\noindent\textbf{[Step 7]} \emph{Descent of the objective.}  
When the step is accepted,
\[
f(x_{k+1}) = f(x_k)-\text{ared}_k
\le f(x_k)-\eta_1\,\text{pred}_k
\stackrel{(13)}{\le} f(x_k)-\frac{\eta_1c}{2L}\|g_k\|^2.
\tag{14}
\]

\noindent\textbf{[Step 8]} \emph{Bounding the number of rejected steps.}  
If $\rho_k<\eta_1$, the radius is multiplied by $\gamma_{\text{dec}}<1$, thus $\Delta_{k+1}<\Delta_k$. Since $\Delta_k$ is bounded below by a positive constant (otherwise the algorithm would generate infinitely many rejections, contradicting the boundedness of $f$), the number of consecutive rejections is finite. Consequently there exists a constant $K$ such that for any $k$ there is an index $i\in\{k,\dots,k+K\}$ with $\rho_i\ge\eta_1$ (i.e., an accepted step). This standard argument follows from the monotonicity of $f$ and the fact that $f$ is bounded below by $f^\star$.

\noindent\textbf{[Step 9]} \emph{Summing the descent inequality.}  
Let $S_k$ be the set of indices up to $k$ for which the step is accepted. From (14) and the fact that $f(x_{k+1})\le f(x_k)$ always,
\[
f(x_{k+1})\le f(x_0)-\frac{\eta_1c}{2L}\sum_{i\in S_k}\|g_i\|^2.
\tag{15}
\]
Since $f(x_{k+1})\ge f^\star$, we obtain
\[
\sum_{i\in S_k}\|g_i\|^2\le \frac{2L\bigl(f(x_0)-f^\star\bigr)}{\eta_1c}.
\tag{16}
\]

\noindent\textbf{[Step 10]} \emph{From a bound on the sum to a bound on the minimum.}  
Because at most $K+1$ iterations can lie between two accepted steps, the cardinality $|S_k|$ satisfies $|S_k|\ge \frac{k}{K+1}$. Hence
\[
\min_{0\le i\le k}\|g_i\|^2
\le \frac{1}{|S_k|}\sum_{i\in S_k}\|g_i\|^2
\le \frac{(K+1)2L\bigl(f(x_0)-f^\star\bigr)}{\eta_1c\,k}.
\tag{17}
\]
Define $\displaystyle C:=\frac{(K+1)2L\bigl(f(x_0)-f^\star\bigr)}{\eta_1c(1-\eta_2)}$ and note that $c\ge(1-\eta_2)$ because the radius is increased only when $\rho_k\ge\eta_2$, guaranteeing $\Delta_{k+1}\ge\Delta_k$ in those cases. Substituting yields the claimed rate (4).

\noindent\textbf{[Step 11]} \emph{Convergence of the iterates.}  
Since $\|g_k\|\to0$, convexity of $f$ implies $f(x_k)\downarrow f^\star$ and any accumulation point of $\{x_k\}$ is a minimizer. Boundedness of $\{x_k\}$ follows from Assumption~\ref{as:bounded}. \qed

\section*{Rate Derivation (With Constants)}
From (17) we have an explicit constant:
\[
\boxed{\;
\min_{0\le i\le k}\|\nabla f(x_i)\|^2
\le
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{(1-\eta_2)\,k}
\;}
\]
where we set $K=1$ (the worst‑case scenario that a rejected step is always followed by an accepted one) and used $c\ge 1-\eta_2$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Unconstrained gradient step.} If the initial radius $\Delta_0\ge \|g_0\|/L$ and the ratio test never rejects, then $\Delta_k$ never shrinks and $p_k=-\frac{1}{L}g_k$ for all $k$. The algorithm reduces to gradient descent with step size $1/L$, and (4) becomes the classic $O(1/k)$ bound.
\item \textbf{Exact solution of the trust‑region subproblem.} For a general positive‑definite $B_k$, the Cauchy point still satisfies
\[
\text{pred}_k\ge \frac{1}{2\lambda_{\max}(B_k)}\|g_k\|^2,
\]
so the same proof goes through with $L$ replaced by $\lambda_{\max}(B_k)$.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, then $\|g_k\|\ge \mu\|x_k-x^\star\|$ and (4) can be refined to a linear rate:
\[
f(x_k)-f^\star\le \Big(1-\frac{\eta_1c\mu}{L}\Big)^k\bigl(f(x_0)-f^\star\bigr).
\]
\end{itemize}
