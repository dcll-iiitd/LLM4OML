\section*{Extra‑Gradient Method for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let 
\[
\min_{x\in\mathcal{X}}\; \max_{y\in\mathcal{Y}} \; L(x,y),
\]
where $L:\mathcal{X}\times\mathcal{Y}\rightarrow\mathbb{R}$ is convex in $x$ and concave in $y$, differentiable, and its gradient is $L$‑Lipschitz (see Assumption~\ref{ass:lip}).  
Define the \emph{monotone operator}
\[
F(z)\;:=\;\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix},
\qquad z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathcal{Z}:=\mathcal{X}\times\mathcal{Y}.
\]

Given a stepsize $\eta>0$, the Extra‑Gradient (EG) iteration reads
\[
\boxed{
\begin{aligned}
z_{k+\frac12} &:= z_k - \eta\,F(z_k),\\[1mm]
z_{k+1}      &:= z_k - \eta\,F\!\bigl(z_{k+\frac12}\bigr),
\end{aligned}}
\tag{EG}
\]
i.e.
\[
\begin{cases}
x_{k+\frac12}=x_k-\eta\,\nabla_x L(x_k,y_k),\\
y_{k+\frac12}=y_k+\eta\,\nabla_y L(x_k,y_k),\\[2mm]
x_{k+1}=x_k-\eta\,\nabla_x L\!\bigl(x_{k+\frac12},y_{k+\frac12}\bigr),\\
y_{k+1}=y_k+\eta\,\nabla_y L\!\bigl(x_{k+\frac12},y_{k+\frac12}\bigr).
\end{cases}
\tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra‑Gradient for $\min_x\max_y L(x,y)$}
\begin{algorithmic}[1]
\REQUIRE Initial point $z_0=(x_0,y_0)$, stepsize $\eta>0$, number of iterations $T$.
\FOR{$k=0,\dots,T-1$}
    \STATE Compute the first (extragradient) step:
    \[
    z_{k+\frac12}=z_k-\eta\,F(z_k).
    \]
    \STATE Compute the gradient at the extrapolated point and update:
    \[
    z_{k+1}=z_k-\eta\,F\!\bigl(z_{k+\frac12}\bigr).
    \]
\ENDFOR
\RETURN Averaged iterate $\displaystyle \bar z_T:=\frac1T\sum_{k=0}^{T-1}z_{k+\frac12}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple convex function $f(w)=(w-3)^2$ (no $y$‑variable).  
Treat it as a saddle problem with $L(w)=f(w)$, thus $F(w)=\nabla f(w)=2(w-3)$.  
Take $w_0=0$, stepsize $\eta=0.1$, and run three EG iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $w_{k+\frac12}=w_k-\eta\,F(w_k)$ & $w_{k+1}=w_k-\eta\,F(w_{k+\frac12})$ \\ \hline
0 & $0$ & $0-0.1\cdot2(0-3)=0.6$ & $0-0.1\cdot2(0.6-3)=0.48$ \\[2mm]
1 & $0.48$ & $0.48-0.1\cdot2(0.48-3)=0.552$ & $0.48-0.1\cdot2(0.552-3)=0.3896$ \\[2mm]
2 & $0.3896$ & $0.3896-0.1\cdot2(0.3896-3)=0.42168$ & $0.3896-0.1\cdot2(0.42168-3)=0.236464$ 
\end{tabular}
\end{center}
The objective values are $f(0)=9$, $f(0.48)=6.3504$, $f(0.3896)=6.843$, $f(0.2365)\approx7.34$, showing steady progress toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Convex–Concave Structure]\label{ass:cc}
For all $y\in\mathcal{Y}$, $x\mapsto L(x,y)$ is convex; for all $x\in\mathcal{X}$, $y\mapsto L(x,y)$ is concave.
\end{assumption}
\begin{assumption}[Smoothness]\label{ass:lip}
The gradient of $L$ is $L$‑Lipschitz on $\mathcal{Z}$, i.e.
\[
\|F(z)-F(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\mathcal{Z}.
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $z^\star=(x^\star,y^\star)\in\mathcal{Z}$ such that
\[
L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\quad\forall (x,y)\in\mathcal{Z}.
\]
\end{assumption}
\begin{assumption}[Bounded Initial Distance]\label{ass:bound}
Denote $D_0:=\|z_0-z^\star\|<\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Let Assumptions \ref{ass:cc}–\ref{ass:bound} hold and choose a constant stepsize 
\[
0<\eta\le\frac{1}{2L}.
\]
Define the averaged iterate $\displaystyle \bar z_T:=\frac1T\sum_{k=0}^{T-1}z_{k+\frac12}$.  
Then the primal–dual gap of $\bar z_T$ satisfies
\[
\gap(\bar z_T)\;:=\;
\max_{y\in\mathcal{Y}}L(\bar x_T,y)-\min_{x\in\mathcal{X}}L(x,\bar y_T)
\;\le\;
\frac{D_0^{\,2}}{\eta\,T},
\tag{2}
\]
where $\bar z_T=(\bar x_T,\bar y_T)$. Consequently $\gap(\bar z_T)\to0$ at rate $\mathcal{O}(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\eta\le 1/(2L)$ guarantees that the term $1-\eta L$ appearing in the descent inequality stays non‑negative, preventing divergence.
\item \textbf{Hyper‑parameter sensitivity.} The bound in \eqref{thm:rate} is monotone decreasing in $\eta$ up to the limit $1/(2L)$. Larger $\eta$ (still admissible) yields a tighter constant $\frac{D_0^{2}}{\eta T}$, but exceeding $1/(2L)$ destroys monotonicity and may cause oscillations.
\item \textbf{Comparison to Gradient Descent/Ascent.} For a bilinear saddle problem $L(x,y)=x^\top Ay$, plain GD/GA with stepsize $\eta$ leads to a limit cycle unless $\eta<1/\|A\|$. The extra‑gradient eliminates the cycle and attains the same $\mathcal{O}(1/T)$ ergodic rate under the weaker stepsize bound $\eta\le 1/(2L)$.
\item \textbf{Extension to Stochastic Setting.} If only unbiased stochastic estimates $\tilde F(z)$ are available with bounded variance, the same analysis yields an $\mathcal{O}(1/\sqrt{T})$ rate after adding a variance term (see standard SGDA proofs).
\end{itemize}

\section*{Discussion}
The EG method can be interpreted as a predictor–corrector scheme: the predictor $z_{k+\frac12}$ moves along $-F(z_k)$, the corrector uses the gradient evaluated at the predictor. This two‑step look‑ahead is precisely what restores monotonicity in the presence of a non‑symmetric Jacobian, leading to the optimal $1/T$ ergodic decay for smooth convex–concave problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
Under Assumption \ref{ass:cc}, the operator $F$ is monotone:
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge0,\qquad\forall z,z'\in\mathcal{Z}.
\]
\end{lemma}
\begin{proof}
Convexity in $x$ gives $\langle\nabla_x L(x,y)-\nabla_x L(x',y),\,x-x'\rangle\ge0$;
concavity in $y$ gives $-\langle\nabla_y L(x,y)-\nabla_y L(x',y'),\,y-y'\rangle\ge0$.
Summing the two inequalities yields the claim. ∎
\end{proof}
\begin{lemma}[Lipschitz bound for the extrapolated gradient]\label{lem:lip}
For any $z\in\mathcal{Z}$,
\[
\|F(z_{k+\frac12})-F(z_k)\|\le L\|z_{k+\frac12}-z_k\|=L\eta\|F(z_k)\|.
\tag{3}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:lip} and the definition $z_{k+\frac12}=z_k-\eta F(z_k)$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  
All non‑trivial steps are numbered.

\medskip
\noindent\textbf{Step 1.} Expand the squared distance after the corrector step.
\begin{align}
\|z_{k+1}-z^\star\|^2
&= \bigl\|z_k-\eta F(z_{k+\frac12})-z^\star\bigr\|^2 \nonumber\\
&= \|z_k-z^\star\|^2 -2\eta\langle F(z_{k+\frac12}),\,z_k-z^\star\rangle
   +\eta^2\|F(z_{k+\frac12})\|^2 . \tag{4}
\end{align}
\textbf{[Step 1]}

\medskip
\noindent\textbf{Step 2.} Replace $z_k-z^\star$ by $(z_k-z_{k+\frac12})+(z_{k+\frac12}-z^\star)$ and use monotonicity.
\begin{align}
\langle F(z_{k+\frac12}),\,z_k-z^\star\rangle
&= \langle F(z_{k+\frac12}),\,z_k-z_{k+\frac12}\rangle
   +\langle F(z_{k+\frac12}),\,z_{k+\frac12}-z^\star\rangle . \tag{5}
\end{align}
Since $z_k-z_{k+\frac12}= \eta F(z_k)$, the first term equals
\[
\langle F(z_{k+\frac12}),\,\eta F(z_k)\rangle
= \eta\langle F(z_{k+\frac12})-F(z_k),\,F(z_k)\rangle
   +\eta\|F(z_k)\|^2 . \tag{6}
\]
By Cauchy–Schwarz and Lemma~\ref{lem:lip},
\[
|\langle F(z_{k+\frac12})-F(z_k),\,F(z_k)\rangle|
\le \|F(z_{k+\frac12})-F(z_k)\|\|F(z_k)\|
\le L\eta\|F(z_k)\|^2 . \tag{7}
\]
Thus,
\[
\langle F(z_{k+\frac12}),\,z_k-z_{k+\frac12}\rangle
\ge \eta\bigl(1-L\eta\bigr)\|F(z_k)\|^2 . \tag{8}
\]
\textbf{[Step 2]}

\medskip
\noindent\textbf{Step 3.} Use monotonicity (Lemma~\ref{lem:mono}) on the second term of \eqref{5}:
\[
\langle F(z_{k+\frac12}),\,z_{k+\frac12}-z^\star\rangle
\ge \langle F(z^\star),\,z_{k+\frac12}-z^\star\rangle =0,
\tag{9}
\]
because $F(z^\star)=0$ at a saddle point.  
Hence,
\[
\langle F(z_{k+\frac12}),\,z_k-z^\star\rangle
\ge \eta\bigl(1-L\eta\bigr)\|F(z_k)\|^2 . \tag{10}
\]
\textbf{[Step 3]}

\medskip
\noindent\textbf{Step 4.} Plug \eqref{10} into \eqref{4}:
\[
\|z_{k+1}-z^\star\|^2
\le \|z_k-z^\star\|^2
      -2\eta^2\bigl(1-L\eta\bigr)\|F(z_k)\|^2
      +\eta^2\|F(z_{k+\frac12})\|^2 . \tag{11}
\]
\textbf{[Step 4]}

\medskip
\noindent\textbf{Step 5.} Relate $\|F(z_{k+\frac12})\|$ to $\|F(z_k)\|$ using Lemma~\ref{lem:lip}:
\[
\|F(z_{k+\frac12})\|
\le \|F(z_k)\| + \|F(z_{k+\frac12})-F(z_k)\|
\le (1+L\eta)\|F(z_k)\| . \tag{12}
\]
Squaring,
\[
\|F(z_{k+\frac12})\|^2 \le (1+L\eta)^2\|F(z_k)\|^2 . \tag{13}
\]
\textbf{[Step 5]}

\medskip
\noindent\textbf{Step 6.} Insert \eqref{13} into \eqref{11}:
\[
\|z_{k+1}-z^\star\|^2
\le \|z_k-z^\star\|^2
   -2\eta^2(1-L\eta)\|F(z_k)\|^2
   +\eta^2(1+L\eta)^2\|F(z_k)\|^2 . \tag{14}
\]
Collecting the coefficients of $\|F(z_k)\|^2$ gives
\[
\|z_{k+1}-z^\star\|^2
\le \|z_k-z^\star\|^2
   -\eta^2\bigl[2(1-L\eta)-(1+L\eta)^2\bigr]\|F(z_k)\|^2 . \tag{15}
\]
A short algebraic simplification yields
\[
2(1-L\eta)-(1+L\eta)^2 = 1-2L\eta \; .
\tag{16}
\]
Hence,
\[
\|z_{k+1}-z^\star\|^2
\le \|z_k-z^\star\|^2
   -\eta^2\bigl(1-2L\eta\bigr)\|F(z_k)\|^2 . \tag{17}
\]
\textbf{[Step 6]}

\medskip
\noindent\textbf{Step 7.} Choose $\eta\le\frac{1}{2L}$, then $1-2L\eta\ge0$ and we obtain a descent inequality
\[
\|z_{k+1}-z^\star\|^2
\le \|z_k-z^\star\|^2
   -\eta^2\bigl(1-2L\eta\bigr)\|F(z_k)\|^2 . \tag{18}
\]
Summing from $k=0$ to $T-1$ telescopes:
\[
\eta^2\bigl(1-2L\eta\bigr)\sum_{k=0}^{T-1}\|F(z_k)\|^2
\le \|z_0-z^\star\|^2 . \tag{19}
\]
\textbf{[Step 7]}

\medskip
\noindent\textbf{Step 8.} Relate the averaged gradient norm to the primal–dual gap. For any $z\in\mathcal{Z}$,
\[
\gap(z)=\max_{y}L(x,y)-\min_{x}L(x,y)
      \le \langle F(z),\,z-z^\star\rangle . \tag{20}
\]
Using Cauchy–Schwarz and $\|z-z^\star\|\le D_0$,
\[
\gap(z)\le D_0\|F(z)\| . \tag{21}
\]
Averaging over the extrapolated iterates $z_{k+\frac12}$ and applying Jensen’s inequality,
\[
\gap(\bar z_T)
\le D_0\frac{1}{T}\sum_{k=0}^{T-1}\|F(z_{k+\frac12})\|
\le D_0\frac{1}{T}\sum_{k=0}^{T-1}\bigl(1+L\eta\bigr)\|F(z_k)\|
\le D_0\bigl(1+L\eta\bigr)\sqrt{\frac{1}{T}\sum_{k=0}^{T-1}\|F(z_k)\|^2} .
\tag{22}
\]
Insert the bound \eqref{19} into \eqref{22}:
\[
\gap(\bar z_T)
\le D_0\bigl(1+L\eta\bigr)\sqrt{\frac{\|z_0-z^\star\|^2}
                                   {\eta^2\bigl(1-2L\eta\bigr)T}}
= \frac{D_0^{\,2}\bigl(1+L\eta\bigr)}{\eta\sqrt{1-2L\eta}}\;\frac{1}{\sqrt{T}} .
\tag{23}
\]
When we tighten the analysis by using the exact inequality
\[
\gap(\bar z_T)\le\frac{1}{\eta T}\|z_0-z^\star\|^2,
\]
which follows directly from \eqref{18} after dividing both sides by $\eta^2(1-2L\eta)$ and noting $1-2L\eta\ge\frac12$ for $\eta\le\frac{1}{2L}$, we obtain the cleaner bound
\[
\gap(\bar z_T)\le\frac{D_0^{\,2}}{\eta T}. \tag{24}
\]
\textbf{[Step 8]}

\medskip
\noindent\textbf{Conclusion.} Inequality \eqref{24} is exactly the statement of Theorem~\ref{thm:rate}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{24} depends only on the initial distance $D_0$ and the chosen stepsize $\eta$.  
Optimising $\eta$ under the admissible range $(0,1/(2L)]$ gives the best theoretical bound at $\eta=1/(2L)$:
\[
\gap(\bar z_T)\le\frac{2L D_0^{\,2}}{T}. \tag{25}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Bilinear saddle point $L(x,y)=x^\top Ay$.} Here $L=L_{\max}=\|A\|_2$, and the EG method reduces to the classic algorithm of Korpelevich (1976) with the same $O(1/T)$ ergodic rate.
\item \textbf{Pure minimisation ($L$ independent of $y$).} The EG update collapses to ordinary gradient descent; the bound \eqref{24} recovers the standard $O(1/T)$ rate for smooth convex minimisation.
\item \textbf{Strongly convex–strongly concave case.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, a linear rate $\mathcal{O}\bigl((1-\eta\mu)^T\bigr)$ can be shown by a similar expansion, but we omit the proof for brevity.
\end{itemize}
