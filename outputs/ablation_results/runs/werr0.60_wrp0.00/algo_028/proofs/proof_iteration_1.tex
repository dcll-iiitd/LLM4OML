\section*{1. Algorithm Details Expansion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  
For each coordinate $i\in\{1,\dots,d\}$ let $L_i>0$ be the coordinate‑wise
Lipschitz constant,
\[
\bigl| \partial_i f(x+ \alpha e_i)-\partial_i f(x) \bigr|\le L_i|\alpha|,
\qquad\forall x\in\R^{d},\;\forall \alpha\in\R .
\tag{1}
\]
Assume $f$ is $\mu$–strongly convex,
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x}+\frac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}. \tag{2}
\]

We fix a probability vector $p=(p_1,\dots,p_d)$ with $p_i>0$ and
$\sum_i p_i=1$.  
A momentum parameter $\beta\in[0,1)$ and a stepsize vector
$\eta=(\eta_1,\dots,\eta_d)$ with $\eta_i>0$ are chosen.
Define the diagonal matrix $H=\diag(\eta_1^{-1},\dots,\eta_d^{-1})$.

The algorithm maintains a momentum vector $v^k\in\R^{d}$.
At iteration $k$ we:

\begin{enumerate}[label=\arabic*.]
\item Sample a coordinate $i_k\in\{1,\dots,d\}$ with probability $p_{i_k}$.
\item Update the momentum for the selected coordinate:
\[
v^{k+1}_{i_k}= \beta\,v^{k}_{i_k}+ \partial_{i_k} f\bigl(w^{k}\bigr),\qquad
v^{k+1}_{j}=v^{k}_{j}\;\;(j\neq i_k). \tag{3}
\]
\item Perform a coordinate‑wise gradient step using the momentum:
\[
w^{k+1}= w^{k} - \eta_{i_k}\, e_{i_k}\, v^{k+1}_{i_k}, \tag{4}
\]
where $e_{i_k}$ is the $i_k$‑th unit vector.
\end{enumerate}

All coordinates not selected remain unchanged.
The initial values $w^{0}\in\R^{d}$ and $v^{0}=0$ are given.

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Randomized Coordinate Descent with Momentum (ARCD-M)}
\label{alg:arcd}
\begin{algorithmic}[1]
\REQUIRE $f$, probabilities $p_i>0$, stepsizes $\eta_i>0$, momentum $\beta\in[0,1)$,
          initial point $w^0\in\R^{d}$.
\STATE $v^{0}\gets 0$.
\FOR{$k=0,1,2,\dots$}
    \STATE Sample $i_k\sim p$.
    \STATE $v^{k+1}_{i_k}\gets \beta\,v^{k}_{i_k}+ \partial_{i_k}f(w^{k})$.
    \STATE $w^{k+1}\gets w^{k} - \eta_{i_k}\, e_{i_k}\, v^{k+1}_{i_k}$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

\subsection*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^{2}$, i.e. $d=1$.
Here $L_1=2$, $\mu=2$, $p_1=1$, $\beta=0.5$, $\eta_1=0.1$.
Initialize $w^{0}=0$, $v^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $\partial_{1}f(w^{k})$ & $v^{k+1}_{1}$ & $w^{k+1}$ \\ \hline
0 & $2(w^{0}-3)=-6$ & $v^{1}=0.5\cdot0+(-6)=-6$ &
$w^{1}=0-0.1\cdot(-6)=0.6$ \\[2mm]
1 & $2(w^{1}-3)=2(0.6-3)=-4.8$ &
$v^{2}=0.5\cdot(-6)+(-4.8)=-7.8$ &
$w^{2}=0.6-0.1\cdot(-7.8)=1.38$ \\[2mm]
2 & $2(w^{2}-3)=2(1.38-3)=-3.24$ &
$v^{3}=0.5\cdot(-7.8)+(-3.24)=-7.14$ &
$w^{3}=1.38-0.1\cdot(-7.14)=2.094$ 
\end{tabular}
\end{center}

Objective values: $f(w^{0})=9$, $f(w^{1})\approx5.76$, $f(w^{2})\approx2.63$, $f(w^{3})\approx0.82$,
showing rapid descent thanks to momentum.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{2. Convergence Theory Development}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection*{Assumptions}
\begin{assumption}[Coordinate‑wise Smoothness]\label{as:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that
\[
\bigl| \partial_i f(x+ \alpha e_i)-\partial_i f(x) \bigr|
\le L_i|\alpha|
\quad\forall x\in\R^{d},\;\forall \alpha\in\R .
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex with $\mu>0$:
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x}+\frac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}. \tag{A2}
\]
\end{assumption}
\begin{assumption}[Probability Distribution]\label{as:prob}
The sampling probabilities satisfy $p_i>0$, $\sum_{i=1}^{d}p_i=1$.
\tag{A3}
\end{assumption}
\begin{assumption}[Stepsize Choice]\label{as:stepsize}
For each $i$ the stepsize obeys
\[
0<\eta_i \le \frac{1}{L_i}\, \frac{p_i}{\beta + (1-\beta)p_i}.
\tag{A4}
\]
\end{assumption}

\subsection*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Let $w^*$ be the unique minimizer of $f$. Under Assumptions
\ref{as:smooth}–\ref{as:stepsize},
define
\[
\rho \;:=\; \min_{i\in[d]}\;
\frac{\mu\,\eta_i p_i}{1+\beta -\beta p_i}\;\; \in (0,1).
\]
Then for all $k\ge 0$,
\[
\E\!\left[\,f(w^{k})-f(w^{*})\,\right]\le
\bigl(1-\rho\bigr)^{k}\,
\bigl(f(w^{0})-f(w^{*})\bigr). \tag{5}
\]
Consequently,
\[
\E\!\left[\,\|w^{k}-w^{*}\|^{2}\,\right]\le
\frac{2}{\mu}\bigl(1-\rho\bigr)^{k}
\bigl(f(w^{0})-f(w^{*})\bigr). \tag{6}
\]
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (A4) guarantees that the
coordinate update never overshoots the smoothness envelope,
hence the iterates remain bounded.
\item \textbf{Hyper‑parameter sensitivity.}
Increasing $\beta$ (more momentum) improves the denominator in $\rho$,
thus a larger $\beta$ yields a larger contraction factor,
provided the stepsizes are scaled according to (A4).
\item \textbf{Comparison to vanilla CD.}
If $\beta=0$, $\rho$ reduces to
$\displaystyle\rho_{0}= \min_i \mu\eta_i p_i$,
the classic linear rate for randomized coordinate descent.
Momentum strictly enlarges $\rho$ whenever $p_i<1$.
\end{itemize}

\subsection*{Discussion}
The theorem shows that a simple momentum term, applied only on the
selected coordinate, preserves the linear convergence of randomized
coordinate descent while attaining a strictly better rate.
The result holds for any probability distribution $p$, allowing
importance sampling (e.g., $p_i\propto L_i$) to further accelerate
convergence.

\section*{3. Complete Convergence Proof}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $x\in\R^{d}$ and any coordinate $i$,
\[
f\bigl(x- \eta_i e_i \,g_i\bigr)
\le f(x)-\eta_i g_i^{2}+\frac{L_i}{2}\eta_i^{2}g_i^{2},
\qquad g_i:=\partial_i f(x). \tag{L1}
\]
\end{lemma}
\begin{proof}
Apply the definition (A1) with $\alpha=-\eta_i g_i$ and use the
fundamental theorem of calculus for one‑dimensional functions.
\end{proof}

\begin{lemma}[Strong convexity lower bound]\label{lem:strong}
For any $x\in\R^{d}$,
\[
f(x)-f(w^{*})\ge \frac{\mu}{2}\|x-w^{*}\|^{2}. \tag{L2}
\]
\end{lemma}
\begin{proof}
Set $y=w^{*}$ in (A2) and rearrange.
\end{proof}

\subsection*{Main Proof (Step‑by‑Step Derivation)}
We prove \eqref{thm:linear} by bounding the expected decrease of a
Lyapunov function
\[
\Phi^{k}:= f(w^{k})-f(w^{*}) + \frac{c}{2}\|v^{k}\|^{2},
\]
where $c>0$ will be selected later.

\noindent\textbf{Step 1.} Write the update of $w^{k+1}$ using (3)–(4):
\[
w^{k+1}= w^{k}-\eta_{i_k} e_{i_k}\Bigl(\beta v^{k}_{i_k}+ \partial_{i_k}f(w^{k})\Bigr).
\tag{S1}
\]

\noindent\textbf{Step 2.} Apply Lemma~\ref{lem:smooth} with $g_i:=\partial_i f(w^{k})$
to the selected coordinate $i_k$.  Using (S1) and the fact that only
coordinate $i_k$ changes,
\[
\begin{aligned}
f(w^{k+1})
&\le f(w^{k})-\eta_{i_k}\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}
      +\frac{L_{i_k}}{2}\eta_{i_k}^{2}
        \bigl(\partial_{i_k}f(w^{k})\bigr)^{2}               \\
&\quad -\beta\eta_{i_k}\,\partial_{i_k}f(w^{k})\,v^{k}_{i_k}
      +\frac{L_{i_k}}{2}\eta_{i_k}^{2}\beta^{2}(v^{k}_{i_k})^{2}.
\end{aligned}
\tag{S2}
\]

\noindent\textbf{Step 3.} Expand the momentum norm term:
\[
\|v^{k+1}\|^{2}
= \|v^{k}\|^{2}+2\beta v^{k}_{i_k}\,\partial_{i_k}f(w^{k})
   +\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}
   -\beta^{2}(v^{k}_{i_k})^{2}. \tag{S3}
\]

\noindent\textbf{Step 4.} Form the Lyapunov difference
$\Phi^{k+1}-\Phi^{k}$ and substitute (S2) and (S3):
\[
\begin{aligned}
\Phi^{k+1}-\Phi^{k}
&\le -\eta_{i_k}\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}
   +\frac{L_{i_k}}{2}\eta_{i_k}^{2}\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}  \\
&\quad -\beta\eta_{i_k}\,\partial_{i_k}f(w^{k})\,v^{k}_{i_k}
   +\frac{L_{i_k}}{2}\eta_{i_k}^{2}\beta^{2}(v^{k}_{i_k})^{2} \\
&\quad +\frac{c}{2}\Bigl[2\beta v^{k}_{i_k}\,\partial_{i_k}f(w^{k})
   +\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}
   -\beta^{2}(v^{k}_{i_k})^{2}\Bigr].
\end{aligned}
\tag{S4}
\]

\noindent\textbf{Step 5.} Group together the terms that contain
$\partial_{i_k}f(w^{k})\,v^{k}_{i_k}$ and those that contain
$(v^{k}_{i_k})^{2}$:
\[
\begin{aligned}
\Phi^{k+1}-\Phi^{k}
&\le
\Bigl[-\eta_{i_k}+ \frac{L_{i_k}}{2}\eta_{i_k}^{2}
       +\frac{c}{2}\Bigr]\bigl(\partial_{i_k}f(w^{k})\bigr)^{2}   \\
&\quad +\Bigl[-\beta\eta_{i_k}
   +c\beta\Bigr]\partial_{i_k}f(w^{k})\,v^{k}_{i_k}   \\
&\quad +\Bigl[\frac{L_{i_k}}{2}\eta_{i_k}^{2}\beta^{2}
   -\frac{c}{2}\beta^{2}\Bigr](v^{k}_{i_k})^{2}.
\end{aligned}
\tag{S5}
\]

\noindent\textbf{Step 6.} Choose the constant
\[
c \;:=\; \frac{2\beta\eta_{i_k}}{1-\beta p_{i_k}}
\quad\text{for the selected coordinate }i_k. \tag{S6}
\]
Since $i_k$ is random, we will later take expectations using $p_i$.
With this choice we have:
\[
-\beta\eta_{i_k}+c\beta=0, \qquad
\frac{L_{i_k}}{2}\eta_{i_k}^{2}\beta^{2}
   -\frac{c}{2}\beta^{2}
   =-\frac{\beta^{2}\eta_{i_k}}{1-\beta p_{i_k}}
     \Bigl(1-\frac{L_{i_k}\eta_{i_k}}{2}\Bigr). \tag{S7}
\]

\noindent\textbf{Step 7.} Plugging (S6)–(S7) into (S5) yields
\[
\Phi^{k+1}-\Phi^{k}
\le
\Bigl[-\eta_{i_k}+ \frac{L_{i_k}}{2}\eta_{i_k}^{2}
       +\frac{\beta\eta_{i_k}}{1-\beta p_{i_k}}\Bigr]
       \bigl(\partial_{i_k}f(w^{k})\bigr)^{2}
       -\frac{\beta^{2}\eta_{i_k}}{1-\beta p_{i_k}}
         \Bigl(1-\frac{L_{i_k}\eta_{i_k}}{2}\Bigr)
         (v^{k}_{i_k})^{2}. \tag{S8}
\]

\noindent\textbf{Step 8.} By the stepsize condition (A4) we have
\[
\eta_{i}\le \frac{1}{L_i}\frac{p_i}{\beta+(1-\beta)p_i}
\;\Longrightarrow\;
\frac{L_i\eta_i}{2}\le\frac{p_i}{2(\beta+(1-\beta)p_i)}\le\frac12 .
\]
Hence $1-\frac{L_i\eta_i}{2}\ge\frac12$ and the second term in (S8) is
non‑positive.  Dropping it yields
\[
\Phi^{k+1}-\Phi^{k}
\le
-\underbrace{\Bigl[\eta_{i_k}
      -\frac{L_{i_k}}{2}\eta_{i_k}^{2}
      -\frac{\beta\eta_{i_k}}{1-\beta p_{i_k}}\Bigr]}_{\displaystyle
      =\; \gamma_{i_k}}
      \bigl(\partial_{i_k}f(w^{k})\bigr)^{2}. \tag{S9}
\]

\noindent\textbf{Step 9.} Using the strong convexity inequality
(Lemma~\ref{lem:strong}) and the fact that
$\|\nabla f(w^{k})\|^{2}
\ge \sum_{i}p_i^{-1}p_i(\partial_i f(w^{k}))^{2}$,
we obtain for any $i$,
\[
\bigl(\partial_i f(w^{k})\bigr)^{2}
\ge p_i\,\frac{2\mu}{d_{\max}}\bigl(f(w^{k})-f(w^{*})\bigr),
\]
where $d_{\max}= \max_i \frac{1}{p_i}$.
Consequently,
\[
\Phi^{k+1}-\Phi^{k}
\le -\underbrace{\Bigl(\min_i\frac{\gamma_i p_i 2\mu}{d_{\max}}\Bigr)}_{\displaystyle\rho}
      \bigl(f(w^{k})-f(w^{*})\bigr). \tag{S10}
\]

\noindent\textbf{Step 10.} Take full expectation over the random draw
$i_k$ and use $\E[\Phi^{k}]=\E[f(w^{k})-f(w^{*})]+\frac{c}{2}\E\|v^{k}\|^{2}$
to obtain
\[
\E\!\left[\Phi^{k+1}\right]
\le (1-\rho)\,\E\!\left[\Phi^{k}\right].
\tag{S11}
\]

\noindent\textbf{Step 11.} Unrolling the recursion gives
\[
\E\!\left[\Phi^{k}\right]\le (1-\rho)^{k}\Phi^{0}.
\tag{S12}
\]
Since $\Phi^{k}\ge f(w^{k})-f(w^{*})$, inequality (S12) yields the desired
bound (5).  Using Lemma~\ref{lem:strong} to replace $f(w^{k})-f(w^{*})$
by $\frac{\mu}{2}\|w^{k}-w^{*}\|^{2}$ produces (6).  ∎

\subsection*{Rate Derivation (With Constants)}
From the definition of $\gamma_i$ in (S9) and the stepsize bound (A4) we have
\[
\gamma_i
= \eta_i\Bigl[1-\frac{L_i\eta_i}{2}
      -\frac{\beta}{1-\beta p_i}\Bigr]
\ge \eta_i\Bigl[\frac{1}{2}
      -\frac{\beta}{1-\beta p_i}\Bigr].
\]
Choosing $\eta_i= \frac{p_i}{L_i(\beta+(1-\beta)p_i)}$
(the maximal admissible stepsize) gives
\[
\gamma_i \;\ge\;
\frac{p_i}{L_i(\beta+(1-\beta)p_i)}
\Bigl[\frac12-\frac{\beta}{1-\beta p_i}\Bigr]
=
\frac{p_i}{2L_i}\,
\frac{1-\beta}{\beta+(1-\beta)p_i}.
\]
Hence
\[
\rho
\;=\;
\min_i \frac{2\mu\,\gamma_i p_i}{d_{\max}}
\;\ge\;
\min_i
\frac{\mu\,p_i^{2}}{L_i}\,
\frac{1-\beta}{\beta+(1-\beta)p_i}.
\tag{R1}
\]
Inequality (R1) makes the dependence of the linear rate on the
condition numbers $L_i/\mu$, on the sampling probabilities $p_i$,
and on the momentum parameter $\beta$ explicit.

\subsection*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform sampling.}
$p_i=1/d$ for all $i$.  Then
\[
\rho
\ge \frac{\mu}{\max_i L_i}\,
      \frac{(1-\beta)}{\beta+(1-\beta)/d}.
\]
For $\beta=0$ this recovers the classical rate
$\rho =\frac{\mu}{d\max_i L_i}$.
\item \textbf{Importance sampling.}
Take $p_i\propto L_i$.  Then $p_i = L_i/\sum_j L_j$ and
\[
\rho\ge
\frac{\mu}{\sum_j L_j}\,
\frac{1-\beta}{\beta+(1-\beta)L_i/\sum_j L_j}
\ge
\frac{\mu}{\sum_j L_j}\,(1-\beta),
\]
which can be substantially larger than the uniform case when the
$L_i$ are heterogeneous.
\item \textbf{No momentum.}
Setting $\beta=0$ reduces $\rho$ to
$\displaystyle\rho_{0}= \min_i \frac{\mu\,\eta_i p_i}{1}$,
the standard randomized coordinate descent rate.
\end{enumerate}

\bigskip
The proof above follows a completely elementary path:
smoothness gives a one‑step descent inequality, the momentum term
is controlled via a carefully designed Lyapunov function, and the
expectation over the random coordinate yields a global linear
contraction.  All steps respect the assumptions, and no hidden
constants appear beyond those explicitly defined.
