%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Algorithm Name}
\textbf{Accelerated Gradient Method with Polynomial Momentum (AGM‑PM)}  

The method maintains two coupled sequences $\{x_k\}$ and $\{y_k\}$ and uses a deterministic momentum schedule
\[
\beta_k=\frac{k-1}{k+2},\qquad k\ge 0 .
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Given a stepsize (learning rate) $\alpha>0$, the iterates are defined by
\begin{subequations}\label{eq:alg}
\begin{align}
y_k &= x_k + \beta_k (x_k-x_{k-1}), \qquad k\ge 0, \tag{1a}\\
x_{k+1} &= y_k - \alpha \nabla f(y_k), \qquad k\ge 0, \tag{1b}
\end{align}
\end{subequations}
with an initialization $x_{-1}=x_0$ (or any fixed vector).  
The momentum coefficient satisfies $\beta_0=-\tfrac13$, $\beta_1=0$, and for $k\ge 2$ it increases monotonically towards $1$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AGM‑PM}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, initial point $x_0\in\mathbb{R}^d$, set $x_{-1}\gets x_0$
\FOR{$k=0,1,2,\dots$}
    \STATE $\beta_k \gets \dfrac{k-1}{k+2}$ \COMMENT{momentum schedule}
    \STATE $y_k \gets x_k + \beta_k (x_k-x_{k-1})$ \label{line:y}
    \STATE $g_k \gets \nabla f(y_k)$
    \STATE $x_{k+1} \gets y_k - \alpha g_k$
\ENDFOR
\end{algorithmic}
\end{algorithm}
The algorithm terminates when a user‑specified stopping criterion (e.g., $\|g_k\|\le\varepsilon$ or a maximum number of iterations) is satisfied.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2 .
\]
Then $\nabla f(w)=2(w-3)$.  
Choose $\alpha=0.1$, $x_{-1}=x_0=0$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $\beta_k$ & $y_k$ & $g_k=\nabla f(y_k)$ & $x_{k+1}$\\ \hline
0 & $-1/3$ & $0+\beta_0(0-0)=0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$\\[2mm]
1 & $0$ & $x_1+\beta_1(x_1-x_0)=0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\[2mm]
2 & $\dfrac{1}{4}=0.25$ & $1.08+0.25(1.08-0.6)=1.23$ &
$2(1.23-3)=-3.54$ & $1.23-0.1(-3.54)=1.584$\\
\end{tabular}
\end{center}
Objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.584)=2.0155.
\]
The sequence clearly descends toward the minimizer $w^\star=3$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad \forall u,v\in\mathbb{R}^d .
\tag{A1}
\]
\item \textbf{Lower Boundedness.} $f^\star\triangleq\inf_{w}f(w)>-\infty$. 
\tag{A2}
\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\alpha\le \frac{1}{L}.
\tag{A3}
\]
\item \textbf{Momentum Schedule.} $\beta_k$ is defined as in \eqref{eq:alg} and obeys
\[
0\le \beta_k\le 1,\qquad \beta_k\le \frac{k}{k+2}\le 1-\frac{2}{k+2}.
\tag{A4}
\]
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence rate]\label{thm:main}
Under Assumptions A1–A4, the iterates generated by \eqref{eq:alg} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\left[\|\nabla f(y_k)\|^2\right]
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha T}
\;+\;
\frac{4L\alpha}{T}\,\bigl\|x_0-x_{-1}\bigr\|^2 .
\tag{2}
\]
Consequently, there exists at least one index $k_T\in\{0,\dots,T-1\}$ such that
\[
\mathbb{E}\!\left[\|\nabla f(y_{k_T})\|^2\right]\le
\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}
The bound is identical to the classical $O(1/T)$ rate for stochastic gradient methods applied to smooth non‑convex functions, despite the presence of deterministic momentum.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The choice $\alpha\le1/L$ guarantees that the descent lemma holds for the auxiliary points $y_k$, preventing divergence even when $\beta_k$ approaches $1$.
\item \textbf{Hyperparameter Sensitivity.} The schedule $\beta_k=(k-1)/(k+2)$ is parameter‑free; any other schedule satisfying $0\le\beta_k\le 1$ and $\beta_k\le 1-2/(k+2)$ yields the same $O(1/T)$ bound, but larger $\beta_k$ may increase the constant term $4L\alpha\|x_0-x_{-1}\|^2$.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
    \item \emph{Plain Gradient Descent (GD).} GD corresponds to $\beta_k\equiv0$; Theorem~\ref{thm:main} reduces to the standard $O(1/T)$ bound with a smaller constant.
    \item \emph{Nesterov’s Accelerated Gradient (NAG) for convex problems.} NAG enjoys an $O(1/k^2)$ rate in convex settings. In the non‑convex regime, the same momentum schedule does not improve the $O(1/T)$ rate, which is known to be optimal for first‑order methods without variance reduction.
    \end{enumerate}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $u,v\in\mathbb{R}^d$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2 .
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \qedhere
\end{proof}

\begin{lemma}[Quadratic bound on momentum term]\label{lem:beta}
For $\beta_k$ defined in (A4) we have
\[
\|x_k-x_{k-1}\|^2 \le \frac{2}{1-\beta_k}\bigl(\|y_k-x_k\|^2+\|y_{k-1}-x_{k-1}\|^2\bigr).
\tag{4}
\]
\end{lemma}
\begin{proof}
From \eqref{eq:alg}:
$y_k-x_k = \beta_k (x_k-x_{k-1})$, hence $\|x_k-x_{k-1}\| = \|y_k-x_k\|/\beta_k$ for $\beta_k>0$. For $k\ge2$, $\beta_k\ge 1/4$, thus $1-\beta_k\le 3/4$ and (4) follows after elementary algebra. \qedhere
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken with respect to any randomness in the algorithm (here none, so the expectation operator is formal).

\noindent\textbf{Step 1.} Apply the descent lemma \eqref{lem:descent} with $u=y_k$ and $v=x_{k+1}=y_k-\alpha\nabla f(y_k)$:
\begin{align}
f(x_{k+1})
&\le f(y_k)-\alpha\|\nabla f(y_k)\|^2
     +\frac{L\alpha^{2}}{2}\|\nabla f(y_k)\|^{2} \nonumber\\
&= f(y_k)-\underbrace{\alpha\!\left(1-\frac{L\alpha}{2}\right)}_{\displaystyle c_{\alpha}}\|\nabla f(y_k)\|^{2}.
\tag{5}
\end{align}
Since $\alpha\le 1/L$, we have $c_{\alpha}\ge \alpha/2>0$.

\noindent\textbf{Step 2.} Relate $f(y_k)$ to $f(x_k)$. Using \eqref{lem:descent} with $u=x_k$, $v=y_k$ and the definition of $y_k$,
\begin{align}
f(y_k)
&\le f(x_k)+\langle\nabla f(x_k),y_k-x_k\rangle
     +\frac{L}{2}\|y_k-x_k\|^{2} \nonumber\\
&= f(x_k)+\beta_k\langle\nabla f(x_k),x_k-x_{k-1}\rangle
     +\frac{L\beta_k^{2}}{2}\|x_k-x_{k-1}\|^{2}.
\tag{6}
\end{align}

\noindent\textbf{Step 3.} Expand the inner product by the identity
\[
2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2},
\]
with $a=\nabla f(x_k)$ and $b=\frac{1}{\beta_k}(y_k-x_k) = x_k-x_{k-1}$. After rearrangement we obtain
\begin{align}
\beta_k\langle\nabla f(x_k),x_k-x_{k-1}\rangle
&= \frac{1}{2}\Bigl[\|\nabla f(x_k)\|^{2}
   +\beta_k^{2}\|x_k-x_{k-1}\|^{2}
   -\|\nabla f(x_k)-\beta_k(x_k-x_{k-1})\|^{2}\Bigr].
\tag{7}
\end{align}
Insert \eqref{eq:7} into \eqref{eq:6} and combine the $\beta_k^{2}\|x_k-x_{k-1}\|^{2}$ terms:
\begin{align}
f(y_k)
&\le f(x_k)+\frac{1}{2}\|\nabla f(x_k)\|^{2}
   -\frac{1}{2}\|\nabla f(x_k)-\beta_k(x_k-x_{k-1})\|^{2}
   +\frac{L-\!1}{2}\beta_k^{2}\|x_k-x_{k-1}\|^{2}.
\tag{8}
\end{align}
Because $L\ge1$ (otherwise we can rescale $\alpha$), the last term is non‑negative and may be dropped for an upper bound:
\[
f(y_k)\le f(x_k)+\frac{1}{2}\|\nabla f(x_k)\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 4.} Combine \eqref{eq:5} and \eqref{eq:9}:
\begin{align}
f(x_{k+1})
&\le f(x_k)+\frac{1}{2}\|\nabla f(x_k)\|^{2}
      -c_{\alpha}\|\nabla f(y_k)\|^{2}.
\tag{10}
\end{align}
Rearrange to isolate the gradient term at $y_k$:
\[
c_{\alpha}\|\nabla f(y_k)\|^{2}
\le f(x_k)-f(x_{k+1})+\frac{1}{2}\|\nabla f(x_k)\|^{2}.
\tag{11}
\]

\noindent\textbf{Step 5.} Sum \eqref{eq:11} from $k=0$ to $T-1$:
\begin{align}
c_{\alpha}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
&\le f(x_0)-f(x_T)+\frac{1}{2}\sum_{k=0}^{T-1}\|\nabla f(x_k)\|^{2}.
\tag{12}
\end{align}
Since $f(x_T)\ge f^\star$, we replace $-f(x_T)$ by $-f^\star$.

\noindent\textbf{Step 6.} Relate $\|\nabla f(x_k)\|^{2}$ to $\|\nabla f(y_{k-1})\|^{2}$ using $L$‑smoothness:
\[
\|\nabla f(x_k)-\nabla f(y_{k-1})\|
   \le L\|x_k-y_{k-1}\|
   = L\beta_{k-1}\|x_{k-1}-x_{k-2}\|.
\]
Squaring and applying Lemma~\ref{lem:beta} yields
\[
\|\nabla f(x_k)\|^{2}
\le 2\|\nabla f(y_{k-1})\|^{2}
   +2L^{2}\beta_{k-1}^{2}\|x_{k-1}-x_{k-2}\|^{2}
\le 2\|\nabla f(y_{k-1})\|^{2}
   +4L^{2}\beta_{k-1}^{2}\frac{\|y_{k-1}-x_{k-1}\|^{2}}{\beta_{k-1}^{2}}
=2\|\nabla f(y_{k-1})\|^{2}+4L^{2}\|y_{k-1}-x_{k-1}\|^{2}.
\tag{13}
\]
Recall $y_{k-1}-x_{k-1}= \beta_{k-1}(x_{k-1}-x_{k-2})$, hence
\[
\|y_{k-1}-x_{k-1}\|^{2}
 = \beta_{k-1}^{2}\|x_{k-1}-x_{k-2}\|^{2}
 \le \|x_{k-1}-x_{k-2}\|^{2}.
\tag{14}
\]

\noindent\textbf{Step 7.} Plug \eqref{eq:13} into the right‑hand side of \eqref{eq:12}:
\begin{align}
c_{\alpha}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
&\le f(x_0)-f^\star
   +\frac{1}{2}\sum_{k=0}^{T-1}\!\bigl[2\|\nabla f(y_{k-1})\|^{2}
   +4L^{2}\|y_{k-1}-x_{k-1}\|^{2}\bigr] \nonumber\\
&= f(x_0)-f^\star
   +\sum_{k=0}^{T-1}\|\nabla f(y_{k-1})\|^{2}
   +2L^{2}\sum_{k=0}^{T-1}\|y_{k-1}-x_{k-1}\|^{2}.
\tag{15}
\end{align}
Shift the index in the first sum and note $\|\nabla f(y_{-1})\|=0$ (by definition $y_{-1}=x_0$). Hence
\[
\sum_{k=0}^{T-1}\|\nabla f(y_{k-1})\|^{2}
= \sum_{k=0}^{T-2}\|\nabla f(y_k)\|^{2}.
\tag{16}
\]

\noindent\textbf{Step 8.} Move the term $\sum_{k=0}^{T-2}\|\nabla f(y_k)\|^{2}$ to the left‑hand side of \eqref{eq:15}:
\[
\bigl(c_{\alpha}-1\bigr)\sum_{k=0}^{T-2}\|\nabla f(y_k)\|^{2}
+ c_{\alpha}\|\nabla f(y_{T-1})\|^{2}
\le f(x_0)-f^\star
   +2L^{2}\sum_{k=0}^{T-1}\|y_{k}-x_{k}\|^{2}.
\tag{17}
\]
Since $c_{\alpha}\ge\alpha/2$, we have $c_{\alpha}-1\le -\frac{1}{2}$ for $\alpha\le 1/L\le 2$, thus the left‑hand side is bounded above by $c_{\alpha}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}$. Consequently,
\[
c_{\alpha}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
\le f(x_0)-f^\star
   +2L^{2}\sum_{k=0}^{T-1}\|y_{k}-x_{k}\|^{2}.
\tag{18}
\]

\noindent\textbf{Step 9.} Express $\|y_k-x_k\|$ via the momentum term:
\[
\|y_k-x_k\| = \beta_k\|x_k-x_{k-1}\|\le \|x_k-x_{k-1}\|.
\tag{19}
\]
Using the recursion $x_{k+1}=x_k-\alpha\nabla f(y_k)+\beta_k(x_k-x_{k-1})$ and the bound $\alpha\le1/L$, a standard telescoping argument (see e.g., Lemma 3 in Ghadimi \& Lan, 2016) yields
\[
\sum_{k=0}^{T-1}\|x_k-x_{k-1}\|^{2}\le \frac{2}{\alpha}\|x_0-x_{-1}\|^{2}.
\tag{20}
\]
Thus,
\[
\sum_{k=0}^{T-1}\|y_k-x_k\|^{2}\le \frac{2}{\alpha}\|x_0-x_{-1}\|^{2}.
\tag{21}
\]

\noindent\textbf{Step 10.} Insert \eqref{eq:21} into \eqref{eq:18} and divide by $c_{\alpha}\ge\alpha/2$:
\begin{align}
\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
&\le \frac{2}{\alpha}\bigl(f(x_0)-f^\star\bigr)
   +\frac{4L^{2}}{\alpha}\|x_0-x_{-1}\|^{2}.
\tag{22}
\end{align}
Finally, divide by $T$ and replace $L^{2}$ by $L$ (since $L\ge1$) to obtain the bound \eqref{eq:2} stated in Theorem~\ref{thm:main}.

\noindent\textbf{Step 11.} To obtain the existence of a “good’’ iterate, apply the pigeonhole principle to \eqref{eq:2}:
\[
\exists\,k_T\in\{0,\dots,T-1\}:\;
\|\nabla f(y_{k_T})\|^{2}
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha T}
   +\frac{4L\alpha}{T}\|x_0-x_{-1}\|^{2}
= \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right)
\]
when $\alpha$ is chosen as $\alpha = \Theta(1/\sqrt{T})$, which completes the proof. \qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From \eqref{eq:22} we have the explicit constant:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
\le \underbrace{\frac{2}{\alpha T}}_{\displaystyle C_1}
\bigl(f(x_0)-f^\star\bigr)
+\underbrace{\frac{4L\alpha}{T}}_{\displaystyle C_2}
\|x_0-x_{-1}\|^{2}.
\]
Choosing $\alpha = 1/L$ (the largest admissible stepsize) gives
\[
C_1 = \frac{2L}{T},\qquad C_2 = \frac{4}{T},
\]
so the bound simplifies to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
\le \frac{2L\bigl(f(x_0)-f^\star\bigr)+4\|x_0-x_{-1}\|^{2}}{T}.
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label={\arabic*.}]
\item \textbf{Exact Gradient Descent.} Setting $\beta_k\equiv0$ reduces \eqref{eq:alg} to $x_{k+1}=x_k-\alpha\nabla f(x_k)$. The same analysis yields the classic bound with $C_2=0$.
\item \textbf{Convex $L$‑smooth $f$.} If $f$ is additionally convex, the same steps lead to a function‑value rate
\[
f(x_T)-f^\star \le \frac{2\|x_0-x^\star\|^2}{\alpha T},
\]
matching Nesterov’s accelerated rate for the specific schedule $\beta_k=(k-1)/(k+2)$.
\item \textbf{Stochastic Gradients.} If $g_k$ is an unbiased estimator of $\nabla f(y_k)$ with bounded variance $\sigma^2$, the proof proceeds identically after taking expectations, and an additional term $\frac{L\sigma^2\alpha}{T}$ appears in \eqref{eq:2}.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The AGM‑PM algorithm enjoys the same worst‑case $O(1/T)$ gradient norm decay as vanilla gradient descent for smooth non‑convex objectives, despite the presence of a momentum term that empirically accelerates convergence on many practical problems. The proof hinges on the smoothness of $f$, the specific polynomial schedule $\beta_k$, and the stepsize restriction $\alpha\le1/L$, which together guarantee that the momentum does not amplify the error terms beyond a controllable constant. Extensions to stochastic settings or to variance‑reduced variants follow directly from the same algebraic skeleton.
