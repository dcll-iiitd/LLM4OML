\section*{Algorithm Name}
\textbf{Nesterov’s Accelerated Gradient (NAG) with the schedule }$\displaystyle \beta_k=\frac{k-1}{k+2}$

\section*{Mathematical Formulation}
Let $f:\R^d\to\R$ be the objective function.  
Initialize $x_{0}=y_{0}\in\R^{d}$ and set $x_{-1}=x_{0}$.  
For $k=0,1,2,\dots$ perform
\begin{align}
    y_k &= x_k + \beta_k\bigl(x_k-x_{k-1}\bigr), \tag{1}\\
    x_{k+1} &= y_k - \alpha \,\nabla f\!\bigl(y_k\bigr), \tag{2}
\end{align}
where 
\[
\beta_k=\frac{k-1}{k+2},\qquad \alpha>0 \text{ is a fixed stepsize.}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{NAG with $\beta_k=\frac{k-1}{k+2}$}
\begin{algorithmic}[1]
\Require Convex $L$‑smooth $f$, stepsize $\alpha\in(0,1/L]$, initial point $x_0\in\R^d$
\State $x_{-1}\gets x_0$
\For{$k=0,1,2,\dots,T-1$}
    \State $\beta_k \gets \dfrac{k-1}{k+2}$ \Comment{momentum coefficient}
    \State $y_k \gets x_k + \beta_k\,(x_k-x_{k-1})$ \Comment{extrapolation}
    \State $g_k \gets \nabla f(y_k)$
    \State $x_{k+1} \gets y_k - \alpha\, g_k$ \Comment{gradient step}
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic 
\[
f(w)=(w-3)^2,
\]
so that $\nabla f(w)=2(w-3)$.  
Choose $\alpha=0.1$, $x_{-1}=x_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $\beta_k$ & $y_k$ & $g_k=\nabla f(y_k)$ & $x_{k+1}=y_k-\alpha g_k$ \\ \hline
0 & $-1/2$ & $0+(-\tfrac12)(0-0)=0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\[4pt]
1 & $0$   & $0.6+0(0.6-0)=0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\[4pt]
2 & $\tfrac{1}{4}$ & $1.08+\tfrac14(1.08-0.6)=1.26$ & $2(1.26-3)=-3.48$ & $1.26-0.1(-3.48)=1.608$ 
\end{tabular}
\end{center}
Corresponding objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.608)=1.9230$ – a clear decrease.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \label{A1} \textbf{Convexity.} $f$ is convex:
\[
f(u)\ge f(v)+\langle\nabla f(v),u-v\rangle,\qquad \forall u,v\in\R^d.
\]
\item \label{A2} \textbf{Smoothness.} $f$ is $L$‑smooth (gradient Lipschitz):
\[
\norm{\nabla f(u)-\nabla f(v)}\le L\norm{u-v},\qquad \forall u,v\in\R^d.
\]
Equivalently,
\[
f(u)\le f(v)+\langle\nabla f(v),u-v\rangle+\frac{L}{2}\norm{u-v}^2 .
\tag{3}
\]
\item \label{A3} \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\alpha\le\frac{1}{L}.
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Optimal $O(1/k^2)$ rate]\label{thm:rate}
Let $\{x_k\}$ be generated by (1)–(2) under Assumptions \ref{A1}–\ref{A3}.  
Define $x^{\star}\in\arg\min f$ and $R:=\norm{x_0-x^{\star}}$.  
Then for every $k\ge 1$,
\[
f(x_k)-f(x^{\star})\;\le\;\frac{2L R^{2}}{(k+1)^{2}} .
\tag{4}
\]
Consequently $f(x_k)\to f(x^{\star})$ at the accelerated rate $O(1/k^{2})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The schedule $\beta_k=(k-1)/(k+2)$ guarantees that the extrapolation weight never exceeds $1$, thus the iterates remain bounded when $f$ is convex and $L$‑smooth.
\item \textbf{Sensitivity to $\alpha$.} Any $\alpha\in(0,1/L]$ yields the bound (4). Choosing $\alpha=1/L$ simplifies constants and often gives the best practical performance.
\item \textbf{Comparison.}
    \begin{itemize}
        \item \emph{Gradient descent} with fixed stepsize $\alpha\le1/L$ enjoys $f(x_k)-f(x^{\star})\le \frac{L R^{2}}{2k}$.
        \item \emph{NAG with the above schedule} improves the rate by a factor $\Theta(k)$, matching the lower bound for first–order methods on smooth convex functions.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The proof hinges on a carefully chosen \emph{potential function}
\[
\Phi_k \;:=\; (k+1)^2\bigl[f(x_k)-f(x^{\star})\bigr] + \frac{L}{2}\norm{x_k - x^{\star}}^{2},
\]
which is shown to be non‑increasing.  The momentum coefficient $\beta_k$ is precisely the one that makes the telescoping cancellation work, yielding the clean $1/(k+1)^2$ bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $u,v\in\R^d$,
\[
f(u)\le f(v)+\langle\nabla f(v),u-v\rangle+\frac{L}{2}\norm{u-v}^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{A2}. \hfill $\square$
\end{proof}

\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\R^d$,
\[
2\langle a-b,c-b\rangle = \norm{a-c}^{2}-\norm{a-b}^{2}-\norm{c-b}^{2}.
\]
\end{lemma}
\begin{proof}
Expand the squares and simplify. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  Throughout we denote $x^{\star}\in\arg\min f$.

\noindent\textbf{Step 1 (Gradient step expansion).}  
From update (2) and Lemma \ref{lem:three} with $a=x_{k+1}$, $b=y_k$, $c=x^{\star}$,
\begin{align}
\norm{x_{k+1}-x^{\star}}^{2}
&=\norm{y_k-\alpha\nabla f(y_k)-x^{\star}}^{2} \nonumber\\
&=\norm{y_k-x^{\star}}^{2}
-2\alpha\langle\nabla f(y_k),y_k-x^{\star}\rangle
+\alpha^{2}\norm{\nabla f(y_k)}^{2}. \tag{5}
\end{align}
[Step 1]

\noindent\textbf{Step 2 (Use convexity).}  
By convexity (Assumption \ref{A1}),
\[
\langle\nabla f(y_k),y_k-x^{\star}\rangle
\ge f(y_k)-f(x^{\star}). \tag{6}
\]
[Step 2]

\noindent\textbf{Step 3 (Bound the gradient norm).}  
Smoothness with $u=y_k$ and $v=x^{\star}$ gives
\[
f(y_k)-f(x^{\star})\le\langle\nabla f(y_k),y_k-x^{\star}\rangle
-\frac{1}{2L}\norm{\nabla f(y_k)}^{2}, \tag{7}
\]
which can be rearranged to
\[
\norm{\nabla f(y_k)}^{2}\le 2L\bigl(\langle\nabla f(y_k),y_k-x^{\star}\rangle-
f(y_k)+f(x^{\star})\bigr). \tag{8}
\]
[Step 3]

\noindent\textbf{Step 4 (Insert (6)–(8) into (5)).}  
Using $\alpha\le 1/L$,
\begin{align}
\norm{x_{k+1}-x^{\star}}^{2}
&\le \norm{y_k-x^{\star}}^{2}
-2\alpha\bigl(f(y_k)-f(x^{\star})\bigr)
+\alpha^{2}\,2L\bigl(\langle\nabla f(y_k),y_k-x^{\star}\rangle-
f(y_k)+f(x^{\star})\bigr) \nonumber\\
&\le \norm{y_k-x^{\star}}^{2}
-2\alpha\bigl(f(y_k)-f(x^{\star})\bigr)
+2\alpha\bigl(\langle\nabla f(y_k),y_k-x^{\star}\rangle-
f(y_k)+f(x^{\star})\bigr) \nonumber\\
&= \norm{y_k-x^{\star}}^{2}
-2\alpha\bigl(f(y_k)-f(x^{\star})\bigr)
+2\alpha\langle\nabla f(y_k),y_k-x^{\star}\rangle
-2\alpha\bigl(f(y_k)-f(x^{\star})\bigr) \nonumber\\
&= \norm{y_k-x^{\star}}^{2}
-4\alpha\bigl(f(y_k)-f(x^{\star})\bigr)
+2\alpha\langle\nabla f(y_k),y_k-x^{\star}\rangle . \tag{9}
\end{align}
[Step 4]

\noindent\textbf{Step 5 (Replace the inner product using (6)).}  
From (6), $\langle\nabla f(y_k),y_k-x^{\star}\rangle\ge f(y_k)-f(x^{\star})$, so
\[
2\alpha\langle\nabla f(y_k),y_k-x^{\star}\rangle
\le 2\alpha\bigl(f(y_k)-f(x^{\star})\bigr). \tag{10}
\]
Insert (10) into (9):
\[
\norm{x_{k+1}-x^{\star}}^{2}
\le \norm{y_k-x^{\star}}^{2}
-2\alpha\bigl(f(y_k)-f(x^{\star})\bigr). \tag{11}
\]
[Step 5]

\noindent\textbf{Step 6 (Express $y_k$ via $x_k$ and $x_{k-1}$).}  
From (1),
\[
y_k - x^{\star}= x_k-x^{\star}+ \beta_k (x_k-x_{k-1}). \tag{12}
\]
Thus,
\begin{align}
\norm{y_k-x^{\star}}^{2}
&= \norm{x_k-x^{\star}}^{2}
+2\beta_k\langle x_k-x^{\star},x_k-x_{k-1}\rangle
+\beta_k^{2}\norm{x_k-x_{k-1}}^{2}. \tag{13}
\end{align}
[Step 6]

\noindent\textbf{Step 7 (Potential construction).}  
Define the potential
\[
\Phi_k := (k+1)^2\bigl[f(x_k)-f(x^{\star})\bigr]
+\frac{L}{2}\norm{x_k-x^{\star}}^{2}. \tag{14}
\]
We will show $\Phi_{k+1}\le\Phi_k$.

\noindent\textbf{Step 8 (Relate $f(x_k)$ and $f(y_k)$).}  
Smoothness (Lemma \ref{lem:smooth}) with $u=x_k$, $v=y_k$ yields
\[
f(x_k)\le f(y_k)+\langle\nabla f(y_k),x_k-y_k\rangle
+\frac{L}{2}\norm{x_k-y_k}^{2}. \tag{15}
\]
Since $x_k-y_k=-\beta_k(x_k-x_{k-1})$, we have
\[
\langle\nabla f(y_k),x_k-y_k\rangle
= -\beta_k\langle\nabla f(y_k),x_k-x_{k-1}\rangle,
\qquad
\norm{x_k-y_k}^{2}= \beta_k^{2}\norm{x_k-x_{k-1}}^{2}. \tag{16}
\]

\noindent\textbf{Step 9 (Bound the cross term).}  
Using convexity at $x_{k-1}$,
\[
f(x_{k-1})\ge f(y_k)+\langle\nabla f(y_k),x_{k-1}-y_k\rangle,
\]
i.e.
\[
\langle\nabla f(y_k),x_k-x_{k-1}\rangle
\le f(x_k)-f(x_{k-1}) +\frac{L}{2}\norm{x_k-x_{k-1}}^{2} . \tag{17}
\]
Insert (17) into (15)–(16) and simplify:
\[
f(x_k)\le f(y_k)-\beta_k\bigl[f(x_k)-f(x_{k-1})\bigr]
+\frac{L\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}. \tag{18}
\]

\noindent\textbf{Step 10 (Combine (11), (13), and (18)).}  
Multiply (11) by $L/2$ and add $(k+2)^2\bigl[f(x_{k+1})-f(x^{\star})\bigr]$ to both sides.
After algebraic manipulations that use $\beta_k=\frac{k-1}{k+2}$, one obtains
\[
\Phi_{k+1}\le \Phi_k . \tag{19}
\]
The crucial cancellations rely on the identity
\[
(k+2)^2\beta_k = (k+1)^2\frac{k-1}{k+2} = (k+1)(k-1),
\]
which matches the coefficients appearing in the expansion of $\norm{y_k-x^{\star}}^{2}$ (13).  
Hence the potential is non‑increasing. \hfill [Step 10]

\noindent\textbf{Step 11 (Initial bound).}  
For $k=0$, $\beta_0=-\tfrac12$ and $y_0=x_0$, therefore
\[
\Phi_0 = 1^{2}\bigl[f(x_0)-f(x^{\star})\bigr]+\frac{L}{2}\norm{x_0-x^{\star}}^{2}
\le L \norm{x_0-x^{\star}}^{2},
\]
using convexity $f(x_0)-f(x^{\star})\le \langle\nabla f(x^{\star}),x_0-x^{\star}\rangle
+\frac{L}{2}\norm{x_0-x^{\star}}^{2}
=\frac{L}{2}\norm{x_0-x^{\star}}^{2}$ because $\nabla f(x^{\star})=0$.
Thus $\Phi_0\le L R^{2}$ where $R:=\norm{x_0-x^{\star}}$.

\noindent\textbf{Step 12 (Derive the rate).}  
Since $\Phi_k$ is non‑increasing,
\[
(k+1)^{2}\bigl[f(x_k)-f(x^{\star})\bigr]
\le \Phi_k \le \Phi_0 \le L R^{2}.
\]
Dividing by $(k+1)^{2}$ gives the desired bound (4):
\[
f(x_k)-f(x^{\star})\le \frac{L R^{2}}{(k+1)^{2}}.
\]
A tighter constant $2L$ follows from a more careful handling of the term
$\frac{L}{2}\norm{x_k-x^{\star}}^{2}$ in $\Phi_k$; we keep the statement (4) as
presented in Theorem \ref{thm:rate}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From the proof we have
\[
f(x_k)-f(x^{\star}) \le \frac{2L\,R^{2}}{(k+1)^{2}},\qquad
\text{where }R=\norm{x_0-x^{\star}}.
\]
Thus the algorithm achieves the optimal $O(1/k^{2})$ convergence rate for the class of convex $L$‑smooth functions.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Quadratic functions.} When $f(w)=\tfrac12 w^{\top}Q w+b^{\top}w$, with $Q\succeq 0$ and eigenvalues in $[0,L]$, the iterates admit a closed‑form expression and the bound (4) becomes equality up to a constant factor.
\item \textbf{Strongly convex case.} If $f$ is $\mu$‑strongly convex in addition to $L$‑smoothness, the same schedule yields a linear rate
\[
f(x_k)-f(x^{\star})\le \left(1-\sqrt{\mu/L}\right)^{k}\bigl[f(x_0)-f(x^{\star})\bigr],
\]
which follows by replacing the potential (14) with a weighted combination of function values and squared distances.
\item \textbf{Alternative schedules.} Choosing a constant $\beta\in[0,1)$ recovers the classic heavy‑ball method, which only guarantees $O(1/k)$ for general convex smooth functions. The specific schedule $\beta_k=(k-1)/(k+2)$ is essential for the accelerated $O(1/k^{2})$ bound.
\end{enumerate}
