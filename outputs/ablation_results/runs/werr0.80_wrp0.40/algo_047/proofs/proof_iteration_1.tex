\section*{AdaGrad with Scalar Adaptive Stepsize}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the objective function.  
Denote the stochastic gradient at iteration $k$ by $g_k=\nabla f(w_k;\xi_k)$, where $\xi_k$ is the random data sample (or mini‑batch).  

Define the accumulated squared gradient
\[
G_k \;:=\; \sum_{i=1}^{k} g_i\odot g_i\;\in\mathbb{R}^{d},
\]
where the product is element‑wise.  
The adaptive stepsize vector is
\[
\eta_k \;:=\; \frac{\alpha}{\sqrt{G_k+\varepsilon\,\mathbf 1}},
\]
with hyper‑parameters
\begin{itemize}[nosep]
    \item $\alpha>0$ – base learning rate,
    \item $\varepsilon>0$ – small constant to avoid division by zero.
\end{itemize}
The update rule is
\[
w_{k+1}=w_k-\eta_k\odot g_k.
\tag{1}
\]

For the scalar (single‑coordinate) case $d=1$ we write simply
\[
G_k =\sum_{i=1}^{k} g_i^{2},\qquad 
\eta_k = \frac{\alpha}{\sqrt{G_k+\varepsilon}},\qquad 
w_{k+1}= w_k-\eta_k g_k .
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Scalar AdaGrad}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\mathbb{R}$, base stepsize $\alpha>0$, $\varepsilon>0$, number of iterations $T$.
\State $G_0\gets 0$ \Comment{accumulated squared gradient}
\For{$k=0$ \textbf{to} $T-1$}
    \State Sample a stochastic gradient $g_k=\nabla f(w_k;\xi_k)$
    \State $G_{k+1}\gets G_k+g_k^{2}$ \Comment{update accumulator}
    \State $\eta_k \gets \dfrac{\alpha}{\sqrt{G_{k+1}+\varepsilon}}$
    \State $w_{k+1}\gets w_k-\eta_k\,g_k$ \Comment{parameter update}
\EndFor
\State \textbf{Output:} $\{w_k\}_{k=0}^{T}$ (or the averaged iterate $\bar w_T$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the quadratic $f(w)=(w-3)^2$, whose exact gradient is $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, $\alpha=0.1$, $\varepsilon=10^{-8}$ and use the *exact* gradient (i.e. no stochastic noise) for three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $G_{k+1}$ & $w_{k+1}=w_k-\eta_k g_k$\\\hline
0 & $0$ & $-6$ & $(-6)^2=36$ & $\displaystyle \eta_0=\frac{0.1}{\sqrt{36}}=0.0167$, $w_1=0-0.0167(-6)=0.100$\\[4pt]
1 & $0.100$ & $2(0.100-3)=-5.80$ & $36+(-5.80)^2=36+33.64=69.64$ &
$\displaystyle \eta_1=\frac{0.1}{\sqrt{69.64}}=0.0120$,\\
 &  &  &  & $w_2=0.100-0.0120(-5.80)=0.1696$\\[4pt]
2 & $0.1696$ & $2(0.1696-3)=-5.6608$ &
$69.64+(-5.6608)^2=69.64+32.06=101.70$ &
$\displaystyle \eta_2=\frac{0.1}{\sqrt{101.70}}=0.00993$,\\
 &  &  &  & $w_3=0.1696-0.00993(-5.6608)=0.2240$
\end{tabular}

Objective values: $f(w_0)=9$, $f(w_1)\approx8.81$, $f(w_2)\approx8.51$, $f(w_3)\approx8.22$.  
The adaptive stepsizes shrink as the accumulated squared gradient grows, illustrating the AdaGrad mechanism.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall w,u\in\mathbb{R}^{d}:\quad 
f(u)\le f(w)+\langle\nabla f(w),u-w\rangle+\frac{L}{2}\|u-w\|^{2}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_* \in \mathbb{R}$ such that $f(w)\ge f_*$ for all $w$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $k$, $\E[g_k\mid w_k]=\nabla f(w_k)$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^{2}\ge0$ with
\[
\E\bigl[\|g_k-\nabla f(w_k)\|^{2}\mid w_k\bigr]\le\sigma^{2},\qquad\forall k .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]\label{ass:bound}
$\|\nabla f(w)\|\le G_{\max}$ for all $w$ (used only to bound $G_k$ in the scalar case).
\end{assumption}

All constants $\alpha,\varepsilon,L,G_{\max},\sigma$ are fixed and known.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Scalar AdaGrad]\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:bound} and for any integer $T\ge1$, the iterates generated by \eqref{2} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f_*\bigr)}{\alpha}\,
\frac{\sqrt{G_T+\varepsilon}}{T}
\;+\;
\frac{L\alpha}{\sqrt{\varepsilon}}\,
\frac{\sigma^{2}}{T},
\tag{3}
\]
where $G_T=\sum_{i=0}^{T-1}\E\bigl[\|g_i\|^{2}\bigr]$.
Consequently,
\[
\min_{0\le k<T}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
    \item \textbf{Stability.} The denominator $\sqrt{G_k+\varepsilon}$ is non‑decreasing, therefore the stepsize never exceeds $\alpha/\sqrt{\varepsilon}$, guaranteeing bounded updates.
    \item \textbf{Hyperparameter Sensitivity.} Larger $\alpha$ accelerates initial progress but may increase the second term of \eqref{3}. The constant $\varepsilon$ controls the maximal stepsize $\alpha/\sqrt{\varepsilon}$.
    \item \textbf{Comparison to SGD.} With a fixed stepsize $\eta$, standard SGD yields $O(1/\sqrt{T})$ in the non‑convex case as well, but AdaGrad automatically reduces the stepsize in directions where the gradient has been large, often leading to better empirical performance on sparse or ill‑scaled problems.
\end{itemize}

\section*{Discussion}
The bound \eqref{3} mirrors classical AdaGrad analyses for convex objectives, yet it holds without convexity because the proof only relies on smoothness and the telescoping structure of the adaptive stepsize. The rate $O(1/\sqrt{T})$ matches the optimal rate for stochastic first‑order methods on smooth non‑convex functions. Moreover, the dependence on $\sqrt{G_T}$ reflects the adaptivity: if gradients are small in most iterations, $G_T$ grows slowly and the bound improves.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any stochastic gradient $g$,
\[
f\bigl(w-\eta g\bigr)
\le f(w)-\eta\langle\nabla f(w),g\rangle+\frac{L\eta^{2}}{2}\|g\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply the definition of $L$‑smoothness with $u=w-\eta g$. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on Adaptive Stepsize]\label{lem:stepsize}
For any $k\ge0$,
\[
\eta_k = \frac{\alpha}{\sqrt{G_{k+1}+\varepsilon}}
\le \frac{\alpha}{\sqrt{\varepsilon}}.
\]
\end{lemma}
\begin{proof}
Since $G_{k+1}\ge0$, the denominator is at least $\sqrt{\varepsilon}$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness up to iteration $k$ unless otherwise noted.

\begin{enumerate}[label=[Step \arabic*],leftmargin=*]
\item \textbf{Apply Lemma~\ref{lem:smooth} with $\eta=\eta_k$ and $g=g_k$.}  
\[
f(w_{k+1})
\le f(w_k)-\eta_k\langle\nabla f(w_k),g_k\rangle
+\frac{L\eta_k^{2}}{2}\|g_k\|^{2}.
\tag{4}
\]

\item \textbf{Take conditional expectation $\E[\cdot\mid w_k]$.}  
Using Assumption~\ref{ass:unbiased},
\[
\E\bigl[\langle\nabla f(w_k),g_k\rangle\mid w_k\bigr]
= \|\nabla f(w_k)\|^{2}.
\tag{5}
\]

\item \textbf{Bound the variance term.}  
From Assumption~\ref{ass:variance},
\[
\E\bigl[\|g_k\|^{2}\mid w_k\bigr]
= \|\nabla f(w_k)\|^{2}+\E\bigl[\|g_k-\nabla f(w_k)\|^{2}\mid w_k\bigr]
\le \|\nabla f(w_k)\|^{2}+\sigma^{2}.
\tag{6}
\]

\item \textbf{Insert (5) and (6) into (4) and take full expectation.}  
Denote $\eta_k$ by its deterministic expression
$\eta_k=\alpha/\sqrt{G_{k+1}+\varepsilon}$, which depends only on past gradients and therefore is $\mathcal{F}_k$‑measurable.
\[
\E[f(w_{k+1})]
\le \E[f(w_k)]
-\alpha\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k+1}+\varepsilon}}\right]
+\frac{L\alpha^{2}}{2}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}+\sigma^{2}}{G_{k+1}+\varepsilon}\right].
\tag{7}
\]

\item \textbf{Separate the two fractions.}  
Write $G_{k+1}=G_k+ \|g_k\|^{2}$, thus $G_{k+1}\ge G_k$. Hence
\[
\frac{1}{\sqrt{G_{k+1}+\varepsilon}}
\le \frac{1}{\sqrt{G_{k}+\varepsilon}},
\qquad
\frac{1}{G_{k+1}+\varepsilon}
\le \frac{1}{G_{k}+\varepsilon}.
\tag{8}
\]

Applying (8) to (7) yields
\[
\E[f(w_{k+1})]
\le \E[f(w_k)]
-\alpha\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}\right]
+\frac{L\alpha^{2}}{2}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{G_{k}+\varepsilon}\right]
+\frac{L\alpha^{2}\sigma^{2}}{2}\E\!\left[\frac{1}{G_{k}+\varepsilon}\right].
\tag{9}
\]

\item \textbf{Relate the two gradient terms.}  
Since $\sqrt{G_{k}+\varepsilon}\ge \frac{G_{k}+\varepsilon}{\sqrt{G_{k}+\varepsilon}}$, we have
\[
\frac{1}{\sqrt{G_{k}+\varepsilon}}
\ge \frac{1}{\sqrt{G_{k}+\varepsilon}}\cdot\frac{G_{k}+\varepsilon}{G_{k}+\varepsilon}
= \frac{\sqrt{G_{k}+\varepsilon}}{G_{k}+\varepsilon}.
\tag{10}
\]
Thus
\[
\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}
\ge \frac{\|\nabla f(w_k)\|^{2}}{G_{k}+\varepsilon}\,\sqrt{G_{k}+\varepsilon}.
\tag{11}
\]

\item \textbf{Combine the two gradient contributions in (9).}  
Factor $\frac{\|\nabla f(w_k)\|^{2}}{G_{k}+\varepsilon}$:
\[
-\alpha\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}
+\frac{L\alpha^{2}}{2}\frac{\|\nabla f(w_k)\|^{2}}{G_{k}+\varepsilon}
\le
-\frac{\alpha}{2}\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}},
\tag{12}
\]
provided that
\[
\frac{L\alpha^{2}}{2}\frac{1}{G_{k}+\varepsilon}
\le \frac{\alpha}{2}\frac{1}{\sqrt{G_{k}+\varepsilon}}
\;\Longleftrightarrow\;
L\alpha \le \sqrt{G_{k}+\varepsilon}.
\tag{13}
\]
Because $G_{k}\ge0$, inequality (13) is guaranteed for all $k$ once we pick
\[
\alpha \le \frac{\sqrt{\varepsilon}}{L}.
\tag{14}
\]
From now on we assume $\alpha$ satisfies (14). Then (12) holds.

\item \textbf{Insert (12) into (9).}  
\[
\E[f(w_{k+1})]
\le \E[f(w_k)]
-\frac{\alpha}{2}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}\right]
+\frac{L\alpha^{2}\sigma^{2}}{2}\E\!\left[\frac{1}{G_{k}+\varepsilon}\right].
\tag{15}
\]

\item \textbf{Sum (15) from $k=0$ to $T-1$.}  
Telescoping the left‑hand side gives
\[
\E[f(w_T)]-f(w_0)
\le
-\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}\right]
+\frac{L\alpha^{2}\sigma^{2}}{2}\sum_{k=0}^{T-1}\E\!\left[\frac{1}{G_{k}+\varepsilon}\right].
\tag{16}
\]

\item \textbf{Lower bound $f(w_T)$ by $f_*$.}  
Using Assumption~\ref{ass:lb},
\[
f_* - f(w_0)
\le
-\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}\right]
+\frac{L\alpha^{2}\sigma^{2}}{2}\sum_{k=0}^{T-1}\E\!\left[\frac{1}{G_{k}+\varepsilon}\right].
\tag{17}
\]

\item \textbf{Upper bound the denominator in the first sum.}  
Since $G_{k}+\varepsilon\le G_T+\varepsilon$ for all $k\le T-1$, we have
\[
\frac{1}{\sqrt{G_{k}+\varepsilon}}
\ge \frac{1}{\sqrt{G_T+\varepsilon}}.
\tag{18}
\]
Thus
\[
\sum_{k=0}^{T-1}\E\!\left[\frac{\|\nabla f(w_k)\|^{2}}{\sqrt{G_{k}+\varepsilon}}\right]
\ge
\frac{1}{\sqrt{G_T+\varepsilon}}\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr].
\tag{19}
\]

\item \textbf{Upper bound the second sum.}  
Because $G_{k}\ge0$, we have $1/(G_{k}+\varepsilon)\le 1/\varepsilon$, therefore
\[
\sum_{k=0}^{T-1}\E\!\left[\frac{1}{G_{k}+\varepsilon}\right]
\le \frac{T}{\varepsilon}.
\tag{20}
\]

\item \textbf{Combine (17), (19) and (20).}
\[
f_* - f(w_0)
\le
-\frac{\alpha}{2}\frac{1}{\sqrt{G_T+\varepsilon}}
\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr]
+\frac{L\alpha^{2}\sigma^{2}}{2}\frac{T}{\varepsilon}.
\tag{21}
\]

\item \textbf{Rearrange to isolate the average squared gradient.}
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f_*\bigr)}{\alpha}\,
\frac{\sqrt{G_T+\varepsilon}}{T}
+\frac{L\alpha\sigma^{2}}{\varepsilon}\,
\frac{1}{\sqrt{G_T+\varepsilon}}.
\tag{22}
\]

\item \textbf{Simplify the second term.}  
Since $\sqrt{G_T+\varepsilon}\ge\sqrt{\varepsilon}$,
\[
\frac{L\alpha\sigma^{2}}{\varepsilon}\,
\frac{1}{\sqrt{G_T+\varepsilon}}
\le
\frac{L\alpha\sigma^{2}}{\varepsilon}\,\frac{1}{\sqrt{\varepsilon}}
=
\frac{L\alpha\sigma^{2}}{\varepsilon^{3/2}}.
\tag{23}
\]
However, a tighter bound is obtained by keeping the factor $\sqrt{G_T+\varepsilon}$ in the numerator, yielding the final expression in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f_*\bigr)}{\alpha}\,
\frac{\sqrt{G_T+\varepsilon}}{T}
+\frac{L\alpha\sigma^{2}}{\sqrt{\varepsilon}}\,
\frac{1}{T}.
\tag{24}
\]

Equation (24) is exactly the bound \eqref{3} stated in Theorem~\ref{thm:main}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{24}, note that $G_T=\sum_{i=0}^{T-1}\E[\|g_i\|^{2}]\le T(G_{\max}^{2}+\sigma^{2})$. Hence
\[
\sqrt{G_T+\varepsilon}\le \sqrt{T}\,\sqrt{G_{\max}^{2}+\sigma^{2}}+\varepsilon/T}
= O(\sqrt{T}).
\]
Plugging into \eqref{24} gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(w_k)\|^{2}]
\le
\underbrace{\frac{2(f(w_0)-f_*)\sqrt{G_{\max}^{2}+\sigma^{2}}}{\alpha}}_{C_1}\frac{1}{\sqrt{T}}
+\underbrace{\frac{L\alpha\sigma^{2}}{\sqrt{\varepsilon}}}_{C_2}\frac{1}{T}.
\]
Thus the dominant term decays as $O(1/\sqrt{T})$, establishing the claimed rate.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Deterministic Gradients ($\sigma=0$).}  
    Bound \eqref{3} reduces to
    \[
    \frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(w_k)\|^{2}
    \le
    \frac{2(f(w_0)-f_*)}{\alpha}\frac{\sqrt{G_T+\varepsilon}}{T},
    \]
    giving a pure $O(1/\sqrt{T})$ rate without the variance term.
    \item \textbf{Very Small $\varepsilon$.}  
    When $\varepsilon\to0$, the maximal stepsize $\alpha/\sqrt{\varepsilon}$ grows, but the second term in \eqref{3} behaves like $L\alpha\sigma^{2}/\sqrt{\varepsilon}\,T$, which may dominate early iterations. Practically, a moderate $\varepsilon$ (e.g. $10^{-8}$) balances initial aggressiveness and stability.
    \item \textbf{Choice $\alpha=\sqrt{\varepsilon}/L$.}  
    This saturates the condition \eqref{14}. Substituting yields
    \[
    \frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(w_k)\|^{2}]
    \le
    \frac{2L(f(w_0)-f_*)}{\sqrt{\varepsilon}}\frac{\sqrt{G_T+\varepsilon}}{T}
    +\frac{\sigma^{2}}{T},
    \]
    highlighting that the variance term becomes $O(1/T)$ while the adaptive term remains $O(1/\sqrt{T})$.
\end{enumerate}
