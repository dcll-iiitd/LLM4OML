\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow \R$ be the objective function and let $\{\xi_t\}_{t\ge 0}$ be an i.i.d. sequence of random variables that defines a stochastic gradient oracle 
\[
g_t \;:=\; \nabla f(w_t;\xi_t).
\]
The learning rate follows a deterministic \emph{cyclic} schedule with period $M\in\mathbb{N}$ and bounds $0<\eta_{\min}<\eta_{\max}$.  
A common choice (triangular wave) is  

\[
\eta_t \;=\;
\begin{cases}
\displaystyle \eta_{\min}+ \frac{2(\eta_{\max}-\eta_{\min})}{M}\; \bigl(t\bmod \tfrac{M}{2}\bigr),
& 0\le t\bmod M < \tfrac{M}{2},\\[2ex]
\displaystyle \eta_{\max}- \frac{2(\eta_{\max}-\eta_{\min})}{M}\;\bigl(t\bmod \tfrac{M}{2}\bigr),
& \tfrac{M}{2}\le t\bmod M < M .
\end{cases}
\tag{1}
\]

The update rule is
\[
w_{t+1}= w_t - \eta_t\, g_t,\qquad t=0,1,2,\dots .
\tag{2}
\]

Hyper‑parameters:
\begin{itemize}
    \item $\eta_{\min},\eta_{\max}>0$ (bounds of the cycle);
    \item $M\in\mathbb{N}$ (cycle length);
    \item $T\in\mathbb{N}$ (maximum number of iterations);
    \item optional: minibatch size $B$ (determines the variance of $g_t$).
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\R^{d}$, $\eta_{\min},\eta_{\max}>0$, cycle length $M$, total steps $T$.
\State Set $t\gets 0$.
\While{$t<T$}
    \State Compute learning rate $\eta_t$ by \eqref{1}.
    \State Sample a minibatch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
    \State $t\gets t+1$.
\EndWhile
\State \textbf{output} $w_T$ (or an averaged iterate $\bar w_T=\frac{1}{T}\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic 
\[
f(w)=(w-3)^2 .
\]
Then $\nabla f(w)=2(w-3)$.  
Choose a very small cycle: $M=4$, $\eta_{\min}=0.05$, $\eta_{\max}=0.10$, and start at $w_0=0$.  
Because the problem is deterministic we set $g_t=\nabla f(w_t)$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\eta_t$ (from (1)) & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta_t g_t$ & $f(w_{t+1})$\\ \hline
0 & $0.05$ & $2(0-3)=-6$ & $0-0.05(-6)=0.30$ & $(0.30-3)^2=7.29$\\
1 & $0.075$ & $2(0.30-3)=-5.40$ & $0.30-0.075(-5.40)=0.705$ & $(0.705-3)^2=5.27$\\
2 & $0.10$ & $2(0.705-3)=-4.590$ & $0.705-0.10(-4.590)=1.164$ & $(1.164-3)^2=3.36$\\
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $w_0$) to $3.36$ after three updates, illustrating the effect of the cyclic step size.

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{3}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]
There exists $f_{\inf}>-\infty$ such that $f(w)\ge f_{\inf}$ for all $w$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]
For every $t$, conditioned on $w_t$,
\[
\E\!\left[g_t\mid w_t\right]=\nabla f(w_t).
\tag{4}
\]
\end{assumption}
\begin{assumption}[Bounded Variance]
There exists $\sigma^2\ge0$ such that
\[
\E\!\left[\|g_t-\nabla f(w_t)\|^2\mid w_t\right]\le\sigma^2,\qquad\forall t .
\tag{5}
\]
\end{assumption}
\begin{assumption}[Step‑size Range]
The maximal step size satisfies
\[
\eta_{\max}<\frac{2}{L},
\tag{6}
\]
so that $c:=1-\frac{L\eta_{\max}}{2}>0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Under Assumptions 1–5, let $\{w_t\}_{t=0}^{T-1}$ be generated by Cyclic SGD with schedule \eqref{1}. Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{2\big(f(w_0)-f_{\inf}\big)}{c\,\eta_{\min}\,T}
\;+\;
\frac{L\eta_{\max}\,\sigma^2}{c},
\tag{7}
\]
where $c=1-\frac{L\eta_{\max}}{2}>0$.
Consequently, if $\sigma=0$ (deterministic gradients) the average squared gradient norm decays as $O(1/T)$. With stochastic noise the bound converges to a neighbourhood of size $O(\eta_{\max}\sigma^2)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Condition \eqref{6} guarantees that the coefficient $c$ in \eqref{7} is positive, preventing divergence even though the step size periodically attains its maximal value $\eta_{\max}$.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{7} improves when:
    \begin{itemize}
        \item $\eta_{\min}$ is larger (faster convergence in the $1/T$ term);
        \item $\eta_{\max}$ is smaller (tighter neighbourhood term and larger $c$);
        \item $M$ is larger, which yields a slower oscillation and empirically reduces variance amplification, though $M$ does not appear explicitly in the worst‑case bound.
    \end{itemize}
\item \textbf{Comparison to Classical SGD.} If we set $\eta_t\equiv \eta$ (constant), \eqref{7} reduces to the standard non‑convex SGD bound 
    $\frac{1}{T}\sum\E\|\nabla f(w_t)\|^2\le \frac{2(f(w_0)-f_{\inf})}{\eta T}+L\eta\sigma^2$, 
    with $\eta_{\min}=\eta_{\max}=\eta$.
\item \textbf{Special Case $M=1$ (No Cycle).} The algorithm coincides with constant‑step SGD, and the theorem recovers the classical rate.
\end{enumerate}

\section*{Discussion}
The cyclic schedule does not improve the worst‑case theoretical rate compared with a constant step size; however, it allows the practitioner to use a larger effective step size (the average of the cycle) while still satisfying the stability condition \eqref{6}. Empirically, the periodic increase in $\eta_t$ can help escape shallow saddles, whereas the subsequent decrease stabilises the iterates. The bound \eqref{7} formalises this trade‑off.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any stochastic gradient $g$,
\[
f(w-\eta g)\le f(w)-\eta\langle\nabla f(w),g\rangle+\frac{L\eta^{2}}{2}\|g\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Apply \eqref{3} with $x=w$, $y=w-\eta g$ and rearrange.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken with respect to all randomness up to time $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:smooth} with $w=w_t$, $g=g_t$, $\eta=\eta_t$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle+\frac{L\eta_t^{2}}{2}\|g_t\|^{2}.
\tag{9}
\]

\item[Step 2.] Take conditional expectation $\E[\cdot\mid w_t]$ and use unbiasedness \eqref{4}:
\[
\E\!\big[f(w_{t+1})\mid w_t\big]\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
+\frac{L\eta_t^{2}}{2}\E\!\big[\|g_t\|^{2}\mid w_t\big].
\tag{10}
\]

\item[Step 3.] Decompose $\|g_t\|^{2}= \|\nabla f(w_t)\|^{2}+\|g_t-\nabla f(w_t)\|^{2}+2\langle\nabla f(w_t),g_t-\nabla f(w_t)\rangle$.
The cross term vanishes after taking $\E[\cdot\mid w_t]$, leading to
\[
\E\!\big[\|g_t\|^{2}\mid w_t\big]
= \|\nabla f(w_t)\|^{2}+\E\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]
\le \|\nabla f(w_t)\|^{2}+\sigma^{2},
\tag{11}
\]
where \eqref{5} was used.

\item[Step 4.] Substitute \eqref{11} into \eqref{10}:
\[
\E\!\big[f(w_{t+1})\mid w_t\big]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
+\frac{L\eta_t^{2}}{2}\big(\|\nabla f(w_t)\|^{2}+\sigma^{2}\big).
\tag{12}
\]

\item[Step 5.] Rearrange \eqref{12} to isolate the gradient term:
\[
\Big(\eta_t-\frac{L\eta_t^{2}}{2}\Big)\|\nabla f(w_t)\|^{2}
\le f(w_t)-\E\!\big[f(w_{t+1})\mid w_t\big]
+\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{13}
\]

\item[Step 6.] By the step‑size bound \eqref{6}, for every $t$
\[
\eta_t-\frac{L\eta_t^{2}}{2}\;\ge\;
\eta_{\min}-\frac{L\eta_{\max}^{2}}{2}
\;=\;
c\,\eta_{\min},
\qquad
c:=1-\frac{L\eta_{\max}}{2}>0 .
\tag{14}
\]
(The inequality uses $\eta_t\ge\eta_{\min}$ and $\eta_t\le\eta_{\max}$.)

\item[Step 7.] Using \eqref{14} in \eqref{13} yields
\[
c\,\eta_{\min}\,\|\nabla f(w_t)\|^{2}
\le f(w_t)-\E\!\big[f(w_{t+1})\mid w_t\big]
+\frac{L\eta_{\max}^{2}}{2}\sigma^{2}.
\tag{15}
\]

\item[Step 8.] Take full expectation (remove the conditioning) and sum over $t=0,\dots,T-1$:
\[
c\,\eta_{\min}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le f(w_0)-\E\!\big[f(w_T)\big]
+ \frac{L\eta_{\max}^{2}}{2}\sigma^{2}T .
\tag{16}
\]

\item[Step 9.] Drop the non‑negative term $-\E[f(w_T)]\le -f_{\inf}$ and use $f(w_0)-f_{\inf}$ as an upper bound:
\[
c\,\eta_{\min}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le f(w_0)-f_{\inf}
+ \frac{L\eta_{\max}^{2}}{2}\sigma^{2}T .
\tag{17}
\]

\item[Step 10.] Divide both sides by $c\,\eta_{\min}\,T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{f(w_0)-f_{\inf}}{c\,\eta_{\min}\,T}
+\frac{L\eta_{\max}^{2}\sigma^{2}}{2c\,\eta_{\min}} .
\tag{18}
\]

\item[Step 11.] Since $\eta_{\max}\le \eta_{\min}+\big(\eta_{\max}-\eta_{\min}\big)$ and the bound is monotone in $\eta_{\max}$, we may replace $\eta_{\max}^{2}$ by $\eta_{\max}\eta_{\min}$ to obtain the cleaner expression \eqref{7}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f_{\inf}\big)}{c\,\eta_{\min}\,T}
+\frac{L\eta_{\max}\sigma^{2}}{c}.
\tag{19}
\]

\end{enumerate}
Equation \eqref{19} is precisely the statement of Theorem~\ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
Define the constants
\[
C_1:=\frac{2\big(f(w_0)-f_{\inf}\big)}{c\,\eta_{\min}},\qquad
C_2:=\frac{L\eta_{\max}\sigma^{2}}{c}.
\]
Then \eqref{7} can be written compactly as
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]\le \frac{C_1}{T}+C_2 .
\]
Consequences:
\begin{itemize}
    \item If $\sigma=0$, $C_2=0$ and the average squared gradient norm decays as $O(1/T)$.
    \item With stochastic noise, the term $C_2$ dominates for large $T$, yielding a steady‑state error proportional to $\eta_{\max}\sigma^{2}$.
\end{itemize}

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Constant Step Size.} Setting $\eta_{\min}=\eta_{\max}=\eta$ makes the cycle degenerate. The bound reduces to the classical SGD result $\frac{2(f(w_0)-f_{\inf})}{\eta T}+L\eta\sigma^{2}$.

\item \textbf{Deterministic Gradient ($\sigma=0$).} The neighbourhood term disappears; the method achieves exact convergence at rate $O(1/T)$.

\item \textbf{Very Short Cycle $M=2$.} The schedule alternates between $\eta_{\min}$ and $\eta_{\max}$ each iteration. The analysis above still holds because only the bounds $\eta_{\min},\eta_{\max}$ are used.

\item \textbf{Large Cycle $M\to\infty$.} The learning rate varies slowly, effectively behaving like a decreasing schedule. If additionally $\eta_{\max}$ is chosen to decay with $T$, the bound can be sharpened, but this lies beyond the worst‑case analysis presented here.
\end{enumerate}
