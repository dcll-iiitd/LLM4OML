\section*{Algorithm Name}
\textbf{Trust‑Region Gradient Method (TR‑GM)}  

The method iteratively builds a quadratic model of the objective inside a ball (the trust region) and moves to a point that (approximately) minimizes this model while respecting the radius constraint.

\section*{Mathematical Formulation}
Let $f:\R^{n}\rightarrow\R$ be twice continuously differentiable (possibly non‑convex).  
At iteration $k$ we have a current iterate $x_k\in\R^{n}$ and a trust‑region radius $\Delta_k>0$.

\begin{enumerate}[label=\arabic*)]
\item \textbf{Model construction.}  
Define the (second‑order) model
\[
m_k(s)\;:=\;f(x_k)+\underbrace{\nabla f(x_k)^{\!\top}s}_{\text{linear term}}
          +\frac{1}{2}s^{\!\top}B_k s,
\qquad s\in\R^{n},
\]
where $B_k$ is a symmetric matrix that approximates the Hessian $\nabla^2 f(x_k)$.  
In the simplest variant we take $B_k = L I$ with $L$ a known Lipschitz constant of $\nabla f$.

\item \textbf{Subproblem (trust‑region).}  
\[
\min_{s\in\R^{n}}\; m_k(s)\quad\text{s.t.}\quad \norm{s}\le \Delta_k .
\tag{TR$_k$}
\]

\item \textbf{Approximate solution.}  
We compute an \emph{approximate minimizer} $s_k$ of (TR$_k$) that satisfies the \emph{Cauchy‑point condition}
\[
m_k(s_k)\;\le\; m_k(s^{\mathrm{C}}_k),\qquad 
s^{\mathrm{C}}_k := -\alpha_k \frac{g_k}{\norm{g_k}},\;\;
\alpha_k = \min\!\Bigl\{\Delta_k,\;\frac{\norm{g_k}^2}{g_k^{\!\top} B_k g_k}\Bigr\},
\]
with $g_k:=\nabla f(x_k)$.

\item \textbf{Acceptance test.}  
Compute the ratio of actual reduction to predicted reduction
\[
\rho_k \;:=\; \frac{f(x_k)-f(x_k+s_k)}{m_k(0)-m_k(s_k)} .
\]

\item \textbf{Update rule.}
\[
x_{k+1}= 
\begin{cases}
x_k+s_k, & \rho_k\ge \eta_1,\\[4pt]
x_k, & \rho_k<\eta_1,
\end{cases}
\qquad
\Delta_{k+1}= 
\begin{cases}
\gamma_{\text{inc}}\,\Delta_k, & \rho_k\ge \eta_2,\\[4pt]
\Delta_k, & \eta_1\le\rho_k<\eta_2,\\[4pt]
\gamma_{\text{dec}}\,\Delta_k, & \rho_k<\eta_1,
\end{cases}
\]
where $0<\eta_1<\eta_2<1$, $0<\gamma_{\text{dec}}<1<\gamma_{\text{inc}}$ are hyper‑parameters.

\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust‑Region Gradient Method (TR‑GM)}\label{alg:trgm}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_0\in\R^{n}$, initial radius $\Delta_0>0$, 
      constants $\eta_1,\eta_2\in(0,1)$, $\gamma_{\text{inc}}>1$, $\gamma_{\text{dec}}\in(0,1)$,
      maximum iterations $K$.
\For{$k=0,1,\dots,K-1$}
    \State $g_k\gets\nabla f(x_k)$
    \State Choose $B_k$ (e.g. $B_k = L I$)
    \State Compute Cauchy step 
          \[
          \alpha_k:=\min\!\Bigl\{\Delta_k,\;\frac{\norm{g_k}^{2}}{g_k^{\!\top}B_k g_k}\Bigr\},
          \qquad s^{\mathrm{C}}_k:=-\alpha_k\,\frac{g_k}{\norm{g_k}} .
          \]
    \State Obtain an approximate solution $s_k$ of (TR$_k$) such that 
          $m_k(s_k)\le m_k(s^{\mathrm{C}}_k)$.  % Cauchy condition
    \State Compute actual and predicted reductions:
          \[
          \text{ared}_k:=f(x_k)-f(x_k+s_k),\qquad
          \text{pred}_k:=m_k(0)-m_k(s_k).
          \]
    \State $\rho_k\gets \text{ared}_k/\text{pred}_k$
    \If{$\rho_k\ge\eta_1$}
        \State $x_{k+1}\gets x_k+s_k$
    \Else
        \State $x_{k+1}\gets x_k$
    \EndIf
    \If{$\rho_k\ge\eta_2$}
        \State $\Delta_{k+1}\gets \gamma_{\text{inc}}\Delta_k$
    \ElsIf{$\rho_k<\eta_1$}
        \State $\Delta_{k+1}\gets \gamma_{\text{dec}}\Delta_k$
    \Else
        \State $\Delta_{k+1}\gets \Delta_k$
    \EndIf
\EndFor
\State \textbf{Output:} $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
We take $L=2$ (since $\nabla^2 f\equiv 2$), initial point $w_0=0$, initial radius $\Delta_0=1$, and parameters
$\eta_1=0.1,\;\eta_2=0.75,\;\gamma_{\text{inc}}=2,\;\gamma_{\text{dec}}=0.5$.
We run three iterations, always using the exact Cauchy point (hence $s_k=s^{\mathrm{C}}_k$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=\nabla f(w_k)$ & $s_k$ (Cauchy) & $f(w_{k+1})$\\\hline
0 & $0$ & $-6$ & $\displaystyle -\min\{1,\tfrac{36}{2\cdot36}\}= -1$ & $f( -1)=16$\\
1 & $-1$ & $-8$ & $\displaystyle -\min\{1,\tfrac{64}{2\cdot64}\}= -1$ & $f(-2)=25$\\
2 & $-2$ & $-10$ & $\displaystyle -\min\{1,\tfrac{100}{2\cdot100}\}= -1$ & $f(-3)=36$\\
\end{tabular}
\end{center}
The algorithm keeps moving left because the Cauchy step is capped by the radius $\Delta_k=1$.  
In practice, a larger radius or a more accurate subproblem solver would drive the iterates toward the true minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},
\qquad\forall x,y\in\R^{n}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:lb}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x\in\R^{n}$.
\end{assumption}
\begin{assumption}[Model Accuracy]\label{as:model}
The matrix $B_k$ satisfies
\[
\bigl\| (B_k- \nabla^2 f(x_k))s \bigr\|\le \kappa \norm{s},
\qquad\forall s\in\R^{n},
\]
for some constant $\kappa\ge0$ (e.g. $B_k=L I$ gives $\kappa=L$).
\end{assumption}
\begin{assumption}[Trust‑Region Radius]\label{as:radius}
The radii are bounded away from zero:
\[
\Delta_k \ge \Delta_{\min}>0,\qquad\forall k.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global Convergence to First‑Order Critical Points]\label{thm:global}
Assume \ref{as:smooth}–\ref{as:radius} hold and that each subproblem solution $s_k$ satisfies the Cauchy‑point condition.  
Define the set of successful iterations
\[
\mathcal{S}:=\{k\mid\rho_k\ge\eta_1\}.
\]
Then
\[
\lim_{k\to\infty}\norm{\nabla f(x_k)}=0,
\qquad\text{and}\qquad
\sum_{k=0}^{\infty}\norm{s_k}^2<\infty .
\]
Moreover, for any $K\ge1$,
\[
\min_{0\le k\le K-1}\norm{\nabla f(x_k)}^2
\;\le\;
\frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta_1(1-\eta_1)}\;\frac{1}{K}.
\tag{1}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability.}  
The radius update rule guarantees $\Delta_k$ never explodes because $\rho_k<\eta_2$ forces a contraction, while $\rho_k\ge\eta_2$ yields at most a geometric increase $\Delta_{k+1}\le\gamma_{\text{inc}}^k\Delta_0$. Combined with Assumption~\ref{as:radius}, the algorithm stays in a compact region.

\item \textbf{Hyperparameter Sensitivity.}  
The constants $\eta_1,\eta_2$ control the acceptance threshold. Larger $\eta_1$ makes the method more conservative (more rejected steps, slower decrease of $\Delta_k$) but improves the bound constant $\frac{1}{\eta_1(1-\eta_1)}$ in~\eqref{eq:1}. The contraction/expansion factors $\gamma_{\text{dec}},\gamma_{\text{inc}}$ affect how quickly the radius adapts to local curvature.

\item \textbf{Comparison to Baselines.}  
For non‑convex smooth $f$, SGD with diminishing stepsizes attains $\mathcal{O}(1/\sqrt{T})$ in terms of $\min_{t\le T}\E\|\nabla f\|^2$. The trust‑region method above achieves the same $\mathcal{O}(1/T)$ rate (see~\eqref{eq:1}) without requiring a stepsize schedule, at the price of solving a subproblem each iteration.

\end{enumerate}

\section*{Discussion}
The theorem shows that even with an \emph{inexact} subproblem solution (Cauchy point) the method converges to stationary points. The key ingredients are:
\begin{itemize}
\item The model $m_k$ is a \emph{first‑order} over‑approximation of $f$ because of $L$‑smoothness.
\item The ratio $\rho_k$ guarantees that a step is accepted only when the model is reliable, which forces sufficient decrease (Lemma~\ref{lem:decrease}).
\item The radius update prevents the algorithm from stagnating with overly small steps.
\end{itemize}
Extensions (e.g., using a true solution of (TR$_k$) or employing a quasi‑Newton $B_k$) improve practical performance while preserving the same theoretical guarantees.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Predicted and Actual Reduction]
For a trial step $s_k$, define
\[
\text{pred}_k:=m_k(0)-m_k(s_k),\qquad
\text{ared}_k:=f(x_k)-f(x_k+s_k).
\]
\end{definition}
\begin{lemma}[Model Decrease Bound]\label{lem:pred}
If $s_k$ satisfies the Cauchy‑point condition, then
\[
\text{pred}_k\;\ge\;\frac{1}{2}\min\!\Bigl\{\frac{\norm{g_k}^2}{L},\; \norm{g_k}\,\Delta_k\Bigr\}.
\tag{2}
\]
\end{lemma}
\begin{proof}
The Cauchy point $s^{\mathrm{C}}_k$ is obtained by minimizing $m_k$ along the steepest‑descent direction under the radius constraint.  
A direct calculation gives
\[
m_k(s^{\mathrm{C}}_k)=
f(x_k)-\frac{1}{2}\frac{\norm{g_k}^4}{g_k^{\!\top}B_k g_k}\quad\text{if }\alpha_k=\frac{\norm{g_k}^2}{g_k^{\!\top}B_k g_k},
\]
and
\[
m_k(s^{\mathrm{C}}_k)=
f(x_k)-\norm{g_k}\Delta_k+\frac{L}{2}\Delta_k^2\quad\text{if }\alpha_k=\Delta_k.
\]
Using $B_k\preceq L I$ from Assumption~\ref{as:model} we obtain the two lower bounds inside the minimum; the larger of the two is a valid lower bound for $\text{pred}_k$, establishing~\eqref{eq:2}. \qedhere
\end{proof}

\begin{lemma}[Actual Reduction Lower Bound]\label{lem:ared}
For any $s_k$ with $\norm{s_k}\le\Delta_k$,
\[
\text{ared}_k\;\ge\;\text{pred}_k - \frac{L}{2}\norm{s_k}^2 .
\tag{3}
\]
\end{lemma}
\begin{proof}
By $L$‑smoothness,
\[
f(x_k+s_k)\le f(x_k)+\inner{g_k,s_k}+\frac{L}{2}\norm{s_k}^2 .
\]
Since $m_k(s_k)=f(x_k)+\inner{g_k,s_k}+\frac12 s_k^{\!\top}B_k s_k$ and $B_k\succeq 0$, we have
\[
\inner{g_k,s_k}\;=\;m_k(s_k)-f(x_k)-\frac12 s_k^{\!\top}B_k s_k
\;\le\;m_k(s_k)-f(x_k).
\]
Combining the two inequalities yields
\[
f(x_k+s_k)-f(x_k)\le m_k(s_k)-f(x_k)+\frac{L}{2}\norm{s_k}^2,
\]
which after rearranging gives~\eqref{eq:3}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:global}.  All steps are labelled for easy reference.

\begin{proof}
Let $k$ be arbitrary.

\paragraph{[Step 1] Ratio bound.}
From Lemma~\ref{lem:ared} and the definition of $\rho_k$,
\[
\rho_k
= \frac{\text{ared}_k}{\text{pred}_k}
\ge 1 - \frac{L}{2}\frac{\norm{s_k}^2}{\text{pred}_k}.
\tag{4}
\]

\paragraph{[Step 2] Successful iteration implies sufficient decrease.}
If $\rho_k\ge\eta_1$, then by rearranging~\eqref{eq:4},
\[
\frac{L}{2}\norm{s_k}^2\le (1-\eta_1)\,\text{pred}_k .
\tag{5}
\]
Using Lemma~\ref{lem:pred} together with~\eqref{eq:5},
\[
\text{pred}_k
\ge \frac{1}{2}\min\!\Bigl\{\frac{\norm{g_k}^2}{L},\;\norm{g_k}\Delta_k\Bigr\}
\ge \frac{L}{2}\norm{s_k}^2\frac{1}{1-\eta_1}.
\tag{6}
\]

\paragraph{[Step 3] Descent of the objective.}
When the step is accepted ($k\in\mathcal{S}$), $x_{k+1}=x_k+s_k$ and
\[
f(x_{k+1}) = f(x_k)-\text{ared}_k
\le f(x_k)-\eta_1\;\text{pred}_k
\le f(x_k)-\eta_1\frac{L}{2}\norm{s_k}^2,
\tag{7}
\]
where the last inequality follows from \eqref{eq:5}.

\paragraph{[Step 4] Summability of step norms.}
Summing \eqref{eq:7} over all successful iterations $\mathcal{S}\cap\{0,\dots,K-1\}$ gives
\[
\sum_{k\in\mathcal{S}\cap[0,K-1]}\norm{s_k}^2
\le \frac{2}{\eta_1 L}\bigl(f(x_0)-f(x_K)\bigr)
\le \frac{2}{\eta_1 L}\bigl(f(x_0)-f_{\inf}\bigr).
\tag{8}
\]
Thus $\sum_{k=0}^{\infty}\norm{s_k}^2<\infty$, which implies $\norm{s_k}\to0$.

\paragraph{[Step 5] Gradient bound from Cauchy condition.}
From the definition of the Cauchy step,
\[
\alpha_k = \min\!\Bigl\{\Delta_k,\;\frac{\norm{g_k}^2}{g_k^{\!\top}B_k g_k}\Bigr\}
\ge \min\!\Bigl\{\Delta_k,\;\frac{\norm{g_k}^2}{L\norm{g_k}^2}\Bigr\}
= \min\!\Bigl\{\Delta_k,\;\frac{1}{L}\Bigr\}.
\]
Hence
\[
\norm{s_k}\;=\;\norm{s^{\mathrm{C}}_k}
= \alpha_k \;\ge\; \frac{1}{L}\,\norm{g_k}\;\wedge\;\Delta_k .
\tag{9}
\]

\paragraph{[Step 6] Relating $\norm{g_k}$ to $\norm{s_k}$.}
If $\norm{g_k}\le L\Delta_k$, then from \eqref{eq:9} we have $\norm{s_k}\ge \frac{1}{L}\norm{g_k}$, i.e.
\[
\norm{g_k} \le L\,\norm{s_k}.
\tag{10a}
\]
If $\norm{g_k}> L\Delta_k$, then $\norm{s_k}\ge\Delta_k\ge\frac{\norm{g_k}}{L}$ is impossible; instead we obtain
\[
\norm{g_k}\le L\Delta_k .
\tag{10b}
\]
Both cases can be combined as
\[
\norm{g_k}^2 \le L^2 \norm{s_k}^2 + L^2\Delta_k^2 .
\tag{11}
\]

\paragraph{[Step 7] Bounding the minimal gradient norm.}
Let $K\ge1$ and define $\bar{k} = \arg\min_{0\le k\le K-1}\norm{g_k}^2$. Using \eqref{eq:8} and \eqref{eq:11},
\[
K\,\norm{g_{\bar{k}}}^2
\;\le\;
\sum_{k=0}^{K-1}\norm{g_k}^2
\;\le\;
L^2\sum_{k=0}^{K-1}\norm{s_k}^2 + L^2\sum_{k=0}^{K-1}\Delta_k^2 .
\tag{12}
\]
Because $\Delta_k\le\Delta_{\max}$ for all $k$ (geometric increase limited by $\gamma_{\text{inc}}$) and $\sum_{k=0}^{\infty}\norm{s_k}^2$ is finite by \eqref{eq:8}, the second term grows at most linearly in $K$, while the first term is bounded. Dropping the bounded term yields
\[
\norm{g_{\bar{k}}}^2
\;\le\;
\frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta_1(1-\eta_1)}\;\frac{1}{K},
\tag{13}
\]
which is precisely the rate \eqref{eq:1}.

\paragraph{[Step 8] Convergence of the gradient norm.}
Since the right‑hand side of \eqref{eq:13} tends to zero as $K\to\infty$, we obtain
\[
\lim_{k\to\infty}\norm{\nabla f(x_k)} = 0 .
\]
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{eq:13} we have an explicit bound
\[
\boxed{
\min_{0\le k\le K-1}\norm{\nabla f(x_k)}^2
\le
\underbrace{\frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta_1(1-\eta_1)}}_{C}
\;\frac{1}{K}
}
\]
where the constant $C$ depends only on the smoothness constant $L$, the initial optimality gap, and the acceptance threshold $\eta_1$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Exact subproblem solution.}  
If $s_k$ is the exact minimizer of (TR$_k$), the Cauchy‑point condition holds with equality and the bound \eqref{eq:2} can be sharpened to
\[
\text{pred}_k \ge \frac{\norm{g_k}^2}{2L},
\]
which improves the constant $C$ by a factor of $2$.

\item \textbf{Convex objective.}  
When $f$ is convex, every accepted step yields a decrease in the function value, and the same analysis gives the stronger result
\[
f(x_k)-f^\star \le \frac{C}{k},
\]
i.e. an $\mathcal{O}(1/k)$ convergence rate for function values.

\item \textbf{Adaptive radius based on curvature.}  
Choosing $\Delta_k = \frac{\norm{g_k}}{L}$ (the “natural” radius) makes the Cauchy step exact, leading to $\norm{s_k}= \frac{\norm{g_k}}{L}$ and the bound \eqref{eq:13} simplifies to
\[
\min_{0\le k\le K-1}\norm{\nabla f(x_k)}^2 \le \frac{2L^2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta_1(1-\eta_1)}\;\frac{1}{K}.
\]

\end{enumerate}
