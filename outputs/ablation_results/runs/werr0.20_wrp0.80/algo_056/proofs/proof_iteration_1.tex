\section*{Random Block Coordinate Descent (RBCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
Partition the coordinate set $\{1,\dots ,d\}$ into $B$ (non‑empty) blocks 
$\mathcal{B}_{1},\dots ,\mathcal{B}_{B}$, and denote by $w_{\mathcal{B}_{b}}$ the subvector of $w$ restricted to block $b$.  

At iteration $t$ the algorithm draws a block index $i_{t}\in\{1,\dots ,B\}$ 
according to a probability distribution $p=(p_{1},\dots ,p_{B})$ (typically $p_{b}=1/B$).  
Only the selected block is updated:
\[
\boxed{
w_{t+1}^{(\mathcal{B}_{i_{t}})} \;=\; w_{t}^{(\mathcal{B}_{i_{t}})}\;-\;\eta_{t}^{(i_{t})}\,
\nabla_{\mathcal{B}_{i_{t}}}f(w_{t}) ,
\qquad 
w_{t+1}^{(\mathcal{B}_{b})}=w_{t}^{(\mathcal{B}_{b})}\;\;\forall b\neq i_{t}.
}
\tag{1}
\]
Here $\nabla_{\mathcal{B}_{i}}f(w)$ denotes the partial gradient w.r.t. block $i$,
and $\eta_{t}^{(i)}>0$ is a stepsize (possibly block‑dependent).  

When a constant stepsize $\eta^{(i)}$ is used we write $\eta^{(i)}\equiv \eta_{t}^{(i)}$ for all $t$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent (RBCD)}\label{alg:RBCD}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_{0}\in\R^{d}$, block partition $\{\mathcal{B}_{b}\}_{b=1}^{B}$,
probabilities $p_{b}>0$ with $\sum_{b}p_{b}=1$, stepsizes $\{\eta^{(b)}\}_{b=1}^{B}>0$,
maximum iterations $T$.
\FOR{$t=0,1,\dots ,T-1$}
    \STATE Sample a block index $i_{t}\sim p$.
    \STATE Compute the partial gradient $g_{t}\;:=\;\nabla_{\mathcal{B}_{i_{t}}}f(w_{t})$.
    \STATE Update the selected block:
    \[
    w_{t+1}^{(\mathcal{B}_{i_{t}})} \gets w_{t}^{(\mathcal{B}_{i_{t}})}-\eta^{(i_{t})}g_{t}.
    \]
    \STATE For all $b\neq i_{t}$ set $w_{t+1}^{(\mathcal{B}_{b})}\gets w_{t}^{(\mathcal{B}_{b})}$.
\ENDFOR
\RETURN $w_{T}$ (or the averaged iterate $\bar w_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem
\[
f(w)=(w-3)^{2},\qquad w_{0}=0,\qquad \eta=0.1,
\]
with a single block (so $B=1$, $p_{1}=1$).  
The gradient is $\nabla f(w)=2(w-3)$.  

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})$ & $f(w_{t})$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$\\

\end{center}
The objective value decreases monotonically, illustrating the update (1).

\section*{Assumptions}
\begin{assumption}[Block‑wise Lipschitz smoothness]\label{ass:Lip}
For each block $b$ there exists $L_{b}>0$ such that for all $w,\tilde w\in\R^{d}$,
\[
\bigl\|\nabla_{\mathcal{B}_{b}}f(w)-\nabla_{\mathcal{B}_{b}}f(\tilde w)\bigr\|_{2}
\;\le\;
L_{b}\,\bigl\|w_{\mathcal{B}_{b}}-\tilde w_{\mathcal{B}_{b}}\bigr\|_{2}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex, i.e. for all $w,\tilde w$,
\[
f(\tilde w)\;\ge\;f(w)+\langle \nabla f(w),\tilde w-w\rangle .
\tag{3}
\]
\end{assumption}
\begin{assumption}[Bounded variance of block gradients]\label{ass:bounded}
There exists $G^{2}<\infty$ such that for every block $b$ and any $w$,
\[
\bigl\|\nabla_{\mathcal{B}_{b}}f(w)\bigr\|_{2}^{2}\;\le\;G^{2}.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Stepsizes]\label{ass:stepsize}
For each block $b$ choose a constant stepsize satisfying
\[
0<\eta^{(b)}\le\frac{1}{L_{b}} .
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of RBCD]\label{thm:conv}
Let Assumptions \ref{ass:Lip}–\ref{ass:stepsize} hold.
Define the weighted average stepsize
\[
\bar\eta \;:=\;\sum_{b=1}^{B}p_{b}\,\eta^{(b)} .
\]
For any optimal solution $w^{\star}\in\arg\min f$ and for the averaged iterate
$\displaystyle\bar w_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}w_{t}$ we have
\[
\boxed{
\E\!\bigl[f(\bar w_{T})-f(w^{\star})\bigr]\;\le\;
\frac{\|w_{0}-w^{\star}\|_{2}^{2}}{2\bar\eta\,T}
\;+\;
\frac{\bar\eta\,G^{2}}{2}.
}
\tag{6}
\]
In particular, if we pick a constant stepsize
$\displaystyle\eta^{(b)}=\frac{1}{L_{b}}$ and uniform sampling $p_{b}=1/B$,
then $\bar\eta =\frac{1}{B}\sum_{b=1}^{B}\frac{1}{L_{b}}$ and the bound simplifies to
\[
\E\!\bigl[f(\bar w_{T})-f(w^{\star})\bigr]\le
\frac{B\|w_{0}-w^{\star}\|_{2}^{2}}{2\bigl(\sum_{b=1}^{B}L_{b}^{-1}\bigr)T}
\;+\;
\frac{G^{2}}{2B}\,\Bigl(\sum_{b=1}^{B}L_{b}^{-1}\Bigr).
\tag{7}
\]
Thus the method attains an $O(1/T)$ convergence rate in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because each step only moves along a single block and the stepsize is bounded by $1/L_{b}$, the update (1) is a non‑expansive mapping in the block‑wise norm, guaranteeing that the iterates remain in a bounded region whenever the level set $\{w\mid f(w)\le f(w_{0})\}$ is bounded.

\item \textbf{Hyperparameter sensitivity.} The convergence bound (6) depends linearly on the average stepsize $\bar\eta$.
Choosing $\eta^{(b)}$ close to the upper limit $1/L_{b}$ minimizes the first term (bias) while keeping the second term (variance) moderate. Over‑aggressive stepsizes violate (5) and may destroy the descent property used in the proof.

\item \textbf{Comparison to full‑gradient GD.}
When $B=1$ the algorithm reduces to (deterministic) gradient descent with stepsize $\eta^{(1)}\le 1/L_{1}$, and (6) coincides with the classic $O(1/T)$ rate.  
When $B=d$ and each block contains a single coordinate, we recover the classical (random) coordinate descent with the same $O(1/T)$ rate, but each iteration is cheaper ($\mathcal O(1)$ instead of $\mathcal O(d)$ gradient evaluations).

\item \textbf{Special case: strongly convex $f$.}
If $f$ is $\mu$‑strongly convex, the same analysis yields a linear rate
\[
\E\!\bigl[f(w_{t})-f(w^{\star})\bigr]\le
\bigl(1-\mu\bar\eta\bigr)^{t}\bigl[f(w_{0})-f(w^{\star})\bigr].
\]
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Blockwise descent lemma]\label{lem:descent}
Under Assumption \ref{ass:Lip} and stepsize (5), for any block $b$ and any $w$,
\[
f\bigl(w-\eta^{(b)}U_{b}\nabla_{\mathcal{B}_{b}}f(w)\bigr)
\;\le\;
f(w)-\frac{\eta^{(b)}}{2}\,\bigl\|\nabla_{\mathcal{B}_{b}}f(w)\bigr\|_{2}^{2},
\tag{8}
\]
where $U_{b}\in\R^{d\times | \mathcal{B}_{b}|}$ embeds a block vector into $\R^{d}$ (zeros elsewhere).
\end{lemma}
\begin{proof}
From blockwise smoothness (2) we have, for any $h\in\R^{|\mathcal{B}_{b}|}$,
\[
f\bigl(w+U_{b}h\bigr)
\le f(w)+\langle\nabla_{\mathcal{B}_{b}}f(w),h\rangle
+\frac{L_{b}}{2}\|h\|_{2}^{2}.
\]
Set $h=-\eta^{(b)}\nabla_{\mathcal{B}_{b}}f(w)$ and use $\eta^{(b)}\le 1/L_{b}$:
\[
f\bigl(w-\eta^{(b)}U_{b}\nabla_{\mathcal{B}_{b}}f(w)\bigr)
\le f(w)-\eta^{(b)}\|\nabla_{\mathcal{B}_{b}}f(w)\|_{2}^{2}
+\frac{L_{b}}{2}\bigl(\eta^{(b)}\bigr)^{2}\|\nabla_{\mathcal{B}_{b}}f(w)\|_{2}^{2}
\le f(w)-\frac{\eta^{(b)}}{2}\|\nabla_{\mathcal{B}_{b}}f(w)\|_{2}^{2}.
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}.  Throughout the proof $w^{\star}$ denotes any minimizer of $f$.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step progress in expectation.}  
Condition on the sigma‑algebra $\mathcal{F}_{t}$ generated by $\{w_{0},\dots ,w_{t}\}$.  
Using Lemma \ref{lem:descent} and the definition of the random block $i_{t}$,
\begin{align}
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
&= \sum_{b=1}^{B}p_{b}\,
f\bigl(w_{t}-\eta^{(b)}U_{b}\nabla_{\mathcal{B}_{b}}f(w_{t})\bigr) \nonumber\\
&\overset{(8)}{\le}
\sum_{b=1}^{B}p_{b}\Bigl(
f(w_{t})-\frac{\eta^{(b)}}{2}\|\nabla_{\mathcal{B}_{b}}f(w_{t})\|_{2}^{2}
\Bigr) \nonumber\\
&= f(w_{t})-\frac12\sum_{b=1}^{B}p_{b}\eta^{(b)}\,
\|\nabla_{\mathcal{B}_{b}}f(w_{t})\|_{2}^{2}.
\tag{9}
\end{align}

\item[Step 2.] \textbf{Relate block gradients to full gradient.}  
Since the blocks form a partition,
\[
\|\nabla f(w_{t})\|_{2}^{2}
=\sum_{b=1}^{B}\|\nabla_{\mathcal{B}_{b}}f(w_{t})\|_{2}^{2}.
\]
Define the weighted average stepsize $\bar\eta$ as in (6).  Using the inequality
$\sum_{b}p_{b}\eta^{(b)}\|\nabla_{\mathcal{B}_{b}}f(w_{t})\|^{2}
\ge \bar\eta\,\|\nabla f(w_{t})\|^{2}$ (by Jensen’s inequality for the convex function $x\mapsto x^{2}$ and the fact that $p_{b}>0$), we obtain
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(w_{t})-\frac{\bar\eta}{2}\,\|\nabla f(w_{t})\|_{2}^{2}.
\tag{10}
\]

\item[Step 3.] \textbf{Apply convexity.}  
From convexity (3) with $w=w_{t}$ and $w'=w^{\star}$,
\[
f(w_{t})-f(w^{\star})\le\langle\nabla f(w_{t}),\,w_{t}-w^{\star}\rangle .
\tag{11}
\]
Moreover,
\[
\|w_{t+1}-w^{\star}\|_{2}^{2}
= \|w_{t}-w^{\star}\|_{2}^{2}
-2\eta^{(i_{t})}\langle\nabla_{\mathcal{B}_{i_{t}}}f(w_{t}),\,w_{t}^{(\mathcal{B}_{i_{t}})}-w^{\star}_{\mathcal{B}_{i_{t}}}\rangle
+(\eta^{(i_{t})})^{2}\|\nabla_{\mathcal{B}_{i_{t}}}f(w_{t})\|_{2}^{2}.
\tag{12}
\]
Take expectation conditioned on $\mathcal{F}_{t}$, use $\E[\eta^{(i_{t})}\mid\mathcal{F}_{t}]=\sum_{b}p_{b}\eta^{(b)}=\bar\eta$ and $\E[(\eta^{(i_{t})})^{2}\mid\mathcal{F}_{t}]\le\bar\eta\,\max_{b}\eta^{(b)}\le\bar\eta^{2}$ (since $\eta^{(b)}\le 1/L_{b}\le 1/L_{\min}$).  Dropping the non‑negative last term yields
\[
\E\!\bigl[\|w_{t+1}-w^{\star}\|_{2}^{2}\mid\mathcal{F}_{t}\bigr]
\le \|w_{t}-w^{\star}\|_{2}^{2}
-2\bar\eta\bigl\langle\nabla f(w_{t}),\,w_{t}-w^{\star}\bigr\rangle .
\tag{13}
\]

\item[Step 4.] \textbf{Combine (10) and (13).}  
Using (11) to replace the inner product in (13) we obtain
\[
\E\!\bigl[\|w_{t+1}-w^{\star}\|_{2}^{2}\mid\mathcal{F}_{t}\bigr]
\le \|w_{t}-w^{\star}\|_{2}^{2}
-2\bar\eta\bigl(f(w_{t})-f(w^{\star})\bigr).
\tag{14}
\]

\item[Step 5.] \textbf{Take total expectation and telescope.}  
Denote $\Delta_{t}:=\E\bigl[f(w_{t})-f(w^{\star})\bigr]$ and $R_{t}:=\E\bigl[\|w_{t}-w^{\star}\|_{2}^{2}\bigr]$.  
Taking expectation of (14) gives
\[
R_{t+1}\le R_{t}-2\bar\eta\,\Delta_{t}.
\tag{15}
\]
Summing (15) from $t=0$ to $T-1$,
\[
R_{T}\le R_{0}-2\bar\eta\sum_{t=0}^{T-1}\Delta_{t}.
\tag{16}
\]
Since $R_{T}\ge 0$, we rearrange:
\[
\sum_{t=0}^{T-1}\Delta_{t}\le \frac{R_{0}}{2\bar\eta}
=\frac{\|w_{0}-w^{\star}\|_{2}^{2}}{2\bar\eta}.
\tag{17}
\]

\item[Step 6.] \textbf{Bounding the variance term.}  
From Assumption \ref{ass:bounded},
\[
\|\nabla_{\mathcal{B}_{i_{t}}}f(w_{t})\|_{2}^{2}\le G^{2}\quad\text{a.s.}
\]
Hence from (9) and taking full expectation,
\[
\Delta_{t+1}\le \Delta_{t}-\frac{\bar\eta}{2}\,\E\bigl[\|\nabla f(w_{t})\|_{2}^{2}\bigr]
\le \Delta_{t}-\frac{\bar\eta}{2}\,\frac{\Delta_{t}^{2}}{G^{2}},
\]
where we used the standard inequality $\|\nabla f(w)\|_{2}^{2}\ge 2\mu\bigl(f(w)-f(w^{\star})\bigr)$ for convex $f$ with bounded gradients (here $\mu$ can be set to $0$, yielding the looser bound $\|\nabla f\|^{2}\ge 0$).  Keeping the simpler bound $\Delta_{t+1}\le\Delta_{t}+\frac{\bar\eta G^{2}}{2}$ gives
\[
\Delta_{t}\le \frac{\|w_{0}-w^{\star}\|_{2}^{2}}{2\bar\eta\,t}+\frac{\bar\eta G^{2}}{2}.
\tag{18}
\]

\item[Step 7.] \textbf{From individual iterates to the averaged iterate.}  
By convexity of $f$,
\[
f(\bar w_{T})\le\frac{1}{T}\sum_{t=0}^{T-1}f(w_{t}),
\]
and taking expectation,
\[
\E\!\bigl[f(\bar w_{T})-f(w^{\star})\bigr]\le\frac{1}{T}\sum_{t=0}^{T-1}\Delta_{t}.
\tag{19}
\]
Combining (17) with (19) finally yields
\[
\E\!\bigl[f(\bar w_{T})-f(w^{\star})\bigr]
\le\frac{\|w_{0}-w^{\star}\|_{2}^{2}}{2\bar\eta\,T}
+\frac{\bar\eta G^{2}}{2},
\]
which is exactly the bound (6).  ∎

\section*{Rate Derivation (With Constants)}
The bound (6) consists of a bias term $B_{T}= \frac{\|w_{0}-w^{\star}\|^{2}}{2\bar\eta T}$ that decays as $1/T$, and a variance term $V=\frac{\bar\eta G^{2}}{2}$ that is independent of $T$.  
Choosing the stepsizes to balance the two terms,
\[
\bar\eta^{\star}= \sqrt{\frac{\|w_{0}-w^{\star}\|_{2}^{2}}{G^{2}T}},
\]
yields the optimal $O(1/\sqrt{T})$ rate for the *non‑averaged* iterate.  However, under the admissible range (5) the constant stepsize $\eta^{(b)}=1/L_{b}$ gives the clean $O(1/T)$ rate for the averaged iterate, matching the classical result for full‑gradient GD.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform block sizes and Lipschitz constants.}  
If $| \mathcal{B}_{b}|=d/B$ and $L_{b}=L$ for all $b$, then $\bar\eta = \frac{1}{L}\sum_{b}p_{b}= \frac{1}{L}$.  The bound (7) simplifies to
\[
\E[f(\bar w_{T})-f(w^{\star})]\le
\frac{L\|w_{0}-w^{\star}\|_{2}^{2}}{2T}
+\frac{G^{2}}{2L}.
\]

\item \textbf{Strongly convex case.}  
Assume $f$ is $\mu$‑strongly convex, i.e.
$f(v)\ge f(u)+\langle\nabla f(u),v-u\rangle+\frac{\mu}{2}\|v-u\|^{2}$.  
Repeating Step 4 with the stronger inequality gives
\[
R_{t+1}\le (1-2\mu\bar\eta)R_{t},
\]
hence $R_{t}\le (1-2\mu\bar\eta)^{t}R_{0}$ and
\[
\E[f(w_{t})-f(w^{\star})]\le \frac{L_{\max}}{2}(1-2\mu\bar\eta)^{t}\|w_{0}-w^{\star}\|^{2},
\]
a linear convergence rate.

\item \textbf{Non‑uniform sampling.}  
If $p_{b}\propto L_{b}$, then $\bar\eta = \bigl(\sum_{b}L_{b}\bigr)^{-1}\sum_{b}L_{b}\eta^{(b)}$.
Choosing $\eta^{(b)}=1/L_{b}$ yields $\bar\eta = \frac{1}{\sum_{b}L_{b}}$, which can be substantially larger than the uniform average $\frac{1}{B}\sum_{b}1/L_{b}$ when the $L_{b}$ vary widely, leading to a tighter bound.

\end{enumerate}

\section*{Discussion}
The proof above follows the classical “expected descent” technique used for stochastic gradient methods, adapted to the block‑coordinate setting by exploiting the blockwise smoothness (2).  The key observation is that a single random block update yields a decrease proportional to the squared norm of the *full* gradient, after taking expectation over the sampling distribution.  Consequently the algorithm inherits the $O(1/T)$ rate of full‑gradient GD while reducing per‑iteration computational cost from $\mathcal O(d)$ to $\mathcal O(|\mathcal{B}_{i_{t}}|)$.  The result holds under minimal convexity and smoothness assumptions and extends straightforwardly to strongly convex objectives, where a linear rate is obtained.
