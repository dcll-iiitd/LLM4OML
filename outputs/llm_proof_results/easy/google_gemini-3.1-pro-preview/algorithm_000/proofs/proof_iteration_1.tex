\section*{Algorithm Name}
Minibatch Stochastic Gradient Descent (Minibatch SGD) with Fixed Step Size.

\section*{Mathematical Formulation}
The objective is to minimize a function $f(w) = \mathbb{E}_{z \sim \mathcal{D}}[f(w; z)]$, where $w \in \mathbb{R}^d$ represents the model parameters and $z$ represents a data sample drawn from the distribution $\mathcal{D}$. 

At each iteration $t$, the algorithm draws a random minibatch of samples $B_t = \{z_{t,1}, z_{t,2}, \dots, z_{t,b}\}$ of size $b = |B_t|$. The stochastic gradient is computed as the empirical average of the gradients over the minibatch:
\begin{equation}
    g_t = \frac{1}{b} \sum_{i=1}^b \nabla f(w_t; z_{t,i})
\end{equation}
The parameters are then updated by taking a step in the negative direction of this stochastic gradient using a fixed learning rate (step size) $\eta > 0$:
\begin{equation}
    w_{t+1} = w_t - \eta g_t
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Minibatch Stochastic Gradient Descent (Fixed Step Size)}
\begin{algorithmic}[1]
\REQUIRE Objective function $f$, initial weights $w_0 \in \mathbb{R}^d$, learning rate $\eta > 0$, batch size $b$, total iterations $T$
\FOR{$t = 0, 1, \dots, T-1$}
    \STATE Sample a minibatch $B_t = \{z_{t,1}, \dots, z_{t,b}\}$ independently from data distribution $\mathcal{D}$
    \STATE Compute the minibatch gradient: $g_t = \frac{1}{b} \sum_{i=1}^b \nabla f(w_t; z_{t,i})$
    \STATE Update parameters: $w_{t+1} = w_t - \eta g_t$
\ENDFOR
\RETURN $\bar{w}_T$ chosen uniformly at random from $\{w_0, \dots, w_{T-1}\}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the deterministic objective function $f(w) = (w-3)^2$. We will perform 3 iterations of the algorithm with $w_0 = 0$, a fixed step size $\eta = 0.1$, and a batch size of $b=1$ (since the function is deterministic, the stochastic gradient equals the exact gradient). The exact gradient is $\nabla f(w) = 2(w-3)$.

\textbf{Initialization:} $w_0 = 0$, $f(w_0) = (0-3)^2 = 9$.

\textbf{Iteration 1 ($t=0$):}
\begin{itemize}
    \item Compute gradient: $g_0 = \nabla f(w_0) = 2(0-3) = -6$.
    \item Update parameter: $w_1 = w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6$.
    \item Objective value: $f(w_1) = (0.6-3)^2 = (-2.4)^2 = 5.76$.
\end{itemize}

\textbf{Iteration 2 ($t=1$):}
\begin{itemize}
    \item Compute gradient: $g_1 = \nabla f(w_1) = 2(0.6-3) = -4.8$.
    \item Update parameter: $w_2 = w_1 - \eta g_1 = 0.6 - 0.1(-4.8) = 1.08$.
    \item Objective value: $f(w_2) = (1.08-3)^2 = (-1.92)^2 = 3.6864$.
\end{itemize}

\textbf{Iteration 3 ($t=2$):}
\begin{itemize}
    \item Compute gradient: $g_2 = \nabla f(w_2) = 2(1.08-3) = -3.84$.
    \item Update parameter: $w_3 = w_2 - \eta g_2 = 1.08 - 0.1(-3.84) = 1.464$.
    \item Objective value: $f(w_3) = (1.464-3)^2 = (-1.536)^2 = 2.359296$.
\end{itemize}
The objective value successfully decreases monotonically.

\section*{Assumptions}
To establish the convergence theory for Minibatch SGD on non-convex objectives, we make the following standard assumptions:

\textbf{Assumption 1 (Lower Boundedness):} The objective function $f(w)$ is bounded from below by some scalar $f^* > -\infty$. That is, $f(w) \geq f^*$ for all $w \in \mathbb{R}^d$.

\textbf{Assumption 2 ($L$-Smoothness):} The objective function $f$ is continuously differentiable, and its gradient is Lipschitz continuous with constant $L > 0$. For all $x, y \in \mathbb{R}^d$:
\begin{equation}
    \|\nabla f(x) - \nabla f(y)\| \leq L \|x - y\|
\end{equation}

\textbf{Assumption 3 (Unbiased Gradient):} The stochastic gradient computed on a single sample is an unbiased estimator of the true full gradient. For any $w$ and a random sample $z$:
\begin{equation}
    \mathbb{E}_z[\nabla f(w; z)] = \nabla f(w)
\end{equation}
By linearity of expectation, the minibatch gradient $g_t$ is also unbiased: $\mathbb{E}[g_t | w_t] = \nabla f(w_t)$.

\textbf{Assumption 4 (Bounded Variance):} The variance of the stochastic gradient evaluated on a single sample $z$ is bounded by $\sigma^2$:
\begin{equation}
    \mathbb{E}_z[\|\nabla f(w; z) - \nabla f(w)\|^2] \leq \sigma^2
\end{equation}
For a minibatch of size $b$ drawn with replacement (or independently), the variance of the minibatch gradient $g_t$ is bounded by $\frac{\sigma^2}{b}$.

\section*{Main Convergence Theorem}
\textbf{Theorem 1 (Convergence of Minibatch SGD for Non-Convex Functions):} Under Assumptions 1-4, if we run Minibatch SGD for $T$ iterations with a fixed step size $\eta$ satisfying $0 < \eta \leq \frac{1}{L}$, and batch size $b \geq 1$, the expected squared norm of the gradients satisfies:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2[f(w_0) - f^*]}{\eta T} + \frac{\eta L \sigma^2}{b}
\end{equation}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Convergence to a Stationary Point:} For non-convex functions, Minibatch SGD cannot guarantee convergence to a global minimum. Instead, the theorem guarantees that the average expected gradient norm converges to a neighborhood of zero. This implies the algorithm finds an approximate stationary point.
    \item \textbf{Variance Reduction via Minibatching:} The residual error term $\frac{\eta L \sigma^2}{b}$ is inversely proportional to the batch size $b$. Increasing the batch size directly reduces the noise in the stochastic gradient, allowing the algorithm to converge to a tighter neighborhood around the stationary point.
    \item \textbf{Hyperparameter Trade-offs:} The learning rate $\eta$ controls a trade-off. A larger $\eta$ decreases the first term (optimization error) faster but increases the second term (stochastic error). To achieve an $\epsilon$-stationary point (where $\frac{1}{T}\sum \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \epsilon$), one typically sets $\eta = \mathcal{O}(1/\sqrt{T})$ and scales $T$ appropriately.
\end{itemize}

\section*{Discussion}
Compared to standard SGD ($b=1$), Minibatch SGD allows for more stable updates and highly efficient parallel computation on modern hardware (like GPUs). Compared to full-batch Gradient Descent (GD), Minibatch SGD avoids computing gradients over the entire dataset, maintaining a low per-iteration computational cost while introducing stochasticity which can empirically help escape shallow local minima or saddle points.

\section*{Preliminaries (Definitions and Lemmas)}
\textbf{Lemma 1 (Descent Lemma):} For any $L$-smooth function $f$, the following inequality holds for all $x, y \in \mathbb{R}^d$:
\begin{equation}
    f(y) \leq f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2} \|y - x\|^2
\end{equation}
\textit{Proof of Lemma 1:} This is a standard consequence of the fundamental theorem of calculus and the $L$-Lipschitz continuity of the gradient (Assumption 2).

\textbf{Lemma 2 (Minibatch Variance):} Let $g_t = \frac{1}{b} \sum_{i=1}^b \nabla f(w_t; z_{t,i})$. Under Assumptions 3 and 4, we have:
\begin{equation}
    \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 \mid w_t] \leq \frac{\sigma^2}{b}
\end{equation}
\textit{Proof of Lemma 2:} Since the samples $z_{t,i}$ are drawn independently, the variance of the empirical mean of $b$ independent random variables is $1/b$ times the variance of a single sample.

\section*{Main Proof (Step-by-Step Derivation)}

We now provide the rigorous step-by-step derivation of Theorem 1. Let $\mathbb{E}_t[\cdot]$ denote the conditional expectation $\mathbb{E}[\cdot \mid w_t]$ given the parameter state at iteration $t$.

\begin{itemize}
    \item \textbf{[Step 1]} By applying the Descent Lemma (Lemma 1) to the points $w_{t+1}$ and $w_t$, we obtain:
    \begin{equation}
        f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t), w_{t+1} - w_t \rangle + \frac{L}{2} \|w_{t+1} - w_t\|^2
    \end{equation}
    
    \item \textbf{[Step 2]} Substitute the algorithm's update rule $w_{t+1} - w_t = -\eta g_t$ into the inequality:
    \begin{equation}
        f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t), -\eta g_t \rangle + \frac{L}{2} \|-\eta g_t\|^2
    \end{equation}
    
    \item \textbf{[Step 3]} Expand the squared norm and pull out the constant $-\eta$ from the inner product:
    \begin{equation}
        f(w_{t+1}) \leq f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L \eta^2}{2} \|g_t\|^2
    \end{equation}
    
    \item \textbf{[Step 4]} Take the conditional expectation $\mathbb{E}_t[\cdot]$ on both sides of the inequality:
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \eta \langle \nabla f(w_t), \mathbb{E}_t[g_t] \rangle + \frac{L \eta^2}{2} \mathbb{E}_t[\|g_t\|^2]
    \end{equation}
    Note that $f(w_t)$ and $\nabla f(w_t)$ are deterministic given $w_t$, so they pass through the conditional expectation.
    
    \item \textbf{[Step 5]} Apply the unbiased gradient assumption (Assumption 3), $\mathbb{E}_t[g_t] = \nabla f(w_t)$, to the inner product term:
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \eta \langle \nabla f(w_t), \nabla f(w_t) \rangle + \frac{L \eta^2}{2} \mathbb{E}_t[\|g_t\|^2]
    \end{equation}
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \mathbb{E}_t[\|g_t\|^2]
    \end{equation}
    
    \item \textbf{[Step 6]} Decompose the expected squared norm of the stochastic gradient using the bias-variance decomposition $\mathbb{E}[\|X\|^2] = \|\mathbb{E}[X]\|^2 + \mathbb{E}[\|X - \mathbb{E}[X]\|^2]$:
    \begin{equation}
        \mathbb{E}_t[\|g_t\|^2] = \|\mathbb{E}_t[g_t]\|^2 + \mathbb{E}_t[\|g_t - \mathbb{E}_t[g_t]\|^2]
    \end{equation}
    Substituting $\mathbb{E}_t[g_t] = \nabla f(w_t)$ yields:
    \begin{equation}
        \mathbb{E}_t[\|g_t\|^2] = \|\nabla f(w_t)\|^2 + \mathbb{E}_t[\|g_t - \nabla f(w_t)\|^2]
    \end{equation}
    
    \item \textbf{[Step 7]} Apply the minibatch variance bound (Lemma 2 and Assumption 4) to bound the second term:
    \begin{equation}
        \mathbb{E}_t[\|g_t\|^2] \leq \|\nabla f(w_t)\|^2 + \frac{\sigma^2}{b}
    \end{equation}
    
    \item \textbf{[Step 8]} Substitute this upper bound on $\mathbb{E}_t[\|g_t\|^2]$ back into the conditional expected descent inequality from Step 5:
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \left( \|\nabla f(w_t)\|^2 + \frac{\sigma^2}{b} \right)
    \end{equation}
    
    \item \textbf{[Step 9]} Group the terms involving $\|\nabla f(w_t)\|^2$ together:
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \left( \eta - \frac{L \eta^2}{2} \right) \|\nabla f(w_t)\|^2 + \frac{L \eta^2 \sigma^2}{2b}
    \end{equation}
    
    \item \textbf{[Step 10]} We assume the learning rate is chosen such that $0 < \eta \leq \frac{1}{L}$. This implies $L\eta \leq 1$, and therefore $\frac{L \eta^2}{2} \leq \frac{\eta}{2}$. We can bound the coefficient of the gradient norm:
    \begin{equation}
        \eta - \frac{L \eta^2}{2} \geq \eta - \frac{\eta}{2} = \frac{\eta}{2}
    \end{equation}
    Substituting this into the inequality gives:
    \begin{equation}
        \mathbb{E}_t[f(w_{t+1})] \leq f(w_t) - \frac{\eta}{2} \|\nabla f(w_t)\|^2 + \frac{L \eta^2 \sigma^2}{2b}
    \end{equation}
\end{itemize}

\section*{Rate Derivation (With Constants)}

\begin{itemize}
    \item \textbf{[Step 11]} Rearrange the inequality from Step 10 to isolate the gradient norm term on the left-hand side:
    \begin{equation}
        \frac{\eta}{2} \|\nabla f(w_t)\|^2 \leq f(w_t) - \mathbb{E}_t[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2b}
    \end{equation}
    
    \item \textbf{[Step 12]} Take the total expectation (over all randomness in the trajectory from $t=0$) on both sides. By the law of total expectation $\mathbb{E}[\mathbb{E}_t[\cdot]] = \mathbb{E}[\cdot]$:
    \begin{equation}
        \frac{\eta}{2} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2b}
    \end{equation}
    
    \item \textbf{[Step 13]} Sum the inequalities from $t = 0$ to $T - 1$. This creates a telescoping sum for the objective function terms:
    \begin{equation}
        \sum_{t=0}^{T-1} \frac{\eta}{2} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \sum_{t=0}^{T-1} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \right) + \sum_{t=0}^{T-1} \frac{L \eta^2 \sigma^2}{2b}
    \end{equation}
    Evaluating the telescoping sum yields:
    \begin{equation}
        \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq f(w_0) - \mathbb{E}[f(w_T)] + \frac{T L \eta^2 \sigma^2}{2b}
    \end{equation}
    
    \item \textbf{[Step 14]} Apply Assumption 1 ($f(w) \geq f^*$ for all $w$, so $-\mathbb{E}[f(w_T)] \leq -f^*$):
    \begin{equation}
        \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq f(w_0) - f^* + \frac{T L \eta^2 \sigma^2}{2b}
    \end{equation}
    
    \item \textbf{[Step 15]} Divide the entire inequality by $\frac{\eta T}{2}$ to obtain the average gradient norm bound:
    \begin{equation}
        \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2(f(w_0) - f^*)}{\eta T} + \frac{L \eta \sigma^2}{b}
    \end{equation}
\end{itemize}
This concludes the proof, establishing the convergence rate of $\mathcal{O}(1/T) + \mathcal{O}(\eta/b)$.

\section*{Special Cases}

\textbf{Case 1: Standard Stochastic Gradient Descent ($b=1$)}
When the batch size is $1$, the variance term is unmitigated by averaging. The convergence bound becomes:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2(f(w_0) - f^*)}{\eta T} + L \eta \sigma^2
\end{equation}
To ensure convergence to $0$, one must decay $\eta \to 0$ as $T \to \infty$ (e.g., $\eta = 1/\sqrt{T}$).

\textbf{Case 2: Full-Batch Gradient Descent ($b = N \to \infty$ or $\sigma^2 = 0$)}
If the full dataset is used, the gradient is exact and deterministic, meaning the variance $\sigma^2 = 0$. The stochastic error term vanishes entirely:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \|\nabla f(w_t)\|^2 \leq \frac{2(f(w_0) - f^*)}{\eta T}
\end{equation}
With a fixed step size $\eta = 1/L$, this yields a deterministic convergence rate of $\mathcal{O}(1/T)$ to a stationary point, recovering standard non-convex optimization rates for Gradient Descent.
