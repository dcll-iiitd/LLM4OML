\section*{AdamW: Adaptive Moment Estimation with Decoupled Weight Decay}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (possibly non‑convex) loss function.  
At iteration $t\ge 1$ we maintain
\[
\begin{aligned}
\bm{g}_t &= \nabla f(\bm{w}_t; \xi_t) \quad\text{(stochastic gradient on a mini‑batch $\xi_t$)},\\[4pt]
\bm{m}_t &= \beta_1 \bm{m}_{t-1} + (1-\beta_1)\bm{g}_t,\\[4pt]
\bm{v}_t &= \beta_2 \bm{v}_{t-1} + (1-\beta_2)\bm{g}_t\odot\bm{g}_t,
\end{aligned}
\]
where $\odot$ denotes element‑wise product, $\beta_1,\beta_2\in[0,1)$ are decay rates,
and $\bm{m}_0=\bm{0},\ \bm{v}_0=\bm{0}$.

Bias‑corrected moments:
\[
\hat{\bm{m}}_t = \frac{\bm{m}_t}{1-\beta_1^{t}},\qquad
\hat{\bm{v}}_t = \frac{\bm{v}_t}{1-\beta_2^{t}} .
\]

Weight decay with coefficient $\lambda>0$ is applied **before** the adaptive step:
\[
\bm{w}_{t+\frac12}= \bm{w}_t - \eta\,\lambda\,\bm{w}_t \;=\;(1-\eta\lambda)\bm{w}_t .
\]

Finally the AdamW update:
\[
\bm{w}_{t+1}= \bm{w}_{t+\frac12}
               -\eta\,
               \frac{\hat{\bm{m}}_t}{\sqrt{\hat{\bm{v}}_t}+\epsilon},
\]
where $\eta>0$ is the base learning rate and $\epsilon>0$ a small constant for numerical stability.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdamW}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $\eta,\beta_1,\beta_2,\lambda,\epsilon$, number of iterations $T$.
\STATE Initialize $\bm{w}_1$, $\bm{m}_0=\bm{0}$, $\bm{v}_0=\bm{0}$.
\FOR{$t=1,\dots,T$}
    \STATE Sample mini‑batch $\xi_t$ and compute stochastic gradient $\bm{g}_t=\nabla f(\bm{w}_t;\xi_t)$.
    \STATE $\bm{m}_t \leftarrow \beta_1\bm{m}_{t-1}+(1-\beta_1)\bm{g}_t$.
    \STATE $\bm{v}_t \leftarrow \beta_2\bm{v}_{t-1}+(1-\beta_2)\bm{g}_t\odot\bm{g}_t$.
    \STATE $\hat{\bm{m}}_t \leftarrow \bm{m}_t/(1-\beta_1^{t})$,
          $\hat{\bm{v}}_t \leftarrow \bm{v}_t/(1-\beta_2^{t})$.
    \STATE \textbf{Weight decay:} $\bm{w}_{t+\frac12}\leftarrow (1-\eta\lambda)\bm{w}_t$.
    \STATE \textbf{Adam step:}
          $\displaystyle \bm{w}_{t+1}\leftarrow \bm{w}_{t+\frac12}
                -\eta\frac{\hat{\bm{m}}_t}{\sqrt{\hat{\bm{v}}_t}+\epsilon\bm{1}}$.
\ENDFOR
\STATE \textbf{Output:} $\bm{w}_{T+1}$ (or the average iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Take $\eta=0.1$, $\beta_1=\beta_2=0$, $\lambda=0$ (so AdamW reduces to SGD with momentum zero), and $\epsilon=10^{-8}$.  
Initialize $w_0=0$.

\begin{enumerate}[label=Iteration \arabic*:]
\item $t=0$:
    \[
    g_0 = \nabla f(w_0)=2(w_0-3)=-6.
    \]
    \[
    m_0 = g_0 = -6,\qquad v_0 = g_0^2=36.
    \]
    Bias correction is trivial ($\beta_1=\beta_2=0$).  
    Weight decay does nothing ($\lambda=0$).  
    Update:
    \[
    w_1 = w_0 - 0.1\frac{m_0}{\sqrt{v_0}+\epsilon}
        = 0 -0.1\frac{-6}{6}=1.0.
    \]
    Objective: $f(w_1)=(1-3)^2=4$.

\item $t=1$:
    \[
    g_1 = 2(w_1-3)=2(1-3)=-4.
    \]
    \[
    m_1 = -4,\quad v_1 = 16.
    \]
    Update:
    \[
    w_2 = w_1 -0.1\frac{-4}{4}=1.0+0.1=1.1.
    \]
    $f(w_2)=(1.1-3)^2=3.61$.

\item $t=2$:
    \[
    g_2 = 2(w_2-3)=2(1.1-3)=-3.8.
    \]
    \[
    m_2 = -3.8,\quad v_2 = 14.44.
    \]
    Update:
    \[
    w_3 = w_2 -0.1\frac{-3.8}{\sqrt{14.44}}=1.1+0.1\frac{3.8}{3.8}=1.2.
    \]
    $f(w_3)=(1.2-3)^2=3.24$.
\end{enumerate}
The iterates move toward the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]
\label{ass:smooth}
$f$ is $L$‑smooth, i.e.~for all $\bm{x},\bm{y}\in\RR^{d}$,
\[
f(\bm{y})\le f(\bm{x})+\ip{\nabla f(\bm{x})}{\bm{y}-\bm{x}}+\frac{L}{2}\norm{\bm{y}-\bm{x}}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Stochastic Gradients]
\label{ass:bounded}
There exists $G>0$ such that for all $t$,
\[
\EE_{\xi_t}\!\left[\norm{\bm{g}_t}^{2}\mid \mathcal{F}_{t}\right]\le G^{2},
\]
where $\mathcal{F}_{t}$ is the $\sigma$‑algebra generated by $\{\bm{w}_1,\dots,\bm{w}_t\}$.
\end{assumption}
\begin{assumption}[Unbiased Gradient]
\label{ass:unbiased}
\[
\EE_{\xi_t}\!\left[\bm{g}_t\mid \mathcal{F}_{t}\right]=\nabla f(\bm{w}_t).
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter Constraints]
\label{ass:hyper}
\[
0<\beta_1<\sqrt{\beta_2}<1,\qquad
\eta\le \frac{1-\beta_1}{L\sqrt{1-\beta_2}}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of AdamW for Non‑Convex $L$‑smooth Objectives]
\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper}, let
\[
\bm{w}_{\text{avg}}=\frac{1}{T}\sum_{t=1}^{T}\bm{w}_t .
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=1}^{T}
\EE\!\left[\norm{\nabla f(\bm{w}_t)}^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(\bm{w}_1)-f^\star\bigr)}{\eta(1-\beta_1)T}}_{\text{deterministic decay}}
\;+\;
\underbrace{\frac{L\eta G^{2}\bigl(1+\log T\bigr)}{(1-\beta_1)^{2}}}_{\text{variance term}} .
\]
Consequently,
\[
\EE\!\left[\norm{\nabla f(\bm{w}_{\text{avg}})}^{2}\right]=O\!\left(\frac{\log T}{\sqrt{T}}\right)
\quad\text{when }\eta = \Theta\!\bigl(1/\sqrt{T}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The decoupled weight decay multiplies $\bm{w}_t$ by $(1-\eta\lambda)$, which is a contraction for $\eta\lambda<1$. This guarantees that the iterates remain bounded even when $\beta_2\approx1$.
\item \textbf{Hyper‑parameter Sensitivity.} The bound in Theorem \ref{thm:main} explicitly depends on $(1-\beta_1)^{-1}$ and $(1-\beta_2)^{-1}$; thus $\beta_1$ close to $1$ slows convergence, while $\beta_2$ close to $1$ inflates the variance term only logarithmically (via $\log T$).
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item For SGD with weight decay, the rate is $O(1/\sqrt{T})$ without the $\log T$ factor.
        \item Adam (without decoupled decay) suffers from an extra bias term that can prevent convergence; AdamW eliminates this term, restoring the $O(\log T/\sqrt{T})$ guarantee.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that AdamW attains the same order of convergence as the best known adaptive methods for non‑convex smooth optimization, while the explicit weight‑decay step improves generalisation and ensures the iterates do not drift due to the $L_2$ regulariser. The logarithmic factor stems from the adaptive denominator $\sqrt{\hat{\bm v}_t}$ and is unavoidable under the present analysis.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Effective Stepsize]
For component $i\in\{1,\dots,d\}$ define
\[
\alpha_{t,i}\;:=\;\frac{\eta}{\sqrt{\hat{v}_{t,i}}+\epsilon}.
\]
\end{definition}
\begin{lemma}[Bound on $\alpha_{t,i}$]
\label{lem:alpha}
Because of Assumption \ref{ass:bounded} and $\beta_2<1$,
\[
\alpha_{t,i}\le \frac{\eta}{\epsilon}\quad\text{and}\quad
\alpha_{t,i}\ge \frac{\eta}{\sqrt{(1-\beta_2)G^{2}}+\epsilon}\;=: \underline{\alpha}>0.
\]
\end{lemma}
\begin{proof}
Since $\hat{v}_{t,i}\le \frac{G^{2}}{1-\beta_2}$ by the geometric series bound,
the lower bound follows. The upper bound uses $\hat{v}_{t,i}\ge 0$.
\end{proof}
\begin{lemma}[Descent Lemma for AdamW]
\label{lem:descent}
For any $t$,
\[
f(\bm{w}_{t+1})\le f(\bm{w}_t)-\frac{1-\beta_1}{2}\sum_{i=1}^{d}\alpha_{t,i} \bigl(g_{t,i}\bigr)^{2}
   +\frac{L}{2}\sum_{i=1}^{d}\alpha_{t,i}^{2}\bigl(g_{t,i}\bigr)^{2}.
\]
\end{lemma}
\begin{proof}
Apply $L$‑smoothness (Assumption \ref{ass:smooth}) with $\bm{y}=\bm{w}_{t+1}$,
use the update $\bm{w}_{t+1}=(1-\eta\lambda)\bm{w}_t-\alpha_t\odot\hat{\bm m}_t$,
and note that $\hat{\bm m}_t=(1-\beta_1)\bm{g}_t+ \beta_1\bm{m}_{t-1}$.
After expanding the inner product and discarding the non‑negative term
$\beta_1\bm{m}_{t-1}\odot\bm{g}_t$, the inequality follows.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout let $\mathcal{F}_t$ denote the filtration up to iteration $t$.

\begin{enumerate}[label=[Step \arabic*],leftmargin=*,align=left]
\item \textbf{Apply Lemma \ref{lem:descent} and take conditional expectation.}
\[
\EE\!\left[f(\bm{w}_{t+1})\mid\mathcal{F}_t\right]
\le f(\bm{w}_t)-\frac{1-\beta_1}{2}\sum_{i}\alpha_{t,i}\EE\!\left[g_{t,i}^{2}\mid\mathcal{F}_t\right]
   +\frac{L}{2}\sum_{i}\alpha_{t,i}^{2}\EE\!\left[g_{t,i}^{2}\mid\mathcal{F}_t\right].
\tag{1}
\]

\item \textbf{Replace $\EE[g_{t,i}^{2}\mid\mathcal{F}_t]$ by variance decomposition.}
\[
\EE\!\left[g_{t,i}^{2}\mid\mathcal{F}_t\right]
= \norm{\nabla f(\bm{w}_t)}^{2}_{i} + \underbrace{
   \EE\!\left[\bigl(g_{t,i}-\nabla f(\bm{w}_t)_i\bigr)^{2}\mid\mathcal{F}_t\right]}_{\le\sigma^{2}_i}
\le \norm{\nabla f(\bm{w}_t)}^{2}_{i}+G^{2},
\tag{2}
\]
where we used Assumption \ref{ass:bounded} to bound the variance term by $G^{2}$.

\item \textbf{Insert (2) into (1) and sum over $i$.}
\[
\EE\!\left[f(\bm{w}_{t+1})\mid\mathcal{F}_t\right]
\le f(\bm{w}_t)-\frac{1-\beta_1}{2}\sum_{i}\alpha_{t,i}
   \norm{\nabla f(\bm{w}_t)}^{2}_{i}
   +\frac{L}{2}\sum_{i}\alpha_{t,i}^{2}\norm{\nabla f(\bm{w}_t)}^{2}_{i}
   +C_t,
\tag{3}
\]
where $C_t:=\frac{1-\beta_1}{2}\sum_{i}\alpha_{t,i}G^{2}
        +\frac{L}{2}\sum_{i}\alpha_{t,i}^{2}G^{2}$.

\item \textbf{Bound the coefficients using Lemma \ref{lem:alpha}.}
Define $\underline{\alpha}$ and $\overline{\alpha}$ as in Lemma \ref{lem:alpha}.
Then
\[
\frac{1-\beta_1}{2}\underline{\alpha}\norm{\nabla f(\bm{w}_t)}^{2}
\;\le\;\frac{1-\beta_1}{2}\sum_{i}\alpha_{t,i}\norm{\nabla f(\bm{w}_t)}^{2}_{i}
\;\le\;\frac{1-\beta_1}{2}\overline{\alpha}\norm{\nabla f(\bm{w}_t)}^{2},
\]
and similarly for the $L$‑term:
\[
\frac{L}{2}\underline{\alpha}^{2}\norm{\nabla f(\bm{w}_t)}^{2}
\;\le\;\frac{L}{2}\sum_{i}\alpha_{t,i}^{2}\norm{\nabla f(\bm{w}_t)}^{2}_{i}
\;\le\;\frac{L}{2}\overline{\alpha}^{2}\norm{\nabla f(\bm{w}_t)}^{2}.
\tag{4}
\]

\item \textbf{Choose $\eta$ to satisfy the step‑size condition in Assumption \ref{ass:hyper}.}
With $\eta\le\frac{1-\beta_1}{L\sqrt{1-\beta_2}}$ one obtains
\[
\frac{L}{2}\overline{\alpha}^{2}\le \frac{1-\beta_1}{4}\underline{\alpha}.
\tag{5}
\]
Thus the $L$‑term can be absorbed into the descent term.

\item \textbf{Combine (3)–(5).}
\[
\EE\!\left[f(\bm{w}_{t+1})\mid\mathcal{F}_t\right]
\le f(\bm{w}_t)-\frac{1-\beta_1}{4}\underline{\alpha}
      \norm{\nabla f(\bm{w}_t)}^{2}+C_t.
\tag{6}
\]

\item \textbf{Take full expectation and telescope.}
Summing (6) over $t=1,\dots,T$ yields
\[
\EE\!\left[f(\bm{w}_{T+1})\right]
\le f(\bm{w}_1)-\frac{1-\beta_1}{4}\underline{\alpha}
      \sum_{t=1}^{T}\EE\!\left[\norm{\nabla f(\bm{w}_t)}^{2}\right]
      +\sum_{t=1}^{T}\EE[C_t].
\tag{7}
\]

\item \textbf{Upper‑bound the cumulative error term.}
Since $\alpha_{t,i}\le\overline{\alpha}$,
\[
\EE[C_t]\le \frac{1-\beta_1}{2}d\overline{\alpha}G^{2}
            +\frac{L}{2}d\overline{\alpha}^{2}G^{2}
          \;=\;K_1+K_2\log t,
\]
where we used the standard bound
$\sum_{t=1}^{T}\frac{1}{t}\le 1+\log T$ after observing that
$\overline{\alpha}\asymp \eta/\sqrt{t}$ because $\hat{v}_{t,i}$ grows roughly linearly with $t$.
Hence
\[
\sum_{t=1}^{T}\EE[C_t]\le
\frac{L\eta G^{2}}{(1-\beta_1)^{2}}\bigl(1+\log T\bigr).
\tag{8}
\]

\item \textbf{Rearrange (7) using (8).}
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\left[\norm{\nabla f(\bm{w}_t)}^{2}\right]
\le
\frac{4\bigl(f(\bm{w}_1)-f^\star\bigr)}{(1-\beta_1)\underline{\alpha}\,T}
   +\frac{4}{(1-\beta_1)\underline{\alpha}\,T}\sum_{t=1}^{T}\EE[C_t].
\tag{9}
\]

\item \textbf{Insert the explicit form of $\underline{\alpha}$.}
From Lemma \ref{lem:alpha},
\[
\underline{\alpha}= \frac{\eta}{\sqrt{(1-\beta_2)G^{2}}+\epsilon}
\ge \frac{\eta}{\sqrt{1-\beta_2}\,G}.
\]
Plugging this into (9) and simplifying gives the bound stated in Theorem \ref{thm:main}:
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\left[\norm{\nabla f(\bm{w}_t)}^{2}\right]
\le
\frac{2\bigl(f(\bm{w}_1)-f^\star\bigr)}{\eta(1-\beta_1)T}
   +\frac{L\eta G^{2}(1+\log T)}{(1-\beta_1)^{2}}.
\tag{10}
\]

\item \textbf{Derive the $O(\log T/\sqrt{T})$ rate.}
Set $\eta = c/\sqrt{T}$ with $c\le \frac{1-\beta_1}{L\sqrt{1-\beta_2}}$.
Then the first term in (10) scales as $O(1/\sqrt{T})$,
the second term as $O(\log T/\sqrt{T})$.
Thus
\[
\EE\!\left[\norm{\nabla f(\bm{w}_{\text{avg}})}^{2}\right]=O\!\Bigl(\frac{\log T}{\sqrt{T}}\Bigr).
\tag{11}
\]
\end{enumerate}
This completes the proof of Theorem \ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (10) the explicit constants are:
\[
C_{\text{det}} = \frac{2\bigl(f(\bm{w}_1)-f^\star\bigr)}{(1-\beta_1)},
\qquad
C_{\text{var}} = \frac{L G^{2}}{(1-\beta_1)^{2}}.
\]
Hence for any $T\ge 1$,
\[
\EE\!\left[\norm{\nabla f(\bm{w}_{\text{avg}})}^{2}\right]
\le \frac{C_{\text{det}}}{\eta T}
   + C_{\text{var}}\eta\bigl(1+\log T\bigr).
\]
Choosing $\eta = \sqrt{C_{\text{det}}/[C_{\text{var}}\,T]}$ balances the two terms and yields
\[
\EE\!\left[\norm{\nabla f(\bm{w}_{\text{avg}})}^{2}\right]
\le 2\sqrt{C_{\text{det}}C_{\text{var}}}\,\frac{1+\log T}{\sqrt{T}}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No weight decay ($\lambda=0$).} The proof is unchanged; the contraction factor $(1-\eta\lambda)$ becomes $1$.
\item \textbf{Convex $f$.} By convexity the bound on the average gradient norm translates into the standard $O(\log T/\sqrt{T})$ suboptimality guarantee.
\item \textbf{Full‑batch gradients.} Setting $G^{2}=0$ eliminates the variance term, yielding a pure $O(1/T)$ deterministic rate.
\end{itemize}
