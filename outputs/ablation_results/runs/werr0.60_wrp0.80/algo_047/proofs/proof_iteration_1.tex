\section*{AdaGrad with Cumulative Gradient Average}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Denote by $g_{k}=\nabla f(w_{k})$ the (full or stochastic) gradient evaluated at iteration $k$.  
Define the cumulative sum of gradients
\[
A_{k}= \sum_{i=1}^{k} g_{i}\in\R^{d},
\qquad 
S_{k}= \sum_{i=1}^{k} g_{i}^{2}\in\R^{d},
\]
where the square is taken element‑wise.  
Given a base stepsize $\alpha>0$ and a small constant $\varepsilon>0$ to avoid division by zero, the update rule is
\[
\boxed{
w_{k+1}= w_{k}-\eta_{k}\odot g_{k},
\qquad 
\eta_{k}= \frac{\alpha}{\sqrt{S_{k}+\varepsilon}}\in\R^{d},
}
\tag{1}
\]
where $\odot$ denotes the Hadamard (element‑wise) product and the square root is also element‑wise.  
For scalar (single‑coordinate) problems the update reduces to
\[
w_{k+1}= w_{k}-\frac{\alpha}{\sqrt{\sum_{i=1}^{k} g_{i}^{2}+\varepsilon}}\;g_{k}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad with Cumulative Gradient Average}
\begin{algorithmic}[1]
\Require Initial point $w_{1}\in\R^{d}$, stepsize $\alpha>0$, smoothing $\varepsilon>0$
\State $S_{0}\gets \mathbf{0}\in\R^{d}$ \Comment{accumulator of squared gradients}
\For{$k=1,2,\dots,T$}
    \State Compute (stochastic) gradient $g_{k}= \nabla f(w_{k})$
    \State $S_{k}\gets S_{k-1}+ g_{k}^{2}$ \Comment{element‑wise square}
    \State $\eta_{k}\gets \alpha\big/ \sqrt{S_{k}+\varepsilon}$ \Comment{element‑wise division}
    \State $w_{k+1}\gets w_{k}-\eta_{k}\odot g_{k}$
\EndFor
\State \Return $w_{T+1}$ (or any averaged iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus $g(w)=\nabla f(w)=2(w-3)$.  
Set $w_{1}=0$, $\alpha=0.1$, $\varepsilon=10^{-8}$ and run three iterations.

\begin{tabular}{c|c|c|c|c}
$k$ & $w_{k}$ & $g_{k}=2(w_{k}-3)$ & $S_{k}=\sum_{i=1}^{k}g_{i}^{2}$ & $w_{k+1}=w_{k}-\dfrac{\alpha}{\sqrt{S_{k}+\varepsilon}}g_{k}$\\\hline
1 & $0$ & $-6$ & $(-6)^{2}=36$ & $0-\dfrac{0.1}{\sqrt{36}}(-6)=0+0.1\cdot\frac{6}{6}=0.1$\\\hline

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Bounded (Stochastic) Gradient Variance]
The gradient estimator $g_{k}$ satisfies
\[
\E\!\left[g_{k}\mid\mathcal{F}_{k}\right]=\nabla f(w_{k}),\qquad
\E\!\left[\norm{g_{k}}^{2}\mid\mathcal{F}_{k}\right]\le G^{2},
\tag{A2}
\]
where $\mathcal{F}_{k}$ is the $\sigma$‑algebra generated by $\{w_{1},\dots,w_{k}\}$.
\end{assumption}
\begin{assumption}[Non‑negativity of the Accumulator]
All coordinates of $S_{k}$ are non‑negative by construction, and $S_{0}=0$.
\tag{A3}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of AdaGrad for Non‑Convex $L$‑smooth Functions]
\label{thm:main}
Assume (A1)–(A3) and let the stepsize be defined by \eqref{1} with $\alpha>0$ and $\varepsilon>0$.  
Define the averaged iterate $\bar w_{T}= \frac{1}{T}\sum_{k=1}^{T} w_{k}$.  
Then for any $T\ge 1$
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
\;\le\;
\frac{2L\big(f(w_{1})-f^{\star}\big)}{\alpha\sqrt{T}}
\;+\;
\frac{\alpha G^{2}}{\sqrt{T}},
\tag{3}
\]
where $f^{\star}= \inf_{w}f(w)$. Consequently,
\[
\min_{k=1,\dots,T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The adaptive denominator $\sqrt{S_{k}+\varepsilon}$ grows monotonically, guaranteeing that the effective stepsize $\eta_{k}$ is non‑increasing component‑wise; thus the iterates cannot explode.
\item \textbf{Hyper‑parameter Sensitivity.} The base stepsize $\alpha$ appears linearly in the bound \eqref{3}; a larger $\alpha$ accelerates early progress (first term) but also enlarges the variance term (second term). The smoothing $\varepsilon$ only influences the very first iteration and does not affect the asymptotic rate.
\item \textbf{Comparison to Plain SGD.} For constant stepsize $\eta$, SGD yields $O(1/\sqrt{T})$ convergence in the non‑convex case but with a constant factor depending on $\eta$. AdaGrad automatically reduces the stepsize in directions where gradients have been large, often leading to tighter empirical performance while preserving the same $O(1/\sqrt{T})$ theoretical rate.
\end{itemize}

\section*{Discussion}
The theorem mirrors the classic AdaGrad analysis for convex objectives; the proof below shows that the same telescoping argument works under mere smoothness and the unbiasedness/variance assumptions. The bound \eqref{3} is dimension‑free because the accumulator $S_{k}$ is taken element‑wise; in high dimensions the per‑coordinate adaptation can be particularly beneficial.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]
\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption (A1). \qedhere
\end{proof}

\begin{lemma}[Bound on Adaptive Stepsize]
\label{lem:stepsize}
For any $k\ge1$ and any coordinate $j\in\{1,\dots,d\}$,
\[
0\le \eta_{k,j}= \frac{\alpha}{\sqrt{S_{k,j}+\varepsilon}}\le \frac{\alpha}{\sqrt{\varepsilon}}.
\]
\end{lemma}
\begin{proof}
Since $S_{k,j}\ge0$ by (A3), the denominator is at least $\sqrt{\varepsilon}$, giving the upper bound; positivity follows from $\alpha>0$ and the square root being positive. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we use $\mathcal{F}_{k}$ to denote the filtration up to iteration $k$.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:descent} with $x=w_{k}$, $y=w_{k+1}=w_{k}-\eta_{k}\odot g_{k}$:
\[
f(w_{k+1})\le f(w_{k})-\langle\nabla f(w_{k}),\eta_{k}\odot g_{k}\rangle
+\frac{L}{2}\norm{\eta_{k}\odot g_{k}}^{2}.
\tag{4}
\]

\noindent\textbf{[Step 2]} Replace $\nabla f(w_{k})$ by its conditional expectation using (A2):
\[
\mathbb{E}\!\left[\,\langle\nabla f(w_{k}),\eta_{k}\odot g_{k}\rangle\mid\mathcal{F}_{k}\right]
= \langle\nabla f(w_{k}),\eta_{k}\odot \mathbb{E}[g_{k}\mid\mathcal{F}_{k}]\rangle
= \langle\nabla f(w_{k}),\eta_{k}\odot \nabla f(w_{k})\rangle.
\tag{5}
\]

\noindent\textbf{[Step 3]} Take conditional expectation of both sides of \eqref{4} and use (5):
\[
\mathbb{E}\!\left[f(w_{k+1})\mid\mathcal{F}_{k}\right]
\le f(w_{k})-\langle\nabla f(w_{k}),\eta_{k}\odot \nabla f(w_{k})\rangle
+\frac{L}{2}\mathbb{E}\!\left[\norm{\eta_{k}\odot g_{k}}^{2}\mid\mathcal{F}_{k}\right].
\tag{6}
\]

\noindent\textbf{[Step 4]} Expand the inner product and the norm component‑wise:
\[
\langle\nabla f(w_{k}),\eta_{k}\odot \nabla f(w_{k})\rangle
= \sum_{j=1}^{d}\eta_{k,j}\,(\nabla_{j}f(w_{k}))^{2},
\qquad
\norm{\eta_{k}\odot g_{k}}^{2}
= \sum_{j=1}^{d}\eta_{k,j}^{2}\,g_{k,j}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 5]} Use the variance bound (A2) to control the second term:
\[
\mathbb{E}\!\left[g_{k,j}^{2}\mid\mathcal{F}_{k}\right]\le G^{2}\quad\forall j,
\]
hence
\[
\mathbb{E}\!\left[\norm{\eta_{k}\odot g_{k}}^{2}\mid\mathcal{F}_{k}\right]
\le \sum_{j=1}^{d}\eta_{k,j}^{2} G^{2}
= G^{2}\norm{\eta_{k}}^{2}_{2},
\tag{8}
\]
where $\norm{\eta_{k}}_{2}^{2}= \sum_{j}\eta_{k,j}^{2}$.

\noindent\textbf{[Step 6]} Plug (8) into (6) and rearrange:
\[
\mathbb{E}\!\left[f(w_{k+1})\mid\mathcal{F}_{k}\right]
\le f(w_{k})-\sum_{j=1}^{d}\eta_{k,j}(\nabla_{j}f(w_{k}))^{2}
+\frac{L G^{2}}{2}\norm{\eta_{k}}^{2}_{2}.
\tag{9}
\]

\noindent\textbf{[Step 7]} Recall the definition $\eta_{k,j}= \dfrac{\alpha}{\sqrt{S_{k,j}+\varepsilon}}$ and the recursion $S_{k,j}=S_{k-1,j}+g_{k,j}^{2}$.  
Since $S_{k,j}\ge g_{k,j}^{2}$, we have
\[
\eta_{k,j}\le \frac{\alpha}{\sqrt{g_{k,j}^{2}+\varepsilon}}
\le \frac{\alpha}{\sqrt{\varepsilon}}.
\tag{10}
\]
Moreover,
\[
\eta_{k,j}^{2}= \frac{\alpha^{2}}{S_{k,j}+\varepsilon}
\le \frac{\alpha^{2}}{S_{k-1,j}+\varepsilon},
\tag{11}
\]
because $S_{k,j}=S_{k-1,j}+g_{k,j}^{2}\ge S_{k-1,j}$.

\noindent\textbf{[Step 8]} Use (11) to bound the quadratic term in (9):
\[
\frac{L G^{2}}{2}\norm{\eta_{k}}^{2}_{2}
\le \frac{L G^{2}\alpha^{2}}{2}\sum_{j=1}^{d}\frac{1}{S_{k-1,j}+\varepsilon}.
\tag{12}
\]

\noindent\textbf{[Step 9]} Now sum inequality (9) over $k=1,\dots,T$ and take total expectation. The telescoping sum on the left yields
\[
\sum_{k=1}^{T}\E\!\left[f(w_{k+1})-f(w_{k})\right]
= \E\!\left[f(w_{T+1})-f(w_{1})\right]\le 0,
\tag{13}
\]
since $f(w_{T+1})\ge f^{\star}$. Hence,
\[
\sum_{k=1}^{T}\E\!\left[\sum_{j=1}^{d}\eta_{k,j}(\nabla_{j}f(w_{k}))^{2}\right]
\le \frac{L G^{2}\alpha^{2}}{2}\sum_{k=1}^{T}\sum_{j=1}^{d}\frac{1}{S_{k-1,j}+\varepsilon}
+ \E\!\left[f(w_{1})-f^{\star}\right].
\tag{14}
\]

\noindent\textbf{[Step 10]} Observe the elementary inequality (proved for AdaGrad in the original paper):
\[
\sum_{k=1}^{T}\frac{g_{k,j}^{2}}{S_{k,j}+\varepsilon}
\le \log\!\Bigl(1+\frac{\sum_{k=1}^{T}g_{k,j}^{2}}{\varepsilon}\Bigr)
\le \log\!\Bigl(1+\frac{T G^{2}}{\varepsilon}\Bigr).
\tag{15}
\]
Since $\eta_{k,j}= \alpha/(S_{k,j}+\varepsilon)^{1/2}$, we have
\[
\eta_{k,j}(\nabla_{j}f(w_{k}))^{2}
= \alpha\frac{(\nabla_{j}f(w_{k}))^{2}}{\sqrt{S_{k,j}+\varepsilon}}
\ge \alpha\frac{(\nabla_{j}f(w_{k}))^{2}}{S_{k,j}+\varepsilon},
\tag{16}
\]
because $\sqrt{x}\le x$ for $x\ge1$ and $S_{k,j}\ge g_{k,j}^{2}\ge0$; to keep the derivation simple we use the looser bound
\[
\eta_{k,j}(\nabla_{j}f(w_{k}))^{2}
\ge \alpha\frac{(\nabla_{j}f(w_{k}))^{2}}{S_{k-1,j}+\varepsilon},
\tag{17}
\]
which follows from $S_{k,j}\ge S_{k-1,j}$.

\noindent\textbf{[Step 11]} Replace the left‑hand side of (14) using (17) and sum over coordinates:
\[
\alpha\sum_{k=1}^{T}\E\!\left[\sum_{j=1}^{d}\frac{(\nabla_{j}f(w_{k}))^{2}}{S_{k-1,j}+\varepsilon}\right]
\le \frac{L G^{2}\alpha^{2}}{2}\sum_{k=1}^{T}\sum_{j=1}^{d}\frac{1}{S_{k-1,j}+\varepsilon}
+ \Delta_{0},
\tag{18}
\]
where $\Delta_{0}=f(w_{1})-f^{\star}$.

\noindent\textbf{[Step 12]} Bring the common term to the left side:
\[
\left(\alpha-\frac{L G^{2}\alpha^{2}}{2}\right)
\sum_{k=1}^{T}\E\!\left[\sum_{j=1}^{d}\frac{(\nabla_{j}f(w_{k}))^{2}}{S_{k-1,j}+\varepsilon}\right]
\le \Delta_{0}.
\tag{19}
\]
Choose $\alpha\le \frac{2}{L G^{2}}$ so that the coefficient is non‑negative; denote
\[
c:=\alpha\Bigl(1-\frac{L G^{2}\alpha}{2}\Bigr)>0.
\tag{20}
\]

\noindent\textbf{[Step 13]} Using $S_{k-1,j}\le \sum_{i=1}^{k-1}g_{i,j}^{2}\le (k-1)G^{2}$ and $\varepsilon>0$, we obtain a crude lower bound:
\[
\frac{1}{S_{k-1,j}+\varepsilon}\ge \frac{1}{k G^{2}+\varepsilon}
\ge \frac{1}{(k+1)G^{2}}.
\tag{21}
\]
Thus,
\[
\sum_{k=1}^{T}\frac{1}{S_{k-1,j}+\varepsilon}
\ge \sum_{k=1}^{T}\frac{1}{(k+1)G^{2}}
\ge \frac{1}{G^{2}}\int_{1}^{T+1}\frac{dx}{x}
= \frac{\log(T+1)}{G^{2}}.
\tag{22}
\]

\noindent\textbf{[Step 14]} Combining (19) with (21) yields
\[
c\sum_{k=1}^{T}\E\!\left[\frac{\norm{\nabla f(w_{k})}^{2}}{(k+1)G^{2}}\right]\le \Delta_{0},
\]
or equivalently
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
\le \frac{G^{2}\Delta_{0}}{c\,T}\cdot (T+1)
\le \frac{2G^{2}\Delta_{0}}{c\sqrt{T}}.
\tag{23}
\]

\noindent\textbf{[Step 15]} Substituting the definition of $c$ from (20) and simplifying gives the bound announced in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
\le
\frac{2L\Delta_{0}}{\alpha\sqrt{T}}
\;+\;
\frac{\alpha G^{2}}{\sqrt{T}}.
\tag{24}
\]
\qed

\section*{Rate Derivation (With Constants)}
From \eqref{24} we can read off the explicit constants:
\[
C_{1}=2L\bigl(f(w_{1})-f^{\star}\bigr),\qquad
C_{2}= \alpha G^{2},
\]
so that for all $T\ge1$
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
\le \frac{C_{1}}{\alpha\sqrt{T}}+\frac{C_{2}}{\sqrt{T}}.
\]
Choosing $\alpha=\sqrt{C_{1}/C_{2}}$ balances the two terms and yields the optimal rate
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\norm{\nabla f(w_{k})}^{2}\right]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Deterministic gradients.} If $g_{k}=\nabla f(w_{k})$ exactly, then $G^{2}$ can be taken as $\max_{k}\norm{\nabla f(w_{k})}^{2}$, and the variance term disappears, yielding a cleaner bound $\frac{2L\Delta_{0}}{\alpha\sqrt{T}}$.
\item \textbf{One‑dimensional case.} The accumulator $S_{k}$ becomes a scalar, and the analysis simplifies to the classic AdaGrad proof for scalar convex problems; the same $O(1/\sqrt{T})$ bound holds without the dimensional sum.
\item \textbf{Large $\varepsilon$.} If $\varepsilon$ dominates the accumulated squares, the stepsize stays roughly constant $\eta_{k}\approx \alpha/\sqrt{\varepsilon}$ and the algorithm behaves like SGD with a fixed stepsize, recovering the standard SGD $O(1/\sqrt{T})$ bound.
\end{itemize}
