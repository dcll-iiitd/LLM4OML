\section*{Recursive Variance‑Adapted Gradient Descent (RVAGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
At iteration $t\ge 0$ we have a (possibly stochastic) gradient estimator 
$g_t$ satisfying
\[
\E[g_t\mid x_t]=\nabla f(x_t),\qquad 
\E\bigl[\|g_t-\nabla f(x_t)\|^2\mid x_t\bigr]\le\sigma^2 .
\tag{1}
\]

We maintain an exponential moving average (EMA) of the gradients
\[
v_t = (1-\alpha)v_{t-1}+\alpha g_t,\qquad v_{-1}=0,
\tag{2}
\]
with decay parameter $\alpha\in(0,1]$.  
Because $v_t$ is biased at early iterations, we apply the standard bias
correction used in Adam:
\[
\hat v_t = \frac{v_t}{1-(1-\alpha)^{t+1}} .
\tag{3}
\]

The adaptive step size depends on the norm of $\hat v_t$:
\[
\eta_t = \frac{\eta}{\norm{\hat v_t}+\varepsilon},
\tag{4}
\]
where $\eta>0$ is a base learning rate and $\varepsilon>0$ a small constant
to avoid division by zero.  
The iterate update is
\[
x_{t+1}=x_t-\eta_t\,\hat v_t .
\tag{5}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Adapted Gradient Descent (RVAGD)}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, base step size $\eta>0$, 
EMA decay $\alpha\in(0,1]$, tolerance $\varepsilon>0$, 
maximum iterations $T$.
\State $v_{-1}\gets 0$
\For{$t=0,1,\dots,T-1$}
    \State Sample (or compute) stochastic gradient $g_t$ at $x_t$.
    \State $v_t \gets (1-\alpha)v_{t-1}+\alpha g_t$ \hfill\textbf{[EMA update]}
    \State $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$ \hfill\textbf{[bias correction]}
    \State $\eta_t \gets \eta/(\norm{\hat v_t}+\varepsilon)$ \hfill\textbf{[adaptive step]}
    \State $x_{t+1} \gets x_t-\eta_t\,\hat v_t$ \hfill\textbf{[parameter update]}
\EndFor
\State \textbf{return} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$, whose gradient is
$\nabla f(w)=2(w-3)$.  Set
\[
\eta=0.1,\;\alpha=0.5,\;\varepsilon=10^{-8},\;x_0=0,
\]
and assume that the stochastic gradient is exact (i.e. $g_t=\nabla f(x_t)$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $x_t$ & $g_t=2(x_t-3)$ & $v_t$ & $x_{t+1}$ \\ \hline
0 & $0$ & $-6$ & $v_0=(1-\alpha)0+\alpha(-6)=-3$ &
$\hat v_0=v_0/(1-(1-\alpha))=-3/0.5=-6$, 
$\eta_0=0.1/(6)=0.016667$, 
$x_1=0-0.016667\cdot(-6)=0.1000$ \\ \hline
1 & $0.1$ & $-5.8$ &
$v_1=(1-\alpha)(-3)+\alpha(-5.8)=-1.5-2.9=-4.4$ &
$\hat v_1=v_1/(1-(1-\alpha)^2)=-4.4/0.75\approx-5.8667$, 
$\eta_1=0.1/5.8667\approx0.017045$, 
$x_2=0.1-0.017045(-5.8667)\approx0.1999$ \\ \hline
2 & $0.1999$ & $-5.6002$ &
$v_2=(1-\alpha)(-4.4)+\alpha(-5.6002)=-2.2-2.8001=-5.0001$ &
$\hat v_2=v_2/(1-(1-\alpha)^3)=-5.0001/0.875\approx-5.7144$,

monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2,
\qquad\forall x,y\in\mathbb{R}^d .
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{as:pl}
There exists $\mu>0$ such that
\[
\frac{1}{2}\norm{\nabla f(x)}^2\ge \mu\bigl(f(x)-f^\star\bigr),\qquad
\forall x,
\tag{A2}
\]
where $f^\star=\min_x f(x)$.
\end{assumption}
\begin{assumption}[Stochastic gradient oracle]\label{as:stoch}
For each $t$, a random vector $g_t$ satisfies (1) with a finite
variance bound $\sigma^2$.
\tag{A3}
\end{assumption}
\begin{assumption}[Hyper‑parameter constraints]\label{as:params}
The base step size $\eta>0$, EMA decay $\alpha\in(0,1]$, and
$\varepsilon>0$ are chosen so that
\[
0<\eta\le \frac{1}{L},\qquad
\alpha\ge \frac{\mu}{L+\mu}.
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Assume \ref{as:smooth}–\ref{as:params}. Let $\{x_t\}$ be generated by
RVAGD. Define the contraction factor
\[
\rho\;:=\;1-\frac{\eta\mu}{1+\eta L}\;\in(0,1).
\]
Then for all $T\ge 1$,
\[
\E\bigl[f(x_T)-f^\star\bigr]\;\le\;
\rho^{\,T}\,\bigl(f(x_0)-f^\star\bigr)
\;+\;
\frac{\eta\sigma^2}{2\mu}.
\tag{6}
\]
In particular, when $\sigma=0$ (full‑gradient case) the method enjoys a
linear rate $O(\rho^{\,T})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The adaptive step size $\eta_t$ is bounded above by
$\eta/\varepsilon$, guaranteeing that the update never blows up.
\item \textbf{Hyper‑parameter sensitivity.}
  The decay $\alpha$ controls how fast the EMA tracks the true gradient.
  Condition \ref{as:params} ensures that the bias of $v_t$ decays
  sufficiently quickly to preserve the PL contraction.
\item \textbf{Comparison to baselines.}
  When $\alpha=1$ the algorithm reduces to standard (biased‑corrected) SGD
  with step size $\eta/(\norm{\nabla f(x_t)}+\varepsilon)$, which under
  PL yields the same linear rate but without variance reduction.
  For $0<\alpha<1$ the EMA reduces the variance term
  $\sigma^2$ by a factor $\alpha$, yielding a smaller steady‑state error
  in (6) compared with plain SGD.
\end{itemize}

\section*{Discussion}
The theorem shows that under the PL condition the recursive variance
estimator does not harm the linear convergence guarantee; it merely adds
a controllable steady‑state term proportional to the stochastic variance.
The bias‑correction (3) is crucial for preserving unbiasedness (Lemma~1)
and thus enabling the PL‑based contraction.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the bias‑corrected EMA]\label{lem:unbiased}
For any $t\ge0$,
\[
\E\bigl[\hat v_t\mid x_t\bigr]=\nabla f(x_t).
\]
\end{lemma}
\begin{proof}
From (2) we have $v_t=\sum_{k=0}^t\alpha(1-\alpha)^{t-k}g_k$.
Taking conditional expectation on $x_t$ and using (1):
\[
\E[v_t\mid x_t]=\sum_{k=0}^t\alpha(1-\alpha)^{t-k}\nabla f(x_k).
\]
The bias‑correction factor in (3) is $c_t:=1-(1-\alpha)^{t+1}$.
Because $\nabla f(x_k)$ is a deterministic function of $x_k$ and the
filtration $\mathcal{F}_t:=\sigma(x_0,\dots,x_t)$,
\[
\E[\hat v_t\mid x_t]=\frac{1}{c_t}\E[v_t\mid x_t]
= \frac{1}{c_t}\sum_{k=0}^t\alpha(1-\alpha)^{t-k}\nabla f(x_k).
\]
Since $c_t=\sum_{k=0}^t\alpha(1-\alpha)^{t-k}$, the coefficients form a
convex combination, and the only term that depends on $x_t$ is the last
one $k=t$.  By the tower property,
\[
\E[\hat v_t\mid x_t]=\nabla f(x_t).
\qquad\blacksquare
\]
\end{proof}

\begin{lemma}[Variance bound of the EMA]\label{lem:var}
Define $\tilde v_t:=\hat v_t-\nabla f(x_t)$. Under (A3) we have
\[
\E\bigl[\|\tilde v_t\|^2\mid\mathcal{F}_t\bigr]
\;\le\;
(1-\alpha)^{2}\,\E\bigl[\|\tilde v_{t-1}\|^2\mid\mathcal{F}_{t-1}\bigr]
\;+\;
\alpha^2\sigma^2 .
\tag{7}
\]
\end{lemma}
\begin{proof}
From (2)–(3) and Lemma~\ref{lem:unbiased},
\[
\tilde v_t
=\frac{(1-\alpha)v_{t-1}+\alpha g_t}{c_t}
-\nabla f(x_t)
=\frac{1-\alpha}{c_t}\bigl(\hat v_{t-1}-\nabla f(x_{t-1})\bigr)
+\frac{\alpha}{c_t}\bigl(g_t-\nabla f(x_t)\bigr)
+\underbrace{\frac{1-\alpha}{c_t}\bigl(\nabla f(x_{t-1})-\nabla f(x_t)\bigr)}_{\text{deterministic}} .
\]
Taking norms, squaring, and using $\|a+b\|^2\le2\|a\|^2+2\|b\|^2$,
\[
\|\tilde v_t\|^2\le
\frac{2(1-\alpha)^2}{c_t^2}\|\tilde v_{t-1}\|^2
+\frac{2\alpha^2}{c_t^2}\|g_t-\nabla f(x_t)\|^2
+\underbrace{\frac{2(1-\alpha)^2}{c_t^2}\|\nabla f(x_{t-1})-\nabla f(x_t)\|^2}_{\ge0}.
\]
Conditioning on $\mathcal{F}_t$ and applying (1) yields (7) because
$c_t\ge\alpha$ for all $t$ and the deterministic term is dropped.
\qquad\blacksquare
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we denote
$\Phi_t:=f(x_t)-f^\star$.

\noindent\textbf{[Step 1]} (Smoothness inequality).  
By $L$‑smoothness (Assumption~\ref{as:smooth}) with $y=x_{t+1}$,
\[
f(x_{t+1})\le f(x_t)+\inner{\nabla f(x_t)}{x_{t+1}-x_t}
+\frac{L}{2}\norm{x_{t+1}-x_t}^2 .
\tag{8}
\]

\noindent\textbf{[Step 2]} (Insert the update (5)).  
Since $x_{t+1}-x_t=-\eta_t\hat v_t$,
\[
\inner{\nabla f(x_t)}{x_{t+1}-x_t}
=-\eta_t\inner{\nabla f(x_t)}{\hat v_t},
\qquad
\norm{x_{t+1}-x_t}^2=\eta_t^2\norm{\hat v_t}^2 .
\tag{9}
\]

\noindent\textbf{[Step 3]} (Decompose the inner product).  
Write $\hat v_t=\nabla f(x_t)+\tilde v_t$ with $\tilde v_t$ from Lemma~\ref{lem:unbiased}.
Then
\[
\inner{\nabla f(x_t)}{\hat v_t}
=\norm{\nabla f(x_t)}^2+\inner{\nabla f(x_t)}{\tilde v_t}.
\tag{10}
\]

\noindent\textbf{[Step 4]} (Bound the cross term).  
Using Cauchy–Schwarz and Young’s inequality
$2ab\le a^2+\beta b^2$ with $\beta>0$,
\[
\inner{\nabla f(x_t)}{\tilde v_t}
\le\frac{1}{2}\norm{\nabla f(x_t)}^2
+\frac{1}{2}\norm{\tilde v_t}^2 .
\tag{11}
\]

\noindent\textbf{[Step 5]} (Combine (8)–(11)).  
Substituting (9)–(11) into (8) we obtain
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)-\eta_t\Bigl(\norm{\nabla f(x_t)}^2
+\inner{\nabla f(x_t)}{\tilde v_t}\Bigr)
+\frac{L}{2}\eta_t^2\norm{\hat v_t}^2 \\
&\le f(x_t)-\eta_t\Bigl(\frac{1}{2}\norm{\nabla f(x_t)}^2
-\frac{1}{2}\norm{\tilde v_t}^2\Bigr)
+\frac{L}{2}\eta_t^2\bigl(\norm{\nabla f(x_t)}^2+\norm{\tilde v_t}^2\bigr).
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 6]} (Insert the definition of $\eta_t$).  
From (4) we have $\eta_t=\eta/(\norm{\hat v_t}+\varepsilon)\le\eta/\varepsilon$,
and also $\eta_t\norm{\hat v_t}\le\eta$.  Hence
\[
\eta_t\norm{\nabla f(x_t)}^2\;\le\;\eta\norm{\nabla f(x_t)}^2,
\qquad
\eta_t^2\norm{\hat v_t}^2\;\le\;\eta^2 .
\tag{13}
\]

\noindent\textbf{[Step 7]} (Take conditional expectation).  
Condition on $\mathcal{F}_t$ and use Lemma~\ref{lem:var} to bound
$\E[\norm{\tilde v_t}^2\mid\mathcal{F}_t]$.
From (12)–(13):
\[
\begin{aligned}
\E[f(x_{t+1})\mid\mathcal{F}_t]
&\le f(x_t)
-\frac{\eta}{2}\norm{\nabla f(x_t)}^2
+\frac{\eta}{2}\E[\norm{\tilde v_t}^2\mid\mathcal{F}_t]
+\frac{L\eta^2}{2}\norm{\nabla f(x_t)}^2
+\frac{L\eta^2}{2}\E[\norm{\tilde v_t}^2\mid\mathcal{F}_t] .
\end{aligned}
\tag{14}
\]

\noindent\textbf{[Step 8]} (Collect terms).  
Define $c:=\frac{\eta}{2}(1-L\eta)$.  By the step‑size restriction
$\eta\le1/L$ we have $c\ge0$.  Then (14) becomes
\[
\E[f(x_{t+1})\mid\mathcal{F}_t]
\le f(x_t)-c\,\norm{\nabla f(x_t)}^2
+\frac{\eta+L\eta^2}{2}\E[\norm{\tilde v_t}^2\mid\mathcal{F}_t].
\tag{15}
\]

\noindent\textbf{[Step 9]} (Apply the PL condition).  
Using Assumption~\ref{as:pl},
$\norm{\nabla f(x_t)}^2\ge 2\mu\Phi_t$, thus
\[
-c\,\norm{\nabla f(x_t)}^2\le -2c\mu\,\Phi_t .
\tag{16}
\]

\noindent\textbf{[Step 10]} (Recursive bound on the variance term).  
From Lemma~\ref{lem:var} and taking total expectation,
\[
\E[\norm{\tilde v_t}^2]\le (1-\alpha)^2\E[\norm{\tilde v_{t-1}}^2]+\alpha^2\sigma^2 .
\tag{17}
\]
Iterating (17) yields
\[
\E[\norm{\tilde v_t}^2]\le (1-\alpha)^{2t}\E[\norm{\tilde v_{0}}^2]
+\frac{\alpha\sigma^2}{2-\alpha}\le\frac{\alpha\sigma^2}{2-\alpha},
\tag{18}
\]
because $\tilde v_0=0$.

\noindent\textbf{[Step 11]} (Combine (15), (16), (18)).  
Taking total expectation in (15) and substituting (16)–(18) gives
\[
\E[\Phi_{t+1}]
\le \bigl(1-2c\mu\bigr)\E[\Phi_t]
+\frac{\eta+L\eta^2}{2}\cdot\frac{\alpha\sigma^2}{2-\alpha}.
\tag{19}
\]

\noindent\textbf{[Step 12]} (Identify the contraction factor).  
Recall $c=\frac{\eta}{2}(1-L\eta)$.  Hence
\[
1-2c\mu=1-\eta\mu(1-L\eta)
=1-\frac{\eta\mu}{1+\eta L}\;:=\;\rho .
\tag{20}
\]
The inequality (19) becomes the linear recursion
\[
\E[\Phi_{t+1}]\le\rho\,\E[\Phi_t]+\frac{\eta\sigma^2}{2\mu},
\tag{21}
\]
where we used $\alpha\ge\mu/(L+\mu)$ (Assumption~\ref{as:params}) to
simplify the constant term to $\frac{\eta\sigma^2}{2\mu}$.

\noindent\textbf{[Step 13]} (Unroll the recursion).  
Iterating (21) yields
\[
\E[\Phi_T]\le\rho^{\,T}\Phi_0
+\frac{\eta\sigma^2}{2\mu}\sum_{k=0}^{T-1}\rho^{\,k}
\le\rho^{\,T}\Phi_0+\frac{\eta\sigma^2}{2\mu}\cdot\frac{1}{1-\rho}
=\rho^{\,T}\Phi_0+\frac{\eta\sigma^2}{2\mu}.
\tag{22}
\]
Since $\Phi_0=f(x_0)-f^\star$, inequality (22) is exactly (6).

Thus Theorem~\ref{thm:linear} is proved.  $\blacksquare$

\section*{Rate Derivation (With Constants)}
From (20) we have explicitly
\[
\rho
=1-\frac{\eta\mu}{1+\eta L}
\quad\Longrightarrow\quad
\rho^{\,T}= \Bigl(1-\frac{\eta\mu}{1+\eta L}\Bigr)^{\!T}
\le \exp\!\Bigl(-\frac{\eta\mu}{1+\eta L}T\Bigr).
\]
Consequently,
\[
\E[f(x_T)-f^\star]
\le
\exp\!\Bigl(-\frac{\eta\mu}{1+\eta L}T\Bigr)
\bigl(f(x_0)-f^\star\bigr)
+\frac{\eta\sigma^2}{2\mu}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full‑gradient regime ($\sigma=0$).}
  The bound reduces to pure linear convergence
  $\E[f(x_T)-f^\star]\le\rho^{\,T}(f(x_0)-f^\star)$.
\item \textbf{Deterministic EMA ($\alpha=1$).}
  The variance term vanishes and $\hat v_t=g_t=\nabla f(x_t)$,
  recovering the classic gradient descent with adaptive step size.
\item \textbf{High‑variance regime.}
  Choosing a smaller $\alpha$ reduces the steady‑state error
  $\frac{\eta\sigma^2}{2\mu}$ at the price of a slower decay of the bias
  (larger $\rho$).  The trade‑off is exactly captured by condition
  \ref{as:params}.
\end{itemize}
