\section*{Algorithm Name}
\textbf{Clipped Momentum Gradient Descent (CMGD)}  

The method first clips the raw gradient to a prescribed norm bound $C>0$, then applies an exponential moving‑average (momentum) to the clipped gradients before the parameter update.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the objective.  
For $t=0,1,2,\dots$ the iterates $\{x_t\}$ are generated as follows:

\begin{align}
    g_t^{\text{raw}} &:= \nabla f(x_t), \label{eq:rawgrad}\\
    \alpha_t &:= \max\!\Bigl\{1,\;\frac{\|g_t^{\text{raw}}\|}{C}\Bigr\}, \qquad 
    g_t := \frac{g_t^{\text{raw}}}{\alpha_t}\;\;\;(\text{gradient clipping})\label{eq:clip}\\
    m_t &:= \beta\, m_{t-1} + (1-\beta)\, g_t ,\qquad m_{-1}=0,\;\; \beta\in[0,1) \label{eq:momentum}\\
    x_{t+1} &:= x_t - \eta\, m_t ,\qquad \eta>0 \text{ (stepsize)}. \label{eq:update}
\end{align}

The clipping operator can be written compactly as  
\[
\operatorname{Clip}_C(v):=\frac{v}{\max\{1,\|v\|/C\}} .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Momentum Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, clipping radius $C>0$, momentum coefficient $\beta\in[0,1)$
\State $m_{-1}\gets 0$
\For{$t=0,1,\dots,T-1$}
    \State $g^{\text{raw}}_t \gets \nabla f(x_t)$
    \State $\alpha_t \gets \max\{1,\|g^{\text{raw}}_t\|/C\}$
    \State $g_t \gets g^{\text{raw}}_t / \alpha_t$ \Comment{Clipped gradient}
    \State $m_t \gets \beta\, m_{t-1} + (1-\beta)\, g_t$
    \State $x_{t+1} \gets x_t - \eta\, m_t$
\EndFor
\State \textbf{Output:} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Set $w_0=0$, learning rate $\eta=0.1$, clipping bound $C=1$, momentum $\beta=0.5$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\nabla f(w_t)$ & $\alpha_t$ & $g_t$ (clipped) & $w_{t+1}$ \\ \hline
0 & $2(w_0-3)=-6$ & $\max\{1,6/1\}=6$ & $-6/6=-1$ & $0-0.1\cdot[-1]=0.1$ \\
1 & $2(0.1-3)=-5.8$ & $5.8$ & $-5.8/5.8=-1$ & $0.1-0.1\cdot[0.5\cdot(-1)+(0.5)(-1)]=0.2$ \\
2 & $2(0.2-3)=-5.6$ & $5.6$ & $-5.6/5.6=-1$ & $0.2-0.1\cdot[0.5\cdot(-1)+0.5\cdot(-1)]=0.3$ \\
\end{tabular}
\end{center}

Objective values:
\[
f(w_0)=9,\; f(w_1)=(0.1-3)^2=8.41,\; f(w_2)=(0.2-3)^2=7.84,\; f(w_3)=(0.3-3)^2=7.29.
\]
The iterates move steadily toward the minimiser $w^\star=3$ while the gradient norm is kept at the clipping threshold $C=1$.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .\tag{A1'}
\]

\item \textbf{Lower boundedness.} $\displaystyle f^\star:=\inf_{x}f(x)>-\infty$. \tag{A2}

\item \textbf{Clipping radius.} $C>0$ is a fixed constant. \tag{A3}

\item \textbf{Momentum parameter.} $\beta\in[0,1)$ and stepsize $\eta>0$ satisfy
\[
0<\eta\le\frac{(1-\beta)C}{L(1+\beta)} .\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationary‑point convergence of CMGD]\label{thm:main}
Under Assumptions (A1)–(A4), for any integer $T\ge1$ the iterates produced by CMGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(x_t)\|^2\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta(1-\beta)T}
\;+\;
\frac{2L\eta C^2}{1-\beta}\; .
\]
Consequently, choosing $\eta =\Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\min_{0\le t<T}\mathbb{E}\bigl[\|\nabla f(x_t)\|^2\bigr]=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Gradient clipping guarantees $\|g_t\|\le C$ for all $t$, preventing exploding updates even when $\|\nabla f(x_t)\|$ is large.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ reduces the bias introduced by clipping (the algorithm behaves like vanilla momentum GD) but weakens the stability guarantee.
        \item The bound \eqref{eq:update} together with (A4) shows that $\eta$ must scale linearly with $C$; a too‑large $\eta$ violates the descent condition.
        \item The momentum factor $\beta$ appears only through the factor $(1-\beta)^{-1}$ in the convergence rate, reflecting the well‑known trade‑off between acceleration and noise amplification.
    \end{itemize}
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Without clipping ($C=\infty$) the bound reduces to the standard $O(1/\sqrt{T})$ rate for momentum SGD on $L$‑smooth non‑convex functions.
        \item With aggressive clipping (small $C$) the second term $2L\eta C^2/(1-\beta)$ dominates, yielding a slower but still sub‑linear decay; the algorithm becomes robust to heavy‑tailed stochastic gradients.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem demonstrates that CMGD preserves the optimal $O(T^{-1/2})$ convergence rate of first‑order methods for smooth non‑convex optimisation while adding a hard bound on the per‑iteration update magnitude. This property is particularly valuable in deep learning settings where gradient explosions can destabilise training. The proof proceeds by a careful combination of the descent lemma, the clipping inequality $\|g_t\|\le C$, and the algebra of the momentum recursion.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^2 .
\]
\end{lemma}
\begin{proof}
Directly from (A1') with $y=x-d$. \hfill $\square$
\end{proof}

\begin{lemma}[Momentum recursion bound]\label{lem:momentum}
For the sequence $\{m_t\}$ defined in \eqref{eq:momentum} with $m_{-1}=0$,
\[
\|m_t\|\le \frac{1-\beta^{t+1}}{1-\beta}\,C\le\frac{C}{1-\beta},\qquad\forall t\ge0 .
\]
\end{lemma}
\begin{proof}
Since $\|g_t\|\le C$ by clipping, we have by induction:
\[
\|m_t\|\le\beta\|m_{t-1}\|+(1-\beta)C
\le\beta\Bigl(\frac{1-\beta^{t}}{1-\beta}C\Bigr)+(1-\beta)C
=\frac{1-\beta^{t+1}}{1-\beta}C .
\] \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout we suppress the expectation $\mathbb{E}[\cdot]$ for brevity, noting that all random quantities are measurable with respect to the filtration generated by $\{x_s\}_{s\le t}$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=\eta m_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),m_t\rangle+\frac{L\eta^2}{2}\|m_t\|^2 .\tag{1}
\]

\item[Step 2.] Decompose $m_t$ using the momentum definition \eqref{eq:momentum}:
\[
m_t=\beta m_{t-1}+(1-\beta)g_t .
\]
Hence
\[
\langle\nabla f(x_t),m_t\rangle
=\beta\langle\nabla f(x_t),m_{t-1}\rangle+(1-\beta)\langle\nabla f(x_t),g_t\rangle .\tag{2}
\]

\item[Step 3.] Since $g_t$ is the clipped version of the raw gradient,
\[
g_t = \frac{\nabla f(x_t)}{\alpha_t},\qquad 
\alpha_t:=\max\{1,\|\nabla f(x_t)\|/C\}\ge 1 .
\]
Therefore
\[
\langle\nabla f(x_t),g_t\rangle = \frac{\|\nabla f(x_t)\|^2}{\alpha_t}
\ge \frac{\|\nabla f(x_t)\|^2}{\alpha_t}\ge \frac{\|\nabla f(x_t)\|^2}{\max\{1,\|\nabla f(x_t)\|/C\}} .
\]
Two cases arise:

\begin{itemize}
\item If $\|\nabla f(x_t)\|\le C$, then $\alpha_t=1$ and $\langle\nabla f(x_t),g_t\rangle=\|\nabla f(x_t)\|^2$.
\item If $\|\nabla f(x_t)\|> C$, then $\alpha_t=\|\nabla f(x_t)\|/C$ and
\[
\langle\nabla f(x_t),g_t\rangle = \frac{\|\nabla f(x_t)\|^2}{\|\nabla f(x_t)\|/C}=C\|\nabla f(x_t)\|\ge C^2 .
\]
\end{itemize}
Both cases can be compactly expressed as
\[
\langle\nabla f(x_t),g_t\rangle \ge \frac{\|\nabla f(x_t)\|^2}{\alpha_t}
\ge \frac{\|\nabla f(x_t)\|^2}{1+\|\nabla f(x_t)\|/C}
\ge \frac{1}{2}\min\bigl\{\|\nabla f(x_t)\|^2,\,C^2\bigr\}. \tag{3}
\]

\item[Step 4.] Plug \eqref{eq:momentum} and \eqref{eq:clip} into (1) and use (2):
\[
f(x_{t+1})\le f(x_t)-\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle
-\eta(1-\beta)\langle\nabla f(x_t),g_t\rangle
+\frac{L\eta^2}{2}\|m_t\|^2 .\tag{4}
\]

\item[Step 5.] Rearrange (4) to isolate the term involving $\langle\nabla f(x_t),g_t\rangle$:
\[
\eta(1-\beta)\langle\nabla f(x_t),g_t\rangle
\le f(x_t)-f(x_{t+1})+\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle
+\frac{L\eta^2}{2}\|m_t\|^2 .\tag{5}
\]

\item[Step 6.] Upper‑bound the cross term $\langle\nabla f(x_t),m_{t-1}\rangle$ by Cauchy–Schwarz and Lemma~\ref{lem:momentum}:
\[
\langle\nabla f(x_t),m_{t-1}\rangle\le\|\nabla f(x_t)\|\,\|m_{t-1}\|
\le\|\nabla f(x_t)\|\frac{C}{1-\beta}. \tag{6}
\]

\item[Step 7.] Upper‑bound $\|m_t\|^2$ using Lemma~\ref{lem:momentum}:
\[
\|m_t\|^2\le\Bigl(\frac{C}{1-\beta}\Bigr)^2 .\tag{7}
\]

\item[Step 8.] Substitute (6) and (7) into (5) and divide by $\eta(1-\beta)$:
\[
\langle\nabla f(x_t),g_t\rangle
\le \frac{f(x_t)-f(x_{t+1})}{\eta(1-\beta)}
+\frac{\beta C}{(1-\beta)^2}\,\|\nabla f(x_t)\|
+\frac{L\eta C^2}{2(1-\beta)^3}. \tag{8}
\]

\item[Step 9.] Apply the lower bound (3) to the left‑hand side of (8). Using $\min\{a,b\}\le a$ we obtain
\[
\frac12\min\{\|\nabla f(x_t)\|^2,\,C^2\}
\le \frac{f(x_t)-f(x_{t+1})}{\eta(1-\beta)}
+\frac{\beta C}{(1-\beta)^2}\,\|\nabla f(x_t)\|
+\frac{L\eta C^2}{2(1-\beta)^3}. \tag{9}
\]

\item[Step 10.] Treat the mixed term $\|\nabla f(x_t)\|$ by Young’s inequality:
\[
\frac{\beta C}{(1-\beta)^2}\,\|\nabla f(x_t)\|
\le \frac{1}{4}\|\nabla f(x_t)\|^2
+\frac{\beta^2 C^2}{(1-\beta)^4}. \tag{10}
\]

\item[Step 11.] Insert (10) into (9) and multiply the whole inequality by $2$:
\[
\min\{\|\nabla f(x_t)\|^2,\,C^2\}
\le \frac{2\bigl(f(x_t)-f(x_{t+1})\bigr)}{\eta(1-\beta)}
+\frac{1}{2}\|\nabla f(x_t)\|^2
+\frac{2\beta^2 C^2}{(1-\beta)^4}
+\frac{L\eta C^2}{(1-\beta)^3}. \tag{11}
\]

\item[Step 12.] Move the $\frac12\|\nabla f(x_t)\|^2$ term to the left side:
\[
\frac12\|\nabla f(x_t)\|^2
\le \frac{2\bigl(f(x_t)-f(x_{t+1})\bigr)}{\eta(1-\beta)}
+\frac{2\beta^2 C^2}{(1-\beta)^4}
+\frac{L\eta C^2}{(1-\beta)^3}. \tag{12}
\]

\item[Step 13.] Sum (12) over $t=0,\dots,T-1$:
\[
\frac12\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^2
\le \frac{2\bigl(f(x_0)-f(x_T)\bigr)}{\eta(1-\beta)}
+T\Bigl(\frac{2\beta^2 C^2}{(1-\beta)^4}
+\frac{L\eta C^2}{(1-\beta)^3}\Bigr). \tag{13}
\]

\item[Step 14.] Drop the non‑negative term $f(x_T)-f^\star\ge0$ and use $f(x_0)-f^\star\ge f(x_0)-f(x_T)$:
\[
\frac12\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^2
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta(1-\beta)}
+T\Bigl(\frac{2\beta^2 C^2}{(1-\beta)^4}
+\frac{L\eta C^2}{(1-\beta)^3}\Bigr). \tag{14}
\]

\item[Step 15.] Divide both sides of (14) by $T$ and use $\beta\in[0,1)$ to bound the coefficient of $\beta^2$ by $\frac{2}{(1-\beta)^2}$ (since $\beta^2\le\beta$ and $1-\beta\le1$). This yields the cleaner bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^2
\le
\frac{4\bigl(f(x_0)-f^\star\bigr)}{\eta(1-\beta)T}
+\frac{2L\eta C^2}{(1-\beta)} .
\tag{15}
\]

\item[Step 16.] Finally, taking expectations (the inequality holds pointwise) gives the statement of Theorem~\ref{thm:main}. \hfill $\blacksquare$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (15) we have
\[
\mathbb{E}\bigl[\|\nabla f(\tilde{x})\|^2\bigr]
\le
\frac{4\Delta_0}{\eta(1-\beta)T}
+\frac{2L\eta C^2}{1-\beta},
\qquad
\Delta_0:=f(x_0)-f^\star .
\]
Choosing $\eta = \sqrt{\frac{2\Delta_0}{L C^2 T}}$ (which respects (A4) because of the $C$ factor) balances the two terms and yields
\[
\mathbb{E}\bigl[\|\nabla f(\tilde{x})\|^2\bigr]
\le
\frac{4\sqrt{2L\Delta_0}\,C}{(1-\beta)\sqrt{T}} .
\]
Thus the algorithm attains the optimal $O(T^{-1/2})$ rate up to the constant $C/(1-\beta)$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No clipping ($C\to\infty$).} The clipping operator becomes the identity, $g_t=\nabla f(x_t)$. Lemma~\ref{lem:momentum} still holds with $C$ replaced by $\|\nabla f(x_t)\|$, and (15) reduces to the classic momentum SGD bound $O(1/(\eta T))$.
\item \textbf{Zero momentum ($\beta=0$).} The update reduces to plain clipped gradient descent:
\[
x_{t+1}=x_t-\eta\,\operatorname{Clip}_C\bigl(\nabla f(x_t)\bigr).
\]
The bound (15) simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\bigl[\|\nabla f(x_t)\|^2\bigr]
\le
\frac{4\Delta_0}{\eta T}+2L\eta C^2,
\]
exactly the rate known for gradient clipping without momentum.
\item \textbf{Large clipping radius ($C\ge \max_t\|\nabla f(x_t)\|$).} The algorithm behaves identically to standard momentum GD, and the extra $C$‑dependent term disappears from the bound.
\end{itemize}
