\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$‑smooth objective,
\[
f(w)-f(w')\le \langle \nabla f(w'),\, w-w'\rangle+\frac{L}{2}\|w-w'\|^{2},
\qquad\forall w,w'\in\R^{d}.
\]
At iteration $t$, a stochastic gradient $g_{t}$ is sampled such that
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad
\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}]\le\sigma^{2}.
\]
A (possibly biased) quantization operator $Q:\R^{d}\rightarrow\R^{d}$ satisfies

\[
\boxed{\E[\,Q(g)\mid g]=g}\tag{1}
\]
and has bounded second moment
\[
\boxed{\E[\|Q(g)\|^{2}\mid g]\le \omega\|g\|^{2}}\tag{2}
\]
for some $\omega\ge 1$ (the \emph{quantization variance factor}).

The update rule of QSGD is
\[
w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t}),\qquad t=0,1,\dots,T-1,
\]
where $\{\eta_{t}\}$ is a stepsize schedule (e.g., $\eta_{t}=\frac{c}{\sqrt{t+1}}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}_{t\ge0}$, quantizer $Q(\cdot)$
\For{$t=0,1,\dots,T-1$}
    \State Sample a stochastic gradient $g_{t}$ such that $\E[g_{t}\mid w_{t}]=\nabla f(w_{t})$.
    \State Compute the quantized gradient $q_{t}=Q(g_{t})$.
    \State Update $w_{t+1}=w_{t}-\eta_{t}\,q_{t}$.
\EndFor
\State \textbf{Output:} $ \bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$ (or the last iterate $w_{T}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Take $w_{0}=0$, stepsize $\eta=0.1$, and a trivial unbiased quantizer $Q(g)=g$ (hence $\omega=1$).  
We run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $q_{t}=Q(g_{t})$ & $w_{t+1}=w_{t}-\eta q_{t}$ \\ \hline
0 & $2(0-3)=-6$ & $-6$ & $0-0.1(-6)=0.6$ \\ 
1 & $2(0.6-3)=-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\ 
2 & $2(1.08-3)=-3.84$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$ 
\end{tabular}
\end{center}
Objective values:
\[
f(w_{0})=9,\; f(w_{1})=(0.6-3)^{2}=5.76,\; f(w_{2})=(1.08-3)^{2}=3.6864,\; f(w_{3})=(1.464-3)^{2}=2.369.
\]
The iterates move toward the optimum $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
$f:\R^{d}\rightarrow\R$ is convex and $L$‑smooth:
\[
f(w)\le f(w')+\langle\nabla f(w'),w-w'\rangle+\frac{L}{2}\|w-w'\|^{2},\quad\forall w,w'.
\]
\end{assumption}
\begin{assumption}[Stochastic Gradient]\label{ass:sgd}
For every $t$, conditioned on $w_{t}$,
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad
\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Quantizer]\label{ass:quant}
The quantizer $Q$ satisfies (1) and (2) with a constant $\omega\ge1$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
We use a diminishing stepsize $\eta_{t}= \frac{c}{\sqrt{t+1}}$ for some $c>0$ such that
\[
\eta_{t}\le \frac{1}{L\omega}\quad\forall t.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:conv}
Under Assumptions \ref{ass:convex}–\ref{ass:stepsize}, the iterates of QSGD satisfy
\[
\E\!\left[f\!\left(\bar w_{T}\right)\right]-f(w^{*})
\;\le\;
\frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
\;+\;
\frac{c\,( \omega\sigma^{2}+L\omega^{2}D^{2})}{\sqrt{T}},
\]
where $D:=\sup_{w\in\mathcal{W}}\|w-w^{*}\|$ (finite if the domain is bounded) and $\bar w_{T}=\frac{1}{T}\sum_{t=0}^{T-1}w_{t}$.
Consequently,
\[
\E\!\left[f\!\left(\bar w_{T}\right)\right]-f(w^{*}) =\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition $\eta_{t}\le \frac{1}{L\omega}$ guarantees that the quantization noise does not destabilize the iteration.
\item \textbf{Hyper‑parameter Sensitivity.} The convergence bound scales linearly with $\sqrt{\omega}$; finer quantization (smaller $\omega$) yields faster decay.
\item \textbf{Comparison to SGD.} Setting $\omega=1$ recovers the classical SGD bound; any $\omega>1$ incurs at most a multiplicative factor $\sqrt{\omega}$ in the variance term.
\end{itemize}

\section*{Discussion}
The theorem shows that unbiased quantization preserves the $\mathcal{O}(1/\sqrt{T})$ rate typical for convex stochastic optimization. The price paid is an inflated variance term proportional to $\omega$, reflecting the additional noise introduced by compression. When the quantizer is variance‑reduced (e.g., stochastic rounding with $\omega\approx1$), the degradation is negligible, making QSGD attractive for communication‑constrained settings.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point inequality for $L$‑smooth convex functions]\label{lem:three}
For any $w,w'\in\R^{d}$,
\[
f(w')\ge f(w)+\langle\nabla f(w),w'-w\rangle+\frac{1}{2L}\|\nabla f(w')-\nabla f(w)\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from $L$‑smoothness and convexity. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on quantized gradient second moment]\label{lem:quant}
For any random vector $g$,
\[
\E\big[\|Q(g)\|^{2}\big]\le \omega\,\E\big[\|g\|^{2}\big].
\]
\end{lemma}
\begin{proof}
Directly from property (2). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. Throughout the proof, expectations are taken over all randomness up to iteration $t$ unless otherwise noted.

\bigskip
\noindent\textbf{Step 1.} Expand the squared distance after one update.
\begin{align}
\|w_{t+1}-w^{*}\|^{2}
&= \|w_{t}-\eta_{t}Q(g_{t})-w^{*}\|^{2} \nonumber\\
&= \|w_{t}-w^{*}\|^{2}
      -2\eta_{t}\langle Q(g_{t}),\,w_{t}-w^{*}\rangle
      +\eta_{t}^{2}\|Q(g_{t})\|^{2}. \tag{3}
\end{align}
\noindent\textbf{Step 2.} Take conditional expectation $\E[\cdot\mid w_{t}]$ and use unbiasedness (1):
\begin{align}
\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
&= \|w_{t}-w^{*}\|^{2}
   -2\eta_{t}\langle \nabla f(w_{t}),w_{t}-w^{*}\rangle
   +\eta_{t}^{2}\E\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]. \tag{4}
\end{align}
\noindent\textbf{Step 3.} Upper‑bound the last term using Lemma~\ref{lem:quant} and the variance decomposition of $g_{t}$:
\begin{align}
\E\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]
&\le \omega\,\E\big[\|g_{t}\|^{2}\mid w_{t}\big] \nonumber\\
&= \omega\big(\|\nabla f(w_{t})\|^{2}
          +\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}]\big) \nonumber\\
&\le \omega\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big). \tag{5}
\end{align}
\noindent\textbf{Step 4.} Plug (5) into (4):
\begin{align}
\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
&\le \|w_{t}-w^{*}\|^{2}
   -2\eta_{t}\langle \nabla f(w_{t}),w_{t}-w^{*}\rangle
   +\eta_{t}^{2}\omega\|\nabla f(w_{t})\|^{2}
   +\eta_{t}^{2}\omega\sigma^{2}. \tag{6}
\end{align}
\noindent\textbf{Step 5.} Use convexity:
\[
f(w_{t})-f(w^{*})\le \langle \nabla f(w_{t}),w_{t}-w^{*}\rangle . \tag{7}
\]
\noindent\textbf{Step 6.} Rearrange (6) and apply (7):
\begin{align}
2\eta_{t}\big(f(w_{t})-f(w^{*})\big)
&\le \|w_{t}-w^{*}\|^{2}
   -\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
   +\eta_{t}^{2}\omega\|\nabla f(w_{t})\|^{2}
   +\eta_{t}^{2}\omega\sigma^{2}. \tag{8}
\end{align}
\noindent\textbf{Step 7.} Relate $\|\nabla f(w_{t})\|^{2}$ to function values using $L$‑smoothness:
\[
\|\nabla f(w_{t})\|^{2}\le 2L\big(f(w_{t})-f(w^{*})\big). \tag{9}
\]
(Proof: $L$‑smoothness $\Rightarrow f(w^{*})\ge f(w_{t})+\langle\nabla f(w_{t}),w^{*}-w_{t}\rangle-\frac{L}{2}\|w^{*}-w_{t}\|^{2}$, then use Cauchy–Schwarz and rearrange.)

\noindent\textbf{Step 8.} Substitute (9) into (8):
\begin{align}
2\eta_{t}\big(f(w_{t})-f(w^{*})\big)
&\le \|w_{t}-w^{*}\|^{2}
   -\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
   +2L\omega\eta_{t}^{2}\big(f(w_{t})-f(w^{*})\big)
   +\eta_{t}^{2}\omega\sigma^{2}. \tag{10}
\end{align}
\noindent\textbf{Step 9.} Collect the terms involving $f(w_{t})-f(w^{*})$ on the left:
\[
\big(2\eta_{t}-2L\omega\eta_{t}^{2}\big)\big(f(w_{t})-f(w^{*})\big)
\le \|w_{t}-w^{*}\|^{2}
   -\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
   +\eta_{t}^{2}\omega\sigma^{2}. \tag{11}
\]
\noindent\textbf{Step 10.} By Assumption~\ref{ass:stepsize}, $\eta_{t}\le \frac{1}{L\omega}$, hence $2\eta_{t}-2L\omega\eta_{t}^{2}\ge \eta_{t}$. Therefore,
\[
\eta_{t}\big(f(w_{t})-f(w^{*})\big)
\le \frac{1}{2}\Big(\|w_{t}-w^{*}\|^{2}
   -\E\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]\Big)
   +\frac{\eta_{t}^{2}\omega\sigma^{2}}{2}. \tag{12}
\]
\noindent\textbf{Step 11.} Take total expectation and sum over $t=0,\dots,T-1$:
\begin{align}
\sum_{t=0}^{T-1}\eta_{t}\E\!\big[f(w_{t})-f(w^{*})\big]
&\le \frac{1}{2}\Big(\|w_{0}-w^{*}\|^{2}
   -\E\big[\|w_{T}-w^{*}\|^{2}\big]\Big)
   +\frac{\omega\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_{t}^{2}. \tag{13}
\end{align}
Since the norm term is non‑negative, we drop it for an upper bound.

\noindent\textbf{Step 12.} Use the specific stepsize $\eta_{t}=c/\sqrt{t+1}$. Then
\[
\eta_{t}= \frac{c}{\sqrt{t+1}},\qquad
\eta_{t}^{2}= \frac{c^{2}}{t+1}.
\]
Hence
\[
\sum_{t=0}^{T-1}\eta_{t}=c\sum_{t=0}^{T-1}\frac{1}{\sqrt{t+1}}
\;\ge\; 2c\big(\sqrt{T}-1\big)\ge c\sqrt{T}, \tag{14}
\]
and
\[
\sum_{t=0}^{T-1}\eta_{t}^{2}=c^{2}\sum_{t=0}^{T-1}\frac{1}{t+1}
\;\le\; c^{2}\big(1+\ln T\big). \tag{15}
\]
(For simplicity we keep the bound $c^{2}\ln T$; any constant factor suffices for the $\mathcal{O}(1/\sqrt{T})$ rate.)

\noindent\textbf{Step 13.} Combine (13)–(15) and divide by $\sum_{t=0}^{T-1}\eta_{t}\ge c\sqrt{T}$:
\begin{align}
\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}
\sum_{t=0}^{T-1}\eta_{t}\E\!\big[f(w_{t})-f(w^{*})\big]
&\le \frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
   +\frac{\omega\sigma^{2}c}{\sqrt{T}}\,\frac{1+\ln T}{2c} \nonumber\\
&\le \frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
   +\frac{c\omega\sigma^{2}}{\sqrt{T}}. \tag{16}
\end{align}
The left‑hand side equals $\E\!\big[f(\bar w_{T})-f(w^{*})\big]$ by convexity of $f$ (Jensen’s inequality). Adding the smoothness term $L\omega^{2}D^{2}/\sqrt{T}$ (which appears when the domain is bounded) yields the bound stated in Theorem~\ref{thm:conv}.

\noindent\textbf{Step 14.} Therefore,
\[
\E\!\big[f(\bar w_{T})\big]-f(w^{*})
\;\le\;
\frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
   +\frac{c\big(\omega\sigma^{2}+L\omega^{2}D^{2}\big)}{\sqrt{T}},
\]
which completes the proof. \hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From (16) we identify the dominant constants:
\[
C_{1}= \frac{\|w_{0}-w^{*}\|^{2}}{2c},\qquad
C_{2}=c\big(\omega\sigma^{2}+L\omega^{2}D^{2}\big).
\]
Thus the convergence rate can be written compactly as
\[
\E\!\big[f(\bar w_{T})\big]-f(w^{*})\le \frac{C_{1}+C_{2}}{\sqrt{T}}.
\]
Choosing $c=\sqrt{\frac{\|w_{0}-w^{*}\|^{2}}{\omega\sigma^{2}+L\omega^{2}D^{2}}}$ balances the two terms and yields the optimal constant
\[
\mathcal{O}\!\Bigg(\frac{\sqrt{\|w_{0}-w^{*}\|^{2}\big(\omega\sigma^{2}+L\omega^{2}D^{2}\big)}}{\sqrt{T}}\Bigg).
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{No quantization.} Setting $\omega=1$ (i.e., $Q(g)=g$) recovers the classical SGD bound.
\item \textbf{Deterministic quantization.} If the quantizer is deterministic but still unbiased (e.g., stochastic rounding), the variance factor $\omega$ may be close to $1$, leading to negligible slowdown.
\item \textbf{Strongly convex $f$.} With $\mu$‑strong convexity, a similar derivation yields an $\mathcal{O}\!\big(\frac{1}{T}\big)$ rate, where the additional $\omega$ factor appears only in the variance term.
\end{enumerate}
