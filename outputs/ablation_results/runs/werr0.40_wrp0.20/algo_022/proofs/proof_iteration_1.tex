\section*{Proximal Gradient Descent (PGD) for Non‑Convex Smooth $f$}

\section*{Mathematical Formulation}
Let  
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}
\item $f:\R^{d}\to\R$ is \textbf{differentiable} and $L$–smooth (i.e., $\|\grad f(x)-\grad f(y)\|\le L\|x-y\|$ for all $x,y$).  
\item $g:\R^{d}\to\R\cup\{+\infty\}$ is \textbf{proper, closed, convex} (possibly non‑smooth).  
\end{itemize}

For a stepsize $\eta>0$, the \emph{proximal operator} of $g$ is
\[
\prox_{\eta g}(v)\;:=\;\arg\min_{u\in\R^{d}}\Big\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]

The PGD iteration is
\begin{equation}
\boxed{%
x_{k+1}\;=\;\prox_{\eta g}\!\big(x_{k}-\eta\,\grad f(x_{k})\big),\qquad k=0,1,\dots
}\tag{1}
\end{equation}
Define the \emph{gradient mapping} (also called proximal gradient) 
\[
\mathcal{G}_{\eta}(x):=\frac{1}{\eta}\Big(x-\prox_{\eta g}\!\big(x-\eta\grad f(x)\big)\Big).
\] 
When $g\equiv0$, $\mathcal{G}_{\eta}(x)=\grad f(x)$ and (1) reduces to ordinary gradient descent.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\R^{d}$, max iterations $K$ (or tolerance).
\For{$k=0$ \textbf{to} $K-1$}
    \State $v_{k}\gets x_{k}-\eta\,\grad f(x_{k})$ \Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(v_{k})$ \Comment{proximal step}
\EndFor
\State \textbf{Output:} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=(w-3)^{2},\qquad g\equiv0,\qquad \eta=0.1,\qquad w_{0}=0.
\]
Since $g=0$, $\prox_{\eta g}$ is the identity. The iteration becomes $w_{k+1}=w_{k}-\eta\,\grad f(w_{k})$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:lip}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{2}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g$ is proper, closed and convex, so its proximal operator is \emph{firmly non‑expansive}:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|^{2}
\le \langle u-v,\prox_{\eta g}(u)-\prox_{\eta g}(v)\rangle,\qquad\forall u,v\in\R^{d}. \tag{3}
\]
In particular, $\prox_{\eta g}$ is non‑expansive:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|. \tag{4}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of PGD]\label{thm:main}
Let $\{x_{k}\}_{k\ge0}$ be generated by \eqref{1} under Assumptions~\ref{as:lip}--\ref{as:step}. Define the gradient mapping $\mathcal{G}_{\eta}(x_{k})$ as above. Then for any $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_{k})\|^{2}
\;\le\;
\frac{2\big(F(x_{0})-F^{*}\big)}{\eta K},\tag{5}
\]
where $F^{*}:=\inf_{x}F(x)$. Consequently,
\[
\min_{0\le k\le K-1}\|\mathcal{G}_{\eta}(x_{k})\|^{2}
\;\le\;
\frac{2\big(F(x_{0})-F^{*}\big)}{\eta K}=O\!\left(\frac{1}{K}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Descent.} Each iteration yields a non‑increase of the composite objective:
\[
F(x_{k+1})\le F(x_{k})-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_{k})\|^{2}. \tag{6}
\]
\item \textbf{Stability w.r.t. stepsize.} The bound \eqref{5} holds for any $\eta\in(0,1/L]$. Larger $\eta$ (up to $1/L$) yields a potentially faster decrease because the factor $\eta$ appears in the denominator of the right‑hand side.
\item \textbf{Special cases.}
\begin{enumerate}
\item If $g\equiv0$, then $\mathcal{G}_{\eta}(x)=\grad f(x)$ and \eqref{5} reduces to the classic $O(1/K)$ guarantee for gradient descent on $L$‑smooth (possibly non‑convex) functions.
\item If $f\equiv0$, the method becomes the proximal point algorithm for a convex $g$, and (6) reduces to the standard proximal point descent.
\end{enumerate}
\item \textbf{Comparison.} Compared with plain gradient descent on $f+g$ (which would require $g$ to be smooth), PGD enjoys the same $O(1/K)$ rate while handling a nonsmooth convex regularizer via the proximal step.
\end{itemize}

\section*{Discussion}
The theorem guarantees that the norm of the gradient mapping converges to zero at the rate $O(1/K)$. Since $\|\mathcal{G}_{\eta}(x)\|$ can be interpreted as a measure of stationarity for the composite problem, any limit point of $\{x_{k}\}$ is a \emph{critical point} satisfying
\[
0\in\grad f(\bar x)+\partial g(\bar x).
\]
The proof crucially uses the non‑expansiveness (3)–(4) of the proximal operator; without convexity of $g$ this property fails and the guarantee no longer holds.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see Assumption~\ref{as:lip}. \qedhere
\end{proof}

\begin{lemma}[Optimality condition of proximal step]\label{lem:prox-opt}
Let $u=\prox_{\eta g}(v)$. Then
\[
0\in\partial g(u)+\frac{1}{\eta}(u-v). \tag{7}
\]
Equivalently,
\[
\frac{1}{\eta}(v-u)\in\partial g(u). 
\]
\end{lemma}
\begin{proof}
By definition of the proximal operator, $u$ minimizes $g(z)+\frac{1}{2\eta}\|z-v\|^{2}$, hence the subgradient inclusion (7) follows from first‑order optimality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof fix $\eta\in(0,1/L]$.

\noindent\textbf{[Step 1]} (Apply Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$).  
\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \tag{8}
\]

\noindent\textbf{[Step 2]} (Rewrite $x_{k+1}$ using the proximal definition).  
From \eqref{1},
\[
x_{k+1}= \prox_{\eta g}\!\big(x_{k}-\eta\grad f(x_{k})\big).
\]
Define the intermediate point
\[
v_{k}:=x_{k}-\eta\grad f(x_{k}). \tag{9}
\]
Then $x_{k+1}= \prox_{\eta g}(v_{k})$.

\noindent\textbf{[Step 3]} (Use optimality condition Lemma~\ref{lem:prox-opt}).  
Applying Lemma~\ref{lem:prox-opt} with $u=x_{k+1}$ and $v=v_{k}$ gives
\[
\frac{1}{\eta}\big(v_{k}-x_{k+1}\big)\in\partial g(x_{k+1}). \tag{10}
\]

\noindent\textbf{[Step 4]} (Form the composite objective difference).  
Add $g(x_{k+1})$ to both sides of \eqref{8} and use convexity of $g$ together with (10).  
For any $s\in\partial g(x_{k+1})$ we have
\[
g(x_{k})\ge g(x_{k+1})+\langle s, x_{k}-x_{k+1}\rangle.
\]
Choosing $s=\frac{1}{\eta}(v_{k}-x_{k+1})$ from (10) yields
\[
g(x_{k})\ge g(x_{k+1})+\frac{1}{\eta}\langle v_{k}-x_{k+1},x_{k}-x_{k+1}\rangle. \tag{11}
\]

\noindent\textbf{[Step 5]} (Combine \eqref{8} and \eqref{11}).  
Adding the two inequalities gives
\[
\begin{aligned}
F(x_{k+1})
&=f(x_{k+1})+g(x_{k+1})\\
&\le f(x_{k})+g(x_{k})
+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&\qquad
+\frac{1}{\eta}\langle v_{k}-x_{k+1},x_{k}-x_{k+1}\rangle .
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 6]} (Express $v_{k}$ via \eqref{9}).  
Since $v_{k}=x_{k}-\eta\grad f(x_{k})$, we have
\[
v_{k}-x_{k+1}=x_{k}-\eta\grad f(x_{k})-x_{k+1}
=-(x_{k+1}-x_{k})-\eta\grad f(x_{k}). \tag{13}
\]

\noindent\textbf{[Step 7]} (Plug \eqref{13} into the inner product term of \eqref{12}).  
\[
\begin{aligned}
\frac{1}{\eta}\langle v_{k}-x_{k+1},x_{k}-x_{k+1}\rangle
&=\frac{1}{\eta}\langle -(x_{k+1}-x_{k})-\eta\grad f(x_{k}),x_{k}-x_{k+1}\rangle\\
&=-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
-\langle\grad f(x_{k}),x_{k}-x_{k+1}\rangle. \tag{14}
\end{aligned}
\]

\noindent\textbf{[Step 8]} (Insert (14) into (12)).  
The terms $\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle$ and $-\langle\grad f(x_{k}),x_{k}-x_{k+1}\rangle$ cancel, leaving
\[
F(x_{k+1})\le F(x_{k})+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}. \tag{15}
\]

\noindent\textbf{[Step 9]} (Use the stepsize condition $\eta\le 1/L$).  
Since $\eta\le 1/L$, we have $L\le 1/\eta$, thus
\[
\frac{L}{2}-\frac{1}{\eta}\le -\frac{1}{2\eta}.
\]
Applying this inequality to \eqref{15} yields the descent relation
\[
F(x_{k+1})\le F(x_{k})-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}. \tag{16}
\]

\noindent\textbf{[Step 10]} (Relate the step difference to the gradient mapping).  
From the definition of $\mathcal{G}_{\eta}$,
\[
\mathcal{G}_{\eta}(x_{k})=\frac{1}{\eta}\big(x_{k}-x_{k+1}\big)
\quad\Longrightarrow\quad
\|x_{k+1}-x_{k}\|^{2}= \eta^{2}\|\mathcal{G}_{\eta}(x_{k})\|^{2}. \tag{17}
\]

\noindent\textbf{[Step 11]} (Insert (17) into (16)).  
\[
F(x_{k+1})\le F(x_{k})-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_{k})\|^{2}. \tag{18}
\]
This is precisely the descent inequality (6).

\noindent\textbf{[Step 12]} (Sum over $k=0,\dots,K-1$).  
Telescoping the inequality \eqref{18} gives
\[
F(x_{K})\le F(x_{0})-\frac{\eta}{2}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_{k})\|^{2}. \tag{19}
\]

\noindent\textbf{[Step 13]} (Use $F(x_{K})\ge F^{*}$).  
Rearranging \eqref{19} yields
\[
\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_{k})\|^{2}
\le \frac{2\big(F(x_{0})-F^{*}\big)}{\eta}. \tag{20}
\]

\noindent\textbf{[Step 14]} (Divide by $K$ to obtain the average bound).  
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_{k})\|^{2}
\le\frac{2\big(F(x_{0})-F^{*}\big)}{\eta K}, 
\]
which is exactly \eqref{5}. \qed

\section*{Rate Derivation (With Constants)}
The constant in \eqref{5} is explicit:
\[
C:=\frac{2\big(F(x_{0})-F^{*}\big)}{\eta}.
\]
Hence the algorithm attains an $O(1/K)$ rate for the squared norm of the gradient mapping. If $F$ is bounded below and $\eta$ is chosen as the maximal admissible stepsize $1/L$, the constant becomes $2L\big(F(x_{0})-F^{*}\big)$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure gradient descent ($g\equiv0$).}  
$\mathcal{G}_{\eta}(x)=\grad f(x)$, and \eqref{5} reduces to the standard smooth non‑convex guarantee
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_{k})\|^{2}\le\frac{2\big(f(x_{0})-f^{*}\big)}{\eta K}.
\]

\item \textbf{Convex $f$, possibly nonsmooth $g$.}  
When $f$ is convex, the same proof yields a stronger result:
\[
F(x_{K})-F^{*}\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta K},
\]
i.e., an $O(1/K)$ rate in function suboptimality.

\item \textbf{Proximal point algorithm ($f\equiv0$).}  
The update becomes $x_{k+1}= \prox_{\eta g}(x_{k})$, and the descent inequality reduces to
\[
g(x_{k+1})\le g(x_{k})-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\]
Thus the same $O(1/K)$ bound holds for the distance between successive iterates.
\end{enumerate}
