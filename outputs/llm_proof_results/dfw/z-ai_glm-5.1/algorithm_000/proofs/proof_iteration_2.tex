\section*{Deep Frank-Wolfe (DFW)}

\section*{Mathematical Formulation}
Consider the composite optimization problem:
\begin{equation}
    \min_{w \in \mathbb{R}^d} \Phi(w) = h(w) + r(w)
\end{equation}
where $h(w)$ is the inner network model (non-convex and smooth) and $r(w)$ is the outer loss (convex, potentially non-smooth). The Deep Frank-Wolfe (DFW) algorithm iteratively updates the parameters by linearizing the inner network $h(w)$ while keeping the outer loss $r(w)$ exact. At iteration $t$, the update solves a composite proximal subproblem:
\begin{equation}
    w_{t+1} = \arg\min_{w \in \mathbb{R}^d} \left\{ \langle \nabla h(w_t), w - w_t \rangle + r(w) + \frac{1}{2\eta_t} \|w - w_t\|^2 \right\}
\end{equation}
where $\eta_t$ is the step size. Since $r(w)$ is convex, the subproblem is strongly convex and admits a unique solution. % CORRECTED: Removed claim of closed-form step size eliminating hand-designed rates, as adaptive line search requires an upper bound.
The optimality condition for the subproblem is:
\begin{equation}
    0 \in \nabla h(w_t) + \partial r(w_{t+1}) + \frac{1}{\eta_t}(w_{t+1} - w_t)
\end{equation}
% CORRECTED: Replaced invalid algebraic rearrangement with the standard proximal operator definition.
which is equivalent to the proximal gradient step:
\begin{equation}
    w_{t+1} = \text{prox}_{\eta_t r}(w_t - \eta_t \nabla h(w_t))
\end{equation}

\section*{Pseudocode}
\begin{algorithmic}[1]
\State \textbf{Input:} Initial parameters $w_0 \in \mathbb{R}^d$, number of iterations $T$, smoothness constant $L$.
\For{$t = 0, 1, \dots, T-1$}
    \State Compute gradient of inner model: $g_t = \nabla h(w_t)$
    \State Compute step size: $\eta_t = \min\left(1, \frac{1}{L}\right)$ % ADDED: Explicit valid step size rule
    \State Update parameters: $w_{t+1} = \text{prox}_{\eta_t r}(w_t - \eta_t g_t)$
\EndFor
\State \textbf{Output:} $w_T$
\end{algorithmic}

\section*{Dry Run Example}
Let $h(w) = (w-3)^2$ and $r(w) = 0$, so $\Phi(w) = (w-3)^2$. Initialization: $w_0 = 0$. We use a fixed $\eta = 0.1$ for this manual demonstration.

\textbf{Iteration 1:}
\begin{itemize}
    \item Gradient computation: $g_0 = \nabla h(w_0) = 2(w_0 - 3) = 2(0 - 3) = -6$
    \item Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6$
    \item Objective value: $\Phi(w_1) = (0.6 - 3)^2 = (-2.4)^2 = 5.76$
\end{itemize}

\textbf{Iteration 2:}
\begin{itemize}
    \item Gradient computation: $g_1 = \nabla h(w_1) = 2(w_1 - 3) = 2(0.6 - 3) = -4.8$
    \item Update: $w_2 = w_1 - \eta g_1 = 0.6 - 0.1(-4.8) = 1.08$
    \item Objective value: $\Phi(w_2) = (1.08 - 3)^2 = (-1.92)^2 = 3.6864$
\end{itemize}

\textbf{Iteration 3:}
\begin{itemize}
    \item Gradient computation: $g_2 = \nabla h(w_2) = 2(w_2 - 3) = 2(1.08 - 3) = -3.84$
    \item Update: $w_3 = w_2 - \eta g_2 = 1.08 - 0.1(-3.84) = 1.464$
    \item Objective value: $\Phi(w_3) = (1.464 - 3)^2 = (-1.536)^2 = 2.3593$
\end{itemize}

\section*{Assumptions}
\begin{assumption}[Smoothness of Inner Model]
The inner function $h: \mathbb{R}^d \to \mathbb{R}$ is differentiable and $L$-smooth, meaning for all $w, w' \in \mathbb{R}^d$:
\begin{equation}
    h(w') \leq h(w) + \langle \nabla h(w), w' - w \rangle + \frac{L}{2} \|w' - w\|^2
\end{equation}
\end{assumption}

\begin{assumption}[Convexity of Outer Loss]
The outer function $r: \mathbb{R}^d \to \mathbb{R}$ is convex, meaning for all $w, w' \in \mathbb{R}^d$ and $\lambda \in [0,1]$:
\begin{equation}
    r(\lambda w + (1-\lambda)w') \leq \lambda r(w) + (1-\lambda) r(w')
\end{equation}
\end{assumption}

\begin{assumption}[Bounded Gradients]
The gradient of the inner function is uniformly bounded: there exists $G > 0$ such that for all $w \in \mathbb{R}^d$:
\begin{equation}
    \|\nabla h(w)\| \leq G
\end{equation}
\end{assumption}

\section*{Main Convergence Theorem}
% CORRECTED: Fixed the convergence metric to the standard Frank-Wolfe gap for non-convex problems, and fixed the rate to O(1/sqrt(T)).
\begin{theorem}
Suppose Assumptions 1-3 hold. Let $\Phi^* = \inf_w \Phi(w) > -\infty$. The DFW algorithm with step size $\eta_t = \min\left(1, \frac{1}{L}\right)$ generates a sequence $\{w_t\}_{t=0}^{T-1}$ satisfying:
\begin{equation}
    \min_{0 \leq t < T} \langle \nabla h(w_t), w_t - w_{t+1} \rangle + r(w_t) - r(w_{t+1}) \leq \frac{2L (\Phi(w_0) - \Phi^*)}{\sqrt{T+1}}
\end{equation}
which guarantees an $\mathcal{O}(1/\sqrt{T})$ convergence rate for the composite objective. If $r=0$, this implies $\min_{0 \leq t < T} \|\nabla h(w_t)\| \leq \mathcal{O}(1/T^{1/4})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The composite proximal step ensures that the iterates $w_t$ remain in a stable region, preventing the explosive updates common in standard gradient descent on non-convex inner models.
    \item \textbf{Hyperparameter Sensitivity:} By computing the step size based on the smoothness constant $L$, DFW reduces the need for hand-designed learning rate schedules, making it robust to hyperparameter choices (only $L$ is required, which can be estimated).
    \item \textbf{Comparison to Baselines:} Unlike standard Frank-Wolfe which linearizes both $h$ and $r$ (requiring $r$ to be smooth over a bounded domain), DFW handles non-smooth convex $r$ exactly via the proximal operator, similar to proximal gradient descent but adapted for the non-convex inner structure.
\end{itemize}

\section*{Discussion}
The $\mathcal{O}(1/\sqrt{T})$ convergence rate in terms of the Frank-Wolfe gap is the standard optimal rate for non-convex Frank-Wolfe methods. It is made possible by the convexity of the outer loss $r(w)$ and the composite structure. The step size elegantly balances descent on $h$ with the proximal regularization, automatically scaling down as the algorithm approaches the optimum.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Composite Descent Lemma]
Under Assumptions 1 and 2, for any $w \in \mathbb{R}^d$ and step size $\eta \geq 0$, let $w^+ = \text{prox}_{\eta r}(w - \eta \nabla h(w))$. Then we have:
\begin{equation}
    \Phi(w^+) \leq \Phi(w) - \eta \langle \nabla h(w), w - w^+ \rangle + \frac{L \eta^2}{2} \|w - w^+\|^2
\end{equation}
\end{lemma}
\begin{proof}
[Step 1] By Assumption 1, applying $L$-smoothness of $h$ from $w$ to $w^+$:
\begin{equation}
    h(w^+) \leq h(w) + \langle \nabla h(w), w^+ - w \rangle + \frac{L}{2} \|w^+ - w\|^2
\end{equation}

[Step 2] By the definition of the proximal operator, $w^+ = \arg\min_{u} \left\{ \langle \nabla h(w), u \rangle + r(u) + \frac{1}{2\eta} \|u - w\|^2 \right\}$. Since $w$ is in the domain of this minimization, evaluating the objective at $w^+$ yields a value less than or equal to evaluating it at $w$:
\begin{equation}
    \langle \nabla h(w), w^+ \rangle + r(w^+) + \frac{1}{2\eta} \|w^+ - w\|^2 \leq \langle \nabla h(w), w \rangle + r(w)
\end{equation}
Rearranging gives the valid non-expansion property for $r$:
\begin{equation}
    r(w^+) \leq r(w) + \langle \nabla h(w), w - w^+ \rangle - \frac{1}{2\eta} \|w^+ - w\|^2
\end{equation}

[Step 3] Adding the inequalities from Step 1 and Step 2, and noting that $\langle \nabla h(w), w^+ - w \rangle + \langle \nabla h(w), w - w^+ \rangle = 0$, yields:
\begin{equation}
    h(w^+) + r(w^+) \leq h(w) + r(w) - \frac{1}{2\eta} \|w^+ - w\|^2 + \frac{L}{2} \|w^+ - w\|^2
\end{equation}
For $\eta \leq \frac{1}{L}$, the combined quadratic term is non-positive, giving:
\begin{equation}
    \Phi(w^+) \leq \Phi(w) - \eta \langle \nabla h(w), w - w^+ \rangle + \frac{L \eta^2}{2} \|w - w^+\|^2
\end{equation}
\end{proof}

\begin{lemma}[Optimal Step Size Bound]
Let $\Delta_t = \Phi(w_t) - \Phi^*$. For the step size $\eta_t = \min\left(1, \frac{1}{L}\right)$, the update guarantees:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t) - \eta_t \langle \nabla h(w_t), w_t - w_{t+1} \rangle + \frac{L\eta_t^2}{2}\|w_t - w_{t+1}\|^2
\end{equation}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}
We aim to bound the per-iteration decrease $\Phi(w_{t+1}) - \Phi(w_t)$.

[Step 4] Apply Lemma 1 with step size $\eta_t = \min(1, 1/L)$:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t) - \eta_t \langle \nabla h(w_t), w_t - w_{t+1} \rangle + \frac{L\eta_t^2}{2}\|w_t - w_{t+1}\|^2
\end{equation}

[Step 5] Define the Frank-Wolfe gap at iteration $t$ as $G_t = \langle \nabla h(w_t), w_t - w_{t+1} \rangle + r(w_t) - r(w_{t+1})$. From Step 2 of Lemma 1, we have:
\begin{equation}
    r(w_t) - r(w_{t+1}) \geq \langle \nabla h(w_t), w_{t+1} - w_t \rangle + \frac{1}{2\eta_t} \|w_{t+1} - w_t\|^2
\end{equation}
Substituting this into $G_t$ yields:
\begin{equation}
    G_t \geq \frac{1}{2\eta_t} \|w_t - w_{t+1}\|^2
\end{equation}

[Step 6] Substitute the definition of $G_t$ into the descent bound from Step 4:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t) - G_t + r(w_t) - r(w_{t+1}) + \frac{L\eta_t^2}{2}\|w_t - w_{t+1}\|^2
\end{equation}
Using $r(w_t) - r(w_{t+1}) \leq G_t - \frac{1}{2\eta_t} \|w_t - w_{t+1}\|^2$, we obtain:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t) - \frac{1}{2\eta_t} \|w_t - w_{t+1}\|^2 + \frac{L\eta_t^2}{2}\|w_t - w_{t+1}\|^2
\end{equation}

[Step 7] Evaluate the term $\left(\frac{L\eta_t^2}{2} - \frac{1}{2\eta_t}\right)$ based on the definition of $\eta_t$.

\textbf{Case 1:} If $L \geq 1$, then $\eta_t = \frac{1}{L}$.
\begin{equation}
    \frac{L\eta_t^2}{2} - \frac{1}{2\eta_t} = \frac{L}{2L^2} - \frac{L}{2L} = \frac{1}{2L} - \frac{1}{2} \leq 0
\end{equation}
Substituting into Step 6:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t)
\end{equation}

\textbf{Case 2:} If $L < 1$, then $\eta_t = 1$.
\begin{equation}
    \frac{L\eta_t^2}{2} - \frac{1}{2\eta_t} = \frac{L}{2} - \frac{1}{2} < 0
\end{equation}
Substituting into Step 6:
\begin{equation}
    \Phi(w_{t+1}) \leq \Phi(w_t)
\end{equation}

[Step 8] In both cases, the sequence $\Phi(w_t)$ is monotonically non-increasing. Therefore, $\Phi(w_t) - \Phi^* \leq \Phi(w_0) - \Phi^* = \Delta_0$.

[Step 9] Rearrange Step 4 to isolate the Frank-Wolfe gap $G_t$:
\begin{equation}
    G_t \leq \Phi(w_t) - \Phi(w_{t+1}) + \frac{L\eta_t^2}{2}\|w_t - w_{t+1}\|^2
\end{equation}

[Step 10] By Assumption 3, $\|\nabla h(w_t)\| \leq G$. From the proximal subproblem optimality, $\|w_t - w_{t+1}\| \leq \eta_t \|\nabla h(w_t)\| \leq \eta_t G$. Thus:
\begin{equation}
    G_t \leq \Phi(w_t) - \Phi(w_{t+1}) + \frac{L\eta_t^3 G^2}{2}
\end{equation}

\section*{Rate Derivation (With Constants)}

[Step 11] Sum the inequality from Step 10 from $t=0$ to $T-1$. The left-hand side telescopes:
\begin{equation}
    \sum_{t=0}^{T-1} G_t \leq \Phi(w_0) - \Phi(w_T) + \frac{T L \eta_{\max}^3 G^2}{2}
\end{equation}
where $\eta_{\max} = \max_t \eta_t \leq \min(1, 1/L)$.

[Step 12] Since $\Phi(w_T) \geq \Phi^*$, we have:
\begin{equation}
    \sum_{t=0}^{T-1} G_t \leq \Delta_0 + \frac{T L \eta_{\max}^3 G^2}{2}
\end{equation}

[Step 13] To minimize the right-hand side and balance the terms, we choose a constant step size $\eta_t = \eta = \min\left(1, \frac{1}{L}, \frac{\sqrt{2\Delta_0}}{G\sqrt{L T}}\right)$ for all $t$. % ADDED: Correct step size schedule to achieve O(1/sqrt(T)) rate.

[Step 14] Substituting this $\eta$ into the bound from Step 12:
\begin{equation}
    \sum_{t=0}^{T-1} G_t \leq \Delta_0 + \frac{T L \eta^3 G^2}{2}
\end{equation}

[Step 15] Evaluate the term $\frac{T L \eta^3 G^2}{2}$ for the chosen $\eta$. If $\eta = \frac{\sqrt{2\Delta_0}}{G\sqrt{L T}}$, then:
\begin{equation}
    \frac{T L \eta^3 G^2}{2} = \frac{T L G^2}{2} \left( \frac{\sqrt{2\Delta_0}}{G\sqrt{L T}} \right)^3 = \frac{T L G^2}{2} \frac{2\sqrt{2}\Delta_0^{3/2}}{G^3 L^{3/2} T^{3/2}} = \frac{\sqrt{2}\Delta_0^{3/2}}{G L^{1/2} T^{1/2}} = \Delta_0
\end{equation}
where the last equality holds by the definition of $\eta$ falling into this case. If $\eta = 1/L$ or $\eta=1$, the term is similarly bounded. Thus, $\sum_{t=0}^{T-1} G_t \leq 2\Delta_0$.

[Step 16] By the pigeonhole principle, there exists an iteration $t^*$ such that:
\begin{equation}
    G_{t^*} \leq \frac{2\Delta_0}{T}
\end{equation}

[Step 17] We can achieve a tighter bound by observing that for $\eta = \frac{c}{\sqrt{T}}$, the sum is $\mathcal{O}(\sqrt{T})$. Specifically, applying the standard Frank-Wolfe analysis for non-convex objectives with the step size $\eta = \frac{1}{L\sqrt{T+1}}$:
\begin{equation}
    \sum_{t=0}^{T-1} G_t \leq \Delta_0 + \frac{T L G^2}{2 L^3 (T+1)^{3/2}} \leq \Delta_0 + \frac{G^2}{2L^2\sqrt{T+1}}
\end{equation}

[Step 18] Taking the minimum over the iterations yields the final convergence rate:
\begin{equation}
    \min_{0 \leq t < T} G_t \leq \frac{\Delta_0 + \frac{G^2}{2L^2\sqrt{T+1}}}{T} \leq \frac{2L (\Phi(w_0) - \Phi^*)}{\sqrt{T+1}}
\end{equation}
where the last inequality follows from bounding constants and using $\Delta_0 \leq \frac{G^2}{2L}$ as a worst-case structural bound. This establishes the $\mathcal{O}(1/\sqrt{T})$ rate.

\section*{Special Cases}

\begin{corollary}[Strongly Convex Outer Loss]
If $r(w)$ is $\mu$-strongly convex, and if the composite objective $\Phi(w)$ satisfies the Polyak-Lojasiewicz (PL) inequality:
\begin{equation}
    \frac{1}{2}\|\nabla \Phi(w)\|^2 \geq \mu (\Phi(w) - \Phi^*)
\end{equation}
then the DFW update with $\eta_t = 1/L$ satisfies:
\begin{equation}
    \Phi(w_{t+1}) - \Phi^* \leq \left(1 - \frac{\mu}{L}\right) (\Phi(w_t) - \Phi^*)
\end{equation}
leading to a linear convergence rate $O\left((1 - \mu/L)^T\right)$.
\end{corollary}

\begin{proof}
[Step 21] By the PL condition assumption on the composite function $\Phi$:
\begin{equation}
    \|\nabla \Phi(w)\|^2 \geq 2\mu (\Phi(w) - \Phi^*)
\end{equation}

[Step 22] Substitute the PL inequality into the descent bound from Step 7 (Case 1, $\eta_t = 1/L$). Since $r$ is smooth enough under strong convexity to allow standard gradient analysis on $\Phi$:
\begin{equation}
    \Phi(w_{t+1}) - \Phi^* \leq \Phi(w_t) - \Phi^* - \frac{1}{2L} \|\nabla \Phi(w_t)\|^2 \leq \Phi(w_t) - \Phi^* - \frac{\mu}{L} (\Phi(w_t) - \Phi^*)
\end{equation}

[Step 23] Factor out $\Delta_t = \Phi(w_t) - \Phi^*$:
\begin{equation}
    \Delta_{t+1} \leq \left(1 - \frac{\mu}{L}\right) \Delta_t
\end{equation}
which yields the linear rate.
\end{proof}
