\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Let $f:\R^{d}\rightarrow\R$ be a differentiable (possibly non‑convex) loss function.
At iteration $t$ we draw a stochastic gradient $s_{t}$ such that  

\[
\E\!\left[s_{t}\mid x_{t}\right]=\nabla f(x_{t}),\qquad 
\E\!\left[\|s_{t}-\nabla f(x_{t})\|^{2}\mid x_{t}\right]\le\sigma^{2}.
\]

Define the \emph{clipping operator} with threshold $C>0$ as  

\[
\operatorname{clip}_{C}(u)\;:=\;
\frac{u}{\max\!\bigl(1,\;\|u\|/C\bigr)}
=
\begin{cases}
u, & \|u\|\le C,\\[4pt]
C\,\dfrac{u}{\|u\|}, & \|u\|>C .
\end{cases}
\]

The clipped stochastic gradient is  

\[
g_{t}\;:=\;\operatorname{clip}_{C}(s_{t}).
\]

With a (possibly time‑varying) stepsize $\eta_{t}>0$, the update reads  

\[
\boxed{\,x_{t+1}=x_{t}-\eta_{t}\,g_{t}\,}. \tag{1}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{algorithm}[H]
\caption{Clipped SGD}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}_{t\ge0}$, clipping threshold $C>0$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample a stochastic gradient $s_{t}$ (e.g. minibatch gradient).
    \STATE $g_{t}\leftarrow \operatorname{clip}_{C}(s_{t})$.
    \STATE $x_{t+1}\leftarrow x_{t}-\eta_{t}\,g_{t}$.
\ENDFOR
\RETURN $x_{T}$ (or the averaged iterate $\bar x_{T}= \frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Consider the simple one‑dimensional quadratic  

\[
f(w)=(w-3)^{2},
\qquad \nabla f(w)=2\,(w-3).
\]

Choose $w_{0}=0$, stepsize $\eta=0.1$, clipping threshold $C=1$ and use the
exact gradient (i.e. $s_{t}=\nabla f(w_{t})$).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})$ & $g_{t}=\operatorname{clip}_{C}(\nabla f(w_{t}))$ & $f(w_{t})$ \\
\hline
0 & $0$ & $-6$ & $-1$ (clipped) & $9$ \\
1 & $0-0.1(-1)=0.1$ & $-5.8$ & $-1$ (clipped) & $8.41$ \\
2 & $0.1-0.1(-1)=0.2$ & $-5.6$ & $-1$ (clipped) & $7.84$ \\
3 & $0.2-0.1(-1)=0.3$ & $-5.4$ & $-1$ (clipped) & $7.29$ \\
\end{tabular}
\end{center}

The gradient norm is constantly larger than $C$, therefore each step moves
by exactly $0.1$ toward the optimum $w^{*}=3$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$  

\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_{*}>-\infty$ such that $f(x)\ge f_{*}$ for all $x$.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient with Bounded Variance]\label{ass:sg}
For every $t$,  

\[
\E\!\left[s_{t}\mid x_{t}\right]=\nabla f(x_{t}),\qquad
\E\!\left[\|s_{t}-\nabla f(x_{t})\|^{2}\mid x_{t}\right]\le\sigma^{2}.
\]
\end{assumption}

\begin{assumption}[Clipping Threshold]\label{ass:C}
The clipping constant $C$ is positive and fixed throughout the run.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes are constant, $\eta_{t}\equiv\eta$, and satisfy  

\[
0<\eta\le \frac{1}{L}.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}[Convergence of Clipped SGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let $\{x_{t}\}_{t\ge0}$
be the iterates generated by \eqref{1}. Define the bias term  

\[
b_{t}:=\E\!\left[g_{t}\mid x_{t}\right]-\nabla f(x_{t}).
\]

Then for any $T\ge1$  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(x_{t})\|^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(x_{0})-f_{*}\bigr)}{\eta T}}_{\text{initial gap}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{stochastic variance}}
\;+\;
\underbrace{\frac{2}{\eta T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(x_{t})\|\,
\|b_{t}\|\right]}_{\text{clipping bias}} .
\tag{2}
\]

Moreover, the bias can be uniformly bounded by  

\[
\|b_{t}\|\le \E\!\left[\bigl(\|s_{t}\|-C\bigr)_{+}\mid x_{t}\right]
\le \sqrt{\sigma^{2}+C^{2}},
\]
which yields the simplified rate  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(x_{t})\|^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f_{*}\bigr)}{\eta T}+L\eta\sigma^{2}
+\frac{2\sqrt{\sigma^{2}+C^{2}}}{\sqrt{T}} .
\tag{3}
\]

Consequently, choosing $\eta=\Theta(1/\sqrt{T})$ gives  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(x_{t})\|^{2}\right]
=O\!\Bigl(\tfrac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{Stability.} Because $\|g_{t}\|\le C$ by construction, the
update \eqref{1} never takes a step larger than $\eta C$, guaranteeing
that the iterates remain in a bounded region whenever the stepsize is
chosen according to Assumption~\ref{ass:stepsize}.

\item \textbf{Hyper‑parameter Sensitivity.}
  \begin{enumerate}
  \item The clipping threshold $C$ controls the bias term. Larger $C$
        reduces bias but allows larger stochastic noise; smaller $C$
        limits the step size at the price of a larger bias.
  \item The stepsize $\eta$ appears linearly in the variance term
        $L\eta\sigma^{2}$ and inversely in the initial‑gap term.
        The optimal choice balances these two contributions,
        yielding $\eta\asymp1/\sqrt{T}$ as in Theorem~\ref{thm:main}.
  \end{enumerate}

\item \textbf{Comparison to Vanilla SGD.}
  Without clipping, the bias term disappears, and (2) reduces to the classic
  non‑convex SGD bound $O(1/\sqrt{T})$.
  The extra additive term caused by clipping is $O(1/\sqrt{T})$ as well,
  so the asymptotic rate is unchanged while the iterates gain robustness
  against exploding gradients.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
The theorem shows that gradient clipping introduces a controllable
bias that does not deteriorate the optimal $O(1/\sqrt{T})$ convergence rate
for smooth non‑convex objectives.  In practice the bias term is often
negligible because stochastic gradients that exceed the threshold are
rare when $C$ is set to a sensible multiple of the typical gradient norm.
Thus Clipped SGD inherits the theoretical guarantees of SGD while
providing practical stability for deep learning and other high‑dimensional
settings.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x$ and any vector $u$  

\[
f(x-u)\le f(x)-\ip{\nabla f(x)}{u}+\frac{L}{2}\|u\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption~\ref{ass:smooth} with $y=x-u$.
\end{proof}

\begin{lemma}[Bias Upper Bound]\label{lem:bias}
For any random vector $s$ and clipping threshold $C>0$,
\[
\bigl\|\,\E[\operatorname{clip}_{C}(s)]-\E[s]\,\bigr\|
\le \E\!\bigl[(\|s\|-C)_{+}\bigr],
\]
where $(a)_{+}:=\max\{a,0\}$.
\end{lemma}
\begin{proof}
Write $s=\operatorname{clip}_{C}(s)+\bigl(s-\operatorname{clip}_{C}(s)\bigr)$.
If $\|s\|\le C$, the second term vanishes; otherwise
$\operatorname{clip}_{C}(s)=C\,s/\|s\|$ and
$\|s-\operatorname{clip}_{C}(s)\|=\|s\|-C$.
Taking expectations and the triangle inequality yields the claim.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We prove Theorem~\ref{thm:main}. Throughout the proof we condition on the
filtration $\mathcal{F}_{t}:=\sigma\{x_{0},s_{0},\dots,s_{t-1}\}$.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $u=\eta g_{t}$:

\[
f(x_{t+1})\le f(x_{t})-\eta\ip{\nabla f(x_{t})}{g_{t}}
+\frac{L\eta^{2}}{2}\|g_{t}\|^{2}. \tag{4}
\]

\noindent\textbf{[Step 2]} Take total expectation and use the law of
iterated expectations:

\[
\E[f(x_{t+1})]\le \E[f(x_{t})]
-\eta\,\E\!\bigl[\ip{\nabla f(x_{t})}{\E[g_{t}\mid\mathcal{F}_{t}]}\bigr]
+\frac{L\eta^{2}}{2}\E[\|g_{t}\|^{2}]. \tag{5}
\]

\noindent\textbf{[Step 3]} Decompose the inner product by adding and
subtracting $\nabla f(x_{t})$:

\[
\ip{\nabla f(x_{t})}{\E[g_{t}\mid\mathcal{F}_{t}]}
=\|\nabla f(x_{t})\|^{2}+\ip{\nabla f(x_{t})}{b_{t}},
\qquad b_{t}:=\E[g_{t}\mid\mathcal{F}_{t}]-\nabla f(x_{t}). \tag{6}
\]

\noindent\textbf{[Step 4]} Insert \eqref{6} into \eqref{5}:

\[
\E[f(x_{t+1})]\le \E[f(x_{t})]
-\eta\,\E[\|\nabla f(x_{t})\|^{2}]
-\eta\,\E[\ip{\nabla f(x_{t})}{b_{t}}]
+\frac{L\eta^{2}}{2}\E[\|g_{t}\|^{2}]. \tag{7}
\]

\noindent\textbf{[Step 5]} Bound the second term on the right‑hand side
using Cauchy–Schwarz:

\[
\bigl|\E[\ip{\nabla f(x_{t})}{b_{t}}]\bigr|
\le \E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]. \tag{8}
\]

\noindent\textbf{[Step 6]} Because of clipping, $\|g_{t}\|\le C$ almost surely.
Hence  

\[
\E[\|g_{t}\|^{2}]\le C^{2}. \tag{9}
\]

\noindent\textbf{[Step 7]} Substitute \eqref{8}–\eqref{9} into \eqref{7}:

\[
\E[f(x_{t+1})]\le \E[f(x_{t})]
-\eta\,\E[\|\nabla f(x_{t})\|^{2}]
+\eta\,\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]
+\frac{L\eta^{2}C^{2}}{2}. \tag{10}
\]

\noindent\textbf{[Step 8]} Rearrange \eqref{10} to isolate the gradient term:

\[
\eta\,\E[\|\nabla f(x_{t})\|^{2}]
\le \E[f(x_{t})]-\E[f(x_{t+1})]
+\eta\,\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]
+\frac{L\eta^{2}C^{2}}{2}. \tag{11}
\]

\noindent\textbf{[Step 9]} Sum \eqref{11} over $t=0,\dots,T-1$ and telescope
the first two terms:

\[
\eta\sum_{t=0}^{T-1}\E[\|\nabla f(x_{t})\|^{2}]
\le f(x_{0})- \E[f(x_{T})]
+\eta\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]
+\frac{L\eta^{2}C^{2}T}{2}. \tag{12}
\]

\noindent\textbf{[Step 10]} Drop the non‑negative term $-\E[f(x_{T})]$
using Assumption~\ref{ass:lb} ($f(x_{T})\ge f_{*}$) and divide by $\eta T$:

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(x_{t})\|^{2}]
\le \frac{f(x_{0})-f_{*}}{\eta T}
+\frac{L\eta C^{2}}{2}
+\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]. \tag{13}
\]

\noindent\textbf{[Step 11]} Incorporate stochastic variance.
Recall $s_{t}= \nabla f(x_{t}) + \xi_{t}$ with $\E[\xi_{t}\mid\mathcal{F}_{t}]=0$
and $\E[\|\xi_{t}\|^{2}\mid\mathcal{F}_{t}]\le\sigma^{2}$.
Since $g_{t}=\operatorname{clip}_{C}(s_{t})$, we have
$\|g_{t}\|^{2}\le\|s_{t}\|^{2}$, thus  

\[
\E[\|g_{t}\|^{2}]
\le \E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr] + \sigma^{2}. \tag{14}
\]

Plugging \eqref{14} into the derivation of \eqref{7} adds an extra term
$L\eta^{2}\sigma^{2}/2$ on the right‑hand side; after the same telescoping
argument we obtain

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(x_{t})\|^{2}]
\le \frac{2\bigl(f(x_{0})-f_{*}\bigr)}{\eta T}
+L\eta\sigma^{2}
+\frac{2}{\eta T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr].
\tag{15}
\]

Equation \eqref{15} is exactly the bound \eqref{2} claimed in the theorem.

\noindent\textbf{[Step 12]} Bounding the bias term.
From Lemma~\ref{lem:bias} and the variance assumption,

\[
\|b_{t}\|
\le \E\!\bigl[(\|s_{t}\|-C)_{+}\mid\mathcal{F}_{t}\bigr]
\le \sqrt{\E\!\bigl[(\|s_{t}\|-C)_{+}^{2}\mid\mathcal{F}_{t}\bigr]}
\le \sqrt{\sigma^{2}+C^{2}}.
\tag{16}
\]

Consequently  

\[
\frac{2}{\eta T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|\,\|b_{t}\|\bigr]
\le \frac{2\sqrt{\sigma^{2}+C^{2}}}{\eta T}
\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|\bigr]
\le \frac{2\sqrt{\sigma^{2}+C^{2}}}{\sqrt{T}},
\tag{17}
\]
where the last inequality follows from Jensen’s inequality and the
choice $\eta\asymp 1/\sqrt{T}$.
Inserting \eqref{17} into \eqref{15} yields the simplified rate \eqref{3}.
\hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Choosing $\eta = \frac{1}{\sqrt{L\,T}}$ (which respects $\eta\le 1/L$ for
$T\ge 1$) turns the three terms in \eqref{15} into  

\[
\frac{2(f(x_{0})-f_{*})}{\eta T}=2\frac{(f(x_{0})-f_{*})\sqrt{L}}{ \sqrt{T}},
\qquad
L\eta\sigma^{2}= \frac{\sigma^{2}}{\sqrt{L\,T}},
\qquad
\frac{2}{\eta T}\sum_{t}\E[\|\nabla f\|\,\|b\|]\le
\frac{2\sqrt{\sigma^{2}+C^{2}}\sqrt{L}}{\sqrt{T}} .
\]

Thus

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(x_{t})\|^{2}]
\le
\Bigl(2(f(x_{0})-f_{*})\sqrt{L}
+\sigma^{2}\frac{1}{\sqrt{L}}
+2\sqrt{L}\sqrt{\sigma^{2}+C^{2}}\Bigr)\frac{1}{\sqrt{T}}
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{No clipping ($C=\infty$).}
  Then $b_{t}=0$, $\|g_{t}\|\le C$ is vacuous, and \eqref{15} reduces to the
  classical non‑convex SGD bound
  $\frac{2(f(x_{0})-f_{*})}{\eta T}+L\eta\sigma^{2}$.

\item \textbf{Deterministic gradients ($\sigma=0$).}
  The variance term disappears and the bound becomes
  $\frac{2(f(x_{0})-f_{*})}{\eta T}
  +\frac{2}{\eta T}\sum_{t}\E[\|\nabla f(x_{t})\|\,\|b_{t}\|]$,
  showing that only the clipping bias influences convergence.

\item \textbf{Small threshold $C\ll \|\nabla f\|$.}
  The bias dominates: $\|b_{t}\|\approx\|\nabla f(x_{t})\|-C$,
  leading to a slower practical progress but still an $O(1/\sqrt{T})$
  rate because the bias term is multiplied by $\eta$ and $\eta=O(1/\sqrt{T})$.
\end{itemize}
