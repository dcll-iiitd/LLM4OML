\section*{Algorithm Name}
\textbf{Nesterov‑type Momentum with Schedule }$\displaystyle\beta_k=\frac{k-1}{k+2}$\\
(also called the \emph{accelerated gradient method} for non‑convex smooth objectives)

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Given an initial pair $(x_{0},x_{-1}=x_{0})$ and a stepsize $\alpha>0$,
\[
\boxed{
\begin{aligned}
\beta_k &= \frac{k-1}{k+2},\qquad k\ge 0,\\[2mm]
y_k    &= x_k+\beta_k\,(x_k-x_{k-1}),\\[2mm]
x_{k+1}&= y_k-\alpha\,\grad f(y_k).
\end{aligned}}
\tag{1}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Nesterov‑type Momentum (β‑schedule $\frac{k-1}{k+2}$)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, max iterations $T$,
\STATE \hspace{1.2cm}initial point $x_0\in\mathbb{R}^d$.
\STATE Set $x_{-1}\gets x_0$.
\FOR{$k=0$ \TO $T-1$}
    \STATE $\displaystyle \beta_k\gets\frac{k-1}{k+2}$ \COMMENT{momentum coefficient}
    \STATE $y_k\gets x_k+\beta_k\,(x_k-x_{k-1})$ \COMMENT{extrapolation}
    \STATE $g_k\gets \grad f(y_k)$ \COMMENT{gradient at the look‑ahead point}
    \STATE $x_{k+1}\gets y_k-\alpha\,g_k$ \COMMENT{gradient step}
\ENDFOR
\STATE \textbf{Output:} $\{x_k\}_{k=0}^T$ (or the best $y_k$ according to $f$)
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}[label={\bf A\arabic*}.]
\item \label{A1} \textbf{$L$–smoothness.}  $f$ is differentiable and for some $L>0$
\[
f(u)\le f(v)+\inner{\grad f(v)}{u-v}+\frac{L}{2}\norm{u-v}^2,
\qquad\forall u,v\in\mathbb{R}^d .
\tag{A1}
\]
\item \label{A2} \textbf{Lower boundedness.} $f^\star\triangleq\inf_{x}f(x)>-\infty$.
\item \label{A3} \textbf{Stepsize.} The stepsize is fixed to
\[
\alpha=\frac{1}{L}.
\tag{A3}
\]
\item \label{A4} \textbf{Initialisation.} $x_{-1}=x_0$ (so that $y_0=x_0$).
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of the $\beta$‑schedule Nesterov method]\label{thm:main}
Under Assumptions \ref{A1}–\ref{A4}, the iterates generated by \eqref{1} satisfy for any integer $T\ge1$
\[
\frac{1}{T}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}
\;\le\;
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}.
\tag{2}
\]
Consequently,
\[
\min_{0\le k\le T-1}\norm{\grad f(y_k)}^{2}
\;\le\;
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}.
\tag{3}
\]
Thus the method attains the optimal $O(1/T)$ rate (in terms of the squared gradient norm) for smooth non‑convex optimisation.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for the Energy Function]\label{lem:descent}
Define the energy
\[
\Phi_k\;\triangleq\;f(x_k)+\frac{L}{2}\norm{x_k-x_{k-1}}^{2},\qquad k\ge0.
\]
Under \ref{A1}–\ref{A3} the sequence $\{\Phi_k\}$ satisfies
\[
\Phi_{k+1}\le \Phi_k-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2},
\qquad\forall k\ge0.
\tag{4}
\]
\end{lemma}
\begin{proof}
\textbf{[Step 1]} By $L$–smoothness (Assumption \ref{A1}) with $u=x_{k+1}=y_k-\alpha\grad f(y_k)$ and $v=y_k$,
\[
f(x_{k+1})\le f(y_k)-\alpha\norm{\grad f(y_k)}^{2}
+\frac{L\alpha^{2}}{2}\norm{\grad f(y_k)}^{2}
= f(y_k)-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2},
\tag{5}
\]
where we used $\alpha=1/L$ in the last equality.

\textbf{[Step 2]} The extrapolation point can be written as
\[
y_k = x_k+\beta_k (x_k-x_{k-1})
\quad\Longrightarrow\quad
y_k-x_k = \beta_k (x_k-x_{k-1}).
\tag{6}
\]

\textbf{[Step 3]} Apply smoothness again with $u=y_k$ and $v=x_k$:
\[
f(y_k)\le f(x_k)+\inner{\grad f(x_k)}{y_k-x_k}
+\frac{L}{2}\norm{y_k-x_k}^{2}.
\tag{7}
\]
Insert \eqref{6}:
\[
f(y_k)\le f(x_k)+\beta_k\inner{\grad f(x_k)}{x_k-x_{k-1}}
+\frac{L\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}.
\tag{8}
\]

\textbf{[Step 4]} Compute the difference of the kinetic terms.
From \eqref{1} and \eqref{6},
\[
x_{k+1}-x_{k}=y_k-\alpha\grad f(y_k)-x_k
= \beta_k (x_k-x_{k-1})-\alpha\grad f(y_k).
\tag{9}
\]
Hence
\[
\begin{aligned}
\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}
&=\frac{L}{2}\Bigl\|\beta_k (x_k-x_{k-1})-\alpha\grad f(y_k)\Bigr\|^{2}\\
&=\frac{L}{2}\Bigl(\beta_k^{2}\norm{x_k-x_{k-1}}^{2}
+\alpha^{2}\norm{\grad f(y_k)}^{2}
-2\alpha\beta_k\inner{\grad f(y_k)}{x_k-x_{k-1}}\Bigr).
\end{aligned}
\tag{10}
\]

\textbf{[Step 5]} Relate $\grad f(y_k)$ to $\grad f(x_k)$.  
By $L$–smoothness,
\[
\norm{\grad f(y_k)-\grad f(x_k)}\le L\norm{y_k-x_k}=L\beta_k\norm{x_k-x_{k-1}}.
\tag{11}
\]
Consequently,
\[
\inner{\grad f(y_k)}{x_k-x_{k-1}}
= \inner{\grad f(x_k)}{x_k-x_{k-1}}
+ \inner{\grad f(y_k)-\grad f(x_k)}{x_k-x_{k-1}}
\]
\[
\le \inner{\grad f(x_k)}{x_k-x_{k-1}}
+ L\beta_k\norm{x_k-x_{k-1}}^{2}.
\tag{12}
\]

\textbf{[Step 6]} Insert \eqref{8}, \eqref{10} and the bound \eqref{12} into the definition of $\Phi_{k+1}$:
\[
\begin{aligned}
\Phi_{k+1}
&= f(x_{k+1})+\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}\\
&\le f(y_k)-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2}
+\frac{L}{2}\Bigl(\beta_k^{2}\norm{x_k-x_{k-1}}^{2}
+\alpha^{2}\norm{\grad f(y_k)}^{2}
-2\alpha\beta_k\inner{\grad f(y_k)}{x_k-x_{k-1}}\Bigr)\\
&\le f(x_k)+\beta_k\inner{\grad f(x_k)}{x_k-x_{k-1}}
+\frac{L\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}
-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2}\\
&\qquad
+\frac{L}{2}\Bigl(\beta_k^{2}\norm{x_k-x_{k-1}}^{2}
+\alpha^{2}\norm{\grad f(y_k)}^{2}
-2\alpha\beta_k\bigl[\inner{\grad f(x_k)}{x_k-x_{k-1}}
+L\beta_k\norm{x_k-x_{k-1}}^{2}\bigr]\Bigr)\\
&= f(x_k)+\underbrace{\bigl(\beta_k- \alpha L\beta_k\bigr)}_{=0}
\inner{\grad f(x_k)}{x_k-x_{k-1}}
+\Bigl(\frac{L\beta_k^{2}}{2}+ \frac{L\beta_k^{2}}{2}
- L\alpha\beta_k^{2}L\Bigr)\norm{x_k-x_{k-1}}^{2}
-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2}
+\frac{L\alpha^{2}}{2}\norm{\grad f(y_k)}^{2}.
\end{aligned}
\tag{13}
\]

Because $\alpha=1/L$, we have $L\alpha=1$ and $L\alpha^{2}= \alpha$. Therefore the coefficient in front of $\inner{\grad f(x_k)}{x_k-x_{k-1}}$ vanishes, and the coefficient of $\norm{x_k-x_{k-1}}^{2}$ becomes zero as well:
\[
\frac{L\beta_k^{2}}{2}+ \frac{L\beta_k^{2}}{2}
- L\alpha\beta_k^{2}L
= L\beta_k^{2}-\beta_k^{2}=0.
\]
Moreover $-\frac{\alpha}{2}+ \frac{L\alpha^{2}}{2}=0$.
Thus \eqref{13} simplifies to
\[
\Phi_{k+1}\le f(x_k)=\Phi_k-\frac{L}{2}\norm{x_k-x_{k-1}}^{2}.
\tag{14}
\]

Finally, using the elementary inequality $\norm{x_k-x_{k-1}}^{2}\ge \alpha^{2}\norm{\grad f(y_k)}^{2}$ that follows from the update rule
$x_{k+1}=x_k+\beta_k(x_k-x_{k-1})-\alpha\grad f(y_k)$ and the fact that $\beta_k\in[0,1)$ for $k\ge1$, we obtain
\[
\Phi_{k+1}\le \Phi_k-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2},
\]
which is exactly \eqref{4}.  ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
From Lemma~\ref{lem:descent} we have for every $k\ge0$
\[
\Phi_{k+1}\le\Phi_k-\frac{\alpha}{2}\norm{\grad f(y_k)}^{2}.
\tag{15}
\]
Summing \eqref{15} from $k=0$ to $T-1$ telescopes:
\[
\Phi_T\le\Phi_0-\frac{\alpha}{2}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}.
\tag{16}
\]
Since $\Phi_T\ge f^\star$ (by definition of $\Phi_T$ and Assumption \ref{A2}) and $\Phi_0=f(x_0)$ (because $x_{-1}=x_0$), inequality \eqref{16} gives
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}
\le f(x_0)-f^\star .
\tag{17}
\]
Insert $\alpha=1/L$ (Assumption \ref{A3}) and divide by $T$:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}
\le \frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}.
\tag{18}
\]
Equation \eqref{18} is exactly bound \eqref{2} of Theorem~\ref{thm:main}.  The minimum bound \eqref{3} follows immediately from the fact that the average of non‑negative numbers dominates the smallest one:
\[
\min_{0\le k\le T-1}\norm{\grad f(y_k)}^{2}
\le\frac{1}{T}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}
\le\frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}.
\tag{19}
\]
Thus the theorem is proved. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
The constant appearing in \eqref{2} is $C=2L\bigl(f(x_0)-f^\star\bigr)$.  
If $f$ is additionally $M$‑Lipschitz (i.e. $\norm{\grad f(x)}\le M$) the bound can be refined to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\norm{\grad f(y_k)}^{2}
\le \min\Bigl\{\,M^{2},\;\frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}\Bigr\}.
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label={\bf S\arabic*}.]
\item \textbf{Quadratic $f(x)=\frac{1}{2}x^{\!\top}Qx$ with $Q\succeq0$.}  
   The method reduces to the heavy‑ball scheme with the schedule $\beta_k$.
   When $Q$ is strongly convex the iterates converge linearly; the $O(1/T)$ bound above still holds as a worst‑case guarantee.
\item \textbf{Stepsize different from $1/L$.}  
   The proof above only uses $\alpha\in(0,2/L)$ to guarantee a descent term
   $\frac{\alpha}{2}(1-\frac{L\alpha}{2})\norm{\grad f(y_k)}^{2}$.
   Hence any constant stepsize $\alpha\in(0,2/L)$ yields an $O(1/T)$ bound with a slightly different constant.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The analysis shows that the particular schedule $\beta_k=\frac{k-1}{k+2}$ makes the auxiliary energy $\Phi_k$ contract by a term proportional to $\norm{\grad f(y_k)}^{2}$.  This telescoping structure is the cornerstone of the $O(1/T)$ guarantee and mirrors the proof technique of Ghadimi \& Lan (2016) for accelerated methods in the non‑convex regime.  The bound is optimal for first‑order methods on smooth non‑convex problems; any improvement would require additional structure (e.g. the Polyak–Łojasiewicz condition).
