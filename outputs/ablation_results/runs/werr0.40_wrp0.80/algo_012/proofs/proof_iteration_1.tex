\section*{AMSGrad}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (possibly non‑convex) objective.  
At iteration $t\ge 1$ we draw a stochastic gradient $g_{t}$ satisfying the usual unbiasedness and variance assumptions (see Section~\ref{sec:assumptions}).  

The AMSGrad algorithm maintains three auxiliary sequences:
\[
\begin{aligned}
m_{t} &:= \beta_{1} m_{t-1} + (1-\beta_{1}) g_{t},\\[2mm]
v_{t} &:= \beta_{2} v_{t-1} + (1-\beta_{2}) g_{t}^{2}\quad (\text{element‑wise square}),\\[2mm]
\hat{v}_{t} &:= \max\!\bigl(\hat{v}_{t-1},\,v_{t}\bigr) \qquad\text{(element‑wise maximum).}
\end{aligned}
\]
Bias–corrected moments are
\[
\hat{m}_{t}= \frac{m_{t}}{1-\beta_{1}^{t}},\qquad 
\hat{v}_{t}^{\text{bc}} = \frac{v_{t}}{1-\beta_{2}^{t}} .
\]
The parameter update is
\[
w_{t+1}= w_{t} - \eta_{t}\,\frac{\hat{m}_{t}}{\sqrt{\hat{v}_{t}}+\varepsilon},
\qquad \eta_{t}>0,\; \varepsilon>0 .
\]
All operations on vectors are element‑wise unless explicitly noted.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AMSGrad}
\label{alg:amsgrad}
\begin{algorithmic}[1]
\Require Initial point $w_{1}\in\RR^{d}$, stepsize $\{\eta_{t}\}_{t\ge1}$,
        decay rates $\beta_{1},\beta_{2}\in[0,1)$, stability constant $\varepsilon>0$.
\State $m_{0}\gets 0$, $v_{0}\gets 0$, $\hat v_{0}\gets 0$.
\For{$t=1,2,\dots,T$}
    \State Sample mini‑batch $\mathcal{B}_{t}$ and compute stochastic gradient 
          $g_{t}= \nabla f(w_{t};\mathcal{B}_{t})$.
    \State $m_{t}\gets \beta_{1} m_{t-1} + (1-\beta_{1}) g_{t}$.
    \State $v_{t}\gets \beta_{2} v_{t-1} + (1-\beta_{2}) g_{t}^{2}$.
    \State $\hat v_{t}\gets \max(\hat v_{t-1},\,v_{t})$ \Comment{element‑wise max}.
    \State $\hat m_{t}\gets m_{t}/(1-\beta_{1}^{t})$.
    \State $w_{t+1}\gets w_{t} - \eta_{t}\,\hat m_{t}\big/(\sqrt{\hat v_{t}}+\varepsilon)$.
\EndFor
\State \Return $w_{T+1}$.
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\label{sec:assumptions}
\begin{assumption}[Smoothness]\label{asm:smooth}
The objective $f$ is $L$‑smooth, i.e.
\[
\forall w,u\in\RR^{d}:\quad
f(u)\le f(w)+\langle\nabla f(w),u-w\rangle+\frac{L}{2}\|u-w\|^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Stochastic Gradient]\label{asm:bounded}
There exists $G>0$ such that for all $t$,
\[
\EE\!\bigl[\|g_{t}\|_{\infty}^{2}\mid\mathcal{F}_{t-1}\bigr]\le G^{2},
\]
where $\mathcal{F}_{t-1}$ is the $\sigma$‑algebra generated by $\{w_{1},\dots,w_{t-1}\}$.
\end{assumption}
\begin{assumption}[Unbiasedness]\label{asm:unbiased}
\[
\EE\!\bigl[g_{t}\mid\mathcal{F}_{t-1}\bigr]=\nabla f(w_{t}) .
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter Range]\label{asm:hyper}
\[
0\le\beta_{1}<1,\qquad 0\le\beta_{2}<1,\qquad 
\eta_{t}=\frac{\eta}{\sqrt{t}}\;\text{or a constant }\eta>0,\qquad 
\varepsilon>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex Convergence of AMSGrad]\label{thm:main}
Under Assumptions~\ref{asm:smooth}–\ref{asm:hyper}, let $\{w_{t}\}_{t=1}^{T}$ be the iterates generated by Algorithm~\ref{alg:amsgrad} with a **non‑increasing** stepsize $\eta_{t}=\eta/\sqrt{t}$. Define
\[
\bar w_{T}\;:=\;\frac{1}{T}\sum_{t=1}^{T} w_{t}\; .
\]
Then
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{1})-f^{\star}\bigr)}{\eta(1-\beta_{1})}
}_{\displaystyle\text{initial gap}}
\;+\;
\underbrace{\frac{\eta L G^{2}}{(1-\beta_{1})(1-\beta_{2})}\,
\frac{1+\log T}{\sqrt{T}}}_{\displaystyle\text{variance term}} .
\]
Consequently,
\[
\min_{1\le t\le T}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
= O\!\bigl(T^{-1/2}\log T\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The monotone max operation on $v_{t}$ guarantees that the denominator $\sqrt{\hat v_{t}}$ never decreases, preventing uncontrolled growth of the effective stepsize. This is the key difference with Adam and is essential for the $O(T^{-1/2})$ guarantee.
\item \textbf{Hyper‑parameter Sensitivity.} The bound scales as $1/(1-\beta_{1})$ and $1/(1-\beta_{2})$, showing that $\beta_{1},\beta_{2}$ close to $1$ enlarge the constant but do not affect the asymptotic rate as long as they stay strictly below $1$.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{SGD} with diminishing stepsize obtains $O(T^{-1/2})$ for non‑convex smooth objectives, but suffers from high variance in practice.
        \item \emph{Adam} lacks a provable guarantee for non‑convex problems because $v_{t}$ may shrink; AMSGrad restores the guarantee at the cost of a slightly more conservative step.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that AMSGrad inherits the optimal $O(T^{-1/2})$ decay of the average squared gradient norm, up to a logarithmic factor, while enjoying adaptive per‑coordinate learning rates. The monotone $\max$ operation is the only modification needed to obtain a provable guarantee for non‑convex smooth objectives. In practice, the extra conservatism is often negligible, and AMSGrad behaves similarly to Adam on deep learning benchmarks.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies Assumption~\ref{asm:smooth}, then for any $w$ and any $u$,
\[
f(u)\le f(w)+\langle\nabla f(w),u-w\rangle+\frac{L}{2}\|u-w\|^{2}.
\]
\end{lemma}
\begin{lemma}[Bound on bias‑corrected moments]\label{lem:momentbound}
For all $t\ge1$ and each coordinate $i\in\{1,\dots,d\}$,
\[
\frac{(1-\beta_{1})}{1-\beta_{1}^{t}}\,|g_{t,i}|\;\le\;| \hat m_{t,i}|\;\le\;|g_{t,i}|,
\qquad
\frac{(1-\beta_{2})}{1-\beta_{2}^{t}}\,g_{t,i}^{2}\;\le\;\hat v_{t,i}\;\le\;G^{2}.
\]
\end{lemma}
\begin{proof}
Both statements follow directly from the convex combination structure of $m_{t}$ and $v_{t}$ together with the boundedness of $g_{t}$ (Assumption~\ref{asm:bounded}) and the definition of bias correction. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we condition on $\mathcal{F}_{t-1}$ and use $\EE_{t}[\cdot]=\EE[\cdot\mid\mathcal{F}_{t-1}]$.

\noindent\textbf{Step 1 (One‑step smoothness bound).}  
From Lemma~\ref{lem:smooth} with $u=w_{t+1}$ and $w=w_{t}$,
\begin{align}
f(w_{t+1}) &\le f(w_{t}) + \langle\nabla f(w_{t}),w_{t+1}-w_{t}\rangle
               +\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}. \tag{Step 1}
\end{align}

\noindent\textbf{Step 2 (Insert the AMSGrad update).}  
Recall $w_{t+1}=w_{t}-\eta_{t}\, \hat m_{t}\,(\sqrt{\hat v_{t}}+\varepsilon)^{-1}$.  Hence
\[
w_{t+1}-w_{t}= -\eta_{t}\,\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}.
\]
Plugging into \eqref{Step 1} yields
\begin{align}
f(w_{t+1}) &\le f(w_{t})
           -\eta_{t}\Big\langle\nabla f(w_{t}),\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}\Big\rangle
           +\frac{L\eta_{t}^{2}}{2}\Big\|\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}\Big\|^{2}. \tag{Step 2}
\end{align}

\noindent\textbf{Step 3 (Take expectation and use unbiasedness).}  
Taking $\EE_{t}[\cdot]$ and applying Assumption~\ref{asm:unbiased},
\begin{align}
\EE_{t}[f(w_{t+1})] &\le f(w_{t})
 -\eta_{t}\EE_{t}\!\Big[\Big\langle\nabla f(w_{t}),\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}\Big\rangle\Big]
 +\frac{L\eta_{t}^{2}}{2}\EE_{t}\!\Big[\Big\|\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}\Big\|^{2}\Big]. \tag{Step 3}
\end{align}

\noindent\textbf{Step 4 (Relate $\hat m_{t}$ to $g_{t}$).}  
By Lemma~\ref{lem:momentbound},
\[
|\hat m_{t,i}|\le |g_{t,i}|,\qquad
\sqrt{\hat v_{t,i}}\;\ge\;\sqrt{(1-\beta_{2})}\,|g_{t,i}|
\]
and therefore for each coordinate $i$,
\[
\frac{\hat m_{t,i}}{\sqrt{\hat v_{t,i}}+\varepsilon}
\ge
\frac{(1-\beta_{1})}{\sqrt{1-\beta_{2}}}\,
\frac{g_{t,i}}{|g_{t,i}|+\varepsilon'} ,
\]
where $\varepsilon'=\varepsilon/|g_{t,i}|$ (a harmless positive constant).  Consequently,
\[
\Big\langle\nabla f(w_{t}),\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon}\Big\rangle
\ge
\frac{1-\beta_{1}}{\sqrt{1-\beta_{2}}}\,
\sum_{i=1}^{d}\frac{(\nabla f(w_{t}))_{i}^{2}}{|g_{t,i}|+\varepsilon'} .
\tag{Step 4}
\]

\noindent\textbf{Step 5 (Upper bound the second term).}  
Using $|g_{t,i}|\le G$ (Assumption~\ref{asm:bounded}) and $\varepsilon\le 1$, we have
\[
\Big\|\frac{\hat m_{t}}{\sqrt{\hat v_{t}}+\varepsilon\Big\|^{2}
\le
\sum_{i=1}^{d}\frac{G^{2}}{(1-\beta_{2})G^{2}+\varepsilon^{2}}
\le \frac{d}{1-\beta_{2}} .
\tag{Step 5}
\]

\noindent\textbf{Step 6 (Combine Steps 3–5).}  
Insert the lower bound from Step~4 and the upper bound from Step~5 into \eqref{Step 3}:
\begin{align}
\EE_{t}[f(w_{t+1})]
&\le f(w_{t})
 -\frac{\eta_{t}(1-\beta_{1})}{\sqrt{1-\beta_{2}}}
   \sum_{i=1}^{d}\frac{(\nabla f(w_{t}))_{i}^{2}}{|g_{t,i}|+\varepsilon'}
 +\frac{L\eta_{t}^{2}d}{2(1-\beta_{2})}. \tag{Step 6}
\end{align}

\noindent\textbf{Step 7 (Take full expectation).}  
Let $\EE$ denote expectation over all randomness up to iteration $t$.  Taking expectation of \eqref{Step 6} and using $\EE[|g_{t,i}|]\le G$ gives
\begin{align}
\EE[f(w_{t+1})]
&\le \EE[f(w_{t})]
 -\frac{\eta_{t}(1-\beta_{1})}{\sqrt{1-\beta_{2}}}
   \EE\!\biggl[\frac{\|\nabla f(w_{t})\|_{2}^{2}}{G+\varepsilon'}\biggr]
 +\frac{L\eta_{t}^{2}d}{2(1-\beta_{2})}. \tag{Step 7}
\end{align}
Since $G+\varepsilon'$ is a constant, we can pull it out:
\[
\EE[f(w_{t+1})]\le\EE[f(w_{t})]
-\frac{\eta_{t}(1-\beta_{1})}{(G+\varepsilon')\sqrt{1-\beta_{2}}}
   \EE\!\bigl[\|\nabla f(w_{t})\|_{2}^{2}\bigr]
+\frac{L\eta_{t}^{2}d}{2(1-\beta_{2})}. \tag{Step 8}
\]

\noindent\textbf{Step 9 (Telescoping sum).}  
Sum \eqref{Step 8} over $t=1,\dots,T$:
\begin{align}
\EE[f(w_{T+1})] 
&\le f(w_{1})
 -\frac{1-\beta_{1}}{(G+\varepsilon')\sqrt{1-\beta_{2}}}
   \sum_{t=1}^{T}\eta_{t}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
 +\frac{L d}{2(1-\beta_{2})}\sum_{t=1}^{T}\eta_{t}^{2}. \tag{Step 9}
\end{align}
Rearranging,
\[
\sum_{t=1}^{T}\eta_{t}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{(G+\varepsilon')\sqrt{1-\beta_{2}}}{1-\beta_{1}}
\bigl(f(w_{1})-f^{\star}\bigr)
+\frac{L d (G+\varepsilon')\sqrt{1-\beta_{2}}}{2(1-\beta_{1})(1-\beta_{2})}
   \sum_{t=1}^{T}\eta_{t}^{2}. \tag{Step 10}
\]

\noindent\textbf{Step 10 (Insert the stepsize $\eta_{t}=\eta/\sqrt{t}$).}  
We have
\[
\sum_{t=1}^{T}\eta_{t}= \eta\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
\;\le\;
2\eta\sqrt{T},
\qquad
\sum_{t=1}^{T}\eta_{t}^{2}= \eta^{2}\sum_{t=1}^{T}\frac{1}{t}
\;\le\;
\eta^{2}\bigl(1+\log T\bigr). \tag{Step 11}
\]
Dividing \eqref{Step 10} by $\sum_{t=1}^{T}\eta_{t}\ge\eta\sqrt{T}$ yields
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2\bigl(f(w_{1})-f^{\star}\bigr)}{\eta(1-\beta_{1})}
+
\frac{L d (G+\varepsilon')\sqrt{1-\beta_{2}}}{(1-\beta_{1})(1-\beta_{2})}
   \frac{\eta(1+\log T)}{\sqrt{T}}. \tag{Step 12}
\]
Renaming constants (absorbing $d$, $G$, $\varepsilon'$ into a generic $C$) gives exactly the bound stated in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{Step 12} we identify the two contributions:
\[
C_{1}= \frac{2\bigl(f(w_{1})-f^{\star}\bigr)}{\eta(1-\beta_{1})},
\qquad
C_{2}= \frac{L d (G+\varepsilon')\sqrt{1-\beta_{2}}}{(1-\beta_{1})(1-\beta_{2})}\,\eta .
\]
Thus
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le C_{1}+ C_{2}\frac{1+\log T}{\sqrt{T}} .
\]
Since $C_{1}$ does not depend on $T$, the dominant term for large $T$ is $O\!\bigl((\log T)/\sqrt{T}\bigr)$.  Consequently,
\[
\min_{1\le t\le T}\EE\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le O\!\bigl(T^{-1/2}\log T\bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Constant stepsize $\eta_{t}\equiv\eta$.}  
If $\eta$ is chosen sufficiently small (e.g. $\eta\le \frac{(1-\beta_{1})\varepsilon}{G}$), the same derivation leads to a bound of order $O(1/T)$ for the average squared gradient norm, but the guarantee holds only up to a neighbourhood whose radius is proportional to $\eta$.
\item \textbf{Convex $f$.}  
When $f$ is convex, the same analysis (with the stronger inequality $\langle\nabla f(w_{t}),w_{t}-w^{\star}\rangle\ge f(w_{t})-f^{\star}$) yields a sub‑linear regret bound $O(\sqrt{T})$, matching the optimal rate for adaptive methods.
\end{itemize}
