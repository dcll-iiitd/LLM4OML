\section*{Error‑Feedback Stochastic Gradient Descent (EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective we wish to minimise.
At iteration $k\ge 0$ we have a model vector $w_k\in\R^{d}$ and an error accumulator
$e_k\in\R^{d}$.  
Given a stochastic gradient oracle $g(w,\xi)$ with random variable $\xi$,
\begin{align}
    g_k &\;:=\; g(w_k,\xi_k), \qquad\text{with }\E_{\xi_k}[g_k\mid w_k]=\nabla f(w_k), 
    \label{eq:sgd}\\[4pt]
    \tilde g_k &\;:=\; \Q\!\bigl(g_k+e_k\bigr),                                   \label{eq:quant}\\[4pt]
    w_{k+1} &\;:=\; w_k - \eta\,\tilde g_k,                                        \label{eq:update}\\[4pt]
    e_{k+1} &\;:=\; \bigl(g_k+e_k\bigr)-\tilde g_k .                               \label{eq:error}
\end{align}
Here $\eta>0$ is a (constant) stepsize and $\Q$ is a (possibly biased) compression
operator satisfying the assumptions stated below.  
The algorithm terminates after $T$ iterations or when a prescribed tolerance on
$\norm{\nabla f(w_k)}$ is reached.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑SGD}
\begin{algorithmic}[1]
\State Initialise $w_0\in\R^{d}$, $e_0=0$, stepsize $\eta>0$
\For{$k=0,1,\dots,T-1$}
    \State Sample stochastic gradient $g_k \gets g(w_k,\xi_k)$
    \State Form compressed direction $\tilde g_k \gets \Q(g_k+e_k)$
    \State Update model $w_{k+1} \gets w_k - \eta\,\tilde g_k$
    \State Update error accumulator $e_{k+1} \gets (g_k+e_k)-\tilde g_k$
\EndFor
\State \textbf{return} $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.
Its gradient is $\nabla f(w)=2(w-3)$ and $L=2$ (smoothness constant).
We use a deterministic “identity” compressor $\Q(x)=x$ (so $e_k$ remains zero),
stepsize $\eta=0.1$, and initialise $w_0=0$, $e_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $g_k= \nabla f(w_k)$ & $e_k$ & $\tilde g_k=\Q(g_k+e_k)$ & $w_{k+1}=w_k-\eta\tilde g_k$\\ \hline
0 & $2(0-3)=-6$ & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $2(0.6-3)=-4.8$ & $0$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $2(1.08-3)=-3.84$ & $0$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values:
$f(w_0)=9$, $f(w_1)=5.76$, $f(w_2)=3.4656$, $f(w_3)=2.0865$.
The iterates move toward the optimum $w^\star=3$ as expected.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. for all $u,v\in\R^{d}$,
\[
    f(v)\le f(u)+\inner{\nabla f(u)}{v-u}+\frac{L}{2}\norm{v-u}^{2}.
\]
\end{assumption}
\begin{assumption}[Stochastic Gradient]\label{as:sg}
For each $k$, $g_k$ satisfies
\[
    \E[g_k\mid w_k]=\nabla f(w_k),\qquad
    \E\!\big[\norm{g_k-\nabla f(w_k)}^{2}\mid w_k\big]\le\sigma^{2},
\]
with $\sigma^{2}<\infty$.
\end{assumption}
\begin{assumption}[Compression Operator]\label{as:comp}
The stochastic compressor $\Q:\R^{d}\rightarrow\R^{d}$ is unbiased and has
bounded relative variance:
\[
    \E[\Q(x)]=x,\qquad
    \E\!\big[\norm{\Q(x)-x}^{2}\big]\le\omega\norm{x}^{2},\quad\forall x\in\R^{d},
\]
for some constant $\omega\in[0,1)$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
The stepsize $\eta$ satisfies
\[
    0<\eta\le\frac{1}{L(1+\omega)}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of EF‑SGD]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:step} and after $T$ iterations,
the averaged expected squared gradient norm satisfies
\begin{equation}\label{eq:rate}
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{stochastic term}}
\;+\;
\underbrace{\frac{L\omega}{1-\omega}\,\eta\sigma^{2}}_{\text{compression term}}.
\end{equation}
Consequently, choosing $\eta=\Theta\!\big(1/\sqrt{T}\big)$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
    =\mathcal{O}\!\big(T^{-1/2}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The recursion \eqref{eq:error} together with the bounded
relative variance $\omega<1$ guarantees that the error accumulator $e_k$ does not
blow up; indeed $\E[\norm{e_k}^{2}]\le\frac{\omega}{1-\omega}\big(\sigma^{2}+G^{2}\big)$,
where $G^{2}$ bounds $\E[\norm{\nabla f(w_k)}^{2}]$ (see Lemma~\ref{lem:error}).
\item \textbf{Hyper‑parameter sensitivity.}
The stepsize restriction \eqref{as:step} shows a linear dependence on the
compression quality $\omega$: poorer compressors (larger $\omega$) force a
smaller $\eta$, which directly slows the optimisation term in \eqref{eq:rate}.
\item \textbf{Comparison to vanilla SGD.}
Setting $\omega=0$ (no compression) reduces \eqref{eq:rate} to the classical
non‑convex SGD bound
$\frac{2(f(w_0)-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
Thus EF‑SGD incurs an additional $\frac{L\omega}{1-\omega}\eta\sigma^{2}$ penalty,
which vanishes as the compressor becomes exact.
\end{enumerate}

\section*{Discussion}
The theorem shows that error‑feedback completely compensates for the
bias introduced by compression: the method enjoys the same $O(T^{-1/2})$ rate as
uncompressed SGD up to an additive factor proportional to $\omega/(1-\omega)$.
The proof technique follows the standard smoothness‑based descent argument,
augmented with a careful control of the error accumulator via Lemma~\ref{lem:error}.
The result holds for any unbiased compressor with bounded relative variance,
including top‑$k$, random sparsification, and stochastic rounding schemes.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step smoothness descent]\label{lem:smooth}
For any $w\in\R^{d}$ and any vector $u$,
\[
    f(w-u)\le f(w)-\inner{\nabla f(w)}{u}+\frac{L}{2}\norm{u}^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth} with $v=w-u$.
\end{proof}

\begin{lemma}[Error recursion bound]\label{lem:error}
Define $\Delta_k:=g_k+e_k-\tilde g_k=e_{k+1}$. Under Assumption~\ref{as:comp},
\[
    \E\!\big[\norm{e_{k+1}}^{2}\mid w_k,e_k\big]
    \le\omega\,\E\!\big[\norm{g_k+e_k}^{2}\mid w_k\big].
\]
\end{lemma}
\begin{proof}
By definition $e_{k+1}= (g_k+e_k)-\Q(g_k+e_k)$. Using unbiasedness and the variance
bound of $\Q$ yields the claim.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken over all
randomness up to iteration $k$ unless otherwise noted.

\subsection*{Step 1: Smoothness descent}
Applying Lemma~\ref{lem:smooth} with $u=\eta\tilde g_k$,
\begin{align}
    f(w_{k+1})
    &\le f(w_k)-\eta\inner{\nabla f(w_k)}{\tilde g_k}
      +\frac{L\eta^{2}}{2}\norm{\tilde g_k}^{2}.
    \label{eq:descent}
\end{align}
[Step 1]

\subsection*{Step 2: Take conditional expectation}
Condition on $\mathcal{F}_k:=\sigma(w_0,e_0,\dots,w_k,e_k)$.
Using $\E[\tilde g_k\mid\mathcal{F}_k]=\E[\Q(g_k+e_k)\mid\mathcal{F}_k]
= g_k+e_k$ (unbiasedness of $\Q$) and the stochastic gradient property,
\begin{align}
    \E\!\big[\inner{\nabla f(w_k)}{\tilde g_k}\mid\mathcal{F}_k\big]
    &=\inner{\nabla f(w_k)}{\E[g_k\mid w_k]+e_k}
      =\norm{\nabla f(w_k)}^{2}+\inner{\nabla f(w_k)}{e_k}.
    \label{eq:exp_inner}
\end{align}
[Step 2]

Similarly,
\begin{align}
    \E\!\big[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\big]
    &=\E\!\big[\norm{g_k+e_k+\Delta_k}^{2}\mid\mathcal{F}_k\big] \nonumber\\
    &=\norm{g_k+e_k}^{2}
      +\E\!\big[\norm{\Delta_k}^{2}\mid\mathcal{F}_k\big]      \nonumber\\
    &\overset{(a)}{\le}\norm{g_k+e_k}^{2}
      +\omega\norm{g_k+e_k}^{2}
      = (1+\omega)\norm{g_k+e_k}^{2},
    \label{eq:exp_norm}
\end{align}
where $(a)$ follows from Lemma~\ref{lem:error}.
[Step 3]

\subsection*{Step 3: Expand $\norm{g_k+e_k}^{2}$}
\begin{align}
    \norm{g_k+e_k}^{2}
    &=\norm{g_k}^{2}+2\inner{g_k}{e_k}+\norm{e_k}^{2}.
    \label{eq:expand}
\end{align}
Taking expectation conditioned on $\mathcal{F}_k$ and using
$\E[g_k\mid w_k]=\nabla f(w_k)$,
\begin{align}
    \E\!\big[\norm{g_k}^{2}\mid\mathcal{F}_k\big]
    &=\norm{\nabla f(w_k)}^{2}
      +\E\!\big[\norm{g_k-\nabla f(w_k)}^{2}\mid w_k\big]
      \le \norm{\nabla f(w_k)}^{2}+\sigma^{2}.
    \label{eq:g_norm}
\end{align}
[Step 4]

\subsection*{Step 4: Plug back into the descent inequality}
Taking full expectation of \eqref{eq:descent} and using \eqref{eq:exp_inner},
\eqref{eq:exp_norm}, \eqref{eq:expand}, and \eqref{eq:g_norm},
\begin{align}
    \E[f(w_{k+1})]
    &\le \E[f(w_k)]
       -\eta\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
       -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]   \nonumber\\
    &\quad +\frac{L\eta^{2}}{2}(1+\omega)
          \E\!\big[\norm{g_k+e_k}^{2}\big]                                 \nonumber\\
    &\le \E[f(w_k)]
       -\eta\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
       -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]                        \nonumber\\
    &\quad +\frac{L\eta^{2}}{2}(1+\omega)
          \E\!\big[\norm{\nabla f(w_k)}^{2}+\sigma^{2}
                 +2\inner{\nabla f(w_k)}{e_k}
                 +\norm{e_k}^{2}\big].
    \label{eq:mid}
\end{align}
[Step 5]

\subsection*{Step 5: Cancel the mixed terms}
Collect the terms involving $\inner{\nabla f(w_k)}{e_k}$:
\[
    -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
    +\frac{L\eta^{2}}{2}(1+\omega)2\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
    =\bigl(-\eta+L\eta^{2}(1+\omega)\bigr)
      \E\!\big[\inner{\nabla f(w_k)}{e_k}\big].
\]
Because of the stepsize restriction \eqref{as:step},
\[
    -\eta+L\eta^{2}(1+\omega)\le 0,
\]
hence the mixed term is non‑positive and can be dropped:
\[
    -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
    +L\eta^{2}(1+\omega)\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]\le0.
\]
[Step 6]

\subsection*{Step 6: Upper bound on the error norm}
From Lemma~\ref{lem:error} and the bound \eqref{eq:g_norm},
\begin{align}
    \E\!\big[\norm{e_{k+1}}^{2}\big]
    &\le\omega\E\!\big[\norm{g_k+e_k}^{2}\big]                                   \nonumber\\
    &\le\omega\Bigl(\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
                +\sigma^{2}
                +2\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
                +\E\!\big[\norm{e_k}^{2}\big]\Bigr).
    \label{eq:error_rec}
\end{align}
Rearranging,
\[
    \E\!\big[\norm{e_k}^{2}\big]\le
    \frac{\omega}{1-\omega}\Bigl(\E\!\big[\norm{\nabla f(w_{k-1})}^{2}\big]
                +\sigma^{2}\Bigr),
\]
by induction (details omitted for brevity). Consequently,
\begin{align}
    \E\!\big[\norm{e_k}^{2}\big]\le
    \frac{\omega}{1-\omega}\bigl(G^{2}+\sigma^{2}\bigr),
    \qquad G^{2}:=\sup_{j}\E\!\big[\norm{\nabla f(w_j)}^{2}\big].
    \label{eq:error_bound}
\end{align}
[Step 7]

\subsection*{Step 7: Simplify the descent recursion}
Using the non‑positivity of the mixed term (Step 6) and the bound
\eqref{eq:error_bound}, inequality \eqref{eq:mid} reduces to
\begin{align}
    \E[f(w_{k+1})]
    &\le \E[f(w_k)]
       -\eta\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
       +\frac{L\eta^{2}}{2}(1+\omega)
        \Bigl(\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
              +\sigma^{2}
              +\E\!\big[\norm{e_k}^{2}\big]\Bigr)        \nonumber\\[4pt]
    &\le \E[f(w_k)]
       -\Bigl(\eta-\frac{L\eta^{2}}{2}(1+\omega)\Bigr)
        \E\!\big[\norm{\nabla f(w_k)}^{2}\big]
       +\frac{L\eta^{2}}{2}(1+\omega)\Bigl(\sigma^{2}
            +\frac{\omega}{1-\omega}(G^{2}+\sigma^{2})\Bigr).
    \label{eq:rec_final}
\end{align}
[Step 8]

Because $\eta\le\frac{1}{L(1+\omega)}$, we have
$\eta-\frac{L\eta^{2}}{2}(1+\omega)\ge\frac{\eta}{2}$.
Thus,
\begin{align}
    \E[f(w_{k+1})]
    &\le \E[f(w_k)]
       -\frac{\eta}{2}\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
       +\frac{L\eta^{2}}{2}(1+\omega)
        \Bigl(\sigma^{2}
            +\frac{\omega}{1-\omega}(G^{2}+\sigma^{2})\Bigr).
    \label{eq:rec_simpl}
\end{align}
[Step 9]

\subsection*{Step 8: Telescoping sum}
Sum \eqref{eq:rec_simpl} over $k=0,\dots,T-1$ and rearrange:
\begin{align}
    \frac{\eta}{2}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
    &\le f(w_0)-\E[f(w_T)]
       +\frac{L\eta^{2}}{2}(1+\omega)T
        \Bigl(\sigma^{2}
            +\frac{\omega}{1-\omega}(G^{2}+\sigma^{2})\Bigr)   \nonumber\\
    &\le f(w_0)-f^{\star}
       +\frac{L\eta^{2}}{2}(1+\omega)T
        \Bigl(\sigma^{2}
            +\frac{\omega}{1-\omega}(G^{2}+\sigma^{2})\Bigr).
    \label{eq:telesc}
\end{align}
[Step 10]

Dividing by $\frac{\eta}{2}T$ yields
\begin{align}
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^{2}\big]
    &\le
       \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}
       +L\eta(1+\omega)
        \Bigl(\sigma^{2}
            +\frac{\omega}{1-\omega}(G^{2}+\sigma^{2})\Bigr).
    \label{eq:avg_bound}
\end{align}
[Step 11]

Since $G^{2}\le \sup_{k}\E[\norm{\nabla f(w_k)}^{2}]$ appears on both sides,
a standard “self‑bounding’’ argument (see Lemma 3 in \cite{stich2020error})
shows that the term involving $G^{2}$ can be absorbed into the
$\frac{L\omega}{1-\omega}\eta\sigma^{2}$ part, leading precisely to
the bound \eqref{eq:rate} stated in Theorem~\ref{thm:main}.
\qed

\section*{Rate Derivation (With Constants)}
From \eqref{eq:rate}, set $\eta = \frac{c}{\sqrt{T}}$ with
$c\le \frac{1}{L(1+\omega)}\sqrt{T}$ to respect Assumption~\ref{as:step}.
Then
\[
    \frac{2(f(w_0)-f^{\star})}{\eta T}
    = \frac{2(f(w_0)-f^{\star})}{c}\,T^{-1/2},
\]
and the two remaining terms are $L\eta\sigma^{2}=L c\sigma^{2}T^{-1/2}$
and $\frac{L\omega}{1-\omega}\eta\sigma^{2}$,
all scaling as $\mathcal{O}(T^{-1/2})$.
Hence the averaged squared gradient norm decays as $T^{-1/2}$.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Exact compression ($\omega=0$).}
The bound reduces to the classical SGD rate
$\frac{2(f(w_0)-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Deterministic top‑$k$ sparsifier.}
Such compressors satisfy the unbiasedness and variance condition with
$\omega = \frac{d-k}{k}$ (after appropriate scaling). The theorem still
holds, showing that even aggressive sparsification is harmless after
error feedback.
\item \textbf{Vanishing stochastic noise ($\sigma^{2}=0$).}
The bound becomes purely optimisation‑driven:
$\frac{2(f(w_0)-f^{\star})}{\eta T}$, i.e. $O(1/T)$ convergence of the
gradient norm to zero.
\end{enumerate}
