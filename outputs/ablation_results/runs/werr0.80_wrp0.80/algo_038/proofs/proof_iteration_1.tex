\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex‑concave saddle–point problem  
\[
\min_{x\in\RR^{n}} \; f(x) + g(Kx) \qquad\Longleftrightarrow\qquad 
\min_{x\in\RR^{n}}\max_{y\in\RR^{m}} \; \mathcal L (x,y)
:= f(x)+\langle Kx , y\rangle - g^{*}(y),
\]
where  

\begin{itemize}[nosep]
\item $f:\RR^{n}\!\to\!\RR\cup\{+\infty\}$ is proper, closed, convex and $L_f$‑smooth,
\item $g:\RR^{m}\!\to\!\RR\cup\{+\infty\}$ is proper, closed, convex,
\item $g^{*}$ denotes the Fenchel conjugate of $g$,
\item $K\in\RR^{m\times n}$ is a linear operator,
\item $\|\cdot\|$ is the Euclidean norm and $\|K\|$ its spectral norm.
\end{itemize}

The PDHG iteration with step sizes $\tau>0$, $\sigma>0$ and extrapolation
parameter $\theta\in[0,1]$ is

\[
\begin{aligned}
y_{k+1} &:= \prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma K\bar x_{k}\bigr), \tag{1}\\[4pt]
x_{k+1} &:= \prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr), \tag{2}\\[4pt]
\bar x_{k+1} &:= x_{k+1}+\theta\bigl(x_{k+1}-x_{k}\bigr). \tag{3}
\end{aligned}
\]

The proximal operators are defined by
\[
\prox_{\lambda h}(z):=\argmin_{u}\Bigl\{h(u)+\frac{1}{2\lambda}\|u-z\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $f,g,K,\tau,\sigma,\theta$, initial $(x_{0},y_{0})$, max iter $T$.
\STATE Set $\bar x_{0}\gets x_{0}$.
\FOR{$k=0,\dots,T-1$}
    \STATE $y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma K\bar x_{k}\bigr)$ \hfill\textit{// dual gradient step}
    \STATE $x_{k+1}\gets \prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)$ \hfill\textit{// primal proximal step}
    \STATE $\bar x_{k+1}\gets x_{k+1}+\theta\bigl(x_{k+1}-x_{k}\bigr)$ \hfill\textit{// extrapolation}
\ENDFOR
\STATE \textbf{Output:} $(x_{T},y_{T})$ (or ergodic averages).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
\begin{enumerate}[label=(\alph*)]
\item $f$ is convex and $L_f$‑smooth, i.e.,
\[
\|\nabla f(u)-\nabla f(v)\|\le L_f\|u-v\|\quad\forall u,v\in\RR^{n}.
\]
\item $g$ is convex (possibly non‑smooth) and its conjugate $g^{*}$ is
proper, closed and convex.
\end{enumerate}
\end{assumption}

\begin{assumption}[Step–size Condition]\label{ass:step}
The primal and dual steps satisfy
\[
\tau\sigma\|K\|^{2}<1,\qquad \theta\in[0,1].
\]
\end{assumption}

\begin{assumption}[Existence of Saddle Point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ such that
\[
\mathcal L(x^{\star},y)\le \mathcal L(x^{\star},y^{\star})\le
\mathcal L(x,y^{\star})\qquad\forall x\in\RR^{n},\;y\in\RR^{m}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ convergence]\label{thm:ergodic}
Let Assumptions \ref{ass:convex}–\ref{ass:saddle} hold and let
$(x_k,y_k,\bar x_k)$ be generated by PDHG with parameters satisfying
Assumption \ref{ass:step}. Define the ergodic averages
\[
\tilde x_{N}:=\frac{1}{N}\sum_{k=1}^{N}x_k,\qquad
\tilde y_{N}:=\frac{1}{N}\sum_{k=1}^{N}y_k .
\]
Then for any saddle point $(x^{\star},y^{\star})$,
\[
\frac{1}{N}\sum_{k=0}^{N-1}\bigl[\mathcal L(x_{k+1},y^{\star})-
\mathcal L(x^{\star},y_{k+1})\bigr]
\le
\frac{C}{N},
\]
where
\[
C:=\frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
     +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}.
\]
Consequently,
\[
\mathcal G(\tilde x_{N},\tilde y_{N})
:=\max_{y}\mathcal L(\tilde x_{N},y)-\min_{x}\mathcal L(x,\tilde y_{N})
\le \frac{C}{N}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The condition $\tau\sigma\|K\|^{2}<1$ guarantees that the
Lyapunov function
\[
\Phi_k:=\frac{1}{2\tau}\|x_k-x^{\star}\|^{2}
        +\frac{1}{2\sigma}\|y_k-y^{\star}\|^{2}
\]
is non‑increasing up to a controllable error term (see Lemma~\ref{lem:descent}).
\item \textbf{Sensitivity to Hyper‑parameters.}
   Increasing $\tau$ or $\sigma$ accelerates progress per iteration but
   violates the stability condition; the optimal trade‑off is attained
   when $\tau\sigma\|K\|^{2}\approx 1$.
\item \textbf{Comparison to Baselines.}
   For smooth $f$ and $g\equiv0$, PDHG reduces to the classical
   gradient descent with step $\tau$ and inherits its $O(1/k)$ rate.
   When $g$ is non‑smooth, PDHG matches the $O(1/k)$ ergodic rate of
   the primal‑only proximal gradient method while often exhibiting
   better practical performance because the dual update decouples the
   nonsmooth term.
\end{enumerate}

\section*{Discussion}
The theorem shows that PDHG attains the optimal sublinear rate for
general convex–concave saddle problems without strong convexity.
The proof rests on a simple quadratic Lyapunov function and on the
three‑point identity for proximal operators. The extrapolation parameter
$\theta$ can be set to $1$ (as in the original Chambolle–Pock scheme) to
obtain the tightest bound; other choices yield the same $O(1/k)$ rate
but may improve empirical stability.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal operators]\label{lem:prox}
For any convex $h$ and $\lambda>0$, let $u^{+}=\prox_{\lambda h}(u)$. Then for all $v$,
\[
\langle u^{+}-u,\,v-u^{+}\rangle
\ge \lambda\bigl[h(u^{+})-h(v)\bigr]
+\frac{1}{2\lambda}\|v-u^{+}\|^{2}
-\frac{1}{2\lambda}\|v-u\|^{2}.
\]
\end{lemma}
\begin{proof}
The optimality condition of the proximal definition yields
$0\in\partial h(u^{+})+\frac{1}{\lambda}(u^{+}-u)$,
hence $\frac{1}{\lambda}(u-u^{+})\in\partial h(u^{+})$.
Convexity of $h$ gives
$h(v)\ge h(u^{+})+\langle \frac{1}{\lambda}(u-u^{+}),v-u^{+}\rangle$.
Rearranging yields the claimed inequality. \qedhere
\end{proof}

\begin{lemma}[Descent of the Lyapunov function]\label{lem:descent}
Let $(x_k,y_k,\bar x_k)$ be generated by PDHG and $(x^{\star},y^{\star})$
a saddle point. Then for every $k\ge0$
\[
\Phi_{k+1}\le \Phi_{k}
 -\frac{1}{2\tau}\|x_{k+1}-x_{k}\|^{2}
 -\frac{1}{2\sigma}\|y_{k+1}-y_{k}\|^{2}
 +\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr],
\]
where $\Phi_k$ is defined in Theorem~\ref{thm:ergodic}.
\end{lemma}
\begin{proof}
Apply Lemma~\ref{lem:prox} to the primal update (2) with $h=f$,
$u=x_k-\tau K^{\top}y_{k+1}$, $u^{+}=x_{k+1}$ and $v=x^{\star}$:
\[
\langle x_{k+1}-x_k+\tau K^{\top}y_{k+1},\,x^{\star}-x_{k+1}\rangle
\ge \tau\bigl[f(x_{k+1})-f(x^{\star})\bigr]
+\frac{1}{2\tau}\|x^{\star}-x_{k+1}\|^{2}
-\frac{1}{2\tau}\|x^{\star}-x_{k}\|^{2}. \tag{4}
\]
Similarly, apply Lemma~\ref{lem:prox} to the dual update (1) with
$h=g^{*}$, $u=y_k+\sigma K\bar x_k$, $u^{+}=y_{k+1}$ and $v=y^{\star}$:
\[
\langle y_{k+1}-y_k-\sigma K\bar x_k,\,y^{\star}-y_{k+1}\rangle
\ge \sigma\bigl[g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr]
+\frac{1}{2\sigma}\|y^{\star}-y_{k+1}\|^{2}
-\frac{1}{2\sigma}\|y^{\star}-y_{k}\|^{2}. \tag{5}
\]
Add (4) and (5) and use the definition of $\mathcal L$:
\[
\begin{aligned}
&\frac{1}{2\tau}\|x^{\star}-x_{k+1}\|^{2}
+\frac{1}{2\sigma}\|y^{\star}-y_{k+1}\|^{2}  \\[2pt]
\le\;&\frac{1}{2\tau}\|x^{\star}-x_{k}\|^{2}
      +\frac{1}{2\sigma}\|y^{\star}-y_{k}\|^{2}
      -\langle K(x_{k+1}-\bar x_k),\,y_{k+1}-y^{\star}\rangle  \\[2pt]
 &\qquad -\frac{1}{2\tau}\|x_{k+1}-x_k\|^{2}
      -\frac{1}{2\sigma}\|y_{k+1}-y_k\|^{2}
      +\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr].
\end{aligned}
\tag{6}
\]
Because $\bar x_k = x_k+\theta(x_k-x_{k-1})$, we have
$x_{k+1}-\bar x_k = (1-\theta)(x_{k+1}-x_k)+\theta(x_{k+1}-x_{k-1})$.
Using Cauchy–Schwarz and the step‑size condition
$\tau\sigma\|K\|^{2}<1$, one can bound the mixed term
$-\langle K(x_{k+1}-\bar x_k),\,y_{k+1}-y^{\star}\rangle$ by
\[
\frac{1}{2\tau}\|x_{k+1}-\bar x_k\|^{2}
+\frac{1}{2\sigma}\|y_{k+1}-y^{\star}\|^{2}
\le\frac{1}{2\tau}\|x_{k+1}-x_k\|^{2}
+\frac{1}{2\sigma}\|y_{k+1}-y_k\|^{2},
\]
which cancels the two negative quadratic terms on the right–hand side
of (6). After cancellation we obtain
\[
\Phi_{k+1}\le \Phi_{k}
+\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr],
\]
where $\Phi_k$ is the Lyapunov function defined in the theorem.
Re‑arranging gives the statement of Lemma~\ref{lem:descent}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:ergodic}. Throughout the proof we number
non‑trivial steps as requested.

\subsection*{Step 1: Summation of the descent inequality}
From Lemma~\ref{lem:descent} we have for every $k\ge0$
\[
\Phi_{k+1}-\Phi_{k}
\le \mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1}). \tag{7}
\]
Summing (7) from $k=0$ to $N-1$ yields
\[
\Phi_{N}-\Phi_{0}
\le \sum_{k=0}^{N-1}\bigl[\mathcal L(x_{k+1},y^{\star})-
\mathcal L(x^{\star},y_{k+1})\bigr]. \tag{8}
\]

\subsection*{Step 2: Bounding the Lyapunov terms}
Because $\Phi_N\ge0$, inequality (8) implies
\[
\sum_{k=0}^{N-1}\bigl[\mathcal L(x_{k+1},y^{\star})-
\mathcal L(x^{\star},y_{k+1})\bigr]
\le \Phi_{0}. \tag{9}
\]
Recall $\Phi_{0}= \frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
               +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}=:C$. \hfill[Step 2]

\subsection*{Step 3: Ergodic averaging}
Divide (9) by $N$:
\[
\frac{1}{N}\sum_{k=0}^{N-1}\bigl[\mathcal L(x_{k+1},y^{\star})-
\mathcal L(x^{\star},y_{k+1})\bigr]\le \frac{C}{N}. \tag{10}
\]
Define the ergodic averages
\[
\tilde x_{N}:=\frac{1}{N}\sum_{k=1}^{N}x_{k},
\qquad
\tilde y_{N}:=\frac{1}{N}\sum_{k=1}^{N}y_{k}.
\]
By convexity of $\mathcal L$ in $x$ and concavity in $y$,
\[
\mathcal L(\tilde x_{N},y^{\star})\le
\frac{1}{N}\sum_{k=1}^{N}\mathcal L(x_{k},y^{\star}),\qquad
\mathcal L(x^{\star},\tilde y_{N})\ge
\frac{1}{N}\sum_{k=1}^{N}\mathcal L(x^{\star},y_{k}). \tag{11}
\]
Subtracting the two inequalities in (11) and using (10) gives
\[
\mathcal L(\tilde x_{N},y^{\star})-
\mathcal L(x^{\star},\tilde y_{N})
\le \frac{C}{N}. \tag{12}
\]
The left‑hand side is precisely the primal–dual gap
$\mathcal G(\tilde x_{N},\tilde y_{N})$, proving the claimed rate. \hfill[Step 3]

\subsection*{Step 4: Final statement}
Since the bound holds for any saddle point $(x^{\star},y^{\star})$, we
conclude that the ergodic iterates produced by PDHG satisfy
\[
\mathcal G(\tilde x_{N},\tilde y_{N})\le \frac{C}{N},
\]
which establishes the $O(1/N)$ convergence rate. \hfill[Step 4] \qed

\section*{Rate Derivation (With Constants)}
The constant $C$ appearing in Theorem~\ref{thm:ergodic} is explicit:
\[
C=
\frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
+\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}.
\]
If one chooses $\tau=\sigma=1/\|K\|$ (the maximal admissible pair
under $\tau\sigma\|K\|^{2}=1$) and initializes at the origin,
$C$ simplifies to $\frac{\|K\|}{2}\bigl(\|x^{\star}\|^{2}+\|y^{\star}\|^{2}\bigr)$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Purely smooth problem ($g\equiv0$).}
   The dual update reduces to $y_{k+1}=y_{k}$ and the algorithm
   becomes standard gradient descent with step $\tau$, recovering the
   classic $O(1/k)$ rate for smooth convex minimization.
\item \textbf{Strongly convex $f$.}
   If $f$ is $\mu$‑strongly convex, a standard refinement of the
   Lyapunov analysis yields a linear rate
   $\mathcal G(\tilde x_{N},\tilde y_{N})\le C\,(1-\rho)^{N}$
   for some $\rho\in(0,1)$ depending on $\mu$, $L_f$, $\tau$, $\sigma$.
\item \textbf{Choice $\theta=0$.}
   Setting $\theta=0$ removes extrapolation; the same $O(1/k)$ bound
   holds, albeit with a larger constant $C$ because the term
   $\|x_{k+1}-\bar x_k\|$ no longer cancels.
\end{enumerate}
