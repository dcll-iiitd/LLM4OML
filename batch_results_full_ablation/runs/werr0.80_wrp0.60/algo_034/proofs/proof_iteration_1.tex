\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
At iteration $t$, given current iterate $x_t\in\R^{d}$, draw a stochastic gradient $g_t$ such that  

\[
\E\!\left[g_t\mid x_t\right]=\nabla f(x_t),\qquad 
\E\!\left[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\right]\le\sigma^{2}.
\]

Define the \emph{clipping operator} with clipping level $C>0$  

\[
\tilde g_t \;:=\; \clip{g_t}{C}
      =\frac{g_t}{\max\bigl\{1,\;\|g_t\|/C\bigr\}},
\tag{1}
\]

so that $\|\tilde g_t\|\le C$ always holds.  
The update rule is  

\[
x_{t+1}=x_t-\eta_t\,\tilde g_t,
\tag{2}
\]

where $\{\eta_t\}_{t\ge0}$ is a stepsize schedule (e.g., $\eta_t=\eta/\sqrt{t+1}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, stepsize schedule $\{\eta_t\}$, clipping level $C>0$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample stochastic gradient $g_t$ (e.g. minibatch gradient).
    \STATE $\displaystyle \tilde g_t\gets\frac{g_t}{\max\{1,\;\|g_t\|/C\}}$ \hfill\COMMENT{Clip}
    \STATE $x_{t+1}\gets x_t-\eta_t\,\tilde g_t$.
\ENDFOR
\STATE \textbf{Output:} $\displaystyle \hat x_T\gets\frac{1}{T}\sum_{t=0}^{T-1}x_t$ (or a random iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.
\begin{itemize}
    \item $f'(w)=2(w-3)$.
    \item Choose $C=1$, stepsize $\eta=0.1$, and deterministic gradients (no noise).
\end{itemize}
\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=\nabla f(w_t)$ & $\tilde g_t$ (clipped) & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $-1$ (since $|-6|/C=6>1$) & $9$\\
1 & $0-0.1(-1)=0.1$ & $-5.8$ & $-1$ & $(0.1-3)^2=8.41$\\
2 & $0.1-0.1(-1)=0.2$ & $-5.6$ & $-1$ & $(0.2-3)^2=7.84$\\
3 & $0.2-0.1(-1)=0.3$ & $-5.4$ & $-1$ & $(0.3-3)^2=7.29$
\end{tabular}
\end{center}
The clipping keeps the update magnitude constant at $0.1$, while the true gradient magnitude shrinks as $w_t$ approaches $3$.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} $f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]
\item \textbf{Unbiased stochastic gradients.} $\E[g_t\mid x_t]=\nabla f(x_t)$.
\item \textbf{Bounded variance.} $\E[\|g_t-\nabla f(x_t)\|^{2}\mid x_t]\le\sigma^{2}$.
\item \textbf{Clipping level.} $C>0$ is a fixed constant; the clipped gradient satisfies $\|\tilde g_t\|\le C$.
\item \textbf{Stepsize.} $\eta_t=\eta/\sqrt{t+1}$ with $\eta\le \frac{1}{L}$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Under assumptions (A1)–(A5), let $\{x_t\}$ be generated by Clipped SGD with clipping level $C$ and stepsizes $\eta_t=\eta/\sqrt{t+1}$. Define a random index $R$ uniformly drawn from $\{0,\dots,T-1\}$. Then

\[
\E\!\bigl[\|\nabla f(x_R)\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\eta\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{2L\eta C^{2}}{\sqrt{T}}}_{\text{clipping variance term}}
\;+\;
\underbrace{2\sigma^{2}\frac{\eta}{\sqrt{T}}}_{\text{stochastic variance term}}.
\tag{3}
\]

Consequently, $\E[\|\nabla f(x_R)\|^{2}]=\mathcal O\!\bigl(1/\sqrt{T}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\|\tilde g_t\|\le C$, the iterates cannot explode even if raw stochastic gradients have heavy tails.
\item \textbf{Hyperparameter Sensitivity.} The bound (3) shows a linear dependence on $C$; larger $C$ reduces clipping bias but may increase variance. Choosing $C\approx\sqrt{\sigma^{2}+L(f(x_0)-f^{\star})}$ balances the terms.
\item \textbf{Comparison to vanilla SGD.} For $C\ge \max_t\|g_t\|$, clipping is inactive and (3) reduces to the classical $\mathcal O(1/\sqrt{T})$ bound for SGD. When gradients are occasionally very large, clipping removes the $ \|\!g_t\!\|^{2}$ term that would otherwise dominate the bound.
\end{itemize}

\section*{Discussion}
The theorem proves that clipping does not deteriorate the optimal $\mathcal O(1/\sqrt{T})$ rate for non‑convex smooth objectives. The extra $L\eta C^{2}$ term quantifies the bias introduced by the clipping operator; it vanishes as $C\to\infty$ (no clipping) and is harmless for moderate $C$ because it is of the same order as the stochastic variance term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]
\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\tag{4}
\]
\end{lemma}

\begin{lemma}[Bias of Clipping]
\label{lem:bias}
For any $v\in\R^{d}$ and clipping level $C>0$,
\[
\bigl\|\,\clip{v}{C}-v\,\bigr\|\le \bigl(\|v\|-C\bigr)_{+},
\qquad 
\bigl\|\clip{v}{C}\bigr\|\le C.
\tag{5}
\]
\end{lemma}
\begin{proof}
If $\|v\|\le C$, the denominator in (1) equals $1$, therefore $\clip{v}{C}=v$ and the bound holds.
If $\|v\|>C$, then $\clip{v}{C}=C\,v/\|v\|$, and the difference is $v-Cv/\|v\|$, whose norm equals $\|v\|-C$. The second inequality follows directly from the definition of the operator. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken with respect to the randomness of the stochastic gradients $\{g_t\}$.

\noindent\textbf{Step 1.} Apply Lemma~\ref{lem:descent} with $d=\eta_t\tilde g_t$:
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\nabla f(x_t),\tilde g_t\rangle
                +\frac{L\eta_t^{2}}{2}\|\tilde g_t\|^{2}.
\tag{6}
\]

\noindent\textbf{Step 2.} Decompose $\tilde g_t$ into the unbiased part and the clipping bias:
\[
\tilde g_t = g_t - b_t,\qquad 
b_t:=g_t-\tilde g_t.
\tag{7}
\]
By Lemma~\ref{lem:bias}, $\|b_t\|\le (\|g_t\|-C)_{+}\le\|g_t\|$ and $\|\tilde g_t\|\le C$.

\noindent\textbf{Step 3.} Substitute (7) into (6) and take conditional expectation given $x_t$:
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
&\le f(x_t)-\eta_t\E\!\bigl[\langle\nabla f(x_t),g_t\rangle\mid x_t\bigr]
      +\eta_t\E\!\bigl[\langle\nabla f(x_t),b_t\rangle\mid x_t\bigr] \\
&\qquad +\frac{L\eta_t^{2}}{2}\E\!\bigl[\|\tilde g_t\|^{2}\mid x_t\bigr].
\end{aligned}
\tag{8}
\]

\noindent\textbf{Step 4.} Use unbiasedness $\E[g_t\mid x_t]=\nabla f(x_t)$:
\[
\E\!\bigl[\langle\nabla f(x_t),g_t\rangle\mid x_t\bigr]=\|\nabla f(x_t)\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 5.} Upper‑bound the bias term with Cauchy–Schwarz and Lemma~\ref{lem:bias}:
\[
\begin{aligned}
\E\!\bigl[\langle\nabla f(x_t),b_t\rangle\mid x_t\bigr]
&\le \|\nabla f(x_t)\|\;\E\!\bigl[\|b_t\|\mid x_t\bigr]   \\
&\le \|\nabla f(x_t)\|\;\E\!\bigl[(\|g_t\|-C)_{+}\mid x_t\bigr]
   \;\le\; \|\nabla f(x_t)\|\;\E\!\bigl[\|g_t\|\mid x_t\bigr].
\end{aligned}
\tag{10}
\]

\noindent\textbf{Step 6.} Apply the variance bound: for any random vector $u$,
\[
\E\!\bigl[\|u\|^{2}\bigr]=\|\E[u]\|^{2}+\E\!\bigl[\|u-\E[u]\|^{2}\bigr].
\]
With $u=g_t$ and using (A2)–(A3) we obtain
\[
\E\!\bigl[\|g_t\|^{2}\mid x_t\bigr]\le \|\nabla f(x_t)\|^{2}+\sigma^{2}.
\tag{11}
\]
Hence $\E[\|g_t\|\mid x_t]\le\sqrt{\|\nabla f(x_t)\|^{2}+\sigma^{2}}\le \|\nabla f(x_t)\|+\sigma$.

\noindent\textbf{Step 7.} Plug (9)–(11) into (8). Using $\|\tilde g_t\|^{2}\le C^{2}$ we get
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
&\le f(x_t)-\eta_t\|\nabla f(x_t)\|^{2}
    +\eta_t\|\nabla f(x_t)\|\bigl(\|\nabla f(x_t)\|+\sigma\bigr)  \\
&\qquad +\frac{L\eta_t^{2}}{2}C^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{Step 8.} Rearrange (12):
\[
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
\le f(x_t)-\eta_t\|\nabla f(x_t)\|^{2}
      +\eta_t\|\nabla f(x_t)\|^{2}
      +\eta_t\sigma\|\nabla f(x_t)\|
      +\frac{L\eta_t^{2}}{2}C^{2}.
\]
The first two terms cancel, leaving
\[
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
\le f(x_t)+\eta_t\sigma\|\nabla f(x_t)\|
      +\frac{L\eta_t^{2}}{2}C^{2}.
\tag{13}
\]

\noindent\textbf{Step 9.} Apply Young’s inequality $ab\le \frac{a^{2}}{2\lambda}+\frac{\lambda b^{2}}{2}$ with $a=\sigma$, $b=\|\nabla f(x_t)\|$, and $\lambda = \eta_t$:
\[
\eta_t\sigma\|\nabla f(x_t)\|
\le \frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
   +\frac{\eta_t\sigma^{2}}{2}.
\tag{14}
\]

\noindent\textbf{Step 10.} Insert (14) into (13):
\[
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
\le f(x_t)-\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
      +\frac{\eta_t\sigma^{2}}{2}
      +\frac{L\eta_t^{2}}{2}C^{2}.
\tag{15}
\]

\noindent\textbf{Step 11.} Take full expectation and sum over $t=0,\dots,T-1$:
\[
\begin{aligned}
f(x_0)-\E[f(x_T)]
&\ge \sum_{t=0}^{T-1}\frac{\eta_t}{2}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
   -\sum_{t=0}^{T-1}\frac{\eta_t\sigma^{2}}{2}
   -\sum_{t=0}^{T-1}\frac{L\eta_t^{2}}{2}C^{2}.
\end{aligned}
\tag{16}
\]

\noindent\textbf{Step 12.} Rearrange (16) to isolate the average gradient norm:
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_t}
   +\sigma^{2}
   +L C^{2}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{17}
\]

\noindent\textbf{Step 13.} For the stepsize $\eta_t=\eta/\sqrt{t+1}$,
\[
\sum_{t=0}^{T-1}\eta_t \ge \eta\int_{0}^{T}\frac{1}{\sqrt{s}}\,ds
      =2\eta\sqrt{T},
\qquad
\sum_{t=0}^{T-1}\eta_t^{2}\le \eta^{2}\sum_{t=0}^{\infty}\frac{1}{t+1}
      =\eta^{2}(1+\log T).
\tag{18}
\]
Using the crude bound $\sum_{t=0}^{T-1}\eta_t^{2}\le \eta^{2}T$ also yields a valid $\mathcal O(1/\sqrt{T})$ rate; we keep the simpler expression $\eta^{2}T$.

\noindent\textbf{Step 14.} Plug (18) into (17):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\eta\sqrt{T}}
+\frac{L\eta C^{2}}{\sqrt{T}}
+\frac{\sigma^{2}\eta}{\sqrt{T}}.
\tag{19}
\]

\noindent\textbf{Step 15.} Choose a random iterate $R$ uniformly from $\{0,\dots,T-1\}$; then $\E[\|\nabla f(x_R)\|^{2}]$ equals the left‑hand side of (19). This yields exactly the bound (3) stated in Theorem~\ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (19) we read off the three contributions:

\[
\begin{aligned}
\text{Optimization term: }& \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\eta\sqrt{T}},\\[2mm]
\text{Clipping bias term: }& \frac{L\eta C^{2}}{\sqrt{T}},\\[2mm]
\text{Stochastic variance term: }& \frac{\sigma^{2}\eta}{\sqrt{T}}.
\end{aligned}
\]

Setting $\eta = \min\!\bigl\{1/L,\; \sqrt{(f(x_0)-f^{\star})}/(\sigma+C\sqrt{L})\bigr\}$ balances the three terms and yields the clean rate

\[
\E\!\bigl[\|\nabla f(x_R)\|^{2}\bigr]\le
\mathcal O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{No clipping ($C\to\infty$).} Lemma~\ref{lem:bias} gives $b_t=0$, thus the clipping bias term disappears and (3) reduces to the standard SGD bound.
\item \textbf{Deterministic gradients ($\sigma=0$).} The stochastic variance term vanishes; the rate is governed only by the optimization and clipping terms.
\item \textbf{Large clipping level $C\ge \max_t\|g_t\|$.} The operator becomes identity, and the algorithm coincides with vanilla SGD.
\item \textbf{Heavy‑tailed noise.} Because $\|\tilde g_t\|\le C$, the algorithm is robust to occasional extremely large stochastic gradients; the bound (3) remains valid without assuming higher‑order moments.
\end{enumerate}
