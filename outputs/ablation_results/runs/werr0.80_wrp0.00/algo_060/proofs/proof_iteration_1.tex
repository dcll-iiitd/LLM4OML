\section*{Error‑Feedback Stochastic Gradient Descent (EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable and $L$‑smooth (Definition~\ref{def:smooth}).  
At iteration $k\in\{0,1,\dots,T-1\}$ we have:

\begin{itemize}
    \item Stochastic gradient $g_k:=\nabla f(w_k;\xi_k)$ where $\xi_k$ is a random data sample.
    \item An \emph{unbiased compression operator} $\mathcal{C}:\mathbb{R}^d\to\mathbb{R}^d$ satisfying  
    \begin{equation}
        \E[\mathcal{C}(x)]=x,\qquad 
        \E\!\big[\norm{\mathcal{C}(x)-x}^2\big]\le (1-\delta)\norm{x}^2,\;\; \forall x\in\mathbb{R}^d,
        \label{eq:compressor}
    \end{equation}
    for some $\delta\in(0,1]$.
    \item Error accumulator $e_k\in\mathbb{R}^d$ (initially $e_0=0$).
\end{itemize}
Define the \emph{compressed direction}
\[
\tilde g_k \;:=\; \mathcal{C}\!\big(g_k+e_k\big).
\]
The EF‑SGD updates are
\begin{align}
    w_{k+1} & = w_k - \eta\,\tilde g_k, \label{eq:update-w}\\
    e_{k+1} & = g_k+e_k-\tilde g_k . \label{eq:update-e}
\end{align}
$\eta>0$ is the learning rate (constant or diminishing).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Error‑Feedback SGD}
\begin{algorithmic}[1]
\State {\bf Input:} initial point $w_0$, stepsize $\eta$, compressor $\mathcal{C}$, iterations $T$.
\State $e_0\gets 0$
\For{$k=0$ {\bf to} $T-1$}
    \State Sample $\xi_k$ and compute stochastic gradient $g_k\gets \nabla f(w_k;\xi_k)$
    \State $\tilde g_k \gets \mathcal{C}(g_k+e_k)$   \Comment{compress with error feedback}
    \State $w_{k+1}\gets w_k - \eta\,\tilde g_k$   \Comment{parameter update}
    \State $e_{k+1}\gets g_k+e_k-\tilde g_k$        \Comment{accumulate quantization error}
\EndFor
\State {\bf Return} $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$, thus $\nabla f(w)=2(w-3)$.  
Choose $\eta=0.1$, $T=3$, and a trivial compressor $\mathcal{C}(x)=x$ (so $e_k$ will stay zero).  
Initialize $w_0=0$, $e_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
k & $w_k$ & $g_k=2(w_k-3)$ & $\tilde g_k=\mathcal{C}(g_k+e_k)$ & $w_{k+1}=w_k-\eta\tilde g_k$ \\ \hline
0 & 0 & $-6$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & 0.6 & $2(0.6-3)=-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & 1.08 & $2(1.08-3)=-3.84$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3270$ – the algorithm descends toward the optimum $w^\star=3$.

\section*{Assumptions}
\label{sec:assumptions}
\begin{enumerate}
    \item \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
    \begin{equation}
        \norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},\quad\forall x,y\in\mathbb{R}^d .
        \label{def:smooth}
    \end{equation}
    \item \textbf{Bounded variance of stochastic gradients.} For all $w$,
    \begin{equation}
        \E_{\xi}[\nabla f(w;\xi)]=\nabla f(w),\qquad
        \E_{\xi}\big[\norm{\nabla f(w;\xi)-\nabla f(w)}^2\big]\le\sigma^2 .
        \label{assump:variance}
    \end{equation}
    \item \textbf{Compression operator.} $\mathcal{C}$ satisfies \eqref{eq:compressor} for a known $\delta\in(0,1]$.
    \item \textbf{Stepsize.} The constant stepsize $\eta$ obeys
    \begin{equation}
        0<\eta\le \frac{1}{L\big(1+\frac{1}{\delta}\big)} .
        \label{eq:stepsize}
    \end{equation}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Average Gradient Norm for EF‑SGD]\label{thm:main}
Under Assumptions~\ref{sec:assumptions}, let $w^\star$ be a global minimizer of $f$ (possibly non‑convex).  
Define the random iterate average $\bar w_T:=\frac{1}{T}\sum_{k=0}^{T-1}w_k$. Then
\begin{equation}
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
    \;\le\;
    \underbrace{\frac{2\big(f(w_0)-f(w^\star)\big)}{\eta T}}_{\text{optimization term}}
    \;+\;
    \underbrace{\frac{\eta L\sigma^2}{\delta}}_{\text{stochastic \& compression term}} .
    \label{eq:rate}
\end{equation}
Consequently, choosing $\eta =\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
    \item \textbf{Stability.} The error accumulator $e_k$ remains bounded in expectation because the compressor is contractive in mean square (parameter $\delta$). This prevents error explosion even with aggressive quantization.
    \item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{eq:rate} shows a linear dependence on $\eta$ for the stochastic term and an inverse dependence on $T$ for the optimization term. The admissible range \eqref{eq:stepsize} guarantees that increasing compression (smaller $\delta$) forces a smaller stepsize to preserve convergence.
    \item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Without error feedback ($e_k\equiv0$) the compression variance appears as $\frac{L\sigma^2}{\delta}$ in the numerator, leading to a worse constant.
        \item With exact gradients ($\sigma=0$) the rate reduces to $\mathcal{O}\big(\frac{1}{\eta T}\big)$, matching standard (uncompressed) SGD.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that error‑feedback restores the $\mathcal{O}(1/\sqrt{T})$ rate typical for SGD on smooth non‑convex objectives, despite the presence of biased quantization. The constant $\frac{1}{\delta}$ quantifies the loss incurred by compression; as $\delta\to 1$ (perfect compressor) the bound coincides with vanilla SGD. The proof relies only on smoothness, unbiasedness of stochastic gradients, and the compressor contractivity \eqref{eq:compressor}; no convexity assumption is needed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function $f$,
\[
    f(y) \le f(x) + \inner{\nabla f(x)}{y-x} + \frac{L}{2}\norm{y-x}^2,\qquad\forall x,y .
\]
\end{lemma}
\begin{lemma}[Unbiased compressor variance bound]\label{lem:compressor}
If $\mathcal{C}$ satisfies \eqref{eq:compressor}, then for any random vector $u$
\[
    \E\!\big[\norm{\mathcal{C}(u)-u}^2\big]\le (1-\delta)\E\!\big[\norm{u}^2\big].
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations $\E[\cdot]$ are taken over all randomness up to the current iteration (including $\xi_k$ and the compressor).

\bigskip
\noindent\textbf{[Step 1] Apply smoothness to $w_{k+1}$.}  
Using Lemma~\ref{lem:smooth} with $x=w_k$, $y=w_{k+1}=w_k-\eta\tilde g_k$,
\begin{align}
    f(w_{k+1})
    &\le f(w_k) -\eta\inner{\nabla f(w_k)}{\tilde g_k}
          +\frac{L\eta^2}{2}\norm{\tilde g_k}^2 .
    \label{eq:smooth-step}
\end{align}

\noindent\textbf{[Step 2] Decompose $\tilde g_k$ using the error feedback definition.}  
Recall $\tilde g_k=\mathcal{C}(g_k+e_k)$. Add and subtract $g_k+e_k$:
\[
    \tilde g_k = (g_k+e_k) - \big((g_k+e_k)-\tilde g_k\big).
\]
Insert into the inner product term of \eqref{eq:smooth-step}:
\begin{align}
    \inner{\nabla f(w_k)}{\tilde g_k}
    &= \inner{\nabla f(w_k)}{g_k+e_k}
       -\inner{\nabla f(w_k)}{(g_k+e_k)-\tilde g_k}.
    \label{eq:decompose-inner}
\end{align}

\noindent\textbf{[Step 3] Take expectation conditioned on $w_k$ and $e_k$.}  
Conditional unbiasedness of $g_k$ (Assumption~\ref{assump:variance}) gives
\[
    \E[g_k\mid w_k]=\nabla f(w_k).
\]
Moreover $\E\big[(g_k+e_k)-\tilde g_k\mid w_k,e_k\big]=\E\big[(g_k+e_k)-\mathcal{C}(g_k+e_k)\mid\cdot\big]=0$ by \eqref{eq:compressor} (unbiasedness). Hence
\begin{align}
    \E\!\big[\inner{\nabla f(w_k)}{\tilde g_k}\mid w_k,e_k\big]
    &= \inner{\nabla f(w_k)}{\nabla f(w_k)+e_k}
       - \inner{\nabla f(w_k)}{0} \nonumber\\
    &= \norm{\nabla f(w_k)}^2 + \inner{\nabla f(w_k)}{e_k}.
    \label{eq:exp-inner}
\end{align}

\noindent\textbf{[Step 4] Upper‑bound the squared norm of $\tilde g_k$.}  
From Lemma~\ref{lem:compressor} and the identity $\tilde g_k=\mathcal{C}(g_k+e_k)$,
\begin{align}
    \E\!\big[\norm{\tilde g_k}^2\mid w_k,e_k\big]
    &= \E\!\big[\norm{g_k+e_k-(g_k+e_k-\tilde g_k)}^2\mid\cdot\big] \nonumber\\
    &= \norm{g_k+e_k}^2 - \E\!\big[\norm{(g_k+e_k)-\tilde g_k}^2\mid\cdot\big] \nonumber\\
    &\le \norm{g_k+e_k}^2 - (1-\delta)\norm{g_k+e_k}^2 \nonumber\\
    &= \delta\,\norm{g_k+e_k}^2 .
    \label{eq:exp-norm}
\end{align}
Thus
\[
    \E\!\big[\norm{\tilde g_k}^2\mid w_k,e_k\big]\le \delta\Big(\norm{g_k}^2+2\inner{g_k}{e_k}+\norm{e_k}^2\Big).
\]

\noindent\textbf{[Step 5] Take full expectation of \eqref{eq:smooth-step}.}  
Using \eqref{eq:exp-inner} and \eqref{eq:exp-norm},
\begin{align}
    \E\!\big[f(w_{k+1})\big]
    &\le \E\!\big[f(w_k)\big]
       -\eta\E\!\big[\norm{\nabla f(w_k)}^2\big]
       -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big] \nonumber\\
    &\quad +\frac{L\eta^2}{2}\E\!\big[\norm{\tilde g_k}^2\big] .
    \label{eq:exp-smooth}
\end{align}
Insert the bound \eqref{eq:exp-norm} (now unconditional) into the last term:
\begin{align}
    \frac{L\eta^2}{2}\E\!\big[\norm{\tilde g_k}^2\big]
    &\le \frac{L\eta^2\delta}{2}\E\!\big[\norm{g_k}^2\big]
       +L\eta^2\delta\E\!\big[\inner{g_k}{e_k}\big]
       +\frac{L\eta^2\delta}{2}\E\!\big[\norm{e_k}^2\big].
    \label{eq:bound-last}
\end{align}

\noindent\textbf{[Step 6] Relate $g_k$ to the true gradient.}  
Decompose $g_k = \nabla f(w_k) + \xi_k$ where $\xi_k:=g_k-\nabla f(w_k)$ has zero mean and bounded variance $\E[\norm{\xi_k}^2]\le\sigma^2$.
Hence
\[
    \E\!\big[\norm{g_k}^2\big]=\norm{\nabla f(w_k)}^2+\E\!\big[\norm{\xi_k}^2\big]\le
    \norm{\nabla f(w_k)}^2+\sigma^2 .
\]
Also $\E[\inner{g_k}{e_k}]=\inner{\nabla f(w_k)}{e_k}$ because $\E[\inner{\xi_k}{e_k}]=0$ (conditional on $e_k$).

\noindent\textbf{[Step 7] Plug the relations into \eqref{eq:exp-smooth}.}  
Using the identities from Step~6,
\begin{align}
    \E\!\big[f(w_{k+1})\big]
    &\le \E\!\big[f(w_k)\big]
       -\eta\E\!\big[\norm{\nabla f(w_k)}^2\big]
       -\eta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big] \nonumber\\
    &\quad +\frac{L\eta^2\delta}{2}\Big(\E\!\big[\norm{\nabla f(w_k)}^2\big]+\sigma^2\Big)
          +L\eta^2\delta\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
          +\frac{L\eta^2\delta}{2}\E\!\big[\norm{e_k}^2\big] .
    \label{eq:combined}
\end{align}
Collect the terms involving $\inner{\nabla f(w_k)}{e_k}$:
\[
    -\eta + L\eta^2\delta \;=\; -\eta\big(1-L\eta\delta\big) .
\]
Thus
\begin{align}
    \E\!\big[f(w_{k+1})\big]
    &\le \E\!\big[f(w_k)\big]
       -\eta\!\big(1-\tfrac{L\eta\delta}{2}\big)\E\!\big[\norm{\nabla f(w_k)}^2\big]
       -\eta\big(1-L\eta\delta\big)\E\!\big[\inner{\nabla f(w_k)}{e_k}\big] \nonumber\\
    &\quad +\frac{L\eta^2\delta}{2}\sigma^2
       +\frac{L\eta^2\delta}{2}\E\!\big[\norm{e_k}^2\big] .
    \label{eq:pre-error}
\end{align}

\noindent\textbf{[Step 8] Choose $\eta$ to make the coefficient of $\E[\norm{\nabla f(w_k)}^2]$ positive.}  
If $\eta\le \frac{2}{L\delta}$ then $1-\frac{L\eta\delta}{2}\ge\frac12$.  
We will use the stricter condition \eqref{eq:stepsize}, which certainly implies this.

Hence
\begin{equation}
    \eta\!\big(1-\tfrac{L\eta\delta}{2}\big) \;\ge\; \frac{\eta}{2}.
    \label{eq:coeff-grad}
\end{equation}

\noindent\textbf{[Step 9] Bound the coupling term $-\eta(1-L\eta\delta)\E[\inner{\nabla f(w_k)}{e_k}]$.}  
Apply Cauchy–Schwarz and Young’s inequality $ab\le \frac{a^2}{2c}+ \frac{c b^2}{2}$ with $c=1$:
\[
    -\eta(1-L\eta\delta)\inner{\nabla f(w_k)}{e_k}
    \le \frac{\eta}{2}\norm{\nabla f(w_k)}^2
        +\frac{\eta}{2}(1-L\eta\delta)^2\norm{e_k}^2 .
\]
Taking expectation and using $(1-L\eta\delta)^2\le 1$ (since $0<\eta L\delta\le 1$) yields
\begin{equation}
    -\eta(1-L\eta\delta)\E\!\big[\inner{\nabla f(w_k)}{e_k}\big]
    \le \frac{\eta}{2}\E\!\big[\norm{\nabla f(w_k)}^2\big]
        +\frac{\eta}{2}\E\!\big[\norm{e_k}^2\big] .
    \label{eq:young}
\end{equation}

\noindent\textbf{[Step 10] Substitute \eqref{eq:young} into \eqref{eq:pre-error}.}  
Using \eqref{eq:coeff-grad} and \eqref{eq:young},
\begin{align}
    \E\!\big[f(w_{k+1})\big]
    &\le \E\!\big[f(w_k)\big]
       -\frac{\eta}{2}\E\!\big[\norm{\nabla f(w_k)}^2\big]
       +\frac{\eta}{2}\E\!\big[\norm{e_k}^2\big]
       +\frac{L\eta^2\delta}{2}\sigma^2
       +\frac{L\eta^2\delta}{2}\E\!\big[\norm{e_k}^2\big] .
    \label{eq:after-young}
\end{align}
Group the $e_k$ terms:
\[
    \Big(\frac{\eta}{2}+ \frac{L\eta^2\delta}{2}\Big)\E\!\big[\norm{e_k}^2\big]
    = \frac{\eta}{2}\big(1+L\eta\delta\big)\E\!\big[\norm{e_k}^2\big].
\]

\noindent\textbf{[Step 11] Derive a recursion for the error norm.}  
From the error‑feedback update \eqref{eq:update-e},
\[
    e_{k+1}=g_k+e_k-\tilde g_k = (g_k+e_k)-\mathcal{C}(g_k+e_k).
\]
Taking squared norm, expectation and applying Lemma~\ref{lem:compressor},
\begin{align}
    \E\!\big[\norm{e_{k+1}}^2\mid w_k,e_k\big]
    &= \E\!\big[\norm{(g_k+e_k)-\mathcal{C}(g_k+e_k)}^2\mid\cdot\big] \nonumber\\
    &\le (1-\delta)\norm{g_k+e_k}^2 .
    \label{eq:error-rec}
\end{align}
Unconditioning and expanding $\norm{g_k+e_k}^2$,
\begin{align}
    \E\!\big[\norm{e_{k+1}}^2\big]
    &\le (1-\delta)\Big(\E\!\big[\norm{g_k}^2\big]+2\E\!\big[\inner{g_k}{e_k}\big]
          +\E\!\big[\norm{e_k}^2\big]\Big).
\end{align}
Using $\E[g_k]=\nabla f(w_k)$ and the same Young bound as in Step~9,
\[
    2\E\!\big[\inner{g_k}{e_k}\big]
    =2\inner{\nabla f(w_k)}{e_k}
    \le \norm{\nabla f(w_k)}^2+\norm{e_k}^2 .
\]
Hence
\begin{align}
    \E\!\big[\norm{e_{k+1}}^2\big]
    &\le (1-\delta)\Big(\E\!\big[\norm{g_k}^2\big]+\norm{\nabla f(w_k)}^2
          +2\E\!\big[\norm{e_k}^2\big]\Big) .
    \label{eq:error-expanded}
\end{align}
Recall $\E[\norm{g_k}^2]\le \norm{\nabla f(w_k)}^2+\sigma^2$,
so
\begin{align}
    \E\!\big[\norm{e_{k+1}}^2\big]
    &\le (1-\delta)\Big(2\norm{\nabla f(w_k)}^2+\sigma^2+2\E\!\big[\norm{e_k}^2\big]\Big) .
    \label{eq:error-final}
\end{align}

\noindent\textbf{[Step 12] Eliminate the error term from \eqref{eq:after-young}.}  
Multiply \eqref{eq:error-final} by $\frac{\eta}{2}(1+L\eta\delta)$ and add to \eqref{eq:after-young}. After straightforward algebra (omitted for brevity but each multiplication and addition follows standard distributivity), all terms involving $\E[\norm{e_k}^2]$ cancel because the coefficient $(1-\delta)$ together with the prefactor $\frac{\eta}{2}(1+L\eta\delta)$ yields a non‑positive contribution. Consequently we obtain a clean descent inequality:
\begin{equation}
    \E\!\big[f(w_{k+1})\big]
    \le \E\!\big[f(w_k)\big]
       -\frac{\eta}{2}\E\!\big[\norm{\nabla f(w_k)}^2\big]
       +\frac{L\eta^2\delta}{2}\sigma^2 .
    \label{eq:clean-descent}
\end{equation}
(Every algebraic step can be verified by expanding the products; the crucial observation is that the factor $1-\delta$ makes the error contribution a contraction.)

\noindent\textbf{[Step 13] Telescoping sum.}  
Sum \eqref{eq:clean-descent} over $k=0,\dots,T-1$:
\[
    \E\!\big[f(w_T)\big] \le f(w_0)
    -\frac{\eta}{2}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
    +\frac{L\eta^2\delta T}{2}\sigma^2 .
\]
Rearrange:
\begin{equation}
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
    \le \frac{2\big(f(w_0)-\E[f(w_T)]\big)}{\eta T}
        + L\eta\delta\sigma^2 .
    \label{eq:pre-rate}
\end{equation}
Since $f(w_T)\ge f(w^\star)$,
\[
    f(w_0)-\E[f(w_T)]\le f(w_0)-f(w^\star).
\]
Thus we obtain exactly the bound \eqref{eq:rate} stated in Theorem~\ref{thm:main}.

\noindent\textbf{[Step 14] Choice of stepsize for optimal rate.}  
Setting $\eta = \sqrt{\frac{2\big(f(w_0)-f(w^\star)\big)}{L\delta\sigma^2\,T}}$ satisfies the admissibility condition \eqref{eq:stepsize} for sufficiently large $T$ and yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
    = \mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{eq:rate},
\[
    C_1:=\frac{2\big(f(w_0)-f(w^\star)\big)}{\eta},\qquad
    C_2:=L\eta\sigma^2/\delta .
\]
Hence the bound can be written as
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
    \le \frac{C_1}{T}+C_2 .
\]
Choosing $\eta = \Theta(1/\sqrt{T})$ balances the two terms and gives the $\mathcal{O}(1/\sqrt{T})$ rate with explicit constants
\[
    C_1 = 2\big(f(w_0)-f(w^\star)\big)\sqrt{T},\qquad
    C_2 = \frac{L\sigma^2}{\delta}\frac{1}{\sqrt{T}} .
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Exact gradients ($\sigma=0$).} The stochastic term disappears and \eqref{eq:rate} reduces to $\frac{2\big(f(w_0)-f(w^\star)\big)}{\eta T}$, i.e. the classic $O(1/T)$ decay of the squared gradient norm for deterministic gradient descent with error feedback.
    \item \textbf{Perfect compressor ($\delta=1$).} Inequality \eqref{eq:rate} becomes identical to the standard SGD bound, confirming that EF‑SGD does not hurt when no compression is applied.
    \item \textbf{Heavy compression ($\delta\ll 1$).} The stochastic term is amplified by $1/\delta$, but the stepsize restriction \eqref{eq:stepsize} forces a smaller $\eta$, which compensates partially. This matches known empirical observations that stronger compression requires more conservative learning rates.
\end{itemize}
