\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.  
At iteration $t\in\{0,1,\dots,T-1\}$ we draw a random sample $z_{t}$ and form an unbiased stochastic gradient
\[
g_{t}\;:=\;\nabla f(w_{t};z_{t})\,,\qquad 
\E\!\left[g_{t}\mid w_{t}\right]=\nabla F(w_{t})\,,\quad 
F(w):=\E_{z}[f(w;z)] .
\]

The learning rate follows a *cyclic* schedule of period $M\in\mathbb{N}$:
\[
\eta_{t}\;=\;\eta_{\min}+(\eta_{\max}-\eta_{\min})\,
\phi\!\left(\frac{t\bmod M}{M-1}\right),\qquad 
\phi:[0,1]\rightarrow[0,1]
\]
where $\phi$ is a deterministic shape function (e.g. the triangular wave
$\phi(u)=1-2|u-0.5|$).  Consequently $\eta_{t}\in[\eta_{\min},\eta_{\max}]$ for every $t$.

The update rule is
\[
\boxed{\,w_{t+1}=w_{t}-\eta_{t}g_{t}\,}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\Require Initial point $w_{0}\in\R^{d}$, period $M$, bounds $0<\eta_{\min}\le\eta_{\max}$, shape $\phi(\cdot)$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $z_{t}\sim\mathcal{D}$ and compute $g_{t}=\nabla f(w_{t};z_{t})$
    \State Compute cyclic step size
    \[
    \eta_{t}= \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
    \phi\!\Big(\frac{t\bmod M}{M-1}\Big)
    \]
    \State Update $w_{t+1}=w_{t}-\eta_{t}g_{t}$
\EndFor
\State \textbf{return} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$F$ is $L$‑smooth: for all $x,y\in\R^{d}$
\[
F(y)\le F(x)+\inner{\nabla F(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $t$, $\E[g_{t}\mid w_{t}]=\nabla F(w_{t})$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\!\left[\norm{g_{t}-\nabla F(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2},
\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Cyclic Step‑Size Bounds]\label{as:step}
The learning rates satisfy
\[
0<\eta_{\min}\le\eta_{t}\le\eta_{\max}\qquad\forall t,
\]
and the period $M$ is a fixed positive integer.
\end{assumption}
\begin{assumption}[Step‑size Upper‑Bound]\label{as:stepmax}
The maximal step size is chosen such that
\[
\eta_{\max}\le\frac{1}{L}.
\]
\end{assumption}
Assumption~\ref{as:stepmax} guarantees that the factor
$\eta_{t}-\frac{L}{2}\eta_{t}^{2}$ is non‑negative (see Step 6).

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Under Assumptions~\ref{as:smooth}–\ref{as:stepmax}, let
\[
\tilde\eta_T:=\frac{1}{T}\sum_{t=0}^{T-1}\eta_{t}
\qquad(\text{empirical average step size over the first $T$ iterates}).
\]
For any $T\ge1$, the iterates generated by Cyclic SGD satisfy
\begin{equation}\label{eq:bound}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla F(w_{t})}^{2}\right]
\;\le\;
\frac{2\big(F(w_{0})-F^{*}\big)}{\eta_{\min}\,T}
\;+\;
\frac{L\,\eta_{\max}^{2}}{\eta_{\min}}\;\sigma^{2},
\end{equation}
where $F^{*}:=\inf_{w}F(w)$.  Consequently,
\[
\min_{0\le t<T}\E\!\left[\norm{\nabla F(w_{t})}^{2}\right]
=O\!\left(\frac{1}{T}\right)+O\!\left(\eta_{\max}\right)
\quad\text{provided that }\eta_{\min}=c\,\eta_{\max}\text{ for some }c\in(0,1].
\]
\end{theorem}

\section*{Preliminaries}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $F$ is $L$‑smooth, then for any $w$ and any stochastic gradient $g$,
\[
F(w-\eta g)
\le F(w)-\eta\inner{\nabla F(w)}{g}
+\frac{L\eta^{2}}{2}\norm{g}^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption~\ref{as:smooth} with $x=w$, $y=w-\eta g$ and rearrange. ∎
\end{proof}

\begin{lemma}[Quadratic Step‑size Bound]\label{lem:quad}
For any $t$ we have $\eta_{t}^{2}\le\eta_{\max}\,\eta_{t}$.
\end{lemma}
\begin{proof}
Since $0\le\eta_{t}\le\eta_{\max}$, multiplying the inequality $\eta_{t}\le\eta_{\max}$ by the non‑negative $\eta_{t}$ yields $\eta_{t}^{2}\le\eta_{\max}\eta_{t}$. ∎
\end{proof}

\section*{Proof of Theorem~\ref{thm:main}}
We keep the original step numbering and correct the problematic steps.

\noindent\textbf{Step 1 (Apply smoothness).}  
Using Lemma~\ref{lem:smooth} with $g=g_{t}$ and $\eta=\eta_{t}$,
\[
F(w_{t+1})\le F(w_{t})-\eta_{t}\inner{\nabla F(w_{t})}{g_{t}}
+\frac{L\eta_{t}^{2}}{2}\norm{g_{t}}^{2}.
\tag{2}
\]

\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $w_{t}$ and use unbiasedness (Assumption~\ref{as:unbiased}) and the variance bound (Assumption~\ref{as:variance}):
\[
\begin{aligned}
\E\!\big[F(w_{t+1})\mid w_{t}\big]
&\le F(w_{t})-\eta_{t}\norm{\nabla F(w_{t})}^{2}
      +\frac{L\eta_{t}^{2}}{2}\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big] .
\end{aligned}
\tag{3}
\]

\noindent\textbf{Step 3 (Expand $\E[\|g_{t}\|^{2}]$).}  
\[
\begin{aligned}
\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]
&=\norm{\nabla F(w_{t})}^{2}
  +\E\!\big[\norm{g_{t}-\nabla F(w_{t})}^{2}\mid w_{t}\big] \\
&\le \norm{\nabla F(w_{t})}^{2}+\sigma^{2}.
\end{aligned}
\tag{4}
\]

\noindent\textbf{Step 4 (Insert (4) into (3)).}
\[
\E\!\big[F(w_{t+1})\mid w_{t}\big]
\le F(w_{t})-\eta_{t}\norm{\nabla F(w_{t})}^{2}
+\frac{L\eta_{t}^{2}}{2}\big(\norm{\nabla F(w_{t})}^{2}+\sigma^{2}\big).
\tag{5}
\]

\noindent\textbf{Step 5 (Collect gradient terms).}
\[
\E\!\big[F(w_{t+1})\mid w_{t}\big]
\le F(w_{t})-\Big(\eta_{t}-\frac{L\eta_{t}^{2}}{2}\Big)\norm{\nabla F(w_{t})}^{2}
+\frac{L\eta_{t}^{2}}{2}\sigma^{2}.
\tag{6}
\]

\noindent\textbf{Step 6 (Lower bound the coefficient).}  
Because of Assumption~\ref{as:stepmax} we have $\eta_{t}\le\eta_{\max}\le 1/L$. Hence
\[
\eta_{t}-\frac{L\eta_{t}^{2}}{2}
= \eta_{t}\Big(1-\frac{L\eta_{t}}{2}\Big)
\ge \eta_{t}\Big(1-\frac{1}{2}\Big)
= \frac{\eta_{t}}{2}
\ge \frac{\eta_{\min}}{2}.
\tag{7}
\]
% CORRECTED: the inequality now follows from the stronger step‑size bound $\eta_{\max}\le 1/L$.

\noindent\textbf{Step 7 (Take full expectation).}  
Taking expectation over all randomness up to time $t$ in (6) and using (7) gives
\[
\E\!\big[F(w_{t+1})\big]
\le \E\!\big[F(w_{t})\big]
-\frac{\eta_{\min}}{2}\,\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
+\frac{L}{2}\,\E\!\big[\eta_{t}^{2}\big]\sigma^{2}.
\tag{8}
\]

\noindent\textbf{Step 8 (Replace $\eta_{t}^{2}$ by $\eta_{\max}\eta_{t}$).}  
By Lemma~\ref{lem:quad}, $\eta_{t}^{2}\le\eta_{\max}\eta_{t}$, therefore
\[
\frac{L}{2}\,\E\!\big[\eta_{t}^{2}\big]\sigma^{2}
\le
\frac{L\eta_{\max}}{2}\,\E\!\big[\eta_{t}\big]\sigma^{2}.
\tag{9}
\]

\noindent\textbf{Step 9 (Insert (9) into (8)).}
\[
\E\!\big[F(w_{t+1})\big]
\le \E\!\big[F(w_{t})\big]
-\frac{\eta_{\min}}{2}\,\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
+\frac{L\eta_{\max}}{2}\,\E\!\big[\eta_{t}\big]\sigma^{2}.
\tag{10}
\]

\noindent\textbf{Step 10 (Telescoping sum).}  
Sum (10) over $t=0,\dots,T-1$:
\[
\E\!\big[F(w_{T})\big]
\le F(w_{0})
-\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
+\frac{L\eta_{\max}}{2}\sigma^{2}\sum_{t=0}^{T-1}\E\!\big[\eta_{t}\big].
\tag{11}
\]

\noindent\textbf{Step 11 (Rearrange).}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\le
\frac{2\big(F(w_{0})-\E[F(w_{T})]\big)}{\eta_{\min}T}
+ \frac{L\eta_{\max}}{\eta_{\min}}\,
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\eta_{t}\big]\sigma^{2}.
\tag{12}
\]

\noindent\textbf{Step 12 (Bound the remaining terms).}  
Since $F^{*}\le \E[F(w_{T})]$ and $\eta_{t}\le\eta_{\max}$ for every $t$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\eta_{t}\big]\le\eta_{\max}.
\]
Plugging these two facts into (12) yields exactly the bound~\eqref{eq:bound}.

\noindent\textbf{Step 13 (Deriving the rate).}  
Inequality~\eqref{eq:bound} directly implies
\[
\min_{0\le t<T}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\;\le\;
\frac{2\big(F(w_{0})-F^{*}\big)}{\eta_{\min}T}
\;+\;
\frac{L\,\eta_{\max}^{2}}{\eta_{\min}}\;\sigma^{2}.
\]
If the cyclic schedule is such that $\eta_{\min}=c\,\eta_{\max}$ for a fixed
constant $c\in(0,1]$, the second term scales as $O(\eta_{\max})$.
Choosing a diminishing maximal step size, e.g. $\eta_{\max}=c'/\sqrt{T}$,
gives the optimal $O(1/\sqrt{T})$ rate for non‑convex stochastic optimization. ∎

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because every $\eta_{t}\le\eta_{\max}\le 1/L$, the smoothness inequality guarantees that $F(w_{t})$ does not increase in expectation.
\item \textbf{Hyper‑parameter Sensitivity.}  The bound~\eqref{eq:bound} depends on the \emph{minimum} step size $\eta_{\min}$ in the $1/T$ term and on the \emph{maximum} step size $\eta_{\max}$ (quadratically) in the variance term.  When $\eta_{\min}$ is a fixed fraction of $\eta_{\max}$, the variance term behaves as $O(\eta_{\max})$.
\item \textbf{Comparison to Classical SGD.}  For a constant step size $\eta$ the standard guarantee reads 
\(
\frac{1}{T}\sum_{t}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\le \frac{2(F(w_{0})-F^{*})}{\eta T}+L\eta\sigma^{2}.
\)
Cyclic SGD replaces $\eta$ by $\eta_{\min}$ in the first term and by $\eta_{\max}^{2}/\eta_{\min}$ in the second, preserving the optimal $O(1/T)$ decay while allowing periodic exploration via larger step sizes.
\end{itemize}

\section*{Special Cases}
\begin{itemize}
\item \textbf{Constant learning rate.}  Setting $\eta_{\min}=\eta_{\max}=\eta$ reduces the cyclic schedule to classical SGD and Theorem~\ref{thm:main} collapses to the standard bound.
\item \textbf{Two‑phase schedule.}  If $\phi$ is a step function that spends a fraction $\alpha\in(0,1)$ of each period at $\eta_{\max}$ and $(1-\alpha)$ at $\eta_{\min}$, then $\tilde\eta_T=\alpha\eta_{\max}+(1-\alpha)\eta_{\min}$.  The theorem still holds with the same proof.
\item \textbf{Vanishing amplitude.}  When $\eta_{\max}-\eta_{\min}\to0$, the cyclic scheme converges to constant‑step SGD, and the variance term $L\eta_{\max}^{2}/\eta_{\min}\,\sigma^{2}$ shrinks accordingly.
\end{itemize}
