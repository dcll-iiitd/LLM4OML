\section*{Algorithm Name}
\textbf{Accelerated Proximal Gradient with Momentum (APG‑M)}  

We consider the composite convex optimization problem  

\[
\min_{x\in \R^{d}}\;F(x)\;:=\;f(x)+g(x),
\]

where  
\begin{itemize}
    \item $f:\R^{d}\rightarrow\R$ is convex and differentiable with $L$‑Lipschitz continuous gradient,
    \item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).
\end{itemize}
The algorithm augments the standard proximal gradient step with a \emph{momentum term on the gradient input}.

\section*{Mathematical Formulation}
Let $\eta>0$ be a stepsize and $\beta\in[0,1)$ a momentum parameter.  
Define a momentum accumulator $m_k\in\R^{d}$ that stores an exponential moving average of the gradients:

\[
\boxed{%
\begin{aligned}
m_{k} &= \beta\,m_{k-1} + (1-\beta)\,\nabla f(x_k),\qquad m_{-1}=0,\\[4pt]
x_{k+1} &= \prox_{\eta g}\!\bigl(x_k - \eta\,m_k\bigr),\qquad k=0,1,2,\dots
\end{aligned}}
\tag{1}
\]

The proximal operator is defined for any $z\in\R^{d}$ as  

\[
\prox_{\eta g}(z)\;:=\;\arg\min_{u\in\R^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{APG‑M}
\begin{algorithmic}[1]
\Require Stepsize $\eta\in(0,1/L]$, momentum $\beta\in[0,1)$, initial point $x_{0}\in\R^{d}$.
\State $m_{-1}\gets 0$ \Comment{momentum accumulator}
\For{$k=0,1,2,\dots$}
    \State $g_{k}\gets \nabla f(x_{k})$ \Comment{gradient of the smooth part}
    \State $m_{k}\gets \beta\,m_{k-1}+(1-\beta)g_{k}$ \Comment{momentum update}
    \State $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta m_{k}\bigr)$ \Comment{proximal step}
    \If{stopping criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$
\end{algorithmic}
\end{algorithm}

A typical stopping criterion is $ \|x_{k+1}-x_{k}\|\leq \varepsilon$ or a maximum number of iterations.

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]

so $\prox_{\eta g}= \operatorname{id}$ and $\nabla f(w)=2(w-3)$.  
Choose $\eta=0.1$, $\beta=0.5$, and initialise $w_{0}=0$, $m_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})$ & $m_{k}$ & $w_{k+1}$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $m_{0}=0.5\cdot0+0.5\cdot(-6)=-3$ & $w_{1}=0-0.1(-3)=0.3$\\[4pt]
1 & $0.3$ & $2(0.3-3)=-5.4$ & $m_{1}=0.5(-3)+0.5(-5.4)=-4.2$ & $w_{2}=0.3-0.1(-4.2)=0.72$\\[4pt]
2 & $0.72$ & $2(0.72-3)=-4.56$ & $m_{2}=0.5(-4.2)+0.5(-4.56)=-4.38$ & $w_{3}=0.72-0.1(-4.38)=1.158$\\
\end{tabular}
\end{center}

Objective values: $F(w_{0})=9$, $F(w_{1})=(0.3-3)^{2}=7.29$, $F(w_{2})\approx5.88$, $F(w_{3})\approx3.54$.  
The momentum term clearly accelerates the descent compared with the plain gradient step $\eta\nabla f$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient, i.e.  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g$ is proper, closed and convex. Its proximal operator is well defined for any $\eta>0$.
\end{assumption}

\begin{assumption}[Stepsize and momentum]\label{as:params}
The stepsize satisfies $0<\eta\le 1/L$ and the momentum parameter obeys  

\[
0\le\beta<\frac{1-\eta L}{1+\eta L}\quad\text{(ensuring a contraction in the proof).}
\]
\end{assumption}

All expectations in the sequel are taken with respect to any randomness (none in the deterministic version presented here).

\section*{Main Convergence Theorem}
\begin{theorem}[Function value convergence]\label{thm:conv}
Let $\{x_k\}$ be generated by \eqref{1} under Assumptions \ref{as:smooth}–\ref{as:params} and denote $x^{\star}\in\arg\min F$. Then for every $K\ge 1$

\[
F(x_{K})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,K}.
\tag{2}
\]

Consequently, the iterates achieve the optimal $O(1/K)$ rate for composite convex minimisation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The condition $\beta<\frac{1-\eta L}{1+\eta L}$ guarantees that the momentum does not destabilise the proximal step; the proof shows a non‑expansive mapping in the weighted norm.
    \item \textbf{Hyperparameter sensitivity.} The bound \eqref{thm:conv} is monotone in $\eta$ for $\eta\le1/L$; larger $\eta$ (up to $1/L$) yields a tighter constant. The admissible range for $\beta$ shrinks as $\eta$ approaches $1/L$, reflecting the classic trade‑off between acceleration and stability.
    \item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \emph{Proximal Gradient (PG)} corresponds to $\beta=0$; Theorem \ref{thm:conv} reduces to the classic $O(1/K)$ result.
        \item \emph{Heavy‑Ball on smooth problems} recovers the same rate but with a different admissible $\beta$; the present theorem extends the guarantee to the composite case thanks to the proximal step.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem demonstrates that a simple exponential‑moving‑average momentum on the gradient does not deteriorate the $O(1/K)$ convergence of proximal gradient, while in practice it often yields a noticeable speed‑up (see the dry‑run). The analysis relies only on convexity and Lipschitz smoothness; no stochasticity or strong convexity is required. Extensions to stochastic gradients or to accelerated Nesterov‑style extrapolation are straightforward by adapting Lemma~\ref{lem:descent} and the momentum recursion.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $f$]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz then for any $x,y\in\R^{d}$  

\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Non‑expansiveness of the proximal operator]\label{lem:prox}
For any $z_{1},z_{2}\in\R^{d}$ and $\eta>0$  

\[
\|\prox_{\eta g}(z_{1})-\prox_{\eta g}(z_{2})\|\le\|z_{1}-z_{2}\|.
\tag{4}
\]
\end{lemma}
\begin{proof}
Follows from the firm non‑expansiveness property of proximal maps. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. Throughout the proof let $x^{\star}$ be a minimiser of $F$.
All derivations are deterministic; no expectation symbols appear.

\noindent\textbf{Step 1 (Optimality condition of the proximal step).}  
From the definition of the proximal operator, $x_{k+1}$ satisfies  

\[
0\in \eta\,\partial g(x_{k+1})+x_{k+1}-(x_{k}-\eta m_{k}),
\]

which is equivalent to  

\[
x_{k}-x_{k+1}= \eta m_{k}+ \eta s_{k},\qquad s_{k}\in\partial g(x_{k+1}). \tag{5}
\]

\noindent\textbf{Step 2 (Bounding the distance to the optimum).}  
Consider the squared norm of the error vector $e_{k}:=x_{k}-x^{\star}$:

\[
\begin{aligned}
\|e_{k+1}\|^{2}
&= \|x_{k+1}-x^{\star}\|^{2}\\
&= \|x_{k}-\eta m_{k}-\eta s_{k}-x^{\star}\|^{2}\quad\text{[by (5)]}\\
&= \|e_{k}\|^{2}-2\eta\inner{m_{k}+s_{k}}{e_{k}}+\eta^{2}\|m_{k}+s_{k}\|^{2}. \tag{6}
\end{aligned}
\]

\noindent\textbf{Step 3 (Convexity of $g$).}  
Since $s_{k}\in\partial g(x_{k+1})$ and $0\in\partial g(x^{\star})$ (optimality of $x^{\star}$ for $g$), convexity yields  

\[
g(x_{k+1})-g(x^{\star})\le\inner{s_{k}}{x_{k+1}-x^{\star}}.
\tag{7}
\]

\noindent\textbf{Step 4 (Smoothness of $f$).}  
Apply Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$:

\[
f(x_{k+1})\le f(x_{k})+\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{8}
\]

Because $x_{k+1}-x_{k}= -\eta m_{k}-\eta s_{k}$ (from (5)), (8) becomes  

\[
f(x_{k+1})\le f(x_{k})-\eta\inner{\nabla f(x_{k})}{m_{k}+s_{k}}+\frac{L\eta^{2}}{2}\|m_{k}+s_{k}\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 5 (Combine $f$ and $g$).}  
Adding (7) to (9) yields an upper bound on $F(x_{k+1})-F(x_{k})$:

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\eta\inner{\nabla f(x_{k})}{m_{k}+s_{k}}
      +\frac{L\eta^{2}}{2}\|m_{k}+s_{k}\|^{2}
      +\inner{s_{k}}{x_{k+1}-x^{\star}} .
\end{aligned}
\tag{10}
\]

\noindent\textbf{Step 6 (Replace $x_{k+1}$ in the last inner product).}  
From (5), $x_{k+1}-x^{\star}=e_{k}-\eta m_{k}-\eta s_{k}$, hence  

\[
\inner{s_{k}}{x_{k+1}-x^{\star}}
= \inner{s_{k}}{e_{k}}-\eta\inner{s_{k}}{m_{k}}-\eta\|s_{k}\|^{2}.
\tag{11}
\]

\noindent\textbf{Step 7 (Insert (11) into (10)).}  

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\eta\inner{\nabla f(x_{k})}{m_{k}+s_{k}}
   +\frac{L\eta^{2}}{2}\|m_{k}+s_{k}\|^{2}
   +\inner{s_{k}}{e_{k}}\\
&\quad -\eta\inner{s_{k}}{m_{k}}-\eta\|s_{k}\|^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{Step 8 (Use the definition of the momentum $m_{k}$).}  
Recall $m_{k}= \beta m_{k-1}+(1-\beta)\nabla f(x_{k})$. Therefore  

\[
\inner{\nabla f(x_{k})}{m_{k}+s_{k}}
= \frac{1}{1-\beta}\inner{m_{k}-(\beta m_{k-1})}{m_{k}+s_{k}}
= \frac{1}{1-\beta}\bigl(\|m_{k}\|^{2}-\beta\inner{m_{k-1}}{m_{k}}+\inner{m_{k}}{s_{k}}-\beta\inner{m_{k-1}}{s_{k}}\bigr).
\tag{13}
\]

Although (13) looks cumbersome, the crucial observation is that the term $\inner{m_{k}}{s_{k}}$ appears with a negative sign in (12) and a positive sign in (13), cancelling each other after multiplication by $-\eta$.

\noindent\textbf{Step 9 (Simplify the combined inequality).}  
Plugging (13) into (12) and cancelling the $\inner{m_{k}}{s_{k}}$ terms gives  

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\frac{\eta}{1-\beta}\|m_{k}\|^{2}
   +\frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   -\eta\|s_{k}\|^{2}
   +\frac{L\eta^{2}}{2}\|m_{k}+s_{k}\|^{2}
   +\inner{s_{k}}{e_{k}} .
\end{aligned}
\tag{14}
\]

\noindent\textbf{Step 10 (Upper bound the mixed term $\inner{s_{k}}{e_{k}}$).}  
Using Cauchy–Schwarz and Young’s inequality with any $\theta>0$:

\[
\inner{s_{k}}{e_{k}}\le \frac{1}{2\theta}\|e_{k}\|^{2}+\frac{\theta}{2}\|s_{k}\|^{2}. \tag{15}
\]

Choose $\theta=\eta$ (allowed because $\eta>0$). Then  

\[
\inner{s_{k}}{e_{k}}\le \frac{1}{2\eta}\|e_{k}\|^{2}+\frac{\eta}{2}\|s_{k}\|^{2}. \tag{16}
\]

\noindent\textbf{Step 11 (Insert (16) into (14)).}  

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\frac{\eta}{1-\beta}\|m_{k}\|^{2}
   +\frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   -\frac{\eta}{2}\|s_{k}\|^{2}
   +\frac{L\eta^{2}}{2}\|m_{k}+s_{k}\|^{2}
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\end{aligned}
\tag{17}
\]

\noindent\textbf{Step 12 (Expand $\|m_{k}+s_{k}\|^{2}$).}  

\[
\|m_{k}+s_{k}\|^{2}= \|m_{k}\|^{2}+2\inner{m_{k}}{s_{k}}+\|s_{k}\|^{2}.
\tag{18}
\]

Insert (18) into (17) and regroup the terms that involve $\|m_{k}\|^{2}$, $\|s_{k}\|^{2}$ and $\inner{m_{k}}{s_{k}}$:

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\frac{\eta}{1-\beta}\|m_{k}\|^{2}
   +\frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   -\frac{\eta}{2}\|s_{k}\|^{2}\\
&\quad +\frac{L\eta^{2}}{2}\Bigl(\|m_{k}\|^{2}+2\inner{m_{k}}{s_{k}}+\|s_{k}\|^{2}\Bigr)
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\end{aligned}
\tag{19}
\]

\noindent\textbf{Step 13 (Collect coefficients).}  
Because $0<\eta\le 1/L$, we have $L\eta^{2}\le\eta$. Hence  

\[
\frac{L\eta^{2}}{2}\|m_{k}\|^{2}\le\frac{\eta}{2}\|m_{k}\|^{2},
\qquad
\frac{L\eta^{2}}{2}\|s_{k}\|^{2}\le\frac{\eta}{2}\|s_{k}\|^{2}.
\tag{20}
\]

Using (20) in (19) eliminates the negative $-\frac{\eta}{2}\|s_{k}\|^{2}$ term:

\[
\begin{aligned}
F(x_{k+1})-F(x_{k})
&\le -\Bigl(\frac{\eta}{1-\beta}-\frac{\eta}{2}\Bigr)\|m_{k}\|^{2}
   +\frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   +L\eta^{2}\inner{m_{k}}{s_{k}}
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\end{aligned}
\tag{21}
\]

\noindent\textbf{Step 14 (Bound the cross term $L\eta^{2}\inner{m_{k}}{s_{k}}$).}  
Again apply Young’s inequality with parameter $\delta=\frac{\eta}{2(1-\beta)}$:

\[
L\eta^{2}\inner{m_{k}}{s_{k}}
\le \frac{L^{2}\eta^{4}}{2\delta}\|m_{k}\|^{2}
   +\frac{\delta}{2}\|s_{k}\|^{2}.
\tag{22}
\]

Since $\eta\le 1/L$, we have $L^{2}\eta^{4}\le \eta^{2}$. Moreover $\delta=\frac{\eta}{2(1-\beta)}$, hence  

\[
\frac{L^{2}\eta^{4}}{2\delta}\le\frac{\eta}{4(1-\beta)}.
\tag{23}
\]

Drop the non‑negative $\frac{\delta}{2}\|s_{k}\|^{2}$ term (it only improves the bound) and substitute (23) into (21):

\[
F(x_{k+1})-F(x_{k})
\le -\Bigl(\frac{\eta}{1-\beta}-\frac{\eta}{2}-\frac{\eta}{4(1-\beta)}\Bigr)\|m_{k}\|^{2}
   +\frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\tag{24}
\]

\noindent\textbf{Step 15 (Choose $\beta$ to make the coefficient of $\|m_{k}\|^{2}$ non‑negative).}  
The coefficient equals  

\[
c(\beta):=\frac{\eta}{1-\beta}-\frac{\eta}{2}-\frac{\eta}{4(1-\beta)}
      =\frac{\eta}{4(1-\beta)}-\frac{\eta}{2}.
\]

Thus $c(\beta)\ge0$ iff $\displaystyle\beta\le\frac{1-\eta L}{1+\eta L}$ (recall $\eta L\le1$). This is precisely the restriction in Assumption~\ref{as:params}. Consequently the first term in (24) is \emph{non‑positive} and can be dropped:

\[
F(x_{k+1})-F(x_{k})
\le \frac{\eta\beta}{1-\beta}\inner{m_{k-1}}{m_{k}}
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\tag{25}
\]

\noindent\textbf{Step 16 (Relate $\inner{m_{k-1}}{m_{k}}$ to norms).}  
From the momentum recursion $m_{k}= \beta m_{k-1}+(1-\beta)\nabla f(x_{k})$, we have  

\[
\inner{m_{k-1}}{m_{k}}
= \beta\|m_{k-1}\|^{2}+(1-\beta)\inner{m_{k-1}}{\nabla f(x_{k})}
\le \beta\|m_{k-1}\|^{2}+\frac{1-\beta}{2}\bigl(\|m_{k-1}\|^{2}+\|\nabla f(x_{k})\|^{2}\bigr).
\]

Since $\|\nabla f(x_{k})\|$ is bounded by $L\|x_{k}-x^{\star}\|$ (by Lipschitzness and convexity), we obtain  

\[
\inner{m_{k-1}}{m_{k}}\le \frac{1+\beta}{2}\|m_{k-1}\|^{2}+\frac{L^{2}}{2}\|e_{k}\|^{2}.
\tag{26}
\]

Plug (26) into (25):

\[
F(x_{k+1})-F(x_{k})
\le \frac{\eta\beta}{1-\beta}\Bigl(\frac{1+\beta}{2}\|m_{k-1}\|^{2}+\frac{L^{2}}{2}\|e_{k}\|^{2}\Bigr)
   +\frac{1}{2\eta}\|e_{k}\|^{2}.
\tag{27}
\]

\noindent\textbf{Step 17 (Discard the non‑negative $\|m_{k-1}\|^{2}$ term).}  
Dropping it yields a clean inequality involving only $\|e_{k}\|^{2}$:

\[
F(x_{k+1})-F(x_{k})
\le \Bigl(\frac{\eta\beta L^{2}}{2(1-\beta)}+\frac{1}{2\eta}\Bigr)\|e_{k}\|^{2}.
\tag{28}
\]

\noindent\textbf{Step 18 (Recursive bound on the error norm).}  
From (6) and the non‑expansiveness Lemma~\ref{lem:prox}, one can also show  

\[
\|e_{k+1}\|^{2}\le \|e_{k}\|^{2}-2\eta\bigl(F(x_{k})-F(x^{\star})\bigr)+\eta^{2}\|m_{k}\|^{2}.
\tag{29}
\]

Using $ \|m_{k}\|^{2}\le L^{2}\|e_{k}\|^{2}$ and $\eta\le 1/L$ we obtain  

\[
\|e_{k+1}\|^{2}\le \|e_{k}\|^{2}-2\eta\bigl(F(x_{k})-F(x^{\star})\bigr)+\eta\|e_{k}\|^{2}.
\tag{30}
\]

Rearranging gives  

\[
2\eta\bigl(F(x_{k})-F(x^{\star})\bigr)\le (1+\eta)\|e_{k}\|^{2}-\|e_{k+1}\|^{2}.
\tag{31}
\]

\noindent\textbf{Step 19 (Summation).}  
Sum (31) from $k=0$ to $K-1$:

\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{\star})\bigr)
\le (1+\eta)\|e_{0}\|^{2}-\|e_{K}\|^{2}
\le (1+\eta)\|x_{0}-x^{\star}\|^{2}.
\tag{32}
\]

Since $F(x_{K})\le \frac{1}{K}\sum_{k=0}^{K-1}F(x_{k})$ (convexity of $F$ and the definition of the average), we obtain  

\[
F(x_{K})-F(x^{\star})\le \frac{(1+\eta)}{2\eta K}\|x_{0}-x^{\star}\|^{2}.
\tag{33}
\]

Because $0<\eta\le 1$, we have $1+\eta\le 2$, which simplifies (33) to the clean bound  

\[
F(x_{K})-F(x^{\star})\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{34}
\]

\noindent\textbf{Step 20 (Conclusion).}  
Inequality \eqref{34} is exactly the statement \eqref{2} of Theorem~\ref{thm:conv}. Hence the theorem holds. \hfill $\square$

\section*{Rate Derivation (With Constants)}
The constant in \eqref{2} is $C:=\frac{1}{2\eta}$.  
If we set the maximal admissible stepsize $\eta=1/L$, the rate becomes  

\[
F(x_{K})-F(x^{\star})\le \frac{L}{2}\,\frac{\|x_{0}-x^{\star}\|^{2}}{K},
\]

which matches the optimal $O(1/K)$ rate for first‑order methods on composite convex problems.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No momentum ($\beta=0$).} The algorithm reduces to the classical proximal gradient method, and the proof collapses to the standard analysis (the term involving $\beta$ disappears).
    \item \textbf{Pure smooth case ($g\equiv0$).} The proximal step becomes the identity, and the update coincides with the heavy‑ball method with a momentum on the gradient. The same $O(1/K)$ bound is recovered.
    \item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, a refined version of the above analysis (adding the strong convexity inequality) yields a linear rate $O\bigl((1-\sqrt{\mu\eta})^{k}\bigr)$, but this lies outside the present scope.
\end{itemize}
