\section*{Subsampled Stochastic Newton (SSN)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a twice differentiable (possibly non‑convex) loss.  
At iteration $t$ we draw a random mini‑batch $\mathcal{B}_{t}$ and form

\[
g_{t}\;:=\;\frac{1}{|\mathcal{B}_{t}|}\sum_{i\in\mathcal{B}_{t}}\grad f_{i}(x_{t}),\qquad
H_{t}\;:=\;\frac{1}{|\mathcal{S}_{t}|}\sum_{j\in\mathcal{S}_{t}}\hess f_{j}(x_{t}),
\]

where $\{f_{i}\}$ are the per‑sample losses and $\mathcal{S}_{t}$ is a (possibly different) subsample used for a Hessian estimate.  
The SSN update is

\begin{equation}\label{eq:update}
x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}g_{t},
\end{equation}
with a deterministic step size $\alpha>0$ (or a prescribed schedule $\alpha_{t}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\DontPrintSemicolon
\SetAlgoLined
\KwIn{Initial point $x_{0}\in\mathbb{R}^{d}$, step size $\alpha>0$, batch sizes $b_{g},b_{H}$}
\For{$t=0,1,\dots,T-1$}{
    Sample a gradient mini‑batch $\mathcal{B}_{t}$ of size $b_{g}$\;
    Compute stochastic gradient $g_{t}=\frac{1}{b_{g}}\sum_{i\in\mathcal{B}_{t}}\grad f_{i}(x_{t})$\;
    Sample a Hessian mini‑batch $\mathcal{S}_{t}$ of size $b_{H}$\;
    Compute stochastic Hessian $H_{t}=\frac{1}{b_{H}}\sum_{j\in\mathcal{S}_{t}}\hess f_{j}(x_{t})$\;
    \tcp{Regularise if needed: $H_{t}\leftarrow H_{t}+\lambda I$}
    Update $x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}g_{t}$\;
}
\Return $x_{T}$
\caption{Subsampled Stochastic Newton (SSN)}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$.  
Exact gradients and Hessians are $\grad f(w)=2(w-3)$ and $\hess f(w)=2$.  
We use the true (deterministic) gradient and Hessian as stochastic estimates, i.e. $g_{t}=\grad f(w_{t})$, $H_{t}=2$.  
Take $w_{0}=0$, $\alpha=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\alpha H_{t}^{-1}g_{t}$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1\cdot(1/2)\cdot(-6)=0.3$ & $(0.3-3)^{2}=7.29$\\
1 & $2(0.3-3)=-5.4$ & $0.3-0.1\cdot(1/2)\cdot(-5.4)=0.57$ & $(0.57-3)^{2}=5.9049$\\
2 & $2(0.57-3)=-4.86$ & $0.57-0.1\cdot(1/2)\cdot(-4.86)=0.813$ & $(0.813-3)^{2}=4.7777$\\
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the SSN dynamics.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ has $L$‑Lipschitz gradient:
\[
\norm{\grad f(x)-\grad f(y)}\le L\norm{x-y},\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Hessian Lipschitz]\label{ass:hesslip}
The Hessian is $\rho$‑Lipschitz:
\[
\norm{\hess f(x)-\hess f(y)}\le\rho\norm{x-y},\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{ass:gradunb}
\[
\E[g_{t}\mid x_{t}]=\grad f(x_{t}),\qquad
\E\big[\norm{g_{t}-\grad f(x_{t})}^{2}\mid x_{t}\big]\le\sigma_{g}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic Hessian]\label{ass:hessunb}
\[
\E[H_{t}\mid x_{t}]=\hess f(x_{t}),\qquad
\E\big[\norm{H_{t}-\hess f(x_{t})}^{2}\mid x_{t}\big]\le\sigma_{H}^{2}.
\]
\end{assumption}
\begin{assumption}[Uniformly bounded Hessian]\label{ass:hessbound}
There exist constants $0<\mu\le L$ such that for all $x$,
\[
\mu I\preceq \hess f(x)\preceq L I .
\]
Consequently $\norm{H_{t}^{-1}}\le 1/\mu$ almost surely.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:hessbound} hold and let the step size be constant
\[
0<\alpha\le\frac{\mu}{L^{2}+ \mu\rho R},
\]
where $R\ge\max_{0\le t<T}\norm{x_{t}-x^{*}}$ (any finite bound on the iterates). Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(x_{t})}^{2}\big]
\;\le\;
\frac{2\big(f(x_{0})-f^{*}\big)}{\alpha\mu\,T}
\;+\;
\underbrace{\frac{2\alpha\sigma_{g}^{2}}{\mu}}_{\text{gradient variance term}}
\;+\;
\underbrace{\frac{2\alpha^{2}L^{2}\sigma_{H}^{2}}{\mu^{3}}}_{\text{Hessian variance term}} .
\]
Consequently, choosing $\alpha=O(T^{-1/2})$ yields the optimal non‑convex rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(x_{t})}^{2}\big]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound on $\norm{H_{t}^{-1}}$ (Assumption~\ref{ass:hessbound}) guarantees that the Newton step does not explode even when the Hessian is poorly conditioned.
\item \textbf{Hyper‑parameter sensitivity.} The admissible range for $\alpha$ depends on $\mu$, $L$, $\rho$, and a radius $R$ that controls the deviation of iterates. In practice a modest $\alpha$ (e.g. $10^{-2}$) works well when $L$ and $\rho$ are of order $1$.
\item \textbf{Comparison to SGD.} For SGD the standard bound is $O(1/\sqrt{T})$ on the average gradient norm with a constant stepsize. SSN enjoys the same asymptotic rate but with a smaller constant because the pre‑conditioner $H_{t}^{-1}$ reduces the effective smoothness from $L$ to $\mu$.
\item \textbf{Special case – exact Hessian.} If $H_{t}\equiv\hess f(x_{t})$ (no variance) and $\alpha=1$, inequality \eqref{eq:descent} reduces to the classic deterministic Newton decrease, giving a quadratic convergence neighbourhood.
\end{itemize}

\section*{Discussion}
The theorem shows that a subsampled Newton method attains the same $O(T^{-1/2})$ non‑convex rate as first‑order methods while exploiting curvature information to improve the leading constant. The proof follows the standard smoothness‑descent template but requires careful handling of the stochastic inverse Hessian. The line‑by‑line annotation below makes each manipulation transparent.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness descent]\label{lem:smooth}
For any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Direct consequence of $L$‑Lipschitz gradient (integrate the gradient bound). \qedhere
\end{proof}

\begin{lemma}[Bound on stochastic Newton step]\label{lem:stepbound}
Under Assumption~\ref{ass:hessbound},
\[
\norm{H_{t}^{-1}g_{t}}\le\frac{1}{\mu}\norm{g_{t}}.
\]
\end{lemma}
\begin{proof}
Since $H_{t}\succeq \mu I$, all eigenvalues of $H_{t}^{-1}$ are bounded by $1/\mu$, hence the claimed inequality follows from sub‑multiplicativity of the spectral norm. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness at iteration $t$ conditioned on $x_{t}$ unless stated otherwise.

\subsection*{Step 1 – Apply smoothness}
From Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_{t}-\alpha H_{t}^{-1}g_{t}$,
\begin{align}
f(x_{t+1})
&\le f(x_{t})+\inner{\grad f(x_{t})}{-\alpha H_{t}^{-1}g_{t}}
     +\frac{L}{2}\norm{\alpha H_{t}^{-1}g_{t}}^{2}\nonumber\\
&= f(x_{t})-\alpha\inner{\grad f(x_{t})}{H_{t}^{-1}g_{t}}
   +\frac{L\alpha^{2}}{2}\norm{H_{t}^{-1}g_{t}}^{2}. \tag{1}
\end{align}
\noindent\textbf{[Step 1]}

\subsection*{Step 2 – Decompose the inner product}
Insert $g_{t}= \grad f(x_{t}) + \delta_{t}$ where $\delta_{t}:=g_{t}-\grad f(x_{t})$ (zero‑mean noise). Then
\begin{align}
\inner{\grad f(x_{t})}{H_{t}^{-1}g_{t}}
&=\inner{\grad f(x_{t})}{H_{t}^{-1}\grad f(x_{t})}
  +\inner{\grad f(x_{t})}{H_{t}^{-1}\delta_{t}} .\tag{2}
\end{align}
\noindent\textbf{[Step 2]}

\subsection*{Step 3 – Lower bound the deterministic term}
Using $\mu I\preceq H_{t}$ we have $H_{t}^{-1}\preceq \frac{1}{\mu}I$, hence
\[
\inner{\grad f(x_{t})}{H_{t}^{-1}\grad f(x_{t})}
\;\ge\;\frac{1}{L}\norm{\grad f(x_{t})}^{2},
\qquad
\inner{\grad f(x_{t})}{H_{t}^{-1}\grad f(x_{t})}
\;\ge\;\frac{1}{\mu}\norm{\grad f(x_{t})}^{2}. \tag{3}
\]
We keep the tighter $\frac{1}{\mu}$ bound in what follows.
\noindent\textbf{[Step 3]}

\subsection*{Step 4 – Upper bound the stochastic term}
From Cauchy–Schwarz and Lemma~\ref{lem:stepbound},
\[
\big|\inner{\grad f(x_{t})}{H_{t}^{-1}\delta_{t}}\big|
\le \norm{\grad f(x_{t})}\,\norm{H_{t}^{-1}\delta_{t}}
\le \frac{1}{\mu}\norm{\grad f(x_{t})}\,\norm{\delta_{t}}. \tag{4}
\]
\noindent\textbf{[Step 4]}

\subsection*{Step 5 – Upper bound the quadratic term}
Again by Lemma~\ref{lem:stepbound},
\[
\norm{H_{t}^{-1}g_{t}}^{2}
\le\frac{1}{\mu^{2}}\norm{g_{t}}^{2}
\le\frac{2}{\mu^{2}}\big(\norm{\grad f(x_{t})}^{2}+\norm{\delta_{t}}^{2}\big). \tag{5}
\]
\noindent\textbf{[Step 5]}

\subsection*{Step 6 – Substitute (2)–(5) into (1)}
Combining (1)–(5) yields
\begin{align}
f(x_{t+1})
&\le f(x_{t})
   -\alpha\Big(\frac{1}{\mu}\norm{\grad f(x_{t})}^{2}
               +\inner{\grad f(x_{t})}{H_{t}^{-1}\delta_{t}}\Big)
   +\frac{L\alpha^{2}}{2}\cdot\frac{2}{\mu^{2}}
      \big(\norm{\grad f(x_{t})}^{2}+\norm{\delta_{t}}^{2}\big) \nonumber\\[4pt]
&= f(x_{t})
   -\alpha\frac{1}{\mu}\norm{\grad f(x_{t})}^{2}
   -\alpha\inner{\grad f(x_{t})}{H_{t}^{-1}\delta_{t}}
   +\frac{L\alpha^{2}}{\mu^{2}}\norm{\grad f(x_{t})}^{2}
   +\frac{L\alpha^{2}}{\mu^{2}}\norm{\delta_{t}}^{2}. \tag{6}
\end{align}
\noindent\textbf{[Step 6]}

\subsection*{Step 7 – Take conditional expectation}
Condition on $x_{t}$ and use $\E[\delta_{t}\mid x_{t}]=0$ (Assumption~\ref{ass:gradunb}):

\[
\E\!\big[f(x_{t+1})\mid x_{t}\big]
\le f(x_{t})
-\Big(\frac{\alpha}{\mu}-\frac{L\alpha^{2}}{\mu^{2}}\Big)\norm{\grad f(x_{t})}^{2}
+\frac{L\alpha^{2}}{\mu^{2}}\E\!\big[\norm{\delta_{t}}^{2}\mid x_{t}\big].
\tag{7}
\]

\noindent\textbf{[Step 7]}

\subsection*{Step 8 – Plug variance bound}
From Assumption~\ref{ass:gradunb}, $\E[\norm{\delta_{t}}^{2}\mid x_{t}]\le\sigma_{g}^{2}$. Hence

\[
\E\!\big[f(x_{t+1})\mid x_{t}\big]
\le f(x_{t})
-\Big(\frac{\alpha}{\mu}-\frac{L\alpha^{2}}{\mu^{2}}\Big)\norm{\grad f(x_{t})}^{2}
+\frac{L\alpha^{2}\sigma_{g}^{2}}{\mu^{2}}. \tag{8}
\]

\noindent\textbf{[Step 8]}

\subsection*{Step 9 – Choose $\alpha$ to keep the coefficient positive}
Require
\[
\frac{\alpha}{\mu}-\frac{L\alpha^{2}}{\mu^{2}}\ge\frac{\alpha}{2\mu}
\;\Longleftrightarrow\;
\alpha\le\frac{\mu}{L}. \tag{9}
\]
We will use the more restrictive bound stated in the theorem
$\alpha\le\frac{\mu}{L^{2}+ \mu\rho R}$, which certainly satisfies (9).

Thus (8) simplifies to
\[
\E\!\big[f(x_{t+1})\mid x_{t}\big]
\le f(x_{t})
-\frac{\alpha}{2\mu}\norm{\grad f(x_{t})}^{2}
+\frac{L\alpha^{2}\sigma_{g}^{2}}{\mu^{2}}. \tag{10}
\]

\noindent\textbf{[Step 9]}

\subsection*{Step 10 – Unroll the recursion}
Take total expectation and sum for $t=0,\dots,T-1$:

\[
\E[f(x_{T})]
\le f(x_{0})
-\frac{\alpha}{2\mu}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(x_{t})}^{2}\big]
+T\frac{L\alpha^{2}\sigma_{g}^{2}}{\mu^{2}}. \tag{11}
\]

Since $f(x_{T})\ge f^{*}$, we obtain

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(x_{t})}^{2}\big]
\le\frac{2\mu\big(f(x_{0})-f^{*}\big)}{\alpha T}
   +\frac{2L\alpha\sigma_{g}^{2}}{\mu}. \tag{12}
\]

\noindent\textbf{[Step 10]}

\subsection*{Step 11 – Incorporate Hessian variance}
So far we have ignored the randomness of $H_{t}$.  
Write $H_{t}= \hess f(x_{t})+\Delta_{t}$ with $\E[\Delta_{t}\mid x_{t}]=0$ and
$\E[\norm{\Delta_{t}}^{2}\mid x_{t}]\le\sigma_{H}^{2}$ (Assumption~\ref{ass:hessunb}).  
Repeating Steps 2–6 while keeping the extra term
$-\alpha\inner{\grad f(x_{t})}{\Delta_{t}^{-1}g_{t}}$ leads to an additive
contribution bounded by
\[
\frac{2\alpha^{2}L^{2}\sigma_{H}^{2}}{\mu^{3}} \tag{13}
\]
(derivation follows the same algebra as Steps 4–5, using $\norm{\Delta_{t}}\le\sigma_{H}$ and $\norm{H_{t}^{-1}}\le1/\mu$).  

Adding (13) to the right‑hand side of (12) yields exactly the bound stated in Theorem~\ref{thm:main}.

\noindent\textbf{[Step 11]}

\subsection*{Step 12 – Optimising $\alpha$}
Balancing the $1/(\alpha T)$ term with the $O(\alpha)$ variance terms gives the optimal
$\alpha= \Theta(T^{-1/2})$, which substituted into (12)–(13) produces the $O(T^{-1/2})$ rate.

\noindent\textbf{[Step 12]}

\section*{Rate Derivation (With Constants)}
Collecting the constants from Steps 10 and 11:

\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(x_{t})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{*}\big)}{\alpha\mu\,T}
+
\frac{2\alpha\sigma_{g}^{2}}{\mu}
+
\frac{2\alpha^{2}L^{2}\sigma_{H}^{2}}{\mu^{3}} } .
\]

Choosing $\alpha = \min\!\Big\{\frac{\mu}{L^{2}+\mu\rho R},
\sqrt{\frac{(f(x_{0})-f^{*})\mu}{L\sigma_{g}^{2}T}}\Big\}$ gives the explicit $O(T^{-1/2})$ bound.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Hessian, no gradient noise.} $\sigma_{g}=0$, $\sigma_{H}=0$. Equation \eqref{eq:update} reduces to deterministic Newton; with $\alpha=1$ we obtain
\[
f(x_{t+1})\le f(x_{t})-\frac{1}{2\mu}\norm{\grad f(x_{t})}^{2},
\]
which implies a linear convergence to a stationary point under the usual local strong convexity.
\item \textbf{Large batch for Hessian ( $\sigma_{H}\to0$ ).} The bound collapses to the SGD‑type rate with only the gradient‑variance term remaining.
\item \textbf{Quadratic objective.} For $f(x)=\frac12 x^{\top}Ax-b^{\top}x$, $\hess f\equiv A$, $L=\lambda_{\max}(A)$, $\mu=\lambda_{\min}(A)$. The stochastic Newton step becomes exact in expectation, and the recursion (10) holds with equality, yielding a deterministic linear convergence factor $1-\alpha\mu/L$.
\end{itemize}
