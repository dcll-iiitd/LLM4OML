\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable and possibly non‑convex.  
At iteration $t$ a stochastic gradient $g_{t}$ is sampled
\[
g_{t}= \nabla f(w_{t};\xi_{t}) ,
\]
where $\xi_{t}$ denotes the random data sample (or mini‑batch).  
Define the deterministic \emph{top‑K sparsifier}
\[
\topK{K}{v}\;:=\; \bigl[\,v_{i}\,\mathbf{1}\{i\in\mathcal{I}_{K}(v)\}\,\bigr]_{i=1}^{d},
\qquad 
\mathcal{I}_{K}(v)=\text{indices of the $K$ largest }|v_{i}|\text{ in }v .
\]
The communicated (sparse) gradient is
\[
\tilde g_{t}= \topK{K}{g_{t}} .
\]
The global model is updated by
\begin{equation}
\boxed{ \; w_{t+1}= w_{t}-\eta_{t}\,\tilde g_{t}\; } \label{eq:update}
\end{equation}
with stepsize $\eta_{t}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}$, sparsity level $K\le d$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample mini‑batch $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$.
    \STATE $\tilde g_{t}\leftarrow \topK{K}{g_{t}}$ (keep $K$ largest magnitude components, set others to $0$).
    \STATE $w_{t+1}\leftarrow w_{t}-\eta_{t}\tilde g_{t}$.
\ENDFOR
\STATE \textbf{Return} $w_{T}$ (or the averaged iterate $\bar w_{T}= \frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, $w_{0}=0$, constant stepsize $\eta=0.1$, and $K=1$ (the only coordinate is always kept).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\ \hline
0 & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3690$
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the mechanics of \eqref{eq:update}.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} $f$ has $L$‑Lipschitz gradients:
\[
\forall x,y\in\R^{d},\qquad 
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y}. \tag{A1}
\]
\item \textbf{Bounded variance of stochastic gradients.}
\[
\E\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2}. \tag{A2}
\]
\item \textbf{Bounded second moment.}
\[
\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]\le G^{2}. \tag{A3}
\]
\item \textbf{Sparsification error bound.} For any $v\in\R^{d}$,
\[
\norm{v-\topK{K}{v}}^{2}\le \frac{d-K}{K}\,\norm{v}^{2}. \tag{A4}
\]
(The bound follows from the fact that at most $d-K$ coordinates are discarded and each discarded
coordinate is no larger in magnitude than the smallest kept one.)
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of Top‑K SGD]\label{thm:main}
Assume (A1)–(A4) and choose a constant stepsize
\[
\eta_{t}\equiv\eta\le \frac{1}{L}.
\]
Define $f^{*}:=\inf_{w}f(w)$ and let $T\ge1$. Then
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{*}\bigr)}{\eta T}
\;+\;
L\eta\Bigl(1+\tfrac{d-K}{K}\Bigr) G^{2}
\;+\;
L\eta\sigma^{2}. \label{eq:rate}
\end{equation}
Consequently, with the diminishing stepsize $\eta=\frac{c}{\sqrt{T}}$ for a constant $c\le\frac{1}{L}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound \eqref{eq:rate} holds for any $K\in\{1,\dots ,d\}$; larger $K$ reduces the sparsification term $\frac{d-K}{K}$, yielding a tighter bound.
\item \textbf{Hyper‑parameter sensitivity.} The stepsize must satisfy $\eta\le 1/L$; otherwise the smoothness inequality used in the proof fails. The constant $c$ in the diminishing schedule controls the trade‑off between the $1/(\eta T)$ term and the $L\eta$ terms.
\item \textbf{Comparison to full‑gradient SGD.} Setting $K=d$ makes $\topK{K}{v}=v$ and the term $\frac{d-K}{K}=0$, recovering the classic non‑convex SGD rate $O(1/\sqrt{T})$.
\end{itemize}

\section*{Discussion}
The theorem shows that Top‑K sparsification introduces an additional additive error proportional to $\frac{d-K}{K}G^{2}$.
When $K$ grows linearly with $d$ (e.g. $K=\alpha d$, $0<\alpha<1$) the factor $\frac{d-K}{K}= \frac{1-\alpha}{\alpha}$ is a constant, and the asymptotic rate remains $O(1/\sqrt{T})$.
Thus, significant communication reduction can be achieved without deteriorating the convergence order, at the price of a larger constant.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies (A1), then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradients; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Sparsification error]\label{lem:sparse}
For any $v\in\R^{d}$ the top‑$K$ operator obeys (A4).
\end{lemma}
\begin{proof}
Let $\mathcal{I}$ be the index set of the $K$ kept entries. By definition,
$|v_{i}|\le |v_{j}|$ for all $i\notin\mathcal{I}$ and $j\in\mathcal{I}$. Hence
\[
\sum_{i\notin\mathcal{I}}v_{i}^{2}\le \frac{d-K}{K}\sum_{j\in\mathcal{I}}v_{j}^{2}
\le \frac{d-K}{K}\sum_{i=1}^{d}v_{i}^{2},
\]
which is exactly \eqref{A4}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof $ \mathcal{F}_{t}= \sigma(w_{0},\xi_{0},\dots ,\xi_{t-1})$ denotes the filtration up to iteration $t$.

\bigskip
\noindent\textbf{[Step 1] Smoothness applied to the update.}
Using Lemma \ref{lem:smooth} with $x=w_{t}$ and $y=w_{t+1}=w_{t}-\eta\tilde g_{t}$,
\begin{align}
f(w_{t+1}) &\le f(w_{t})-\eta\langle\nabla f(w_{t}),\tilde g_{t}\rangle
+\frac{L\eta^{2}}{2}\norm{\tilde g_{t}}^{2}. \label{eq:smooth}
\end{align}

\noindent\textbf{[Step 2] Decompose the sparse gradient.}
Write
\[
\tilde g_{t}=g_{t}+(\topK{K}{g_{t}}-g_{t})
= \nabla f(w_{t}) + \underbrace{(g_{t}-\nabla f(w_{t}))}_{\text{stochastic error}}
+ \underbrace{(\topK{K}{g_{t}}-g_{t})}_{\text{sparsification error}} .
\]
Define the total error term
\[
\varepsilon_{t}:=(g_{t}-\nabla f(w_{t})) + (\topK{K}{g_{t}}-g_{t}).
\]
Thus $\tilde g_{t}= \nabla f(w_{t})+\varepsilon_{t}$.

\noindent\textbf{[Step 3] Substitute into \eqref{eq:smooth}.}
\begin{align}
f(w_{t+1}) &\le f(w_{t})-\eta\langle\nabla f(w_{t}),\nabla f(w_{t})+\varepsilon_{t}\rangle
+\frac{L\eta^{2}}{2}\norm{\nabla f(w_{t})+\varepsilon_{t}}^{2}\nonumber\\
&= f(w_{t})-\eta\norm{\nabla f(w_{t})}^{2}
-\eta\langle\nabla f(w_{t}),\varepsilon_{t}\rangle
+\frac{L\eta^{2}}{2}\bigl(\norm{\nabla f(w_{t})}^{2}+2\langle\nabla f(w_{t}),\varepsilon_{t}\rangle
+\norm{\varepsilon_{t}}^{2}\bigr). \label{eq:expand}
\end{align}

\noindent\textbf{[Step 4] Rearrange terms.}
Collect the inner–product contributions:
\[
-\eta\langle\nabla f(w_{t}),\varepsilon_{t}\rangle
+\frac{L\eta^{2}}{2}\cdot 2\langle\nabla f(w_{t}),\varepsilon_{t}\rangle
=
\bigl(-\eta+L\eta^{2}\bigr)\langle\nabla f(w_{t}),\varepsilon_{t}\rangle .
\]
Hence
\begin{align}
f(w_{t+1}) &\le f(w_{t})
-\bigl(\eta-\tfrac{L\eta^{2}}{2}\bigr)\norm{\nabla f(w_{t})}^{2}
+\bigl(-\eta+L\eta^{2}\bigr)\langle\nabla f(w_{t}),\varepsilon_{t}\rangle
+\frac{L\eta^{2}}{2}\norm{\varepsilon_{t}}^{2}. \label{eq:rearr}
\end{align}

\noindent\textbf{[Step 5] Take conditional expectation.}
Condition on $\mathcal{F}_{t}$ and use $\E[\varepsilon_{t}\mid\mathcal{F}_{t}]
= \E[g_{t}-\nabla f(w_{t})\mid\mathcal{F}_{t}]
+ \E[\topK{K}{g_{t}}-g_{t}\mid\mathcal{F}_{t}]$.
The stochastic part is unbiased, i.e. $\E[g_{t}-\nabla f(w_{t})\mid\mathcal{F}_{t}]=0$.
The sparsification part is deterministic given $g_{t}$, thus
$\E[\topK{K}{g_{t}}-g_{t}\mid\mathcal{F}_{t}]
= \topK{K}{\E[g_{t}\mid\mathcal{F}_{t}]}-\E[g_{t}\mid\mathcal{F}_{t}]
= \topK{K}{\nabla f(w_{t})}-\nabla f(w_{t})$.
Consequently
\[
\E\!\bigl[\langle\nabla f(w_{t}),\varepsilon_{t}\rangle\mid\mathcal{F}_{t}\bigr]
= \langle\nabla f(w_{t}),\topK{K}{\nabla f(w_{t})}-\nabla f(w_{t})\rangle
\le 0,
\]
because $\topK{K}{\nabla f(w_{t})}$ keeps the largest‑magnitude components,
so the inner product with the discarded part is non‑positive.
Hence the second term in \eqref{eq:rearr} is \emph{non‑positive} and can be dropped.

\noindent\textbf{[Step 6] Bound the error norm.}
From the definition of $\varepsilon_{t}$ and the triangle inequality,
\[
\norm{\varepsilon_{t}}^{2}
\le 2\norm{g_{t}-\nabla f(w_{t})}^{2}
+2\norm{\topK{K}{g_{t}}-g_{t}}^{2}.
\]
Taking expectation and using (A2) and Lemma \ref{lem:sparse} gives
\begin{align}
\E\!\bigl[\norm{\varepsilon_{t}}^{2}\mid\mathcal{F}_{t}\bigr]
&\le 2\sigma^{2}+2\frac{d-K}{K}\,\E\!\bigl[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\bigr]\nonumber\\
&\le 2\sigma^{2}+2\frac{d-K}{K}\,G^{2}. \label{eq:errorbound}
\end{align}

\noindent\textbf{[Step 7] Combine the pieces.}
Taking total expectation in \eqref{eq:rearr} and applying \eqref{eq:errorbound} yields
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_{t})]
-\bigl(\eta-\tfrac{L\eta^{2}}{2}\bigr)\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
+\frac{L\eta^{2}}{2}\Bigl(2\sigma^{2}+2\frac{d-K}{K}G^{2}\Bigr) \nonumber\\
&= \E[f(w_{t})]
-\bigl(\eta-\tfrac{L\eta^{2}}{2}\bigr)\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
+L\eta^{2}\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr). \label{eq:recursion}
\end{align}
Because $\eta\le 1/L$, we have $\eta-\frac{L\eta^{2}}{2}\ge \frac{\eta}{2}$, and thus
\begin{equation}
\E[f(w_{t+1})]\le \E[f(w_{t})]
-\frac{\eta}{2}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
+L\eta^{2}\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr). \label{eq:recursion2}
\end{equation}

\noindent\textbf{[Step 8] Telescope the inequality.}
Summing \eqref{eq:recursion2} over $t=0,\dots,T-1$ gives
\[
\E[f(w_{T})]\le f(w_{0})
-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
+TL\eta^{2}\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr).
\]
Rearranging,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le
\frac{2\bigl(f(w_{0})-f^{*}\bigr)}{\eta T}
+2L\eta\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr).
\]
Since $f^{*}\le \E[f(w_{T})]$, we replace $f(w_{0})- \E[f(w_{T})]$ by $f(w_{0})-f^{*}$, obtaining exactly \eqref{eq:rate}.

\noindent\textbf{[Step 9] Choice of diminishing stepsize.}
Set $\eta=c/\sqrt{T}$ with $c\le 1/L$. Then
\[
\frac{2(f(w_{0})-f^{*})}{\eta T}=O\!\left(\frac{1}{\sqrt{T}}\right),\qquad
2L\eta\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr)=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
Thus the average squared gradient norm decays as $O(1/\sqrt{T})$, completing the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
For a concrete constant choice, let $\eta = \frac{1}{L\sqrt{T}}$.
Plugging into \eqref{eq:rate} yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le
\frac{2L\bigl(f(w_{0})-f^{*}\bigr)}{\sqrt{T}}
+\frac{1}{\sqrt{T}}\Bigl(\sigma^{2}+ \tfrac{d-K}{K}G^{2}\Bigr).
\]
All constants are explicit in terms of $L$, $G$, $\sigma$, $d$, $K$, and the initial optimality gap.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Full gradient (no sparsification).} $K=d$ gives $\frac{d-K}{K}=0$ and the bound reduces to the classical SGD result.
\item \textbf{Extreme sparsification.} $K=1$ yields $\frac{d-1}{1}=d-1$, inflating the constant by a factor $\approx d$. The rate order is unchanged but the bound may become loose for very high dimensional problems.
\item \textbf{Large‑scale regime $K=\alpha d$ with fixed $\alpha\in(0,1]$.} The factor $\frac{d-K}{K}= \frac{1-\alpha}{\alpha}$ is independent of $d$, so communication is reduced by a factor $1/\alpha$ while the convergence constant grows only by $\frac{1-\alpha}{\alpha}$.
\end{enumerate}
