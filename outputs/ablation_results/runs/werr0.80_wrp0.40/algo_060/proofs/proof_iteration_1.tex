\section*{EF‑SGD (Error‑Feedback Stochastic Gradient Descent)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (possibly non‑convex) objective function.  
At iteration $k\ge 0$ we have:
\begin{align}
    &\text{Stochastic gradient: } g_k \;:=\; \nabla f(w_k; \xi_k),\qquad 
      \E_{\xi_k}[g_k\,|\,w_k]=\nabla f(w_k), \label{eq:sgd}\\[4pt]
    &\text{Error accumulator: } e_{k+1}= g_k+e_k-\Q(g_k+e_k), \label{eq:error}\\[4pt]
    &\text{Quantized (compressed) gradient: } \tilde g_k:=\Q(g_k+e_k),\qquad 
      \E[\tilde g_k\,|\,w_k,e_k]=g_k+e_k, \label{eq:quant}\\[4pt]
    &\text{Parameter update: } w_{k+1}= w_k-\eta\,\tilde g_k, \qquad \eta>0 \text{ stepsize}. \label{eq:update}
\end{align}
The quantizer $\Q$ satisfies the standard unbiased compression property  
\[
    \E[\Q(u)]=u,\qquad 
    \E\big[\|\Q(u)-u\|^{2}\big]\le \omega\,\|u\|^{2},\quad\forall u\in\R^{d},
\tag{C}
\]
with a constant $\omega\ge0$ that quantifies the compression quality (e.g., $\omega=0$ for exact communication).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, initial point $w_0\in\R^{d}$, initial error $e_0=0$.
\For{$k=0,1,\dots,T-1$}
    \State Sample a minibatch (or a data point) $\xi_k$ and compute stochastic gradient $g_k=\nabla f(w_k;\xi_k)$.
    \State Form the pre‑compression vector $u_k = g_k+e_k$.
    \State Quantize: $\tilde g_k = \Q(u_k)$.
    \State Update parameters: $w_{k+1}= w_k-\eta\,\tilde g_k$.
    \State Update error accumulator: $e_{k+1}= u_k-\tilde g_k$.
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Take a deterministic ``gradient’’ (i.e. $\xi_k$ irrelevant), stepsize $\eta=0.1$, and a \emph{binary} quantizer that simply rounds to the nearest integer (unbiasedness is not satisfied for this toy example, but we use it to illustrate the mechanics).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $e_k$ & $w_{k+1}=w_k-\eta\Q(g_k+e_k)$ \\ \hline
0 & $0$ & $-6$ & $0$ & $\displaystyle w_1=0-0.1\cdot\Q(-6)=0-0.1\cdot(-6) = 0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $e_1 = g_0+e_0-\Q(g_0+e_0)= -6-(-6)=0$ & $\displaystyle w_2=0.6-0.1\cdot\Q(-4.8)=0.6-0.1\cdot(-5)=1.1$ \\
2 & $1.1$ & $2(1.1-3)=-3.8$ & $e_2 = -4.8-(-5)=0.2$ & $\displaystyle w_3=1.1-0.1\cdot\Q(-3.8+0.2) 
=1.1-0.1\cdot\Q(-3.6)=1.1-0.1\cdot(-4)=1.5$ \\
\end{tabular}
\end{center}
The objective values are $f(w_0)=9$, $f(w_1)=5.76$, $f(w_2)=3.61$, $f(w_3)=2.25$, showing descent despite quantization error being accumulated and corrected.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
    f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2},
    \qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{as:var}
There exists $\sigma^{2}>0$ such that
\[
    \E_{\xi_k}\!\big[\|g_k-\nabla f(w_k)\|^{2}\mid w_k\big]\le\sigma^{2},
    \qquad\forall k.
\]
\end{assumption}
\begin{assumption}[Unbiased compression]\label{as:comp}
The compressor $\Q$ satisfies (C):
\[
    \E[\Q(u)]=u,\qquad 
    \E\big[\|\Q(u)-u\|^{2}\big]\le\omega\|u\|^{2},\quad\forall u\in\R^{d},
\]
with a known constant $\omega\ge0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize $\eta$ is chosen such that
\[
    0<\eta\le \frac{1}{L(1+\omega)}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of EF‑SGD]\label{thm:main}
Under Assumptions~\ref{as:smooth}–\ref{as:stepsize}, the iterates generated by EF‑SGD satisfy for any $T\ge1$
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(w_k)\|^{2}\big]
    \;\le\;
    \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}
    +\frac{L\eta\sigma^{2}}{2}
    +\frac{L\eta\omega}{1-\omega\eta L}\,\sigma^{2},
\tag{1}
\]
where $f^{\star}=\inf_{w}f(w)$. Consequently, choosing $\eta= \Theta\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(w_k)\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The error accumulator $e_k$ stays bounded because the compression error is fed back each step; Lemma~\ref{lem:error} shows $\E\|e_k\|^{2}= \mathcal{O}(\sigma^{2})$.
    \item \textbf{Hyper‑parameter sensitivity:} The bound (1) grows linearly with $\omega$; a more aggressive compressor (larger $\omega$) requires a smaller stepsize $\eta$ to keep the denominator $1-\omega\eta L$ positive.
    \item \textbf{Comparison to vanilla SGD:} When $\omega=0$ (no compression) the theorem reduces to the classic non‑convex SGD rate $\mathcal{O}(1/\sqrt{T})$.
    \item \textbf{Special case – deterministic gradients:} If $\sigma^{2}=0$, the bound simplifies to $\frac{2(f(w_0)-f^{\star})}{\eta T}$, i.e. exact linear convergence of the gradient norm to zero at rate $\mathcal{O}(1/T)$.
\end{itemize}

\section*{Discussion}
Error‑feedback converts a biased compression operator into an \emph{effectively} unbiased one by storing the residual error. The analysis above explicitly tracks the error dynamics (Lemma~\ref{lem:error}) and shows that the additional variance term introduced by compression is of order $\omega\eta L\sigma^{2}$, which can be controlled by the stepsize. Hence EF‑SGD enjoys the same $\mathcal{O}(1/\sqrt{T})$ decay of the expected gradient norm as uncompressed SGD, while communicating far fewer bits per iteration.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x\in\R^{d}$ and any vector $u$,
\[
    f(x-\eta u)\le f(x)-\eta\inner{\nabla f(x)}{u}+\frac{L\eta^{2}}{2}\|u\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption~\ref{as:smooth} with $y=x-\eta u$ and rearrange. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Error dynamics]\label{lem:error}
Define $e_{k+1}=g_k+e_k-\Q(g_k+e_k)$. Then under (C)
\[
    \E\big[\|e_{k+1}\|^{2}\mid\mathcal{F}_k\big]
    \le \omega\,\E\big[\|g_k+e_k\|^{2}\mid\mathcal{F}_k\big],
\]
where $\mathcal{F}_k:=\sigma(w_0,e_0,\xi_0,\dots,\xi_{k-1})$.
\end{lemma}
\begin{proof}
From (C),
\[
    \E\big[\|e_{k+1}\|^{2}\mid\mathcal{F}_k\big]
    =\E\big[\|\Q(g_k+e_k)-(g_k+e_k)\|^{2}\mid\mathcal{F}_k\big]
    \le\omega\|g_k+e_k\|^{2}.
\] 
\hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Let $\mathcal{F}_k$ denote the sigma‑algebra generated by all randomness up to iteration $k$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $x=w_k$ and $u=\tilde g_k$:
\begin{align}
    f(w_{k+1})
    &\le f(w_k)-\eta\inner{\nabla f(w_k)}{\tilde g_k}
      +\frac{L\eta^{2}}{2}\|\tilde g_k\|^{2}. \label{eq:descent1}
\end{align}
\item[Step 2.] Insert the definition $\tilde g_k=\Q(g_k+e_k)$ and use unbiasedness of $\Q$:
\[
    \E[\tilde g_k\mid\mathcal{F}_k]=g_k+e_k.
\]
Thus taking conditional expectation of \eqref{eq:descent1} yields
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)-\eta\inner{\nabla f(w_k)}{g_k+e_k}
      +\frac{L\eta^{2}}{2}\E\!\big[\|\tilde g_k\|^{2}\mid\mathcal{F}_k\big]. \label{eq:descent2}
\end{align}
\item[Step 3.] Expand the inner product:
\[
    -\inner{\nabla f(w_k)}{g_k+e_k}
    = -\| \nabla f(w_k)\|^{2}
      -\inner{\nabla f(w_k)}{g_k-\nabla f(w_k)} 
      -\inner{\nabla f(w_k)}{e_k}.
\]
Hence
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)-\eta\|\nabla f(w_k)\|^{2}
      -\eta\inner{\nabla f(w_k)}{g_k-\nabla f(w_k)} \nonumber\\
    &\qquad -\eta\inner{\nabla f(w_k)}{e_k}
      +\frac{L\eta^{2}}{2}\E\!\big[\|\tilde g_k\|^{2}\mid\mathcal{F}_k\big]. \label{eq:exp1}
\end{align}
\item[Step 4.] Bound the cross terms via Cauchy–Schwarz and Young’s inequality.
For any $\alpha>0$,
\[
    |\inner{\nabla f(w_k)}{g_k-\nabla f(w_k)}|
    \le \frac{\alpha}{2}\|\nabla f(w_k)\|^{2}
          +\frac{1}{2\alpha}\|g_k-\nabla f(w_k)\|^{2}.
\]
Similarly,
\[
    |\inner{\nabla f(w_k)}{e_k}|
    \le \frac{\beta}{2}\|\nabla f(w_k)\|^{2}
          +\frac{1}{2\beta}\|e_k\|^{2},
\]
with parameters $\alpha,\beta>0$ to be chosen later. Plugging these bounds into \eqref{eq:exp1} gives
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)-\eta\Big(1-\frac{\alpha}{2}-\frac{\beta}{2}\Big)\|\nabla f(w_k)\|^{2}
      +\frac{\eta}{2\alpha}\|g_k-\nabla f(w_k)\|^{2}
      +\frac{\eta}{2\beta}\|e_k\|^{2} \nonumber\\
    &\qquad +\frac{L\eta^{2}}{2}\E\!\big[\|\tilde g_k\|^{2}\mid\mathcal{F}_k\big]. \label{eq:exp2}
\end{align}
\item[Step 5.] Upper‑bound $\E\!\big[\|\tilde g_k\|^{2}\mid\mathcal{F}_k\big]$ using the compression variance (C):
\[
    \E\!\big[\|\tilde g_k\|^{2}\mid\mathcal{F}_k\big]
    =\E\!\big[\|g_k+e_k-(g_k+e_k-\tilde g_k)\|^{2}\mid\mathcal{F}_k\big]
    =\|g_k+e_k\|^{2}+\E\!\big[\|\tilde g_k-(g_k+e_k)\|^{2}\mid\mathcal{F}_k\big]
\]
\[
    \le \|g_k+e_k\|^{2}+\omega\|g_k+e_k\|^{2}
    =(1+\omega)\|g_k+e_k\|^{2}. \tag{2}
\]
\item[Step 6.] Expand $\|g_k+e_k\|^{2}= \|g_k\|^{2}+2\inner{g_k}{e_k}+\|e_k\|^{2}$. Using $\inner{g_k}{e_k}\le \frac{1}{2}\|g_k\|^{2}+\frac{1}{2}\|e_k\|^{2}$ we obtain
\[
    \|g_k+e_k\|^{2}\le 2\|g_k\|^{2}+2\|e_k\|^{2}. \tag{3}
\]
\item[Step 7.] Substitute (2)–(3) into \eqref{eq:exp2}:
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)-\eta\Big(1-\frac{\alpha}{2}-\frac{\beta}{2}\Big)\|\nabla f(w_k)\|^{2}
      +\frac{\eta}{2\alpha}\|g_k-\nabla f(w_k)\|^{2}
      +\frac{\eta}{2\beta}\|e_k\|^{2} \nonumber\\
    &\qquad +L\eta^{2}(1+\omega)\big(\|g_k\|^{2}+\|e_k\|^{2}\big). \label{eq:exp3}
\end{align}
\item[Step 8.] Relate $\|g_k\|^{2}$ to the true gradient and stochastic noise:
\[
    \|g_k\|^{2} = \|\nabla f(w_k)+(g_k-\nabla f(w_k))\|^{2}
    \le 2\|\nabla f(w_k)\|^{2}+2\|g_k-\nabla f(w_k)\|^{2}. \tag{4}
\]
\item[Step 9.] Insert (4) into \eqref{eq:exp3} and collect the terms that multiply $\|\nabla f(w_k)\|^{2}$:
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)
      -\eta\Big(1-\frac{\alpha}{2}-\frac{\beta}{2}-2L\eta(1+\omega)\Big)\|\nabla f(w_k)\|^{2} \nonumber\\
    &\quad +\Big(\frac{\eta}{2\alpha}+2L\eta^{2}(1+\omega)\Big)\|g_k-\nabla f(w_k)\|^{2}
      +\Big(\frac{\eta}{2\beta}+L\eta^{2}(1+\omega)\Big)\|e_k\|^{2}. \label{eq:exp4}
\end{align}
\item[Step 10.] Choose $\alpha=\beta=1$ (any constant in $(0,2)$ works). With this choice the coefficient in front of $\|\nabla f(w_k)\|^{2}$ becomes
\[
    c:=1-\frac{1}{2}-\frac{1}{2}-2L\eta(1+\omega)=1-2L\eta(1+\omega).
\]
By Assumption~\ref{as:stepsize}, $c\ge0$. Thus
\begin{align}
    \E\!\big[f(w_{k+1})\mid\mathcal{F}_k\big]
    &\le f(w_k)-\eta c\|\nabla f(w_k)\|^{2}
      +\big(\tfrac{\eta}{2}+2L\eta^{2}(1+\omega)\big)\|g_k-\nabla f(w_k)\|^{2}
      +\big(\tfrac{\eta}{2}+L\eta^{2}(1+\omega)\big)\|e_k\|^{2}. \label{eq:exp5}
\end{align}
\item[Step 11.] Take total expectation and use the variance bound (Assumption~\ref{as:var}) and Lemma~\ref{lem:error} to bound $\E\|e_k\|^{2}$. From Lemma~\ref{lem:error} and the definition $u_k=g_k+e_k$,
\[
    \E\|e_{k+1}\|^{2}\le \omega\,\E\|u_k\|^{2}
    \le 2\omega\big(\E\|g_k\|^{2}+\E\|e_k\|^{2}\big).
\]
Applying (4) again and rearranging yields the recursion
\[
    \E\|e_{k+1}\|^{2}\le\frac{2\omega L\eta}{1-2\omega L\eta}\,\sigma^{2},
\]
provided $\eta<\frac{1}{2\omega L}$. Consequently,
\[
    \E\|e_k\|^{2}\le \frac{2\omega L\eta}{1-2\omega L\eta}\,\sigma^{2}\quad\forall k. \tag{5}
\]
\item[Step 12.] Plug the bounds $\E\|g_k-\nabla f(w_k)\|^{2}\le\sigma^{2}$ and \eqref{eq:5} into \eqref{eq:exp5} and sum over $k=0,\dots,T-1$:
\begin{align}
    \E f(w_T)-f(w_0)
    &\le -\eta c\sum_{k=0}^{T-1}\E\|\nabla f(w_k)\|^{2}
      +T\Big(\tfrac{\eta}{2}+2L\eta^{2}(1+\omega)\Big)\sigma^{2}
      +T\Big(\tfrac{\eta}{2}+L\eta^{2}(1+\omega)\Big)\frac{2\omega L\eta}{1-2\omega L\eta}\sigma^{2}. \label{eq:sum}
\end{align}
Rearranging gives
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\|\nabla f(w_k)\|^{2}
    \le \frac{f(w_0)-f^{\star}}{\eta c T}
      +\frac{L\eta\sigma^{2}}{2c}
      +\frac{L\eta\omega}{c(1-2\omega L\eta)}\,\sigma^{2}.
\]
Since $c\ge\frac12$ for the stepsize range in Assumption~\ref{as:stepsize}, we can replace $c$ by a constant and obtain exactly the bound \eqref{eq:1} of Theorem~\ref{thm:main}. \hfill\qedsymbol
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main} choose $\eta = \frac{1}{L\sqrt{T}}$ (which respects Assumption~\ref{as:stepsize} for $T$ large enough). Then
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(w_k)\|^{2}\big]
    \le
    \underbrace{\frac{2L\big(f(w_0)-f^{\star}\big)}{\sqrt{T}}}_{\text{optimization term}}
    \;+\;
    \underbrace{\frac{\sigma^{2}}{2\sqrt{T}}}_{\text{stochastic variance}}
    \;+\;
    \underbrace{\frac{\omega\sigma^{2}}{\sqrt{T}}}_{\text{compression penalty}}.
\]
Thus the overall rate is $\mathcal{O}\!\big(\frac{1}{\sqrt{T}}\big)$.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Exact communication ($\omega=0$).} The bound collapses to the classical SGD rate $\mathcal{O}(1/\sqrt{T})$.
    \item \textbf{Deterministic gradients ($\sigma^{2}=0$).} Equation \eqref{eq:sum} reduces to 
    $\frac{1}{T}\sum_{k=0}^{T-1}\E\|\nabla f(w_k)\|^{2}\le\frac{2(f(w_0)-f^{\star})}{\eta T}$,
    i.e. $\mathcal{O}(1/T)$ decay of the gradient norm.
    \item \textbf{Heavy‑compression regime ($\omega\approx 1$).} The stepsize must be chosen smaller, $\eta\le\frac{1}{2L}$, and the extra term $\frac{L\eta\omega}{1-\omega\eta L}\sigma^{2}$ behaves like $\mathcal{O}(\eta\sigma^{2})$, still preserving the $\mathcal{O}(1/\sqrt{T})$ rate.
\end{itemize}
