\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^d\to\R$ be a (possibly non‑convex) loss function.  
At iteration $t$ we draw a random mini‑batch $\xi_t$ and compute a stochastic gradient
\[
g_t \;:=\; \grad f(w_t;\xi_t)\in\R^d .
\]
A (random) \emph{quantizer} $\Q:\R^d\to\R^d$ is applied to $g_t$ before communication:
\[
\tilde g_t \;:=\; \Q(g_t).
\]
The iterate is updated by
\begin{equation}
\boxed{%
w_{t+1}=w_t-\eta_t \,\tilde g_t
}
\tag{1}
\end{equation}
where $\eta_t>0$ is the learning rate.  
The quantizer satisfies the following unbiasedness and variance properties (see Assumptions).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\R^d$, learning‑rate schedule $\{\eta_t\}_{t\ge0}$, quantizer $\Q(\cdot)$
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample mini‑batch $\xi_t$ and compute stochastic gradient $g_t=\grad f(w_t;\xi_t)$
    \STATE Quantize: $\tilde g_t =\Q(g_t)$
    \STATE Update: $w_{t+1}=w_t-\eta_t \tilde g_t$
\ENDFOR
\RETURN $w_T$ (or the average $\bar w_T=\tfrac1T\sum_{t=0}^{T-1}w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple scalar quadratic
\[
f(w)=(w-3)^2,\qquad \grad f(w)=2(w-3).
\]
We take a deterministic “mini‑batch’’ (so $g_t=\grad f(w_t)$) and a *deterministic* quantizer that simply returns the exact value (i.e. $\Q(g)=g$).  
Set $w_0=0$ and a constant stepsize $\eta=0.1$.

\begin{tabular}{c|c|c|c}

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\forall x,y\in\R^d,\qquad
f(y)\le f(x)+\ip{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\tag{A1}
\]
\item \textbf{Lower boundedness.} $f^{\star}:=\inf_{w}f(w)>-\infty$.
\tag{A2}
\item \textbf{Stochastic gradient.} For every $t$,
\[
\E_{\xi_t}[g_t\mid w_t]=\grad f(w_t),\qquad
\E_{\xi_t}\!\big[\norm{g_t-\grad f(w_t)}^2\mid w_t\big]\le\sigma^2 .
\tag{A3}
\]
\item \textbf{Unbiased quantizer with bounded variance.} There exists $\omega\ge0$ such that for any $u\in\R^d$,
\[
\E_{\Q}\!\big[\Q(u)\mid u\big]=u,\qquad
\E_{\Q}\!\big[\norm{\Q(u)}^2\mid u\big]\le (1+\omega)\norm{u}^2 .
\tag{A4}
\]
\item \textbf{Stepsize.} The learning rate is constant $\eta_t\equiv\eta$ satisfying
\[
0<\eta\le\frac{1}{L(1+\omega)} .
\tag{A5}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of QSGD]\label{thm:main}
Under Assumptions A1–A5, for any $T\ge1$ the iterates generated by \eqref{1} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\,(1+\omega)\,\sigma^2}_{\text{variance term}} .
\tag{2}
\]
Consequently,
\[
\min_{0\le t<T}\E\!\big[\norm{\grad f(w_t)}^2\big]\;=\;O\!\Big(\frac{1}{\sqrt{T}}\Big)
\quad\text{when }\eta\asymp T^{-1/2}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound \eqref{2} holds for any $\eta$ respecting (A5); larger $\omega$ (more aggressive quantization) inflates the variance term by the factor $1+\omega$.
\item \textbf{Hyper‑parameter sensitivity.} The optimal constant stepsize is
\[
\eta^{\star}= \sqrt{\frac{2\big(f(w_0)-f^{\star}\big)}{L(1+\omega)\,\sigma^2\,T}} ,
\]
which satisfies (A5) provided $T\ge \frac{2L(1+\omega)\big(f(w_0)-f^{\star}\big)}{\sigma^2}$.
\item \textbf{Comparison.} Setting $\omega=0$ recovers the standard non‑convex SGD bound; thus quantization incurs a multiplicative factor $1+\omega$ in the variance term.
\item \textbf{Special case – deterministic gradients.} If $\sigma=0$, the bound reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T},
\]
the same rate as exact gradient descent, because the unbiased quantizer does not add extra variance.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that unbiased quantization does not destroy the classical $O(1/\sqrt{T})$ convergence guarantee for smooth non‑convex optimization; the price paid is a factor $1+\omega$ in the variance term. This matches intuition: quantization behaves like an additional source of noise. The result is tight up to constants and aligns with existing analyses of QSGD \cite{Alistarh2017}.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^d$,
\[
f(y)\le f(x)+\ip{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz gradient condition. \qedhere
\end{proof}

\begin{lemma}[Bound on quantized second moment]\label{lem:quant}
For any random vector $u$,
\[
\E\!\big[\norm{\Q(u)}^2\mid u\big]\le (1+\omega)\norm{u}^2 .
\]
\end{lemma}
\begin{proof}
Directly from Assumption A4. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness of the mini‑batch $\xi_t$ and the quantizer $\Q$ unless otherwise noted.

\subsection*{Step 1 – Smoothness applied to the update}
From Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta\tilde g_t$,
\begin{align}
f(w_{t+1})
&\le f(w_t)+\ip{\grad f(w_t)}{-\eta\tilde g_t}
   +\frac{L}{2}\norm{-\eta\tilde g_t}^2 \nonumber\\
&= f(w_t)-\eta\ip{\grad f(w_t)}{\tilde g_t}
   +\frac{L\eta^2}{2}\norm{\tilde g_t}^2 .
\tag{3}
\end{align}
\textbf{[Step 1]}

\subsection*{Step 2 – Take conditional expectation given $w_t$}
Condition on $w_t$ and use unbiasedness of the stochastic gradient (A3) and of the quantizer (A4):
\begin{align}
\E\!\big[f(w_{t+1})\mid w_t\big]
&\le f(w_t)-\eta\,\E\!\big[\ip{\grad f(w_t)}{\tilde g_t}\mid w_t\big]
   +\frac{L\eta^2}{2}\E\!\big[\norm{\tilde g_t}^2\mid w_t\big] .
\tag{4}
\end{align}
\textbf{[Step 2]}

\subsection*{Step 3 – Expand the inner product}
Because $\tilde g_t=\Q(g_t)$ and $\E[\Q(g_t)\mid g_t]=g_t$, we have
\[
\E\!\big[\ip{\grad f(w_t)}{\tilde g_t}\mid w_t\big]
 =\ip{\grad f(w_t)}{\E[g_t\mid w_t]}
 =\norm{\grad f(w_t)}^2 .
\tag{5}
\]
\textbf{[Step 3]}

\subsection*{Step 4 – Bound the second‑moment term}
Write $\tilde g_t = g_t + (\tilde g_t-g_t)$. Using Lemma~\ref{lem:quant},
\begin{align}
\E\!\big[\norm{\tilde g_t}^2\mid w_t\big]
&= \E\!\big[\norm{g_t}^2\mid w_t\big]
   +\E\!\big[\norm{\tilde g_t-g_t}^2\mid w_t\big] \nonumber\\
&\le \E\!\big[\norm{g_t}^2\mid w_t\big]
   +\omega\,\E\!\big[\norm{g_t}^2\mid w_t\big]  \quad\text{(by A4)}\nonumber\\
&= (1+\omega)\,\E\!\big[\norm{g_t}^2\mid w_t\big] .
\tag{6}
\end{align}
\textbf{[Step 4]}

\subsection*{Step 5 – Decompose $\E[\norm{g_t}^2\mid w_t]$}
Using the variance bound (A3):
\[
\E\!\big[\norm{g_t}^2\mid w_t\big]
= \norm{\grad f(w_t)}^2 + \E\!\big[\norm{g_t-\grad f(w_t)}^2\mid w_t\big]
\le \norm{\grad f(w_t)}^2 + \sigma^2 .
\tag{7}
\]
\textbf{[Step 5]}

\subsection*{Step 6 – Plug (5)–(7) into (4)}
Combining (4)–(7) yields
\begin{align}
\E\!\big[f(w_{t+1})\mid w_t\big]
&\le f(w_t)-\eta\norm{\grad f(w_t)}^2
   +\frac{L\eta^2}{2}(1+\omega)\big(\norm{\grad f(w_t)}^2+\sigma^2\big) .
\tag{8}
\end{align}
\textbf{[Step 6]}

\subsection*{Step 7 – Rearrange terms}
Move the gradient term to the left‑hand side:
\begin{align}
\Big(\eta-\frac{L\eta^2}{2}(1+\omega)\Big)\norm{\grad f(w_t)}^2
&\le f(w_t)-\E\!\big[f(w_{t+1})\mid w_t\big]
   +\frac{L\eta^2}{2}(1+\omega)\sigma^2 .
\tag{9}
\end{align}
\textbf{[Step 7]}

\subsection*{Step 8 – Use the stepsize condition (A5)}
Since $0<\eta\le\frac{1}{L(1+\omega)}$, we have
\[
\eta-\frac{L\eta^2}{2}(1+\omega)
\;\ge\;\frac{\eta}{2}.
\tag{10}
\]
\textbf{[Step 8]}

\subsection*{Step 9 – Lower bound the left side of (9)}
Applying (10) to (9):
\[
\frac{\eta}{2}\norm{\grad f(w_t)}^2
\le f(w_t)-\E\!\big[f(w_{t+1})\mid w_t\big]
   +\frac{L\eta^2}{2}(1+\omega)\sigma^2 .
\tag{11}
\]
\textbf{[Step 9]}

\subsection*{Step 10 – Take full expectation and sum over $t$}
Take expectation over all randomness and sum from $t=0$ to $T-1$:
\begin{align}
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
&\le \E\!\big[f(w_0)\big]-\E\!\big[f(w_T)\big]
   +\frac{L\eta^2}{2}(1+\omega)\sigma^2 T .
\tag{12}
\end{align}
Since $f(w_T)\ge f^{\star}$, we replace it by $f^{\star}$:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le f(w_0)-f^{\star}
   +\frac{L\eta^2}{2}(1+\omega)\sigma^2 T .
\tag{13}
\]
\textbf{[Step 10]}

\subsection*{Step 11 – Divide by $\eta T$}
Dividing (13) by $\eta T$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}
   +L\eta\,(1+\omega)\sigma^2 .
\tag{14}
\]
\textbf{[Step 11]}

\subsection*{Step 12 – Remarks on the variance term}
% CORRECTED: The previous version replaced $(1+\omega)\sigma^2$ by $\sigma^2+\omega G^2$,
% which required an additional (unavailable) bounded‑gradient assumption.
% Here we retain the term $L\eta(1+\omega)\sigma^2$, which follows directly from
% the assumptions A1–A5.  If one wishes to express the bound in terms of a
% uniform bound $G^2\ge\sup_t\E[\|g_t\|^2]$, the inequality
% $L\eta(1+\omega)\sigma^2\le L\eta\big(\sigma^2+\omega G^2\big)$ holds, but it
% yields a *looser* bound and therefore is unnecessary for the main result.
\textbf{[Step 12]}

\subsection*{Step 13 – Optimizing the stepsize}
The right‑hand side of \eqref{14} is
\[
\Phi(\eta)=\frac{2\Delta}{\eta T}+L\eta(1+\omega)\sigma^2,
\qquad\Delta:=f(w_0)-f^{\star}.
\]
Minimising $\Phi$ over $\eta>0$ gives
\[
\eta^{\star}= \sqrt{\frac{2\Delta}{L(1+\omega)\sigma^2\,T}} .
\tag{15}
\]
% CORRECTED: we now provide an explicit condition guaranteeing $\eta^{\star}\le 1/[L(1+\omega)]$.
From (A5) we need $\eta^{\star}\le\frac{1}{L(1+\omega)}$, which is equivalent to
\[
T\;\ge\;\frac{2L(1+\omega)\Delta}{\sigma^2}.
\tag{16}
\]
Thus, for any $T$ satisfying \eqref{16}, the choice \eqref{15} is admissible and yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\;\le\;2\sqrt{\frac{L(1+\omega)\Delta\,\sigma^2}{T}}
\;=\;O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\textbf{[Step 13]}

Thus every step of the theorem’s statement has been derived from the assumptions, completing the proof. \qed

\section*{Rate Derivation (With Constants)}
Collecting the constants appearing in \eqref{2}:
\[
C_1:=2\big(f(w_0)-f^{\star}\big),\qquad
C_2:=L(1+\omega)\,\sigma^2 .
\]
The bound can be written compactly as
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le \frac{C_1}{\eta T}+C_2\eta .
\]
Minimising the right‑hand side over $\eta\in(0,1/(L(1+\omega))]$ yields the optimal $\eta^{\star}= \sqrt{C_1/(C_2 T)}$, producing the $O(T^{-1/2})$ rate with explicit constant $\sqrt{C_1 C_2}$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact gradients ($\sigma=0$).} The bound reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T},
\]
which is the classic deterministic gradient‑descent rate.
\item \textbf{No quantization ($\omega=0$).} The theorem collapses to the classic non‑convex SGD guarantee
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\grad f(w_t)}^2\big]
\le \frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}+L\eta\sigma^2 .
\]
\item \textbf{High‑precision quantization.} If $\omega = \Theta(2^{-b})$ for $b$ bits, the additional variance factor $1+\omega$ approaches $1$ exponentially fast, showing that a modest number of bits suffices to keep the convergence degradation negligible.
\end{enumerate}
