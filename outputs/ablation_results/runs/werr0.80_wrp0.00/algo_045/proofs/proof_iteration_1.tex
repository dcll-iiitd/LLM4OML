\section*{Algorithm Name}
\textbf{Differentially Private Stochastic Gradient Descent (DP‑SGD) with Gaussian Noise}  

The algorithm performs standard SGD updates on a (possibly non‑convex) smooth loss
and adds i.i.d. Gaussian noise to each stochastic gradient in order to obtain
$(\varepsilon,\delta)$‑differential privacy.

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (population) loss, and let $\mathcal{B}_{t}\subseteq\{1,\dots,n\}$ be a minibatch
sampled uniformly (with or without replacement) at iteration $t$.
Define the (clipped) stochastic gradient
\[
g_{t}
\;=\;
\frac{1}{|\mathcal{B}_{t}|}\sum_{i\in\mathcal{B}_{t}}
\operatorname{clip}_{C}\!\big(\nabla f_{i}(x_{t})\big)
\;+\;
\xi_{t},
\qquad 
\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d}),
\]
where $\operatorname{clip}_{C}(v)=v\cdot\min\!\big\{1,\;C/\norm{v}\big\}$ bounds the $\ell_{2}$‑norm of each per‑example
gradient by a constant $C>0$.

The parameter update is
\[
\boxed{%
x_{t+1}=x_{t}-\eta_{t}\,g_{t}}, \qquad t=0,1,\dots,T-1,
\]
with learning‑rate schedule $\{\eta_{t}\}_{t\ge0}$.

Privacy accounting (e.g. moments accountant) is performed on the sequence
$\{\mathcal{B}_{t}\}$ and the Gaussian noise scale $\sigma$, but it does not affect the
convergence analysis below.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, learning rates $\{\eta_{t}\}_{t=0}^{T-1}$,
        clipping norm $C$, Gaussian noise scale $\sigma>0$, batch size $b$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample a minibatch $\mathcal{B}_{t}$ of size $b$ uniformly from $\{1,\dots,n\}$.
    \STATE Compute per‑example gradients $\nabla f_{i}(x_{t})$, $i\in\mathcal{B}_{t}$.
    \STATE Clip each gradient:
          $\tilde g_{i}= \operatorname{clip}_{C}\!\big(\nabla f_{i}(x_{t})\big)$.
    \STATE Form the averaged clipped gradient:
          $\bar g_{t}= \frac{1}{b}\sum_{i\in\mathcal{B}_{t}}\tilde g_{i}$.
    \STATE Sample Gaussian noise $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$.
    \STATE Set noisy gradient $g_{t}= \bar g_{t}+ \xi_{t}$.
    \STATE Update parameters: $x_{t+1}=x_{t}-\eta_{t}g_{t}$.
\ENDFOR
\RETURN $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
No minibatching or clipping is needed; we set $C=\infty$.
Take $w_{0}=0$, constant step size $\eta=0.1$, and Gaussian noise variance
$\sigma^{2}=0.01$.  We illustrate three iterations (the random noise values are
illustrative).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $\xi_{t}\sim\mathcal{N}(0,0.01)$ & $w_{t+1}=w_{t}-\eta(\nabla f(w_{t})+\xi_{t})$\\\hline
0 & $2(0-3)=-6$ & $0.05$ & $w_{1}=0-0.1(-6+0.05)=0.595$\\
1 & $2(0.595-3)=-4.81$ & $-0.02$ & $w_{2}=0.595-0.1(-4.81-0.02)=1.077$\\
2 & $2(1.077-3)=-3.846$ & $0.03$ & $w_{3}=1.077-0.1(-3.846+0.03)=1.460$\\
\end{tabular}
\end{center}

The corresponding objective values are
$f(w_{0})=9$, $f(w_{1})\approx5.46$, $f(w_{2})\approx3.80$, $f(w_{3})\approx2.92$,
showing descent despite the injected noise.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
The loss $f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\qquad
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Variance]\label{ass:variance}
For the (clipped) stochastic gradient $\bar g_{t}$ we have
\[
\E\!\big[\bar g_{t}\mid x_{t}\big]=\nabla f(x_{t}),\qquad
\E\!\big[\norm{\bar g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\big]\le \tau^{2},
\]
for some constant $\tau\ge0$ (the variance coming from minibatching).
\end{assumption}
\begin{assumption}[Gaussian Noise]\label{ass:gauss}
The added noise $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$ is independent of all past
randomness and $\E[\xi_{t}]=0$, $\E\!\big[\norm{\xi_{t}}^{2}\big]=d\sigma^{2}$.
\end{assumption}
\begin{assumption}[Learning Rate]\label{ass:lr}
The step sizes satisfy $0<\eta_{t}\le \frac{1}{L}$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:lr}, let $T\ge1$ and define the
average step size $\bar\eta=\frac{1}{T}\sum_{t=0}^{T-1}\eta_{t}$.
Then the iterates produced by DP‑SGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\;\le\;
\frac{2\big(f(x_{0})-f^{\star}\big)}{\bar\eta\,T}
\;+\;
L\Big(\tau^{2}+d\sigma^{2}\Big)\,\bar\eta,
\]
where $f^{\star}=\inf_{x}f(x)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound is linear in the total noise variance
$d\sigma^{2}$; thus a larger privacy budget (larger $\sigma$) degrades the
steady‑state term $L(d\sigma^{2})\bar\eta$ but does not affect the
$O(1/T)$ term.
\item \textbf{Hyper‑parameter Sensitivity.}
Choosing a constant step size $\eta_{t}\equiv\eta\le 1/L$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
+L\big(\tau^{2}+d\sigma^{2}\big)\eta.
\]
The optimal $\eta$ balancing the two terms is
$\eta^{\star}= \sqrt{\frac{2\big(f(x_{0})-f^{\star}\big)}{L\big(\tau^{2}+d\sigma^{2}\big)T}}$,
yielding a convergence rate of $O\!\big(T^{-1/2}\big)$.
\item \textbf{Comparison to Non‑Private SGD.}
When $\sigma=0$ the bound reduces to the classic non‑convex SGD guarantee
with variance $\tau^{2}$.
\item \textbf{Special Cases.}
\begin{enumerate}
\item If $f$ is convex, the same proof yields an $O(1/T)$ bound on sub‑optimality.
\item For a decreasing schedule $\eta_{t}=c/\sqrt{t+1}$, the bound simplifies to
$O\!\big(\log T/\sqrt{T}\big)$.
\end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that DP‑SGD retains the standard non‑convex SGD convergence
behaviour up to an additive term proportional to the privacy‑induced noise
variance. By selecting the clipping norm $C$ and the batch size appropriately,
the total variance $\tau^{2}+d\sigma^{2}$ can be kept modest, preserving
practical performance while guaranteeing $(\varepsilon,\delta)$‑DP.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Unbiasedness and Variance of Noisy Gradient]\label{lem:unbias}
Define $g_{t}= \bar g_{t}+\xi_{t}$ as in the algorithm.
Then
\[
\E[g_{t}\mid x_{t}]=\nabla f(x_{t}),\qquad
\E\!\big[\norm{g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\big]
\le \tau^{2}+d\sigma^{2}.
\]
\end{lemma}
\begin{proof}
By Assumption~\ref{ass:variance},
$\E[\bar g_{t}\mid x_{t}]=\nabla f(x_{t})$ and
$\E[\norm{\bar g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}]\le\tau^{2}$.
Since $\xi_{t}$ is independent with zero mean and variance $d\sigma^{2}$,
\[
\E[g_{t}\mid x_{t}]
= \E[\bar g_{t}\mid x_{t}]+\E[\xi_{t}]
= \nabla f(x_{t}),
\]
and using independence,
\[
\begin{aligned}
\E\!\big[\norm{g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\big]
&= \E\!\big[\norm{(\bar g_{t}-\nabla f(x_{t}))+\xi_{t}}^{2}\mid x_{t}\big]\\
&= \E\!\big[\norm{\bar g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\big]
   +\E\!\big[\norm{\xi_{t}}^{2}\big] \\
&\le \tau^{2}+d\sigma^{2}.
\end{aligned}
\]
\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken over the randomness
up to iteration $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.]  Apply Lemma~\ref{lem:descent} with $x=x_{t}$ and $y=x_{t+1}=x_{t}-\eta_{t}g_{t}$:
\[
f(x_{t+1})\le f(x_{t})-\eta_{t}\inner{\nabla f(x_{t})}{g_{t}}
               +\frac{L}{2}\eta_{t}^{2}\norm{g_{t}}^{2}.
\tag{1}
\]

\item[Step 2.]  Expand the inner product using $g_{t}= \nabla f(x_{t})+(g_{t}-\nabla f(x_{t}))$:
\[
\inner{\nabla f(x_{t})}{g_{t}}
= \norm{\nabla f(x_{t})}^{2}
  +\inner{\nabla f(x_{t})}{g_{t}-\nabla f(x_{t})}.
\tag{2}
\]

\item[Step 3.]  Substitute (2) into (1):
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})-\eta_{t}\norm{\nabla f(x_{t})}^{2}
   -\eta_{t}\inner{\nabla f(x_{t})}{g_{t}-\nabla f(x_{t})}
   +\frac{L}{2}\eta_{t}^{2}\norm{g_{t}}^{2}.
\end{aligned}
\tag{3}
\]

\item[Step 4.]  Take conditional expectation $\E[\cdot\mid x_{t}]$ on both sides.
Using Lemma~\ref{lem:unbias}, $\E[g_{t}\mid x_{t}]=\nabla f(x_{t})$, hence
\[
\E\!\big[\inner{\nabla f(x_{t})}{g_{t}-\nabla f(x_{t})}\mid x_{t}\big]=0.
\tag{4}
\]

\item[Step 5.]  For the last term, write
\[
\norm{g_{t}}^{2}
= \norm{\nabla f(x_{t})}^{2}
  +2\inner{\nabla f(x_{t})}{g_{t}-\nabla f(x_{t})}
  +\norm{g_{t}-\nabla f(x_{t})}^{2}.
\tag{5}
\]

\item[Step 6.]  Plug (5) into the expectation of the last term in (3) and use (4):
\[
\begin{aligned}
\E\!\big[\norm{g_{t}}^{2}\mid x_{t}\big]
&= \norm{\nabla f(x_{t})}^{2}
   +\E\!\big[\norm{g_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\big] \\
&\le \norm{\nabla f(x_{t})}^{2}+ \tau^{2}+d\sigma^{2}.
\end{aligned}
\tag{6}
\]

\item[Step 7.]  Take expectation of (3) using (4) and (6):
\[
\begin{aligned}
\E\!\big[f(x_{t+1})\mid x_{t}\big]
&\le f(x_{t})
   -\eta_{t}\norm{\nabla f(x_{t})}^{2}
   +\frac{L}{2}\eta_{t}^{2}
     \Big(\norm{\nabla f(x_{t})}^{2}+ \tau^{2}+d\sigma^{2}\Big).
\end{aligned}
\tag{7}
\]

\item[Step 8.]  Rearrange (7) to isolate $\norm{\nabla f(x_{t})}^{2}$:
\[
\begin{aligned}
\Big(\eta_{t}-\frac{L}{2}\eta_{t}^{2}\Big)\norm{\nabla f(x_{t})}^{2}
\le f(x_{t})-\E\!\big[f(x_{t+1})\mid x_{t}\big]
      +\frac{L}{2}\eta_{t}^{2}\big(\tau^{2}+d\sigma^{2}\big).
\end{aligned}
\tag{8}
\]

\item[Step 9.]  By Assumption~\ref{ass:lr}, $\eta_{t}\le 1/L$, which implies
$\eta_{t}-\frac{L}{2}\eta_{t}^{2}\ge \frac{\eta_{t}}{2}$.  Hence from (8):
\[
\frac{\eta_{t}}{2}\norm{\nabla f(x_{t})}^{2}
\le f(x_{t})-\E\!\big[f(x_{t+1})\mid x_{t}\big]
   +\frac{L}{2}\eta_{t}^{2}\big(\tau^{2}+d\sigma^{2}\big).
\tag{9}
\]

\item[Step 10.]  Take total expectation over all randomness up to iteration $t$ and
sum (9) for $t=0,\dots,T-1$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_{t}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\le f(x_{0})- \E\!\big[f(x_{T})\big]
   +\frac{L}{2}\big(\tau^{2}+d\sigma^{2}\big)\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{10}
\]

\item[Step 11.]  Since $f(x_{T})\ge f^{\star}$,
\[
f(x_{0})- \E[f(x_{T})]\le f(x_{0})-f^{\star}.
\tag{11}
\]

\item[Step 12.]  Use the elementary inequality $\eta_{t}^{2}\le \eta_{t}\,\bar\eta$
(where $\bar\eta:=\frac{1}{T}\sum_{t=0}^{T-1}\eta_{t}$) to bound the sum of squares:
\[
\sum_{t=0}^{T-1}\eta_{t}^{2}\le \bar\eta\sum_{t=0}^{T-1}\eta_{t}
= \bar\eta\,T\bar\eta = T\bar\eta^{2}.
\tag{12}
\]

\item[Step 13.]  Plug (11) and (12) into (10) and divide by $T\bar\eta$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{\star}\big)}{T\bar\eta}
   + L\big(\tau^{2}+d\sigma^{2}\big)\bar\eta .
\tag{13}
\]

\item[Step 14.]  Equation (13) is exactly the claim of Theorem~\ref{thm:main}.
\qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
For a constant step size $\eta_{t}\equiv\eta\le 1/L$, we have $\bar\eta=\eta$ and
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
+L\big(\tau^{2}+d\sigma^{2}\big)\eta.
\]
Minimising the right‑hand side w.r.t. $\eta$ yields
\[
\eta^{\star}
=
\sqrt{\frac{2\big(f(x_{0})-f^{\star}\big)}
           {L\big(\tau^{2}+d\sigma^{2}\big)T}},
\]
and substituting $\eta^{\star}$ gives the $O(T^{-1/2})$ rate:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_{t})}^{2}\big]
\;\le\;
2\sqrt{\frac{L\big(\tau^{2}+d\sigma^{2}\big)
               \big(f(x_{0})-f^{\star}\big)}\;T^{-1/2}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Zero Privacy Noise ($\sigma=0$).} The bound reduces to the classic
non‑private SGD guarantee with variance $\tau^{2}$.
\item \textbf{Full‑Batch Gradient ($\tau=0$).} Only the Gaussian noise contributes to
the variance term, yielding
$\frac{1}{T}\sum\E[\norm{\nabla f(x_{t})}^{2}]
\le \frac{2\Delta f}{\eta T}+L d\sigma^{2}\eta$.
\item \textbf{Convex $f$.} By convexity, $f(x_{t})-f^{\star}\ge\inner{\nabla f(x_{t})}{x_{t}-x^{\star}}$,
and a standard telescoping argument gives
$\E[f(\bar x_{T})]-f^{\star}\le
\frac{\norm{x_{0}-x^{\star}}^{2}}{2\eta T}
+ \frac{L}{2}\big(\tau^{2}+d\sigma^{2}\big)\eta$,
where $\bar x_{T}=\frac{1}{T}\sum_{t}x_{t}$.
\end{enumerate}
