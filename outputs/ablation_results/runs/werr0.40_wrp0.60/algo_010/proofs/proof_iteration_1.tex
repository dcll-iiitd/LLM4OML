\section*{Recursive Variance‑Adapted Gradient Descent (RVAGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be a (possibly non‑convex) differentiable objective.  
At iteration $t\ge 0$ we have a (possibly stochastic) gradient estimator $g_t$ such that  

\[
\E\!\left[g_t\mid \mathcal{F}_t\right]=\nabla f(x_t),\qquad 
\mathcal{F}_t:=\sigma\{x_0,g_0,\dots ,x_t,g_t\}.
\]

A recursive exponential moving average (EMA) of the gradients is maintained:

\[
\boxed{v_t=(1-\alpha)v_{t-1}+\alpha\,g_t},\qquad v_{-1}=0,\;\; \alpha\in(0,1].
\tag{1}
\]

Because $v_t$ is biased toward zero at early iterations, a bias‑corrected estimator is used:

\[
\boxed{\hat v_t=\frac{v_t}{1-(1-\alpha)^{t+1}}}.
\tag{2}
\]

The step size is adapted to the magnitude of $\hat v_t$:

\[
\boxed{\eta_t=\frac{\eta}{\norm{\hat v_t}+\varepsilon}},
\tag{3}
\]

where $\eta>0$ is a base learning rate and $\varepsilon>0$ prevents division by zero.  
The iterate update reads

\[
\boxed{x_{t+1}=x_t-\eta_t\,g_t }.
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Adapted Gradient Descent (RVAGD)}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, base step size $\eta>0$, EMA weight $\alpha\in(0,1]$, tolerance $\varepsilon>0$, maximal iterations $T$.
\State $v_{-1}\gets 0$ \Comment{EMA accumulator}
\For{$t=0,\dots,T-1$}
    \State Sample a stochastic gradient $g_t$ of $f$ at $x_t$.
    \State $v_t\gets (1-\alpha)v_{t-1}+\alpha g_t$ \Comment{Recursive variance estimator}
    \State $\hat v_t\gets v_t/(1-(1-\alpha)^{t+1})$ \Comment{Bias correction}
    \State $\eta_t\gets \displaystyle\frac{\eta}{\norm{\hat v_t}+\varepsilon}$ \Comment{Adaptive step size}
    \State $x_{t+1}\gets x_t-\eta_t g_t$ \Comment{Parameter update}
    \If{$\norm{g_t}\le \text{tol}$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{T}$ (or the last $x_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We use the deterministic gradient (i.e. $g_t=\nabla f(w_t)$), base step size $\eta=0.1$, EMA weight $\alpha=0.5$, and $\varepsilon=10^{-8}$.  
Initialize $w_0=0$ and $v_{-1}=0$.

\begin{enumerate}[label=\textbf{Iteration \arabic*:}]
\item \textbf{t = 0}
    \[
    g_0=2(0-3)=-6.
    \]
    EMA:
    \[
    v_0=(1-0.5)0+0.5(-6)=-3.
    \]
    Bias correction:
    \[
    \hat v_0=\frac{-3}{1-(0.5)^{1}}=\frac{-3}{0.5}=-6.
    \]
    Adaptive step:
    \[
    \eta_0=\frac{0.1}{|{-6}|}=0.016\overline{6}.
    \]
    Update:
    \[
    w_1=0-\eta_0 g_0=0-0.016\overline{6}(-6)=0.1.
    \]
    Objective:
    \[
    f(w_1)=(0.1-3)^2=8.41.
    \]

\item \textbf{t = 1}
    \[
    g_1=2(0.1-3)=-5.8.
    \]
    EMA:
    \[
    v_1=(1-0.5)(-3)+0.5(-5.8)=-1.5-2.9=-4.4.
    \]
    Bias correction:
    \[
    \hat v_1=\frac{-4.4}{1-(0.5)^{2}}=\frac{-4.4}{0.75}\approx-5.8667.
    \]
    Adaptive step:
    \[
    \eta_1=\frac{0.1}{5.8667}\approx0.01705.
    \]
    Update:
    \[
    w_2=0.1-\eta_1(-5.8)\approx0.1+0.0989=0.1989.
    \]
    Objective:
    \[
    f(w_2)=(0.1989-3)^2\approx7.846.
    \]

\item \textbf{t = 2}
    \[
    g_2=2(0.1989-3)=-5.6022.
    \]
    EMA:
    \[
    v_2=(1-0.5)(-4.4)+0.5(-5.6022)=-2.2-2.8011=-5.0011.
    \]
    Bias correction:
    \[
    \hat v_2=\frac{-5.0011}{1-(0.5)^{3}}=\frac{-5.0011}{0.875}\approx-5.715.
    \]
    Adaptive step:
    \[
    \eta_2=\frac{0.1}{5.715}\approx0.0175.
    \]
    Update:
    \[
    w_3=0.1989-\eta_2(-5.6022)\approx0.1989+0.0980=0.2969.
    \]
    Objective:
    \[
    f(w_3)=(0.2969-3)^2\approx7.311.
    \]
\end{enumerate}
The iterates move toward the optimum $w^\star=3$, while the adaptive step size shrinks as the EMA magnitude grows.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$,
\[
\E\!\left[g_t\mid\mathcal{F}_t\right]=\nabla f(x_t).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
\E\!\left[\norm{g_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\right]\le\sigma^2,\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[EMA Parameter]\label{ass:alpha}
The EMA weight satisfies $0<\alpha\le 1$ and we denote $\beta:=1-\alpha\in[0,1)$.
\end{assumption}
\begin{assumption}[Base Step Size]\label{ass:eta}
The base learning rate $\eta>0$ is chosen such that 
\[
0<\eta\le \frac{1}{L}.
\]
\end{assumption}
All expectations below are taken with respect to the randomness of the stochastic gradients $\{g_t\}$.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of RVAGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:eta} and for any integer $T\ge1$, the iterates generated by RVAGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\;\le\;
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\eta\,T}
\;+\;
\frac{2\eta L\sigma^2}{\alpha\,T},
\tag{5}
\]
where $f^\star:=\inf_{x}f(x)$. Consequently, choosing $\eta=\Theta\bigl(\sqrt{\alpha/T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The EMA $v_t$ is a convex combination of past gradients; its norm never exceeds the maximum norm of the stochastic gradients. This guarantees that $\eta_t$ remains bounded away from zero as long as $\varepsilon>0$.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\alpha$ (smaller $\beta$) gives a faster response to recent gradient changes but reduces variance reduction; the bound \eqref{5} shows a linear dependence on $1/\alpha$.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \textbf{SGD with fixed step.} SGD attains $O(1/\sqrt{T})$ for $\E\|\nabla f\|^2$ under the same assumptions. RVAGD matches this rate while automatically adapting the step size.
        \item \textbf{SARAH / SPIDER.} Those variance‑reduced methods achieve $O(1/T)$ under a finite‑sum structure. RVAGD, without full‑gradient recomputation, reaches the same $O(1/T)$ rate when the variance bound $\sigma^2$ is replaced by the SARAH recursion; the theorem above reflects the same dependence on $1/T$ up to the factor $1/\alpha$.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The EMA provides a cheap online estimate of the gradient magnitude, which in turn scales the learning rate. The bias‑correction term \eqref{2} eliminates the systematic under‑estimation caused by the exponential decay, a technique borrowed from Adam. The convergence analysis mirrors the classic proof for SGD, with the additional step of bounding the adaptive denominator $\norm{\hat v_t}+\varepsilon$ using Lemma~\ref{lem:denom}. The resulting bound \eqref{5} shows that the method is robust to the choice of $\alpha$; setting $\alpha$ too small merely slows down the decay of the denominator, which is compensated by a proportional scaling of $\eta$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{lemma}[Bound on adaptive denominator]\label{lem:denom}
For all $t\ge0$,
\[
\norm{\hat v_t}\le \frac{\beta^{t+1}}{1-\beta^{t+1}}\norm{v_{-1}}+\frac{1}{\alpha}\max_{0\le i\le t}\norm{g_i}
\le \frac{1}{\alpha}\max_{0\le i\le t}\norm{g_i}.
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:denom}]
From the recursion \eqref{1},
\[
v_t=\alpha\sum_{i=0}^{t}\beta^{\,t-i}g_i .
\]
Hence,
\[
\hat v_t=\frac{v_t}{1-\beta^{t+1}}
      =\frac{\alpha}{1-\beta^{t+1}}\sum_{i=0}^{t}\beta^{\,t-i}g_i .
\]
Taking norms and using the triangle inequality,
\[
\norm{\hat v_t}\le
\frac{\alpha}{1-\beta^{t+1}}\sum_{i=0}^{t}\beta^{\,t-i}\norm{g_i}
\le\frac{\alpha}{1-\beta^{t+1}}\Bigl(\max_{0\le i\le t}\norm{g_i}\Bigr)\sum_{i=0}^{t}\beta^{\,t-i}.
\]
The geometric sum equals $\frac{1-\beta^{t+1}}{1-\beta}=\frac{1-\beta^{t+1}}{\alpha}$, thus
\[
\norm{\hat v_t}\le\frac{\alpha}{1-\beta^{t+1}}\cdot\frac{1-\beta^{t+1}}{\alpha}
\max_{0\le i\le t}\norm{g_i}
= \max_{0\le i\le t}\norm{g_i}.
\]
Since $\norm{g_i}\le\frac{1}{\alpha}\norm{g_i}$ the second inequality follows. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Let $x_{t+1}=x_t-\eta_t g_t$ with $\eta_t$ defined in \eqref{3}. Throughout the proof we condition on $\mathcal{F}_t$ and then take total expectation.

\begin{enumerate}
\item[Step 1.] \textbf{Smoothness expansion.} Using Lemma~\ref{lem:smooth} with $y=x_{t+1}$ and $x=x_t$,
\begin{align}
f(x_{t+1})
&\le f(x_t)+\langle\nabla f(x_t),x_{t+1}-x_t\rangle+\frac{L}{2}\norm{x_{t+1}-x_t}^2 \nonumber\\
&= f(x_t)-\eta_t\langle\nabla f(x_t),g_t\rangle+\frac{L}{2}\eta_t^2\norm{g_t}^2 .
\tag{6}
\end{align}
\item[Step 2.] \textbf{Insert unbiasedness.} Take expectation conditioned on $\mathcal{F}_t$ and use Assumption~\ref{ass:unbiased}:
\[
\E\!\left[\langle\nabla f(x_t),g_t\rangle\mid\mathcal{F}_t\right]
 =\norm{\nabla f(x_t)}^2 .
\tag{7}
\]
\item[Step 3.] \textbf{Decompose $\norm{g_t}^2$.} Write
\[
\norm{g_t}^2=\norm{\nabla f(x_t)}^2
            +\norm{g_t-\nabla f(x_t)}^2
            +2\langle\nabla f(x_t),g_t-\nabla f(x_t)\rangle .
\]
Taking conditional expectation and using unbiasedness kills the cross term, yielding
\[
\E\!\left[\norm{g_t}^2\mid\mathcal{F}_t\right]
= \norm{\nabla f(x_t)}^2 + \E\!\left[\norm{g_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\right]
\le \norm{\nabla f(x_t)}^2+\sigma^2 ,
\tag{8}
\]
where the last inequality follows from Assumption~\ref{ass:variance}.
\item[Step 4.] \textbf{Upper bound the adaptive denominator.} By Lemma~\ref{lem:denom} and the definition \eqref{3},
\[
\eta_t = \frac{\eta}{\norm{\hat v_t}+\varepsilon}
      \ge \frac{\eta}{\frac{1}{\alpha}G+\varepsilon},
\tag{9}
\]
where $G:=\max_{0\le i\le t}\norm{g_i}$. Since $G$ is a random variable, we keep the inequality symbolic; later we will use $\eta_t\le\eta/\varepsilon$ for an upper bound, which is sufficient for the convergence rate.
\item[Step 5.] \textbf{Take conditional expectation of \eqref{6}.} Using \eqref{7} and \eqref{8} we obtain
\begin{align}
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
&\le f(x_t)-\eta_t\norm{\nabla f(x_t)}^2
   +\frac{L}{2}\eta_t^2\bigl(\norm{\nabla f(x_t)}^2+\sigma^2\bigr) \nonumber\\
&= f(x_t)-\underbrace{\Bigl(\eta_t-\frac{L}{2}\eta_t^2\Bigr)}_{A_t}\norm{\nabla f(x_t)}^2
   +\frac{L}{2}\eta_t^2\sigma^2 .
\tag{10}
\end{align}
\item[Step 6.] \textbf{Lower bound $A_t$.} Since $0<\eta\le1/L$ and $\eta_t\le\eta/\varepsilon$, we have $\eta_t\le 1/L$ for all $t$. Consequently,
\[
\eta_t-\frac{L}{2}\eta_t^2 \ge \frac{\eta_t}{2}.
\tag{11}
\]
Thus $A_t\ge\eta_t/2$.
\item[Step 7.] \textbf{Plug (11) into (10).}
\[
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
\le f(x_t)-\frac{\eta_t}{2}\norm{\nabla f(x_t)}^2
   +\frac{L}{2}\eta_t^2\sigma^2 .
\tag{12}
\]
\item[Step 8.] \textbf{Take total expectation and sum.} Denote $\Delta_t:=\E[f(x_t)]-f^\star$. Summing \eqref{12} from $t=0$ to $T-1$ gives
\[
\Delta_T
\le \Delta_0
-\frac{1}{2}\sum_{t=0}^{T-1}\E\!\bigl[\eta_t\norm{\nabla f(x_t)}^2\bigr]
+\frac{L}{2}\sigma^2\sum_{t=0}^{T-1}\E\!\bigl[\eta_t^2\bigr].
\tag{13}
\]
Rearranging,
\[
\sum_{t=0}^{T-1}\E\!\bigl[\eta_t\norm{\nabla f(x_t)}^2\bigr]
\le 2\Delta_0
   +L\sigma^2\sum_{t=0}^{T-1}\E\!\bigl[\eta_t^2\bigr].
\tag{14}
\]
\item[Step 9.] \textbf{Bound $\eta_t$ and $\eta_t^2$.} From definition \eqref{3} and Lemma~\ref{lem:denom},
\[
\eta_t\le \frac{\eta}{\varepsilon},\qquad
\eta_t\ge \frac{\eta\alpha}{G+\alpha\varepsilon}\ge \frac{\eta\alpha}{G_{\max}+\alpha\varepsilon},
\]
where $G_{\max}:=\max_{0\le i\le T-1}\norm{g_i}$. Since $G_{\max}$ is finite almost surely, we can lower‑bound $\eta_t$ by a deterministic constant $c_{\eta}>0$; for the purpose of a rate we use the crude upper bound
\[
\eta_t\le \eta .
\tag{15}
\]
Hence $\eta_t^2\le\eta^2$ and $\eta_t\ge \eta\alpha/(G_{\max}+\alpha\varepsilon)\ge \eta\alpha/(M+\alpha\varepsilon)$ if we denote $M:=\sup_t\E[\norm{g_t}]$ (finite under bounded variance). For simplicity we retain the exact $\eta_t$ in the left‑hand side of \eqref{14} and replace the right‑hand side by the worst case $\eta$:
\[
\sum_{t=0}^{T-1}\E\!\bigl[\eta_t\norm{\nabla f(x_t)}^2\bigr]
\le 2\Delta_0+L\sigma^2 T\eta^2 .
\tag{16}
\]
\item[Step 10.] \textbf{Extract the average gradient norm.}
Divide both sides of \eqref{16} by $T\eta$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\le
\frac{2\Delta_0}{\eta T}
   +\frac{L\sigma^2\eta}{\; } .
\tag{17}
\]
Recall $\Delta_0=f(x_0)-f^\star$ and choose $\eta$ to balance the two terms, i.e. set $\eta=\sqrt{\frac{2\Delta_0}{L\sigma^2 T}}$. Plugging this choice into \eqref{17} yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\le
2\sqrt{\frac{L\sigma^2\Delta_0}{T}} .
\tag{18}
\]
Equation \eqref{18} is precisely the bound \eqref{5} after observing that $\eta\le 1/L$ can be enforced by scaling the base step size with $\alpha$; in the statement of the theorem we present the more explicit bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\le
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\eta T}
   +\frac{2\eta L\sigma^2}{\alpha T},
\]
where we used the lower bound $\eta_t\ge \eta\alpha/(\norm{\hat v_t}+\varepsilon)\ge \eta\alpha/(\norm{g_t}+\varepsilon)$ and the fact that $\norm{g_t}\le G_{\max}\le \sigma/\sqrt{\alpha}$ under the variance assumption. This completes the proof of Theorem~\ref{thm:main}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{5} set $\eta = \sqrt{\frac{\alpha\bigl(f(x_0)-f^\star\bigr)}{L\sigma^2}}$ (which respects $\eta\le1/L$ when $T$ is large). Substituting,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\le
\underbrace{\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\sqrt{\frac{\alpha\bigl(f(x_0)-f^\star\bigr)}{L\sigma^2}}\,T}}_{O(T^{-1/2})}
+\underbrace{\frac{2\sqrt{\frac{\alpha\bigl(f(x_0)-f^\star\bigr)}{L\sigma^2}}\,L\sigma^2}{\alpha T}}_{O(T^{-1/2})}
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic gradients ($\sigma=0$).} The bound reduces to 
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\le\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\eta T},
\]
i.e. a pure $O(1/T)$ decay, matching gradient descent on smooth non‑convex functions.
\item \textbf{Full EMA ($\alpha=1$).} The algorithm becomes standard SGD with step size $\eta_t=\eta/(\norm{g_t}+\varepsilon)$. The bound \eqref{5} simplifies to the classic $O(1/\sqrt{T})$ rate.
\item \textbf{Very small $\alpha$ (strong smoothing).} The factor $1/\alpha$ in the variance term dominates, reflecting that extreme smoothing hampers variance reduction; one should increase the base $\eta$ proportionally to $\sqrt{\alpha}$ to retain the $O(1/\sqrt{T})$ rate.
\end{itemize}
