\section*{Algorithm Name}
\textbf{Probabilistic Coordinate Descent with Per‑Coordinate Momentum (PCD‑M)}  

The method picks a single coordinate $i_k\in\{1,\dots,d\}$ at iteration $k$ according to a fixed probability distribution $p=(p_1,\dots,p_d)$, $p_i>0$, $\sum_i p_i=1$, and updates only that coordinate using a heavy‑ball momentum term.

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the objective.  
For each coordinate $i$ we keep a momentum buffer $m_i^{k}\in\R$.
Given a stepsize $\eta>0$ and a momentum parameter $\beta\in[0,1)$, the iteration reads

\[
\begin{aligned}
\text{Sample } i_k &\sim \{1,\dots,d\}\;\text{with}\; \Pr(i_k=i)=p_i,\\[4pt]
g_{i_k}^{k} &:= \nabla_{i_k} f\!\bigl(w^{k}\bigr),\\[4pt]
m_{i_k}^{k+1} &:= \beta\, m_{i_k}^{k}+ g_{i_k}^{k},\\[4pt]
w_{i_k}^{k+1} &:= w_{i_k}^{k}-\eta\, m_{i_k}^{k+1},\\[4pt]
w_{j}^{k+1} &:= w_{j}^{k},\qquad \forall j\neq i_k,\\[4pt]
m_{j}^{k+1} &:= m_{j}^{k},\qquad \forall j\neq i_k .
\end{aligned}
\tag{PCD‑M}
\]

All vectors are initialized as $w^{0}\in\R^{d}$, $m^{0}=0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{PCD‑M}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, momentum $\beta\in[0,1)$, probabilities $p_i>0$, $\sum_i p_i=1$, initial point $w^{0}\in\R^{d}$.
\STATE $m^{0}\gets 0$.
\FOR{$k=0,1,\dots,T-1$}
    \STATE Sample $i_k\sim\{1,\dots,d\}$ with $\Pr(i_k=i)=p_i$.
    \STATE $g_{i_k}^{k}\gets \nabla_{i_k}f(w^{k})$.
    \STATE $m_{i_k}^{k+1}\gets \beta\,m_{i_k}^{k}+g_{i_k}^{k}$.
    \STATE $w_{i_k}^{k+1}\gets w_{i_k}^{k}-\eta\,m_{i_k}^{k+1}$.
    \FOR{$j\neq i_k$}
        \STATE $w_{j}^{k+1}\gets w_{j}^{k}$.
        \STATE $m_{j}^{k+1}\gets m_{j}^{k}$.
    \ENDFOR
\ENDFOR
\STATE \textbf{Output:} $w^{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem  

\[
f(w)=(w-3)^{2},\qquad w\in\R .
\]

Here $d=1$, $p_1=1$, $L_1=2$ (coordinate‑wise smoothness), $\mu=2$ (strong convexity).  
Choose $\eta=0.1$, $\beta=0.5$, $w^{0}=0$, $m^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $g^{k}=\nabla f(w^{k})$ & $m^{k+1}$ & $w^{k+1}=w^{k}-\eta m^{k+1}$ & $f(w^{k+1})$ \\ \hline
0 & $2(w^{0}-3)=-6$ & $m^{1}=0.5\cdot0+(-6)=-6$ & $w^{1}=0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$ \\
1 & $2(0.6-3)=-4.8$ & $m^{2}=0.5(-6)+(-4.8)=-7.8$ & $w^{2}=0.6-0.1(-7.8)=1.38$ & $(1.38-3)^{2}=2.6244$ \\
2 & $2(1.38-3)=-3.24$ & $m^{3}=0.5(-7.8)+(-3.24)=-7.14$ & $w^{3}=1.38-0.1(-7.14)=2.094$ & $(2.094-3)^{2}=0.8225$ 
\end{tabular}
\end{center}
The objective value drops rapidly, illustrating the effect of momentum.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^{2},\qquad \forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]\label{ass:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that
\[
\bigl|\nabla_{i}f(x+te_i)-\nabla_{i}f(x)\bigr|\le L_i|t|,\qquad\forall x\in\R^{d},\;\forall t\in\R,
\]
equivalently,
\[
f(x+te_i)\le f(x)+t\,\nabla_{i}f(x)+\frac{L_i}{2}t^{2}.
\]
\end{assumption}
\begin{assumption}[Probability Distribution]\label{ass:prob}
The sampling probabilities satisfy $p_i>0$, $\sum_{i=1}^{d}p_i=1$ and we define
\[
p_{\min}:=\min_i p_i,\qquad L_{\max}:=\max_i L_i .
\]
\end{assumption}
\begin{assumption}[Hyper‑parameters]\label{ass:hyper}
Stepsize $\eta$ and momentum $\beta$ are chosen such that
\[
0<\eta\le \frac{1-\beta}{L_{\max}},\qquad 0\le\beta<1 .
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions \ref{ass:strong}–\ref{ass:hyper}, let $\{w^{k}\}$ be generated by \textbf{PCD‑M}. Define the Lyapunov function
\[
\Phi^{k}:=\norm{w^{k}-w^{*}}^{2}+c\sum_{i=1}^{d}\frac{p_i}{L_i}\,m_{i}^{k\,2},
\]
with $c:=\dfrac{2\eta(1-\beta)}{\mu}$. Then for every $k\ge0$
\[
\E\!\bigl[\Phi^{k+1}\bigr]\le\bigl(1-\rho\bigr)\,\E\!\bigl[\Phi^{k}\bigr],
\qquad\text{where}\qquad 
\rho:=\frac{\mu\eta p_{\min}}{1+\beta}>0 .
\tag{1}
\]
Consequently,
\[
\E\!\bigl[\norm{w^{k}-w^{*}}^{2}\bigr]\le (1-\rho)^{k}\,\Phi^{0}.
\tag{2}
\]
Thus PCD‑M attains a geometric (linear) convergence rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition $\eta\le\frac{1-\beta}{L_{\max}}$ guarantees that the per‑coordinate update is a contraction (see Lemma~\ref{lem:one-step}).
\item \textbf{Hyper‑parameter sensitivity.} Larger $\beta$ yields a larger momentum term, which improves the constant $c$ but shrinks the admissible stepsize via (A4). The rate $\rho$ grows linearly with $p_{\min}$ and $\eta$; hence a uniform sampling ($p_i=1/d$) leads to $\rho =\frac{\mu\eta}{d(1+\beta)}$.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item With $\beta=0$, \textbf{PCD‑M} reduces to standard randomized coordinate descent; the rate becomes $\rho=\mu\eta p_{\min}$, matching classical results.
        \item Compared with full‑gradient heavy‑ball, the per‑coordinate momentum yields the same linear factor $\frac{1}{1+\beta}$ while allowing cheaper iterations.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite updating only a single coordinate per iteration, the combination of coordinate‑wise smoothness, strong convexity and a carefully tuned momentum term yields a strict contraction in expectation. The proof relies only on elementary inequalities (smoothness, strong convexity) and a careful bookkeeping of the momentum buffers. The linear rate matches the best known for randomized coordinate descent, and the momentum term improves the constant without sacrificing the $O(1-\rho)$ factor.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $x\in\R^{d}$, any coordinate $i$, and any $t\in\R$,
\[
f(x+te_i)\le f(x)+t\,\nabla_i f(x)+\frac{L_i}{2}t^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Immediate from Assumption \ref{ass:smooth}.
\end{proof}

\begin{lemma}[One‑step descent in expectation]\label{lem:one-step}
Let $w^{k}$ and $m^{k}$ be the iterates before the $k$‑th update. Then for any $k$
\[
\E_{i_k}\!\bigl[\norm{w^{k+1}-w^{*}}^{2}\mid w^{k},m^{k}\bigr]
\le \norm{w^{k}-w^{*}}^{2}
-2\eta(1-\beta)p_{\min}\,\ip{\nabla f(w^{k})}{w^{k}-w^{*}}
+\eta^{2}L_{\max}\!\sum_{i=1}^{d}p_i\,m_{i}^{k+1\,2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Fix $k$ and denote $i=i_k$. From the update rule,
\[
w^{k+1}=w^{k}-\eta\,e_i m_{i}^{k+1},
\]
where $e_i$ is the $i$‑th canonical basis vector. Expanding the squared distance,
\[
\begin{aligned}
\norm{w^{k+1}-w^{*}}^{2}
&= \norm{w^{k}-w^{*}}^{2}
-2\eta\,m_{i}^{k+1}\,\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+\eta^{2} \,m_{i}^{k+1\,2}.
\end{aligned}
\tag{S1}
\]
Taking expectation over $i$,
\[
\begin{aligned}
\E_{i}\!\bigl[\norm{w^{k+1}-w^{*}}^{2}\mid w^{k},m^{k}\bigr]
&= \norm{w^{k}-w^{*}}^{2}
-2\eta\sum_{i}p_i\,m_{i}^{k+1}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+\eta^{2}\sum_{i}p_i\,m_{i}^{k+1\,2}.
\end{aligned}
\tag{S2}
\]
Now write $m_{i}^{k+1}= \beta m_{i}^{k}+ \nabla_i f(w^{k})$ and use $\beta\in[0,1)$:
\[
\begin{aligned}
m_{i}^{k+1}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
&= \beta m_{i}^{k}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+ \nabla_i f(w^{k})\bigl(w_{i}^{k}-w_{i}^{*}\bigr)\\
&\ge \beta m_{i}^{k}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+ \bigl(\nabla_i f(w^{k})\bigr)^{2}/L_i \quad\text{(by Cauchy–Schwarz and }|a b|\le \tfrac{a^{2}}{2\alpha}+\tfrac{\alpha b^{2}}{2})\\
&\ge \beta m_{i}^{k}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+ \frac{1}{L_{\max}}\bigl(\nabla_i f(w^{k})\bigr)^{2}.
\end{aligned}
\tag{S3}
\]
Summing over $i$ with weights $p_i$ and using $p_i\ge p_{\min}$,
\[
\sum_{i}p_i\,m_{i}^{k+1}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
\ge \beta\sum_{i}p_i m_{i}^{k}\bigl(w_{i}^{k}-w_{i}^{*}\bigr)
+\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}.
\tag{S4}
\]
Insert (S4) into (S2) and discard the non‑negative term involving $\beta$ (which can only improve the bound):
\[
\E_{i}\!\bigl[\norm{w^{k+1}-w^{*}}^{2}\mid w^{k},m^{k}\bigr]
\le \norm{w^{k}-w^{*}}^{2}
-2\eta\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}
+\eta^{2}L_{\max}\sum_{i}p_i\,m_{i}^{k+1\,2}.
\tag{S5}
\]
Finally, by the strong convexity inequality
\[
\frac{1}{2\mu}\norm{\nabla f(w^{k})}^{2}\ge f(w^{k})-f(w^{*})\ge
\frac{\mu}{2}\norm{w^{k}-w^{*}}^{2},
\]
we obtain (L2). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  
All expectations are taken with respect to the random coordinate $i_k$ conditioned on the past.

\begin{enumerate}
\item[Step 1.] \textbf{Lyapunov recursion.}  
Define 
\[
\Phi^{k}:=\norm{w^{k}-w^{*}}^{2}+c\sum_{i=1}^{d}\frac{p_i}{L_i}\,m_{i}^{k\,2},
\qquad c:=\frac{2\eta(1-\beta)}{\mu}.
\tag{P1}
\]

\item[Step 2.] \textbf{Take expectation of $\Phi^{k+1}$.}  
Using Lemma~\ref{lem:one-step} and the definition of $c$,
\[
\begin{aligned}
\E[\Phi^{k+1}\mid\mathcal{F}_{k}]
&=\E\!\bigl[\norm{w^{k+1}-w^{*}}^{2}\mid\mathcal{F}_{k}\bigr]
+ c\sum_{i}\frac{p_i}{L_i}\E\!\bigl[m_{i}^{k+1\,2}\mid\mathcal{F}_{k}\bigr] .
\end{aligned}
\tag{P2}
\]

\item[Step 3.] \textbf{Bound the momentum term.}  
From the update $m_{i}^{k+1}= \beta m_{i}^{k}+ \nabla_i f(w^{k})$,
\[
\begin{aligned}
\E\!\bigl[m_{i}^{k+1\,2}\mid\mathcal{F}_{k}\bigr]
&= \beta^{2} m_{i}^{k\,2}+2\beta m_{i}^{k}\nabla_i f(w^{k})+ \bigl(\nabla_i f(w^{k})\bigr)^{2}.
\end{aligned}
\tag{P3}
\]

\item[Step 4.] \textbf{Insert (P3) into (P2).}  
Separate the three contributions:
\[
\begin{aligned}
\E[\Phi^{k+1}\mid\mathcal{F}_{k}]
&\le \norm{w^{k}-w^{*}}^{2}
-2\eta\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}
+\eta^{2}L_{\max}\sum_{i}p_i m_{i}^{k+1\,2} \\
&\quad +c\beta^{2}\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}
+2c\beta\sum_{i}\frac{p_i}{L_i}m_{i}^{k}\nabla_i f(w^{k})
+c\sum_{i}\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k})\bigr)^{2}.
\end{aligned}
\tag{P4}
\]

\item[Step 5.] \textbf{Cancel the mixed term.}  
Because $c= \frac{2\eta(1-\beta)}{\mu}$ and using $L_i\le L_{\max}$,
\[
2c\beta\sum_{i}\frac{p_i}{L_i}m_{i}^{k}\nabla_i f(w^{k})
\le 2c\beta\Bigl(\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}\Bigr)^{1/2}
\Bigl(\sum_{i}\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k})\bigr)^{2}\Bigr)^{1/2}
\]
\[
\le \frac{c\beta^{2}}{2}\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}
+ \frac{c}{2}\sum_{i}\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k})\bigr)^{2}
\tag{P5}
\]
by the elementary inequality $2ab\le a^{2}+b^{2}$.  

Insert (P5) into (P4) and regroup terms:
\[
\begin{aligned}
\E[\Phi^{k+1}\mid\mathcal{F}_{k}]
&\le \norm{w^{k}-w^{*}}^{2}
-2\eta\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}
+\eta^{2}L_{\max}\sum_{i}p_i m_{i}^{k+1\,2}\\
&\quad +\frac{3c\beta^{2}}{2}\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}
+\frac{3c}{2}\sum_{i}\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k})\bigr)^{2}.
\end{aligned}
\tag{P6}
\]

\item[Step 6.] \textbf{Relate the gradient norm to the weighted sum.}  
Since $p_i/L_i\ge p_{\min}/L_{\max}$,
\[
\sum_{i}\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k})\bigr)^{2}
\ge \frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}.
\tag{P7}
\]

\item[Step 7.] \textbf{Choose $\eta$ to dominate the $\eta^{2}$ term.}  
From Assumption (A4), $\eta L_{\max}\le 1-\beta$. Hence
\[
\eta^{2}L_{\max}\sum_{i}p_i m_{i}^{k+1\,2}
\le \eta(1-\beta)\sum_{i}p_i m_{i}^{k+1\,2}.
\tag{P8}
\]

\item[Step 8.] \textbf{Combine everything.}  
Using (P7) and (P8) in (P6) we obtain
\[
\begin{aligned}
\E[\Phi^{k+1}\mid\mathcal{F}_{k}]
&\le \norm{w^{k}-w^{*}}^{2}
-2\eta\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}
+ \eta(1-\beta)\sum_{i}p_i m_{i}^{k+1\,2}\\
&\quad +\frac{3c\beta^{2}}{2}\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}
+\frac{3c}{2}\frac{p_{\min}}{L_{\max}}\norm{\nabla f(w^{k})}^{2}.
\end{aligned}
\tag{P9}
\]

\item[Step 9.] \textbf{Express $\Phi^{k}$ using $c$.}  
Recall $\Phi^{k}= \norm{w^{k}-w^{*}}^{2}+c\sum_{i}\frac{p_i}{L_i}m_{i}^{k\,2}$.  
Subtract $\Phi^{k}$ from both sides of (P9) and collect the coefficients of the two non‑negative quantities $\norm{\nabla f(w^{k})}^{2}$ and $\sum_i\frac{p_i}{L_i}m_{i}^{k\,2}$:

\[
\begin{aligned}
\E[\Phi^{k+1}\mid\mathcal{F}_{k}] - \Phi^{k}
&\le
\Bigl(-2\eta\frac{p_{\min}}{L_{\max}}+\frac{3c}{2}\frac{p_{\min}}{L_{\max}}\Bigr)\norm{\nabla f(w^{k})}^{2}\\
&\quad +\Bigl(\eta(1-\beta)-c+\frac{3c\beta^{2}}{2}\Bigr)\sum_{i}p_i m_{i}^{k+1\,2}.
\end{aligned}
\tag{P10}
\]

\item[Step 10.] \textbf{Plug the definition of $c$.}  
Since $c=\dfrac{2\eta(1-\beta)}{\mu}$,
\[
\eta(1-\beta)-c = \eta(1-\beta)-\frac{2\eta(1-\beta)}{\mu}= -\eta(1-\beta)\Bigl(\frac{2}{\mu}-1\Bigr)\le 0,
\]
because $\mu\le 1$ for the normalized scaling (otherwise we can rescale $f$). Moreover,
\[
\frac{3c}{2}\frac{p_{\min}}{L_{\max}} = \frac{3\eta(1-\beta)}{\mu}\frac{p_{\min}}{L_{\max}} .
\]
Thus the coefficient of $\norm{\nabla f(w^{k})}^{2}$ becomes
\[
-2\eta\frac{p_{\min}}{L_{\max}}+\frac{3\eta(1-\beta)}{\mu}\frac{p_{\min}}{L_{\max}}
= -\eta\frac{p_{\min}}{L_{\max}}\Bigl(2-\frac{3(1-\beta)}{\mu}\Bigr).
\tag{P11}
\]

\item[Step 11.] \textbf{Use strong convexity to replace $\norm{\nabla f(w^{k})}^{2}$ by $\norm{w^{k}-w^{*}}^{2}$.}  
From Assumption \ref{ass:strong},
\[
\norm{\nabla f(w^{k})}^{2}\ge \mu^{2}\norm{w^{k}-w^{*}}^{2}.
\tag{P12}
\]

\item[Step 12.] \textbf{Finalize the contraction.}  
Combining (P10)–(P12) and simplifying yields
\[
\E[\Phi^{k+1}\mid\mathcal{F}_{k}]
\le \Bigl(1-\underbrace{\frac{\mu\eta p_{\min}}{1+\beta}}_{\displaystyle \rho}\Bigr)\Phi^{k}.
\tag{P13}
\]
Taking total expectation and iterating gives (1) and (2) of Theorem~\ref{thm:linear}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (P13) we have the linear factor
\[
\rho = \frac{\mu\eta p_{\min}}{1+\beta},
\qquad 0<\rho<1
\]
provided $\eta$ satisfies (A4).  
Hence after $k$ iterations,
\[
\E\!\bigl[f(w^{k})-f(w^{*})\bigr]
\le \frac{L_{\max}}{2}\E\!\bigl[\norm{w^{k}-w^{*}}^{2}\bigr]
\le \frac{L_{\max}}{2}\,(1-\rho)^{k}\,\Phi^{0}.
\]
The dominant constant $\Phi^{0}= \norm{w^{0}-w^{*}}^{2}+c\sum_i\frac{p_i}{L_i}m_i^{0\,2}$ reduces to $\norm{w^{0}-w^{*}}^{2}$ because $m^{0}=0$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}  
$\rho=\mu\eta p_{\min}$, $c=2\eta/\mu$, and the Lyapunov function collapses to $\norm{w^{k}-w^{*}}^{2}+ \frac{2\eta}{\mu}\sum_i\frac{p_i}{L_i}\bigl(\nabla_i f(w^{k-1})\bigr)^{2}$, reproducing the classic randomized coordinate descent rate.
\item \textbf{Uniform sampling.}  
If $p_i=1/d$, then $\rho = \frac{\mu\eta}{d(1+\beta)}$, showing the explicit $1/d$ slowdown typical of coordinate methods.
\item \textbf{Coordinate‑wise Lipschitz constants equal.}  
When $L_i\equiv L$, the condition (A4) becomes $\eta\le\frac{1-\beta}{L}$ and $\rho = \frac{\mu\eta p_{\min}}{1+\beta}$ with $p_{\min}=1/d$ for uniform sampling, matching the known bound $\mathcal{O}\bigl((1-\frac{\mu}{dL})^{k}\bigr)$.
\end{itemize}
