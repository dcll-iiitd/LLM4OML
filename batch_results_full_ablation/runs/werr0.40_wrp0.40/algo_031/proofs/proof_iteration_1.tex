\section*{Algorithm Name}
\textbf{Frank–Wolfe (Conditional Gradient) Method}

\section*{Mathematical Formulation}
Consider the constrained convex optimisation problem  

\[
\min_{x\in\mathcal C}\; f(x),
\tag{P}
\]

where  

\begin{itemize}
\item $f:\mathbb R^{d}\rightarrow\mathbb R$ is convex and $L$‑smooth (i.e. differentiable and its gradient is $L$‑Lipschitz);
\item $\mathcal C\subset\mathbb R^{d}$ is a non‑empty, convex and compact set with (Euclidean) diameter  

\[
D \;:=\; \max_{u,v\in\mathcal C}\|u-v\| .
\]
\end{itemize}

At iteration $k\ge 0$ the algorithm performs the following two steps  

\begin{enumerate}
\item \textbf{Linear minimisation oracle (LMO)}  

\[
s_k \;:=\; \arg\min_{s\in\mathcal C}\;\langle \nabla f(x_k),\,s\rangle .
\tag{1}
\]

\item \textbf{Convex combination update}  

\[
x_{k+1}\;:=\;(1-\gamma_k)\,x_k+\gamma_k\,s_k ,
\tag{2}
\]

where $\{\gamma_k\}_{k\ge0}\subset(0,1]$ is a stepsize sequence (either a prescribed schedule
$\gamma_k =\frac{2}{k+2}$ or a line‑search value).  
\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe (Conditional Gradient)}\label{alg:FW}
\begin{algorithmic}[1]
\REQUIRE Convex smooth $f$, feasible set $\mathcal C$, initial point $x_0\in\mathcal C$, stepsize rule $\{\gamma_k\}$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k \gets \nabla f(x_k)$.
    \STATE \textbf{LMO:} $s_k \gets \arg\min_{s\in\mathcal C}\langle g_k,s\rangle$.
    \STATE Choose stepsize $\gamma_k$ (e.g. $\gamma_k=2/(k+2)$ or via line‑search).
    \STATE Update $x_{k+1}\gets (1-\gamma_k)x_k+\gamma_k s_k$.
    \IF{termination criterion satisfied (e.g. duality gap $<\varepsilon$)}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimise the one‑dimensional quadratic  

\[
f(w)=(w-3)^2 ,\qquad \mathcal C=[0,5] .
\]

The gradient is $\nabla f(w)=2(w-3)$.  
Take $w_0=0$ and a \emph{fixed} stepsize $\gamma_k=\gamma=0.1$ for three iterations.

\medskip
\noindent\textbf{Iteration $0$}
\[
\begin{aligned}
g_0 &= \nabla f(w_0)=2(0-3)=-6,\\[2mm]
s_0 &= \arg\min_{s\in[0,5]}\langle -6,s\rangle = \arg\min_{s\in[0,5]}(-6s)=5,\\[2mm]
w_1 &= (1-\gamma)w_0+\gamma s_0 =0.9\cdot0+0.1\cdot5 =0.5,\\[2mm]
f(w_1) &= (0.5-3)^2 = 6.25 .
\end{aligned}
\]

\noindent\textbf{Iteration $1$}
\[
\begin{aligned}
g_1 &= 2(0.5-3) = -5,\\
s_1 &= \arg\min_{s\in[0,5]}(-5s)=5,\\
w_2 &= 0.9\cdot0.5+0.1\cdot5 =0.95,\\
f(w_2) &= (0.95-3)^2 \approx 4.2025 .
\end{aligned}
\]

\noindent\textbf{Iteration $2$}
\[
\begin{aligned}
g_2 &= 2(0.95-3) = -4.1,\\
s_2 &= \arg\min_{s\in[0,5]}(-4.1s)=5,\\
w_3 &= 0.9\cdot0.95+0.1\cdot5 =1.355,\\
f(w_3) &= (1.355-3)^2 \approx 2.698 .
\end{aligned}
\]

The objective value decreases monotonically, illustrating the algorithm’s progress.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex-smooth}
The objective $f$ is convex and $L$‑smooth on an open set containing $\mathcal C$, i.e.  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathcal C .
\]
\end{assumption}

\begin{assumption}[Compact Feasible Set]\label{ass:compact}
$\mathcal C$ is convex, compact and has finite diameter  

\[
D:=\max_{u,v\in\mathcal C}\|u-v\|<\infty .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize sequence is either
\[
\gamma_k = \frac{2}{k+2},\qquad k\ge0,
\tag{3}
\]
or any sequence satisfying $0<\gamma_k\le 1$ and $\sum_{k=0}^{\infty}\gamma_k =\infty$, $\sum_{k=0}^{\infty}\gamma_k^2<\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal Gap $O(1/k)$]\label{thm:fw-rate}
Let $x^\star\in\mathcal C$ be a minimiser of $f$ over $\mathcal C$.  
Under Assumptions~\ref{ass:convex-smooth}–\ref{ass:stepsize} and with the stepsize $\gamma_k=2/(k+2)$,
the iterates generated by the Frank–Wolfe method satisfy  

\[
f(x_k)-f(x^\star)\;\le\;\frac{2L D^{2}}{k+2},
\qquad\forall k\ge0 .
\tag{4}
\]

Consequently the duality gap $g(x_k):=\max_{s\in\mathcal C}\langle\nabla f(x_k),x_k-s\rangle$ also decays as $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability w.r.t.\ stepsize.} The bound (4) holds for the canonical schedule (3). Any stepsize satisfying Assumption~\ref{ass:stepsize} yields the same $O(1/k)$ rate, possibly with a different constant.
\item \textbf{Comparison to projected gradient.} Frank–Wolfe avoids projection onto $\mathcal C$; its rate matches the optimal $O(1/k)$ for first‑order methods on smooth convex problems, while each iteration only requires solving a linear problem (the LMO).
\item \textbf{Sensitivity to diameter $D$.} The constant $2LD^{2}$ shows that a tighter feasible set (smaller $D$) directly improves the worst‑case bound.
\item \textbf{Special case – strongly convex $f$.} If $f$ is $\mu$‑strongly convex, the primal gap decays linearly to a neighbourhood of size $O(D^{2}L^{2}/\mu)$ (see e.g. Jaggi, 2013). 
\end{itemize}

\section*{Discussion}
The proof below follows the classical analysis of Jaggi (2013) but expands every algebraic manipulation and annotates each non‑trivial step with a line number for easy reference.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For any $x,y\in\mathcal C$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Diameter Bound]\label{lem:diameter}
For any $u,v\in\mathcal C$, $\|u-v\|\le D$.
\tag{6}
\]
\end{lemma}
\begin{proof}
By definition of $D$. \qedhere
\end{proof}

\begin{lemma}[LMO Optimality]\label{lem:lmo}
For the point $s_k$ defined in (1) and any $x\in\mathcal C$,
\[
\langle\nabla f(x_k),s_k-x\rangle\le\langle\nabla f(x_k),x_k-x\rangle .
\tag{7}
\]
\end{lemma}
\begin{proof}
Since $s_k$ minimises the linear functional $\langle\nabla f(x_k),\cdot\rangle$ over $\mathcal C$, we have
\[
\langle\nabla f(x_k),s_k\rangle\le\langle\nabla f(x_k),x\rangle\quad\forall x\in\mathcal C.
\]
Rearranging yields (7). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw-rate}. Let $x^\star\in\arg\min_{x\in\mathcal C}f(x)$.  
For brevity denote $g_k:=\nabla f(x_k)$.

\bigskip
\noindent\textbf{[Step 1]} Apply the smoothness inequality (5) with $x=x_k$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle g_k, x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{8}
\]

\noindent\textbf{[Step 2]} Substitute the update rule (2):
\[
x_{k+1}-x_k = \gamma_k(s_k-x_k).
\tag{9}
\]

\noindent\textbf{[Step 3]} Insert (9) into (8):
\[
f(x_{k+1})\le f(x_k)+\gamma_k\langle g_k, s_k-x_k\rangle
      +\frac{L}{2}\gamma_k^{2}\|s_k-x_k\|^{2}.
\tag{10}
\]

\noindent\textbf{[Step 4]} Use Lemma~\ref{lem:lmo} (7) with $x=x^\star$:
\[
\langle g_k, s_k-x_k\rangle
= -\langle g_k, x_k-s_k\rangle
\le -\langle g_k, x_k-x^\star\rangle .
\tag{11}
\]

\noindent\textbf{[Step 5]} Plug (11) into (10):
\[
f(x_{k+1})\le f(x_k)-\gamma_k\langle g_k, x_k-x^\star\rangle
            +\frac{L}{2}\gamma_k^{2}\|s_k-x_k\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 6]} By convexity of $f$,
\[
f(x_k)-f(x^\star)\le \langle g_k, x_k-x^\star\rangle .
\tag{13}
\]

\noindent\textbf{[Step 7]} Combine (12) and (13):
\[
\begin{aligned}
f(x_{k+1})-f(x^\star)
&\le f(x_k)-f(x^\star)
   -\gamma_k\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{L}{2}\gamma_k^{2}\|s_k-x_k\|^{2}\\[1mm]
&= (1-\gamma_k)\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{L}{2}\gamma_k^{2}\|s_k-x_k\|^{2}.
\end{aligned}
\tag{14}
\]

\noindent\textbf{[Step 8]} Bound the distance term using Lemma~\ref{lem:diameter} (6):
\[
\|s_k-x_k\|\le D\quad\Longrightarrow\quad
\|s_k-x_k\|^{2}\le D^{2}.
\tag{15}
\]

\noindent\textbf{[Step 9]} Insert (15) into (14):
\[
f(x_{k+1})-f(x^\star)\le (1-\gamma_k)\bigl(f(x_k)-f(x^\star)\bigr)
    +\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{16}
\]

\noindent\textbf{[Step 10]} Choose the canonical stepsize $\displaystyle\gamma_k=\frac{2}{k+2}$.
Then $1-\gamma_k = \frac{k}{k+2}$ and $\gamma_k^{2}= \frac{4}{(k+2)^{2}}$.  
Substituting into (16) yields

\[
f(x_{k+1})-f(x^\star)\le
\frac{k}{k+2}\bigl(f(x_k)-f(x^\star)\bigr)
+\frac{2L D^{2}}{(k+2)^{2}}.
\tag{17}
\]

\noindent\textbf{[Step 11]} Define $h_k:=f(x_k)-f(x^\star)$.  
Inequality (17) becomes

\[
h_{k+1}\le\frac{k}{k+2}h_k+\frac{2L D^{2}}{(k+2)^{2}}.
\tag{18}
\]

\noindent\textbf{[Step 12]} Multiply both sides of (18) by $(k+2)(k+1)$:

\[
(k+2)(k+1)h_{k+1}\le k(k+1)h_k+2L D^{2}.
\tag{19}
\]

\noindent\textbf{[Step 13]} Define the auxiliary sequence $A_k:=k(k+1)h_k$.  
From (19) we obtain the simple recursion  

\[
A_{k+1}\le A_k+2L D^{2}.
\tag{20}
\]

\noindent\textbf{[Step 14]} Unfolding (20) from $0$ to $k$ gives  

\[
A_{k}\le A_{0}+2L D^{2}k .
\tag{21}
\]

\noindent\textbf{[Step 15]} Since $x_0\in\mathcal C$, $h_0$ is finite; moreover $A_0=0\cdot1\cdot h_0=0$.  
Thus $A_k\le 2L D^{2}k$.  Dividing by $k(k+1)$ yields

\[
h_k\le \frac{2L D^{2}}{k+1}.
\tag{22}
\]

\noindent\textbf{[Step 16]} Re‑indexing $k\leftarrow k-1$ provides the bound stated in (4):

\[
f(x_k)-f(x^\star)\le \frac{2L D^{2}}{k+2},\qquad\forall k\ge0 .
\]

This completes the proof. \qed

\section*{Rate Derivation (With Constants)}
The derivation above shows that the constant in front of $1/k$ is exactly $2L D^{2}$.  
If a different stepsize rule $\gamma_k=\frac{c}{k+1}$ with $c\in(0,2]$ is used, the same algebra leads to  

\[
f(x_k)-f(x^\star)\le \frac{c\,L D^{2}}{k+1},
\]

so the optimal choice $c=2$ minimises the worst‑case bound.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Polyhedral feasible set.} When $\mathcal C$ is a polytope, the LMO can be solved by a linear program; the $O(1/k)$ rate still holds.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, combining (16) with the strong convexity inequality yields linear convergence to a neighbourhood of $x^\star$ of radius $O(L D^{2}/\mu)$.
\item \textbf{Exact line‑search stepsize.} Choosing $\gamma_k$ that minimises $f\big((1-\gamma)x_k+\gamma s_k\big)$ gives a bound identical to (4) but often with a smaller empirical constant.
\end{itemize}
