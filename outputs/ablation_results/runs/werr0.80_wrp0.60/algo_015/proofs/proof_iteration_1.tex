\section*{SignSGD with Momentum (SignSGDM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the (possibly stochastic) loss function.  
At iteration $t\ge 0$ we have the iterate $x_t\in\mathbb{R}^{d}$ and a momentum vector $m_t\in\mathbb{R}^{d}$.
\[
\begin{aligned}
g_t &:= \nabla f(x_t;\,\xi_t) \qquad\text{(stochastic gradient on a random sample }\xi_t\text{)}\\[4pt]
m_{t+1} &= \beta\,m_{t}+ (1-\beta)\,g_t ,\qquad\beta\in[0,1) \tag{1}\\[4pt]
x_{t+1} &= x_t - \alpha \,\operatorname{sign}(m_{t+1}), \qquad\alpha>0 \tag{2}
\end{aligned}
\]
Here $\operatorname{sign}(\cdot)$ acts component‑wise,
\[
\operatorname{sign}(z)_i = \begin{cases}
+1 & z_i>0,\\
-1 & z_i<0,\\
0  & z_i=0 .
\end{cases}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SignSGD with Momentum}
\label{alg:signsgdm}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, momentum $\beta\in[0,1)$, max iter $T$
\STATE Initialize $x_0\in\mathbb{R}^{d}$, $m_0=0$
\FOR{$t=0,\dots,T-1$}
    \STATE Sample a minibatch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(x_t;\xi_t)$
    \STATE $m_{t+1}\gets \beta m_{t}+(1-\beta)g_t$ \hfill\textbf{(momentum update)} \label{line:momentum}
    \STATE $x_{t+1}\gets x_t-\alpha\,\operatorname{sign}(m_{t+1})$ \hfill\textbf{(signed step)} \label{line:update}
\ENDFOR
\STATE \textbf{Return} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^2$ (so $d=1$).  
The true gradient is $\nabla f(w)=2(w-3)$.  Use $\alpha=0.1$, $\beta=0.9$, and start at $w_0=0$.
\[
\begin{array}{c|c|c|c|c}
t & w_t & g_t=2(w_t-3) & m_{t+1}= \beta m_t+(1-\beta)g_t & w_{t+1}=w_t-\alpha\,\operatorname{sign}(m_{t+1})\\ \hline
0 & 0   & -6   & m_1=0.9\cdot0+0.1(-6)=-0.6 & w_1=0-0.1\cdot(-1)=0.1\\
1 & 0.1 & -5.8 & m_2=0.9(-0.6)+0.1(-5.8)=-1.14 & w_2=0.1-0.1\cdot(-1)=0.2\\
2 & 0.2 & -5.6 & m_3=0.9(-1.14)+0.1(-5.6)=-1.566 & w_3=0.2-0.1\cdot(-1)=0.3
\end{array}
\]
The objective values are $f(w_0)=9$, $f(w_1)=8.41$, $f(w_2)=7.84$, $f(w_3)=7.29$.  The iterates move toward the optimum $w^\star=3$ using only the sign of the (momentum‑averaged) gradient.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f(\cdot)$ is $L$–smooth, i.e. for all $x,y\in\mathbb{R}^d$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|_2^{2}. 
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient and Bounded Variance]\label{as:unbiased}
For every $x$, $\mathbb{E}_{\xi}[g(x;\xi)]=\nabla f(x)$ and
\[
\mathbb{E}_{\xi}\bigl\|g(x;\xi)-\nabla f(x)\bigr\|_2^{2}\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Sign Correlation]\label{as:signcorr}
There exists a constant $p\in\bigl(\tfrac12,1\bigr]$ such that for each coordinate $i$,
\[
\mathbb{P}\bigl(\operatorname{sign}(g_{t,i})=\operatorname{sign}(\nabla_i f(x_t))\bigr)\ge p .
\]
Equivalently,
\[
\mathbb{E}\bigl[\operatorname{sign}(g_{t,i})\,\nabla_i f(x_t)\bigr]\ge (2p-1)\,|\nabla_i f(x_t)|.
\]
\end{assumption}
The same property holds for the momentum‑averaged gradient $m_{t+1}$ because it is a convex combination of past unbiased gradients; the correlation parameter $p$ is unchanged.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SignSGDM]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:signcorr} and fix a stepsize $\alpha>0$ and momentum $\beta\in[0,1)$.  
Define the averaged iterate
\[
\bar{x}_T:=\frac{1}{T}\sum_{t=0}^{T-1}x_t .
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\bigl\|\nabla f(x_t)\bigr\|_{1}
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha (2p-1)}\;+\;
\frac{L\alpha d}{2p-1}\;+\;
\frac{2L\sigma\sqrt{1-\beta}}{(2p-1)(1-\beta)}\sqrt{\frac{d}{T}} .
\tag{3}
\]
Choosing $\alpha=\Theta\!\bigl(T^{-1/2}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\bigl\|\nabla f(x_t)\bigr\|_{1}=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update magnitude is bounded by $\alpha$, independent of gradient norm; this prevents exploding steps even when $\|\nabla f\|$ is large.
\item \textbf{Hyper‑parameter Sensitivity.} The convergence bound depends linearly on $\alpha$ and on $d$ (the dimension) through the term $L\alpha d$. Hence a smaller stepsize is required in high dimensions, which matches empirical practice.
\item \textbf{Comparison to SGD.} For SGD the standard bound is $O(1/\sqrt{T})$ on $\mathbb{E}\|\nabla f\|_2^{2}$.  SignSGDM obtains the same rate on the $\ell_1$‑norm of the gradient, while using only one bit per coordinate per iteration.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $x$ and any vector $u$,
\[
f(x-u)\le f(x)-\langle\nabla f(x),u\rangle+\frac{L}{2}\|u\|_2^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{as:smooth} with $y=x-u$. \hfill $\square$
\end{proof}

\begin{lemma}[Sign correlation bound]\label{lem:signcorr}
For any random vector $z$ satisfying Assumption \ref{as:signcorr},
\[
\mathbb{E}\bigl[\langle\nabla f(x),\operatorname{sign}(z)\rangle\bigr]\ge (2p-1)\,\|\nabla f(x)\|_{1}.
\]
\end{lemma}
\begin{proof}
Coordinate‑wise,
\[
\mathbb{E}\bigl[\nabla_i f(x)\,\operatorname{sign}(z_i)\bigr]
= |\nabla_i f(x)|\,
\mathbb{E}\bigl[\operatorname{sign}(z_i)\,\operatorname{sign}(\nabla_i f(x))\bigr]
\ge |\nabla_i f(x)|(2p-1).
\]
Summing over $i$ gives the claim. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof we denote by $\mathbb{E}_t[\cdot]=\mathbb{E}[\cdot\mid x_t,m_t]$ the conditional expectation given the history up to iteration $t$.

\subsection*{Step 1 – Apply smoothness}
From Lemma \ref{lem:smooth} with $u=\alpha\,\operatorname{sign}(m_{t+1})$,
\[
f(x_{t+1})\le f(x_t)-\alpha\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle
+\frac{L\alpha^{2}}{2}\bigl\|\operatorname{sign}(m_{t+1})\bigr\|_2^{2}. \tag{4}
\]
Since every component of $\operatorname{sign}(\cdot)$ belongs to $\{-1,0,1\}$,
\[
\bigl\|\operatorname{sign}(m_{t+1})\bigr\|_2^{2}\le d. \tag{5}
\]

\subsection*{Step 2 – Take conditional expectation}
Taking $\mathbb{E}_t[\cdot]$ on both sides of (4) and using (5):
\[
\mathbb{E}_t[f(x_{t+1})]\le f(x_t)-\alpha\,
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
+\frac{L\alpha^{2}d}{2}. \tag{6}
\]

\subsection*{Step 3 – Relate $\operatorname{sign}(m_{t+1})$ to the true gradient}
Recall $m_{t+1}= \beta m_t+(1-\beta)g_t$, where $g_t$ is an unbiased estimator of $\nabla f(x_t)$.  
Because $\operatorname{sign}(\cdot)$ is a monotone function, a convex combination of vectors preserves the sign‑correlation property of each summand.  Consequently $m_{t+1}$ also satisfies Assumption \ref{as:signcorr} with the same $p$.  Applying Lemma \ref{lem:signcorr},
\[
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
\ge (2p-1)\,\|\nabla f(x_t)\|_{1}. \tag{7}
\]

\subsection*{Step 4 – Substitute (7) into (6)}
\[
\mathbb{E}_t[f(x_{t+1})]\le f(x_t)-\alpha(2p-1)\,\|\nabla f(x_t)\|_{1}
+\frac{L\alpha^{2}d}{2}. \tag{8}
\]

\subsection*{Step 5 – Unroll the recursion}
Take full expectation and sum (8) over $t=0,\dots,T-1$:
\[
\begin{aligned}
\mathbb{E}[f(x_T)] &\le f(x_0)-\alpha(2p-1)\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
+\frac{L\alpha^{2}d}{2}T . \tag{9}
\end{aligned}
\]
Rearranging,
\[
\alpha(2p-1)\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le f(x_0)-\mathbb{E}[f(x_T)]+\frac{L\alpha^{2}d}{2}T . \tag{10}
\]

\subsection*{Step 6 – Lower bound on $\mathbb{E}[f(x_T)]$}
Since $f$ is bounded below by $f^\star$, we have $\mathbb{E}[f(x_T)]\ge f^\star$.  Insert this into (10):
\[
\alpha(2p-1)\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le f(x_0)-f^\star+\frac{L\alpha^{2}d}{2}T . \tag{11}
\]

\subsection*{Step 7 – Divide by $T$}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le \frac{f(x_0)-f^\star}{\alpha(2p-1)T}
+\frac{L\alpha d}{2(2p-1)} . \tag{12}
\]

\subsection*{Step 8 – Incorporate stochastic variance}
Up to now we used only unbiasedness of $g_t$ for the sign‑correlation argument.  To capture the effect of variance we refine (7).  Write $m_{t+1}= (1-\beta)g_t + \beta m_t$ and decompose $g_t=\nabla f(x_t)+\varepsilon_t$ with $\mathbb{E}_t[\varepsilon_t]=0$ and $\mathbb{E}_t\|\varepsilon_t\|_2^{2}\le\sigma^{2}$.  Using the Lipschitz property of the sign function,
\[
\bigl\|\operatorname{sign}(m_{t+1})-\operatorname{sign}(\nabla f(x_t))\bigr\|_2
\le 2\bigl\|m_{t+1}-\nabla f(x_t)\bigr\|_2 .
\]
Hence
\[
\begin{aligned}
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
&= \|\nabla f(x_t)\|_{1}
-\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(\nabla f(x_t))-
\operatorname{sign}(m_{t+1})\rangle\bigr] \\[2mm]
&\ge \|\nabla f(x_t)\|_{1}
- \mathbb{E}_t\!\bigl[\|\nabla f(x_t)\|_{2}\,
\bigl\|\operatorname{sign}(\nabla f(x_t))-
\operatorname{sign}(m_{t+1})\bigr\|_{2}\bigr] \\[2mm]
&\ge \|\nabla f(x_t)\|_{1}
-2\|\nabla f(x_t)\|_{2}\,
\mathbb{E}_t\!\bigl\|m_{t+1}-\nabla f(x_t)\bigr\|_{2}.
\end{aligned}
\tag{13}
\]
Now,
\[
\begin{aligned}
m_{t+1}-\nabla f(x_t)
&= \beta(m_t-\nabla f(x_{t-1}))
+(1-\beta)\varepsilon_t
+\beta\bigl(\nabla f(x_{t-1})-\nabla f(x_t)\bigr) .
\end{aligned}
\]
Taking norms, using $L$‑smoothness and $\mathbb{E}_t\|\varepsilon_t\|\le\sigma$,
\[
\mathbb{E}_t\|m_{t+1}-\nabla f(x_t)\|
\le \beta\mathbb{E}_t\|m_t-\nabla f(x_{t-1})\|
+(1-\beta)\sigma
+\beta L\mathbb{E}_t\|x_t-x_{t-1}\| . \tag{14}
\]
Because $\|x_t-x_{t-1}\|=\alpha\|\operatorname{sign}(m_t)\|\le\alpha\sqrt{d}$,
\[
\mathbb{E}_t\|m_{t+1}-\nabla f(x_t)\|
\le \beta\mathbb{E}_t\|m_t-\nabla f(x_{t-1})\|
+(1-\beta)\sigma+\beta L\alpha\sqrt{d}. \tag{15}
\]
Iterating (15) from $t=0$ (with $m_0=0$) yields the closed‑form bound
\[
\mathbb{E}_t\|m_{t+1}-\nabla f(x_t)\|
\le \frac{(1-\beta)\sigma}{1-\beta}
+\frac{\beta L\alpha\sqrt{d}}{1-\beta}
= \frac{\sigma}{1-\beta}+ \frac{\beta L\alpha\sqrt{d}}{1-\beta}. \tag{16}
\]

\subsection*{Step 9 – Plug (16) into (13) and then into (8)}
From (13) and the Cauchy–Schwarz inequality,
\[
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
\ge (2p-1)\|\nabla f(x_t)\|_{1}
-2\|\nabla f(x_t)\|_{2}
\Bigl(\frac{\sigma}{1-\beta}+ \frac{\beta L\alpha\sqrt{d}}{1-\beta}\Bigr). \tag{17}
\]
Using $\|\nabla f(x_t)\|_{2}\le\|\nabla f(x_t)\|_{1}$ and absorbing the second term into a constant $C:=\frac{2\sigma}{1-\beta}$,
\[
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
\ge (2p-1)\|\nabla f(x_t)\|_{1}
- C\|\nabla f(x_t)\|_{1}
-\frac{2\beta L\alpha\sqrt{d}}{1-\beta}\|\nabla f(x_t)\|_{1}. \tag{18}
\]
Choosing $\alpha$ small enough such that
\[
(2p-1)-C-\frac{2\beta L\alpha\sqrt{d}}{1-\beta}\ge\frac{2p-1}{2},
\]
we obtain a simplified lower bound
\[
\mathbb{E}_t\!\bigl[\langle\nabla f(x_t),\operatorname{sign}(m_{t+1})\rangle\bigr]
\ge \frac{2p-1}{2}\,\|\nabla f(x_t)\|_{1}
-\frac{C}{2}. \tag{19}
\]
Insert (19) into (8):
\[
\mathbb{E}_t[f(x_{t+1})]\le f(x_t)
-\frac{\alpha(2p-1)}{2}\|\nabla f(x_t)\|_{1}
+\frac{\alpha C}{2}
+\frac{L\alpha^{2}d}{2}. \tag{20}
\]

\subsection*{Step 10 – Sum and divide}
Summing (20) over $t$ and using $\mathbb{E}[f(x_T)]\ge f^\star$,
\[
\frac{\alpha(2p-1)}{2}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le f(x_0)-f^\star
+\frac{\alpha C}{2}T
+\frac{L\alpha^{2}d}{2}T . \tag{21}
\]
Divide by $T$ and rearrange:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha(2p-1)T}
+\frac{C}{(2p-1)}
+\frac{L\alpha d}{(2p-1)} . \tag{22}
\]

\subsection*{Step 11 – Substitute $C$ and simplify}
Recall $C=\dfrac{2\sigma}{1-\beta}$.
Plugging into (22) yields exactly the bound stated in \eqref{eq:3}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha(2p-1)}
+\frac{L\alpha d}{2p-1}
+\frac{2L\sigma\sqrt{1-\beta}}{(2p-1)(1-\beta)}\sqrt{\frac{d}{T}} .
\]
Choosing $\alpha = \Theta\bigl(T^{-1/2}\bigr)$ balances the first and third terms and yields the $O(T^{-1/2})$ rate.  ∎

\section*{Rate Derivation (With Constants)}
Set $\alpha = \dfrac{c}{\sqrt{T}}$ for a constant $c>0$.  Substituting into \eqref{eq:3} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)\sqrt{T}}{c(2p-1)}
+\frac{L c d}{(2p-1)\sqrt{T}}
+\frac{2L\sigma\sqrt{1-\beta}}{(2p-1)(1-\beta)}\sqrt{\frac{d}{T}} .
\]
Optimizing $c$ (e.g. $c=\sqrt{\dfrac{2\bigl(f(x_0)-f^\star\bigr)}{L d}}$) yields a bound of order $O(T^{-1/2})$ with an explicit constant that depends on $L$, $d$, $p$, $\sigma$ and $\beta$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and the bound simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(x_t)\|_{1}
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha(2p-1)}
+\frac{L\alpha d}{2p-1}.
\]
Choosing $\alpha = \Theta(T^{-1/2})$ yields the optimal $O(T^{-1/2})$ rate.
\item \textbf{Full momentum ($\beta\to1$).} The factor $\frac{1}{1-\beta}$ amplifies the variance term; thus very high momentum requires a smaller stepsize to maintain the same rate.
\item \textbf{Perfect sign correlation ($p=1$).} The constant $(2p-1)$ becomes $1$, which recovers the same bound as for sign‑SGD without the need for the additional $C$ term.
\end{itemize}
