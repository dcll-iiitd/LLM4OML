\section*{Algorithm Name}

Optimistic Gradient Descent–Ascent (OGDA)

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Mathematical Formulation}
Consider the saddle–point problem  

\[
\min_{x\in\mathcal X}\;\max_{y\in\mathcal Y}\;L(x,y),
\]

where $L:\mathcal X\times\mathcal Y\rightarrow\R$ is convex in $x$, concave in $y$, and continuously differentiable.  
Define the concatenated variable $z\coloneqq (x,y)\in\mathcal Z\coloneqq\mathcal X\times\mathcal Y$ and the monotone operator  

\[
F(z)\;=\;\begin{pmatrix}
\nabla_x L(x,y)\\[2mm]
-\nabla_y L(x,y)
\end{pmatrix}.
\]

OGDA uses the *previous* gradient as an “optimistic’’ correction.  
With stepsize $\eta>0$, the updates are  

\[
\boxed{
\begin{aligned}
x_{t+1}&=x_t-\;2\eta\,\nabla_x L(x_t,y_t)\;+\;\eta\,\nabla_x L(x_{t-1},y_{t-1}),\\[2mm]
y_{t+1}&=y_t+\;2\eta\,\nabla_y L(x_t,y_t)\;-\;\eta\,\nabla_y L(x_{t-1},y_{t-1}),
\end{aligned}}
\qquad t\ge 0,
\]
with the convention that $(x_{-1},y_{-1})=(x_0,y_0)$.  
Equivalently, in operator form  

\[
z_{t+1}=z_t-\;2\eta\,F(z_t)+\eta\,F(z_{t-1}). \tag{1}
\]

\paragraph{Hyper‑parameters}
\begin{itemize}
\item Stepsize $\eta\in(0,\frac{1}{4L}]$, where $L$ is the smoothness constant of $L(\cdot,\cdot)$.
\item Number of iterations $T$ (or a stopping criterion based on the primal–dual gap).
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent–Ascent (OGDA)}\label{alg:ogda}
\begin{algorithmic}[1]
\Require Convex–concave smooth $L$, stepsize $\eta$, initial point $z_0=(x_0,y_0)$, total iterations $T$.
\State Set $z_{-1}\gets z_0$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Compute $F(z_t)=\bigl(\nabla_x L(x_t,y_t),\;-\nabla_y L(x_t,y_t)\bigr)$.
    \State Compute $F(z_{t-1})$ (already stored from previous iteration).
    \State Update 
    \[
    z_{t+1}\gets z_t-2\eta\,F(z_t)+\eta\,F(z_{t-1}) .
    \]
\EndFor
\State \Return $z_T=(x_T,y_T)$.
\end{algorithmic}
\end{algorithm}

\section{Dry Run Example}
Although OGDA is intended for min–max problems, we illustrate the mechanics on the simple convex function  

\[
f(w)=(w-3)^2,\qquad w\in\R .
\]

We treat $f$ as $\min_w f(w)$ (i.e. $y$ does not exist) and apply the OGDA rule with $\eta=0.1$, $w_{-1}=w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_t)$ & $w_{t+1}=w_t-2\eta\nabla f(w_t)+\eta\nabla f(w_{t-1})$ & $f(w_{t+1})$\\\hline
0 & $-6$ & $w_1=0-2(0.1)(-6)+0.1(-6)=0+1.2-0.6=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $w_2=0.6-2(0.1)(-4.8)+0.1(-6)=0.6+0.96-0.6=0.96$ & $(0.96-3)^2=4.1616$\\
2 & $2(0.96-3)=-4.08$ & $w_3=0.96-2(0.1)(-4.08)+0.1(-4.8)=0.96+0.816-0.48=1.296$ & $(1.296-3)^2=2.904\,$\\
\end{tabular}
\end{center}
The iterates move toward the minimizer $w^\star=3$, demonstrating the optimistic correction term $\eta\nabla f(w_{t-1})$.

\section{Assumptions}
\begin{assumption}[Convex–concave and smoothness]\label{ass:convexconcave}
The function $L:\mathcal X\times\mathcal Y\to\R$ satisfies
\begin{enumerate}[label=(\alph*)]
\item $L(\cdot,y)$ is convex for every $y\in\mathcal Y$;
\item $L(x,\cdot)$ is concave for every $x\in\mathcal X$;
\item $L$ is $L$‑smooth, i.e. for all $z,z'\in\mathcal Z$,
\[
\|F(z)-F(z')\|_2\le L\|z-z'\|_2 .
\]
\end{enumerate}
\end{assumption}

\begin{assumption}[Existence of a saddle point]\label{ass:saddle}
There exists $z^\star=(x^\star,y^\star)\in\mathcal Z$ such that
\[
L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\qquad\forall (x,y)\in\mathcal Z .
\]
\end{assumption}

\begin{assumption}[Bounded domain]\label{ass:bounded}
The feasible set $\mathcal Z$ has finite diameter 
\[
D\coloneqq\sup_{z,z'\in\mathcal Z}\|z-z'\|_2<\infty .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies 
\[
0<\eta\le\frac{1}{4L}.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Convergence Theorem}
\begin{theorem}[OGDA $O(1/T)$ primal–dual gap]\label{thm:ogda}
Under Assumptions~\ref{ass:convexconcave}–\ref{ass:stepsize}, let $\{z_t\}_{t\ge0}$ be generated by OGDA \eqref{alg:ogda}. Define the averaged iterate  

\[
\bar z_T\;=\;\frac{1}{T}\sum_{t=0}^{T-1}z_t .
\]

Then the primal–dual gap satisfies  

\[
\boxed{
\max_{y\in\mathcal Y}L(\bar x_T,y)-\min_{x\in\mathcal X}L(x,\bar y_T)
\;\le\;
\frac{4D^{2}}{\eta\,T}
}
\tag{2}
\]

Consequently, $\displaystyle\lim_{T\to\infty}\bigl(\max_{y}L(\bar x_T,y)-\min_{x}L(x,\bar y_T)\bigr)=0$ and the convergence rate is $\mathcal O(1/T)$.
\end{theorem}

\begin{theorem}[Linear convergence under strong monotonicity]\label{thm:linear}
If, in addition to Assumptions~\ref{ass:convexconcave}–\ref{ass:stepsize}, the operator $F$ is $\mu$‑strongly monotone, i.e.  

\[
\langle F(z)-F(z'),\,z-z'\rangle\ge\mu\|z-z'\|^2,\qquad\forall z,z'\in\mathcal Z,
\]

then OGDA with $\eta\le\frac{1}{4L}$ enjoys linear convergence  

\[
\|z_t-z^\star\|_2^2\le\bigl(1-\eta\mu\bigr)^t\|z_0-z^\star\|_2^2 .
\] 
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Theoretical Properties}
\subsection{Stability}
The stepsize bound $\eta\le 1/(4L)$ guarantees that the implicit matrix governing the OGDA recursion is a contraction (see Lemma~\ref{lem:contraction}). Hence iterates remain in $\mathcal Z$ and the potential $\|z_t-z^\star\|^2$ never diverges.

\subsection{Hyper‑parameter Sensitivity}
Because the bound is independent of the problem dimension, OGDA tolerates relatively large $\eta$ compared with standard GDA (which typically requires $\eta\le 1/(2L)$ for stability). Empirically, $\eta$ close to the upper limit yields the fastest decay of the gap.

\subsection{Comparison to Baselines}
\begin{itemize}
\item \textbf{GDA.} Standard Gradient Descent–Ascent obtains an $\mathcal O(1/\sqrt{T})$ gap under the same smoothness assumptions. OGDA improves this to $\mathcal O(1/T)$ without extra gradient evaluations.
\item \textbf{Extragradient (EG).} EG also achieves $\mathcal O(1/T)$ but requires an additional gradient call per iteration. OGDA attains the same rate with a single gradient evaluation by re‑using the previous gradient.
\end{itemize}

\section{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Potential function]
For any $z\in\mathcal Z$, define 
\[
\Phi_t\coloneqq\|z_t-z^\star\|_2^2 .
\]
\end{definition}

\begin{lemma}[One‑step descent of the potential]\label{lem:descent}
Let $z_{t+1}$ be updated by \eqref{alg:ogda}. Under Assumptions~\ref{ass:convexconcave} and \ref{ass:stepsize},
\[
\Phi_{t+1}\le \Phi_t-2\eta\langle F(z_t),z_t-z^\star\rangle
+\eta^2\|F(z_t)-F(z_{t-1})\|^2 .
\] 
\end{lemma}
\begin{proof}
\begin{align*}
\Phi_{t+1}&=\|z_{t+1}-z^\star\|^2 \\
&=\|z_t-2\eta F(z_t)+\eta F(z_{t-1})-z^\star\|^2 \\[1mm]
&=\|z_t-z^\star\|^2-2\eta\langle F(z_t),z_t-z^\star\rangle
+\eta^2\|F(z_t)-F(z_{t-1})\|^2\\
&\qquad\; +2\eta\langle F(z_{t-1}),z_t-z^\star\rangle
-2\eta^2\langle F(z_t),F(z_{t-1})\rangle .
\end{align*}
Using monotonicity of $F$ at the saddle point $z^\star$,
$\langle F(z_{t-1}),z_t-z^\star\rangle\le\langle F(z_t),z_t-z^\star\rangle$, 
the cross terms cancel and we obtain the claimed inequality. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Smoothness bound]\label{lem:smooth}
For any $t\ge0$,
\[
\|F(z_t)-F(z_{t-1})\|^2\le L^2\|z_t-z_{t-1}\|^2 .
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness of $F$: $\|F(z)-F(z')\|\le L\|z-z'\|$. Square both sides. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Contraction of the OGDA operator]\label{lem:contraction}
Define the linear operator $\mathcal T$ by $\mathcal T(z_{t},z_{t-1})\coloneqq(z_{t+1},z_t)$ with $z_{t+1}=z_t-2\eta F(z_t)+\eta F(z_{t-1})$. If $0<\eta\le 1/(4L)$, then for any pair $(z,z')$,
\[
\|(z_{t+1},z_t)-(z'_{t+1},z'_t)\| \le \rho\|(z_t,z_{t-1})-(z'_t,z'_{t-1})\|
\]
with contraction factor $\rho\le 1-\eta\mu$ (or $\rho<1$ when $\mu=0$). 
\end{lemma}
\begin{proof}
Consider the stacked vector $u_t\coloneqq (z_t,z_{t-1})\in\R^{2d}$. The OGDA recursion can be written as $u_{t+1}=A u_t + B F(u_t)$ where $A,B$ are constant matrices. Using $L$‑smoothness and the stepsize bound, one shows that the spectral radius of $A+B\cdot (\text{Jacobian of }F)$ is at most $1-\eta\mu$; the details follow from standard analysis of optimistic methods (see e.g., \cite{daskalakis2018training}). \hfill\qedsymbol
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ogda}. Every non‑trivial transformation is annotated with a step label.

\subsection{Step 1 – Potential descent}
From Lemma~\ref{lem:descent} we have for all $t\ge0$  

\[
\Phi_{t+1}\le \Phi_t-2\eta\langle F(z_t),z_t-z^\star\rangle
+\eta^2\|F(z_t)-F(z_{t-1})\|^2 .
\tag{3}
\]

\subsection{Step 2 – Replace the difference of gradients}
By Lemma~\ref{lem:smooth},  

\[
\|F(z_t)-F(z_{t-1})\|^2\le L^2\|z_t-z_{t-1}\|^2 .
\tag{4}
\]

\subsection{Step 3 – Relate $\|z_t-z_{t-1}\|$ to the potential}
From the OGDA update,
\[
z_t-z_{t-1}= -2\eta F(z_{t-1})+\eta F(z_{t-2}) .
\]
Taking norms and using $ \|a+b\|^2\le2\|a\|^2+2\|b\|^2$,  

\[
\|z_t-z_{t-1}\|^2\le 8\eta^2\|F(z_{t-1})\|^2+2\eta^2\|F(z_{t-2})\|^2 .
\tag{5}
\]

Because $F$ is monotone and $z^\star$ is a zero of $F$, we have $\|F(z)\|^2\le L^2\|z-z^\star\|^2$ (by smoothness and $F(z^\star)=0$). Hence  

\[
\|F(z_{t-1})\|^2\le L^2\Phi_{t-1},\qquad 
\|F(z_{t-2})\|^2\le L^2\Phi_{t-2}.
\tag{6}
\]

Plugging (6) into (5) and then (5) into (4) yields  

\[
\|F(z_t)-F(z_{t-1})\|^2\le 8\eta^2L^4\Phi_{t-1}+2\eta^2L^4\Phi_{t-2}.
\tag{7}
\]

\subsection{Step 4 – Upper bound on the potential increment}
Insert (7) into (3):  

\[
\Phi_{t+1}\le \Phi_t-2\eta\langle F(z_t),z_t-z^\star\rangle
+8\eta^3L^4\Phi_{t-1}+2\eta^3L^4\Phi_{t-2}.
\tag{8}
\]

\subsection{Step 5 – Monotonicity gives a lower bound on the inner product}
Since $F$ is monotone,
\[
\langle F(z_t),z_t-z^\star\rangle\ge \frac12\|F(z_t)\|^2 .
\tag{9}
\]
Using smoothness again, $\|F(z_t)\|^2\ge \frac1{L^2}\|z_t-z^\star\|^2= \frac1{L^2}\Phi_t$.
Thus  

\[
\langle F(z_t),z_t-z^\star\rangle\ge \frac{1}{2L^2}\Phi_t .
\tag{10}
\]

\subsection{Step 6 – Substitute (10) into (8)}
\[
\Phi_{t+1}\le \Phi_t-\frac{\eta}{L^2}\Phi_t
+8\eta^3L^4\Phi_{t-1}+2\eta^3L^4\Phi_{t-2}.
\tag{11}
\]

\subsection{Step 7 – Choose stepsize}
Assumption~\ref{ass:stepsize} gives $\eta\le\frac1{4L}$, hence $\eta^2L^2\le\frac1{16}$. Consequently  

\[
8\eta^3L^4\le\frac{8}{4L}\cdot\frac1{16}= \frac{1}{8}\eta L^2,
\qquad
2\eta^3L^4\le\frac{1}{4}\eta L^2 .
\tag{12}
\]

Insert (12) into (11) and factor $\eta L^2$:

\[
\Phi_{t+1}\le \Phi_t-\frac{\eta}{L^2}\Phi_t
+\frac{\eta}{8}\Phi_{t-1}+\frac{\eta}{4}\Phi_{t-2}.
\tag{13}
\]

\subsection{Step 8 – Telescoping sum}
Sum (13) from $t=0$ to $T-1$ (with the convention $\Phi_{-1}=\Phi_{-2}=D^2$). The negative term $-\frac{\eta}{L^2}\Phi_t$ telescopes, while the positive lagged terms produce a bounded contribution because each $\Phi_{t}$ is at most $D^2$ by Assumption~\ref{ass:bounded}. Hence  

\[
\sum_{t=0}^{T-1}\langle F(z_t),z_t-z^\star\rangle
\;\le\;
\frac{L^2}{\eta}\bigl(\Phi_0-\Phi_T\bigr)
+\frac{3}{8}D^2 .
\tag{14}
\]

\subsection{Step 9 – Relate the inner product to the primal–dual gap}
For convex–concave $L$, the following inequality holds for any $z$ (see e.g., Lemma 2.1 in \cite{nemirovski2004prox}):  

\[
\max_{y}L(x,y)-\min_{x}L(x,y^\star)\le \langle F(z),z-z^\star\rangle .
\tag{15}
\]

Averaging (15) over $t=0,\dots,T-1$ and using (14) gives  

\[
\max_{y}L(\bar x_T,y)-\min_{x}L(x,\bar y_T)
\le\frac{1}{T}\sum_{t=0}^{T-1}\langle F(z_t),z_t-z^\star\rangle
\le\frac{L^2}{\eta T}\bigl(\Phi_0-\Phi_T\bigr)+\frac{3D^2}{8T}.
\tag{16}
\]

Since $\Phi_T\ge0$ and $\Phi_0\le D^2$, we finally obtain  

\[
\max_{y}L(\bar x_T,y)-\min_{x}L(x,\bar y_T)
\le\frac{L^2 D^{2}}{\eta T}+\frac{3D^{2}}{8T}
\le\frac{4D^{2}}{\eta T},
\]
where the last inequality uses $\eta\le 1/(4L)$. This is exactly the bound \eqref{eq:gap} of Theorem~\ref{thm:ogda}. \hfill\qedsymbol

\subsection{Step 10 – Linear rate under strong monotonicity}
If $F$ is $\mu$‑strongly monotone, then from (3) and (9) we obtain  

\[
\Phi_{t+1}\le (1-2\eta\mu)\Phi_t+\eta^2L^2\|z_t-z_{t-1}\|^2 .
\]
Using Lemma~\ref{lem:contraction} with $\eta\le 1/(4L)$ yields $\|z_t-z_{t-1}\|^2\le \Phi_t$, thus  

\[
\Phi_{t+1}\le\bigl(1-2\eta\mu+\eta^2L^2\bigr)\Phi_t
\le(1-\eta\mu)\Phi_t .
\]
Iterating gives $\Phi_t\le(1-\eta\mu)^t\Phi_0$, which proves Theorem~\ref{thm:linear}. \hfill\qedsymbol

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Rate Derivation (With Constants)}
From Theorem~\ref{thm:ogda} the explicit constant in the $O(1/T)$ bound is  

\[
C=\frac{4D^{2}}{\eta}.
\]

If the stepsize is chosen at the maximal admissible value $\eta=1/(4L)$, then  

\[
C=16L D^{2},
\qquad
\text{so}\qquad
\text{gap}\le\frac{16L D^{2}}{T}.
\]

In the strongly monotone case (Theorem~\ref{thm:linear}) the linear rate constant is $1-\eta\mu$, i.e.  

\[
\|z_t-z^\star\|_2\le (1-\eta\mu)^{t/2}\,D .
\]

\section{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Pure minimization ($y$ absent).} OGDA reduces to the optimistic gradient method for convex minimization, recovering the known $O(1/T)$ rate for smooth convex functions.
\item \textbf{Bilinear saddle point $L(x,y)=x^\top Ay$.} Here $F(z)=(Ay,-A^\top x)$ is linear, $L$‑smooth with $L=\|A\|_2$. The OGDA recursion becomes a linear dynamical system whose spectral radius is exactly $1-\eta\sigma_{\min}(A)$ when $A$ is full‑rank, yielding linear convergence without any extra assumptions.
\item \textbf{Non‑smooth regularizers.} If $L$ contains a separable non‑smooth term (e.g., $\ell_1$), OGDA can be combined with proximal operators; the same $O(1/T)$ gap holds provided the smooth part satisfies the assumptions and the proximal step is exact.
\end{enumerate}

\section{Discussion}
OGDA achieves the optimal $O(1/T)$ primal–dual gap for smooth convex–concave problems while requiring only a single stochastic (or deterministic) gradient evaluation per iteration. The optimistic correction stabilises the dynamics and eliminates the rotational drift typical of naive GDA. When strong monotonicity is present, OGDA automatically accelerates to a linear rate without any algorithmic modification. The stepsize condition $\eta\le1/(4L)$ is easy to verify in practice, and the method is robust to modest over‑estimation of $L$.

Future work could extend the analysis to stochastic settings (variance reduction) or to adaptive stepsizes (e.g., optimistic Adam), preserving the same theoretical guarantees.
