\section*{Algorithm Name}
\textbf{Stochastic Gradient Descent with Warm Restarts (SGDR)} – a variant of SGD in which the learning‑rate schedule is periodically reset to a maximal value $\eta_{\max }$ and then decays within each cycle of length $M$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\begin{itemize}
    \item Parameter vector: $w_t\in\R^d$ at iteration $t$.
    \item Stochastic gradient: $g_t\triangleq\nabla f(w_t;\xi_t)$ where $\xi_t$ is a random data sample (or mini‑batch) drawn i.i.d. from a distribution $\mathcal{D}$.
    \item Learning‑rate schedule (linear decay with warm restart):
    \[
        \eta_t \;=\;
        \eta_{\max}\!\left(1-\frac{t\bmod M}{M}\right),
        \qquad t=0,1,2,\dots
    \tag{1}
    \]
    i.e. at the beginning of each cycle ($t\bmod M=0$) the step size is reset to $\eta_{\max}$ and then decreases linearly to $0$ at the end of the cycle.
    \item Update rule:
    \[
        w_{t+1}=w_t-\eta_t\,g_t .
    \tag{2}
    \]
\end{itemize}
Hyper‑parameters:
\begin{enumerate}[label=\arabic*)]
    \item $\eta_{\max}>0$ – maximal learning rate.
    \item $M\in\mathbb{N}$ – length of a restart cycle (number of iterations between two resets).
    \item Optionally a minimal learning rate $\eta_{\min}\ge0$ can be added:
    $\eta_t=\eta_{\min}+(\eta_{\max}-\eta_{\min})\!\bigl(1-\frac{t\bmod M}{M}\bigr)$.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\State \textbf{Input:} $\eta_{\max}>0$, cycle length $M$, initial point $w_0$,
        number of iterations $T$.
\For{$t=0$ {\bf to} $T-1$}
    \State Sample a mini‑batch $\xi_t\sim\mathcal{D}$.
    \State Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State Compute learning rate 
    \[
        \eta_t\gets\eta_{\max}\Bigl(1-\frac{t\bmod M}{M}\Bigr).
    \] 
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or e.g. the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^2$ (so $\nabla f(w)=2(w-3)$).  
Take $w_0=0$, $\eta_{\max}=0.1$, $M=3$, and assume the stochastic gradient is exact (i.e. $g_t=\nabla f(w_t)$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $\eta_t$ & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & $0$   & $-6$  & $0.1\bigl(1-0/3\bigr)=0.1$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $-4.8$& $0.1\bigl(1-1/3\bigr)=0.0667$ & $0.6-0.0667(-4.8)=1.0$ \\
2 & $1.0$ & $-4.0$& $0.1\bigl(1-2/3\bigr)=0.0333$ & $1.0-0.0333(-4.0)=1.1333$ \\
\end{tabular}
\end{center}
Objective values:
\[
f(w_0)=9,\; f(w_1)=5.76,\; f(w_2)=4.00,\; f(w_3)=3.56.
\]
The schedule resets after $t=3$, where $\eta_3$ would again be $0.1$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
The objective $f:\R^d\to\R$ is $L$‑smooth:
\[
    \forall u,v\in\R^d,\qquad 
    f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, $\E[g_t\,|\,w_t]=\nabla f(w_t)$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
    \E\bigl[\|g_t-\nabla f(w_t)\|^2\;\big|\;w_t\bigr]\le\sigma^2,
    \qquad\forall t .
\]
\end{assumption}
\begin{assumption}[Bounded Second Moment]\label{ass:bound}
There is $G>0$ with $\E[\|g_t\|^2]\le G^2$ for all $t$.
\end{assumption}
\begin{assumption}[Learning‑rate schedule]\label{ass:schedule}
The schedule (1) satisfies
\[
    0\le\eta_t\le\eta_{\max},\qquad 
    \sum_{t=0}^{T-1}\eta_t\ge \frac{\eta_{\max}}{2}T,
    \qquad 
    \sum_{t=0}^{T-1}\eta_t^{2}\le\frac{\eta_{\max}^{2}}{3}M\,\Big\lceil\frac{T}{M}\Big\rceil .
\tag{3}
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity guarantee for SGDR]\label{thm:sgdr}
Let Assumptions \ref{ass:smooth}–\ref{ass:schedule} hold and run SGDR for $T$ iterations.
Define the (uniform) average iterate $\bar w_T\triangleq\frac1T\sum_{t=0}^{T-1}w_t$.
Then
\[
\frac1T\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T}
\;+\;
\frac{L\sigma^{2}\eta_{\max}M}{3T},
\tag{4}
\]
where $f^{\star}= \inf_{w}f(w)$.
Consequently,
\[
\min_{0\le t<T}\E\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;=\;
O\!\left(\frac{1}{T}\right) .
\]
\end{theorem}

\begin{theorem}[Special case: deterministic gradients]\label{thm:det}
If $\sigma^{2}=0$ (exact gradients), the bound (4) reduces to
\[
\frac1T\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^{2}
\;\le\;
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T},
\]
i.e. SGDR attains the same $O(1/T)$ rate as classical gradient descent with a constant stepsize $\eta_{\max}/2$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Stability.} The linear decay guarantees $\eta_t\le\eta_{\max}$, preventing exploding updates. The reset prevents $\eta_t$ from becoming arbitrarily small, which would stall progress.
    \item \textbf{Sensitivity to $M$.} Larger $M$ yields longer decay phases, increasing $\sum_{t}\eta_t^{2}$ relative to $\sum_{t}\eta_t$ and thus slightly worsening the variance term in (4). Conversely, very small $M$ leads to frequent restarts, which may improve exploration in non‑convex landscapes.
    \item \textbf{Comparison to baseline SGD.} For a constant stepsize $\eta$, the standard SGD bound contains $\frac{L\sigma^{2}\eta}{2}$ as the variance term. SGDR replaces $\eta$ by the average $\frac{1}{T}\sum\eta_t\approx\eta_{\max}/2$, while the additional factor $\frac{M}{T}$ in (4) is negligible for large $T$, yielding essentially the same asymptotic rate but with an implicit learning‑rate annealing that often improves practical performance.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that the warm‑restart schedule does not deteriorate the classical $O(1/T)$ convergence guarantee for finding an \emph{approximate stationary point} in non‑convex smooth optimization. The proof hinges only on the two elementary sums (3) that any schedule satisfying a linear decay and periodic reset enjoys. Hence the result extends to cosine‑annealing or other monotone decays within each cycle, provided the same order of magnitude for the sums holds.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $w$ and any vector $d$,
\[
    f(w-d)\le f(w)-\langle\nabla f(w),d\rangle+\frac{L}{2}\|d\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:smooth} with $u=w$, $v=w-d$.
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $g$ with $\E[g]=\mu$,
\[
    \E\|g\|^{2}= \|\mu\|^{2}+ \E\|g-\mu\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Expand $\|g\|^{2}=\|g-\mu+\mu\|^{2}$ and take expectation.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:sgdr}.  All expectations are taken w.r.t.\ the randomness up to iteration $t$ unless otherwise noted.

\paragraph{Step 1 (Smoothness applied to the update).}
Using Lemma \ref{lem:smooth} with $d=\eta_t g_t$,
\[
f(w_{t+1})
\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle
     +\frac{L\eta_t^{2}}{2}\|g_t\|^{2}.
\tag{7}
\]

\paragraph{Step 2 (Take conditional expectation).}
Condition on $w_t$ and employ Assumption \ref{ass:unbiased}:
\[
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
     +\frac{L\eta_t^{2}}{2}\E\!\bigl[\|g_t\|^{2}\mid w_t\bigr].
\tag{8}
\]

\paragraph{Step 3 (Bound the second‑moment of $g_t$).}
From Lemma \ref{lem:var} and Assumptions \ref{ass:unbiased}, \ref{ass:variance},
\[
\E\!\bigl[\|g_t\|^{2}\mid w_t\bigr]
= \|\nabla f(w_t)\|^{2}+ \E\!\bigl[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\bigr]
\le \|\nabla f(w_t)\|^{2}+ \sigma^{2}.
\tag{9}
\]

\paragraph{Step 4 (Insert (9) into (8)).}
\[
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
   +\frac{L\eta_t^{2}}{2}\bigl(\|\nabla f(w_t)\|^{2}+\sigma^{2}\bigr).
\tag{10}
\]

\paragraph{Step 5 (Rearrange).}
Bring the gradient term to the left:
\[
\bigl(\eta_t-\tfrac{L}{2}\eta_t^{2}\bigr)\|\nabla f(w_t)\|^{2}
\le f(w_t)-\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
      +\tfrac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{11}
\]

\paragraph{Step 6 (Use the bound $\eta_t\le\eta_{\max}\le\frac{2}{L}$).}
Choose $\eta_{\max}\le\frac{2}{L}$ (this can always be enforced by scaling). Then
\[
\eta_t-\frac{L}{2}\eta_t^{2}\ge\frac{\eta_t}{2}.
\tag{12}
\]
Plugging (12) into (11) yields
\[
\frac{\eta_t}{2}\,\|\nabla f(w_t)\|^{2}
\le f(w_t)-\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
      +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{13}
\]

\paragraph{Step 7 (Take full expectation).}
Denote $\Delta_t\triangleq\E[f(w_t)]$. Taking expectation of (13) gives
\[
\frac{\eta_t}{2}\,\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le \Delta_t-\Delta_{t+1}
      +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{14}
\]

\paragraph{Step 8 (Sum over $t=0,\dots,T-1$).}
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\,\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le \Delta_0-\Delta_T
      +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{15}
\]

\paragraph{Step 9 (Lower bound $\Delta_0-\Delta_T$).}
Since $f\ge f^{\star}$,
\[
\Delta_0-\Delta_T\le f(w_0)-f^{\star}.
\tag{16}
\]

\paragraph{Step 10 (Insert (16) and use the schedule sums (3)).}
From (15)–(16):
\[
\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le 2\bigl(f(w_0)-f^{\star}\bigr)
     +L\sigma^{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{17}
\]

\paragraph{Step 11 (Divide by $\sum_{t=0}^{T-1}\eta_t$).}
Using the lower bound from (3),
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_t}
   +L\sigma^{2}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{18}
\]

\paragraph{Step 12 (Insert the explicit sums from (3)).}
\[
\sum_{t=0}^{T-1}\eta_t\;\ge\;\frac{\eta_{\max}}{2}T,
\qquad
\sum_{t=0}^{T-1}\eta_t^{2}\;\le\;\frac{\eta_{\max}^{2}}{3}M\Big\lceil\frac{T}{M}\Big\rceil
\;\le\;\frac{\eta_{\max}^{2}}{3}M\Big(\frac{T}{M}+1\Big)
\le\frac{2\eta_{\max}^{2}}{3}M\,\frac{T}{M}
= \frac{2\eta_{\max}^{2}}{3}T .
\tag{19}
\]
Insert (19) into (18):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T}
   +\frac{L\sigma^{2}\eta_{\max}M}{3T}.
\tag{20}
\]
Equation (20) is exactly the bound (4) of Theorem \ref{thm:sgdr}. \hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (20) we read off the two contributions:
\begin{itemize}
    \item \textbf{Deterministic part:} $\displaystyle \frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T}=O\!\left(\frac{1}{T}\right)$.
    \item \textbf{Stochastic variance part:} $\displaystyle \frac{L\sigma^{2}\eta_{\max}M}{3T}=O\!\left(\frac{1}{T}\right)$.
\end{itemize}
Thus the overall rate for the average squared gradient norm is $O(1/T)$.  In the common regime where $\eta_{\max}=c/\sqrt{T}$ (to balance the two terms) one recovers the classical $O(1/\sqrt{T})$ rate for SGD; the theorem, however, holds for any fixed $\eta_{\max}$ satisfying $\eta_{\max}\le 2/L$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
    \item \textbf{Exact gradients ($\sigma^{2}=0$).} Equation (20) reduces to the deterministic bound stated in Theorem \ref{thm:det}.
    \item \textbf{Cosine annealing schedule.} If one replaces the linear decay in (1) by the cosine schedule
    \[
        \eta_t=\eta_{\min}+\frac12(\eta_{\max}-\eta_{\min})
               \bigl(1+\cos(\pi (t\bmod M)/M)\bigr),
    \]
    the same order of magnitude for $\sum\eta_t$ and $\sum\eta_t^{2}$ holds, and the proof proceeds unchanged.
    \item \textbf{Mini‑batch size $b$.} Increasing $b$ reduces the variance bound to $\sigma^{2}/b$, improving the second term in (4) proportionally.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Conclusion}
We have presented a complete mathematical description of SGDR, illustrated its execution on a toy problem, and derived—step by step, with every inequality explicitly justified—a convergence guarantee for non‑convex smooth objectives.  The analysis confirms that the warm‑restart schedule preserves the $O(1/T)$ decay of the expected squared gradient norm, while providing practical benefits such as automatic learning‑rate annealing and periodic exploration resets.
