\section*{Algorithm Name}
\textbf{Mirror Descent with Bregman Proximal Mapping}  
(the update is sometimes called the \emph{Bregman gradient step}).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{n}\rightarrow\mathbb{R}$ be a convex differentiable objective and let
$h:\mathbb{R}^{n}\rightarrow\mathbb{R}$ be a \emph{prox‑function} that is
$\alpha$–strongly convex with respect to a norm $\|\cdot\|$.
The associated Bregman divergence is  

\[
\DD{x}{y}\;=\;h(x)-h(y)-\ip{\grad h(y)}{x-y},\qquad x,y\in\mathbb{R}^{n}.
\]

Given a stepsize sequence $\{\eta_k\}_{k\ge 0}$, the Mirror Descent iteration is  

\[
\boxed{
x_{k+1}\;=\;\argmin_{x\in\mathbb{R}^{n}}
\Big\{\,\ip{\grad f(x_k)}{x}\;+\;\frac{1}{\eta_k}\,\DD{x}{x_k}\Big\}
}\tag{MD}
\]

The optimality condition of the inner minimisation is  

\[
\grad f(x_k)\;+\;\frac{1}{\eta_k}\big(\grad h(x_{k+1})-\grad h(x_k)\big)=0.
\tag{1}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent (Bregman Proximal Mapping)}\label{alg:MD}
\begin{algorithmic}[1]
\Require Convex $f$, prox‑function $h$, stepsize schedule $\{\eta_k\}$,
initial point $x_0\in\operatorname{dom}h$.
\For{$k=0,1,2,\dots$}
    \State Compute gradient $g_k\gets\grad f(x_k)$.
    \State Solve the proximal subproblem
    \[
        x_{k+1}\gets\argmin_{x}\Big\{\,\ip{g_k}{x}
               +\frac{1}{\eta_k}\big(h(x)-h(x_k)-\ip{\grad h(x_k)}{x-x_k}\big)\Big\}.
    \]
\EndFor
\State \textbf{output} $x_T$ (or the average $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimise \(f(w)=(w-3)^2\) on the real line.  
Choose the Euclidean prox‑function \(h(w)=\tfrac12 w^2\) (so \(\DD{u}{v}= \tfrac12 (u-v)^2\))
and a constant stepsize \(\eta_k=\eta=0.1\).  
Initial point \(w_0=0\).

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $k$ & $\grad f(w_k)=2(w_k-3)$ & $w_{k+1}$ (solution of (MD)) & $f(w_{k+1})$\\ \hline
0 & $-6$ &
\(
w_{1}= \argmin_w\{-6w+\tfrac1{0.1}\tfrac12(w-0)^2\}=0-0.1(-6)=0.6
\) & $(0.6-3)^2=5.76$\\[4pt]
1 & $2(0.6-3)=-4.8$ &
\(
w_{2}=0.6-0.1(-4.8)=1.08
\) & $(1.08-3)^2=3.6864$\\[4pt]
2 & $2(1.08-3)=-3.84$ &
\(
w_{3}=1.08-0.1(-3.84)=1.464
\) & $(1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}

divergence reduces to a squared norm.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]\label{ass:f}
$f$ is convex on $\mathbb{R}^{n}$ and $L$–smooth with respect to the Bregman
divergence of $h$, i.e. for all $x,y$,
\[
f(x)\le f(y)+\ip{\grad f(y)}{x-y}+\frac{L}{2}\,\DD{x}{y}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Strong Convexity of $h$]\label{ass:h}
The prox‑function $h$ is $\alpha$–strongly convex w.r.t. $\|\cdot\|$:
\[
\DD{x}{y}\;\ge\;\frac{\alpha}{2}\,\|x-y\|^{2},
\qquad\forall x,y\in\mathbb{R}^{n}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:eta}
The stepsizes satisfy \(0<\eta_k\le \frac{\alpha}{L}\) for all $k$.
\tag{A3}
\end{assumption}

\begin{assumption}[Existence of a minimiser]\label{ass:opt}
There exists at least one optimal point $x^{\star}\in\arg\min f$ and
$x^{\star}\in\operatorname{dom}h$.
\tag{A4}
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) convergence of Mirror Descent]\label{thm:rate}
Under Assumptions \ref{ass:f}–\ref{ass:opt} and a constant stepsize
\(\eta_k\equiv\eta\le\alpha/L\), the iterates produced by \eqref{MD} satisfy  

\[
f(x_{k})-f(x^{\star})
\;\le\;
\frac{\DD{x^{\star}}{x_{0}}}{\eta\,k},
\qquad\forall k\ge 1.
\tag{2}
\]

Equivalently, after $T$ iterations the averaged iterate
\(\bar x_T:=\frac1T\sum_{k=0}^{T-1}x_k\) obeys  

\[
f(\bar x_T)-f(x^{\star})\le\frac{\DD{x^{\star}}{x_{0}}}{\eta\,T}.
\tag{3}
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (2) depends only on the Bregman distance
between the start point and any optimal solution; no explicit dependence on the
dimension appears.
\item \textbf{Stepsize Sensitivity.} The admissible range
\(\eta\le\alpha/L\) guarantees that the descent inequality (derived below)
holds. Choosing $\eta$ smaller merely scales the constant in (2) but does not
affect the $1/k$ rate.
\item \textbf{Comparison to Gradient Descent.}
If $h(w)=\tfrac12\|w\|_2^{2}$, then $\DD{x}{y}=\tfrac12\|x-y\|_2^{2}$ and
Mirror Descent reduces to classical gradient descent with step $\eta$,
recovering the well‑known $O(1/k)$ rate for convex $L$‑smooth functions.
\item \textbf{Adaptivity.}
Because the divergence can be chosen to reflect geometry of the feasible set,
the algorithm can enjoy faster practical convergence on problems with
non‑Euclidean structure (e.g. simplex constraints using the entropy prox‑function).
\end{itemize}

\section*{Discussion}
The theorem shows that Mirror Descent inherits the optimal sublinear rate for
deterministic convex optimisation while allowing the practitioner to encode
problem geometry via $h$.  The proof is elementary yet relies crucially on the
three‑point identity of Bregman divergences and the smoothness inequality (A1).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:3pt}
For any $x,y,z$,
\[
\DD{x}{z}-\DD{x}{y}-\DD{y}{z}
   =\ip{\grad h(y)-\grad h(z)}{x-y}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Expand each divergence using the definition of $D_h$ and cancel terms.
\end{proof}

\begin{lemma}[Descent Lemma w.r.t. Bregman divergence]\label{lem:descent}
If $f$ satisfies (A1), then for any $x,y$,
\[
f(x)\le f(y)+\ip{\grad f(y)}{x-y}+\frac{L}{2}\DD{x}{y}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Directly the statement of Assumption \ref{ass:f}.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate} for a constant stepsize
$\eta_k\equiv\eta\le\alpha/L$.

\noindent\textbf{[Step 1]}  Optimality condition of the subproblem \eqref{MD}:
\[
\grad f(x_k)+\frac1\eta\big(\grad h(x_{k+1})-\grad h(x_k)\big)=0.
\tag{4}
\]

\noindent\textbf{[Step 2]}  Rearrange (4) and take inner product with $x_{k+1}-x^{\star}$:
\[
\ip{\grad f(x_k)}{x_{k+1}-x^{\star}}
   =-\frac1\eta\ip{\grad h(x_{k+1})-\grad h(x_k)}{x_{k+1}-x^{\star}}.
\tag{5}
\]

\noindent\textbf{[Step 3]}  Apply Lemma \ref{lem:3pt} with
$(x,y,z)=(x^{\star},x_{k+1},x_k)$ to the right‑hand side of (5):
\[
\ip{\grad h(x_{k+1})-\grad h(x_k)}{x_{k+1}-x^{\star}}
   =\DD{x^{\star}}{x_k}-\DD{x^{\star}}{x_{k+1}}-\DD{x_{k+1}}{x_k}.
\tag{6}
\]

\noindent\textbf{[Step 4]}  Substitute (6) into (5) and multiply by $\eta$:
\[
\eta\ip{\grad f(x_k)}{x_{k+1}-x^{\star}}
   =-\big(\DD{x^{\star}}{x_k}-\DD{x^{\star}}{x_{k+1}}-\DD{x_{k+1}}{x_k}\big).
\tag{7}
\]

\noindent\textbf{[Step 5]}  By convexity of $f$,
\[
f(x_{k})-f(x^{\star})\le\ip{\grad f(x_k)}{x_{k}-x^{\star}}.
\tag{8}
\]

\noindent\textbf{[Step 6]}  Add and subtract $\ip{\grad f(x_k)}{x_{k+1}-x^{\star}}$ to the right‑hand side of (8):
\[
f(x_k)-f(x^{\star})
\le\ip{\grad f(x_k)}{x_{k+1}-x^{\star}}
   +\ip{\grad f(x_k)}{x_{k}-x_{k+1}}.
\tag{9}
\]

\noindent\textbf{[Step 7]}  Use the smoothness Lemma \ref{lem:descent} with
$(x,y)=(x_{k+1},x_k)$:
\[
f(x_{k+1})\le f(x_k)+\ip{\grad f(x_k)}{x_{k+1}-x_k}
               +\frac{L}{2}\DD{x_{k+1}}{x_k}.
\tag{10}
\]
Rearrange (10) to obtain
\[
\ip{\grad f(x_k)}{x_{k}-x_{k+1}}
   \le f(x_k)-f(x_{k+1})+\frac{L}{2}\DD{x_{k+1}}{x_k}.
\tag{11}
\]

\noindent\textbf{[Step 8]}  Plug (11) into (9) and then use (7):
\[
\begin{aligned}
f(x_k)-f(x^{\star})
&\le \ip{\grad f(x_k)}{x_{k+1}-x^{\star}}
      +f(x_k)-f(x_{k+1})+\frac{L}{2}\DD{x_{k+1}}{x_k}\\[2mm]
&= f(x_k)-f(x_{k+1})
   +\frac{L}{2}\DD{x_{k+1}}{x_k}
   +\frac1\eta\!\left(\DD{x^{\star}}{x_k}-\DD{x^{\star}}{x_{k+1}}
                     -\DD{x_{k+1}}{x_k}\right) \quad\text{by (7)}\\[2mm]
&= f(x_k)-f(x_{k+1})
   +\Big(\frac{L}{2}-\frac1\eta\Big)\DD{x_{k+1}}{x_k}
   +\frac1\eta\big(\DD{x^{\star}}{x_k}-\DD{x^{\star}}{x_{k+1}}\big).
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 9]}  By the stepsize condition $\eta\le\alpha/L\le 1/L$,
the coefficient $\frac{L}{2}-\frac1\eta\le0$.  Hence we can drop the non‑positive
term to obtain the fundamental descent inequality
\[
f(x_k)-f(x^{\star})\le
\frac1\eta\big(\DD{x^{\star}}{x_k}-\DD{x^{\star}}{x_{k+1}}\big).
\tag{13}
\]

\noindent\textbf{[Step 10]}  Sum (13) for $k=0,\dots,T-1$:
\[
\sum_{k=0}^{T-1}\big(f(x_k)-f(x^{\star})\big)
\le\frac1\eta\big(\DD{x^{\star}}{x_0}-\DD{x^{\star}}{x_T}\big)
\le\frac{\DD{x^{\star}}{x_0}}{\eta}.
\tag{14}
\]

\noindent\textbf{[Step 11]}  Since each term on the left of (14) is non‑negative,
the smallest term is bounded by the average:
\[
\min_{0\le k<T}\big(f(x_k)-f(x^{\star})\big)
\le\frac1T\sum_{k=0}^{T-1}\big(f(x_k)-f(x^{\star})\big)
\le\frac{\DD{x^{\star}}{x_0}}{\eta\,T}.
\tag{15}
\]
Choosing the last iterate $x_T$ (or the average $\bar x_T$) yields exactly the
bound (2) and (3) stated in Theorem \ref{thm:rate}.
\hfill\qedsymbol

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (13) we have for any $k\ge1$,
\[
f(x_k)-f(x^{\star})\le\frac{\DD{x^{\star}}{x_0}}{\eta\,k},
\]
because the telescoping sum in (14) gives
\(\sum_{i=0}^{k-1}(f(x_i)-f(x^{\star}))\le\frac{\DD{x^{\star}}{x_0}}{\eta}\).
Thus the explicit constant in the $O(1/k)$ rate is $\DD{x^{\star}}{x_0}/\eta$.
If $h$ is $\alpha$‑strongly convex, then
\(\DD{x^{\star}}{x_0}\ge \frac{\alpha}{2}\|x^{\star}-x_0\|^{2}\), providing a
Euclidean‑style constant.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean setting.} $h(w)=\tfrac12\|w\|_2^{2}$,
$\DD{x}{y}= \tfrac12\|x-y\|_2^{2}$, $\alpha=1$.
The algorithm reduces to classical gradient descent and the bound
\(f(x_k)-f(x^{\star})\le\frac{\|x_0-x^{\star}\|_2^{2}}{2\eta k}\) matches the
standard result with $\eta\le1/L$.
\item \textbf{Entropy prox‑function on the simplex.}
Take $h(p)=\sum_{i}p_i\log p_i$, $\DD{x}{y}=\sum_i x_i\log(x_i/y_i)$.
Mirror Descent becomes the exponentiated gradient method; the same
$O(1/k)$ guarantee holds with the Bregman distance being the Kullback–Leibler
divergence.
\item \textbf{Variable stepsizes.}
If $\eta_k=\frac{2}{L(k+1)}$, the same telescoping argument yields
\(f(x_k)-f(x^{\star})\le\frac{2L\,\DD{x^{\star}}{x_0}}{k(k+1)}\),
i.e. an $O(1/k^{2})$ rate for strongly convex $f$ (not covered here).
\end{itemize}
