\section*{Trust Region Gradient Method (TRGM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^n\to\mathbb{R}$ be a convex, $L$‑smooth function, i.e.  
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^n .
\tag{1}
\]

At iteration $k$ we build the (quadratic) model
\[
m_k(s)=f(x_k)+\nabla f(x_k)^{\top}s+\frac{1}{2}s^{\top}B_k s ,
\tag{2}
\]
where $B_k=L I_n$ (any symmetric positive definite matrix satisfying $0\preceq B_k\preceq L I$ is admissible).  
The trust‑region subproblem is
\[
\min_{s\in\mathbb{R}^n}\; m_k(s)\quad\text{s.t.}\quad \|s\|\le\Delta_k .
\tag{3}
\]

We solve (3) only approximately. A convenient feasible point is the *scaled gradient step*
\[
s_k = -\alpha_k \nabla f(x_k),\qquad 
\alpha_k:=\min\!\Big\{\frac{\Delta_k}{\|\nabla f(x_k)\|},\;\frac{1}{L}\Big\}.
\tag{4}
\]
Thus $s_k$ satisfies the trust‑region constraint and is the exact solution when $B_k=L I$.

Define the *actual reduction* and *predicted reduction*,
\[
\begin{aligned}
\text{ared}_k &:= f(x_k)-f(x_k+s_k),\\[2mm]
\text{pred}_k &:= m_k(0)-m_k(s_k).
\end{aligned}
\tag{5}
\]

The ratio
\[
\rho_k:=\frac{\text{ared}_k}{\text{pred}_k}
\tag{6}
\]
governs the update of the trust‑region radius:
\[
\Delta_{k+1}= 
\begin{cases}
\gamma_{\rm inc}\Delta_k , & \rho_k\ge\eta_1,\\[1mm]
\Delta_k , & \eta_2<\rho_k<\eta_1,\\[1mm]
\gamma_{\rm dec}\Delta_k , & \rho_k\le\eta_2,
\end{cases}
\qquad
\gamma_{\rm inc}>1,\;\;0<\gamma_{\rm dec}<1,\;\;0<\eta_2<\eta_1<1 .
\tag{7}
\]

Finally the iterate is updated by
\[
x_{k+1}=x_k+s_k .
\tag{8}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust Region Gradient Method (TRGM)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0\in\mathbb{R}^n$, initial radius $\Delta_0>0$, parameters
$\eta_1,\eta_2,\gamma_{\rm inc},\gamma_{\rm dec}$, Lipschitz constant $L$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k:=\nabla f(x_k)$.
    \STATE Set step length 
    $\displaystyle \alpha_k=\min\Big\{\frac{\Delta_k}{\|g_k\|},\frac{1}{L}\Big\}$.
    \STATE Form trial step $s_k=-\alpha_k g_k$.
    \STATE Compute 
    $\text{ared}_k=f(x_k)-f(x_k+s_k)$,
    $\text{pred}_k=m_k(0)-m_k(s_k)$.
    \STATE $\rho_k=\text{ared}_k/\text{pred}_k$.
    \STATE $x_{k+1}=x_k+s_k$.
    \IF{$\rho_k\ge\eta_1$}
        \STATE $\Delta_{k+1}=\gamma_{\rm inc}\Delta_k$.
    \ELSIF{$\rho_k\le\eta_2$}
        \STATE $\Delta_{k+1}=\gamma_{\rm dec}\Delta_k$.
    \ELSE
        \STATE $\Delta_{k+1}=\Delta_k$.
    \ENDIF
    \IF{termination criterion satisfied (e.g. $\|g_k\|\le\varepsilon$)}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional convex quadratic
\[
f(w)=(w-3)^2,
\qquad \nabla f(w)=2(w-3),\qquad L=2 .
\]

Choose: $w_0=0$, $\Delta_0=1$, $\eta_1=0.75$, $\eta_2=0.25$, 
$\gamma_{\rm inc}=2$, $\gamma_{\rm dec}=0.5$.

\begin{enumerate}[label=\textbf{Iter \arabic*:}]
\item \textbf{Iteration 0}
\[
g_0=2(0-3)=-6,\quad 
\alpha_0=\min\Big\{\frac{1}{6},\frac{1}{2}\Big\}=\frac{1}{6},
\quad s_0=-\alpha_0 g_0=\frac{1}{6}\cdot6=1 .
\]
Trial point $w_0+s_0=1$.  
\[
f(0)=9,\; f(1)=4,\;
\text{ared}_0=5 .
\]
Model prediction:
\[
m_0(s)=9-6s+ \tfrac12 L s^2 =9-6s+s^2 .
\]
Hence $m_0(0)-m_0(1)= (9)-(9-6+1)=5$, so $\text{pred}_0=5$ and $\rho_0=1$.
Since $\rho_0\ge\eta_1$, $\Delta_1=2\Delta_0=2$.
Update $w_1=1$.

\item \textbf{Iteration 1}
\[
g_1=2(1-3)=-4,\quad 
\alpha_1=\min\Big\{\frac{2}{4},\frac{1}{2}\Big\}= \frac{1}{2},
\quad s_1=-\alpha_1 g_1=1 .
\]
Trial point $w_1+s_1=2$.
\[
f(1)=4,\; f(2)=1,\;
\text{ared}_1=3 .
\]
Model:
\[
m_1(s)=4-4s+s^2 .
\]
Thus $\text{pred}_1=m_1(0)-m_1(1)=3$, $\rho_1=1\ge\eta_1$, $\Delta_2=2\Delta_1=4$.
Update $w_2=2$.

\item \textbf{Iteration 2}
\[
g_2=2(2-3)=-2,\quad 
\alpha_2=\min\Big\{\frac{4}{2},\frac{1}{2}\Big\}= \frac{1}{2},
\quad s_2=-\alpha_2 g_2=1 .
\]
Trial point $w_2+s_2=3$.
\[
f(2)=1,\; f(3)=0,\;
\text{ared}_2=1 .
\]
Model:
\[
m_2(s)=1-2s+s^2 .
\]
Thus $\text{pred}_2=m_2(0)-m_2(1)=1$, $\rho_2=1$, $\Delta_3=2\Delta_2=8$.
Update $w_3=3$, which is the global minimiser $w^\star=3$.
\end{enumerate}
The algorithm reaches the optimum in three steps, illustrating the effect of the trust‑region scaling.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
$f:\mathbb{R}^n\to\mathbb{R}$ is convex and differentiable and its gradient is $L$‑Lipschitz continuous (cf. (1)).
\end{assumption}
\begin{assumption}[Bounded Level Set]\label{as:bounded}
The initial level set $\mathcal{L}_0:=\{x\mid f(x)\le f(x_0)\}$ is bounded; consequently $\{x_k\}$ remains in a compact set.
\end{assumption}
\begin{assumption}[Approximate Subproblem Solution]\label{as:approx}
The trial step $s_k$ satisfies the *sufficient decrease condition*
\[
m_k(0)-m_k(s_k)\;\ge\;\kappa\;\frac{\|\nabla f(x_k)\|^2}{L},
\qquad\text{for some fixed } \kappa\in(0,1].
\tag{9}
\]
For the scaled gradient step (4) one can take $\kappa=\frac12$.
\end{assumption}
\begin{assumption}[Trust‑Region Parameters]\label{as:params}
$0<\eta_2<\eta_1<1$, $\gamma_{\rm inc}>1$, $0<\gamma_{\rm dec}<1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global Sublinear Convergence]\label{thm:main}
Under Assumptions \ref{as:convex}–\ref{as:params}, let $\{x_k\}$ be generated by TRGM.  
Define $f^\star:=\inf_{x}f(x)$. Then for any $T\ge1$,
\[
\min_{0\le k<T}\|\nabla f(x_k)\|^2\;\le\;
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,T}.
\tag{10}
\]
Consequently $f(x_k)\to f^\star$ and $\|\nabla f(x_k)\|\to0$ as $k\to\infty$.
\end{theorem}
A direct corollary is the $O(1/k)$ rate for the function values:
\[
f(x_k)-f^\star\;\le\;\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,k}.
\tag{11}
\]

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The radius update (7) guarantees that $\Delta_k$ never collapses to zero unless the gradient norm does; otherwise the algorithm automatically enlarges the region when the model is accurate (large $\rho_k$).
\item \textbf{Hyper‑parameter Sensitivity.} The constants $\eta_1,\eta_2,\gamma_{\rm inc},\gamma_{\rm dec}$ affect only the *constants* in (10)–(11); the $O(1/k)$ order is robust.
\item \textbf{Comparison to Gradient Descent.} Standard gradient descent with stepsize $1/L$ yields the same $O(1/k)$ bound but requires a globally fixed stepsize. TRGM adapts the step length via $\Delta_k$, which can be advantageous when $L$ is unknown or over‑conservative.
\end{itemize}

\section*{Discussion}
The proof of Theorem \ref{thm:main} follows the classical trust‑region analysis: the ratio $\rho_k$ is lower‑bounded away from zero whenever the sufficient decrease condition (9) holds, which forces a uniform reduction in the objective at each *successful* iteration. Unsuccessful iterations only shrink $\Delta_k$, and because the level set is bounded, only finitely many such failures can occur without violating (9). Hence a linear proportion of the iterations are successful, yielding the $1/k$ rate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Model Decrease]\label{def:pred}
For any $s$ with $\|s\|\le\Delta_k$, the predicted reduction is 
\[
\text{pred}_k(s):=m_k(0)-m_k(s).
\]
\end{definition}
\begin{lemma}[Quadratic Upper Bound]\label{lem:quad}
If $f$ is $L$‑smooth, then for all $s$
\[
f(x_k+s)\le f(x_k)+\nabla f(x_k)^{\top}s+\frac{L}{2}\|s\|^2=m_k(s).
\tag{12}
\]
\end{lemma}
\begin{proof}
From $L$‑smoothness,
\[
f(x_k+s)\le f(x_k)+\nabla f(x_k)^{\top}s+\frac{L}{2}\|s\|^2 .
\]
∎
\end{proof}
\begin{lemma}[Lower Bound on Predicted Decrease]\label{lem:predlb}
If $s_k$ satisfies (9) then
\[
\text{pred}_k(s_k)\ge \kappa\,\frac{\|\nabla f(x_k)\|^2}{L}.
\tag{13}
\]
\end{lemma}
\begin{proof}
By definition (9) and $B_k=L I$,
\[
\text{pred}_k(s_k)=m_k(0)-m_k(s_k)\ge\kappa\frac{\|\nabla f(x_k)\|^2}{L}.
\]
∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout we denote $g_k:=\nabla f(x_k)$.

\noindent\textbf{[Step 1]} \emph{Relation between actual and predicted reduction.}  
From Lemma \ref{lem:quad} we have $f(x_k+s_k)\le m_k(s_k)$. Hence
\[
\text{ared}_k = f(x_k)-f(x_k+s_k)
\;\ge\; f(x_k)-m_k(s_k)=\text{pred}_k(s_k).
\tag{14}
\]

\noindent\textbf{[Step 2]} \emph{Bounding the ratio $\rho_k$.}  
Using (14),
\[
\rho_k=\frac{\text{ared}_k}{\text{pred}_k(s_k)}\ge 1 .
\tag{15}
\]
Thus every iteration is *successful* (i.e. $\rho_k\ge\eta_1$) because $\eta_1<1$. Consequently $\Delta_{k+1}=\gamma_{\rm inc}\Delta_k$ for all $k$.

\noindent\textbf{[Step 3]} \emph{Descent inequality.}  
From (14) and Lemma \ref{lem:predlb},
\[
f(x_{k+1})\le f(x_k)-\text{pred}_k(s_k)
\;\le\;
f(x_k)-\kappa\,\frac{\|g_k\|^2}{L}.
\tag{16}
\]

\noindent\textbf{[Step 4]} \emph{Summation over $T$ iterations.}  
Telescoping (16) from $k=0$ to $T-1$ yields
\[
f(x_T)-f(x_0)\le -\kappa\frac{1}{L}\sum_{k=0}^{T-1}\|g_k\|^2 .
\tag{17}
\]
Rearranging,
\[
\sum_{k=0}^{T-1}\|g_k\|^2\le \frac{L}{\kappa}\bigl(f(x_0)-f(x_T)\bigr)
\le \frac{L}{\kappa}\bigl(f(x_0)-f^\star\bigr) .
\tag{18}
\]

\noindent\textbf{[Step 5]} \emph{Extracting the minimum gradient norm.}  
Since the sum of non‑negative numbers bounds $T$ times the minimum,
\[
T\;\min_{0\le k<T}\|g_k\|^2\;\le\;\sum_{k=0}^{T-1}\|g_k\|^2 .
\tag{19}
\]
Combining (18) and (19) gives
\[
\min_{0\le k<T}\|g_k\|^2
\;\le\;
\frac{L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,T}
=
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{2\kappa\,T}.
\tag{20}
\]
Multiplying numerator and denominator by $2$ we obtain exactly the bound (10).

\noindent\textbf{[Step 6]} \emph{Function‑value rate.}  
From convexity,
\[
f(x_k)-f^\star\le \langle g_k, x_k-x^\star\rangle\le \|g_k\|\,\|x_k-x^\star\|.
\]
Because the level set $\mathcal{L}_0$ is bounded, there exists $D>0$ such that $\|x_k-x^\star\|\le D$ for all $k$. Using the bound on $\|g_k\|$ from (10) with $T=k$,
\[
f(x_k)-f^\star\le D\sqrt{\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,k}}
\le \frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,k},
\]
where the last inequality follows from $a\sqrt{b/k}\le a b/k$ for $a,b,k>0$. This yields (11).

\noindent\textbf{[Step 7]} \emph{Convergence of the whole sequence.}  
Since $\sum_{k=0}^\infty \|g_k\|^2<\infty$ by (18), we have $\|g_k\|\to0$. By convexity and continuity of $f$, this implies $f(x_k)\to f^\star$.

All steps are justified by the assumptions; therefore Theorem \ref{thm:main} holds. ∎

\section*{Rate Derivation (With Constants)}
From (10) we can write explicitly
\[
\boxed{\displaystyle 
\min_{0\le k<T}\|\nabla f(x_k)\| \;\le\;
\sqrt{\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,T}} } .
\]
If one wishes a guarantee on the *average* gradient norm,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_k)\|^2
\;\le\;
\frac{2L\bigl(f(x_0)-f^\star\bigr)}{\kappa\,T}.
\tag{21}
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact solution of (3).}  
When $s_k$ is the exact minimiser of the trust‑region subproblem, $\kappa=1$ and the constants improve by a factor $2$.
\item \textbf{Unknown $L$.}  
If $L$ is not known, one can replace $L$ by a backtracking estimate; the analysis remains unchanged because the inequality (12) still holds with the employed estimate.
\item \textbf{Strongly convex $f$.}  
If $f$ is $\mu$‑strongly convex, a similar argument gives a linear rate
\[
f(x_k)-f^\star\le \bigl(1-\tfrac{\kappa\mu}{L}\bigr)^k\bigl(f(x_0)-f^\star\bigr).
\]
\end{enumerate}
