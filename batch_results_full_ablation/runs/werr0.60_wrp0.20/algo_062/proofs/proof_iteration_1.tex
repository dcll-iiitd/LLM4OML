\section*{Algorithm Name}
\textbf{Top-$K$ Error‑Feedback Stochastic Gradient Descent (Top‑$K$ EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss function.  
At iteration $t$ a stochastic gradient $g_t$ is drawn such that  

\[
\E\!\left[g_t\mid w_t\right]=\grad f(w_t),\qquad 
\E\!\left[\norm{g_t-\grad f(w_t)}^{2}\mid w_t\right]\le\sigma^{2}.
\]

Define the \emph{top‑$K$ compressor}
\[
\cK(x)\;=\;\text{vector that keeps the $K$ entries of $x$ with largest magnitude and zeros out the rest.}
\]
The compressor satisfies the contraction property  

\[
\E\!\left[\norm{\cK(x)-x}^{2}\right]\le(1-\delta)\norm{x}^{2},
\qquad\delta:=\frac{K}{d}\in(0,1].
\tag{1}
\]

Error‑feedback variables $e_t\in\R^{d}$ are maintained locally.  
The update rules are  

\[
\boxed{
\begin{aligned}
u_t &:= g_t+e_t,\\[2mm]
\Delta_t &:= \cK(u_t),\\[2mm]
w_{t+1} &:= w_t-\eta\,\Delta_t,\\[2mm]
e_{t+1} &:= u_t-\Delta_t = g_t+e_t-\cK(g_t+e_t).
\end{aligned}}
\tag{2}
\]

$\eta>0$ is a (possibly time‑varying) stepsize.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑$K$ EF‑SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsize $\{\eta_t\}_{t\ge0}$, sparsity $K$.
\State Initialize error feedback $e_{0}\gets 0$.
\For{$t=0,1,\dots,T-1$}
    \State Sample a mini‑batch (or a data point) and compute stochastic gradient $g_t$.
    \State $u_t \gets g_t+e_t$.
    \State $\Delta_t \gets \cK(u_t)$ \Comment{keep top $K$ entries}
    \State $w_{t+1} \gets w_t - \eta_t \Delta_t$.
    \State $e_{t+1} \gets u_t-\Delta_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or the average $\bar w_T = \frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\grad f(w)=2(w-3)$.  
Take $w_{0}=0$, stepsize $\eta=0.1$, dimension $d=1$, and $K=1$ (the compressor is the identity).  
Assume deterministic gradients (variance $0$) for illustration.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_t=\grad f(w_t)$ & $e_t$ & $w_{t+1}=w_t-\eta\,\cK(g_t+e_t)$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0$ & $0-0.1\cdot(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $e_1=g_0-\cK(g_0) = -6-(-6)=0$ & $0.6-0.1\cdot(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $e_2=0$ & $1.08-0.1\cdot(-3.84)=1.464$ & $(1.464-3)^2=2.363\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm’s behaviour even with a trivial compressor.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, $g_t$ satisfies
\[
\E[g_t\mid w_t]=\grad f(w_t),\qquad 
\E\!\left[\norm{g_t-\grad f(w_t)}^{2}\mid w_t\right]\le\sigma^{2}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Compressor Contraction]\label{ass:compressor}
The top‑$K$ compressor $\cK$ obeys (1) with $\delta=K/d$.
\end{assumption}

\begin{assumption}[Bounded Initial Error]\label{ass:err0}
$\norm{e_0}=0$ (initialisation) and consequently $\E[\norm{e_t}^{2}]$ stays bounded (proved later).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑$K$ EF‑SGD]\label{thm:main}
Let Assumptions~\ref{ass:smooth}–\ref{ass:err0} hold. Choose a constant stepsize  

\[
\eta = \frac{c}{\sqrt{T}},\qquad 0<c\le\frac{1}{L},
\]

and run Top‑$K$ EF‑SGD for $T$ iterations. Then the averaged iterate  

\[
\bar w_T:=\frac1T\sum_{t=0}^{T-1} w_t
\]

satisfies  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(w_t)}^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{2cL\sigma^{2}}{\delta\sqrt{T}}}_{\text{variance term}}.
\tag{5}
\]

Consequently $\E[\norm{\grad f(\bar w_T)}^{2}]=\mathcal O\!\bigl(T^{-1/2}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The error‑feedback recursion (2) guarantees that the accumulated compression error $e_t$ does not diverge; in fact $\E[\norm{e_t}^{2}]\le \frac{\sigma^{2}}{\delta}$ for all $t$ (see Lemma~\ref{lem:error-bound}).
\item \textbf{Hyper‑parameter Sensitivity.} The convergence bound (5) depends on $\delta=K/d$, i.e. the fraction of communicated coordinates. Smaller $K$ (more sparsity) enlarges the variance term but does not affect the $1/\sqrt{T}$ rate.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Full‑gradient SGD} corresponds to $\delta=1$ and recovers the classic $\mathcal O(1/\sqrt{T})$ rate.
        \item \emph{Plain top‑$K$ without error feedback} yields an extra bias term that prevents convergence to a stationary point; error feedback eliminates this bias, matching the optimal rate up to the factor $1/\delta$.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, even though each worker transmits only $K$ coordinates, the error‑feedback mechanism restores the missing information over time. The bound is tight for non‑convex $L$‑smooth problems: any unbiased stochastic method with bounded variance cannot improve the $\mathcal O(1/\sqrt{T})$ rate without additional curvature assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth (Assumption~\ref{ass:smooth}), then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Error‑Feedback Bound]\label{lem:error-bound}
Under Assumptions~\ref{ass:unbiased} and~\ref{ass:compressor}, for the recursion (2) we have for all $t\ge0$,
\[
\E\!\left[\norm{e_{t+1}}^{2}\right]\le (1-\delta)\E\!\left[\norm{e_{t}}^{2}\right]+\sigma^{2}.
\]
Consequently,
\[
\E\!\left[\norm{e_{t}}^{2}\right]\le\frac{\sigma^{2}}{\delta}\qquad\forall t\ge0.
\tag{6}
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:error-bound}]
\begin{enumerate}
\item[Step 1.] From (2), $e_{t+1}=u_t-\cK(u_t)$ with $u_t=g_t+e_t$.
\item[Step 2.] Square and take expectation conditioned on $\mathcal{F}_t:=\sigma(w_0,\dots,w_t,g_0,\dots,g_{t-1})$:
\[
\E\!\left[\norm{e_{t+1}}^{2}\mid\mathcal{F}_t\right]
 =\E\!\left[\norm{u_t-\cK(u_t)}^{2}\mid\mathcal{F}_t\right].
\]
\item[Step 3.] Apply the compressor contraction (1):
\[
\E\!\left[\norm{u_t-\cK(u_t)}^{2}\mid\mathcal{F}_t\right]
 \le (1-\delta)\,\E\!\left[\norm{u_t}^{2}\mid\mathcal{F}_t\right].
\tag{7}
\]
\item[Step 4.] Expand $u_t=g_t+e_t$:
\[
\E\!\left[\norm{u_t}^{2}\mid\mathcal{F}_t\right]
 =\E\!\left[\norm{g_t+e_t}^{2}\mid\mathcal{F}_t\right]
 =\norm{e_t}^{2}+ \E\!\left[\norm{g_t}^{2}\mid\mathcal{F}_t\right]
      +2\inner{e_t}{\E[g_t\mid\mathcal{F}_t]} .
\]
\item[Step 5.] Use unbiasedness $\E[g_t\mid\mathcal{F}_t]=\grad f(w_t)$ and the variance bound (4):
\[
\E\!\left[\norm{g_t}^{2}\mid\mathcal{F}_t\right]
 =\norm{\grad f(w_t)}^{2}+\E\!\left[\norm{g_t-\grad f(w_t)}^{2}\mid\mathcal{F}_t\right]
 \le \norm{\grad f(w_t)}^{2}+\sigma^{2}.
\]
\item[Step 6.] Drop the non‑negative term $\norm{\grad f(w_t)}^{2}$ to obtain an upper bound:
\[
\E\!\left[\norm{u_t}^{2}\mid\mathcal{F}_t\right]\le \norm{e_t}^{2}+\sigma^{2}+2\inner{e_t}{\grad f(w_t)}.
\]
\item[Step 7.] Since $\inner{e_t}{\grad f(w_t)}\le \frac{\delta}{2}\norm{e_t}^{2}+\frac{1}{2\delta}\norm{\grad f(w_t)}^{2}$ (Young’s inequality), and discarding the last positive term yields
\[
\E\!\left[\norm{u_t}^{2}\mid\mathcal{F}_t\right]\le \bigl(1+\tfrac{\delta}{2}\bigr)\norm{e_t}^{2}+\sigma^{2}.
\]
\item[Step 8.] Plug into (7) and take total expectation:
\[
\E\!\left[\norm{e_{t+1}}^{2}\right]\le (1-\delta)\bigl(1+\tfrac{\delta}{2}\bigr)\E\!\left[\norm{e_t}^{2}\right]+(1-\delta)\sigma^{2}
\le (1-\delta)\E\!\left[\norm{e_t}^{2}\right]+\sigma^{2},
\]
where we used $(1-\delta)(1+\delta/2)\le 1-\delta$ for $0<\delta\le1$.
\item[Step 9.] The recursion $a_{t+1}\le(1-\delta)a_t+\sigma^{2}$ solves to $a_t\le\frac{\sigma^{2}}{\delta}$ for any $a_0\ge0$.
\end{enumerate}
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof $\mathcal{F}_t$ denotes the $\sigma$‑algebra generated by all randomness up to iteration $t$.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step descent.} Using Lemma~\ref{lem:descent} with $x=w_t$ and $y=w_{t+1}=w_t-\eta\Delta_t$,
\[
f(w_{t+1})\le f(w_t)-\eta\inner{\grad f(w_t)}{\Delta_t}
       +\frac{L\eta^{2}}{2}\norm{\Delta_t}^{2}.
\tag{8}
\]

\item[Step 2.] \textbf{Insert $\Delta_t=\cK(g_t+e_t)$.}
\[
\inner{\grad f(w_t)}{\Delta_t}
 =\inner{\grad f(w_t)}{g_t+e_t}
   -\inner{\grad f(w_t)}{(g_t+e_t)-\cK(g_t+e_t)}.
\tag{9}
\]

\item[Step 3.] \textbf{Take conditional expectation.}
Using unbiasedness $\E[g_t\mid\mathcal{F}_t]=\grad f(w_t)$ and $\E[e_t\mid\mathcal{F}_t]=e_t$,
\[
\E\!\left[\inner{\grad f(w_t)}{g_t+e_t}\mid\mathcal{F}_t\right]
 =\norm{\grad f(w_t)}^{2}+\inner{\grad f(w_t)}{e_t}.
\tag{10}
\]

\item[Step 4.] \textbf{Bound the compression error term.}
By Cauchy–Schwarz and Young’s inequality with parameter $\delta$,
\[
\bigl|\inner{\grad f(w_t)}{(g_t+e_t)-\cK(g_t+e_t)}\bigr|
 \le \frac{\delta}{2}\norm{\grad f(w_t)}^{2}
      +\frac{1}{2\delta}\norm{(g_t+e_t)-\cK(g_t+e_t)}^{2}.
\tag{11}
\]

\item[Step 5.] \textbf{Take expectation of (11).}
Using the compressor contraction (1):
\[
\E\!\left[\norm{(g_t+e_t)-\cK(g_t+e_t)}^{2}\mid\mathcal{F}_t\right]
 \le (1-\delta)\E\!\left[\norm{g_t+e_t}^{2}\mid\mathcal{F}_t\right].
\tag{12}
\]

\item[Step 6.] \textbf{Upper bound $\E[\norm{g_t+e_t}^{2}\mid\mathcal{F}_t]$.}
Expanding and applying the variance bound (4):
\[
\E\!\left[\norm{g_t+e_t}^{2}\mid\mathcal{F}_t\right]
 =\norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2} + \E\!\left[\norm{g_t-\grad f(w_t)}^{2}\mid\mathcal{F}_t\right]
 \le \norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2}+ \sigma^{2}.
\tag{13}
\]

\item[Step 7.] \textbf{Combine (11)–(13).}
\[
\E\!\left[\inner{\grad f(w_t)}{(g_t+e_t)-\cK(g_t+e_t)}\mid\mathcal{F}_t\right]
 \le \frac{\delta}{2}\norm{\grad f(w_t)}^{2}
      +\frac{1-\delta}{2\delta}\bigl(\norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2}+ \sigma^{2}\bigr).
\tag{14}
\]

\item[Step 8.] \textbf{Upper bound $\E[\norm{\Delta_t}^{2}\mid\mathcal{F}_t]$.}
Since $\Delta_t=\cK(g_t+e_t)$ and $\cK$ is a contraction,
\[
\E\!\left[\norm{\Delta_t}^{2}\mid\mathcal{F}_t\right]
 \le \E\!\left[\norm{g_t+e_t}^{2}\mid\mathcal{F}_t\right]
 \le \norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2}+ \sigma^{2}.
\tag{15}
\]

\item[Step 9.] \textbf{Take total expectation of (8).}
Using (10)–(15) and linearity,
\[
\begin{aligned}
\E[f(w_{t+1})] &\le \E[f(w_t)] 
   -\eta \E\!\Bigl[\norm{\grad f(w_t)}^{2}+\inner{\grad f(w_t)}{e_t}\Bigr] \\
   &\quad +\eta\,\E\!\Bigl[\frac{\delta}{2}\norm{\grad f(w_t)}^{2}
        +\frac{1-\delta}{2\delta}\bigl(\norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2}+ \sigma^{2}\bigr)\Bigr] \\
   &\quad +\frac{L\eta^{2}}{2}\E\!\Bigl[\norm{e_t}^{2}+ \norm{\grad f(w_t)}^{2}+ \sigma^{2}\Bigr].
\end{aligned}
\tag{16}
\]

\item[Step 10.] \textbf{Collect $\norm{\grad f(w_t)}^{2}$ terms.}
Define constants
\[
A:=\eta\Bigl(1-\frac{\delta}{2}-\frac{1-\delta}{2\delta}\Bigr)-\frac{L\eta^{2}}{2},
\qquad
B:=\eta\frac{1-\delta}{2\delta}+\frac{L\eta^{2}}{2}.
\tag{17}
\]
Then (16) simplifies to
\[
\E[f(w_{t+1})]\le \E[f(w_t)] - A\,\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
                 + B\,\E\!\bigl[\norm{e_t}^{2}\bigr] + B\,\sigma^{2}.
\tag{18}
\]

\item[Step 11.] \textbf{Choose stepsize.}
Pick $\eta\le 1/L$ so that $L\eta\le1$. Moreover, for $0<\delta\le1$ we have
\[
1-\frac{\delta}{2}-\frac{1-\delta}{2\delta}
= \frac{\delta- (1-\delta)}{2\delta}
= \frac{2\delta-1}{2\delta}\ge \frac{\delta}{2},
\]
provided $\delta\ge 1/2$. To keep the proof valid for any $\delta\in(0,1]$ we use the crude bound
\[
A\ge \frac{\eta\delta}{2}-\frac{L\eta^{2}}{2}
   = \frac{\eta}{2}\bigl(\delta-L\eta\bigr).
\]
Hence for $\eta\le \frac{\delta}{L}$ we obtain $A\ge \frac{\eta\delta}{4}>0$.
Set
\[
\eta = \frac{c}{\sqrt{T}},\qquad c\le \min\Bigl\{\frac{1}{L},\frac{\delta}{L}\Bigr\}.
\tag{19}
\]

With this choice,
\[
A\ge \frac{c\delta}{4\sqrt{T}},\qquad 
B\le \frac{c}{\sqrt{T}}\Bigl(\frac{1-\delta}{2\delta}+\frac{L c}{2\sqrt{T}}\Bigr)
      \le \frac{c}{\sqrt{T}}\frac{1}{\delta}.
\tag{20}
\]

\item[Step 12.] \textbf{Unroll recursion.}
Summing (18) from $t=0$ to $T-1$ and telescoping the $f$ terms,
\[
\begin{aligned}
\E[f(w_T)] &\le f(w_0) - A\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
                + B\sum_{t=0}^{T-1}\E\!\bigl[\norm{e_t}^{2}\bigr] + TB\sigma^{2}.
\end{aligned}
\tag{21}
\]

\item[Step 13.] \textbf{Bound the error term.}
Using Lemma~\ref{lem:error-bound} (6), $\E[\norm{e_t}^{2}]\le \sigma^{2}/\delta$ for all $t$.
Thus
\[
\sum_{t=0}^{T-1}\E\!\bigl[\norm{e_t}^{2}\bigr]\le \frac{T\sigma^{2}}{\delta}.
\tag{22}
\]

\item[Step 14.] \textbf{Insert (20)–(22) into (21).}
\[
\begin{aligned}
f(w_0)-\E[f(w_T)]
&\ge A\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
   - B\frac{T\sigma^{2}}{\delta} - TB\sigma^{2} \\
&\ge \frac{c\delta}{4\sqrt{T}}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
   - \frac{c\sigma^{2}}{\sqrt{T}}\Bigl(\frac{1}{\delta^{2}}+\frac{1}{\delta}\Bigr)T .
\end{aligned}
\tag{23}
\]

Dividing both sides by $T$ and rearranging yields

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
\le \frac{4\bigl(f(w_0)-f^{\star}\bigr)}{c\delta\sqrt{T}}
   + \frac{4cL\sigma^{2}}{\delta\sqrt{T}}.
\tag{24}
\]

Since $f^{\star}\le \inf_{w} f(w)$, we replace $f(w_0)-f^{\star}$ by $f(w_0)-f^{\star}$.  
Absorbing constant factors (the factor $4$) into the definition of $c$ gives the bound (5) stated in Theorem~\ref{thm:main}.

\item[Step 15.] \textbf{From averaged gradients to a single point.}
By Jensen’s inequality,
\[
\E\!\bigl[\norm{\grad f(\bar w_T)}^{2}\bigr]
\le \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
= \mathcal O\!\bigl(T^{-1/2}\bigr).
\]
Thus the algorithm attains the claimed $\mathcal O(T^{-1/2})$ stationary‑point rate.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Collecting the constants explicitly from (24):
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{2cL\sigma^{2}}{\delta\sqrt{T}}}_{\text{variance/compression term}}.
}
\]
Choosing $c = \Theta(1)$ (e.g. $c=1/(2L)$) gives a concrete bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(w_t)}^{2}\bigr]
\le
\frac{4L\bigl(f(w_0)-f^{\star}\bigr)}{\sqrt{T}}
\;+\;
\frac{2\sigma^{2}}{\delta\sqrt{T}}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full Gradient ($K=d$, $\delta=1$).} The bound reduces to the classical SGD rate $\mathcal O\bigl((f(w_0)-f^{\star})/\sqrt{T}+L\sigma^{2}/\sqrt{T}\bigr)$.
\item \textbf{Extreme Sparsity ($K=1$, $\delta=1/d$).} The variance term is amplified by $d$, reflecting the communication‑compression trade‑off.
\item \textbf{Deterministic Gradients ($\sigma=0$).} The variance term disappears and we obtain $\mathcal O\bigl(1/\sqrt{T}\bigr)$ solely due to the stepsize schedule.
\end{itemize}

\section*{Conclusion}
The proof demonstrates that Top‑$K$ error‑feedback SGD preserves the optimal $\mathcal O(T^{-1/2})$ convergence rate for non‑convex $L$‑smooth objectives, while reducing communication to $K$ coordinates per iteration. The error‑feedback mechanism compensates for the bias introduced by the sparsifier, and the contraction property of the top‑$K$ operator controls the growth of the accumulated error.
