\section*{Algorithm Name}
\textbf{Probabilistic Coordinate Descent (PCD)}  

The method randomly selects a single coordinate according to a prescribed discrete distribution and performs a gradient step only along that coordinate.  To obtain an unbiased descent direction the step size is scaled by the inverse of the selection probability.

%=====================================================================
\section*{Mathematical Formulation}
We consider the unconstrained optimization problem
\[
\min_{w\in\mathbb{R}^{d}} f(w),
\]
where $f:\mathbb{R}^{d}\!\to\!\mathbb{R}$ is \emph{non‑convex} but \emph{coordinate‑wise $L_i$‑smooth}:
\begin{equation}
\label{eq:coord_smooth}
\bigl|\partial_i f(x+\alpha e_i)-\partial_i f(x)\bigr|\le L_i|\alpha|
\qquad\forall x\in\mathbb{R}^{d},\;\forall\alpha\in\mathbb{R},
\end{equation}
with $e_i$ the $i$‑th canonical basis vector and $L_i>0$ a Lipschitz constant for the $i$‑th partial derivative.

Let $p=(p_1,\dots,p_d)$ be a probability vector:
\[
p_i>0,\qquad \sum_{i=1}^{d}p_i=1 .
\]

Given a stepsize parameter $\eta>0$, the PCD iteration is
\begin{equation}
\label{eq:update}
w_{k+1}= w_k-\frac{\eta}{p_{i_k}}\;\partial_{i_k} f(w_k)\,e_{i_k},
\qquad i_k\sim \text{Categorical}(p).
\end{equation}
The factor $1/p_{i_k}$ guarantees that the expected update equals the full gradient scaled by $\eta$:
\[
\mathbb{E}\!\bigl[w_{k+1}\mid w_k\bigr]=w_k-\eta\,\nabla f(w_k).
\]

%=====================================================================
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent (PCD)}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^{d}$, stepsize $\eta>0$, probability vector $p\in\Delta^{d-1}$, max iterations $T$.
\For{$k=0,1,\dots,T-1$}
    \State Sample $i_k\in\{1,\dots,d\}$ with $\Pr(i_k=i)=p_i$.
    \State Compute partial gradient $g_{i_k}= \partial_{i_k}f(w_k)$.
    \State Update the selected coordinate
    \[
    w_{k+1}= w_k-\frac{\eta}{p_{i_k}}\, g_{i_k}\,e_{i_k}.
    \]
\EndFor
\State \Return $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize the one‑dimensional quadratic $f(w)=(w-3)^2$.
\begin{itemize}
    \item Dimension $d=1$, thus $p_1=1$, $L_1=2$ (since $\partial^2 f=2$).
    \item Choose $\eta=0.1$ and $w_0=0$.
\end{itemize}
Because $d=1$, the algorithm always selects the sole coordinate.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise Smoothness]
\label{as:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that \eqref{eq:coord_smooth} holds.
\end{assumption}

\begin{assumption}[Bounded Below]
\label{as:lower}
$f$ is bounded from below: $f^\star:=\inf_{w}f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Stepsize]
\label{as:stepsize}
The stepsize $\eta$ satisfies
\[
0<\eta\le \min_{i}\frac{p_i}{L_i}.
\]
\end{assumption}

All expectations are taken with respect to the random index $i_k$ conditioned on the current iterate $w_k$.

%=====================================================================
\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Let $\{w_k\}_{k\ge0}$ be generated by PCD under Assumptions \ref{as:smooth}–\ref{as:stepsize}. Define the averaged squared partial gradient
\[
\Phi_T:=\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr],
\qquad 
\|\nabla f(w_k)\|_{p^{-1}}^{2}:=\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(w_k)\bigr)^{2}.
\]
Then
\begin{equation}
\label{eq:rate}
\Phi_T\;\le\;
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta\,T}.
\end{equation}
Consequently,
\[
\min_{0\le k<T}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]\;\le\;\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta\,T}.
\]
Thus the method attains an $\mathcal{O}(1/T)$ decay of the expected (weighted) gradient norm.
\end{theorem}

%=====================================================================
\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Unbiased Descent Direction.} By construction,
\[
\mathbb{E}\!\bigl[w_{k+1}-w_k\mid w_k\bigr]=-\eta\nabla f(w_k).
\]
Hence the method behaves like full‑gradient descent in expectation.

\item \textbf{Stability Condition.} The stepsize bound in Assumption \ref{as:stepsize} guarantees that the coefficient
\[
\frac{\eta}{p_i}-\frac{L_i\eta^{2}}{2p_i^{2}}\ge\frac{\eta}{2p_i}>0,
\]
which is essential for a guaranteed decrease of the objective (see Lemma \ref{lem:descent}).

\item \textbf{Effect of the Probability Vector.} The weighted norm $\|\cdot\|_{p^{-1}}$ naturally penalises coordinates that are selected rarely (small $p_i$).  Choosing $p_i\propto L_i^{-1/2}$ equalises the per‑iteration progress across coordinates (cf. importance sampling).

\item \textbf{Comparison to Uniform Random CD.} If $p_i=1/d$, the bound \eqref{eq:rate} reduces to the standard $\mathcal{O}(d/T)$ rate for uniform coordinate descent.  Non‑uniform sampling can improve the constant by a factor up to $\sqrt{d}$.

\end{enumerate}

\section*{Discussion}
The theorem provides a sub‑linear convergence guarantee for \emph{non‑convex} problems, measured by the weighted gradient norm.  Unlike convex analyses, no global optimality is claimed; instead the result ensures that the algorithm finds an \emph{approximate stationary point}.  The bound matches the optimal $\mathcal{O}(1/T)$ rate for stochastic first‑order methods under the same smoothness assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate Descent Smoothness Inequality]
\label{lem:smooth}
Under Assumption \ref{as:smooth}, for any $w\in\mathbb{R}^{d}$, any coordinate $i$ and any scalar $\alpha$,
\[
f(w+\alpha e_i)\le f(w)+\alpha\,\partial_i f(w)+\frac{L_i}{2}\alpha^{2}.
\]
\end{lemma}
\begin{proof}
Apply the fundamental theorem of calculus to the univariate function $\phi(\alpha)=f(w+\alpha e_i)$ and use the Lipschitz bound \eqref{eq:coord_smooth}.  This yields the standard quadratic upper bound.  $\square$
\end{proof}

\begin{lemma}[One‑step Expected Descent]
\label{lem:descent}
Let $w_{k+1}$ be defined by \eqref{eq:update}.  Under Assumptions \ref{as:smooth}–\ref{as:stepsize},
\[
\mathbb{E}\!\bigl[f(w_{k+1})\mid w_k\bigr]\le
f(w_k)-\frac{\eta}{2}\,\|\nabla f(w_k)\|_{p^{-1}}^{2}.
\]
\end{lemma}
\begin{proof}
Condition on $w_k$ and apply Lemma \ref{lem:smooth} with $\alpha=-\frac{\eta}{p_i}\partial_i f(w_k)$ for the randomly chosen index $i=i_k$:
\[
f\bigl(w_k-\tfrac{\eta}{p_i}\partial_i f(w_k)e_i\bigr)
\le f(w_k)-\frac{\eta}{p_i}\bigl(\partial_i f(w_k)\bigr)^{2}
+\frac{L_i}{2}\Bigl(\tfrac{\eta}{p_i}\partial_i f(w_k)\Bigr)^{2}.
\tag{A}
\]
Rearrange (A):
\[
f(w_{k+1})\le f(w_k)-\Bigl(\frac{\eta}{p_i}-\frac{L_i\eta^{2}}{2p_i^{2}}\Bigr)\bigl(\partial_i f(w_k)\bigr)^{2}.
\tag{B}
\]
By Assumption \ref{as:stepsize},
\[
\frac{\eta}{p_i}-\frac{L_i\eta^{2}}{2p_i^{2}}
\ge\frac{\eta}{2p_i}\qquad\forall i.
\tag{C}
\]
Insert (C) into (B) and take expectation w.r.t.\ $i_k$:
\[
\mathbb{E}\!\bigl[f(w_{k+1})\mid w_k\bigr]
\le f(w_k)-\frac{\eta}{2}\sum_{i=1}^{d}\frac{1}{p_i}p_i\bigl(\partial_i f(w_k)\bigr)^{2}
= f(w_k)-\frac{\eta}{2}\,\|\nabla f(w_k)\|_{p^{-1}}^{2}.
\tag{D}
\]
Thus the claimed inequality holds. $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  All steps are explicitly numbered for easy reference.

\begin{proof}
\textbf{[Step 1]} Define the optimality gap $\Delta_k:=f(w_k)-f^\star$, where $f^\star$ denotes the infimum of $f$ (Assumption \ref{as:lower}).  

\textbf{[Step 2]} Apply Lemma \ref{lem:descent} and take total expectation:
\[
\mathbb{E}\bigl[f(w_{k+1})\bigr]\le\mathbb{E}\bigl[f(w_k)\bigr]-\frac{\eta}{2}\,
\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr].
\tag{1}
\]

\textbf{[Step 3]} Rearrange (1) to isolate the gradient term:
\[
\frac{\eta}{2}\,
\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]
\le \mathbb{E}\bigl[f(w_k)\bigr]-\mathbb{E}\bigl[f(w_{k+1})\bigr].
\tag{2}
\]

\textbf{[Step 4]} Sum inequality (2) over $k=0,\dots,T-1$:
\[
\frac{\eta}{2}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]
\le \mathbb{E}\bigl[f(w_0)\bigr]-\mathbb{E}\bigl[f(w_T)\bigr].
\tag{3}
\]

\textbf{[Step 5]} Since $f(w_T)\ge f^\star$, we can replace the right‑hand side by $f(w_0)-f^\star$:
\[
\frac{\eta}{2}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]
\le f(w_0)-f^\star.
\tag{4}
\]

\textbf{[Step 6]} Divide both sides of (4) by $T$ and multiply by $2/\eta$ to obtain the average bound:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta\,T}.
\tag{5}
\]
The left‑hand side is precisely $\Phi_T$ defined in the theorem statement, which yields \eqref{eq:rate}.

\textbf{[Step 7]} To derive the minimum‑gradient guarantee, note that for any non‑negative numbers $a_0,\dots,a_{T-1}$,
\[
\min_{k} a_k \le \frac{1}{T}\sum_{k=0}^{T-1}a_k.
\]
Apply this with $a_k:=\mathbb{E}\!\bigl[\|\nabla f(w_k)\|_{p^{-1}}^{2}\bigr]$ and combine with (5) to obtain the second inequality of the theorem. $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
The bound \eqref{eq:rate} can be rewritten as
\[
\mathbb{E}\!\bigl[\|\nabla f(w_{\tilde{k}})\|_{p^{-1}}^{2}\bigr]
\le \underbrace{\frac{2}{\eta}}_{C_1}\,
\underbrace{\bigl(f(w_0)-f^\star\bigr)}_{C_2}\,
\underbrace{\frac{1}{T}}_{C_3},
\]
where:
\begin{itemize}
\item $C_1$ depends inversely on the stepsize.  Larger $\eta$ (subject to \eqref{as:stepsize}) yields a tighter bound.
\item $C_2$ is the initial optimality gap.
\item $C_3$ shows the explicit $\mathcal{O}(1/T)$ decay.
\end{itemize}
If one chooses the \emph{optimal} stepsize
\[
\eta^{\star}= \min_i \frac{p_i}{L_i},
\]
then $C_1=2\max_i \frac{L_i}{p_i}$, providing a clean expression in terms of the smoothness constants and sampling probabilities.

%=====================================================================
\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Uniform Sampling.} $p_i=1/d$ for all $i$.  The weighted norm reduces to $d\|\nabla f(w)\|^{2}$ and the rate becomes
\[
\frac{2d\bigl(f(w_0)-f^\star\bigr)}{\eta T},
\]
which coincides with the classical analysis of uniform random coordinate descent.

\item \textbf{Importance Sampling.} Set $p_i\propto L_i^{1/2}$.  Then $\frac{L_i}{p_i}$ is constant across coordinates, minimising the constant $C_1$ in the rate.  This choice is optimal for the bound in the sense of minimizing the worst‑case per‑iteration progress.

\item \textbf{Deterministic Cyclic CD.} If $p_i=1$ for a single coordinate and $0$ for the rest, the algorithm reduces to a standard gradient step on that coordinate.  The theorem still holds (trivially) with $d=1$.

\end{enumerate}

%=====================================================================
\section*{Conclusion}
We have presented a fully detailed analysis of Probabilistic Coordinate Descent for non‑convex, coordinate‑wise smooth objectives.  By scaling the step size with the inverse sampling probability, the method enjoys an unbiased descent direction and attains an $\mathcal{O}(1/T)$ convergence rate in expectation of a weighted gradient norm.  The proof is elementary, relies only on the coordinate smoothness property, and highlights how the probability vector $p$ influences both stability and the convergence constant.
