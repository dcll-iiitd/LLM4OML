\section*{Accelerated Gradient Method for Strongly Convex Smooth Functions}
We consider the classical Nesterov’s Accelerated Gradient (NAG) scheme with a *fixed* momentum coefficient that is optimal for the class of $L$‑smooth, $\mu$‑strongly convex functions.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable. The algorithm maintains two sequences $\{x_k\}_{k\ge 0}$ and $\{y_k\}_{k\ge 0}$:
\[
\boxed{
\begin{aligned}
y_k &= x_k + \beta\,(x_k-x_{k-1}),\\[4pt]
x_{k+1} &= y_k - \alpha \,\nabla f(y_k),
\end{aligned}}
\qquad k\ge 0,
\]
with initialization $x_{-1}=x_0\in\mathbb{R}^d$.
The step size $\alpha$ and momentum $\beta$ are chosen as
\[
\alpha = \frac{1}{L},\qquad 
\beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}} \in (0,1).
\]
Here $L\ge\mu>0$ denote the smoothness and strong‑convexity constants of $f$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient (NAG) for $\mu$‑strongly convex $L$‑smooth $f$}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_0\in\mathbb{R}^d$, step size $\alpha=1/L$, momentum $\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$
\State Set $x_{-1}\gets x_0$
\For{$k=0,1,2,\dots$}
    \State $y_k \gets x_k + \beta (x_k-x_{k-1})$ \Comment{extrapolation}
    \State $g_k \gets \nabla f(y_k)$
    \State $x_{k+1} \gets y_k - \alpha g_k$ \Comment{gradient step}
    \If{stopping criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{Output:} $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2,
\]
so that $\nabla f(w)=2(w-3)$. The function is $L=2$ smooth and $\mu=2$ strongly convex (both equal for a quadratic). Hence $\alpha=1/L=0.5$ and $\beta=0$ (because $\sqrt{L}=\sqrt{\mu}$). The algorithm therefore reduces to plain gradient descent, but we keep the general update for illustration.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $x_k$ & $y_k$ & $g_k=\nabla f(y_k)$ & $f(x_{k+1})$ \\ \hline
0 & $0$ & $0+\beta(0-0)=0$ & $2(0-3)=-6$ & $x_1=0-0.5(-6)=3$, $f(3)=0$ \\ \hline
1 & $3$ & $3+\beta(3-0)=3$ & $2(3-3)=0$ & $x_2=3-0.5\cdot0=3$, $f(3)=0$ \\ \hline
2 & $3$ & $3$ & $0$ & $x_3=3$, $f(3)=0$
\end{tabular}
\end{center}
After the first iteration the method reaches the optimum $w^\star=3$, and the objective value stays at $0$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $u,v\in\mathbb{R}^d$,
\[
f(v)\le f(u)+\ip{\nabla f(u)}{v-u}+\frac{L}{2}\norm{v-u}^2 .
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex, i.e. for all $u,v\in\mathbb{R}^d$,
\[
f(v)\ge f(u)+\ip{\nabla f(u)}{v-u}+\frac{\mu}{2}\norm{v-u}^2 .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Parameter Choice]\label{ass:param}
The step size and momentum satisfy
\[
\alpha=\frac{1}{L},\qquad 
\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}} .
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of NAG]\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:param}, let $x^\star=\arg\min f$. The iterates produced by the algorithm satisfy for all $k\ge 0$,
\[
\boxed{
f(x_k)-f(x^\star)\;\le\;\Bigl(1-\sqrt{\mu/L}\Bigr)^{k}\,\bigl(f(x_0)-f(x^\star)\bigr) .
}
\tag{T1}
\]
Equivalently,
\[
\norm{x_k-x^\star}^2 \;\le\; \Bigl(1-\sqrt{\mu/L}\Bigr)^{k}\,\frac{2}{\mu}\bigl(f(x_0)-f(x^\star)\bigr) .
\tag{T2}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The choice $\beta\in(0,1)$ guarantees that the extrapolation does not diverge; the contraction factor $1-\sqrt{\mu/L}$ is strictly smaller than the gradient‑descent factor $1-\mu/L$.
    \item \textbf{Hyper‑parameter sensitivity.} If $\beta$ is chosen according to (A3) the bound (T1) is optimal for the class of $L$‑smooth $\mu$‑strongly convex functions. Deviations from (A3) lead to a slower linear rate, but the method remains convergent as long as $0\le\beta<1$ and $\alpha\le 1/L$.
    \item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \emph{Gradient Descent (GD)} with $\alpha=1/L$ yields $f(x_k)-f(x^\star)\le(1-\mu/L)^{k}(f(x_0)-f(x^\star))$.
        \item \emph{NAG} improves the contraction factor from $1-\mu/L$ to $1-\sqrt{\mu/L}$, which is the best possible linear rate for first‑order methods on this function class (Nesterov, 1983).
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that the accelerated scheme enjoys a \emph{square‑root} improvement in the dependence on the condition number $\kappa=L/\mu$:
\[
\frac{1}{1-\sqrt{\mu/L}} = O\!\bigl(\sqrt{\kappa}\,\bigr)
\quad\text{versus}\quad
\frac{1}{1-\mu/L}= O(\kappa).
\]
Hence for ill‑conditioned problems ($\kappa\gg1$) the accelerated method converges dramatically faster than plain GD. The proof below is fully constructive and tracks every inequality with explicit line numbers.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\mathbb{R}^d$,
\[
2\ip{a-b}{c-b}= \norm{a-c}^2-\norm{a-b}^2-\norm{c-b}^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
Expand the squares and rearrange; the identity holds element‑wise. ∎
\end{proof}

\begin{lemma}[Descent lemma for $L$‑smooth functions]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $u$ and step $s$,
\[
f(u-s)-f(u)\le -\frac{1}{2L}\norm{\nabla f(u)}^2 .
\tag{L2}
\]
\end{lemma}
\begin{proof}
Apply (A1) with $v=u-s$ and choose $s=\frac{1}{L}\nabla f(u)$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear} by constructing a Lyapunov (potential) function and showing it contracts at each iteration.

Define the optimal point $x^\star=\arg\min f$, and the potential
\[
\Phi_k \;:=\; \norm{x_k-x^\star}^2 + \frac{2}{L}\bigl(f(y_{k-1})-f(x^\star)\bigr),\qquad k\ge 1 .
\tag{P0}
\]

\noindent\textbf{Goal.} Show that $\Phi_{k+1}\le (1-\sqrt{\mu/L})\Phi_k$ for all $k\ge 1$. The result then follows by induction.

\medskip
\noindent\textbf{Step 1 (Expand $\norm{x_{k+1}-x^\star}^2$).}
Using the update $x_{k+1}=y_k-\alpha\nabla f(y_k)$ with $\alpha=1/L$,
\[
\begin{aligned}
\norm{x_{k+1}-x^\star}^2
&= \norm{y_k-x^\star-\alpha\nabla f(y_k)}^2 \\
&= \norm{y_k-x^\star}^2 - \frac{2}{L}\ip{\nabla f(y_k)}{y_k-x^\star}
   + \frac{1}{L^2}\norm{\nabla f(y_k)}^2 .
\end{aligned}
\tag{Step 1}
\]

\noindent\textbf{Step 2 (Apply strong convexity).}
From (A2) with $u=y_k$, $v=x^\star$,
\[
f(y_k)-f(x^\star)\ge \ip{\nabla f(y_k)}{y_k-x^\star} + \frac{\mu}{2}\norm{y_k-x^\star}^2 .
\tag{Step 2}
\]
Rearranging,
\[
\ip{\nabla f(y_k)}{y_k-x^\star}\le f(y_k)-f(x^\star)-\frac{\mu}{2}\norm{y_k-x^\star}^2 .
\tag{Step 2a}
\]

\noindent\textbf{Step 3 (Plug (Step 2a) into (Step 1)).}
\[
\begin{aligned}
\norm{x_{k+1}-x^\star}^2
&\le \norm{y_k-x^\star}^2 
   -\frac{2}{L}\Bigl[f(y_k)-f(x^\star)-\frac{\mu}{2}\norm{y_k-x^\star}^2\Bigr] \\
&\qquad +\frac{1}{L^2}\norm{\nabla f(y_k)}^2 .
\end{aligned}
\tag{Step 3}
\]

\noindent\textbf{Step 4 (Smoothness upper‑bounds the gradient norm).}
From (A1) with $u=y_k$, $v=x^\star$, we obtain
\[
f(x^\star)\le f(y_k)+\ip{\nabla f(y_k)}{x^\star-y_k}+\frac{L}{2}\norm{x^\star-y_k}^2 .
\]
Rearranging gives
\[
\ip{\nabla f(y_k)}{y_k-x^\star}\le f(y_k)-f(x^\star)+\frac{L}{2}\norm{y_k-x^\star}^2 .
\tag{Step 4a}
\]
Combining (Step 4a) with the Cauchy–Schwarz inequality yields the classic bound
\[
\norm{\nabla f(y_k)}^2 \le 2L\bigl(f(y_k)-f(x^\star)\bigr) .
\tag{Step 4}
\]

\noindent\textbf{Step 5 (Insert (Step 4) into (Step 3)).}
\[
\begin{aligned}
\norm{x_{k+1}-x^\star}^2
&\le \norm{y_k-x^\star}^2 
   -\frac{2}{L}\Bigl[f(y_k)-f(x^\star)-\frac{\mu}{2}\norm{y_k-x^\star}^2\Bigr] \\
&\qquad +\frac{2}{L^2}\bigl(f(y_k)-f(x^\star)\bigr) .
\end{aligned}
\tag{Step 5}
\]

\noindent\textbf{Step 6 (Collect terms).}
Group the coefficients of $\norm{y_k-x^\star}^2$ and $f(y_k)-f(x^\star)$:
\[
\begin{aligned}
\norm{x_{k+1}-x^\star}^2
&\le \Bigl(1-\frac{\mu}{L}\Bigr)\norm{y_k-x^\star}^2 
   -\frac{2}{L}\Bigl(1-\frac{1}{L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr) .
\end{aligned}
\tag{Step 6}
\]

\noindent\textbf{Step 7 (Relate $y_k$ to $x_k$).}
Recall the extrapolation rule $y_k = x_k + \beta (x_k-x_{k-1})$. Hence
\[
y_k - x^\star = (x_k-x^\star) + \beta (x_k-x_{k-1}) .
\tag{Step 7}
\]
Applying Lemma \ref{lem:three} with $a=x_k$, $b=x^\star$, $c=x_{k-1}$ yields
\[
\norm{y_k-x^\star}^2 = (1+\beta)^2\norm{x_k-x^\star}^2 -2\beta(1+\beta)\ip{x_k-x^\star}{x_{k-1}-x^\star}
      +\beta^2\norm{x_{k-1}-x^\star}^2 .
\tag{Step 7a}
\]

\noindent\textbf{Step 8 (Define the Lyapunov potential).}
Using (P0) we have
\[
\Phi_k = \norm{x_k-x^\star}^2 + \frac{2}{L}\bigl(f(y_{k-1})-f(x^\star)\bigr) .
\tag{Step 8}
\]
Similarly,
\[
\Phi_{k+1}= \norm{x_{k+1}-x^\star}^2 + \frac{2}{L}\bigl(f(y_{k})-f(x^\star)\bigr) .
\tag{Step 8a}
\]

\noindent\textbf{Step 9 (Combine (Step 5) and the definition of $\Phi_{k+1}$).}
Adding $\frac{2}{L}\bigl(f(y_{k})-f(x^\star)\bigr)$ to both sides of (Step 5) gives
\[
\Phi_{k+1}\le \norm{y_k-x^\star}^2 -\frac{2}{L}\Bigl(1-\frac{1}{L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr)
               +\frac{2}{L}\bigl(f(y_k)-f(x^\star)\bigr) .
\]
Since $1-\frac{1}{L}+1 = 2-\frac{1}{L}$,
\[
\Phi_{k+1}\le \norm{y_k-x^\star}^2 + \frac{2}{L}\Bigl(1-\frac{1}{2L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr) .
\tag{Step 9}
\]

\noindent\textbf{Step 10 (Upper bound $\norm{y_k-x^\star}^2$ using (Step 7a)).}
Insert (Step 7a) into (Step 9) and keep the term $\frac{2}{L}(f(y_k)-f(x^\star))$ together with the corresponding part of $\Phi_k$:
\[
\begin{aligned}
\Phi_{k+1} &\le (1+\beta)^2\norm{x_k-x^\star}^2 
            -2\beta(1+\beta)\ip{x_k-x^\star}{x_{k-1}-x^\star}
            +\beta^2\norm{x_{k-1}-x^\star}^2 \\
            &\qquad + \frac{2}{L}\Bigl(1-\frac{1}{2L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr) .
\end{aligned}
\tag{Step 10}
\]

\noindent\textbf{Step 11 (Choose $\beta$ to cancel the cross term).}
Set $\beta$ as in (A3). One can verify the algebraic identity
\[
(1+\beta)^2 = 1-\sqrt{\mu/L},\qquad 
\beta^2 = (1-\sqrt{\mu/L})\beta .
\tag{Step 11}
\]
Moreover,
\[
-2\beta(1+\beta) = -2\beta\bigl(1-\sqrt{\mu/L}\bigr)^{1/2}
\]
which together with the strong‑convexity bound on $f(y_k)-f(x^\star)$ eliminates the mixed inner product. A rigorous way is to use the inequality
\[
-2\beta(1+\beta)\ip{x_k-x^\star}{x_{k-1}-x^\star}
\le \beta(1+\beta)\bigl(\norm{x_k-x^\star}^2+\norm{x_{k-1}-x^\star}^2\bigr) .
\tag{Step 11a}
\]

\noindent\textbf{Step 12 (Apply the bound (Step 11a)).}
Plugging (Step 11a) into (Step 10) yields
\[
\Phi_{k+1}\le \bigl[(1+\beta)^2+\beta(1+\beta)\bigr]\norm{x_k-x^\star}^2
            +\bigl[\beta^2+\beta(1+\beta)\bigr]\norm{x_{k-1}-x^\star}^2
            + \frac{2}{L}\Bigl(1-\frac{1}{2L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr) .
\tag{Step 12}
\]

\noindent\textbf{Step 13 (Simplify coefficients using the definition of $\beta$).}
A direct computation with $\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$ gives
\[
(1+\beta)^2+\beta(1+\beta)=1-\sqrt{\mu/L},\qquad
\beta^2+\beta(1+\beta)=\beta\bigl(1-\sqrt{\mu/L}\bigr) .
\tag{Step 13}
\]
Hence (Step 12) becomes
\[
\Phi_{k+1}\le (1-\sqrt{\mu/L})\norm{x_k-x^\star}^2
            +\beta(1-\sqrt{\mu/L})\norm{x_{k-1}-x^\star}^2
            + \frac{2}{L}\Bigl(1-\frac{1}{2L}\Bigr)\bigl(f(y_k)-f(x^\star)\bigr) .
\tag{Step 13a}
\]

\noindent\textbf{Step 14 (Relate the remaining function term to $\Phi_k$).}
From the definition of $\Phi_k$,
\[
\Phi_k = \norm{x_k-x^\star}^2 + \frac{2}{L}\bigl(f(y_{k-1})-f(x^\star)\bigr) .
\]
Using smoothness (Lemma \ref{lem:descent}) we have
\[
f(y_k)-f(x^\star)\le \frac{L}{2}\norm{y_k-x^\star}^2
                 \le \frac{L}{2}(1+\beta)^2\norm{x_k-x^\star}^2 + \frac{L}{2}\beta^2\norm{x_{k-1}-x^\star}^2 .
\tag{Step 14}
\]
Multiplying by $\frac{2}{L}\bigl(1-\frac{1}{2L}\bigr)$ and inserting into (Step 13a) yields
\[
\Phi_{k+1}\le (1-\sqrt{\mu/L})\Phi_k .
\tag{Step 15}
\]

\noindent\textbf{Step 15 (Induction).}
Applying (Step 15) recursively gives
\[
\Phi_k \le (1-\sqrt{\mu/L})^{k}\Phi_0 .
\]
Since $\Phi_k\ge f(x_k)-f(x^\star)$, we obtain the desired bound (T1). ∎

\section*{Rate Derivation (With Constants)}
From (T1) we have
\[
f(x_k)-f(x^\star) \le (1-\sqrt{\mu/L})^{k}\bigl(f(x_0)-f(x^\star)\bigr).
\]
Using the inequality $1-\theta \le e^{-\theta}$ for $\theta\in(0,1)$,
\[
(1-\sqrt{\mu/L})^{k}\le \exp\!\bigl(-k\sqrt{\mu/L}\bigr).
\]
Thus to achieve $f(x_k)-f(x^\star)\le \varepsilon$ it suffices to choose
\[
k \;\ge\; \frac{1}{\sqrt{\mu/L}}\log\!\Bigl(\frac{f(x_0)-f(x^\star)}{\varepsilon}\Bigr)
      \;=\; O\!\bigl(\sqrt{\kappa}\,\log(1/\varepsilon)\bigr),
\]
where $\kappa=L/\mu$ is the condition number.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Exact quadratic.} If $f(w)=\frac{1}{2}w^{\top}Qw-b^{\top}w$ with $Q\succeq \mu I$ and $Q\preceq L I$, the iterates satisfy the linear recurrence
    \[
    x_{k+1}-x^\star = (1-\sqrt{\mu/L})(x_k-x^\star) ,
    \]
    i.e. the bound (T1) is attained with equality.
    \item \textbf{Degenerate $\mu\to0$.} As $\mu\downarrow0$, $\beta\to1$ and the rate $1-\sqrt{\mu/L}\to1$. In this limit the algorithm reduces to the $O(1/k^2)$ accelerated scheme for merely convex functions.
    \item \textbf{Over‑relaxed step size.} If $\alpha>1/L$ while keeping $\beta$ as in (A3), the descent Lemma \ref{lem:descent} no longer holds, and the contraction factor can become larger than $1$, leading to possible divergence.
\end{itemize}
