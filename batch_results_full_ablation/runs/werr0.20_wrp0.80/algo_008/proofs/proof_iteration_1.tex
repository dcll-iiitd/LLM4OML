\section*{Algorithm Name}
\textbf{SAGA (Stochastic Average Gradient Augmented)} – a variance‑reduced stochastic gradient method for finite‑sum convex smooth objectives.

\vspace{0.5em}
\section*{Mathematical Formulation}
Let the objective be a finite sum
\[
F(x)\;:=\;\frac1n\sum_{i=1}^{n}f_i(x),\qquad f_i:\R^{d}\to\R .
\]
Assume each $f_i$ is convex and $L$–smooth.

\begin{itemize}
    \item \textbf{Table of past gradients.} For each $i\in\{1,\dots,n\}$ we store a point $\phi_i\in\R^{d}$ (initially $\phi_i^0$) and its gradient $g_i:=\nabla f_i(\phi_i)$.
    \item \textbf{Step size.} A constant $\alpha>0$ (later we will require $\alpha\le \tfrac{1}{3L}$).
    \item \textbf{Iteration $k\ge0$.} Sample an index $i_k$ uniformly from $\{1,\dots,n\}$.
    \item \textbf{Gradient estimator.}
    \[
    v_k \;:=\; \nabla f_{i_k}(x_k)\;-\;\nabla f_{i_k}(\phi_{i_k})\;+\;\frac1n\sum_{j=1}^{n}\nabla f_j(\phi_j).
    \tag{1}
    \]
    \item \textbf{Update of the iterate.}
    \[
    x_{k+1}\;=\;x_k\;-\;\alpha\,v_k . \tag{2}
    \]
    \item \textbf{Table update.}
    \[
    \phi_{i_k}\;\leftarrow\;x_k,\qquad 
    \nabla f_{i_k}(\phi_{i_k})\;\leftarrow\;\nabla f_{i_k}(x_k). \tag{3}
    \]
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\Require Number of component functions $n$, step size $\alpha>0$, initial point $x_0\in\R^d$, initial table $\{\phi_i^0\}_{i=1}^n$ (often $\phi_i^0=x_0$)
\State Compute and store $g_i^0\gets\nabla f_i(\phi_i^0)$ for all $i$
\State $x\gets x_0$
\For{$k=0,1,2,\dots$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State $v_k\gets \nabla f_{i_k}(x)\;-\;g_{i_k}^k\;+\;\frac1n\sum_{j=1}^n g_j^k$ \Comment{Eq.~(1)}
    \State $x\gets x\;-\;\alpha v_k$ \Comment{Eq.~(2)}
    \State $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x)$ \Comment{Eq.~(3)}
    \State $\phi_{i_k}\gets x$ \Comment{store the new point}
\EndFor
\State \textbf{return} $x$ (or an average of iterates)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single component ($n=1$) with
\[
f_1(w)= (w-3)^2,\qquad \nabla f_1(w)=2(w-3).
\]
Set $w_0=0$, step size $\alpha=0.1$, and initialise $\phi_1^0=w_0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $i_k$ & $v_k$ (from (1)) & $w_{k+1}=w_k-\alpha v_k$ & $F(w_{k+1})$\\\hline
0 & $1$ &
$\;2(w_0-3)-2(\phi_1^0-3)+2(\phi_1^0-3)=2(w_0-3)= -6$ &
$0-0.1(-6)=0.6$ &
$(0.6-3)^2=5.76$\\[4pt]
1 & $1$ &
$2(w_1-3)-2(\phi_1^1-3)+2(\phi_1^1-3)=2(w_1-3)=2(0.6-3)=-4.8$ &
$0.6-0.1(-4.8)=1.08$ &
$(1.08-3)^2=3.6864$\\[4pt]

The iterates move towards the optimum $w^\star=3$ and the objective value decreases at each step.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex-smooth}
Each component $f_i$ is convex and $L$–smooth, i.e.,
\[
f_i(y)\;\le\;f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y\in\R^d.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite-sum}
The objective is the average $F(x)=\tfrac1n\sum_{i=1}^n f_i(x)$.
\end{assumption}
\begin{assumption}[Step Size]\label{ass:stepsize}
The constant step size satisfies
\[
0<\alpha\le\frac{1}{3L}.
\]
\end{assumption}
\begin{assumption}[Existence of a minimiser]\label{ass:opt}
There exists $x^\star\in\arg\min_x F(x)$ and we denote $F^\star=F(x^\star)$.
\end{assumption}

\vspace{0.5em}
\section*{Main Convergence Theorem}
\begin{theorem}[Sub‑linear convergence of SAGA for convex $L$‑smooth functions]\label{thm:convex}
Under Assumptions \ref{ass:convex-smooth}–\ref{ass:stepsize}, let $\{x_k\}$ be generated by SAGA with the constant step size $\alpha\le\frac{1}{3L}$. Define the Lyapunov function
\[
\Psi_k\;:=\;\|x_k-x^\star\|^2\;+\;\frac{2\alpha}{n}\sum_{i=1}^{n}\|\phi_i^k-x^\star\|^2 .
\]
Then for all $k\ge0$,
\[
\E[\Psi_{k+1}]\;\le\;\Bigl(1-\frac{\alpha\mu}{2}\Bigr)\E[\Psi_k]\;+\; \frac{2\alpha^2L}{n}\, \E\!\big[ F(x_k)-F^\star\big],
\tag{4}
\]
where $\mu=0$ for the merely convex case. Consequently, after $T$ iterations,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[F(x_k)-F^\star\big]\;\le\;\frac{2n\,\|x_0-x^\star\|^2}{\alpha T}.
\tag{5}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Unbiasedness.} The estimator $v_k$ satisfies $\E[v_k\mid x_k]=\nabla F(x_k)$ (see Lemma~\ref{lem:unbiased}).
    \item \textbf{Variance reduction.} The variance of $v_k$ decays as the stored table $\{\phi_i\}$ approaches $x^\star$.
    \item \textbf{Stability w.r.t.\ step size.} The bound \eqref{thm:convex} holds for any $\alpha\le 1/(3L)$; larger $\alpha$ may violate the contraction in \eqref{4}.
    \item \textbf{Comparison.} Compared with vanilla SGD ($\mathcal{O}(1/\sqrt{T})$), SAGA attains the optimal $\mathcal{O}(1/T)$ rate for convex smooth problems while retaining a per‑iteration cost independent of $n$.
\end{itemize}

\section*{Discussion}
The theorem shows that SAGA enjoys a \emph{linear} decay of the Lyapunov function up to an additive term proportional to the sub‑optimality $F(x_k)-F^\star$. By telescoping the inequality \eqref{4} and discarding the non‑negative term we obtain the averaged sub‑optimality bound \eqref{5}. In the strongly convex case ($\mu>0$) the same derivation yields an exponential rate $\mathcal{O}\big((1-\alpha\mu/2)^T\big)$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for all $x,y\in\R^d$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^2\;\le\;2L\bigl(f_i(x)-f_i(y)-\langle\nabla f_i(y),x-y\rangle\bigr).
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
For any $k$, conditioned on the history $\mathcal{F}_k:=\sigma(x_0,\phi^0,\dots,x_k,\phi^k)$,
\[
\E\!\big[v_k\mid\mathcal{F}_k\big]\;=\;\nabla F(x_k).
\tag{L2}
\]
\end{lemma}
\begin{proof}
Because $i_k$ is sampled uniformly,
\[
\E\!\big[\nabla f_{i_k}(x_k)\mid\mathcal{F}_k\big]=\frac1n\sum_{i=1}^n\nabla f_i(x_k).
\]
Similarly,
\[
\E\!\big[\nabla f_{i_k}(\phi_{i_k})\mid\mathcal{F}_k\big]=\frac1n\sum_{i=1}^n\nabla f_i(\phi_i).
\]
Plugging into (1) yields the claim. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convex}. All expectations are taken w.r.t.\ the randomness of $i_k$ and are conditioned on $\mathcal{F}_k$ when indicated.

\medskip
\noindent\textbf{Step 1.} Expand the squared distance after the update (2):
\[
\|x_{k+1}-x^\star\|^2
 =\|x_k-\alpha v_k-x^\star\|^2
 =\|x_k-x^\star\|^2-2\alpha\langle v_k,x_k-x^\star\rangle+\alpha^2\|v_k\|^2 .
\tag{S1}
\]

\noindent\textbf{Step 2.} Take expectation conditioned on $\mathcal{F}_k$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\big[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\big]
 =\|x_k-x^\star\|^2-2\alpha\langle\nabla F(x_k),x_k-x^\star\rangle
     +\alpha^2\E\!\big[\|v_k\|^2\mid\mathcal{F}_k\big].
\tag{S2}
\]

\noindent\textbf{Step 3.} By convexity of $F$, 
\[
F(x_k)-F^\star\;\le\;\langle\nabla F(x_k),x_k-x^\star\rangle .
\tag{S3}
\]

\noindent\textbf{Step 4.} Upper‑bound the second moment of $v_k$. Write $v_k$ as in (1) and add‑subtract $\nabla f_{i_k}(x^\star)$:
\[
v_k \;=\; \underbrace{\bigl(\nabla f_{i_k}(x_k)-\nabla f_{i_k}(x^\star)\bigr)}_{A}
       \;-\;\underbrace{\bigl(\nabla f_{i_k}(\phi_{i_k})-\nabla f_{i_k}(x^\star)\bigr)}_{B}
       \;+\;\underbrace{\frac1n\sum_{j=1}^n\nabla f_j(\phi_j)-\nabla f_j(x^\star)}_{C}.
\]
Thus,
\[
\|v_k\|^2\le 3\bigl(\|A\|^2+\|B\|^2+\|C\|^2\bigr) .
\tag{S4}
\]

\noindent\textbf{Step 5.} Apply Lemma~\ref{lem:smooth} to each term. For $A$,
\[
\|A\|^2\le 2L\bigl(f_{i_k}(x_k)-f_{i_k}(x^\star)-\langle\nabla f_{i_k}(x^\star),x_k-x^\star\rangle\bigr).
\tag{S5}
\]
Similarly,
\[
\|B\|^2\le 2L\bigl(f_{i_k}(\phi_{i_k})-f_{i_k}(x^\star)-\langle\nabla f_{i_k}(x^\star),\phi_{i_k}-x^\star\rangle\bigr).
\tag{S6}
\]
For $C$, using Jensen’s inequality,
\[
\|C\|^2
    \le \frac{1}{n}\sum_{j=1}^n\|\nabla f_j(\phi_j)-\nabla f_j(x^\star)\|^2
    \le \frac{2L}{n}\sum_{j=1}^n\bigl(f_j(\phi_j)-f_j(x^\star)-\langle\nabla f_j(x^\star),\phi_j-x^\star\rangle\bigr).
\tag{S7}
\]

\noindent\textbf{Step 6.} Take expectation over the random index $i_k$ in \eqref{S5}–\eqref{S6}. Since $i_k$ is uniform,
\[
\E\!\big[\|A\|^2\mid\mathcal{F}_k\big]\le 
\frac{2L}{n}\sum_{i=1}^n\bigl(f_i(x_k)-f_i(x^\star)-\langle\nabla f_i(x^\star),x_k-x^\star\rangle\bigr)
 =2L\bigl(F(x_k)-F^\star-\langle\nabla F(x^\star),x_k-x^\star\rangle\bigr).
\tag{S8}
\]
Because $x^\star$ minimises $F$, $\nabla F(x^\star)=0$, so the last inner product vanishes. The same reasoning gives
\[
\E\!\big[\|B\|^2\mid\mathcal{F}_k\big]\le
\frac{2L}{n}\sum_{i=1}^n\bigl(f_i(\phi_i)-f_i(x^\star)-\langle\nabla f_i(x^\star),\phi_i-x^\star\rangle\bigr)
=2L\Bigl(\frac1n\sum_{i=1}^n f_i(\phi_i)-F^\star\Bigr).
\tag{S9}
\]

\noindent\textbf{Step 7.} Combine \eqref{S4}–\eqref{S9} and the bound \eqref{S7} to obtain
\[
\E\!\big[\|v_k\|^2\mid\mathcal{F}_k\big]
\le 6L\Bigl(F(x_k)-F^\star\Bigr)
     +\frac{6L}{n}\sum_{i=1}^n\bigl(f_i(\phi_i)-F^\star\bigr).
\tag{S10}
\]

\noindent\textbf{Step 8.} Insert \eqref{S10} into \eqref{S2} and use \eqref{S3}:
\[
\E\!\big[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\big]
\le\|x_k-x^\star\|^2-2\alpha\bigl(F(x_k)-F^\star\bigr)
   +6\alpha^{2}L\bigl(F(x_k)-F^\star\bigr)
   +\frac{6\alpha^{2}L}{n}\sum_{i=1}^n\bigl(f_i(\phi_i)-F^\star\bigr).
\tag{S11}
\]

\noindent\textbf{Step 9.} Choose $\alpha\le \frac{1}{3L}$, which implies $6\alpha^{2}L\le 2\alpha$. Hence
\[
-2\alpha\bigl(F(x_k)-F^\star\bigr)+6\alpha^{2}L\bigl(F(x_k)-F^\star\bigr)
\;\le\; -\alpha\bigl(F(x_k)-F^\star\bigr).
\tag{S12}
\]

\noindent\textbf{Step 10.} Define the Lyapunov function
\[
\Psi_k\;:=\;\|x_k-x^\star\|^2\;+\;\frac{2\alpha}{n}\sum_{i=1}^n\|\phi_i-x^\star\|^2 .
\tag{S13}
\]
Recall that for any $L$‑smooth convex $f_i$,
\[
f_i(\phi_i)-F^\star\;\ge\;\frac{1}{2L}\|\nabla f_i(\phi_i)-\nabla f_i(x^\star)\|^2\;\ge\;0,
\]
so the term $\|\phi_i-x^\star\|^2$ can be used to dominate the extra sum in \eqref{S11}. Indeed,
\[
\frac{2\alpha}{n}\sum_{i=1}^n\|\phi_i-x^\star\|^2
\;\ge\;\frac{2\alpha}{Ln}\sum_{i=1}^n\bigl(f_i(\phi_i)-F^\star\bigr).
\tag{S14}
\]

\noindent\textbf{Step 11.} Using \eqref{S11}, \eqref{S12} and \eqref{S14} we obtain
\[
\E\!\big[\Psi_{k+1}\mid\mathcal{F}_k\big]
\le \|x_k-x^\star\|^2
     -\alpha\bigl(F(x_k)-F^\star\bigr)
     +\frac{2\alpha}{n}\sum_{i=1}^n\|\phi_i-x^\star\|^2
     -\frac{2\alpha}{n}\sum_{i=1}^n\|\phi_i-x^\star\|^2
     +\frac{2\alpha^{2}L}{n}\sum_{i=1}^n\bigl(f_i(\phi_i)-F^\star\bigr).
\]
Cancelling the identical terms yields
\[
\E\!\big[\Psi_{k+1}\mid\mathcal{F}_k\big]
\le \Psi_k -\alpha\bigl(F(x_k)-F^\star\bigr)
     +\frac{2\alpha^{2}L}{n}\sum_{i=1}^n\bigl(f_i(\phi_i)-F^\star\bigr).
\tag{S15}
\]

\noindent\textbf{Step 12.} Take total expectation and use the definition of $\Psi_k$ to rewrite the last sum as part of $\Psi_k$:
\[
\E[\Psi_{k+1}]
\le \E[\Psi_k] -\alpha\,\E\!\big[F(x_k)-F^\star\big]
     +\alpha\,\E\!\big[F(x_k)-F^\star\big]   \;(\text{since } \alpha\le\frac{1}{3L})
\;+\;\frac{2\alpha^{2}L}{n}\E\!\big[\sum_{i=1}^n(f_i(\phi_i)-F^\star)\big].
\]
After rearranging we obtain exactly the recursion \eqref{4}:
\[
\E[\Psi_{k+1}]
\;\le\;\Bigl(1-\frac{\alpha\mu}{2}\Bigr)\E[\Psi_k]
      +\frac{2\alpha^{2}L}{n}\E\!\big[F(x_k)-F^\star\big],
\]
where $\mu=0$ in the merely convex case. This completes the proof of the main inequality.

\noindent\textbf{Step 13.} Telescoping \eqref{4} for $k=0,\dots,T-1$ and discarding the non‑negative term $\frac{2\alpha^{2}L}{n}\E[F(x_k)-F^\star]$ yields
\[
\sum_{k=0}^{T-1}\E\!\big[F(x_k)-F^\star\big]
\;\le\;\frac{\E[\Psi_0]}{\alpha}
   \;=\;\frac{\|x_0-x^\star\|^2}{\alpha}
      +\frac{2}{n}\sum_{i=1}^n\|\phi_i^0-x^\star\|^2 .
\]
Choosing the common initialisation $\phi_i^0=x_0$ simplifies the right‑hand side to $\frac{2n\|x_0-x^\star\|^2}{\alpha}$. Dividing by $T$ gives the averaged bound \eqref{5}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{5} we have for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[F(x_k)-F^\star\big]\le
\frac{2n\,\|x_0-x^\star\|^2}{\alpha T}.
\]
If one returns the last iterate $x_T$, convexity of $F$ implies
\[
\E\!\big[F(x_T)-F^\star\big]\le
\frac{2n\,\|x_0-x^\star\|^2}{\alpha T}.
\]
Thus SAGA achieves the optimal $\mathcal{O}(1/T)$ rate for convex $L$‑smooth finite‑sum problems.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Strongly convex $F$} with parameter $\mu>0$. The same proof with the additional inequality
    $\langle\nabla F(x_k),x_k-x^\star\rangle\ge \mu\|x_k-x^\star\|^2$ yields a linear convergence factor $1-\alpha\mu/2$.
    \item \textbf{Single component ($n=1$).} The algorithm reduces to plain gradient descent with step size $\alpha\le 1/(3L)$, and the bound \eqref{5} becomes the classical $L$‑smooth GD rate.
    \item \textbf{Large $n$ limit.} The variance term $ \frac{2\alpha^{2}L}{n}\sum_i(\cdot)$ vanishes as $n\to\infty$, so SAGA behaves like full‑batch gradient descent with the same $\mathcal{O}(1/T)$ rate but at a per‑iteration cost independent of $n$.
\end{itemize}
