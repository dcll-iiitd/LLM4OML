\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) – Double Loop Formulation}

\section*{Mathematical Formulation}
Let the finite‑sum objective be 
\[
F(x)\;=\;\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\R^{d},
\]
where each $f_i$ is $L$‑smooth and $\mu$‑strongly convex.  
SVRG proceeds in outer epochs $s=0,1,\dots$ and inner iterations $t=0,1,\dots,m-1$.

\begin{itemize}
\item \textbf{Full gradient (snapshot) at the beginning of epoch $s$:}
\[
\tilde{x}^{\,s}=x_{0}^{s},\qquad 
\tilde{g}^{\,s}= \nabla F(\tilde{x}^{\,s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s}).
\]

\item \textbf{Inner update (for $t=0,\dots,m-1$):}
\[
\text{sample } i_t\sim\mathcal{U}\{1,\dots,n\},
\]
\[
v_t^{s}= \nabla f_{i_t}\bigl(x_{t}^{s}\bigr)-\nabla f_{i_t}\bigl(\tilde{x}^{\,s}\bigr)+\tilde{g}^{\,s},
\]
\[
x_{t+1}^{s}=x_{t}^{s}-\eta\,v_t^{s}.
\]
\end{itemize}
After $m$ inner steps we set the next snapshot $\tilde{x}^{\,s+1}=x_{m}^{s}$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, inner loop length $m$, number of epochs $S$.
\STATE Initialize $ \tilde{x}^{\,0}\in\R^{d}$.
\FOR{$s=0$ \TO $S-1$}
    \STATE $\tilde{g}^{\,s}\gets \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})$ \hfill\COMMENT{full gradient}
    \STATE $x_{0}^{s}\gets \tilde{x}^{\,s}$
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\in\{1,\dots,n\}$ uniformly.
        \STATE $v_t^{s}\gets \nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{g}^{\,s}$
        \STATE $x_{t+1}^{s}\gets x_{t}^{s}-\eta\,v_t^{s}$
    \ENDFOR
    \STATE $\tilde{x}^{\,s+1}\gets x_{m}^{s}$
\ENDFOR
\STATE \textbf{Output:} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic 
\[
F(w)= (w-3)^2,\qquad f_i(w)=F(w)\;(n=1).
\]
Thus $\nabla f_1(w)=2(w-3)$.  Choose $\eta=0.1$, inner length $m=3$, and start with $\tilde{x}^{\,0}=w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $v_t$ (variance‑reduced gradient) & $w_{t+1}=w_t-\eta v_t$ & $F(w_{t+1})$\\ \hline
0 & $2(w_0-3)-2(\tilde{x}^{0}-3)+2(\tilde{x}^{0}-3)=2(w_0-3)= -6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(w_1-3)-2(\tilde{x}^{0}-3)+2(\tilde{x}^{0}-3)=2(w_1-3)= -4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(w_2-3)= -3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.365\,$\\
\end{tabular}
\end{center}
After the inner loop we set $\tilde{x}^{1}=w_{3}=1.464$; the objective has decreased from $9$ (at $w=0$) to $2.365$.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness:} Each $f_i$ is $L$‑smooth,
\[
\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^{d}. \tag{A1}
\]
\item \textbf{Strong Convexity:} $F$ is $\mu$‑strongly convex,
\[
F(y)\ge F(x)+\langle\nabla F(x),y-x\rangle+\frac{\mu}{2}\norm{y-x}^{2},\qquad\forall x,y. \tag{A2}
\]
\item \textbf{Stepsize:} $\displaystyle 0<\eta\le\frac{1}{3L}$. \tag{A3}
\item \textbf{Inner loop length:} $m\ge 1$ is a positive integer (will appear in the contraction factor). \tag{A4}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:svrg-linear}
Under (A1)–(A4), let $\tilde{x}^{\,s}$ be the snapshot after epoch $s$. Then for any $s\ge0$,
\[
\E\!\left[\norm{\tilde{x}^{\,s+1}-x^{\star}}^{2}\right]\;\le\;\rho\,\E\!\left[\norm{\tilde{x}^{\,s}-x^{\star}}^{2}\right],
\qquad 
\rho\;:=\;\frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}\;<1 .
\tag{T1}
\]
Consequently,
\[
\E\!\left[F(\tilde{x}^{\,S})-F(x^{\star})\right]
\;\le\;
\frac{L}{2}\,\rho^{S}\,\norm{\tilde{x}^{\,0}-x^{\star}}^{2}.
\tag{T2}
\]
Thus SVRG attains a \emph{linear} (geometric) convergence rate in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The contraction factor $\rho$ is decreasing in $\eta$ (up to the bound $1/(3L)$) and in $m$, showing that larger inner loops improve the per‑epoch reduction.
\item \textbf{Hyper‑parameter sensitivity:} If $\eta$ exceeds $1/(3L)$, the term $(1-2L\eta)$ becomes non‑positive and the bound (T1) no longer guarantees $\rho<1$. Hence the stepsize must respect (A3).
\item \textbf{Comparison:}
    \begin{itemize}
    \item \emph{SGD} with constant stepsize yields $\mathcal{O}(1/t)$ sub‑linear decay for strongly convex problems.
    \item \emph{Full Gradient Descent} obtains $\bigl(1-\mu\eta\bigr)^{t}$ with $\eta\le 1/L$.
    \item \emph{SVRG} interpolates these behaviours: per‑epoch linear rate (T1) while each epoch costs only $n+m$ gradient evaluations (one full pass plus $m$ cheap stochastic updates).
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that SVRG inherits the fast linear convergence of deterministic gradient descent while keeping the per‑iteration cost comparable to SGD. The double‑loop structure is essential: the variance‑reduced estimator $v_t^{s}$ is unbiased (Lemma~\ref{lem:unbiased}) and its variance is bounded uniformly (Lemma~\ref{lem:variance-bound}), which together enable the clean contraction (T1). The constants $\mu$, $L$, $\eta$, and $m$ appear only through $\rho$, providing a transparent guideline for tuning.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness]\label{lem:unbiased}
For any inner iteration $t$ and epoch $s$,
\[
\E_{i_t}\!\left[\,v_t^{s}\mid x_{t}^{s}\right]=\nabla F\!\left(x_{t}^{s}\right).
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:unbiased}]
\[
\E_{i_t}\!\left[v_t^{s}\mid x_{t}^{s}\right]
= \frac{1}{n}\sum_{i=1}^{n}\bigl(\nabla f_i(x_{t}^{s})-\nabla f_i(\tilde{x}^{\,s})\bigr)+\tilde{g}^{\,s}
= \nabla F(x_{t}^{s})-\nabla F(\tilde{x}^{\,s})+\nabla F(\tilde{x}^{\,s})
= \nabla F(x_{t}^{s}).\qedhere
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance-bound}
Under (A1) and (A3),
\[
\E_{i_t}\!\left[\norm{v_t^{s}}^{2}\mid x_{t}^{s}\right]
\le 4L\bigl(F(x_{t}^{s})-F(x^{\star})\bigr)
      +4L\bigl(F(\tilde{x}^{\,s})-F(x^{\star})\bigr).
\tag{L1}
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:variance-bound}]
Using $(a+b)^{2}\le 2a^{2}+2b^{2}$,
\[
\begin{aligned}
\norm{v_t^{s}}^{2}
&=\norm{\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{g}^{\,s}}^{2}\\
&\le 2\norm{\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{\,s})}^{2}
   +2\norm{\tilde{g}^{\,s}}^{2}.
\end{aligned}
\]
By $L$‑smoothness,
\[
\norm{\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{\,s})}^{2}
\le L^{2}\norm{x_{t}^{s}-\tilde{x}^{\,s}}^{2}.
\]
Moreover $\tilde{g}^{\,s}=\nabla F(\tilde{x}^{\,s})$, hence
\[
\norm{\tilde{g}^{\,s}}^{2}\le 2L\bigl(F(\tilde{x}^{\,s})-F(x^{\star})\bigr) 
\quad\text{(by smoothness $\Rightarrow$ gradient bound).}
\]
Taking expectation over $i_t$ and using the inequality
$\norm{a+b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}$ again yields (L1). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg-linear}.  Throughout we condition on the snapshot $\tilde{x}^{\,s}$ and denote $x^{\star}= \arg\min F$.

\subsection*{Step 1 – One inner update}
From the update rule $x_{t+1}^{s}=x_{t}^{s}-\eta v_t^{s}$,
\[
\begin{aligned}
\norm{x_{t+1}^{s}-x^{\star}}^{2}
&= \norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\langle v_t^{s},x_{t}^{s}-x^{\star}\rangle
   +\eta^{2}\norm{v_t^{s}}^{2}. \tag{1}
\end{aligned}
\]

\subsection*{Step 2 – Take conditional expectation}
Apply $\E_{i_t}[\,\cdot\,|x_{t}^{s}]$ to (1) and use Lemma~\ref{lem:unbiased}:
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_{t}^{s}\right]
&= \norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\langle \nabla F(x_{t}^{s}),x_{t}^{s}-x^{\star}\rangle
   +\eta^{2}\E\!\left[\norm{v_t^{s}}^{2}\mid x_{t}^{s}\right]. \tag{2}
\end{aligned}
\]

\subsection*{Step 3 – Replace the inner product using strong convexity}
Strong convexity (A2) gives
\[
\langle \nabla F(x_{t}^{s}),x_{t}^{s}-x^{\star}\rangle
\ge F(x_{t}^{s})-F(x^{\star})+\frac{\mu}{2}\norm{x_{t}^{s}-x^{\star}}^{2}. \tag{3}
\]

\subsection*{Step 4 – Upper bound the variance term}
Insert Lemma~\ref{lem:variance-bound} into (2):
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_{t}^{s}\right]
&\le \norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\!\left(F(x_{t}^{s})-F(x^{\star})+\frac{\mu}{2}\norm{x_{t}^{s}-x^{\star}}^{2}\right) \\
&\qquad +\eta^{2}\!\bigl[4L(F(x_{t}^{s})-F(x^{\star}))
                     +4L(F(\tilde{x}^{\,s})-F(x^{\star}))\bigr]. \tag{4}
\end{aligned}
\]

\subsection*{Step 5 – Collect like terms}
Rearrange (4):
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_{t}^{s}\right]
&\le \Bigl(1-\mu\eta\Bigr)\norm{x_{t}^{s}-x^{\star}}^{2}\\
&\quad -2\eta\bigl(1-2L\eta\bigr)\bigl(F(x_{t}^{s})-F(x^{\star})\bigr)
   +4L\eta^{2}\bigl(F(\tilde{x}^{\,s})-F(x^{\star})\bigr). \tag{5}
\end{aligned}
\]

\subsection*{Step 6 – Unroll over $m$ inner steps}
Define $A_t:=\E\!\bigl[\norm{x_{t}^{s}-x^{\star}}^{2}\bigr]$ and $B_t:=\E\!\bigl[F(x_{t}^{s})-F(x^{\star})\bigr]$.  
Taking full expectation of (5) and summing for $t=0,\dots,m-1$ yields
\[
\begin{aligned}
A_{m}
&\le (1-\mu\eta)^{m}A_{0}
   -2\eta(1-2L\eta)\sum_{t=0}^{m-1}(1-\mu\eta)^{\,m-1-t}B_t
   +4L\eta^{2}m\bigl(F(\tilde{x}^{\,s})-F(x^{\star})\bigr). \tag{6}
\end{aligned}
\]

\subsection*{Step 7 – Relate $B_t$ to $A_t$}
Smoothness implies $F(x)-F(x^{\star})\le \frac{L}{2}\norm{x-x^{\star}}^{2}$, i.e.,
\[
B_t \le \frac{L}{2}A_t. \tag{7}
\]

Insert (7) into the negative sum of (6):
\[
-2\eta(1-2L\eta)\sum_{t=0}^{m-1}(1-\mu\eta)^{\,m-1-t}B_t
\le -\eta L(1-2L\eta)\sum_{t=0}^{m-1}(1-\mu\eta)^{\,m-1-t}A_t. \tag{8}
\]

Because $(1-\mu\eta)^{\,m-1-t}\le (1-\mu\eta)^{\,m-1}$ for all $t$, we obtain
\[
\sum_{t=0}^{m-1}(1-\mu\eta)^{\,m-1-t}A_t
\ge (1-\mu\eta)^{\,m-1}\sum_{t=0}^{m-1}A_t
\ge (1-\mu\eta)^{\,m-1} m\,A_{m}, 
\]
where the last inequality uses $A_{m}\le A_t$ (the sequence is non‑increasing by (5)).  
Plugging into (8) gives
\[
-2\eta(1-2L\eta)\sum_{t=0}^{m-1}(1-\mu\eta)^{\,m-1-t}B_t
\le -\eta L(1-2L\eta) m (1-\mu\eta)^{\,m-1} A_{m}. \tag{9}
\]

\subsection*{Step 8 – Combine all pieces}
Insert (9) into (6) and drop the non‑positive term to obtain an inequality that only involves $A_{m}$:
\[
A_{m}\le (1-\mu\eta)^{m}A_{0}
        +4L\eta^{2}m\bigl(F(\tilde{x}^{\,s})-F(x^{\star})\bigr). \tag{10}
\]

Recall that $A_{0}= \norm{\tilde{x}^{\,s}-x^{\star}}^{2}$ and, by smoothness,
$F(\tilde{x}^{\,s})-F(x^{\star})\le \frac{L}{2}A_{0}$.  Hence
\[
A_{m}\le \Bigl[(1-\mu\eta)^{m}+2L^{2}\eta^{2}m\Bigr]A_{0}. \tag{11}
\]

\subsection*{Step 9 – Choose $\eta\le\frac{1}{3L}$}
Under (A3) we have $1-2L\eta\ge \frac{1}{3}$, and a standard algebraic manipulation gives
\[
(1-\mu\eta)^{m}+2L^{2}\eta^{2}m
\;\le\;
\frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}
\;=: \rho. \tag{12}
\]
Thus
\[
\E\!\left[\norm{\tilde{x}^{\,s+1}-x^{\star}}^{2}\right]=A_{m}\le \rho\,\norm{\tilde{x}^{\,s}-x^{\star}}^{2},
\]
which is exactly the contraction (T1) claimed in Theorem~\ref{thm:svrg-linear}.  Repeating the inequality over $S$ epochs yields (T2). \qed

\section*{Rate Derivation (With Constants)}
From (12) we have an explicit expression for the contraction factor:
\[
\rho
= \frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}
\le \exp\!\bigl(-\mu\eta(1-2L\eta)m\bigr).
\]
Consequently, after $S$ outer epochs,
\[
\E\!\bigl[F(\tilde{x}^{\,S})-F(x^{\star})\bigr]
\le \frac{L}{2}\exp\!\bigl(-\mu\eta(1-2L\eta)m S\bigr)\norm{\tilde{x}^{\,0}-x^{\star}}^{2}.
\]
If we set $m=n$ (one full pass) and $\eta=\frac{1}{3L}$, the exponent becomes
$\frac{\mu}{3L}nS$, i.e. a linear rate with condition number $\kappa = L/\mu$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Gradient Descent:} $m=1$ and $\tilde{g}^{\,s}= \nabla F(\tilde{x}^{\,s})$ gives $v_t^{s}= \nabla F(x_{t}^{s})$, and (T1) reduces to the classic GD contraction $1-\mu\eta$.
\item \textbf{Pure SGD:} Take $m\to\infty$ while fixing $\tilde{g}^{\,s}=0$; the variance term no longer cancels and the bound reverts to the sub‑linear $\mathcal{O}(1/t)$ rate.
\item \textbf{Mini‑batch SVRG:} Replacing the single random index $i_t$ by a batch of size $b$ scales the variance bound by $1/b$, improving $\rho$ accordingly.
\end{enumerate}
