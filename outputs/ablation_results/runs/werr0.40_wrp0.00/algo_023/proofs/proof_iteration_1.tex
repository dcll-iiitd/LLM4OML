\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be convex and differentiable with $L$-Lipschitz continuous gradient,
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad\forall u,v\in\R^{d}.
\]
Let $g:\R^{d}\to\R\cup\{+\infty\}$ be a proper, closed, convex (possibly non‑smooth) function.
Define the composite objective
\[
F(x):=f(x)+g(x),\qquad x\in\R^{d}.
\]
For a stepsize $\eta>0$ the proximal operator of $g$ is
\[
\prox_{\eta g}(z):=\arg\min_{u\in\R^{d}}\Big\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Big\}.
\]
The \textbf{proximal gradient update} reads
\begin{equation}
\label{eq:update}
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),\qquad k=0,1,2,\dots
\end{equation}
We shall assume throughout that the stepsize satisfies
\begin{equation}
\label{eq:stepsize}
0<\eta\le \frac{1}{L}.
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta>0$ satisfying \eqref{eq:stepsize}, max iter $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_{k}:=\nabla f(x_{k})$.
    \State $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta g_{k}\bigr)$.
\EndFor
\State \Return $x_{K}$ (or the best iterate seen).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a one‑dimensional problem with  
\[
f(w)=(w-3)^{2},\qquad g(w)=0\;( \text{so }\prox_{\eta g}= \text{id}),
\]
stepsize $\eta=0.1$, and initial point $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values $F(w_{k})=f(w_{k})$ decrease:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.7712)=1.5635.
\]

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]
\label{ass:smooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad\forall u,v\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]
\label{ass:gconvex}
$g$ is proper, closed and convex (possibly non‑smooth). Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]
\label{ass:stepsize}
The stepsize $\eta$ satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Existence of a minimizer]
\label{ass:opt}
The composite problem $\min_{x\in\R^{d}}F(x)$ admits at least one optimal solution $x^{\star}$, and $F(x^{\star})$ is finite.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear $O(1/k)$ rate]
\label{thm:sublinear}
Under Assumptions~\ref{ass:smooth}--\ref{ass:opt} and with a stepsize $\eta\in(0,1/L]$, the iterates generated by \eqref{eq:update} satisfy for all $k\ge1$
\[
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,k}.
\]
Consequently $F(x_{k})\to F(x^{\star})$ and any limit point of $\{x_{k}\}$ is an optimal solution.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The non‑expansiveness of $\prox_{\eta g}$ guarantees that the mapping $x\mapsto \prox_{\eta g}(x-\eta\nabla f(x))$ is firmly non‑expansive for $\eta\le1/L$, preventing divergence even when $g$ is non‑smooth.
\item \textbf{Hyperparameter sensitivity:} The bound \eqref{eq:stepsize} is tight; choosing $\eta>1/L$ may violate the descent property (see Step~[5] of the proof). Smaller $\eta$ yields a larger denominator in the rate, i.e. slower convergence, but improves numerical stability.
\item \textbf{Comparison to gradient descent:} When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and Theorem~\ref{thm:sublinear} recovers the classical $O(1/k)$ rate for gradient descent on smooth convex functions.
\item \textbf{Extension to accelerated schemes:} By embedding Nesterov’s momentum, the $O(1/k^{2})$ rate can be achieved while still relying on the same proximal step.
\end{itemize}

\section*{Discussion}
The proximal gradient method unifies gradient descent (smooth part) and proximal mapping (non‑smooth part). The proof below is entirely deterministic, uses only the non‑expansiveness of the proximal operator and the descent lemma for $f$. No stochastic assumptions are required.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]
\label{lem:descent}
For any $u,v\in\R^{d}$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; omitted for brevity. \qedhere
\end{proof}

\begin{lemma}[Non‑expansiveness of the proximal operator]
\label{lem:prox_ne}
For any $u,v\in\R^{d}$ and $\eta>0$,
\[
\big\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\big\|\le\|u-v\|.
\]
\end{lemma}
\begin{proof}
Follows from firm non‑expansiveness of the resolvent of a maximal monotone operator. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:sublinear}. Throughout the proof we fix an optimal solution $x^{\star}\in\arg\min F$.

\begin{proof}
Let $x_{k+1}$ be defined by \eqref{eq:update}. Define the auxiliary point
\[
y_{k}:=x_{k}-\eta\nabla f(x_{k}) .
\]
By definition of the proximal operator,
\[
x_{k+1}= \prox_{\eta g}(y_{k}) = \arg\min_{u}\bigl\{ g(u)+\tfrac{1}{2\eta}\|u-y_{k}\|^{2}\bigr\}.
\]
The optimality condition of this strongly convex subproblem yields
\begin{equation}
\label{eq:opt_cond}
0\in \partial g(x_{k+1})+\frac{1}{\eta}(x_{k+1}-y_{k}),
\qquad\text{or equivalently}\qquad
\frac{1}{\eta}(y_{k}-x_{k+1})\in\partial g(x_{k+1}).
\end{equation}

\medskip
\noindent\textbf{[Step 1]}  Use convexity of $g$:
\[
g(x^{\star})\ge g(x_{k+1})+\langle s_{k+1},\,x^{\star}-x_{k+1}\rangle,
\quad\text{with } s_{k+1}:=\frac{1}{\eta}(y_{k}-x_{k+1})\in\partial g(x_{k+1}) \text{ by } \eqref{eq:opt_cond}.
\tag{1}
\]

\noindent\textbf{[Step 2]}  Rearrange (1) and add $f(x^{\star})$ to both sides:
\[
F(x^{\star})\ge f(x^{\star})+g(x_{k+1})+\langle s_{k+1},\,x^{\star}-x_{k+1}\rangle .
\tag{2}
\]

\noindent\textbf{[Step 3]}  Apply the descent lemma (Lemma~\ref{lem:descent}) with $u=x_{k}$ and $v=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{3}
\]

\noindent\textbf{[Step 4]}  Combine (2) and (3) to bound the optimality gap:
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&= f(x_{k+1})+g(x_{k+1})-F(x^{\star})\\
&\le f(x_{k})+g(x_{k+1})-F(x^{\star})
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&\le \underbrace{\bigl[f(x_{k})-f(x^{\star})\bigr]}_{\text{(A)}}
   +\underbrace{\langle s_{k+1},x_{k+1}-x^{\star}\rangle}_{\text{(B)}}
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2},
\end{aligned}
\tag{4}
\]
where we used (2) to replace $-F(x^{\star})$ by $-f(x^{\star})-g(x_{k+1})-\langle s_{k+1},x^{\star}-x_{k+1}\rangle$ and rearranged signs.

\noindent\textbf{[Step 5]}  Observe that $s_{k+1}=\frac{1}{\eta}(y_{k}-x_{k+1})$ and $y_{k}=x_{k}-\eta\nabla f(x_{k})$. Hence
\[
s_{k+1}= \frac{1}{\eta}\bigl(x_{k}-\eta\nabla f(x_{k})-x_{k+1}\bigr)
       =\frac{x_{k}-x_{k+1}}{\eta}-\nabla f(x_{k}).
\tag{5}
\]

\noindent\textbf{[Step 6]}  Substitute (5) into the sum of the inner products in (4):
\[
\begin{aligned}
\langle s_{k+1},x_{k+1}-x^{\star}\rangle
+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
&= \Big\langle\frac{x_{k}-x_{k+1}}{\eta}-\nabla f(x_{k}),\,x_{k+1}-x^{\star}\Big\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle\\
&= \frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
   -\langle\nabla f(x_{k}),x_{k+1}-x^{\star}\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle\\
&= \frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
   -\langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 7]}  Insert (6) into (4):
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le \bigl[f(x_{k})-f(x^{\star})\bigr]
   +\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
   -\langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{7}
\]

\noindent\textbf{[Step 8]}  Use convexity of $f$:
\[
f(x_{k})-f(x^{\star})\le\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle.
\tag{8}
\]

\noindent\textbf{[Step 9]}  Plug (8) into (7); the terms $\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle$ cancel:
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le \frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 10]}  Expand the inner product:
\[
\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
= \frac{1}{2}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\Bigr).
\tag{10}
\]

\noindent\textbf{[Step 11]}  Substitute (10) into (9):
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le \frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}
                         -\|x_{k+1}-x_{k}\|^{2}\Bigr)
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&= \frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Bigr)
   +\frac{1}{2}\Bigl(L-\frac{1}{\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{11}
\]

\noindent\textbf{[Step 12]}  By the stepsize condition \eqref{eq:stepsize}, $L-\frac{1}{\eta}\le 0$, therefore the second term on the right‑hand side is non‑positive and can be dropped:
\[
F(x_{k+1})-F(x^{\star})\le \frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\tag{12}
\]

\noindent\textbf{[Step 13]}  Rearranging (12) gives a telescoping inequality:
\[
2\eta\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}.
\tag{13}
\]

\noindent\textbf{[Step 14]}  Sum (13) from $k=0$ to $k=K-1$:
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}
\le \|x_{0}-x^{\star}\|^{2}.
\tag{14}
\]

\noindent\textbf{[Step 15]}  Since each term $F(x_{k+1})-F(x^{\star})\ge0$, the average is bounded by the last term:
\[
F(x_{K})-F(x^{\star})\le \frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{15}
\]

\noindent\textbf{[Step 16]}  Setting $k=K$ yields exactly the claimed bound:
\[
F(x_{k})-F(x^{\star})\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,k},\qquad\forall k\ge1.
\tag{16}
\]

Thus the sequence $\{F(x_{k})\}$ converges to $F(x^{\star})$ at the rate $O(1/k)$. Moreover, the boundedness of $\{x_{k}\}$ follows from (13), implying existence of limit points; any such point satisfies the optimality condition and therefore belongs to $\arg\min F$. \qed
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{eq:stepsize} we have $0<\eta\le 1/L$. The constant in the $O(1/k)$ bound is explicitly $\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}$. Choosing the largest admissible stepsize $\eta=1/L$ yields the tightest bound:
\[
F(x_{k})-F(x^{\star})\le \frac{L\,\|x_{0}-x^{\star}\|^{2}}{2k}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure smooth case ($g\equiv0$).} The proximal step reduces to the identity, and the proof collapses to the classical gradient descent analysis with the same $O(1/k)$ rate.
\item \textbf{Indicator of a convex set $C$ ($g=\iota_{C}$).} Then $\prox_{\eta g}$ is the Euclidean projection onto $C$. The algorithm becomes projected gradient descent, and the same rate holds.
\item \textbf{L1‑regularization ($g(x)=\lambda\|x\|_{1}$).} The proximal operator is soft‑thresholding; the bound remains unchanged, providing a simple guarantee for sparse optimisation problems.
\end{itemize}
