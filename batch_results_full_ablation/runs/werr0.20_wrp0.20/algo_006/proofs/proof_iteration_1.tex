\section*{SVRG for Non‑Convex Optimization}

\section*{Mathematical Formulation}
Let 
\[
f(\mathbf{x})\;=\;\frac1n\sum_{i=1}^{n} f_i(\mathbf{x}),\qquad 
\mathbf{x}\in\RR^{d},
\]
where each component $f_i$ is differentiable and $L$–smooth (i.e., $\|\nabla f_i(\mathbf{u})-\nabla f_i(\mathbf{v})\|\le L\|\mathbf{u}-\mathbf{v}\|$ for all $\mathbf{u},\mathbf{v}$).  
The algorithm proceeds in \emph{epochs}.  
At the beginning of epoch $s$ we fix a \emph{snapshot point} $\tilde{\mathbf{x}}^{\,s}$ and compute the \emph{full gradient}
\[
\tilde{\mathbf{g}}^{\,s}\;:=\;\nabla f(\tilde{\mathbf{x}}^{\,s})\;=\;\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{\mathbf{x}}^{\,s}).
\]

Inside epoch $s$ we perform $m$ inner iterations.  
Given the current iterate $\mathbf{x}_t$, we draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and form the \emph{variance‑reduced estimator}
\[
\mathbf{v}_t\;:=\;\nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{\,s})+\tilde{\mathbf{g}}^{\,s}.
\tag{1}
\]
The update is a standard stochastic gradient step
\[
\mathbf{x}_{t+1}\;=\;\mathbf{x}_t-\eta\,\mathbf{v}_t,
\tag{2}
\]
where $\eta>0$ is a constant stepsize.  
After $m$ inner steps we set the snapshot for the next epoch as the last inner iterate:
\[
\tilde{\mathbf{x}}^{\,s+1}\;:=\;\mathbf{x}_{m}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex)}\label{alg:svrg}
\begin{algorithmic}[1]
\REQUIRE Initial point $\mathbf{x}_0$, stepsize $\eta$, epoch length $m$, number of epochs $S$
\STATE $\tilde{\mathbf{x}}^{\,0}\gets\mathbf{x}_0$
\FOR{$s=0,\dots,S-1$}
    \STATE Compute full gradient $\tilde{\mathbf{g}}^{\,s}\gets \nabla f(\tilde{\mathbf{x}}^{\,s})$
    \STATE $\mathbf{x}_0^{(s)}\gets \tilde{\mathbf{x}}^{\,s}$
    \FOR{$t=0,\dots,m-1$}
        \STATE Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \STATE $\mathbf{v}_t\gets \nabla f_{i_t}(\mathbf{x}_t^{(s)})-\nabla f_{i_t}(\tilde{\mathbf{x}}^{\,s})+\tilde{\mathbf{g}}^{\,s}$ \COMMENT{(1)}
        \STATE $\mathbf{x}_{t+1}^{(s)}\gets \mathbf{x}_t^{(s)}-\eta\,\mathbf{v}_t$ \COMMENT{(2)}
    \ENDFOR
    \STATE $\tilde{\mathbf{x}}^{\,s+1}\gets \mathbf{x}_m^{(s)}$
\ENDFOR
\RETURN $\hat{\mathbf{x}}\;$ (any iterate uniformly chosen from all inner iterates)
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(\mathbf{u})-\nabla f_i(\mathbf{v})\|\le L\|\mathbf{u}-\mathbf{v}\|,\qquad\forall\mathbf{u},\mathbf{v}\in\RR^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded Variance at Snapshot]\label{ass:var}
For any snapshot $\tilde{\mathbf{x}}$, the stochastic gradient variance is bounded:
\[
\EE_{i\sim\mathcal{U}[n]}\bigl[\|\nabla f_i(\tilde{\mathbf{x}})-\nabla f(\tilde{\mathbf{x}})\|^{2}\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Lower Bounded Objective]\label{ass:lb}
The function $f$ is bounded below, i.e., there exists $f^{\inf}\in\RR$ such that $f(\mathbf{x})\ge f^{\inf}$ for all $\mathbf{x}$.
\end{assumption}
All constants $L,\sigma$ are known; the stepsize will be chosen as
\[
0<\eta\le\frac{1}{3L}.
\tag{3}
\]

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:lb} and stepsize satisfying (3), let $\{\tilde{\mathbf{x}}^{\,s}\}_{s=0}^{S}$ be the sequence of snapshots produced by Algorithm~\ref{alg:svrg} with epoch length $m$. Define the random output $\hat{\mathbf{x}}$ as a uniform choice among all $Sm$ inner iterates. Then
\[
\EE\bigl[\|\nabla f(\hat{\mathbf{x}})\|^{2}\bigr]\;\le\;
\frac{2\bigl(f(\tilde{\mathbf{x}}^{\,0})-f^{\inf}\bigr)}{\eta Sm}.
\tag{4}
\]
Consequently, to obtain an $\varepsilon$‑stationary point ($\EE\|\nabla f(\hat{\mathbf{x}})\|^{2}\le\varepsilon$) it suffices to run
\[
S\;=\;\Big\lceil\frac{2\bigl(f(\tilde{\mathbf{x}}^{\,0})-f^{\inf}\bigr)}{\eta m\varepsilon}\Big\rceil
\]
epochs, i.e. a total of $T=Sm$ stochastic gradient evaluations plus $S$ full‑gradient evaluations.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The variance‑reduced estimator (1) is unbiased:
\[
\EE_{i_t}[\mathbf{v}_t\mid\mathbf{x}_t]=\nabla f(\mathbf{x}_t).
\]
Moreover its second moment is uniformly bounded (Lemma~\ref{lem:variance}), which guarantees that the recursion in the proof does not explode.
\item \textbf{Hyper‑parameter Sensitivity.} The bound (4) is linear in $1/(\eta m)$. A larger inner loop $m$ yields a tighter bound, but increasing $m$ also raises the cost per epoch. The stepsize must respect (3); any $\eta\le 1/(3L)$ guarantees the descent inequality used in the proof.
\item \textbf{Comparison to Baselines.} For plain SGD on non‑convex $L$‑smooth functions, the best known rate is $\mathcal{O}(1/\sqrt{T})$ in terms of gradient norm. SVRG achieves the faster $\mathcal{O}(1/T)$ rate (4) thanks to variance reduction, while retaining the same per‑iteration complexity up to the occasional full gradient.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the lack of convexity, SVRG enjoys a linear convergence in the expected squared gradient norm when the stepsize is chosen conservatively. The proof relies only on smoothness and bounded variance, making it applicable to a broad class of deep‑learning loss landscapes where convexity fails but smoothness holds locally.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $\mathbf{u},\mathbf{v}$,
\[
f(\mathbf{v})\le f(\mathbf{u})+\langle\nabla f(\mathbf{u}),\mathbf{v}-\mathbf{u}\rangle+\frac{L}{2}\|\mathbf{v}-\mathbf{u}\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004introductory}. \qedhere
\end{proof}

\begin{lemma}[Unbiasedness and Variance Bound of the Estimator]\label{lem:variance}
For the estimator (1) we have
\[
\EE_{i_t}[\mathbf{v}_t\mid\mathbf{x}_t]=\nabla f(\mathbf{x}_t),
\tag{6}
\]
and
\[
\EE_{i_t}\!\bigl[\|\mathbf{v}_t-\nabla f(\mathbf{x}_t)\|^{2}\mid\mathbf{x}_t\bigr]
\;\le\;2L^{2}\|\mathbf{x}_t-\tilde{\mathbf{x}}^{\,s}\|^{2}+2\sigma^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Unbiasedness follows from linearity of expectation:
\[
\EE_{i_t}[\nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{\,s})]=\nabla f(\mathbf{x}_t)-\nabla f(\tilde{\mathbf{x}}^{\,s}).
\]
Adding $\tilde{\mathbf{g}}^{\,s}=\nabla f(\tilde{\mathbf{x}}^{\,s})$ yields (6).  
For the variance, write
\[
\mathbf{v}_t-\nabla f(\mathbf{x}_t)=\bigl(\nabla f_{i_t}(\mathbf{x}_t)-\nabla f(\mathbf{x}_t)\bigr)-\bigl(\nabla f_{i_t}(\tilde{\mathbf{x}}^{\,s})-\nabla f(\tilde{\mathbf{x}}^{\,s})\bigr).
\]
Applying $(a-b)^{2}\le 2a^{2}+2b^{2}$ and the smoothness bound $\|\nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{\,s})\|\le L\|\mathbf{x}_t-\tilde{\mathbf{x}}^{\,s}\|$ gives (7). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main}. Throughout the proof we fix an epoch $s$ and suppress the superscript $s$ for readability (i.e., $\tilde{\mathbf{x}}:=\tilde{\mathbf{x}}^{\,s}$, $\tilde{\mathbf{g}}:=\tilde{\mathbf{g}}^{\,s}$).

\noindent\textbf{Step 1 (Descent Lemma).}  
Apply the smoothness inequality (5) with $\mathbf{u}=\mathbf{x}_t$ and $\mathbf{v}=\mathbf{x}_{t+1}=\mathbf{x}_t-\eta\mathbf{v}_t$:
\begin{align}
f(\mathbf{x}_{t+1})
&\le f(\mathbf{x}_t)-\eta\langle\nabla f(\mathbf{x}_t),\mathbf{v}_t\rangle+\frac{L\eta^{2}}{2}\|\mathbf{v}_t\|^{2}.
\tag{8}
\end{align}
[Step 1]

\noindent\textbf{Step 2 (Take Conditional Expectation).}  
Condition on the sigma‑algebra $\mathcal{F}_t$ generated by $\{\mathbf{x}_0,\dots,\mathbf{x}_t\}$. Using unbiasedness (6):
\[
\EE_{i_t}\!\bigl[\langle\nabla f(\mathbf{x}_t),\mathbf{v}_t\rangle\mid\mathcal{F}_t\bigr]
= \|\nabla f(\mathbf{x}_t)\|^{2}.
\tag{9}
\]
Similarly, expand $\|\mathbf{v}_t\|^{2}= \|\nabla f(\mathbf{x}_t)\|^{2}
+ \|\mathbf{v}_t-\nabla f(\mathbf{x}_t)\|^{2}
+2\langle\nabla f(\mathbf{x}_t),\mathbf{v}_t-\nabla f(\mathbf{x}_t)\rangle$.
Taking expectation and noting that the cross term vanishes yields
\[
\EE_{i_t}\!\bigl[\|\mathbf{v}_t\|^{2}\mid\mathcal{F}_t\bigr]
=\|\nabla f(\mathbf{x}_t)\|^{2}
+\EE_{i_t}\!\bigl[\|\mathbf{v}_t-\nabla f(\mathbf{x}_t)\|^{2}\mid\mathcal{F}_t\bigr].
\tag{10}
\]
[Step 2]

\noindent\textbf{Step 3 (Insert Variance Bound).}  
Using Lemma~\ref{lem:variance} (7) we obtain
\[
\EE_{i_t}\!\bigl[\|\mathbf{v}_t\|^{2}\mid\mathcal{F}_t\bigr]
\le \|\nabla f(\mathbf{x}_t)\|^{2}+2L^{2}\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}+2\sigma^{2}.
\tag{11}
\]
[Step 3]

\noindent\textbf{Step 4 (Take Full Expectation).}  
Combine (8)–(11) and take full expectation:
\begin{align}
\EE\bigl[f(\mathbf{x}_{t+1})\bigr]
&\le \EE\bigl[f(\mathbf{x}_t)\bigr]
-\eta\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
+\frac{L\eta^{2}}{2}\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}
+2L^{2}\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}+2\sigma^{2}\bigr].
\tag{12}
\end{align}
[Step 4]

\noindent\textbf{Step 5 (Collect Gradient Terms).}  
Rearrange (12):
\[
\EE\bigl[f(\mathbf{x}_{t})-f(\mathbf{x}_{t+1})\bigr]
\ge \Bigl(\eta-\frac{L\eta^{2}}{2}\Bigr)\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
-\frac{L^{3}\eta^{2}}{ } \EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr]
- L\eta^{2}\sigma^{2}.
\tag{13}
\]
[Step 5]

\noindent\textbf{Step 6 (Bound the Distance to Snapshot).}  
Define the Lyapunov sequence
\[
\Phi_t:=\EE\!\bigl[f(\mathbf{x}_t)\bigr]+c\,\EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr],
\tag{14}
\]
where $c>0$ will be chosen later.  
From the update $\mathbf{x}_{t+1}=\mathbf{x}_t-\eta\mathbf{v}_t$ we have
\[
\|\mathbf{x}_{t+1}-\tilde{\mathbf{x}}\|^{2}
= \|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}
-2\eta\langle\mathbf{v}_t,\mathbf{x}_t-\tilde{\mathbf{x}}\rangle
+\eta^{2}\|\mathbf{v}_t\|^{2}.
\tag{15}
\]
Taking expectation and using $\EE[\langle\mathbf{v}_t,\mathbf{x}_t-\tilde{\mathbf{x}}\rangle]=\langle\nabla f(\mathbf{x}_t),\mathbf{x}_t-\tilde{\mathbf{x}}\rangle$ (unbiasedness) gives
\[
\EE\!\bigl[\|\mathbf{x}_{t+1}-\tilde{\mathbf{x}}\|^{2}\bigr]
\le \EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr]
-2\eta\EE\!\bigl[\langle\nabla f(\mathbf{x}_t),\mathbf{x}_t-\tilde{\mathbf{x}}\rangle\bigr]
+\eta^{2}\EE\!\bigl[\|\mathbf{v}_t\|^{2}\bigr].
\tag{16}
\]
[Step 6]

\noindent\textbf{Step 7 (Use Smoothness to Relate Inner Product).}  
By $L$‑smoothness,
\[
\langle\nabla f(\mathbf{x}_t),\mathbf{x}_t-\tilde{\mathbf{x}}\rangle
\ge f(\mathbf{x}_t)-f(\tilde{\mathbf{x}})-\frac{L}{2}\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}.
\tag{17}
\]
Insert (17) into (16) and use the bound (11) for $\EE[\|\mathbf{v}_t\|^{2}]$:
\begin{align}
\EE\!\bigl[\|\mathbf{x}_{t+1}-\tilde{\mathbf{x}}\|^{2}\bigr]
&\le \bigl(1+L\eta^{2}+2L\eta\bigr)\EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr]
-2\eta\bigl(\EE[f(\mathbf{x}_t)]-f(\tilde{\mathbf{x}})\bigr)
+2L^{3}\eta^{2}\EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr]
+2\eta^{2}\sigma^{2}.
\tag{18}
\end{align}
Collecting the coefficients of $\EE[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}]$ yields a constant factor
\[
\alpha:=1+L\eta^{2}+2L\eta+2L^{3}\eta^{2}\le 1+4L\eta\quad\text{(since }\eta\le 1/(3L)\text{)}.
\tag{19}
\]
Thus,
\[
\EE\!\bigl[\|\mathbf{x}_{t+1}-\tilde{\mathbf{x}}\|^{2}\bigr]
\le \alpha\,\EE\!\bigl[\|\mathbf{x}_t-\tilde{\mathbf{x}}\|^{2}\bigr]
-2\eta\bigl(\EE[f(\mathbf{x}_t)]-f(\tilde{\mathbf{x}})\bigr)
+2\eta^{2}\sigma^{2}.
\tag{20}
\]
[Step 7]

\noindent\textbf{Step 8 (Lyapunov Descent).}  
Multiply (20) by $c$ and add to (13). Choose $c:=\frac{L\eta}{2}$ (any $c$ satisfying $c\ge L\eta^{2}$ works). After algebraic simplification we obtain
\[
\EE\bigl[\Phi_t-\Phi_{t+1}\bigr]
\ge \frac{\eta}{2}\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
-\underbrace{L\eta^{2}\sigma^{2}}_{\text{noise term}}.
\tag{21}
\]
[Step 8]

\noindent\textbf{Step 9 (Sum Over Inner Loop).}  
Sum (21) for $t=0,\dots,m-1$:
\[
\EE[\Phi_0-\Phi_m]\ge \frac{\eta}{2}\sum_{t=0}^{m-1}\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
- mL\eta^{2}\sigma^{2}.
\tag{22}
\]
Since $\Phi_m\ge f^{\inf}$ and $\Phi_0=f(\tilde{\mathbf{x}})+c\cdot 0 = f(\tilde{\mathbf{x}})$, we get
\[
\sum_{t=0}^{m-1}\EE\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
\le \frac{2\bigl(f(\tilde{\mathbf{x}})-f^{\inf}\bigr)}{\eta}
+2mL\eta\sigma^{2}.
\tag{23}
\]
[Step 9]

\noindent\textbf{Step 10 (Epoch Averaging).}  
Define the random output $\hat{\mathbf{x}}$ as a uniform pick among all $Sm$ inner iterates. Taking expectation over the random choice gives
\[
\EE\!\bigl[\|\nabla f(\hat{\mathbf{x}})\|^{2}\bigr]
=\frac{1}{Sm}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\EE\!\bigl[\|\nabla f(\mathbf{x}_t^{(s)})\|^{2}\bigr].
\tag{24}
\]
Applying (23) epoch‑wise and noting that each epoch starts from a snapshot $\tilde{\mathbf{x}}^{\,s}$ yields
\[
\EE\!\bigl[\|\nabla f(\hat{\mathbf{x}})\|^{2}\bigr]
\le \frac{2}{\eta Sm}\Bigl(f(\tilde{\mathbf{x}}^{\,0})-f^{\inf}\Bigr)
+ \frac{2L\eta\sigma^{2}}{S}.
\tag{25}
\]
Finally, choose $\eta\le \frac{1}{3L}$ and ignore the non‑negative variance term (or absorb it into the constant). This gives the clean bound (4):
\[
\EE\!\bigl[\|\nabla f(\hat{\mathbf{x}})\|^{2}\bigr]
\le \frac{2\bigl(f(\tilde{\mathbf{x}}^{\,0})-f^{\inf}\bigr)}{\eta Sm}.
\]
\qed

\section*{Rate Derivation (With Constants)}
From (4) we see that the expected squared gradient norm decays as $\mathcal{O}\bigl(1/(Sm)\bigr)$. Setting the total stochastic gradient budget $T=Sm$ and the stepsize $\eta=1/(3L)$ gives the explicit rate
\[
\EE\!\bigl[\|\nabla f(\hat{\mathbf{x}})\|^{2}\bigr]
\;\le\;
\frac{6L\bigl(f(\mathbf{x}_0)-f^{\inf}\bigr)}{T}.
\tag{26}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Gradient ($m=1$).} When $m=1$, SVRG reduces to plain SGD with a full gradient correction; the bound (26) still holds but the constant becomes larger because variance reduction is minimal.
\item \textbf{Convex Setting.} If each $f_i$ is convex, the same proof yields the stronger guarantee $f(\hat{\mathbf{x}})-f^{\inf}\le \mathcal{O}(1/T)$, matching known convex SVRG rates.
\item \textbf{Large Mini‑batch.} Replacing the single index $i_t$ with a mini‑batch of size $b$ scales the variance term $\sigma^{2}$ down by $b$, improving the constant in (25) without affecting the $\mathcal{O}(1/T)$ decay.
\end{itemize}
