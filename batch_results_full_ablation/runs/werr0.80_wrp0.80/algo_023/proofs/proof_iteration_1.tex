\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let 
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}
    \item $f:\R^{d}\to\R$ is convex, differentiable and has $L$–Lipschitz continuous gradient:
    \[
    \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\quad\forall x,y\in\R^{d};
    \]
    \item $g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).
\end{itemize}
Given a stepsize $\eta>0$, the \textbf{proximal gradient iteration} is
\begin{equation}
\boxed{%
x_{k+1}\;=\;\prox_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),\qquad k=0,1,\dots
}
\label{eq:update}
\end{equation}
where the proximal operator of $g$ with parameter $\eta$ is
\[
\prox_{\eta g}(v)\;:=\;\arg\min_{u\in\R^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Convex $f$, convex $g$, stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\R^{d}$, max iterations $K$
\State $k\gets 0$
\While{$k<K$}
    \State $v_{k}\gets x_{k}-\eta\nabla f(x_{k})$ \Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(v_{k})$ \Comment{proximal step}
    \State $k\gets k+1$
\EndWhile
\State \textbf{return} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a scalar problem  
\[
f(w)= (w-3)^{2},\qquad g\equiv 0,
\]
so $F(w)=f(w)$.  
Take $w_{0}=0$, stepsize $\eta=0.1$, and perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})=2(w_{k}-3)$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values: $F(w_{0})=9$, $F(w_{1})=(0.6-3)^{2}=5.76$, $F(w_{2})\approx4.19$, $F(w_{3})\approx3.44$,
showing monotone decrease toward the optimum $F^{*}=0$ at $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{ass:convexg}
$g$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is well defined for all $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le\frac{1}{L}.
\]
\end{assumption}
\begin{assumption}[Existence of a minimizer]\label{ass:opt}
The composite problem $\min_{x}F(x)$ admits at least one solution $x^{*}$ and we denote $F^{*}=F(x^{*})$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{eq:update} under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}. Then for any $k\ge 1$
\[
F(x_{k})-F^{*}\;\le\;\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k}.
\]
Consequently $F(x_{k})\to F^{*}$ and the sequence $\{x_{k}\}$ is bounded.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The iteration is a composition of a gradient step (non‑expansive for $\eta\le 1/L$) and a proximal step (firmly non‑expansive). Hence the whole map is non‑expansive, guaranteeing that the iterates stay in a compact set whenever the level set $\{x:F(x)\le F(x_{0})\}$ is bounded.
    \item \textbf{Hyperparameter Sensitivity.} The bound \eqref{thm:rate} degrades linearly with $\eta$; a larger stepsize (up to $1/L$) yields a smaller constant $\frac{1}{2\eta}$, but exceeding $1/L$ destroys the descent property.
    \item \textbf{Comparison to Gradient Descent.} When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and the theorem collapses to the classic $O(1/k)$ rate for gradient descent on smooth convex functions.
    \item \textbf{Special case $g$ indicator of a convex set $C$.} The method becomes the projected gradient method; the same $O(1/k)$ guarantee holds.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Non‑expansiveness of the proximal operator]\label{lem:nonexp}
For any $u,v\in\R^{d}$ and $\eta>0$,
\[
\bigl\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\bigr\|\le\|u-v\|.
\]
\end{lemma}
\begin{lemma}[Optimality condition of the proximal step]\label{lem:optcond}
Let $x^{+}=\prox_{\eta g}(x-\eta\nabla f(x))$. Then
\[
\frac{1}{\eta}\bigl(x-x^{+}\bigr)-\nabla f(x)\in\partial g(x^{+}).
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:rate} by establishing a recursive inequality and telescoping it.

\noindent\textbf{Notation.} Throughout the proof we denote $x_{k+1}$ as in \eqref{eq:update} and fix an arbitrary optimal solution $x^{*}\in\arg\min F$.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{Proximal optimality.} By Lemma~\ref{lem:optcond} applied to $x_{k}$,
\[
\frac{1}{\eta}\bigl(x_{k}-x_{k+1}\bigr)-\nabla f(x_{k})\in\partial g(x_{k+1}).
\tag{1}
\]

\item \textbf{Convexity of $g$.} For any $u\in\partial g(x_{k+1})$ and any $z$,
\[
g(z)\ge g(x_{k+1})+\langle u,\,z-x_{k+1}\rangle .
\]
Choosing $u$ as the element in (1) and $z=x^{*}$ yields
\[
g(x^{*})\ge g(x_{k+1})+\Big\langle \frac{1}{\eta}(x_{k}-x_{k+1})-\nabla f(x_{k}),\,x^{*}-x_{k+1}\Big\rangle .
\tag{2}
\]

\item \textbf{Rearranging (2).} Multiply (2) by $\eta$ and bring all terms to the left:
\[
\eta\bigl(g(x_{k+1})-g(x^{*})\bigr)
\le \langle x_{k}-x_{k+1},\,x_{k+1}-x^{*}\rangle-\eta\langle\nabla f(x_{k}),\,x^{*}-x_{k+1}\rangle .
\tag{3}
\]

\item \textbf{Algebraic identity.} For any vectors $a,b$,
\[
\langle a-b,\,b-c\rangle =\tfrac12\bigl(\|a-c\|^{2}-\|a-b\|^{2}-\|b-c\|^{2}\bigr).
\]
Apply this with $a=x_{k}$, $b=x_{k+1}$, $c=x^{*}$ to the first inner product in (3):
\[
\langle x_{k}-x_{k+1},\,x_{k+1}-x^{*}\rangle
=\tfrac12\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k}-x_{k+1}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr).
\tag{4}
\]

\item \textbf{Descent lemma for $f$.} By Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),\,x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{5}
\]

\item \textbf{Combine (3)–(5).} Add $f(x_{k+1})-f(x^{*})$ to both sides of (3) and use (5) to replace $f(x_{k+1})$:
\begin{align*}
&\bigl[f(x_{k+1})+g(x_{k+1})\bigr]-F^{*}\\
&\le f(x_{k})-f(x^{*})+\langle\nabla f(x_{k}),\,x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&\quad +\frac{1}{\eta}\langle x_{k}-x_{k+1},\,x_{k+1}-x^{*}\rangle
-\langle\nabla f(x_{k}),\,x^{*}-x_{k+1}\rangle .
\end{align*}
Cancel the terms $\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$ and $-\langle\nabla f(x_{k}),x^{*}-x_{k+1}\rangle$ (they sum to $-\langle\nabla f(x_{k}),x^{*}-x_{k}\rangle$), obtaining
\[
F(x_{k+1})-F^{*}\le f(x_{k})-f(x^{*})-\langle\nabla f(x_{k}),x^{*}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{*}\rangle .
\tag{6}
\]

\item \textbf{Use convexity of $f$.} Convexity yields
\[
f(x_{k})-f(x^{*})\le\langle\nabla f(x_{k}),\,x_{k}-x^{*}\rangle .
\tag{7}
\]
Insert (7) into (6). The two inner‑product terms cancel exactly, leaving
\[
F(x_{k+1})-F^{*}\le \frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{*}\rangle .
\tag{8}
\]

\item \textbf{Replace the inner product via (4).} Using (4) in the second term of (8) gives
\[
\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{*}\rangle
=\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr).
\tag{9}
\]

\item \textbf{Combine (8) and (9).}
\begin{align*}
F(x_{k+1})-F^{*}
&\le\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr)\\[4pt]
&=\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr)
+\Bigl(\frac{L}{2}-\frac{1}{2\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\end{align*}
\tag{10}
\item \textbf{Stepsize condition.} By Assumption~\ref{ass:stepsize}, $\eta\le1/L$, i.e.
\[
\frac{L}{2}-\frac{1}{2\eta}\le0.
\]
Hence the second term in (10) is non‑positive and can be dropped:
\[
F(x_{k+1})-F^{*}\le\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr).
\tag{11}
\]

\item \textbf{Telescoping sum.} Summing inequality (11) for $k=0,\dots, T-1$ yields
\[
\sum_{k=0}^{T-1}\bigl(F(x_{k+1})-F^{*}\bigr)
\le\frac{1}{2\eta}\bigl(\|x_{0}-x^{*}\|^{2}-\|x_{T}-x^{*}\|^{2}\bigr)
\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}.
\tag{12}
\]

\item \textbf{Monotonicity and final bound.} Since $F(x_{k})$ is non‑increasing (from (11)), each term in the sum is bounded by the average:
\[
T\bigl(F(x_{T})-F^{*}\bigr)\le\sum_{k=0}^{T-1}\bigl(F(x_{k+1})-F^{*}\bigr)
\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}.
\]
Dividing by $T$ gives the claimed rate:
\[
F(x_{T})-F^{*}\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,T},\qquad\forall\,T\ge1.
\tag{13}
\]

\item \textbf{Conclusion.} Inequality (13) is exactly the statement of Theorem~\ref{thm:rate}. $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The constant in the $O(1/k)$ bound is $\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}$.  
Choosing the maximal admissible stepsize $\eta=1/L$ minimizes this constant, yielding
\[
F(x_{k})-F^{*}\le\frac{L\,\|x_{0}-x^{*}\|^{2}}{2k}.
\]

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Pure gradient descent} ($g\equiv0$).  $\prox_{\eta g}$ becomes the identity, and the proof collapses to the classic analysis for smooth convex minimization.
    \item \textbf{Projected gradient method}.  If $g=\iota_{C}$ is the indicator of a closed convex set $C$, then $\prox_{\eta g}=P_{C}$, the Euclidean projection.  All steps above remain valid, giving the same $O(1/k)$ rate for constrained convex problems.
    \item \textbf{Lasso (elastic net) regularization}.  For $g(x)=\lambda\|x\|_{1}$ the proximal operator is soft‑thresholding.  The theorem guarantees that the objective value converges at $O(1/k)$ regardless of the non‑smooth $\ell_{1}$ term.
\end{enumerate}
