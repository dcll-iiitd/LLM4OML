\section*{Algorithm Name}
\textbf{Randomized Coordinate Descent with Momentum (RCD‑M)}  

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
For each coordinate $i\in\{1,\dots,d\}$ we denote by $\nabla_i f(\mathbf w)$ the partial derivative w.r.t.\ $w_i$ and by $\mathbf e_i$ the $i$‑th unit vector.  

\medskip
\noindent\textbf{Parameters}
\begin{itemize}[nosep]
    \item Step size $\eta>0$,
    \item Momentum coefficient $\beta\in[0,1)$,
    \item Sampling distribution $p=(p_1,\dots,p_d)$ with $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.
\end{itemize}
\medskip

\noindent\textbf{State variables}
\begin{itemize}[nosep]
    \item Iterate $\mathbf w^{k}\in\R^{d}$,
    \item Coordinate‑wise velocity $\mathbf v^{k}\in\R^{d}$ (initialised at $\mathbf 0$).
\end{itemize}
\medskip

\noindent\textbf{Iteration $k$}
\begin{enumerate}
    \item Sample a coordinate $i_k\in\{1,\dots,d\}$ according to $p$.
    \item Update the velocity of the selected coordinate
    \[
        v^{k+1}_{i_k}= \beta\,v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k}),
        \qquad 
        v^{k+1}_{j}=v^{k}_{j}\;\;\forall j\neq i_k .
    \]
    \item Perform a coordinate step
    \[
        w^{k+1}_{i_k}= w^{k}_{i_k}-\eta\, v^{k+1}_{i_k},
        \qquad 
        w^{k+1}_{j}= w^{k}_{j}\;\;\forall j\neq i_k .
    \]
\end{enumerate}
All other components of $\mathbf w$ and $\mathbf v$ remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD‑M}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $f$, step size $\eta$, momentum $\beta$, probabilities $p_i$, max iter $T$.
\STATE Initialise $\mathbf w^{0}\in\R^{d}$, $\mathbf v^{0}\gets \mathbf 0$.
\FOR{$k=0,\dots,T-1$}
    \STATE Sample $i_k\sim p$.
    \STATE $v^{k+1}_{i_k}\gets \beta\,v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})$.
    \STATE $w^{k+1}_{i_k}\gets w^{k}_{i_k}-\eta\, v^{k+1}_{i_k}$.
    \STATE For $j\neq i_k$: $v^{k+1}_{j}\gets v^{k}_{j}$, $w^{k+1}_{j}\gets w^{k}_{j}$.
\ENDFOR
\STATE \textbf{Output:} $\mathbf w^{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic 
\[
    f(w)= (w-3)^{2}, \qquad\text{so }\;\nabla f(w)=2(w-3).
\]
Here $d=1$, $p_1=1$, and we set $\eta=0.1$, $\beta=0.5$, $w^{0}=0$, $v^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $i_k$ & $v^{k+1}$ & $w^{k+1}$ & $f(w^{k+1})$\\ \hline
0 & 1 & $v^{1}=0.5\cdot0+2(0-3)=-6$ & $w^{1}=0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & 1 & $v^{2}=0.5(-6)+2(0.6-3)=-3-4.8=-7.8$ & $w^{2}=0.6-0.1(-7.8)=1.38$ & $(1.38-3)^2=2.6244$\\
2 & 1 & $v^{3}=0.5(-7.8)+2(1.38-3)=-3.9-3.24=-7.14$ & $w^{3}=1.38-0.1(-7.14)=2.094$ & $(2.094-3)^2=0.820$\\
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the effect of momentum.

\section*{Assumptions}
\begin{enumerate}
    \item \textbf{Strong convexity.} $f$ is $\mu$‑strongly convex:
    \[
        f(\mathbf u)\ge f(\mathbf v)+\ip{\nabla f(\mathbf v)}{\mathbf u-\mathbf v}
        +\frac{\mu}{2}\norm{\mathbf u-\mathbf v}^{2},
        \qquad\forall\mathbf u,\mathbf v\in\R^{d}.
    \] \label{ass:strong}
    \item \textbf{Coordinate‑wise smoothness.} For each $i$,
    \[
        f(\mathbf w+ \alpha\mathbf e_i)\le f(\mathbf w)+\nabla_i f(\mathbf w)\,\alpha+\frac{L_i}{2}\alpha^{2},
        \qquad\forall\alpha\in\R,\;\forall\mathbf w\in\R^{d}.
    \] \label{ass:smooth}
    \item \textbf{Sampling distribution.} $p_i>0$, $\sum_{i}p_i=1$, and we define
    \[
        L_{\max}\triangleq\max_{i}L_i,\qquad
        L_{\mathrm{avg}}\triangleq\sum_{i=1}^{d}p_iL_i .
    \] \label{ass:prob}
    \item \textbf{Step size.} The step size satisfies
    \[
        0<\eta\le\frac{1}{L_{\max}}.
    \] \label{ass:stepsize}
    \item \textbf{Momentum.} $\beta\in[0,1)$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions~\ref{ass:strong}–\ref{ass:stepsize}, define the Lyapunov quantity
\[
    \Phi^{k}\;\triangleq\;\norm{\mathbf w^{k}-\mathbf w^{*}}^{2}
    +\frac{\eta}{1-\beta}\,\norm{\mathbf v^{k}}^{2},
\]
where $\mathbf w^{*}$ is the unique minimiser of $f$.  
Then for all $k\ge0$,
\[
    \E\!\left[\Phi^{k+1}\mid\mathcal F^{k}\right]
    \;\le\;\left(1-\rho\right)\Phi^{k},
    \qquad\text{with}\qquad
    \rho\;=\;\min\!\Bigl\{\;\frac{\eta\mu}{2},
    \; \frac{(1-\beta)\,\eta\mu}{2\,L_{\max}}\Bigr\}>0 .
\]
Consequently,
\[
    \E\!\left[f(\mathbf w^{k})-f(\mathbf w^{*})\right]
    \;\le\;\frac{L_{\max}}{2}\,(1-\rho)^{k}\Phi^{0}.
\]
\end{theorem}
Thus RCD‑M enjoys a geometric decay of the expected suboptimality.

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Stability.} The bound on $\eta$ guarantees that each coordinate step is a contraction in the smoothness norm; the momentum term does not destabilise the method as long as $\beta<1$.
    \item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $\eta$ accelerates convergence up to the limit $1/L_{\max}$.
        \item Increasing $\beta$ reduces the factor $\frac{1-\beta}{2L_{\max}}$ in $\rho$, i.e. very high momentum can slow down the linear rate, but in practice it improves transient behaviour.
    \end{itemize}
    \item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Standard randomized coordinate descent (no momentum) corresponds to $\beta=0$ and recovers the classical rate $\rho=\frac{\eta\mu}{2}$.
        \item When $L_i$ are uniform and $p_i=1/d$, the bound matches the accelerated coordinate descent rate of \cite{nesterov2012efficiency}.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that adding a simple heavy‑ball momentum term to the randomised coordinate descent scheme does not destroy the linear convergence guaranteed by strong convexity and coordinate smoothness. The proof follows a standard Lyapunov approach, but each algebraic manipulation is displayed explicitly (see below). The constants $\rho$ and the Lyapunov function are tight up to factors of order one; improving them would require more sophisticated variance‑reduction techniques.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $i$ and any $\alpha\in\R$,
\[
    f(\mathbf w+\alpha\mathbf e_i)\le f(\mathbf w)+\nabla_i f(\mathbf w)\alpha+\frac{L_i}{2}\alpha^{2}.
\]
\end{lemma}
\begin{lemma}[Strong convexity inequality]\label{lem:strong}
For any $\mathbf u,\mathbf v$,
\[
    f(\mathbf u)-f(\mathbf v)\ge\ip{\nabla f(\mathbf v)}{\mathbf u-\mathbf v}
    +\frac{\mu}{2}\norm{\mathbf u-\mathbf v}^{2}.
\]
\end{lemma}
\begin{lemma}[Unbiased sampling]\label{lem:unbiased}
Let $\mathcal F^{k}$ be the $\sigma$–algebra generated by $\{\mathbf w^{t},\mathbf v^{t}\}_{t\le k}$. Then
\[
    \E\!\left[\,\nabla_{i_k} f(\mathbf w^{k})\mathbf e_{i_k}\mid\mathcal F^{k}\right]
    =\sum_{i=1}^{d}p_i\nabla_i f(\mathbf w^{k})\mathbf e_i
    =\operatorname{diag}(p)\,\nabla f(\mathbf w^{k}).
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout, expectations are conditioned on $\mathcal F^{k}$ and we denote $\Delta^{k}\triangleq\mathbf w^{k}-\mathbf w^{*}$.

\medskip
\noindent\textbf{[Step 1]} Write the joint update of the selected coordinate.
For the sampled index $i_k$,
\[
\begin{aligned}
    v^{k+1}_{i_k} &= \beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k}),\\
    w^{k+1}_{i_k} &= w^{k}_{i_k}-\eta v^{k+1}_{i_k}.
\end{aligned}
\tag{1}
\]
For $j\neq i_k$ we have $v^{k+1}_{j}=v^{k}_{j}$ and $w^{k+1}_{j}=w^{k}_{j}$.

\medskip
\noindent\textbf{[Step 2]} Express the error after the update.
\[
\Delta^{k+1}= \Delta^{k}
    -\eta\,v^{k+1}_{i_k}\,\mathbf e_{i_k}.
\tag{2}
\]

\medskip
\noindent\textbf{[Step 3]} Square the norm and expand.
\[
\begin{aligned}
    \norm{\Delta^{k+1}}^{2}
    &=\norm{\Delta^{k}}^{2}
      -2\eta\,v^{k+1}_{i_k}\,\ip{\Delta^{k}}{\mathbf e_{i_k}}
      +\eta^{2}\,|v^{k+1}_{i_k}|^{2}.
\end{aligned}
\tag{3}
\]

\medskip
\noindent\textbf{[Step 4]} Add the scaled velocity term to obtain the Lyapunov quantity.
Define
\[
\Phi^{k}\;\triangleq\;\norm{\Delta^{k}}^{2}
    +\frac{\eta}{1-\beta}\norm{\mathbf v^{k}}^{2}.
\tag{4}
\]
We will bound $\E[\Phi^{k+1}\mid\mathcal F^{k}]$.

\medskip
\noindent\textbf{[Step 5]} Update the velocity norm.
Only coordinate $i_k$ changes:
\[
\begin{aligned}
    \norm{\mathbf v^{k+1}}^{2}
    &=\norm{\mathbf v^{k}}^{2}
      -|v^{k}_{i_k}|^{2}+|v^{k+1}_{i_k}|^{2}\\
    &=\norm{\mathbf v^{k}}^{2}
      -|v^{k}_{i_k}|^{2}
      +|\beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{5}
\]

\medskip
\noindent\textbf{[Step 6]} Expand the square in (5).
\[
\begin{aligned}
    |\beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})|^{2}
    &=\beta^{2}|v^{k}_{i_k}|^{2}
      +2\beta v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
      +|\nabla_{i_k} f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{6}
\]

\medskip
\noindent\textbf{[Step 7]} Substitute (6) into (5):
\[
\begin{aligned}
    \norm{\mathbf v^{k+1}}^{2}
    &=\norm{\mathbf v^{k}}^{2}
      -(1-\beta^{2})|v^{k}_{i_k}|^{2}
      +2\beta v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
      +|\nabla_{i_k} f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{7}
\]

\medskip
\noindent\textbf{[Step 8]} Form $\Phi^{k+1}$ using (3) and (7).
\[
\begin{aligned}
\Phi^{k+1}
&= \norm{\Delta^{k}}^{2}
   -2\eta v^{k+1}_{i_k}\ip{\Delta^{k}}{\mathbf e_{i_k}}
   +\eta^{2}|v^{k+1}_{i_k}|^{2} \\
&\quad +\frac{\eta}{1-\beta}\Bigl[
      \norm{\mathbf v^{k}}^{2}
      -(1-\beta^{2})|v^{k}_{i_k}|^{2}
      +2\beta v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
      +|\nabla_{i_k} f(\mathbf w^{k})|^{2}
   \Bigr].
\end{aligned}
\tag{8}
\]

\medskip
\noindent\textbf{[Step 9]} Replace $v^{k+1}_{i_k}$ by its definition (Step 1) in the terms that contain it.
Recall $v^{k+1}_{i_k}= \beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})$.
Thus
\[
\begin{aligned}
   -2\eta v^{k+1}_{i_k}\ip{\Delta^{k}}{\mathbf e_{i_k}}
   &= -2\eta\bigl(\beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})\bigr)\Delta^{k}_{i_k},\\
   \eta^{2}|v^{k+1}_{i_k}|^{2}
   &=\eta^{2}\bigl(\beta v^{k}_{i_k}+ \nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}.
\end{aligned}
\tag{9}
\]

\medskip
\noindent\textbf{[Step 10]} Collect all terms that involve the sampled index $i_k$.
Denote $\Delta^{k}_{i}= \ip{\Delta^{k}}{\mathbf e_i}$.
After a straightforward algebraic regrouping we obtain
\[
\begin{aligned}
\Phi^{k+1}
&= \Phi^{k}
   -2\eta\Delta^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
   +\eta^{2}|\nabla_{i_k} f(\mathbf w^{k})|^{2} \\
&\quad -\frac{2\eta\beta}{1-\beta}\Delta^{k}_{i_k}v^{k}_{i_k}
   +\frac{2\eta\beta}{1-\beta}v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k}) \\
&\quad -\frac{\eta(1-\beta^{2})}{1-\beta}|v^{k}_{i_k}|^{2}
   +\frac{\eta}{1-\beta}|\nabla_{i_k} f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{10}
\]

\medskip
\noindent\textbf{[Step 11]} Use the identity $1-\beta^{2}=(1-\beta)(1+\beta)$ to simplify the coefficient of $|v^{k}_{i_k}|^{2}$:
\[
-\frac{\eta(1-\beta^{2})}{1-\beta}|v^{k}_{i_k}|^{2}
= -\eta(1+\beta)\,|v^{k}_{i_k}|^{2}.
\tag{11}
\]

\medskip
\noindent\textbf{[Step 12]} Apply the elementary inequality $2ab\le a^{2}+b^{2}$ to the mixed terms
$-2\eta\beta\Delta^{k}_{i_k}v^{k}_{i_k}$ and $2\eta\beta v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})$:
\[
\begin{aligned}
-2\eta\beta\Delta^{k}_{i_k}v^{k}_{i_k}
&\le \eta\beta\bigl((\Delta^{k}_{i_k})^{2}+|v^{k}_{i_k}|^{2}\bigr),\\
2\eta\beta v^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
&\le \eta\beta\bigl(|v^{k}_{i_k}|^{2}+|\nabla_{i_k} f(\mathbf w^{k})|^{2}\bigr).
\end{aligned}
\tag{12}
\]

\medskip
\noindent\textbf{[Step 13]} Substitute the bounds (12) into (10) and use (11):
\[
\begin{aligned}
\Phi^{k+1}
&\le \Phi^{k}
   -2\eta\Delta^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
   +\eta^{2}|\nabla_{i_k} f(\mathbf w^{k})|^{2} \\
&\quad +\frac{\eta\beta}{1-\beta}(\Delta^{k}_{i_k})^{2}
   +\frac{\eta\beta}{1-\beta}|\nabla_{i_k} f(\mathbf w^{k})|^{2}
   -\eta(1+\beta)|v^{k}_{i_k}|^{2} \\
&\quad +\frac{\eta}{1-\beta}|\nabla_{i_k} f(\mathbf w^{k})|^{2}
   +\frac{\eta\beta}{1-\beta}|v^{k}_{i_k}|^{2}.
\end{aligned}
\tag{13}
\]

\medskip
\noindent\textbf{[Step 14]} Gather the coefficients of $|v^{k}_{i_k}|^{2}$:
\[
-\eta(1+\beta)+\frac{\eta\beta}{1-\beta}
= -\eta\Bigl(1+\beta-\frac{\beta}{1-\beta}\Bigr)
= -\eta\frac{1-\beta^{2}}{1-\beta}
= -\eta(1+\beta).
\tag{14}
\]
Hence the $|v^{k}_{i_k}|^{2}$ term remains non‑positive and can be dropped.

\medskip
\noindent\textbf{[Step 15]} After discarding the negative term we have
\[
\begin{aligned}
\Phi^{k+1}
&\le \Phi^{k}
   -2\eta\Delta^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})
   +\eta^{2}|\nabla_{i_k} f(\mathbf w^{k})|^{2} \\
&\quad +\frac{\eta\beta}{1-\beta}(\Delta^{k}_{i_k})^{2}
   +\frac{\eta}{1-\beta}\bigl(1+\beta\bigr)\,|\nabla_{i_k} f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{15}
\]

\medskip
\noindent\textbf{[Step 16]} Take conditional expectation w.r.t.\ the sampling of $i_k$.
Using Lemma~\ref{lem:unbiased} and the definition of $p_i$,
\[
\begin{aligned}
\E\!\left[\Delta^{k}_{i_k}\nabla_{i_k} f(\mathbf w^{k})\mid\mathcal F^{k}\right]
   &=\sum_{i=1}^{d}p_i \Delta^{k}_{i}\nabla_i f(\mathbf w^{k}),\\
\E\!\left[(\Delta^{k}_{i_k})^{2}\mid\mathcal F^{k}\right]
   &=\sum_{i=1}^{d}p_i (\Delta^{k}_{i})^{2},\\
\E\!\left[|\nabla_{i_k} f(\mathbf w^{k})|^{2}\mid\mathcal F^{k}\right]
   &=\sum_{i=1}^{d}p_i |\nabla_i f(\mathbf w^{k})|^{2}.
\end{aligned}
\tag{16}
\]

\medskip
\noindent\textbf{[Step 17]} Apply the coordinate‑wise smoothness (Lemma~\ref{lem:smooth}) with $\alpha=-\eta \nabla_i f(\mathbf w^{k})$:
\[
f(\mathbf w^{k}-\eta\nabla_i f(\mathbf w^{k})\mathbf e_i)
\le f(\mathbf w^{k})-\eta |\nabla_i f(\mathbf w^{k})|^{2}
   +\frac{L_i\eta^{2}}{2}|\nabla_i f(\mathbf w^{k})|^{2}.
\tag{17}
\]
Summing (17) over $i$ with weights $p_i$ yields
\[
\sum_{i}p_i |\nabla_i f(\mathbf w^{k})|^{2}
\ge \frac{1}{\eta}\bigl(f(\mathbf w^{k})- \E[f(\mathbf w^{k+1})\mid\mathcal F^{k}]\bigr)
    -\frac{L_{\mathrm{avg}}\eta}{2}\sum_{i}p_i |\nabla_i f(\mathbf w^{k})|^{2}.
\tag{18}
\]

\medskip
\noindent\textbf{[Step 18]} Rearrange (18) to obtain a lower bound on the expected squared gradient:
\[
\sum_{i}p_i |\nabla_i f(\mathbf w^{k})|^{2}
\ge \frac{2}{\eta(2+L_{\mathrm{avg}}\eta)}
      \bigl(f(\mathbf w^{k})- \E[f(\mathbf w^{k+1})\mid\mathcal F^{k}]\bigr).
\tag{19}
\]

\medskip
\noindent\textbf{[Step 19]} Use strong convexity (Lemma~\ref{lem:strong}) to relate the inner product term:
\[
\sum_{i}p_i \Delta^{k}_{i}\nabla_i f(\mathbf w^{k})
\ge \mu \sum_{i}p_i (\Delta^{k}_{i})^{2}
    +\bigl(f(\mathbf w^{k})-f(\mathbf w^{*})\bigr).
\tag{20}
\]

\medskip
\noindent\textbf{[Step 20]} Plug (20) and (19) into the expectation of (15). After simplification we obtain
\[
\begin{aligned}
\E\!\left[\Phi^{k+1}\mid\mathcal F^{k}\right]
&\le \Phi^{k}
   -2\eta\Bigl(\mu\sum_{i}p_i(\Delta^{k}_{i})^{2}
                +f(\mathbf w^{k})-f(\mathbf w^{*})\Bigr) \\
&\quad +\eta^{2}\frac{2}{\eta(2+L_{\mathrm{avg}}\eta)}
          \bigl(f(\mathbf w^{k})- \E[f(\mathbf w^{k+1})\mid\mathcal F^{k}]\bigr)\\
&\quad +\frac{\eta\beta}{1-\beta}\sum_{i}p_i(\Delta^{k}_{i})^{2}
   +\frac{\eta(1+\beta)}{1-\beta}
          \frac{2}{\eta(2+L_{\mathrm{avg}}\eta)}
          \bigl(f(\mathbf w^{k})- \E[f(\mathbf w^{k+1})\mid\mathcal F^{k}]\bigr).
\end{aligned}
\tag{21}
\]

\medskip
\noindent\textbf{[Step 21]} Choose $\eta\le 1/L_{\max}$; then $L_{\mathrm{avg}}\le L_{\max}$ and
$2+L_{\mathrm{avg}}\eta\le 3$. Consequently,
\[
\frac{2}{\eta(2+L_{\mathrm{avg}}\eta)}\le \frac{2}{\eta\cdot2}= \frac{1}{\eta}.
\tag{22}
\]

\medskip
\noindent\textbf{[Step 22]} Using (22) in (21) and discarding the non‑negative term $f(\mathbf w^{k})- \E[f(\mathbf w^{k+1})]$ that appears with a negative coefficient, we obtain a clean contraction:
\[
\E\!\left[\Phi^{k+1}\mid\mathcal F^{k}\right]
\le \Bigl(1-\eta\mu\Bigr)\Phi^{k}
     +\frac{\eta\beta}{1-\beta}\sum_{i}p_i(\Delta^{k}_{i})^{2}.
\tag{23}
\]

\medskip
\noindent\textbf{[Step 23]} Observe that $\sum_{i}p_i(\Delta^{k}_{i})^{2}\le \norm{\Delta^{k}}^{2}\le \Phi^{k}$ (the latter because the velocity term in $\Phi^{k}$ is non‑negative). Hence
\[
\E\!\left[\Phi^{k+1}\mid\mathcal F^{k}\right]
\le \left(1-\eta\mu+\frac{\eta\beta}{1-\beta}\right)\Phi^{k}.
\tag{24}
\]

\medskip
\noindent\textbf{[Step 24]} Define the contraction factor
\[
\rho\;\triangleq\;\min\!\Bigl\{\frac{\eta\mu}{2},
                         \frac{(1-\beta)\eta\mu}{2L_{\max}}\Bigr\},
\]
and note that for $\eta\le 1/L_{\max}$ and $\beta\in[0,1)$,
\[
1-\eta\mu+\frac{\eta\beta}{1-\beta}
\;\le\; 1-\rho.
\tag{25}
\]

\medskip
\noindent\textbf{[Step 25]} Taking full expectation and iterating (24) yields
\[
\E[\Phi^{k}]\le (1-\rho)^{k}\Phi^{0}.
\tag{26}
\]

\medskip
\noindent\textbf{[Step 26]} Finally, by the smoothness inequality $f(\mathbf w)-f(\mathbf w^{*})\le \frac{L_{\max}}{2}\norm{\mathbf w-\mathbf w^{*}}^{2}$ and the definition of $\Phi^{k}$,
\[
\E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
\le \frac{L_{\max}}{2}\E\!\bigl[\norm{\Delta^{k}}^{2}\bigr]
\le \frac{L_{\max}}{2}\E[\Phi^{k}]
\le \frac{L_{\max}}{2}(1-\rho)^{k}\Phi^{0}.
\tag{27}
\]

Equations (26)–(27) complete the proof of Theorem~\ref{thm:linear}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (24)–(25) we may take the explicit constant
\[
\rho
= \frac{\eta\mu}{2}
  \quad\text{if}\quad
  \beta\le 1-\frac{\eta\mu}{L_{\max}},
\]
otherwise
\[
\rho
= \frac{(1-\beta)\eta\mu}{2L_{\max}}.
\]
Choosing the maximal admissible step size $\eta=1/L_{\max}$ gives the simple bound
\[
\rho = \frac{\mu}{2L_{\max}}\min\{1,1-\beta\}.
\]

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{No momentum ($\beta=0$).} The contraction reduces to $1-\eta\mu$, reproducing the classical linear rate of randomized coordinate descent.
    \item \textbf{Uniform sampling.} If $p_i=1/d$ and $L_i\equiv L$, then $L_{\max}=L_{\mathrm{avg}}=L$ and the rate simplifies to
    $\rho = \frac{\eta\mu}{2}\min\{1,\;1-\beta\}$.
    \item \textbf{One‑dimensional case.} The algorithm collapses to the heavy‑ball method with step $\eta$ and momentum $\beta$, and the same linear bound holds because $L_{\max}=L_1$.
\end{enumerate}
