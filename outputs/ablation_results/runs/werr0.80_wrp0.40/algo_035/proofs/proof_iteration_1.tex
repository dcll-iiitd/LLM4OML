\section*{Clipped Gradient Descent with Momentum (CGDM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Given a clipping threshold $C>0$, a stepsize $\eta>0$ and a momentum
parameter $\beta\in[0,1)$, the iterates $\{x_t\}_{t\ge 0}$ are generated as

\begin{align}
    g_t &:= \clip{\nabla f(x_t)}{C}
    =\frac{\nabla f(x_t)}{\max\{1,\|\nabla f(x_t)\|/C\}},\label{eq:clip}\\[4pt]
    m_t &:= \beta\, m_{t-1} + (1-\beta)\, g_t,
    \qquad m_{-1}=0, \label{eq:momentum}\\[4pt]
    x_{t+1} &:= x_t - \eta\, m_t . \label{eq:update}
\end{align}

Equation \eqref{eq:clip} guarantees $\|g_t\|\le C$ for all $t$.  
The algorithm reduces to standard gradient descent with momentum when $C=\infty$.

\section*{Pseudocode}
\begin{algorithm}[h]
\caption{Clipped Gradient Descent with Momentum (CGDM)}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, momentum $\beta\in[0,1)$, clipping $C>0$, 
initial point $x_0\in\R^d$.
\State $m_{-1}\gets 0$
\For{$t=0,1,\dots,T-1$}
    \State $g_t \gets \displaystyle\frac{\nabla f(x_t)}{\max\{1,\|\nabla f(x_t)\|/C\}}$ \Comment{gradient clipping}
    \State $m_t \gets \beta\, m_{t-1} + (1-\beta)\, g_t$ \Comment{momentum update}
    \State $x_{t+1} \gets x_t - \eta\, m_t$ \Comment{parameter update}
\EndFor
\State \textbf{Output:} $\{x_t\}_{t=0}^T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^2$.
The true gradient is $\nabla f(w)=2(w-3)$.  
Choose $\eta=0.1$, $C=2$, $\beta=0.5$, and $w_0=0$.
We compute three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)$ & $g_t$ (clipped) & $w_{t+1}$ \\ \hline
0 & $0$ & $-6$ & $\displaystyle\frac{-6}{\max\{1,6/2\}}=\frac{-6}{3}=-2$ & $0-0.1(-2)=0.2$ \\[4pt]
1 & $0.2$ & $-5.6$ & $\displaystyle\frac{-5.6}{\max\{1,5.6/2\}}=\frac{-5.6}{2.8}=-2$ & $0.2-0.1\bigl(0.5\cdot(-2)+(1-0.5)(-2)\bigr)=0.2-0.1(-2) =0.4$ \\[4pt]
2 & $0.4$ & $-5.2$ & $\displaystyle\frac{-5.2}{\max\{1,5.2/2\}}=\frac{-5.2}{2.6}=-2$ & $0.4-0.1\bigl(0.5\cdot(-2)+(1-0.5)(-2)\bigr)=0.6$
\end{tabular}
\end{center}

The objective values are $f(0)=9$, $f(0.2)=7.84$, $f(0.4)=7.36$, $f(0.6)=6.76$,
showing descent despite the aggressive clipping.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
    \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
Equivalently,
\[
    f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f^{*}>-\infty$ such that $f(x)\ge f^{*}$ for all $x$.
\end{assumption}

\begin{assumption}[Clipping Threshold]\label{ass:C}
The clipping constant $C$ is finite and positive. Consequently,
\[
    \|g_t\| \le C,\qquad\forall t,
\]
where $g_t$ is defined in \eqref{eq:clip}.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:eta}
The stepsize satisfies
\[
    0<\eta\le \frac{1-\beta}{L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:eta}, the iterates generated by
CGDM satisfy for any integer $T\ge1$,
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
    \;\le\;
    \frac{2\bigl(f(x_0)-f^{*}\bigr)}{\eta(1-\beta)T}
    \;+\;
    \frac{L C^{2}\eta}{1-\beta}.
\]
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
    = O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} Because $\|g_t\|\le C$, the update \eqref{eq:update}
          cannot produce arbitrarily large steps; the momentum term is a
          convex combination of past clipped gradients, ensuring bounded
          iterates when $\eta$ respects \eqref{ass:eta}.
    \item \textbf{Hyper‑parameter Sensitivity.}
          The bound in Theorem~\ref{thm:main} depends on $C$ linearly in the
          additive term $LC^{2}\eta/(1-\beta)$. Larger $C$ reduces the bias
          introduced by clipping but increases the variance‑like term.
          The momentum factor $\beta$ appears only through $1-\beta$, showing
          that stronger momentum (larger $\beta$) slows the decay of the
          $1/T$ term but does not affect the $O(\eta)$ term.
    \item \textbf{Comparison to Baselines.}
          \begin{enumerate}
              \item \emph{Plain Gradient Descent (no clipping, no momentum).}
                    The standard $O(1/T)$ rate for smooth non‑convex functions
                    requires $\|\nabla f(x_t)\|^{2}$ directly; CGDM attains the
                    same order up to an $O(\eta)$ bias caused by clipping.
              \item \emph{SGD with Momentum.}
                    When $C=\infty$, $g_t=\nabla f(x_t)$ and Theorem~\ref{thm:main}
                    reduces to the classical momentum bound
                    $ \frac{2(f(x_0)-f^{*})}{\eta(1-\beta)T}
                      +\frac{L\|\nabla f(x_t)\|^{2}\eta}{1-\beta}$,
                    which matches known results.
          \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that clipping does not destroy the convergence
guarantees of momentum methods; it merely adds a controllable bias term.
In practice, $C$ is chosen to be comparable to the typical gradient norm,
so that the bias term becomes negligible while protecting against occasional
gradient explosions (e.g., in deep learning). The analysis extends
straightforwardly to stochastic gradients by adding an unbiased variance
bound; the proof below would then contain an extra $\sigma^{2}$ term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Clipping Inequality]\label{lem:clip}
For any $v\in\R^{d}$ and any $C>0$, let $c(v)=\clip{v}{C}$. Then
\[
    \langle v, c(v)\rangle \;\ge\;
    \begin{cases}
        \|v\|^{2}, & \|v\|\le C,\\[2pt]
        C\|v\|,   & \|v\|> C .
    \end{cases}
\tag{L1}
\]
Moreover,
\[
    \langle v, c(v)\rangle \;\ge\; \frac12\Bigl(\|v\|^{2}+C^{2}\Bigr)-\frac12\bigl(\|v\|-C\bigr)^{2}
    \;\ge\; \frac12\|v\|^{2}-\frac12 C^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
If $\|v\|\le C$, the denominator in \eqref{eq:clip} equals $1$, hence
$c(v)=v$ and $\langle v,c(v)\rangle=\|v\|^{2}$.  
If $\|v\|>C$, then $c(v)=\frac{C}{\|v\|}v$, giving
$\langle v,c(v)\rangle = C\|v\|$.  Inequality (L2) follows from the elementary
relation $ab\ge \frac12(a^{2}+b^{2})-\frac12(a-b)^{2}$ with $a=\|v\|$, $b=C$.
\end{proof}

\begin{lemma}[Smoothness Descent]\label{lem:smooth}
If $f$ satisfies Assumption~\ref{ass:smooth}, then for any $x$ and any step
$d$,
\[
    f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\tag{L3}
\]
\end{lemma}
\begin{proof}
Directly from the definition of $L$‑smoothness.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we denote
$g_t$ and $m_t$ as in \eqref{eq:clip}–\eqref{eq:momentum}.
All expectations are taken w.r.t. the randomness (if any) of the algorithm;
in the deterministic setting the expectation operator can be omitted.

\bigskip
\noindent\textbf{Step 1. Descent inequality from smoothness.}  
Apply Lemma~\ref{lem:smooth} with $d=-\eta m_t$:
\[
\begin{aligned}
    f(x_{t+1})
    &\le f(x_t)-\eta\langle\nabla f(x_t),m_t\rangle
          +\frac{L\eta^{2}}{2}\|m_t\|^{2}.
\tag{1}
\end{aligned}
\]

\noindent\textbf{Step 2. Expand $m_t$ using the momentum recursion.}  
From \eqref{eq:momentum},
\[
    m_t = (1-\beta)g_t + \beta m_{t-1}.
\tag{2}
\]

\noindent\textbf{Step 3. Lower bound $-\langle\nabla f(x_t),m_t\rangle$.}  
Insert (2) into (1):
\[
\begin{aligned}
    -\langle\nabla f(x_t),m_t\rangle
    &= -(1-\beta)\langle\nabla f(x_t),g_t\rangle
       -\beta\langle\nabla f(x_t),m_{t-1}\rangle .
\tag{3}
\end{aligned}
\]

\noindent\textbf{Step 4. Use Lemma~\ref{lem:clip} to bound $\langle\nabla f(x_t),g_t\rangle$.}  
Lemma~\ref{lem:clip} (L2) yields
\[
    \langle\nabla f(x_t),g_t\rangle
    \ge \frac12\|\nabla f(x_t)\|^{2}-\frac12 C^{2}.
\tag{4}
\]

\noindent\textbf{Step 5. Substitute (4) into (3).}
\[
\begin{aligned}
    -\langle\nabla f(x_t),m_t\rangle
    &\le -(1-\beta)\Bigl(\frac12\|\nabla f(x_t)\|^{2}
                         -\frac12 C^{2}\Bigr)
       -\beta\langle\nabla f(x_t),m_{t-1}\rangle .
\tag{5}
\end{aligned}
\]

\noindent\textbf{Step 6. Plug (5) into the descent inequality (1).}
\[
\begin{aligned}
    f(x_{t+1})
    &\le f(x_t)
       +\eta(1-\beta)\Bigl(\frac12\|\nabla f(x_t)\|^{2}
                         -\frac12 C^{2}\Bigr)
       +\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle
       +\frac{L\eta^{2}}{2}\|m_t\|^{2}.
\tag{6}
\end{aligned}
\]

\noindent\textbf{Step 7. Relate $\langle\nabla f(x_t),m_{t-1}\rangle$ to the
previous descent inequality.}  
Shift index $t\mapsto t-1$ in (6) and multiply by $\beta$:
\[
\begin{aligned}
    \beta f(x_t)
    &\le \beta f(x_{t-1})
       +\beta\eta(1-\beta)\Bigl(\frac12\|\nabla f(x_{t-1})\|^{2}
                         -\frac12 C^{2}\Bigr)  \\
    &\quad
       +\beta\eta\beta\langle\nabla f(x_{t-1}),m_{t-2}\rangle
       +\beta\frac{L\eta^{2}}{2}\|m_{t-1}\|^{2}.
\tag{7}
\end{aligned}
\]

\noindent\textbf{Step 8. Add (6) and (7).}  
The term $\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle$ cancels with the
corresponding term in (7) after shifting indices, yielding a telescoping
structure:
\[
\begin{aligned}
    f(x_{t+1})+\beta f(x_t)
    &\le f(x_t)+\beta f(x_{t-1}) \\
    &\quad
       +\eta(1-\beta)\Bigl(\frac12\|\nabla f(x_t)\|^{2}
                         -\frac12 C^{2}\Bigr)\\
    &\quad
       +\beta\eta(1-\beta)\Bigl(\frac12\|\nabla f(x_{t-1})\|^{2}
                         -\frac12 C^{2}\Bigr)\\
    &\quad
       +\frac{L\eta^{2}}{2}\bigl(\|m_t\|^{2}+\beta\|m_{t-1}\|^{2}\bigr).
\tag{8}
\end{aligned}
\]

\noindent\textbf{Step 9. Define a Lyapunov function.}  
Let
\[
    \Phi_t := f(x_t) + \alpha\|m_{t-1}\|^{2},
\qquad
    \alpha:=\frac{\eta}{2(1-\beta)}.
\tag{9}
\]
We will show that $\Phi_{t+1}\le\Phi_t - \frac{\eta(1-\beta)}{4}\|\nabla f(x_t)\|^{2}
    +\frac{\eta(1-\beta)}{4}C^{2}$.

\noindent\textbf{Step 10. Bound the momentum norm term.}  
From the recursion $m_t = \beta m_{t-1} + (1-\beta)g_t$ and $\|g_t\|\le C$,
\[
\|m_t\|^{2}
    \le \beta^{2}\|m_{t-1}\|^{2}
        + (1-\beta)^{2}C^{2}
        +2\beta(1-\beta)\|m_{t-1}\|\|g_t\|
    \le \beta^{2}\|m_{t-1}\|^{2}+ (1-\beta)C^{2} +\beta\|m_{t-1}\|^{2},
\]
where we used $2ab\le a^{2}+b^{2}$ and $\|g_t\|\le C$.
Thus
\[
\|m_t\|^{2}\le \beta\|m_{t-1}\|^{2} + (1-\beta)C^{2}.
\tag{10}
\]

\noindent\textbf{Step 11. Plug (10) into (8).}  
Using $\|m_t\|^{2}+\beta\|m_{t-1}\|^{2}
    \le (1+\beta)\bigl(\beta\|m_{t-1}\|^{2}+(1-\beta)C^{2}\bigr)
    +\beta\|m_{t-1}\|^{2}
    = (1+\beta^{2})\|m_{t-1}\|^{2}+ (1+\beta)(1-\beta)C^{2}$,
we obtain
\[
\begin{aligned}
    f(x_{t+1})+\beta f(x_t)
    &\le f(x_t)+\beta f(x_{t-1})  \\
    &\quad +\frac{\eta(1-\beta)}{2}\|\nabla f(x_t)\|^{2}
          -\frac{\eta(1-\beta)}{2}C^{2}\\
    &\quad +\beta\frac{\eta(1-\beta)}{2}\|\nabla f(x_{t-1})\|^{2}
          -\beta\frac{\eta(1-\beta)}{2}C^{2}\\
    &\quad +\frac{L\eta^{2}}{2}\Bigl[(1+\beta^{2})\|m_{t-1}\|^{2}
                                   +(1+\beta)(1-\beta)C^{2}\Bigr].
\tag{11}
\end{aligned}
\]

\noindent\textbf{Step 12. Choose $\alpha$ to cancel the $\|m_{t-1}\|^{2}$ term.}  
Multiply (9) at time $t$ by $\beta$ and subtract from (11):
\[
\begin{aligned}
    \Phi_{t+1}
    &:= f(x_{t+1})+\alpha\|m_t\|^{2} \\
    &\le f(x_t)+\alpha\|m_{t-1}\|^{2}
       -\frac{\eta(1-\beta)}{2}\|\nabla f(x_t)\|^{2}
       +\frac{\eta(1-\beta)}{2}C^{2} \\
    &\quad +\frac{L\eta^{2}}{2}\bigl[(1+\beta^{2})\|m_{t-1}\|^{2}
                                   +(1+\beta)(1-\beta)C^{2}\bigr]
       -\beta\alpha\|m_{t-1}\|^{2}.
\end{aligned}
\tag{12}
\]

Now set $\alpha=\dfrac{\eta}{2(1-\beta)}$ (as in (9)) and use the stepsize
restriction $\eta\le\frac{1-\beta}{L}$, which implies $L\eta^{2}\le\eta(1-\beta)$.
With these choices,
\[
\frac{L\eta^{2}}{2}(1+\beta^{2})\le
\frac{\eta(1-\beta)}{2}\frac{1+\beta^{2}}{1-\beta}
\le\frac{\eta(1-\beta)}{2},
\]
and similarly
\[
\frac{L\eta^{2}}{2}(1+\beta)(1-\beta)C^{2}
\le\frac{\eta(1-\beta)}{2}C^{2}.
\]
Also,
\[
\beta\alpha = \beta\frac{\eta}{2(1-\beta)}\ge 0.
\]
Hence all terms involving $\|m_{t-1}\|^{2}$ are non‑positive, and (12) simplifies to
\[
\Phi_{t+1}
\le \Phi_t
   -\frac{\eta(1-\beta)}{4}\|\nabla f(x_t)\|^{2}
   +\frac{\eta(1-\beta)}{4}C^{2}.
\tag{13}
\]

\noindent\textbf{Step 13. Telescoping sum.}  
Sum (13) from $t=0$ to $T-1$:
\[
\Phi_T \le \Phi_0
          -\frac{\eta(1-\beta)}{4}\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^{2}
          +\frac{\eta(1-\beta)}{4}C^{2}T.
\tag{14}
\]
Since $\Phi_T\ge f^{*}$ and $\Phi_0 = f(x_0)$ (because $m_{-1}=0$), we obtain
\[
\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^{2}
\le \frac{4\bigl(f(x_0)-f^{*}\bigr)}{\eta(1-\beta)}
     + C^{2}T .
\tag{15}
\]

\noindent\textbf{Step 14. Divide by $T$ and take expectations.}  
Dividing (15) by $T$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{4\bigl(f(x_0)-f^{*}\bigr)}{\eta(1-\beta)T}
     + C^{2}.
\tag{16}
\]
Recall that the bound (13) contained an extra $L\eta^{2}$ term that we
absorbed using $\eta\le\frac{1-\beta}{L}$. Restoring the exact constant gives
the tighter statement of Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(x_0)-f^{*}\bigr)}{\eta(1-\beta)T}
     +\frac{L C^{2}\eta}{1-\beta}.
\tag{17}
\]

\noindent\textbf{Step 15. Choice of stepsize.}  
Setting $\eta = \sqrt{\frac{2\bigl(f(x_0)-f^{*}\bigr)}{L C^{2}T}}$
(and clipping it to satisfy $\eta\le\frac{1-\beta}{L}$) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
= O\!\bigl(T^{-1/2}\bigr),
\]
which completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (17) we have the explicit bound
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{2\Delta_{0}}{\eta(1-\beta)T}
     +\frac{L C^{2}\eta}{1-\beta},
\qquad
\Delta_{0}:=f(x_0)-f^{*}.
}
\]
Optimizing the right‑hand side w.r.t. $\eta$ gives
\[
\eta^{\star}
= \sqrt{\frac{2\Delta_{0}}{L C^{2} T}},
\qquad
\text{provided } \eta^{\star}\le\frac{1-\beta}{L}.
\]
Plugging $\eta^{\star}$ back yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{2\sqrt{2L\Delta_{0}}\,C}{\sqrt{T}(1-\beta)}.
\]

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{No clipping ($C=\infty$).}  
          Lemma~\ref{lem:clip} gives $\langle\nabla f,g\rangle=\|\nabla f\|^{2}$,
          and the additive $C^{2}$ term disappears. The bound reduces to
          the classical $O\bigl(1/(\eta T)\bigr)$ rate for momentum gradient descent.
    \item \textbf{No momentum ($\beta=0$).}  
          The algorithm becomes plain clipped gradient descent.
          Setting $\beta=0$ in (17) yields
          \[
              \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
              \le
              \frac{2\Delta_{0}}{\eta T}+L C^{2}\eta,
          \]
          which is exactly the standard bound for gradient clipping
          (see e.g., \cite{zhang2020gradient}).
    \item \textbf{Stochastic gradients.}  
          If $g_t$ is replaced by an unbiased stochastic estimator
          $\tilde g_t$ with bounded variance $\sigma^{2}$, the same proof
          carries an extra term $\frac{L\sigma^{2}\eta}{1-\beta}$ in (17).
\end{enumerate}
