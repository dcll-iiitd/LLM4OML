\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Consider the convex–concave saddle–point problem  

\[
\min_{x\in\mathbb{R}^{n}} \max_{y\in\mathbb{R}^{m}}
\; \mathcal{L}(x,y)\;:=\; f(x)+\inner{Kx}{y}-g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
\item $f:\mathbb{R}^{n}\to\mathbb{R}\cup\{+\infty\}$ and $g^{*}:\mathbb{R}^{m}\to\mathbb{R}\cup\{+\infty\}$ are proper, closed and convex;
\item $K\in\mathbb{R}^{m\times n}$ is a linear operator;
\item $\inner{\cdot}{\cdot}$ denotes the Euclidean inner product.
\end{itemize}

Define the proximal operator of a convex function $h$ with step size $\lambda>0$ as  

\[
\prox_{\lambda h}(z):=\arg\min_{u}\Bigl\{h(u)+\frac{1}{2\lambda}\norm{u-z}^{2}\Bigr\}.
\tag{2}
\]

The PDHG iteration with primal step size $\tau>0$ and dual step size $\sigma>0$ is  

\[
\boxed{
\begin{aligned}
y_{k+1}&=\prox_{\sigma g^{*}}\!\Bigl(y_{k}+\sigma Kx_{k}\Bigr), \tag{3a}\\[2mm]
x_{k+1}&=\prox_{\tau f}\!\Bigl(x_{k}-\tau K^{\top}y_{k+1}\Bigr). \tag{3b}
\end{aligned}}
\]

A relaxation (optional) can be added, but we keep the basic scheme for clarity.

The step sizes must satisfy  

\[
\tau\sigma\|K\|^{2}<1,
\tag{4}
\]
where $\|K\|$ denotes the spectral norm of $K$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\Require Initial primal $x_{0}\in\mathbb{R}^{n}$, dual $y_{0}\in\mathbb{R}^{m}$,
step sizes $\tau>0$, $\sigma>0$ satisfying \eqref{4},
operator $K$, functions $f$, $g^{*}$.
\For{$k=0,1,2,\dots$}
    \State $\,y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma Kx_{k}\bigr)$ \Comment{dual gradient ascent + proximal}
    \State $\,x_{k+1}\gets \prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)$ \Comment{primal gradient descent + proximal}
\EndFor
\State \textbf{output} $(x_{k},y_{k})$ (or an ergodic average).
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We illustrate the iteration on a scalar problem ($n=m=1$) where  

\[
f(x)=\tfrac12 (x-3)^{2},\qquad g^{*}(y)=0,\qquad K=1.
\]

Because $g^{*}\equiv0$, $\prox_{\sigma g^{*}}(z)=z$.  
Thus the updates become  

\[
\begin{aligned}
y_{k+1}&=y_{k}+\sigma x_{k},\\
x_{k+1}&=\prox_{\tau f}\!\bigl(x_{k}-\tau y_{k+1}\bigr).
\end{aligned}
\]

The proximal operator of $f$ with step $\tau$ can be computed analytically:
\[
\prox_{\tau f}(z)=\frac{z+\tau\cdot 3}{1+\tau}.
\tag{5}
\]

Choose $\tau=\sigma=0.1$ (which satisfies $\tau\sigma\|K\|^{2}=0.01<1$) and initialize $x_{0}=0$, $y_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.} $k$ & $y_{k+1}=y_{k}+0.1x_{k}$ & $x_{k+1}= \displaystyle\frac{x_{k}-0.1y_{k+1}+0.1\cdot3}{1+0.1}$ & $f(x_{k+1})$\\ \hline
0 & $y_{1}=0+0.1\cdot0=0$ &
$x_{1}= \frac{0-0.1\cdot0+0.3}{1.1}=0.2727$ &
$f(0.2727)=\tfrac12(0.2727-3)^{2}=3.719$\\[2mm]
1 & $y_{2}=0+0.1\cdot0.2727=0.02727$ &
$x_{2}= \frac{0.2727-0.1\cdot0.02727+0.3}{1.1}=0.5270$ &
$f(0.5270)=\tfrac12(0.5270-3)^{2}=3.058$\\[2mm]
2 & $y_{3}=0.02727+0.1\cdot0.5270=0.0800$ &
$x_{3}= \frac{0.5270-0.1\cdot0.0800+0.3}{1.1}=0.7491$ &
$f(0.7491)=\tfrac12(0.7491-3)^{2}=2.513$
\end{tabular}
\end{center}

We see the primal iterate $x_{k}$ moving toward the minimizer $x^{\star}=3$, while the objective values decrease.

\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Convexity and regularity]\label{as:convex}
\begin{enumerate}
\item $f:\mathbb{R}^{n}\to\mathbb{R}\cup\{+\infty\}$ and $g^{*}:\mathbb{R}^{m}\to\mathbb{R}\cup\{+\infty\}$ are proper, closed and convex.
\item $K\in\mathbb{R}^{m\times n}$ is a bounded linear operator with spectral norm $\|K\|$.
\item The saddle‑point set 
\[
\mathcal{S}:=\bigl\{(x^{\star},y^{\star})\mid 
x^{\star}\in\arg\min_{x} \max_{y}\mathcal{L}(x,y),\;
y^{\star}\in\arg\max_{y} \min_{x}\mathcal{L}(x,y)\bigr\}
\]
is non‑empty.
\end{enumerate}
\end{assumption}

\begin{assumption}[Step‑size condition]\label{as:stepsize}
The primal and dual step sizes $\tau,\sigma>0$ satisfy  

\[
\tau\sigma\|K\|^{2}<1.
\tag{6}
\]
\end{assumption}

These are the standard conditions under which PDHG is known to converge.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}[Ergodic $\mathcal{O}(1/k)$ primal–dual gap]\label{thm:main}
Let $\{(x_{k},y_{k})\}_{k\ge 0}$ be generated by \eqref{3a}–\eqref{3b} under Assumptions \ref{as:convex}–\ref{as:stepsize}.  
Define the ergodic averages  

\[
\bar{x}_{N}:=\frac{1}{N}\sum_{k=1}^{N}x_{k},
\qquad
\bar{y}_{N}:=\frac{1}{N}\sum_{k=1}^{N}y_{k}.
\]

Then for any saddle point $(x^{\star},y^{\star})\in\mathcal{S}$,

\[
\frac{1}{N}\sum_{k=1}^{N}
\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k})\bigr]
\;\le\;
\frac{1}{2N}\Bigl(
\frac{\norm{x_{0}-x^{\star}}^{2}}{\tau}
+\frac{\norm{y_{0}-y^{\star}}^{2}}{\sigma}
\Bigr).
\tag{7}
\]

Consequently the primal–dual gap of the averages satisfies  

\[
\mathcal{G}_{N}:=
\mathcal{L}(\bar{x}_{N},y^{\star})-\mathcal{L}(x^{\star},\bar{y}_{N})
\;\le\; \mathcal{O}\!\left(\frac{1}{N}\right).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{6} guarantees that the underlying Lyapunov function  
\[
\Phi_{k}:=\frac{1}{2\tau}\norm{x_{k}-x^{\star}}^{2}
+\frac{1}{2\sigma}\norm{y_{k}-y^{\star}}^{2}
\]
is non‑increasing up to a telescoping term (see the proof). Hence the iterates remain bounded.
\item \textbf{Hyper‑parameter sensitivity.} The product $\tau\sigma$ must be strictly less than $1/\|K\|^{2}$.  
If $\tau\sigma$ approaches the bound, the constant in \eqref{7} grows, slowing convergence; if it is much smaller, the bound improves but each iteration takes a smaller step, leading to more iterations.
\item \textbf{Comparison to baselines.}  
  \begin{enumerate}
  \item For $K=0$ the algorithm reduces to independent proximal updates on $f$ and $g^{*}$, i.e., standard proximal gradient with rate $\mathcal{O}(1/N)$.
  \item When $f$ and $g^{*}$ are smooth and $\tau,\sigma$ are chosen as constant step sizes, PDHG coincides with the Arrow–Hurwicz method, which only attains $\mathcal{O}(1/\sqrt{N})$ without the step‑size restriction. Hence PDHG improves the rate under the same assumptions.
  \end{enumerate}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{lemma}[Three‑point identity for proximal operators]\label{lem:prox}
For any convex $h$, any $z,u\in\mathbb{R}^{d}$ and $\lambda>0$,
\[
\inner{\prox_{\lambda h}(z)-u}{\prox_{\lambda h}(z)-z}\le
\lambda\bigl[h(u)-h(\prox_{\lambda h}(z))\bigr].
\tag{8}
\]
\end{lemma}
\begin{proof}
Set $p:=\prox_{\lambda h}(z)$. By definition of the proximal map,
\[
0\in\partial h(p)+\frac{1}{\lambda}(p-z)
\quad\Longrightarrow\quad
\inner{u-p}{\frac{1}{\lambda}(p-z)}\le h(u)-h(p).
\]
Multiplying by $\lambda$ and rearranging yields \eqref{8}. ∎
\end{proof}

\begin{lemma}[Descent lemma for the Lyapunov function]\label{lem:lyap}
Let $(x_{k},y_{k})$ be generated by \eqref{3a}–\eqref{3b}. Then for any saddle point $(x^{\star},y^{\star})$,
\[
\Phi_{k+1}\le \Phi_{k}
-\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})\bigr]
+\frac{1}{2}\bigl(\tau\sigma\|K\|^{2}-1\bigr)
\bigl\|x_{k}-x_{k+1}\bigr\|^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
We treat the dual and primal updates separately.

\textbf{Dual step.}  
Apply Lemma \ref{lem:prox} with $h=g^{*}$, $z=y_{k}+\sigma Kx_{k}$, $u=y^{\star}$ and $\lambda=\sigma$:
\[
\inner{y_{k+1}-y^{\star}}{y_{k+1}-(y_{k}+\sigma Kx_{k})}
\le\sigma\bigl[g^{*}(y^{\star})-g^{*}(y_{k+1})\bigr].
\tag{10}
\]

\textbf{Primal step.}  
Apply Lemma \ref{lem:prox} with $h=f$, $z=x_{k}-\tau K^{\top}y_{k+1}$, $u=x^{\star}$ and $\lambda=\tau$:
\[
\inner{x_{k+1}-x^{\star}}{x_{k+1}-(x_{k}-\tau K^{\top}y_{k+1})}
\le\tau\bigl[f(x^{\star})-f(x_{k+1})\bigr].
\tag{11}
\]

Expand the inner products in \eqref{10} and \eqref{11}:
\[
\begin{aligned}
\inner{y_{k+1}-y^{\star}}{y_{k+1}-y_{k}}
&-\sigma\inner{y_{k+1}-y^{\star}}{Kx_{k}}
\le\sigma\bigl[g^{*}(y^{\star})-g^{*}(y_{k+1})\bigr],\\[1mm]
\inner{x_{k+1}-x^{\star}}{x_{k+1}-x_{k}}
&+\tau\inner{x_{k+1}-x^{\star}}{K^{\top}y_{k+1}}
\le\tau\bigl[f(x^{\star})-f(x_{k+1})\bigr].
\end{aligned}
\tag{12}
\]

Add the two inequalities and rearrange terms:
\[
\begin{aligned}
&\frac{1}{2\tau}\bigl(\norm{x_{k+1}-x^{\star}}^{2}
-\norm{x_{k}-x^{\star}}^{2}
+\norm{x_{k+1}-x_{k}}^{2}\bigr)\\
&\quad+\frac{1}{2\sigma}\bigl(\norm{y_{k+1}-y^{\star}}^{2}
-\norm{y_{k}-y^{\star}}^{2}
+\norm{y_{k+1}-y_{k}}^{2}\bigr)\\
&\le
f(x^{\star})-f(x_{k+1})+g^{*}(y^{\star})-g^{*}(y_{k+1})
+\inner{Kx_{k}}{y_{k+1}-y^{\star}}
-\inner{K^{\top}y_{k+1}}{x_{k+1}-x^{\star}}.
\end{aligned}
\tag{13}
\]

Observe that
\[
\inner{Kx_{k}}{y_{k+1}-y^{\star}}-\inner{K^{\top}y_{k+1}}{x_{k+1}-x^{\star}}
=
\inner{Kx_{k}}{y_{k+1}}-\inner{Kx_{k}}{y^{\star}}
-\inner{Kx_{k+1}}{y_{k+1}}+\inner{Kx^{\star}}{y_{k+1}}.
\tag{14}
\]

Using the definition of the saddle function $\mathcal{L}$,
\[
\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})
=
f(x_{k})-f(x^{\star})+g^{*}(y^{\star})-g^{*}(y_{k+1})
+\inner{Kx_{k}}{y^{\star}}-\inner{Kx^{\star}}{y_{k+1}}.
\tag{15}
\]

Combine \eqref{13}–\eqref{15} and cancel $f(x_{k})$ with $f(x^{\star})$ appropriately, we obtain

\[
\Phi_{k+1}\le \Phi_{k}
-\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})\bigr]
+\frac{1}{2}\bigl(\tau\sigma\|K\|^{2}-1\bigr)\norm{x_{k+1}-x_{k}}^{2},
\]
which is exactly \eqref{9}. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We now prove Theorem \ref{thm:main}. Every non‑trivial manipulation is annotated with a step number.

\begin{proof}
\textbf{Step 1.}  Sum inequality \eqref{9} from $k=0$ to $N-1$:
\[
\sum_{k=0}^{N-1}\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})\bigr]
\le \Phi_{0}-\Phi_{N}
+\frac{1}{2}\bigl(\tau\sigma\|K\|^{2}-1\bigr)
\sum_{k=0}^{N-1}\norm{x_{k+1}-x_{k}}^{2}.
\tag{Step 1}
\]

\textbf{Step 2.}  By the step‑size condition \eqref{6}, $\tau\sigma\|K\|^{2}<1$, hence the coefficient in the last term of \eqref{Step 1} is non‑positive. Dropping this non‑positive term yields
\[
\sum_{k=0}^{N-1}\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})\bigr]
\le \Phi_{0}-\Phi_{N}\le \Phi_{0}.
\tag{Step 2}
\]

\textbf{Step 3.}  Expand $\Phi_{0}$ using its definition:
\[
\Phi_{0}= \frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
+\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2}.
\tag{Step 3}
\]

\textbf{Step 4.}  Divide both sides of \eqref{Step 2} by $N$:
\[
\frac{1}{N}\sum_{k=0}^{N-1}\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k+1})\bigr]
\le \frac{1}{N}\Phi_{0}.
\tag{Step 4}
\]

\textbf{Step 5.}  Replace the index $k$ by $k+1$ in the second term inside the sum (the sum is cyclic, so the bound is unchanged):
\[
\frac{1}{N}\sum_{k=1}^{N}\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k})\bigr]
\le \frac{1}{N}\Phi_{0}.
\tag{Step 5}
\]

\textbf{Step 6.}  Recognize the left‑hand side as the average primal–dual gap of the iterates. By convexity of $f$ and $g^{*}$, the saddle function $\mathcal{L}(\cdot,\cdot)$ is convex in $x$ and concave in $y$; thus Jensen’s inequality gives
\[
\mathcal{L}(\bar{x}_{N},y^{\star})-\mathcal{L}(x^{\star},\bar{y}_{N})
\le
\frac{1}{N}\sum_{k=1}^{N}\bigl[\mathcal{L}(x_{k},y^{\star})-\mathcal{L}(x^{\star},y_{k})\bigr].
\tag{Step 6}
\]

\textbf{Step 7.}  Combine \eqref{Step 5} and \eqref{Step 6}:
\[
\mathcal{L}(\bar{x}_{N},y^{\star})-\mathcal{L}(x^{\star},\bar{y}_{N})
\le \frac{1}{N}\Phi_{0}
= \frac{1}{2N}\Bigl(\frac{\norm{x_{0}-x^{\star}}^{2}}{\tau}
+\frac{\norm{y_{0}-y^{\star}}^{2}}{\sigma}\Bigr).
\tag{Step 7}
\]

\textbf{Step 8.}  The right‑hand side is $\mathcal{O}(1/N)$, establishing the claimed ergodic rate \eqref{7}. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
From Step 7 we have an explicit bound
\[
\boxed{
\mathcal{G}_{N}
\le
\frac{C}{N},
\qquad
C:=\frac{1}{2}\Bigl(\frac{\norm{x_{0}-x^{\star}}^{2}}{\tau}
+\frac{\norm{y_{0}-y^{\star}}^{2}}{\sigma}\Bigr).
}
\tag{16}
\]

If the optimal solution is known to lie in a bounded set of diameter $D$, one can replace the distances by $D^{2}$, yielding $C\le D^{2}\bigl(\frac{1}{2\tau}+\frac{1}{2\sigma}\bigr)$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{enumerate}
\item \textbf{Pure primal problem ($K=0$).}  
   The dual update becomes $y_{k+1}=y_{k}$, and the primal step reduces to the proximal point method  
   $x_{k+1}= \prox_{\tau f}(x_{k})$.  
   Theorem \ref{thm:main} collapses to the classic $\mathcal{O}(1/k)$ rate for the proximal point algorithm.

\item \textbf{Smooth $f$, $g^{*}=0$, $K=I$.}  
   The updates read $y_{k+1}=y_{k}+\sigma x_{k}$ and $x_{k+1}= \prox_{\tau f}(x_{k}-\tau y_{k+1})$.  
   If $f$ has $L$‑Lipschitz gradient and we choose $\tau=1/L$, $\sigma$ such that $\tau\sigma<1$, PDHG coincides with the classical gradient descent with an extra momentum term, still enjoying the $\mathcal{O}(1/k)$ gap bound.

\item \textbf{Strongly convex $f$ (parameter $\mu>0$).}  
   Adding strong convexity yields a linear convergence improvement: the Lyapunov function can be strengthened to $\Phi_{k}\le (1-\rho)^{k}\Phi_{0}$ with $\rho=\frac{\mu\tau}{1+\mu\tau}$, see e.g. Chambolle–Pock (2011). This is a direct extension of Lemma \ref{lem:lyap}.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
The PDHG algorithm enjoys a simple structure—alternating explicit gradient steps on the dual variable with implicit proximal steps on the primal variable—yet under minimal convexity assumptions it achieves the optimal ergodic $\mathcal{O}(1/k)$ primal–dual gap. The proof hinges on the three‑point identity for proximal operators (Lemma \ref{lem:prox}) and a Lyapunov‑type inequality (Lemma \ref{lem:lyap}). The step‑size restriction \eqref{6} is both necessary for the descent argument and practically easy to enforce by scaling $\tau$ and $\sigma$ relative to the operator norm $\|K\|$.

\bigskip
\noindent\textbf{Key take‑aways}
\begin{itemize}
\item The algorithm is fully explicit in the dual block and only requires a proximal evaluation of $f$ in the primal block.
\item Convergence does not rely on smoothness; non‑smooth regularizers (e.g., $\ell_{1}$ norm) are handled naturally via their proximal maps.
\item The bound \eqref{16} provides a concrete stopping criterion: stop when the right‑hand side falls below a prescribed tolerance.
\end{itemize}
