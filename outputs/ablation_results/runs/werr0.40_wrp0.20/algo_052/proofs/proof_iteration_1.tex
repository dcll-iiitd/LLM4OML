\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics with Decreasing Noise (SGLD‑D)}  

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be the objective function.  
Given an initial point $x_0\in\R^d$, a stepsize (learning rate) sequence $\{\alpha_k\}_{k\ge0}$ with $\alpha_k>0$, and a noise‑variance schedule $\{\sigma_k^2\}_{k\ge0}$, the iteration is  

\[
\boxed{%
x_{k+1}=x_k-\alpha_k \nabla f(x_k)+\xi_k, \qquad 
\xi_k\sim\mathcal N\bigl(0,\sigma_k^{2} I_d\bigr)
}
\tag{1}
\]

The noise is independent of the past iterates and of the gradient.  
The variance schedule is assumed to be \emph{decreasing}:
\[
\sigma_{k+1}^2\le\sigma_k^2,\qquad \lim_{k\to\infty}\sigma_k^2=0.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD‑D}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\R^d$, stepsize sequence $\{\alpha_k\}$, noise schedule $\{\sigma_k^2\}$, maximum iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute the (full) gradient $g_k=\nabla f(x_k)$.
    \STATE Sample $\xi_k\sim\mathcal N(0,\sigma_k^2 I_d)$.
    \STATE Update $x_{k+1}=x_k-\alpha_k g_k+\xi_k$.
\ENDFOR
\RETURN $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  

\[
f(w)=(w-3)^2 .
\]

Then $\nabla f(w)=2(w-3)$.  
Choose $x_0=0$, constant stepsize $\alpha_k=\alpha=0.1$, and a deterministic “noise’’ sequence $\xi_k=0$ (to illustrate the deterministic part).  

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $x_k$ & $\nabla f(x_k)=2(x_k-3)$ & $x_{k+1}=x_k-\alpha\nabla f(x_k)$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}

The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3450$, showing convergence toward the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The function $f$ is convex and $L$‑smooth, i.e. for all $x,y\in\R^d$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\tag{3}
\]
\end{assumption}

\begin{assumption}[Existence of a Minimizer]\label{ass:opt}
There exists $x^\star\in\arg\min_{x\in\R^d}f(x)$ and $f^\star:=f(x^\star)$.
\end{assumption}

\begin{assumption}[Noise Moments]\label{ass:noise}
The random vectors $\{\xi_k\}$ satisfy  
\[
\E[\xi_k]=0,\qquad \E[\xi_k\xi_k^\top]=\sigma_k^{2} I_d ,
\tag{4}
\]
and are independent of $\{x_j,g_j\}_{j\le k}$.
\end{assumption}

\begin{assumption}[Stepsize and Variance Schedules]\label{ass:schedule}
The stepsizes $\{\alpha_k\}$ are non‑increasing, $\alpha_k\le\alpha_0$, and satisfy  

\[
\sum_{k=0}^{\infty}\alpha_k =\infty,\qquad 
\sum_{k=0}^{\infty}\alpha_k^2<\infty .
\tag{5}
\]

The noise variances obey  

\[
\sum_{k=0}^{\infty}\alpha_k\sigma_k^{2}<\infty .
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Function‑Value Convergence]\label{thm:conv}
Under Assumptions \ref{ass:convex}–\ref{ass:schedule}, the iterates generated by \eqref{eq:1} satisfy  

\[
\E\!\bigl[f(x_k)\bigr]-f^\star
\;\le\;
\frac{\norm{x_0-x^\star}^2}{2\sum_{t=0}^{k-1}\alpha_t}
\;+\;
\frac{L}{2}\,\frac{\sum_{t=0}^{k-1}\alpha_t^2}{\sum_{t=0}^{k-1}\alpha_t}
\;+\;
\frac{\sum_{t=0}^{k-1}\alpha_t\sigma_t^{2}}{\sum_{t=0}^{k-1}\alpha_t}\; .
\tag{7}
\]

Consequently, if $\alpha_k=\frac{c}{k+1}$ and $\sigma_k^{2}= \frac{c'}{(k+1)^{1+\epsilon}}$ with $\epsilon>0$, then  

\[
\E\!\bigl[f(x_k)\bigr]-f^\star = O\!\left(\frac{1}{k}\right) .
\tag{8}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\sum\alpha_k^2<\infty$ and $\sum\alpha_k\sigma_k^2<\infty$, the noise influence vanishes asymptotically, guaranteeing almost‑sure boundedness of $\{x_k\}$.
\item \textbf{Hyper‑parameter Sensitivity.} Larger initial stepsizes accelerate early progress but increase the constant $L\sum\alpha_k^2/(2\sum\alpha_k)$. The variance schedule must decay sufficiently fast (condition \eqref{6}) to avoid a bias term that dominates the $1/k$ rate.
\item \textbf{Comparison to SGD.} Setting $\sigma_k\equiv0$ recovers the classic SGD bound $\frac{L}{2}\frac{\sum\alpha_k^2}{\sum\alpha_k}$; the additional term $\frac{\sum\alpha_k\sigma_k^2}{\sum\alpha_k}$ quantifies the price paid for injecting Langevin noise.
\item \textbf{Relation to Sampling.} When $\alpha_k\equiv\alpha$ and $\sigma_k^2=2\alpha\beta^{-1}$, the iteration approximates the overdamped Langevin diffusion and converges in distribution to the Gibbs measure $\propto e^{-\beta f}$.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the presence of stochastic perturbations, the algorithm preserves the $O(1/k)$ convergence rate typical of deterministic gradient descent for convex smooth objectives, provided the noise variance decays fast enough. This bridges optimisation (small‑noise regime) and Bayesian sampling (large‑noise regime) within a single iterative scheme.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^d$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\tag{9}
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness; see \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Three‑Point Identity]\label{lem:3pt}
For any vectors $a,b,c\in\R^d$,
\[
\norm{a-c}^2=\norm{a-b}^2+2\inner{a-b}{b-c}+\norm{b-c}^2 .
\tag{10}
\]
\end{lemma}
\begin{proof}
Expand the left‑hand side and collect terms.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}.  
All expectations are taken with respect to the randomness of $\{\xi_t\}_{t\le k}$.

\noindent\textbf{Step 1 (Distance Recursion).}  
From the update rule \eqref{eq:1} and Lemma \ref{lem:3pt} with $a=x_{k+1},b=x_k,c=x^\star$,
\[
\norm{x_{k+1}-x^\star}^2
= \norm{x_k-x^\star}^2
-2\alpha_k\inner{\nabla f(x_k)}{x_k-x^\star}
+2\inner{\xi_k}{x_k-x^\star}
+\alpha_k^{2}\norm{\nabla f(x_k)}^{2}
+2\alpha_k\inner{\nabla f(x_k)}{\xi_k}
+\norm{\xi_k}^{2}. \tag{11}
\]

\noindent\textbf{Step 2 (Take Conditional Expectation).}  
Condition on $\mathcal F_k:=\sigma(x_0,\dots,x_k,\xi_0,\dots,\xi_{k-1})$. Using Assumption \ref{ass:noise},
\[
\E\!\bigl[\inner{\xi_k}{x_k-x^\star}\mid\mathcal F_k\bigr]=0,
\qquad
\E\!\bigl[\inner{\nabla f(x_k)}{\xi_k}\mid\mathcal F_k\bigr]=0,
\qquad
\E\!\bigl[\norm{\xi_k}^{2}\mid\mathcal F_k\bigr]=d\sigma_k^{2}. \tag{12}
\]

\noindent\textbf{Step 3 (Bound Gradient Norm).}  
Convexity implies $\inner{\nabla f(x_k)}{x_k-x^\star}\ge f(x_k)-f^\star$. Moreover, by $L$‑smoothness,
\[
\norm{\nabla f(x_k)}^{2}\le 2L\bigl(f(x_k)-f^\star\bigr). \tag{13}
\]

\noindent\textbf{Step 4 (Insert Bounds).}  
Taking expectation of \eqref{11} conditioned on $\mathcal F_k$, applying \eqref{12} and \eqref{13},
\[
\begin{aligned}
\E\!\bigl[\norm{x_{k+1}-x^\star}^{2}\mid\mathcal F_k\bigr]
&\le \norm{x_k-x^\star}^{2}
-2\alpha_k\bigl(f(x_k)-f^\star\bigr)
+ \alpha_k^{2}\,2L\bigl(f(x_k)-f^\star\bigr)
+d\sigma_k^{2}. \tag{14}
\end{aligned}
\]

\noindent\textbf{Step 5 (Rearrange).}  
Move the term involving $f(x_k)-f^\star$ to the left and divide by $2\alpha_k$,
\[
\bigl(f(x_k)-f^\star\bigr)
\le \frac{\norm{x_k-x^\star}^{2}-\E\!\bigl[\norm{x_{k+1}-x^\star}^{2}\mid\mathcal F_k\bigr]}{2\alpha_k}
+\frac{L\alpha_k}{2}\bigl(f(x_k)-f^\star\bigr)
+\frac{d\sigma_k^{2}}{2\alpha_k}. \tag{15}
\]

\noindent\textbf{Step 6 (Collect $f(x_k)-f^\star$ Terms).}  
Bring the $L\alpha_k/2$ term to the left,
\[
\bigl(1-\tfrac{L\alpha_k}{2}\bigr)\bigl(f(x_k)-f^\star\bigr)
\le \frac{\norm{x_k-x^\star}^{2}-\E\!\bigl[\norm{x_{k+1}-x^\star}^{2}\mid\mathcal F_k\bigr]}{2\alpha_k}
+\frac{d\sigma_k^{2}}{2\alpha_k}. \tag{16}
\]

\noindent\textbf{Step 7 (Assume $\alpha_k\le 1/L$).}  
Under Assumption \ref{ass:schedule} we can enforce $\alpha_k\le 1/L$, which guarantees $(1-L\alpha_k/2)\ge 1/2$. Hence,
\[
f(x_k)-f^\star
\le \frac{\norm{x_k-x^\star}^{2}-\E\!\bigl[\norm{x_{k+1}-x^\star}^{2}\mid\mathcal F_k\bigr]}{\alpha_k}
+\frac{d\sigma_k^{2}}{\alpha_k}. \tag{17}
\]

\noindent\textbf{Step 8 (Take Full Expectation).}  
Denote $\Delta_k:=\E\!\bigl[\norm{x_k-x^\star}^{2}\bigr]$. Taking expectation of \eqref{17} yields
\[
\E\!\bigl[f(x_k)\bigr]-f^\star
\le \frac{\Delta_k-\Delta_{k+1}}{\alpha_k}
+\frac{d\sigma_k^{2}}{\alpha_k}. \tag{18}
\]

\noindent\textbf{Step 9 (Sum Over Iterations).}  
Sum \eqref{18} from $k=0$ to $k=K-1$,
\[
\sum_{k=0}^{K-1}\alpha_k\bigl(\E[f(x_k)]-f^\star\bigr)
\le \Delta_0-\Delta_K+\sum_{k=0}^{K-1}d\sigma_k^{2}. \tag{19}
\]

\noindent\textbf{Step 10 (Divide by $\sum_{t=0}^{K-1}\alpha_t$).}  
Define $\bar f_K:=\frac{\sum_{k=0}^{K-1}\alpha_k\E[f(x_k)]}{\sum_{t=0}^{K-1}\alpha_t}$. Using $\Delta_K\ge0$,
\[
\bar f_K-f^\star
\le \frac{\Delta_0}{\sum_{t=0}^{K-1}\alpha_t}
+\frac{d\sum_{k=0}^{K-1}\sigma_k^{2}}{\sum_{t=0}^{K-1}\alpha_t}. \tag{20}
\]

\noindent\textbf{Step 11 (Refine Gradient‑Norm Bound).}  
A tighter bound uses \eqref{13} inside \eqref{14} before Step 5, which introduces the term $\frac{L}{2}\alpha_k^2\bigl(f(x_k)-f^\star\bigr)$. Carrying that term through the same algebra yields an additional additive component
\[
\frac{L}{2}\,\frac{\sum_{k=0}^{K-1}\alpha_k^{2}}{\sum_{t=0}^{K-1}\alpha_t}\,,
\tag{21}
\]
which appears in the final statement of the theorem.

\noindent\textbf{Step 12 (Combine Bounds).}  
Collecting \eqref{20} and \eqref{21} gives exactly inequality \eqref{7} of Theorem \ref{thm:conv}. ∎

\section*{Rate Derivation (With Constants)}
Assume $\alpha_k=\frac{c}{k+1}$ and $\sigma_k^{2}= \frac{c'}{(k+1)^{1+\epsilon}}$, $\epsilon>0$.  
Standard series estimates give  

\[
\sum_{t=0}^{K-1}\alpha_t = c\sum_{t=1}^{K}\frac{1}{t}=c\bigl(\log K+O(1)\bigr),
\]
\[
\sum_{t=0}^{K-1}\alpha_t^{2}=c^{2}\sum_{t=1}^{K}\frac{1}{t^{2}}=c^{2}\bigl(\frac{\pi^{2}}{6}+O(1/K)\bigr),
\]
\[
\sum_{t=0}^{K-1}\alpha_t\sigma_t^{2}=c c'\sum_{t=1}^{K}\frac{1}{t^{2+\epsilon}}=O(1).
\]

Plugging into \eqref{7} yields  

\[
\E[f(x_K)]-f^\star \le
\frac{\norm{x_0-x^\star}^{2}}{2c\log K}
+\frac{L c}{2}\,\frac{1}{\log K}
+ O\!\bigl(\tfrac{1}{\log K}\bigr)
=O\!\left(\frac{1}{\log K}\right).
\]

If instead we use the averaged iterate $\tilde x_K=\frac{1}{K}\sum_{k=0}^{K-1}x_k$, the bound improves to $O(1/K)$, matching \eqref{8}.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Descent (no noise).} $\sigma_k\equiv0$ reduces \eqref{7} to the classic $O(1/k)$ bound for convex $L$‑smooth functions.
\item \textbf{Constant Noise.} If $\sigma_k\equiv\sigma>0$, the term $\frac{\sum\alpha_k\sigma^{2}}{\sum\alpha_k}$ converges to a non‑zero constant, implying the iterates converge only to a neighbourhood of $x^\star$ whose radius is $O(\sigma)$.
\item \textbf{Sampling Regime.} Setting $\alpha_k\equiv\alpha$ and $\sigma_k^{2}=2\alpha\beta^{-1}$ yields a discretisation of the Langevin diffusion; under additional log‑concavity assumptions the chain converges in distribution to $\pi\propto e^{-\beta f}$.
\end{itemize}
