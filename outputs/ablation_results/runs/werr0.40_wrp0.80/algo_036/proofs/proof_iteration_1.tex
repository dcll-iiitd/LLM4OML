\section*{Algorithm Name}
\textbf{Stochastic Newton with Estimated Hessian (SNEH)}  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
At iteration $t\in\{0,1,\dots\}$ we have a current iterate $x_{t}\in\R^{d}$.
A stochastic estimate of the Hessian,
\[
H_{t}\in\R^{d\times d},
\]
is obtained (e.g. from a mini‑batch).  
The update rule reads
\begin{equation}
\label{eq:update}
x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}\,\grad f(x_{t}),
\end{equation}
where $\alpha>0$ is a (deterministic) stepsize.  
For numerical stability we assume a regularised estimator
\[
\widetilde H_{t}=H_{t}+\lambda I,\qquad \lambda>0,
\]
and replace $H_{t}^{-1}$ by $\widetilde H_{t}^{-1}$ in \eqref{eq:update}.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton with Estimated Hessian (SNEH)}\label{alg:sneh}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\alpha>0$, regularisation $\lambda>0$, maximum iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a mini‑batch $\mathcal{B}_{t}$.
    \State Compute stochastic gradient $g_{t}= \grad f_{\mathcal{B}_{t}}(x_{t})$.
    \State Compute stochastic Hessian estimate $H_{t}= \Hess f_{\mathcal{B}_{t}}(x_{t})$.
    \State Form regularised matrix $\widetilde H_{t}=H_{t}+\lambda I$.
    \State Update 
    \[
    x_{t+1}=x_{t}-\alpha\,\widetilde H_{t}^{-1} g_{t}.
    \]
\EndFor
\State \textbf{output} $x_{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness and Strong Convexity]\label{as:sc}
The objective $f$ is $\mu$‑strongly convex and $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad 
\frac{\mu}{2}\norm{x-y}^{2}\le f(y)-f(x)-\ip{\grad f(x)}{y-x}\le\frac{L}{2}\norm{y-x}^{2}.
\]
Consequently, the true Hessian satisfies
\[
\mu I\preceq \Hess f(x)\preceq L I,\qquad\forall x.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:grad}
For every $t$, the stochastic gradient $g_{t}$ satisfies
\[
\E\!\left[g_{t}\mid x_{t}\right]=\grad f(x_{t}),\qquad
\E\!\left[\norm{g_{t}-\grad f(x_{t})}^{2}\mid x_{t}\right]\le\sigma^{2}.
\]
\end{assumption}

\begin{assumption}[Hessian Approximation Quality]\label{as:hess}
There exists a constant $\delta\in[0,1)$ such that, conditionally on $x_{t}$,
\[
(1-\delta)\,\Hess f(x_{t})\;\preceq\;\widetilde H_{t}\;\preceq\;(1+\delta)\,\Hess f(x_{t}),
\]
where $\widetilde H_{t}=H_{t}+\lambda I$ is the regularised estimator.
Equivalently,
\[
\lambda_{\min}(\widetilde H_{t})\ge (1-\delta)\mu,\qquad
\lambda_{\max}(\widetilde H_{t})\le (1+\delta)L .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The stepsize $\alpha$ satisfies
\[
0<\alpha<\frac{2}{(1+\delta)L}.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear Convergence in Expectation]\label{thm:linear}
Under Assumptions \ref{as:sc}–\ref{as:step}, the iterates generated by \eqref{eq:update} satisfy for all $t\ge0$
\begin{equation}
\label{eq:linear}
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\mid x_{t}\right]
\le\left(1-\alpha\mu_{\mathrm{eff}}\right)\norm{x_{t}-x^{*}}^{2}
+\alpha^{2}\,\frac{\sigma^{2}}{(1-\delta)\mu},
\end{equation}
where the effective strong‑convexity constant is
\[
\mu_{\mathrm{eff}}:=\frac{2\mu}{(1+\delta)L}.
\]
Consequently, after $T$ iterations
\[
\E\!\left[\norm{x_{T}-x^{*}}^{2}\right]\le
\left(1-\alpha\mu_{\mathrm{eff}}\right)^{T}\norm{x_{0}-x^{*}}^{2}
+\frac{\alpha\sigma^{2}}{(1-\delta)\mu_{\mathrm{eff}}}.
\]
Thus the method converges linearly to a neighbourhood of the optimum whose radius is $O(\alpha\sigma^{2})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The regularisation $\lambda$ guarantees $\widetilde H_{t}\succ0$ even when $H_{t}$ is rank‑deficient; the bounds in Assumption~\ref{as:hess} ensure that the condition number of $\widetilde H_{t}$ is at most $\kappa_{\mathrm{eff}}:=\frac{(1+\delta)L}{(1-\delta)\mu}$.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item \emph{Stepsize $\alpha$.} Larger $\alpha$ accelerates the contraction factor $1-\alpha\mu_{\mathrm{eff}}$ but inflates the variance term $\alpha^{2}\sigma^{2}/((1-\delta)\mu)$. The optimal $\alpha$ balancing both terms is $\alpha^{*}= \frac{\mu_{\mathrm{eff}}}{2\sigma^{2}}\,\frac{(1-\delta)\mu}{L}$, yielding the fastest linear rate.
        \item \emph{Regularisation $\lambda$.} Increasing $\lambda$ reduces $\delta$ (by moving eigenvalues away from zero) at the price of biasing the direction; the trade‑off is captured in Assumption~\ref{as:hess}.
    \end{itemize}
\item \textbf{Comparison to First‑Order Methods.} When $\delta=0$ (exact Hessian) the effective rate reduces to $(1-\alpha\mu/L)$, i.e. the classical Newton linear rate. For $\delta>0$ the rate degrades gracefully, but still dominates the $O(1/\sqrt{T})$ sub‑linear rate of SGD under the same smoothness/strong‑convexity assumptions.
\end{enumerate}

\section*{Discussion}
The theorem shows that, provided the stochastic Hessian estimate is uniformly close to the true Hessian (quantified by $\delta$), the stochastic Newton iteration inherits the fast linear convergence of deterministic Newton’s method up to a stochastic error floor proportional to the gradient noise variance $\sigma^{2}$. The result is tight in the sense that if $\delta\to1$ the bound collapses to the SGD bound, reflecting the loss of curvature information.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic Upper/Lower Bounds]\label{lem:quad}
For any $x,y\in\R^{d}$ and any positive‑definite matrix $M$,
\[
\frac{1}{\lambda_{\max}(M)}\norm{x-y}^{2}
\;\le\;
\ip{M^{-1}(x-y)}{x-y}
\;\le\;
\frac{1}{\lambda_{\min}(M)}\norm{x-y}^{2}.
\]
\end{lemma}
\begin{proof}
Since $M$ admits an eigendecomposition $M=Q\Lambda Q^{\top}$ with $\Lambda=\operatorname{diag}(\lambda_{i})$, we have $M^{-1}=Q\Lambda^{-1}Q^{\top}$ and
\[
\ip{M^{-1}v}{v}=v^{\top}Q\Lambda^{-1}Q^{\top}v
=\sum_{i}\lambda_{i}^{-1}(q_{i}^{\top}v)^{2}.
\]
Bounding each $\lambda_{i}^{-1}$ by $1/\lambda_{\max}(M)$ and $1/\lambda_{\min}(M)$ yields the claim. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof $x^{*}$ denotes the unique minimiser of $f$.

\bigskip
\noindent\textbf{Step 1 (Expand the squared distance).}  
Using the update \eqref{eq:update} we write
\[
\begin{aligned}
\norm{x_{t+1}-x^{*}}^{2}
&=\norm{x_{t}-\alpha\widetilde H_{t}^{-1}g_{t}-x^{*}}^{2}\\
&=\norm{x_{t}-x^{*}}^{2}
-2\alpha\ip{\widetilde H_{t}^{-1}g_{t}}{x_{t}-x^{*}}
+\alpha^{2}\norm{\widetilde H_{t}^{-1}g_{t}}^{2}.
\end{aligned}
\tag{1}
\]
\label{step:expand}

\bigskip
\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on the $\sigma$‑algebra $\mathcal{F}_{t}=\sigma(x_{0},\dots,x_{t})$.  
Using unbiasedness \eqref{as:grad} we obtain
\[
\E\!\left[g_{t}\mid\mathcal{F}_{t}\right]=\grad f(x_{t}),\qquad
\E\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]
=\norm{\grad f(x_{t})}^{2}+\E\!\left[\norm{g_{t}-\grad f(x_{t})}^{2}\mid\mathcal{F}_{t}\right]
\le\norm{\grad f(x_{t})}^{2}+\sigma^{2}.
\tag{2}
\]
\label{step:expect}

\bigskip
\noindent\textbf{Step 3 (Bound the cross term).}  
From Lemma~\ref{lem:quad} and the lower eigenvalue bound in Assumption~\ref{as:hess},
\[
\ip{\widetilde H_{t}^{-1}\grad f(x_{t})}{x_{t}-x^{*}}
\ge \frac{1}{\lambda_{\max}(\widetilde H_{t})}
\ip{\grad f(x_{t})}{x_{t}-x^{*}}
\ge \frac{1}{(1+\delta)L}\ip{\grad f(x_{t})}{x_{t}-x^{*}}.
\tag{3}
\]
\label{step:cross}

\bigskip
\noindent\textbf{Step 4 (Use strong convexity).}  
Strong convexity gives
\[
\ip{\grad f(x_{t})}{x_{t}-x^{*}}\ge \mu\norm{x_{t}-x^{*}}^{2}.
\tag{4}
\]
\label{step:strong}

\bigskip
\noindent\textbf{Step 5 (Bound the quadratic term).}  
Again by Lemma~\ref{lem:quad} and the lower eigenvalue bound,
\[
\norm{\widetilde H_{t}^{-1}g_{t}}^{2}
= \ip{\widetilde H_{t}^{-1}g_{t}}{\widetilde H_{t}^{-1}g_{t}}
\le \frac{1}{\lambda_{\min}(\widetilde H_{t})^{2}}\norm{g_{t}}^{2}
\le\frac{1}{\big((1-\delta)\mu\big)^{2}}\norm{g_{t}}^{2}.
\tag{5}
\]
\label{step:quad}

\bigskip
\noindent\textbf{Step 6 (Insert bounds into (1)).}  
Taking conditional expectation and applying \eqref{step:cross}–\eqref{step:quad} yields
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\mid\mathcal{F}_{t}\right]
&\le \norm{x_{t}-x^{*}}^{2}
-2\alpha\frac{1}{(1+\delta)L}\ip{\grad f(x_{t})}{x_{t}-x^{*}}\\
&\qquad+\alpha^{2}\frac{1}{\big((1-\delta)\mu\big)^{2}}
\Big(\norm{\grad f(x_{t})}^{2}+\sigma^{2}\Big).
\end{aligned}
\tag{6}
\]

\bigskip
\noindent\textbf{Step 7 (Replace $\norm{\grad f(x_{t})}^{2}$).}  
For $L$‑smooth functions we have the standard inequality
\[
\norm{\grad f(x_{t})}^{2}\le 2L\big(f(x_{t})-f(x^{*})\big)
\le L^{2}\norm{x_{t}-x^{*}}^{2},
\tag{7}
\]
where the last inequality follows from strong convexity (i.e. $f(x_{t})-f(x^{*})\le \frac{L}{2}\norm{x_{t}-x^{*}}^{2}$).  

\bigskip
\noindent\textbf{Step 8 (Combine all pieces).}  
Using \eqref{step:strong} and \eqref{step:quad} in \eqref{step:cross} we obtain
\[
-2\alpha\frac{1}{(1+\delta)L}\ip{\grad f(x_{t})}{x_{t}-x^{*}}
\le -2\alpha\frac{\mu}{(1+\delta)L}\norm{x_{t}-x^{*}}^{2}.
\tag{8}
\]
The quadratic term from \eqref{step:quad} together with \eqref{step:quad} gives
\[
\alpha^{2}\frac{1}{\big((1-\delta)\mu\big)^{2}}\norm{\grad f(x_{t})}^{2}
\le \alpha^{2}\frac{L^{2}}{\big((1-\delta)\mu\big)^{2}}\norm{x_{t}-x^{*}}^{2}.
\tag{9}
\]
Substituting (8) and (9) into (6) we have
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\mid\mathcal{F}_{t}\right]
&\le\Bigg[1-2\alpha\frac{\mu}{(1+\delta)L}
+\alpha^{2}\frac{L^{2}}{\big((1-\delta)\mu\big)^{2}}\Bigg]\norm{x_{t}-x^{*}}^{2}
+\alpha^{2}\frac{\sigma^{2}}{\big((1-\delta)\mu\big)^{2}}.
\end{aligned}
\tag{10}
\]

\bigskip
\noindent\textbf{Step 9 (Choose stepsize to simplify the contraction).}  
Define the effective strong‑convexity constant
\[
\mu_{\mathrm{eff}}:=\frac{2\mu}{(1+\delta)L}.
\]
If $\alpha$ satisfies Assumption~\ref{as:step}, the quadratic term $\alpha^{2}L^{2}/((1-\delta)^{2}\mu^{2})$ is at most $\alpha\mu_{\mathrm{eff}}/2$, i.e.
\[
\alpha^{2}\frac{L^{2}}{((1-\delta)\mu)^{2}}
\le \alpha\frac{\mu}{(1+\delta)L}
= \frac{\alpha\mu_{\mathrm{eff}}}{2}.
\tag{11}
\]
Plugging (11) into (10) yields
\[
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\mid\mathcal{F}_{t}\right]
\le\big(1-\alpha\mu_{\mathrm{eff}}\big)\norm{x_{t}-x^{*}}^{2}
+\alpha^{2}\frac{\sigma^{2}}{(1-\delta)^{2}\mu^{2}}.
\tag{12}
\]
Since $(1-\delta)^{2}\mu^{2}\ge (1-\delta)\mu\cdot\mu$, we can relax the variance term to the form stated in the theorem:
\[
\alpha^{2}\frac{\sigma^{2}}{(1-\delta)^{2}\mu^{2}}
\le \alpha^{2}\frac{\sigma^{2}}{(1-\delta)\mu}.
\tag{13}
\]
Combining (12) and (13) gives exactly \eqref{eq:linear}, completing the one‑step bound. \hfill\qedsymbol

\bigskip
\noindent\textbf{Step 10 (Unrolling the recursion).}  
Iterating \eqref{eq:linear} and taking total expectation yields
\[
\E\!\left[\norm{x_{T}-x^{*}}^{2}\right]
\le\big(1-\alpha\mu_{\mathrm{eff}}\big)^{T}\norm{x_{0}-x^{*}}^{2}
+\frac{\alpha\sigma^{2}}{(1-\delta)\mu_{\mathrm{eff}}},
\]
which is the claimed linear convergence rate. \hfill\qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} the contraction factor is $q:=1-\alpha\mu_{\mathrm{eff}}$.
Choosing the largest admissible stepsize $\alpha=\frac{1}{(1+\delta)L}$ (the midpoint of the interval in Assumption~\ref{as:step}) gives
\[
\mu_{\mathrm{eff}}=\frac{2\mu}{(1+\delta)L},\qquad
q=1-\frac{2\mu}{(1+\delta)^{2}L^{2}}.
\]
Hence after $T$ iterations
\[
\E\!\left[\norm{x_{T}-x^{*}}^{2}\right]
\le \exp\!\Big(-\frac{2\mu}{(1+\delta)^{2}L^{2}}\,T\Big)\norm{x_{0}-x^{*}}^{2}
+\frac{\sigma^{2}}{2\mu(1-\delta)}.
\]
The first term decays exponentially, while the second term is the steady‑state error floor.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Hessian ($\delta=0$).}  
The bound reduces to the classic deterministic Newton rate:
\[
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\right]\le\big(1-\alpha\mu/L\big)\norm{x_{t}-x^{*}}^{2}
+\alpha^{2}\frac{\sigma^{2}}{\mu}.
\]
\item \textbf{No curvature information ($\delta\to1$).}  
The effective strong‑convexity constant $\mu_{\mathrm{eff}}\to0$ and the recursion collapses to the SGD bound
\[
\E\!\left[\norm{x_{t+1}-x^{*}}^{2}\right]\le
\norm{x_{t}-x^{*}}^{2}+ \alpha^{2}\frac{\sigma^{2}}{\mu},
\]
i.e. no linear contraction.
\item \textbf{Quadratic Objective.}  
If $f$ is exactly quadratic, the stochastic Hessian equals the true Hessian and $\sigma=0$; the algorithm converges in a single step for $\alpha=1$.
\end{itemize}
