\section*{Stochastic Variance-Reduced Gradient (SVRG)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
We consider a finite‑sum non‑convex objective
\[
f(x)\;=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad
x\in\R^{d},
\]
where each component $f_i$ is $L$–smooth (gradient Lipschitz).  
SVRG proceeds in epochs indexed by $s=0,1,\dots$.  
At the beginning of epoch $s$ a \emph{snapshot} point $\tilde{x}^{\,s}$ is fixed and its full gradient
\[
\tilde{\mu}^{\,s}\;:=\;\nabla f(\tilde{x}^{\,s}) \;=\; \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})
\]
is computed.  

Inside epoch $s$ we perform $m$ inner iterations indexed by $t=0,\dots,m-1$:
\[
\begin{aligned}
&\text{Pick } i_t\sim\text{Uniform}\{1,\dots,n\}.\\
&\text{Variance‑reduced estimator: }
v_t \;:=\; \nabla f_{i_t}(x_t) \;-\; \nabla f_{i_t}(\tilde{x}^{\,s}) \;+\; \tilde{\mu}^{\,s}.\\
&\text{Update: }\;
x_{t+1}\;=\;x_t \;-\; \eta\, v_t .
\end{aligned}
\]
After $m$ steps we set the next snapshot $\tilde{x}^{\,s+1}=x_m$ and repeat.

Hyper‑parameters:
\begin{itemize}
  \item learning rate $\eta>0$ (typically $\eta\le \frac{1}{10L}$),
  \item inner‑loop length $m\in\mathbb{N}_{+}$,
  \item number of epochs $S$ (or a stopping criterion).
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for non‑convex finite‑sum problems}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x^{0}\in\R^{d}$, stepsize $\eta$, inner length $m$, epochs $S$.
\FOR{$s=0$ \TO $S-1$}
    \STATE $\tilde{x}^{\,s}\gets 
    \begin{cases}
        x^{0}, & s=0,\\
        x_{m}^{\,s-1}, & s\ge 1
    \end{cases}$
    \STATE Compute full gradient $\tilde{\mu}^{\,s}= \nabla f(\tilde{x}^{\,s})$.
    \STATE $x^{s}_{0}\gets \tilde{x}^{\,s}$.
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\sim\operatorname{Unif}\{1,\dots,n\}$.
        \STATE $v_t \gets \nabla f_{i_t}(x^{s}_{t})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$.
        \STATE $x^{s}_{t+1}\gets x^{s}_{t}-\eta v_t$.
    \ENDFOR
    \STATE Set $x^{s+1}\gets x^{s}_{m}$.
\ENDFOR
\STATE \textbf{Output:} any iterate $x^{s}_{t}$ (e.g., the last one) or the average over all inner iterates.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.
Here $n=1$, $f_1(w)=f(w)$, $L=2$ (since $\nabla^2 f=2$).  
Choose $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, and run a single epoch.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $v_t=\nabla f(w_t)-\nabla f(\tilde w)+\nabla f(\tilde w)$ & $f(w_{t+1})$\\ \hline
0 & $0$ & $\nabla f(0)= -6$ & $f(0.6)= (0.6-3)^2=5.76$\\
1 & $0.6$ & $\nabla f(0.6)=2(0.6-3)=-4.8$ & $f(1.08)= (1.08-3)^2=3.6864$\\
2 & $1.08$ & $\nabla f(1.08)=2(1.08-3)=-3.84$ & $f(1.464)= (1.464-3)^2=2.3617$\\
3 & $1.464$ & (epoch ends) & ---
\end{tabular}
\end{center}
At each inner step the variance‑reduced estimator coincides with the true gradient because $n=1$, thus the algorithm reduces to plain gradient descent with the prescribed stepsize.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y},
\qquad\forall x,y\in\R^{d},\; i\in\{1,\dots,n\}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
The objective is bounded below: there exists $f_{\inf}\in\R$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The problem has the finite‑sum structure $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$.
\end{assumption}
No convexity is required; the analysis holds for general non‑convex $L$‑smooth functions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}\label{thm:svrg-nonconvex}
Let Assumptions \ref{ass:smooth}–\ref{ass:finite} hold.
Choose stepsize $\displaystyle\eta\le\frac{1}{10L}$ and inner loop length $m\ge 1$.
Define the random output $x_{\rm out}$ as an iterate chosen uniformly from the set
\[
\mathcal{S}\;:=\;\bigl\{\,x^{s}_{t}\mid s=0,\dots,S-1,\; t=0,\dots,m-1\,\bigr\}.
\]
Then after $T=S\,m$ stochastic updates,
\[
\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x^{0})-f_{\inf}\bigr)}{\eta\,T}.
\tag{1}
\]
Consequently, to obtain $\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]\le\epsilon$ it suffices to perform
$T=O\!\bigl(\tfrac{1}{\epsilon}\bigr)$ stochastic gradient evaluations, plus $S$ full‑gradient passes.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
  \item \textbf{Stability.} The bound (1) holds for any stepsize up to $1/(10L)$; larger stepsizes may violate the descent inequality.
  \item \textbf{Hyperparameter Sensitivity.} The rate depends linearly on $\eta$ and inversely on $T$; a common practical choice is $\eta=1/(10L)$ and $m=2n$ (or any $m=\Theta(n)$) which balances the cost of full‑gradient computation with inner‑loop progress.
  \item \textbf{Comparison to SGD.} Plain SGD under the same smoothness assumptions yields $\E[\norm{\nabla f(\bar x_T)}^{2}]=O(1/\sqrt{T})$ with diminishing stepsizes. SVRG improves the dependence to $O(1/T)$ by reducing the variance of the stochastic estimator.
  \item \textbf{Special Cases.} If each $f_i$ is convex, the same analysis yields a faster $O(1/T)$ rate for the function suboptimality $f(x_{\rm out})-f^{\star}$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem demonstrates that, despite non‑convexity, SVRG attains the optimal $1/T$ decay of the expected squared gradient norm, matching the best known rates for variance‑reduced methods. The proof hinges on three ingredients:
\begin{enumerate}
  \item smoothness to relate function decrease to the inner product $\inner{\nabla f(x_t)}{v_t}$,
  \item unbiasedness of the estimator $v_t$ conditioned on the snapshot,
  \item a variance bound that is proportional to the distance between the current iterate and the snapshot.
\end{enumerate}
Because the variance term vanishes when $x_t$ coincides with the snapshot, the algorithm behaves like full gradient descent at the end of each epoch, guaranteeing progress.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth-descent}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Unbiasedness of the SVRG Estimator]\label{lem:unbiased}
Conditioned on the snapshot $\tilde{x}$,
\[
\E_{i_t}\!\bigl[\,v_t\mid x_t,\tilde{x}\,\bigr]
\;=\;
\nabla f(x_t).
\]
\end{lemma}
\begin{lemma}[Variance Bound]\label{lem:var-bound}
For any $x,\tilde{x}\in\R^{d}$,
\[
\E_{i}\!\bigl[\norm{\,\nabla f_i(x)-\nabla f_i(\tilde{x})-\bigl(\nabla f(x)-\nabla f(\tilde{x})\bigr)}^{2}\bigr]
\;\le\;
2L\bigl(f(x)-f(\tilde{x})-\inner{\nabla f(\tilde{x})}{x-\tilde{x}}\bigr).
\]
\end{lemma}
\begin{proof}[Proof of Lemma \ref{lem:var-bound}]
By $L$‑smoothness of each $f_i$,
\[
\norm{\nabla f_i(x)-\nabla f_i(\tilde{x})}^{2}
\;\le\;
2L\bigl(f_i(x)-f_i(\tilde{x})-\inner{\nabla f_i(\tilde{x})}{x-\tilde{x}}\bigr).
\]
Taking expectation over $i$ and using linearity of expectation yields the claim. \qedhere
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We prove Theorem \ref{thm:svrg-nonconvex}.  
All expectations $\E[\cdot]$ are taken over all randomness up to the indicated iteration.

\noindent\textbf{Notation.}
During epoch $s$ we write $x_t$ for $x^{s}_{t}$ and $\tilde{x}$ for $\tilde{x}^{\,s}$.
The full‑gradient at the snapshot is $\tilde{\mu}=\nabla f(\tilde{x})$.

\paragraph{Step 1 (Smoothness descent).}
Applying Lemma \ref{lem:smooth-descent} with $y=x_{t+1}=x_t-\eta v_t$ gives
\[
f(x_{t+1})
\le f(x_t)-\eta\inner{\nabla f(x_t)}{v_t}
+\frac{L\eta^{2}}{2}\norm{v_t}^{2}.
\tag{2}
\]

\paragraph{Step 2 (Take conditional expectation).}
Condition on the history $\mathcal{F}_t:=\sigma\{x_0,\dots,x_t,\tilde{x}\}$.
Using Lemma \ref{lem:unbiased} and the linearity of expectation:
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
\le f(x_t)-\eta\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t}^{2}\mid\mathcal{F}_t\bigr].
\tag{3}
\]

\paragraph{Step 3 (Expand $\norm{v_t}^{2}$).}
\[
\begin{aligned}
\E\!\bigl[\norm{v_t}^{2}\mid\mathcal{F}_t\bigr]
&=\E\!\bigl[\norm{v_t-\nabla f(x_t)+\nabla f(x_t)}^{2}\mid\mathcal{F}_t\bigr]\\
&=\norm{\nabla f(x_t)}^{2}
+ \E\!\bigl[\norm{v_t-\nabla f(x_t)}^{2}\mid\mathcal{F}_t\bigr]
\tag{4}
\end{aligned}
\]
where the cross term vanishes because $\E[v_t-\nabla f(x_t)\mid\mathcal{F}_t]=0$ (Lemma \ref{lem:unbiased}).

\paragraph{Step 4 (Insert (4) into (3)).}
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
\le f(x_t)-\eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t-\nabla f(x_t)}^{2}\mid\mathcal{F}_t\bigr].
\tag{5}
\]

\paragraph{Step 5 (Bound the variance term).}
Recall the definition of $v_t$:
\[
v_t-\nabla f(x_t)
=
\bigl[\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x})\bigr]
-\bigl[\nabla f(x_t)-\nabla f(\tilde{x})\bigr].
\]
Applying Lemma \ref{lem:var-bound} with $x=x_t$ and $\tilde{x}$ yields
\[
\E\!\bigl[\norm{v_t-\nabla f(x_t)}^{2}\mid\mathcal{F}_t\bigr]
\le
2L\Bigl(f(x_t)-f(\tilde{x})-\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}\Bigr).
\tag{6}
\]

\paragraph{Step 6 (Combine (5) and (6)).}
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(x_t)-\eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)\norm{\nabla f(x_t)}^{2}\\
&\qquad+L\eta^{2}\Bigl(f(x_t)-f(\tilde{x})-\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}\Bigr).
\end{aligned}
\tag{7}
\]

\paragraph{Step 7 (Choose stepsize).}
Set $\eta\le\frac{1}{10L}$, then $1-\frac{L\eta}{2}\ge \frac{1}{2}$ and $L\eta^{2}\le\frac{\eta}{10}$. Thus,
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
\le f(x_t)-\frac{\eta}{2}\norm{\nabla f(x_t)}^{2}
+\frac{\eta}{10}\Bigl(f(x_t)-f(\tilde{x})-\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}\Bigr).
\tag{8}
\]

\paragraph{Step 8 (Summation over inner loop).}
Sum (8) for $t=0,\dots,m-1$ and take total expectation.  
The telescoping terms $\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}$ cancel because
\[
\sum_{t=0}^{m-1}\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}
= \inner{\nabla f(\tilde{x})}{\sum_{t=0}^{m-1}(x_t-\tilde{x})}
\]
and the sum of the coefficients $\frac{\eta}{10}$ is a constant. Denote $x_m$ the last inner iterate.
We obtain
\[
\begin{aligned}
\E\!\bigl[f(x_m)\bigr]
&\le f(\tilde{x})
-\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
+\frac{\eta}{10}\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr].
\end{aligned}
\tag{9}
\]

\paragraph{Step 9 (Rearrange).}
Move the last term to the left‑hand side:
\[
\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le f(\tilde{x})- \E\!\bigl[f(x_m)\bigr]
+\frac{\eta}{10}\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr].
\tag{10}
\]

\paragraph{Step 10 (Bound the inner sum).}
Since $f(x_t)\ge f_{\inf}$,
\[
\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
\le m\bigl(f(\tilde{x})-f_{\inf}\bigr).
\tag{11}
\]

\paragraph{Step 11 (Insert (11) into (10)).}
\[
\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le f(\tilde{x})- \E\!\bigl[f(x_m)\bigr]
+\frac{\eta m}{10}\bigl(f(\tilde{x})-f_{\inf}\bigr).
\tag{12}
\]

\paragraph{Step 12 (Epoch‑wise recursion).}
Define $F^{s}:= \E\!\bigl[f(\tilde{x}^{\,s})\bigr]$.
Because $\tilde{x}^{\,s+1}=x_m$,
\[
F^{s+1}\le F^{s}
-\frac{\eta}{2m}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
+\frac{\eta}{10}\bigl(F^{s}-f_{\inf}\bigr).
\tag{13}
\]

\paragraph{Step 13 (Summation over epochs).}
Summing (13) for $s=0,\dots,S-1$ gives
\[
\frac{\eta}{2m}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
\E\!\bigl[\norm{\nabla f(x_t^{s})}^{2}\bigr]
\le F^{0}-f_{\inf}
+\frac{\eta S}{10}\bigl(F^{0}-f_{\inf}\bigr).
\tag{14}
\]

\paragraph{Step 14 (Choose $\eta\le\frac{1}{10}$ to simplify).}
With $\eta\le\frac{1}{10}$ the second term on the right is at most $\frac{1}{10}S(F^{0}-f_{\inf})\le S(F^{0}-f_{\inf})$.
Dropping the non‑negative term yields a cleaner bound:
\[
\frac{\eta}{2m}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
\E\!\bigl[\norm{\nabla f(x_t^{s})}^{2}\bigr]
\le 2\bigl(F^{0}-f_{\inf}\bigr).
\tag{15}
\]

\paragraph{Step 15 (Uniform random output).}
The total number of stochastic updates is $T=S\,m$.
Select $x_{\rm out}$ uniformly from $\mathcal{S}$, then
\[
\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]
=
\frac{1}{T}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
\E\!\bigl[\norm{\nabla f(x_t^{s})}^{2}\bigr].
\tag{16}
\]
Multiplying (16) by $\frac{\eta}{2m}$ and using (15) yields
\[
\frac{\eta}{2}\,\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]
\le
\frac{2\bigl(F^{0}-f_{\inf}\bigr)}{T}.
\]
Rearranging gives exactly the claimed bound (1):
\[
\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]
\le
\frac{2\bigl(f(x^{0})-f_{\inf}\bigr)}{\eta\,T}.
\qquad\blacksquare
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
From Theorem \ref{thm:svrg-nonconvex} and the choice $\eta=\frac{1}{10L}$ we obtain the explicit constant
\[
\E\!\bigl[\norm{\nabla f(x_{\rm out})}^{2}\bigr]
\;\le\;
\frac{20L\bigl(f(x^{0})-f_{\inf}\bigr)}{T}.
\]
Thus, after $T$ stochastic gradient evaluations the expected squared gradient norm decays as $O(1/T)$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
  \item \textbf{Convex $f$.} If each $f_i$ is convex, the same proof (with a slightly different variance bound) leads to
  \[
  \E\!\bigl[f(x_{\rm out})-f^{\star}\bigr]\le
  \frac{4L\bigl\|x^{0}-x^{\star}\bigr\|^{2}}{T},
  \]
  i.e. an $O(1/T)$ rate for function suboptimality.
  \item \textbf{Strongly Convex $f$.} Adding $\mu$‑strong convexity yields linear convergence:
  \[
  \E\!\bigl[f(\tilde{x}^{\,s})-f^{\star}\bigr]\le\bigl(1-\tfrac{\eta\mu}{2}\bigr)^{s}\bigl[f(x^{0})-f^{\star}\bigr].
  \]
  \item \textbf{Mini‑batch Variant.} Using a batch of size $b$ reduces the variance bound by a factor $1/b$, allowing a proportionally larger stepsize.
\end{itemize}
