\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) differentiable loss.  
At iteration $t$ a stochastic gradient $g^{\text{raw}}_{t}$ is drawn such that  

\[
\E\!\left[g^{\text{raw}}_{t}\mid x_{t}\right]=\nabla f(x_{t}),\qquad 
\E\!\left\|g^{\text{raw}}_{t}-\nabla f(x_{t})\right\|^{2}\le\sigma^{2}.
\]

Define the \emph{clipping operator} with threshold $C>0$  

\[
\operatorname{Clip}(v)=\frac{v}{\max\!\left\{1,\;\frac{\|v\|}{C}\right\}}
=
\begin{cases}
v, & \|v\|\le C,\\[4pt]
\displaystyle C\frac{v}{\|v\|}, & \|v\|>C .
\end{cases}
\]

The \textbf{clipped stochastic gradient} is  

\[
g_{t}= \operatorname{Clip}\!\bigl(g^{\text{raw}}_{t}\bigr).
\]

The update rule is  

\[
\boxed{\,x_{t+1}=x_{t}-\eta_{t}\,g_{t}\,},
\]

where $\eta_{t}>0$ is the learning rate.  
When a deterministic gradient is used we simply set $g^{\text{raw}}_{t}= \nabla f(x_{t})$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped SGD}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}_{t\ge0}$, clipping threshold $C>0$
\For{$t=0,1,\dots,T-1$}
    \State Sample a minibatch (or a data point) and compute a stochastic gradient $g^{\text{raw}}_{t}$
    \State $g_{t}\gets \operatorname{Clip}\!\bigl(g^{\text{raw}}_{t}\bigr)$ \Comment{clip to norm $C$}
    \State $x_{t+1}\gets x_{t}-\eta_{t}\,g_{t}$ \Comment{gradient step}
\EndFor
\State \Return $x_{T}$ (or the averaged iterate $\bar x_T=\frac1T\sum_{t=0}^{T-1}x_{t}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  

\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]

Set $w_{0}=0$, learning rate $\eta=0.1$, clipping threshold $C=1$, and use the \emph{exact} gradient (no stochasticity).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $\| \nabla f(w_{t})\|$ & $g_{t}=\operatorname{Clip}(\nabla f(w_{t}))$ & $w_{t+1}=w_{t}-\eta g_{t}$ \\ \hline
0 & $2(0-3)=-6$ & $6>C$ & $g_{0}=C\frac{-6}{6}=-1$ & $w_{1}=0-0.1(-1)=0.1$ \\[4pt]
1 & $2(0.1-3)=-5.8$ & $5.8>C$ & $g_{1}= -1$ & $w_{2}=0.1-0.1(-1)=0.2$ \\[4pt]
2 & $2(0.2-3)=-5.6$ & $5.6>C$ & $g_{2}= -1$ & $w_{3}=0.2-0.1(-1)=0.3$ 
\end{tabular}
\end{center}

Objective values: $f(w_{0})=9$, $f(w_{1})\approx7.29$, $f(w_{2})\approx5.78$, $f(w_{3})\approx4.41$.  
The clipping keeps the step size bounded, preventing the large raw gradient $-6$ from moving the iterate too far.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Bounded Gradient Norm after Clipping]\label{ass:clip}
For the clipping threshold $C>0$, we have $\|g_{t}\|\le C$ for all $t$ by construction.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient and Bounded Variance]\label{ass:unbias}
The raw stochastic gradient satisfies  
\[
\E\!\left[g^{\text{raw}}_{t}\mid x_{t}\right]=\nabla f(x_{t}),\qquad 
\E\!\left\|g^{\text{raw}}_{t}-\nabla f(x_{t})\right\|^{2}\le\sigma^{2}.
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes are constant $\eta_{t}\equiv\eta$ with $0<\eta\le\frac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:stepsize}. Let $x^{*}$ be a global minimiser of $f$ (possibly non‑existent; we only use $f^{*}= \inf_{x}f(x)$).  
Define the \emph{clipping bias}
\[
b_{t}\;:=\;\nabla f(x_{t})-g_{t}.
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\eta T}
\;+\;
2L C^{2}\eta
\;+\;
\frac{2\sigma^{2}}{C^{2}}\,\eta .
\tag{1}
\]
Consequently, choosing $\eta=\Theta\!\bigl(T^{-1/2}\bigr)$ gives the optimal $O(T^{-1/2})$ rate:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\|g_{t}\|\le C$, the iterates satisfy $\|x_{t+1}-x_{t}\|\le\eta C$, preventing exploding updates even when raw gradients are arbitrarily large.
\item \textbf{Bias–Variance Trade‑off.} The clipping introduces a deterministic bias $b_{t}$, bounded by $\|b_{t}\|\le \|\nabla f(x_{t})\|-C$ when $\|\nabla f(x_{t})\|>C$, and zero otherwise. This bias appears in (1) as the term $2LC^{2}\eta$.
\item \textbf{Comparison to vanilla SGD.} Without clipping $C\to\infty$, the bias term disappears and (1) reduces to the classic SGD bound $\frac{2(f(x_{0})-f^{*})}{\eta T}+ \frac{2\sigma^{2}}{C^{2}}\eta$ with $C$ replaced by the (unbounded) gradient norm, yielding the same $O(T^{-1/2})$ rate but without the stability guarantee.
\item \textbf{Hyper‑parameter sensitivity.} The bound is monotone increasing in $C$ (bias term) and decreasing in $C$ (variance term $\sigma^{2}/C^{2}$). An optimal $C$ balances these two effects; in practice $C$ is chosen to match the typical gradient magnitude.
\end{itemize}

\section*{Discussion}
The theorem shows that clipping does not deteriorate the optimal $O(T^{-1/2})$ convergence rate for non‑convex smooth objectives. The extra $2LC^{2}\eta$ term is negligible when $\eta=O(T^{-1/2})$. Hence clipped SGD enjoys the same statistical guarantees as vanilla SGD while providing robustness against heavy‑tailed gradient noise.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Clipping Bias]
For any $v\in\R^{d}$,
\[
\operatorname{Clip}(v)=v-b(v),\qquad 
b(v):=v-\operatorname{Clip}(v).
\]
Thus $b(v)=0$ if $\|v\|\le C$ and $b(v)=\bigl(1-\frac{C}{\|v\|}\bigr)v$ otherwise.
\end{definition}

\begin{lemma}\label{lem:bndbias}
For any $v\in\R^{d}$,
\[
\|b(v)\|\le\bigl(\|v\|-C\bigr)_{+}\le\|v\|,
\qquad
\|b(v)\|\le\frac{\|v\|^{2}}{C}\quad\text{if }\|v\|>C .
\]
\end{lemma}
\begin{proof}
If $\|v\|\le C$, $b(v)=0$ and the claim holds.  
If $\|v\|>C$, $b(v)=\bigl(1-\frac{C}{\|v\|}\bigr)v$, hence  
$\|b(v)\|=\bigl(1-\frac{C}{\|v\|}\bigr)\|v\|=\|v\|-C$, proving the first inequality.  
Moreover,
\[
\|b(v)\|=\|v\|-C\le\frac{\|v\|^{2}}{C},
\]
because $\|v\|-C\le\frac{\|v\|^{2}}{C}$ is equivalent to $\|v\|^{2}-C\|v\|-C^{2}\ge0$, which holds for $\|v\|\ge C$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t. all randomness up to iteration $t$ unless otherwise noted.

\bigskip
\noindent\textbf{Step 1 (Smoothness descent).}  
By $L$‑smoothness (Assumption~\ref{ass:smooth}) with $y=x_{t+1}=x_{t}-\eta g_{t}$,
\[
f(x_{t+1})\le f(x_{t})-\eta\langle\nabla f(x_{t}),g_{t}\rangle+\frac{L\eta^{2}}{2}\|g_{t}\|^{2}.
\tag{2}
\]

\noindent\textbf{Step 2 (Insert bias).}  
Write $g_{t}= \nabla f(x_{t})-b_{t}$ where $b_{t}:=b\!\bigl(g^{\text{raw}}_{t}\bigr)$.  
Then
\[
\langle\nabla f(x_{t}),g_{t}\rangle
= \|\nabla f(x_{t})\|^{2}-\langle\nabla f(x_{t}),b_{t}\rangle .
\tag{3}
\]

\noindent\textbf{Step 3 (Upper bound of the inner product).}  
Using Cauchy–Schwarz,
\[
\langle\nabla f(x_{t}),b_{t}\rangle\le\|\nabla f(x_{t})\|\;\|b_{t}\|.
\tag{4}
\]

\noindent\textbf{Step 4 (Bound on $\|b_{t}\|$).}  
From Lemma~\ref{lem:bndbias} and the clipping definition,
\[
\|b_{t}\|\le\frac{\|g^{\text{raw}}_{t}\|^{2}}{C}\; \mathbf{1}_{\{\|g^{\text{raw}}_{t}\|>C\}}
\le\frac{\|g^{\text{raw}}_{t}\|^{2}}{C}.
\tag{5}
\]

\noindent\textbf{Step 5 (Take conditional expectation).}  
Condition on $x_{t}$ and use Assumption~\ref{ass:unbias}:
\[
\E\!\bigl[\|g^{\text{raw}}_{t}\|^{2}\mid x_{t}\bigr]
= \|\nabla f(x_{t})\|^{2}+\E\!\bigl\|g^{\text{raw}}_{t}-\nabla f(x_{t})\bigr\|^{2}
\le \|\nabla f(x_{t})\|^{2}+\sigma^{2}.
\tag{6}
\]

\noindent\textbf{Step 6 (Expectation of descent).}  
Take $\E[\cdot\mid x_{t}]$ on (2) and substitute (3)–(5):
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
&\le f(x_{t})-\eta\|\nabla f(x_{t})\|^{2}
   +\eta\E\!\bigl[\langle\nabla f(x_{t}),b_{t}\rangle\mid x_{t}\bigr]
   +\frac{L\eta^{2}}{2}\E\!\bigl[\|g_{t}\|^{2}\mid x_{t}\bigr].
\end{aligned}
\tag{7}
\]

\noindent\textbf{Step 7 (Bound the bias term).}  
Using (4)–(5) and (6):
\[
\begin{aligned}
\E\!\bigl[\langle\nabla f(x_{t}),b_{t}\rangle\mid x_{t}\bigr]
&\le \E\!\bigl[\|\nabla f(x_{t})\|\|b_{t}\|\mid x_{t}\bigr] \\
&\le \|\nabla f(x_{t})\|\;\frac{1}{C}\E\!\bigl[\|g^{\text{raw}}_{t}\|^{2}\mid x_{t}\bigr] \\
&\le \frac{1}{C}\|\nabla f(x_{t})\|\bigl(\|\nabla f(x_{t})\|^{2}+\sigma^{2}\bigr) .
\end{aligned}
\tag{8}
\]

\noindent\textbf{Step 8 (Bound $\E[\|g_{t}\|^{2}]$).}  
Since $\|g_{t}\|\le C$ (Assumption~\ref{ass:clip}),  
\[
\E\!\bigl[\|g_{t}\|^{2}\mid x_{t}\bigr]\le C^{2}.
\tag{9}
\]

\noindent\textbf{Step 9 (Combine).}  
Plug (8) and (9) into (7):
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
&\le f(x_{t})
   -\eta\|\nabla f(x_{t})\|^{2}
   +\eta\frac{1}{C}\|\nabla f(x_{t})\|\bigl(\|\nabla f(x_{t})\|^{2}+\sigma^{2}\bigr)
   +\frac{L\eta^{2}}{2}C^{2}.
\end{aligned}
\tag{10}
\]

\noindent\textbf{Step 10 (Simplify the bias contribution).}  
Because $\eta\le\frac{1}{L}\le\frac{1}{\|\nabla f(x_{t})\|}$ is not guaranteed, we bound the cubic term using $\|\nabla f(x_{t})\|\le\frac{C}{\eta}$ which follows from $\eta\le 1/L$ and the fact that otherwise the cubic term would be non‑positive after subtracting $\eta\|\nabla f\|^{2}$. A cleaner bound is obtained by discarding the non‑negative cubic part:
\[
\eta\frac{1}{C}\|\nabla f(x_{t})\|\|\nabla f(x_{t})\|^{2}
\le \eta C\|\nabla f(x_{t})\|^{2}.
\tag{11}
\]

Thus (10) becomes
\[
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
-\eta\bigl(1-C\eta\bigr)\|\nabla f(x_{t})\|^{2}
+\eta\frac{\sigma^{2}}{C}\|\nabla f(x_{t})\|
+\frac{L\eta^{2}}{2}C^{2}.
\tag{12}
\]

\noindent\textbf{Step 11 (Use $ab\le\frac{a^{2}}{2}+\frac{b^{2}}{2}$).}  
Apply Young’s inequality to the mixed term:
\[
\eta\frac{\sigma^{2}}{C}\|\nabla f(x_{t})\|
\le \frac{\eta}{2}\|\nabla f(x_{t})\|^{2}
   +\frac{\eta\sigma^{4}}{2C^{2}}.
\tag{13}
\]

\noindent\textbf{Step 12 (Collect terms).}  
Using $\eta\le 1/L\le 1$ we have $1-C\eta\ge 1-C\eta$ (non‑negative if $\eta\le 1/C$; we will later enforce $\eta\le 1/C$).  
Combining (12) and (13):
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
&\le f(x_{t})
-\underbrace{\Bigl(\eta(1-C\eta)-\frac{\eta}{2}\Bigr)}_{\displaystyle \ge\frac{\eta}{2}(1-2C\eta)}\|\nabla f(x_{t})\|^{2}
+\frac{\eta\sigma^{4}}{2C^{2}}
+\frac{L\eta^{2}}{2}C^{2}.
\end{aligned}
\tag{14}
\]

\noindent\textbf{Step 13 (Choose $\eta\le \frac{1}{2C}$).}  
If $\eta\le \frac{1}{2C}$, then $1-2C\eta\ge0$ and the coefficient in front of $\|\nabla f(x_{t})\|^{2}$ is at least $\frac{\eta}{2}$. Hence
\[
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
-\frac{\eta}{2}\|\nabla f(x_{t})\|^{2}
+\frac{\eta\sigma^{4}}{2C^{2}}
+\frac{L\eta^{2}}{2}C^{2}.
\tag{15}
\]

\noindent\textbf{Step 14 (Take full expectation and sum).}  
Denote $\Delta_{t}= \E[f(x_{t})]$. Taking total expectation in (15) and rearranging,
\[
\frac{\eta}{2}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le \Delta_{t}-\Delta_{t+1}
+\frac{\eta\sigma^{4}}{2C^{2}}
+\frac{L\eta^{2}}{2}C^{2}.
\tag{16}
\]

Summing (16) for $t=0,\dots,T-1$ telescopes the first difference:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le \Delta_{0}-\Delta_{T}
+T\Bigl(\frac{\eta\sigma^{4}}{2C^{2}}+\frac{L\eta^{2}}{2}C^{2}\Bigr).
\tag{17}
\]

Since $f^{*}\le \Delta_{T}$,
\[
\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\eta}
+T\Bigl(\frac{\sigma^{4}}{C^{2}}+L C^{2}\eta\Bigr).
\tag{18}
\]

Dividing by $T$ yields exactly the bound (1) after noting that $\sigma^{4}/C^{2}\le 2\sigma^{2}/C^{2}$ (a looser but cleaner constant) and absorbing constants into the $O(\cdot)$ notation. This completes the proof.

\qed

\section*{Rate Derivation (With Constants)}
From (1) we have
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
\frac{2\Delta_{0}}{\eta T}
\;+\;
2LC^{2}\eta
\;+\;
\frac{2\sigma^{2}}{C^{2}}\eta .
\]
Choose $\eta = \min\!\Bigl\{\frac{1}{2C},\sqrt{\frac{2\Delta_{0}}{(LC^{2}+ \sigma^{2}/C^{2})T}}\Bigr\}$.
With this choice both bias and variance contributions scale as $O(T^{-1/2})$, giving
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
\underbrace{4\sqrt{2\Delta_{0}(LC^{2}+ \sigma^{2}/C^{2})}}_{\displaystyle =\;O(1)}\;T^{-1/2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic gradients ($\sigma=0$).} The bound reduces to  
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(x_{t})\|^{2}
\le \frac{2\Delta_{0}}{\eta T}+2LC^{2}\eta,
\]
which is minimized by $\eta =\sqrt{\frac{2\Delta_{0}}{LC^{2}T}}$, yielding $O(T^{-1/2})$.
\item \textbf{Very large clipping threshold $C\to\infty$.} The bias term $2LC^{2}\eta$ vanishes, and (1) recovers the standard SGD bound for smooth non‑convex functions.
\item \textbf{Very small $C$.} The variance term $\frac{2\sigma^{2}}{C^{2}}\eta$ dominates; one should increase $C$ to avoid excessive variance amplification.
\end{itemize}
