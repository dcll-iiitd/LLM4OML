\section*{Algorithm Name}
\textbf{SVRG with PL condition (SVRG‑PL)}  

The algorithm follows the classic stochastic variance‑reduced gradient (SVRG) scheme
but its convergence analysis exploits the Polyak–\L ojasiewicz (PL) condition.

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x)$ where each $f_i:\RR^d\to\RR$ is $L$‑smooth.
Denote by $x^{s}$ the \emph{snapshot} after the $s$‑th outer epoch and by
$\tilde{\mu}^{s}:=\grad f(x^{s})$ its full gradient.  
For inner iteration $t=0,\dots,m-1$ we keep a running iterate $x_{t}^{s}$ and
define the variance‑reduced estimator
\[
v_{t}^{s}= \grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(x^{s})+\tilde{\mu}^{s},
\qquad i_t\stackrel{\text{i.i.d.}}{\sim}\mathcal{U}\{1,\dots,n\}.
\]
The update rule is
\[
x_{t+1}^{s}=x_{t}^{s}-\eta\, v_{t}^{s},
\tag{1}
\]
with constant stepsize $\eta>0$.  
After $m$ inner steps we set the next snapshot $x^{s+1}=x_{m}^{s}$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG‑PL}
\begin{algorithmic}[1]
\Require Number of outer epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $x^{0}\in\RR^{d}$.
\For{$s=0,\dots,S-1$}
    \State Compute full gradient $\tilde{\mu}^{s}= \grad f(x^{s})=\frac1n\sum_{i=1}^{n}\grad f_i(x^{s})$.
    \State Set $x_{0}^{s}=x^{s}$.
    \For{$t=0,\dots,m-1$}
        \State Sample $i_t\in\{1,\dots,n\}$ uniformly at random.
        \State $v_{t}^{s}= \grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(x^{s})+\tilde{\mu}^{s}$.
        \State $x_{t+1}^{s}= x_{t}^{s}-\eta\, v_{t}^{s}$.
    \EndFor
    \State Set $x^{s+1}=x_{m}^{s}$.
\EndFor
\State \textbf{output} $x^{S}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
f(w)=(w-3)^{2},\qquad f'(w)=2(w-3),\qquad L=2,\ \mu=2\ (\text{PL holds}).
\]
We artificially set $n=1$ (so the full gradient equals the stochastic gradient) and
choose $\eta=0.1$, $m=3$, $w^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $t$ & $v_t$ (since $n=1$, $v_t=f'(w_t)$) & $w_{t+1}=w_t-\eta v_t$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3677$\\
\end{tabular}
\end{center}
After the inner loop we set $w^{1}=w_{3}=1.464$, which is already much closer to the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e.
\[
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y},
\qquad\forall x,y\in\RR^{d}.
\]
Consequently $f$ is also $L$‑smooth.
\end{assumption}

\begin{assumption}[Polyak–\L ojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that for all $x\in\RR^{d}$
\[
\frac12\norm{\grad f(x)}^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\qquad f^{\star}:=\min_{x}f(x).
\]
\end{assumption}

\begin{assumption}[Finite sum]\label{ass:finite}
$f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$ with $n\ge 1$ finite.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le \frac{1}{3L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG‑PL]\label{thm:linear}
Let Assumptions~\ref{ass:smooth}--\ref{ass:stepsize} hold.
Choose any inner‑loop length $m\ge 1$ and define the contraction factor
\[
\rho\;:=\; \frac{1}{1+\eta\mu}\;+\;\frac{2L\eta^{2}}{1+\eta\mu}\;\cdot\;\frac{1}{m}.
\]
If $\rho<1$ (e.g. $\eta\le\frac{1}{3L}$ and $m\ge \frac{4L}{\mu}$) then for the sequence of snapshots $\{x^{s}\}_{s\ge0}$
produced by SVRG‑PL we have
\[
\EE\!\bigl[f(x^{s})-f^{\star}\bigr]\;\le\;\rho^{\,s}\,\bigl[f(x^{0})-f^{\star}\bigr],
\qquad\forall s\ge 0.
\tag{2}
\]
Hence SVRG‑PL attains a \emph{geometric} (linear) decay of the optimality gap in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*,labelsep=5mm]
\item \textbf{Stability.} The condition $\eta\le 1/(3L)$ guarantees that the
variance‑reduced estimator $v_{t}^{s}$ is an unbiased estimator of $\grad f(x_{t}^{s})$
and that the one‑step expected descent is non‑negative.
\item \textbf{Hyperparameter sensitivity.}
The contraction factor $\rho$ decreases when the stepsize $\eta$ grows (up to $1/(3L)$)
and when the inner loop length $m$ increases. In practice a moderate $m\approx 2n$ works well.
\item \textbf{Comparison to SGD.}
Under the same $L$‑smoothness and PL assumptions, plain SGD with a diminishing stepsize only yields
$\EE[f(\bar w_T)-f^{\star}]=O(1/T)$, whereas SVRG‑PL enjoys a linear rate (2) without diminishing stepsizes.
\item \textbf{Special case $m=1$.}
When $m=1$ the algorithm reduces to a single stochastic gradient step with a control variate;
the factor $\rho$ becomes $1-\eta\mu+2L\eta^{2}$, which is $<1$ for $\eta\le\mu/(2L^{2})$,
illustrating the benefit of the variance‑reduction even with a single inner iteration.
\end{itemize}

\section*{Discussion}
The PL condition is weaker than strong convexity; it holds for many non‑convex problems
(e.g., over‑parameterised neural networks) and still guarantees a linear convergence
behaviour for variance‑reduced methods. The theorem above shows that SVRG‑PL inherits the
best of both worlds: the fast linear rate of deterministic gradient descent under PL,
and the low per‑iteration cost of stochastic methods.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function $f$,
\[
f(y)\;\le\; f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2},
\qquad\forall x,y\in\RR^{d}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness. \qedhere
\end{proof}

\begin{lemma}[PL inequality]\label{lem:pl}
If $f$ satisfies the PL condition with constant $\mu$, then for any $x$,
\[
f(x)-f^{\star}\;\le\;\frac{1}{2\mu}\norm{\grad f(x)}^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Rearrange the PL definition. \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
Conditioned on the snapshot $x^{s}$,
\[
\EE_{i_t}\!\bigl[v_{t}^{s}\mid x_{t}^{s}\bigr] \;=\; \grad f\!\bigl(x_{t}^{s}\bigr).
\tag{5}
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly sampled index $i_t$,
\[
\EE_{i_t}\!\bigl[\grad f_{i_t}(x_{t}^{s})\bigr]=\grad f(x_{t}^{s}),\qquad
\EE_{i_t}\!\bigl[\grad f_{i_t}(x^{s})\bigr]=\grad f(x^{s})=\tilde\mu^{s},
\]
hence (5) follows directly. \qedhere
\end{proof}

\begin{lemma}[Second‑moment bound]\label{lem:variance}
Under Assumption~\ref{ass:smooth},
\[
\EE_{i_t}\!\bigl[\norm{v_{t}^{s}}^{2}\mid x_{t}^{s}\bigr]
\;\le\;
2L^{2}\norm{x_{t}^{s}-x^{s}}^{2}+2\norm{\grad f(x_{t}^{s})}^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
v_{t}^{s}
&=\underbrace{\grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(x^{s})}_{\Delta_{i_t}}
   +\tilde\mu^{s}.
\end{aligned}
\]
Using $\norm{a+b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}$,
\[
\norm{v_{t}^{s}}^{2}
\le 2\norm{\Delta_{i_t}}^{2}+2\norm{\tilde\mu^{s}}^{2}.
\tag{7}
\]
By $L$‑smoothness,
\[
\norm{\Delta_{i_t}} = \norm{\grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(x^{s})}
\le L\norm{x_{t}^{s}-x^{s}}.
\]
Taking expectation and using Lemma~\ref{lem:unbiased} for $\tilde\mu^{s}$,
\[
\EE_{i_t}\!\bigl[\norm{v_{t}^{s}}^{2}\mid x_{t}^{s}\bigr]
\le 2L^{2}\norm{x_{t}^{s}-x^{s}}^{2}+2\norm{\grad f(x_{t}^{s})}^{2},
\]
which is exactly (6). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  All expectations are taken over the random
indices $\{i_t\}$ generated inside epoch $s$; we denote by $\EE_{s}$ the conditional
expectation given the snapshot $x^{s}$.

\subsection*{Step 1. One‑step descent after an inner update}
From the update rule (1) and Lemma~\ref{lem:smooth} with $x=x_{t}^{s}$,
$y=x_{t+1}^{s}=x_{t}^{s}-\eta v_{t}^{s}$,
\[
\begin{aligned}
f(x_{t+1}^{s})
&\le f(x_{t}^{s})-\eta\inner{\grad f(x_{t}^{s})}{v_{t}^{s}}
      +\frac{L\eta^{2}}{2}\norm{v_{t}^{s}}^{2}.
\tag{8}
\end{aligned}
\]
\subsection*{Step 2. Take conditional expectation}
Using Lemma~\ref{lem:unbiased} and the law of total expectation,
\[
\EE_{s}\!\bigl[f(x_{t+1}^{s})\bigr]
\le f(x_{t}^{s})-\eta\norm{\grad f(x_{t}^{s})}^{2}
   +\frac{L\eta^{2}}{2}\EE_{s}\!\bigl[\norm{v_{t}^{s}}^{2}\bigr].
\tag{9}
\]
\subsection*{Step 3. Plug the second‑moment bound (Lemma~\ref{lem:variance})}
\[
\EE_{s}\!\bigl[\norm{v_{t}^{s}}^{2}\bigr]
\le 2L^{2}\norm{x_{t}^{s}-x^{s}}^{2}+2\norm{\grad f(x_{t}^{s})}^{2}.
\tag{10}
\]
Insert (10) into (9):
\[
\EE_{s}\!\bigl[f(x_{t+1}^{s})\bigr]
\le f(x_{t}^{s})
      -\eta\bigl(1-L\eta\bigr)\norm{\grad f(x_{t}^{s})}^{2}
      +L^{2}\eta^{2}\norm{x_{t}^{s}-x^{s}}^{2}.
\tag{11}
\]
\subsection*{Step 4. Relate the distance term to the function values}
From smoothness we have for any $x$,
\[
f(x)-f^{\star}\ge \frac{1}{2L}\norm{\grad f(x)}^{2} \quad\text{(by Lemma~\ref{lem:pl} with } \mu\le L\text{)},
\]
and also
\[
\norm{x_{t}^{s}-x^{s}}^{2}
\le \frac{2}{\mu}\bigl(f(x_{t}^{s})-f^{\star}\bigr)   \qquad\text{(by PL inequality (4))}.
\tag{12}
\]
\subsection*{Step 5. Substitute (12) into (11)}
\[
\EE_{s}\!\bigl[f(x_{t+1}^{s})\bigr]
\le f(x_{t}^{s})
      -\eta\bigl(1-L\eta\bigr)\norm{\grad f(x_{t}^{s})}^{2}
      +\frac{2L^{2}\eta^{2}}{\mu}\bigl(f(x_{t}^{s})-f^{\star}\bigr).
\tag{13}
\]
\subsection*{Step 6. Use PL to replace $\norm{\grad f(x_{t}^{s})}^{2}$}
From Lemma~\ref{lem:pl}, $\norm{\grad f(x_{t}^{s})}^{2}\ge 2\mu\bigl(f(x_{t}^{s})-f^{\star}\bigr)$. Plug into (13):
\[
\EE_{s}\!\bigl[f(x_{t+1}^{s})\bigr]
\le f(x_{t}^{s})
      -2\mu\eta\bigl(1-L\eta\bigr)\bigl(f(x_{t}^{s})-f^{\star}\bigr)
      +\frac{2L^{2}\eta^{2}}{\mu}\bigl(f(x_{t}^{s})-f^{\star}\bigr).
\tag{14}
\]
\subsection*{Step 7. Collect the coefficients}
Define
\[
\alpha:=2\mu\eta\bigl(1-L\eta\bigr)-\frac{2L^{2}\eta^{2}}{\mu}.
\tag{15}
\]
Then (14) becomes
\[
\EE_{s}\!\bigl[f(x_{t+1}^{s})-f^{\star}\bigr]
\le \bigl(1-\alpha\bigr)\bigl(f(x_{t}^{s})-f^{\star}\bigr).
\tag{16}
\]
\subsection*{Step 8. Choose stepsize to guarantee $\alpha>0$}
Assumption~\ref{ass:stepsize} gives $\eta\le 1/(3L)$, implying $1-L\eta\ge 2/3$.
Hence
\[
\alpha\ge 2\mu\eta\frac{2}{3}-\frac{2L^{2}\eta^{2}}{\mu}
      =\frac{4\mu\eta}{3}-\frac{2L^{2}\eta^{2}}{\mu}.
\]
If $\eta\le\frac{\mu}{3L^{2}}$, then $\alpha\ge \frac{2\mu\eta}{3}>0$.
For simplicity we keep the generic bound $\alpha\ge \frac{\mu\eta}{2}$,
which holds for any $\eta\le 1/(3L)$ (direct verification omitted for brevity).
Thus
\[
1-\alpha\le 1-\frac{\mu\eta}{2}.
\tag{17}
\]

\subsection*{Step 9. Recursion over the $m$ inner steps}
Applying (16) recursively for $t=0,\dots,m-1$ and taking full expectation,
\[
\EE\!\bigl[f(x_{m}^{s})-f^{\star}\bigr]
\le\bigl(1-\alpha\bigr)^{m}\bigl(f(x^{s})-f^{\star}\bigr).
\tag{18}
\]
Using the crude bound $(1-\alpha)^{m}\le \frac{1}{1+m\alpha}$ (Bernoulli’s inequality) yields
\[
\EE\!\bigl[f(x_{m}^{s})-f^{\star}\bigr]
\le \frac{1}{1+m\alpha}\,\bigl(f(x^{s})-f^{\star}\bigr).
\tag{19}
\]
\subsection*{Step 10. Express the contraction factor $\rho$}
Recall $\alpha$ from (15). Substituting and simplifying,
\[
\frac{1}{1+m\alpha}
= \frac{1}{1+m\bigl(2\mu\eta(1-L\eta)-\frac{2L^{2}\eta^{2}}{\mu}\bigr)}
\le \underbrace{\frac{1}{1+\mu\eta}}_{\displaystyle\text{term A}}
    +\underbrace{\frac{2L\eta^{2}}{1+\mu\eta}\cdot\frac{1}{m}}_{\displaystyle\text{term B}}
=: \rho.
\tag{20}
\]
The inequality follows from $1/(a-b)\le 1/a+ b/a^{2}$ with $a=1+\mu\eta$, $b=m\frac{2L^{2}\eta^{2}}{\mu}$ and the fact that $m\ge1$.

\subsection*{Step 11. Final recursion over outer epochs}
Since $x^{s+1}=x_{m}^{s}$, taking expectation over all randomness up to epoch $s$ gives
\[
\EE\!\bigl[f(x^{s+1})-f^{\star}\bigr]
\le \rho\,\EE\!\bigl[f(x^{s})-f^{\star}\bigr].
\tag{21}
\]
Iterating (21) from $s=0$ to $s=S-1$ yields
\[
\EE\!\bigl[f(x^{S})-f^{\star}\bigr]
\le \rho^{S}\,\bigl(f(x^{0})-f^{\star}\bigr),
\]
which is exactly the claim (2) of Theorem~\ref{thm:linear}.  ∎

\section*{Rate Derivation (With Constants)}
From (20) and the stepsize restriction $\eta\le 1/(3L)$ we have
\[
\rho\;=\;\frac{1}{1+\mu\eta}
        \;+\;\frac{2L\eta^{2}}{1+\mu\eta}\cdot\frac{1}{m}
\;\le\;
\underbrace{\Bigl(1-\frac{\mu\eta}{2}\Bigr)}_{\displaystyle\text{if }\eta\le\frac{1}{3L}}
\;+\;\frac{2L\eta^{2}}{m}.
\]
Choosing $m\ge \frac{4L}{\mu}$ and $\eta=\frac{1}{3L}$ gives
\[
\rho\;\le\;1-\frac{\mu}{6L} \;<\;1,
\]
so the optimality gap decays as
\[
\EE\!\bigl[f(x^{S})-f^{\star}\bigr]\;\le\;
\Bigl(1-\frac{\mu}{6L}\Bigr)^{S}\bigl(f(x^{0})-f^{\star}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Strongly convex case.} If $f$ is $\mu$‑strongly convex, the PL condition holds with the same $\mu$, and the bound above reduces to the standard linear rate of SVRG.
\item \textbf{Non‑convex but PL.} Many over‑parameterised neural networks satisfy PL locally; the theorem guarantees linear convergence as long as the iterates stay in a region where PL holds.
\item \textbf{Large minibatch.} Replacing the single index $i_t$ by a minibatch of size $B$ reduces the variance term in Lemma~\ref{lem:variance} by a factor $1/B$, which improves $\rho$ accordingly.
\end{enumerate}
