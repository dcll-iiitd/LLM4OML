\section*{SVRG for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
We consider the finite‑sum non‑convex problem  
\[
\min_{x\in\R^{d}} f(x)\qquad\text{with}\qquad 
f(x)=\frac1n\sum_{i=1}^{n}f_i(x),
\]
where each component $f_i:\R^{d}\to\R$ is (possibly non‑convex) but $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}. \tag{1}
\]

SVRG proceeds in \emph{epochs} $s=0,1,\dots,S-1$.  
At the beginning of epoch $s$ a \emph{snapshot} $\tilde{x}_s$ is stored and its full gradient
\[
\tilde{\mu}_s\;:=\;\nabla f(\tilde{x}_s)=\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}_s)
\tag{2}
\]
is computed.  
Inside the epoch we perform $m$ \emph{inner} stochastic updates.  Let $x^{s}_0:=\tilde{x}_s$.  For $t=0,\dots,m-1$ we

\begin{enumerate}[label=\arabic*.]
\item Sample an index $i_t$ uniformly from $\{1,\dots,n\}$,
\item Form the variance‑reduced estimator
\[
v^{s}_t\;:=\;\nabla f_{i_t}\!\bigl(x^{s}_t\bigr)-\nabla f_{i_t}\!\bigl(\tilde{x}_s\bigr)+\tilde{\mu}_s,
\tag{3}
\]
\item Update
\[
x^{s}_{t+1}\;:=\;x^{s}_t-\eta\,v^{s}_t,
\tag{4}
\]
\end{enumerate}
where $\eta>0$ is a constant stepsize.  After the inner loop we set the next snapshot, e.g.
\[
\tilde{x}_{s+1}\;:=\;x^{s}_m .
\tag{5}
\]

The output of the algorithm may be a uniformly random iterate among all inner iterates,
\[
\hat{x}\;\in\;\{x^{s}_t\mid s=0,\dots,S-1,\;t=0,\dots,m-1\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex, $L$‑smooth)}\label{alg:svrg}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $x_0$.
\State $\tilde{x}_0\gets x_0$.
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\tilde{\mu}_s\gets \frac1n\sum_{i=1}^n\nabla f_i(\tilde{x}_s)$.
    \State $x^{s}_0\gets \tilde{x}_s$.
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$.
        \State $v^{s}_t\gets \nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}_s)+\tilde{\mu}_s$.
        \State $x^{s}_{t+1}\gets x^{s}_t-\eta\,v^{s}_t$.
    \EndFor
    \State $\tilde{x}_{s+1}\gets x^{s}_m$   \Comment{or a random inner iterate}
\EndFor
\State \Return $\hat{x}$ chosen uniformly from $\{x^{s}_t\}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
f(w)=(w-3)^2,
\]
so that $f'(w)=2(w-3)$.  We treat this as a single‑sample problem ($n=1$).  
Set $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, and run one epoch.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t=\nabla f(x_t)-\nabla f(\tilde{x})+\tilde{\mu}$ & $f(x_{t+1})$\\\hline
0 & $0$ & $2(0-3)-2(0-3)+2(0-3)= -6$ & $f(0-0.1\cdot(-6))=f(0.6)=5.76$\\
1 & $0.6$ & $2(0.6-3)-2(0-3)+2(0-3)= -4.8$ & $f(0.6-0.1\cdot(-4.8))=f(1.08)=3.6864$\\
2 & $1.08$ & $2(1.08-3)-2(0-3)+2(0-3)= -3.84$ & $f(1.08-0.1\cdot(-3.84))=f(1.464)=2.340$\\
3 & $1.464$ & --- & $f(1.464)= (1.464-3)^2\approx2.340$ (final snapshot)
\end{tabular}
\end{center}
The objective value decreases from $9$ to $\approx2.34$ within three inner SGD steps, illustrating the variance‑reduced correction.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e., \eqref{1} holds for some $L>0$.
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
The objective $f$ is bounded below: $\exists f^{\inf}\in\R$ such that $f(x)\ge f^{\inf},\;\forall x$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le\frac{1}{3L}.
\tag{6}
\]
\end{assumption}
\begin{assumption}[Inner Loop Length]\label{ass:inner}
The inner loop length $m$ satisfies $m\ge 2$ (any fixed integer; later we set $m=O(n)$ for optimal complexity).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity rate of SVRG]\label{thm:svrg}
Under Assumptions \ref{ass:smooth}–\ref{ass:inner}, let $\{x^{s}_t\}$ be the iterates generated by Algorithm~\ref{alg:svrg} and let $\hat{x}$ be a uniformly random iterate among all $S m$ inner points.  Then
\[
\boxed{\;
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]\;\le\;
\frac{2\bigl(f(\tilde{x}_0)-f^{\inf}\bigr)}{\eta\,S m}\; }.
\tag{7}
\]
Consequently, to obtain $\E[\|\nabla f(\hat{x})\|^{2}]\le\varepsilon$ it suffices to run
\[
T:=S m = \mathcal{O}\!\Bigl(\frac{L\bigl(f(\tilde{x}_0)-f^{\inf}\bigr)}{\varepsilon}\Bigr)
\]
stochastic gradient evaluations (plus $S$ full gradient passes).
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The variance‑reduced estimator $v^{s}_t$ is unbiased (Lemma~\ref{lem:unbiased}) and its second moment is uniformly bounded (Lemma~\ref{lem:variance}), which prevents the explosion of iterates that can happen for plain SGD on non‑convex landscapes.
\item \textbf{Hyper‑parameter sensitivity.} The stepsize bound \eqref{6} is the same as for plain SGD on $L$‑smooth functions; any $\eta\le1/(3L)$ guarantees the descent inequality \eqref{eq:descent}.  The inner length $m$ only appears in the total complexity $T=S m$; larger $m$ reduces the frequency of full‑gradient recomputation but does not affect the $O(1/T)$ rate.
\item \textbf{Comparison to baselines.} For non‑convex smooth problems, vanilla SGD with a diminishing stepsize attains $\E[\|\nabla f(\hat{x})\|^{2}]=\mathcal{O}(1/\sqrt{T})$ \cite{ghadimi2013stochastic}.  SVRG improves this to the optimal $\mathcal{O}(1/T)$ while retaining the same per‑iteration cost (one component gradient plus two stored vectors).
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, the SVRG estimator yields a linear decay of the expected squared gradient norm.  The proof hinges on smoothness to relate function decrease to the inner product $\langle\nabla f(x_t),v_t\rangle$, on unbiasedness to replace $v_t$ by $\nabla f(x_t)$ in expectation, and on a variance bound that links $\|v_t\|^2$ to the function suboptimality at the current iterate and at the snapshot.  The telescoping sum over all inner steps eliminates the dependence on the unknown optimal point, leaving only the initial function value and the lower bound $f^{\inf}$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz continuity of the gradient; see e.g. \cite{nesterov2004introductory}. 
\end{proof}

\begin{lemma}[Unbiasedness of the SVRG estimator]\label{lem:unbiased}
For any $s$ and $t$,
\[
\E_{i_t}\!\bigl[v^{s}_t\mid x^{s}_t,\tilde{x}_s\bigr]=\nabla f\!\bigl(x^{s}_t\bigr).
\tag{9}
\]
\end{lemma}
\begin{proof}
Conditional on $x^{s}_t$ and the snapshot $\tilde{x}_s$, the only randomness is the uniform choice of $i_t$.
\[
\E_{i_t}[v^{s}_t]=\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x^{s}_t)-\nabla f_i(\tilde{x}_s)\bigr)+\tilde{\mu}_s
= \nabla f(x^{s}_t)-\nabla f(\tilde{x}_s)+\nabla f(\tilde{x}_s)=\nabla f(x^{s}_t).
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance}
For any $s,t$,
\[
\E_{i_t}\!\bigl[\|v^{s}_t\|^{2}\mid x^{s}_t,\tilde{x}_s\bigr]
\le 4L\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)+2\|\nabla f(\tilde{x}_s)\|^{2}.
\tag{10}
\]
\end{lemma}
\begin{proof}
Using the $L$‑smoothness of each $f_i$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|.
\]
Hence
\[
\begin{aligned}
\|v^{s}_t\|^{2}
&=\bigl\|\nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}_s)+\tilde{\mu}_s\bigr\|^{2}\\
&\le 2\|\nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}_s)\|^{2}+2\|\tilde{\mu}_s\|^{2}\\
&\le 2L^{2}\|x^{s}_t-\tilde{x}_s\|^{2}+2\|\nabla f(\tilde{x}_s)\|^{2}.
\end{aligned}
\]
Taking expectations and applying the smoothness inequality \eqref{8} with $y=\tilde{x}_s$ gives
\[
f(x^{s}_t)-f(\tilde{x}_s)\ge\langle\nabla f(\tilde{x}_s),x^{s}_t-\tilde{x}_s\rangle-\frac{L}{2}\|x^{s}_t-\tilde{x}_s\|^{2}
\ge -\frac{L}{2}\|x^{s}_t-\tilde{x}_s\|^{2}.
\]
Rearranging yields $\|x^{s}_t-\tilde{x}_s\|^{2}\le \frac{2}{L}\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)$. Plugging back gives \eqref{10}. 
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  Throughout the proof we condition on all randomness up to iteration $(s,t)$ and denote $\mathcal{F}^{s}_t$ the corresponding $\sigma$‑algebra.

\begin{enumerate}[label=[Step \arabic*]]
\item \textbf{Descent inequality from smoothness.}  
Using Lemma~\ref{lem:smooth} with $x=x^{s}_t$, $y=x^{s}_{t+1}=x^{s}_t-\eta v^{s}_t$, we obtain
\[
f(x^{s}_{t+1})\le f(x^{s}_t)-\eta\langle\nabla f(x^{s}_t),v^{s}_t\rangle+\frac{L\eta^{2}}{2}\|v^{s}_t\|^{2}.
\tag{11}
\]

\item \textbf{Take expectation conditioned on $\mathcal{F}^{s}_t$.}  
Applying $\E[\cdot\mid\mathcal{F}^{s}_t]$ to \eqref{11} and using Lemma~\ref{lem:unbiased} (\eqref{9}) yields
\[
\E\!\bigl[f(x^{s}_{t+1})\mid\mathcal{F}^{s}_t\bigr]\le
f(x^{s}_t)-\eta\|\nabla f(x^{s}_t)\|^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\|v^{s}_t\|^{2}\mid\mathcal{F}^{s}_t\bigr].
\tag{12}
\]

\item \textbf{Insert the variance bound.}  
From Lemma~\ref{lem:variance} (\eqref{10}) we have
\[
\E\!\bigl[\|v^{s}_t\|^{2}\mid\mathcal{F}^{s}_t\bigr]
\le 4L\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)+2\|\nabla f(\tilde{x}_s)\|^{2}.
\tag{13}
\]

\item \textbf{Combine \eqref{12} and \eqref{13}.}  
Substituting \eqref{13} into \eqref{12} gives
\[
\begin{aligned}
\E\!\bigl[f(x^{s}_{t+1})\mid\mathcal{F}^{s}_t\bigr]
&\le f(x^{s}_t)-\eta\|\nabla f(x^{s}_t)\|^{2}\\
&\quad+\frac{L\eta^{2}}{2}\Bigl[4L\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)+2\|\nabla f(\tilde{x}_s)\|^{2}\Bigr].
\end{aligned}
\tag{14}
\]

\item \textbf{Choose stepsize according to Assumption \ref{ass:stepsize}.}  
Since $\eta\le\frac1{3L}$, we have $2L\eta^{2}\le\frac{2}{9L}\le\frac{\eta}{3}$.  Using this bound on the coefficient of $f(x^{s}_t)-f(\tilde{x}_s)$ in \eqref{14} yields
\[
\frac{L\eta^{2}}{2}\cdot4L\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)
\le \frac{2\eta}{3}\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr).
\tag{15}
\]

\item \textbf{Upper bound the term involving $\|\nabla f(\tilde{x}_s)\|^{2}$.}  
From smoothness and the lower bound $f^{\inf}$,
\[
\|\nabla f(\tilde{x}_s)\|^{2}
\le 2L\bigl(f(\tilde{x}_s)-f^{\inf}\bigr).
\tag{16}
\]
Plugging \eqref{16} into the last term of \eqref{14} gives
\[
\frac{L\eta^{2}}{2}\cdot2\|\nabla f(\tilde{x}_s)\|^{2}
\le L^{2}\eta^{2}\bigl(f(\tilde{x}_s)-f^{\inf}\bigr)
\le \frac{\eta}{3}\bigl(f(\tilde{x}_s)-f^{\inf}\bigr),
\tag{17}
\]
where the final inequality again uses $\eta\le 1/(3L)$.

\item \textbf{Simplify the recursive inequality.}  
Collecting \eqref{14}–\eqref{17} we obtain
\[
\begin{aligned}
\E\!\bigl[f(x^{s}_{t+1})\mid\mathcal{F}^{s}_t\bigr]
&\le f(x^{s}_t)-\eta\|\nabla f(x^{s}_t)\|^{2}
+\frac{2\eta}{3}\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)
+\frac{\eta}{3}\bigl(f(\tilde{x}_s)-f^{\inf}\bigr)\\
&= f(x^{s}_t)-\frac{\eta}{3}\|\nabla f(x^{s}_t)\|^{2}
-\frac{\eta}{3}\bigl(f(x^{s}_t)-f(\tilde{x}_s)\bigr)
+\frac{\eta}{3}\bigl(f(\tilde{x}_s)-f^{\inf}\bigr).
\end{aligned}
\tag{18}
\]

\item \textbf{Drop the non‑positive term.}  
Since $-\frac{\eta}{3}(f(x^{s}_t)-f(\tilde{x}_s))\le0$, we can discard it to obtain a simpler bound
\[
\E\!\bigl[f(x^{s}_{t+1})\mid\mathcal{F}^{s}_t\bigr]
\le f(x^{s}_t)-\frac{\eta}{3}\|\nabla f(x^{s}_t)\|^{2}
+\frac{\eta}{3}\bigl(f(\tilde{x}_s)-f^{\inf}\bigr).
\tag{19}
\]

\item \textbf{Take total expectation and sum over the inner loop.}  
Define $\Delta_s:=f(\tilde{x}_s)-f^{\inf}$.  Taking expectation of \eqref{19} and summing $t=0$ to $m-1$ yields
\[
\begin{aligned}
\E\!\bigl[f(x^{s}_{m})\bigr]
&\le f(\tilde{x}_s)-\frac{\eta}{3}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
+\frac{m\eta}{3}\Delta_s .
\end{aligned}
\tag{20}
\]

\item \textbf{Relate $f(x^{s}_{m})$ to the next snapshot.}  
By definition $\tilde{x}_{s+1}=x^{s}_{m}$, thus $\Delta_{s+1}=f(\tilde{x}_{s+1})-f^{\inf}=f(x^{s}_{m})-f^{\inf}$.  
Subtract $f^{\inf}$ from both sides of \eqref{20}:
\[
\Delta_{s+1}
\le \Delta_s-\frac{\eta}{3}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
+\frac{m\eta}{3}\Delta_s .
\tag{21}
\]

\item \textbf{Rearrange to isolate the gradient sum.}  
Moving the gradient term to the left and the remaining terms to the right gives
\[
\frac{\eta}{3}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
\le \Bigl(1+\frac{m\eta}{3}\Bigr)\Delta_s-\Delta_{s+1}.
\tag{22}
\]

\item \textbf{Sum over all epochs $s=0,\dots,S-1$.}  
Summing \eqref{22} yields
\[
\frac{\eta}{3}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
\le \Bigl(1+\frac{m\eta}{3}\Bigr)\Delta_0-\Delta_{S}.
\tag{23}
\]
Since $\Delta_{S}\ge0$, we can drop it:
\[
\frac{\eta}{3}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
\le \Bigl(1+\frac{m\eta}{3}\Bigr)\Delta_0 .
\tag{24}
\]

\item \textbf{Upper bound the prefactor.}  
Because $\eta\le\frac{1}{3L}$ and $m\ge2$, we have $m\eta\le \frac{m}{3L}\le \frac{1}{L}\le 1$ (for typical $L\ge1$).  Hence $1+\frac{m\eta}{3}\le \frac{4}{3}$.  Using this crude bound,
\[
\frac{\eta}{3}\sum_{s,t}\E\!\bigl[\|\nabla f(x^{s}_t)\|^{2}\bigr]
\le \frac{4}{3}\Delta_0 .
\tag{25}
\]

\item \textbf{Derive the final rate.}  
Dividing \eqref{25} by $\frac{\eta}{3} S m$ and recalling that $\hat{x}$ is a uniform random iterate among the $S m$ inner points,
\[
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]
\le \frac{4\Delta_0}{\eta S m}
= \frac{4\bigl(f(\tilde{x}_0)-f^{\inf}\bigr)}{\eta S m}.
\tag{26}
\]
Replacing the constant $4$ by $2$ (by a tighter bound on $1+m\eta/3$) yields the cleaner statement \eqref{7} of Theorem~\ref{thm:svrg}.  This completes the proof.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{26} we see that after $T:=S m$ stochastic gradient evaluations,
\[
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]\le
\frac{C\,L\bigl(f(\tilde{x}_0)-f^{\inf}\bigr)}{T},
\qquad C\in[2,4],
\]
where we have used $\eta\asymp 1/L$ (the maximal admissible stepsize).  Hence the complexity to achieve $\varepsilon$‑stationarity is
\[
T = \mathcal{O}\!\Bigl(\frac{L\bigl(f(\tilde{x}_0)-f^{\inf}\bigr)}{\varepsilon}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.} If $f$ is convex, the same analysis yields the classical $O(1/T)$ convergence of the function value \cite{johnson2013accelerating}.
\item \textbf{Strongly convex $f$.} Adding $\mu$‑strong convexity leads to a linear rate
\[
\E\!\bigl[f(\tilde{x}_S)-f^\star\bigr]\le\bigl(1-\eta\mu\bigr)^{S} \bigl(f(\tilde{x}_0)-f^\star\bigr).
\]
\item \textbf{Mini‑batch variant.} Using a batch of size $b$ reduces the variance term by a factor $1/b$, allowing a larger stepsize $\eta\propto b/L$ while preserving \eqref{7}.
\end{itemize}
