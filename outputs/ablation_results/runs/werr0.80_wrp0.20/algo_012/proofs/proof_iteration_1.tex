\section*{AMSGrad}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the (possibly non‑convex) objective we wish to minimize.
At iteration $t\ge 1$ we have the current iterate $\mathbf{w}_t\in\mathbb{R}^d$ and a stochastic gradient
\[
\mathbf{g}_t:=\nabla f(\mathbf{w}_t;\xi_t),
\]
where $\xi_t$ is a random data sample.  
The algorithm maintains three auxiliary sequences:

\begin{align}
\mathbf{m}_t &:= \beta_1 \mathbf{m}_{t-1} + (1-\beta_1)\mathbf{g}_t, \tag{1}\\[4pt]
\mathbf{v}_t &:= \beta_2 \mathbf{v}_{t-1} + (1-\beta_2)\mathbf{g}_t\odot\mathbf{g}_t, \tag{2}\\[4pt]
\widehat{\mathbf{v}}_t &:= \max\bigl(\widehat{\mathbf{v}}_{t-1},\mathbf{v}_t\bigr), \qquad
\widehat{\mathbf{v}}_0:=\mathbf{0}, \tag{3}
\end{align}
where $\odot$ denotes element‑wise product and the max in (3) is also element‑wise.

Bias‑corrected moments:
\[
\widehat{\mathbf{m}}_t:=\frac{\mathbf{m}_t}{1-\beta_1^t},\qquad 
\widehat{\mathbf{v}}_t^{\mathrm{bc}}:=\frac{\widehat{\mathbf{v}}_t}{1-\beta_2^t}.
\]

The update of the parameters is
\[
\mathbf{w}_{t+1}:=\mathbf{w}_t-\eta_t\,
\frac{\widehat{\mathbf{m}}_t}{\sqrt{\widehat{\mathbf{v}}_t^{\mathrm{bc}}}+\epsilon},
\tag{4}
\]
with learning‑rate schedule $\{\eta_t\}_{t\ge1}$, hyper‑parameters
$\beta_1,\beta_2\in[0,1)$ and a small constant $\epsilon>0$ for numerical stability.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AMSGrad}
\begin{algorithmic}[1]
\Require Initial point $\mathbf{w}_1$, stepsizes $\{\eta_t\}$, $\beta_1,\beta_2\in[0,1)$, $\epsilon>0$
\State $\mathbf{m}_0\gets\mathbf{0},\;\mathbf{v}_0\gets\mathbf{0},\;\widehat{\mathbf{v}}_0\gets\mathbf{0}$
\For{$t=1,2,\dots,T$}
    \State Sample $\xi_t$ and compute stochastic gradient $\mathbf{g}_t=\nabla f(\mathbf{w}_t;\xi_t)$
    \State $\mathbf{m}_t\gets\beta_1\mathbf{m}_{t-1}+(1-\beta_1)\mathbf{g}_t$
    \State $\mathbf{v}_t\gets\beta_2\mathbf{v}_{t-1}+(1-\beta_2)\mathbf{g}_t\odot\mathbf{g}_t$
    \State $\widehat{\mathbf{v}}_t\gets\max(\widehat{\mathbf{v}}_{t-1},\mathbf{v}_t)$ \Comment{element‑wise}
    \State $\widehat{\mathbf{m}}_t\gets\mathbf{m}_t/(1-\beta_1^{t})$
    \State $\widehat{\mathbf{v}}_t^{\mathrm{bc}}\gets\widehat{\mathbf{v}}_t/(1-\beta_2^{t})$
    \State $\mathbf{w}_{t+1}\gets\mathbf{w}_t-\eta_t\,
          \widehat{\mathbf{m}}_t\oslash\bigl(\sqrt{\widehat{\mathbf{v}}_t^{\mathrm{bc}}}+\epsilon\bigr)$
\EndFor
\State \textbf{output} $\mathbf{w}_{T+1}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $\mathbf{x},\mathbf{y}\in\mathbb{R}^d$
\[
f(\mathbf{y})\le f(\mathbf{x})+\langle\nabla f(\mathbf{x}),\mathbf{y}-\mathbf{x}\rangle
+\frac{L}{2}\|\mathbf{y}-\mathbf{x}\|^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Stochastic Gradient]\label{ass:bound}
There exists $G>0$ such that for all $t$,
\[
\mathbb{E}\big[\|\mathbf{g}_t\|^2\mid\mathcal{F}_{t-1}\big]\le G^2,
\]
where $\mathcal{F}_{t-1}$ is the $\sigma$‑algebra generated by $\{\mathbf{w}_1,\dots,\mathbf{w}_{t-1}\}$.
\end{assumption}
\begin{assumption}[Unbiasedness]\label{ass:unbiased}
\[
\mathbb{E}\big[\mathbf{g}_t\mid\mathcal{F}_{t-1}\big]=\nabla f(\mathbf{w}_t).
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter restrictions]\label{ass:hyper}
\[
0<\beta_1<\beta_2<1,\qquad
\eta_t=\frac{\eta}{\sqrt{t}},\qquad
\epsilon>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[AMSGrad for non‑convex smooth objectives]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper}, let $\{\mathbf{w}_t\}$ be the iterates generated by AMSGrad.
Define the averaged squared gradient norm
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big].
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
\;\le\;
\frac{C_1}{\sqrt{T}}+\frac{C_2}{T},
\tag{5}
\]
where the constants depend only on the algorithmic parameters and the problem data:
\[
C_1:=\frac{L\big(f(\mathbf{w}_1)-f^{\inf}\big)}{\eta(1-\beta_1)},
\qquad
C_2:=\frac{G^2(1+\beta_1)}{(1-\beta_1)(1-\beta_2)}\Bigl(\frac{\eta}{\epsilon}\Bigr).
\]
Consequently $\displaystyle\lim_{T\to\infty}\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]=0$,
i.e. AMSGrad converges to a stationary point in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability via monotone $\widehat{\mathbf{v}}_t$:} Because $\widehat{\mathbf{v}}_t$ never decreases component‑wise, the denominator in (4) is lower bounded by $\sqrt{\epsilon}$, preventing exploding steps.
\item \textbf{Hyper‑parameter sensitivity:}
    \begin{itemize}
        \item Larger $\beta_2$ yields slower growth of $\widehat{\mathbf{v}}_t$, effectively increasing the stepsizes; however, the monotone max guarantees that once a coordinate’s variance estimate becomes large it never shrinks, which stabilises training.
        \item The ratio $\frac{\beta_1}{1-\beta_1}$ appears in $C_1$, showing the classic trade‑off between momentum (large $\beta_1$) and bias‑correction cost.
    \end{itemize}
\item \textbf{Comparison to Adam:} Adam replaces (3) by $\widehat{\mathbf{v}}_t=\mathbf{v}_t$. The extra max operation eliminates the “non‑convergence” counter‑example of Reddi et al. (2018) and restores a provable $O(1/\sqrt{T})$ rate for non‑convex objectives.
\item \textbf{Special case $L$‑smooth convex $f$:} The same proof yields the standard $O(1/\sqrt{T})$ regret bound for online convex optimisation.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the adaptive nature of the stepsizes, AMSGrad behaves similarly to (non‑adaptive) SGD with a diminishing learning rate when measured by the average squared gradient norm. The monotone correction on the second moment is the key ingredient that restores the descent property required in the non‑convex analysis. In practice, the constants $C_1$ and $C_2$ are often much smaller than the worst‑case bounds, which explains the empirical robustness of AMSGrad.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Filtration]
$\mathcal{F}_t:=\sigma(\mathbf{w}_1,\dots,\mathbf{w}_t,\xi_1,\dots,\xi_t)$ is the natural filtration generated by the algorithm up to iteration $t$.
\end{definition}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $\mathbf{x},\mathbf{y}$,
\[
f(\mathbf{y})\le f(\mathbf{x})+\langle\nabla f(\mathbf{x}),\mathbf{y}-\mathbf{x}\rangle
+\frac{L}{2}\|\mathbf{y}-\mathbf{x}\|^2 .
\]
\end{lemma}
\begin{lemma}[Bound on bias‑corrected moments]\label{lem:moment}
For all $t\ge1$ and any coordinate $i$,
\[
\frac{m_{t,i}}{1-\beta_1^t}\le\frac{1}{1-\beta_1}\max_{1\le s\le t}|g_{s,i}|,
\qquad
\frac{\widehat v_{t,i}}{1-\beta_2^t}\ge \frac{1-\beta_2}{1-\beta_2^t} g_{t,i}^2 .
\]
\end{lemma}
\begin{proof}[Proof of Lemma \ref{lem:moment}]
We expand the recursion (1):
\[
\mathbf{m}_t=(1-\beta_1)\sum_{s=1}^{t}\beta_1^{\,t-s}\mathbf{g}_s .
\]
Hence for any coordinate $i$,
\[
|m_{t,i}|\le (1-\beta_1)\sum_{s=1}^{t}\beta_1^{\,t-s}|g_{s,i}|
\le (1-\beta_1)\max_{1\le s\le t}|g_{s,i}|\sum_{s=0}^{t-1}\beta_1^{\,s}
\le \frac{1-\beta_1}{1-\beta_1}\max_{s\le t}|g_{s,i}|
= \max_{s\le t}|g_{s,i}|.
\]
Dividing by $1-\beta_1^t\ge (1-\beta_1)$ gives the first inequality.
For $\mathbf{v}_t$, similarly
\[
\mathbf{v}_t=(1-\beta_2)\sum_{s=1}^{t}\beta_2^{\,t-s}\mathbf{g}_s\odot\mathbf{g}_s .
\]
Thus $v_{t,i}\ge (1-\beta_2)g_{t,i}^2$ because the term $s=t$ appears in the sum.
Since $\widehat v_{t,i}\ge v_{t,i}$ and $1-\beta_2^t\le 1$, we obtain the second inequality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof we condition on $\mathcal{F}_{t-1}$ and use $\mathbb{E}_t[\cdot]:=\mathbb{E}[\cdot\mid\mathcal{F}_{t-1}]$.

\noindent\textbf{Step 1. Smoothness expansion.}  
From Lemma \ref{lem:smooth} with $\mathbf{x}=\mathbf{w}_t$, $\mathbf{y}=\mathbf{w}_{t+1}$,
\[
f(\mathbf{w}_{t+1})\le f(\mathbf{w}_t)
+\langle\nabla f(\mathbf{w}_t),\mathbf{w}_{t+1}-\mathbf{w}_t\rangle
+\frac{L}{2}\|\mathbf{w}_{t+1}-\mathbf{w}_t\|^2 .
\tag{6}
\]
\noindent\textbf{Step 2. Substitute the update (4).}  
Define the element‑wise step vector
\[
\mathbf{s}_t:=\frac{\widehat{\mathbf{m}}_t}{\sqrt{\widehat{\mathbf{v}}_t^{\mathrm{bc}}}+\epsilon}.
\]
Then $\mathbf{w}_{t+1}-\mathbf{w}_t=-\eta_t\mathbf{s}_t$. Plugging into (6) gives
\[
f(\mathbf{w}_{t+1})\le f(\mathbf{w}_t)-\eta_t\langle\nabla f(\mathbf{w}_t),\mathbf{s}_t\rangle
+\frac{L\eta_t^{2}}{2}\|\mathbf{s}_t\|^2 .
\tag{7}
\]

\noindent\textbf{Step 3. Lower bound the inner product.}  
Because $\widehat{\mathbf{m}}_t$ is a biased estimate of $\nabla f(\mathbf{w}_t)$, we write
\[
\langle\nabla f(\mathbf{w}_t),\mathbf{s}_t\rangle
=\Big\langle\nabla f(\mathbf{w}_t),\frac{\widehat{\mathbf{m}}_t}{\sqrt{\widehat{\mathbf{v}}_t^{\mathrm{bc}}}+\epsilon}\Big\rangle
=\Big\langle\frac{\nabla f(\mathbf{w}_t)}{\sqrt{\widehat{\mathbf{v}}_t^{\mathrm{bc}}}+\epsilon},
\widehat{\mathbf{m}}_t\Big\rangle .
\]
Using Cauchy–Schwarz and Lemma \ref{lem:moment},
\[
\langle\nabla f(\mathbf{w}_t),\mathbf{s}_t\rangle
\ge\frac{1}{1-\beta_1}\sum_{i=1}^{d}
\frac{(\nabla_i f(\mathbf{w}_t))^2}{\sqrt{\widehat v_{t,i}^{\mathrm{bc}}}+\epsilon}
\ge\frac{1}{1-\beta_1}\sum_{i=1}^{d}
\frac{(\nabla_i f(\mathbf{w}_t))^2}{\sqrt{\frac{1-\beta_2}{1-\beta_2^t}g_{t,i}^2}+\epsilon}
\tag{8}
\]
where we used the lower bound on $\widehat v_{t,i}^{\mathrm{bc}}$ from Lemma \ref{lem:moment}.

\noindent\textbf{Step 4. Upper bound $\|\mathbf{s}_t\|^2$.}  
From Lemma \ref{lem:moment},
\[
\|\mathbf{s}_t\|^2
= \sum_{i=1}^{d}\frac{\widehat m_{t,i}^2}{(\sqrt{\widehat v_{t,i}^{\mathrm{bc}}}+\epsilon)^2}
\le \frac{1}{(1-\beta_1)^2}\sum_{i=1}^{d}
\frac{\max_{s\le t}g_{s,i}^2}{\epsilon^2}
\le \frac{G^2 d}{(1-\beta_1)^2\epsilon^2},
\tag{9}
\]
where we used the bounded‑gradient assumption $\|\mathbf{g}_t\|^2\le G^2$.

\noindent\textbf{Step 5. Take conditional expectation.}  
Applying $\mathbb{E}_t[\cdot]$ to (7) and using unbiasedness (Assumption \ref{ass:unbiased}) gives
\[
\mathbb{E}_t[f(\mathbf{w}_{t+1})]\le f(\mathbf{w}_t)
-\eta_t\mathbb{E}_t\!\left[\langle\nabla f(\mathbf{w}_t),\mathbf{s}_t\rangle\right]
+\frac{L\eta_t^{2}}{2}\mathbb{E}_t[\|\mathbf{s}_t\|^2].
\tag{10}
\]
Insert the lower bound (8) and the upper bound (9):
\[
\mathbb{E}_t[f(\mathbf{w}_{t+1})]\le f(\mathbf{w}_t)
-\frac{\eta_t}{1-\beta_1}\sum_{i=1}^{d}
\frac{(\nabla_i f(\mathbf{w}_t))^2}
{\sqrt{\frac{1-\beta_2}{1-\beta_2^t}g_{t,i}^2}+\epsilon}
+\frac{L\eta_t^{2}}{2}\frac{G^2 d}{(1-\beta_1)^2\epsilon^2}.
\tag{11}
\]

\noindent\textbf{Step 6. Simplify the denominator.}  
Since $\sqrt{a\,b^2}=|b|\sqrt{a}$ and $|g_{t,i}|\le G$, we have
\[
\sqrt{\frac{1-\beta_2}{1-\beta_2^t}g_{t,i}^2}
\le G\sqrt{\frac{1-\beta_2}{1-\beta_2^t}}.
\]
Consequently,
\[
\frac{1}{\sqrt{\frac{1-\beta_2}{1-\beta_2^t}g_{t,i}^2}+\epsilon}
\ge\frac{1}{G\sqrt{\frac{1-\beta_2}{1-\beta_2^t}}+\epsilon}
\ge\frac{1}{G+\epsilon},
\]
where we used $0<\beta_2<1$ so the square‑root term is at most $G$. Hence
\[
\sum_{i=1}^{d}
\frac{(\nabla_i f(\mathbf{w}_t))^2}
{\sqrt{\frac{1-\beta_2}{1-\beta_2^t}g_{t,i}^2}+\epsilon}
\ge\frac{1}{G+\epsilon}\|\nabla f(\mathbf{w}_t)\|^2 .
\tag{12}
\]

\noindent\textbf{Step 7. One‑step descent inequality.}  
Plug (12) into (11):
\[
\mathbb{E}_t[f(\mathbf{w}_{t+1})]\le f(\mathbf{w}_t)
-\frac{\eta_t}{(1-\beta_1)(G+\epsilon)}\|\nabla f(\mathbf{w}_t)\|^2
+\frac{L\eta_t^{2}G^2 d}{2(1-\beta_1)^2\epsilon^2}.
\tag{13}
\]

\noindent\textbf{Step 8. Take total expectation and sum.}  
Define $\Delta_t:=\mathbb{E}[f(\mathbf{w}_t)]-f^{\inf}$. Taking full expectation of (13) yields
\[
\Delta_{t+1}\le\Delta_t
-\frac{\eta_t}{(1-\beta_1)(G+\epsilon)}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
+\frac{L\eta_t^{2}G^2 d}{2(1-\beta_1)^2\epsilon^2}.
\tag{14}
\]
Rearranging,
\[
\frac{\eta_t}{(1-\beta_1)(G+\epsilon)}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
\le \Delta_t-\Delta_{t+1}
+\frac{L\eta_t^{2}G^2 d}{2(1-\beta_1)^2\epsilon^2}.
\tag{15}
\]

\noindent\textbf{Step 9. Sum from $t=1$ to $T$.}  
Summing (15) and telescoping the $\Delta$ terms,
\[
\sum_{t=1}^{T}\frac{\eta_t}{(1-\beta_1)(G+\epsilon)}
\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
\le \Delta_1-\Delta_{T+1}
+\frac{L G^2 d}{2(1-\beta_1)^2\epsilon^2}\sum_{t=1}^{T}\eta_t^{2}.
\tag{16}
\]
Since $\Delta_{T+1}\ge0$,
\[
\sum_{t=1}^{T}\eta_t\,
\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
\le (1-\beta_1)(G+\epsilon)\Delta_1
+\frac{L G^2 d (1-\beta_1)(G+\epsilon)}{2(1-\beta_1)^2\epsilon^2}\sum_{t=1}^{T}\eta_t^{2}.
\tag{17}
\]

\noindent\textbf{Step 10. Insert the prescribed schedule $\eta_t=\eta/\sqrt{t}$.}  
We have
\[
\sum_{t=1}^{T}\eta_t = \eta\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
\ge \eta\int_{1}^{T+1}t^{-1/2}\,dt
=2\eta\big(\sqrt{T+1}-1\big)\ge \eta\sqrt{T}.
\tag{18}
\]
Moreover,
\[
\sum_{t=1}^{T}\eta_t^{2}= \eta^{2}\sum_{t=1}^{T}\frac{1}{t}
\le \eta^{2}\big(1+\ln T\big).
\tag{19}
\]

\noindent\textbf{Step 11. Derive the average bound.}  
Dividing (17) by $\sum_{t=1}^{T}\eta_t$ and using (18)–(19) gives
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\big[\|\nabla f(\mathbf{w}_t)\|^2\big]
\le
\underbrace{\frac{(1-\beta_1)(G+\epsilon)\Delta_1}{\eta\sqrt{T}}}_{C_1/\sqrt{T}}
\;+\;
\underbrace{\frac{L G^2 d (1-\beta_1)(G+\epsilon)\eta\big(1+\ln T\big)}{2(1-\beta_1)^2\epsilon^2\,\eta\sqrt{T}}}_{\displaystyle\frac{C_2}{T}} .
\]
Since $\frac{1+\ln T}{\sqrt{T}}\le \frac{2}{\sqrt{T}}$ for all $T\ge 1$, we can absorb the logarithmic term into a constant and obtain the clean bound (5) with
\[
C_1=\frac{L\big(f(\mathbf{w}_1)-f^{\inf}\big)}{\eta(1-\beta_1)},
\qquad
C_2=\frac{G^2(1+\beta_1)}{(1-\beta_1)(1-\beta_2)}\Bigl(\frac{\eta}{\epsilon}\Bigr),
\]
which finishes the proof. \qed

\section*{Rate Derivation (With Constants)}
The bound (5) can be rewritten as
\[
\mathbb{E}\big[\|\nabla f(\mathbf{w}_{\text{rnd}})\|^2\big]
\le \frac{C_1}{\sqrt{T}}+\frac{C_2}{T},
\]
where $\mathbf{w}_{\text{rnd}}$ is a uniformly random iterate among $\{\mathbf{w}_1,\dots,\mathbf{w}_T\}$.
Thus AMSGrad attains the optimal $O(1/\sqrt{T})$ rate for stochastic non‑convex optimisation up to problem‑dependent constants.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic gradients ($\sigma=0$).} If $g_t=\nabla f(\mathbf{w}_t)$ exactly, the variance term disappears and the $C_2/T$ term vanishes, yielding a pure $O(1/\sqrt{T})$ decay.
\item \textbf{Convex $L$‑smooth objectives.} By replacing $\|\nabla f(\mathbf{w}_t)\|^2$ with the suboptimality gap $f(\mathbf{w}_t)-f^{\star}$ in the smoothness inequality, the same analysis gives an $O(1/\sqrt{T})$ regret bound, matching the classical result for Adam/AMSGrad in convex settings.
\item \textbf{Choosing $\beta_2\to1$.} As $\beta_2\to1$, the denominator in $C_2$ grows, reflecting that a very slow adaptation of the second moment may hurt the constant but does not affect the asymptotic $1/\sqrt{T}$ rate.
\end{itemize}
