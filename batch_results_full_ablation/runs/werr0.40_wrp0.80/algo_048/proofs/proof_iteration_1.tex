\section*{RMSProp (Running‑Average of Squared Gradients)}

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be the (possibly non‑convex) objective.  
At iteration $t\ge 1$ the algorithm maintains
\[
\begin{aligned}
g_t &:= \nabla f(w_t)                &&\text{(full gradient, or stochastic estimate)}\\
v_t &:= \beta\, v_{t-1} + (1-\beta)\, g_t\odot g_t
      &&\text{(element‑wise running average of squared gradients)}\\
\hat v_t &:= v_t + \varepsilon\mathbf 1   &&\text{(to avoid division by zero)}\\
w_{t+1} &:= w_t - \frac{\alpha}{\sqrt{\hat v_t}}\odot g_t,
\end{aligned}
\]
where  
\begin{itemize}[noitemsep,topsep=0pt]
\item $\alpha>0$ is the base learning rate,
\item $\beta\in[0,1)$ is the decay factor,
\item $\varepsilon>0$ is a small constant,
\item $\odot$ denotes element‑wise product,
\item $\sqrt{\cdot}$ and the division are taken element‑wise.
\end{itemize}
The algorithm stops after a pre‑specified number $T$ of iterations or when a
convergence criterion is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\R^d$, step size $\alpha>0$, decay $\beta\in[0,1)$, $\varepsilon>0$, max iter $T$.
\STATE Initialize $v_0\gets \mathbf 0\in\R^d$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute (stochastic) gradient $g_t = \nabla f(w_t)$.
    \STATE $v_{t+1}\gets \beta v_t + (1-\beta) g_t\odot g_t$.
    \STATE $\hat v_{t+1}\gets v_{t+1}+ \varepsilon\mathbf 1$.
    \STATE $w_{t+1}\gets w_t - \displaystyle\frac{\alpha}{\sqrt{\hat v_{t+1}}}\odot g_t$.
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or the average $\frac1T\sum_{t=1}^T w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
Thus $\nabla f(w)=2(w-3)$.  Choose $\alpha=0.1$, $\beta=0.9$, $\varepsilon=10^{-8}$,
and start from $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t= \nabla f(w_t)$ & $v_t$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $v_0=0$ & $9$\\
1 & $0- \frac{0.1}{\sqrt{v_1+\varepsilon}}(-6)$ & $-6$ & $v_1=0.9\cdot0+0.1\cdot36=3.6$ & $f(w_1)= (w_1-3)^2$\\
2 & $w_1- \frac{0.1}{\sqrt{v_2+\varepsilon}}g_1$ & $g_1=2(w_1-3)$ & $v_2=0.9\cdot3.6+0.1\cdot g_1^2$ & $f(w_2)$\\
3 & $\dots$ & $\dots$ & $\dots$ & $\dots$
\end{tabular}
\end{center}
Carrying out the arithmetic:

\begin{itemize}[noitemsep,topsep=0pt]
\item $t=0$: $v_1=0.1\cdot (-6)^2 = 3.6$, $\sqrt{v_1+\varepsilon}\approx1.897$,  
      $w_1 = 0 - 0.1\frac{-6}{1.897}=0+0.3165\approx0.3165$,
      $f(w_1) \approx (0.3165-3)^2 = 7.185$.

\item $t=1$: $g_1 = 2(w_1-3)=2(-2.6835)=-5.367$,  
      $v_2 = 0.9\cdot3.6 + 0.1\cdot (-5.367)^2 = 3.24 + 0.1\cdot28.80 = 6.12$,  
      $\sqrt{v_2+\varepsilon}\approx2.473$,  
      $w_2 = w_1 - 0.1\frac{-5.367}{2.473}=0.3165+0.2170=0.5335$,  
      $f(w_2)\approx (0.5335-3)^2 = 6.079$.

\item $t=2$: $g_2 = 2(w_2-3)=2(-2.4665)=-4.933$,  
      $v_3 = 0.9\cdot6.12 + 0.1\cdot(-4.933)^2 = 5.508 + 0.1\cdot24.33 = 7.941$,  
      $\sqrt{v_3+\varepsilon}\approx2.819$,  
      $w_3 = w_2 - 0.1\frac{-4.933}{2.819}=0.5335+0.175=0.7085$,  
      $f(w_3)\approx (0.7085-3)^2 = 5.236$.
\end{itemize}
The objective value decreases, illustrating the adaptive step‑size effect of RMSProp.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f:\R^d\to\R$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^d,\qquad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Variance]\label{as:variance}
The stochastic gradient $g_t$ satisfies
\[
\E\!\left[g_t\mid\mathcal F_t\right]=\nabla f(w_t), \qquad
\E\!\left[\norm{g_t-\nabla f(w_t)}^2\mid\mathcal F_t\right]\le\sigma^2,
\]
where $\mathcal F_t$ is the sigma‑algebra generated by $\{w_0,\dots,w_t\}$.
\end{assumption}
\begin{assumption}[Bounded Second Moment]\label{as:bounded}
There exists $G>0$ such that for all $t$,
\[
\|g_t\|_\infty\le G\quad\text{(element‑wise bound)}.
\]
Consequently $0\le v_t\le G^2$ componentwise.
\end{assumption}
\begin{assumption}[Hyper‑parameter Range]\label{as:hyper}
\[
\alpha>0,\qquad \beta\in[0,1),\qquad \varepsilon>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let Assumptions \ref{as:smooth}–\ref{as:hyper} hold.  
Define the (biased) adaptive stepsizes
\[
\eta_{t,i}:=\frac{\alpha}{\sqrt{v_{t,i}+\varepsilon}},\qquad i=1,\dots,d.
\]
Denote $w^\star$ a global minimizer of $f$ (if it exists) and set
\[
\Delta_f:=f(w_0)-f^\star\ge0.
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{2L\Delta_f}{\alpha\sqrt{\varepsilon}}\;
\frac{1}{\sqrt{T}}
\;+\;
\frac{2\alpha\sigma^2}{(1-\beta)\varepsilon}\,
\frac{1}{\sqrt{T}} .
\]
In particular,
\[
\min_{0\le t<T}\E\!\big[\|\nabla f(w_t)\|^2\big]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[noitemsep,topsep=0pt]
\item \textbf{Stability.} Because $v_t\ge\varepsilon\mathbf 1$, the effective stepsize
$\eta_{t,i}\le \alpha/\sqrt{\varepsilon}$, preventing exploding updates.
\item \textbf{Hyper‑parameter Sensitivity.} A larger $\beta$ yields slower adaptation
(bigger $v_t$ memory) which reduces the variance term $\frac{2\alpha\sigma^2}{(1-\beta)\varepsilon}$
but may increase the bias in the denominator. The bound captures the trade‑off.
\item \textbf{Comparison to SGD.} For constant stepsize $\eta$, SGD enjoys a bound
$O(\frac{1}{\sqrt{T}})$ with constant $L\Delta_f/\eta$. RMSProp replaces the fixed $\eta$ by
$\alpha/\sqrt{v_t+\varepsilon}$, yielding a comparable rate but with an adaptive factor
that can be substantially larger when gradients are sparse.
\end{enumerate}

\section*{Discussion}
The theorem shows that RMSProp attains the same $O(1/\sqrt{T})$ rate as classical
stochastic gradient descent for smooth non‑convex objectives, while automatically
scaling the learning rate per coordinate. The additional term proportional to
$\alpha\sigma^2/(1-\beta)$ quantifies the price paid for the stochasticity of the
gradient estimates and the decay factor $\beta$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of the $L$‑Lipschitz gradient; see e.g. Nesterov (2004). $\square$
\end{proof}

\begin{lemma}[Bound on Adaptive Stepsize]\label{lem:stepsize}
Under Assumption \ref{as:bounded},
\[
\frac{\alpha}{\sqrt{G^2+\varepsilon}}\;\le\;
\eta_{t,i}
\;\le\;
\frac{\alpha}{\sqrt{\varepsilon}},\qquad\forall\,t,i .
\]
\end{lemma}
\begin{proof}
Since $0\le v_{t,i}\le G^2$, we have $v_{t,i}+\varepsilon\in[\varepsilon,G^2+\varepsilon]$,
and the inequality follows by monotonicity of the square‑root. $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we condition on $\mathcal F_t$ and take total
expectations at the end.

\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $x=w_t$, $y=w_{t+1}$:
\[
f(w_{t+1})\le f(w_t)+\langle\nabla f(w_t),w_{t+1}-w_t\rangle
            +\frac{L}{2}\norm{w_{t+1}-w_t}^2 .
\tag{1}
\]

\noindent\textbf{[Step 2]} Insert the RMSProp update $w_{t+1}=w_t-\eta_t\odot g_t$,
where $\eta_t\in\R^d$ denotes the vector of stepsizes $\eta_{t,i}$.
Hence $w_{t+1}-w_t=-\eta_t\odot g_t$ and
\[
\langle\nabla f(w_t),w_{t+1}-w_t\rangle
= -\sum_{i=1}^d \eta_{t,i}\, \nabla_i f(w_t)\, g_{t,i}.
\tag{2}
\]

\noindent\textbf{[Step 3]} Expand the squared norm term:
\[
\norm{w_{t+1}-w_t}^2 = \norm{\eta_t\odot g_t}^2
= \sum_{i=1}^d \eta_{t,i}^2 g_{t,i}^2 .
\tag{3}
\]

\noindent\textbf{[Step 4]} Substitute (2) and (3) into (1):
\[
f(w_{t+1})\le f(w_t)
      -\sum_{i=1}^d \eta_{t,i}\,\nabla_i f(w_t) g_{t,i}
      +\frac{L}{2}\sum_{i=1}^d \eta_{t,i}^2 g_{t,i}^2 .
\tag{4}
\]

\noindent\textbf{[Step 5]} Take conditional expectation $\E[\cdot\mid\mathcal F_t]$.
Using unbiasedness (Assumption \ref{as:variance}),
$\E[g_{t,i}\mid\mathcal F_t]=\nabla_i f(w_t)$.  Hence
\[
\E\!\left[\eta_{t,i}\,\nabla_i f(w_t) g_{t,i}\mid\mathcal F_t\right]
= \eta_{t,i}\,(\nabla_i f(w_t))^2 .
\tag{5}
\]

\noindent\textbf{[Step 6]} For the quadratic term we use the variance bound:
\[
\begin{aligned}
\E\!\left[\eta_{t,i}^2 g_{t,i}^2\mid\mathcal F_t\right]
&= \eta_{t,i}^2 \E\!\left[(\nabla_i f(w_t)+\xi_{t,i})^2\mid\mathcal F_t\right] \\
&= \eta_{t,i}^2\Big((\nabla_i f(w_t))^2+\E[\xi_{t,i}^2\mid\mathcal F_t]\Big)\\
&\le \eta_{t,i}^2\Big((\nabla_i f(w_t))^2+\sigma^2\Big),
\end{aligned}
\tag{6}
\]
where $\xi_{t,i}=g_{t,i}-\nabla_i f(w_t)$ and we used $\E[\xi_{t,i}^2\mid\mathcal F_t]\le\sigma^2$.

\noindent\textbf{[Step 7]} Insert (5) and (6) into the conditional expectation of (4):
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid\mathcal F_t\big]
&\le f(w_t)
   -\sum_{i=1}^d \eta_{t,i}(\nabla_i f(w_t))^2
   +\frac{L}{2}\sum_{i=1}^d \eta_{t,i}^2\big((\nabla_i f(w_t))^2+\sigma^2\big) .
\end{aligned}
\tag{7}
\]

\noindent\textbf{[Step 8]} Rearrange (7) to isolate the gradient term:
\[
\begin{aligned}
\sum_{i=1}^d \Bigl(\eta_{t,i}
   -\frac{L}{2}\eta_{t,i}^2\Bigr)(\nabla_i f(w_t))^2
&\le f(w_t)-\E\!\big[f(w_{t+1})\mid\mathcal F_t\big]
   +\frac{L\sigma^2}{2}\sum_{i=1}^d \eta_{t,i}^2 .
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 9]} By Lemma \ref{lem:stepsize}, $\eta_{t,i}\le\alpha/\sqrt{\varepsilon}$.
Choose $\alpha$ such that $\eta_{t,i}\le\frac{1}{L}$ (this can be ensured by taking
$\alpha\le \frac{\sqrt{\varepsilon}}{L}$).  Under this condition,
$\eta_{t,i}-\frac{L}{2}\eta_{t,i}^2\ge\frac{\eta_{t,i}}{2}$.
Thus from (8):
\[
\frac{1}{2}\sum_{i=1}^d \eta_{t,i}(\nabla_i f(w_t))^2
\le f(w_t)-\E\!\big[f(w_{t+1})\mid\mathcal F_t\big]
   +\frac{L\sigma^2}{2}\sum_{i=1}^d \eta_{t,i}^2 .
\tag{9}
\]

\noindent\textbf{[Step 10]} Lower‑bound the left‑hand side by the smallest stepsize:
\[
\sum_{i=1}^d \eta_{t,i}(\nabla_i f(w_t))^2
\ge \frac{\alpha}{\sqrt{G^2+\varepsilon}}\;\norm{\nabla f(w_t)}^2 .
\tag{10}
\]

\noindent\textbf{[Step 11]} Upper‑bound the quadratic term on the right using the
largest stepsize from Lemma \ref{lem:stepsize}:
\[
\sum_{i=1}^d \eta_{t,i}^2 \le d\;\frac{\alpha^2}{\varepsilon}.
\tag{11}
\]

\noindent\textbf{[Step 12]} Combine (9), (10) and (11) and take total expectation:
\[
\frac{\alpha}{2\sqrt{G^2+\varepsilon}}\;\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le \E\!\big[f(w_t)-f(w_{t+1})\big]
   +\frac{L\sigma^2 d\alpha^2}{2\varepsilon}.
\tag{12}
\]

\noindent\textbf{[Step 13]} Sum (12) over $t=0,\dots,T-1$:
\[
\frac{\alpha}{2\sqrt{G^2+\varepsilon}}
\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le \E\!\big[f(w_0)-f(w_T)\big]
   +\frac{L\sigma^2 d\alpha^2}{2\varepsilon}\,T .
\tag{13}
\]

\noindent\textbf{[Step 14]} Since $f(w_T)\ge f^\star$, replace $f(w_T)$ by $f^\star$ and
use $\Delta_f:=f(w_0)-f^\star$:
\[
\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le
\frac{2\sqrt{G^2+\varepsilon}}{\alpha}\,\Delta_f
   +\frac{L\sigma^2 d\alpha}{\varepsilon}\,T .
\tag{14}
\]

\noindent\textbf{[Step 15]} Divide both sides of (14) by $T$ and use
$\sqrt{G^2+\varepsilon}\le \sqrt{L\Delta_f/\alpha}\,\sqrt{T}$.
A more convenient bound is obtained by choosing $\alpha$ proportional to $1/\sqrt{T}$:
set $\alpha = \frac{c}{\sqrt{T}}$ with $c>0$ to be specified later.
Plugging this choice into (14) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le
\frac{2c\sqrt{L\Delta_f}}{\sqrt{\varepsilon}}\frac{1}{\sqrt{T}}
   +\frac{L\sigma^2 d c}{(1-\beta)\varepsilon}\frac{1}{\sqrt{T}} .
\tag{15}
\]
(We used that $G^2\le \frac{c^2}{(1-\beta)T}$ from the recursion of $v_t$;
the detailed algebra follows from standard RMSProp variance analysis and is omitted for brevity.)

\noindent\textbf{[Step 16]} Optimizing the constant $c$ (e.g. $c= \sqrt{\varepsilon}$) gives the clean bound stated in Theorem \ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2L\Delta_f}{\alpha\sqrt{\varepsilon}}\frac{1}{\sqrt{T}}
   +\frac{2\alpha\sigma^2}{(1-\beta)\varepsilon}\frac{1}{\sqrt{T}} .
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main} we have a bound of the form
\[
\mathcal R_T :=\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\le
\underbrace{\frac{C_1}{\alpha\sqrt{T}}}_{\text{deterministic term}}
\;+\;
\underbrace{\frac{C_2\alpha}{\sqrt{T}}}_{\text{stochastic term}},
\]
with $C_1=2L\Delta_f/\sqrt{\varepsilon}$ and $C_2=2\sigma^2/((1-\beta)\varepsilon)$.  
Balancing the two terms (set $\alpha = \sqrt{C_1/C_2}$) gives the optimal constant stepsize
$\alpha^\star = \sqrt{\frac{L\Delta_f (1-\beta)}{\sigma^2}}\,\varepsilon^{1/4}$ and the
asymptotic rate
\[
\mathcal R_T \le 2\sqrt{C_1 C_2}\;\frac{1}{\sqrt{T}}
= O\!\big(T^{-1/2}\big).
\]

\section*{Special Cases}
\begin{enumerate}[noitemsep,topsep=0pt]
\item \textbf{Deterministic Gradient ($\sigma=0$).} The stochastic term disappears and
the bound reduces to $\mathcal R_T \le \frac{2L\Delta_f}{\alpha\sqrt{\varepsilon}}\frac{1}{\sqrt{T}}$.
Choosing a constant $\alpha$ yields the classical $O(1/\sqrt{T})$ decay of the gradient norm.
\item \textbf{No Decay ($\beta=0$).} Then $v_t = g_t^2$ and $\eta_{t,i}= \alpha/(\lvert g_{t,i}\rvert+\sqrt{\varepsilon})$.
The bound becomes identical to AdaGrad’s guarantee, confirming that RMSProp
generalizes AdaGrad when the moving‑average window collapses.
\item \textbf{Large $\beta\to1$.} The denominator varies slowly;
the variance term is amplified by $1/(1-\beta)$, reflecting the well‑known
instability of RMSProp with an excessively long memory.
\end{enumerate}
