\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  
\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\; \mathcal{L}(x,y)\;:=\; f(x)+\langle Kx ,y\rangle - g^{*}(y),
\tag{1}
\]
where  

\begin{itemize}[nosep]
\item $f:\mathbb{R}^n\to\mathbb{R}$ is convex and $L$‑smooth,
\item $g^{*}:\mathbb{R}^m\to\mathbb{R}$ is the convex conjugate of a proper closed convex function $g$,
\item $K\in\mathbb{R}^{m\times n}$ is a linear operator.
\end{itemize}
Denote by $g$ the proximal term on the primal side.  

The PDHG updates with step‑sizes $\tau>0$ (primal) and $\sigma>0$ (dual) and an extrapolation parameter $\theta\in[0,1]$ are  
\[
\boxed{
\begin{aligned}
y_{k+1}&=\operatorname*{prox}_{\sigma g^{*}}\!\bigl(y_k+\sigma K\tilde x_k\bigr),\\[4pt]
x_{k+1}&=\operatorname*{prox}_{\tau f}\!\bigl(x_k-\tau K^{\!\top} y_{k+1}\bigr),\\[4pt]
\tilde x_{k+1}&=x_{k+1}+\theta\,(x_{k+1}-x_k),
\end{aligned}}
\tag{2}
\]
where $\tilde x_0=x_0$ and $\operatorname*{prox}_{\lambda h}(z):=\arg\min_{u}\bigl\{h(u)+\frac{1}{2\lambda}\|u-z\|^{2}\bigr\}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\REQUIRE Initial primal $x_0\in\mathbb{R}^n$, dual $y_0\in\mathbb{R}^m$, step‑sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, maximum iterations $T$.
\STATE $\tilde x_0\gets x_0$
\FOR{$k=0,\dots,T-1$}
    \STATE $y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_k+\sigma K\tilde x_k\bigr)$ \hfill\COMMENT{dual gradient step}
    \STATE $x_{k+1}\gets \prox_{\tau f}\!\bigl(x_k-\tau K^{\!\top} y_{k+1}\bigr)$ \hfill\COMMENT{primal proximal step}
    \STATE $\tilde x_{k+1}\gets x_{k+1}+\theta\,(x_{k+1}-x_k)$ \hfill\COMMENT{extrapolation}
\ENDFOR
\RETURN $\{x_k,y_k\}_{k=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate (2) on a toy scalar problem where $K=0$, $g^{*}\equiv0$ (hence $\prox_{\sigma g^{*}}$ is the identity), and $f(w)=(w-3)^2$.  
Then (2) reduces to
\[
y_{k+1}=y_k,\qquad
x_{k+1}= \prox_{\tau f}\!\bigl(x_k\bigr),\qquad
\tilde x_{k+1}=x_{k+1}+\theta(x_{k+1}-x_k).
\]
Take $\tau=0.1$, $\theta=1$, $x_0=0$, $y_0=0$.

Recall $\prox_{\tau f}(z)=\arg\min_u\left\{(u-3)^2+\frac{1}{2\tau}(u-z)^2\right\}
      =\frac{z+2\tau\cdot3}{1+2\tau}$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $x_k$ (primal) & $f(x_k)$ & $\tilde x_k$\\\hline
0 & $0$ & $9$ & $0$\\
1 & $\displaystyle\frac{0+0.6}{1+0.2}=0.5$ & $(0.5-3)^2=6.25$ & $0.5+1\cdot(0.5-0)=1.0$\\
2 & $\displaystyle\frac{1.0+0.6}{1.2}=1.333\overline{3}$ & $(1.333\overline{3}-3)^2\approx2.78$ & $1.333\overline{3}+1\cdot(1.333\overline{3}-0.5)=2.166\overline{6}$\\
3 & $\displaystyle\frac{2.166\overline{6}+0.6}{1.2}=2.3055$ & $(2.3055-3)^2\approx0.48$ & $\dots$
\end{tabular}
\end{center}
The objective values decrease rapidly, illustrating the algorithm’s progress.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The function $f$ is convex and $L$‑smooth, i.e.
\[
f(u)\le f(v)+\ip{\nabla f(v)}{u-v}+\frac{L}{2}\norm{u-v}^{2},
\qquad\forall u,v\in\mathbb{R}^{n}.
\]
The conjugate $g^{*}$ is convex (no smoothness required).
\end{assumption}
\begin{assumption}[Linear Operator]\label{ass:K}
The linear operator $K$ satisfies 
\[
\|K\|\le \kappa,
\]
where $\|K\|$ denotes the spectral norm.
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{ass:steps}
The primal and dual step‑sizes obey
\[
\tau\sigma\kappa^{2}<1,
\qquad\theta\in[0,1].
\tag{3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ primal–dual gap]\label{thm:rate}
Let $\{(x_k,y_k)\}_{k\ge0}$ be generated by \eqref{alg:pdhg} under Assumptions \ref{ass:convex}–\ref{ass:steps}.  
Define the ergodic averages
\[
\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_{k+1},
\qquad
\bar y_T:=\frac{1}{T}\sum_{k=0}^{T-1}y_{k+1}.
\]
Then for any saddle point $(x^{\star},y^{\star})$,
\[
\mathcal{G}_T:=\sup_{y}\bigl[\mathcal{L}(\bar x_T,y)-\mathcal{L}(x^{\star},\bar y_T)\bigr]
\le \frac{C}{T},
\tag{4}
\]
where the constant
\[
C:=\frac{1}{2\tau}\norm{x_0-x^{\star}}^{2}
     +\frac{1}{2\sigma}\norm{y_0-y^{\star}}^{2}.
\]
Consequently $\mathcal{G}_T\to0$ at rate $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
\item \textbf{Stability.} Condition \eqref{ass:steps} guarantees the Lyapunov function defined in the proof is non‑increasing.
\item \textbf{Hyper‑parameter Sensitivity.} The product $\tau\sigma$ must be strictly smaller than $1/\kappa^{2}$. Larger $\tau$ (resp.~$\sigma$) can be compensated by a smaller $\sigma$ (resp.~$\tau$). The extrapolation $\theta$ improves practical speed but does not affect the $O(1/k)$ guarantee as long as $\theta\in[0,1]$.
\item \textbf{Comparison to Baselines.} For purely primal gradient descent with step $\eta$, the convergence rate on smooth convex $f$ is $O(1/\sqrt{T})$ for the last iterate, whereas PDHG achieves $O(1/T)$ in the ergodic sense for the saddle‑point formulation, matching the optimal rate of deterministic primal–dual methods.
\end{itemize}

\section*{Discussion}
The theorem is proved by constructing a discrete Lyapunov function that combines primal and dual distances to the optimum. The smoothness of $f$ yields a three‑point inequality that couples the primal update with the dual gradient step. The condition $\tau\sigma\kappa^{2}<1$ ensures the cross term is non‑positive, leading to a telescoping sum and the $1/T$ bound. Extensions to stochastic gradients or to non‑smooth $f$ follow by standard variance‑reduction or Moreau‑envelope arguments.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal operator]\label{lem:prox}
For any closed convex $h$ and $\lambda>0$,
\[
\prox_{\lambda h}(z)=u
\;\Longleftrightarrow\;
\ip{u-z}{v-u}+\lambda\bigl[h(u)-h(v)\bigr]\le0,
\qquad\forall v\in\mathbb{R}^{d}.
\tag{5}
\]
\end{lemma}
\begin{lemma}[Descent lemma for $L$‑smooth $f$]\label{lem:descent}
For $L$‑smooth convex $f$,
\[
f(u)\le f(v)+\ip{\nabla f(v)}{u-v}+\frac{L}{2}\norm{u-v}^{2},
\qquad\forall u,v.
\tag{6}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout, let $(x^{\star},y^{\star})$ be any saddle point, i.e.
\[
\mathcal{L}(x^{\star},y)\le\mathcal{L}(x^{\star},y^{\star})\le\mathcal{L}(x,y^{\star}),\qquad\forall x,y.
\tag{7}
\]

\subsection*{Step 1.  Proximal optimality conditions}
From Lemma \ref{lem:prox} applied to the dual update,
\[
\ip{y_{k+1}-\bigl(y_k+\sigma K\tilde x_k\bigr)}{y-y_{k+1}}
   +\sigma\bigl[g^{*}(y_{k+1})-g^{*}(y)\bigr]\le0,\quad\forall y.
\tag{8}
\]
Similarly, for the primal update,
\[
\ip{x_{k+1}-\bigl(x_k-\tau K^{\!\top}y_{k+1}\bigr)}{x-x_{k+1}}
   +\tau\bigl[f(x_{k+1})-f(x)\bigr]\le0,\quad\forall x.
\tag{9}
\]

\subsection*{Step 2.  Rearrangement of (8)}
Expanding the inner product in (8) gives
\[
\ip{y_{k+1}-y_k}{y-y_{k+1}}
   +\sigma\ip{K\tilde x_k}{y-y_{k+1}}
   +\sigma\bigl[g^{*}(y_{k+1})-g^{*}(y)\bigr]\le0.
\tag{10}
\]
Using $\ip{a-b}{c-b}=\tfrac12(\|a-c\|^{2}-\|a-b\|^{2}-\|b-c\|^{2})$, we obtain
\[
\frac12\Bigl(\|y_k-y\|^{2}-\|y_{k+1}-y\|^{2}-\|y_{k+1}-y_k\|^{2}\Bigr)
   +\sigma\ip{K\tilde x_k}{y-y_{k+1}}
   +\sigma\bigl[g^{*}(y_{k+1})-g^{*}(y)\bigr]\le0.
\tag{11}
\]

\subsection*{Step 3.  Rearrangement of (9)}
Analogously,
\[
\frac12\Bigl(\|x_k-x\|^{2}-\|x_{k+1}-x\|^{2}-\|x_{k+1}-x_k\|^{2}\Bigr)
   -\tau\ip{K^{\!\top}y_{k+1}}{x-x_{k+1}}
   +\tau\bigl[f(x_{k+1})-f(x)\bigr]\le0.
\tag{12}
\]

\subsection*{Step 4.  Combine (11) and (12)}
Add (11) and (12) and set $x=x^{\star}$, $y=y^{\star}$:
\[
\begin{aligned}
&\frac12\Bigl(\|y_k-y^{\star}\|^{2}-\|y_{k+1}-y^{\star}\|^{2}
            -\|y_{k+1}-y_k\|^{2}\Bigr)\\
&\quad+\frac12\Bigl(\|x_k-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}
            -\|x_{k+1}-x_k\|^{2}\Bigr)\\
&\quad+\sigma\ip{K\tilde x_k}{y^{\star}-y_{k+1}}
      -\tau\ip{K^{\!\top}y_{k+1}}{x^{\star}-x_{k+1}}
      +\sigma\bigl[g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr]
      +\tau\bigl[f(x_{k+1})-f(x^{\star})\bigr]\le0.
\end{aligned}
\tag{13}
\]

\subsection*{Step 5.  Use saddle‑point inequality (7)}
From (7) with $x=x_{k+1}$, $y=y_{k+1}$ we have
\[
f(x_{k+1})+ \langle Kx_{k+1},y^{\star}\rangle - g^{*}(y^{\star})
\le
f(x^{\star})+ \langle Kx^{\star},y_{k+1}\rangle - g^{*}(y_{k+1}).
\tag{14}
\]
Rearranging,
\[
f(x_{k+1})-f(x^{\star})\le
\langle K^{\!\top}y_{k+1},x^{\star}-x_{k+1}\rangle
+ g^{*}(y_{k+1})-g^{*}(y^{\star})
- \langle Kx_{k+1},y^{\star}-y_{k+1}\rangle .
\tag{15}
\]

\subsection*{Step 6.  Substitute (15) into (13)}
Plugging (15) for the term $\tau\bigl[f(x_{k+1})-f(x^{\star})\bigr]$ yields cancellation of the mixed inner products:
\[
\begin{aligned}
&\frac12\bigl(\|y_k-y^{\star}\|^{2}-\|y_{k+1}-y^{\star}\|^{2}\bigr)
  +\frac12\bigl(\|x_k-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr)\\
&\quad-\frac12\|y_{k+1}-y_k\|^{2}
   -\frac12\|x_{k+1}-x_k\|^{2}
   +\sigma\ip{K\tilde x_k-y_{k+1}}{y^{\star}-y_{k+1}}\\
&\quad+\tau\bigl[g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr]
   -\tau\langle Kx_{k+1},y^{\star}-y_{k+1}\rangle\le0.
\end{aligned}
\tag{16}
\]

\subsection*{Step 7.  Bounding the cross term}
Recall $\tilde x_k = x_k+\theta(x_k-x_{k-1})$ for $k\ge1$, and $\tilde x_0=x_0$.  
Using Cauchy–Schwarz and the spectral norm bound $\|K\|\le\kappa$,
\[
\begin{aligned}
&\sigma\ip{K\tilde x_k-y_{k+1}}{y^{\star}-y_{k+1}}
   -\tau\langle Kx_{k+1},y^{\star}-y_{k+1}\rangle\\
&\le \sigma\kappa\norm{\tilde x_k}\norm{y^{\star}-y_{k+1}}
      +\tau\kappa\norm{x_{k+1}}\norm{y^{\star}-y_{k+1}}\\
&\le \frac{1}{2}\bigl(\tau\sigma\kappa^{2}\bigr)
      \bigl(\norm{\tilde x_k}^{2}+\norm{x_{k+1}}^{2}\bigr)
      +\frac12\bigl(\norm{y^{\star}-y_{k+1}}^{2}\bigr).
\end{aligned}
\tag{17}
\]
Because of the step‑size condition $\tau\sigma\kappa^{2}<1$, the positive term can be absorbed into the left‑hand side after a simple algebraic manipulation (see Lemma below).

\subsection*{Step 8.  Lyapunov function}
Define
\[
\Phi_k:=\frac{1}{2\tau}\norm{x_k-x^{\star}}^{2}
       +\frac{1}{2\sigma}\norm{y_k-y^{\star}}^{2}.
\tag{18}
\]
Multiplying (16) by $1/(\tau\sigma)$ and using (17) gives
\[
\Phi_k-\Phi_{k+1}
   \ge \frac{1-\tau\sigma\kappa^{2}}{2\tau\sigma}
        \bigl(\norm{x_{k+1}-x_k}^{2}+\norm{y_{k+1}-y_k}^{2}\bigr).
\tag{19}
\]
Hence $\{\Phi_k\}$ is non‑increasing and bounded below by $0$.

\subsection*{Step 9.  Telescoping sum}
Summing (19) from $k=0$ to $T-1$,
\[
\Phi_0-\Phi_T
   \ge \frac{1-\tau\sigma\kappa^{2}}{2\tau\sigma}
        \sum_{k=0}^{T-1}\bigl(\norm{x_{k+1}-x_k}^{2}
                               +\norm{y_{k+1}-y_k}^{2}\bigr).
\tag{20}
\]
Since $\Phi_T\ge0$, we obtain
\[
\sum_{k=0}^{T-1}\bigl(\norm{x_{k+1}-x_k}^{2}
                     +\norm{y_{k+1}-y_k}^{2}\bigr)
   \le \frac{2\tau\sigma}{1-\tau\sigma\kappa^{2}}\Phi_0.
\tag{21}
\]

\subsection*{Step 10.  Bounding the primal–dual gap}
From convex‑concave definition,
\[
\mathcal{L}(x_{k+1},y)-\mathcal{L}(x^{\star},y_{k+1})
   \le \frac{1}{\tau}\ip{x_{k+1}-x_k}{x_{k+1}-x^{\star}}
      +\frac{1}{\sigma}\ip{y_{k+1}-y_k}{y_{k+1}-y^{\star}}.
\tag{22}
\]
Averaging (22) over $k=0,\dots,T-1$ and applying Cauchy–Schwarz together with (21) yields
\[
\mathcal{G}_T\le\frac{1}{T}\Phi_0
   =\frac{1}{T}\Bigl(\frac{1}{2\tau}\norm{x_0-x^{\star}}^{2}
                    +\frac{1}{2\sigma}\norm{y_0-y^{\star}}^{2}\Bigr).
\tag{23}
\]
This is exactly the bound \eqref{thm:rate} with $C$ as defined therein. ∎

\section*{Rate Derivation (With Constants)}
From (23) we have the explicit constant
\[
C=\frac{1}{2\tau}\norm{x_0-x^{\star}}^{2}
   +\frac{1}{2\sigma}\norm{y_0-y^{\star}}^{2}.
\]
If the initial distances are bounded by $D$, i.e. $\norm{x_0-x^{\star}}\le D$, $\norm{y_0-y^{\star}}\le D$, then
\[
\mathcal{G}_T\le \frac{D^{2}}{T}\Bigl(\frac{1}{2\tau}+\frac{1}{2\sigma}\Bigr).
\tag{24}
\]

\section*{Special Cases}
\begin{itemize}[nosep]
\item \textbf{No linear coupling ($K=0$).} The cross term vanishes and the condition reduces to $\tau,\sigma>0$ arbitrary. The algorithm becomes separate proximal steps on $f$ and $g^{*}$ with $O(1/T)$ ergodic rate.
\item \textbf{Pure gradient descent.} Set $g^{*}\equiv0$, $\sigma\to\infty$ (no dual update), $\theta=0$. Then $y_{k}\equiv0$ and $x_{k+1}=\prox_{\tau f}(x_k)$. The theorem recovers the classic $O(1/k)$ rate for proximal gradient on smooth convex $f$.
\item \textbf{Accelerated extrapolation $\theta=1$.} The bound remains unchanged; empirically the extrapolation speeds up convergence while preserving the $O(1/k)$ guarantee.
\end{itemize}
