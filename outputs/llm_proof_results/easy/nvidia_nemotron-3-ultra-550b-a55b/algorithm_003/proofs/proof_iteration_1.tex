\section*{Algorithm Name}
Online Gradient Descent (OGD)

\section*{Mathematical Formulation}
Let $\mathcal{W} \subseteq \R^d$ be a closed convex set (the decision domain). At each round $t = 1, 2, \dots, T$:
\begin{enumerate}
    \item The learner chooses a point $w_t \in \mathcal{W}$.
    \item The environment reveals a convex loss function $f_t : \mathcal{W} \to \R$.
    \item The learner incurs loss $f_t(w_t)$ and observes a subgradient $g_t \in \partial f_t(w_t)$.
    \item The learner updates the parameter via projected gradient descent:
    \begin{equation}
        w_{t+1} = \Pi_{\mathcal{W}} \bigl( w_t - \eta_t g_t \bigr),
    \end{equation}
    where $\Pi_{\mathcal{W}}(u) = \arg\min_{w \in \mathcal{W}} \norm{w - u}_2$ is the Euclidean projection onto $\mathcal{W}$, and $\eta_t > 0$ is the learning rate (step size).
\end{enumerate}
The performance metric is \textit{regret} against any fixed comparator $u \in \mathcal{W}$:
\begin{equation}
    R_T(u) = \sum_{t=1}^T f_t(w_t) - \sum_{t=1}^T f_t(u).
\end{equation}
The goal is to achieve sublinear regret: $R_T(u) = o(T)$ for all $u \in \mathcal{W}$.

\textbf{Hyperparameters:}
\begin{itemize}
    \item Initial point $w_1 \in \mathcal{W}$ (often $w_1 = 0$ or arbitrary).
    \item Learning rate schedule $\{\eta_t\}_{t=1}^T$. Common choices:
    \begin{itemize}
        \item Fixed: $\eta_t = \eta$.
        \item Decreasing: $\eta_t = \frac{\eta}{\sqrt{t}}$.
    \end{itemize}
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}
\caption{Online Gradient Descent (OGD)}
\begin{algorithmic}[1]
\REQUIRE Convex set $\mathcal{W}$, learning rates $\{\eta_t\}_{t=1}^T$, initial $w_1 \in \mathcal{W}$.
\FOR{$t = 1$ to $T$}
    \STATE Play $w_t$, observe convex loss $f_t$, receive subgradient $g_t \in \partial f_t(w_t)$.
    \STATE Compute gradient step: $\tilde{w}_{t+1} = w_t - \eta_t g_t$.
    \STATE Project onto $\mathcal{W}$: $w_{t+1} = \Pi_{\mathcal{W}}(\tilde{w}_{t+1})$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the fixed convex function $f(w) = (w-3)^2$ (so $f_t = f$ for all $t$), unconstrained domain $\mathcal{W} = \R$, initial point $w_1 = 0$, fixed learning rate $\eta = 0.1$. The subgradient is $g_t = f'(w_t) = 2(w_t - 3)$. We run 3 iterations.

\begin{center}
\begin{tabular}{c c c c c c}
\toprule
$t$ & $w_t$ & $g_t = 2(w_t-3)$ & $\tilde{w}_{t+1} = w_t - 0.1 g_t$ & $w_{t+1}$ (no projection) & $f(w_t)$ \\
\midrule
1 & 0.0 & $2(0-3) = -6$ & $0 - 0.1(-6) = 0.6$ & 0.6 & $(0-3)^2 = 9$ \\
2 & 0.6 & $2(0.6-3) = -4.8$ & $0.6 - 0.1(-4.8) = 1.08$ & 1.08 & $(0.6-3)^2 = 5.76$ \\
3 & 1.08 & $2(1.08-3) = -3.84$ & $1.08 - 0.1(-3.84) = 1.464$ & 1.464 & $(1.08-3)^2 = 3.6864$ \\
\bottomrule
\end{tabular}
\end{center}
After 3 iterations, $w_4 = 1.464$, $f(w_4) = (1.464-3)^2 \approx 2.359$. The algorithm moves toward the minimizer $w^* = 3$.

\section*{Assumptions}
We make the following standard assumptions for the regret analysis.

\begin{assumption}[Convexity] \label{asm:convex}
Each loss function $f_t : \mathcal{W} \to \R$ is convex. Hence for any $w, u \in \mathcal{W}$ and any $g_t \in \partial f_t(w)$,
\begin{equation}
    f_t(u) \ge f_t(w) + \inner{g_t}{u - w}.
\end{equation}
\end{assumption}

\begin{assumption}[Lipschitz Continuity / Bounded Subgradients] \label{asm:lipschitz}
There exists a constant $G > 0$ such that for all $t$ and all $w \in \mathcal{W}$,
\begin{equation}
    \norm{g}_2 \le G \quad \forall g \in \partial f_t(w).
\end{equation}
Equivalently, each $f_t$ is $G$-Lipschitz w.r.t. the Euclidean norm.
\end{assumption}

\begin{assumption}[Bounded Domain] \label{asm:bounded}
The decision set $\mathcal{W}$ is convex and compact with finite diameter $D$:
\begin{equation}
    D = \max_{w, u \in \mathcal{W}} \norm{w - u}_2 < \infty.
\end{equation}
\end{assumption}

\begin{assumption}[Learning Rate] \label{asm:lr}
The learning rates $\eta_t > 0$ are non-increasing: $\eta_{t+1} \le \eta_t$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Regret Bound for OGD] \label{thm:regret}
Under Assumptions \ref{asm:convex}--\ref{asm:lr}, for any comparator $u \in \mathcal{W}$, the regret of OGD after $T$ rounds satisfies
\begin{equation}
    R_T(u) \le \frac{\norm{w_1 - u}^2}{2\eta_1} + \sum_{t=2}^T \frac{\norm{w_t - u}^2}{2} \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right) + \frac{G^2}{2} \sum_{t=1}^T \eta_t.
\end{equation}
In particular, if $\eta_t = \eta$ (fixed), then
\begin{equation}
    R_T(u) \le \frac{\norm{w_1 - u}^2}{2\eta} + \frac{\eta G^2 T}{2} \le \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}.
\end{equation}
Choosing $\eta = \frac{D}{G\sqrt{T}}$ yields
\begin{equation}
    R_T(u) \le D G \sqrt{T}.
\end{equation}
If $\eta_t = \frac{D}{G\sqrt{t}}$ (decreasing), then
\begin{equation}
    R_T(u) \le \frac{3}{2} D G \sqrt{T}.
\end{equation}
\end{theorem}

\section*{Theoretical Properties}
\begin{property}[Sublinear Regret]
The regret grows as $O(\sqrt{T})$, which is sublinear in $T$. Hence the average regret $\frac{R_T}{T} \to 0$ as $T \to \infty$.
\end{property}

\begin{property}[Optimality]
For convex Lipschitz losses, the $O(\sqrt{T})$ regret is minimax optimal (cannot be improved in worst case).
\end{property}

\begin{property}[Stability]
The projection step ensures $w_t \in \mathcal{W}$ for all $t$, guaranteeing bounded iterates even with noisy gradients.
\end{property}

\begin{property}[Hyperparameter Sensitivity]
The fixed learning rate $\eta$ requires knowledge of $T$, $D$, $G$ to optimize the bound. The decreasing schedule $\eta_t \propto 1/\sqrt{t}$ is adaptive and achieves the same rate without knowing $T$.
\end{property}

\begin{property}[Comparison to Baselines]
OGD is the foundational online learning algorithm. It matches the regret of Follow-the-Leader (FTL) for convex losses but with lower computational cost (one gradient step per round vs. full optimization). For strongly convex losses, OGD achieves $O(\log T)$ regret with appropriate learning rates.
\end{property}

\section*{Discussion}
The regret bound holds for any sequence of convex Lipschitz functions, even if chosen adversarially. The proof relies on the Pythagorean property of projections and the convexity inequality. The $O(\sqrt{T})$ rate is tight for general convex functions. If the functions are strongly convex, the rate improves to $O(\log T)$. The algorithm is also the basis for many adaptive methods (AdaGrad, Adam) which achieve data-dependent bounds.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Subgradient]
For a convex function $f$, a vector $g$ is a subgradient at $w$ if $f(u) \ge f(w) + \inner{g}{u-w}$ for all $u$.
\end{definition}

\begin{lemma}[Projection Property] \label{lem:projection}
For any closed convex set $\mathcal{W}$ and any $u \in \mathcal{W}$, $y \in \R^d$, let $x = \Pi_{\mathcal{W}}(y)$. Then
\begin{equation}
    \norm{x - u}^2 \le \norm{y - u}^2 - \norm{y - x}^2 \le \norm{y - u}^2.
\end{equation}
\end{lemma}

\begin{lemma}[Convexity Bound] \label{lem:convexity}
Under Assumption \ref{asm:convex}, for any $w_t, u \in \mathcal{W}$ and $g_t \in \partial f_t(w_t)$,
\begin{equation}
    f_t(w_t) - f_t(u) \le \inner{g_t}{w_t - u}.
\end{equation}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}

We fix an arbitrary comparator $u \in \mathcal{W}$. The proof proceeds by bounding the instantaneous regret $f_t(w_t) - f_t(u)$ using the update rule and telescoping sums.

\step{1} By Lemma \ref{lem:convexity} (convexity of $f_t$),
\begin{equation}
    f_t(w_t) - f_t(u) \le \inner{g_t}{w_t - u}. \tag{1}
\end{equation}

\step{2} Let $\tilde{w}_{t+1} = w_t - \eta_t g_t$ be the unprojected update. Then $w_{t+1} = \Pi_{\mathcal{W}}(\tilde{w}_{t+1})$. By Lemma \ref{lem:projection} (projection property) with $y = \tilde{w}_{t+1}$ and $x = w_{t+1}$,
\begin{equation}
    \norm{w_{t+1} - u}^2 \le \norm{\tilde{w}_{t+1} - u}^2. \tag{2}
\end{equation}

\step{3} Expand the squared norm of the unprojected update:
\begin{align}
    \norm{\tilde{w}_{t+1} - u}^2 &= \norm{w_t - \eta_t g_t - u}^2 \\
    &= \norm{w_t - u}^2 - 2\eta_t \inner{g_t}{w_t - u} + \eta_t^2 \norm{g_t}^2. \tag{3}
\end{align}

\step{4} Combine (2) and (3):
\begin{equation}
    \norm{w_{t+1} - u}^2 \le \norm{w_t - u}^2 - 2\eta_t \inner{g_t}{w_t - u} + \eta_t^2 \norm{g_t}^2. \tag{4}
\end{equation}

\step{5} Rearrange (4) to isolate the inner product term:
\begin{equation}
    2\eta_t \inner{g_t}{w_t - u} \le \norm{w_t - u}^2 - \norm{w_{t+1} - u}^2 + \eta_t^2 \norm{g_t}^2. \tag{5}
\end{equation}

\step{6} Divide both sides by $2\eta_t$ (since $\eta_t > 0$):
\begin{equation}
    \inner{g_t}{w_t - u} \le \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t} + \frac{\eta_t}{2} \norm{g_t}^2. \tag{6}
\end{equation}

\step{7} Substitute the convexity bound (1) into (6):
\begin{equation}
    f_t(w_t) - f_t(u) \le \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t} + \frac{\eta_t}{2} \norm{g_t}^2. \tag{7}
\end{equation}

\step{8} Apply Assumption \ref{asm:lipschitz} ($\norm{g_t} \le G$) to (7):
\begin{equation}
    f_t(w_t) - f_t(u) \le \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t} + \frac{\eta_t G^2}{2}. \tag{8}
\end{equation}

\step{9} Sum inequality (8) over $t = 1$ to $T$:
\begin{equation}
    \sum_{t=1}^T \bigl( f_t(w_t) - f_t(u) \bigr) \le \sum_{t=1}^T \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t} + \frac{G^2}{2} \sum_{t=1}^T \eta_t. \tag{9}
\end{equation}

\step{10} The first sum on the right-hand side is a telescoping sum with varying coefficients. We rewrite it by separating the $t=1$ term and shifting indices for $t \ge 2$:
\begin{align}
    \sum_{t=1}^T \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t}
    &= \frac{\norm{w_1 - u}^2}{2\eta_1} + \sum_{t=2}^T \frac{\norm{w_t - u}^2}{2\eta_t} - \sum_{t=1}^{T-1} \frac{\norm{w_{t+1} - u}^2}{2\eta_t} - \frac{\norm{w_{T+1} - u}^2}{2\eta_T} \\
    &= \frac{\norm{w_1 - u}^2}{2\eta_1} + \sum_{t=2}^T \norm{w_t - u}^2 \left( \frac{1}{2\eta_t} - \frac{1}{2\eta_{t-1}} \right) - \frac{\norm{w_{T+1} - u}^2}{2\eta_T}. \tag{10}
\end{align}

\step{11} Since $\norm{w_{T+1} - u}^2 \ge 0$ and $\eta_T > 0$, the last term is non-positive. Dropping it gives an upper bound:
\begin{equation}
    \sum_{t=1}^T \frac{\norm{w_t - u}^2 - \norm{w_{t+1} - u}^2}{2\eta_t} \le \frac{\norm{w_1 - u}^2}{2\eta_1} + \sum_{t=2}^T \frac{\norm{w_t - u}^2}{2} \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right). \tag{11}
\end{equation}

\step{12} Substitute (11) into (9) to obtain the general regret bound:
\begin{equation}
    R_T(u) \le \frac{\norm{w_1 - u}^2}{2\eta_1} + \sum_{t=2}^T \frac{\norm{w_t - u}^2}{2} \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right) + \frac{G^2}{2} \sum_{t=1}^T \eta_t. \tag{12}
\end{equation}
This proves the first statement of Theorem \ref{thm:regret}.

\section*{Rate Derivation (With Constants)}

\step{13} \textbf{Fixed learning rate $\eta_t = \eta$:} Then $\frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} = 0$ for $t \ge 2$. Using $\norm{w_t - u} \le D$ (Assumption \ref{asm:bounded}),
\begin{equation}
    R_T(u) \le \frac{\norm{w_1 - u}^2}{2\eta} + \frac{\eta G^2 T}{2} \le \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}. \tag{13}
\end{equation}

\step{14} Minimize the RHS of (13) w.r.t. $\eta$: derivative $\frac{d}{d\eta} \left( \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2} \right) = -\frac{D^2}{2\eta^2} + \frac{G^2 T}{2} = 0 \implies \eta^* = \frac{D}{G\sqrt{T}}$.
Plugging $\eta^*$ back:
\begin{equation}
    R_T(u) \le \frac{D^2}{2} \frac{G\sqrt{T}}{D} + \frac{D}{G\sqrt{T}} \frac{G^2 T}{2} = \frac{DG\sqrt{T}}{2} + \frac{DG\sqrt{T}}{2} = D G \sqrt{T}. \tag{14}
\end{equation}

\step{15} \textbf{Decreasing learning rate $\eta_t = \frac{D}{G\sqrt{t}}$:} We have $\eta_1 = \frac{D}{G}$. The difference term:
\begin{equation}
    \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} = \frac{G}{D}(\sqrt{t} - \sqrt{t-1}) = \frac{G}{D} \frac{1}{\sqrt{t} + \sqrt{t-1}} \le \frac{G}{2D\sqrt{t-1}}. \tag{15}
\end{equation}
Using $\norm{w_t - u} \le D$,
\begin{align}
    \sum_{t=2}^T \frac{\norm{w_t - u}^2}{2} \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right)
    &\le \sum_{t=2}^T \frac{D^2}{2} \cdot \frac{G}{2D\sqrt{t-1}} = \frac{DG}{4} \sum_{t=1}^{T-1} \frac{1}{\sqrt{t}}. \tag{16}
\end{align}
Bound the sum by integral: $\sum_{t=1}^{T-1} \frac{1}{\sqrt{t}} \le 1 + \int_1^{T-1} \frac{dx}{\sqrt{x}} = 1 + 2(\sqrt{T-1} - 1) \le 2\sqrt{T}$.
Thus the difference term is $\le \frac{DG}{4} \cdot 2\sqrt{T} = \frac{DG\sqrt{T}}{2}$.
The first term: $\frac{\norm{w_1-u}^2}{2\eta_1} \le \frac{D^2}{2} \frac{G}{D} = \frac{DG}{2}$.
The last term: $\frac{G^2}{2} \sum_{t=1}^T \eta_t = \frac{G^2}{2} \frac{D}{G} \sum_{t=1}^T \frac{1}{\sqrt{t}} \le \frac{DG}{2} \cdot 2\sqrt{T} = DG\sqrt{T}$.
Summing all three parts:
\begin{equation}
    R_T(u) \le \frac{DG}{2} + \frac{DG\sqrt{T}}{2} + DG\sqrt{T} \le \frac{3}{2} DG\sqrt{T} \quad (\text{for } T \ge 1). \tag{17}
\end{equation}

\section*{Special Cases}

\begin{corollary}[Unconstrained Domain with Decreasing Learning Rate]
If $\mathcal{W} = \R^d$ (no projection) and we assume $\norm{w_t - u} \le R$ for all $t$ (e.g., via bounded gradients and decreasing $\eta_t$), the same bounds hold with $D$ replaced by $R$.
\end{corollary}

\begin{corollary}[Strongly Convex Losses]
If each $f_t$ is $\alpha$-strongly convex, then using $\eta_t = \frac{1}{\alpha t}$ yields $R_T(u) \le \frac{G^2}{2\alpha} (1 + \log T)$.
\end{corollary}

\begin{corollary}[Stochastic Online Learning]
If $f_t(w) = \ell(w; z_t)$ with i.i.d. data $z_t$, then OGD with $\eta_t \propto 1/\sqrt{t}$ achieves expected excess risk $O(1/\sqrt{T})$.
\end{corollary}
