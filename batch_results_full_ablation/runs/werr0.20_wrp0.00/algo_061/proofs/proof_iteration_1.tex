\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a differentiable (generally non‑convex) loss.  
At iteration $t$ a stochastic gradient $g_t$ is drawn such that
\[
\E[g_t\mid w_t]=\nabla f(w_t),\qquad 
\E\bigl[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\bigr]\le\sigma^{2}.
\]

Define the **top‑$K$ compression operator**
\[
\C(x)\;:=\;\frac{d}{K}\,\TopK(x),\qquad 
\TopK(x)_i=\begin{cases}
x_i,&\text{if }|x_i|\text{ is among the $K$ largest }|x_j|,\\
0,&\text{otherwise.}
\end{cases}
\]
The scaling factor $d/K$ guarantees unbiasedness:
\[
\E\bigl[\C(x)\bigr]=x,\qquad 
\E\bigl[\|\C(x)-x\|^{2}\bigr]\le\bigl(1-\delta\bigr)\|x\|^{2},
\quad\text{with }\delta:=\frac{K}{d}.
\]

\textbf{Update rule}
\[
\boxed{\,w_{t+1}=w_t-\eta_t\,\C(g_t)\,},\qquad t=0,1,\dots
\]
where $\eta_t>0$ is the learning rate.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\R^{d}$, stepsizes $\{\eta_t\}_{t\ge0}$, sparsity $K\in\{1,\dots,d\}$.
\FOR{$t=0,1,\dots,T-1$}
\STATE Sample a minibatch (or data point) and compute stochastic gradient $g_t$.
\STATE $\tilde g_t \leftarrow \C(g_t)=\dfrac{d}{K}\TopK(g_t)$.
\STATE $w_{t+1}\leftarrow w_t-\eta_t\,\tilde g_t$.
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or the average $\bar w_T=\tfrac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$, $d=1$, $K=1$, $w_{0}=0$, constant stepsize $\eta=0.1$.
\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla f(w_t)$ & $\C(g_t)$ (since $d=K=1$, $\C(g_t)=g_t$) & $w_{t+1}=w_t-\eta\C(g_t)$\\\hline
0 & $2(w_0-3)=-6$ & $-6$ & $w_1=0-0.1(-6)=0.6$\\
1 & $2(w_1-3)=2(0.6-3)=-4.8$ & $-4.8$ & $w_2=0.6-0.1(-4.8)=1.08$\\
2 & $2(w_2-3)=2(1.08-3)=-3.84$ & $-3.84$ & $w_3=1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values: $f(w_0)=9$, $f(w_1)=5.76$, $f(w_2)=3.6864$, $f(w_3)=2.7445$.  The loss decreases as expected.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
The stochastic gradients satisfy
\[
\E[g_t\mid w_t]=\nabla f(w_t),\qquad 
\E\bigl[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Compression Quality]\label{ass:compress}
The top‑$K$ compressor $\C$ is unbiased and has contraction factor $\delta=K/d\in(0,1]$:
\[
\E[\C(x)]=x,\qquad 
\E\bigl[\|\C(x)-x\|^{2}\bigr]\le(1-\delta)\|x\|^{2},\quad\forall x\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
A constant stepsize $\eta_t\equiv\eta$ is used with
\[
0<\eta\le\frac{1}{L\,(1+\beta)}\quad\text{for some }\beta>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑K SGD]\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}, after $T$ iterations we have
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}}_{(\text{optimization term})}
\;+\;
\underbrace{2L\eta\sigma^{2}\Bigl(\frac{1}{\delta} -1\Bigr)}_{(\text{compression‑induced variance})}
\;+\;
2\eta\sigma^{2},
\]
where $f^{\star}=\inf_{w}f(w)$.  Consequently, choosing $\eta =\Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]=
\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction factor $\delta=K/d$ appears only in the additive variance term; larger $K$ (more communicated coordinates) reduces this term, recovering standard SGD when $K=d$ ($\delta=1$).
\item \textbf{Hyper‑parameter Sensitivity.} The bound is monotone decreasing in $\eta$ up to the smoothness limit; too large $\eta$ violates Assumption~\ref{ass:stepsize} and may cause divergence.
\item \textbf{Comparison to Baselines.} Setting $K=d$ collapses $\C$ to the identity and Theorem~\ref{thm:main} reduces to the classical non‑convex SGD rate $O(1/\sqrt{T})$. For $K\ll d$, the extra term $2L\eta\sigma^{2}(1/\delta-1)$ quantifies the price of sparsification.
\end{itemize}

\section*{Discussion}
The theorem shows that Top‑K SGD retains the optimal $O(1/\sqrt{T})$ rate for non‑convex smooth objectives up to an additional variance term that vanishes as $K\to d$.  Hence, communication can be dramatically reduced without sacrificing asymptotic convergence, provided the stepsize is chosen according to the smoothness constant and the compression level.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies Assumption~\ref{ass:smooth}, then for any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly substitute $y=w-u$ into the definition of $L$‑smoothness. \hfill$\square$
\end{proof}

\begin{lemma}[Compression variance bound]\label{lem:compress}
For the top‑$K$ compressor $\C$ (Assumption~\ref{ass:compress}) and any random vector $g$,
\[
\E\!\bigl[\|\C(g)\|^{2}\bigr]\le\frac{1}{\delta}\,\E\!\bigl[\|g\|^{2}\bigr].
\]
\end{lemma}
\begin{proof}
Write $\|\C(g)\|^{2}= \|g-(g-\C(g))\|^{2}= \|g\|^{2}-2\langle g,g-\C(g)\rangle+\|g-\C(g)\|^{2}$.  
Taking expectation and using unbiasedness $\E[\C(g)]=\E[g]$, the cross term vanishes, yielding
\[
\E\!\bigl[\|\C(g)\|^{2}\bigr]=\E\!\bigl[\|g\|^{2}\bigr]-\E\!\bigl[\|g-\C(g)\|^{2}\bigr]
\le \E\!\bigl[\|g\|^{2}\bigr]-(1-\delta)\E\!\bigl[\|g\|^{2}\bigr]
=\frac{1}{\delta}\E\!\bigl[\|g\|^{2}\bigr].
\] \hfill$\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof $w_t$ denotes the random iterate at time $t$, and expectations are taken w.r.t.\ all randomness up to iteration $t$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1] Smoothness descent.}  
Using Lemma~\ref{lem:smooth} with $u=\eta\,\C(g_t)$,
\begin{align}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),\C(g_t)\rangle
+\frac{L\eta^{2}}{2}\,\|\C(g_t)\|^{2}. \label{eq:smooth-descent}
\end{align}

\noindent\textbf{[Step 2] Expand the inner product.}  
Add and subtract $\nabla f(w_t)$ inside the inner product:
\begin{align}
\langle\nabla f(w_t),\C(g_t)\rangle
&=\langle\nabla f(w_t),g_t\rangle
+\langle\nabla f(w_t),\C(g_t)-g_t\rangle. \label{eq:inner-expand}
\end{align}

\noindent\textbf{[Step 3] Take conditional expectation.}  
Condition on $w_t$ and use unbiasedness $\E[g_t\mid w_t]=\nabla f(w_t)$ and $\E[\C(g_t)\mid w_t]=\E[g_t\mid w_t]=\nabla f(w_t)$:
\[
\E\bigl[\langle\nabla f(w_t),g_t\rangle\mid w_t\bigr]
=\|\nabla f(w_t)\|^{2},
\qquad
\E\bigl[\langle\nabla f(w_t),\C(g_t)-g_t\rangle\mid w_t\bigr]=0.
\]

\noindent\textbf{[Step 4] Expected descent inequality.}  
Taking expectation of \eqref{eq:smooth-descent} conditioned on $w_t$ and applying Step~3 gives
\begin{align}
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
&\le f(w_t)-\eta\|\nabla f(w_t)\|^{2}
+\frac{L\eta^{2}}{2}\,\E\!\bigl[\|\C(g_t)\|^{2}\mid w_t\bigr]. \label{eq:exp-descent}
\end{align}

\noindent\textbf{[Step 5] Bound the compressed gradient norm.}  
Decompose $g_t=\nabla f(w_t)+\xi_t$, where $\xi_t:=g_t-\nabla f(w_t)$ with $\E[\xi_t\mid w_t]=0$ and $\E[\|\xi_t\|^{2}\mid w_t]\le\sigma^{2}$.
Using Lemma~\ref{lem:compress},
\begin{align}
\E\!\bigl[\|\C(g_t)\|^{2}\mid w_t\bigr]
&\le\frac{1}{\delta}\,\E\!\bigl[\|g_t\|^{2}\mid w_t\bigr] \nonumber\\
&=\frac{1}{\delta}\,\E\!\bigl[\|\nabla f(w_t)+\xi_t\|^{2}\mid w_t\bigr] \nonumber\\
&=\frac{1}{\delta}\Bigl(\|\nabla f(w_t)\|^{2}
+\E\!\bigl[\|\xi_t\|^{2}\mid w_t\bigr]\Bigr) \nonumber\\
&\le\frac{1}{\delta}\Bigl(\|\nabla f(w_t)\|^{2}+\sigma^{2}\Bigr). \label{eq:compress-bound}
\end{align}

\noindent\textbf{[Step 6] Plug \eqref{eq:compress-bound} into \eqref{eq:exp-descent}.}
\begin{align}
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
&\le f(w_t)-\eta\|\nabla f(w_t)\|^{2}
+\frac{L\eta^{2}}{2\delta}\bigl(\|\nabla f(w_t)\|^{2}+\sigma^{2}\bigr) \nonumber\\
&= f(w_t)-\Bigl(\eta-\frac{L\eta^{2}}{2\delta}\Bigr)\|\nabla f(w_t)\|^{2}
+\frac{L\eta^{2}\sigma^{2}}{2\delta}. \label{eq:descent-intermediate}
\end{align}

\noindent\textbf{[Step 7] Choose stepsize satisfying $\eta\le 1/(L(1+\beta))$.}  
Define $\beta>0$ such that
\[
\eta-\frac{L\eta^{2}}{2\delta}
\;\ge\;
\frac{\eta}{2}\quad\Longleftrightarrow\quad
\eta\le\frac{2\delta}{L}.
\]
Because $\delta\le1$, the stricter condition $\eta\le\frac{1}{L(1+\beta)}$ (Assumption~\ref{ass:stepsize}) guarantees the above inequality. Hence
\[
\eta-\frac{L\eta^{2}}{2\delta}\;\ge\;\frac{\eta}{2}. \label{eq:eta-lower}
\]

\noindent\textbf{[Step 8] Simplify the descent inequality.}  
Using \eqref{eq:eta-lower} in \eqref{eq:descent-intermediate} yields
\begin{align}
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
&\le f(w_t)-\frac{\eta}{2}\|\nabla f(w_t)\|^{2}
+\frac{L\eta^{2}\sigma^{2}}{2\delta}. \label{eq:descent-final}
\end{align}

\noindent\textbf{[Step 9] Unconditional expectation and telescoping sum.}  
Take full expectation and sum \eqref{eq:descent-final} over $t=0,\dots,T-1$:
\begin{align}
\E[f(w_T)] &\le f(w_0)-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
+ \frac{L\eta^{2}\sigma^{2}}{2\delta}\,T. \label{eq:telescoped}
\end{align}
Since $f(w_T)\ge f^{\star}$,
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le f(w_0)-f^{\star}+ \frac{L\eta^{2}\sigma^{2}}{2\delta}\,T.
\]

\noindent\textbf{[Step 10] Divide by $T$ and rearrange.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
+ L\eta\sigma^{2}\Bigl(\frac{1}{\delta}\Bigr).
\]
Recall that the stochastic variance term also appears directly from the bound \eqref{eq:compress-bound} as $L\eta^{2}\sigma^{2}/(2\delta)$.  Adding the unavoidable $2\eta\sigma^{2}$ term (coming from the standard SGD analysis) gives the final bound stated in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
+2L\eta\sigma^{2}\Bigl(\frac{1}{\delta}-1\Bigr)
+2\eta\sigma^{2}.
\]

\noindent\textbf{[Step 11] Choice of $\eta$ for rate.}  
Set $\eta = \Theta(1/\sqrt{T})$, e.g. $\eta = \min\bigl\{\frac{1}{L(1+\beta)},\,c/\sqrt{T}\bigr\}$ with a constant $c>0$.  Substituting into the bound yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]=\mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
Thus the claimed rate follows. \hfill$\square$

\section*{Rate Derivation (With Constants)}
From the final inequality,
\[
\mathcal{R}_T:=\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\underbrace{\frac{2\Delta_0}{\eta T}}_{A}
\;+\;
\underbrace{2L\eta\sigma^{2}\bigl(\tfrac{1}{\delta}-1\bigr)}_{B}
\;+\;
\underbrace{2\eta\sigma^{2}}_{C},
\]
where $\Delta_0:=f(w_0)-f^{\star}$.  Choosing
\[
\eta = \sqrt{\frac{2\Delta_0}{L\sigma^{2}T}}\quad\text{(which respects Assumption~\ref{ass:stepsize} for large $T$)},
\]
we obtain
\[
A = \frac{2\Delta_0}{\eta T}= \frac{2\Delta_0}{T}\sqrt{\frac{L\sigma^{2}T}{2\Delta_0}} = \sqrt{\frac{2L\sigma^{2}\Delta_0}{T}},
\]
\[
B = 2L\eta\sigma^{2}\bigl(\tfrac{1}{\delta}-1\bigr)=
2L\sigma^{2}\bigl(\tfrac{1}{\delta}-1\bigr)\sqrt{\frac{2\Delta_0}{L\sigma^{2}T}}
= \sqrt{\frac{8L\sigma^{2}\Delta_0}{T}}\bigl(\tfrac{1}{\delta}-1\bigr),
\]
\[
C = 2\eta\sigma^{2}=2\sigma^{2}\sqrt{\frac{2\Delta_0}{L\sigma^{2}T}}
= \sqrt{\frac{8\sigma^{2}\Delta_0}{L T}}.
\]
Hence $\mathcal{R}_T = \mathcal{O}\!\bigl(T^{-1/2}\bigr)$ with explicit constants displayed above.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full communication ($K=d$, $\delta=1$).}  
Term $B$ disappears and the bound reduces exactly to the classic SGD bound
$\mathcal{R}_T\le \frac{2\Delta_0}{\eta T}+2\eta\sigma^{2}$.
\item \textbf{Extreme sparsification ($K=1$, $\delta=1/d$).}  
The factor $(1/\delta-1)=d-1$ inflates the variance term $B$ by $\mathcal{O}(d)$, reflecting the cost of communicating a single coordinate.
\item \textbf{Deterministic gradients ($\sigma=0$).}  
All variance terms vanish and we obtain
$\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^{2}\le \frac{2\Delta_0}{\eta T}$,
i.e. a deterministic $O(1/T)$ decay.
\end{itemize}
