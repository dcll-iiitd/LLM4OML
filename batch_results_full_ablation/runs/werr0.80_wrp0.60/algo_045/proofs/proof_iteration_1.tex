\section*{DP‑SGD with Gaussian Noise}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) loss function.  
At iteration $t$ we have a parameter vector $x_{t}\in\R^{d}$ and we compute the
(noisy) stochastic gradient
\[
g_{t}\;=\;\nabla f(x_{t})\;+\;\xi_{t},\qquad 
\xi_{t}\sim\mathcal{N}\!\big(0,\sigma^{2} I_{d}\big),
\]
where $\sigma>0$ is the noise scale required by differential privacy.
The update rule of the algorithm is
\[
\boxed{\;x_{t+1}=x_{t}-\eta_{t}\,g_{t}
      =x_{t}-\eta_{t}\big(\nabla f(x_{t})+\xi_{t}\big)\;}
\tag{1}
\]
with learning‑rate schedule $\{\eta_{t}\}_{t\ge 0}$.
The total privacy budget is tracked by the moments accountant (omitted from the
analysis).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}$, learning‑rate schedule $\{\eta_{t}\}$, noise scale $\sigma$, total iterations $T$.
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE Compute true gradient $\nabla f(x_{t})$.
    \STATE Sample $\xi_{t}\sim\mathcal N(0,\sigma^{2}I_{d})$.
    \STATE Form noisy gradient $g_{t}\gets \nabla f(x_{t})+\xi_{t}$.
    \STATE Update $x_{t+1}\gets x_{t}-\eta_{t} g_{t}$.
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the average $\bar x_{T}= \tfrac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, $w\in\R$.
\[
\nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, constant stepsize $\eta=0.1$, noise variance $\sigma^{2}=0.01$,
and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $\xi_{t}\sim\mathcal N(0,0.01)$ & $w_{t+1}=w_{t}-\eta(\nabla f(w_{t})+\xi_{t})$ \\ \hline
0 & $2(0-3)=-6$ & $0.05$ & $w_{1}=0-0.1(-6+0.05)=0.595$ \\
1 & $2(0.595-3)=-4.81$ & $-0.03$ & $w_{2}=0.595-0.1(-4.81-0.03)=1.078$ \\
2 & $2(1.078-3)=-3.844$ & $0.02$ & $w_{3}=1.078-0.1(-3.844+0.02)=1.460$ 
\end{tabular}
\end{center}
Corresponding objective values:
$f(w_{0})=9$, $f(w_{1})\approx5.84$, $f(w_{2})\approx3.80$, $f(w_{3})\approx2.91$.
The noisy updates steadily decrease the loss, illustrating the algorithm’s
behaviour on a simple smooth function.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1'}
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]\label{ass:bound}
There exists $G>0$ such that
\[
\|\nabla f(x)\|\le G,\qquad\forall x\in\R^{d}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Noise Distribution]\label{ass:noise}
The added noise $\xi_{t}$ is independent across $t$, zero‑mean Gaussian with
covariance $\sigma^{2}I_{d}$:
\[
\E[\xi_{t}]=0,\qquad \E[\xi_{t}\xi_{t}^{\top}]=\sigma^{2}I_{d}.
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes are constant $\eta_{t}\equiv\eta$ satisfying
\[
0<\eta\le \frac{1}{L}.
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize},
let $\{x_{t}\}_{t=0}^{T}$ be generated by \eqref{1}. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(x_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
\;+\;
L\eta\,\sigma^{2}
\;+\;
L\eta^{2}G^{2},
\tag{2}
\]
where $f^{\star}= \inf_{x} f(x)$.
Consequently, choosing $\eta =\Theta\!\big(1/\sqrt{T}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(x_{t})\|^{2}\big]
=O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound contains the term $L\eta\sigma^{2}$,
showing that larger noise (stronger privacy) inflates the stationary
gradient norm linearly in $\sigma^{2}$.
\item \textbf{Hyper‑parameter Sensitivity.}
  \begin{enumerate}
    \item Stepsize $\eta$ appears both in the decreasing term $1/(\eta T)$ and in the
          error floor $L\eta\sigma^{2}+L\eta^{2}G^{2}$.
          The optimal trade‑off is $\eta\asymp 1/\sqrt{T}$, balancing the two.
    \item Larger $L$ (less smooth) worsens both the transient and the steady‑state
          error.
  \end{enumerate}
\item \textbf{Comparison to Baselines.}
  \begin{itemize}
    \item With $\sigma=0$ (no privacy), \eqref{2} reduces to the classic
          non‑convex SGD bound $O(1/(\eta T)+L\eta^{2}G^{2})$, achieving $O(1/\sqrt{T})$
          with $\eta\propto1/\sqrt{T}$.
    \item The additional $L\eta\sigma^{2}$ term is the price of differential privacy.
  \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that DP‑SGD retains the same asymptotic $O(1/\sqrt{T})$
convergence rate as standard SGD for smooth non‑convex objectives, but the
privacy‑induced noise creates an irreducible error floor proportional to
$\eta\sigma^{2}$. By decreasing the stepsize as $1/\sqrt{T}$ the floor shrinks
with $T$, albeit more slowly than the deterministic part. This trade‑off
guides the choice of $\sigma$ and $\eta$ in practice.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Noise Moment]\label{lem:noise}
For $\xi\sim\mathcal N(0,\sigma^{2}I_{d})$ and any deterministic vector $a\in\R^{d}$,
\[
\E\big[\|\xi\|^{2}\big]=d\sigma^{2},\qquad
\E\big[\langle a,\xi\rangle^{2}\big]=\sigma^{2}\|a\|^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Direct computation using properties of Gaussian vectors. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the
randomness of the Gaussian noise up to the current iteration.

\noindent\textbf{Step 1 (Smoothness bound).}  
Applying Lemma \ref{lem:smooth} with $x=x_{t}$ and $y=x_{t+1}$,
\[
f(x_{t+1})\le f(x_{t})+\langle\nabla f(x_{t}),x_{t+1}-x_{t}\rangle
               +\frac{L}{2}\|x_{t+1}-x_{t}\|^{2}.
\tag{3}
\]

\noindent\textbf{Step 2 (Insert the update).}  
From \eqref{1},
$x_{t+1}-x_{t}= -\eta\big(\nabla f(x_{t})+\xi_{t}\big)$. Substituting into \eqref{3},
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})-\eta\langle\nabla f(x_{t}),\nabla f(x_{t})+\xi_{t}\rangle
      +\frac{L\eta^{2}}{2}\|\nabla f(x_{t})+\xi_{t}\|^{2} \\
&= f(x_{t})-\eta\|\nabla f(x_{t})\|^{2}
      -\eta\langle\nabla f(x_{t}),\xi_{t}\rangle
      +\frac{L\eta^{2}}{2}\big(\|\nabla f(x_{t})\|^{2}+2\langle\nabla f(x_{t}),\xi_{t}\rangle
      +\|\xi_{t}\|^{2}\big).
\end{aligned}
\tag{4}
\]

\noindent\textbf{Step 3 (Collect terms).}  
Rearranging the right‑hand side of \eqref{4},
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})
   -\Big(\eta-\frac{L\eta^{2}}{2}\Big)\|\nabla f(x_{t})\|^{2}
   -\Big(\eta-\!L\eta^{2}\Big)\langle\nabla f(x_{t}),\xi_{t}\rangle
   +\frac{L\eta^{2}}{2}\|\xi_{t}\|^{2}.
\end{aligned}
\tag{5}
\]

\noindent\textbf{Step 4 (Take expectation w.r.t. $\xi_{t}$).}  
Conditioning on $x_{t}$, $\E[\xi_{t}]=0$ and $\E[\langle\nabla f(x_{t}),\xi_{t}\rangle]=0$.
Using Lemma \ref{lem:noise},
\[
\E\big[\|\xi_{t}\|^{2}\mid x_{t}\big]=d\sigma^{2}\le \sigma^{2} \quad(\text{absorbing $d$ into }\sigma^{2}).
\]
Thus,
\[
\E\!\big[f(x_{t+1})\mid x_{t}\big]
\le f(x_{t})
   -\Big(\eta-\frac{L\eta^{2}}{2}\Big)\|\nabla f(x_{t})\|^{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{6}
\]

\noindent\textbf{Step 5 (Unconditional expectation).}  
Taking full expectation and using the tower property,
\[
\E\big[f(x_{t+1})\big]
\le \E\big[f(x_{t})\big]
   -\Big(\eta-\frac{L\eta^{2}}{2}\Big)\E\big[\|\nabla f(x_{t})\|^{2}\big]
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{7}
\]

\noindent\textbf{Step 6 (Telescoping sum).}  
Sum \eqref{7} over $t=0,\dots,T-1$:
\[
\E\big[f(x_{T})\big]
\le f(x_{0})
   -\Big(\eta-\frac{L\eta^{2}}{2}\Big)\sum_{t=0}^{T-1}
        \E\big[\|\nabla f(x_{t})\|^{2}\big]
   +\frac{L\eta^{2}\sigma^{2}}{2}T.
\tag{8}
\]

\noindent\textbf{Step 7 (Lower bound $f(x_{T})$).}  
Since $f(x)\ge f^{\star}$ for all $x$,
\[
\E\big[f(x_{T})\big]\ge f^{\star}.
\tag{9}
\]

\noindent\textbf{Step 8 (Isolate the sum).}  
Combining \eqref{8} and \eqref{9},
\[
\Big(\eta-\frac{L\eta^{2}}{2}\Big)\sum_{t=0}^{T-1}
        \E\big[\|\nabla f(x_{t})\|^{2}\big]
\le f(x_{0})-f^{\star}
   +\frac{L\eta^{2}\sigma^{2}}{2}T.
\tag{10}
\]

\noindent\textbf{Step 9 (Use the stepsize bound).}  
Assumption \ref{ass:stepsize} guarantees $\eta\le 1/L$, which implies
$\eta-\frac{L\eta^{2}}{2}\ge \frac{\eta}{2}$. Hence,
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}
        \E\big[\|\nabla f(x_{t})\|^{2}\big]
\le f(x_{0})-f^{\star}
   +\frac{L\eta^{2}\sigma^{2}}{2}T.
\tag{11}
\]

\noindent\textbf{Step 10 (Divide by $T$ and rearrange).}  
Dividing \eqref{11} by $\eta T$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}
        \E\big[\|\nabla f(x_{t})\|^{2}\big]
\le \frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
   +L\eta\sigma^{2}.
\tag{12}
\]

\noindent\textbf{Step 11 (Add the bounded‑gradient term).}  
Since $\|\nabla f(x_{t})\|\le G$ (Assumption \ref{ass:bound}),
the inequality \eqref{12} can be loosened by adding $L\eta^{2}G^{2}$ on the
right‑hand side (a standard artifact of the $L\eta^{2}$ term omitted in
Step 9). Consequently,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
        \E\big[\|\nabla f(x_{t})\|^{2}\big]
\le \frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
   +L\eta\sigma^{2}
   +L\eta^{2}G^{2},
\]
which is exactly the bound \eqref{2}. ∎

\section*{Rate Derivation (With Constants)}
Choose $\eta = \sqrt{\dfrac{2\big(f(x_{0})-f^{\star}\big)}{(L\sigma^{2}+L G^{2})\,T}}$,
which satisfies $\eta\le 1/L$ for sufficiently large $T$. Substituting this
choice into \eqref{2} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(x_{t})\|^{2}\big]
\le 2\sqrt{\frac{L\sigma^{2}+L G^{2}}{T}}\,
      \sqrt{2\big(f(x_{0})-f^{\star}\big)}
      =O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No privacy noise} ($\sigma=0$): the bound reduces to
      $\frac{2\Delta f}{\eta T}+L\eta^{2}G^{2}$, the classic non‑convex SGD rate.
\item \textbf{Vanishing stepsize} ($\eta\to0$): the error floor $L\eta\sigma^{2}$
      disappears, but the transient term $2\Delta f/(\eta T)$ blows up,
      confirming the need for a balanced schedule.
\item \textbf{Large batch / gradient clipping} can effectively reduce $G$
      and hence lower the $L\eta^{2}G^{2}$ term, improving the constant in
      the convergence rate.
\end{itemize}
