\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.  
At iteration $t\in\{0,1,\dots,T-1\}$ we draw a random sample $z_{t}$ and form an unbiased stochastic gradient
\[
g_{t}\;:=\;\nabla f(w_{t};z_{t})\,,\qquad 
\E\!\left[g_{t}\mid w_{t}\right]=\nabla F(w_{t})\,,\quad 
F(w):=\E_{z}[f(w;z)] .
\]

The learning rate follows a *cyclic* schedule of period $M\in\mathbb{N}$:
\[
\eta_{t}\;=\;\eta_{\min}+(\eta_{\max}-\eta_{\min})\,
\phi\!\left(\frac{t\bmod M}{M-1}\right),\qquad 
\phi:[0,1]\rightarrow[0,1]
\]
where $\phi$ is a deterministic shape function (e.g. the triangular wave
$\phi(u)=1-2|u-0.5|$).  Consequently $\eta_{t}\in[\eta_{\min},\eta_{\max}]$ for every $t$.

The update rule is
\[
\boxed{\,w_{t+1}=w_{t}-\eta_{t}g_{t}\,}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\Require Initial point $w_{0}\in\R^{d}$, period $M$, bounds $0<\eta_{\min}\le\eta_{\max}$, shape $\phi(\cdot)$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $z_{t}\sim\mathcal{D}$ and compute $g_{t}=\nabla f(w_{t};z_{t})$
    \State Compute cyclic step size
    \[
    \eta_{t}= \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
    \phi\!\Big(\frac{t\bmod M}{M-1}\Big)
    \]
    \State Update $w_{t+1}=w_{t}-\eta_{t}g_{t}$
\EndFor
\State \textbf{return} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus $F(w)=f(w)$ and $\nabla F(w)=2(w-3)$.  
Set $w_{0}=0$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$, $M=4$, and use a triangular shape $\phi(u)=1-2|u-0.5|$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $t\bmod M$ & $\eta_{t}$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta_{t}g_{t}$\\\hline
0 & 0 & $0.05$ & $2(0-3)=-6$ & $0-0.05(-6)=0.30$\\
1 & 1 & $0.10$ & $2(0.30-3)=-5.40$ & $0.30-0.10(-5.40)=0.84$\\
2 & 2 & $0.15$ & $2(0.84-3)=-4.32$ & $0.84-0.15(-4.32)=1.50$\\
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.30)=7.29$, $f(0.84)=4.66$, $f(1.50)=2.25$, showing a monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$F$ is $L$‑smooth: for all $x,y\in\R^{d}$
\[
F(y)\le F(x)+\inner{\nabla F(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $t$, $\E[g_{t}\mid w_{t}]=\nabla F(w_{t})$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\!\left[\norm{g_{t}-\nabla F(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2},\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Cyclic Step‑Size Bounds]\label{as:step}
The learning rates satisfy
\[
0<\eta_{\min}\le\eta_{t}\le\eta_{\max}\qquad\forall t,
\]
and the period $M$ is a fixed positive integer.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Under Assumptions~\ref{as:smooth}–\ref{as:step}, let
\[
\bar{\eta}:=\frac{1}{M}\sum_{k=0}^{M-1}\eta_{k}
          =\frac{\eta_{\min}+\eta_{\max}}{2}
\]
(the average step size over one cycle).  
For any $T\ge1$, the iterates generated by Cyclic SGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla F(w_{t})}^{2}\right]
\;\le\;
\frac{2\big(F(w_{0})-F^{*}\big)}{\bar{\eta}\,T}
\;+\;
L\,\eta_{\max}\,\sigma^{2},
\tag{1}
\]
where $F^{*}:=\inf_{w}F(w)$. Consequently,
\[
\min_{0\le t<T}\E\!\left[\norm{\nabla F(w_{t})}^{2}\right]
=O\!\left(\frac{1}{T}\right)+O\!\left(\eta_{\max}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because every $\eta_{t}$ is bounded by $\eta_{\max}<\frac{1}{L}$, the smoothness inequality guarantees that the sequence $\{F(w_{t})\}$ is non‑increasing in expectation.
\item \textbf{Hyper‑parameter Sensitivity.}  The bound (1) depends only on the *average* step size $\bar{\eta}$ (appearing in the $1/T$ term) and on the *maximum* step size $\eta_{\max}$ (appearing in the variance term).  Hence enlarging the amplitude $\eta_{\max}-\eta_{\min}$ improves the descent speed (larger $\bar{\eta}$) but may increase the steady‑state error through the $L\eta_{\max}\sigma^{2}$ term.
\item \textbf{Comparison to Classical SGD.}  For a constant step size $\eta$ the standard guarantee reads 
\(
\frac{1}{T}\sum_{t}\E\!\left[\norm{\nabla F(w_{t})}^{2}\right]
\le \frac{2(F(w_{0})-F^{*})}{\eta T}+L\eta\sigma^{2}.
\)
Cyclic SGD replaces $\eta$ by $\bar{\eta}$ in the first term and by $\eta_{\max}$ in the variance term, thereby preserving the optimal $O(1/T)$ decay while allowing periodic exploration via larger step sizes.
\end{itemize}

\section*{Discussion}
The theorem shows that cyclic learning rates do not deteriorate the worst‑case convergence order for smooth non‑convex problems.  The proof only uses the boundedness of the step sizes and the unbiasedness of the stochastic gradients; therefore the result holds for any deterministic shape $\phi$, including triangular, sinusoidal, or sawtooth waves.  In practice the periodic increase of $\eta_{t}$ can help escape shallow saddles, an effect that is not captured by the worst‑case bound but is compatible with the theoretical guarantee.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $F$ is $L$‑smooth, then for any $w$ and any stochastic gradient $g$,
\[
F(w-\eta g)
\le F(w)-\eta\inner{\nabla F(w)}{g}
+\frac{L\eta^{2}}{2}\norm{g}^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption~\ref{as:smooth} with $x=w$, $y=w-\eta g$ and rearrange. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof all expectations are taken with respect to the randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{Step 1 (Apply smoothness).}  
Using Lemma~\ref{lem:smooth} with $g=g_{t}$ and $\eta=\eta_{t}$,
\[
F(w_{t+1})\le F(w_{t})-\eta_{t}\inner{\nabla F(w_{t})}{g_{t}}
+\frac{L\eta_{t}^{2}}{2}\norm{g_{t}}^{2}.
\tag{2}
\]

\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $w_{t}$ and use unbiasedness (Assumption~\ref{as:unbiased}) and variance bound (Assumption~\ref{as:variance}):
\[
\begin{aligned}
\E\!\big[F(w_{t+1})\mid w_{t}\big]
&\le F(w_{t})-\eta_{t}\norm{\nabla F(w_{t})}^{2}
      +\frac{L\eta_{t}^{2}}{2}\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big] .
\end{aligned}
\tag{3}
\]

\noindent\textbf{Step 3 (Expand $\E[\|g_{t}\|^{2}]$).}  
\[
\begin{aligned}
\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]
&=\E\!\big[\norm{\nabla F(w_{t})+(g_{t}-\nabla F(w_{t}))}^{2}\mid w_{t}\big] \\
&=\norm{\nabla F(w_{t})}^{2}
  +\E\!\big[\norm{g_{t}-\nabla F(w_{t})}^{2}\mid w_{t}\big] \\
&\le \norm{\nabla F(w_{t})}^{2}+\sigma^{2}.
\end{aligned}
\tag{4}
\]

\noindent\textbf{Step 4 (Insert (4) into (3)).}
\[
\E\!\big[F(w_{t+1})\mid w_{t}\big]
\le F(w_{t})-\eta_{t}\norm{\nabla F(w_{t})}^{2}
+\frac{L\eta_{t}^{2}}{2}\big(\norm{\nabla F(w_{t})}^{2}+\sigma^{2}\big).
\tag{5}
\]

\noindent\textbf{Step 5 (Collect gradient terms).}
\[
\E\!\big[F(w_{t+1})\mid w_{t}\big]
\le F(w_{t})-\Big(\eta_{t}-\frac{L\eta_{t}^{2}}{2}\Big)\norm{\nabla F(w_{t})}^{2}
+\frac{L\eta_{t}^{2}}{2}\sigma^{2}.
\tag{6}
\]

\noindent\textbf{Step 6 (Use the bound $\eta_{t}\le \eta_{\max}<\frac{2}{L}$).}  
Since $\eta_{t}\le\eta_{\max}$ and $0<\eta_{\min}$, we have
\[
\eta_{t}-\frac{L\eta_{t}^{2}}{2}
\;\ge\;
\frac{\eta_{t}}{2}
\;\ge\;
\frac{\eta_{\min}}{2}.
\tag{7}
\]

\noindent\textbf{Step 7 (Take full expectation).}  
Taking expectation over all randomness up to time $t$ in (6) and applying (7) yields
\[
\E\!\big[F(w_{t+1})\big]
\le \E\!\big[F(w_{t})\big]
-\frac{\eta_{\min}}{2}\,\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
+\frac{L\eta_{\max}^{2}}{2}\sigma^{2}.
\tag{8}
\]

\noindent\textbf{Step 8 (Telescoping sum).}  
Sum (8) over $t=0,\dots,T-1$:
\[
\E\!\big[F(w_{T})\big]
\le F(w_{0})
-\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
+ \frac{L\eta_{\max}^{2}}{2}\sigma^{2}T .
\tag{9}
\]

\noindent\textbf{Step 9 (Rearrange).}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\le
\frac{2\big(F(w_{0})- \E[F(w_{T})]\big)}{\eta_{\min}T}
+L\eta_{\max}^{2}\sigma^{2}.
\tag{10}
\]

\noindent\textbf{Step 10 (Replace $\eta_{\min}$ by the average $\bar{\eta}$).}  
Because $\bar{\eta}= (\eta_{\min}+\eta_{\max})/2\ge \eta_{\min}/2$, we have
\[
\frac{2}{\eta_{\min}} \le \frac{4}{\eta_{\min}+\eta_{\max}} = \frac{2}{\bar{\eta}} .
\]
Insert this inequality into (10) to obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\le
\frac{2\big(F(w_{0})-F^{*}\big)}{\bar{\eta}\,T}
+L\eta_{\max}\sigma^{2},
\]
where we used $F^{*}\le \E[F(w_{T})]$.  This is exactly bound (1) of Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From (1) we directly read the asymptotic rate:
\[
\min_{0\le t<T}\E\!\big[\norm{\nabla F(w_{t})}^{2}\big]
\le
\frac{2\big(F(w_{0})-F^{*}\big)}{\bar{\eta}\,T}
+L\eta_{\max}\sigma^{2}.
\]
If we choose $\eta_{\max}=O(1/\sqrt{T})$ (while keeping $\eta_{\min}=c\,\eta_{\max}$ for a constant $c\in(0,1]$), the second term becomes $O(1/\sqrt{T})$ and the overall bound is $O(1/\sqrt{T})$, matching the optimal rate for non‑convex SGD.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Constant learning rate.}  Setting $\eta_{\min}=\eta_{\max}=\eta$ reduces the cyclic schedule to classical SGD and (1) collapses to the standard bound.
\item \textbf{Two‑phase schedule.}  If $\phi$ is a step function that spends a fraction $\alpha\in(0,1)$ of each period at $\eta_{\max}$ and $(1-\alpha)$ at $\eta_{\min}$, then $\bar{\eta}= \alpha\eta_{\max}+(1-\alpha)\eta_{\min}$.  The theorem still holds with the same proof.
\item \textbf{Vanishing amplitude.}  When $\eta_{\max}-\eta_{\min}\to0$, the cyclic scheme converges to constant‑step SGD, and the variance term $L\eta_{\max}\sigma^{2}$ shrinks accordingly.
\end{itemize}
