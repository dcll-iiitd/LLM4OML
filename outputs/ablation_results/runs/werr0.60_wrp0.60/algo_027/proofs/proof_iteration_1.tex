\section*{Randomized Coordinate Descent (RCD) for Non‑Convex Smooth Functions}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
For each coordinate $i\in\{1,\dots,d\}$ denote by $L_i>0$ the \emph{coordinate‑wise Lipschitz constant},
\[
\big| \nabla_i f(x+ \alpha e_i)-\nabla_i f(x) \big|\le L_i|\alpha|,
\qquad\forall x\in\R^d,\;\forall\alpha\in\R .
\tag{1}
\]
Here $e_i$ is the $i$‑th canonical basis vector and $\nabla_i f(x)$ the $i$‑th partial derivative.

The algorithm maintains a iterate $w_t\in\R^d$.
At iteration $t$ a coordinate $i_t$ is drawn independently from a fixed distribution
\[
\mathbb{P}(i_t=i)=p_i,\qquad p_i>0,\;\sum_{i=1}^{d}p_i=1 .
\tag{2}
\]
Only the selected coordinate is updated:
\[
w_{t+1}=w_t-\eta_i\,e_i\,\nabla_i f(w_t),\qquad 
\eta_i:=\frac{1}{L_i}\, .
\tag{3}
\]
All other coordinates remain unchanged.  

Define the \emph{weighted gradient norm}
\[
\| \nabla f(w)\|_{p^{-1}}^2:=\sum_{i=1}^{d}\frac{1}{p_i}\,\big(\nabla_i f(w)\big)^2 .
\tag{4}
\]

--------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Coordinate Descent (RCD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, probabilities $\{p_i\}_{i=1}^{d}$, coordinate Lipschitz constants $\{L_i\}_{i=1}^{d}$, maximum iterations $T$.
\State \textbf{Pre‑compute:} stepsizes $\eta_i\gets 1/L_i$.
\For{$t=0,\dots,T-1$}
    \State Sample $i_t\sim\{1,\dots,d\}$ with $\Pr(i_t=i)=p_i$.
    \State Compute partial gradient $g_{t}:=\nabla_{i_t} f(w_t)$.
    \State Update coordinate $i_t$:
    \[
    w_{t+1}=w_t-\eta_{i_t}\,e_{i_t}\,g_{t}.
    \]
\EndFor
\State \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or any $w_{\tau}$ selected uniformly at random).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional function $f(w)=(w-3)^2$.
Here $d=1$, $L_1=2$ (since $f''(w)=2$), $p_1=1$, $\eta_1=1/L_1=0.5$.
Initialize $w_0=0$, learning‑rate factor $\gamma=0.1$ (we multiply the canonical step by $\gamma$ for illustration).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)$ & Update $w_{t+1}=w_t-\gamma\eta_1\nabla f(w_t)$ & $f(w_{t+1})$\\ \hline
0 & $0$ & $-6$ & $0-0.1\cdot0.5\cdot(-6)=0.3$ & $(0.3-3)^2=7.29$\\
1 & $0.3$ & $-5.4$ & $0.3-0.1\cdot0.5\cdot(-5.4)=0.57$ & $(0.57-3)^2=5.9025$\\
2 & $0.57$ & $-4.86$ & $0.57-0.1\cdot0.5\cdot(-4.86)=0.813$ & $(0.813-3)^2=4.7849$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the effect of a single‑coordinate update.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise smoothness]\label{ass:smooth}
For each $i\in\{1,\dots,d\}$ the partial gradient $\nabla_i f(\cdot)$ is $L_i$‑Lipschitz, i.e. \eqref{1} holds.
\end{assumption}

\begin{assumption}[Bounded variance of stochastic coordinate selection]\label{ass:prob}
The sampling distribution $\{p_i\}$ satisfies $p_i>0$ for all $i$ and $\sum_i p_i=1$.
\end{assumption}

\begin{assumption}[Lower bounded objective]\label{ass:lb}
There exists $f_{\inf}>-\infty$ such that $f(w)\ge f_{\inf}$ for all $w\in\R^{d}$.
\end{assumption}

No convexity is required; $f$ may be non‑convex.

--------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:lb} and run RCD for $T$ iterations with stepsizes $\eta_i=1/L_i$.
Define the random index $t$ drawn uniformly from $\{0,\dots,T-1\}$ and let $w_t$ be the corresponding iterate.
Then
\[
\E\!\left[\big\|\nabla f(w_t)\big\|_{p^{-1}}^{\,2}\right]
\;\le\;
\frac{2\big(f(w_0)-f_{\inf}\big)}{T}\; .
\tag{5}
\]
Consequently,
\[
\min_{0\le t<T}\E\!\left[\big\|\nabla f(w_t)\big\|_{p^{-1}}^{\,2}\right]
=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\begin{remark}
The norm $\|\cdot\|_{p^{-1}}$ weights each coordinate by $1/p_i$. Hence coordinates that are sampled rarely (small $p_i$) are amplified, which is exactly the quantity that naturally appears in the expectation analysis.
\end{remark}

--------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because each update uses the exact partial gradient and a stepsize $\eta_i=1/L_i$, the descent inequality \eqref{1} guarantees that the objective never increases in expectation.
\item \textbf{Hyper‑parameter sensitivity.} The only tunable parameters are the sampling probabilities $p_i$. Choosing $p_i\propto L_i$ yields the tightest bound \eqref{5} (the constant $2$ cannot be improved without additional assumptions).
\item \textbf{Comparison to full‑gradient SGD.} For $p_i=1/d$ the bound becomes $d\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]\le 2d\,(f(w_0)-f_{\inf})/T$, i.e. the same $O(1/T)$ rate as SGD for smooth non‑convex problems but with a per‑iteration cost reduced from $O(d)$ to $O(1)$.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness descent]\label{lem:descent}
For any $w\in\R^{d}$ and any coordinate $i$,
\[
f\!\big(w - \eta_i e_i \nabla_i f(w)\big)
\le f(w)-\frac{1}{2L_i}\big(\nabla_i f(w)\big)^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Apply the definition of $L_i$–smoothness with step $\alpha=-\eta_i\nabla_i f(w)=-\frac{1}{L_i}\nabla_i f(w)$:
\[
f(w+ \alpha e_i)\le f(w)+\nabla_i f(w)\alpha+\frac{L_i}{2}\alpha^{2}
= f(w)-\frac{1}{L_i}(\nabla_i f(w))^{2}+\frac{L_i}{2}\frac{1}{L_i^{2}}(\nabla_i f(w))^{2}
= f(w)-\frac{1}{2L_i}(\nabla_i f(w))^{2}.
\]
\end{proof}

--------------------------------------------------------------------

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove \eqref{5} by telescoping the expected descent obtained from Lemma~\ref{lem:descent}.

\bigskip
\noindent\textbf{Step 1.}  Write the one‑step improvement conditioned on the current iterate $w_t$:
\[
\E\!\big[ f(w_{t+1})\mid w_t\big]
= \sum_{i=1}^{d}p_i\, f\!\big(w_t-\eta_i e_i \nabla_i f(w_t)\big).
\tag{7}
\]

\noindent\textbf{Step 2.}  Apply Lemma~\ref{lem:descent} inside the sum:
\[
f\!\big(w_t-\eta_i e_i \nabla_i f(w_t)\big)
\le f(w_t)-\frac{1}{2L_i}\big(\nabla_i f(w_t)\big)^{2}.
\tag{8}
\]

\noindent\textbf{Step 3.}  Substitute \eqref{8} into \eqref{7} and use linearity of expectation:
\[
\E\!\big[ f(w_{t+1})\mid w_t\big]
\le f(w_t)-\frac{1}{2}\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\nabla_i f(w_t)\big)^{2}.
\tag{9}
\]

\noindent\textbf{Step 4.}  Multiply both sides of \eqref{9} by $2$ and rearrange:
\[
\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\nabla_i f(w_t)\big)^{2}
\le 2\big(f(w_t)-\E[f(w_{t+1})\mid w_t]\big).
\tag{10}
\]

\noindent\textbf{Step 5.}  Take total expectation (over all randomness up to time $t$) and note that $L_i$ are constants:
\[
\E\!\left[\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\nabla_i f(w_t)\big)^{2}\right]
\le 2\,\E\!\big[ f(w_t)-f(w_{t+1})\big].
\tag{11}
\]

\noindent\textbf{Step 6.}  Sum \eqref{11} over $t=0,\dots,T-1$ and telescope:
\[
\sum_{t=0}^{T-1}\E\!\left[\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\nabla_i f(w_t)\big)^{2}\right]
\le 2\big(f(w_0)-\E[f(w_T)]\big)
\le 2\big(f(w_0)-f_{\inf}\big).
\tag{12}
\]

\noindent\textbf{Step 7.}  Exchange the order of summations in the left‑hand side:
\[
\sum_{i=1}^{d}\frac{p_i}{L_i}\,\sum_{t=0}^{T-1}\E\!\big[(\nabla_i f(w_t))^{2}\big]
\le 2\big(f(w_0)-f_{\inf}\big).
\tag{13}
\]

\noindent\textbf{Step 8.}  Define the random index $t$ uniformly on $\{0,\dots,T-1\}$, i.e.
$\Pr(t=t')=1/T$.  Then for any fixed $i$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[(\nabla_i f(w_t))^{2}\big]
= \E\!\big[(\nabla_i f(w_t))^{2}\big].
\tag{14}
\]

\noindent\textbf{Step 9.}  Divide \eqref{13} by $T$ and use \eqref{14}:
\[
\sum_{i=1}^{d}\frac{p_i}{L_i}\,\E\!\big[(\nabla_i f(w_t))^{2}\big]
\le \frac{2\big(f(w_0)-f_{\inf}\big)}{T}.
\tag{15}
\]

\noindent\textbf{Step 10.}  Recall that the stepsizes are $\eta_i=1/L_i$.  Multiply both sides of \eqref{15} by $2$ and rewrite the left side in terms of the weighted norm \eqref{4}:
\[
\E\!\left[\sum_{i=1}^{d}\frac{1}{p_i}\big(\nabla_i f(w_t)\big)^{2}\right]
= \E\!\big[\|\nabla f(w_t)\|_{p^{-1}}^{\,2}\big]
\le \frac{2\big(f(w_0)-f_{\inf}\big)}{T}.
\tag{16}
\]

\noindent\textbf{Step 11.}  Inequality \eqref{16} is exactly the claim \eqref{5} of Theorem~\ref{thm:main}.  $\square$

--------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From \eqref{16} we have an explicit constant:
\[
C:=2\big(f(w_0)-f_{\inf}\big).
\]
Thus for any $T\ge1$,
\[
\E\!\big[\|\nabla f(w_{\text{rand}})\|_{p^{-1}}^{2}\big]\le \frac{C}{T},
\]
where $w_{\text{rand}}$ is a uniformly random iterate among $\{w_0,\dots,w_{T-1}\}$.
If one is interested in the ordinary Euclidean norm, the Cauchy–Schwarz inequality together with $p_{\min}:=\min_i p_i$ yields
\[
\E\!\big[\|\nabla f(w_{\text{rand}})\|^{2}\big]
\le p_{\min}\,\E\!\big[\|\nabla f(w_{\text{rand}})\|_{p^{-1}}^{2}\big]
\le \frac{C\,p_{\min}}{T}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$.  Then $\|\cdot\|_{p^{-1}}^{2}=d\|\cdot\|^{2}$ and \eqref{5} becomes
\[
\E\!\big[\|\nabla f(w_t)\|^{2}\big]\le \frac{2d\big(f(w_0)-f_{\inf}\big)}{T}.
\]
\item \textbf{Importance sampling.} Choose $p_i = L_i/\sum_j L_j$.  Since $\frac{p_i}{L_i}=1/\sum_j L_j$ is constant,
\eqref{15} simplifies to
\[
\frac{1}{\sum_j L_j}\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]\le \frac{2\big(f(w_0)-f_{\inf}\big)}{T},
\]
which is the tightest possible bound under the present analysis.
\end{itemize}

\section*{Discussion}
The theorem shows that a simple randomized coordinate update, despite being applied to a possibly non‑convex landscape, drives the weighted gradient norm to zero at the optimal $O(1/T)$ rate.  
The weighting by $1/p_i$ is unavoidable: coordinates that are selected rarely contribute proportionally more to the expectation, guaranteeing an unbiased estimate of the full gradient in the sense
\[
\E\!\big[ \tfrac{1}{p_{i_t}} e_{i_t}\nabla_{i_t} f(w_t) \mid w_t\big]=\nabla f(w_t).
\]
Consequently, the analysis mirrors that of stochastic gradient descent, but the per‑iteration computational cost drops from $O(d)$ to $O(1)$, making RCD attractive for high‑dimensional problems where coordinate smoothness constants are known or can be estimated.
