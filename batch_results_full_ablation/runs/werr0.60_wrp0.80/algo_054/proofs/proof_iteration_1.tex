\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.  
At iteration $t\ge 0$ we draw a stochastic gradient $g_{t}$ such that  

\[
g_{t}= \nabla f(w_{t};\xi_{t}),\qquad 
\xi_{t}\sim\mathcal{D}\ \text{(data sample)} .
\]

The learning rate $\eta_{t}$ follows a \emph{cyclic schedule} of period $M\in\mathbb{N}$,
oscillating between a minimum $\eta_{\min}>0$ and a maximum $\eta_{\max}\ge\eta_{\min}$.
A common triangular schedule is  

\[
\boxed{\;
\eta_{t}= \eta_{\min}
       +(\eta_{\max}-\eta_{\min})\Bigl(1-
       \Bigl|\frac{t\bmod (2M)}{M}-1\Bigr|\Bigr)
\;}
\tag{1}
\]

so that $\eta_{t}\in[\eta_{\min},\eta_{\max}]$ and $\eta_{t+M}= \eta_{t}$.

The update rule is the standard SGD step with the time‑varying step size:

\[
\boxed{\; w_{t+1}= w_{t}-\eta_{t}\, g_{t}\; } \tag{2}
\]

Hyper‑parameters: $\eta_{\min},\eta_{\max},M$, batch size (implicit in $g_{t}$), and a stopping
criterion (maximum iterations $T$ or tolerance).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}$, $\eta_{\min},\eta_{\max},M$, total iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a minibatch $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$.
    \State Compute cyclic step size $\eta_{t}$ by \eqref{1}.
    \State Update $w_{t+1}= w_{t}-\eta_{t} g_{t}$.
\EndFor
\State \textbf{Output:} $w_{T}$ (or optionally the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1} w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^{2}$, whose gradient is $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, a deterministic gradient (i.e. $g_{t}=\nabla f(w_{t})$), $M=2$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$
so that the schedule (1) yields the sequence $\eta_{0}=0.15$, $\eta_{1}=0.05$, $\eta_{2}=0.15$, …  

\begin{tabular}{c|c|c|c}
$t$ & $\eta_{t}$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta_{t}g_{t}$ \\ \hline
0 & $0.15$ & $2(0-3)=-6$ & $w_{1}=0-0.15(-6)=0.9$ \\
1 & $0.05$ & $2(0.9-3)=-4.2$ & $w_{2}=0.9-0.05(-4.2)=1.11$ \\
2 & $0.15$ & $2(1.11-3)=-3.78$ & $w_{3}=1.11-0.15(-3.78)=1.687$ 
\end{tabular}

Corresponding objective values: $f(w_{0})=9$, $f(w_{1})=(0.9-3)^{2}=4.41$, $f(w_{2})\approx3.56$, $f(w_{3})\approx1.77$,
showing descent despite the cyclic learning rates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $t$, conditional on $w_{t}$,
\[
\E\!\big[ g_{t}\mid w_{t}\big]=\nabla f(w_{t}).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}\ge0$ such that
\[
\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2},
\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Cyclic Step Size]\label{as:cyclic}
The schedule \eqref{1} satisfies
\[
0<\eta_{\min}\le \eta_{t}\le \eta_{\max}<\frac{1}{L},
\qquad\forall t,
\]
and $\eta_{t+M}=\eta_{t}$ (periodicity).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:cyclic}, let $w_{0}$ be arbitrary and run Cyclic SGD for $T$ iterations.
Define the averaged iterate $\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$.
Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}
\;+\;
L\eta_{\max}\sigma^{2},
\tag{3}
\]
where $f^{\star}= \inf_{w}f(w)$.
Consequently,
\[
\lim_{T\to\infty}\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]=L\eta_{\max}\sigma^{2},
\]
and if $\sigma^{2}=0$ (full‑batch case) the right‑hand side decays as $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because $\eta_{\max}<1/L$, each individual step respects the standard
smoothness‑based descent condition; the cyclic modulation only changes constants.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{3} depends linearly on $\eta_{\max}$
and inversely on $\eta_{\min}$. Larger $\eta_{\max}$ accelerates early descent but inflates the
asymptotic variance term $L\eta_{\max}\sigma^{2}$; larger $\eta_{\min}$ improves the $O(1/T)$ term.
\item \textbf{Comparison to Constant‑Step SGD.} For a constant step size $\eta\in(0,1/L)$,
the classic bound is $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.  
Cyclic SGD attains the same order with $\eta$ replaced by the *worst‑case* $\eta_{\max}$ in the
variance term and by the *best‑case* $\eta_{\min}$ in the $1/T$ term, thus potentially offering
faster practical progress while preserving the same theoretical rate.
\end{itemize}

\section*{Discussion}
The theorem shows that a deterministic cyclic modulation does not worsen the
asymptotic convergence order for non‑convex smooth objectives.  The proof follows the
standard smooth‑function descent argument; the only extra ingredient is the
periodic lower and upper bounds on $\eta_{t}$, which allow us to replace $\eta_{t}$ by
$\eta_{\min}$ (for the $1/T$ term) and by $\eta_{\max}$ (for the variance term).  This
explains why practitioners observe “learning‑rate warm‑up” and “cool‑down’’ effects
without sacrificing theoretical guarantees.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any vector $d$,
\[
f(w-d)\le f(w)-\langle \nabla f(w),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Immediate from Assumption \ref{as:smooth} with $y=w-d$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  All expectations are taken with respect to the randomness
of the stochastic gradients up to the current iteration.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $d=\eta_{t}g_{t}$:
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),g_{t}\rangle
               +\frac{L\eta_{t}^{2}}{2}\|g_{t}\|^{2}.
\tag{4}
\]

\noindent\textbf{[Step 2]} Take conditional expectation w.r.t. $g_{t}$ given $w_{t}$.
Using unbiasedness (Assumption \ref{as:unbiased}):

\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]
\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta_{t}^{2}}{2}\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big].
\tag{5}
\]

\noindent\textbf{[Step 3]} Decompose $\|g_{t}\|^{2}$:
\[
\|g_{t}\|^{2}= \|\nabla f(w_{t})\|^{2}
            +\|g_{t}-\nabla f(w_{t})\|^{2}
            +2\langle\nabla f(w_{t}),g_{t}-\nabla f(w_{t})\rangle .
\]
Taking conditional expectation, the cross term vanishes (by unbiasedness), yielding
\[
\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big]
= \|\nabla f(w_{t})\|^{2}
  +\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2},
\tag{6}
\]
where the last inequality uses Assumption \ref{as:variance}.

\noindent\textbf{[Step 4]} Substitute \eqref{6} into \eqref{5}:
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]
\le f(w_{t})
   -\eta_{t}\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta_{t}^{2}}{2}\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big).
\tag{7}
\]

\noindent\textbf{[Step 5]} Rearrange \eqref{7} to isolate the gradient term:
\[
\big(\eta_{t}-\tfrac{L}{2}\eta_{t}^{2}\big)\|\nabla f(w_{t})\|^{2}
\le f(w_{t})-\E\!\big[f(w_{t+1})\mid w_{t}\big]
   +\tfrac{L}{2}\eta_{t}^{2}\sigma^{2}.
\tag{8}
\]

\noindent\textbf{[Step 6]} Because $\eta_{t}\le\eta_{\max}<1/L$, we have
$\eta_{t}-\frac{L}{2}\eta_{t}^{2}\ge\frac{\eta_{t}}{2}\ge\frac{\eta_{\min}}{2}$.
Thus,
\[
\frac{\eta_{\min}}{2}\,\|\nabla f(w_{t})\|^{2}
\le f(w_{t})-\E\!\big[f(w_{t+1})\mid w_{t}\big]
   +\frac{L}{2}\eta_{\max}^{2}\sigma^{2}.
\tag{9}
\]

\noindent\textbf{[Step 7]} Take total expectation and sum \eqref{9} over $t=0,\dots,T-1$:
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \E\!\big[f(w_{0})-f(w_{T})\big]
   +\frac{L}{2}\eta_{\max}^{2}\sigma^{2}T .
\tag{10}
\]

\noindent\textbf{[Step 8]} Since $f(w_{T})\ge f^{\star}$,
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le f(w_{0})-f^{\star}
   +\frac{L}{2}\eta_{\max}^{2}\sigma^{2}T .
\tag{11}
\]

\noindent\textbf{[Step 9]} Divide both sides of \eqref{11} by $T$ and by $\eta_{\min}/2$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}
   +\frac{L\eta_{\max}^{2}}{\eta_{\min}}\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 10]} Because $\eta_{\max}<1/L$, we may simplify the constant factor:
\[
\frac{L\eta_{\max}^{2}}{\eta_{\min}}\le L\eta_{\max},
\]
which yields the cleaner bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}
   +L\eta_{\max}\sigma^{2}.
\tag{13}
\]

Equation \eqref{13} is exactly the statement \eqref{3} in Theorem \ref{thm:main}.
Thus the theorem follows. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{13} we identify two regimes:

\begin{enumerate}
\item \textbf{Noise‑free case ($\sigma^{2}=0$).}  
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}
= O\!\left(\frac{1}{T}\right).
\]

\item \textbf{Stochastic case ($\sigma^{2}>0$).}  
The bound consists of an $O(1/T)$ term plus a steady‑state floor $L\eta_{\max}\sigma^{2}$.
Choosing $\eta_{\max}=O(1/\sqrt{T})$ (as in a diminishing cyclic schedule) drives the floor
to $O(1/\sqrt{T})$, reproducing the classical $\mathcal{O}(1/\sqrt{T})$ rate for non‑convex SGD.
\end{enumerate}

\section*{Special Cases}
\begin{itemize}
\item \textbf{Constant learning rate.} Setting $\eta_{\min}=\eta_{\max}=\eta$ reduces
\eqref{13} to the standard SGD bound $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Triangular schedule with $M=1$.} The schedule alternates between
$\eta_{\min}$ and $\eta_{\max}$ each iteration; the proof above still holds because
the bounds \eqref{Step 6} only require knowledge of the global minima and maxima.
\item \textbf{Cosine annealing.} Any periodic schedule satisfying
$\eta_{\min}\le\eta_{t}\le\eta_{\max}$ and $\eta_{\max}<1/L$ can be plugged into the same analysis.
\end{itemize}
