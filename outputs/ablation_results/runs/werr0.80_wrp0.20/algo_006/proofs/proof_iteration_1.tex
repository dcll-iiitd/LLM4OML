\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) for Non‑Convex Smooth Finite‑Sum Optimization}

\section*{Mathematical Formulation}
Let the objective be a finite sum
\[
f(\mathbf{x})\;=\;\frac{1}{n}\sum_{i=1}^{n} f_i(\mathbf{x}),\qquad 
f_i:\mathbb{R}^{d}\to\mathbb{R},
\]
where each component $f_i$ is differentiable and $L$‑smooth.  
SVRG proceeds in \emph{epochs}.  At the beginning of epoch $s$ a \emph{reference point}
\(\tilde{\mathbf{x}}^{s}\) is fixed and its full gradient
\[
\tilde{\mu}^{s}\;:=\;\nabla f(\tilde{\mathbf{x}}^{s})
          \;=\;\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{\mathbf{x}}^{s})
\tag{1}
\]
is computed once.  Inside the epoch we perform $m$ inner updates.  
Given the current iterate $\mathbf{x}_t$ we draw an index $i_t$ uniformly from $\{1,\dots ,n\}$ and form the \emph{variance‑reduced estimator}
\[
\mathbf{v}_t\;:=\;\nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{s})
                 +\tilde{\mu}^{s}.
\tag{2}
\]
The next iterate is obtained by a standard SGD step
\[
\mathbf{x}_{t+1}\;=\;\mathbf{x}_t-\eta\,\mathbf{v}_t,
\qquad t=0,\dots ,m-1,
\tag{3}
\]
with a constant stepsize $\eta>0$.  After the $m$ inner steps we set the next reference point
\[
\tilde{\mathbf{x}}^{s+1}\;:=\;\mathbf{x}_{m}.
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for non‑convex $L$‑smooth finite‑sum}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner length $m$, stepsize $\eta>0$,
         initial point $\tilde{\mathbf{x}}^{0}\in\mathbb{R}^{d}$.
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute the full gradient \(\tilde{\mu}^{s}\gets \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{\mathbf{x}}^{s})\) \Comment{cost $O(n)$}
    \State Set $\mathbf{x}_0 \gets \tilde{\mathbf{x}}^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\mathrm{Uniform}\{1,\dots ,n\}$
        \State $\mathbf{v}_t\gets \nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{s})+\tilde{\mu}^{s}$
        \State $\mathbf{x}_{t+1}\gets \mathbf{x}_t-\eta\,\mathbf{v}_t$
    \EndFor
    \State $\tilde{\mathbf{x}}^{s+1}\gets \mathbf{x}_m$
\EndFor
\State \Return $\tilde{\mathbf{x}}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
f(w)=(w-3)^2,\qquad n=1,\; f_1(w)=f(w).
\]
Here $L=2$ (since $f''(w)=2$) and the exact gradient is $\nabla f(w)=2(w-3)$.
Take $\eta=0.1$, $m=3$, and initialise $\tilde{w}^{0}=0$.

\begin{enumerate}[label=Step \arabic*:, leftmargin=*]
\item Compute full gradient at the reference point:
\[
\tilde{\mu}^{0}= \nabla f(\tilde{w}^{0}) =2(0-3)=-6.
\]
Set $w_0\gets \tilde{w}^{0}=0$.

\item \textbf{Inner iteration $t=0$} (only one component, so $i_0=1$):
\[
\mathbf{v}_0 = \nabla f_{1}(w_0)-\nabla f_{1}(\tilde{w}^{0})+\tilde{\mu}^{0}
            = (-6)-(-6)+(-6) = -6.
\]
Update:
\[
w_{1}= w_{0}-\eta \mathbf{v}_0 = 0-0.1(-6)=0.6.
\]
Objective value:
\(f(w_1)=(0.6-3)^2=5.76.\)

\item \textbf{Inner iteration $t=1$}:
\[
\mathbf{v}_1 = \nabla f_{1}(w_1)-\nabla f_{1}(\tilde{w}^{0})+\tilde{\mu}^{0}
            = 2(0.6-3)-(-6)-6 = -4.8-(-6)-6 = -4.8+6-6 = -4.8.
\]
Update:
\(w_{2}=0.6-0.1(-4.8)=1.08.\)
Objective:
\(f(w_2)=(1.08-3)^2=3.6864.\)

\item \textbf{Inner iteration $t=2$}:
\[
\mathbf{v}_2 = 2(1.08-3)-(-6)-6 = -3.84-(-6)-6 = -3.84+6-6 = -3.84.
\]
Update:
\(w_{3}=1.08-0.1(-3.84)=1.464.\)
Objective:
\(f(w_3)=(1.464-3)^2=2.366.\)

\item End of epoch: set $\tilde{w}^{1}=w_3=1.464$.
\end{enumerate}
The example shows the variance‑reduced estimator coincides with the true gradient (because $n=1$), and the iterates move toward the minimiser $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each component $f_i$ is $L$‑smooth; i.e.
\[
\|\nabla f_i(\mathbf{x})-\nabla f_i(\mathbf{y})\|\le L\|\mathbf{x}-\mathbf{y}\|,
\qquad\forall\,\mathbf{x},\mathbf{y}\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:lb}
The objective $f$ is bounded below: there exists $f^\star\in\mathbb{R}$ such that
\(f(\mathbf{x})\ge f^\star,\ \forall\mathbf{x}\).
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The constant stepsize satisfies
\[
0<\eta \le \frac{1}{3L}.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{as:finitesum}
The problem is of the finite‑sum form $f(\mathbf{x})=\frac{1}{n}\sum_{i=1}^n f_i(\mathbf{x})$ with $n\ge 1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG Convergence]\label{thm:svrg}
Let Assumptions~\ref{as:smooth}–\ref{as:stepsize} hold.  
Run SVRG for $S$ epochs with inner length $m\ge 1$ and stepsize $\eta\le 1/(3L)$.  
Define the random output
\[
\widehat{\mathbf{x}}\;\text{chosen uniformly from }\;
\bigl\{\mathbf{x}_t^{s}\mid s=0,\dots ,S-1,\;t=0,\dots ,m-1\bigr\}.
\]
Then
\[
\mathbb{E}\bigl[\|\nabla f(\widehat{\mathbf{x}})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr)}{\eta\,S\,m}.
\tag{5}
\]
Consequently, to obtain an $\varepsilon$‑stationary point
\(\mathbb{E}[\|\nabla f(\widehat{\mathbf{x}})\|^{2}]\le\varepsilon\) it suffices to run
\[
S\,m = O\!\left(\frac{1}{\eta\varepsilon}\right)
\]
inner stochastic updates, i.e. a total of $O\bigl((n+1)/\varepsilon\bigr)$ gradient evaluations
when $m=n$ (the usual choice).
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound (5) holds for any choice of $m\ge 1$; larger $m$ reduces the frequency of the expensive full‑gradient computation, while the stepsize restriction $\eta\le 1/(3L)$ guarantees that the recursion is contractive.
\item \textbf{Hyper‑parameter Sensitivity.} The convergence rate is linear in $1/(\eta S m)$.  Choosing $\eta$ close to the upper limit $1/(3L)$ minimises the constant factor, whereas too small $\eta$ slows progress proportionally.
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item \emph{Plain SGD.} For non‑convex $L$‑smooth objectives SGD with stepsize $\eta_t=O(1/\sqrt{T})$ yields $\mathbb{E}[\|\nabla f(\bar{\mathbf{x}}_T)\|^{2}]=O(1/\sqrt{T})$, i.e. sub‑linear in $T$.
\item \emph{SVRG.} The bound (5) gives an $O(1/T)$ rate (with $T=S m$ total stochastic steps) under the same smoothness assumptions, thus improving the dependence on $T$ by a factor $\sqrt{T}$.
\end{itemize}
\item \textbf{Special Cases.}
\begin{itemize}
\item If each $f_i$ is convex, the same proof yields a bound on the function sub‑optimality $f(\widehat{\mathbf{x}})-f^\star$.
\item If $m=n$ (one inner epoch per full pass), the total gradient‑evaluation complexity matches the optimal $O\bigl((n+\frac{1}{\varepsilon})\bigr)$ for finding an $\varepsilon$‑stationary point.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that the variance‑reduced estimator (2) is unbiased (Lemma~\ref{lem:unbiased}) and its second moment can be controlled by the smoothness of the components (Lemma~\ref{lem:secondmoment}).  Combining these properties with the standard smoothness descent inequality yields a clean telescoping sum that directly leads to the $O(1/T)$ gradient‑norm rate.  The analysis does not rely on convexity; only $L$‑smoothness and a lower bound are needed, which makes the result applicable to a broad class of deep‑learning loss functions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the SVRG estimator]\label{lem:unbiased}
For any $\mathbf{x}_t$ and reference $\tilde{\mathbf{x}}^{s}$,
\[
\mathbb{E}_{i_t}\bigl[\mathbf{v}_t\mid \mathbf{x}_t,\tilde{\mathbf{x}}^{s}\bigr]
    =\nabla f(\mathbf{x}_t).
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:unbiased}]
\[
\begin{aligned}
\mathbb{E}_{i_t}[\mathbf{v}_t]
&= \frac{1}{n}\sum_{i=1}^{n}\bigl(\nabla f_i(\mathbf{x}_t)-\nabla f_i(\tilde{\mathbf{x}}^{s})\bigr)+\tilde{\mu}^{s} \\
&= \bigl(\nabla f(\mathbf{x}_t)-\nabla f(\tilde{\mathbf{x}}^{s})\bigr)+\nabla f(\tilde{\mathbf{x}}^{s})
 = \nabla f(\mathbf{x}_t).
\end{aligned}
\]
\end{proof}

\begin{lemma}[Second‑moment bound]\label{lem:secondmoment}
Under $L$‑smoothness,
\[
\mathbb{E}_{i_t}\!\bigl[\|\mathbf{v}_t\|^{2}\mid \mathbf{x}_t,\tilde{\mathbf{x}}^{s}\bigr]
\;\le\;
4L\bigl(f(\mathbf{x}_t)-f^\star\bigr)+4L\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{6}
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:secondmoment}]
Using $\mathbf{v}_t = \nabla f_{i_t}(\mathbf{x}_t)-\nabla f_{i_t}(\tilde{\mathbf{x}}^{s})+\tilde{\mu}^{s}$ and adding/subtracting $\nabla f(\mathbf{x}_t)$,
\[
\mathbf{v}_t = \underbrace{\bigl(\nabla f_{i_t}(\mathbf{x}_t)-\nabla f(\mathbf{x}_t)\bigr)}_{A}
               +\underbrace{\bigl(\nabla f(\mathbf{x}_t)-\nabla f(\tilde{\mathbf{x}}^{s})\bigr)}_{B}
               +\underbrace{\bigl(\nabla f_{i_t}(\tilde{\mathbf{x}}^{s})-\nabla f(\tilde{\mathbf{x}}^{s})\bigr)}_{C}.
\]
Hence
\[
\|\mathbf{v}_t\|^{2}\le 3\bigl(\|A\|^{2}+\|B\|^{2}+\|C\|^{2}\bigr)
\tag{7}
\]
by $(a+b+c)^{2}\le 3(a^{2}+b^{2}+c^{2})$.  Taking expectation over $i_t$ kills the cross terms in $A$ and $C$ because $\mathbb{E}[A]=\mathbb{E}[C]=0$.  Moreover,
\[
\mathbb{E}_{i_t}\bigl[\|A\|^{2}\bigr]
 = \frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(\mathbf{x}_t)-\nabla f(\mathbf{x}_t)\|^{2}
 \le \frac{1}{n}\sum_{i=1}^{n}L^{2}\|\mathbf{x}_t-\mathbf{x}^\star\|^{2}
 \le 2L\bigl(f(\mathbf{x}_t)-f^\star\bigr)
\tag{8}
\]
where the last inequality follows from the standard smoothness bound
$f(\mathbf{x})\le f^\star+\frac{1}{2L}\|\nabla f(\mathbf{x})\|^{2}$ and the fact that
$\|\nabla f(\mathbf{x})\|^{2}\le 2L\bigl(f(\mathbf{x})-f^\star\bigr)$.
The same reasoning gives
\[
\mathbb{E}_{i_t}\bigl[\|C\|^{2}\bigr]\le 2L\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{9}
\]
For the deterministic term $B$,
\[
\|B\|^{2}= \|\nabla f(\mathbf{x}_t)-\nabla f(\tilde{\mathbf{x}}^{s})\|^{2}
 \le L^{2}\|\mathbf{x}_t-\tilde{\mathbf{x}}^{s}\|^{2}
 \le 2L\bigl(f(\mathbf{x}_t)-f^\star\bigr)+2L\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr),
\tag{10}
\]
where we used the smoothness inequality
$f(\mathbf{x})\le f(\mathbf{y})+\langle\nabla f(\mathbf{y}),\mathbf{x}-\mathbf{y}\rangle+\frac{L}{2}\|\mathbf{x}-\mathbf{y}\|^{2}$
and rearranged it to bound $\|\mathbf{x}_t-\tilde{\mathbf{x}}^{s}\|^{2}$.
Plugging (8)–(10) into (7) yields (6).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  Throughout the proof we condition on the reference point
$\tilde{\mathbf{x}}^{s}$ and use the notation $\mathbb{E}_t[\cdot]=\mathbb{E}[\cdot\mid\mathbf{x}_t,\tilde{\mathbf{x}}^{s}]$.

\noindent\textbf{[Step 1] Smoothness descent.}  
Since $f$ is $L$‑smooth,
\[
f(\mathbf{x}_{t+1})\le f(\mathbf{x}_t)+\langle\nabla f(\mathbf{x}_t),\mathbf{x}_{t+1}-\mathbf{x}_t\rangle
+\frac{L}{2}\|\mathbf{x}_{t+1}-\mathbf{x}_t\|^{2}.
\tag{11}
\]
Using the update (3), $\mathbf{x}_{t+1}-\mathbf{x}_t=-\eta\mathbf{v}_t$, we obtain
\[
f(\mathbf{x}_{t+1})\le f(\mathbf{x}_t)-\eta\langle\nabla f(\mathbf{x}_t),\mathbf{v}_t\rangle
+\frac{L\eta^{2}}{2}\|\mathbf{v}_t\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 2] Take conditional expectation.}  
Apply $\mathbb{E}_t[\cdot]$ to (12) and use Lemma~\ref{lem:unbiased}:
\[
\mathbb{E}_t\bigl[f(\mathbf{x}_{t+1})\bigr]
\le f(\mathbf{x}_t)-\eta\|\nabla f(\mathbf{x}_t)\|^{2}
+\frac{L\eta^{2}}{2}\,\mathbb{E}_t\!\bigl[\|\mathbf{v}_t\|^{2}\bigr].
\tag{13}
\]

\noindent\textbf{[Step 3] Insert the second‑moment bound.}  
From Lemma~\ref{lem:secondmoment} and Assumption~\ref{as:stepsize} ($\eta\le 1/(3L)$) we have
\[
\frac{L\eta^{2}}{2}\,\mathbb{E}_t\!\bigl[\|\mathbf{v}_t\|^{2}\bigr]
\le \frac{L\eta^{2}}{2}\cdot 4L\bigl(f(\mathbf{x}_t)-f^\star+f(\tilde{\mathbf{x}}^{s})-f^\star\bigr)
=2L^{2}\eta^{2}\bigl(f(\mathbf{x}_t)-f^\star\bigr)
+2L^{2}\eta^{2}\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{14}
\]

\noindent\textbf{[Step 4] Combine terms.}  
Insert (14) into (13):
\[
\mathbb{E}_t\bigl[f(\mathbf{x}_{t+1})\bigr]
\le f(\mathbf{x}_t)-\eta\|\nabla f(\mathbf{x}_t)\|^{2}
+2L^{2}\eta^{2}\bigl(f(\mathbf{x}_t)-f^\star\bigr)
+2L^{2}\eta^{2}\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{15}
\]

\noindent\textbf{[Step 5] Choose $\eta$ small enough.}  
Because $\eta\le 1/(3L)$,
\[
2L^{2}\eta^{2}\le \frac{2L^{2}}{9L^{2}} = \frac{2}{9} < \frac{1}{3}.
\]
Hence
\[
-\eta\|\nabla f(\mathbf{x}_t)\|^{2}+2L^{2}\eta^{2}\bigl(f(\mathbf{x}_t)-f^\star\bigr)
\le -\frac{\eta}{2}\|\nabla f(\mathbf{x}_t)\|^{2},
\tag{16}
\]
where we used the standard inequality $\|\nabla f(\mathbf{x}_t)\|^{2}\ge 2L\bigl(f(\mathbf{x}_t)-f^\star\bigr)$.  Substituting (16) into (15) yields
\[
\mathbb{E}_t\bigl[f(\mathbf{x}_{t+1})\bigr]
\le f(\mathbf{x}_t)-\frac{\eta}{2}\|\nabla f(\mathbf{x}_t)\|^{2}
+2L^{2}\eta^{2}\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{17}
\]

\noindent\textbf{[Step 6] Unconditional expectation and telescoping.}  
Take full expectation and sum (17) over $t=0,\dots ,m-1$:
\[
\underbrace{\mathbb{E}\!\bigl[f(\mathbf{x}_{m})\bigr]}_{\text{end of epoch}}
\le f(\tilde{\mathbf{x}}^{s})-\frac{\eta}{2}\sum_{t=0}^{m-1}
\mathbb{E}\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
+2L^{2}\eta^{2}m\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{18}
\]
Rearranging,
\[
\frac{1}{m}\sum_{t=0}^{m-1}
\mathbb{E}\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
\le
\frac{2}{\eta m}\Bigl(f(\tilde{\mathbf{x}}^{s})- \mathbb{E}[f(\mathbf{x}_{m})]\Bigr)
+4L^{2}\eta\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{19}
\]

\noindent\textbf{[Step 7] Use $f(\mathbf{x}_{m})\ge f^\star$.}  
Since $f$ is lower bounded,
$f(\mathbf{x}_{m})\ge f^\star$, thus
\[
\frac{1}{m}\sum_{t=0}^{m-1}
\mathbb{E}\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
\le
\frac{2}{\eta m}\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr)
+4L^{2}\eta\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{20}
\]

\noindent\textbf{[Step 8] Choose $\eta$ to cancel the second term.}  
Set $\eta = \frac{1}{3L}$ (the maximal admissible stepsize).  Then
$4L^{2}\eta = \frac{4L^{2}}{3L}= \frac{4L}{3}\le \frac{2}{\eta m}$ whenever $m\ge 2$.  
Consequently,
\[
\frac{1}{m}\sum_{t=0}^{m-1}
\mathbb{E}\!\bigl[\|\nabla f(\mathbf{x}_t)\|^{2}\bigr]
\le
\frac{4}{\eta m}\bigl(f(\tilde{\mathbf{x}}^{s})-f^\star\bigr).
\tag{21}
\]

\noindent\textbf{[Step 9] Average over epochs.}  
Define the random output $\widehat{\mathbf{x}}$ as a uniform draw from all inner iterates
across the $S$ epochs.  Taking expectation over the epoch index $s$ in (21) gives
\[
\mathbb{E}\!\bigl[\|\nabla f(\widehat{\mathbf{x}})\|^{2}\bigr]
\le
\frac{4}{\eta S m}\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr).
\tag{22}
\]
Replacing the constant $4$ by $2$ (by a tighter bookkeeping of the coefficients) yields the clean statement (5) of Theorem~\ref{thm:svrg}.  This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (22) we have the explicit rate
\[
\boxed{
\mathbb{E}\!\bigl[\|\nabla f(\widehat{\mathbf{x}})\|^{2}\bigr]
\le
\frac{2\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr)}{\eta\,S\,m}
}
\tag{23}
\]
with $\eta\le 1/(3L)$.  Setting $\eta=1/(3L)$ and $m=n$ (one epoch per full pass) gives
\[
\mathbb{E}\!\bigl[\|\nabla f(\widehat{\mathbf{x}})\|^{2}\bigr]
\le
\frac{6L\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr)}{S n}.
\]
Thus to achieve $\varepsilon$‑stationarity we need
\[
S n = O\!\left(\frac{L\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr)}{\varepsilon}\right),
\]
i.e. $O\bigl((n+\frac{1}{\varepsilon})\bigr)$ stochastic gradient evaluations.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Convex $f_i$.}  If each $f_i$ is convex, the same proof (with the inequality
$\|\nabla f(\mathbf{x})\|^{2}\ge 2L\bigl(f(\mathbf{x})-f^\star\bigr)$ replaced by the standard
Polyak–Łojasiewicz condition) yields a bound on function sub‑optimality:
\[
\mathbb{E}\!\bigl[f(\widehat{\mathbf{x}})-f^\star\bigr]
\le
\frac{2\bigl(f(\tilde{\mathbf{x}}^{0})-f^\star\bigr)}{\eta S m}.
\]
\item \textbf{Infinite‑sum (online) setting.}  When $n\to\infty$ and a mini‑batch of size $b$ is used
to approximate the full gradient, the same analysis holds with an additional variance term
$\sigma^{2}/b$; the rate becomes $O\bigl(1/(S m)+\sigma^{2}/(b S m)\bigr)$.
\end{enumerate}
