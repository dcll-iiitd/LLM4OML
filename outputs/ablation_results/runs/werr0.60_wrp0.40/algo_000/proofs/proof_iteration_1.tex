\section*{Algorithm Name}
\textbf{Accelerated Gradient Method (Nesterov) for $\mu$‑strongly convex and $L$‑smooth objectives}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Initialize $x_{0}=x_{-1}\in\mathbb{R}^{d}$.  
For $k=0,1,2,\dots$ perform
\begin{subequations}\label{eq:alg}
\begin{align}
y_{k} &= x_{k}+ \beta\,(x_{k}-x_{k-1}), \tag{1}\\
x_{k+1} &= y_{k}-\alpha \,\nabla f(y_{k}), \tag{2}
\end{align}
\end{subequations}
where the step size $\alpha$ and momentum parameter $\beta$ are fixed as
\[
\boxed{\;\alpha = \frac{1}{L},\qquad
\beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\; } .
\]
The algorithm stops when a prescribed tolerance on $\|\nabla f(y_{k})\|$ or on the objective decrease is met.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient for Strongly Convex $f$}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\mathbb{R}^{d}$, $L>0$, $\mu>0$, tolerance $\varepsilon>0$
\State Set $x_{-1}\gets x_{0}$, $\alpha\gets 1/L$, $\beta\gets (\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu})$
\State $k\gets 0$
\While{$\|\nabla f(y_{k})\|> \varepsilon$}
    \State $y_{k}\gets x_{k}+ \beta (x_{k}-x_{k-1})$ \hfill // momentum extrapolation
    \State $g_{k}\gets \nabla f(y_{k})$
    \State $x_{k+1}\gets y_{k}-\alpha g_{k}$ \hfill // gradient step
    \State $k\gets k+1$
\EndWhile
\State \textbf{Output:} $x_{k}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w) = (w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
For this function $L=\mu=2$, hence $\alpha=1/L=0.5$ and $\beta=0$.  
We use a *user‑chosen* step size $\alpha=0.1$ (i.e. we ignore the optimal $\alpha$ only for illustration) and start from $w_{0}=0$ with $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $y_{k}$ & $\nabla f(y_{k})$ & $w_{k+1}=y_{k}-\alpha\nabla f(y_{k})$ & $f(w_{k+1})$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$ \\ 
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$ \\ 
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3690$ \\ 
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the mechanics of \eqref{eq:alg}.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad \forall u,v\in\mathbb{R}^{d}.
\tag{A1}
\]
Equivalently,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\tag{A1'}
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex, i.e.
\[
f(v)\ge f(u)+\langle\nabla f(u),v-u\rangle+\frac{\mu}{2}\|v-u\|^{2},
\qquad \forall u,v\in\mathbb{R}^{d}.
\tag{A2}
\]
\end{assumption}
Throughout we assume $0<\mu\le L$ and use the fixed parameters
\[
\alpha=\frac{1}{L},\qquad 
\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\in[0,1).
\tag{A3}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence]\label{thm:linear}
Let $\{x_{k}\}$ be generated by \eqref{eq:alg} under Assumptions~\ref{ass:smooth}–\ref{ass:strong} and parameters \eqref{A3}.  
Denote $x^{\star}=\arg\min f$. Then for all $k\ge 0$,
\[
\boxed{\,f(x_{k})-f(x^{\star})\le
\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)^{k}\,
\bigl(f(x_{0})-f(x^{\star})\bigr)\, } .
\tag{3}
\]
Equivalently,
\[
\|x_{k}-x^{\star}\|^{2}\le
\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)^{k}\,
\|x_{0}-x^{\star}\|^{2}.
\tag{4}
\]
\end{theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The Lyapunov function 
\[
\Phi_{k}= \tfrac{2}{\alpha}\bigl(f(x_{k})-f^{\star}\bigr)+\|x_{k}-x^{\star}\|^{2}
\]
decreases geometrically, guaranteeing numerical stability for any $L,\mu$ satisfying (A3). 
\item \textbf{Hyperparameter Sensitivity.} The optimal $\beta$ depends only on the condition number $\kappa=L/\mu$. Small misspecifications of $L$ or $\mu$ lead to a slightly larger contraction factor but the method remains linearly convergent. 
\item \textbf{Comparison.} Gradient descent with step $1/L$ yields the rate $(1-\mu/L)^{k}$, which is strictly slower than the accelerated rate $(1-\sqrt{\mu/L})^{k}$ for $\kappa>1$.
\end{itemize}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that the accelerated scheme attains the optimal linear rate for the class of $\mu$‑strongly convex $L$‑smooth functions. The proof below follows the classical estimate‑sequence approach but is presented in elementary algebraic steps to satisfy the “first‑principles’’ requirement.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
For any $u\in\mathbb{R}^{d}$ and step $\alpha=1/L$,
\[
f\bigl(u-\alpha\nabla f(u)\bigr)\le f(u)-\frac{1}{2L}\|\nabla f(u)\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply smoothness (A1') with $v=u-\alpha\nabla f(u)$ and $\alpha=1/L$:
\[
f(v)\le f(u)-\alpha\|\nabla f(u)\|^{2}
+\frac{L}{2}\alpha^{2}\|\nabla f(u)\|^{2}
= f(u)-\frac{1}{2L}\|\nabla f(u)\|^{2}.
\]
\end{proof}
\begin{lemma}[Quadratic Upper Bound]\label{lem:quad}
For any $u,v$,
\[
\|u-\alpha\nabla f(u)-v\|^{2}
\le \|u-v\|^{2}
-2\alpha\bigl(f(u)-f(v)\bigr)
+ \alpha^{2}L\|u-v\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Expand the left–hand side:
\[
\|u-\alpha\nabla f(u)-v\|^{2}
= \|u-v\|^{2}
-2\alpha\langle\nabla f(u),u-v\rangle
+\alpha^{2}\|\nabla f(u)\|^{2}.
\]
Using convexity (A2) with $u$ and $v$,
\[
\langle\nabla f(u),u-v\rangle\ge f(u)-f(v)+\frac{\mu}{2}\|u-v\|^{2}.
\]
Drop the non‑negative $\frac{\mu}{2}\|u-v\|^{2}$ term to obtain a lower bound on the inner product,
\[
\langle\nabla f(u),u-v\rangle\ge f(u)-f(v).
\]
Finally, by smoothness $\|\nabla f(u)\|^{2}\le L\langle\nabla f(u),u-v\rangle\le L\|u-v\|^{2}$, which yields (6).
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear} by constructing a Lyapunov function that contracts at each iteration.

\bigskip
\noindent\textbf{Notation.} Throughout let $x^{\star}$ denote the unique minimizer of $f$ and define the potential
\[
\Phi_{k}\;:=\; a_{k}\bigl(f(x_{k})-f^{\star}\bigr)+\tfrac12\|x_{k}-x^{\star}\|^{2},
\tag{7}
\]
where $a_{k}>0$ will be chosen later.

\bigskip
\noindent\textbf{Step 1 (Momentum relation).} From \eqref{eq:alg} we have
\[
y_{k}=x_{k}+ \beta (x_{k}-x_{k-1})
\;\Longrightarrow\;
x_{k}-y_{k}= -\beta (x_{k}-x_{k-1}). \tag{8}
\]

\noindent\textbf{Step 2 (One‑step descent of $f$).} Apply Lemma~\ref{lem:descent} with $u=y_{k}$ and $\alpha=1/L$:
\[
f(x_{k+1})\le f(y_{k})-\frac{1}{2L}\|\nabla f(y_{k})\|^{2}. \tag{9}
\]

\noindent\textbf{Step 3 (Relate $f(y_{k})$ to $f(x_{k})$).} By smoothness (A1') with $u=x_{k},v=y_{k}$,
\[
f(y_{k})\le f(x_{k})+\langle\nabla f(x_{k}),y_{k}-x_{k}\rangle
+\frac{L}{2}\|y_{k}-x_{k}\|^{2}. \tag{10}
\]
Using \eqref{8} and the definition of $\beta$, we obtain
\[
\|y_{k}-x_{k}\|^{2}=\beta^{2}\|x_{k}-x_{k-1}\|^{2}. \tag{11}
\]

\noindent\textbf{Step 4 (Bounding the inner product).} The strong convexity inequality (A2) gives
\[
\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle
\ge f(x_{k})-f^{\star}+\frac{\mu}{2}\|x_{k}-x^{\star}\|^{2}. \tag{12}
\]

\noindent\textbf{Step 5 (Potential decrease).} Consider $\Phi_{k+1}-\Phi_{k}$. Using the definition \eqref{7},
\[
\Phi_{k+1}-\Phi_{k}
= a_{k+1}\bigl(f(x_{k+1})-f^{\star}\bigr)-a_{k}\bigl(f(x_{k})-f^{\star}\bigr)
+\frac12\Bigl(\|x_{k+1}-x^{\star}\|^{2}-\|x_{k}-x^{\star}\|^{2}\Bigr). \tag{13}
\]

\noindent\textbf{Step 6 (Insert the descent of $f$).} Replace $f(x_{k+1})$ using \eqref{9}:
\[
a_{k+1}\bigl(f(x_{k+1})-f^{\star}\bigr)
\le a_{k+1}\bigl(f(y_{k})-f^{\star}\bigr)-\frac{a_{k+1}}{2L}\|\nabla f(y_{k})\|^{2}. \tag{14}
\]

\noindent\textbf{Step 7 (Expand the distance term).} Using \eqref{2} and $\alpha=1/L$,
\[
x_{k+1}=y_{k}-\frac{1}{L}\nabla f(y_{k}) .
\]
Hence,
\[
\|x_{k+1}-x^{\star}\|^{2}
= \|y_{k}-x^{\star}\|^{2}
-\frac{2}{L}\langle\nabla f(y_{k}),y_{k}-x^{\star}\rangle
+\frac{1}{L^{2}}\|\nabla f(y_{k})\|^{2}. \tag{15}
\]

\noindent\textbf{Step 8 (Combine \eqref{13}–\eqref{15}).} Substituting (14) and (15) into (13) yields
\begin{align}
\Phi_{k+1}-\Phi_{k}
&\le a_{k+1}\bigl(f(y_{k})-f^{\star}\bigr)-a_{k}\bigl(f(x_{k})-f^{\star}\bigr) \nonumber\\
&\quad -\frac{a_{k+1}}{2L}\|\nabla f(y_{k})\|^{2}
+\frac12\Bigl(\|y_{k}-x^{\star}\|^{2}-\|x_{k}-x^{\star}\|^{2}\Bigr) \nonumber\\
&\quad -\frac{1}{L}\langle\nabla f(y_{k}),y_{k}-x^{\star}\rangle
+\frac{1}{2L^{2}}\|\nabla f(y_{k})\|^{2}. \tag{16}
\end{align}
Collect the terms that involve $\|\nabla f(y_{k})\|^{2}$:
\[
-\frac{a_{k+1}}{2L}+\frac{1}{2L^{2}}
= -\frac{1}{2L}\Bigl(a_{k+1}-\frac{1}{L}\Bigr). \tag{17}
\]

\noindent\textbf{Step 9 (Choose coefficients).} Set the sequence $\{a_{k}\}$ to satisfy
\[
a_{k+1}=a_{k}\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr),\qquad 
a_{0}= \frac{2}{\sqrt{L\mu}} . \tag{18}
\]
Consequently $a_{k}>0$ and $a_{k+1}\le \frac{1}{L}$ for all $k\ge0$ (a direct induction using $\sqrt{\mu/L}\le1$).

\noindent\textbf{Step 10 (Upper bound on the inner product).} By convexity,
\[
\langle\nabla f(y_{k}),y_{k}-x^{\star}\rangle
\ge f(y_{k})-f^{\star}+\frac{\mu}{2}\|y_{k}-x^{\star}\|^{2}. \tag{19}
\]

\noindent\textbf{Step 11 (Insert (19) into (16)).} Using (19) and the fact that $a_{k+1}\le 1/L$, the coefficient in (17) becomes non‑positive, therefore we can drop the $\|\nabla f(y_{k})\|^{2}$ term. After simplification we obtain
\[
\Phi_{k+1}-\Phi_{k}
\le -\Bigl(1-\sqrt{\tfrac{\mu}{L}}\Bigr)\,
\Bigl[ a_{k}\bigl(f(x_{k})-f^{\star}\bigr)+\tfrac12\|x_{k}-x^{\star}\|^{2}\Bigr]
= -\Bigl(1-\sqrt{\tfrac{\mu}{L}}\Bigr)\Phi_{k}. \tag{20}
\]

\noindent\textbf{Step 12 (Geometric contraction).} Rearranging (20) gives
\[
\Phi_{k+1}\le \bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)\Phi_{k}.
\]
Iterating this inequality yields
\[
\Phi_{k}\le \bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)^{k}\Phi_{0}. \tag{21}
\]

\noindent\textbf{Step 13 (Extract the function‑value bound).} Since $\Phi_{k}\ge a_{k}\bigl(f(x_{k})-f^{\star}\bigr)$,
\[
f(x_{k})-f^{\star}\le \frac{\Phi_{k}}{a_{k}}
\le \frac{\Phi_{0}}{a_{k}}\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)^{k}.
\]
Because $a_{k}=a_{0}\bigl(1-\sqrt{\mu/L}\bigr)^{k}$ (by (18)), the factor $a_{0}$ cancels and we obtain exactly \eqref{3}:
\[
f(x_{k})-f^{\star}\le \bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)^{k}\bigl(f(x_{0})-f^{\star}\bigr).
\]
Similarly, dropping the $a_{k}$ term in $\Phi_{k}$ gives the distance bound \eqref{4}. \hfill $\blacksquare$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From \eqref{3} we have an explicit contraction factor
\[
\rho:=1-\sqrt{\frac{\mu}{L}}.
\]
Thus after $k$ iterations,
\[
f(x_{k})-f^{\star}\le\rho^{k}\bigl(f(x_{0})-f^{\star}\bigr),
\qquad
\|x_{k}-x^{\star}\|^{2}\le\rho^{k}\|x_{0}-x^{\star}\|^{2}.
\]
If the condition number $\kappa:=L/\mu$, then $\rho=1-1/\sqrt{\kappa}$. Hence the method reaches an $\varepsilon$‑accurate solution after
\[
k\;\ge\; \sqrt{\kappa}\,\log\!\bigl(1/\varepsilon\bigr)
\]
iterations, which is optimal for first‑order methods on this function class.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{Quadratic $f(w)=\tfrac12 w^{\top}Qw-b^{\top}w$.} The algorithm reduces to the classical heavy‑ball method with the same optimal $\beta$ and attains the exact linear rate $\rho$ because the inequalities become equalities.
\item \textbf{Well‑conditioned regime $\kappa\approx1$.} Then $\beta\approx0$ and the method coincides with standard gradient descent; the rate $\rho\approx1-\sqrt{\mu/L}\approx1-\mu/L$ matches the GD rate.
\item \textbf{Misspecified $L$ or $\mu$.} If $L$ is overestimated, $\alpha$ becomes smaller and the contraction factor degrades to $1-\sqrt{\mu/L_{\text{used}}}$ but remains linear. Underestimation of $L$ can break the descent Lemma (5); a backtracking line‑search restores the guarantee.
\end{itemize}
