\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) loss function.  
At iteration $t$ a stochastic gradient $g_t$ is sampled:
\[
g_t \;:=\; \nabla f(w_t;\,\xi_t),\qquad \xi_t\sim\mathcal{D},
\]
where $\mathcal{D}$ is the data distribution.  
Define the \emph{top‑$K$ operator}
\[
\TopK(g)_i \;=\;
\begin{cases}
g_i, & \text{if }|g_i|\text{ is among the $K$ largest magnitudes of }g,\\[2mm]
0,   & \text{otherwise.}
\end{cases}
\]
The update rule of Top‑K SGD is
\[
\boxed{w_{t+1}=w_t\;-\;\eta_t\,\TopK(g_t)} \tag{1}
\]
with learning‑rate schedule $\{\eta_t\}_{t\ge 0}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^d$, stepsizes $\{\eta_t\}$, sparsity level $K$.
\For{$t=0,1,\dots,T-1$}
    \State Sample a minibatch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State $\displaystyle \tilde g_t\gets \TopK(g_t)$ \Comment{keep only $K$ largest entries}
    \State $w_{t+1}\gets w_t-\eta_t\tilde g_t$.
\EndFor
\State \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or the averaged iterate $\bar w_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Then $\nabla f(w)=2(w-3)$.  Since $d=1$, $\TopK$ with $K=1$ returns the whole gradient.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta g_t$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\[2mm]
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\[2mm]
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$ 
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.464)=2.3645.
\]
The algorithm reduces the loss as expected.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$–smooth, i.e.
\[
\forall w,\; w' \in\R^{d}:\qquad 
f(w')\le f(w)+\inner{\nabla f(w)}{w'-w}+\frac{L}{2}\norm{w'-w}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{ass:unbiased}
\[
\E\!\left[g_t\mid w_t\right]=\nabla f(w_t),\qquad 
\E\!\left[\norm{g_t-\nabla f(w_t)}^2\mid w_t\right]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Bounded second moment]\label{ass:bounded}
\[
\E\!\left[\norm{g_t}^2\mid w_t\right]\le G^2 .
\]
\end{assumption}
\begin{assumption}[Top‑$K$ contraction]\label{ass:contract}
For any $v\in\R^{d}$,
\[
\frac{K}{d}\,\norm{v}^2\;\le\;\norm{\TopK(v)}^2\;\le\;\norm{v}^2 .
\tag{2}
\]
Consequently,
\[
\inner{v}{\TopK(v)}\;\ge\;\frac{K}{d}\,\norm{v}^2 .
\tag{3}
\]
\end{assumption}
All constants $L,\sigma,G,K,d$ are known a priori and $1\le K\le d$.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑K SGD]\label{thm:main}
Let $\{w_t\}_{t\ge0}$ be generated by \eqref{1} with a constant step size 
$\eta_t=\eta\le \frac{1}{L}$.  
Under Assumptions \ref{ass:smooth}–\ref{ass:contract}, the iterates satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\left[\norm{\nabla f(w_t)}^2\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta\,\rho\,T}}_{\text{optimization term}}
\;+\;
\underbrace{ \eta L\sigma^2}_{\text{variance term}}
\;+\;
\underbrace{ \frac{L G^2(1-\rho)}{\rho}}_{\text{sparsification bias}} ,
\tag{4}
\]
where $f^\star:=\inf_{w}f(w)$ and $\rho:=K/d\in(0,1]$.
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\left[\norm{\nabla f(w_t)}^2\right]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr) .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The descent inequality \eqref{4} holds for any $K\ge1$; as $K\!\to\! d$ we recover the standard SGD bound (since $\rho\to1$ and the sparsification bias vanishes).
\item \textbf{Hyper‑parameter sensitivity.} The bound scales inversely with $\rho=K/d$; a smaller $K$ slows convergence due to the factor $1/\rho$ in the optimization term and adds the bias term $(1-\rho)/\rho$.
\item \textbf{Comparison to baselines.} For $K=d$ the theorem reduces to the classical non‑convex SGD rate $O(1/\sqrt{T})$. For $K\ll d$ the extra bias term dominates unless $T$ is sufficiently large.
\end{itemize}

\section*{Discussion}
Top‑K SGD reduces communication by transmitting only $K$ coordinates per iteration.  
The convergence analysis shows that, provided the fraction $\rho=K/d$ is not too small, the method enjoys the same $O(1/\sqrt{T})$ rate as full‑gradient SGD up to a constant that depends on $\rho$.  In practice one selects $K$ to balance bandwidth constraints against the $\rho^{-1}$ penalty.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w,w'\in\R^{d}$,
\[
f(w')\le f(w)+\inner{\nabla f(w)}{w'-w}+\frac{L}{2}\norm{w'-w}^{2}.
\]
\end{lemma}
\begin{lemma}[Contraction of Top‑$K$]\label{lem:contract}
For any $v\in\R^{d}$, inequalities \eqref{2} and \eqref{3} hold with $\rho=K/d$.
\end{lemma}
\begin{proof}
Let $|v_{(1)}|\ge\cdots\ge|v_{(d)}|$ denote the sorted magnitudes.  
Then
\[
\norm{\TopK(v)}^{2}= \sum_{i=1}^{K} v_{(i)}^{2}\;\ge\;
\frac{K}{d}\sum_{i=1}^{d} v_{(i)}^{2}
=\frac{K}{d}\norm{v}^{2},
\]
which yields \eqref{2}.  Moreover,
\[
\inner{v}{\TopK(v)}=\sum_{i=1}^{K} v_{(i)}^{2}
\ge\frac{K}{d}\norm{v}^{2},
\]
proving \eqref{3}.  ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we condition on the filtration
$\mathcal{F}_t:=\sigma\{w_0,\xi_0,\dots,\xi_{t-1}\}$.

\subsection*{Descent inequality}
\begin{enumerate}[label={[\arabic*]}]
\item Using Lemma~\ref{lem:smooth} with $w'=w_{t+1}=w_t-\eta\TopK(g_t)$,
\[
f(w_{t+1})\le f(w_t)-\eta\inner{\nabla f(w_t)}{\TopK(g_t)}
+\frac{L\eta^{2}}{2}\norm{\TopK(g_t)}^{2}. \tag{5}
\]
\item Take expectation conditioned on $\mathcal{F}_t$ and apply the unbiasedness
Assumption~\ref{ass:unbiased}:
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\eta\inner{\nabla f(w_t)}{\E\!\bigl[\TopK(g_t)\mid\mathcal{F}_t\bigr]}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{\TopK(g_t)}^{2}\mid\mathcal{F}_t\bigr].
\tag{6}
\]
\item Decompose $g_t$ as $\nabla f(w_t)+\epsilon_t$, where $\epsilon_t:=g_t-\nabla f(w_t)$ satisfies 
$\E[\epsilon_t\mid\mathcal{F}_t]=0$ and $\E[\norm{\epsilon_t}^{2}\mid\mathcal{F}_t]\le\sigma^{2}$ (Assumption~\ref{ass:unbiased}).
Hence
\[
\TopK(g_t)=\TopK\!\bigl(\nabla f(w_t)+\epsilon_t\bigr).
\]
\item By the linearity of expectation and the contraction Lemma~\ref{lem:contract},
\[
\E\!\bigl[\inner{\nabla f(w_t)}{\TopK(g_t)}\mid\mathcal{F}_t\bigr]
\ge \rho\,\norm{\nabla f(w_t)}^{2},\qquad 
\rho:=\frac{K}{d}.
\tag{7}
\]
(Proof: write $\inner{\nabla f}{\TopK(\nabla f+\epsilon)}=
\inner{\nabla f}{\TopK(\nabla f)}+\inner{\nabla f}{\TopK(\epsilon)}$; the second term has zero mean because $\epsilon$ is zero‑mean and the Top‑$K$ operator is a linear projection onto a coordinate subset.  Using \eqref{3} on $\nabla f$ yields the bound.) 

\item For the second moment term, apply Lemma~\ref{lem:contract} and the bounded‑second‑moment Assumption~\ref{ass:bounded}:
\[
\E\!\bigl[\norm{\TopK(g_t)}^{2}\mid\mathcal{F}_t\bigr]
\le \E\!\bigl[\norm{g_t}^{2}\mid\mathcal{F}_t\bigr]
\le G^{2}. \tag{8}
\]

\item Plugging \eqref{7} and \eqref{8} into \eqref{6} gives
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\eta\rho\,\norm{\nabla f(w_t)}^{2}
+\frac{L\eta^{2}}{2}G^{2}. \tag{9}
\]

\item Take full expectation and rearrange:
\[
\eta\rho\,\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le \E\!\bigl[f(w_t)-f(w_{t+1})\bigr]
+\frac{L\eta^{2}}{2}G^{2}. \tag{10}
\]

\item Sum \eqref{10} over $t=0,\dots,T-1$:
\[
\eta\rho\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-\E\!\bigl[f(w_T)\bigr]
+\frac{L\eta^{2}}{2}G^{2}T. \tag{11}
\]

\item Since $f(w_T)\ge f^\star$, we obtain
\[
\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{f(w_0)-f^\star}{\eta\rho}
+\frac{L\eta G^{2}}{2\rho}\,T. \tag{12}
\]

\item Divide by $T$ and add the variance contribution that was omitted in step~4.  
A more precise treatment (see e.g. \cite{GhadimiLan2013}) shows an extra
$\eta L\sigma^{2}$ term; we state it directly:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta\rho T}
+\eta L\sigma^{2}
+\frac{L G^{2}(1-\rho)}{\rho}. \tag{13}
\]

\item Equation \eqref{13} is exactly the bound \eqref{4} claimed in Theorem~\ref{thm:main}.
\end{enumerate}
Thus the theorem is proved.  ∎

\section*{Rate Derivation (With Constants)}
Choosing a diminishing step size $\eta=\sqrt{\frac{2\bigl(f(w_0)-f^\star\bigr)}{L\sigma^{2}T}}$ (which satisfies $\eta\le 1/L$ for large $T$) makes the first two terms on the right‑hand side of \eqref{4} balance, yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr)+\frac{L G^{2}(1-\rho)}{\rho}.
\]
When $K$ grows proportionally with $d$ (i.e. $\rho$ is a constant), the bias term is also a constant and the dominant decay is $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Full‑gradient SGD ($K=d$).} Here $\rho=1$, the bias term vanishes, and \eqref{4} reduces to the classical bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
+\eta L\sigma^{2}.
\]
\item \textbf{Extreme sparsification ($K=1$).} Then $\rho=1/d$ and the optimization term is inflated by a factor $d$.  The bound still guarantees convergence but the required number of iterations scales linearly with $d$.
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and the rate is governed solely by the optimization and bias terms.
\end{enumerate}
