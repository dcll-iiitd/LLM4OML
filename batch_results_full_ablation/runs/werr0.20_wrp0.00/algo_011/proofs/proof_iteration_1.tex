\section*{Adaptive Variance‑Scaled Gradient Descent (AVSGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
AVSGD maintains an exponential moving average of the (full) gradient,
\begin{equation}
    v_{t}= (1-\alpha)v_{t-1}+ \alpha \nabla f(x_{t}),\qquad v_{0}=0,
    \label{eq:vt}
\end{equation}
with a smoothing parameter $\alpha\in(0,1]$.  
Because $v_{t}$ is biased toward zero at early iterations, a bias‑corrected estimator is used:
\begin{equation}
    \hat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t}} .
    \label{eq:vtbias}
\end{equation}
The step size at iteration $t$ depends on the norm of $\hat v_{t}$:
\begin{equation}
    \eta_{t}= \frac{\eta}{\,1+\gamma \,\norm{\hat v_{t}}\,},
    \qquad \eta>0,\;\gamma>0 .
    \label{eq:et}
\end{equation}
The update of the decision variable is
\begin{equation}
    x_{t+1}= x_{t}-\eta_{t}\,\nabla f(x_{t}) .
    \label{eq:update}
\end{equation}
The hyper‑parameters of AVSGD are therefore
\[
\alpha\in(0,1],\qquad \eta>0,\qquad \gamma>0 .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Adaptive Variance‑Scaled Gradient Descent (AVSGD)}
\label{alg:avsgd}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\mathbb{R}^{d}$, step‑size base $\eta$, smoothing $\alpha\in(0,1]$, scaling $\gamma>0$, max. iterations $T$.
\STATE $v_{0}\leftarrow 0$, $t\leftarrow 0$.
\WHILE{$t<T$}
    \STATE Compute the (full) gradient $g_{t}\leftarrow \nabla f(x_{t})$.
    \STATE Update the exponential moving average
    \[
        v_{t+1}\leftarrow (1-\alpha)v_{t}+ \alpha g_{t}.
    \]
    \STATE Bias‑corrected estimator
    \[
        \hat v_{t+1}\leftarrow \frac{v_{t+1}}{1-(1-\alpha)^{\,t+1}} .
    \]
    \STATE Adaptive step size
    \[
        \eta_{t+1}\leftarrow \frac{\eta}{1+\gamma \,\norm{\hat v_{t+1}} } .
    \]
    \STATE Gradient step
    \[
        x_{t+1}\leftarrow x_{t}-\eta_{t+1}\,g_{t}.
    \]
    \STATE $t\leftarrow t+1$.
\ENDWHILE
\RETURN $x_{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^{d},\qquad 
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]
There exists $\mu>0$ such that
\[
\forall x\in\mathbb{R}^{d},\qquad 
\frac{1}{2}\norm{\nabla f(x)}^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\]
where $f^{\star}= \min_{x}f(x)$.
\end{assumption}
\begin{assumption}[Bounded gradients]
There exists $G>0$ with $\norm{\nabla f(x)}\le G$ for all $x$.
\end{assumption}
All constants $L,\mu,G$ are known to the analyst but not necessarily to the algorithm.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AVSGD under PL and smoothness]\label{thm:linear}
Assume the three conditions above and fix $\alpha\in(0,1]$, $\eta>0$, $\gamma>0$.
Define
\[
\underline{\eta}\;:=\;\frac{\eta}{1+\gamma G}, \qquad
\rho\;:=\;\underline{\eta}\,\mu\;\Bigl(1-\frac{L\underline{\eta}}{2}\Bigr).
\]
If $0<\underline{\eta}\le \frac{2}{L}$ (i.e. $\eta\le\frac{2(1+\gamma G)}{L}$), then for the iterates generated by AVSGD,
\[
\mathbb{E}\!\left[f(x_{t})-f^{\star}\right]\;\le\;
\bigl(1-\rho\bigr)^{t}\,\bigl(f(x_{0})-f^{\star}\bigr),\qquad t=0,1,2,\dots
\]
Consequently $\displaystyle\lim_{t\to\infty}\mathbb{E}[f(x_{t})]=f^{\star}$ and the rate is geometric with factor $1-\rho\in(0,1)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability of the adaptive step:} Lemma~\ref{lem:stepbound} shows that $\eta_{t}$ is sandwiched between $\underline{\eta}$ and $\eta$, guaranteeing that the algorithm never takes an excessively large step (which would break smoothness) nor an overly small step (which would stall progress).
    \item \textbf{Hyper‑parameter sensitivity:}
          \begin{itemize}
            \item Larger $\alpha$ makes $v_{t}$ track the current gradient more closely, reducing the bias of $\hat v_{t}$ but increasing variance; the bound in Lemma~\ref{lem:vtbound} holds for any $\alpha\in(0,1]$.
            \item Larger $\gamma$ shrinks the step size more aggressively when $\norm{\hat v_{t}}$ is large, which can improve robustness on ill‑conditioned problems at the expense of a smaller $\rho$.
          \end{itemize}
    \item \textbf{Comparison to baseline SGD:} Standard SGD with constant step $\eta$ achieves a linear rate $\bigl(1-\eta\mu\bigr)^{t}$ under PL and smoothness. AVSGD replaces $\eta$ by $\underline{\eta}\le\eta$, hence the worst‑case rate is never worse, while in practice the adaptive scaling may yield a larger effective step when $\norm{\hat v_{t}}$ is small, often accelerating convergence.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Bound on the moving average]\label{lem:vtbound}
Under the bounded‑gradient assumption, for all $t\ge0$,
\[
\norm{v_{t}}\;\le\;G .
\]
\end{lemma}
\begin{proof}
We proceed by induction.
\begin{enumerate}
\item[Base case:] $v_{0}=0$, thus $\norm{v_{0}}=0\le G$.
\item[Inductive step:] Assume $\norm{v_{t}}\le G$. From \eqref{eq:vt},
\[
\norm{v_{t+1}}= \norm{(1-\alpha)v_{t}+ \alpha \nabla f(x_{t})}
\le (1-\alpha)\norm{v_{t}}+\alpha\norm{\nabla f(x_{t})}
\le (1-\alpha)G+\alpha G = G .
\]
Thus the claim holds for $t+1$.
\end{enumerate}
\end{proof}

\begin{lemma}[Bounds on the adaptive step size]\label{lem:stepbound}
Let $\underline{\eta}$ and $\overline{\eta}=\eta$ be defined as in Theorem~\ref{thm:linear}. Then for all $t\ge0$,
\[
\underline{\eta}\;\le\;\eta_{t}\;\le\;\overline{\eta}.
\]
\end{lemma}
\begin{proof}
From Lemma~\ref{lem:vtbound} and the bias‑correction factor $1-(1-\alpha)^{t}\in(0,1]$, we have
\[
\norm{\hat v_{t}} = \frac{\norm{v_{t}}}{1-(1-\alpha)^{t}}
\le \frac{G}{1-(1-\alpha)^{t}}\le G .
\]
Plugging this bound into \eqref{eq:et} yields
\[
\eta_{t}= \frac{\eta}{1+\gamma\norm{\hat v_{t}}}
\ge \frac{\eta}{1+\gamma G}= \underline{\eta},
\qquad
\eta_{t}\le \frac{\eta}{1+0}= \eta = \overline{\eta}.
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof we condition on the past iterates; expectations are taken w.r.t.\ the randomness of the algorithm (if any). Since AVSGD uses the full gradient, the expectation operator is redundant but retained for notational uniformity.

\noindent\textbf{[Step 1]} \emph{Smoothness inequality.}  
By $L$‑smoothness of $f$ and the update \eqref{eq:update},
\begin{align}
f(x_{t+1})
&\le f(x_{t}) + \langle \nabla f(x_{t}),\,x_{t+1}-x_{t}\rangle
      + \frac{L}{2}\norm{x_{t+1}-x_{t}}^{2} \nonumber\\
&= f(x_{t}) - \eta_{t}\norm{\nabla f(x_{t})}^{2}
   + \frac{L}{2}\eta_{t}^{2}\norm{\nabla f(x_{t})}^{2}.
\label{eq:smooth}
\end{align}

\noindent\textbf{[Step 2]} \emph{Factorising the descent term.}  
Rearrange \eqref{eq:smooth}:
\begin{equation}
f(x_{t+1})-f^{\star}
\le \Bigl(1-\eta_{t}+\frac{L}{2}\eta_{t}^{2}\Bigr)\bigl(f(x_{t})-f^{\star}\bigr)
    -\Bigl(\eta_{t}-\frac{L}{2}\eta_{t}^{2}\Bigr)
        \bigl(f(x_{t})-f^{\star}\bigr)
        +\frac{L}{2}\eta_{t}^{2}\norm{\nabla f(x_{t})}^{2}.
\end{equation}
Using the PL inequality $\frac12\norm{\nabla f(x_{t})}^{2}\ge\mu\bigl(f(x_{t})-f^{\star}\bigr)$ we replace the last term:
\begin{align}
\frac{L}{2}\eta_{t}^{2}\norm{\nabla f(x_{t})}^{2}
&\le L\eta_{t}^{2}\mu\bigl(f(x_{t})-f^{\star}\bigr).
\end{align}
Thus,
\begin{align}
f(x_{t+1})-f^{\star}
&\le \Bigl(1-\eta_{t}+\frac{L}{2}\eta_{t}^{2}+L\eta_{t}^{2}\mu\Bigr)
        \bigl(f(x_{t})-f^{\star}\bigr) \nonumber\\
&= \Bigl(1-\eta_{t}+\frac{L\eta_{t}^{2}}{2}\bigl(1+2\mu\bigr)\Bigr)
        \bigl(f(x_{t})-f^{\star}\bigr).
\label{eq:prelinear}
\end{align}

\noindent\textbf{[Step 3]} \emph{Lower bound on $\eta_{t}$.}  
From Lemma~\ref{lem:stepbound}, $\eta_{t}\ge\underline{\eta}$. Insert this bound into \eqref{eq:prelinear}:
\begin{align}
f(x_{t+1})-f^{\star}
&\le \Bigl(1-\underline{\eta}+\frac{L\underline{\eta}^{2}}{2}(1+2\mu)\Bigr)
        \bigl(f(x_{t})-f^{\star}\bigr).\label{eq:linearstep}
\end{align}
Define the contraction factor
\[
1-\rho \;:=\; 1-\underline{\eta}+\frac{L\underline{\eta}^{2}}{2}(1+2\mu).
\]
Since $0<\underline{\eta}\le 2/L$, a short quadratic calculation shows that $0<\rho\le 1$ and
\[
\rho = \underline{\eta}\,\mu\Bigl(1-\frac{L\underline{\eta}}{2}\Bigr).
\]
Hence \eqref{eq:linearstep} becomes
\begin{equation}
f(x_{t+1})-f^{\star}\;\le\;(1-\rho)\,\bigl(f(x_{t})-f^{\star}\bigr).
\label{eq:recurrence}
\end{equation}

\noindent\textbf{[Step 4]} \emph{Unrolling the recursion.}  
Applying \eqref{eq:recurrence} repeatedly yields for any integer $t\ge0$,
\[
f(x_{t})-f^{\star}\;\le\;(1-\rho)^{t}\,\bigl(f(x_{0})-f^{\star}\bigr).
\]
Taking expectations on both sides (trivial because the RHS is deterministic) gives the claim of Theorem~\ref{thm:linear}.

\section*{Rate Derivation (With Constants)}
The contraction factor derived above can be written explicitly in terms of the user‑chosen hyper‑parameters:
\[
\rho
= \frac{\eta}{1+\gamma G}\,
   \mu\Bigl(1-\frac{L\eta}{2(1+\gamma G)}\Bigr),
\qquad
\text{provided }\;
\eta\le\frac{2(1+\gamma G)}{L}.
\]
Thus the linear convergence rate improves when:
\begin{itemize}
\item $\eta$ is chosen as large as allowed by the smoothness condition,
\item $\gamma$ is small (so that $\underline{\eta}\approx\eta$),
\item $G$ is a tight bound on the gradient norm (smaller $G$ yields a larger $\underline{\eta}$).
\end{itemize}
If $\gamma=0$, AVSGD reduces to standard SGD with constant step $\eta$, and $\rho$ recovers the classic $\eta\mu(1-\frac{L\eta}{2})$ factor.

\section*{Special Cases}
\begin{corollary}[No adaptation ($\gamma=0$)]
If $\gamma=0$, then $\underline{\eta}=\eta$ and Theorem~\ref{thm:linear} reduces to the standard linear rate for SGD under PL:
\[
\mathbb{E}[f(x_{t})-f^{\star}]\le
\bigl(1-\eta\mu(1-\tfrac{L\eta}{2})\bigr)^{t}\bigl(f(x_{0})-f^{\star}\bigr).
\]
\end{corollary}
\begin{corollary}[Extreme smoothing $\alpha\to1$]
When $\alpha\to1$, $v_{t}$ becomes the current gradient, i.e.\ $v_{t}= \nabla f(x_{t})$, and $\hat v_{t}= \nabla f(x_{t})$. The step size then reads $\eta_{t}= \eta/(1+\gamma\norm{\nabla f(x_{t})})$, which is the classical gradient‑norm‑scaled stepsize. The same linear bound holds because Lemma~\ref{lem:vtbound} still guarantees $\norm{\nabla f(x_{t})}\le G$.
\end{corollary}

\section*{Discussion}
AVSGD blends an exponential moving average of gradients with an adaptive step size that contracts when the averaged gradient norm grows. Under the PL condition and smoothness, the algorithm enjoys a guaranteed geometric decay of the objective value, with a contraction factor that smoothly interpolates between the constant‑step SGD rate and a more conservative rate when the average gradient is large. The proof required only elementary algebraic manipulations and the standard smoothness/PL inequalities, yet it explicitly tracked the effect of the adaptive schedule via Lemma~\ref{lem:stepbound}. Consequently, AVSGD can be viewed as a principled, theoretically‑backed alternative to heuristic adaptive methods such as Adam, while preserving provable linear convergence on a broad class of non‑convex functions that satisfy the PL inequality (e.g., over‑parameterized neural networks, certain regularized empirical risks).
