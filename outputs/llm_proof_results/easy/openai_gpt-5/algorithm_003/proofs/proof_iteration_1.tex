\section*{Algorithm Name}
Online Gradient Descent (OGD) for Sublinear Regret

\section*{Mathematical Formulation}
We operate in an online convex optimization (OCO) setting over a closed, convex, bounded decision set $\mathcal{W}\subset \R^d$ with Euclidean norm $\|\cdot\|$. At each round $t=1,2,\dots,T$:
\begin{itemize}[leftmargin=2em]
\item The learner chooses $w_t\in \mathcal{W}$.
\item An adversary reveals a convex loss $f_t:\mathcal{W}\to \R$, and the learner incurs loss $f_t(w_t)$.
\item The learner observes a subgradient $g_t\in \partial f_t(w_t)$ and updates
\[
w_{t+1} \;=\; \Pi_{\mathcal{W}}\!\bigl(w_t - \eta\, g_t\bigr),
\]
where $\Pi_{\mathcal{W}}$ is the Euclidean projection onto $\mathcal{W}$ and $\eta>0$ is a stepsize (learning rate).
\end{itemize}
Hyperparameters:
\begin{itemize}[leftmargin=2em]
\item Stepsize $\eta>0$ (constant in the main analysis; extensions to time-varying $\eta_t$ are standard).
\end{itemize}
Performance metric (regret) against any comparator $u\in \mathcal{W}$:
\[
R_T(u) \;=\; \sum_{t=1}^T \bigl(f_t(w_t) - f_t(u)\bigr), 
\qquad 
R_T \;=\; R_T(w^\star), 
\quad w^\star \in \arg\min_{w\in \mathcal{W}} \sum_{t=1}^T f_t(w).
\]

\section*{Pseudocode}
\begin{enumerate}[leftmargin=2em]
\item Input: stepsize $\eta>0$, decision set $\mathcal{W}$, horizon $T$.
\item Initialize: choose any $w_1\in \mathcal{W}$ (e.g., $w_1=\arg\min_{w\in \mathcal{W}}\|w\|$).
\item For $t=1,2,\dots,T$:
\begin{enumerate}
\item Play $w_t\in \mathcal{W}$ and observe loss $f_t:\mathcal{W}\to\R$.
\item Obtain a subgradient $g_t\in \partial f_t(w_t)$.
\item Update: $w_{t+1} \leftarrow \Pi_{\mathcal{W}}\!\bigl(w_t - \eta\, g_t\bigr)$.
\end{enumerate}
\item Output: sequence $(w_t)_{t=1}^T$ or an average $\bar{w}_T = \frac{1}{T}\sum_{t=1}^T w_t$ if desired.
\end{enumerate}

\section*{Dry Run Example}
Consider a 1D example with a fixed convex function $f(w)=(w-3)^2$ (so $f_t\equiv f$ for illustration), unconstrained domain $\mathcal{W}=\R$ (projection is identity), stepsize $\eta=0.1$, and initialization $w_1=w_0=0$.
\begin{itemize}[leftmargin=2em]
\item Iteration $t=1$:
\[
g_1 \;=\; \nabla f(w_1) \;=\; 2(w_1-3) \;=\; 2(0-3) \;=\; -6.
\]
Update:
\[
w_2 \;=\; w_1 - \eta g_1 \;=\; 0 - 0.1(-6) \;=\; 0.6.
\]
Objective values: $f(w_1)=(0-3)^2=9$, $f(w_2)=(0.6-3)^2=5.76$.
\item Iteration $t=2$:
\[
g_2 \;=\; 2(w_2-3) \;=\; 2(0.6-3) \;=\; -4.8,
\quad
w_3 \;=\; 0.6 - 0.1(-4.8) \;=\; 1.08.
\]
Objective: $f(w_3)=(1.08-3)^2 = (-1.92)^2 = 3.6864$.
\item Iteration $t=3$:
\[
g_3 \;=\; 2(w_3-3) \;=\; 2(1.08-3) \;=\; -3.84,
\quad
w_4 \;=\; 1.08 - 0.1(-3.84) \;=\; 1.464.
\]
Objective: $f(w_4)=(1.464-3)^2 = (-1.536)^2 = 2.359296.
\]
\end{itemize}

\section*{Assumptions}
\begin{assumption}[Convex and Lipschitz losses]
For each $t\in\{1,\dots,T\}$, $f_t:\mathcal{W}\to \R$ is convex and admits a subgradient at $w_t$ with a uniform bound $\|g_t\| \le G$ for some $G>0$.
\end{assumption}
\begin{assumption}[Bounded convex decision set]
$\mathcal{W}\subset \R^d$ is nonempty, closed, convex, and bounded with diameter
\[
D \;=\; \sup_{x,y\in \mathcal{W}} \|x-y\| \;<\; \infty.
\]
\end{assumption}
\begin{assumption}[Projection]
$\Pi_{\mathcal{W}}(x) \in \arg\min_{w\in \mathcal{W}} \|w-x\|$ denotes the Euclidean projection onto $\mathcal{W}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Regret of OGD with constant stepsize]
\label{thm:ogd}
Under the above assumptions, let $w_{t+1}=\Pi_{\mathcal{W}}(w_t-\eta g_t)$ with any $\eta>0$, and choose any $w^\star \in \arg\min_{w\in \mathcal{W}} \sum_{t=1}^T f_t(w)$. Then
\[
R_T \;=\; \sum_{t=1}^T \bigl(f_t(w_t)-f_t(w^\star)\bigr)
\;\le\; \frac{\|w_1-w^\star\|^2}{2\eta} + \frac{\eta}{2}\sum_{t=1}^T \|g_t\|^2
\;\le\; \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}.
\]
In particular, with $\eta^\star = \frac{D}{G\sqrt{T}}$,
\[
R_T \;\le\; D G \sqrt{T}.
\]
The same bound holds in expectation if randomness is present (in the algorithm or the environment).
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=2em]
\item Stability: The projection $\Pi_{\mathcal{W}}$ is non-expansive, ensuring the iterates remain in $\mathcal{W}$ and do not move excessively between rounds (formalized in Lemma~\ref{lem:projection}).
\item Hyperparameter sensitivity: The regret upper bound $\frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}$ is convex in $\eta$ and minimized at $\eta^\star = D/(G\sqrt{T})$. Overly large $\eta$ inflates the variance term $\frac{\eta G^2 T}{2}$; overly small $\eta$ inflates the bias term $\frac{D^2}{2\eta}$.
\item Comparison to baselines: Any static comparator $u\in\mathcal{W}$ suffers no-regret asymptotically since $R_T/T \to 0$ as $T\to\infty$. The $\mathcal{O}(\sqrt{T})$ rate matches lower bounds for adversarial OCO with Lipschitz convex losses under Euclidean geometry.
\end{itemize}

\section*{Discussion}
The algorithm optimizes decisions online from sequential gradients and aims for sublinear regret rather than convergence to a single fixed point. The bound in Theorem~\ref{thm:ogd} guarantees average regret $R_T/T = \mathcal{O}(1/\sqrt{T})$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Regret]
For any comparator $u\in\mathcal{W}$, the (static) regret is
\[
R_T(u) \;=\; \sum_{t=1}^T \bigl(f_t(w_t) - f_t(u)\bigr).
\]
We focus on $R_T = \min_{u\in \mathcal{W}} R_T(u)$ achieved by $w^\star$.
\end{definition}

\begin{lemma}[Projection non-expansiveness]
\label{lem:projection}
For any $a,b\in \R^d$ and any nonempty closed convex $\mathcal{W}\subset \R^d$,
\[
\bigl\|\Pi_{\mathcal{W}}(a) - \Pi_{\mathcal{W}}(b)\bigr\| \;\le\; \|a-b\|.
\]
\end{lemma}
\begin{proof}
Let $x=\Pi_{\mathcal{W}}(a)$ and $y=\Pi_{\mathcal{W}}(b)$. By the first-order optimality condition for Euclidean projection onto a closed convex set, we have
\[
\langle a-x, z-x\rangle \le 0 \quad \forall z\in \mathcal{W}, 
\qquad 
\langle b-y, z-y\rangle \le 0 \quad \forall z\in \mathcal{W}.
\]
In particular, choose $z=y$ in the first inequality and $z=x$ in the second:
\[
\langle a-x, y-x\rangle \le 0, 
\qquad 
\langle b-y, x-y\rangle \le 0.
\]
Adding these inequalities gives
\[
\langle a-b, x-y\rangle - \|x-y\|^2 \;\ge\; 0 \quad \text{[Step 1: add and regroup inner products using bilinearity]}.
\]
Therefore,
\[
\|x-y\|^2 \;\le\; \langle a-b, x-y\rangle \;\le\; \|a-b\|\,\|x-y\|
\quad \text{[Step 2: Cauchy--Schwarz inequality on }\langle a-b, x-y\rangle\text{]}.
\]
If $x\neq y$, divide both sides by $\|x-y\|$ to obtain $\|x-y\|\le \|a-b\|$; if $x=y$, the claim is trivial. \qedhere
\end{proof}

\begin{lemma}[One-step progress inequality]
\label{lem:one-step}
Fix any $u\in \mathcal{W}$. For the OGD update $w_{t+1}=\Pi_{\mathcal{W}}(w_t-\eta g_t)$ with $\eta>0$, we have
\[
\langle g_t, w_t - u\rangle \;\le\; \frac{\|w_t-u\|^2 - \|w_{t+1}-u\|^2}{2\eta} \;+\; \frac{\eta}{2}\,\|g_t\|^2.
\]
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:projection} with $a=w_t-\eta g_t$ and $b=u$,
\[
\|w_{t+1}-u\|^2 \;\le\; \|w_t-\eta g_t - u\|^2 
\quad \text{[Step 3: non-expansiveness of }\Pi_{\mathcal{W}}\text{]}.
\]
Expand the right-hand side using $\|x-y\|^2=\|x\|^2-2\langle x,y\rangle + \|y\|^2$ with $x=w_t-u$ and $y=\eta g_t$:
\begin{align*}
\|w_t-\eta g_t - u\|^2
&= \|w_t-u\|^2 - 2\eta \langle g_t, w_t-u\rangle + \eta^2 \|g_t\|^2 
\quad \text{[Step 4: polarization identity and bilinearity].}
\end{align*}
Combining with the previous inequality and rearranging terms gives
\begin{align*}
-2\eta \langle g_t, w_t-u\rangle 
&\;\ge\; \|w_{t+1}-u\|^2 - \|w_t-u\|^2 - \eta^2 \|g_t\|^2 
\quad \text{[Step 5: move terms across inequality]},\\
\langle g_t, w_t-u\rangle 
&\;\le\; \frac{\|w_t-u\|^2 - \|w_{t+1}-u\|^2}{2\eta} + \frac{\eta}{2}\|g_t\|^2 
\quad \text{[Step 6: divide by }2\eta>0\text{]}. 
\end{align*}
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
\begin{proof}[Proof of Theorem~\ref{thm:ogd}]
Fix $w^\star\in\arg\min_{w\in\mathcal{W}}\sum_{t=1}^T f_t(w)$. For each $t$, by convexity of $f_t$ and any subgradient $g_t\in \partial f_t(w_t)$,
\[
f_t(w_t) - f_t(w^\star) \;\le\; \langle g_t, w_t - w^\star\rangle 
\quad \text{[Step 7: subgradient inequality for convex }f_t\text{]}.
\]
Applying Lemma~\ref{lem:one-step} with $u=w^\star$,
\[
\langle g_t, w_t - w^\star\rangle 
\;\le\; \frac{\|w_t-w^\star\|^2 - \|w_{t+1}-w^\star\|^2}{2\eta} + \frac{\eta}{2}\|g_t\|^2 
\quad \text{[Step 8: invoke Lemma~\ref{lem:one-step}]}.
\]
Summing these inequalities over $t=1$ to $T$ yields
\begin{align*}
\sum_{t=1}^T \bigl(f_t(w_t) - f_t(w^\star)\bigr)
&\le \sum_{t=1}^T \left[\frac{\|w_t-w^\star\|^2 - \|w_{t+1}-w^\star\|^2}{2\eta} + \frac{\eta}{2}\|g_t\|^2\right]
\quad \text{[Step 9: sum both sides]}\\
&= \frac{\|w_1-w^\star\|^2 - \|w_{T+1}-w^\star\|^2}{2\eta} + \frac{\eta}{2}\sum_{t=1}^T \|g_t\|^2 
\quad \text{[Step 10: telescoping sum on norms]}\\
&\le \frac{\|w_1-w^\star\|^2}{2\eta} + \frac{\eta}{2}\sum_{t=1}^T \|g_t\|^2 
\quad \text{[Step 11: drop }-\|w_{T+1}-w^\star\|^2 \le 0\text{]}.
\end{align*}
By Assumption 2, $\|w_1-w^\star\|\le D$, and by Assumption 1, $\|g_t\|\le G$ for all $t$, hence
\begin{align*}
R_T 
&= \sum_{t=1}^T \bigl(f_t(w_t) - f_t(w^\star)\bigr) \\
&\le \frac{D^2}{2\eta} + \frac{\eta}{2} \sum_{t=1}^T G^2 
\quad \text{[Step 12: substitute the uniform bounds]}\\
&= \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2} 
\quad \text{[Step 13: simplify the summation]}.
\end{align*}
This proves the stated regret bound for any $\eta>0$. \qedhere
\end{proof}

\section*{Rate Derivation (With Constants)}
We optimize the upper bound $B(\eta)=\frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}$ over $\eta>0$.
\begin{align*}
\frac{d}{d\eta} B(\eta) 
&= -\frac{D^2}{2\eta^2} + \frac{G^2 T}{2} \quad \text{[Step 14: derivative of }D^2/(2\eta)\text{ is }-D^2/(2\eta^2)],\\
0 &= -\frac{D^2}{2\eta^{\star 2}} + \frac{G^2 T}{2} 
\quad \Rightarrow \quad 
\eta^\star \;=\; \frac{D}{G\sqrt{T}}
\quad \text{[Step 15: solve for the stationary point]}.
\end{align*}
Substitute $\eta^\star$ back into $B(\eta)$:
\begin{align*}
B(\eta^\star) 
&= \frac{D^2}{2}\cdot \frac{G\sqrt{T}}{D} + \frac{1}{2}\cdot \frac{D}{G\sqrt{T}} \cdot G^2 T
\quad \text{[Step 16: substitute }\eta^\star=D/(G\sqrt{T})],\\
&= \frac{DG\sqrt{T}}{2} + \frac{DG\sqrt{T}}{2} 
\;=\; DG\sqrt{T}
\quad \text{[Step 17: simplify products]}.
\end{align*}
Therefore, the regret satisfies $R_T \le DG\sqrt{T}$ with the optimal constant stepsize $\eta^\star$.

Expectation version: If any randomness is present (algorithmic or environmental), the inequality
\[
R_T(\omega) \;\le\; \frac{D^2}{2\eta} + \frac{\eta}{2}\sum_{t=1}^T \|g_t(\omega)\|^2
\]
holds for every outcome $\omega$ in the probability space
\quad \text{[Step 18: pathwise validity of deterministic inequalities]}.
Taking expectations and using linearity of expectation,
\[
\E[R_T] \;\le\; \frac{D^2}{2\eta} + \frac{\eta}{2}\sum_{t=1}^T \E\bigl[\|g_t\|^2\bigr] 
\quad \text{[Step 19: linearity of }\E\text{]},
\]
and if $\|g_t\|\le G$ almost surely, then $\E\|g_t\|^2 \le G^2$, yielding the same bound as above
\quad \text{[Step 20: monotonicity of expectation]}.

\section*{Special Cases}
\subsection*{Strongly Convex Losses}
Assume in addition that each $f_t$ is $\mu$-strongly convex on $\mathcal{W}$ with respect to $\|\cdot\|$, i.e.,
\[
f_t(w) \;\ge\; f_t(u) + \langle g, w-u\rangle + \frac{\mu}{2}\|w-u\|^2
\quad \forall u,w\in\mathcal{W},\; g\in \partial f_t(u)
\quad \text{[Step 21: definition of strong convexity]}.
\]
Then for $g_t\in\partial f_t(w_t)$,
\[
f_t(w_t) - f_t(w^\star) \;\le\; \langle g_t, w_t - w^\star\rangle - \frac{\mu}{2}\|w_t-w^\star\|^2 
\quad \text{[Step 22: rearrange strong convexity inequality]}.
\]
By Lemma~\ref{lem:one-step} (with general $\eta_t>0$ now) and setting $u=w^\star$,
\[
\langle g_t, w_t - w^\star\rangle 
\;\le\; \frac{\|w_t-w^\star\|^2 - \|w_{t+1}-w^\star\|^2}{2\eta_t} + \frac{\eta_t}{2}\|g_t\|^2 
\quad \text{[Step 23: one-step bound with }\eta_t].
\]
Combining the last two displays gives
\begin{align*}
f_t(w_t) - f_t(w^\star)
&\le \frac{\|w_t-w^\star\|^2 - \|w_{t+1}-w^\star\|^2}{2\eta_t} 
    + \frac{\eta_t}{2}\|g_t\|^2 
    - \frac{\mu}{2}\|w_t-w^\star\|^2 
\quad \text{[Step 24: sum of RHS bounds]}\\
&= \left(\frac{1}{2\eta_t} - \frac{\mu}{2}\right)\|w_t-w^\star\|^2 - \frac{1}{2\eta_t}\|w_{t+1}-w^\star\|^2 + \frac{\eta_t}{2}\|g_t\|^2
\quad \text{[Step 25: regroup terms].}
\end{align*}
Choose the canonical decaying stepsize $\eta_t=\frac{1}{\mu t}$. Then 
\[
\frac{1}{2\eta_t} - \frac{\mu}{2} \;=\; \frac{\mu t}{2} - \frac{\mu}{2} \;=\; \frac{\mu}{2}(t-1)
\quad \text{[Step 26: compute coefficient]},\qquad
\frac{1}{2\eta_t} \;=\; \frac{\mu t}{2}
\quad \text{[Step 27: reciprocal of }\eta_t].
\]
Hence,
\begin{align*}
f_t(w_t) - f_t(w^\star)
&\le \frac{\mu}{2}(t-1)\|w_t-w^\star\|^2 - \frac{\mu t}{2}\|w_{t+1}-w^\star\|^2 + \frac{1}{2\mu t}\|g_t\|^2 
\quad \text{[Step 28: substitute }\eta_t]\\
&= \frac{\mu}{2}\Bigl[(t-1)\|w_t-w^\star\|^2 - t\|w_{t+1}-w^\star\|^2\Bigr] + \frac{1}{2\mu t}\|g_t\|^2
\quad \text{[Step 29: factor }\mu/2].
\end{align*}
Summing over $t=1$ to $T$ yields a weighted telescoping:
\begin{align*}
\sum_{t=1}^T \bigl(f_t(w_t) - f_t(w^\star)\bigr)
&\le \frac{\mu}{2}\sum_{t=1}^T \Bigl[(t-1)\|w_t-w^\star\|^2 - t\|w_{t+1}-w^\star\|^2\Bigr]
     + \sum_{t=1}^T \frac{1}{2\mu t}\|g_t\|^2 
\quad \text{[Step 30: sum both sides]}\\
&= \frac{\mu}{2}\Bigl[-T\|w_{T+1}-w^\star\|^2\Bigr]
   + \sum_{t=1}^T \frac{1}{2\mu t}\|g_t\|^2
\quad \text{[Step 31: weighted telescoping; all intermediate terms cancel]}\\
&\le \sum_{t=1}^T \frac{1}{2\mu t}\|g_t\|^2 
\quad \text{[Step 32: drop the nonpositive term }- \frac{\mu}{2} T\|w_{T+1}-w^\star\|^2].
\end{align*}
Using $\|g_t\|\le G$,
\[
R_T \;=\; \sum_{t=1}^T \bigl(f_t(w_t)-f_t(w^\star)\bigr) 
\;\le\; \frac{G^2}{2\mu}\sum_{t=1}^T \frac{1}{t} 
\;\le\; \frac{G^2}{2\mu}\bigl(1+\ln T\bigr)
\quad \text{[Step 33: harmonic sum bound } \sum_{t=1}^T \frac{1}{t}\le 1+\ln T].
\]
Thus, for $\mu$-strongly convex losses and $\eta_t=1/(\mu t)$, OGD achieves $R_T=\mathcal{O}\!\left(\frac{G^2}{\mu}\log T\right)$ regret.
