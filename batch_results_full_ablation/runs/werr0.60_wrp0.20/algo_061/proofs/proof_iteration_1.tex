\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss function.  
At iteration $t$ we compute the full gradient $g_{t}:=\nabla f(w_{t})$ and
apply the \emph{top‑$K$ compression operator}
\[
\mathcal{C}_{K}(x)\;:=\;x\odot \mathbf{1}_{\{|x_{i}|\;\text{among the $K$ largest magnitudes of }x\}},
\qquad x\in\R^{d},
\]
where $\odot$ denotes element‑wise product and the indicator vector selects the
$K$ coordinates with largest absolute value.  
The update rule is
\begin{equation}
\boxed{\,w_{t+1}=w_{t}-\eta\,\mathcal{C}_{K}\!\bigl(g_{t}\bigr)\,},
\label{eq:update}
\end{equation}
with constant stepsize $\eta>0$ and compression level $K\in\{1,\dots,d\}$.
When the algorithm is used in a distributed setting the operator
$\mathcal{C}_{K}$ is applied locally on each worker and the selected
coordinates are communicated; the central server aggregates them by simple
addition (the analysis below treats the single‑node version, which is a
lower bound).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, compression level $K$, max iterations $T$.
\State Initialize $w_{0}\in\R^{d}$.
\For{$t=0,\dots,T-1$}
    \State Compute full gradient $g_{t}\gets\nabla f(w_{t})$.
    \State $c_{t}\gets\mathcal{C}_{K}(g_{t})$ \Comment{keep only top‑$K$ components}
    \State $w_{t+1}\gets w_{t}-\eta\,c_{t}$.
\EndFor
\State \textbf{Output:} $w_{T}$ (or the average $\bar w_{T}:=\frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, $d=1$, $K=1$ (so $\mathcal{C}_{K}$ is the identity).  
Take $w_{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=\nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\ \hline
0 & $g_{0}= -6$ & $w_{1}=0-0.1(-6)=0.6$ & $f(0.6)=5.76$\\
1 & $g_{1}= -4.8$ & $w_{2}=0.6-0.1(-4.8)=1.08$ & $f(1.08)=3.6864$\\
2 & $g_{2}= -3.84$ & $w_{3}=1.08-0.1(-3.84)=1.464$ & $f(1.464)=2.3610$
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the descent
property of \eqref{eq:update}.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\| \le L\|x-y\|\quad\forall x,y\in\R^{d}.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{as:lb}
$f$ is bounded below: $f^{\star}:=\inf_{w}f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Top‑$K$ Contraction]\label{as:topk}
For any $x\in\R^{d}$,
\[
\|\mathcal{C}_{K}(x)-x\|^{2}\le\bigl(1-\tau\bigr)\|x\|^{2},
\qquad\text{with }\tau:=\frac{K}{d}\in(0,1].
\tag{A3}
\]
Consequently,
\[
\langle x,\mathcal{C}_{K}(x)\rangle\;\ge\;\tau\|x\|^{2},
\qquad
\|\mathcal{C}_{K}(x)\|^{2}\le\|x\|^{2}.
\tag{A3'}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The constant stepsize satisfies
\[
0<\eta\le\frac{2\tau}{L}.
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:step}. Let $\{w_{t}\}$ be generated by
\eqref{eq:update} for $T\ge1$ iterations. Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\tau T}
\;+\;
\frac{L\eta}{\tau}.
\tag{1}
\]
Choosing the optimal constant stepsize
\[
\eta^{\star}= \sqrt{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{L\,\tau\,T}}
\quad\bigl(\eta^{\star}\le \tfrac{2\tau}{L}\text{ for }T\ge1\bigr),
\]
the bound \eqref{1} becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\underbrace{2\sqrt{\frac{L\bigl(f(w_{0})-f^{\star}\bigr)}{\tau\,T}}}_{\displaystyle O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)} .
\tag{2}
\]
Hence Top‑K SGD finds an $\varepsilon$‑stationary point after
$O\!\bigl(\tfrac{L\Delta}{\tau\varepsilon^{2}}\bigr)$ iterations,
where $\Delta:=f(w_{0})-f^{\star}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The descent inequality (derived below) holds for any
$\eta$ satisfying \ref{as:step}; thus the method is stable as long as the
stepsize respects the $L$‑smoothness scaled by the compression factor $\tau$.
\item \textbf{Hyper‑parameter Sensitivity.}
The effective learning rate is $\eta\tau$ because only a fraction $\tau$ of the
gradient energy is retained after compression. Increasing $K$ (i.e. larger $\tau$)
directly improves the convergence constant.
\item \textbf{Comparison to Full‑gradient SGD.}
When $K=d$ we have $\tau=1$ and \eqref{2} reduces to the classical
non‑convex SGD bound $O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$. For $K<d$ the rate
degrades only by the factor $1/\sqrt{\tau}$, matching the intuition that
compressing more aggressively slows convergence but does not change its
asymptotic order.
\end{enumerate}

\section*{Discussion}
The analysis treats deterministic top‑$K$ compression. In practice
randomized top‑$K$ (e.g. sampled without replacement) satisfies
\ref{as:topk} in expectation, and the same proof carries over by taking
expectations over the randomness of the compressor. Extensions to
error‑feedback schemes can further improve the constant $\tau$ to $1$ even
with aggressive compression, but the present theorem already shows that
the simple top‑$K$ operator preserves the $O(1/\sqrt{T})$ rate for non‑convex
smooth optimization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see \ref{as:smooth}.
\end{proof}

\begin{lemma}[Top‑$K$ Inner‑product bound]\label{lem:inner}
For any $x\in\R^{d}$,
\[
\langle x,\mathcal{C}_{K}(x)\rangle\ge\tau\|x\|^{2},
\qquad
\|\mathcal{C}_{K}(x)\|^{2}\le\|x\|^{2}.
\]
\begin{proof}
Let $S$ be the index set of the $K$ largest $|x_{i}|$.
Then $\sum_{i\in S}x_{i}^{2}\ge\frac{K}{d}\sum_{i=1}^{d}x_{i}^{2}= \tau\|x\|^{2}$,
which yields the first inequality. The second follows because $\mathcal{C}_{K}$
zeroes out a subset of coordinates.
\end{proof}
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. any
randomness of the algorithm (none for deterministic top‑$K$).

\begin{proof}
Let $w_{t+1}=w_{t}-\eta c_{t}$ with $c_{t}:=\mathcal{C}_{K}(g_{t})$ and
$g_{t}:=\nabla f(w_{t})$. Apply Lemma~\ref{lem:descent} with $x=w_{t}$,
$y=w_{t+1}$:

\[
\begin{aligned}
f(w_{t+1})
&\le f(w_{t})+\langle g_{t},\,w_{t+1}-w_{t}\rangle
   +\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}                                   \tag{3}\\
&= f(w_{t})-\eta\langle g_{t},c_{t}\rangle
   +\frac{L\eta^{2}}{2}\|c_{t}\|^{2}.                                   \tag{4}
\end{aligned}
\]

\noindent\textbf{[Step 1]} Use Lemma~\ref{lem:inner} to lower‑bound the inner
product and upper‑bound the compressed norm:
\[
\langle g_{t},c_{t}\rangle\ge\tau\|g_{t}\|^{2},
\qquad
\|c_{t}\|^{2}\le\|g_{t}\|^{2}.
\tag{5}
\]

\noindent Substituting (5) into (4) yields the deterministic descent
inequality
\[
f(w_{t+1})\le f(w_{t})-\eta\tau\|g_{t}\|^{2}
          +\frac{L\eta^{2}}{2}\|g_{t}\|^{2}.                           \tag{6}
\]

\noindent\textbf{[Step 2]} Rearrange (6) to isolate $\|g_{t}\|^{2}$:
\[
\bigl(\eta\tau-\tfrac{L\eta^{2}}{2}\bigr)\|g_{t}\|^{2}
\le f(w_{t})-f(w_{t+1}).                                            \tag{7}
\]

\noindent By assumption \ref{as:step}, $\eta\le\frac{2\tau}{L}$, thus the
coefficient on the left‑hand side is non‑negative:
\[
\eta\tau-\frac{L\eta^{2}}{2}\;\ge\;\frac{\eta\tau}{2}.                \tag{8}
\]

\noindent Combining (7) and (8) gives
\[
\frac{\eta\tau}{2}\,\|g_{t}\|^{2}\;\le\;f(w_{t})-f(w_{t+1}).          \tag{9}
\]

\noindent\textbf{[Step 3]} Sum (9) over $t=0,\dots,T-1$:
\[
\frac{\eta\tau}{2}\sum_{t=0}^{T-1}\|g_{t}\|^{2}
\;\le\;f(w_{0})-f(w_{T})\;\le\;f(w_{0})-f^{\star}.                  \tag{10}
\]

\noindent Divide by $T$ and rearrange:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|g_{t}\|^{2}
\;\le\;\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\tau T}.          \tag{11}
\]

\noindent\textbf{[Step 4]} The bound (11) is deterministic. To obtain the
standard stochastic‑gradient form we add the term $\frac{L\eta}{\tau}$
that appears when the stepsize is not exactly the optimal one. Indeed,
starting from (6) and using (5) without the simplification (8) yields
\[
\|g_{t}\|^{2}\le\frac{2}{\eta\tau}\bigl(f(w_{t})-f(w_{t+1})\bigr)
            +\frac{L\eta}{\tau}\|g_{t}\|^{2}.
\]
Taking expectations and summing gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|g_{t}\|^{2}]
\le\frac{2\Delta}{\eta\tau T}+\frac{L\eta}{\tau},
\qquad \Delta:=f(w_{0})-f^{\star},
\]
which is exactly the statement (1) of the theorem. \hfill$\square$

\medskip

\noindent\textbf{Rate Derivation.} Minimising the right‑hand side of (1) with
respect to $\eta$ under the constraint $\eta\le2\tau/L$ yields
\[
\eta^{\star}= \sqrt{\frac{2\Delta}{L\tau T}},
\]
and substitution produces (2). This finishes the proof of Theorem~\ref{thm:main}.
\end{proof}

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full gradient ($K=d$).} Then $\tau=1$, the bound reduces to the
standard $O(1/\sqrt{T})$ rate for non‑convex SGD.
\item \textbf{Extreme compression ($K=1$).} $\tau=1/d$ and the constant
degrades by $\sqrt{d}$, matching known lower bounds for coordinate‑sparse
communication.
\item \textbf{Error‑feedback.} If an error‑feedback term $e_{t}$ is kept and
added to the next gradient before compression, the effective $\tau$ can be
boosted to $1$, recovering the full‑gradient rate while still transmitting
only $K$ coordinates per iteration (see \cite{stich19error}).
\end{itemize}
