\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) differentiable loss function.  
At iteration $t$ we have a parameter vector $w_{t}\in\R^{d}$.  
A stochastic gradient $g_{t}$ is obtained by sampling a random mini‑batch
$\xi_{t}$:
\[
g_{t}\;:=\;\nabla f(w_{t};\xi_{t}) .
\]
Instead of transmitting $g_{t}$, each worker applies a (random) quantization
operator $Q:\R^{d}\rightarrow\R^{d}$ satisfying

\[
\boxed{\E[Q(g_{t})\mid w_{t}]=g_{t}},\qquad 
\boxed{\E\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]\leq (1+\omega)\,\|g_{t}\|^{2}}
\tag{1}
\]
for some constant $\omega\ge 0$ that captures the quality of the quantizer.

The global learner updates the parameters with a (possibly time‑varying) step
size $\eta_{t}>0$:
\[
\boxed{w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t})}. \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, learning‑rate schedule
$\{\eta_{t}\}_{t\ge0}$, number of iterations $T$.
\For{$t=0,\dots,T-1$}
    \State Sample a mini‑batch $\xi_{t}$ and compute stochastic gradient 
          $g_{t}= \nabla f(w_{t};\xi_{t})$.
    \State Quantize: $q_{t}=Q(g_{t})$   \hfill\Comment{unbiased, (1)}
    \State Update: $w_{t+1}= w_{t}-\eta_{t}\,q_{t}$   \hfill\Comment{(2)}
\EndFor
\State \textbf{Return} $w_{T}$ (or optionally the averaged iterate
$\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus
$\nabla f(w)=2(w-3)$.  We use a deterministic ``quantizer'' $Q(g)=g$ (so
$\omega=0$) to keep the arithmetic simple.  Parameters:
$w_{0}=0$, constant step size $\eta=0.1$, $T=3$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$ \\
\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$ \\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$ \\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3645$ \\
\end{tabular}
\end{center}
The objective value decreases from $9$ to $2.36$ after three updates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Bounded stochastic gradient variance]\label{ass:var}
For every $t$,
\[
\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2}.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Quantizer variance]\label{ass:quant}
The quantizer $Q$ satisfies (1) with a constant $\omega\ge0$ that is
independent of $t$ and $w_{t}$.
\end{assumption}
\begin{assumption}[Step‑size]\label{ass:stepsize}
The learning rates are constant, $\eta_{t}\equiv\eta$, and satisfy
\[
0<\eta\le\frac{1}{L(1+\omega)}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let the iterates
$\{w_{t}\}$ be generated by QSGD (2).  For any integer $T\ge1$,
the averaged iterate $\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$ satisfies
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta\,T}
\;+\;
2L\eta\,(1+\omega)\,\sigma^{2}
},
\tag{6}
\]
where $f^{\star}\!=\!\inf_{w} f(w)$.  Consequently, choosing
$\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
= O\!\big(\tfrac{1}{\sqrt{T}}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition (5) guarantees that the
quantized update does not diverge even though the gradient is distorted;
the factor $(1+\omega)$ inflates the effective Lipschitz constant.
\item \textbf{Hyper‑parameter sensitivity.}
The bound (6) shows a linear dependence on $\eta$ in the variance term and an
inverse dependence in the bias term.  The optimal $\eta^{\star}
=\sqrt{\frac{2\big(f(w_{0})-f^{\star}\big)}{L(1+\omega)\sigma^{2}}}\,T^{-1/2}$
balances the two contributions.
\item \textbf{Comparison to unquantized SGD.}
When $\omega=0$ (exact communication) the theorem reduces to the classic
non‑convex SGD rate $O(1/\sqrt{T})$.  A larger $\omega$ merely enlarges the
constant factor, confirming that quantization incurs only a mild penalty.
\end{itemize}

\section*{Discussion}
The result matches the best known rates for stochastic first‑order methods on
smooth non‑convex objectives.  The proof hinges on smoothness, unbiasedness of
$Q$, and a simple variance bound (1).  Extensions to diminishing step sizes,
biased quantizers, or momentum are possible with minor modifications.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth (Assumption \ref{ass:smooth}), then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Unbiased quantization variance]\label{lem:quant}
For any random vector $g$, the quantizer $Q$ satisfies
\[
\E\!\big[\|Q(g)\|^{2}\big]\le (1+\omega)\,\E\!\big[\|g\|^{2}\big].
\]
\end{lemma}
\begin{lemma}[Tower property]\label{lem:tower}
For any random variables $X,Y$,
\[
\E[X]=\E\big[\E[X\mid Y]\big].
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof, expectations are taken
with respect to all randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{Step 1 (Smoothness expansion).}  
From Lemma \ref{lem:smooth} with $x=w_{t}$ and $y=w_{t+1}=w_{t}-\eta Q(g_{t})$,
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_{t})+\langle\nabla f(w_{t}),\,w_{t+1}-w_{t}\rangle
   +\frac{L}{2}\|w_{t+1}-w_{t}\|^{2} \\
&= f(w_{t})-\eta\langle\nabla f(w_{t}),\,Q(g_{t})\rangle
   +\frac{L\eta^{2}}{2}\|Q(g_{t})\|^{2}. \tag{7}
\end{aligned}
\]

\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $w_{t}$ and use unbiasedness of $Q$ (Assumption \ref{ass:quant}):
\[
\E\!\big[\langle\nabla f(w_{t}),Q(g_{t})\rangle\mid w_{t}\big]
   =\langle\nabla f(w_{t}),\E[Q(g_{t})\mid w_{t}]\rangle
   =\langle\nabla f(w_{t}),\nabla f(w_{t})\rangle
   =\|\nabla f(w_{t})\|^{2}. \tag{8}
\]

\noindent\textbf{Step 3 (Bound the second moment of $Q(g_{t})$).}  
From Lemma \ref{lem:quant} and the variance decomposition,
\[
\begin{aligned}
\E\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]
&= \E\!\big[\|Q(g_{t})-g_{t}+g_{t}\|^{2}\mid w_{t}\big] \\
&= \E\!\big[\|Q(g_{t})-g_{t}\|^{2}\mid w_{t}\big]
   +2\langle g_{t},\E[Q(g_{t})-g_{t}\mid w_{t}]\rangle
   +\|g_{t}\|^{2} \\
&= \E\!\big[\|Q(g_{t})-g_{t}\|^{2}\mid w_{t}\big]+\|g_{t}\|^{2} \\
&\le \omega\|g_{t}\|^{2}+\|g_{t}\|^{2}
   =(1+\omega)\|g_{t}\|^{2}. \tag{9}
\end{aligned}
\]
(Here we used $\E[Q(g_{t})-g_{t}\mid w_{t}]=0$ and the definition of $\omega$
in (1).)

\noindent\textbf{Step 4 (Insert (8)–(9) into (7) and take total expectation).}
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})-\eta\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta^{2}}{2}(1+\omega)\,\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big]. \tag{10}
\end{aligned}
\]

\noindent\textbf{Step 5 (Decompose $\|g_{t}\|^{2}$).}  
Write $g_{t}= \nabla f(w_{t}) + \xi_{t}$ where
$\xi_{t}=g_{t}-\nabla f(w_{t})$ satisfies $\E[\xi_{t}\mid w_{t}]=0$ and
$\E[\|\xi_{t}\|^{2}\mid w_{t}]\le\sigma^{2}$ (Assumption \ref{ass:var}). Then
\[
\begin{aligned}
\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big]
&= \E\!\big[\|\nabla f(w_{t})+\xi_{t}\|^{2}\mid w_{t}\big] \\
&= \|\nabla f(w_{t})\|^{2}
   +2\langle\nabla f(w_{t}),\E[\xi_{t}\mid w_{t}]\rangle
   +\E\!\big[\|\xi_{t}\|^{2}\mid w_{t}\big] \\
&= \|\nabla f(w_{t})\|^{2} + \E\!\big[\|\xi_{t}\|^{2}\mid w_{t}\big] \\
&\le \|\nabla f(w_{t})\|^{2} + \sigma^{2}. \tag{11}
\end{aligned}
\]

\noindent\textbf{Step 6 (Plug (11) into (10)).}
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})-\eta\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta^{2}}{2}(1+\omega)
      \big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big) \\
&= f(w_{t})
   -\underbrace{\Big(\eta-\frac{L\eta^{2}}{2}(1+\omega)\Big)}_{\displaystyle
      \alpha}\|\nabla f(w_{t})\|^{2}
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2}. \tag{12}
\end{aligned}
\]

\noindent\textbf{Step 7 (Choose $\eta$ so that $\alpha\ge\frac{\eta}{2}$).}  
Assumption \ref{ass:stepsize} guarantees $0<\eta\le\frac{1}{L(1+\omega)}$,
hence $L\eta(1+\omega)\le1$ and
\[
\eta-\frac{L\eta^{2}}{2}(1+\omega)
   \ge \eta-\frac{\eta}{2}= \frac{\eta}{2}. \tag{13}
\]
Thus $\alpha\ge \eta/2$.

\noindent\textbf{Step 8 (Take full expectation and sum).}  
Applying the tower property (Lemma \ref{lem:tower}) to (12) and using (13):
\[
\E[f(w_{t+1})]\le \E[f(w_{t})]
   -\frac{\eta}{2}\,\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2}. \tag{14}
\]
Rearrange (14):
\[
\frac{\eta}{2}\,\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \E[f(w_{t})]-\E[f(w_{t+1})]
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2}. \tag{15}
\]

\noindent\textbf{Step 9 (Sum over $t=0,\dots,T-1$).}
\[
\begin{aligned}
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
&\le \sum_{t=0}^{T-1}\big(\E[f(w_{t})]-\E[f(w_{t+1})]\big)
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2} T \\
&= \E[f(w_{0})]-\E[f(w_{T})]
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2} T. \tag{16}
\end{aligned}
\]

\noindent\textbf{Step 10 (Lower bound $f(w_{T})\ge f^{\star}$).}
Since $f^{\star}\le f(w_{T})$, (16) yields
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le f(w_{0})-f^{\star}
   +\frac{L\eta^{2}}{2}(1+\omega)\sigma^{2} T. \tag{17}
\]

\noindent\textbf{Step 11 (Divide by $\eta T$).}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2\big(f(w_{0})-f^{\star}\big)}{\eta\,T}
   + L\eta\,(1+\omega)\sigma^{2}. \tag{18}
\]

\noindent\textbf{Step 12 (Conclusion).}  
Inequality (18) is exactly the bound (6) stated in Theorem
\ref{thm:main}.  Choosing $\eta = \sqrt{\frac{2\big(f(w_{0})-f^{\star}\big)}
{L(1+\omega)\sigma^{2}}}\,T^{-1/2}$ (which respects (5) for sufficiently large
$T$) gives the $O(1/\sqrt{T})$ rate.

\hfill $\square$

\section*{Rate Derivation (With Constants)}
From (18) we define
\[
C_{1}=2\big(f(w_{0})-f^{\star}\big),\qquad
C_{2}=L(1+\omega)\sigma^{2}.
\]
Then for any constant step size $\eta$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{C_{1}}{\eta T}+C_{2}\eta .
\]
Optimizing the RHS over $\eta>0$ gives $\eta^{\star}= \sqrt{C_{1}/(C_{2}T)}$,
leading to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le 2\sqrt{C_{1}C_{2}}\;T^{-1/2}
= O\!\big(T^{-1/2}\big).
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Exact communication ($\omega=0$).}  
The bound collapses to the classic SGD result with constant $C_{2}=L\sigma^{2}$.
\item \textbf{Deterministic gradients ($\sigma^{2}=0$).}  
If the stochastic gradient is exact, (18) reduces to
$\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2\big(f(w_{0})-f^{\star}\big)}{\eta T}$,
i.e. a $O(1/T)$ decay of the gradient norm.
\item \textbf{Diminishing step size $\eta_{t}= \frac{c}{\sqrt{t+1}}$.}  
Repeating the analysis with $\eta_{t}$ yields the same $O(1/\sqrt{T})$ rate
without the need to tune $c$, provided $c\le 1/(L\sqrt{1+\omega})$.
\end{itemize}
