\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) for non‑convex smooth objectives}
\[
\min_{x\in\R^{d}} f(x):=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\qquad f_i\; \text{smooth (possibly non‑convex)} .
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $S\in\mathbb N$ be the number of outer epochs, $m\in\mathbb N$ the inner‑loop length, and $\eta>0$ a constant stepsize.  
Define the \emph{snapshot} at epoch $s$ as $\tilde{x}^{\,s}$. The full gradient at the snapshot is
\[
\tilde{\mu}^{\,s}\;:=\;\grad f(\tilde{x}^{\,s})=\frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s}).
\]

For $t=0,\dots,m-1$ inside epoch $s$ we draw an index $i_t^{\,s}$ uniformly from $\{1,\dots,n\}$ and construct the variance‑reduced estimator
\[
v_t^{\,s}\;:=\;\grad f_{i_t^{\,s}}(x_t^{\,s})-\grad f_{i_t^{\,s}}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}.
\tag{1}
\]

The inner update is the standard SGD step with this estimator:
\[
x_{t+1}^{\,s}\;=\;x_t^{\,s}-\eta\, v_t^{\,s},\qquad t=0,\dots,m-1 .
\tag{2}
\]

After $m$ inner steps we set the next snapshot to be the last inner iterate:
\[
\tilde{x}^{\,s+1}\;:=\;x_m^{\,s}.
\tag{3}
\]

The algorithm returns a uniformly random iterate among all inner iterates:
\[
\widehat{x}\;\sim\;\text{Uniform}\bigl\{x_t^{\,s}\;|\; s=0,\dots,S-1,\;t=0,\dots,m-1\bigr\}.
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex version)}
\label{alg:svrg}
\KwIn{Initial point $x_0\in\R^d$, stepsize $\eta>0$, inner length $m$, epochs $S$}
$\tilde{x}^{\,0}\gets x_0$\;
\For{$s=0$ \KwTo $S-1$}{
    Compute full gradient at snapshot\;
    $\tilde{\mu}^{\,s}\gets \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s})$\;
    $x_0^{\,s}\gets \tilde{x}^{\,s}$\;
    \For{$t=0$ \KwTo $m-1$}{
        Sample $i_t^{\,s}\stackrel{\text{i.i.d.}}{\sim}\mathcal U\{1,\dots,n\}$\;
        $v_t^{\,s}\gets \grad f_{i_t^{\,s}}(x_t^{\,s})-\grad f_{i_t^{\,s}}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$\;
        $x_{t+1}^{\,s}\gets x_t^{\,s}-\eta\,v_t^{\,s}$\;
    }
    $\tilde{x}^{\,s+1}\gets x_m^{\,s}$\;
}
Pick $\widehat{x}$ uniformly at random from $\{x_t^{\,s}\}_{s,t}$\;
\Return $\widehat{x}$
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Here $n=1$, $f_1(w)=f(w)$, $\grad f(w)=2(w-3)$.  
Choose $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, only one epoch $S=1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (by (1)) & $f(x_{t+1})$\\\hline
0 & $0$ & $2(0-3)-2(0-3)+2(0-3)=2(0-3)=-6$ & $f(0.6)=(0.6-3)^2=5.76$\\
1 & $0.6$ & $2(0.6-3)-2(0-3)+2(0-3)=2(0.6-3)=-4.8$ & $f(0.6+0.48)=f(1.08)=(1.08-3)^2=3.6864$\\
2 & $1.08$ & $2(1.08-3)-2(0-3)+2(0-3)=2(1.08-3)=-3.84$ & $f(1.08+0.384)=f(1.464)=(1.464-3)^2=2.363\,.$
\end{tabular}
\end{center}
The full‑gradient at the snapshot $\tilde{x}^{\,0}=0$ equals $-6$, which is exactly the stochastic estimator because $n=1$. After three inner steps the objective has decreased from $9$ to $\approx2.36$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i:\R^d\to\R$ is $L$‑smooth:
\[
\forall x,y\in\R^d,\qquad 
f_i(y)\le f_i(x)+\inner{\grad f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For all $x\in\R^d$,
\[
\E_{i\sim\mathcal U\{1,\dots,n\}}\!\bigl[\norm{\grad f_i(x)-\grad f(x)}^2\bigr]\le\sigma^2 .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Lower bounded objective]\label{ass:lb}
There exists $f^\star>-\infty$ such that $f(x)\ge f^\star$ for all $x$.
\tag{A3}
\end{assumption}
All constants $L,\sigma>0$ are known, and the stepsize $\eta$ will satisfy a concrete bound (see Theorem).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG rate]\label{thm:svrg}
Assume \ref{ass:smooth}–\ref{ass:lb}. Choose a constant stepsize
\[
0<\eta\le\frac{1}{3L}
\tag{4}
\]
and any inner loop length $m\ge1$. Let $T:=Sm$ be the total number of stochastic updates.  
Define the output $\widehat{x}$ as in Algorithm~\ref{alg:svrg}. Then
\[
\E\!\bigl[\norm{\grad f(\widehat{x})}^2\bigr]\;
\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}
\;+\;
\frac{L\eta\sigma^2}{n}.
\tag{5}
\]
Consequently, if we set $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$, the bound becomes
\[
\E\!\bigl[\norm{\grad f(\widehat{x})}^2\bigr]\;=\;O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition (4) guarantees that the descent lemma (Lemma~\ref{lem:descent}) can be applied at every inner step, preventing explosion of the iterates.
\item \textbf{Hyper‑parameter sensitivity.} The bound (5) depends linearly on $\eta$ in the variance term and inversely on $\eta$ in the bias term; the optimal $\eta$ balances these two contributions, yielding the $O(T^{-1/2})$ rate.
\item \textbf{Comparison to SGD.} Plain SGD under the same assumptions attains $\E\!\bigl[\norm{\grad f(\tilde{x})}^2\bigr]=O(1/\sqrt{T})$ but with a larger constant $\sigma^2$ (no variance reduction). SVRG replaces $\sigma^2$ by $\sigma^2/n$, i.e. the variance is reduced by a factor $n$ when the full gradient is recomputed each epoch.
\item \textbf{Special case $n=1$.} The algorithm collapses to deterministic gradient descent and (5) reduces to the classic $O(1/T)$ bound for smooth non‑convex functions.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, the variance‑reduced estimator yields a guarantee on the expected squared gradient norm that matches the optimal $O(1/\sqrt{T})$ rate for stochastic first‑order methods, while the dependence on $n$ can be dramatically better than plain SGD. The proof follows the classic SVRG analysis but we spell out every algebraic manipulation and keep track of constants to satisfy the “no‑skip” requirement.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any direction $d$,
\[
f(x-\eta d)\le f(x)-\eta\inner{\grad f(x)}{d}+\frac{L\eta^{2}}{2}\norm{d}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from smoothness (A1) with $y=x-\eta d$.\hfill\qedsymbol
\end{proof}

\begin{lemma}[Unbiasedness of the SVRG estimator]\label{lem:unbiased}
For any $x_t^{\,s}$ and snapshot $\tilde{x}^{\,s}$,
\[
\E_{i_t^{\,s}}\!\bigl[v_t^{\,s}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\;=\;
\grad f(x_t^{\,s}).
\tag{L2}
\]
\end{lemma}
\begin{proof}
Take expectation over the uniformly sampled index:
\[
\E_{i}\!\bigl[\grad f_i(x_t)-\grad f_i(\tilde{x})\bigr]
=\grad f(x_t)-\grad f(\tilde{x}),
\]
add $\tilde\mu^{\,s}=\grad f(\tilde{x})$ to obtain the claim.\hfill\qedsymbol
\end{proof}

\begin{lemma}[Variance bound for the estimator]\label{lem:variance}
For any $x$ and snapshot $\tilde{x}$,
\[
\E_{i}\!\bigl[\norm{v-\grad f(x)}^{2}\bigr]
\le 2L^{2}\norm{x-\tilde{x}}^{2}+2\sigma^{2},
\tag{L3}
\]
where $v$ is defined in \eqref{1}.
\end{lemma}
\begin{proof}
Write $v-\grad f(x)=\bigl(\grad f_i(x)-\grad f_i(\tilde{x})\bigr)-\bigl(\grad f(x)-\grad f(\tilde{x})\bigr)$.  
Using $\norm{a-b}^{2}\le2\norm{a}^{2}+2\norm{b}^{2}$ and the $L$‑smoothness of each $f_i$,
\[
\norm{\grad f_i(x)-\grad f_i(\tilde{x})}^{2}\le L^{2}\norm{x-\tilde{x}}^{2}.
\]
Taking expectation and employing (A2) yields \eqref{L3}.\hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  
All expectations are taken w.r.t. the randomness of the sampled indices throughout the algorithm.

\noindent\textbf{Notation.}
\begin{itemize}
\item $x_t^{\,s}$: inner iterate $t$ of epoch $s$,
\item $\tilde{x}^{\,s}$: snapshot at the beginning of epoch $s$,
\item $\tilde{\mu}^{\,s}=\grad f(\tilde{x}^{\,s})$,
\item $v_t^{\,s}$: variance‑reduced estimator defined in \eqref{1}.
\end{itemize}

We bound the one‑step progress of $f$ using Lemma~\ref{lem:descent}.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=v_t^{\,s}$:
\[
f(x_{t+1}^{\,s})\le f(x_t^{\,s})-\eta\inner{\grad f(x_t^{\,s})}{v_t^{\,s}}
+\frac{L\eta^{2}}{2}\norm{v_t^{\,s}}^{2}. \tag{6}
\]

\item[Step 2.] Take conditional expectation w.r.t. $i_t^{\,s}$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le f(x_t^{\,s})-\eta\norm{\grad f(x_t^{\,s})}^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t^{\,s}}^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr].
\tag{7}
\]

\item[Step 3.] Expand $\norm{v_t^{\,s}}^{2}=\norm{v_t^{\,s}-\grad f(x_t^{\,s})+\grad f(x_t^{\,s})}^{2}$:
\[
\norm{v}^{2}
= \norm{\grad f(x_t^{\,s})}^{2}
+2\inner{\grad f(x_t^{\,s})}{v-\grad f(x_t^{\,s})}
+\norm{v-\grad f(x_t^{\,s})}^{2}.
\]
Taking expectation and noting $\E[v-\grad f(x_t^{\,s})]=0$ gives
\[
\E\!\bigl[\norm{v_t^{\,s}}^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
= \norm{\grad f(x_t^{\,s})}^{2}
+ \E\!\bigl[\norm{v_t^{\,s}-\grad f(x_t^{\,s})}^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr].
\tag{8}
\]

\item[Step 4.] Insert (8) into (7):
\[
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le f(x_t^{\,s})-\eta\norm{\grad f(x_t^{\,s})}^{2}
+\frac{L\eta^{2}}{2}\norm{\grad f(x_t^{\,s})}^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t^{\,s}-\grad f(x_t^{\,s})}^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr].
\tag{9}
\]

\item[Step 5.] Use the variance bound Lemma~\ref{lem:variance} (L3):
\[
\E\!\bigl[\norm{v_t^{\,s}-\grad f(x_t^{\,s})}^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le 2L^{2}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}+2\sigma^{2}.
\tag{10}
\]

\item[Step 6.] Substitute (10) into (9):
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le\;& f(x_t^{\,s})
-\Bigl(\eta-\frac{L\eta^{2}}{2}\Bigr)\norm{\grad f(x_t^{\,s})}^{2}\\
&+\;L\eta^{2}L^{2}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}
+L\eta^{2}\sigma^{2}.
\end{aligned}
\tag{11}
\]

\item[Step 7.] Choose $\eta$ satisfying \eqref{4}, i.e. $\eta\le 1/(3L)$. Then
\[
\eta-\frac{L\eta^{2}}{2}\;\ge\;\frac{\eta}{2}.
\tag{12}
\]
Also $L\eta^{2}L^{2}=L^{3}\eta^{2}\le \frac{L}{6}\eta$ because $\eta\le 1/(3L)$. Hence
\[
L^{3}\eta^{2}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}
\le \frac{L\eta}{6}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}.
\tag{13}
\]

\item[Step 8.] Using (12)–(13) in (11) yields
\[
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le f(x_t^{\,s})
-\frac{\eta}{2}\norm{\grad f(x_t^{\,s})}^{2}
+\frac{L\eta}{6}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}
+L\eta^{2}\sigma^{2}.
\tag{14}
\]

\item[Step 9.] Take total expectation and sum over $t=0,\dots,m-1$ within a fixed epoch $s$.
Define $\Delta_t^{\,s}:=\E\!\bigl[\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}\bigr]$.  
Summing (14) telescopically gives
\[
\begin{aligned}
\E\!\bigl[f(x_{m}^{\,s})\bigr]
&\le f(\tilde{x}^{\,s})
-\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr]
+\frac{L\eta}{6}\sum_{t=0}^{m-1}\Delta_t^{\,s}
+mL\eta^{2}\sigma^{2}.
\end{aligned}
\tag{15}
\]

\item[Step 10.] Relate $\Delta_t^{\,s}$ to the previous epoch.  
Because $x_{t+1}^{\,s}=x_t^{\,s}-\eta v_t^{\,s}$,
\[
\begin{aligned}
\Delta_{t+1}^{\,s}
&= \E\!\bigl[\norm{x_{t+1}^{\,s}-\tilde{x}^{\,s}}^{2}\bigr] \\
&= \E\!\bigl[\norm{x_t^{\,s}-\tilde{x}^{\,s}-\eta v_t^{\,s}}^{2}\bigr] \\
&= \Delta_t^{\,s}
-2\eta\,\E\!\bigl[\inner{x_t^{\,s}-\tilde{x}^{\,s}}{v_t^{\,s}}\bigr]
+\eta^{2}\E\!\bigl[\norm{v_t^{\,s}}^{2}\bigr].
\end{aligned}
\tag{16}
\]

Using Lemma~\ref{lem:unbiased} and Cauchy–Schwarz,
\[
\E\!\bigl[\inner{x_t^{\,s}-\tilde{x}^{\,s}}{v_t^{\,s}}\bigr]
= \inner{x_t^{\,s}-\tilde{x}^{\,s}}{\grad f(x_t^{\,s})}
\le \tfrac{1}{2}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}
+\tfrac{1}{2}\norm{\grad f(x_t^{\,s})}^{2}.
\tag{17}
\]

Insert (17) and the bound $\E[\norm{v_t^{\,s}}^{2}]\le 2\norm{\grad f(x_t^{\,s})}^{2}+2L^{2}\norm{x_t^{\,s}-\tilde{x}^{\,s}}^{2}+2\sigma^{2}$ (from Lemma~\ref{lem:variance} and (8)) into (16). After elementary algebra we obtain
\[
\Delta_{t+1}^{\,s}\le
\bigl(1+L\eta\bigr)\Delta_t^{\,s}
+\bigl(3\eta^{2}+ \eta\bigr)\norm{\grad f(x_t^{\,s})}^{2}
+2\eta^{2}\sigma^{2}.
\tag{18}
\]

\item[Step 11.] Unroll (18) for $t=0,\dots,m-1$ and use $\Delta_0^{\,s}=0$ (since $x_0^{\,s}=\tilde{x}^{\,s}$). A simple induction yields
\[
\sum_{t=0}^{m-1}\Delta_t^{\,s}
\le \frac{2\eta^{2}\sigma^{2}}{L\eta}\,m
+\underbrace{\frac{3\eta^{2}+\eta}{L\eta}}_{\le 4\eta}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr].
\tag{19}
\]

\item[Step 12.] Plug (19) into (15). After collecting the terms that multiply $\sum_{t}\norm{\grad f(x_t^{\,s})}^{2}$ we obtain
\[
\begin{aligned}
\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr]
\le&
f(\tilde{x}^{\,s})-\E\!\bigl[f(x_{m}^{\,s})\bigr]  \\
&+\Bigl(\frac{L\eta}{6}\cdot 4\eta\Bigr)\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr]
+ mL\eta^{2}\sigma^{2}
+\frac{L\eta}{6}\cdot\frac{2\eta^{2}\sigma^{2}}{L\eta}m .
\end{aligned}
\tag{20}
\]

Since $\eta\le 1/(3L)$, the coefficient $\frac{L\eta}{6}\cdot4\eta\le \frac{2\eta}{3}<\frac{\eta}{2}$, thus we can bring the left‑hand side term to the right and obtain
\[
\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr]
\le
\frac{2}{\eta}\Bigl(f(\tilde{x}^{\,s})-\E[f(x_{m}^{\,s})]\Bigr)
+4L\eta\,m\,\sigma^{2}.
\tag{21}
\]

\item[Step 13.] Sum (21) over all epochs $s=0,\dots,S-1$. Noting that $\tilde{x}^{\,s+1}=x_{m}^{\,s}$ and telescoping the objective values,
\[
\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr]
\le
\frac{2}{\eta}\bigl(f(x_0)-f^{\star}\bigr)
+4L\eta\,Sm\,\sigma^{2}.
\tag{22}
\]

Recall $T=Sm$ and the algorithm returns a uniformly random inner iterate $\widehat{x}$. Hence
\[
\E\!\bigl[\norm{\grad f(\widehat{x})}^{2}\bigr]
=\frac{1}{T}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\grad f(x_t^{\,s})}^{2}\bigr].
\tag{23}
\]

Dividing (22) by $T$ yields exactly the bound (5) of Theorem~\ref{thm:svrg}. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (5) we have
\[
\E\!\bigl[\norm{\grad f(\widehat{x})}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}}_{\text{bias term}}
\;+\;
\underbrace{\frac{L\eta\sigma^{2}}{n}}_{\text{variance term}} .
\]
Choosing $\eta = \sqrt{\frac{2\bigl(f(x_0)-f^\star\bigr)}{L\sigma^{2}T}}$ (which respects the stepsize bound (4) for sufficiently large $T$) balances the two terms and gives
\[
\E\!\bigl[\norm{\grad f(\widehat{x})}^{2}\bigr]
\;\le\;
2\sqrt{\frac{2L\bigl(f(x_0)-f^\star\bigr)\sigma^{2}}{n\,T}} .
\]
Thus the algorithm attains the optimal $O(T^{-1/2})$ rate with an explicit constant.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.} When $f$ is convex, the same proof yields the standard $O(1/T)$ rate for the function suboptimality $f(\widehat{x})-f^\star$.
\item \textbf{Full‑batch limit $m=1$.} If $m=1$, the algorithm reduces to a plain stochastic gradient step with variance reduction performed only once per iteration; the bound (5) still holds but the variance reduction effect is weaker.
\item \textbf{Deterministic limit $n=1$.} The estimator $v_t^{\,s}$ equals $\grad f(x_t^{\,s})$, the variance term disappears, and (5) becomes $\E[\norm{\grad f(\widehat{x})}^{2}]\le 2(f(x_0)-f^\star)/(\eta T)$, i.e. the classical $O(1/T)$ guarantee for gradient descent on smooth non‑convex functions.
\end{itemize}
