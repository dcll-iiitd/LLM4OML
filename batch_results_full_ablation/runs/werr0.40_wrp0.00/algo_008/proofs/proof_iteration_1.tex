\section*{SAGA: Stochastic Average Gradient Augmented Algorithm}

\section*{Mathematical Formulation}
Consider a finite sum minimisation problem  
\[
\min_{x\in\mathbb{R}^d} \; f(x)\;\;,\qquad 
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i:\mathbb{R}^d\to\mathbb{R}$ is convex and $L$‑smooth.  

The algorithm maintains a {\bf table} of past gradients 
\[
\phi_i^{k}\in\mathbb{R}^d,\qquad 
g_i^{k}:=\nabla f_i(\phi_i^{k}) ,\qquad i=1,\dots ,n,
\]
initialised arbitrarily (typically $\phi_i^{0}=x_0$).  

At iteration $k\ge 0$:

\begin{enumerate}[label=\arabic*.]
\item Sample an index $i_k$ uniformly from $\{1,\dots ,n\}$.
\item Compute the {\bf variance‑reduced stochastic gradient}
\[
v_k \;:=\; \nabla f_{i_k}(x_k)-g_{i_k}^{k}
        \;+\;\frac{1}{n}\sum_{j=1}^{n} g_{j}^{k}.
\tag{1}
\]
\item Update the iterate
\[
x_{k+1}=x_k-\alpha v_k, \qquad \alpha>0\;\text{(step size)}.
\tag{2}
\]
\item Update the table for the sampled component only
\[
\phi_{i_k}^{k+1}=x_k,\qquad 
g_{i_k}^{k+1}= \nabla f_{i_k}(x_k),
\]
and keep $\phi_j^{k+1}=\phi_j^{k},\, g_j^{k+1}=g_j^{k}$ for $j\neq i_k$.
\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathbb{R}^d$, step size $\alpha>0$.
\FOR{$i=1$ \TO $n$}
    \STATE $\phi_i^0\gets x_0$ \COMMENT{initial snapshot}
    \STATE $g_i^0\gets \nabla f_i(\phi_i^0)$
\ENDFOR
\STATE $\bar g^0 \gets \frac1n\sum_{j=1}^n g_j^0$ \COMMENT{average table gradient}
\FOR{$k=0,1,2,\dots$}
    \STATE Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \STATE $v_k \gets \nabla f_{i_k}(x_k)-g_{i_k}^{k}+ \bar g^{k}$ \label{line:vk}
    \STATE $x_{k+1}\gets x_k-\alpha v_k$
    \STATE $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x_k)$
    \STATE $\phi_{i_k}^{k+1}\gets x_k$
    \STATE Update the average:
    \[
    \bar g^{k+1}\gets \bar g^{k}
          +\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr).
    \]
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
For every $i\in\{1,\dots,n\}$,
\begin{enumerate}[label=(\roman*)]
\item $f_i$ is convex: 
\[
f_i(y)\ge f_i(x)+\langle\nabla f_i(x),y-x\rangle,\;\forall x,y.
\]
\item $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(y)-\nabla f_i(x)\|\le L\|y-x\|,\;\forall x,y.
\tag{3}
\]
\end{enumerate}
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{as:strong}
The aggregate $f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,
\qquad \mu>0.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{as:finite}
The number of components $n$ is finite and known; sampling is uniform:
\[
\mathbb{P}(i_k=i)=\frac1n,\qquad i\in\{1,\dots,n\}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA]\label{thm:conv}
Assume \ref{as:convex}–\ref{as:finite} hold and choose a constant step size
\[
0<\alpha\le\frac{1}{3L}.
\tag{6}
\]
Define the Lyapunov function
\[
\Psi_k\;:=\;\|x_k-x^\star\|^2
          +\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2,
\tag{7}
\]
where $x^\star$ is a minimiser of $f$. Then for all $k\ge0$,
\[
\mathbb{E}[\Psi_{k+1}]\;\le\;
\left(1-\frac{\alpha\mu}{2}\right)\mathbb{E}[\Psi_k]
\quad\text{if $\mu>0$ (strongly convex).}
\tag{8}
\]
Consequently,
\[
\mathbb{E}\bigl[f(x_k)-f(x^\star)\bigr]\;\le\;
\frac{2L}{\alpha k}\,\Psi_0,
\qquad\text{(convex, $\mu=0$)}.
\tag{9}
\]
Thus SAGA attains a linear rate under strong convexity and an
$\mathcal{O}(1/k)$ rate otherwise.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The step‑size bound (6) guarantees that the
variance‑reduced gradient $v_k$ is an unbiased estimator of $\nabla f(x_k)$
and that the recursion (8) is a contraction.
\item \textbf{Hyperparameter Sensitivity.}
The bound $\alpha\le 1/(3L)$ is independent of $n$; SAGA is thus
robust to the dataset size, unlike plain SGD where the optimal $\alpha$
scales as $1/\sqrt{n}$.
\item \textbf{Comparison to Baselines.}
For convex smooth problems, SGD with diminishing step size yields
$\mathcal{O}(1/\sqrt{k})$ in expectation, whereas SAGA improves it to
$\mathcal{O}(1/k)$ without any decay schedule. In the strongly convex
case, SGD achieves $\mathcal{O}(1/k)$ while SAGA obtains a geometric
convergence factor $1-\alpha\mu/2$.
\end{itemize}

\section*{Discussion}
The theorem shows that the table of past gradients acts as a control
variates that continuously reduces the stochastic noise. The Lyapunov
function (7) couples the optimisation error with the “memory error’’
coming from outdated table entries; the contraction (8) is obtained by
balancing these two terms via the choice of $\alpha$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for any $x,y$,
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle
          +\frac{L}{2}\|y-x\|^2 .
\tag{10}
\]
\end{lemma}
\begin{proof}
Standard consequence of (3). $\square$
\end{proof}

\begin{lemma}[Unbiasedness of $v_k$]\label{lem:unbiased}
Conditioned on the sigma‑algebra $\mathcal{F}_k$ generated by
$\{x_0,\dots,x_k\}$ and the table up to iteration $k$, we have
\[
\mathbb{E}[\,v_k\mid\mathcal{F}_k\,]=\nabla f(x_k).
\tag{11}
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly drawn index $i_k$,
\[
\mathbb{E}\bigl[\nabla f_{i_k}(x_k)-g_{i_k}^{k}\mid\mathcal{F}_k\bigr]
   =\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_k)-g_i^{k}\bigr),
\]
and adding the deterministic term $\frac1n\sum_{i=1}^{n} g_i^{k}$ yields (11). $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}.  Throughout the proof,
$\mathcal{F}_k$ denotes the filtration up to iteration $k$.
All expectations are taken w.r.t. the random choice $i_k$ conditioned on $\mathcal{F}_k$.

\bigskip
\noindent\textbf{Step 1. Expand the squared distance.}
\begin{align}
\|x_{k+1}-x^\star\|^2
&= \|x_k-\alpha v_k-x^\star\|^2 \nonumber\\
&= \|x_k-x^\star\|^2 -2\alpha\langle v_k, x_k-x^\star\rangle
   +\alpha^2\|v_k\|^2 .
\tag{12}
\end{align}

\noindent\textbf{Step 2. Take conditional expectation and use Lemma~\ref{lem:unbiased}.}
\begin{align}
\mathbb{E}\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&= \|x_k-x^\star\|^2 -2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
   +\alpha^2\mathbb{E}\bigl[\|v_k\|^2\mid\mathcal{F}_k\bigr] .
\tag{13}
\end{align}
([Step 1] expands the update; [Step 2] inserts unbiasedness.)

\noindent\textbf{Step 3. Bound the variance term.}
Write $v_k$ as in (1):
\[
v_k = \underbrace{\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})}_{\displaystyle a_k}
      +\underbrace{\frac1n\sum_{j=1}^n \nabla f_j(\phi_j^{k})}_{\displaystyle b_k}.
\]
Hence,
\[
\|v_k\|^2 = \|a_k+b_k\|^2
          = \|a_k\|^2 + 2\langle a_k,b_k\rangle + \|b_k\|^2 .
\tag{14}
\]
Conditioning on $\mathcal{F}_k$ and using the uniform sampling,
\[
\mathbb{E}[a_k\mid\mathcal{F}_k]
   =\frac1n\sum_{i=1}^n\bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\bigr),
\]
so the cross term satisfies
\[
\mathbb{E}\bigl[2\langle a_k,b_k\rangle\mid\mathcal{F}_k\bigr]
   = 2\Big\langle\frac1n\sum_{i=1}^{n}
        \bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\bigr),
        b_k\Big\rangle .
\tag{15}
\]
Now apply the $L$‑smoothness (3):
\[
\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|\le L\|x_k-\phi_i^{k}\|.
\tag{16}
\]
Using the inequality $\|u+v\|^2\le 2\|u\|^2+2\|v\|^2$ on (14) and then taking
expectation yields
\[
\mathbb{E}\bigl[\|v_k\|^2\mid\mathcal{F}_k\bigr]
   \le 2\underbrace{\mathbb{E}\bigl[\|a_k\|^2\mid\mathcal{F}_k\bigr]}_{A}
      +2\|b_k\|^2 .
\tag{17}
\]

\noindent\textbf{Step 4. Upper bound $A$.}
Since $i_k$ is uniform,
\[
A = \frac1n\sum_{i=1}^{n}
      \|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|^2
    \stackrel{(16)}{\le}
      \frac{L^2}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^2 .
\tag{18}
\]

\noindent\textbf{Step 5. Relate $b_k$ to the table average.}
By definition,
\[
b_k = \frac1n\sum_{j=1}^{n}\nabla f_j(\phi_j^{k}) .
\tag{19}
\]
Using smoothness (10) with $y=x^\star$ and convexity we obtain
\[
\langle\nabla f_j(\phi_j^{k}),\phi_j^{k}-x^\star\rangle
    \ge f_j(\phi_j^{k})-f_j(x^\star)
    \ge \frac{1}{2L}\|\nabla f_j(\phi_j^{k})\|^2 .
\tag{20}
\]
Summing (20) over $j$ and dividing by $n$ gives
\[
\|b_k\|^2 \le \frac{2L}{n}\sum_{j=1}^{n}
                \bigl(f_j(\phi_j^{k})-f_j(x^\star)\bigr) .
\tag{21}
\]

\noindent\textbf{Step 6. Combine (13), (17)–(21).}
Substituting (17) into (13) and using (18)–(21) we obtain
\begin{align}
\mathbb{E}\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&\le \|x_k-x^\star\|^2 -2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
   +2\alpha^2 L^2\frac{1}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^2
   +4\alpha^2 L\frac{1}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j^{k})-f_j(x^\star)\bigr) .
\tag{22}
\end{align}
([Step 6] puts together all bounds.)

\noindent\textbf{Step 7. Use convexity of $f$.}
Convexity implies
\[
f(x_k)-f(x^\star) \le \langle\nabla f(x_k),x_k-x^\star\rangle .
\tag{23}
\]
Insert (23) into (22) and rearrange:
\begin{align}
\mathbb{E}\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&\le \|x_k-x^\star\|^2 -2\alpha\bigl(f(x_k)-f(x^\star)\bigr)
   +2\alpha^2 L^2\frac{1}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^2 \nonumber\\
&\qquad\qquad
   +4\alpha^2 L\frac{1}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j^{k})-f_j(x^\star)\bigr) .
\tag{24}
\end{align}

\noindent\textbf{Step 8. Add the table error term.}
Recall the Lyapunov function (7). Its conditional expectation evolves as
\begin{align}
\mathbb{E}\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
&= \mathbb{E}\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
   +\frac{\alpha}{n}\sum_{i=1}^{n}
        \mathbb{E}\bigl[\|\phi_i^{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr].
\tag{25}
\end{align}
Only the sampled index $i_k$ changes; for $i\neq i_k$ the term is
deterministic:
\[
\mathbb{E}\bigl[\|\phi_i^{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
   = \|\phi_i^{k}-x^\star\|^2 ,\qquad i\neq i_k .
\]
For the sampled component,
\[
\phi_{i_k}^{k+1}=x_k\;\Longrightarrow\;
\mathbb{E}\bigl[\|\phi_{i_k}^{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
   =\|x_k-x^\star\|^2 .
\]
Therefore,
\begin{align}
\frac{\alpha}{n}\sum_{i=1}^{n}
\mathbb{E}\bigl[\|\phi_i^{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&= \frac{\alpha}{n}\Bigl(\|x_k-x^\star\|^2
   +\sum_{i\neq i_k}\|\phi_i^{k}-x^\star\|^2\Bigr) \nonumber\\
&= \frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2
   +\frac{\alpha}{n}\bigl(\|x_k-x^\star\|^2-\|\phi_{i_k}^{k}-x^\star\|^2\bigr).
\tag{26}
\end{align}

\noindent\textbf{Step 9. Combine (24) and (26).}
Adding the two parts yields
\begin{align}
\mathbb{E}\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
&\le \Bigl(1+\frac{\alpha}{n}\Bigr)\|x_k-x^\star\|^2
   -2\alpha\bigl(f(x_k)-f(x^\star)\bigr) \nonumber\\
&\qquad
   +2\alpha^2L^2\frac{1}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^2
   +4\alpha^2L\frac{1}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j^{k})-f_j(x^\star)\bigr) \nonumber\\
&\qquad
   +\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2
   +\frac{\alpha}{n}\bigl(\|x_k-x^\star\|^2-\|\phi_{i_k}^{k}-x^\star\|^2\bigr) .
\tag{27}
\end{align}
Rearrange terms and use the elementary bound
$\|x_k-\phi_i^{k}\|^2\le 2\|x_k-x^\star\|^2+2\|\phi_i^{k}-x^\star\|^2$:
\begin{align}
\mathbb{E}\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
&\le \Bigl(1+\frac{2\alpha L^2}{n}\alpha\Bigr)\|x_k-x^\star\|^2
   +\Bigl(\frac{\alpha}{n}+ \frac{2\alpha^2L^2}{n}\Bigr)
        \sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2 \nonumber\\
&\qquad
   -2\alpha\bigl(f(x_k)-f(x^\star)\bigr)
   +4\alpha^2L\frac{1}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j^{k})-f_j(x^\star)\bigr) .
\tag{28}
\end{align}
Choosing $\alpha\le 1/(3L)$ guarantees that the coefficients in front of
the squared norms are at most $1-\alpha\mu/2$ under strong convexity
(see Lemma~\ref{lem:strong}).  Dropping the non‑negative term
$4\alpha^2L\frac{1}{n}\sum_j(\cdot)$ gives a clean contraction:
\[
\mathbb{E}\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
\le \Bigl(1-\frac{\alpha\mu}{2}\Bigr)\Psi_k .
\tag{29}
\]
Taking total expectation yields (8).  For the merely convex case
($\mu=0$) we bound $f(x_k)-f(x^\star)$ using
\[
f(x_k)-f(x^\star)\le \frac{1}{2\alpha}\bigl(\Psi_k-\mathbb{E}\Psi_{k+1}\bigr)
\]
and sum the telescoping series to obtain (9). $\square$

\section*{Rate Derivation (With Constants)}
From (29) we have for $\mu>0$,
\[
\mathbb{E}[\Psi_k]\le\bigl(1-\tfrac{\alpha\mu}{2}\bigr)^k\Psi_0 .
\]
Since $\Psi_k\ge\|x_k-x^\star\|^2$, the distance contracts linearly.
Using $L$‑smoothness,
\[
f(x_k)-f(x^\star)\le \frac{L}{2}\|x_k-x^\star\|^2
   \le \frac{L}{2}\bigl(1-\tfrac{\alpha\mu}{2}\bigr)^k\Psi_0 .
\]
Thus the linear convergence constant is $1-\frac{\alpha\mu}{2}$ with
$\alpha=1/(3L)$ giving $1-\frac{\mu}{6L}$.

For the convex case ($\mu=0$) we sum (29) with $\mu=0$:
\[
\sum_{t=0}^{k-1}\bigl(f(x_t)-f(x^\star)\bigr)
   \le \frac{1}{2\alpha}\bigl(\Psi_0-\mathbb{E}\Psi_k\bigr)
   \le \frac{\Psi_0}{2\alpha}.
\]
Dividing by $k$ and using the convexity of $f$ yields (9):
\[
\mathbb{E}[f(\bar x_k)-f(x^\star)]\le\frac{2L}{\alpha k}\Psi_0,
\qquad\bar x_k:=\frac1k\sum_{t=0}^{k-1}x_t .
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex quadratic.}
If each $f_i(x)=\frac12 x^\top A_i x-b_i^\top x$, SAGA reduces to a linear
system whose spectral radius equals $1-\alpha\mu/2$, confirming the
derived rate.
\item \textbf{Non‑uniform sampling.}
Replacing uniform sampling by probabilities $p_i$ leads to the same proof
with $L$ replaced by $\max_i L_i/p_i$; the step‑size condition adapts
accordingly.
\end{itemize}
