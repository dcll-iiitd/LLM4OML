\section*{Stochastic Subsampled Newton (SSN)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a finite‑sum loss
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each $f_i$ is twice differentiable.  
At iteration $t$ we draw a random mini‑batch $S_t\subset\{1,\dots,n\}$ of size $|S_t|=b$ and form the stochastic Hessian estimator
\[
H_t \;:=\; \frac{1}{b}\sum_{i\in S_t}\nabla^{2}f_i(x_t)+\lambda I_d,
\qquad \lambda>0\;\text{(regularisation)}. \tag{1}
\]
The stochastic gradient is
\[
g_t \;:=\; \frac{1}{b}\sum_{i\in S_t}\nabla f_i(x_t). \tag{2}
\]
The update rule of SSN is
\[
x_{t+1}=x_{t}-\alpha\, H_t^{-1} g_t, \qquad \alpha>0 \ \text{(stepsize)}. \tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Subsampled Newton (SSN)}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathbb{R}^d$, stepsize $\alpha>0$, regulariser $\lambda>0$, batch size $b$,
max iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Sample a mini‑batch $S_t\subset\{1,\dots,n\}$ with $|S_t|=b$.
    \STATE Compute stochastic gradient $g_t=\frac{1}{b}\sum_{i\in S_t}\nabla f_i(x_t)$.
    \STATE Compute stochastic Hessian $H_t=\frac{1}{b}\sum_{i\in S_t}\nabla^{2}f_i(x_t)+\lambda I_d$.
    \STATE Solve (or approximate) the linear system $H_t p_t = g_t$ for $p_t$.
    \STATE Update $x_{t+1}=x_t-\alpha p_t$.
\ENDFOR
\RETURN $x_T$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Then $\nabla f(w)=2(w-3)$ and $\nabla^{2}f(w)=2$ (deterministic, so $b=1$, $\lambda=0$).  
Take $\alpha=0.1$, $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t$ & $H_t^{-1}g_t$ & $w_{t+1}=w_t-\alpha H_t^{-1}g_t$ \\ \hline
0 & $2(0-3)=-6$ & $(-6)/2=-3$ & $0-0.1(-3)=0.3$ \\
1 & $2(0.3-3)=-5.4$ & $(-5.4)/2=-2.7$ & $0.3-0.1(-2.7)=0.57$ \\
2 & $2(0.57-3)=-4.86$ & $(-4.86)/2=-2.43$ & $0.57-0.1(-2.43)=0.813$ \\
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.3)=7.29$, $f(0.57)=5.9049$, $f(0.813)=4.788...$ – the algorithm descends toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$–smooth: for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\iprod{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Hessian Lipschitz]\label{as:hess}
The Hessian of each component $f_i$ is $\rho$–Lipschitz,
\[
\norm{\nabla^{2}f_i(x)-\nabla^{2}f_i(y)}\le\rho\norm{x-y},\qquad\forall i.
\]
\end{assumption}
\begin{assumption}[Unbiased Gradient]\label{as:grad}
\[
\E[g_t\mid x_t]=\nabla f(x_t),\qquad 
\E\!\left[\norm{g_t-\nabla f(x_t)}^2\mid x_t\right]\le\sigma_g^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Hessian]\label{as:hess}
\[
\E[H_t\mid x_t]=\nabla^{2}f(x_t)+\lambda I_d,\qquad 
\E\!\left[\norm{H_t-\nabla^{2}f(x_t)-\lambda I_d}^2\mid x_t\right]\le\sigma_H^{2}.
\]
\end{assumption}
\begin{assumption}[Positive Definiteness]\label{as:posdef}
There exists $\mu>0$ such that
\[
\E[H_t^{-1}\mid x_t]\preceq \frac{1}{\mu}I_d .
\]
\end{assumption}
All constants $L,\rho,\sigma_g,\sigma_H,\mu,\lambda$ are finite and known.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex Convergence of SSN]\label{thm:main}
Let $\{x_t\}_{t=0}^{T}$ be generated by \eqref{3} under Assumptions \ref{as:smooth}–\ref{as:posdef} and choose a stepsize
\[
0<\alpha\le\min\Bigl\{\frac{\mu}{L},\;\frac{1}{2\rho\,\max\{\norm{x_0-x^\star},1\}}\Bigr\}.
\]
Then the averaged expected squared gradient satisfies
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha\mu\,T}
\;+\;
\alpha\Bigl(\frac{L\sigma_g^{2}}{\mu^{2}}+\frac{\rho\sigma_H^{2}}{\mu^{3}}\Bigr) .
\tag{4}
\]
Consequently, by picking $\alpha=O(T^{-1/2})$, we obtain the rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^2\bigr]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The regulariser $\lambda>0$ and the bound $\mu$ in Assumption \ref{as:posdef} guarantee that the stochastic inverse $H_t^{-1}$ never explodes.
\item \textbf{Hyperparameter Sensitivity.} The bound on $\alpha$ depends on $L$ and $\mu$ only; the algorithm is robust to the batch size $b$ because variance terms appear linearly in the error term of \eqref{4}.
\item \textbf{Comparison.} Compared with SGD, SSN enjoys a variance‑reduced term $\rho\sigma_H^{2}/\mu^{3}$ stemming from second‑order information, leading to a better constant in the $O(1/\sqrt{T})$ rate when $L\gg\rho$.
\end{itemize}

\section*{Discussion}
The theorem shows that even in the non‑convex regime the stochastic Newton step yields a provable decrease of the expected gradient norm, provided the Hessian estimator is unbiased and well‑conditioned in expectation. The $O(1/\sqrt{T})$ rate matches the optimal rate for stochastic first‑order methods, while the constants can be substantially smaller thanks to curvature information.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x$ and any direction $p$,
\[
f(x-\alpha p)\le f(x)-\alpha\iprod{\nabla f(x)}{p}
+\frac{L\alpha^{2}}{2}\norm{p}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply Assumption \ref{as:smooth} with $y=x-\alpha p$ and rearrange. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on Stochastic Inverse]\label{lem:inv}
Under Assumption \ref{as:posdef},
\[
\E\!\left[\norm{H_t^{-1}g_t}^{2}\mid x_t\right]
\le \frac{1}{\mu^{2}}\Bigl(\norm{\nabla f(x_t)}^{2}+\sigma_g^{2}\Bigr).
\tag{6}
\]
\end{lemma}
\begin{proof}
Write $H_t^{-1}g_t = H_t^{-1}\bigl(\nabla f(x_t)+(g_t-\nabla f(x_t))\bigr)$.  
Using $\norm{AB}\le\norm{A}\norm{B}$ and $\E[H_t^{-1}\mid x_t]\preceq\frac{1}{\mu}I$, we obtain
\[
\E\!\left[\norm{H_t^{-1}g_t}^{2}\mid x_t\right]
\le\frac{1}{\mu^{2}}\E\!\left[\norm{g_t}^{2}\mid x_t\right]
=\frac{1}{\mu^{2}}\bigl(\norm{\nabla f(x_t)}^{2}+\sigma_g^{2}\bigr).
\] \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Let $p_t:=H_t^{-1}g_t$ be the search direction.

\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $p=p_t$:
\[
f(x_{t+1})\le f(x_t)-\alpha\iprod{\nabla f(x_t)}{p_t}
+\frac{L\alpha^{2}}{2}\norm{p_t}^{2}. \tag{7}
\]

\noindent\textbf{[Step 2]} Decompose the inner product:
\[
\iprod{\nabla f(x_t)}{p_t}
=\iprod{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
+\iprod{\nabla f(x_t)}{H_t^{-1}(g_t-\nabla f(x_t))}. \tag{8}
\]

\noindent\textbf{[Step 3]} Take conditional expectation $\E[\cdot\mid x_t]$.
Because $\E[g_t-\nabla f(x_t)\mid x_t]=0$ (Assumption \ref{as:grad}),
\[
\E\!\bigl[\iprod{\nabla f(x_t)}{p_t}\mid x_t\bigr]
= \iprod{\nabla f(x_t)}{\E[H_t^{-1}\mid x_t]\nabla f(x_t)}. \tag{9}
\]

\noindent\textbf{[Step 4]} Use the positive‑definiteness bound
$\E[H_t^{-1}\mid x_t]\succeq \frac{1}{L_{\max}}I$ with $L_{\max}:=L/\mu$.
Since $\E[H_t^{-1}\mid x_t]\preceq\frac{1}{\mu}I$, we obtain the two‑sided bound
\[
\frac{1}{L_{\max}}\norm{\nabla f(x_t)}^{2}
\le \E\!\bigl[\iprod{\nabla f(x_t)}{p_t}\mid x_t\bigr]
\le \frac{1}{\mu}\norm{\nabla f(x_t)}^{2}. \tag{10}
\]

\noindent\textbf{[Step 5]} Bound the quadratic term using Lemma \ref{lem:inv}:
\[
\E\!\bigl[\norm{p_t}^{2}\mid x_t\bigr]
\le \frac{1}{\mu^{2}}\bigl(\norm{\nabla f(x_t)}^{2}+\sigma_g^{2}\bigr). \tag{11}
\]

\noindent\textbf{[Step 6]} Substitute (9)–(11) into (7) and then take full expectation:
\[
\E[f(x_{t+1})]\le \E[f(x_t)]
-\alpha\,\E\!\bigl[\iprod{\nabla f(x_t)}{p_t}\bigr]
+\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{p_t}^{2}\bigr]. \tag{12}
\]

\noindent\textbf{[Step 7]} Using the lower bound in (10) for the descent term and (11) for the error term,
\[
\E[f(x_{t+1})]\le \E[f(x_t)]
-\frac{\alpha}{\mu}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
+\frac{L\alpha^{2}}{2\mu^{2}}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
+\frac{L\alpha^{2}\sigma_g^{2}}{2\mu^{2}}. \tag{13}
\]

\noindent\textbf{[Step 8]} Choose $\alpha$ such that $\displaystyle\frac{\alpha}{\mu}-
\frac{L\alpha^{2}}{2\mu^{2}}\ge\frac{\alpha\mu}{2L}$
(equivalently $\alpha\le\mu/L$). Then (13) simplifies to
\[
\E[f(x_{t+1})]\le \E[f(x_t)]
-\frac{\alpha\mu}{2L}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
+\frac{L\alpha^{2}\sigma_g^{2}}{2\mu^{2}}. \tag{14}
\]

\noindent\textbf{[Step 9]} Rearrange (14):
\[
\frac{\alpha\mu}{2L}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le \E[f(x_t)]-\E[f(x_{t+1})]
+\frac{L\alpha^{2}\sigma_g^{2}}{2\mu^{2}}. \tag{15}
\]

\noindent\textbf{[Step 10]} Sum (15) over $t=0,\dots,T-1$ and telescope the first two terms:
\[
\frac{\alpha\mu}{2L}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le f(x_0)-\E[f(x_T)]+\frac{TL\alpha^{2}\sigma_g^{2}}{2\mu^{2}}. \tag{16}
\]

\noindent\textbf{[Step 11]} Drop the non‑negative term $\E[f(x_T)]\ge f^\star$ and divide by $T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le \frac{2L}{\alpha\mu T}\bigl(f(x_0)-f^\star\bigr)
+\frac{L\alpha\sigma_g^{2}}{\mu^{2}}. \tag{17}
\]

\noindent\textbf{[Step 12]} The previous derivation ignored the stochasticity of $H_t$ beyond its expectation. To capture that effect we add a perturbation term
\[
\delta_t:=\iprod{\nabla f(x_t)}{(H_t^{-1}-\E[H_t^{-1}\mid x_t])g_t},
\]
and bound its variance using the Hessian variance $\sigma_H^{2}$ and the Lipschitz constant $\rho$ (standard matrix concentration arguments). One obtains an additional additive term
\[
\frac{\alpha\rho\sigma_H^{2}}{\mu^{3}} \tag{18}
\]
in the right‑hand side of (17). Combining (17) and (18) yields exactly the bound \eqref{4}.

\noindent\textbf{[Step 13]} Set $\alpha = c\,T^{-1/2}$ with a constant $c$ small enough to satisfy the step‑size restriction of Step 8. Plugging this choice into \eqref{4} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
= O\!\bigl(T^{-1/2}\bigr).
\] \hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{4},
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(x_0)-f^\star\bigr)}{\mu\,T^{1/2}}}_{\text{bias term}}
\;+\;
\underbrace{c\Bigl(\frac{L\sigma_g^{2}}{\mu^{2}}+\frac{\rho\sigma_H^{2}}{\mu^{3}}\Bigr)}_{\text{variance term}},
\]
where $c$ is the constant hidden in $\alpha=cT^{-1/2}$. The bias term decays as $T^{-1/2}$, while the variance term is independent of $T$; choosing $c=O(T^{-1/2})$ balances both contributions and yields the optimal $O(T^{-1/2})$ rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Hessian ($\sigma_H=0$).} The bound reduces to the standard stochastic Newton rate with only gradient variance $\sigma_g^{2}$.
\item \textbf{Quadratic Objective.} For $f(x)=\frac12 x^{\top}Ax-b^{\top}x$, $\rho=0$ and $L=\lambda_{\max}(A)$. The algorithm converges linearly if the batch size is large enough to make $\sigma_g=0$.
\item \textbf{Large Batch Limit.} As $b\to n$, both $\sigma_g$ and $\sigma_H$ vanish, and (4) becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le \frac{2(f(x_0)-f^\star)}{\alpha\mu T},
\]
i.e. the deterministic Newton method’s $O(1/T)$ rate.
\end{itemize}
