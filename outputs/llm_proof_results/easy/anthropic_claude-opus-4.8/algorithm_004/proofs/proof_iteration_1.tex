\section*{Algorithm Name}
\textbf{Online Gradient Descent (OGD) with adaptive (strongly-convex) step size.}

The learner faces a sequence of loss functions $f_1, f_2, \dots, f_T$ revealed one at a time. 
At round $t$ the learner picks $w_t \in \mathcal{K}$ (a convex feasible set), then observes $f_t$ 
and suffers loss $f_t(w_t)$. The goal is to minimize the \emph{regret}
\[
R_T = \sum_{t=1}^T f_t(w_t) - \min_{w \in \mathcal{K}} \sum_{t=1}^T f_t(w),
\]
i.e.\ to achieve \emph{sublinear} regret rather than convergence to a fixed minimizer.

%==================================================================
\section*{Mathematical Formulation}

\textbf{Update rule.} With a Euclidean projection $\Pi_{\mathcal{K}}(\cdot)$ onto $\mathcal{K}$,
\[
g_t = \nabla f_t(w_t), \qquad
w_{t+1} = \Pi_{\mathcal{K}}\!\left( w_t - \eta_t\, g_t \right).
\]

\textbf{Step-size schedule.} For $\mu$-strongly convex losses we use
\[
\eta_t = \frac{1}{\mu\, t}.
\]

\textbf{Hyperparameters.}
\begin{itemize}
  \item $\mu > 0$: strong-convexity modulus (lower bound used in the step size).
  \item Feasible set $\mathcal{K} \subseteq \mathbb{R}^d$ convex, closed.
  \item Initialization $w_1 \in \mathcal{K}$.
\end{itemize}

\textbf{Projection property.} Since $\mathcal{K}$ is convex and closed, $\Pi_{\mathcal{K}}$ is nonexpansive:
\[
\|\Pi_{\mathcal{K}}(x) - \Pi_{\mathcal{K}}(y)\| \le \|x - y\|, \qquad \forall x,y .
\]

%==================================================================
\section*{Pseudocode}
\begin{verbatim}
Input: feasible set K, strong-convexity modulus mu > 0, horizon T
Initialize: w_1 in K  (arbitrary)
for t = 1, 2, ..., T do
    Play w_t
    Observe loss function f_t and incur cost f_t(w_t)
    Compute gradient g_t = grad f_t(w_t)
    Set step size eta_t = 1 / (mu * t)
    Update (gradient step):  y_{t+1} = w_t - eta_t * g_t
    Project:                 w_{t+1} = Proj_K( y_{t+1} )
end for
Termination: stop after T rounds (horizon reached)
Output: sequence (w_1, ..., w_T)
\end{verbatim}

\section*{Dry Run Example}
We illustrate the mechanics on a \emph{stationary} loss
$f_t(w) = f(w) = (w-3)^2$ for all $t$, with $\mathcal{K} = \mathbb{R}$ 
(so projection is the identity), $w_0 = 0$, and a \emph{fixed} step
$\eta = 0.1$ (as specified by the dry-run instructions). 
The gradient is $\nabla f(w) = 2(w-3)$.

\begin{itemize}
  \item \textbf{Iteration 1.} $w_0 = 0$.
        \[
        g_0 = 2(0-3) = -6, \qquad
        w_1 = 0 - 0.1\cdot(-6) = 0.6, \qquad
        f(w_1) = (0.6-3)^2 = 5.76 .
        \]
  \item \textbf{Iteration 2.} $w_1 = 0.6$.
        \[
        g_1 = 2(0.6-3) = -4.8, \qquad
        w_2 = 0.6 - 0.1\cdot(-4.8) = 1.08, \qquad
        f(w_2) = (1.08-3)^2 = 3.6864 .
        \]
  \item \textbf{Iteration 3.} $w_2 = 1.08$.
        \[
        g_2 = 2(1.08-3) = -3.84, \qquad
        w_3 = 1.08 - 0.1\cdot(-3.84) = 1.464, \qquad
        f(w_3) = (1.464-3)^2 = 2.359296 .
        \]
\end{itemize}
The iterates $0 \to 0.6 \to 1.08 \to 1.464$ move monotonically toward the 
minimizer $w^\star = 3$, and the objective decreases 
$9 \to 5.76 \to 3.6864 \to 2.3593$, illustrating the descent behavior.

\section*{Assumptions}

\begin{assumption}[Convex feasible set]
$\mathcal{K} \subseteq \mathbb{R}^d$ is nonempty, closed, and convex.
\end{assumption}

\begin{assumption}[Strong convexity]
Each $f_t:\mathcal{K}\to\mathbb{R}$ is differentiable and $\mu$-strongly convex with $\mu>0$:
\[
f_t(y) \ge f_t(x) + \langle \nabla f_t(x), y-x\rangle + \frac{\mu}{2}\|y-x\|^2,
\qquad \forall x,y \in \mathcal{K}.
\]
\end{assumption}

\begin{assumption}[Bounded gradients]
There exists $G>0$ such that
\[
\|\nabla f_t(w)\| \le G, \qquad \forall w\in\mathcal{K},\ \forall t.
\]
\end{assumption}

\noindent\textbf{Comparator.} Let
$w^\star \in \arg\min_{w\in\mathcal{K}} \sum_{t=1}^T f_t(w)$
denote the best fixed decision in hindsight.

%==================================================================
\section*{Main Convergence Theorem}

\begin{theorem}[Logarithmic regret of OGD]
Under Assumptions 1--3, OGD with step size $\eta_t = 1/(\mu t)$ guarantees
\[
R_T = \sum_{t=1}^T f_t(w_t) - \sum_{t=1}^T f_t(w^\star)
\;\le\; \frac{G^2}{2\mu}\bigl(1 + \log T\bigr).
\]
Consequently $R_T = O\!\left(\dfrac{G^2}{\mu}\log T\right)$, and the
\emph{average} regret satisfies
\[
\frac{R_T}{T} \le \frac{G^2(1+\log T)}{2\mu\, T} = O\!\left(\frac{\log T}{T}\right)\to 0.
\]
\end{theorem}

%==================================================================
\section*{Theoretical Properties}

\begin{itemize}
  \item \textbf{Sublinearity.} $R_T = O(\log T)$ is sublinear, hence average regret $\to 0$;
        OGD is a \emph{no-regret} algorithm under strong convexity.
  \item \textbf{Optimality.} For strongly convex online losses, $\Omega(\log T)$ regret is
        a known lower bound, so OGD is rate-optimal in $T$.
  \item \textbf{Hyperparameter sensitivity.} The schedule $\eta_t=1/(\mu t)$ requires only the
        modulus $\mu$. Underestimating $\mu$ (larger steps) increases the constant; the
        dependence is $G^2/\mu$.
  \item \textbf{Comparison to convex-only OGD.} For merely convex losses with $\eta_t \propto 1/\sqrt{t}$
        one gets $R_T = O(\sqrt{T})$; strong convexity improves this exponentially in $T$ to $O(\log T)$.
  \item \textbf{Stability.} The nonexpansive projection guarantees iterates remain in $\mathcal{K}$ and
        the recursion contracts the distance to $w^\star$ aggressively.
\end{itemize}

%==================================================================
\section*{Discussion}
OGD targets adversarial/sequential data: no statistical assumption on $f_t$ is required,
only convexity/strong convexity and bounded gradients. The strongly convex step size 
$1/(\mu t)$ shrinks fast enough to stabilize updates yet slowly enough (harmonic sum)
to leave only a logarithmic cumulative penalty.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Nonexpansiveness of projection]\label{lem:proj}
For closed convex $\mathcal{K}$ and any $x \in \mathbb{R}^d$, $u\in\mathcal{K}$,
\[
\|\Pi_{\mathcal{K}}(x) - u\| \le \|x - u\|.
\]
\end{lemma}
\begin{proof}
The projection satisfies the obtuse-angle (variational) inequality
$\langle x - \Pi_{\mathcal{K}}(x),\, u - \Pi_{\mathcal{K}}(x)\rangle \le 0$ for all $u\in\mathcal{K}$.
Writing $p=\Pi_{\mathcal{K}}(x)$,
\[
\|x-u\|^2 = \|x-p\|^2 + 2\langle x-p,\, p-u\rangle + \|p-u\|^2 
\ge \|p-u\|^2,
\]
since $\langle x-p, p-u\rangle = -\langle x-p, u-p\rangle \ge 0$.
\end{proof}

\begin{lemma}[Strong convexity bound on instantaneous regret]\label{lem:sc}
For each $t$, by Assumption 2 applied with $x=w_t$, $y=w^\star$,
\[
f_t(w_t) - f_t(w^\star) \le \langle \nabla f_t(w_t),\, w_t - w^\star\rangle 
- \frac{\mu}{2}\|w_t - w^\star\|^2 .
\]
\end{lemma}
\begin{proof}
Strong convexity gives
$f_t(w^\star) \ge f_t(w_t) + \langle \nabla f_t(w_t), w^\star - w_t\rangle + \frac{\mu}{2}\|w^\star-w_t\|^2$.
Rearranging,
\[
f_t(w_t)-f_t(w^\star) \le -\langle \nabla f_t(w_t), w^\star-w_t\rangle - \frac{\mu}{2}\|w^\star-w_t\|^2
= \langle \nabla f_t(w_t), w_t-w^\star\rangle - \frac{\mu}{2}\|w_t-w^\star\|^2.
\]
\end{proof}

%==================================================================
\section*{Main Proof (Step-by-Step Derivation)}

Let $d_t := \|w_t - w^\star\|^2$ and $g_t = \nabla f_t(w_t)$.

\medskip
\noindent\textbf{[Step 1]} (Expand the projected update.)
By the update rule $w_{t+1} = \Pi_{\mathcal{K}}(w_t - \eta_t g_t)$ and 
Lemma~\ref{lem:proj} with $x = w_t - \eta_t g_t$, $u = w^\star\in\mathcal{K}$:
\[
\|w_{t+1} - w^\star\|^2 \le \|w_t - \eta_t g_t - w^\star\|^2 .
\]

\noindent\textbf{[Step 2]} (Algebraic expansion of the squared norm.)
\[
\|w_t - \eta_t g_t - w^\star\|^2 
= \|w_t - w^\star\|^2 - 2\eta_t \langle g_t,\, w_t - w^\star\rangle + \eta_t^2 \|g_t\|^2 .
\]
Hence
\[
d_{t+1} \le d_t - 2\eta_t \langle g_t,\, w_t - w^\star\rangle + \eta_t^2 \|g_t\|^2 .
\]

\noindent\textbf{[Step 3]} (Isolate the inner product.)
Rearranging,
\[
\langle g_t,\, w_t - w^\star\rangle 
\le \frac{d_t - d_{t+1}}{2\eta_t} + \frac{\eta_t}{2}\|g_t\|^2 .
\]

\noindent\textbf{[Step 4]} (Insert the strong-convexity regret bound.)
By Lemma~\ref{lem:sc}, $f_t(w_t)-f_t(w^\star) \le \langle g_t, w_t-w^\star\rangle - \frac{\mu}{2}d_t$.
Combining with [Step 3]:
\[
f_t(w_t) - f_t(w^\star)
\le \frac{d_t - d_{t+1}}{2\eta_t} + \frac{\eta_t}{2}\|g_t\|^2 - \frac{\mu}{2} d_t .
\]

\noindent\textbf{[Step 5]} (Apply the gradient bound, Assumption 3.)
Since $\|g_t\|\le G$, $\|g_t\|^2 \le G^2$, so
\[
f_t(w_t) - f_t(w^\star)
\le \frac{d_t - d_{t+1}}{2\eta_t} - \frac{\mu}{2} d_t + \frac{\eta_t}{2} G^2 .
\]

\noindent\textbf{[Step 6]} (Sum over $t = 1,\dots,T$.)
\[
R_T = \sum_{t=1}^T \bigl(f_t(w_t) - f_t(w^\star)\bigr)
\le \underbrace{\sum_{t=1}^T \left( \frac{d_t - d_{t+1}}{2\eta_t} - \frac{\mu}{2} d_t \right)}_{=:A}
\;+\; \underbrace{\frac{G^2}{2}\sum_{t=1}^T \eta_t}_{=:B}.
\]

\noindent\textbf{[Step 7]} (Telescope term $A$ using $\eta_t = 1/(\mu t)$, so $1/\eta_t = \mu t$.)
\[
A = \sum_{t=1}^T \left( \frac{\mu t}{2} d_t - \frac{\mu(t)}{2} d_{t+1}\cdot\frac{1}{1}  \right)
\quad\text{— rewrite carefully below.}
\]
We expand $A$ explicitly:
\[
A = \sum_{t=1}^T \frac{d_t - d_{t+1}}{2\eta_t} - \frac{\mu}{2}\sum_{t=1}^T d_t
  = \frac{1}{2}\sum_{t=1}^T \frac{1}{\eta_t}\bigl(d_t - d_{t+1}\bigr) - \frac{\mu}{2}\sum_{t=1}^T d_t .
\]
With $1/\eta_t = \mu t$,
\[
\frac{1}{2}\sum_{t=1}^T \mu t\,(d_t - d_{t+1})
= \frac{\mu}{2}\sum_{t=1}^T t\,d_t - \frac{\mu}{2}\sum_{t=1}^T t\,d_{t+1}.
\]
Re-index the second sum ($s=t+1$):
\[
\frac{\mu}{2}\sum_{t=1}^T t\,d_{t+1} = \frac{\mu}{2}\sum_{s=2}^{T+1} (s-1)\,d_{s}.
\]
Therefore
\[
\frac{1}{2}\sum_{t=1}^T \mu t (d_t - d_{t+1})
= \frac{\mu}{2}\Bigl[ \sum_{t=1}^T t\,d_t - \sum_{t=2}^{T+1}(t-1)d_t \Bigr]
= \frac{\mu}{2}\Bigl[ 1\cdot d_1 + \sum_{t=2}^{T} \bigl(t-(t-1)\bigr)d_t - T\,d_{T+1} \Bigr].
\]

\noindent\textbf{[Step 8]} (Simplify the telescoped expression.)
Since $t-(t-1)=1$,
\[
\frac{1}{2}\sum_{t=1}^T \mu t (d_t - d_{t+1})
= \frac{\mu}{2}\Bigl[ d_1 + \sum_{t=2}^{T} d_t - T\,d_{T+1} \Bigr]
= \frac{\mu}{2}\Bigl[ \sum_{t=1}^{T} d_t - T\,d_{T+1} \Bigr].
\]
Subtracting $\frac{\mu}{2}\sum_{t=1}^T d_t$ (the second part of $A$):
\[
A = \frac{\mu}{2}\Bigl[ \sum_{t=1}^{T} d_t - T\,d_{T+1} \Bigr] - \frac{\mu}{2}\sum_{t=1}^T d_t
  = -\frac{\mu T}{2} d_{T+1} \le 0,
\]
because $d_{T+1} = \|w_{T+1}-w^\star\|^2 \ge 0$.

\noindent\textbf{[Step 9]} (Discard the nonpositive term $A$.)
\[
R_T \le A + B \le B = \frac{G^2}{2}\sum_{t=1}^T \eta_t = \frac{G^2}{2}\sum_{t=1}^T \frac{1}{\mu t}
= \frac{G^2}{2\mu}\sum_{t=1}^T \frac{1}{t}.
\]

\noindent\textbf{[Step 10]} (Bound the harmonic sum.)
Using $\sum_{t=1}^T \frac{1}{t} \le 1 + \int_1^T \frac{dx}{x} = 1 + \log T$, we obtain
\[
\boxed{\,R_T \le \frac{G^2}{2\mu}\bigl(1 + \log T\bigr).\,}
\]
This completes the proof. $\qquad\blacksquare$

%==================================================================
\section*{Rate Derivation (With Constants)}

From [Step 10],
\[
R_T \le \frac{G^2}{2\mu}(1+\log T)
= \frac{G^2}{2\mu} + \frac{G^2}{2\mu}\log T .
\]
\begin{itemize}
  \item \textbf{Leading constant:} $\dfrac{G^2}{2\mu}$.
  \item \textbf{Cumulative regret:} $R_T = O\!\left(\dfrac{G^2}{\mu}\log T\right)$.
  \item \textbf{Average regret:}
        \[
        \frac{R_T}{T} \le \frac{G^2(1+\log T)}{2\mu T} = O\!\left(\frac{\log T}{T}\right) \xrightarrow[T\to\infty]{} 0.
        \]
\end{itemize}

\section*{Special Cases}

\textbf{(a) Stationary loss ($f_t \equiv f$).}
Then $\sum_t f(w_t) - T f(w^\star) \le \frac{G^2}{2\mu}(1+\log T)$. Dividing by $T$ and using
Jensen / convexity on the average iterate $\bar w_T = \frac1T\sum_t w_t$,
\[
f(\bar w_T) - f(w^\star) \le \frac{1}{T}\sum_{t=1}^T \bigl(f(w_t)-f(w^\star)\bigr)
\le \frac{G^2(1+\log T)}{2\mu T} = O\!\left(\frac{\log T}{T}\right),
\]
recovering a (near) $1/T$ optimization rate for strongly convex offline minimization.

\medskip
\noindent\textbf{(b) Larger constant step underestimating $\mu$.}
If one uses $\eta_t = 1/(\hat\mu t)$ with $\hat\mu \le \mu$, term $A$ remains
nonpositive (the $-\frac{\mu}{2}d_t$ term dominates the telescoping driven by $\hat\mu$ whenever
$\hat\mu \le \mu$), and $B = \frac{G^2}{2\hat\mu}(1+\log T)$, giving regret
$O\!\left(\frac{G^2}{\hat\mu}\log T\right)$ — only the constant degrades.

\medskip
\noindent\textbf{(c) Merely convex losses ($\mu \to 0$).}
The $-\frac{\mu T}{2}d_{T+1}$ cushion vanishes; one instead chooses $\eta_t = D/(G\sqrt{t})$
(with $D=\operatorname{diam}(\mathcal{K})$) and obtains the classical $R_T = O(G D\sqrt{T})$,
showing strong convexity is what upgrades $\sqrt{T}$ to $\log T$.
