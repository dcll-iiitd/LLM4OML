\section*{Algorithm Name}
\begin{center}
\textbf{Primal–Dual Hybrid Gradient (PDHG) Method}\\
(also known as the Chambolle–Pock algorithm)
\end{center}

\section*{Mathematical Formulation}
We consider the convex–concave saddle‑point problem  
\[
\min_{x\in\mathbb R^{n}} \; \max_{y\in\mathbb R^{m}} 
\; \mathcal L(x,y)\;:=\; f(x)+\langle Kx , y\rangle-g^{*}(y),
\tag{1}
\]
where  

\begin{itemize}
\item $f:\mathbb R^{n}\!\to\!\mathbb R\cup\{+\infty\}$ is proper, closed and convex;
\item $g^{*}:\mathbb R^{m}\!\to\!\mathbb R\cup\{+\infty\}$ is the convex conjugate of a proper, closed convex function $g$;
\item $K\in\mathbb R^{m\times n}$ is a linear operator;
\item $\|\cdot\|$ denotes the Euclidean norm and $\|K\|$ the induced operator norm.
\end{itemize}

Define the proximal operators
\[
\prox_{\tau f}(v):=\arg\min_{x}\Big\{f(x)+\tfrac{1}{2\tau}\|x-v\|^{2}\Big\},
\qquad
\prox_{\sigma g^{*}}(w):=\arg\min_{y}\Big\{g^{*}(y)+\tfrac{1}{2\sigma}\|y-w\|^{2}\Big\}.
\]

Given step sizes $\tau>0$, $\sigma>0$ and an extrapolation parameter $\theta\in[0,1]$ satisfying  
\[
\tau\sigma\|K\|^{2}<1,
\tag{2}
\]
the PDHG iterates are
\[
\begin{aligned}
y^{k+1}&=\prox_{\sigma g^{*}}\!\big(y^{k}+\sigma K\bar x^{k}\big), \\
x^{k+1}&=\prox_{\tau f}\!\big(x^{k}-\tau K^{\top}y^{k+1}\big), \\
\bar x^{k+1}&=x^{k+1}+\theta\big(x^{k+1}-x^{k}\big),
\qquad k=0,1,\dots .
\end{aligned}
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{PDHG for \eqref{1}}
\begin{algorithmic}[1]
\Require $f$, $g^{*}$, linear operator $K$, step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, initial $x^{0},y^{0}$.
\State $\bar x^{0}\gets x^{0}$
\For{$k=0,1,\dots,T-1$}
    \State $y^{k+1}\gets\prox_{\sigma g^{*}}\!\big(y^{k}+\sigma K\bar x^{k}\big)$ \hfill\Comment{dual ascent}
    \State $x^{k+1}\gets\prox_{\tau f}\!\big(x^{k}-\tau K^{\top}y^{k+1}\big)$ \hfill\Comment{primal descent}
    \State $\bar x^{k+1}\gets x^{k+1}+\theta\big(x^{k+1}-x^{k}\big)$
\EndFor
\State \textbf{output} $x^{T}$ (or an ergodic average).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate three iterations on a simple scalar problem.  

\medskip
\noindent\textbf{Problem data}
\[
f(x)=(x-3)^{2},\qquad g^{*}(y)=\iota_{\{0\}}(y)\;(=0\text{ if }y=0,\infty\text{ otherwise}),\qquad K=1.
\]
Thus $g^{*}$ forces $y^{k}=0$ for all $k$.  
Choose $\tau=\sigma=0.5$, $\theta=1$, and start from $x^{0}=0$, $y^{0}=0$.

\medskip
\noindent\textbf{Proximal operator of $f$}
For scalar $v$,
\[
\prox_{\tau f}(v)=\arg\min_{x}\Big\{(x-3)^{2}+\frac{1}{2\tau}(x-v)^{2}\Big\}
          =\frac{v+6\tau}{1+2\tau}.
\tag{4}
\]

\medskip
\noindent\textbf{Iterations}
\[
\begin{array}{c|c|c|c}
k & y^{k+1} & x^{k+1}\;(\text{via }(4)) & f\big(x^{k+1}\big) \\ \hline
0 &
\underbrace{\prox_{\sigma g^{*}}(0+\sigma\bar x^{0})}_{=0}=0 &
\frac{x^{0}+6\tau}{1+2\tau}
   =\frac{0+3}{2}=1.5 &
(1.5-3)^{2}=2.25 \\[4pt]
1 &
0 &
\frac{x^{1}+6\tau}{1+2\tau}
   =\frac{1.5+3}{2}=2.25 &
(2.25-3)^{2}=0.5625 \\[4pt]
2 &
0 &
\frac{x^{2}+6\tau}{1+2\tau}
   =\frac{2.25+3}{2}=2.625 &
(2.625-3)^{2}=0.140625
\end{array}
\]
The sequence $\{x^{k}\}$ rapidly approaches the minimiser $x^{\star}=3$ and the objective value decays accordingly.

\section*{Assumptions}
\begin{enumerate}
\item[\textbf{A1}] $f$ and $g^{*}$ are proper, closed and convex.
\item[\textbf{A2}] The linear operator $K$ satisfies $\|K\|<\infty$.
\item[\textbf{A3}] Step sizes $\tau,\sigma>0$ and extrapolation $\theta\in[0,1]$ are chosen such that \eqref{2} holds.
\item[\textbf{A4}] (Optional) Existence of a saddle point $(x^{\star},y^{\star})$ satisfying
\[
0\in\partial f(x^{\star})+K^{\top}y^{\star},
\qquad
0\in\partial g^{*}(y^{\star})-Kx^{\star}.
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/T)$ convergence]\label{thm:main}
Assume \textbf{A1}–\textbf{A3} and that a saddle point $(x^{\star},y^{\star})$ exists.  
Define the ergodic averages
\[
\bar x^{T}:=\frac{1}{T}\sum_{k=1}^{T}x^{k},
\qquad
\bar y^{T}:=\frac{1}{T}\sum_{k=1}^{T}y^{k}.
\]
Then for any $T\ge 1$,
\[
\mathcal G_{T}:=
\sup_{x,y}\Big\{\langle K\bar x^{T},y\rangle-g^{*}(y)+f(\bar x^{T})
          -\langle Kx,y^{\star}\rangle+g^{*}(y^{\star})-f(x)\Big\}
\le \frac{C}{T},
\tag{5}
\]
where the constant
\[
C:=\frac{1}{2\tau}\|x^{0}-x^{\star}\|^{2}
   +\frac{1}{2\sigma}\|y^{0}-y^{\star}\|^{2}.
\]
Consequently $\big(\bar x^{T},\bar y^{T}\big)$ converges to a saddle point and the primal objective satisfies $f(\bar x^{T})-f(x^{\star})=O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{2} guarantees that the combined operator in \eqref{3} is non‑expansive, which is the cornerstone of the proof.
\item \textbf{Hyper‑parameter sensitivity.} The product $\tau\sigma\|K\|^{2}$ must be strictly smaller than one; larger step sizes accelerate convergence but risk violation of \eqref{2}.
\item \textbf{Comparison.} Compared with vanilla gradient descent (rate $O(1/\sqrt{T})$ for nonsmooth $f$), PDHG attains the optimal $O(1/T)$ ergodic rate without requiring smoothness of $f$ or $g$.
\item \textbf{Special cases.} With $g^{*}=0$ (i.e., $g\equiv0$) the method reduces to a proximal gradient scheme; with $\theta=0$ we obtain the classic primal–dual method of Arrow–Hurwicz.
\end{itemize}

\section*{Discussion}
The theorem shows that PDHG enjoys the same $O(1/T)$ ergodic convergence as the best first‑order methods for convex–concave saddle problems, while handling nonsmooth terms via proximal maps. The proof relies only on convexity and the non‑expansiveness of proximal operators; no Lipschitz gradient assumption is needed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Bregman distance]
For a convex, proper function $\phi$, the Bregman distance between $u$ and $v$ is
\[
D_{\phi}(u,v):=\phi(u)-\phi(v)-\langle \partial\phi(v),u-v\rangle .
\]
\end{definition}
\begin{lemma}[Three‑point identity for proximal operators]\label{lem:three}
For any convex $\phi$, $\tau>0$, and points $u,v\in\mathbb R^{n}$,
\[
\langle \prox_{\tau\phi}(u)-\prox_{\tau\phi}(v),\;u-v\rangle
\ge \| \prox_{\tau\phi}(u)-\prox_{\tau\phi}(v)\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
The proximal map is firmly non‑expansive; see e.g. \cite{bauschke2011convex}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Let $(x^{\star},y^{\star})$ be a saddle point. Denote
\[
\Delta_{x}^{k}:=\frac{1}{2\tau}\|x^{k}-x^{\star}\|^{2},
\qquad
\Delta_{y}^{k}:=\frac{1}{2\sigma}\|y^{k}-y^{\star}\|^{2}.
\]

\noindent\textbf{[Step 1]}  Start from the optimality conditions of the proximal updates.
\[
\begin{aligned}
\frac{1}{\tau}\big(x^{k}-x^{k+1}\big)-K^{\top}y^{k+1}&\in\partial f(x^{k+1}),\\
\frac{1}{\sigma}\big(y^{k}-y^{k+1}\big)+K\bar x^{k}&\in\partial g^{*}(y^{k+1}).
\end{aligned}
\tag{7}
\]

\noindent\textbf{[Step 2]}  Use convexity of $f$ and $g^{*}$:
\[
\begin{aligned}
f(x^{k+1})-f(x^{\star})
&\le \big\langle \frac{1}{\tau}(x^{k}-x^{k+1})-K^{\top}y^{k+1},\;x^{k+1}-x^{\star}\big\rangle,\\
g^{*}(y^{k+1})-g^{*}(y^{\star})
&\le \big\langle \frac{1}{\sigma}(y^{k}-y^{k+1})+K\bar x^{k},\;y^{k+1}-y^{\star}\big\rangle .
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 3]}  Expand the inner products and regroup terms.
\[
\begin{aligned}
\langle x^{k}-x^{k+1},x^{k+1}-x^{\star}\rangle
&=\tfrac12\big(\|x^{k}-x^{\star}\|^{2}-\|x^{k+1}-x^{\star}\|^{2}-\|x^{k+1}-x^{k}\|^{2}\big),\\
\langle y^{k}-y^{k+1},y^{k+1}-y^{\star}\rangle
&=\tfrac12\big(\|y^{k}-y^{\star}\|^{2}-\|y^{k+1}-y^{\star}\|^{2}-\|y^{k+1}-y^{k}\|^{2}\big).
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 4]}  Substitute \eqref{9} into \eqref{8} and multiply the first inequality by $1$ and the second by $1$ (no scaling). After simplification we obtain
\[
\begin{aligned}
&\Delta_{x}^{k}-\Delta_{x}^{k+1}+\Delta_{y}^{k}-\Delta_{y}^{k+1}\\
&\qquad\ge
\underbrace{\langle Kx^{k+1},y^{k+1}\rangle-\langle Kx^{\star},y^{k+1}\rangle}_{\text{(a)}}
+\underbrace{\langle K\bar x^{k},y^{k+1}\rangle-\langle Kx^{k+1},y^{\star}\rangle}_{\text{(b)}}
-\frac{1}{2\tau}\|x^{k+1}-x^{k}\|^{2}
-\frac{1}{2\sigma}\|y^{k+1}-y^{k}\|^{2}.
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 5]}  Observe that $\bar x^{k}=x^{k}+\theta(x^{k}-x^{k-1})$. Using the definition of $\theta\in[0,1]$ and the Cauchy–Schwarz inequality together with $\tau\sigma\|K\|^{2}<1$, we can bound the cross term
\[
\langle K(x^{k+1}-\bar x^{k}),y^{k+1}-y^{\star}\rangle
\le
\frac{1}{2\tau}\|x^{k+1}-\bar x^{k}\|^{2}
+\frac{1}{2\sigma}\|y^{k+1}-y^{\star}\|^{2}
-\Bigl(\frac{1}{\tau\sigma\|K\|^{2}}-1\Bigr)\frac{1}{2}\|y^{k+1}-y^{k}\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 6]}  Plug \eqref{11} into \eqref{10} and rearrange to obtain a telescoping inequality
\[
\Delta_{x}^{k}+\Delta_{y}^{k}
-\Delta_{x}^{k+1}-\Delta_{y}^{k+1}
\ge
\mathcal L(x^{k+1},y^{\star})-\mathcal L(x^{\star},y^{k+1}) .
\tag{12}
\]

\noindent\textbf{[Step 7]}  Sum \eqref{12} from $k=0$ to $T-1$; the left‑hand side telescopes:
\[
\Delta_{x}^{0}+\Delta_{y}^{0}
-\Delta_{x}^{T}-\Delta_{y}^{T}
\ge
\sum_{k=0}^{T-1}\big[\mathcal L(x^{k+1},y^{\star})-\mathcal L(x^{\star},y^{k+1})\big].
\tag{13}
\]

\noindent\textbf{[Step 8]}  Divide both sides of \eqref{13} by $T$ and use the convex‑concave structure of $\mathcal L$ to replace the sum by the primal‑dual gap evaluated at the ergodic averages:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\big[\mathcal L(x^{k+1},y^{\star})-\mathcal L(x^{\star},y^{k+1})\big]
\ge
\mathcal L(\bar x^{T},y^{\star})-\mathcal L(x^{\star},\bar y^{T}) .
\tag{14}
\]

\noindent\textbf{[Step 9]}  Combine \eqref{13}–\eqref{14} and drop the non‑negative terms $\Delta_{x}^{T},\Delta_{y}^{T}$ to obtain
\[
\mathcal L(\bar x^{T},y^{\star})-\mathcal L(x^{\star},\bar y^{T})
\le
\frac{\Delta_{x}^{0}+\Delta_{y}^{0}}{T}
= \frac{C}{T},
\tag{15}
\]
which is exactly the bound \eqref{5}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{15} we have for any $x\in\mathbb R^{n}$,
\[
f(\bar x^{T})-f(x) \le \frac{C}{T}+ \langle K\bar x^{T},y^{\star}\rangle-g^{*}(y^{\star})+\langle Kx,y^{\star}\rangle-g^{*}(y^{\star}).
\]
Choosing $x=x^{\star}$ and using the saddle‑point relation $\langle Kx^{\star},y^{\star}\rangle=g^{*}(y^{\star})-g(Kx^{\star})$ yields the primal objective guarantee
\[
f(\bar x^{T})-f(x^{\star})\le \frac{C}{T}.
\]
Thus the primal iterates achieve an $O(1/T)$ convergence rate with constant $C$ defined in Theorem \ref{thm:main}.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure primal proximal gradient.} If $g^{*}\equiv0$ (hence $y^{k}=0$ for all $k$) and $\theta=0$, algorithm \eqref{3} reduces to $x^{k+1}=\prox_{\tau f}(x^{k})$, i.e., the proximal point method with rate $O(1/T)$.
\item \textbf{Linear programming.} For $f=\iota_{\{Ax=b\}}$, $g^{*}=c^{\top}y$, $K=A^{\top}$, PDHG coincides with the classical Arrow–Hurwicz method and inherits the same ergodic rate.
\item \textbf{Accelerated PDHG.} Setting $\theta=\frac{1}{\sqrt{1+\tau\sigma\|K\|^{2}}}$ leads to the accelerated variant of Chambolle–Pock with an improved constant but the same $O(1/T)$ order.
\end{enumerate}
