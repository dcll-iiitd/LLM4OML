\section*{Algorithm Name}
\textbf{Proximal Gradient Descent (PGD) for Non‑Convex Composite Optimization}  

We consider the composite problem  

\[
\min_{x\in\R^{d}} \; \Phi(x)\;:=\; f(x)+g(x),
\]

where $f$ is smooth (possibly non‑convex) and $g$ is proper, lower‑semicontinuous (possibly non‑convex and non‑smooth).  

The algorithm updates a point $x_k$ by taking a gradient step on $f$ followed by a proximal step on $g$.

\section*{Mathematical Formulation}
Given a stepsize $\eta>0$, the iteration reads  

\[
\boxed{
x_{k+1}= \prox_{\eta g}\bigl(x_k-\eta \,\grad f(x_k)\bigr)},
\qquad k=0,1,2,\dots
\tag{1}
\]

The proximal operator of $g$ with parameter $\eta$ is defined by  

\[
\prox_{\eta g}(u):=\arg\min_{z\in\R^{d}}\Bigl\{ g(z)+\frac{1}{2\eta}\norm{z-u}^{2}\Bigr\}.
\tag{2}
\]

\section*{Pseudocode}
\begin{enumerate}[label=Step~\arabic*:, leftmargin=*]
\item \textbf{Input:} initial point $x_0\in\R^{d}$, stepsize $\eta>0$, maximum iterations $K$ (or tolerance $\varepsilon$).
\item \textbf{For} $k=0,\dots,K-1$ \textbf{do}
\begin{enumerate}[label=\arabic*.]
\item Compute the (full) gradient $g_k:=\grad f(x_k)$.
\item Form the gradient step $u_k:=x_k-\eta g_k$.
\item Compute the proximal point 
\[
x_{k+1}:= \prox_{\eta g}(u_k).
\]
\item (Optional) If $\norm{x_{k+1}-x_k}\le \varepsilon$ \textbf{break}.
\end{enumerate}
\item \textbf{Output:} $x_{K}$ (or the last iterate).
\end{enumerate}

\section*{Dry Run Example}
Minimize the scalar function $f(w)=(w-3)^2$ with $g\equiv0$ (so $\prox_{\eta g}(u)=u$).  
Take $w_0=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\grad f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta\grad f(w_k)$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ 
\end{tabular}
\end{center}

Objective values: $\Phi(w_0)=9$, $\Phi(w_1)=5.76$, $\Phi(w_2)=3.6864$, $\Phi(w_3)=2.3630\ldots$ – a clear descent.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f:\R^{d}\to\R$ is continuously differentiable and its gradient is $L$–Lipschitz, i.e.  

\[
\norm{\grad f(x)-\grad f(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Lower boundedness]\label{as:lb}
The composite objective $\Phi$ is bounded below: $\Phi_{\inf}:=\inf_{x\in\R^{d}}\Phi(x)>-\infty$.
\tag{A2}
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies  

\[
0<\eta\le \frac{1}{L}.
\tag{A3}
\]
\end{assumption}

No convexity of $f$ or $g$ is required beyond the above.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of PGD]\label{thm:main}
Let $\{x_k\}$ be generated by \eqref{1} under Assumptions \ref{as:smooth}–\ref{as:stepsize}.  
Define the \emph{gradient mapping}  

\[
\mathcal{G}_{\eta}(x):=\frac{1}{\eta}\bigl(x-\prox_{\eta g}(x-\eta\grad f(x))\bigr).
\tag{3}
\]

Then for every $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\norm{\mathcal{G}_{\eta}(x_k)}^{2}
\;\le\;
\frac{2\bigl(\Phi(x_0)-\Phi_{\inf}\bigr)}{\eta K}.
\tag{4}
\]
Consequently,
\[
\min_{0\le k\le K-1}\norm{\mathcal{G}_{\eta}(x_k)}^{2}
\;\le\;
\frac{2\bigl(\Phi(x_0)-\Phi_{\inf}\bigr)}{\eta K}
=O\!\left(\frac{1}{K}\right).
\tag{5}
\]
In particular, any limit point of $\{x_k\}$ is a \emph{stationary point}, i.e. $0\in\grad f(\bar x)+\partial g(\bar x)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Descent property.} From the proof we obtain the one‑step inequality  

\[
\Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\norm{\mathcal{G}_{\eta}(x_k)}^{2}.
\tag{6}
\]

\item \textbf{Stability w.r.t.\ stepsize.} The bound \eqref{4} holds for any $\eta\in(0,1/L]$; a larger $\eta$ (up to $1/L$) yields a tighter constant $1/\eta$.

\item \textbf{Comparison.} When $g\equiv0$, $\mathcal{G}_{\eta}(x)=\grad f(x)$ and \eqref{4} reduces to the classic $O(1/K)$ guarantee for gradient descent on smooth (possibly non‑convex) functions.  
When $g$ is convex, \eqref{6} coincides with the standard proximal descent lemma.

\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on $\grad f$.
\end{proof}

\begin{lemma}[Non‑expansiveness of the proximal operator]\label{lem:nonexp}
For any $u,v\in\R^{d}$ and any $\eta>0$,
\[
\norm{\prox_{\eta g}(u)-\prox_{\eta g}(v)}\le \norm{u-v}.
\tag{8}
\]
\end{lemma}
\begin{proof}
The proximal map is the resolvent of the sub‑differential $\partial g$, which is firmly non‑expansive; see e.g. \cite{proxbook}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof we fix $k\ge0$ and denote  

\[
u_k:=x_k-\eta\grad f(x_k),\qquad
x_{k+1}:=\prox_{\eta g}(u_k).
\]

\noindent\textbf{[Step 1] Optimality condition of the proximal step.}\\
By definition \eqref{2}, $x_{k+1}$ satisfies  

\[
0\in \partial g(x_{k+1})+\frac{1}{\eta}(x_{k+1}-u_k).
\tag{9}
\]

Equivalently,
\[
\frac{1}{\eta}(u_k-x_{k+1})\in\partial g(x_{k+1}).
\tag{10}
\]

\noindent\textbf{[Step 2] Relating the gradient mapping to the sub‑gradient.}\\
Recall the definition \eqref{3} of $\mathcal{G}_{\eta}$; using $u_k=x_k-\eta\grad f(x_k)$ we obtain  

\[
\mathcal{G}_{\eta}(x_k)=\frac{1}{\eta}\bigl(x_k-x_{k+1}\bigr)
= \grad f(x_k)-\frac{1}{\eta}(u_k-x_{k+1}).
\tag{11}
\]

Insert \eqref{10} into \eqref{11} to get  

\[
\mathcal{G}_{\eta}(x_k)=\grad f(x_k)-s_{k+1},
\qquad s_{k+1}\in \partial g(x_{k+1}).
\tag{12}
\]

Thus $0\in \grad f(x_k)-\mathcal{G}_{\eta}(x_k)+\partial g(x_{k+1})$.

\noindent\textbf{[Step 3] Apply the descent lemma to $f$ at $(x_k,x_{k+1})$.}\\
Using Lemma \ref{lem:descent} with $x=x_k$, $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_k)+\inner{\grad f(x_k)}{x_{k+1}-x_k}
+\frac{L}{2}\norm{x_{k+1}-x_k}^{2}.
\tag{13}
\]

\noindent\textbf{[Step 4] Add $g(x_{k+1})$ to both sides and use convexity of the sub‑gradient inequality.}\\
Since $s_{k+1}\in\partial g(x_{k+1})$, we have  

\[
g(x_{k+1})\le g(x_k)+\inner{s_{k+1}}{x_{k+1}-x_k}.
\tag{14}
\]

Summing \eqref{13} and \eqref{14} yields  

\[
\Phi(x_{k+1})\le \Phi(x_k)
+\inner{\grad f(x_k)+s_{k+1}}{x_{k+1}-x_k}
+\frac{L}{2}\norm{x_{k+1}-x_k}^{2}.
\tag{15}
\]

\noindent\textbf{[Step 5] Substitute the expression from \eqref{12}.}\\
From \eqref{12}, $\grad f(x_k)+s_{k+1}= \mathcal{G}_{\eta}(x_k)$. Hence  

\[
\Phi(x_{k+1})\le \Phi(x_k)
+\inner{\mathcal{G}_{\eta}(x_k)}{x_{k+1}-x_k}
+\frac{L}{2}\norm{x_{k+1}-x_k}^{2}.
\tag{16}
\]

\noindent\textbf{[Step 6] Express $x_{k+1}-x_k$ via the gradient mapping.}\\
From the definition of $\mathcal{G}_{\eta}$,  

\[
x_{k+1}-x_k = -\eta \mathcal{G}_{\eta}(x_k).
\tag{17}
\]

Insert \eqref{17} into \eqref{16}:

\[
\Phi(x_{k+1})\le \Phi(x_k)
-\eta\norm{\mathcal{G}_{\eta}(x_k)}^{2}
+\frac{L\eta^{2}}{2}\norm{\mathcal{G}_{\eta}(x_k)}^{2}.
\tag{18}
\]

\noindent\textbf{[Step 7] Use the stepsize condition $\eta\le 1/L$.}\\
Because $\eta\le 1/L$, we have $L\eta^{2}\le\eta$. Therefore  

\[
-\eta+\frac{L\eta^{2}}{2}\;\le\; -\frac{\eta}{2}.
\tag{19}
\]

Applying \eqref{19} to \eqref{18} gives the one‑step descent inequality  

\[
\boxed{\Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\norm{\mathcal{G}_{\eta}(x_k)}^{2}}.
\tag{20}
\]

This is precisely \eqref{6}.

\noindent\textbf{[Step 8] Telescoping sum.}\\
Summing \eqref{20} from $k=0$ to $K-1$ yields  

\[
\Phi(x_{K})\le \Phi(x_{0})
-\frac{\eta}{2}\sum_{k=0}^{K-1}\norm{\mathcal{G}_{\eta}(x_k)}^{2}.
\tag{21}
\]

Rearranging and using the lower bound $\Phi_{\inf}\le\Phi(x_{K})$ (Assumption \ref{as:lb}) we obtain  

\[
\frac{1}{K}\sum_{k=0}^{K-1}\norm{\mathcal{G}_{\eta}(x_k)}^{2}
\le \frac{2\bigl(\Phi(x_{0})-\Phi_{\inf}\bigr)}{\eta K},
\tag{22}
\]

which is exactly the claim \eqref{4} of Theorem \ref{thm:main}.  

\noindent\textbf{[Step 9] Convergence of the gradient mapping.}\\
Since the right‑hand side of \eqref{22} tends to zero as $K\to\infty$, the sequence $\{\norm{\mathcal{G}_{\eta}(x_k)}\}$ possesses a subsequence converging to $0$. Any limit point $\bar x$ of $\{x_k\}$ then satisfies $0\in\grad f(\bar x)+\partial g(\bar x)$, i.e. $\bar x$ is a stationary point.

The theorem is proved. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{22} we can write an explicit bound for the \emph{best} iterate among the first $K$ steps:

\[
\min_{0\le k\le K-1}\norm{\mathcal{G}_{\eta}(x_k)}^{2}
\le \frac{2\bigl(\Phi(x_{0})-\Phi_{\inf}\bigr)}{\eta K}.
\]

If one defines the \emph{stationarity measure} $\epsilon_K:=\min_{0\le k<K}\norm{\mathcal{G}_{\eta}(x_k)}$, then to guarantee $\epsilon_K\le\varepsilon$ it suffices to run  

\[
K\ge \frac{2\bigl(\Phi(x_{0})-\Phi_{\inf}\bigr)}{\eta\varepsilon^{2}}
\]

iterations. The dependence on $1/\eta$ reflects the stepsize sensitivity discussed in Section 5.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal step becomes the identity, $\mathcal{G}_{\eta}(x)=\grad f(x)$, and \eqref{20} reduces to the classic smooth‑nonconvex descent lemma $f(x_{k+1})\le f(x_k)-\frac{\eta}{2}\norm{\grad f(x_k)}^{2}$.
\item \textbf{Convex $g$.} The same bound holds; however, under additional convexity of $g$ one can also prove that the function values $\Phi(x_k)$ converge to the global optimum.
\item \textbf{Indicator of a convex set $C$.} If $g=\iota_{C}$, then $\prox_{\eta g}$ is the Euclidean projection onto $C$. The algorithm becomes projected gradient descent and the proof above specializes to the standard projected‑gradient convergence result.
\end{enumerate}

\section*{Discussion}
The proof relies solely on the smoothness of $f$ (via Lemma \ref{lem:descent}) and the non‑expansiveness of the proximal map (Lemma \ref{lem:nonexp}). No convexity of $f$ or $g$ is required, which makes the result applicable to a broad class of non‑convex learning problems (e.g., neural networks with $\ell_{1}$ regularization).  

The $O(1/K)$ rate is optimal for first‑order methods in the non‑convex smooth setting. Faster rates would require stronger assumptions such as the Polyak–Łojasiewicz condition or Kurdyka–Łojasiewicz property, which are beyond the present scope.
