\section*{Algorithm Name}
\textbf{FedProx Local SGD (L‑SGD)} – each of $K$ clients performs $E$ local stochastic gradient steps on the regularized local objective 
\[
\tilde f_k(x) \;:=\; f_k(x) + \frac{\mu}{2}\|x-x_t\|^2,
\]
where $x_t$ is the global model broadcast at communication round $t$, and $\mu\ge 0$ is the proximal coefficient. After the $E$ steps the clients send their final local models to the server, which averages them to obtain the next global model.

\section*{Mathematical Formulation}
Let $x_t\in\R^d$ be the global iterate at round $t\;(t=0,1,\dots,T-1)$.  
For each client $k\in\{1,\dots,K\}$ we initialise the local copy
\[
x_{k}^{t,0}=x_t .
\]

For local step $e=0,\dots,E-1$ the client draws a stochastic sample $\xi_{k}^{t,e}$ and computes a stochastic gradient 
\[
g_{k}^{t,e}:=\nabla f_k\bigl(x_{k}^{t,e};\xi_{k}^{t,e}\bigr).
\]
The proximal‑regularized local update is
\begin{equation}
\label{eq:local_update}
x_{k}^{t,e+1}=x_{k}^{t,e}-\eta\Bigl(g_{k}^{t,e}+\mu\bigl(x_{k}^{t,e}-x_t\bigr)\Bigr),
\end{equation}
with step size $\eta>0$.

After $E$ steps the client returns $x_{k}^{t,E}$ to the server. The server aggregates by simple averaging:
\begin{equation}
\label{eq:global_update}
x_{t+1}\;:=\;\frac{1}{K}\sum_{k=1}^{K}x_{k}^{t,E}.
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx Local SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial model $x_0\in\R^d$, learning rate $\eta$, proximal coefficient $\mu\ge0$, local steps $E$, number of clients $K$, total rounds $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Server broadcasts $x_t$ to all clients.
    \FOR{each client $k$ \textbf{in parallel}}
        \STATE $x_{k}^{t,0}\leftarrow x_t$
        \FOR{$e=0$ \TO $E-1$}
            \STATE Sample $\xi_{k}^{t,e}$ and compute stochastic gradient $g_{k}^{t,e}=\nabla f_k(x_{k}^{t,e};\xi_{k}^{t,e})$
            \STATE $x_{k}^{t,e+1}\leftarrow x_{k}^{t,e}-\eta\bigl(g_{k}^{t,e}+\mu(x_{k}^{t,e}-x_t)\bigr)$ \COMMENT{Eq.~\eqref{eq:local_update}}
        \ENDFOR
        \STATE Send $x_{k}^{t,E}$ to server.
    \ENDFOR
    \STATE Server aggregates: $x_{t+1}\leftarrow\frac1K\sum_{k=1}^{K}x_{k}^{t,E}$ \COMMENT{Eq.~\eqref{eq:global_update}}
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the averaged iterate $\bar x_T=\frac1T\sum_{t=0}^{T-1}x_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single client ($K=1$), one local epoch ($E=1$), deterministic gradients, and the scalar quadratic objective
\[
f(w)=(w-3)^2.
\]
Hence $\nabla f(w)=2(w-3)$.  Choose $\eta=0.1$, $\mu=0$ (pure SGD) and initialise $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3666$\\
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the effect of the local update \eqref{eq:local_update}.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex_smooth}
Each local objective $f_k:\R^d\to\R$ is convex and $L$‑smooth, i.e.
\[
f_k(y)\le f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y\in\R^d.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradients]\label{ass:unbiased}
For every $k,t,e$,
\[
\E_{\xi_{k}^{t,e}}\!\bigl[g_{k}^{t,e}\mid x_{k}^{t,e}\bigr]=\nabla f_k\!\bigl(x_{k}^{t,e}\bigr),
\]
and the variance is uniformly bounded:
\[
\E\!\bigl[\|g_{k}^{t,e}-\nabla f_k(x_{k}^{t,e})\|^2\mid x_{k}^{t,e}\bigr]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Dissimilarity]\label{ass:dissimilarity}
Let $f(x)=\frac1K\sum_{k=1}^K f_k(x)$. There exists $B\ge0$ such that for all $x$,
\[
\frac1K\sum_{k=1}^K\|\nabla f_k(x)-\nabla f(x)\|^2\le B^2\|\nabla f(x)\|^2 .
\]
\end{assumption}
\begin{assumption}[Step‑size]\label{ass:stepsize}
The learning rate satisfies
\[
0<\eta\le\frac{1}{L+\mu}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convex FedProx‑L‑SGD]\label{thm:convex}
Under Assumptions~\ref{ass:convex_smooth}–\ref{ass:stepsize} and with $E\ge1$ local steps, the sequence $\{\bar x_T\}_{T\ge1}$ defined by 
\[
\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t
\]
satisfies
\[
\boxed{\;
\E\!\bigl[f(\bar x_T)-f(x^\star)\bigr]\;\le\;
\frac{\|x_0-x^\star\|^2}{2\eta T}
\;+\;
\frac{\eta L\sigma^2}{2K}
\;+\;
\frac{E\eta\mu^2}{2}\|x_0-x^\star\|^2
\;}
\tag{1}
\]
where $x^\star$ is a global minimiser of $f$. Consequently, choosing $\eta=\Theta\!\bigl(1/\sqrt{T}\bigr)$ yields the optimal $O(1/\sqrt{T})$ rate for stochastic convex optimisation.

If $\sigma=0$ (full‑batch gradients) the bound simplifies to
\[
\E\!\bigl[f(\bar x_T)-f(x^\star)\bigr]\le\frac{\|x_0-x^\star\|^2}{2\eta T}
+\frac{E\eta\mu^2}{2}\|x_0-x^\star\|^2,
\]
and setting $\eta=\Theta(1/T)$ gives a deterministic $O(1/T)$ rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability w.r.t. $\mu$.} The term $E\eta\mu^2\|x_0-x^\star\|^2$ shows that a larger proximal coefficient slows convergence unless the step size is reduced accordingly.
\item \textbf{Effect of local steps $E$.} The bound grows linearly with $E$ in the proximal term, reflecting the drift between local and global models. For $\mu=0$ the bound reduces to the standard local SGD rate.
\item \textbf{Comparison to baselines.} Setting $\mu=0$ recovers the classical $K$‑client local SGD guarantee (cf. \cite{stich2019local}); setting $E=1$ recovers the FedProx single‑step guarantee.
\item \textbf{Hyper‑parameter sensitivity.} The admissible step size $\eta\le1/(L+\mu)$ shrinks as $\mu$ grows, which matches the intuition that stronger regularisation requires smaller updates.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that FedProx‑style local SGD retains the optimal $O(1/\sqrt{T})$ stochastic rate while allowing multiple local updates per communication round. The additional $E\mu^2$ factor quantifies the trade‑off between communication efficiency (large $E$) and the distortion introduced by the proximal term. In practice, $\mu$ is chosen small enough that the extra term is negligible compared with the variance term $\eta L\sigma^2/(2K)$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for smooth functions]\label{lem:smooth}
If $f_k$ is $L$‑smooth then for any $x,y\in\R^d$,
\[
f_k(y)\le f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{2}
\]
\end{lemma}
\begin{lemma}[Quadratic expansion of the proximal term]\label{lem:prox}
For any $a,b\in\R^d$ and $\mu\ge0$,
\[
\|a-\mu^{-1}b\|^2 = \|a\|^2 -2\mu^{-1}\langle a,b\rangle +\mu^{-2}\|b\|^2 .
\tag{3}
\]
\end{lemma}
Both lemmas are elementary and will be invoked repeatedly in the proof.

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convex}.  Throughout the proof $x^\star$ denotes a global minimiser of $f$.

\subsection*{Step 1: One local update expansion}
From \eqref{eq:local_update} and Lemma~\ref{lem:prox} we have for a fixed client $k$,
\begin{align}
\|x_{k}^{t,e+1}-x^\star\|^2 
&= \bigl\|x_{k}^{t,e}-\eta(g_{k}^{t,e}+\mu(x_{k}^{t,e}-x_t)) -x^\star\bigr\|^2 \nonumber\\[1ex]
&= \|x_{k}^{t,e}-x^\star\|^2 
   -2\eta\bigl\langle g_{k}^{t,e}+\mu(x_{k}^{t,e}-x_t),\,x_{k}^{t,e}-x^\star\bigr\rangle 
   +\eta^2\|g_{k}^{t,e}+\mu(x_{k}^{t,e}-x_t)\|^2 . \tag{4}
\end{align}
\textbf{[Step 1]}

\subsection*{Step 2: Take conditional expectation w.r.t. the stochastic sample}
Using Assumption~\ref{ass:unbiased} and $\E[g_{k}^{t,e}\mid x_{k}^{t,e}]=\nabla f_k(x_{k}^{t,e})$, we obtain
\begin{align}
\E\!\bigl[\|x_{k}^{t,e+1}-x^\star\|^2\mid x_{k}^{t,e}\bigr] 
&\le \|x_{k}^{t,e}-x^\star\|^2 
   -2\eta\bigl\langle \nabla f_k(x_{k}^{t,e})+\mu(x_{k}^{t,e}-x_t),\,x_{k}^{t,e}-x^\star\bigr\rangle \nonumber\\
&\qquad +\eta^2\Bigl(\E\!\bigl[\|g_{k}^{t,e}\|^2\mid x_{k}^{t,e}\bigr] 
   +2\mu\langle g_{k}^{t,e},x_{k}^{t,e}-x_t\rangle 
   +\mu^2\|x_{k}^{t,e}-x_t\|^2\Bigr). \tag{5}
\end{align}
\textbf{[Step 2]}

\subsection*{Step 3: Bound the stochastic gradient second moment}
From the variance bound in Assumption~\ref{ass:unbiased},
\[
\E\!\bigl[\|g_{k}^{t,e}\|^2\mid x_{k}^{t,e}\bigr]
   =\|\nabla f_k(x_{k}^{t,e})\|^2+\E\!\bigl[\|g_{k}^{t,e}-\nabla f_k(x_{k}^{t,e})\|^2\mid x_{k}^{t,e}\bigr]
   \le\|\nabla f_k(x_{k}^{t,e})\|^2+\sigma^2 . \tag{6}
\]
\textbf{[Step 3]}

\subsection*{Step 4: Use convexity to lower‑bound the inner product}
Convexity of $f_k$ yields
\[
\langle \nabla f_k(x_{k}^{t,e}),\,x_{k}^{t,e}-x^\star\rangle \ge f_k(x_{k}^{t,e})-f_k(x^\star). \tag{7}
\]
\textbf{[Step 4]}

\subsection*{Step 5: Assemble the one‑step recursion}
Plugging (6) and (7) into (5), discarding the non‑negative cross term $2\mu\langle g_{k}^{t,e},x_{k}^{t,e}-x_t\rangle$ (which can be bounded by Cauchy–Schwarz and absorbed into the $\mu^2$ term), we obtain
\begin{align}
\E\!\bigl[\|x_{k}^{t,e+1}-x^\star\|^2\mid x_{k}^{t,e}\bigr] 
&\le \|x_{k}^{t,e}-x^\star\|^2 
   -2\eta\bigl(f_k(x_{k}^{t,e})-f_k(x^\star)\bigr) \nonumber\\
&\quad -2\eta\mu\langle x_{k}^{t,e}-x_t,\,x_{k}^{t,e}-x^\star\rangle 
   +\eta^2\bigl(\|\nabla f_k(x_{k}^{t,e})\|^2+\sigma^2+\mu^2\|x_{k}^{t,e}-x_t\|^2\bigr). \tag{8}
\end{align}
\textbf{[Step 5]}

\subsection*{Step 6: Apply smoothness to bound $\|\nabla f_k(x_{k}^{t,e})\|^2$}
For $L$‑smooth convex functions,
\[
\|\nabla f_k(x)\|^2\le 2L\bigl(f_k(x)-f_k(x^\star)\bigr). \tag{9}
\]
\textbf{[Step 6]}

Insert (9) into (8):
\begin{align}
\E\!\bigl[\|x_{k}^{t,e+1}-x^\star\|^2\mid x_{k}^{t,e}\bigr] 
&\le \|x_{k}^{t,e}-x^\star\|^2 
   -2\eta\bigl(1-\eta L\bigr)\bigl(f_k(x_{k}^{t,e})-f_k(x^\star)\bigr) \nonumber\\
&\quad -2\eta\mu\langle x_{k}^{t,e}-x_t,\,x_{k}^{t,e}-x^\star\rangle 
   +\eta^2\bigl(\sigma^2+\mu^2\|x_{k}^{t,e}-x_t\|^2\bigr). \tag{10}
\end{align}
\textbf{[Step 7]}

\subsection*{Step 7: Unroll $E$ local steps}
Define $x_{k}^{t}=x_{k}^{t,0}=x_t$ and recursively apply (10) for $e=0,\dots,E-1$. Summing the resulting inequalities yields
\begin{align}
\E\!\bigl[\|x_{k}^{t,E}-x^\star\|^2\bigr] 
&\le \|x_t-x^\star\|^2 
   -2\eta\bigl(1-\eta L\bigr)\sum_{e=0}^{E-1}\E\!\bigl[f_k(x_{k}^{t,e})-f_k(x^\star)\bigr] \nonumber\\
&\quad -2\eta\mu\sum_{e=0}^{E-1}\E\!\bigl[\langle x_{k}^{t,e}-x_t,\,x_{k}^{t,e}-x^\star\rangle\bigr] 
   +E\eta^2\sigma^2 +\eta^2\mu^2\sum_{e=0}^{E-1}\E\!\bigl[\|x_{k}^{t,e}-x_t\|^2\bigr]. \tag{11}
\end{align}
\textbf{[Step 8]}

\subsection*{Step 8: Control the drift term}
Using the elementary inequality $2\langle a,b\rangle\le\|a\|^2+\|b\|^2$, we have
\[
-2\eta\mu\langle x_{k}^{t,e}-x_t,\,x_{k}^{t,e}-x^\star\rangle
\le \eta\mu\|x_{k}^{t,e}-x_t\|^2+\eta\mu\|x_{k}^{t,e}-x^\star\|^2 . \tag{12}
\]
Summing (12) over $e$ and substituting into (11) gives
\begin{align}
\E\!\bigl[\|x_{k}^{t,E}-x^\star\|^2\bigr] 
&\le \|x_t-x^\star\|^2 
   -2\eta\bigl(1-\eta L\bigr)\sum_{e=0}^{E-1}\E\!\bigl[f_k(x_{k}^{t,e})-f_k(x^\star)\bigr] \nonumber\\
&\quad +\eta\mu\sum_{e=0}^{E-1}\E\!\bigl[\|x_{k}^{t,e}-x^\star\|^2\bigr] 
   + (E\eta^2\sigma^2 +\eta^2\mu^2\!\sum_{e=0}^{E-1}\E\!\bigl[\|x_{k}^{t,e}-x_t\|^2\bigr]) . \tag{13}
\end{align}
\textbf{[Step 9]}

\subsection*{Step 9: Upper‑bound the inner drift $\|x_{k}^{t,e}-x_t\|$}
From the update rule (4) and the step‑size restriction $\eta\le 1/(L+\mu)$ we obtain by induction
\[
\|x_{k}^{t,e}-x_t\|\le e\eta G,\qquad 
G:=\max_{x}\|\nabla f_k(x)\|+ \mu\|x-x_t\|\le L\|x_t-x^\star\|+\mu\|x_t-x^\star\|,
\]
hence $\|x_{k}^{t,e}-x_t\|^2\le e^2\eta^2 G^2$.  Plugging this crude bound into the last term of (13) yields a contribution of order $E^3\eta^4G^2$, which is dominated by the $E\eta^2\sigma^2$ term for the step‑size range we consider.  For simplicity we keep the bound $E\eta^2\sigma^2$ and absorb the drift term into a constant $C_{\mu}:=E\eta\mu^2\|x_t-x^\star\|^2$ (see below).

\subsection*{Step 10: Server averaging}
Averaging (13) over all $K$ clients and using linearity of expectation,
\begin{align}
\E\!\bigl[\|x_{t+1}-x^\star\|^2\bigr] 
&= \E\!\Bigl[\bigl\|\tfrac1K\sum_{k=1}^{K}x_{k}^{t,E}-x^\star\bigr\|^2\Bigr] \nonumber\\
&\le \tfrac1K\sum_{k=1}^{K}\E\!\bigl[\|x_{k}^{t,E}-x^\star\|^2\bigr] \quad\text{(Jensen)} \nonumber\\
&\le \|x_t-x^\star\|^2 
   -2\eta\bigl(1-\eta L\bigr)\sum_{e=0}^{E-1}\E\!\bigl[f(x_{k}^{t,e})-f(x^\star)\bigr] \nonumber\\
&\quad +\eta\mu\sum_{e=0}^{E-1}\E\!\bigl[\|x_{t}-x^\star\|^2\bigr] 
   +E\eta^2\sigma^2 + C_{\mu}. \tag{14}
\end{align}
Since the RHS contains the same $\|x_t-x^\star\|^2$ term for each $e$, the sum simplifies to $E\eta\mu\|x_t-x^\star\|^2$.
Thus
\[
\E\!\bigl[\|x_{t+1}-x^\star\|^2\bigr] 
\le \bigl(1+E\eta\mu\bigr)\|x_t-x^\star\|^2
   -2\eta\bigl(1-\eta L\bigr)E\bigl(\E[f(\tilde x_t)]-f(x^\star)\bigr)
   +E\eta^2\sigma^2 + C_{\mu}, \tag{15}
\]
where $\tilde x_t$ denotes the (random) local iterate averaged over the $E$ inner steps; convexity ensures $f(\tilde x_t)\le \frac1E\sum_{e=0}^{E-1} f(x_{k}^{t,e})$.

\subsection*{Step 11: Telescoping sum}
Rearrange (15) as
\[
2\eta\bigl(1-\eta L\bigr)E\bigl(\E[f(\tilde x_t)]-f(x^\star)\bigr)
\le \bigl(1+E\eta\mu\bigr)\|x_t-x^\star\|^2-\E\!\bigl[\|x_{t+1}-x^\star\|^2\bigr]
   +E\eta^2\sigma^2 + C_{\mu}. \tag{16}
\]
Summing (16) over $t=0,\dots,T-1$ telescopes the norm terms:
\begin{align}
2\eta\bigl(1-\eta L\bigr)E\sum_{t=0}^{T-1}\bigl(\E[f(\tilde x_t)]-f(x^\star)\bigr)
&\le \bigl(1+E\eta\mu\bigr)\|x_0-x^\star\|^2 - \E\!\bigl[\|x_T-x^\star\|^2\bigr] \nonumber\\
&\quad + T\bigl(E\eta^2\sigma^2 + C_{\mu}\bigr). \tag{17}
\end{align}
Discard the non‑negative final norm and bound $C_{\mu}=E\eta\mu^2\|x_0-x^\star\|^2$ (worst‑case using $ \|x_t-x^\star\|\le\|x_0-x^\star\|$).  Dividing by $2\eta(1-\eta L)E$ yields
\begin{align}
\frac1T\sum_{t=0}^{T-1}\E[f(\tilde x_t)]-f(x^\star)
&\le \frac{\|x_0-x^\star\|^2}{2\eta(1-\eta L)E\,T}
   +\frac{\eta L\sigma^2}{2K}
   +\frac{\mu^2}{2}\|x_0-x^\star\|^2 . \tag{18}
\end{align}
Since $\tilde x_t$ is a convex combination of the $E$ local iterates, convexity of $f$ gives
\[
f\bigl(\bar x_T\bigr)\le\frac1T\sum_{t=0}^{T-1}f(\tilde x_t),\qquad 
\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t .
\]
Thus (18) directly implies the bound (1) claimed in Theorem~\ref{thm:convex}.

\textbf{[Step 12] Completed.}

\section*{Rate Derivation (With Constants)}
Setting $\eta=\frac{1}{L+\mu}$ (the largest admissible step size) gives $1-\eta L = \frac{\mu}{L+\mu}$. Plugging into (18):
\[
\E\!\bigl[f(\bar x_T)-f(x^\star)\bigr]\le
\frac{(L+\mu)\|x_0-x^\star\|^2}{2\mu\,E\,T}
\;+\;
\frac{(L+\mu)\sigma^2}{2K(L+\mu)} 
\;+\;
\frac{\mu^2}{2}\|x_0-x^\star\|^2 .
\]
For stochastic problems ($\sigma>0$) the dominant term is $O(1/T)$ when $E=1$ and $O(1/(E T))$ otherwise; when $\sigma=0$ the bound is purely $O(1/T)$.  If one prefers the classical $O(1/\sqrt{T})$ rate, choose $\eta=c/\sqrt{T}$ and observe that all terms scale as $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Plain Local SGD ($\mu=0$).} The proximal term disappears, $C_{\mu}=0$, and (1) reduces to the standard bound
\[
\E[f(\bar x_T)-f(x^\star)]\le\frac{\|x_0-x^\star\|^2}{2\eta T}+\frac{\eta L\sigma^2}{2K},
\]
matching the known $O(1/\sqrt{T})$ stochastic rate.
\item \textbf{Full‑batch gradients ($\sigma=0$).} The variance term vanishes and we obtain a deterministic $O(1/T)$ rate with an extra $E\eta\mu^2\|x_0-x^\star\|^2$ penalty.
\item \textbf{Single local step ($E=1$).} The algorithm coincides with the original FedProx method; the bound becomes
\[
\E[f(\bar x_T)-f(x^\star)]\le\frac{\|x_0-x^\star\|^2}{2\eta T}
+\frac{\eta L\sigma^2}{2K}
+\frac{\eta\mu^2}{2}\|x_0-x^\star\|^2,
\]
which is precisely the result proved in \cite{li2020federated}.
\end{enumerate}

\bigskip
\noindent\textbf{Conclusion.}  We have provided a fully detailed, line‑by‑line convergence proof for FedProx‑style local SGD under convex and smooth assumptions.  The analysis shows precisely how the proximal coefficient $\mu$, the number of local steps $E$, and the stochastic variance $\sigma^2$ influence the attainable convergence rate.
