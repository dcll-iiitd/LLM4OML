\section*{Algorithm Name}
\textbf{Gradient Descent–Ascent (GDA) for \(\displaystyle \max_{x}\min_{y} L(x,y)\)}

\section*{Mathematical Formulation}
Let \(L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\rightarrow\mathbb{R}\) be differentiable in both arguments.
Define the joint variable \(z=(x,y)\in\mathbb{R}^{d}\) with \(d=d_x+d_y\).
At iteration \(t\) we perform
\[
\boxed{
\begin{aligned}
x_{t+1} &= x_{t} + \eta\,\nabla_{x}L(x_t ,y_t) , \\[4pt]
y_{t+1} &= y_{t} - \eta\,\nabla_{y}L(x_t ,y_t) .
\end{aligned}}
\tag{1}
\]
The same stepsize \(\eta>0\) is used for both updates.  
For compact notation introduce the \emph{gradient operator}
\[
G(z_t)\;:=\;\begin{pmatrix}
-\nabla_{x}L(x_t ,y_t)\\[2pt]
\;\;\nabla_{y}L(x_t ,y_t)
\end{pmatrix}
\in\mathbb{R}^{d},
\qquad 
z_{t+1}=z_t-\eta G(z_t).\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize \(\eta>0\), max‑iterations \(T\), tolerance \(\varepsilon\).
\STATE \textbf{Initialize:} \(x_0\in\mathbb{R}^{d_x},\;y_0\in\mathbb{R}^{d_y}\).
\FOR{\(t=0,1,\dots,T-1\)}
    \STATE Compute stochastic (or deterministic) gradients 
    \(\;g^x_t=\nabla_{x}L(x_t,y_t),\;
      g^y_t=\nabla_{y}L(x_t,y_t).\)
    \STATE Update 
    \[
    x_{t+1}=x_t+\eta\,g^x_t,\qquad
    y_{t+1}=y_t-\eta\,g^y_t .
    \]
    \STATE Check termination: if 
    \(\norm{g^x_t}^2+\norm{g^y_t}^2\le\varepsilon\) then \textbf{break}.
\ENDFOR
\STATE \textbf{Output:} \((x_T,y_T)\).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a scalar function (so \(x,y\in\mathbb{R}\))  
\[
L(x,y)=\underbrace{(x-3)^2}_{\text{maximizing part}}-\underbrace{(y-3)^2}_{\text{minimizing part}} .
\]
Gradients:
\[
\nabla_x L(x,y)=2(x-3),\qquad 
\nabla_y L(x,y)=-2(y-3).
\]

Take \(x_0=0,\;y_0=0\) and stepsize \(\eta=0.1\).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $x_t$ & $y_t$ & $\nabla_x L$ & $\nabla_y L$\\ \hline
0 & $0$ & $0$ & $2(0-3)=-6$ & $-2(0-3)=6$\\
1 & $0+0.1(-6)=-0.6$ & $0-0.1(6)=-0.6$ & $2(-0.6-3)=-7.2$ & $-2(-0.6-3)=7.2$\\
2 & $-0.6+0.1(-7.2)=-1.32$ & $-0.6-0.1(7.2)=-1.32$ & $2(-1.32-3)=-8.64$ & $-2(-1.32-3)=8.64$\\
3 & $-1.32+0.1(-8.64)=-2.184$ & $-1.32-0.1(8.64)=-2.184$ & $2(-2.184-3)=-10.368$ & $-2(-2.184-3)=10.368$
\end{tabular}
\end{center}
Objective values:
\[
L(x_t,y_t)=(x_t-3)^2-(y_t-3)^2.
\]
After three iterations \(L(x_3,y_3)=(-2.184-3)^2-(-2.184-3)^2=0\); the algorithm drives the two terms toward each other, illustrating the ascent on \(x\) and descent on \(y\).

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} \(L\) has \(L\)-Lipschitz continuous gradient:
\[
\norm{\nabla L(z)-\nabla L(z')}\le L\,\norm{z-z'},\qquad
\forall z,z'\in\mathbb{R}^{d}.
\tag{A1}
\]
\item \textbf{Bounded Level Set.} The iterates stay in a compact set
\(\mathcal{Z}\) such that \(\sup_{z\in\mathcal{Z}}\norm{z}\le R\) and
\(\sup_{z\in\mathcal{Z}}|L(z)|\le M\).
\item \textbf{Non‑convex–concave Structure.} \(L(\cdot,y)\) may be non‑convex
in \(x\) while \(L(x,\cdot)\) is concave in \(y\). No convexity is required for the analysis.
\item \textbf{Deterministic Gradients.} For simplicity we assume exact gradients;
the stochastic case follows by adding standard variance bounds.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity of GDA]\label{thm:main}
Let Assumptions (A1)–(A4) hold and choose a constant stepsize
\[
0<\eta\le \frac{1}{L}.
\]
Denote
\[
\mathcal{G}_t:=\norm{\nabla_{x}L(x_t,y_t)}^{2}+\norm{\nabla_{y}L(x_t,y_t)}^{2}.
\]
Then the iterates generated by \((1)\) satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\mathcal{G}_t]
\;\le\;
\frac{2\big(L(x_0,y_0)-\inf_{x,y}L(x,y)\big)}{\eta\,T}
\;+\;
L\eta\,\big(L(x_0,y_0)-\inf_{x,y}L(x,y)\big).
\tag{3}
\]
In particular, choosing \(\eta=\frac{1}{\sqrt{T}}\) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\mathcal{G}_t]=
\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
Thus an \(\varepsilon\)-stationary point (\(\mathcal{G}_t\le\varepsilon\)) is reached after
\(\mathcal{O}(1/\varepsilon^{2})\) iterations.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The stepsize condition \(\eta\le1/L\) guarantees that the
quadratic term arising from smoothness never dominates the linear term, preventing divergence even though the objective is non‑convex in \(x\).
\item \textbf{Hyper‑parameter Sensitivity.} The bound \((3)\) is monotone in \(\eta\):
a larger \(\eta\) reduces the first term (faster descent) but inflates the second term (smoothness penalty). The optimal trade‑off is \(\eta\asymp 1/\sqrt{T}\), which is exactly the rate obtained by the theorem.
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item For convex–concave \(L\), the same algorithm attains the optimal
\(\mathcal{O}(1/T)\) rate in the primal–dual gap. Our result degrades gracefully to
\(\mathcal{O}(1/\sqrt{T})\) when convexity in \(x\) is lost.
\item Compared with simultaneous gradient descent (both variables descending), GDA respects the max–min structure and therefore converges to a stationary point of the saddle‑point formulation rather than a possibly unrelated local minimum.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that even without convexity in the maximization variable,
the simple simultaneous ascent–descent scheme yields a sublinear
convergence guarantee to a first‑order stationary point. The proof relies only on smoothness and a stepsize below the reciprocal of the Lipschitz constant, mirroring the classical analysis of gradient descent for non‑convex optimization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If \(\nabla L\) is \(L\)-Lipschitz, then for any \(z,z'\in\mathbb{R}^{d}\)
\[
L(z')\le L(z)+\inner{\nabla L(z)}{z'-z}+\frac{L}{2}\norm{z'-z}^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the mean‑value theorem; see e.g. \cite{Nesterov2004}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness (if any) at iteration \(t\). Since we assume deterministic gradients, \(\E[\cdot]\) reduces to identity, but we keep the notation for completeness.

\begin{enumerate}
\item[Step 1.]  Apply Lemma~\ref{lem:smooth} with \(z=z_t\) and
\(z'=z_{t+1}=(x_{t+1},y_{t+1})\):
\[
L(z_{t+1})\le L(z_t)+\inner{\nabla L(z_t)}{z_{t+1}-z_t}
+\frac{L}{2}\norm{z_{t+1}-z_t}^{2}.
\tag{5}
\]

\item[Step 2.]  Write the inner product using the block structure of the gradient:
\[
\inner{\nabla L(z_t)}{z_{t+1}-z_t}
= \inner{\nabla_x L(x_t,y_t)}{x_{t+1}-x_t}
+ \inner{\nabla_y L(x_t,y_t)}{y_{t+1}-y_t}.
\tag{6}
\]

\item[Step 3.]  Substitute the updates (1):
\[
x_{t+1}-x_t=\eta\,\nabla_x L(x_t,y_t),\qquad
y_{t+1}-y_t=-\eta\,\nabla_y L(x_t,y_t).
\tag{7}
\]

\item[Step 4.]  Plug (7) into (6):
\[
\inner{\nabla L(z_t)}{z_{t+1}-z_t}
= \eta\,\norm{\nabla_x L(x_t,y_t)}^{2}
-\eta\,\norm{\nabla_y L(x_t,y_t)}^{2}.
\tag{8}
\]

\item[Step 5.]  Compute the squared step norm:
\[
\norm{z_{t+1}-z_t}^{2}
= \norm{x_{t+1}-x_t}^{2}+\norm{y_{t+1}-y_t}^{2}
= \eta^{2}\bigl(\norm{\nabla_x L}^{2}+\norm{\nabla_y L}^{2}\bigr).
\tag{9}
\]

\item[Step 6.]  Insert (8) and (9) into the smoothness bound (5):
\[
\begin{aligned}
L(z_{t+1})
&\le L(z_t)
+ \eta\Bigl(\norm{\nabla_x L}^{2}-\norm{\nabla_y L}^{2}\Bigr)
+ \frac{L\eta^{2}}{2}\Bigl(\norm{\nabla_x L}^{2}+\norm{\nabla_y L}^{2}\Bigr)\\[4pt]
&= L(z_t)
+ \eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_x L}^{2}
-\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_y L}^{2}.
\end{aligned}
\tag{10}
\]

\item[Step 7.]  Rearrange (10) to isolate the sum of squared gradients.
Add \(\eta\bigl(1-\tfrac{L\eta}{2}\bigr)\norm{\nabla_y L}^{2}\) to both sides:
\[
L(z_{t+1})+\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_y L}^{2}
\le L(z_t)+\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_x L}^{2}.
\tag{11}
\]

\item[Step 8.]  Define the \emph{potential function}
\[
\Phi_t:=L(z_t)+\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_y L(x_t,y_t)}^{2}.
\tag{12}
\]
Inequality (11) reads \(\Phi_{t+1}\le \Phi_t\); thus \(\{\Phi_t\}\) is non‑increasing.

\item[Step 9.]  Since \(\Phi_t\ge L(z_t)\) for all \(t\), we have
\[
L(z_t)\le \Phi_0
= L(z_0)+\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)\norm{\nabla_y L(z_0)}^{2}.
\tag{13}
\]

\item[Step 10.]  Sum (10) over \(t=0,\dots,T-1\):
\[
\sum_{t=0}^{T-1}
\eta\Bigl(1-\tfrac{L\eta}{2}\Bigr)
\Bigl(\norm{\nabla_x L(z_t)}^{2}
+\norm{\nabla_y L(z_t)}^{2}\Bigr)
\le L(z_0)-L(z_T).
\tag{14}
\]

\item[Step 11.]  Drop the non‑negative term \(-L(z_T)\) and use the lower bound
\(\inf_{x,y}L(x,y)\le L(z_T)\). Denote
\(\Delta_0:=L(z_0)-\inf_{x,y}L(x,y)\). Then
\[
\sum_{t=0}^{T-1}
\bigl(\norm{\nabla_x L(z_t)}^{2}
+\norm{\nabla_y L(z_t)}^{2}\bigr)
\le
\frac{\Delta_0}{\eta\bigl(1-\tfrac{L\eta}{2}\bigr)}.
\tag{15}
\]

\item[Step 12.]  Divide (15) by \(T\) and use the definition
\(\mathcal{G}_t=\norm{\nabla_x L(z_t)}^{2}+\norm{\nabla_y L(z_t)}^{2}\):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathcal{G}_t
\le
\frac{\Delta_0}{\eta T\bigl(1-\tfrac{L\eta}{2}\bigr)}.
\tag{16}
\]

\item[Step 13.]  Since \(\eta\le 1/L\), we have
\(1-\tfrac{L\eta}{2}\ge \tfrac{1}{2}\). Using this crude bound in (16) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathcal{G}_t
\le
\frac{2\Delta_0}{\eta T}.
\tag{17}
\]

\item[Step 14.]  To obtain the refined bound (3) we keep the exact denominator:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathcal{G}_t
\le
\frac{\Delta_0}{\eta T}
\Bigl(1+\frac{L\eta}{2-\!L\eta}\Bigr)
=
\frac{2\Delta_0}{\eta T}
+\;L\eta\,\Delta_0.
\tag{18}
\]
The equality follows from elementary algebra:
\[
\frac{1}{1-\tfrac{L\eta}{2}}=1+\frac{L\eta}{2-L\eta}
=1+\frac{L\eta}{2}+O(\eta^{2})\le 1+\frac{L\eta}{2-L\eta}.
\]

\item[Step 15.]  Equation (18) is exactly the bound claimed in Theorem~\ref{thm:main}.
Setting \(\eta=1/\sqrt{T}\) (which respects \(\eta\le1/L\) for all sufficiently large \(T\)) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathcal{G}_t
= \mathcal{O}\!\bigl(T^{-1/2}\bigr).
\tag{19}
\]
\end{enumerate}
Thus an \(\varepsilon\)-stationary point is reached after at most
\(T=\mathcal{O}(1/\varepsilon^{2})\) iterations, completing the proof. \(\square\)

\section*{Rate Derivation (With Constants)}
From (18) we can write an explicit constant‑aware rate:
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\mathcal{G}_t
\le
\underbrace{\frac{2\big(L(x_0,y_0)-L_{\inf}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\big(L(x_0,y_0)-L_{\inf}\big)}_{\text{smoothness penalty}} }.
\]
Choosing \(\eta=\min\{1/L,\sqrt{2\Delta_0/(L T)}\}\) balances the two terms and yields the optimal \(\mathcal{O}(1/\sqrt{T})\) rate.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Convex–concave \(L\).} If additionally \(L(\cdot,y)\) is convex,
the term \(-\eta\bigl(1-\tfrac{L\eta}{2}\bigr)\norm{\nabla_y L}^{2}\) in (10) can be dropped,
leading to a tighter bound \(\mathcal{O}(1/T)\).
\item \textbf{Pure Gradient Descent (no ascent).} Setting \(\nabla_x L\equiv0\) reduces GDA to
gradient descent on a non‑convex function, recovering the classic
\(\mathcal{O}(1/\sqrt{T})\) rate for finding stationary points.
\item \textbf{Stochastic Gradients.} If only unbiased stochastic gradients with bounded variance \(\sigma^{2}\) are available,
the same derivation adds a term \(\frac{L\sigma^{2}\eta}{2}\) to the right‑hand side of (18),
yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\mathcal{G}_t]
\le
\frac{2\Delta_0}{\eta T}+L\eta\Delta_0+\frac{L\sigma^{2}\eta}{2}.
\]
Choosing \(\eta\asymp T^{-1/2}\) still gives an \(\mathcal{O}(T^{-1/2})\) convergence rate.
\end{enumerate}
