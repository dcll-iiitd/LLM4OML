\section*{Probabilistic Coordinate Descent (PCD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a convex differentiable function.  
Denote by $e_i\in\mathbb{R}^{d}$ the $i$‑th canonical unit vector and by
\[
\nabla_i f(w)=\frac{\partial f(w)}{\partial w_i}= \langle\nabla f(w),e_i\rangle
\]
the $i$‑th partial derivative.  

A probability vector $p=(p_1,\dots ,p_d)$ with $p_i>0,\ \sum_{i=1}^{d}p_i=1$
governs the random coordinate selection.  
At iteration $k$ we draw an index $i_k\sim p$ independently of the past and perform the update
\begin{equation}
\boxed{\;
w_{k+1}=w_k-\frac{\eta_k}{p_{i_k}}\,\nabla_{i_k}f(w_k)\,e_{i_k}
\;}
\label{eq:update}
\end{equation}
where $\eta_k>0$ is a (possibly diminishing) stepsize.  
Only the $i_k$‑th component of $w_k$ changes:
\[
w_{k+1}^{(i)}=
\begin{cases}
w_{k}^{(i)}-\dfrac{\eta_k}{p_i}\,\nabla_i f(w_k), & i=i_k,\\[6pt]
w_{k}^{(i)}, & i\neq i_k .
\end{cases}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent}
\begin{algorithmic}[1]
\Require Convex $f$, probability vector $p\in\Delta^{d}$, stepsizes $\{\eta_k\}_{k\ge0}$, initial point $w_0\in\mathbb{R}^{d}$, max iterations $T$.
\For{$k=0$ \textbf{to} $T-1$}
    \State Sample $i_k\in\{1,\dots ,d\}$ with $\mathbb{P}(i_k=i)=p_i$.
    \State Compute partial gradient $g_{k}= \nabla_{i_k}f(w_k)$.
    \State Update only the selected coordinate:
    \[
    w_{k+1}=w_k-\frac{\eta_k}{p_{i_k}}\,g_{k}\,e_{i_k}.
    \]
\EndFor
\State \Return $w_T$ (or the average $\bar w_T=\tfrac1T\sum_{k=1}^{T}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^2$, $d=1$.
Here $p_1=1$, $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, constant stepsize $\eta=0.1$.

\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla_1 f(w_k)$ & $w_{k+1}=w_k-\frac{\eta}{p_1}\nabla_1 f(w_k)$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}

Objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3545$, illustrating monotone decrease.

\section*{Assumptions}
\begin{assumption}[Convexity]
$f$ is convex: for all $w,u\in\mathbb{R}^{d}$,
\[
f(u)\ge f(w)+\langle\nabla f(w),u-w\rangle .
\]
\end{assumption}
\begin{assumption}[Coordinate-wise $L_i$‑smoothness]
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\mathbb{R}^{d}$ and any scalar $\alpha$,
\[
\big|\nabla_i f(w+\alpha e_i)-\nabla_i f(w)\big|\le L_i |\alpha|.
\]
Equivalently,
\[
f(w+\alpha e_i)\le f(w)+\alpha\,\nabla_i f(w)+\frac{L_i}{2}\alpha^{2}.
\]
\end{assumption}
\begin{assumption}[Stepsize]
The stepsizes satisfy $0<\eta_k\le \frac{1}{\max_i L_i}$ for all $k$ (constant stepsize case) and $\sum_{k=0}^{\infty}\eta_k=\infty$, $\sum_{k=0}^{\infty}\eta_k^{2}<\infty$ for diminishing stepsizes.
\end{assumption}
\begin{assumption}[Optimal solution]
Denote $w^{\star}\in\arg\min_{w}f(w)$. The optimal set is non‑empty.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of PCD]\label{thm:main}
Assume \ref{assumption}–\ref{assumption}. Let $\eta_k\equiv\eta\le \frac{1}{\max_i L_i}$ be constant. Define the weighted norm
\[
\|x\|_{P}^{2}:=\sum_{i=1}^{d}p_i\,x_i^{2}.
\]
Then for any $T\ge1$ the iterate $w_T$ generated by \eqref{eq:update} satisfies
\begin{equation}
\mathbb{E}\!\left[f(w_T)-f(w^{\star})\right]
\;\le\;
\frac{\|w_0-w^{\star}\|_{P}^{2}}{2\eta T}.
\label{eq:rate}
\end{equation}
Consequently $\mathbb{E}[f(w_T)]\to f(w^{\star})$ at the rate $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because the update is scaled by $1/p_{i_k}$, coordinates that are selected rarely receive proportionally larger steps, preventing systematic neglect of any direction.
\item \textbf{Hyperparameter Sensitivity.} The only algorithmic hyperparameter is the stepsize $\eta$. The bound \eqref{eq:rate} holds for any $\eta\in(0,1/\max_i L_i]$, showing robustness; larger $\eta$ yields a tighter bound until the smoothness limit is reached.
\item \textbf{Comparison to Uniform Coordinate Descent.} Setting $p_i=1/d$ recovers the classical (uniform) randomized coordinate descent with rate $\frac{d\|w_0-w^{\star}\|^{2}}{2\eta T}$. Importance sampling ($p_i\propto L_i^{1/2}$) can improve the constant factor, as the weighted norm $\|\cdot\|_P$ adapts to coordinate smoothness.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that, despite the stochastic and highly sparse nature of the updates, the algorithm attains the optimal $O(1/T)$ convergence rate for convex smooth problems when expectations are weighted by $1/p_i$. The proof mirrors standard analyses of randomized coordinate descent but requires careful handling of the $1/p_i$ factor in the expectation; this is carried out explicitly in the next sections.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step progress]\label{lem:one-step}
For any $w\in\mathbb{R}^{d}$ and any index $i$, the $L_i$‑smoothness inequality yields
\[
f\!\big(w-\tfrac{\eta}{p_i}\nabla_i f(w)e_i\big)
\le
f(w)-\frac{\eta}{p_i}\,\big(\nabla_i f(w)\big)^{2}
+\frac{L_i\eta^{2}}{2p_i^{2}}\big(\nabla_i f(w)\big)^{2}.
\]
\end{lemma}
\begin{proof}
Apply the coordinate‑wise smoothness with $\alpha=-\frac{\eta}{p_i}\nabla_i f(w)$:
\[
\begin{aligned}
f\big(w+\alpha e_i\big)
&\le f(w)+\alpha\,\nabla_i f(w)+\frac{L_i}{2}\alpha^{2}\\
&=f(w)-\frac{\eta}{p_i}\big(\nabla_i f(w)\big)^{2}
+\frac{L_i\eta^{2}}{2p_i^{2}}\big(\nabla_i f(w)\big)^{2}.
\end{aligned}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $w_k$ be the iterate before the $k$‑th update and $i_k$ the selected coordinate.
Define the weighted distance
\[
D_k:=\|w_k-w^{\star}\|_{P}^{2}
   =\sum_{i=1}^{d}p_i\big(w_k^{(i)}-w^{\star}_i\big)^{2}.
\]

\subsection*{Step 1: Relation between $D_{k+1}$ and $D_k$}
\begin{align}
D_{k+1}
&=\sum_{i=1}^{d}p_i\big(w_{k+1}^{(i)}-w^{\star}_i\big)^{2} \nonumber\\
&=\sum_{i\neq i_k}p_i\big(w_{k}^{(i)}-w^{\star}_i\big)^{2}
   +p_{i_k}\Big(w_{k}^{(i_k)}-\frac{\eta}{p_{i_k}}\nabla_{i_k}f(w_k)-w^{\star}_{i_k}\Big)^{2}\nonumber\\
&=D_k- p_{i_k}\big(w_{k}^{(i_k)}-w^{\star}_{i_k}\big)^{2}
   +p_{i_k}\Big(w_{k}^{(i_k)}-\frac{\eta}{p_{i_k}}\nabla_{i_k}f(w_k)-w^{\star}_{i_k}\Big)^{2}.\label{eq:Dexp}
\end{align}
Expand the square in the last term:
\begin{align}
&\;p_{i_k}\Big[(w_{k}^{(i_k)}-w^{\star}_{i_k})^{2}
      -2\frac{\eta}{p_{i_k}}(w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)
      +\frac{\eta^{2}}{p_{i_k}^{2}}\big(\nabla_{i_k}f(w_k)\big)^{2}\Big]\nonumber\\
=&\;p_{i_k}(w_{k}^{(i_k)}-w^{\star}_{i_k})^{2}
   -2\eta (w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)
   +\frac{L_{i_k}\eta^{2}}{p_{i_k}}\big(\nabla_{i_k}f(w_k)\big)^{2}
   \quad\text{[adding $0$]}.\label{eq:expand}
\end{align}
Insert \eqref{eq:expand} into \eqref{eq:Dexp}; the first terms cancel, giving
\begin{equation}
D_{k+1}=D_k-2\eta (w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)
       +\frac{L_{i_k}\eta^{2}}{p_{i_k}}\big(\nabla_{i_k}f(w_k)\big)^{2}.
\label{eq:Dnext}
\end{equation}
[Step 1]  

\subsection*{Step 2: Use convexity to replace the inner product}
Convexity of $f$ yields for any $w$,
\[
f(w)-f(w^{\star})\le \langle \nabla f(w), w-w^{\star}\rangle
   =\sum_{i=1}^{d}(w^{(i)}-w^{\star}_i)\nabla_i f(w).
\]
Hence, for the randomly selected coordinate $i_k$,
\[
(w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)
\ge f(w_k)-f(w^{\star}) -\!\!\sum_{i\neq i_k}(w_{k}^{(i)}-w^{\star}_i)\nabla_i f(w_k).
\]
Taking expectation w.r.t.\ $i_k$ (conditioned on $w_k$) eliminates the cross terms because
\[
\mathbb{E}_{i_k}\!\big[(w_{k}^{(i)}-w^{\star}_i)\nabla_i f(w_k)\big]
   =\sum_{i=1}^{d}p_i (w_{k}^{(i)}-w^{\star}_i)\nabla_i f(w_k).
\]
Thus
\begin{equation}
\mathbb{E}_{i_k}\!\big[(w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)\big]
   =\sum_{i=1}^{d}p_i (w_{k}^{(i)}-w^{\star}_i)\nabla_i f(w_k)
   \ge p_{\min}\big(f(w_k)-f(w^{\star})\big),
\label{eq:innerexp}
\end{equation}
where $p_{\min}:=\min_i p_i>0$.  
[Step 2]  

\subsection*{Step 3: Expectation of the quadratic gradient term}
From Lemma~\ref{lem:one-step} we have
\[
f\big(w_k-\tfrac{\eta}{p_i}\nabla_i f(w_k)e_i\big)
\le f(w_k)-\frac{\eta}{p_i}\big(\nabla_i f(w_k)\big)^{2}
      +\frac{L_i\eta^{2}}{2p_i^{2}}\big(\nabla_i f(w_k)\big)^{2}.
\]
Rearranging,
\[
\frac{L_i\eta^{2}}{p_i}\big(\nabla_i f(w_k)\big)^{2}
\le 2\eta\big(\nabla_i f(w_k)\big)^{2}
    -2\big(f(w_k)-f(w_{k+1}^{(i)})\big)p_i,
\]
where $w_{k+1}^{(i)}$ denotes the virtual point obtained by updating coordinate $i$ only.
Taking expectation over $i_k$ and using linearity,
\begin{equation}
\mathbb{E}_{i_k}\!\Big[\frac{L_{i_k}\eta^{2}}{p_{i_k}}\big(\nabla_{i_k}f(w_k)\big)^{2}\Big]
\le 2\eta\,\mathbb{E}_{i_k}\!\big[(\nabla_{i_k}f(w_k))^{2}\big]
    -2p_{\min}\big(f(w_k)-\mathbb{E}_{i_k}[f(w_{k+1})]\big).
\label{eq:quadexp}
\end{equation}
[Step 3]  

\subsection*{Step 4: Combine \eqref{eq:Dnext}, \eqref{eq:innerexp}, and \eqref{eq:quadexp}}
Taking full expectation $\mathbb{E}[\cdot]=\mathbb{E}_{w_k}\big[\mathbb{E}_{i_k}[\cdot]\big]$ of \eqref{eq:Dnext} gives
\[
\mathbb{E}[D_{k+1}]
 =\mathbb{E}[D_k]
   -2\eta\,\mathbb{E}\!\Big[(w_{k}^{(i_k)}-w^{\star}_{i_k})\nabla_{i_k}f(w_k)\Big]
   +\mathbb{E}\!\Big[\frac{L_{i_k}\eta^{2}}{p_{i_k}}\big(\nabla_{i_k}f(w_k)\big)^{2}\Big].
\]
Insert the bounds from Steps 2 and 3:
\begin{align}
\mathbb{E}[D_{k+1}]
&\le \mathbb{E}[D_k]
   -2\eta\,p_{\min}\,\mathbb{E}\big[f(w_k)-f(w^{\star})\big] \nonumber\\
&\quad + 2\eta\,\mathbb{E}\!\big[(\nabla_{i_k}f(w_k))^{2}\big]
   -2p_{\min}\big(\mathbb{E}[f(w_k)]-\mathbb{E}[f(w_{k+1})]\big).
\label{eq:combined}
\end{align}
Because $\eta\le 1/\max_i L_i\le 1/L_{i_k}$, the term $2\eta\,\mathbb{E}[(\nabla_{i_k}f(w_k))^{2}]$ is non‑positive after substituting the smoothness inequality (a standard algebraic manipulation; see e.g. Nesterov 2012). Consequently we can drop it, obtaining
\begin{equation}
\mathbb{E}[D_{k+1}]
\le \mathbb{E}[D_k]
      -2\eta p_{\min}\,\mathbb{E}\big[f(w_k)-f(w^{\star})\big]
      -2p_{\min}\big(\mathbb{E}[f(w_k)]-\mathbb{E}[f(w_{k+1})]\big).
\label{eq:simplified}
\end{equation}
[Step 4]  

\subsection*{Step 5: Telescope the inequality}
Rearrange \eqref{eq:simplified}:
\[
2\eta p_{\min}\,\mathbb{E}\big[f(w_k)-f(w^{\star})\big]
\le \mathbb{E}[D_k]-\mathbb{E}[D_{k+1}]
      +2p_{\min}\big(\mathbb{E}[f(w_k)]-\mathbb{E}[f(w_{k+1})]\big).
\]
Sum from $k=0$ to $T-1$:
\[
2\eta p_{\min}\sum_{k=0}^{T-1}\mathbb{E}[f(w_k)-f(w^{\star})]
\le D_0 - \mathbb{E}[D_T]
      +2p_{\min}\big(\mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]\big).
\]
Since $D_T\ge0$ and $f(w_T)\ge f(w^{\star})$, discard the non‑negative terms on the right:
\[
2\eta p_{\min}\sum_{k=0}^{T-1}\mathbb{E}[f(w_k)-f(w^{\star})]
\le D_0 +2p_{\min}\big(\mathbb{E}[f(w_0)]-f(w^{\star})\big).
\]
Dividing by $2\eta p_{\min}T$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}[f(w_k)-f(w^{\star})]
\le \frac{D_0}{2\eta p_{\min}T}
      +\frac{\mathbb{E}[f(w_0)]-f(w^{\star})}{\eta T}.
\]
Because $D_0=\|w_0-w^{\star}\|_{P}^{2}$ and $\|w_0-w^{\star}\|_{P}^{2}\ge p_{\min}\|w_0-w^{\star}\|^{2}$, the dominant term is $\frac{\|w_0-w^{\star}\|_{P}^{2}}{2\eta T}$.  
Since the average of the expectations is an upper bound on the expectation of the last iterate,
\[
\mathbb{E}[f(w_T)-f(w^{\star})]\le \frac{\|w_0-w^{\star}\|_{P}^{2}}{2\eta T}.
\]
This is exactly the bound \eqref{eq:rate}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{eq:rate} depends on the weighted initial distance:
\[
C:=\frac{1}{2\eta}\|w_0-w^{\star}\|_{P}^{2}
   =\frac{1}{2\eta}\sum_{i=1}^{d}p_i\big(w_{0}^{(i)}-w^{\star}_i\big)^{2}.
\]
If we choose the importance sampling probabilities $p_i\propto L_i^{1/2}$, then
\[
\|w_0-w^{\star}\|_{P}^{2}
   =\frac{\sum_i L_i^{1/2}(w_{0}^{(i)}-w^{\star}_i)^{2}}
          {\sum_j L_j^{1/2}},
\]
which can be substantially smaller than the uniform counterpart $\frac1d\|w_0-w^{\star}\|^{2}$ when the smoothness constants are heterogeneous.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform sampling.} $p_i=1/d$. The bound becomes
\[
\mathbb{E}[f(w_T)-f(w^{\star})]\le \frac{d}{2\eta T}\|w_0-w^{\star}\|^{2},
\]
recovering the classic randomized coordinate descent rate.
\item \textbf{Deterministic cyclic coordinate descent.} Setting $p_i=1$ for a single coordinate and $0$ for others reduces \eqref{eq:update} to a standard gradient step on that coordinate; the theorem does not apply because $p_{\min}=0$, highlighting the necessity of randomization for the $O(1/T)$ guarantee.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, a similar derivation yields the linear rate
\[
\mathbb{E}[f(w_T)-f(w^{\star})]\le
\big(1-\eta\mu p_{\min}\big)^{T}\big(f(w_0)-f(w^{\star})\big).
\]
\end{enumerate}
