\section*{SignSGD}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.  
At iteration $k\in\{0,1,\dots\}$ we draw a stochastic gradient $g_k\in\R^{d}$ satisfying the
median‑based variance condition described in the Assumptions (Section~\ref{sec:ass}).  
The update rule of **SignSGD** is
\begin{equation}
\boxed{ \; x_{k+1}=x_{k}-\alpha\,\sign(g_k)\; }, \qquad
\sign(g_k)_i=
\begin{cases}
+1 & \text{if } (g_k)_i>0,\\[2mm]
-1 & \text{if } (g_k)_i<0,\\[2mm]
0  & \text{if } (g_k)_i=0,
\end{cases}
\label{eq:update}
\end{equation}
where $\alpha>0$ is a (scalar) stepsize common to all coordinates.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SignSGD with median‑based variance bound}
\label{alg:signsgd}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha>0$, max iterations $T$, initial point $x_0\in\R^{d}$
\For{$k=0$ {\bf to} $T-1$}
    \State Sample a stochastic gradient $g_k$ at $x_k$
    \State Compute coordinate‑wise sign $\; s_k \gets \sign(g_k)$
    \State Update \; $x_{k+1}\gets x_k-\alpha\,s_k$
\EndFor
\State \textbf{Output:} $x_T$ (or the average iterate $\bar x_T$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Take $w_0=0$, stepsize $\alpha=0.1$, and assume the stochastic gradient is exact,
i.e. $g_k=\nabla f(w_k)$.  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $\sign(g_k)$ & $w_{k+1}=w_k-\alpha\sign(g_k)$\\ \hline
0 & $0$   & $-6$ & $-1$ & $0-0.1(-1)=0.1$\\[2mm]
1 & $0.1$ & $-5.8$ & $-1$ & $0.1-0.1(-1)=0.2$\\[2mm]
2 & $0.2$ & $-5.6$ & $-1$ & $0.2-0.1(-1)=0.3$\\[2mm]
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.1)=8.41$, $f(0.2)=7.84$, $f(0.3)=7.29$,
showing a descent in each iteration.

\section*{Assumptions}
\label{sec:ass}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{ass:unbiased}
For every $k$, $\E[g_k\mid x_k]=\nabla f(x_k)$.
\end{assumption}
\begin{assumption}[Median‑based variance bound]\label{ass:median}
There exists a constant $\mu\in\bigl(0,\tfrac12\bigr]$ such that for each coordinate $i\in\{1,\dots,d\}$
\[
\Pr\bigl[\sign(g_k)_i=\sign(\nabla f(x_k)_i)\mid x_k\bigr]\;\ge\;\tfrac12+\mu .
\tag{A3}
\]
Equivalently,
\[
\Pr\bigl[\sign(g_k)_i\neq\sign(\nabla f(x_k)_i)\mid x_k\bigr]\le\tfrac12-\mu .
\]
\end{assumption}
\begin{assumption}[Bounded gradient norm]\label{ass:bounded}
There exists $G>0$ such that $\norm{\nabla f(x)}_2\le G$ for all $x$ in the domain of interest.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SignSGD]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:bounded} hold and fix a stepsize $\alpha\le \dfrac{2\mu}{L\sqrt{d}}$.
Let $\{x_k\}_{k=0}^T$ be the iterates generated by \eqref{eq:update}.
Then
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
\;\le\;
\frac{f(x_0)-f^\star}{\alpha\mu\sqrt{d}\,T}
\;+\;
\frac{L\alpha\sqrt{d}}{2\mu},
\tag{1}
\]
where $f^\star=\inf_{x}f(x)$.  
Choosing $\alpha = \dfrac{c}{\sqrt{T}}$ with $c\le \dfrac{2\mu}{L\sqrt{d}}$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The stepsize restriction $\alpha\le 2\mu/(L\sqrt d)$ guarantees that the
quadratic term in the smoothness inequality does not dominate the linear descent term.
\item \textbf{Hyper‑parameter sensitivity.} The convergence rate depends linearly on the
median bias $\mu$: larger $\mu$ (more reliable signs) yields faster descent.
\item \textbf{Comparison with SGD.} For the same $\alpha$, the bound \eqref{1} replaces the
$\norm{\nabla f}_2^2$ term of classic SGD by $\norm{\nabla f}_2$, reflecting the coarse
information carried by signs. Nonetheless the $O(1/\sqrt{T})$ rate is retained under the
median condition.
\end{itemize}

\section*{Discussion}
The theorem shows that even with the extreme quantisation to a single bit per coordinate,
SignSGD converges to a stationary point of a smooth non‑convex function as long as the
stochastic signs are more often correct than wrong (parameter $\mu$).  The median‑based
assumption is weaker than a bounded variance assumption and is satisfied, for example,
by symmetric heavy‑tailed noise distributions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate‑wise sign expectation]\label{lem:signexp}
Under Assumption~\ref{ass:median},
\[
\E\!\bigl[\sign(g_k)_i\,\nabla f_i(x_k)\mid x_k\bigr]
=|\nabla f_i(x_k)|(2p_i-1)\ge 2\mu\,|\nabla f_i(x_k)|
\]
where $p_i=\Pr[\sign(g_k)_i=\sign(\nabla f_i(x_k))\mid x_k]$.
\end{lemma}
\begin{proof}
By definition,
\[
\E[\sign(g_k)_i\,\nabla f_i\mid x_k]
= \nabla f_i\bigl(p_i\cdot 1 + (1-p_i)\cdot(-1)\bigr)
= \nabla f_i(2p_i-1).
\]
If $\nabla f_i\ge0$, then $|\nabla f_i|=\nabla f_i$ and the claim follows; the case
$\nabla f_i<0$ is symmetric.  Using $p_i\ge\tfrac12+\mu$ yields the lower bound.
\end{proof}

\begin{lemma}[Relation between $\ell_1$ and $\ell_2$ norms]\label{lem:l1l2}
For any vector $v\in\R^{d}$,
\[
\norm{v}_1 \;\ge\; \frac{1}{\sqrt d}\,\norm{v}_2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
Cauchy–Schwarz gives $\norm{v}_1 = \ip{|v|}{\mathbf{1}}\le
\norm{v}_2\norm{\mathbf{1}}_2 = \norm{v}_2\sqrt d$, whence the reverse inequality.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t.\ the randomness of
$g_k$ conditional on $x_k$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1] (Smoothness inequality).}
From Assumption~\ref{ass:smooth} with $y=x_{k+1}=x_k-\alpha\sign(g_k)$,
\[
f(x_{k+1})\le f(x_k)-\alpha\ip{\nabla f(x_k)}{\sign(g_k)}+\frac{L\alpha^2}{2}\norm{\sign(g_k)}^2 .
\tag{2}
\]

\noindent\textbf{[Step 2] (Taking conditional expectation).}
Apply $\E[\cdot\mid x_k]$ to \eqref{2}:
\[
\E\!\bigl[f(x_{k+1})\mid x_k\bigr]
\le f(x_k)-\alpha\,\E\!\bigl[\ip{\nabla f(x_k)}{\sign(g_k)}\mid x_k\bigr]
+\frac{L\alpha^2}{2}\E\!\bigl[\norm{\sign(g_k)}^2\mid x_k\bigr].
\tag{3}
\]

\noindent\textbf{[Step 3] (Bounding the inner product).}
Using Lemma~\ref{lem:signexp} coordinate‑wise and summing,
\[
\E\!\bigl[\ip{\nabla f(x_k)}{\sign(g_k)}\mid x_k\bigr]
=\sum_{i=1}^d \E\!\bigl[\sign(g_k)_i\,\nabla f_i(x_k)\mid x_k\bigr]
\ge 2\mu\sum_{i=1}^d |\nabla f_i(x_k)|
=2\mu\,\norm{\nabla f(x_k)}_1 .
\tag{4}
\]

\noindent\textbf{[Step 4] (Bounding the norm of the sign vector).}
Since each coordinate of $\sign(g_k)$ belongs to $\{-1,0,1\}$,
\[
\norm{\sign(g_k)}^2 = \sum_{i=1}^d \sign(g_k)_i^2 \le d .
\tag{5}
\]
Hence $\E[\norm{\sign(g_k)}^2\mid x_k]\le d$.

\noindent\textbf{[Step 5] (Insert bounds into \eqref{3}).}
Combining \eqref{3}, \eqref{4}, and \eqref{5} yields
\[
\E\!\bigl[f(x_{k+1})\mid x_k\bigr]
\le f(x_k)-2\alpha\mu\,\norm{\nabla f(x_k)}_1
+\frac{L\alpha^2 d}{2}.
\tag{6}
\]

\noindent\textbf{[Step 6] (Replace $\ell_1$ norm by $\ell_2$ norm).}
By Lemma~\ref{lem:l1l2},
\[
\norm{\nabla f(x_k)}_1\ge \frac{1}{\sqrt d}\,\norm{\nabla f(x_k)}_2 .
\tag{7}
\]
Insert \eqref{7} into \eqref{6}:
\[
\E\!\bigl[f(x_{k+1})\mid x_k\bigr]
\le f(x_k)-\frac{2\alpha\mu}{\sqrt d}\,\norm{\nabla f(x_k)}_2
+\frac{L\alpha^2 d}{2}.
\tag{8}
\]

\noindent\textbf{[Step 7] (Take total expectation).}
Denote $\Phi_k:=\E[f(x_k)]$.  Taking expectation of \eqref{8} over the history up to $k$,
\[
\Phi_{k+1}\le \Phi_k-\frac{2\alpha\mu}{\sqrt d}\,\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
+\frac{L\alpha^2 d}{2}.
\tag{9}
\]

\noindent\textbf{[Step 8] (Summation over $k=0,\dots,T-1$).}
Sum \eqref{9} from $k=0$ to $T-1$:
\[
\Phi_T \le \Phi_0
-\frac{2\alpha\mu}{\sqrt d}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
+\frac{L\alpha^2 d T}{2}.
\tag{10}
\]

\noindent\textbf{[Step 9] (Rearrange).}
Since $\Phi_T\ge f^\star$, we obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
\le
\frac{\Phi_0-f^\star}{\alpha\mu\sqrt d\,T}
+\frac{L\alpha\sqrt d}{2\mu}.
\tag{11}
\]
Recall $\Phi_0=f(x_0)$.  Equation \eqref{11} is exactly the bound \eqref{1} claimed in the theorem.

\noindent\textbf{[Step 10] (Choosing the stepsize).}
Set $\alpha = c/\sqrt T$ with $c\le 2\mu/(L\sqrt d)$.
Then the second term on the right-hand side of \eqref{11} becomes
$\frac{Lc}{2\mu\sqrt T}\le \frac{1}{\sqrt T}$,
and the first term scales as $O(1/\sqrt T)$ as well.
Thus
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
= O\!\Bigl(\frac{1}{\sqrt T}\Bigr).
\]
This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{11},
\[
\boxed{
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
\le
\underbrace{\frac{f(x_0)-f^\star}{\alpha\mu\sqrt d}}_{C_1}
\frac{1}{T}
+
\underbrace{\frac{L\alpha\sqrt d}{2\mu}}_{C_2}.
}
\]
Choosing $\alpha = \dfrac{2\mu}{L\sqrt d}\,\dfrac{1}{\sqrt T}$ gives $C_1 = \dfrac{L(f(x_0)-f^\star)}{2\mu^2}\sqrt d$ and $C_2 = \dfrac{1}{\sqrt T}$, i.e.
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_k)}_2\bigr]
\le
\frac{L(f(x_0)-f^\star)\sqrt d}{2\mu^2}\,\frac{1}{\sqrt T}
\;+\;
\frac{1}{\sqrt T}
=
O\!\Bigl(\frac{1}{\sqrt T}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Exact gradients.} If $g_k=\nabla f(x_k)$ deterministically, then $p_i=1$,
$\mu=\tfrac12$, and the bound simplifies to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\norm{\nabla f(x_k)}_2
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha\sqrt d\,T}
+\frac{L\alpha\sqrt d}{1}.
\]
\item \textbf{High‑dimensional limit.} As $d\to\infty$, the factor $\sqrt d$ appears both in the
denominator of the linear term and the numerator of the quadratic term.  Proper scaling of
$\alpha\propto 1/\sqrt d$ restores the $O(1/\sqrt T)$ rate independent of $d$.
\item \textbf{Convex $f$.} If $f$ is convex, the same derivation yields a bound on the
optimality gap $f(\bar x_T)-f^\star$ of order $O(1/\sqrt T)$, matching the classic SignSGD analysis.
\end{enumerate}
