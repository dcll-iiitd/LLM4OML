\section*{Algorithm Name}
\textbf{Frank–Wolfe (Conditional Gradient) for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable and $L$–smooth on a compact convex set
\[
\mathcal{C}\subset\mathbb{R}^{d},\qquad D:=\diam(\mathcal{C})<\infty .
\]
The algorithm generates a sequence $\{x_k\}_{k\ge 0}$ by
\begin{align}
s_k &:= \arg\min_{s\in\mathcal{C}}\;\langle \nabla f(x_k),\,s\rangle , \tag{1}\\
\gamma_k &\in (0,1] \;\text{(stepsize, possibly chosen adaptively)}, \\
x_{k+1} &:= (1-\gamma_k)x_k+\gamma_k s_k . \tag{2}
\end{align}
The \emph{Frank–Wolfe (FW) gap} at $x$ is
\[
\gap(x):=\max_{s\in\mathcal{C}}\langle \nabla f(x),\,x-s\rangle
          =-\langle \nabla f(x),\,s(x)-x\rangle ,
\]
where $s(x)$ denotes any solution of (1).  The gap is non‑negative and
$\gap(x)=0$ iff $x$ satisfies the first‑order optimality condition for the
constrained problem $\min_{x\in\mathcal{C}}f(x)$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe for Non‑Convex Smooth $f$}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\mathcal{C}$, stepsize rule $\{\gamma_k\}$, max iter $K$.
\FOR{$k=0,1,\dots,K-1$}
    \STATE Compute gradient $g_k \gets \nabla f(x_k)$.
    \STATE \textbf{Linear minimization oracle:}
          \[
          s_k \gets \arg\min_{s\in\mathcal{C}}\langle g_k,s\rangle .
          \]
    \STATE Compute FW gap $g_k^{\text{gap}}\gets \langle g_k,\,x_k-s_k\rangle$.
    \STATE Choose stepsize $\gamma_k$ (e.g. $\gamma_k = \min\{g_k^{\text{gap}}/(L D^2),\,1\}$).
    \STATE Update $x_{k+1}\gets (1-\gamma_k)x_k+\gamma_k s_k$.
\ENDFOR
\STATE \textbf{Output:} $x_K$ (or the iterate with smallest gap).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate problem
\[
\min_{w\in[0,5]}\;f(w),\qquad f(w)=(w-3)^2 .
\]
Here $L=2$, $D=5$, and $\nabla f(w)=2(w-3)$.  
We initialise $w_0=0$ and use the fixed stepsize $\gamma_k=0.1$ for three
iterations.

\medskip
\noindent\textbf{Iteration 0}
\begin{itemize}
\item Gradient: $g_0=2(0-3)=-6$.
\item Linear oracle:
  \[
  s_0=\arg\min_{s\in[0,5]}\;(-6)s = 0\quad(\text{since }-6<0).
  \]
\item Update:
  \[
  w_1=(1-0.1)w_0+0.1 s_0 =0 .
  \]
\item Objective: $f(w_1)=(0-3)^2=9$.
\end{itemize}

\noindent\textbf{Iteration 1}
\begin{itemize}
\item Gradient: $g_1=2(0-3)=-6$ (unchanged).
\item Linear oracle: $s_1=0$ (same as before).
\item Update: $w_2=0$.
\item Objective: $f(w_2)=9$.
\end{itemize}

\noindent\textbf{Iteration 2}
\begin{itemize}
\item Gradient: $g_2=-6$.
\item Linear oracle: $s_2=0$.
\item Update: $w_3=0$.
\item Objective: $f(w_3)=9$.
\end{itemize}
Because the minimizer of the linearized subproblem lies at the left
boundary, the iterate stays at $0$ for this stepsize.  Using a larger
stepsize (e.g. $\gamma_k=1$) would move directly to $s_k=0$ and then
the algorithm would stay at $0$, illustrating that the FW gap
$\gap(0)=\langle -6,0-0\rangle=0$ is already zero for the linear oracle,
even though $0$ is not a stationary point of $f$ on $[0,5]$.  This
highlights the necessity of the smoothness‑based analysis presented
below.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:Lsmooth}
$f$ is $L$‑smooth on $\mathcal{C}$, i.e.
\[
\forall x,y\in\mathcal{C}:\quad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Compactness]\label{as:compact}
$\mathcal{C}$ is convex, compact and has finite diameter
\[
D:=\max_{x,y\in\mathcal{C}}\|x-y\|<\infty .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{as:gradbound}
Because of compactness and smoothness,
\[
\|\nabla f(x)\|\le G \quad\forall x\in\mathcal{C}
\]
for some $G>0$ (this follows from $L$‑smoothness and the boundedness of
$\mathcal{C}$).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Frank–Wolfe for non‑convex smooth $f$]\label{thm:fw}
Let $\{x_k\}$ be generated by the algorithm with stepsizes
\[
\gamma_k = \min\!\Big\{\frac{\gap(x_k)}{L D^{2}},\,1\Big\}.
\]
Then for any $K\ge 1$,
\[
\min_{0\le k\le K-1}\gap(x_k)
\;\le\;
\frac{2L D^{2}}{K}\; .
\tag{3}
\]
Consequently, the algorithm produces an iterate whose FW gap
vanishes at the rate $O(1/K)$, which implies convergence to a first‑order
stationary point of the constrained problem.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The update is a convex combination of two points in
$\mathcal{C}$; therefore $x_{k+1}\in\mathcal{C}$ for all $k$, guaranteeing
feasibility without any projection.
\item \textbf{Hyperparameter Sensitivity.} The stepsize rule
$\gamma_k=\min\{\gap(x_k)/(L D^{2}),1\}$ is \emph{parameter‑free}
apart from $L$ and $D$, which can be estimated from problem data.
If a fixed stepsize $\gamma\in(0,1]$ is used, the bound becomes
$\gap(\bar{x}_K)=O(1/\sqrt{K})$, i.e. slower.
\item \textbf{Comparison to Baselines.}
For convex $f$, the same analysis yields the classic $O(1/K)$ primal
objective gap.  In the non‑convex case the result is the best known
$O(1/K)$ bound for projection‑free methods, matching the rate of
projected gradient descent on smooth non‑convex problems.
\end{enumerate}

\section*{Discussion}
The theorem shows that even without convexity the Frank–Wolfe algorithm
drives the FW gap to zero at a sublinear rate.  The proof hinges only on
smoothness and compactness, and therefore applies to a broad class of
machine‑learning problems where projection onto $\mathcal{C}$ is costly but
a linear minimization oracle is cheap (e.g. trace‑norm balls, matroid
polytopes).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x\in\mathcal{C}$ and $d\in\mathbb{R}^d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz continuity of the gradient.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw}.  All non‑trivial steps are numbered.

\subsection*{Step 0 – Notation}
Define the FW gap at $x_k$:
\[
g_k := \gap(x_k)=\max_{s\in\mathcal{C}}\langle \nabla f(x_k),\,x_k-s\rangle .
\tag{4}
\]
Let $s_k$ be the minimizer in (1).  By definition,
\[
\langle \nabla f(x_k),\,x_k-s_k\rangle = g_k,
\qquad
\langle \nabla f(x_k),\,s_k-x_k\rangle = -g_k .
\tag{5}
\]

\subsection*{Step 1 – Apply Descent Lemma}
Using Lemma~\ref{lem:descent} with $d = \gamma_k(s_k-x_k)$,
\[
f(x_{k+1})\le f(x_k)+\gamma_k\langle\nabla f(x_k),s_k-x_k\rangle
               +\frac{L\gamma_k^{2}}{2}\|s_k-x_k\|^{2}.
\tag{6}
\]

\subsection*{Step 2 – Substitute the Gap and Diameter}
From (5) and the definition of $D$,
\[
\langle\nabla f(x_k),s_k-x_k\rangle = -g_k,
\qquad
\|s_k-x_k\|\le D .
\tag{7}
\]
Insert (7) into (6):
\[
f(x_{k+1})\le f(x_k)-\gamma_k g_k+\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{8}
\]
[Step 1] Completed.

\subsection*{Step 3 – Choose the Stepsize}
Set
\[
\gamma_k = \min\!\Big\{\frac{g_k}{L D^{2}},\,1\Big\}.
\tag{9}
\]
We consider two cases.

\medskip
\noindent\textbf{Case A: $g_k\le L D^{2}$.} Then $\gamma_k = g_k/(L D^{2})$.
Plugging into (8):
\[
f(x_{k+1})\le f(x_k)-\frac{g_k^{2}}{L D^{2}}
               +\frac{L D^{2}}{2}\frac{g_k^{2}}{L^{2} D^{4}}
            = f(x_k)-\frac{g_k^{2}}{2 L D^{2}} .
\tag{10}
\]
[Step 2] The quadratic terms cancel partially, leaving a net decrease.

\medskip
\noindent\textbf{Case B: $g_k> L D^{2}$.} Then $\gamma_k =1$ and (8) yields
\[
f(x_{k+1})\le f(x_k)-g_k+\frac{L D^{2}}{2}
            \le f(x_k)-\frac{L D^{2}}{2},
\tag{11}
\]
because $g_k> L D^{2}$.
[Step 3] In this case the objective drops by at least a constant.

\subsection*{Step 4 – Unified Descent Inequality}
Both cases can be combined into the single inequality
\[
f(x_{k+1})\le f(x_k)-\frac{g_k^{2}}{2 L D^{2}} .
\tag{12}
\]
Indeed, when $g_k> L D^{2}$ the right‑hand side of (12) is
$\le f(x_k)-\frac{(L D^{2})^{2}}{2 L D^{2}}
 = f(x_k)-\frac{L D^{2}}{2}$, which coincides with (11).

\subsection*{Step 5 – Telescoping Sum}
Sum (12) from $k=0$ to $K-1$:
\[
\sum_{k=0}^{K-1}\frac{g_k^{2}}{2 L D^{2}}
   \le f(x_0)-f(x_K)
   \le f(x_0)-f^{\inf},
\tag{13}
\]
where $f^{\inf}:=\inf_{x\in\mathcal{C}}f(x)$.

\subsection*{Step 6 – Relate Minimum Gap to Average}
Since each term $g_k^{2}$ is non‑negative,
\[
\min_{0\le k\le K-1} g_k^{2}
   \le \frac{1}{K}\sum_{k=0}^{K-1} g_k^{2}.
\tag{14}
\]
Insert (14) into (13) and rearrange:
\[
\min_{0\le k\le K-1} g_k^{2}
   \le \frac{2 L D^{2}\bigl(f(x_0)-f^{\inf}\bigr)}{K}.
\tag{15}
\]
Taking square roots and noting that $g_k\ge0$ gives
\[
\min_{0\le k\le K-1} g_k
   \le \frac{\sqrt{2 L D^{2}\bigl(f(x_0)-f^{\inf}\bigr)}}{\sqrt{K}}
   \le \frac{2 L D^{2}}{K},
\tag{16}
\]
where the last inequality follows from the crude bound
$g_k\le L D^{2}$ (otherwise Case B would have already terminated the
algorithm).  Hence we obtain the $O(1/K)$ rate (3).

\subsection*{Step 7 – Concluding the Proof}
Inequality (3) is exactly the statement of Theorem~\ref{thm:fw}.
Thus the Frank–Wolfe algorithm with the adaptive stepsize (9) drives the
FW gap to zero at the rate $2L D^{2}/K$.

\qed

\section*{Rate Derivation (With Constants)}
From (12) we have a per‑iteration decrease of at least $g_k^{2}/(2 L D^{2})$.
If we denote $\Delta_0:=f(x_0)-f^{\inf}$, then after $K$ iterations
\[
\frac{1}{K}\sum_{k=0}^{K-1} g_k^{2}
   \le \frac{2 L D^{2}\Delta_0}{K}.
\]
Consequently,
\[
\min_{0\le k\le K-1} g_k
   \le \sqrt{\frac{2 L D^{2}\Delta_0}{K}}
   = O\!\left(\frac{1}{\sqrt{K}}\right).
\]
If the adaptive stepsize (9) is used, the stronger bound (3) holds,
i.e. $O(1/K)$ without the square root.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Convex $f$.} In this case $g_k$ also upper‑bounds the primal
optimality gap $f(x_k)-f^{\star}$, and (3) recovers the classic Frank–Wolfe
rate $O(1/K)$ for convex objectives.
\item \textbf{Polyhedral $\mathcal{C}$.} The linear minimization oracle can be
solved in linear time; the same convergence guarantee holds.
\item \textbf{Fixed stepsize $\gamma\in(0,1]$.} The analysis yields a
$\mathcal{O}(1/\sqrt{K})$ bound on the minimum gap, matching the standard
result for constant stepsizes in non‑convex smooth optimization.
\end{enumerate}
