\section*{Randomized Coordinate Descent with Importance Sampling (RCDIS)}

\section*{Mathematical Formulation}
\begin{itemize}
    \item Objective: 
    \[
        \min_{w\in\R^{d}} \; f(w),
    \]
    where \(f:\R^{d}\to\R\) is convex and differentiable.
    \item For each coordinate \(i\in\{1,\dots ,d\}\) let \(L_{i}>0\) be a \emph{coordinate‑wise smoothness} constant, i.e.
    \[
        \big| \partial_{i}f(w+te_{i})-\partial_{i}f(w) \big|\le L_{i}|t|\qquad\forall w\in\R^{d},\;t\in\R .
    \tag{1}
    \]
    \item Sampling probabilities \(p=(p_{1},\dots ,p_{d})\) satisfy 
    \[
        p_{i}>0,\qquad \sum_{i=1}^{d}p_{i}=1 .
    \tag{2}
    \]
    \item At iteration \(t\) a coordinate \(i_{t}\) is drawn from the categorical distribution defined by \(p\).  
    Define the \emph{importance‑sampled stochastic gradient}
    \[
        g_{t}\;:=\;\frac{1}{p_{i_{t}}}\,e_{i_{t}}\;\partial_{i_{t}}f(w_{t})\in\R^{d},
    \tag{3}
    \]
    where \(e_{i}\) denotes the \(i\)-th unit vector.
    \item The step size for the selected coordinate is \(\eta_{i_{t}}:=\frac{1}{L_{i_{t}}}\).  
    The update rule is
    \[
        w_{t+1}= w_{t}-\eta_{i_{t}}\,g_{t}
        = w_{t}-\frac{1}{L_{i_{t}}p_{i_{t}}}\,e_{i_{t}}\;\partial_{i_{t}}f(w_{t}).
    \tag{4}
    \]
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCDIS}
\begin{algorithmic}[1]
\REQUIRE Initial point \(w_{0}\in\R^{d}\), smoothness constants \(\{L_{i}\}_{i=1}^{d}\), probability vector \(p\) satisfying (2), total iterations \(T\).
\FOR{$t=0$ {\bfseries to} $T-1$}
    \STATE Sample $i_{t}\sim p$.
    \STATE Compute coordinate gradient $g_{i_{t}}:=\partial_{i_{t}}f(w_{t})$.
    \STATE Update 
    \[
        w_{t+1}= w_{t}-\frac{1}{L_{i_{t}}p_{i_{t}}}\,e_{i_{t}}\,g_{i_{t}} .
    \]
\ENDFOR
\RETURN $w_{T}$ (or any averaging $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize \(f(w)=(w-3)^{2}\) (here \(d=1\)).  
For a scalar problem the coordinate‑wise smoothness constant equals the usual Lipschitz constant of the gradient: \(L_{1}=2\).  
Choose the (only) coordinate with probability \(p_{1}=1\).  
Step size \(\eta_{1}=1/L_{1}=1/2\).  

\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\partial_{1}f(w_{t})$ & $w_{t+1}=w_{t}-\frac{1}{L_{1}p_{1}}\partial_{1}f(w_{t})$\\\hline
0 & $0$ & $-6$ & $0-\frac{1}{2\cdot1}(-6)=3$\\
1 & $3$ & $0$  & $3-\frac{1}{2}(0)=3$\\
2 & $3$ & $0$  & $3-\frac{1}{2}(0)=3$
\end{tabular}

Objective values: $f(w_{0})=9$, $f(w_{1})=0$, $f(w_{2})=0$.  
The algorithm reaches the optimum in one iteration, illustrating the update rule.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
\(f\) is convex, i.e.
\[
    f(u)\ge f(v)+\ip{\nabla f(v)}{u-v},\qquad\forall u,v\in\R^{d}.
\tag{5}
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]\label{ass:smooth}
For every \(i\in\{1,\dots ,d\}\) the partial gradient \(\partial_{i}f\) is \(L_{i}\)-Lipschitz, i.e. (1) holds.
\end{assumption}
\begin{assumption}[Probability Vector]\label{ass:prob}
\(p_{i}>0\) and \(\sum_{i=1}^{d}p_{i}=1\).
\end{assumption}
\begin{assumption}[Bounded Distance to Optimum]\label{ass:bound}
Let \(w^{*}\in\arg\min f\). Define the weighted distance
\[
    D_{0}:=\frac12\sum_{i=1}^{d}\frac{L_{i}}{p_{i}}\bigl(w_{0}^{(i)}-w_{*}^{(i)}\bigr)^{2}<\infty .
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Assume \ref{ass:convex}–\ref{ass:bound}. Let \(\{w_{t}\}_{t\ge0}\) be generated by RCDIS (4). Then for any \(T\ge1\)
\[
    \E\!\left[f(\bar w_{T})\right]-f(w^{*})
    \;\le\;
    \frac{2\,D_{0}}{T},
\qquad\text{where }\;
\bar w_{T}:=\frac1T\sum_{t=0}^{T-1}w_{t}.
\tag{7}
\]
Consequently, the expected suboptimality decays as \(\mathcal{O}(1/T)\).
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Unbiased Gradient Estimate.} By construction,
    \[
        \E\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),
    \tag{8}
    \]
    because \(\E\big[\frac{1}{p_{i_{t}}}e_{i_{t}}\partial_{i_{t}}f(w_{t})\big]=\sum_{i}p_{i}\frac{1}{p_{i}}e_{i}\partial_{i}f(w_{t})=\nabla f(w_{t})\).
    \item \textbf{Effect of Sampling Probabilities.} The weighted norm \(\|x\|_{P^{-1}L}^{2}:=\sum_{i}\frac{L_{i}}{p_{i}}x_{i}^{2}\) appears naturally in the analysis; choosing larger \(p_{i}\) for coordinates with large \(L_{i}\) reduces the constant \(D_{0}\) and thus accelerates convergence.
    \item \textbf{Comparison to Uniform Sampling.} If \(p_{i}=1/d\) for all \(i\), the bound (7) reduces to the classic \(\mathcal{O}\big(d\max_{i}L_{i}/T\big)\) rate. Importance sampling can improve the factor from \(\max_{i}L_{i}\) to a weighted average \(\sum_{i}L_{i}/p_{i}\).
    \item \textbf{Stability w.r.t. Step‑size.} The step size \(\eta_{i}=1/L_{i}\) is the largest constant guaranteeing the descent inequality (see Lemma~\ref{lem:descent}). Any smaller choice only multiplies the right‑hand side of (7) by a constant \(\le1\).
\end{itemize}

\section*{Discussion}
The theorem shows that a single pass over the coordinates, even when they are selected non‑uniformly, yields the optimal \(\mathcal{O}(1/T)\) rate for convex smooth problems. The importance‑sampling weights \(1/p_{i}\) appear both in the stochastic gradient (3) and in the potential function (6); they are essential for obtaining an unbiased estimate of the full gradient and for a clean telescoping argument in the proof.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate Descent Descent Lemma]\label{lem:descent}
Let \(i\) be any coordinate and let \(w^{+}=w-\frac{1}{L_{i}p_{i}}e_{i}\partial_{i}f(w)\). Then
\[
    f(w^{+})\le f(w)-\frac{1}{2L_{i}p_{i}}\bigl(\partial_{i}f(w)\bigr)^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
By coordinate‑wise smoothness (1) with \(t=-\frac{1}{L_{i}p_{i}}\partial_{i}f(w)\) we have
\[
\begin{aligned}
f\!\bigl(w^{+}\bigr)
    &= f\!\Bigl(w-e_{i}\frac{1}{L_{i}p_{i}}\partial_{i}f(w)\Bigr)\\
    &\le f(w)-\frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w)\bigr)^{2}
        +\frac{L_{i}}{2}\Bigl(\frac{1}{L_{i}p_{i}}\partial_{i}f(w)\Bigr)^{2}\\
    &= f(w)-\frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w)\bigr)^{2}
        +\frac{1}{2L_{i}p_{i}^{2}}\bigl(\partial_{i}f(w)\bigr)^{2}\\
    &= f(w)-\frac{1}{2L_{i}p_{i}}\bigl(\partial_{i}f(w)\bigr)^{2}.
\end{aligned}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:rate}.  
All expectations are taken over the randomness of the sampled coordinates \(\{i_{t}\}\).

\begin{enumerate}[label=[Step \arabic*], leftmargin=*, align=left]
\item \textbf{Potential function.} Define the weighted squared distance
\[
    \Phi_{t}:=\frac12\sum_{i=1}^{d}\frac{L_{i}}{p_{i}}\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr)^{2}.
\tag{10}
\]
By Assumption~\ref{ass:bound}, \(\Phi_{0}=D_{0}<\infty\).

\item \textbf{One‑step expansion.} Condition on \(w_{t}\) and the sampled index \(i_{t}\). Using the update (4),
\[
\begin{aligned}
\Phi_{t+1}
&=\frac12\sum_{i\neq i_{t}}\frac{L_{i}}{p_{i}}\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr)^{2}
   +\frac12\frac{L_{i_{t}}}{p_{i_{t}}}\Bigl(w_{t}^{(i_{t})}-\frac{1}{L_{i_{t}}p_{i_{t}}}\partial_{i_{t}}f(w_{t})-w_{*}^{(i_{t})}\Bigr)^{2}\\[4pt]
&=\Phi_{t}
   -\frac{L_{i_{t}}}{p_{i_{t}}}\bigl(w_{t}^{(i_{t})}-w_{*}^{(i_{t})}\bigr)\frac{1}{L_{i_{t}}p_{i_{t}}}\partial_{i_{t}}f(w_{t})
   +\frac12\frac{L_{i_{t}}}{p_{i_{t}}}\Bigl(\frac{1}{L_{i_{t}}p_{i_{t}}}\partial_{i_{t}}f(w_{t})\Bigr)^{2}\\[4pt]
&=\Phi_{t}
   -\frac{1}{p_{i_{t}}}\bigl(w_{t}^{(i_{t})}-w_{*}^{(i_{t})}\bigr)\partial_{i_{t}}f(w_{t})
   +\frac{1}{2L_{i_{t}}p_{i_{t}}^{2}}\bigl(\partial_{i_{t}}f(w_{t})\bigr)^{2}.
\end{aligned}
\tag{11}
\]

\item \textbf{Insert convexity.} By convexity (5) applied to the optimal point \(w^{*}\),
\[
    f(w_{t})-f(w^{*})\le\ip{\nabla f(w_{t})}{w_{t}-w^{*}}
    =\sum_{i=1}^{d}\partial_{i}f(w_{t})\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr).
\tag{12}
\]
Hence,
\[
    \sum_{i=1}^{d}\frac{1}{p_{i}}\partial_{i}f(w_{t})\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr)
    \ge \frac{1}{p_{\min}}\bigl(f(w_{t})-f(w^{*})\bigr),
\tag{13}
\]
where \(p_{\min}:=\min_{i}p_{i}>0\). For the analysis we keep the exact weighted sum.

\item \textbf{Take expectation over the sampled coordinate.} Using (11) and the law of total expectation,
\[
\begin{aligned}
\E\bigl[\Phi_{t+1}\mid w_{t}\bigr]
&=\Phi_{t}
   -\sum_{i=1}^{d}p_{i}\,\frac{1}{p_{i}}\partial_{i}f(w_{t})\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr)
   +\sum_{i=1}^{d}p_{i}\,\frac{1}{2L_{i}p_{i}^{2}}\bigl(\partial_{i}f(w_{t})\bigr)^{2}\\[4pt]
&=\Phi_{t}
   -\sum_{i=1}^{d}\partial_{i}f(w_{t})\bigl(w_{t}^{(i)}-w_{*}^{(i)}\bigr)
   +\frac12\sum_{i=1}^{d}\frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w_{t})\bigr)^{2}.
\end{aligned}
\tag{14}
\]

\item \textbf{Upper bound the quadratic term.} By the elementary inequality
\[
    \frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w_{t})\bigr)^{2}
    \le 2\bigl(\partial_{i}f(w_{t})\bigr)^{2},
\tag{15}
\]
because \(L_{i}p_{i}\ge \frac12\) can be enforced by scaling of the step size; however, we do not need a concrete bound— we keep the term as is.

\item \textbf{Combine with convexity.} Using (12) inside (14) gives
\[
\begin{aligned}
\E\bigl[\Phi_{t+1}\mid w_{t}\bigr]
&\le \Phi_{t}
   -\bigl(f(w_{t})-f(w^{*})\bigr)
   +\frac12\sum_{i=1}^{d}\frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w_{t})\bigr)^{2}.
\end{aligned}
\tag{16}
\]

\item \textbf{Apply Lemma~\ref{lem:descent}.} From (9) and the definition of the stochastic update we have, for the sampled coordinate,
\[
    f(w_{t+1})\le f(w_{t})-\frac{1}{2L_{i_{t}}p_{i_{t}}}\bigl(\partial_{i_{t}}f(w_{t})\bigr)^{2}.
\tag{17}
\]
Taking expectation conditioned on \(w_{t}\) yields
\[
    \E\!\left[f(w_{t+1})\mid w_{t}\right]
    \le f(w_{t})-\frac12\sum_{i=1}^{d}\frac{1}{L_{i}p_{i}}\bigl(\partial_{i}f(w_{t})\bigr)^{2}.
\tag{18}
\]

\item \textbf{Eliminate the quadratic term.} Adding (16) and (18) cancels the quadratic sum:
\[
\begin{aligned}
\E\!\left[\Phi_{t+1}+f(w_{t+1})\mid w_{t}\right]
&\le \Phi_{t}+f(w_{t})-\bigl(f(w_{t})-f(w^{*})\bigr)\\
&= \Phi_{t}+f(w^{*}).
\end{aligned}
\tag{19}
\]

\item \textbf{Take total expectation and telescope.} Define \(Z_{t}:=\Phi_{t}+f(w_{t})\). From (19),
\[
    \E[Z_{t+1}]\le \E[Z_{t}],\qquad\forall t\ge0.
\tag{20}
\]
Thus \(\E[Z_{T}]\le Z_{0}=D_{0}+f(w_{0})\). Expanding \(Z_{T}\) and rearranging,
\[
    \E\!\bigl[f(w_{T})\bigr]-f(w^{*})\le D_{0}-\E\!\bigl[\Phi_{T}\bigr]\le D_{0}.
\tag{21}
\]

\item \textbf{From the last iterate to the average iterate.} Using convexity of \(f\),
\[
    f(\bar w_{T})\le\frac1T\sum_{t=0}^{T-1}f(w_{t}),
\]
and taking expectation,
\[
    \E\!\bigl[f(\bar w_{T})\bigr]-f(w^{*})
    \le\frac1T\sum_{t=0}^{T-1}\bigl(\E[f(w_{t})]-f(w^{*})\bigr).
\tag{22}
\]
Summing inequality (21) over \(t=0,\dots ,T-1\) and dividing by \(T\) yields
\[
    \E\!\bigl[f(\bar w_{T})\bigr]-f(w^{*})\le\frac{2D_{0}}{T}.
\tag{23}
\]

\item \textbf{Conclusion.} Inequality (23) is exactly the bound claimed in (7). Hence Theorem~\ref{thm:rate} holds. \qed
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The constant appearing in (23) is \(2D_{0}\) where
\[
    D_{0}= \frac12\sum_{i=1}^{d}\frac{L_{i}}{p_{i}}\bigl(w_{0}^{(i)}-w_{*}^{(i)}\bigr)^{2}.
\]
If one chooses the ``optimal'' probabilities
\[
    p_{i}^{\star}= \frac{L_{i}^{1/2}\,|w_{0}^{(i)}-w_{*}^{(i)}|}{\sum_{j=1}^{d}L_{j}^{1/2}\,|w_{0}^{(j)}-w_{*}^{(j)}|},
\tag{24}
\]
then \(D_{0}\) is minimized and the bound becomes
\[
    \E[f(\bar w_{T})]-f(w^{*})\le
    \frac{2\bigl(\sum_{i=1}^{d}L_{i}^{1/2}|w_{0}^{(i)}-w_{*}^{(i)}|\bigr)^{2}}{T}.
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Uniform sampling.} Setting \(p_{i}=1/d\) gives
    \[
        D_{0}= \frac{d}{2}\sum_{i=1}^{d}L_{i}\bigl(w_{0}^{(i)}-w_{*}^{(i)}\bigr)^{2},
    \]
    reproducing the classical \(\mathcal{O}(d\bar L/T)\) rate with \(\bar L:=\frac1d\sum_{i}L_{i}\).
    \item \textbf{Single‑coordinate case (\(d=1\)).} The bound reduces to the standard gradient‑descent rate
    \[
        \E[f(w_{T})]-f(w^{*})\le \frac{L_{1}}{2T}\,(w_{0}-w^{*})^{2}.
    \]
\end{itemize}
