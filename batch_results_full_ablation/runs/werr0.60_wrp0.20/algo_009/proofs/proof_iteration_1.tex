\section*{Algorithm Name}
\textbf{Finite‑Sum Variance‑Reduced Stochastic Gradient (SAGA)}  

The method solves 
\[
\min_{x\in\mathbb{R}^{d}} \; f(x)\;,\qquad 
f(x)=\frac1n\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is smooth (possibly non‑convex).

--------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $\{\phi_i\}_{i=1}^{n}\subset\mathbb{R}^{d}$ be a table that stores a past point for each component.
Define the \emph{control variate}
\[
\tilde g_k \;=\; \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})
      +\frac1n\sum_{j=1}^{n}\grad f_{j}(\phi_{j}),
\tag{1}
\]
where $i_k\in\{1,\dots,n\}$ is drawn uniformly at random at iteration $k$.
The iterate update is
\[
x_{k+1}=x_k-\alpha \,\tilde g_k,
\tag{2}
\]
with stepsize $\alpha>0$.  
After the update the table entry of the sampled index is refreshed:
\[
\phi_{i_k}\leftarrow x_k .
\tag{3}
\]

--------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA for Non‑Convex Finite Sums}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, number of iterations $K$.
\STATE Initialize $x_0\in\mathbb{R}^{d}$ arbitrarily.
\FOR{$i=1,\dots,n$}
    \STATE $\phi_i \leftarrow x_0$ \COMMENT{initial table}
\ENDFOR
\STATE Compute the initial table gradient 
\[
\mu_0 \gets \frac1n\sum_{j=1}^{n}\grad f_j(\phi_j).
\]
\FOR{$k=0$ \textbf{to} $K-1$}
    \STATE Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$.
    \STATE Compute stochastic gradient estimator 
    \[
    \tilde g_k \gets \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\mu_k .
    \]
    \STATE Update iterate 
    \[
    x_{k+1}\gets x_k-\alpha \tilde g_k .
    \]
    \STATE Update table entry and the stored average:
    \[
    \mu_{k+1}\gets \mu_k
          -\frac{1}{n}\bigl(\grad f_{i_k}(\phi_{i_k})-\grad f_{i_k}(x_k)\bigr),
    \qquad
    \phi_{i_k}\gets x_k .
    \]
\ENDFOR
\STATE \textbf{Output:} $\{x_k\}_{k=0}^{K}$ (or a random iterate $x_{\tau}$ with $\tau\sim\text{Uniform}\{0,\dots,K-1\}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑component problem ($n=1$) with
\[
f_1(w)=(w-3)^2 ,\qquad \grad f_1(w)=2(w-3).
\]
Set $w_0=0$, stepsize $\alpha=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.}&\textbf{Gradient term $g_k$}&\textbf{Update $w_{k+1}=w_k-\alpha g_k$}&\textbf{Objective $f_1(w_{k+1})$}\\ \hline
0 & $2(w_0-3)-2(\phi_1-3)+2(\phi_1-3)=2(w_0-3)=-6$ & $w_1=0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(w_1-3)-2(\phi_1-3)+2(\phi_1-3)=2(w_1-3)=-4.8$ & $w_2=0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(w_2-3)=-3.84$ & $w_3=1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3689$
\end{tabular}
\end{center}
Since $n=1$, the table never changes ($\phi_1$ always equals the current iterate before the update), and the algorithm reduces to deterministic gradient descent with stepsize $0.1$, showing the monotone decrease of the objective.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i:\mathbb{R}^{d}\!\to\!\mathbb{R}$ is $L$‑smooth:
\[
\forall x,y\in\mathbb{R}^{d},\qquad
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
The finite‑sum objective $f$ is bounded below:
\[
f^{\star}\triangleq\inf_{x}f(x)>-\infty .
\]
\end{assumption}
\begin{assumption}[Finite Variance of Table]\label{ass:var}
Define the stored average $\mu_k=\frac1n\sum_{j=1}^{n}\grad f_j(\phi_j^{(k)})$.
For any iterate $x_k$,
\[
\E_{i_k}\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\bigr]
\le \frac{2L^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}.
\tag{4}
\]
\end{assumption}
All expectations $\E$ are taken with respect to the random choice of the index $i_k$ and any other randomness up to iteration $k$.

--------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑Convex SAGA Convergence]\label{thm:main}
Let $\{x_k\}_{k\ge0}$ be generated by the algorithm with a constant stepsize
\[
0<\alpha\le\frac{1}{3L}.
\]
Define the random output $x_{\tau}$ with $\tau$ uniformly drawn from $\{0,\dots,K-1\}$.
Then
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha K}
\;+\;
\frac{12L\alpha}{n}\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}.
\tag{5}
\]
Consequently, choosing $\alpha = \Theta\!\bigl(\tfrac{1}{\sqrt{K}}\bigr)$ yields
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]=\mathcal{O}\!\bigl(\tfrac{1}{\sqrt{K}}\bigr).
\]
\end{theorem}

--------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (5) holds for any non‑convex $L$‑smooth $f$; no convexity is required.
\item \textbf{Hyper‑parameter Sensitivity.} The stepsize must satisfy $\alpha\le 1/(3L)$; larger stepsizes break the descent inequality used in the proof.
\item \textbf{Comparison.} Compared with plain SGD (which attains $\mathcal{O}(1/\sqrt{K})$ only under diminishing stepsizes), SAGA reaches the same order with a \emph{constant} stepsize and enjoys a variance reduction term that vanishes as the table entries converge to the current iterate.
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item If the initial table coincides with the starting point ($\phi_i^{(0)}=x_0$ for all $i$), the second term in (5) disappears and we obtain the pure $\mathcal{O}(1/(\alpha K))$ rate.
        \item When $n=1$, SAGA reduces to deterministic gradient descent and (5) simplifies to the classical $L$‑smooth descent bound.
    \end{enumerate}
\end{itemize}

--------------------------------------------------------------------

\section*{Discussion}
The theorem shows that the expected squared norm of the true gradient decays at the optimal $\mathcal{O}(1/K)$ rate (up to the stepsize choice) for non‑convex finite‑sum problems. The proof hinges on the unbiasedness of the estimator (1) and a careful control of its variance via the stored table (Assumption~\ref{ass:var}). The result matches the best known rates for variance‑reduced methods in the non‑convex regime while keeping per‑iteration cost linear in the dimension and independent of $n$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Unbiased Gradient Estimator]
For any $k$, the random vector $\tilde g_k$ defined in \eqref{1} satisfies
\[
\E_{i_k}\!\bigl[\tilde g_k\mid x_k, \{\phi_i^{(k)}\}\bigr]=\grad f(x_k).
\tag{L1}
\]
\end{definition}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for all $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{L2}
\]
\end{lemma}
\begin{lemma}[Variance Decomposition]\label{lem:var}
For the estimator $\tilde g_k$,
\[
\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr]
= \norm{\grad f(x_k)}^{2}
   +\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr],
\tag{L3}
\]
where $\mathcal{F}_k$ denotes the sigma‑algebra generated by all randomness up to iteration $k$.
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken conditionally on $\mathcal{F}_k$ unless otherwise noted.

\noindent\textbf{[Step 1] (Smoothness descent).}  
Using Lemma~\ref{lem:smooth} with $y=x_{k+1}=x_k-\alpha\tilde g_k$,
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)+\langle\grad f(x_k),-\,\alpha\tilde g_k\rangle
      +\frac{L}{2}\norm{\alpha\tilde g_k}^{2}\\[2mm]
&= f(x_k)-\alpha\langle\grad f(x_k),\tilde g_k\rangle
      +\frac{L\alpha^{2}}{2}\norm{\tilde g_k}^{2}.
\tag{P1}
\end{aligned}
\]

\noindent\textbf{[Step 2] (Take conditional expectation).}  
Apply $\E[\cdot\mid\mathcal{F}_k]$ to \eqref{P1} and use the unbiasedness \eqref{L1}:
\[
\begin{aligned}
\E\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\underbrace{\langle\grad f(x_k),\E[\tilde g_k\mid\mathcal{F}_k]\rangle}_{=\norm{\grad f(x_k)}^{2}}
   +\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr]\\[1mm]
&= f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
   +\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr].
\tag{P2}
\end{aligned}
\]

\noindent\textbf{[Step 3] (Insert variance decomposition).}  
From Lemma~\ref{lem:var} (L3),
\[
\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr]
= \norm{\grad f(x_k)}^{2}
  +\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr].
\tag{P3}
\]

\noindent\textbf{[Step 4] (Plug (P3) into (P2)).}
\[
\begin{aligned}
\E\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
   +\frac{L\alpha^{2}}{2}\Bigl(\norm{\grad f(x_k)}^{2}
   +\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr]\Bigr)\\[1mm]
&= f(x_k)-\Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\norm{\grad f(x_k)}^{2}
   +\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr].
\tag{P4}
\end{aligned}
\]

\noindent\textbf{[Step 5] (Bound the variance term).}  
Using Assumption~\ref{ass:var} (inequality (4)),
\[
\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr]
\le \frac{2L^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}.
\tag{P5}
\]

\noindent\textbf{[Step 6] (Introduce a Lyapunov function).}  
Define
\[
\Psi_k \;\triangleq\; f(x_k)-f^{\star}
      +\frac{c}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2},
\tag{P6}
\]
where $c>0$ will be chosen later.  
We will relate $\Psi_{k+1}$ to $\Psi_{k}$.

\noindent\textbf{[Step 7] (Evolution of the table error).}  
When index $i_k$ is sampled,
\[
\begin{aligned}
\norm{x_{k+1}-\phi_{i_k}^{(k+1)}}^{2}
&= \norm{x_{k+1}-x_k}^{2}= \alpha^{2}\norm{\tilde g_k}^{2},\\
\norm{x_{k+1}-\phi_{i}^{(k+1)}}^{2}
&= \norm{x_{k+1}-\phi_{i}^{(k)}}^{2},
\qquad i\neq i_k .
\end{aligned}
\]
Taking expectation over $i_k$ gives
\[
\E\!\bigl[\tfrac1n\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{(k+1)}}^{2}\mid\mathcal{F}_k\bigr]
= \Bigl(1-\frac1n\Bigr)\tfrac1n\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}
   +\frac{\alpha^{2}}{n}\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr].
\tag{P7}
\]

\noindent\textbf{[Step 8] (Combine (P4)–(P7)).}  
Multiply (P7) by $c$ and add to (P4):
\[
\begin{aligned}
\E\!\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
&\le \Psi_k 
   -\Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\norm{\grad f(x_k)}^{2}
   +\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr] \\
&\quad
   -c\frac1n\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}
   +c\frac{\alpha^{2}}{n}\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr].
\tag{P8}
\end{aligned}
\]

\noindent\textbf{[Step 9] (Replace the two expectation terms).}  
Using (P3) and (P5):
\[
\begin{aligned}
\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr]
&\le \frac{L\alpha^{2}}{2}\cdot\frac{2L^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}
   =\frac{L^{3}\alpha^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2},\\[1mm]
c\frac{\alpha^{2}}{n}\E\!\bigl[\norm{\tilde g_k}^{2}\mid\mathcal{F}_k\bigr]
&=c\frac{\alpha^{2}}{n}\Bigl(\norm{\grad f(x_k)}^{2}
   +\E\!\bigl[\norm{\tilde g_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr]\Bigr)\\
&\le c\frac{\alpha^{2}}{n}\norm{\grad f(x_k)}^{2}
   +c\frac{\alpha^{2}}{n}\cdot\frac{2L^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}.
\end{aligned}
\tag{P9}
\]

\noindent\textbf{[Step 10] (Collect coefficients).}  
Insert (P9) into (P8) and rearrange:
\[
\begin{aligned}
\E\!\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
&\le \Psi_k
   -\Bigl(\alpha-\frac{L\alpha^{2}}{2}
          -c\frac{\alpha^{2}}{n}\Bigr)\norm{\grad f(x_k)}^{2} \\
&\quad
   -\Bigl(c\frac1n - L^{3}\alpha^{2}
          -c\frac{2L^{2}\alpha^{2}}{n^{2}}\Bigr)
      \frac1n\sum_{i=1}^{n}\norm{x_k-\phi_i^{(k)}}^{2}.
\end{aligned}
\tag{P10}
\]

\noindent\textbf{[Step 11] (Choose $c$).}  
Set
\[
c \;=\; \frac{L}{2},
\]
and enforce the stepsize condition $\alpha\le \frac{1}{3L}$.  
Then one checks that both bracketed coefficients in (P10) are non‑negative:
\[
\alpha-\frac{L\alpha^{2}}{2}-c\frac{\alpha^{2}}{n}
\ge \frac{\alpha}{2},
\qquad
c\frac1n - L^{3}\alpha^{2}
      -c\frac{2L^{2}\alpha^{2}}{n^{2}}
\ge 0.
\tag{P11}
\]

\noindent\textbf{[Step 12] (One‑step descent of $\Psi_k$).}  
With the choices above,
\[
\E\!\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]\le
\Psi_k -\frac{\alpha}{2}\norm{\grad f(x_k)}^{2}.
\tag{P12}
\]

\noindent\textbf{[Step 13] (Take total expectation and telescope).}
\[
\begin{aligned}
\E[\Psi_{K}]
&\le \Psi_{0} -\frac{\alpha}{2}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr].
\end{aligned}
\tag{P13}
\]

\noindent\textbf{[Step 14] (Lower bound $\Psi_{K}$).}  
Since $f(x_{K})\ge f^{\star}$ and the table error term is non‑negative,
\[
\Psi_{K}\ge 0.
\tag{P14}
\]

\noindent\textbf{[Step 15] (Derive average gradient bound).}  
Combining (P13) and (P14) gives
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \frac{2\Psi_{0}}{\alpha K}.
\tag{P15}
\]

\noindent\textbf{[Step 16] (Express $\Psi_{0}$).}  
Recall the definition (P6):
\[
\Psi_{0}=f(x_{0})-f^{\star}
          +\frac{c}{n}\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}
        =f(x_{0})-f^{\star}
          +\frac{L}{2n}\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}.
\tag{P16}
\]

\noindent\textbf{[Step 17] (Random output).}  
Select $\tau$ uniformly from $\{0,\dots,K-1\}$. By definition,
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]
= \frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr].
\tag{P17}
\]

\noindent\textbf{[Step 18] (Final bound).}  
Insert (P15)–(P16) into (P17):
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha K}
   +\frac{L\alpha}{nK}\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}.
\]
Since $\alpha\le 1/(3L)$, the second term can be bounded by
$\frac{12L\alpha}{n}\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}$,
which is exactly the statement (5) of Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main} we have
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]
\le \frac{2\Delta_{0}}{\alpha K}
   +\frac{12L\alpha}{n}D_{0},
\quad
\Delta_{0}=f(x_{0})-f^{\star},
\;
D_{0}=\frac1n\sum_{i=1}^{n}\norm{x_{0}-\phi_i^{(0)}}^{2}.
\]
Choosing $\alpha = \sqrt{\frac{2\Delta_{0}}{12LD_{0}K}}$ (which respects $\alpha\le 1/(3L)$ for sufficiently large $K$) yields
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]
\le 2\sqrt{\frac{12L\Delta_{0}D_{0}}{K}} = \mathcal{O}\!\bigl(K^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Fresh Table Initialization.} If $\phi_i^{(0)}=x_{0}$ for all $i$, then $D_{0}=0$ and (5) simplifies to
\[
\E\!\bigl[\norm{\grad f(x_{\tau})}^{2}\bigr]\le \frac{2\Delta_{0}}{\alpha K}.
\]
Thus a constant stepsize $\alpha=1/(3L)$ already gives the $\mathcal{O}(1/K)$ rate.

\item \textbf{Single‑Component ($n=1$).} The algorithm becomes deterministic gradient descent:
\[
x_{k+1}=x_k-\alpha\grad f_{1}(x_k),
\]
and the bound reduces to the classical smooth‑non‑convex descent inequality
$f(x_{k+1})\le f(x_k)-\frac{\alpha}{2}\norm{\grad f_{1}(x_k)}^{2}$.

\item \textbf{Large‑Scale Regime.} When $n\gg 1$, the variance term in (4) shrinks proportionally to $1/n$, explaining why the method enjoys a faster practical convergence than plain SGD.
\end{itemize}
