\section*{Frank–Wolfe (Conditional Gradient) Method}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be a convex and $L$-smooth function and let 
$C\subset\mathbb{R}^d$ be a non‑empty compact convex set (the feasible domain).  
At iteration $k\ge 0$ the algorithm performs

\[
\begin{aligned}
s_k &:= \arg\min_{s\in C}\;\langle \nabla f(x_k),\,s\rangle , \tag{1}\\[4pt]
\gamma_k &\in[0,1] \quad\text{(step size, see below)},\\[4pt]
x_{k+1} &:= (1-\gamma_k)x_k + \gamma_k s_k . \tag{2}
\end{aligned}
\]

Typical step‑size choices are  

\[
\gamma_k = \frac{2}{k+2}\qquad\text{(standard FW schedule)},
\]
or the exact line‑search
\[
\gamma_k = \arg\min_{\gamma\in[0,1]} f\bigl((1-\gamma)x_k+\gamma s_k\bigr).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe}
\begin{algorithmic}[1]
\STATE \textbf{Input:} convex $f$, feasible set $C$, initial point $x_0\in C$, 
maximum iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE $g_k \gets \nabla f(x_k)$
    \STATE $s_k \gets \displaystyle\arg\min_{s\in C}\langle g_k,s\rangle$ \hfill\COMMENT{Linear minimization oracle}
    \STATE Choose step size $\gamma_k\in[0,1]$ (e.g. $\gamma_k=2/(k+2)$)
    \STATE $x_{k+1} \gets (1-\gamma_k)x_k + \gamma_k s_k$
\ENDFOR
\STATE \textbf{Output:} $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional problem  

\[
f(w) = (w-3)^2,\qquad C=[0,5],\qquad x_0 = 0,
\]
and fix a constant step size $\gamma_k=\gamma=0.1$ for illustration.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $k=0$  
\[
\nabla f(x_0)=2(x_0-3) = -6.
\]
The linear oracle solves $\min_{s\in[0,5]} (-6)s$, whose minimizer is $s_0=5$ (because the coefficient is negative).  
Update:
\[
x_1 = (1-0.1)0 + 0.1\cdot5 = 0.5.
\]
Objective:
\[
f(x_1) = (0.5-3)^2 = 6.25.
\]

\item $k=1$  
\[
\nabla f(x_1)=2(0.5-3) = -5.
\]
Oracle: $\min_{s\in[0,5]} (-5)s$ $\Rightarrow$ $s_1=5$.  
Update:
\[
x_2 = 0.9\cdot0.5 + 0.1\cdot5 = 0.95.
\]
Objective:
\[
f(x_2) = (0.95-3)^2 \approx 4.2025.
\]

\item $k=2$  
\[
\nabla f(x_2)=2(0.95-3) = -4.1.
\]
Oracle: again $s_2=5$.  
Update:
\[
x_3 = 0.9\cdot0.95 + 0.1\cdot5 = 1.355.
\]
Objective:
\[
f(x_3) = (1.355-3)^2 \approx 2.708.
\]
\end{enumerate}
The iterates move monotonically toward the optimum $w^\star=3$ while staying inside $C$.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Convexity.} $f$ is convex on $\mathbb{R}^d$: 
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle,\quad\forall x,y.
\]
\item \textbf{Smoothness.} $f$ has $L$‑Lipschitz gradient:
\[
\|\nabla f(y)-\nabla f(x)\| \le L\|y-x\|,\quad\forall x,y.
\]
Equivalently,
\[
f(y) \le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 . \tag{A2'}
\]
\item \textbf{Compact feasible set.} $C$ is convex, compact and has finite diameter
\[
D:=\max_{x,s\in C}\|x-s\|<\infty .
\]
\item \textbf{Linear minimization oracle.} For any $g\in\mathbb{R}^d$ the subproblem 
$\arg\min_{s\in C}\langle g,s\rangle$ can be solved exactly (producing $s_k$).
\end{enumerate}

Define the \emph{curvature constant} of $f$ on $C$ (cf. Jaggi, 2013) as
\[
C_f := \sup_{\substack{x,s\in C\\ \gamma\in[0,1]}}
\frac{2}{\gamma^{2}}
\Bigl(f\bigl(x+\gamma(s-x)\bigr)-f(x)-\gamma\langle\nabla f(x),s-x\rangle\Bigr).
\tag{3}
\]
Because of $L$‑smoothness and bounded diameter $D$, one has $C_f\le L D^{2}$.

\section*{Main Convergence Theorem}
\begin{theorem}[Primal gap $O(1/k)$]\label{thm:fw}
Let the assumptions (A1)–(A4) hold and let the step size be $\gamma_k = \frac{2}{k+2}$.
Denote $x^\star:=\arg\min_{x\in C}f(x)$. Then for all $k\ge 0$
\[
f(x_k)-f(x^\star) \le \frac{2C_f}{k+2}. \tag{4}
\]
Consequently $f(x_k)\to f(x^\star)$ at a rate $\mathcal{O}(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The iterates remain in $C$ for any $\gamma_k\in[0,1]$ because $C$ is convex.
\item \textbf{Step‑size sensitivity.} The standard schedule $\gamma_k=2/(k+2)$ yields the optimal $O(1/k)$ bound without line‑search. Exact line‑search can improve the constant but not the asymptotic order.
\item \textbf{Comparison.} Compared with projected gradient descent (PGD) which needs a projection onto $C$, FW replaces the projection by a cheap linear oracle and attains the same $O(1/k)$ rate under identical smoothness/convexity assumptions.
\item \textbf{Special cases.} If $f$ is additionally $\mu$‑strongly convex, the primal gap contracts linearly:
\[
f(x_k)-f(x^\star) \le \left(1-\frac{\mu}{L}\right)^{k}\bigl(f(x_0)-f(x^\star)\bigr).
\]
\end{itemize}

\section*{Discussion}
The theorem shows that the Frank–Wolfe method enjoys a sublinear convergence rate that depends only on the curvature constant $C_f$, which in turn is bounded by $L D^2$. The proof relies on a one‑step progress inequality derived from smoothness and the definition of $C_f$. The method is particularly attractive when the linear minimization oracle is cheap (e.g., over a simplex or trace‑norm ball) while a projection would be expensive.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function $f$ and any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{5}
\]
\end{lemma}
\begin{proof}
Standard from the definition of $L$‑Lipschitz gradient (integrate the gradient bound). $\square$
\end{proof}

\begin{lemma}[Curvature bound]\label{lem:curv}
For any $x\in C$, $s\in C$ and $\gamma\in[0,1]$,
\[
f\bigl(x+\gamma(s-x)\bigr) \le f(x)+\gamma\langle\nabla f(x),s-x\rangle+\frac{\gamma^{2}}{2}C_f . \tag{6}
\]
\end{lemma}
\begin{proof}
By definition (3) of $C_f$, rearranging yields (6). $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw}. Let $x^\star\in\arg\min_{x\in C}f(x)$ and denote the \emph{Frank–Wolfe gap}
\[
g_k := \langle \nabla f(x_k),\,x_k-s_k\rangle .
\tag{7}
\]

\noindent\textbf{[Step 1]}~\emph{One‑step progress using curvature.}  
Apply Lemma~\ref{lem:curv} with $x=x_k$, $s=s_k$, and $\gamma=\gamma_k$:
\[
f(x_{k+1}) 
\le f(x_k)+\gamma_k\langle\nabla f(x_k),s_k-x_k\rangle+\frac{\gamma_k^{2}}{2}C_f .
\tag{8}
\]

\noindent\textbf{[Step 2]}~\emph{Express the inner product via the gap.}  
Since $g_k = \langle\nabla f(x_k),x_k-s_k\rangle = -\langle\nabla f(x_k),s_k-x_k\rangle$, (8) becomes
\[
f(x_{k+1}) \le f(x_k)-\gamma_k g_k+\frac{\gamma_k^{2}}{2}C_f .
\tag{9}
\]

\noindent\textbf{[Step 3]}~\emph{Relate $g_k$ to the primal gap.}  
Because $s_k$ solves the linear oracle,
\[
\langle\nabla f(x_k),s_k\rangle \le \langle\nabla f(x_k),x^\star\rangle .
\]
Hence
\[
g_k = \langle\nabla f(x_k),x_k\rangle-\langle\nabla f(x_k),s_k\rangle
\ge \langle\nabla f(x_k),x_k-x^\star\rangle .
\tag{10}
\]
By convexity (A1),
\[
f(x_k)-f(x^\star) \le \langle\nabla f(x_k),x_k-x^\star\rangle .
\tag{11}
\]
Combining (10) and (11) yields
\[
g_k \ge f(x_k)-f(x^\star). \tag{12}
\]

\noindent\textbf{[Step 4]}~\emph{Insert (12) into (9).}
\[
f(x_{k+1})-f(x^\star) \le \bigl(1-\gamma_k\bigr)\bigl[f(x_k)-f(x^\star)\bigr]+\frac{\gamma_k^{2}}{2}C_f .
\tag{13}
\]

\noindent\textbf{[Step 5]}~\emph{Unroll the recursion.}  
Define $\Delta_k:=f(x_k)-f(x^\star)$. Then (13) reads
\[
\Delta_{k+1} \le (1-\gamma_k)\Delta_k+\frac{\gamma_k^{2}}{2}C_f .
\tag{14}
\]

\noindent\textbf{[Step 6]}~\emph{Choose the standard step size $\gamma_k=2/(k+2)$.}  
Observe that $1-\gamma_k = \frac{k}{k+2}$ and $\gamma_k^{2}= \frac{4}{(k+2)^{2}}$. Substituting into (14):
\[
\Delta_{k+1} \le \frac{k}{k+2}\Delta_k+\frac{2}{(k+2)^{2}}C_f .
\tag{15}
\]

\noindent\textbf{[Step 7]}~\emph{Induction to bound $\Delta_k$.}  
We claim that for all $k\ge 0$,
\[
\Delta_k \le \frac{2C_f}{k+2}. \tag{16}
\]
\emph{Base case $k=0$.}  
From definition $\Delta_0 = f(x_0)-f(x^\star)\le C_f$ (since curvature upper‑bounds any single step, see (6) with $\gamma=1$). Hence
\[
\Delta_0 \le C_f \le \frac{2C_f}{2},
\]
so (16) holds for $k=0$.

\emph{Inductive step.} Assume (16) holds for some $k\ge0$. Plug the inductive bound into (15):
\[
\Delta_{k+1}
\le \frac{k}{k+2}\cdot\frac{2C_f}{k+2}+\frac{2C_f}{(k+2)^{2}}
= \frac{2C_f}{(k+2)^{2}}\bigl(k+1\bigr)
= \frac{2C_f}{k+3}.
\tag{17}
\]
Thus (16) holds for $k+1$. By induction, (16) is true for all $k$.

\noindent\textbf{[Step 8]}~\emph{Conclusion.}  
Inequality (16) is precisely the bound (4) claimed in Theorem~\ref{thm:fw}. $\square$

\section*{Rate Derivation (With Constants)}
The curvature constant satisfies $C_f\le LD^{2}$, where $D$ is the diameter of $C$. Hence a fully explicit bound is
\[
f(x_k)-f(x^\star) \le \frac{2L D^{2}}{k+2}. \tag{18}
\]
If the exact line‑search is employed, the factor $2$ can be replaced by $1$, yielding a tighter constant.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, the Frank–Wolfe gap satisfies $g_k\ge \frac{\mu}{2}\|x_k-x^\star\|^{2}$. Combining with (9) and using $\gamma_k=2/(k+2)$ gives a linear convergence rate (see Lacoste‑Julien \& Jaggi, 2015).
\item \textbf{Polytope constraints.} When $C$ is a polytope, each $s_k$ is a vertex, and the algorithm can be interpreted as moving towards extreme points; the bound (4) remains valid.
\item \textbf{Away‑step / pairwise variants.} Modifications that occasionally move away from previously selected vertices improve the rate to $O(1/k^{2})$ under a \emph{pyramidal width} condition.
\end{enumerate}
