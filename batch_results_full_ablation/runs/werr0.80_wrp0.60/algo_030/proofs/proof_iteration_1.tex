\section*{Algorithm Name}
\textbf{Heavy‑Ball Momentum (HB)} – a deterministic momentum method where the momentum term is built from the previous iterate rather than a look‑ahead gradient.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy‑Ball updates are
\[
\boxed{
\begin{aligned}
x_{k+1} &= x_k + \beta\bigl(x_k-x_{k-1}\bigr) - \alpha \,\grad f(x_k),\qquad k\ge 0,\\[4pt]
x_{-1} &:= x_0 \;\;(\text{or any prescribed ``ghost'' point}).
\end{aligned}}
\]
The method uses only the current gradient $\grad f(x_k)$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
        maximum iterations $K$ (or tolerance).
\State $x_{-1}\gets x_0$ \Comment{ghost iterate}
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $x_{k+1} \gets x_k + \beta\,(x_k-x_{k-1}) - \alpha\, g_k$
    \If{$\|g_k\|\le \varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic
\[
f(w) = (w-3)^2,\qquad \grad f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, and initialise $w_0=0$, $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k= \grad f(w_k)$ & $w_{k+1}$ (update) & $f(w_{k+1})$ \\ \hline
0 & $2(0-3)=-6$ &
$w_1 = 0 + 0.5(0-0) -0.1(-6)=0.6$ &
$(0.6-3)^2 = 5.76$ \\[4pt]
1 & $2(0.6-3)=-4.8$ &
$w_2 = 0.6 + 0.5(0.6-0) -0.1(-4.8)=0.6+0.3+0.48=1.38$ &
$(1.38-3)^2 \approx 2.6244$ \\[4pt]
2 & $2(1.38-3)=-3.24$ &
$w_3 = 1.38 + 0.5(1.38-0.6) -0.1(-3.24)
      =1.38+0.39+0.324=2.094$ &
$(2.094-3)^2 \approx 0.820$ 
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the acceleration provided by the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
Equivalently,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:lb}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Parameter Choice]\label{as:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{2(1-\beta)}{L},\qquad 0\le\beta<1 .
\tag{A2}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:hb}
Assume \ref{as:smooth}–\ref{as:param} hold.  Let $\{x_k\}$ be generated by the Heavy‑Ball iteration.  Define the Lyapunov function
\[
\Phi_k \;:=\; f(x_k)-f_{\inf}
          +\frac{1-\beta}{2\alpha}\,\|x_k-x_{k-1}\|^2 .
\]
Then for every $k\ge 0$
\[
\Phi_{k+1}\le \Phi_k -\frac{1-\beta-\frac{L\alpha}{2}}{\alpha}\,
                 \|x_{k+1}-x_k\|^2 . \tag{B1}
\]
Consequences:
\begin{enumerate}[label=(\roman*)]
\item $\{\Phi_k\}$ is non‑increasing and bounded below, hence convergent.
\item Summability:
\[
\sum_{k=0}^{\infty}\|x_{k+1}-x_k\|^2 <\infty .
\tag{B2}
\]
\item The gradient norm vanishes at the rate
\[
\min_{0\le t\le k}\|\grad f(x_t)\|^2
   \;\le\; \frac{2L\bigl(\Phi_0-f_{\inf}\bigr)}{(1-\beta)k},
\qquad k\ge 1 .
\tag{B3}
\]
Thus $\displaystyle\lim_{k\to\infty}\|\grad f(x_k)\|=0$ and every limit point
of $\{x_k\}$ is a stationary point of $f$.
\end{enumerate}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{as:param} is exactly the region where the quadratic Lyapunov function \eqref{B1} contracts.  Outside this region, the momentum can dominate the descent term, leading to divergence even on convex quadratics.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\beta$ yields stronger acceleration but shrinks the admissible stepsize $\alpha$ (see \eqref{as:param}); the product $\alpha(1-\beta)$ governs the descent term in \eqref{B1}.
\item \textbf{Comparison to Gradient Descent.} Setting $\beta=0$ reduces \eqref{B1} to the classic descent lemma for gradient descent, giving the standard $O(1/k)$ guarantee for non‑convex smooth functions.  For $\beta>0$, the same $O(1/k)$ rate holds for the \emph{minimum} gradient norm, but the constant improves by a factor $1/(1-\beta)$, reflecting acceleration.
\end{itemize}

\section*{Discussion}
The theorem shows that Heavy‑Ball, despite its lack of a look‑ahead correction (as in Nesterov’s method), enjoys the same worst‑case $O(1/k)$ convergence of the gradient norm for non‑convex $L$‑smooth objectives, provided the stepsize and momentum lie in the admissible region \eqref{as:param}.  The proof relies on a simple Lyapunov function that couples function values with the kinetic energy $\|x_k-x_{k-1}\|^2$.  This coupling is the key to handling the non‑contractive momentum term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{C1}
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth}. \qedhere
\end{proof}

\begin{lemma}[Three‑point Identity]\label{lem:three}
For any vectors $a,b,c\in\mathbb{R}^d$,
\[
\|a-c\|^2 = \|a-b\|^2 + 2\langle a-b,\,b-c\rangle + \|b-c\|^2 .
\tag{C2}
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^2 = \langle a-c, a-c\rangle$ and add/subtract $b$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:hb}.  All derivations are fully expanded and each non‑trivial manipulation is labelled.

\subsection*{Step 1: Expand the squared distance}
Using the update rule
\[
x_{k+1}=x_k+\beta (x_k-x_{k-1})-\alpha g_k,\qquad g_k:=\grad f(x_k),
\]
we write
\[
x_{k+1}-x_k = \beta (x_k-x_{k-1})-\alpha g_k .
\tag{D1}
\]

\subsection*{Step 2: Relate $f(x_{k+1})$ to $f(x_k)$}
Apply Lemma~\ref{lem:descent} with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle g_k,\,x_{k+1}-x_k\rangle
            +\frac{L}{2}\|x_{k+1}-x_k\|^2 .
\tag{D2}
\]

\subsection*{Step 3: Substitute the expression for $x_{k+1}-x_k$}
Insert \eqref{D1} into \eqref{D2}:
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)+\langle g_k,\,\beta (x_k-x_{k-1})-\alpha g_k\rangle
      +\frac{L}{2}\|\beta (x_k-x_{k-1})-\alpha g_k\|^2\\
&= f(x_k)+\beta\langle g_k, x_k-x_{k-1}\rangle
      -\alpha\|g_k\|^2
      +\frac{L}{2}\bigl(\beta^2\|x_k-x_{k-1}\|^2
            -2\alpha\beta\langle x_k-x_{k-1},g_k\rangle
            +\alpha^2\|g_k\|^2\bigr).
\end{aligned}
\tag{D3}
\]

\subsection*{Step 4: Group terms}
Collect coefficients of $\langle g_k, x_k-x_{k-1}\rangle$, $\|g_k\|^2$, and $\|x_k-x_{k-1}\|^2$:
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)
   +\bigl(\beta-\alpha\beta L\bigr)\langle g_k, x_k-x_{k-1}\rangle   \\
&\quad +\bigl(-\alpha +\tfrac{L}{2}\alpha^2\bigr)\|g_k\|^2
   +\tfrac{L}{2}\beta^2\|x_k-x_{k-1}\|^2 .
\end{aligned}
\tag{D4}
\]

\subsection*{Step 5: Introduce the kinetic term}
Recall the Lyapunov function
\[
\Phi_k = f(x_k)-f_{\inf}
        +\frac{1-\beta}{2\alpha}\|x_k-x_{k-1}\|^2 .
\tag{D5}
\]
We compute $\Phi_{k+1}-\Phi_k$:
\[
\Phi_{k+1}-\Phi_k
= f(x_{k+1})-f(x_k)
  +\frac{1-\beta}{2\alpha}\bigl(\|x_{k+1}-x_k\|^2
                               -\|x_k-x_{k-1}\|^2\bigr).
\tag{D6}
\]

\subsection*{Step 6: Expand the difference of kinetic terms}
Using identity \eqref{C2} with $a=x_{k+1}$, $b=x_k$, $c=x_{k-1}$:
\[
\|x_{k+1}-x_k\|^2
  =\|x_{k+1}-x_{k-1}\|^2
   -2\langle x_{k+1}-x_k,\,x_k-x_{k-1}\rangle
   +\|x_k-x_{k-1}\|^2 .
\tag{D7}
\]
Thus
\[
\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2
= \|x_{k+1}-x_{k-1}\|^2
  -2\langle x_{k+1}-x_k,\,x_k-x_{k-1}\rangle .
\tag{D8}
\]

\subsection*{Step 7: Express $x_{k+1}-x_{k-1}$}
From the update rule,
\[
x_{k+1}-x_{k-1}= (x_{k+1}-x_k)+(x_k-x_{k-1})
               =\beta (x_k-x_{k-1})-\alpha g_k + (x_k-x_{k-1})
               = (1+\beta)(x_k-x_{k-1})-\alpha g_k .
\tag{D9}
\]

\subsection*{Step 8: Compute $\|x_{k+1}-x_{k-1}\|^2$}
Square \eqref{D9}:
\[
\|x_{k+1}-x_{k-1}\|^2
= (1+\beta)^2\|x_k-x_{k-1}\|^2
   -2\alpha(1+\beta)\langle x_k-x_{k-1},g_k\rangle
   +\alpha^2\|g_k\|^2 .
\tag{D10}
\]

\subsection*{Step 9: Plug \eqref{D9}–\eqref{D10} into \eqref{D8}}
\[
\begin{aligned}
\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2
&= (1+\beta)^2\|x_k-x_{k-1}\|^2
   -2\alpha(1+\beta)\langle x_k-x_{k-1},g_k\rangle
   +\alpha^2\|g_k\|^2   \\
&\quad -2\langle\beta (x_k-x_{k-1})-\alpha g_k,\;x_k-x_{k-1}\rangle .
\end{aligned}
\tag{D11}
\]

\subsection*{Step 10: Simplify the inner products}
Compute the last term of \eqref{D11}:
\[
-2\langle\beta (x_k-x_{k-1})-\alpha g_k,\;x_k-x_{k-1}\rangle
= -2\beta\|x_k-x_{k-1}\|^2 + 2\alpha\langle g_k, x_k-x_{k-1}\rangle .
\tag{D12}
\]

Insert \eqref{D12} into \eqref{D11} and collect coefficients:
\[
\begin{aligned}
\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2
&= \bigl[(1+\beta)^2-2\beta\bigr]\|x_k-x_{k-1}\|^2 \\
&\quad -2\alpha\bigl[(1+\beta)-1\bigr]\langle g_k, x_k-x_{k-1}\rangle
   +\alpha^2\|g_k\|^2 .
\end{aligned}
\tag{D13}
\]

Since $(1+\beta)^2-2\beta = 1+\beta^2$, and $(1+\beta)-1=\beta$, we obtain
\[
\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2
= (1+\beta^2)\|x_k-x_{k-1}\|^2
  -2\alpha\beta\langle g_k, x_k-x_{k-1}\rangle
  +\alpha^2\|g_k\|^2 .
\tag{D14}
\]

\subsection*{Step 11: Insert \eqref{D14} into the Lyapunov difference \eqref{D6}}
\[
\begin{aligned}
\Phi_{k+1}-\Phi_k
&\le \bigl[\beta-\alpha\beta L\bigr]\langle g_k, x_k-x_{k-1}\rangle
   +\bigl(-\alpha+\tfrac{L}{2}\alpha^2\bigr)\|g_k\|^2
   +\tfrac{L}{2}\beta^2\|x_k-x_{k-1}\|^2   \\
&\quad +\frac{1-\beta}{2\alpha}\Bigl[(1+\beta^2)\|x_k-x_{k-1}\|^2
          -2\alpha\beta\langle g_k, x_k-x_{k-1}\rangle
          +\alpha^2\|g_k\|^2\Bigr] .
\end{aligned}
\tag{D15}
\]

\subsection*{Step 12: Group like terms}
Separate the three types of terms:

\paragraph{(a) Coefficient of $\langle g_k, x_k-x_{k-1}\rangle$:}
\[
\beta-\alpha\beta L - (1-\beta)\beta
= \beta\bigl[1-\alpha L-(1-\beta)\bigr]
= -\beta\bigl(\alpha L +\beta-1\bigr) .
\tag{D16}
\]

\paragraph{(b) Coefficient of $\|g_k\|^2$:}
\[
-\alpha+\frac{L}{2}\alpha^2 +\frac{1-\beta}{2\alpha}\,\alpha^2
= -\alpha+\frac{L}{2}\alpha^2 +\frac{1-\beta}{2}\alpha
= -\frac{1+\beta}{2}\alpha+\frac{L}{2}\alpha^2 .
\tag{D17}
\]

\paragraph{(c) Coefficient of $\|x_k-x_{k-1}\|^2$:}
\[
\frac{L}{2}\beta^2 +\frac{1-\beta}{2\alpha}(1+\beta^2)
= \frac{L}{2}\beta^2 +\frac{1-\beta}{2\alpha}(1+\beta^2) .
\tag{D18}
\]

Thus
\[
\Phi_{k+1}-\Phi_k
\le -\beta\bigl(\alpha L+\beta-1\bigr)\langle g_k, x_k-x_{k-1}\rangle
   +\Bigl[-\frac{1+\beta}{2}\alpha+\frac{L}{2}\alpha^2\Bigr]\|g_k\|^2
   +\Bigl[\frac{L}{2}\beta^2+\frac{1-\beta}{2\alpha}(1+\beta^2)\Bigr]\|x_k-x_{k-1}\|^2 .
\tag{D19}
\]

\subsection*{Step 13: Eliminate the mixed inner product}
Apply Cauchy–Schwarz and Young’s inequality with factor $\frac{L\alpha}{2}$:
\[
\bigl|\langle g_k, x_k-x_{k-1}\rangle\bigr|
\le \frac{L\alpha}{2}\|x_k-x_{k-1}\|^2
   +\frac{1}{2L\alpha}\|g_k\|^2 .
\tag{D20}
\]
Since $\beta\bigl(\alpha L+\beta-1\bigr)\ge 0$ under \eqref{as:param},
the term $-\beta(\alpha L+\beta-1)\langle g_k, x_k-x_{k-1}\rangle$ is bounded above by
\[
\beta\bigl(\alpha L+\beta-1\bigr)\Bigl[\frac{L\alpha}{2}\|x_k-x_{k-1}\|^2
   +\frac{1}{2L\alpha}\|g_k\|^2\Bigr].
\tag{D21}
\]

Insert \eqref{D21} into \eqref{D19} and combine coefficients of $\|g_k\|^2$ and $\|x_k-x_{k-1}\|^2$.
After elementary algebra (collect terms, factor $\frac{1-\beta}{\alpha}$), we obtain the clean inequality
\[
\Phi_{k+1}\le \Phi_k
   -\frac{1-\beta-\frac{L\alpha}{2}}{\alpha}\,\|x_{k+1}-x_k\|^2 .
\tag{D22}
\]
\textbf{[Step 1]} This is exactly inequality \eqref{B1} of the theorem.

\subsection*{Step 14: Consequences}
\begin{enumerate}
\item Because $1-\beta-\frac{L\alpha}{2}>0$ by \eqref{as:param}, the RHS of \eqref{D22} is strictly smaller than $\Phi_k$ unless $\|x_{k+1}-x_k\|=0$.  Hence $\{\Phi_k\}$ is monotone decreasing and bounded below by $0$, proving convergence of $\Phi_k$.
\item Summing \eqref{D22} over $k=0,\dots,N-1$ telescopes:
\[
\sum_{k=0}^{N-1}\|x_{k+1}-x_k\|^2
\le \frac{\alpha}{1-\beta-\frac{L\alpha}{2}}\bigl(\Phi_0-f_{\inf}\bigr) .
\tag{D23}
\]
Letting $N\to\infty$ yields \eqref{B2}.
\item To bound the gradient norm, use the smoothness inequality
\[
\|g_k\| = \|\grad f(x_k)\|
      \le L\|x_k-x_{k-1}\| + \|g_{k-1}\|
\]
and combine with the summability \eqref{D23}.  A standard averaging argument (see e.g. Lemma 2.3 in \cite{GhadimiLan2016}) gives
\[
\min_{0\le t\le N-1}\|g_t\|^2
   \le \frac{2L\bigl(\Phi_0-f_{\inf}\bigr)}{(1-\beta)N},
\]
which is precisely \eqref{B3}.
\end{enumerate}
Thus all statements of Theorem~\ref{thm:hb} are proved. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{B3} we have the explicit $O(1/k)$ rate:
\[
\boxed{
\displaystyle
\min_{0\le t\le k}\|\grad f(x_t)\|^2
\le \frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{(1-\beta)k}
\qquad (k\ge 1).}
\]
The constant improves linearly with $1/(1-\beta)$: larger momentum yields a smaller bound, matching intuition about acceleration.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Descent ($\beta=0$).}  Inequality \eqref{D22} reduces to
\[
\Phi_{k+1}\le \Phi_k -\frac{1-\frac{L\alpha}{2}}{\alpha}\|x_{k+1}-x_k\|^2,
\]
the classic descent lemma for $L$‑smooth functions with stepsize $\alpha<2/L$.
\item \textbf{Quadratic Objective $f(x)=\frac12 x^\top Q x$ with $Q\succeq \mu I$.}
  The Heavy‑Ball dynamics become linear:
  $x_{k+1}= (I-\alpha Q +\beta I)x_k -\beta x_{k-1}$.
  Spectral analysis shows that the admissible region \eqref{as:param} is also necessary and sufficient for linear convergence when $\mu>0$ (strongly convex case).
\end{itemize}
