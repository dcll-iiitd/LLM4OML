\section*{Algorithm Name}
\textbf{Mirror Descent with Bregman Proximal Step (MD‑B)}  

The algorithm iteratively solves a proximal subproblem defined by a Bregman divergence generated from a strongly convex prox‑function $h$.

\section*{Mathematical Formulation}
Let $f:\R^n\rightarrow\R$ be a convex objective and let $h:\R^n\rightarrow\R$ be a differentiable $1$‑strongly convex function w.r.t. a norm $\|\cdot\|$.  
The Bregman divergence induced by $h$ is
\[
\Breg{x}{y}=h(x)-h(y)-\inner{\grad h(y)}{x-y}.
\]

Given a stepsize $\eta_k>0$, the MD‑B update is
\begin{equation}
\label{eq:update}
x_{k+1}= \argmin_{x\in\R^n}\Big\{ \inner{\grad f(x_k)}{x}+ \frac{1}{\eta_k}\,\Breg{x}{x_k}\Big\}.
\end{equation}
When $h(x)=\tfrac12\|x\|_2^{2}$, $\Breg{x}{y}=\tfrac12\|x-y\|_2^{2}$ and \eqref{eq:update} reduces to the classical gradient step.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent with Bregman Proximal Step}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^n$, stepsize sequence $\{\eta_k\}_{k\ge0}>0$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\grad f(x_k)$.
    \STATE Solve the proximal subproblem
    \[
    x_{k+1}= \argmin_{x}\Big\{ \inner{g_k}{x}+ \frac{1}{\eta_k}\big(h(x)-h(x_k)-\inner{\grad h(x_k)}{x-x_k}\big)\Big\}.
    \]
\ENDFOR
\STATE \textbf{Output:} last iterate $x_T$ (or any averaged iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^2,\qquad h(w)=\tfrac12 w^{2},
\]
so that $\Breg{w}{w_k}= \tfrac12 (w-w_k)^2$.
Take $w_0=0$, constant stepsize $\eta_k=\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\grad f(w_k)$ & $w_{k+1}$ (solution of \eqref{eq:update}) & $f(w_{k+1})$\\ \hline
0 & $g_0=2(w_0-3)=-6$ &
\begin{aligned}
w_{1}&=\argmin_{w}\big\{-6w+\tfrac1{0.1}\tfrac12 (w-0)^2\big\}\\
&=\argmin_{w}\big\{-6w+5 w^{2}\big\}=0.6
\end{aligned}
& $(0.6-3)^2=5.76$\\ \hline
1 & $g_1=2(0.6-3)=-4.8$ &
\begin{aligned}
w_{2}&=\argmin_{w}\big\{-4.8 w+5 (w-0.6)^2\big\}\\
&=0.6+\frac{0.48}{10}=0.648
\end{aligned}
& $(0.648-3)^2\approx5.497$\\ \hline
2 & $g_2=2(0.648-3)=-4.704$ &
\begin{aligned}
w_{3}&=\argmin_{w}\big\{-4.704 w+5 (w-0.648)^2\big\}\\
&=0.648+\frac{0.4704}{10}=0.69504
\end{aligned}
& $(0.69504-3)^2\approx5.308$
\end{tabular}
\end{center}

The iterates move toward the minimiser $w^{\star}=3$, and the objective values decrease.

\section*{Assumptions}
\begin{assumption}[Convexity of $f$]
\label{ass:convex}
$f$ is convex on $\R^n$, i.e.
\[
f(y)\ge f(x)+\inner{\grad f(x)}{y-x},\qquad\forall x,y\in\R^n.
\]
\end{assumption}

\begin{assumption}[Relative Smoothness]
\label{ass:smooth}
There exists $L>0$ such that for all $x,y\in\R^n$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+L\,\Breg{y}{x}.
\]
\end{assumption}

\begin{assumption}[Strong Convexity of $h$]
\label{ass:hsc}
$h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$, i.e.
\[
\Breg{x}{y}\ge \tfrac12\|x-y\|^{2},\qquad\forall x,y\in\R^n.
\]
\end{assumption}

\begin{assumption}[Stepsize]
\label{ass:stepsize}
The stepsizes satisfy $0<\eta_k\le \frac{1}{L}$ for all $k$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) Function‑value Convergence]
\label{thm:convergence}
Let $x^{\star}\in\arg\min f$ and assume \ref{ass:convex}–\ref{ass:stepsize}.  
Define the Bregman distance to the optimum $V:=\Breg{x^{\star}}{x_0}$.  
If the stepsize is constant $\eta_k\equiv\eta\le 1/L$, then for every $K\ge1$,
\[
f(x_K)-f(x^{\star})\le \frac{V}{\eta K}.
\]
If $\eta_k=\frac{2}{L(k+1)}$, the bound improves to $f(x_K)-f(x^{\star})\le\frac{2LV}{K(K+1)}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} Because $h$ is strongly convex, the Bregman term enforces a ``proximal'' pull toward the previous iterate, guaranteeing boundedness of the sequence $\{x_k\}$.
\item \textbf{Hyperparameter Sensitivity:} The convergence rate depends linearly on $1/\eta$. Choosing $\eta$ larger than $1/L$ violates the relative smoothness inequality and may cause divergence.
\item \textbf{Comparison to Gradient Descent:} With $h(x)=\tfrac12\|x\|^2$, $\Breg{x}{y}=\tfrac12\|x-y\|^2$ and Theorem~\ref{thm:convergence} recovers the classic $O(1/k)$ rate for gradient descent on $L$‑smooth convex functions.
\item \textbf{Extension to Stochastic Setting:} Under unbiased stochastic gradients and bounded variance, the same proof skeleton yields an $O(1/\sqrt{K})$ rate after averaging.
\end{itemize}

\section*{Discussion}
The theorem shows that mirror descent with a Bregman proximal step attains the optimal sublinear rate for deterministic convex optimisation under the weaker \emph{relative smoothness} condition. This condition is satisfied, for example, when $f$ is smooth in a non‑Euclidean geometry (e.g., KL‑divergence for probability simplices). The proof relies only on the three‑point identity of Bregman divergences and does not require Lipschitz continuity of $\grad f$ in the Euclidean norm.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]
\label{lem:threepoint}
For any $x,y,z\in\R^n$,
\[
\inner{\grad h(z)-\grad h(y)}{x-z}= \Breg{x}{y}-\Breg{x}{z}-\Breg{z}{y}.
\]
\end{lemma}
\begin{proof}
By definition of $\Breg{\cdot}{\cdot}$,
\[
\Breg{x}{y}=h(x)-h(y)-\inner{\grad h(y)}{x-y},
\]
\[
\Breg{x}{z}=h(x)-h(z)-\inner{\grad h(z)}{x-z},
\]
\[
\Breg{z}{y}=h(z)-h(y)-\inner{\grad h(y)}{z-y}.
\]
Subtracting the last two equalities from the first yields the identity. ∎
\end{proof}

\begin{lemma}[Relative Smoothness Inequality]
\label{lem:relSmooth}
If $f$ satisfies Assumption~\ref{ass:smooth}, then for any $x,y\in\R^n$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+L\,\Breg{y}{x}.
\]
\end{lemma}
\begin{proof}
This is precisely the statement of Assumption~\ref{ass:smooth}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convergence} for a constant stepsize $\eta\le 1/L$.  
All non‑trivial steps are labelled \textbf{[Step N]}.

\subsection*{Step 1 – Optimality Condition of the Subproblem}
From the definition \eqref{eq:update}, $x_{k+1}$ minimizes a convex function, so the first‑order optimality condition reads
\begin{equation}
\label{eq:optcond}
\inner{\grad f(x_k)}{x-x_{k+1}}+\frac{1}{\eta}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x-x_{k+1}}\ge 0,\qquad\forall x\in\R^n.
\end{equation}
\textbf{[Step 1]}

\subsection*{Step 2 – Plug $x=x^{\star}$}
Choosing $x=x^{\star}$ (an optimal point) in \eqref{eq:optcond} gives
\begin{equation}
\label{eq:optstar}
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
\ge \frac{1}{\eta}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x_{k+1}-x^{\star}}.
\end{equation}
\textbf{[Step 2]}

\subsection*{Step 3 – Apply Three‑Point Identity}
Apply Lemma~\ref{lem:threepoint} with $(x,y,z)=(x^{\star},x_k,x_{k+1})$:
\[
\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}
= \Breg{x^{\star}}{x_k}-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}.
\]
Insert this into \eqref{eq:optstar}:
\begin{equation}
\label{eq:optstar2}
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
\ge \frac{1}{\eta}\Big(\Breg{x^{\star}}{x_k}-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}\Big).
\end{equation}
\textbf{[Step 3]}

\subsection*{Step 4 – Convexity of $f$}
Convexity (Assumption~\ref{ass:convex}) yields
\[
f(x_k)-f(x^{\star})\le \inner{\grad f(x_k)}{x_k-x^{\star}}.
\]
Rearrange:
\[
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
= \inner{\grad f(x_k)}{x^{\star}-x_k}+\inner{\grad f(x_k)}{x_k-x_{k+1}}
\le f(x^{\star})-f(x_k)+\inner{\grad f(x_k)}{x_k-x_{k+1}}.
\]
\textbf{[Step 4]}

\subsection*{Step 5 – Combine Step 3 and Step 4}
From \eqref{eq:optstar2} and the bound above,
\[
f(x_k)-f(x^{\star})\le \inner{\grad f(x_k)}{x_k-x_{k+1}}-\frac{1}{\eta}\Big(\Breg{x^{\star}}{x_k}-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}\Big).
\]
Rearrange terms:
\begin{equation}
\label{eq:basic}
f(x_k)-f(x^{\star})\le \inner{\grad f(x_k)}{x_k-x_{k+1}}+\frac{1}{\eta}\Breg{x^{\star}}{x_{k+1}}-\frac{1}{\eta}\Breg{x^{\star}}{x_k}+\frac{1}{\eta}\Breg{x_{k+1}}{x_k}.
\end{equation}
\textbf{[Step 5]}

\subsection*{Step 6 – Relative Smoothness Upper Bound}
By Lemma~\ref{lem:relSmooth} with $x=x_k$, $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_k)+\inner{\grad f(x_k)}{x_{k+1}-x_k}+L\,\Breg{x_{k+1}}{x_k}.
\]
Rearrange:
\[
\inner{\grad f(x_k)}{x_k-x_{k+1}}\le f(x_k)-f(x_{k+1})+L\,\Breg{x_{k+1}}{x_k}.
\]
\textbf{[Step 6]}

\subsection*{Step 7 – Substitute Step 6 into \eqref{eq:basic}}
\[
\begin{aligned}
f(x_k)-f(x^{\star})
&\le \big(f(x_k)-f(x_{k+1})+L\,\Breg{x_{k+1}}{x_k}\big)
   +\frac{1}{\eta}\Breg{x^{\star}}{x_{k+1}}-\frac{1}{\eta}\Breg{x^{\star}}{x_k}
   +\frac{1}{\eta}\Breg{x_{k+1}}{x_k}  \\
&= f(x_k)-f(x_{k+1})+\frac{1}{\eta}\Breg{x^{\star}}{x_{k+1}}-\frac{1}{\eta}\Breg{x^{\star}}{x_k}
   +\Big(L+\frac{1}{\eta}\Big)\Breg{x_{k+1}}{x_k}.
\end{aligned}
\]
\textbf{[Step 7]}

\subsection*{Step 8 – Use the Stepsize Condition}
Assumption~\ref{ass:stepsize} guarantees $\eta\le 1/L$, i.e. $L\le 1/\eta$. Hence
\[
L+\frac{1}{\eta}\le \frac{2}{\eta}.
\]
Thus,
\[
f(x_k)-f(x^{\star})\le f(x_k)-f(x_{k+1})+\frac{1}{\eta}\Breg{x^{\star}}{x_{k+1}}-\frac{1}{\eta}\Breg{x^{\star}}{x_k}
   +\frac{2}{\eta}\Breg{x_{k+1}}{x_k}.
\]
\textbf{[Step 8]}

\subsection*{Step 9 – Drop the Non‑negative Term}
Since $\Breg{x_{k+1}}{x_k}\ge 0$, discarding it yields a cleaner inequality:
\begin{equation}
\label{eq:telescoping}
f(x_k)-f(x^{\star})\le f(x_k)-f(x_{k+1})+\frac{1}{\eta}\big(\Breg{x^{\star}}{x_{k+1}}-\Breg{x^{\star}}{x_k}\big).
\end{equation}
\textbf{[Step 9]}

\subsection*{Step 10 – Rearrangement}
Move $f(x_{k+1})$ to the left side:
\[
f(x_{k+1})-f(x^{\star})\le \frac{1}{\eta}\big(\Breg{x^{\star}}{x_{k}}-\Breg{x^{\star}}{x_{k+1}}\big).
\]
\textbf{[Step 10]}

\subsection*{Step 11 – Telescoping Sum}
Sum \eqref{eq:telescoping} from $k=0$ to $K-1$:
\[
\sum_{k=0}^{K-1}\big(f(x_{k+1})-f(x^{\star})\big)
\le \frac{1}{\eta}\big(\Breg{x^{\star}}{x_{0}}-\Breg{x^{\star}}{x_{K}}\big)
\le \frac{V}{\eta},
\]
where $V:=\Breg{x^{\star}}{x_0}$ and we used $\Breg{x^{\star}}{x_K}\ge0$.
\textbf{[Step 11]}

\subsection*{Step 12 – Derive the Rate}
Since each term on the left is non‑negative,
\[
K\big(f(x_{K})-f(x^{\star})\big)\le \sum_{k=0}^{K-1}\big(f(x_{k+1})-f(x^{\star})\big)
\le \frac{V}{\eta},
\]
hence
\[
f(x_{K})-f(x^{\star})\le \frac{V}{\eta K}.
\]
\textbf{[Step 12]}

This completes the proof of the $O(1/K)$ bound for constant stepsize. The proof for the decreasing stepsize $\eta_k=2/(L(k+1))$ follows the same algebraic steps, with the telescoping sum yielding the tighter $\frac{2LV}{K(K+1)}$ bound.

\section*{Rate Derivation (With Constants)}
For constant $\eta\le 1/L$,
\[
\boxed{f(x_{K})-f(x^{\star})\le \frac{\Breg{x^{\star}}{x_0}}{\eta\,K}}.
\]
If we set $\eta=1/L$, the bound becomes
\[
f(x_{K})-f(x^{\star})\le \frac{L\,\Breg{x^{\star}}{x_0}}{K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean setting:} $h(x)=\tfrac12\|x\|_2^{2}$ gives $\Breg{x}{y}= \tfrac12\|x-y\|_2^{2}$ and the theorem reduces to the classic gradient‑descent rate.
\item \textbf{Entropy prox:} $h(x)=\sum_{i}x_i\log x_i$ on the probability simplex yields the KL‑divergence as Bregman distance; the same $O(1/k)$ rate holds for problems that are $L$‑smooth relative to the KL geometry (e.g., exponential families).
\item \textbf{Strongly convex $f$:} If, in addition, $f$ is $\mu$‑strongly convex, the same proof with the stronger inequality $f(x_k)-f(x^{\star})\ge \frac{\mu}{2}\|x_k-x^{\star}\|^{2}$ leads to a linear rate $f(x_k)-f(x^{\star})\le (1-\eta\mu)^{k}V$.
\end{itemize}
