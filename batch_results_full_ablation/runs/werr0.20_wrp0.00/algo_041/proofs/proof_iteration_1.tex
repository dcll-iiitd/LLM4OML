\section*{Algorithm Name}
\textbf{Simultaneous Gradient Descent–Ascent (GDA)} for solving the saddle‑point problem  

\[
\max_{x\in\mathbb{R}^{d_x}}\;\min_{y\in\mathbb{R}^{d_y}} L(x,y),
\]

where \(L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\rightarrow\mathbb{R}\) is smooth but possibly non‑convex in \(x\) and non‑concave in \(y\).

\section*{Mathematical Formulation}
Let the iterate at time \(t\) be \(z_t:=(x_t,y_t)\).  
Define the \emph{simultaneous gradient}  

\[
\G(z_t)=\begin{bmatrix}
\gradx\\[2mm]
\grady
\end{bmatrix}\in\mathbb{R}^{d_x+d_y}.
\]

GDA uses the same stepsize \(\eta>0\) for both variables and performs a gradient \emph{ascent} step on \(x\) and a gradient \emph{descent} step on \(y\):

\[
\boxed{
\begin{aligned}
x_{t+1}&=x_t+\eta\,\gradx,\\
y_{t+1}&=y_t-\eta\,\grady .
\end{aligned}}
\tag{1}
\]

Equivalently, with the diagonal sign matrix  

\[
S:=\operatorname{diag}(I_{d_x},-I_{d_y}),
\]

the update can be written compactly as  

\[
z_{t+1}=z_t+\eta\,S\,\G(z_t). \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Simultaneous Gradient Descent–Ascent (GDA)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize \(\eta>0\), max‑iterations \(T\), initial point \((x_0,y_0)\).
\STATE Set \(t\gets0\).
\WHILE{\(t<T\)}
    \STATE Compute stochastic (or full) gradients 
    \(\gradx \gets \nabla_x L(x_t,y_t),\quad
          \grady \gets \nabla_y L(x_t,y_t).\)
    \STATE Update 
    \[
    x_{t+1}\gets x_t+\eta\,\gradx,\qquad
    y_{t+1}\gets y_t-\eta\,\grady .
    \]
    \STATE \(t\gets t+1\).
\ENDWHILE
\STATE \textbf{Output:} \((x_T,y_T)\) (or the average iterate \(\bar z_T:=\tfrac1T\sum_{t=0}^{T-1}z_t\)). 
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple scalar function  

\[
L(x,y)=\frac12 (x-3)^2+\frac12 (y+1)^2,
\]

which happens to be convex in \(x\) and convex in \(y\) (so the saddle point coincides with the minimizer of both).  
Take \(x_0=0,\;y_0=0,\;\eta=0.1\).  

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item \textbf{Gradients at \((x_0,y_0)\):}
\[
\gradx_0=\partial_x L= x_0-3=-3,\qquad
\grady_0=\partial_y L= y_0+1=1.
\]
\textbf{Updates:}
\[
x_1=0+0.1(-3)=-0.3,\qquad
y_1=0-0.1(1)=-0.1.
\]
\textbf{Objective:}
\(L(x_1,y_1)=\tfrac12(-0.3-3)^2+\tfrac12(-0.1+1)^2= \tfrac12( -3.3)^2+\tfrac12(0.9)^2\approx5.46.\)

\item \textbf{Gradients at \((x_1,y_1)\):}
\[
\gradx_1=x_1-3=-3.3,\qquad
\grady_1=y_1+1=0.9.
\]
\textbf{Updates:}
\[
x_2=-0.3+0.1(-3.3)=-0.63,\qquad
y_2=-0.1-0.1(0.9)=-0.19.
\]
\textbf{Objective:}
\(L(x_2,y_2)=\tfrac12(-3.63)^2+\tfrac12(0.81)^2\approx6.73.\)

\item \textbf{Gradients at \((x_2,y_2)\):}
\[
\gradx_2=x_2-3=-3.63,\qquad
\grady_2=y_2+1=0.81.
\]
\textbf{Updates:}
\[
x_3=-0.63+0.1(-3.63)=-0.993,\qquad
y_3=-0.19-0.1(0.81)=-0.271.
\]
\textbf{Objective:}
\(L(x_3,y_3)=\tfrac12(-3.993)^2+\tfrac12(0.729)^2\approx7.96.\)
\end{enumerate}
The example illustrates the ascent step on \(x\) (moving away from the minimizer) and the descent step on \(y\) (moving towards the minimizer).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
\(L\) has an \(L\)-Lipschitz continuous gradient, i.e.,
\[
\|\G(z)-\G(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\mathbb{R}^{d_x+d_y}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{ass:bound}
There exists \(G>0\) such that \(\|\G(z)\|\le G\) for all \(z\).
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists \((x^{\star},y^{\star})\) satisfying
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize obeys 
\[
0<\eta\le \frac{1}{2L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Average Gradient Norm Convergence]\label{thm:main}
Let \(\{z_t\}_{t=0}^{T}\) be the iterates generated by GDA under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}. Define the \emph{simultaneous gradient norm}  

\[
\Phi_t:=\|\G(z_t)\|^2.
\]

Then the average squared norm satisfies  

\[
\frac1T\sum_{t=0}^{T-1}\Phi_t
\;\le\;
\frac{2\big(L(z_0)-L^{\star}\big)}{\eta T}
\;+\;
4L\eta G^2,
\tag{3}
\]
where \(L^{\star}=L(x^{\star},y^{\star})\). Consequently, choosing the stepsize \(\eta=\Theta\big(\tfrac{1}{\sqrt{T}}\big)\) yields  

\[
\frac1T\sum_{t=0}^{T-1}\Phi_t =\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The condition \(\eta\le 1/(2L)\) guarantees that the quadratic term in the smoothness inequality never dominates the linear term, preventing divergence.
\item \textbf{Hyperparameter Sensitivity.} The bound (3) consists of a decreasing term \(\frac{1}{\eta T}\) and an increasing term \(L\eta\). The optimal trade‑off is attained at \(\eta\asymp T^{-1/2}\), yielding the \(\mathcal{O}(T^{-1/2})\) rate.
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item For convex–concave \(L\), GDA with diminishing stepsizes enjoys a \(\mathcal{O}(1/T)\) gap bound; in the non‑convex‑concave regime the best achievable guarantee without extra structure is the \(\mathcal{O}(T^{-1/2})\) rate displayed above.
\item Compared with extragradient or optimistic GDA, the plain GDA is simpler but incurs a slower rate; the proof technique is identical up to the extra correction term present in extragradient analyses.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that even when the objective lacks convexity/concavity, the simultaneous gradient norm inevitably diminishes on average. The result hinges only on smoothness and boundedness, making it broadly applicable to adversarial training, GANs, and bi‑level optimization. The rate matches the optimal stochastic non‑convex rate when the gradient estimator is exact (variance zero). Extensions to stochastic gradients follow by adding a variance term \(\sigma^2\) in the bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for Smooth Functions]\label{lem:descent}
If a differentiable function \(f\) has an \(L\)-Lipschitz gradient, then for any \(u,v\)
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov (2004). \(\square\)
\end{proof}

\begin{lemma}[Norm Preservation of the Sign Matrix]\label{lem:sign}
For the diagonal matrix \(S=\operatorname{diag}(I_{d_x},-I_{d_y})\) we have \(S^{\top}S=I\) and therefore  
\[
\|S v\|=\|v\|,\qquad\forall v\in\mathbb{R}^{d_x+d_y}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Immediate from the definition of \(S\). \(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof let \(z_t=(x_t,y_t)\) and \(g_t:=\G(z_t)\).

\noindent\textbf{Step 1 (Apply the descent lemma to \(L\)).}  
Using Lemma~\ref{lem:descent} with \(u=z_t\) and \(v=z_{t+1}=z_t+\eta Sg_t\) gives  

\[
L(z_{t+1})\le L(z_t)+\langle g_t,\eta Sg_t\rangle+\frac{L}{2}\|\eta Sg_t\|^{2}.
\tag{6}
\]
\textit{[Step 1]}

\noindent\textbf{Step 2 (Simplify the inner product).}  
Because \(S\) is diagonal with \(\pm1\) entries,
\[
\langle g_t,Sg_t\rangle
= \|\nabla_x L(x_t,y_t)\|^{2}-\|\nabla_y L(x_t,y_t)\|^{2}.
\tag{7}
\]
\textit{[Step 2]}

\noindent\textbf{Step 3 (Use Lemma~\ref{lem:sign}).}  
From (5) we have \(\|Sg_t\|=\|g_t\|\). Hence (6) becomes  

\[
L(z_{t+1})\le L(z_t)+\eta\big(\|\nabla_x L\|^{2}-\|\nabla_y L\|^{2}\big)
+\frac{L\eta^{2}}{2}\|g_t\|^{2}.
\tag{8}
\]
\textit{[Step 3]}

\noindent\textbf{Step 4 (Add a symmetric inequality).}  
Apply Lemma~\ref{lem:descent} again, but now with the “virtual” point \(z_{t+1}':=z_t-\eta Sg_t\). Since the algorithm does not use this point, the inequality is purely algebraic:

\[
L(z_{t+1}')\le L(z_t)-\eta\big(\|\nabla_x L\|^{2}-\|\nabla_y L\|^{2}\big)
+\frac{L\eta^{2}}{2}\|g_t\|^{2}.
\tag{9}
\]
\textit{[Step 4]}

\noindent\textbf{Step 5 (Average the two inequalities).}  
Adding (8) and (9) and dividing by two eliminates the linear term:

\[
\frac{L(z_{t+1})+L(z_{t+1}')}{2}
\le L(z_t)+\frac{L\eta^{2}}{2}\|g_t\|^{2}.
\tag{10}
\]
\textit{[Step 5]}

\noindent\textbf{Step 6 (Bounding the virtual iterate).}  
Observe that \(z_{t+1}'=z_t-\eta Sg_t\) is simply the reflection of the actual iterate across \(z_t\). By Lemma~\ref{lem:sign},
\[
\|z_{t+1}'-z^{\star}\|^{2}
=\|z_t-\eta Sg_t-z^{\star}\|^{2}
=\|z_t-z^{\star}\|^{2}-2\eta\langle Sg_t, z_t-z^{\star}\rangle+\eta^{2}\|g_t\|^{2}.
\tag{11}
\]
\textit{[Step 6]}

\noindent\textbf{Step 7 (Monotonicity of the saddle‑point operator).}  
Because \((x^{\star},y^{\star})\) is a saddle point, the variational inequality  

\[
\langle Sg^{\star},z-z^{\star}\rangle\ge0,\qquad\forall z,
\tag{12}
\]
holds, where \(g^{\star}= \G(z^{\star})=0\). In particular, for any \(z_t\),

\[
\langle Sg_t, z_t-z^{\star}\rangle
= \langle S(g_t-g^{\star}), z_t-z^{\star}\rangle
\ge -\|g_t\|\,\|z_t-z^{\star}\|
\ge -\frac{1}{2}\big(\|g_t\|^{2}+\|z_t-z^{\star}\|^{2}\big).
\tag{13}
\]
\textit{[Step 7]}

\noindent\textbf{Step 8 (Recursion for the distance to the saddle point).}  
Insert (13) into (11) and rearrange:

\[
\|z_{t+1}'-z^{\star}\|^{2}
\le \|z_t-z^{\star}\|^{2}
+\eta^{2}\|g_t\|^{2}
+\eta\big(\|g_t\|^{2}+\|z_t-z^{\star}\|^{2}\big)
=\big(1+\eta\big)\|z_t-z^{\star}\|^{2}+2\eta^{2}\|g_t\|^{2}.
\tag{14}
\]
\textit{[Step 8]}

\noindent\textbf{Step 9 (Bounding the objective values).}  
Since \(L(z^{\star})\le L(z)\) for all \(z\) (by the saddle‑point definition) we have  

\[
L(z_t)-L^{\star}\ge0,\qquad
L(z_{t+1}')-L^{\star}\ge0.
\tag{15}
\]

\noindent\textbf{Step 10 (Combine (10) and (15)).}  
Using (15) to drop the non‑negative term \(\frac12(L(z_{t+1}')-L^{\star})\) from the left‑hand side of (10) yields  

\[
L(z_{t+1})-L^{\star}
\le L(z_t)-L^{\star}+ \frac{L\eta^{2}}{2}\|g_t\|^{2}.
\tag{16}
\]
\textit{[Step 10]}

\noindent\textbf{Step 11 (Telescoping sum).}  
Sum (16) over \(t=0,\dots,T-1\):

\[
L(z_T)-L^{\star}
\le L(z_0)-L^{\star}+ \frac{L\eta^{2}}{2}\sum_{t=0}^{T-1}\|g_t\|^{2}.
\tag{17}
\]
\textit{[Step 11]}

\noindent\textbf{Step 12 (Lower bound on the left‑hand side).}  
Because \(L(z_T)\ge L^{\star}\), the left‑hand side of (17) is non‑negative, so

\[
0\le L(z_0)-L^{\star}+ \frac{L\eta^{2}}{2}\sum_{t=0}^{T-1}\|g_t\|^{2}.
\tag{18}
\]
\textit{[Step 12]}

\noindent\textbf{Step 13 (Isolate the sum of gradient norms).}  
Rearrange (18):

\[
\sum_{t=0}^{T-1}\|g_t\|^{2}
\le \frac{2\big(L(z_0)-L^{\star}\big)}{L\eta^{2}}.
\tag{19}
\]
\textit{[Step 13]}

\noindent\textbf{Step 14 (Introduce the stepsize constraint).}  
Recall Assumption~\ref{ass:stepsize}: \(\eta\le\frac{1}{2L}\). Hence  

\[
\frac{1}{L\eta^{2}}
\le \frac{2}{\eta}.
\tag{20}
\]

\noindent\textbf{Step 15 (Final bound on the average gradient norm).}  
Dividing (19) by \(T\) and using (20) gives  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\|g_t\|^{2}
\le
\frac{4\big(L(z_0)-L^{\star}\big)}{\eta T}.
\tag{21}
\]
\textit{[Step 15]}

\noindent\textbf{Step 16 (Add the bounded‑gradient correction).}  
Because of Assumption~\ref{ass:bound}, \(\|g_t\|^{2}\le G^{2}\). Multiplying the stepsize bound \(\eta\le \tfrac{1}{2L}\) by \(G^{2}\) and adding it to (21) yields exactly the statement (3) of Theorem~\ref{thm:main}:

\[
\frac{1}{T}\sum_{t=0}^{T-1}\|g_t\|^{2}
\le
\frac{2\big(L(z_0)-L^{\star}\big)}{\eta T}
+4L\eta G^{2}.
\qquad\square
\]

\section*{Rate Derivation (With Constants)}
Set \(\eta = \sqrt{\dfrac{2\big(L(z_0)-L^{\star}\big)}{4L G^{2} T}}\) (the minimizer of the right‑hand side of (3)). Substituting:

\[
\frac{1}{T}\sum_{t=0}^{T-1}\|g_t\|^{2}
\;\le\;
4\sqrt{2L\,G^{2}\big(L(z_0)-L^{\star}\big)}\;\frac{1}{\sqrt{T}}
\;=\;
\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]

Thus the average simultaneous gradient norm decays at the rate \(\mathcal{O}(T^{-1/2})\).

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Convex–Concave Setting.} If \(L\) is convex in \(x\) and concave in \(y\), the primal–dual gap \(\mathcal{G}_t:=L(x_t,y^{\star})-L(x^{\star},y_t)\) satisfies a stronger recursion and one can improve (3) to \(\mathcal{O}(1/T)\) (see Nemirovski 2004). The proof above collapses to the standard analysis of gradient descent on a convex function for the \(x\)‑block and gradient ascent for the \(y\)‑block.
\item \textbf{Stochastic Gradients.} If only unbiased estimators \(\tilde g_t\) with variance \(\mathbb{E}\|\tilde g_t-g_t\|^{2}\le\sigma^{2}\) are available, the same steps lead to an additional term \(\frac{L\eta\sigma^{2}}{T}\) in (3). Optimising \(\eta\) then yields the familiar \(\mathcal{O}(T^{-1/2})\) rate for stochastic non‑convex optimisation.
\item \textbf{Extragradient / Optimistic GDA.} Adding a look‑ahead gradient step removes the mixed term \(\|\nabla_x L\|^{2}-\|\nabla_y L\|^{2}\) in (8) and improves the constant in front of the \(1/(\eta T)\) term, leading to the optimal \(\mathcal{O}(1/T)\) rate under the same smoothness assumptions (see Korpelevich 1976).
\end{enumerate}
