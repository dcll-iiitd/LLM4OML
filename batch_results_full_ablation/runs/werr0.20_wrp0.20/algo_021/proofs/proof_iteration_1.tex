\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let  
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\mathbb R^{d},
\]
where  

\begin{itemize}
\item $f:\mathbb R^{d}\to\mathbb R$ is convex and differentiable with $L$‑Lipschitz continuous gradient,
\item $g:\mathbb R^{d}\to\mathbb R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).
\end{itemize}
For a stepsize $\eta>0$ the proximal operator of $g$ is  
\[
\operatorname{prox}_{\eta g}(v)\;:=\;\arg\min_{u\in\mathbb R^{d}}\Big\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]

The **proximal gradient update** is  
\[
\boxed{\,x_{k+1}= \operatorname{prox}_{\eta g}\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),\qquad k=0,1,\dots\, } \tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, initial point $x_{0}\in\mathbb R^{d}$, max iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \STATE $x_{k+1}\gets \operatorname{prox}_{\eta g}\bigl(x_{k}-\eta g_{k}\bigr)$.
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or the best iterate w.r.t.\ $F$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv 0,\qquad \eta=0.1,\qquad w_{0}=0 .
\]

Since $g\equiv0$, $\operatorname{prox}_{\eta g}(v)=v$.  
The gradient $\nabla f(w)=2(w-3)$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values $F(w_{k})=f(w_{k})$ decrease:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3686$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]\label{as:convex}
$f:\mathbb R^{d}\to\mathbb R$ is convex and differentiable with an $L$‑Lipschitz gradient, i.e.  
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb R^{d}.
\] 
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{as:gconvex}
$g:\mathbb R^{d}\to\mathbb R\cup\{+\infty\}$ is proper, closed and convex.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Existence of a Minimizer]\label{as:exist}
The set of minimizers $\mathcal X^{*}:=\arg\min_{x}F(x)$ is non‑empty; pick any $x^{*}\in\mathcal X^{*}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sub‑linear $O(1/k)$ rate]\label{thm:rate}
Under Assumptions \ref{as:convex}–\ref{as:exist} and for any stepsize $\eta\in(0,1/L]$, the iterates generated by \eqref{eq:pgm} satisfy for all $k\ge1$  
\[
F(x_{k})-F(x^{*})\;\le\;\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k}.
\] 
Consequently $F(x_{k})\to F(x^{*})$ as $k\to\infty$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The update is a composition of a gradient step (non‑expansive for $\eta\le1/L$) and a firmly non‑expansive proximal map; thus the whole mapping is non‑expansive, guaranteeing boundedness of the iterates.
\item \textbf{Hyper‑parameter sensitivity:} The bound \eqref{thm:rate} is monotone decreasing in $\eta$; the largest admissible stepsize $\eta=1/L$ yields the smallest constant $\frac{1}{2\eta}=L/2$.
\item \textbf{Comparison:} Compared with plain gradient descent on $F$ (which would require $g$ smooth), the proximal gradient retains the same $O(1/k)$ rate while handling non‑smooth $g$.
\end{itemize}

\section*{Discussion}
The proof hinges on two elementary facts:
\begin{enumerate}
\item The \emph{descent lemma} for $f$ (Lipschitz gradient).
\item The \emph{firm non‑expansiveness} of the proximal operator, which provides a variational inequality linking $x_{k+1}$, $x_{k}$ and any optimal point $x^{*}$.
\end{enumerate}
Both are invoked without any stochasticity, so the result holds deterministically.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Proximal operator]\label{def:prox}
For $\eta>0$ and convex $g$,  
\[
\operatorname{prox}_{\eta g}(v)=\arg\min_{u}\Big\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]
\end{definition}
\begin{lemma}[Firm non‑expansiveness]\label{lem:firm}
For any $u,v\in\mathbb R^{d}$,
\[
\big\|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\big\|^{2}
\le \langle u-v,\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\rangle .
\] 
Equivalently,
\[
\langle u-\operatorname{prox}_{\eta g}(u)-\bigl(v-\operatorname{prox}_{\eta g}(v)\bigr),
\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\rangle\ge 0 .
\] 
\end{lemma}
\begin{lemma}[Descent lemma for $f$]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y$,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\] 
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  
Let $x^{*}\in\mathcal X^{*}$ be arbitrary but fixed.  
Denote the gradient step before the proximal map as
\[
y_{k}=x_{k}-\eta\nabla f(x_{k}). \tag{2}
\]

\noindent\textbf{[Step 1]} Optimality condition of the proximal map.  
By definition of $\operatorname{prox}_{\eta g}$,
\[
x_{k+1}= \operatorname{prox}_{\eta g}(y_{k})
\;\Longleftrightarrow\;
\frac{1}{\eta}\bigl(y_{k}-x_{k+1}\bigr)\in\partial g(x_{k+1}). \tag{3}
\]

\noindent\textbf{[Step 2]} Subgradient inequality for $g$.  
For any $u\in\mathbb R^{d}$ and any $s\in\partial g(x_{k+1})$,
\[
g(u)\ge g(x_{k+1})+\langle s,\,u-x_{k+1}\rangle .
\] 
Choose $u=x^{*}$ and $s$ as in \eqref{3}:
\[
g(x^{*})\ge g(x_{k+1})+\frac{1}{\eta}\langle y_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle . \tag{4}
\]

\noindent\textbf{[Step 3]} Convexity of $f$.  
For any $u$,
\[
f(u)\ge f(x_{k})+\langle \nabla f(x_{k}),\,u-x_{k}\rangle . \tag{5}
\]
Set $u=x^{*}$ and $u=x_{k+1}$ respectively.

\noindent\textbf{[Step 4]} Combine \eqref{4} and \eqref{5} for $u=x^{*}$.  
Adding the two inequalities gives
\[
F(x^{*})\ge f(x_{k})+g(x_{k+1})
+\langle \nabla f(x_{k}),\,x^{*}-x_{k}\rangle
+\frac{1}{\eta}\langle y_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle . \tag{6}
\]

\noindent\textbf{[Step 5]} Expand $y_{k}=x_{k}-\eta\nabla f(x_{k})$ in the last inner product:
\[
\begin{aligned}
\langle y_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle
&=\langle x_{k}-\eta\nabla f(x_{k})-x_{k+1},\,x^{*}-x_{k+1}\rangle\\
&=\langle x_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle
-\eta\langle \nabla f(x_{k}),\,x^{*}-x_{k+1}\rangle .
\end{aligned} \tag{7}
\]

\noindent\textbf{[Step 6]} Substitute \eqref{7} into \eqref{6} and cancel the gradient terms:
\[
\begin{aligned}
F(x^{*})\ge&\; f(x_{k})+g(x_{k+1})
+\langle \nabla f(x_{k}),\,x^{*}-x_{k}\rangle\\
&\;+\frac{1}{\eta}\langle x_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle
-\langle \nabla f(x_{k}),\,x^{*}-x_{k+1}\rangle .
\end{aligned}
\] 
Observe that 
\[
\langle \nabla f(x_{k}),\,x^{*}-x_{k}\rangle-\langle \nabla f(x_{k}),\,x^{*}-x_{k+1}\rangle
=\langle \nabla f(x_{k}),\,x_{k+1}-x_{k}\rangle . \tag{8}
\]

\noindent\textbf{[Step 7]} Insert \eqref{8}:
\[
F(x^{*})\ge f(x_{k})+g(x_{k+1})
+\langle \nabla f(x_{k}),\,x_{k+1}-x_{k}\rangle
+\frac{1}{\eta}\langle x_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle . \tag{9}
\]

\noindent\textbf{[Step 8]} Apply the descent lemma \eqref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\langle \nabla f(x_{k}),\,x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \tag{10}
\]

\noindent\textbf{[Step 9]} Since $\eta\le 1/L$, we have $\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\le\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}$.  
Replace the first three terms on the right‑hand side of \eqref{9} by $f(x_{k+1})$ using \eqref{10}:
\[
F(x^{*})\ge f(x_{k+1})+g(x_{k+1})
+\frac{1}{\eta}\langle x_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle
-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}. \tag{11}
\]

\noindent\textbf{[Step 10]} Recognize $F(x_{k+1})=f(x_{k+1})+g(x_{k+1})$ and rewrite the inner product:
\[
\langle x_{k}-x_{k+1},\,x^{*}-x_{k+1}\rangle
=\frac{1}{2}\Big(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\Big). \tag{12}
\]

\noindent\textbf{[Step 11]} Substitute \eqref{12} into \eqref{11} and simplify:
\[
\begin{aligned}
F(x^{*})\ge&\; F(x_{k+1})
+\frac{1}{2\eta}\Big(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Big)\\
&\;-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\] 
The two identical $\|x_{k+1}-x_{k}\|^{2}$ terms cancel, yielding
\[
F(x_{k+1})-F(x^{*})\le \frac{1}{2\eta}\Big(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\Big). \tag{13}
\]

\noindent\textbf{[Step 12]} Rearranged, inequality \eqref{13} is the fundamental descent relation:
\[
\boxed{ \; \|x_{k+1}-x^{*}\|^{2}\le \|x_{k}-x^{*}\|^{2}
-2\eta\bigl(F(x_{k+1})-F(x^{*})\bigr). \;}
\] 

\noindent\textbf{[Step 13]} Sum \eqref{13} from $k=0$ to $k=K-1$:
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{*})\bigr)
\le \frac{1}{2\eta}\bigl(\|x_{0}-x^{*}\|^{2}-\|x_{K}-x^{*}\|^{2}\bigr)
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta}. \tag{14}
\]

\noindent\textbf{[Step 14]} Since the left‑hand side is a sum of $K$ non‑negative terms, the smallest term cannot exceed the average:
\[
\min_{1\le k\le K}\bigl(F(x_{k})-F(x^{*})\bigr)
\le \frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{*})\bigr)
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta K}. \tag{15}
\]

Choosing the last iterate $x_{K}$ (or any iterate) yields the same bound, completing the proof.
\qed

\section*{Rate Derivation (With Constants)}
From \eqref{13} we have for every $k$,
\[
F(x_{k})-F(x^{*})\le \frac{\|x_{k-1}-x^{*}\|^{2}}{2\eta}
-\frac{\|x_{k}-x^{*}\|^{2}}{2\eta}
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k},
\]
where the last inequality follows by telescoping the series as in \eqref{14}.  
Hence the explicit constant in the $O(1/k)$ rate is $\frac{1}{2\eta}\|x_{0}-x^{*}\|^{2}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal step reduces to the identity, and \eqref{13} becomes the classical descent inequality for gradient descent with stepsize $\eta\le1/L$.
\item \textbf{Indicator function $g=\iota_{C}$ of a convex set $C$.} Then $\operatorname{prox}_{\eta g}$ is the Euclidean projection onto $C$, and the method recovers the projected gradient algorithm with the same $O(1/k)$ guarantee.
\item \textbf{Lasso ($g(x)=\lambda\|x\|_{1}$).} The proximal map is soft‑thresholding; the theorem guarantees that the objective value converges at rate $1/k$ without any additional assumptions on $\lambda$.
\end{itemize}
