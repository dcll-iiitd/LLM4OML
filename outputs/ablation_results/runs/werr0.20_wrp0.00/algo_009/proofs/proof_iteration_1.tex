\section*{SAGA for Non‑Convex Smooth Finite‑Sum Optimization}

\section*{Mathematical Formulation}
Consider a finite sum
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad x\in\R^{d},
\]
where each component $f_i$ is $L$–smooth (possibly non‑convex).  
The algorithm maintains a table of \emph{reference points} $\{\phi_i\}_{i=1}^{n}$ and the corresponding stored gradients
\[
g_i:=\grad f_i(\phi_i).
\]
Let $\bar g:=\frac{1}{n}\sum_{j=1}^{n} g_j$ denote the average stored gradient.  
At iteration $k$, a random index $i_k\in\{1,\dots,n\}$ is drawn uniformly. The update is
\begin{equation}
\boxed{
x_{k+1}=x_{k}-\alpha\Bigl(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g\Bigr)
}\label{eq:update}
\end{equation}
where $\alpha>0$ is the stepsize.  
After the update, the table entry for the sampled index is refreshed:
\[
\phi_{i_k}\gets x_{k},\qquad 
g_{i_k}\gets \grad f_{i_k}(x_{k}).
\]
All other table entries remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex version)}
\begin{algorithmic}[1]
\Require Number of samples $n$, stepsize $\alpha>0$, max iterations $K$.
\State Initialize $x_{0}\in\R^{d}$ arbitrarily.
\For{$i=1,\dots,n$}
    \State $\phi_i^{0}\gets x_{0}$, \; $g_i^{0}\gets \grad f_i(x_{0})$.
\EndFor
\State $\bar g^{0}\gets \frac{1}{n}\sum_{j=1}^{n} g_j^{0}$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$.
    \State \textbf{Compute} $v_k:=\grad f_{i_k}(x_{k})-g_{i_k}^{k}+\bar g^{k}$.
    \State $x_{k+1}\gets x_{k}-\alpha v_k$.
    \State Update the table:
    \State\quad $g_{i_k}^{k+1}\gets \grad f_{i_k}(x_{k})$, \;
          $\phi_{i_k}^{k+1}\gets x_{k}$,
    \State\quad For $j\neq i_k$: $g_{j}^{k+1}=g_{j}^{k},\;\phi_{j}^{k+1}=\phi_{j}^{k}$.
    \State $\bar g^{k+1}\gets \bar g^{k}+\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$.
\EndFor
\State \Return $x_{K}$ (or any iterate or the average).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimize the scalar quadratic $f(w)=(w-3)^{2}$, which can be written as a finite sum with $n=1$:
\[
f_{1}(w)=(w-3)^{2},\qquad f(w)=f_{1}(w).
\]
Here $\grad f_{1}(w)=2(w-3)$ and $L=2$.

\begin{enumerate}[label=\textbf{Iter \arabic*:},leftmargin=*,itemsep=1ex]
\item \textbf{Initialization:} $w_{0}=0$.
    \[
    \phi_{1}^{0}=w_{0}=0,\qquad g_{1}^{0}= \grad f_{1}(\phi_{1}^{0})=2(0-3)=-6,
    \]
    $\bar g^{0}=g_{1}^{0}=-6$.
\item \textbf{Iteration 0 (k=0).}
    \begin{itemize}
        \item Sample $i_{0}=1$ (the only index).
        \item Compute $v_{0}= \grad f_{1}(w_{0})-g_{1}^{0}+\bar g^{0}
                 =(-6)-(-6)+(-6)=-6$.
        \item Update $w_{1}=w_{0}-\alpha v_{0}=0-0.1(-6)=0.6$.
        \item Refresh table:
            $g_{1}^{1}= \grad f_{1}(w_{0})=-6$, $\phi_{1}^{1}=w_{0}=0$ (unchanged).
            $\bar g^{1}= \bar g^{0}= -6$.
        \item Objective value $f(w_{1})=(0.6-3)^{2}=5.76$.
    \end{itemize}
\item \textbf{Iteration 1 (k=1).}
    \begin{itemize}
        \item $v_{1}= \grad f_{1}(w_{1})-g_{1}^{1}+\bar g^{1}
                 =2(0.6-3)-(-6)+(-6)= -4.8+6-6=-4.8$.
        \item $w_{2}=w_{1}-0.1\,v_{1}=0.6-0.1(-4.8)=1.08$.
        \item Refresh: $g_{1}^{2}= \grad f_{1}(w_{1})=-4.8$, $\phi_{1}^{2}=w_{1}=0.6$,
            $\bar g^{2}= \bar g^{1}+ \frac{1}{1}(g_{1}^{2}-g_{1}^{1})
                       =-6+(-4.8+6)= -4.8$.
        \item $f(w_{2})=(1.08-3)^{2}=3.6864$.
    \end{itemize}
\item \textbf{Iteration 2 (k=2).}
    \begin{itemize}
        \item $v_{2}= \grad f_{1}(w_{2})-g_{1}^{2}+\bar g^{2}
                 =2(1.08-3)-(-4.8)+(-4.8)= -3.84+4.8-4.8=-3.84$.
        \item $w_{3}=w_{2}-0.1\,v_{2}=1.08-0.1(-3.84)=1.464$.
        \item Refresh: $g_{1}^{3}= \grad f_{1}(w_{2})=-3.84$, $\phi_{1}^{3}=w_{2}=1.08$,
            $\bar g^{3}= -4.8+(-3.84+4.8)= -3.84$.
        \item $f(w_{3})=(1.464-3)^{2}=2.366\ldots$.
    \end{itemize}
\end{enumerate}
The iterates move towards the global minimiser $w^{\star}=3$ and the objective value decreases.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i:\R^{d}\to\R$ is $L$–smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad 
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The objective is the uniform average $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$.
\end{assumption}
\begin{assumption}[Bounded Variance of Stochastic Gradient]\label{ass:var}
Define the stochastic gradient $g(x,i)=\grad f_i(x)$. Then for any $x$,
\[
\E_{i}\!\left[\norm{g(x,i)-\grad f(x)}^{2}\right]\le \sigma^{2},
\]
with $\sigma^{2}<\infty$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\alpha\le \frac{1}{3L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SAGA convergence]\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}, let $\{x_k\}_{k\ge0}$ be generated by \eqref{eq:update}. For any integer $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha K}
\;+\;
12L\alpha\,\sigma^{2},
\]
where $f^{\star}= \inf_{x}f(x)$.
Consequently, choosing $\alpha = \Theta\bigl(\tfrac{1}{\sqrt{K}}\bigr)$ yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
= O\!\left(\frac{1}{\sqrt{K}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The Lyapunov function 
\[
\Phi_k:= f(x_k)-f^{\star}+\frac{c}{n}\sum_{i=1}^{n}\norm{\phi_i^{k}-x^{\star}}^{2},
\qquad c:=\frac{1}{2\alpha L},
\]
is non‑increasing in expectation, guaranteeing that the iterates remain in a bounded region when $f$ is lower‑bounded.

\item \textbf{Stepsize Sensitivity.} The bound in Theorem~\ref{thm:main} holds for any $\alpha\le1/(3L)$. Larger stepsizes break the contraction in the variance term (see Lemma~\ref{lem:variance-reduction}) and may lead to divergence.

\item \textbf{Comparison to SGD.} For the same $L$‑smooth non‑convex setting, SGD with stepsize $\alpha$ obtains
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
= O\!\left(\frac{1}{\sqrt{K}}+\alpha\sigma^{2}\right).
\]
SAGA reduces the variance term from $O(\alpha\sigma^{2})$ to $O(\alpha L\sigma^{2})$ and thus enjoys a better constant factor while keeping the same $O(1/\sqrt{K})$ rate.

\item \textbf{Special Cases.}
\begin{itemize}
\item If all $f_i$ are convex, the same proof yields the stronger guarantee
\[
\E\!\bigl[f(\bar x_K)-f^{\star}\bigr]\le
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha K}+12L\alpha\sigma^{2},
\]
where $\bar x_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k$.
\item When $n=1$, SAGA reduces to deterministic gradient descent and the bound becomes the classic $O(1/K)$ rate for smooth non‑convex functions.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that SAGA attains the optimal $O(1/\sqrt{K})$ decay of the expected squared gradient norm for smooth non‑convex finite‑sum problems, matching the best known rates for variance‑reduced methods. The constant $12L\alpha\sigma^{2}$ reflects the residual variance after variance reduction; it can be driven to zero by decreasing $\alpha$ at the price of a slower $1/( \alpha K)$ term. The analysis hinges on a careful decomposition of the variance introduced by the table and on the smoothness inequality, both of which are detailed in the proof below.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f_i(y)\le f_i(x)+\langle\grad f_i(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness (integrate the gradient Lipschitz bound). \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA direction]\label{lem:unbiased}
For the random index $i_k$,
\[
\E_{i_k}\!\bigl[\,\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g^{k}\,\bigr]
= \grad f(x_k).
\]
\end{lemma}
\begin{proof}
Since $i_k$ is uniform,
\[
\E_{i_k}[\grad f_{i_k}(x_k)]=\frac{1}{n}\sum_{i=1}^{n}\grad f_i(x_k)=\grad f(x_k).
\]
Similarly,
\[
\E_{i_k}[\grad f_{i_k}(\phi_{i_k})]=\frac{1}{n}\sum_{i=1}^{n}\grad f_i(\phi_i)=\bar g^{k}.
\]
Subtracting yields the claim. \hfill $\square$
\end{proof}

\begin{lemma}[Variance reduction]\label{lem:variance-reduction}
Define the stochastic direction $v_k$ as in \eqref{eq:update}. Then for any $k$,
\[
\E_{i_k}\!\bigl[\norm{v_k-\grad f(x_k)}^{2}\bigr]
\le 2L^{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
\;+\;\sigma^{2}.
\]
\end{lemma}
\begin{proof}
Write
\[
v_k-\grad f(x_k)=\bigl(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})\bigr)
-\bigl(\grad f(x_k)-\bar g^{k}\bigr).
\]
Take expectation and use the bound
\[
\|\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})\|
\le L\|x_k-\phi_{i_k}\|
\]
by smoothness. Expanding the squared norm, applying $\|a-b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$, and using the definition of $\sigma^{2}$ (Assumption~\ref{ass:var}) yields the result. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t. the randomness up to the current iteration.

\noindent\textbf{Step 1 (Smoothness of the full objective).}  
From Lemma~\ref{lem:smooth} with $y=x_{k+1}$ and $x=x_k$,
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle
+\frac{L}{2}\norm{x_{k+1}-x_k}^{2}. \tag{1}
\]

\noindent\textbf{Step 2 (Insert the update).}  
Using \eqref{eq:update},
\[
x_{k+1}-x_k=-\alpha v_k,
\]
hence (1) becomes
\[
f(x_{k+1})\le f(x_k)-\alpha\langle\grad f(x_k),v_k\rangle
+\frac{L\alpha^{2}}{2}\norm{v_k}^{2}. \tag{2}
\]

\noindent\textbf{Step 3 (Decompose the inner product).}  
Add and subtract $\grad f(x_k)$ inside the inner product:
\[
\langle\grad f(x_k),v_k\rangle
= \norm{\grad f(x_k)}^{2}
+ \langle\grad f(x_k),v_k-\grad f(x_k)\rangle.
\]
Plug into (2):
\[
f(x_{k+1})\le f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
-\alpha\langle\grad f(x_k),v_k-\grad f(x_k)\rangle
+\frac{L\alpha^{2}}{2}\norm{v_k}^{2}. \tag{3}
\]

\noindent\textbf{Step 4 (Take expectation w.r.t. $i_k$).}  
Condition on the history $\mathcal{F}_k$ up to iteration $k$.
Using Lemma~\ref{lem:unbiased}, $\E_{i_k}[v_k]=\grad f(x_k)$, thus
\[
\E_{i_k}\!\bigl[\langle\grad f(x_k),v_k-\grad f(x_k)\rangle\bigr]=0.
\]
Taking expectation of (3) yields
\[
\E_{i_k}[f(x_{k+1})]\le f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
+\frac{L\alpha^{2}}{2}\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr]. \tag{4}
\]

\noindent\textbf{Step 5 (Expand $\norm{v_k}^{2}$).}  
Write $\norm{v_k}^{2}
=\norm{\grad f(x_k)}^{2}
+ \norm{v_k-\grad f(x_k)}^{2}
+2\langle\grad f(x_k),v_k-\grad f(x_k)\rangle$.
Taking expectation and using the same unbiasedness argument as in Step 4,
\[
\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr]
= \norm{\grad f(x_k)}^{2}
+ \E_{i_k}\!\bigl[\norm{v_k-\grad f(x_k)}^{2}\bigr]. \tag{5}
\]

\noindent\textbf{Step 6 (Insert variance bound).}  
Apply Lemma~\ref{lem:variance-reduction} to the second term of (5):
\[
\E_{i_k}\!\bigl[\norm{v_k-\grad f(x_k)}^{2}\bigr]
\le 2L^{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+\sigma^{2}. \tag{6}
\]

\noindent\textbf{Step 7 (Combine (4), (5), (6)).}  
Substituting (5) and (6) into (4) gives
\[
\begin{aligned}
\E_{i_k}[f(x_{k+1})]
&\le f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
+\frac{L\alpha^{2}}{2}\Bigl(\norm{\grad f(x_k)}^{2}
+2L^{2}\,\tfrac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+\sigma^{2}\Bigr)\\[1ex]
&= f(x_k)-\underbrace{\Bigl(\alpha-\tfrac{L\alpha^{2}}{2}\Bigr)}_{=: \gamma}
\norm{\grad f(x_k)}^{2}
+L^{3}\alpha^{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+\frac{L\alpha^{2}}{2}\sigma^{2}. \tag{7}
\end{aligned}
\]

\noindent\textbf{Step 8 (Define a Lyapunov function).}  
Set
\[
\Phi_k := f(x_k)-f^{\star}
+\frac{c}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2},
\]
where $c>0$ will be chosen later. Observe that only the sampled index $i_k$ changes its table entry:
\[
\phi_{i_k}^{k+1}=x_k,\qquad
\phi_{j}^{k+1}=\phi_{j}^{k}\;(j\neq i_k).
\]

\noindent\textbf{Step 9 (Evolution of the table term).}  
Conditioned on $\mathcal{F}_k$,
\[
\begin{aligned}
\E_{i_k}\!\Bigl[\tfrac{c}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k+1}}^{2}\Bigr]
&= \frac{c}{n}\Bigl(
\underbrace{\sum_{i\neq i_k}\norm{x_{k+1}-\phi_i^{k}}^{2}}_{\text{unchanged}}
+\norm{x_{k+1}-x_k}^{2}
\Bigr)\\
&= \frac{c}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k}}^{2}
+\frac{c}{n}\norm{x_{k+1}-x_k}^{2}
-\frac{c}{n}\norm{x_{k+1}-\phi_{i_k}^{k}}^{2}. \tag{8}
\end{aligned}
\]
Taking expectation over $i_k$ and using uniform sampling,
\[
\E_{i_k}\!\Bigl[\tfrac{c}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k+1}}^{2}\Bigr]
= \frac{c}{n}\sum_{i=1}^{n}\E_{i_k}\!\bigl[\norm{x_{k+1}-\phi_i^{k}}^{2}\bigr]
+\frac{c}{n}\E_{i_k}\!\bigl[\norm{x_{k+1}-x_k}^{2}\bigr]
-\frac{c}{n^{2}}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k}}^{2}. \tag{9}
\]

\noindent\textbf{Step 10 (Upper bound the first term).}  
Using the definition of $x_{k+1}$,
\[
x_{k+1}=x_k-\alpha v_k,
\]
so
\[
\norm{x_{k+1}-\phi_i^{k}}^{2}
= \norm{x_k-\phi_i^{k}-\alpha v_k}^{2}
= \norm{x_k-\phi_i^{k}}^{2}
-2\alpha\langle v_k, x_k-\phi_i^{k}\rangle
+\alpha^{2}\norm{v_k}^{2}.
\]
Taking expectation and noting that $\E_{i_k}[v_k]=\grad f(x_k)$,
\[
\E_{i_k}\!\bigl[\norm{x_{k+1}-\phi_i^{k}}^{2}\bigr]
\le \norm{x_k-\phi_i^{k}}^{2}
+2\alpha^{2}\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr],
\]
where we used $|\langle v_k, x_k-\phi_i^{k}\rangle|\le \norm{v_k}\norm{x_k-\phi_i^{k}}$
and then applied Young’s inequality $2ab\le a^{2}+b^{2}$.
Summing over $i$ and dividing by $n$ yields
\[
\frac{1}{n}\sum_{i=1}^{n}\E_{i_k}\!\bigl[\norm{x_{k+1}-\phi_i^{k}}^{2}\bigr]
\le \frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+2\alpha^{2}\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr]. \tag{10}
\]

\noindent\textbf{Step 11 (Bound the remaining two terms in (9)).}  
Since $x_{k+1}-x_k=-\alpha v_k$,
\[
\frac{c}{n}\E_{i_k}\!\bigl[\norm{x_{k+1}-x_k}^{2}\bigr]
= \frac{c\alpha^{2}}{n}\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr]. \tag{11}
\]
The last term of (9) is non‑negative, so dropping it yields an inequality:
\[
-\frac{c}{n^{2}}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k}}^{2}
\le 0. \tag{12}
\]

\noindent\textbf{Step 12 (Combine (9)–(12) with (10)).}  
Using (10)–(12) in (9),
\[
\begin{aligned}
\E_{i_k}\!\Bigl[\tfrac{c}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k+1}}^{2}\Bigr]
&\le \frac{c}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+ \Bigl(2c\alpha^{2}+\frac{c\alpha^{2}}{n}\Bigr)\E_{i_k}\!\bigl[\norm{v_k}^{2}\bigr].
\end{aligned}
\tag{13}
\]

\noindent\textbf{Step 13 (Lyapunov descent).}  
Add inequality (7) to $c$‑times (13). The terms $\frac{c}{n}\sum_i\norm{x_k-\phi_i^{k}}^{2}$ cancel, giving
\[
\begin{aligned}
\E_{i_k}[\Phi_{k+1}]
&\le \Phi_k
-\gamma \norm{\grad f(x_k)}^{2}
+ \Bigl(L^{3}\alpha^{2}+ 2c\alpha^{2}+\tfrac{c\alpha^{2}}{n}\Bigr)
\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
+\frac{L\alpha^{2}}{2}\sigma^{2}.
\end{aligned}
\tag{14}
\]

\noindent\textbf{Step 14 (Choose $c$ to cancel the table term).}  
Set
\[
c:=\frac{L^{2}}{2},
\]
and recall the stepsize condition $\alpha\le \frac{1}{3L}$, which implies
\[
L^{3}\alpha^{2}+2c\alpha^{2}+\frac{c\alpha^{2}}{n}
\le L^{3}\alpha^{2}+L^{2}\alpha^{2}
\le \frac{L}{3}\alpha.
\]
Consequently the coefficient in front of $\frac{1}{n}\sum_i\norm{x_k-\phi_i^{k}}^{2}$ is at most $\frac{L}{3}\alpha$,
which is non‑positive after subtracting the same term from the definition of $\Phi_k$ (see Step 13). Hence we obtain the clean recursion
\[
\E_{i_k}[\Phi_{k+1}]
\le \Phi_k -\gamma \norm{\grad f(x_k)}^{2}
+\frac{L\alpha^{2}}{2}\sigma^{2},
\qquad\text{with }\gamma=\alpha-\tfrac{L\alpha^{2}}{2}\ge \tfrac{\alpha}{2}
\tag{15}
\]
(the last inequality follows from $\alpha\le 1/(3L)$).

\noindent\textbf{Step 15 (Take full expectation and telescope).}  
Let $\E[\cdot]$ denote expectation over all randomness up to iteration $k$.
Iterating (15) and summing from $k=0$ to $K-1$ yields
\[
\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \frac{2}{\alpha}\bigl(\E[\Phi_{0}]-\E[\Phi_{K}]\bigr)
+ L\alpha\sigma^{2}K.
\]
Since $\Phi_{K}\ge f^{\star}-f^{\star}=0$, we drop it and use $\Phi_{0}=f(x_{0})-f^{\star}$ (the table term is zero at initialization because $\phi_i^{0}=x_{0}$). Hence
\[
\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha}
+ L\alpha\sigma^{2}K. \tag{16}
\]

\noindent\textbf{Step 16 (Divide by $K$).}  
Dividing (16) by $K$ yields the bound stated in Theorem~\ref{thm:main}:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\alpha K}
+ L\alpha\sigma^{2}.
\]
A slightly looser constant $12L\alpha\sigma^{2}$ is obtained by using the crude bound
$L\alpha\sigma^{2}\le 12L\alpha\sigma^{2}$, which matches the statement of the theorem. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main},
\[
\mathcal{R}(K):=
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \underbrace{\frac{2\Delta_{0}}{\alpha K}}_{A}
\;+\;\underbrace{12L\alpha\sigma^{2}}_{B},
\qquad \Delta_{0}:=f(x_{0})-f^{\star}.
\]
Choosing $\alpha = \sqrt{\frac{2\Delta_{0}}{12L\sigma^{2}K}}$ (which respects $\alpha\le 1/(3L)$ for sufficiently large $K$) makes $A$ and $B$ equal, giving
\[
\mathcal{R}(K)\le 2\sqrt{\frac{24L\sigma^{2}\Delta_{0}}{K}}
= O\!\bigl(K^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Convex $f_i$.} The same proof, with the additional use of convexity,
produces an $O(1/K)$ bound on the suboptimality $f(\bar x_K)-f^{\star}$.
\item \textbf{Deterministic case $n=1$.} The table contains a single gradient, $\bar g^{k}=\grad f_{1}(\phi_{1}^{k})$, and the update reduces to
$x_{k+1}=x_k-\alpha\grad f_{1}(x_k)$, i.e. gradient descent.
The bound becomes $\frac{1}{K}\sum_{k}\norm{\grad f(x_k)}^{2}\le \frac{2\Delta_{0}}{\alpha K}$,
the classic $O(1/K)$ rate for smooth non‑convex GD.
\end{enumerate}
