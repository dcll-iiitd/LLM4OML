\section*{Stochastic Gradient Descent with Warm Restarts (SGDR)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be the (possibly non‑convex) objective function.
At iteration $t$ we have a parameter vector $w_t\in\mathbb{R}^{d}$ and a stochastic
gradient $g_t$ satisfying
\[
\mathbb{E}[g_t\mid w_t]=\nabla f(w_t), \qquad 
\mathbb{E}\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]\le\sigma^{2}.
\]

The update rule is the standard SGD step
\begin{equation}
\boxed{\,w_{t+1}=w_t-\eta_t g_t\,},
\label{eq:update}
\end{equation}
where the step size $\eta_t$ follows a cosine annealing schedule with periodic
warm restarts.  
Define a sequence of cycle lengths $\{T_i\}_{i\ge 0}$ by
\[
T_0=M,\qquad T_{i+1}= \gamma\, T_i,\qquad \gamma>1,
\]
where $M$ is the length of the first cycle and $\gamma$ the restart multiplier.
Let $c(t)=\max\{i\ge0: \sum_{j=0}^{i-1}T_j < t\}$ be the index of the current
cycle, and $s(t)=t-\sum_{j=0}^{c(t)-1}T_j$ the intra‑cycle iteration (so $0\le s(t)<T_{c(t)}$).
The learning rate at iteration $t$ is
\begin{equation}
\boxed{\;
\eta_t = \eta_{\max}\,
\frac{1}{2}\Bigl(1+\cos\Bigl(\pi\frac{s(t)}{T_{c(t)}}\Bigr)\Bigr)
\;},
\label{eq:lr}
\end{equation}
with $\eta_{\max}>0$ the maximal step size.  
At the end of each cycle ($s(t)=T_{c(t)}$) the schedule ``restarts'' and $\eta_t$
jumps back to $\eta_{\max}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0$, maximal step size $\eta_{\max}$,
first cycle length $M$, restart multiplier $\gamma>1$, total iterations $T$.
\State Initialize $t\gets 0$, $c\gets 0$, $T_c\gets M$, $s\gets 0$.
\While{$t<T$}
    \State Sample a mini‑batch and compute stochastic gradient $g_t$.
    \State Compute learning rate 
    \[
    \eta_t \gets \eta_{\max}\,\frac{1}{2}\Bigl(1+\cos\bigl(\pi s/T_c\bigr)\Bigr).
    \]
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
    \State $t\gets t+1$, $s\gets s+1$.
    \If{$s = T_c$}
        \State \textit{restart}: $c\gets c+1$, $T_c\gets \gamma\,T_c$, $s\gets 0$.
    \EndIf
\EndWhile
\State \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or any averaging scheme).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We use a deterministic gradient (i.e. $g_t=\nabla f(w_t)$) and set
$\eta_{\max}=0.1$, $M=3$, $\gamma=2$ (so the first cycle has $T_0=3$ steps).
Because $T_0=3$, the cosine schedule yields the same learning rate for all
three steps:
\[
\eta_0=\eta_1=\eta_2=\eta_{\max}\frac{1}{2}\bigl(1+\cos 0\bigr)=0.1.
\]

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3665$
\end{tabular}
\end{center}
After three iterations the objective has decreased from $9$ to about $2.37$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\mathbb{R}^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Lower Bounded]\label{as:lb}
$f$ is bounded below: $f^{\star}\triangleq\inf_{w}f(w)>-\infty$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For each $t$, $\mathbb{E}[g_t\mid w_t]=\nabla f(w_t)$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}\ge0$ such that
\[
\mathbb{E}\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]\le\sigma^{2},
\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Learning‑rate Schedule]\label{as:schedule}
The step sizes $\{\eta_t\}$ are generated by \eqref{eq:lr} with parameters
$\eta_{\max}>0$, $M\in\mathbb{N}$ and $\gamma>1$.  Moreover,
\[
0<\eta_t\le\eta_{\max},\qquad 
\sum_{t=0}^{\infty}\eta_t=\infty,\qquad 
\sum_{t=0}^{\infty}\eta_t^{2}<\infty.
\tag{A5}
\]
(The last two relations hold because each cosine segment is bounded below by
a positive constant on a fixed fraction of its length and $T_i$ grows
geometrically.)
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR]\label{thm:sgdr}
Assume \ref{as:smooth}–\ref{as:schedule}.  Let $\{w_t\}$ be generated by
\eqref{eq:update}–\eqref{eq:lr} for $T$ iterations and define
\[
S_T\triangleq\sum_{t=0}^{T-1}\eta_t,\qquad
Q_T\triangleq\sum_{t=0}^{T-1}\eta_t^{2}.
\]
Then
\begin{equation}
\min_{0\le t<T}\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\frac{2\big(f(w_0)-f^{\star}\big)}{S_T}
\;+\;
\frac{L\sigma^{2} Q_T}{S_T}.
\label{eq:mainbound}
\end{equation}
Consequently, because of (A5) we have $S_T=\Theta(T)$ and $Q_T=O(1)$,
which yields the $O(1/T)$ rate for the squared gradient norm:
\[
\min_{0\le t<T}\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}
\begin{remark}
The bound \eqref{eq:mainbound} is identical to the classical result for
SGD with a diminishing stepsize, showing that the cosine‑annealing-with‑restarts
schedule does not deteriorate the worst‑case convergence guarantee.
\end{remark}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Since $\eta_t\le\eta_{\max}$ and $L$‑smoothness
implies a quadratic upper bound on $f$, the iterates remain in a bounded
region as long as $\eta_{\max}<2/L$.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item $\eta_{\max}$ controls the overall magnitude of the updates;
          choosing $\eta_{\max}>2/L$ may violate the descent condition.
        \item $M$ determines how many iterations a single cosine decay spans.
          Small $M$ yields frequent restarts, which empirically can help escape
          shallow basins but does not affect the asymptotic $O(1/T)$ rate.
        \item $\gamma$ governs the growth of cycle lengths; larger $\gamma$
          leads to longer periods and a slower decay of $S_T$, still preserving
          $S_T=\Theta(T)$.
    \end{itemize}
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Constant step SGD.}  With a fixed $\eta$, only a
          $O(1/\sqrt{T})$ bound on $\mathbb{E}[\|\nabla f\|]$ can be proved.
        \item \emph{Polynomial decay.}  SGDR matches the $O(1/T)$ guarantee while
          often achieving better practical performance because restarts
          re‑inject larger step sizes.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that despite the non‑monotone learning‑rate schedule,
the accumulated step sizes grow linearly and the accumulated squares stay
bounded, which is sufficient for the standard SGD analysis.  Therefore,
SGDR inherits the same worst‑case convergence guarantees as SGD with a
classical diminishing schedule, while its periodic restarts can improve
empirical convergence on non‑convex landscapes (e.g. deep neural networks).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Definition \ref{as:smooth} with $y=x-d$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:sgdr}.  All expectations are taken with respect to the
history up to time $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply Lemma \ref{lem:descent} with $d=\eta_t g_t$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle
+\frac{L}{2}\eta_t^{2}\|g_t\|^{2}.
\tag{1}
\]

\item[Step 2.] Take conditional expectation $\mathbb{E}[\cdot\mid w_t]$.
Using Assumption \ref{as:unbiased} and the variance bound,
\[
\mathbb{E}\!\big[\langle\nabla f(w_t),g_t\rangle\mid w_t\big]
= \|\nabla f(w_t)\|^{2},
\qquad
\mathbb{E}\!\big[\|g_t\|^{2}\mid w_t\big]
= \|\nabla f(w_t)\|^{2}+\mathbb{E}\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]
\le \|\nabla f(w_t)\|^{2}+\sigma^{2}.
\tag{2}
\]

\item[Step 3.] Substitute (2) into (1) and then take full expectation.
\[
\mathbb{E}[f(w_{t+1})]\le
\mathbb{E}[f(w_t)]-\eta_t\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
+\frac{L}{2}\eta_t^{2}\Big(\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]+\sigma^{2}\Big).
\tag{3}
\]

\item[Step 4.] Rearrange (3) to isolate the gradient term:
\[
\eta_t\Big(1-\frac{L}{2}\eta_t\Big)\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\mathbb{E}[f(w_t)]-\mathbb{E}[f(w_{t+1})]
+\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{4}
\]

\item[Step 5.] Because $\eta_t\le\eta_{\max}<2/L$ (a standard
restriction on the maximal stepsize), the factor
$1-\frac{L}{2}\eta_t$ is positive.  Define
\[
\alpha_t\triangleq\eta_t\Big(1-\frac{L}{2}\eta_t\Big)\ge\frac{\eta_t}{2},
\]
where the inequality follows from $\eta_t\le\eta_{\max}\le\frac{1}{L}$.
Then (4) yields
\[
\frac{\eta_t}{2}\,\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\mathbb{E}[f(w_t)]-\mathbb{E}[f(w_{t+1})]
+\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{5}
\]

\item[Step 6.] Sum (5) over $t=0,\dots,T-1$.
The telescoping sum on the right gives
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_t\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
f(w_0)-\mathbb{E}[f(w_T)]
+\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{6}
\]

\item[Step 7.] Drop the non‑negative term $-\mathbb{E}[f(w_T)]\le -f^{\star}$,
and multiply both sides of (6) by $2$:
\[
\sum_{t=0}^{T-1}\eta_t\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
2\big(f(w_0)-f^{\star}\big)
+L\sigma^{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{7}
\]

\item[Step 8.] Divide (7) by $S_T\triangleq\sum_{t=0}^{T-1}\eta_t$.
Since the left‑hand side is a convex combination of the terms
$\mathbb{E}[\|\nabla f(w_t)\|^{2}]$, we obtain
\[
\min_{0\le t<T}\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{\star}\big)}{S_T}
+\frac{L\sigma^{2}\,Q_T}{S_T},
\tag{8}
\]
where $Q_T=\sum_{t=0}^{T-1}\eta_t^{2}$.  This is exactly inequality \eqref{eq:mainbound}.
\end{enumerate}
Thus Theorem \ref{thm:sgdr} follows.

\section*{Rate Derivation (With Constants)}
From the definition of the cosine schedule \eqref{eq:lr} we have for any
cycle of length $T_i$,
\[
\eta_t\ge \eta_{\max}\,\frac{1}{2}\Bigl(1+\cos\pi\Bigr)=0,
\]
and more usefully,
\[
\frac{1}{T_i}\sum_{t\in\text{cycle }i}\eta_t
= \frac{\eta_{\max}}{T_i}\int_{0}^{T_i}
\frac{1}{2}\Bigl(1+\cos\bigl(\pi s/T_i\bigr)\Bigr)\,ds
= \frac{\eta_{\max}}{2}.
\tag{9}
\]
Hence each cycle contributes at least $\frac{\eta_{\max}}{2}T_i$ to $S_T$.
Because $T_i$ grows geometrically ($T_i = M\gamma^{i}$) and the number of
completed cycles up to iteration $T$ is $O(\log_{\gamma} T)$, we obtain
\[
S_T \ge \frac{\eta_{\max}}{2}\sum_{i=0}^{\lfloor\log_{\gamma}(T/M)\rfloor}
M\gamma^{i}
= \Theta(\eta_{\max} T).
\tag{10}
\]
Similarly,
\[
\eta_t\le\eta_{\max}\;\Longrightarrow\;
Q_T\le\eta_{\max}^{2} T = O(T).
\tag{11}
\]
Plugging (10)–(11) into \eqref{eq:mainbound} yields
\[
\min_{t<T}\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{C_{1}}{T}+\frac{C_{2}}{T},
\qquad
C_{1}= \frac{2\big(f(w_0)-f^{\star}\big)}{\eta_{\max}/2},
\;
C_{2}= \frac{L\sigma^{2}\eta_{\max}}{2},
\]
i.e. an $O(1/T)$ convergence rate for the squared gradient norm.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No Restarts ($\gamma=1$).}  The schedule reduces to a single
cosine decay over $M$ steps followed by constant $\eta_{\max}=0$;
the sums $S_T$ and $Q_T$ still satisfy (10)–(11) as $T\to\infty$, so the same
rate holds.
\item \textbf{Very Large $\gamma$.}  Cycle lengths increase rapidly, making the
schedule asymptotically equivalent to a simple constant step size for a long
initial period.  The condition $\sum\eta_t^{2}<\infty$ may fail; in that case
the theorem requires an additional decay factor (e.g. multiply $\eta_{\max}$
by $1/\sqrt{i}$ after $i$ restarts) to restore summability.
\item \textbf{Deterministic Gradients ($\sigma=0$).}  The variance term disappears
and the bound simplifies to $\frac{2(f(w_0)-f^{\star})}{S_T}=O(1/T)$.
\end{itemize}
