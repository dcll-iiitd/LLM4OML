\section*{DP‑SGD with Gaussian Noise}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss over a dataset of size $n$.  
At iteration $t$ we sample a minibatch $\mathcal{B}_{t}\subset\{1,\dots ,n\}$ of size $b$ uniformly at random and compute the (un‑noised) stochastic gradient
\[
g^{\text{raw}}_{t}\;:=\;\frac{1}{b}\sum_{i\in\mathcal{B}_{t}}\nabla f_i(x_{t}),
\]
where $f_i$ is the loss contributed by example $i$.  
We then add isotropic Gaussian noise
\[
\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d}),
\qquad 
g_{t}\;:=\;g^{\text{raw}}_{t}+\xi_{t},
\]
and perform a gradient step with learning rate $\eta_{t}>0$:
\begin{equation}
\boxed{\,x_{t+1}=x_{t}-\eta_{t} g_{t}\,}.
\label{eq:update}
\end{equation}
The sequence $\{x_{t}\}_{t=0}^{T}$ is the output of the algorithm.  
The privacy accountant (e.g. moments accountant) uses the noise scale $\sigma$,
the sampling probability $q=b/n$, the number of steps $T$ and a target $\delta$ to compute the overall $(\varepsilon,\delta)$‑DP guarantee; this part is orthogonal to the convergence analysis and is only mentioned for completeness.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, learning‑rate schedule $\{\eta_{t}\}_{t\ge0}$, minibatch size $b$, noise scale $\sigma$, number of iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a minibatch $\mathcal{B}_{t}\subset\{1,\dots ,n\}$, $|\mathcal{B}_{t}|=b$ uniformly.
    \State Compute raw stochastic gradient 
    \[
    g^{\text{raw}}_{t}\gets\frac{1}{b}\sum_{i\in\mathcal{B}_{t}}\nabla f_i(x_{t}) .
    \]
    \State Sample noise $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$.
    \State Form noisy gradient $g_{t}\gets g^{\text{raw}}_{t}+\xi_{t}$.
    \State Update $x_{t+1}\gets x_{t}-\eta_{t} g_{t}$.
\EndFor
\State \textbf{Output:} $x_{T}$ (or the average $\bar x_{T}=\frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$.  
Its gradient is $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, constant step size $\eta=0.1$, noise variance $\sigma^{2}=0.01$, and deterministic ``minibatch’’ (so $g^{\text{raw}}_{t}=\nabla f(w_{t})$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g^{\text{raw}}_{t}$ & $g_{t}=g^{\text{raw}}_{t}+\xi_{t}$ & $w_{t+1}=w_{t}-\eta g_{t}$\\ \hline
0 & $2(0-3)=-6$ & $-6+0.05=-5.95$ & $0-0.1(-5.95)=0.595$\\
1 & $2(0.595-3)=-4.81$ & $-4.81-0.02=-4.83$ & $0.595-0.1(-4.83)=1.078$\\
2 & $2(1.078-3)=-3.844$ & $-3.844+0.01=-3.834$ & $1.078-0.1(-3.834)=1.462$\\
\end{tabular}
\end{center}
After three iterations the objective values are
$f(0.595)=5.78$, $f(1.078)=3.69$, $f(1.462)=2.38$, showing monotone decrease in expectation.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Stochastic Gradient Variance]\label{ass:var}
For any $x$, the raw minibatch gradient satisfies
\[
\E_{\mathcal{B}}\!\left[\norm{g^{\text{raw}}(x)-\nabla f(x)}^{2}\right]\le \zeta^{2},
\]
where the expectation is over the random minibatch.
\end{assumption}
\begin{assumption}[Noise Independence]\label{ass:noise}
The Gaussian noise $\xi_{t}$ is independent of the minibatch and of all past randomness, with
\[
\E[\xi_{t}]=0,\qquad \E[\norm{\xi_{t}}^{2}]=d\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Learning‑rate]\label{ass:lr}
The stepsizes are non‑increasing and satisfy
\[
0<\eta_{t}\le \frac{1}{L},\qquad \sum_{t=0}^{T-1}\eta_{t}= \Theta(\sqrt{T}),\qquad 
\sum_{t=0}^{T-1}\eta_{t}^{2}=O(1).
\]
A convenient choice is $\eta_{t}=\frac{c}{\sqrt{T}}$ with $c\le 1/L$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient‑Norm Convergence]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:lr}, the iterates produced by \eqref{eq:update} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta_{\min} T}
\;+\;
L\bigl(\zeta^{2}+d\sigma^{2}\bigr)\,\frac{\sum_{t=0}^{T-1}\eta_{t}}{T},
\]
where $f^{\star}=\inf_{x}f(x)$ and $\eta_{\min}:=\min_{t}\eta_{t}$.  
Choosing the constant stepsize $\eta_{t}=c/\sqrt{T}$ with $c\le1/L$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The presence of Gaussian noise does not destabilise the algorithm as long as $\eta_{t}\le 1/L$; the extra variance term $d\sigma^{2}$ appears linearly in the bound.
\item \textbf{Hyper‑parameter Sensitivity.} Larger minibatch $b$ reduces $\zeta^{2}$ (variance of the raw gradient) while larger $\sigma$ (required for stronger privacy) inflates the second term. The stepsize $c$ balances the two terms.
\item \textbf{Comparison to Non‑Private SGD.} Setting $\sigma=0$ recovers the classical $O(1/\sqrt{T})$ bound for non‑convex smooth SGD.
\item \textbf{Special Cases.} If $f$ is convex, the same proof yields the standard $O(1/\sqrt{T})$ bound on suboptimality; if $f$ satisfies the Polyak–Łojasiewicz condition, the rate improves to $O(1/T)$ with the same noise penalty.
\end{itemize}

\section*{Discussion}
The theorem shows that adding isotropic Gaussian noise for differential privacy merely adds an additive variance term to the usual SGD convergence bound. By scaling the learning rate as $1/\sqrt{T}$ the algorithm attains the optimal $O(1/\sqrt{T})$ decay of the expected gradient norm, matching the best known rates for private non‑convex optimisation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x$ and any vector $u$,
\[
f(x-u)\le f(x)-\inner{\nabla f(x)}{u}+\frac{L}{2}\norm{u}^{2}.
\]
\end{lemma}
\begin{proof}
Immediate from Assumption \ref{ass:smooth} with $y=x-u$.
\end{proof}

\begin{lemma}[Bound on Noisy Gradient Second Moment]\label{lem:noise}
For the noisy gradient $g_{t}=g^{\text{raw}}_{t}+\xi_{t}$,
\[
\E\!\left[\norm{g_{t}}^{2}\mid x_{t}\right]
\le 2\bigl(\norm{\nabla f(x_{t})}^{2}+\zeta^{2}\bigr)+2d\sigma^{2}.
\]
\end{lemma}
\begin{proof}
Condition on $x_{t}$. Write $g_{t}= \nabla f(x_{t})+(g^{\text{raw}}_{t}-\nabla f(x_{t}))+\xi_{t}$. Using $\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$ twice,
\[
\|g_{t}\|^{2}\le 2\|\nabla f(x_{t})\|^{2}+2\|g^{\text{raw}}_{t}-\nabla f(x_{t})+\xi_{t}\|^{2}.
\]
Again $\|u+v\|^{2}\le2\|u\|^{2}+2\|v\|^{2}$ with $u=g^{\text{raw}}_{t}-\nabla f(x_{t})$, $v=\xi_{t}$. Taking expectation and using Assumptions \ref{ass:var} and \ref{ass:noise} yields the claimed bound.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout, expectations are taken over all randomness up to iteration $t$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $u=\eta_{t}g_{t}$:
\[
f(x_{t+1})\le f(x_{t})-\eta_{t}\inner{\nabla f(x_{t})}{g_{t}}+\frac{L\eta_{t}^{2}}{2}\norm{g_{t}}^{2}.
\tag{1}
\]

\noindent\textbf{[Step 2]} Take conditional expectation $\E[\cdot\mid x_{t}]$ on both sides of (1).  
Because $\E[\xi_{t}\mid x_{t}]=0$ and the minibatch is independent of $\xi_{t}$,
\[
\E\!\left[\inner{\nabla f(x_{t})}{g_{t}}\mid x_{t}\right]
=\inner{\nabla f(x_{t})}{\E[g^{\text{raw}}_{t}\mid x_{t}]}+\inner{\nabla f(x_{t})}{\E[\xi_{t}\mid x_{t}]}
=\norm{\nabla f(x_{t})}^{2}.
\tag{2}
\]

\noindent\textbf{[Step 3]} Use Lemma \ref{lem:noise} to bound the second moment:
\[
\E\!\left[\norm{g_{t}}^{2}\mid x_{t}\right]
\le 2\bigl(\norm{\nabla f(x_{t})}^{2}+\zeta^{2}\bigr)+2d\sigma^{2}.
\tag{3}
\]

\noindent\textbf{[Step 4]} Substitute (2) and (3) into (1) and then take full expectation:
\[
\begin{aligned}
\E[f(x_{t+1})]
&\le \E[f(x_{t})]
-\eta_{t}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
+\frac{L\eta_{t}^{2}}{2}\E\!\left[2\bigl(\norm{\nabla f(x_{t})}^{2}+\zeta^{2}\bigr)+2d\sigma^{2}\right]\\
&= \E[f(x_{t})]
-\eta_{t}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
+L\eta_{t}^{2}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
+L\eta_{t}^{2}\bigl(\zeta^{2}+d\sigma^{2}\bigr).
\end{aligned}
\tag{4}
\]

\noindent\textbf{[Step 5]} Rearrange (4) to isolate the gradient term:
\[
\bigl(\eta_{t}-L\eta_{t}^{2}\bigr)\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le \E[f(x_{t})]-\E[f(x_{t+1})]
+L\eta_{t}^{2}\bigl(\zeta^{2}+d\sigma^{2}\bigr).
\tag{5}
\]

\noindent\textbf{[Step 6]} By Assumption \ref{ass:lr}, $\eta_{t}\le 1/L$, hence $\eta_{t}-L\eta_{t}^{2}\ge \eta_{t}/2$. Therefore,
\[
\frac{\eta_{t}}{2}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le \E[f(x_{t})]-\E[f(x_{t+1})]
+L\eta_{t}^{2}\bigl(\zeta^{2}+d\sigma^{2}\bigr).
\tag{6}
\]

\noindent\textbf{[Step 7]} Sum inequality (6) over $t=0,\dots ,T-1$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_{t}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le f(x_{0})- \E[f(x_{T})]
+L\bigl(\zeta^{2}+d\sigma^{2}\bigr)\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 8]} Since $f(x_{T})\ge f^{\star}$,
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_{t}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le f(x_{0})-f^{\star}
+L\bigl(\zeta^{2}+d\sigma^{2}\bigr)\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 9]} Divide both sides of (8) by $\sum_{t=0}^{T-1}\eta_{t}$ and use $\eta_{\min}:=\min_{t}\eta_{t}$ to obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta_{\min}T}
+L\bigl(\zeta^{2}+d\sigma^{2}\bigr)\,\frac{\sum_{t=0}^{T-1}\eta_{t}}{T},
\]
which is exactly the bound stated in Theorem \ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
Choose a constant stepsize $\eta_{t}=c/\sqrt{T}$ with $c\le 1/L$. Then
\[
\eta_{\min}=c/\sqrt{T},\qquad
\sum_{t=0}^{T-1}\eta_{t}=c\sqrt{T},\qquad
\sum_{t=0}^{T-1}\eta_{t}^{2}=c^{2}.
\]
Plugging into the bound of Theorem \ref{thm:main} yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
\le
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{c}\frac{1}{\sqrt{T}}
+Lc\bigl(\zeta^{2}+d\sigma^{2}\bigr)\frac{1}{\sqrt{T}}
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No privacy noise ($\sigma=0$).} The bound reduces to the standard non‑private SGD result.
\item \textbf{Full‑batch ($b=n$) and $\zeta=0$.} Only the Gaussian term remains, giving
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_{t})}^{2}\right]
=O\!\left(\frac{1}{\sqrt{T}}+ \frac{d\sigma^{2}}{\sqrt{T}}\right).
\]
\item \textbf{Polyak–Łojasiewicz (PL) condition.} If $\frac12\|\nabla f(x)\|^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr)$ for some $\mu>0$, the same steps give a linear recursion leading to
\[
\E\!\left[f(x_{T})-f^{\star}\right]\le (1-\mu\eta)^{T}\bigl(f(x_{0})-f^{\star}\bigr)+\frac{L(\zeta^{2}+d\sigma^{2})}{\mu},
\]
i.e. an $O(1/T)$ decay up to a noise floor.
\end{itemize}
