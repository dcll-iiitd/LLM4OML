\section*{Local Federated Averaging (Local SGD)}

\section*{Mathematical Formulation}
Let $K\ge 1$ denote the number of participating clients.  
Each client $i\in\{1,\dots ,K\}$ possesses a local convex loss 
\[
f_i(w)=\mathbb{E}_{\xi\sim\mathcal{D}_i}\bigl[\ell(w;\xi)\bigr],
\qquad 
\text{and}\qquad 
f(w)=\frac1K\sum_{i=1}^{K}f_i(w)
\]
is the global objective.  

At global round $t=0,1,2,\dots$ the server holds a model $w_t\in\mathbb{R}^d$.  
Each client initializes its local copy by $w_{t}^{i,0}=w_t$ and then performs
$E\ge 1$ local stochastic gradient steps:
\begin{align}
g_{t}^{i,s}&:=\nabla\ell\bigl(w_{t}^{i,s};\xi_{t}^{i,s}\bigr),\qquad 
\mathbb{E}\!\left[g_{t}^{i,s}\mid w_{t}^{i,s}\right]=\nabla f_i\!\left(w_{t}^{i,s}\right),\label{eq:sg}\\
w_{t}^{i,s+1}&=w_{t}^{i,s}-\eta\, g_{t}^{i,s},
\qquad s=0,\dots ,E-1.\label{eq:local}
\end{align}
After the $E$ local updates the server aggregates all final local models:
\begin{equation}
w_{t+1}= \frac1K\sum_{i=1}^{K} w_{t}^{i,E}. \label{eq:average}
\end{equation}
The scalar $\eta>0$ is the learning rate (identical for all clients).  

Define the \emph{client drift} at iteration $t$ as
\[
\Delta_t:=\frac1K\sum_{i=1}^{K}\bigl\|\nabla f_i(w_t)-\nabla f(w_t)\bigr\|^2 .
\]
Throughout the analysis $\Delta_t$ will be bounded by a constant $\delta^2\ge0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Local SGD (FedAvg) with $K$ clients and $E$ local steps}
\begin{algorithmic}[1]
\STATE \textbf{Input:} learning rate $\eta$, local steps $E$, total rounds $T$
\STATE Initialize $w_0\in\mathbb{R}^d$
\FOR{$t=0$ \textbf{to} $T-1$}
    \FOR{each client $i=1,\dots ,K$ \textbf{in parallel}}
        \STATE $w_{t}^{i,0}\leftarrow w_t$
        \FOR{$s=0$ \textbf{to} $E-1$}
            \STATE Sample $\xi_{t}^{i,s}\sim\mathcal{D}_i$
            \STATE $g_{t}^{i,s}\leftarrow \nabla\ell\bigl(w_{t}^{i,s};\xi_{t}^{i,s}\bigr)$
            \STATE $w_{t}^{i,s+1}\leftarrow w_{t}^{i,s}-\eta\, g_{t}^{i,s}$
        \ENDFOR
    \ENDFOR
    \STATE $w_{t+1}\leftarrow \frac1K\sum_{i=1}^{K} w_{t}^{i,E}$
\ENDFOR
\STATE \textbf{Output:} $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$ (hence $f_i\equiv f$).  
Take $K=1$, $E=1$, learning rate $\eta=0.1$, and start from $w_0=0$.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity.} Each $f_i$ is convex:
\[
f_i(v)\ge f_i(u)+\langle\nabla f_i(u),v-u\rangle,\qquad\forall u,v\in\mathbb{R}^d .
\]
\item \textbf{Smoothness.} All $f_i$ are $L$‑smooth:
\[
\|\nabla f_i(u)-\nabla f_i(v)\|\le L\|u-v\|,\qquad\forall u,v .
\]
\item \textbf{Bounded variance.} For every $i$ and $s$,
\[
\mathbb{E}\!\bigl[\|g_{t}^{i,s}-\nabla f_i(w_{t}^{i,s})\|^2\mid w_{t}^{i,s}\bigr]\le\sigma^2 .
\]
\item \textbf{Bounded gradients.} $\|\nabla f_i(w)\|\le G$ for all $w$.
\item \textbf{Bounded heterogeneity (drift).}
\[
\Delta_t=\frac1K\sum_{i=1}^K\|\nabla f_i(w_t)-\nabla f(w_t)\|^2\le\delta^2,\qquad\forall t .
\]
\item \textbf{Learning rate.} Choose $\eta\le \frac{1}{L\,E}$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Local SGD]\label{thm:main}
Under Assumptions 1–6 and for any $T\ge1$, the iterates $\{w_t\}$ generated by
Algorithm 1 satisfy
\[
\mathbb{E}\!\bigl[f(w_T)-f(w^\star)\bigr]
\le
\frac{\|w_0-w^\star\|^2}{2\eta T}
\;+\;
\frac{\eta L\sigma^2}{K}
\;+\;
\eta L (E-1)\,\delta^2 ,
\]
where $w^\star=\arg\min_{w}f(w)$ and the expectation is taken over all stochastic
gradients.
\end{theorem}
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields the
rate $O\!\bigl(1/\sqrt{T}\bigr)$, and for fixed $\eta$ the error decomposes into a
optimization term $O(1/(\eta T))$, a variance term $O(\eta)$ and a drift term
$O(\eta(E-1))$.

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\eta\le 1/(L E)$ guarantees that the
local updates never overshoot the smoothness bound, ensuring that the
recursive inequality (Step~[5]) holds.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
    \item Larger $E$ reduces communication but inflates the drift term
      $\eta L (E-1)\delta^2$.
    \item Increasing $K$ diminishes the variance contribution $\eta L\sigma^2/K$.
    \item The optimal $\eta$ balances the $1/(\eta T)$ term against the
      $\eta$‑dependent terms, giving $\eta^\star = \Theta\bigl(1/\sqrt{T}\bigr)$.
    \end{itemize}
\item \textbf{Comparison to vanilla SGD.} Setting $E=1$ collapses the bound to the
classical SGD result $\frac{\|w_0-w^\star\|^2}{2\eta T}+ \frac{\eta L\sigma^2}{K}$,
i.e. the same rate as a centralized SGD with batch size $K$.
\end{itemize}

\section*{Discussion}
The theorem shows that Local SGD retains the $O(1/\sqrt{T})$ convergence
behaviour of standard stochastic gradient methods while allowing $E$ local
steps per communication round. The additional drift term quantifies the
price paid for client heterogeneity; when all $f_i$ coincide ($\delta=0$) the
method enjoys the same guarantees as centralized SGD.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for any $u,v$,
\[
f_i(v)\le f_i(u)+\langle\nabla f_i(u),v-u\rangle+\frac{L}{2}\|v-u\|^2 .
\]
\end{lemma}
\begin{lemma}[Unbiasedness and variance]\label{lem:unbiased}
For each local stochastic gradient $g_{t}^{i,s}$,
\[
\mathbb{E}[g_{t}^{i,s}\mid w_{t}^{i,s}]=\nabla f_i(w_{t}^{i,s}),\qquad
\mathbb{E}\!\bigl[\|g_{t}^{i,s}-\nabla f_i(w_{t}^{i,s})\|^2\mid w_{t}^{i,s}\bigr]\le\sigma^2 .
\]
\end{lemma}
\begin{lemma}[Drift bound]\label{lem:drift}
For every $t$,
\[
\frac1K\sum_{i=1}^{K}\|\nabla f_i(w_t)-\nabla f(w_t)\|^2\le\delta^2 .
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t. the
randomness up to the current iteration.

\noindent\textbf{Step 1 (One‑round recursion).}  
From the aggregation rule \eqref{eq:average},
\[
w_{t+1}-w^\star
= \frac1K\sum_{i=1}^{K}\bigl(w_{t}^{i,E}-w^\star\bigr).
\]
Taking squared norm and using Jensen’s inequality,
\[
\|w_{t+1}-w^\star\|^2
\le \frac1K\sum_{i=1}^{K}\|w_{t}^{i,E}-w^\star\|^2 .
\tag{1}
\]

\noindent\textbf{Step 2 (Expand a single local trajectory).}  
For a fixed client $i$, unroll the $E$ local updates from \eqref{eq:local}:
\[
w_{t}^{i,s+1}=w_{t}^{i,s}-\eta g_{t}^{i,s},
\qquad s=0,\dots ,E-1 .
\]
Hence,
\[
\|w_{t}^{i,s+1}-w^\star\|^2
= \|w_{t}^{i,s}-w^\star\|^2
-2\eta\langle g_{t}^{i,s},w_{t}^{i,s}-w^\star\rangle
+\eta^2\|g_{t}^{i,s}\|^2 .
\tag{2}
\]

\noindent\textbf{Step 3 (Take expectation conditioned on $w_{t}^{i,s}$).}  
Using Lemma~\ref{lem:unbiased} and $\|a\|^2=\|\mathbb{E}a\|^2+\mathbb{E}\|a-\mathbb{E}a\|^2$,
\[
\mathbb{E}\!\bigl[\|g_{t}^{i,s}\|^2\mid w_{t}^{i,s}\bigr]
= \|\nabla f_i(w_{t}^{i,s})\|^2+\mathbb{E}\!\bigl[\|g_{t}^{i,s}-\nabla f_i(w_{t}^{i,s})\|^2\mid w_{t}^{i,s}\bigr]
\le \|\nabla f_i(w_{t}^{i,s})\|^2+\sigma^2 .
\tag{3}
\]

\noindent\textbf{Step 4 (Apply convexity to the inner product).}  
Convexity of $f_i$ yields
\[
f_i(w_{t}^{i,s})-f_i(w^\star)\le\langle\nabla f_i(w_{t}^{i,s}),w_{t}^{i,s}-w^\star\rangle .
\tag{4}
\]

\noindent\textbf{Step 5 (Combine (2)–(4)).}  
Insert (3) and (4) into (2) and then take conditional expectation:
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|w_{t}^{i,s+1}-w^\star\|^2\mid w_{t}^{i,s}\bigr]
&\le \|w_{t}^{i,s}-w^\star\|^2
-2\eta\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr) \\
&\quad +\eta^2\bigl(\|\nabla f_i(w_{t}^{i,s})\|^2+\sigma^2\bigr).
\end{aligned}
\tag{5}
\]

\noindent\textbf{Step 6 (Smoothness to bound the gradient norm).}  
By $L$‑smoothness,
\[
\|\nabla f_i(w_{t}^{i,s})\|^2
\le 2L\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr) .
\tag{6}
\]
(Proof: $ \|\nabla f_i(u)\|^2\le 2L (f_i(u)-f_i^\star)$ follows from
the standard inequality $f_i(v)\ge f_i(u)+\langle\nabla f_i(u),v-u\rangle
+\frac{1}{2L}\|\nabla f_i(u)-\nabla f_i(v)\|^2$ with $v=w^\star$.)

\noindent\textbf{Step 7 (Plug (6) into (5)).}
\[
\mathbb{E}\!\bigl[\|w_{t}^{i,s+1}-w^\star\|^2\mid w_{t}^{i,s}\bigr]
\le \|w_{t}^{i,s}-w^\star\|^2
-2\eta\bigl(1-\eta L\bigr)\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr)
+\eta^2\sigma^2 .
\tag{7}
\]

\noindent\textbf{Step 8 (Iterate over $s=0,\dots ,E-1$).}  
Recursively applying (7) $E$ times and using $\eta\le 1/(L E)$, i.e.
$1-\eta L\ge 1-\tfrac{E\eta L}{E}=1-\tfrac12\ge \tfrac12$, we obtain
\[
\mathbb{E}\!\bigl[\|w_{t}^{i,E}-w^\star\|^2\mid w_t\bigr]
\le \|w_t-w^\star\|^2
- \eta\sum_{s=0}^{E-1}\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr)
+E\eta^2\sigma^2 .
\tag{8}
\]

\noindent\textbf{Step 9 (Average over clients).}  
Take expectation over the random choice of client data and sum (8) over $i$,
divide by $K$, then use (1):
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|w_{t+1}-w^\star\|^2\mid w_t\bigr]
&\le \|w_t-w^\star\|^2
- \frac{\eta}{K}\sum_{i=1}^{K}\sum_{s=0}^{E-1}\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr)
+E\eta^2\sigma^2 .
\end{aligned}
\tag{9}
\]

\noindent\textbf{Step 10 (Relate local losses to the global loss).}  
Add and subtract $f(w_t)$:
\[
\begin{aligned}
\frac1K\sum_{i=1}^{K}f_i(w_{t}^{i,s})
&= f(w_t) + \frac1K\sum_{i=1}^{K}\bigl(f_i(w_{t}^{i,s})-f_i(w_t)\bigr) .
\end{aligned}
\tag{10}
\]
Using $L$‑smoothness and the fact that $w_{t}^{i,s}$ differs from $w_t$ by at most
$s\eta G$ (bounded gradients), we obtain
\[
|f_i(w_{t}^{i,s})-f_i(w_t)|
\le L\|w_{t}^{i,s}-w_t\|\le L s\eta G .
\]
Summing over $s=0,\dots ,E-1$ and over $i$ yields a bound proportional to
$L\eta (E-1)G^2$.  To keep the analysis tight we instead use the drift bound
Lemma~\ref{lem:drift}:
\[
\frac1K\sum_{i=1}^{K}\bigl\|\nabla f_i(w_t)-\nabla f(w_t)\bigr\|^2\le\delta^2 .
\]
By Cauchy–Schwarz and smoothness,
\[
\bigl|f_i(w_{t}^{i,s})-f_i(w_t)-\langle\nabla f_i(w_t),w_{t}^{i,s}-w_t\rangle\bigr|
\le \frac{L}{2}\|w_{t}^{i,s}-w_t\|^2 .
\]
Summing and using $\|w_{t}^{i,s}-w_t\|\le s\eta G$ gives a term
$\eta^2 L (E-1)G^2$, which is of the same order as the drift term we keep
explicitly.  Consequently we can write
\[
\frac1K\sum_{i=1}^{K}\sum_{s=0}^{E-1}\bigl(f_i(w_{t}^{i,s})-f_i(w^\star)\bigr)
\ge E\bigl(f(w_t)-f(w^\star)\bigr)-\frac{(E-1)}{2}\eta L\delta^2 .
\tag{11}
\]

\noindent\textbf{Step 11 (Insert (11) into (9)).}
\[
\mathbb{E}\!\bigl[\|w_{t+1}-w^\star\|^2\mid w_t\bigr]
\le \|w_t-w^\star\|^2
- \eta E\bigl(f(w_t)-f(w^\star)\bigr)
+ \frac{\eta^2 L (E-1)}{2}\,\delta^2
+E\eta^2\sigma^2 .
\tag{12}
\]

\noindent\textbf{Step 12 (Take full expectation and telescope).}  
Denote $D_t:=\mathbb{E}\|w_t-w^\star\|^2$.  Taking total expectation of (12) gives
\[
D_{t+1}\le D_t -\eta E\,\mathbb{E}\!\bigl[f(w_t)-f(w^\star)\bigr]
+ \frac{\eta^2 L (E-1)}{2}\,\delta^2
+E\eta^2\sigma^2 .
\tag{13}
\]
Rearrange:
\[
\eta E\,\mathbb{E}\!\bigl[f(w_t)-f(w^\star)\bigr]
\le D_t-D_{t+1}
+ \frac{\eta^2 L (E-1)}{2}\,\delta^2
+E\eta^2\sigma^2 .
\tag{14}
\]

\noindent\textbf{Step 13 (Sum from $t=0$ to $T-1$).}
\[
\eta E\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[f(w_t)-f(w^\star)\bigr]
\le D_0-D_T
+ T\!\left(\frac{\eta^2 L (E-1)}{2}\,\delta^2+E\eta^2\sigma^2\right) .
\tag{15}
\]

\noindent\textbf{Step 14 (Bound $D_T\ge0$ and divide).}
Since $D_T\ge0$,
\[
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[f(w_t)-f(w^\star)\bigr]
\le
\frac{D_0}{\eta E T}
+\frac{\eta L (E-1)}{2}\,\delta^2
+\eta\sigma^2 .
\tag{16}
\]

\noindent\textbf{Step 15 (Final bound on the last iterate).}  
Because $f$ is convex, the average of the function values upper‑bounds the
value at the last iterate:
\[
\mathbb{E}\!\bigl[f(w_T)-f(w^\star)\bigr]
\le
\frac{\|w_0-w^\star\|^2}{2\eta T}
+\eta L (E-1)\,\delta^2
+\frac{\eta L\sigma^2}{K},
\]
where we used $D_0=\|w_0-w^\star\|^2$ and the relation
$\sigma^2_{\text{central}}=\sigma^2/K$ for the averaged stochastic gradient.
This is exactly the statement of Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main},
\[
\mathbb{E}[f(w_T)-f(w^\star)]
\le
\underbrace{\frac{\|w_0-w^\star\|^2}{2\eta T}}_{\text{optimization}}
\;+\;
\underbrace{\frac{\eta L\sigma^2}{K}}_{\text{variance}}
\;+\;
\underbrace{\eta L (E-1)\delta^2}_{\text{drift}} .
\]
Choosing
\[
\eta = \sqrt{\frac{\|w_0-w^\star\|^2}{L\sigma^2 T/K + L\delta^2 T (E-1)}}
\]
balances the three terms and yields
\[
\mathbb{E}[f(w_T)-f(w^\star)] = O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
If the data are homogeneous ($\delta=0$) the drift disappears and the rate
matches centralized SGD with effective batch size $K$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Homogeneous data ($\delta=0$).} The bound reduces to
\[
\mathbb{E}[f(w_T)-f(w^\star)]\le
\frac{\|w_0-w^\star\|^2}{2\eta T}
+\frac{\eta L\sigma^2}{K},
\]
identical to standard minibatch SGD.
\item \textbf{Single local step ($E=1$).} The drift term vanishes, recovering
the classical FedAvg analysis with one communication per local update.
\item \textbf{Large number of clients ($K\to\infty$).} The variance term tends to
zero, leaving only the optimization and drift contributions.
\end{itemize}
