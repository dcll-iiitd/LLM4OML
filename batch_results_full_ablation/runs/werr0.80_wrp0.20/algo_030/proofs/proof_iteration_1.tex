\section*{Heavy‑Ball Method with Gradient Momentum}

The method updates the current iterate by adding a \emph{momentum} term that depends on the
difference of the two most recent iterates and then takes a gradient step.  
It is often called the \textbf{Heavy‑Ball (HB)} method.

---------------------------------------------------------------------

\section*{Mathematical Formulation}

Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.
Given an initial pair $(x_{0},x_{-1})\in\mathbb{R}^{d}\times\mathbb{R}^{d}$,
step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the iterates are
\[
\boxed{
x_{k+1}=x_{k}+\beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha \nabla f(x_{k}),\qquad k=0,1,2,\dots
}
\tag{HB}
\]

\noindent
The algorithm uses only the current gradient $\nabla f(x_k)$; no “look‑ahead’’ gradient is required.

---------------------------------------------------------------------

\section*{Pseudocode}

\begin{algorithm}[H]
\caption{Heavy‑Ball Method}
\begin{algorithmic}[1]
\REQUIRE Initial points $x_{0},x_{-1}\in\mathbb{R}^{d}$, step size $\alpha>0$, momentum $\beta\in[0,1)$, maximal iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \STATE Update momentum term $m_{k}\gets \beta\,(x_{k}-x_{k-1})$.
    \STATE $x_{k+1}\gets x_{k}+m_{k}-\alpha\,g_{k}$.
\ENDFOR
\RETURN $x_{K}$
\end{algorithmic}
\end{algorithm}

---------------------------------------------------------------------

\section*{Dry Run Example}

Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$.
\[
\nabla f(w)=2(w-3),\qquad L=2.
\]

Choose $\alpha=0.1$, $\beta=0.5$, and initialise $x_{-1}=0$, $x_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.} $k$ & $x_{k}$ & $\nabla f(x_{k})$ & $f(x_{k})$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$ \\[2mm]
1 & $x_{1}=0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ &
$2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$ \\[2mm]
2 & $x_{2}=0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ &
$2(1.38-3)=-3.24$ & $(1.38-3)^{2}=2.6244$ \\[2mm]
3 & $x_{3}=1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ &
$2(2.094-3)=-1.812$ & $(2.094-3)^{2}=0.8196$
\end{tabular}
\end{center}

The objective value decreases from $9$ to $0.8196$ after three iterations, illustrating the effect of momentum.

---------------------------------------------------------------------

\section*{Assumptions}

\begin{enumerate}[label={\bf A\arabic*}.]
\item \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A1}
\]

\item \textbf{Lower boundedness.} $f$ has a finite infimum $f^{\star}>-\infty$.
\tag{A2}

\item \textbf{Step‑size restriction.} The parameters satisfy
\[
0<\alpha\le\frac{1-\beta}{L},\qquad 0\le\beta<1.
\tag{A3}
\]

\item \textbf{Initialisation.} $x_{-1}=x_{0}$ is allowed; any pair $(x_{0},x_{-1})$ is admissible.
\tag{A4}
\end{enumerate}

---------------------------------------------------------------------

\section*{Main Convergence Theorem}

\begin{theorem}[Average Gradient Norm for Heavy‑Ball]\label{thm:hb}
Assume \textbf{A1}–\textbf{A4}.  Define the Lyapunov function
\[
\Phi_{k}\;:=\;f(x_{k})+\frac{L\beta}{2(1-\beta)}\|x_{k}-x_{k-1}\|^{2},\qquad k\ge0.
\tag{T1}
\]
Then for every $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}
\;\le\;
\frac{2\bigl(\Phi_{0}-f^{\star}\bigr)}{\alpha K}.
\tag{T2}
\]
Consequently $\displaystyle\lim_{K\to\infty}\|\nabla f(x_{K})\|=0$ and every limit point of $\{x_{k}\}$ is a stationary point of $f$.
\end{theorem}

---------------------------------------------------------------------

\section*{Theoretical Properties}

\begin{enumerate}[label={\bf P\arabic*}.]
\item \textbf{Stability.} Condition \textbf{A3} guarantees that the Lyapunov sequence
$\{\Phi_{k}\}$ is non‑increasing, which prevents the iterates from diverging even for non‑convex $f$.

\item \textbf{Hyper‑parameter sensitivity.} Larger $\beta$ accelerates progress when the
gradient direction does not change rapidly, but \textbf{A3} forces $\alpha$ to shrink as $\beta\to1$; otherwise the descent inequality (Step~[5]) may be violated.

\item \textbf{Comparison to Gradient Descent (GD).} Setting $\beta=0$ reduces (HB) to
$x_{k+1}=x_{k}-\alpha\nabla f(x_{k})$, and Theorem~\ref{thm:hb} recovers the standard GD bound
$\frac{2(f(x_{0})-f^{\star})}{\alpha K}$.

\item \textbf{Special case – Polyak–Łojasiewicz (PL) condition.} If $f$ satisfies
\[
\frac{1}{2}\|\nabla f(x)\|^{2}\ge\mu\bigl(f(x)-f^{\star}\bigr),\qquad\mu>0,
\tag{P4}
\]
then (T2) yields a linear rate
\[
f(x_{k})-f^{\star}\le\bigl(1-\alpha\mu\bigr)^{k}\bigl(f(x_{0})-f^{\star}\bigr).
\]
\end{enumerate}

---------------------------------------------------------------------

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for all $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}

\begin{lemma}[Three‑point identity]\label{lem:three}
For any vectors $a,b,c$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+\|b-c\|^{2}+2\langle a-b,b-c\rangle .
\tag{L2}
\]
\end{lemma}

---------------------------------------------------------------------

\section*{Main Proof (Step‑by‑Step Derivation)}

We prove Theorem~\ref{thm:hb}.  All non‑trivial manipulations are numbered.

\bigskip
\noindent\textbf{Step 1 (Apply smoothness).}  
Using Lemma~\ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{S1}
\]

\noindent\textbf{Step 2 (Insert the update (HB)).}  
Because $x_{k+1}-x_{k}= \beta(x_{k}-x_{k-1})-\alpha\nabla f(x_{k})$,
\[
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
= \beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
-\alpha\|\nabla f(x_{k})\|^{2}.
\tag{S2}
\]

\noindent\textbf{Step 3 (Expand the squared norm).}  
\[
\|x_{k+1}-x_{k}\|^{2}
= \|\beta(x_{k}-x_{k-1})-\alpha\nabla f(x_{k})\|^{2}
= \beta^{2}\|x_{k}-x_{k-1}\|^{2}
+\alpha^{2}\|\nabla f(x_{k})\|^{2}
-2\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle .
\tag{S3}
\]

\noindent\textbf{Step 4 (Plug (S2) and (S3) into (S1)).}  
\begin{align*}
f(x_{k+1})
&\le f(x_{k})
+\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
-\alpha\|\nabla f(x_{k})\|^{2}\\
&\quad+\frac{L}{2}\Bigl[\beta^{2}\|x_{k}-x_{k-1}\|^{2}
+\alpha^{2}\|\nabla f(x_{k})\|^{2}
-2\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle\Bigr].
\end{align*}
Collecting the inner‑product terms gives
\[
f(x_{k+1})\le f(x_{k})
-\alpha\!\Bigl(1-\frac{L\alpha}{2}\Bigr)\!\|\nabla f(x_{k})\|^{2}
+\beta\!\Bigl(1-L\alpha\Bigr)\!\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
+\frac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}.
\tag{S4}
\]

\noindent\textbf{Step 5 (Add a weighted momentum term).}  
Consider the Lyapunov function $\Phi_{k}$ defined in (T1).  Using Lemma~\ref{lem:three}
with $a=x_{k+1}$, $b=x_{k}$, $c=x_{k-1}$,
\[
\|x_{k+1}-x_{k-1}\|^{2}= \|x_{k+1}-x_{k}\|^{2}
+\|x_{k}-x_{k-1}\|^{2}+2\langle x_{k+1}-x_{k},x_{k}-x_{k-1}\rangle .
\tag{S5}
\]
Multiplying (S5) by $\frac{L\beta}{2(1-\beta)}$ and adding to (S4) yields
\[
\Phi_{k+1}\le \Phi_{k}
-\underbrace{\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)}_{\displaystyle c_{1}}\|\nabla f(x_{k})\|^{2}
-\underbrace{\frac{L\beta}{2(1-\beta)}\Bigl(1-\beta-\tfrac{L\alpha}{2}\Bigr)}_{\displaystyle c_{2}}\|x_{k}-x_{k-1}\|^{2}.
\tag{S6}
\]

\noindent\textbf{Step 6 (Choose $\alpha$ according to (A3)).}  
By (A3) we have $0<\alpha\le\frac{1-\beta}{L}$, which implies
\[
1-\frac{L\alpha}{2}\ge\frac{1}{2},\qquad
1-\beta-\frac{L\alpha}{2}\ge\frac{1-\beta}{2}>0.
\]
Hence $c_{1}\ge\frac{\alpha}{2}$ and $c_{2}\ge\frac{L\beta}{4(1-\beta)}\ge0$.
Discarding the non‑negative $c_{2}$ term from (S6) we obtain the clean descent inequality
\[
\Phi_{k+1}\le \Phi_{k}-\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}.
\tag{S7}
\]

\noindent\textbf{Step 7 (Telescoping sum).}  
Summing (S7) for $k=0,\dots,K-1$ gives
\[
\Phi_{K}\le \Phi_{0}-\frac{\alpha}{2}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}.
\tag{S8}
\]
Since $\Phi_{K}\ge f^{\star}$ by definition of $\Phi_{K}$ and (A2), we rearrange (S8):
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}
\le \frac{2\bigl(\Phi_{0}-f^{\star}\bigr)}{\alpha K}.
\tag{S9}
\]

\noindent\textbf{Step 8 (Conclusion).}  
Inequality (S9) is exactly (T2).  Because the right‑hand side tends to $0$ as $K\to\infty$,
the average squared gradient norm vanishes, which implies $\|\nabla f(x_{k})\|\to0$ along a subsequence.
Standard arguments (e.g., Bolzano–Weierstrass) then guarantee that every limit point of $\{x_{k}\}$
is a stationary point of $f$.  ∎

---------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}

From (S9) we may write explicitly
\[
\boxed{
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}
\le
\frac{2}{\alpha K}\Bigl(f(x_{0})-f^{\star}
+\tfrac{L\beta}{2(1-\beta)}\|x_{0}-x_{-1}\|^{2}\Bigr).
}
\]
Thus the Heavy‑Ball method enjoys an $O(1/K)$ decay of the \emph{average} squared gradient
norm under the sole smoothness assumption.

---------------------------------------------------------------------

\section*{Special Cases}

\begin{enumerate}[label={\bf S\arabic*}.]
\item \textbf{Pure Gradient Descent ($\beta=0$).}  
$\Phi_{k}=f(x_{k})$, $c_{2}=0$, and (S7) reduces to
$f(x_{k+1})\le f(x_{k})-\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}$,
the classical GD descent lemma.

\item \textbf{Heavy‑Ball with exact PL condition.}  
If $f$ satisfies (P4), then from (T2)
\[
\frac{1}{K}\sum_{k=0}^{K-1}2\mu\bigl(f(x_{k})-f^{\star}\bigr)
\le\frac{2(\Phi_{0}-f^{\star})}{\alpha K},
\]
which yields $f(x_{k})-f^{\star}\le(1-\alpha\mu)^{k}(f(x_{0})-f^{\star})$,
i.e., a linear convergence rate.

\item \textbf{Limit $\beta\to1^{-}$.}  
Condition (A3) forces $\alpha\to0$, so the method degenerates to a
very slow gradient descent; this illustrates the necessity of the
step‑size bound.
\end{enumerate}

---------------------------------------------------------------------

\section*{Discussion}

The Heavy‑Ball method with the gradient‑only momentum term enjoys a simple
Lyapunov‑based analysis that yields an $O(1/K)$ guarantee on the average
gradient norm for arbitrary non‑convex $L$‑smooth objectives.  The result
matches the best known rate for first‑order methods without variance
reduction.  The constants explicitly depend on the momentum $\beta$ through
the additional term $\frac{L\beta}{2(1-\beta)}\|x_{0}-x_{-1}\|^{2}$,
highlighting the trade‑off between acceleration (large $\beta$) and
required step size (small $\alpha$).  When stronger properties such as the
Polyak–Łojasiewicz inequality hold, the same analysis upgrades to a linear
convergence rate, demonstrating that the Heavy‑Ball method is robust across
a wide spectrum of smooth non‑convex problems.
