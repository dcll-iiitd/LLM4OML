\section*{Clipped Momentum Gradient Descent (CMGD)}
%=====================================================================

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
At iteration $t\ge 0$ we have a point $x_t\in\R^{d}$ and a stochastic
gradient estimator $\tilde g_t$ such that  

\[
\EE\!\left[\tilde g_t\mid x_t\right]=\nabla f(x_t),\qquad
\EE\!\left[\|\tilde g_t-\nabla f(x_t)\|^{2}\mid x_t\right]\le\sigma^{2}.
\]

\begin{enumerate}
\item \textbf{Gradient clipping}
\[
g_t\;:=\;\frac{\tilde g_t}{\max\!\Bigl(1,\;\frac{\|\tilde g_t\|}{C}\Bigr)}
      \;=\;
      \begin{cases}
      \tilde g_t, & \|\tilde g_t\|\le C,\\[4pt]
      \displaystyle C\;\frac{\tilde g_t}{\|\tilde g_t\|}, & \|\tilde g_t\|>C,
      \end{cases}
\tag{1}
\]
where $C>0$ is the clipping threshold. Consequently $\|g_t\|\le C$ a.s.

\item \textbf{Momentum update}
\[
m_t \;:=\;\beta\,m_{t-1}+(1-\beta)\,g_t,\qquad\beta\in[0,1),
\tag{2}
\]
with $m_{-1}=0$.

\item \textbf{Parameter update}
\[
x_{t+1}\;:=\;x_t-\eta\,m_t,\qquad\eta>0\ \text{(stepsize)}.
\tag{3}
\]
\end{enumerate}

The algorithm is fully defined by the hyper‑parameters
$(\eta,\beta,C)$ and the stochastic gradient oracle.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Momentum Gradient Descent (CMGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, momentum $\beta\in[0,1)$, clipping threshold $C>0$, 
initial point $x_{0}\in\R^{d}$, maximum iterations $T$.
\STATE $m_{-1}\gets 0$
\FOR{$t=0,\dots,T-1$}
    \STATE Sample stochastic gradient $\tilde g_t$ at $x_t$.
    \STATE \textbf{Clip:} $g_t\gets\tilde g_t/\max\{1,\|\tilde g_t\|/C\}$ \hfill\COMMENT{(1)}
    \STATE \textbf{Momentum:} $m_t\gets\beta m_{t-1}+(1-\beta)g_t$ \hfill\COMMENT{(2)}
    \STATE \textbf{Update:} $x_{t+1}\gets x_t-\eta m_t$ \hfill\COMMENT{(3)}
\ENDFOR
\STATE \textbf{Output:} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Choose $C=1$, $\eta=0.1$, $\beta=0$, and start from $w_{0}=0$.  
Since $\beta=0$, the momentum term reduces to $m_t=g_t$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\nabla f(w_t)$ & $\| \nabla f(w_t)\|/C$ & $g_t$ (clipped) & $w_{t+1}=w_t-\eta g_t$\\\hline
0 & $2(0-3)=-6$ & $6>1$ & $g_0= C\frac{-6}{6}= -1$ & $w_1=0-0.1(-1)=0.1$\\[4pt]
1 & $2(0.1-3)=-5.8$ & $5.8>1$ & $g_1=-1$ & $w_2=0.1-0.1(-1)=0.2$\\[4pt]
2 & $2(0.2-3)=-5.6$ & $5.6>1$ & $g_2=-1$ & $w_3=0.2-0.1(-1)=0.3$
\end{tabular}
\end{center}

The objective values are
$f(w_0)=9$, $f(w_1)=8.41$, $f(w_2)=7.84$, $f(w_3)=7.29$,
showing a steady decrease despite the aggressive clipping.

%=====================================================================
\section*{Assumptions}
%=====================================================================
\begin{enumerate}
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]

\item \textbf{Stochastic gradient oracle.}
For every $x$ there exists a random vector $\tilde g$ such that
\[
\EE[\tilde g\mid x]=\nabla f(x),\qquad
\EE\big[\|\tilde g-\nabla f(x)\|^{2}\mid x\big]\le\sigma^{2}.
\tag{A2}
\]

\item \textbf{Clipping threshold.} $C>0$ is fixed and finite.
Consequently $\|g_t\|\le C$ a.s. (by construction (1)). 
\tag{A3}
\]

\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\eta\le\frac{1}{L}.
\tag{A4}
\]

\item \textbf{Momentum parameter.} $\beta\in[0,1)$.
\end{enumerate}

%=====================================================================
\section*{Main Convergence Theorem}
%=====================================================================
\begin{theorem}[Convergence of CMGD]\label{thm:main}
Under Assumptions (A1)–(A4) and with any $T\ge1$, the iterates produced by
CMGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{2L\eta C^{2}}_{\text{clipping bias term}}
\;+\;
\underbrace{\frac{2\sigma^{2}}{C^{2}}\,\eta}_{\text{stochastic variance term}},
\tag{4}
\]
where $f^{\star}=\inf_{x}f(x)$.
Consequently, for a stepsize $\eta =\Theta\bigl(1/\sqrt{T}\bigr)$,
\[
\min_{0\le t<T}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]=\mathcal O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

%=====================================================================
\section*{Theoretical Properties}
%=====================================================================
\begin{itemize}
\item \textbf{Stability.} Because $\|g_t\|\le C$, the update magnitude
$\|\eta m_t\|\le \eta C/(1-\beta)$ is uniformly bounded; this prevents
exploding steps even when stochastic gradients have heavy tails.

\item \textbf{Hyperparameter sensitivity.}
  \begin{itemize}
  \item Larger $C$ reduces the bias term $2L\eta C^{2}$ but allows larger
        gradient steps, possibly increasing variance.
  \item Momentum $\beta$ only appears in the bound through the factor
        $1/(1-\beta)$ hidden in the derivation of Lemma~\ref{lem:momentum},
        thus a moderate $\beta$ (e.g. $0.9$) is safe.
  \item The stepsize $\eta$ must respect (A4); taking $\eta\propto 1/\sqrt{T}$
        balances the three terms in (4) and yields the optimal
        $\mathcal O(1/\sqrt{T})$ rate.
  \end{itemize}

\item \textbf{Comparison to baselines.}
  \begin{itemize}
  \item Standard SGD (no clipping) obtains the same $\mathcal O(1/\sqrt{T})$
        bound but requires a bounded second moment of the raw gradients.
  \item CMGD relaxes this by forcing $\|g_t\|\le C$, at the price of an
        additional bias term $2L\eta C^{2}$, which vanishes when
        $C\ge\sup_t\|\nabla f(x_t)\|$.
  \end{itemize}
\end{itemize}

%=====================================================================
\section*{Preliminaries (Definitions and Lemmas)}
%=====================================================================
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004introductory}.
\end{proof}

\begin{lemma}[Bound on the bias introduced by clipping]\label{lem:bias}
Define $b_t:=\EE[g_t\mid x_t]-\nabla f(x_t)$. Then
\[
\|b_t\|\le\frac{1}{C}\,\EE\!\big[\|\tilde g_t-\nabla f(x_t)\|^{2}\mid x_t\big]^{1/2}
\le\frac{\sigma}{C}.
\tag{6}
\]
\end{lemma}
\begin{proof}
If $\|\tilde g_t\|\le C$, then $g_t=\tilde g_t$ and $b_t=0$.
Otherwise $\|g_t\|=C$ and
\[
\|b_t\|
 =\Big\|\frac{C}{\|\tilde g_t\|}\tilde g_t-\nabla f(x_t)\Big\|
 \le\Big\|\tilde g_t-\nabla f(x_t)\Big\|
      +\Big\|\tilde g_t-\frac{C}{\|\tilde g_t\|}\tilde g_t\Big\|
 \le\frac{1}{C}\|\tilde g_t-\nabla f(x_t)\|^{2},
\]
where the last inequality uses $\|\tilde g_t\|\ge C$.
Taking conditional expectation and applying (A2) yields (6).
\end{proof}

\begin{lemma}[Momentum recursion]\label{lem:momentum}
For $\beta\in[0,1)$ and any sequence $\{g_t\}$,
\[
\|m_t\|^{2}\le\frac{1-\beta}{1-\beta^{2}}\sum_{k=0}^{t}\beta^{t-k}\|g_k\|^{2}
\le\frac{1}{1-\beta}\max_{0\le k\le t}\|g_k\|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Unfold $m_t$:
$m_t=\sum_{k=0}^{t}(1-\beta)\beta^{t-k}g_k$.
Using Cauchy–Schwarz,
\[
\|m_t\|^{2}\le\Big(\sum_{k=0}^{t}(1-\beta)\beta^{t-k}\|g_k\|\Big)^{2}
\le\Big(\sum_{k=0}^{t}(1-\beta)\beta^{t-k}\Big)
      \Big(\sum_{k=0}^{t}(1-\beta)\beta^{t-k}\|g_k\|^{2}\Big)
      =\frac{1-\beta^{t+1}}{1-\beta}\sum_{k=0}^{t}(1-\beta)\beta^{t-k}\|g_k\|^{2}.
\]
Since $1-\beta^{t+1}\le1$, we obtain the first inequality.
The second follows from $\|g_k\|\le C$ and the fact that
$\sum_{k=0}^{t}(1-\beta)\beta^{t-k}=1-\beta^{t+1}\le1$.
\end{proof}

%=====================================================================
\section*{Main Proof (Step‑by‑Step Derivation)}
%=====================================================================
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. the randomness up to iteration $t$,
unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply the smoothness inequality (Lemma~\ref{lem:smooth}) with
$y=x_{t+1}=x_t-\eta m_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),m_t\rangle
          +\frac{L\eta^{2}}{2}\|m_t\|^{2}.
\tag{8}
\]

\item[Step 2.] Decompose the inner product using $m_t$ from (2):
\[
\langle\nabla f(x_t),m_t\rangle
 =\beta\langle\nabla f(x_t),m_{t-1}\rangle
   +(1-\beta)\langle\nabla f(x_t),g_t\rangle .
\tag{9}
\]

\item[Step 3.] Take conditional expectation $\EE[\cdot\mid x_t]$ of (8).
Because $m_{t-1}$ is $\mathcal{F}_t$‑measurable, we obtain
\[
\EE\!\big[f(x_{t+1})\mid x_t\big]
 \le f(x_t)-\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle
      -\eta(1-\beta)\EE\!\big[\langle\nabla f(x_t),g_t\rangle\mid x_t\big]
      +\frac{L\eta^{2}}{2}\EE\!\big[\|m_t\|^{2}\mid x_t\big].
\tag{10}
\]

\item[Step 4.] Replace $\EE[g_t\mid x_t]$ by $\nabla f(x_t)+b_t$,
where $b_t$ is the clipping bias defined in Lemma~\ref{lem:bias}:
\[
\EE\!\big[\langle\nabla f(x_t),g_t\rangle\mid x_t\big]
   =\langle\nabla f(x_t),\nabla f(x_t)+b_t\rangle
   =\|\nabla f(x_t)\|^{2}+\langle\nabla f(x_t),b_t\rangle.
\tag{11}
\]

\item[Step 5.] Use Cauchy–Schwarz and Lemma~\ref{lem:bias} to bound the
bias term:
\[
\langle\nabla f(x_t),b_t\rangle
   \le\|\nabla f(x_t)\|\,\|b_t\|
   \le\|\nabla f(x_t)\|\frac{\sigma}{C}
   \le\frac{1}{2}\|\nabla f(x_t)\|^{2}
        +\frac{\sigma^{2}}{2C^{2}} .
\tag{12}
\]

\item[Step 6.] Plug (11)–(12) into (10):
\[
\EE\!\big[f(x_{t+1})\mid x_t\big]
 \le f(x_t)-\eta(1-\beta)\Bigl(\frac12\|\nabla f(x_t)\|^{2}
                               -\frac{\sigma^{2}}{2C^{2}}\Bigr)
      -\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle
      +\frac{L\eta^{2}}{2}\EE\!\big[\|m_t\|^{2}\mid x_t\big].
\tag{13}
\]

\item[Step 7.] Upper‑bound $\EE[\|m_t\|^{2}\mid x_t]$ using Lemma~\ref{lem:momentum}
and the clipping bound $\|g_k\|\le C$:
\[
\EE\!\big[\|m_t\|^{2}\mid x_t\big]\le\frac{C^{2}}{1-\beta}.
\tag{14}
\]

\item[Step 8.] Substitute (14) into (13) and drop the non‑positive term
$-\eta\beta\langle\nabla f(x_t),m_{t-1}\rangle$ (it can only improve the bound):
\[
\EE\!\big[f(x_{t+1})\mid x_t\big]
 \le f(x_t)-\frac{\eta(1-\beta)}{2}\|\nabla f(x_t)\|^{2}
      +\frac{\eta(1-\beta)}{2}\frac{\sigma^{2}}{C^{2}}
      +\frac{L\eta^{2}C^{2}}{2(1-\beta)} .
\tag{15}
\]

\item[Step 9.] Take total expectation and sum (15) over $t=0,\dots,T-1$:
\[
\EE[f(x_T)]\le f(x_0)
   -\frac{\eta(1-\beta)}{2}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
   +\frac{T\eta(1-\beta)}{2}\frac{\sigma^{2}}{C^{2}}
   +\frac{TL\eta^{2}C^{2}}{2(1-\beta)} .
\tag{16}
\]

\item[Step 10.] Rearrange (16) and use $f(x_T)\ge f^{\star}$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
 \le \frac{2\big(f(x_0)-f^{\star}\big)}{\eta(1-\beta)T}
      +\frac{ \sigma^{2}}{C^{2}}
      +\frac{ L\eta C^{2}}{(1-\beta)^{2}} .
\tag{17}
\]

\item[Step 11.] Since $0\le\beta<1$, we bound $\frac{1}{1-\beta}\le\frac{2}{1-\beta^{2}}$
and absorb the factor $1/(1-\beta)$ into the constants.
Choosing the convenient upper bound $1/(1-\beta)\le2$ yields the
simplified expression (4) stated in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
 \le
 \frac{2\big(f(x_{0})-f^{\star}\big)}{\eta T}
 +2L\eta C^{2}
 +\frac{2\sigma^{2}}{C^{2}}\eta .
\]
\end{enumerate}
This completes the proof. \hfill $\square$

%=====================================================================
\section*{Rate Derivation (With Constants)}
%=====================================================================
Set the stepsize $\eta = \sqrt{\dfrac{2\big(f(x_{0})-f^{\star}\big)}
                              {L C^{2} T}}$,
which satisfies the admissibility condition (A4) for all sufficiently
large $T$ because $\eta = O(1/\sqrt{T})\le 1/L$.
Plugging this choice into (4) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
   \;\le\;
   2\sqrt{\frac{2L C^{2}\big(f(x_{0})-f^{\star}\big)}{T}}
   +\frac{2\sigma^{2}}{C^{2}}
      \sqrt{\frac{2\big(f(x_{0})-f^{\star}\big)}{L C^{2} T}
   \;=\;
   \mathcal O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
Hence the algorithm attains the optimal $\mathcal O(1/\sqrt{T})$ rate for
non‑convex smooth optimization up to the explicit constants shown.

%=====================================================================
\section*{Special Cases}
%=====================================================================
\begin{itemize}
\item \textbf{No clipping ($C\to\infty$).}
    The bias term disappears ($\sigma^{2}/C^{2}\to0$) and (4) reduces to the
    classic SGD bound $\frac{2(f(x_{0})-f^{\star})}{\eta T}
    +\frac{L\eta\sigma^{2}}{2}$.

\item \textbf{Deterministic gradients ($\sigma=0$).}
    The variance term vanishes, and the bound consists only of the
    optimization and clipping bias terms.

\item \textbf{Heavy‑tailed noise.}
    If the raw stochastic gradients have only bounded second moment,
    clipping guarantees $\|g_t\|\le C$, thus the algorithm remains stable
    whereas vanilla SGD may diverge.
\end{itemize}
