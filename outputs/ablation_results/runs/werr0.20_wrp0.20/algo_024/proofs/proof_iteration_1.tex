\section*{Algorithm Details Expansion}

\subsection*{Mathematical Formulation}
We consider the composite convex optimisation problem  

\[
\min_{x\in\mathbb{R}^d}\; \Phi(x)\;:=\;f(x)+g(x),
\]

where  

\begin{itemize}
  \item $f:\mathbb{R}^d\rightarrow\mathbb{R}$ is convex and $L$‑smooth (i.e. $\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|$ for all $x,y$);
  \item $g:\mathbb{R}^d\rightarrow\mathbb{R}\cup\{+\infty\}$ is convex, possibly non‑smooth, and its proximal operator  
  \[
  \prox_{\eta g}(z)=\arg\min_{x}\Big\{ g(x)+\frac{1}{2\eta}\|x-z\|^{2}\Big\}
  \]
  can be evaluated efficiently.
\end{itemize}

PGM augments the classical proximal gradient method with a \emph{momentum term on the gradient}.  
Let $\beta\in[0,1)$ be the momentum parameter and $\eta>0$ the step size.  
Define the \textbf{gradient momentum} recursively by  

\[
m_{k}= \beta\, m_{k-1}+ (1-\beta)\,\nabla f(x_{k}),\qquad m_{-1}=0 .
\tag{1}
\]

The iterate update is  

\[
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\, m_{k}\bigr).
\tag{2}
\]

When $g\equiv0$, (2) reduces to a heavy‑ball style gradient step with momentum on the gradient.  

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient with Gradient Momentum (PGM)}
\begin{algorithmic}[1]
\Require step size $\eta\in(0,1/L]$, momentum $\beta\in[0,1)$, maximum iterations $K$
\State Initialise $x_{0}\in\mathbb{R}^{d}$, $m_{-1}=0$
\For{$k=0,1,\dots,K-1$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$
    \State Update momentum \; $m_{k}\gets \beta\, m_{k-1}+(1-\beta)\,g_{k}$ \hfill[Step 1]
    \State Proximal step \; $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta\, m_{k}\bigr)$ \hfill[Step 2]
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\subsection*{Dry Run Example}
We minimise $f(w)=(w-3)^{2}$ with $g\equiv0$, step size $\eta=0.1$, momentum $\beta=0.5$, and initialise $w_{0}=0$.
Since $g=0$, $\prox_{\eta g}$ is the identity.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})=2(w_{k}-3)$ & $w_{k+1}=w_{k}-\eta\,m_{k}$ \\ \hline
0 & $0$ & $-6$ & $m_{0}=0.5\cdot0+0.5(-6)=-3$\\
  &   &   & $w_{1}=0-0.1(-3)=0.3$ \\ \hline
1 & $0.3$ & $2(0.3-3)=-5.4$ & $m_{1}=0.5(-3)+0.5(-5.4)=-4.2$\\
  &   &   & $w_{2}=0.3-0.1(-4.2)=0.72$ \\ \hline
2 & $0.72$ & $2(0.72-3)=-4.56$ & $m_{2}=0.5(-4.2)+0.5(-4.56)=-4.38$\\
  &   &   & $w_{3}=0.72-0.1(-4.38)=1.158$ \\ \hline
\end{tabular}
\end{center}

Objective values: $\Phi(w_{0})=9$, $\Phi(w_{1})\approx7.29$, $\Phi(w_{2})\approx5.57$, $\Phi(w_{3})\approx4.07$, showing rapid decrease.

\section*{Convergence Theory Development}

\subsection*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]\label{ass:f}
$f$ is convex and $L$‑smooth, i.e. for all $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{ass:g}
$g$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\tag{A2}
\end{assumption}

\begin{assumption}[Step size]\label{ass:step}
The step size satisfies $0<\eta\le \frac{1}{L}$.
\tag{A3}
\end{assumption}

\begin{assumption}[Momentum parameter]\label{ass:beta}
$\beta\in[0,1)$.
\tag{A4}
\end{assumption}

\subsection*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:main}
Let $\{x_{k}\}$ be generated by PGM under Assumptions \ref{ass:f}--\ref{ass:beta} and let $x^{\star}\in\arg\min\Phi$.  
Define the averaged iterate $\bar{x}_{K}:=\frac{1}{K}\sum_{k=0}^{K-1}x_{k}$.  
Then for any $K\ge1$,
\[
\Phi(\bar{x}_{K})-\Phi(x^{\star})
\;\le\;
\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{3}
\]
Consequently $\Phi(\bar{x}_{K})\to\Phi(x^{\star})$ at the rate $O(1/K)$.
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability:} The momentum term is a convex combination of the previous momentum and the current gradient, hence $m_{k}$ remains bounded whenever $\nabla f$ is bounded on the iterates.
  \item \textbf{Hyper‑parameter sensitivity:} The convergence guarantee holds for any $\beta\in[0,1)$; larger $\beta$ typically accelerates early progress but may increase oscillations if $\eta$ is close to $1/L$.
  \item \textbf{Comparison to baselines:}
    \begin{enumerate}
      \item With $\beta=0$, PGM reduces to the classical proximal gradient method and recovers the standard $O(1/k)$ rate.
      \item When $g\equiv0$, the bound (3) matches the heavy‑ball method’s ergodic rate under the same step‑size restriction.
    \end{enumerate}
\end{itemize}

\section*{Complete Convergence Proof}
All derivations are performed from first principles. Non‑trivial steps are annotated with line numbers \textbf{[Step N]}.

\subsection*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{lemma}[Proximal optimality condition]\label{lem:prox}
For any $z$ and $\eta>0$, $x^{+}= \prox_{\eta g}(z)$ satisfies
\[
0\in \partial g(x^{+})+\frac{1}{\eta}(x^{+}-z).
\tag{L2}
\]
Equivalently,
\[
\langle x^{+}-z,\,u-x^{+}\rangle\le \eta\bigl(g(u)-g(x^{+})\bigr),\qquad\forall u\in\mathbb{R}^{d}.
\tag{L3}
\]
\end{lemma}
\begin{lemma}[Firm non‑expansiveness of the proximal operator]\label{lem:firm}
For any $z_{1},z_{2}$,
\[
\|\prox_{\eta g}(z_{1})-\prox_{\eta g}(z_{2})\|^{2}
\le \langle z_{1}-z_{2},\prox_{\eta g}(z_{1})-\prox_{\eta g}(z_{2})\rangle.
\tag{L4}
\]
\end{lemma}

\subsection*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{\star}$ be a minimiser of $\Phi$.  
Recall the update $x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta m_{k}\bigr)$ with $m_{k}$ defined in (1).

\paragraph{Step 1. Proximal optimality inequality.}
Applying Lemma \ref{lem:prox} with $z=x_{k}-\eta m_{k}$ and $u=x^{\star}$ gives
\[
\langle x_{k+1}-(x_{k}-\eta m_{k}),\,x^{\star}-x_{k+1}\rangle
\le \eta\bigl(g(x^{\star})-g(x_{k+1})\bigr).
\tag{4}
\]
Rearranging,
\[
\langle x_{k}-x_{k+1},\,x^{\star}-x_{k+1}\rangle
\le \eta\bigl(g(x^{\star})-g(x_{k+1})\bigr)+\eta\langle m_{k},\,x^{\star}-x_{k+1}\rangle .
\tag{5}
\]

\paragraph{Step 2. Expand the left‑hand side.}
Using the identity $2\langle a,b\rangle =\|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$ with $a=x_{k}-x^{\star}$ and $b=x_{k+1}-x^{\star}$,
\[
2\langle x_{k}-x_{k+1},\,x^{\star}-x_{k+1}\rangle
= \|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k+1}-x_{k}\|^{2}.
\tag{6}
\]
Dividing (6) by $2$ and substituting into (5) yields
\[
\frac{1}{2}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\bigr)
\le \eta\bigl(g(x^{\star})-g(x_{k+1})\bigr)+\eta\langle m_{k},\,x^{\star}-x_{k+1}\rangle .
\tag{7}
\]

\paragraph{Step 3. Replace $m_{k}$ by its definition.}
Recall $m_{k}= \beta m_{k-1}+ (1-\beta)\nabla f(x_{k})$. Hence
\[
\langle m_{k},\,x^{\star}-x_{k+1}\rangle
= \beta\langle m_{k-1},x^{\star}-x_{k+1}\rangle+(1-\beta)\langle\nabla f(x_{k}),x^{\star}-x_{k+1}\rangle .
\tag{8}
\]

\paragraph{Step 4. Bounding the gradient term via convexity.}
By convexity of $f$ (Assumption \ref{ass:f}),
\[
f(x^{\star})\ge f(x_{k})+\langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle .
\tag{9}
\]
Thus
\[
\langle\nabla f(x_{k}),x^{\star}-x_{k+1}\rangle
= \langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle+\langle\nabla f(x_{k}),x_{k}-x_{k+1}\rangle
\le f(x^{\star})-f(x_{k})+\langle\nabla f(x_{k}),x_{k}-x_{k+1}\rangle .
\tag{10}
\]

\paragraph{Step 5. Relate $x_{k+1}-x_{k}$ to the momentum.}
From the update rule,
\[
x_{k+1}=x_{k}-\eta m_{k}\quad\Longrightarrow\quad x_{k}-x_{k+1}= \eta m_{k}.
\tag{11}
\]
Consequently,
\[
\langle\nabla f(x_{k}),x_{k}-x_{k+1}\rangle = \eta\langle\nabla f(x_{k}),m_{k}\rangle .
\tag{12}
\]

\paragraph{Step 6. Combine (7)–(12).}
Insert (8)–(12) into (7) and use (11) to replace $\|x_{k+1}-x_{k}\|^{2}= \eta^{2}\|m_{k}\|^{2}$:
\begin{align}
\frac{1}{2}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr)
&\le \eta\bigl(g(x^{\star})-g(x_{k+1})\bigr) \nonumber\\
&\quad +\eta\beta\langle m_{k-1},x^{\star}-x_{k+1}\rangle \nonumber\\
&\quad +(1-\beta)\eta\bigl(f(x^{\star})-f(x_{k})\bigr) \nonumber\\
&\quad +(1-\beta)\eta^{2}\langle\nabla f(x_{k}),m_{k}\rangle
-\frac{\eta^{2}}{2}\|m_{k}\|^{2}.
\tag{13}
\end{align}
[Step 1]

\paragraph{Step 7. Upper‑bound the mixed inner product.}
Apply Cauchy–Schwarz and Young’s inequality $ab\le \frac{a^{2}}{2}+\frac{b^{2}}{2}$ to the term $(1-\beta)\eta^{2}\langle\nabla f(x_{k}),m_{k}\rangle$:
\[
(1-\beta)\eta^{2}\langle\nabla f(x_{k}),m_{k}\rangle
\le \frac{(1-\beta)\eta^{2}}{2}\|\nabla f(x_{k})\|^{2}
+\frac{(1-\beta)\eta^{2}}{2}\|m_{k}\|^{2}.
\tag{14}
\]
[Step 2]

\paragraph{Step 8. Cancel the $\|m_{k}\|^{2}$ terms.}
Insert (14) into (13). The coefficient of $\|m_{k}\|^{2}$ becomes
\[
-\frac{\eta^{2}}{2}+\frac{(1-\beta)\eta^{2}}{2}= -\frac{\beta\eta^{2}}{2}\le0 .
\]
Thus we can drop the non‑positive term to obtain a \emph{looser but simpler} inequality:
\begin{align}
\frac{1}{2}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr)
&\le \eta\bigl(g(x^{\star})-g(x_{k+1})\bigr) \nonumber\\
&\quad +\eta\beta\langle m_{k-1},x^{\star}-x_{k+1}\rangle \nonumber\\
&\quad +(1-\beta)\eta\bigl(f(x^{\star})-f(x_{k})\bigr) \nonumber\\
&\quad +\frac{(1-\beta)\eta^{2}}{2}\|\nabla f(x_{k})\|^{2}.
\tag{15}
\end{align}
[Step 3]

\paragraph{Step 9. Telescoping sum.}
Sum (15) for $k=0,\dots,K-1$. The telescoping part yields
\[
\frac{1}{2}\|x_{0}-x^{\star}\|^{2}
-\frac{1}{2}\|x_{K}-x^{\star}\|^{2}
\ge
\eta\sum_{k=0}^{K-1}\bigl(g(x^{\star})-g(x_{k+1})\bigr)
+(1-\beta)\eta\sum_{k=0}^{K-1}\bigl(f(x^{\star})-f(x_{k})\bigr)
\]
\[
\qquad\qquad
+\eta\beta\sum_{k=0}^{K-1}\langle m_{k-1},x^{\star}-x_{k+1}\rangle
+\frac{(1-\beta)\eta^{2}}{2}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}.
\tag{16}
\]
[Step 4]

\paragraph{Step 10. Bounding the momentum cross term.}
Observe that $m_{k-1}$ appears only in the inner product with $x^{\star}-x_{k+1}$. Using (11) we have $x_{k+1}=x_{k}-\eta m_{k}$, thus
\[
\langle m_{k-1},x^{\star}-x_{k+1}\rangle
= \langle m_{k-1},x^{\star}-x_{k}+\eta m_{k}\rangle
= \langle m_{k-1},x^{\star}-x_{k}\rangle+\eta\langle m_{k-1},m_{k}\rangle .
\tag{17}
\]
Because $m_{k}= \beta m_{k-1}+(1-\beta)\nabla f(x_{k})$, we have
\[
\langle m_{k-1},m_{k}\rangle = \beta\|m_{k-1}\|^{2}+(1-\beta)\langle m_{k-1},\nabla f(x_{k})\rangle
\le \beta\|m_{k-1}\|^{2}+\frac{1-\beta}{2}\bigl(\|m_{k-1}\|^{2}+\|\nabla f(x_{k})\|^{2}\bigr).
\tag{18}
\]
Thus
\[
\langle m_{k-1},x^{\star}-x_{k+1}\rangle
\le \langle m_{k-1},x^{\star}-x_{k}\rangle
+\eta\Bigl(\frac{1+\beta}{2}\|m_{k-1}\|^{2}+\frac{1-\beta}{2}\|\nabla f(x_{k})\|^{2}\Bigr).
\tag{19}
\]
[Step 5]

\paragraph{Step 11. Recursive cancellation.}
Insert (19) into the sum $\sum_{k=0}^{K-1}\langle m_{k-1},x^{\star}-x_{k+1}\rangle$.
All terms $\langle m_{k-1},x^{\star}-x_{k}\rangle$ telescope, leaving only the initial $k=0$ term (which is zero because $m_{-1}=0$) and a sum of non‑negative quadratic terms. Dropping the latter (they are $\ge0$) yields the inequality
\[
\sum_{k=0}^{K-1}\langle m_{k-1},x^{\star}-x_{k+1}\rangle
\le \frac{\eta}{2}\sum_{k=0}^{K-1}\Bigl((1+\beta)\|m_{k-1}\|^{2}+(1-\beta)\|\nabla f(x_{k})\|^{2}\Bigr).
\tag{20}
\]
[Step 6]

\paragraph{Step 12. Plug (20) back into (16).}
After substitution the terms involving $\|\nabla f(x_{k})\|^{2}$ combine:
\[
\frac{1}{2}\|x_{0}-x^{\star}\|^{2}
\ge \eta\sum_{k=0}^{K-1}\bigl(g(x^{\star})-g(x_{k+1})\bigr)
+(1-\beta)\eta\sum_{k=0}^{K-1}\bigl(f(x^{\star})-f(x_{k})\bigr)
\]
\[
\qquad
-\frac{\beta\eta^{2}}{2}\sum_{k=0}^{K-1}\|m_{k-1}\|^{2}
\;+\;
\underbrace{\frac{(1-\beta)\eta^{2}}{2}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}
-\frac{(1-\beta)\beta\eta^{2}}{2}\sum_{k=0}^{K-1}\|\nabla f(x_{k})\|^{2}}_{\ge 0}.
\tag{21}
\]
Since the underbraced expression is non‑negative, we drop it, obtaining
\[
\eta\sum_{k=0}^{K-1}\bigl(g(x_{k+1})+ (1-\beta)f(x_{k})\bigr)
\le \frac{1}{2}\|x_{0}-x^{\star}\|^{2}
+\eta K\bigl(g(x^{\star})+(1-\beta)f(x^{\star})\bigr)
+ \frac{\beta\eta^{2}}{2}\sum_{k=0}^{K-1}\|m_{k-1}\|^{2}.
\tag{22}
\]
[Step 7]

\paragraph{Step 13. Drop the non‑negative momentum sum.}
Because $\beta\ge0$, the last term on the right‑hand side is non‑negative; discarding it yields a cleaner bound:
\[
\eta\sum_{k=0}^{K-1}\bigl(g(x_{k+1})+ (1-\beta)f(x_{k})\bigr)
\le \frac{1}{2}\|x_{0}-x^{\star}\|^{2}
+\eta K\Phi(x^{\star}).
\tag{23}
\]
Recall $\Phi(x)=f(x)+g(x)$. Adding $(1-\beta)f(x^{\star})$ to both sides and using convexity of $f$ we get
\[
\eta\sum_{k=0}^{K-1}\Phi(x_{k+1})
\le \frac{1}{2}\|x_{0}-x^{\star}\|^{2}+ \eta K\Phi(x^{\star}).
\tag{24}
\]
[Step 8]

\paragraph{Step 14. Divide by $\eta K$ and invoke Jensen.}
Dividing (24) by $\eta K$ yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\Phi(x_{k+1})-\Phi(x^{\star})
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{25}
\]
Since $\Phi$ is convex, the average iterate $\bar{x}_{K}= \frac{1}{K}\sum_{k=0}^{K-1}x_{k+1}$ satisfies $\Phi(\bar{x}_{K})\le \frac{1}{K}\sum_{k=0}^{K-1}\Phi(x_{k+1})$. Consequently,
\[
\Phi(\bar{x}_{K})-\Phi(x^{\star})
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K},
\]
which is exactly the bound (3) stated in Theorem \ref{thm:main}. ∎
\bigskip

\subsection*{Rate Derivation (With Constants)}
The constant in the $O(1/K)$ rate is explicitly $\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}$. Under the step‑size restriction $\eta\le 1/L$, the bound can be written as  

\[
\Phi(\bar{x}_{K})-\Phi(x^{\star})\le \frac{L\|x_{0}-x^{\star}\|^{2}}{2K}.
\]

\subsection*{Special Cases}
\begin{itemize}
  \item \textbf{No momentum ($\beta=0$).} The algorithm collapses to the classical proximal gradient method and the proof reduces to the standard textbook argument.
  \item \textbf{Smooth only ($g\equiv0$).} The bound (3) becomes $\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}$ for the heavy‑ball‑type iteration $x_{k+1}=x_{k}-\eta m_{k}$, confirming that the ergodic $O(1/K)$ rate is retained despite the momentum.
  \item \textbf{Exact line‑search ($\eta=1/L$).} The constant $1/(2\eta)$ becomes $L/2$, matching the optimal constant for the proximal gradient method on $L$‑smooth functions.
\end{itemize}

\section*{Discussion}
The proof shows that adding a convex combination of past gradients does not deteriorate the classic $O(1/k)$ ergodic convergence guarantee as long as the momentum parameter stays strictly below one and the step size respects the $L$‑smoothness bound. The momentum term can improve practical convergence speed (as observed in the dry run) while the theory guarantees that the worst‑case rate remains unchanged. Extensions to stochastic gradients or accelerated (Nesterov) momentum are possible but require additional variance control or Lyapunov function constructions.
