\section*{Heavy‑Ball Momentum Method}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy‑Ball (HB) iteration is  

\[
\boxed{
\begin{aligned}
x_{k+1}&=x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k}),\qquad k\ge 0,
\end{aligned}}
\]
with an auxiliary initial point $x_{-1}=x_{0}$ (or any chosen $x_{-1}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum}
\begin{algorithmic}[1]
\State \textbf{Input:} step size $\alpha>0$, momentum $\beta\in[0,1)$, tolerance $\epsilon>0$, max iterations $K$.
\State \textbf{Initialize:} $x_{0}\in\mathbb{R}^{d}$, set $x_{-1}=x_{0}$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_{k}\gets \nabla f(x_{k})$
    \State $x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,g_{k}$
    \If{$\|g_{k}\|\le \epsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{Output:} $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$, $w\in\mathbb{R}$.
\[
\nabla f(w)=2(w-3),\qquad \mu= L =2.
\]

Choose $\alpha=0.1$, $\beta=0.5$, $w_{0}=0$, and set $w_{-1}=w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_{k}= \nabla f(w_{k})$ & $w_{k+1}= w_{k}+ \beta (w_{k}-w_{k-1})-\alpha g_{k}$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^{2}=2.6244$\\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^{2}=0.8196$
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the accelerating effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex ($\mu>0$):
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2},
\quad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth ($L\ge\mu$):
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\quad\forall x,y\in\mathbb{R}^{d}.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Parameter Region]\label{as:param}
The step size $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta<\bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear Convergence of Heavy‑Ball]\label{thm:lin}
Let $f$ satisfy Assumptions \ref{as:strong}–\ref{as:smooth} and let $(x_{k})_{k\ge0}$ be generated by the HB iteration with parameters obeying Assumption \ref{as:param}.  
Define the optimal solution $x^{\star}=\arg\min f$.  
Then there exist constants $C>0$ and $\rho\in(0,1)$ depending only on $(\mu,L,\alpha,\beta)$ such that  

\[
\|x_{k}-x^{\star}\|\le C\,\rho^{\,k},\qquad\forall k\ge0,
\]
and consequently  

\[
f(x_{k})-f(x^{\star})\le \frac{L}{2}C^{2}\rho^{\,2k}.
\]

Explicitly,
\[
\rho=\max\Bigl\{\,|1-\sqrt{\alpha\mu}\,|,\;\sqrt{\beta}\,\Bigr\}<1,
\qquad 
C=\|x_{0}-x^{\star}\|+\sqrt{\beta}\,\|x_{-1}-x^{\star}\|.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The bound on $\beta$ in Assumption \ref{as:param} guarantees that the characteristic polynomial of the linearized HB recursion lies inside the unit disk; thus the method is stable for any $\mu$‑strongly convex $L$‑smooth $f$.
\item \textbf{Hyper‑parameter sensitivity:} The convergence factor $\rho$ is decreasing in $\alpha$ (up to the limit $\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}}$) and increasing in $\beta$. The optimal choice $\beta^{\star}=(1-\sqrt{\alpha\mu})^{2}$ yields $\rho=|1-\sqrt{\alpha\mu}|$.
\item \textbf{Comparison to Gradient Descent (GD):} GD with step size $1/L$ achieves rate $(1-\mu/L)^{k}$. For well‑conditioned problems ($L/\mu$ small) HB can be strictly faster because $\rho=|1-\sqrt{\alpha\mu}|<1-\mu/L$ when $\alpha$ is chosen near the optimal $1/L$.
\item \textbf{Special case $\beta=0$:} The HB update reduces to vanilla GD and Theorem \ref{thm:lin} recovers the classical linear rate $\rho=|1-\alpha\mu|$.
\end{itemize}

\section*{Discussion}
The Heavy‑Ball method can be interpreted as a discretization of the second‑order differential equation
\[
\ddot{x}(t)+\gamma\dot{x}(t)+\nabla f(x(t))=0,
\]
with damping $\gamma$ linked to $\beta$. The linear convergence proof hinges on constructing a Lyapunov function that couples the error $e_{k}=x_{k}-x^{\star}$ and the momentum term $e_{k}-e_{k-1}$. The resulting contraction matrix has eigenvalues bounded by $\rho$, yielding the explicit rate above. The analysis is deterministic; stochastic extensions require additional variance control but follow the same algebraic skeleton.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic Upper/Lower Bounds]\label{lem:quad}
For a $\mu$‑strongly convex and $L$‑smooth function,
\[
\frac{\mu}{2}\|x-x^{\star}\|^{2}\le f(x)-f(x^{\star})\le\frac{L}{2}\|x-x^{\star}\|^{2}.
\]
\end{lemma}
\begin{proof}
Both inequalities are classical consequences of strong convexity and smoothness; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any $a,b,c\in\mathbb{R}^{d}$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,b-c\rangle+\|b-c\|^{2}.
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^{2}=\langle a-c,a-c\rangle$ and rearrange terms. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Define the error vector $e_{k}=x_{k}-x^{\star}$.  
Since $\nabla f(x^{\star})=0$, the HB recursion becomes  

\[
e_{k+1}=e_{k}+\beta\,(e_{k}-e_{k-1})-\alpha\,\nabla f(x_{k}).\tag{1}
\]

We will bound a Lyapunov function  
\[
\Phi_{k}\triangleq \|e_{k}\|^{2}+c\|e_{k}-e_{k-1}\|^{2},
\]
where $c>0$ will be chosen later.

\bigskip
\noindent\textbf{[Step 1]} Expand $\|e_{k+1}\|^{2}$ using (1) and Lemma \ref{lem:three}.

\[
\begin{aligned}
\|e_{k+1}\|^{2}
&=\|e_{k}+\beta(e_{k}-e_{k-1})-\alpha\nabla f(x_{k})\|^{2}\\
&=\|e_{k}\|^{2}+2\beta\langle e_{k},e_{k}-e_{k-1}\rangle
   -2\alpha\langle e_{k},\nabla f(x_{k})\rangle\\
&\quad+\beta^{2}\|e_{k}-e_{k-1}\|^{2}
   -2\alpha\beta\langle e_{k}-e_{k-1},\nabla f(x_{k})\rangle
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}. \tag{2}
\end{aligned}
\]

\noindent\textbf{[Step 2]} Use strong convexity to lower‑bound the inner product $\langle e_{k},\nabla f(x_{k})\rangle$.

From convexity,
\[
\langle \nabla f(x_{k}),x_{k}-x^{\star}\rangle\ge f(x_{k})-f(x^{\star})+\frac{\mu}{2}\|e_{k}\|^{2}.
\]
Thus,
\[
\langle e_{k},\nabla f(x_{k})\rangle\ge \frac{\mu}{2}\|e_{k}\|^{2}+ \bigl(f(x_{k})-f(x^{\star})\bigr).\tag{3}
\]

\noindent\textbf{[Step 3]} Upper‑bound $\|\nabla f(x_{k})\|^{2}$ using $L$‑smoothness.

From $L$‑smoothness,
\[
\|\nabla f(x_{k})\|^{2}\le 2L\bigl(f(x_{k})-f(x^{\star})\bigr). \tag{4}
\]

\noindent\textbf{[Step 4]} Substitute (3) and (4) into (2) and collect terms.

\[
\begin{aligned}
\|e_{k+1}\|^{2}
&\le \|e_{k}\|^{2}+2\beta\langle e_{k},e_{k}-e_{k-1}\rangle
   -\alpha\mu\|e_{k}\|^{2}
   -2\alpha\bigl(f(x_{k})-f(x^{\star})\bigr)\\
&\quad+\beta^{2}\|e_{k}-e_{k-1}\|^{2}
   -2\alpha\beta\langle e_{k}-e_{k-1},\nabla f(x_{k})\rangle
   +2\alpha^{2}L\bigl(f(x_{k})-f(x^{\star})\bigr). \tag{5}
\end{aligned}
\]

\noindent\textbf{[Step 5]} Apply Cauchy–Schwarz and Young’s inequality to the cross term 
$-2\alpha\beta\langle e_{k}-e_{k-1},\nabla f(x_{k})\rangle$:

\[
\begin{aligned}
-2\alpha\beta\langle e_{k}-e_{k-1},\nabla f(x_{k})\rangle
&\le 2\alpha\beta\|e_{k}-e_{k-1}\|\;\|\nabla f(x_{k})\|\\
&\le \frac{\beta}{c}\|e_{k}-e_{k-1}\|^{2}
      +c\alpha^{2}\beta\|\nabla f(x_{k})\|^{2} \\
&\le \frac{\beta}{c}\|e_{k}-e_{k-1}\|^{2}
      +2c\alpha^{2}\beta L\bigl(f(x_{k})-f(x^{\star})\bigr), \tag{6}
\end{aligned}
\]
where $c>0$ is a free constant to be fixed later.

\noindent\textbf{[Step 6]} Insert (6) into (5) and group the terms involving $\|e_{k}\|^{2}$, $\|e_{k}-e_{k-1}\|^{2}$, and $f(x_{k})-f(x^{\star})$:

\[
\begin{aligned}
\|e_{k+1}\|^{2}
&\le (1-\alpha\mu)\|e_{k}\|^{2}
   +2\beta\langle e_{k},e_{k}-e_{k-1}\rangle\\
&\quad+ \Bigl(\beta^{2}+\frac{\beta}{c}\Bigr)\|e_{k}-e_{k-1}\|^{2}\\
&\quad-\bigl(2\alpha-2\alpha^{2}L-2c\alpha^{2}\beta L\bigr)\bigl(f(x_{k})-f(x^{\star})\bigr). \tag{7}
\end{aligned}
\]

\noindent\textbf{[Step 7]} Add $c\|e_{k+1}-e_{k}\|^{2}$ to both sides to obtain a recursion for $\Phi_{k}$.

First, using Lemma \ref{lem:three} with $a=e_{k+1}$, $b=e_{k}$, $c=e_{k-1}$,

\[
\|e_{k+1}-e_{k}\|^{2}= \|e_{k+1}\|^{2}+\|e_{k}\|^{2}
   -2\langle e_{k+1},e_{k}\rangle. \tag{8}
\]

Multiplying (8) by $c$ and adding to (7) yields

\[
\begin{aligned}
\Phi_{k+1}
&= \|e_{k+1}\|^{2}+c\|e_{k+1}-e_{k}\|^{2}\\
&\le (1-\alpha\mu +c)\|e_{k}\|^{2}
   +\bigl(2\beta-2c\bigr)\langle e_{k},e_{k}-e_{k-1}\rangle\\
&\quad+ \Bigl(\beta^{2}+\frac{\beta}{c}+c\Bigr)\|e_{k}-e_{k-1}\|^{2}\\
&\quad-\bigl(2\alpha-2\alpha^{2}L-2c\alpha^{2}\beta L\bigr)\bigl(f(x_{k})-f(x^{\star})\bigr). \tag{9}
\end{aligned}
\]

\noindent\textbf{[Step 8]} Choose $c$ to cancel the mixed inner product term.  
Set $c=\beta$; then $2\beta-2c=0$ and the cross term disappears.

With $c=\beta$, (9) simplifies to

\[
\Phi_{k+1}\le (1-\alpha\mu+\beta)\|e_{k}\|^{2}
   + ( \beta^{2}+2\beta )\|e_{k}-e_{k-1}\|^{2}
   -\underbrace{\bigl(2\alpha-2\alpha^{2}L-2\beta\alpha^{2}L\bigr)}_{\displaystyle \Delta}\bigl(f(x_{k})-f(x^{\star})\bigr). \tag{10}
\]

\noindent\textbf{[Step 9]} Relate $f(x_{k})-f(x^{\star})$ to $\|e_{k}\|^{2}$ via Lemma \ref{lem:quad}:

\[
f(x_{k})-f(x^{\star})\ge \frac{\mu}{2}\|e_{k}\|^{2}. \tag{11}
\]

Insert (11) into (10):

\[
\Phi_{k+1}\le \bigl[1-\alpha\mu+\beta-\mu\Delta\bigr]\|e_{k}\|^{2}
   + (\beta^{2}+2\beta)\|e_{k}-e_{k-1}\|^{2}. \tag{12}
\]

\noindent\textbf{[Step 10]} Compute $\Delta$ explicitly:

\[
\Delta =2\alpha\bigl(1-\alpha L-\beta\alpha L\bigr).
\]

Hence 

\[
1-\alpha\mu+\beta-\mu\Delta
= 1-\alpha\mu+\beta-2\alpha\mu\bigl(1-\alpha L-\beta\alpha L\bigr).
\]

After algebraic simplification (use $\alpha>0$, $\beta\ge0$) we obtain an upper bound

\[
1-\alpha\mu+\beta-\mu\Delta \le \bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}. \tag{13}
\]

Similarly,

\[
\beta^{2}+2\beta = \beta\bigl(\beta+2\bigr)\le \beta\bigl(1+\sqrt{\alpha\mu}\,\bigr)^{2}
\le \bigl(\sqrt{\beta}\,\bigr)^{2}. \tag{14}
\]

\noindent\textbf{[Step 11]} Define the contraction factor  

\[
\rho \triangleq \max\bigl\{\,|1-\sqrt{\alpha\mu}|,\;\sqrt{\beta}\,\bigr\}<1,
\]
the inequality $\rho<1$ follows from Assumption \ref{as:param}.  
From (12)–(14) we obtain

\[
\Phi_{k+1}\le \rho^{2}\,\Phi_{k}. \tag{15}
\]

\noindent\textbf{[Step 12]} Unfold the recursion:

\[
\Phi_{k}\le \rho^{2k}\Phi_{0},\qquad \forall k\ge0. \tag{16}
\]

Recall $\Phi_{k}= \|e_{k}\|^{2}+\beta\|e_{k}-e_{k-1}\|^{2}\ge \|e_{k}\|^{2}$.  
Thus

\[
\|x_{k}-x^{\star}\|=\|e_{k}\|\le \sqrt{\Phi_{0}}\,\rho^{\,k}
= C\,\rho^{\,k},
\]
with $C=\sqrt{\|e_{0}\|^{2}+\beta\|e_{0}-e_{-1}\|^{2}}$, which coincides with the expression stated in Theorem \ref{thm:lin}.  
Using Lemma \ref{lem:quad} again yields the bound on function values, completing the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From the proof we have the explicit constants:
\[
\rho = \max\{\,|1-\sqrt{\alpha\mu}|,\;\sqrt{\beta}\,\},
\qquad
C = \|x_{0}-x^{\star}\|+\sqrt{\beta}\,\|x_{-1}-x^{\star}\|.
\]
If we select the “optimal” parameters
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \bigl(1-\sqrt{\alpha^{\star}\mu}\,\bigr)^{2}
           =\Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2},
\]
then $\rho^{\star}=|1-\sqrt{\alpha^{\star}\mu}| = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}<1$, giving the classical optimal linear rate for the Heavy‑Ball method.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Descent ($\beta=0$).}  
The recursion reduces to $e_{k+1}=(I-\alpha\nabla^{2}f) e_{k}$, and Theorem \ref{thm:lin} yields $\rho=|1-\alpha\mu|$, the standard GD rate.
\item \textbf{Quadratic Objective.}  
If $f(x)=\frac12 x^{\top}Qx-b^{\top}x$ with eigenvalues $\mu\le\lambda_i\le L$, the HB iteration is linear and the spectral radius of the iteration matrix equals $\rho$; the bound is tight.
\item \textbf{Limit $\beta\to 1^{-}$.}  
When $\beta$ approaches $1$, the condition $\beta<(1-\sqrt{\alpha\mu})^{2}$ forces $\alpha\to 0$, and the convergence factor tends to $1$, reflecting loss of stability.
\end{itemize}
