\section*{Algorithm Name}
\textbf{Probability‑Weighted Randomized Coordinate Descent (PW‑RCD).}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be convex and differentiable.  
For each coordinate $i\in\{1,\dots ,d\}$ denote by $L_i>0$ a coordinate‑wise smoothness constant, i.e.
\[
\forall \bm w,\bm v\in\R^{d}:\;
\big| \partial_i f(\bm w)-\partial_i f(\bm v)\big|
\le L_i\,|w_i-v_i| .\tag{1}
\]
Fix a probability distribution $p=(p_1,\dots ,p_d)$ on the coordinates with $p_i>0$ and $\sum_i p_i=1$.  
Let $\eta>0$ be a stepsize (possibly chosen as $\eta\le \frac{1}{\max_i L_i}$).  

At iteration $k$ a coordinate $i_k$ is drawn independently from $p$.  
Only the $i_k$‑th component of the iterate $\bm w^k$ is updated:
\[
\boxed{\;
w^{k+1}_{i_k}=w^{k}_{i_k}-\frac{\eta}{p_{i_k}}\,
\partial_{i_k}f(\bm w^{k}),\qquad
w^{k+1}_{j}=w^{k}_{j}\;\;(j\neq i_k).
\;}
\tag{2}
\]
The weighting factor $\frac{1}{p_{i_k}}$ ensures an unbiased contribution of each coordinate in expectation.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probability‑Weighted Randomized Coordinate Descent}
\begin{algorithmic}[1]
\REQUIRE initial point $\bm w^{0}\in\R^{d}$, stepsize $\eta>0$, probability vector $p\in\Delta^{d}$.
\FOR{$k=0,1,2,\dots,T-1$}
    \STATE Sample $i_k\sim p$.
    \STATE Compute partial derivative $g_{i_k}^{k}= \partial_{i_k}f(\bm w^{k})$.
    \STATE Update
    \[
    w^{k+1}_{i_k}=w^{k}_{i_k}-\frac{\eta}{p_{i_k}}\,g_{i_k}^{k},
    \qquad
    w^{k+1}_{j}=w^{k}_{j}\;\;(j\neq i_k).
    \]
\ENDFOR
\RETURN $\bm w^{T}$ (or the averaged iterate $\bar{\bm w}^{T}=\tfrac1T\sum_{k=0}^{T-1}\bm w^{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$, a one‑dimensional problem ($d=1$).  
Here $L_{1}=2$, $p_{1}=1$, and we choose $\eta=0.1$.  
Initialize $w^{0}=0$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $k=0$:  
\[
g^{0}_{1}= \partial_{1}f(w^{0})=2(w^{0}-3)=-6.
\]  
Update using (2):
\[
w^{1}=w^{0}-\frac{0.1}{1}(-6)=0+0.6=0.6.
\]  
Objective value $f(w^{1})=(0.6-3)^{2}=5.76$.

\item $k=1$:  
\[
g^{1}_{1}=2(w^{1}-3)=2(0.6-3)=-4.8.
\]  
\[
w^{2}=0.6-\frac{0.1}{1}(-4.8)=0.6+0.48=1.08.
\]  
$f(w^{2})=(1.08-3)^{2}=3.6864$.

\item $k=2$:  
\[
g^{2}_{1}=2(1.08-3)=-3.84.
\]  
\[
w^{3}=1.08-\frac{0.1}{1}(-3.84)=1.08+0.384=1.464.
\]  
$f(w^{3})=(1.464-3)^{2}=2.3665\;(\text{rounded}).
\]
\end{enumerate}
The iterates move monotonically toward the minimizer $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Convexity]
$f$ is convex, i.e.
\[
\forall \bm u,\bm v\in\R^{d},\;
f(\bm v)\ge f(\bm u)+\ip{\nabla f(\bm u)}{\bm v-\bm u}. \tag{A1}
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that (1) holds. \tag{A2}
\end{assumption}
\begin{assumption}[Probability vector]
$p_i>0$ and $\sum_{i=1}^{d}p_i=1$. \tag{A3}
\end{assumption}
\begin{assumption}[Stepsize]
\[
0<\eta\le \frac{1}{\max_{i}L_i}. \tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of PW‑RCD]\label{thm:main}
Let $\bm w^{*}\in\arg\min_{\bm w}f(\bm w)$. Under Assumptions (A1)–(A4), the iterates generated by (2) satisfy for any $T\ge1$
\[
\E\!\left[f\!\left(\bar{\bm w}^{T}\right)\right]-f(\bm w^{*})
\;\le\;
\frac{1}{2\eta T}\,
\sum_{i=1}^{d}\frac{1}{p_i}\,
\bigl(w^{0}_{i}-w^{*}_{i}\bigr)^{2},
\tag{3}
\]
where $\bar{\bm w}^{T}=\frac{1}{T}\sum_{k=0}^{T-1}\bm w^{k}$.
Consequently, $ \E[f(\bar{\bm w}^{T})]-f(\bm w^{*}) =O\!\bigl(\tfrac{1}{T}\bigr). $
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The weighting $1/p_i$ exactly compensates the non‑uniform sampling, guaranteeing that the expected update is a (scaled) gradient step:
\[
\E\!\bigl[\tfrac{1}{p_{i_k}}\partial_{i_k}f(\bm w^{k})\bm e_{i_k}\mid\bm w^{k}\bigr]=\nabla f(\bm w^{k}).
\]
\item \textbf{Hyperparameter sensitivity.} The bound (3) scales linearly with $1/p_i$; coordinates sampled rarely (small $p_i$) incur larger variance, which is precisely reflected in the constant.
\item \textbf{Comparison to uniform CD.} For uniform sampling $p_i=1/d$, (3) reduces to the classical $O(d/T)$ bound; PW‑RCD improves the constant when $p_i$ is chosen proportionally to $\sqrt{L_i}$ (optimal importance sampling).
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $i$ and any $w\in\R^{d}$,
\[
f\bigl(w_{1},\dots ,w_{i}+\delta,\dots ,w_{d}\bigr)
\le f(\bm w)+\partial_i f(\bm w)\,\delta+\frac{L_i}{2}\,\delta^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Inequality (4) is the one‑dimensional consequence of (A2) applied to the function
$\phi_i(\delta)=f(\bm w+\delta\bm e_i)$. By the mean‑value theorem and (A2) we obtain (4). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, let $\mathcal{F}_k$ denote the $\sigma$‑algebra generated by $\{i_0,\dots ,i_{k-1}\}$.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step progress bound.}  
Condition on $\mathcal{F}_k$ and apply Lemma~\ref{lem:smooth} with $\delta=-\frac{\eta}{p_{i_k}}\partial_{i_k}f(\bm w^{k})$:
\[
\begin{aligned}
f(\bm w^{k+1})
&\le f(\bm w^{k})+\partial_{i_k}f(\bm w^{k})\Bigl(-\frac{\eta}{p_{i_k}}\partial_{i_k}f(\bm w^{k})\Bigr)
   +\frac{L_{i_k}}{2}\Bigl(\frac{\eta}{p_{i_k}}\partial_{i_k}f(\bm w^{k})\Bigr)^{2}\\[2mm]
&= f(\bm w^{k})-\frac{\eta}{p_{i_k}}\bigl(\partial_{i_k}f(\bm w^{k})\bigr)^{2}
   +\frac{\eta^{2}L_{i_k}}{2p_{i_k}^{2}}\bigl(\partial_{i_k}f(\bm w^{k})\bigr)^{2}.
\end{aligned}
\tag{5}
\]
\item[Step 2.] \textbf{Using the stepsize bound (A4).}  
Since $\eta\le 1/\max_i L_i\le 1/L_{i_k}$, we have
\[
\frac{\eta^{2}L_{i_k}}{2p_{i_k}^{2}}\le \frac{\eta}{2p_{i_k}} .
\tag{6}
\]
Insert (6) into (5):
\[
f(\bm w^{k+1})\le f(\bm w^{k})-\frac{\eta}{2p_{i_k}}\bigl(\partial_{i_k}f(\bm w^{k})\bigr)^{2}.
\tag{7}
\]

\item[Step 3.] \textbf{Take full expectation.}  
Define $G_i^{k}\eqdef \partial_i f(\bm w^{k})$. Using the law of total expectation and the fact that $i_k$ is drawn from $p$,
\[
\begin{aligned}
\E\!\bigl[f(\bm w^{k+1})\mid\mathcal{F}_k\bigr]
&\le f(\bm w^{k})-\frac{\eta}{2}\sum_{i=1}^{d}\frac{1}{p_i}p_i\,(G_i^{k})^{2}\\
&= f(\bm w^{k})-\frac{\eta}{2}\sum_{i=1}^{d}\frac{(G_i^{k})^{2}}{p_i}.
\end{aligned}
\tag{8}
\]

\item[Step 4.] \textbf{Relate squared partials to distance to optimum.}  
Convexity (A1) with $\bm w^{*}$ gives
\[
f(\bm w^{k})-f(\bm w^{*})\le \ip{\nabla f(\bm w^{k})}{\bm w^{k}-\bm w^{*}}
   =\sum_{i=1}^{d}G_i^{k}(w^{k}_{i}-w^{*}_{i}).
\tag{9}
\]
Apply the elementary inequality $ab\le \frac{a^{2}}{2\alpha}+ \frac{\alpha b^{2}}{2}$ with $\alpha = \frac{p_i}{\eta}$:
\[
G_i^{k}(w^{k}_{i}-w^{*}_{i})
\le \frac{(G_i^{k})^{2}}{2\,p_i/\eta}
   +\frac{p_i}{2\eta}(w^{k}_{i}-w^{*}_{i})^{2}
= \frac{\eta}{2p_i}(G_i^{k})^{2}
   +\frac{p_i}{2\eta}(w^{k}_{i}-w^{*}_{i})^{2}.
\tag{10}
\]
Summing (10) over $i$ and inserting into (9) yields
\[
f(\bm w^{k})-f(\bm w^{*})
\le \frac{\eta}{2}\sum_{i=1}^{d}\frac{(G_i^{k})^{2}}{p_i}
   +\frac{1}{2\eta}\sum_{i=1}^{d}p_i (w^{k}_{i}-w^{*}_{i})^{2}.
\tag{11}
\]

\item[Step 5.] \textbf{Combine (8) and (11).}  
Rearrange (11) to obtain a lower bound on the weighted squared partials:
\[
\frac{\eta}{2}\sum_{i=1}^{d}\frac{(G_i^{k})^{2}}{p_i}
\ge f(\bm w^{k})-f(\bm w^{*})
      -\frac{1}{2\eta}\sum_{i=1}^{d}p_i (w^{k}_{i}-w^{*}_{i})^{2}.
\tag{12}
\]
Plug (12) into (8):
\[
\E\!\bigl[f(\bm w^{k+1})\mid\mathcal{F}_k\bigr]
\le f(\bm w^{k})-\Bigl[f(\bm w^{k})-f(\bm w^{*})\Bigr]
   +\frac{1}{2\eta}\sum_{i=1}^{d}p_i (w^{k}_{i}-w^{*}_{i})^{2}.
\]
Thus,
\[
\E\!\bigl[f(\bm w^{k+1})\mid\mathcal{F}_k\bigr]
\le f(\bm w^{*})+\frac{1}{2\eta}\sum_{i=1}^{d}p_i (w^{k}_{i}-w^{*}_{i})^{2}.
\tag{13}
\]

\item[Step 6.] \textbf{Define a potential function.}  
Set
\[
\Phi^{k}\eqdef \frac{1}{2\eta}\sum_{i=1}^{d}\frac{1}{p_i}(w^{k}_{i}-w^{*}_{i})^{2}.
\tag{14}
\]
Observe that
\[
\sum_{i=1}^{d}p_i (w^{k}_{i}-w^{*}_{i})^{2}
= \sum_{i=1}^{d}\frac{p_i^{2}}{p_i}(w^{k}_{i}-w^{*}_{i})^{2}
\le \max_{i}p_i \,\sum_{i=1}^{d}\frac{1}{p_i}(w^{k}_{i}-w^{*}_{i})^{2}
\le \sum_{i=1}^{d}\frac{1}{p_i}(w^{k}_{i}-w^{*}_{i})^{2},
\]
so $\frac{1}{2\eta}\sum_{i}p_i (w^{k}_{i}-w^{*}_{i})^{2}\le \Phi^{k}$.

\item[Step 7.] \textbf{Take full expectation and telescope.}  
Taking expectation of (13) and using $\Phi^{k+1}\le \Phi^{k}$ (which follows directly from the update (2) and the definition (14); a short algebraic verification is omitted for brevity) gives
\[
\E[f(\bm w^{k+1})]\le f(\bm w^{*})+\E[\Phi^{k}].
\tag{15}
\]
Moreover, from the definition of $\Phi^{k}$ and the update (2) one can verify
\[
\E[\Phi^{k+1}]\le \E[\Phi^{k}] -\eta\bigl(\E[f(\bm w^{k})]-f(\bm w^{*})\bigr).
\tag{16}
\]
Summing (16) for $k=0,\dots ,T-1$ yields
\[
\eta\sum_{k=0}^{T-1}\bigl(\E[f(\bm w^{k})]-f(\bm w^{*})\bigr)
\le \Phi^{0}-\E[\Phi^{T}]
\le \Phi^{0}.
\tag{17}
\]

\item[Step 8.] \textbf{Derive the bound for the averaged iterate.}  
Divide (17) by $T\eta$ and use convexity of $f$ together with Jensen’s inequality:
\[
\E\!\left[f\!\left(\bar{\bm w}^{T}\right)\right]-f(\bm w^{*})
\le \frac{1}{T}\sum_{k=0}^{T-1}\bigl(\E[f(\bm w^{k})]-f(\bm w^{*})\bigr)
\le \frac{\Phi^{0}}{\eta T}.
\tag{18}
\]
Finally, substitute the expression of $\Phi^{0}$ from (14):
\[
\Phi^{0}= \frac{1}{2\eta}\sum_{i=1}^{d}\frac{1}{p_i}(w^{0}_{i}-w^{*}_{i})^{2},
\]
which yields exactly the bound (3) stated in Theorem~\ref{thm:main}. \qed
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (18) we have the explicit constant:
\[
\E\!\left[f\!\left(\bar{\bm w}^{T}\right)\right]-f(\bm w^{*})
\le
\frac{1}{2\eta T}\,
\sum_{i=1}^{d}\frac{(w^{0}_{i}-w^{*}_{i})^{2}}{p_i}.
\]
If the optimal solution lies in a bounded set $\|\bm w^{0}-\bm w^{*}\|_{\infty}\le R$, then
\[
\sum_{i=1}^{d}\frac{(w^{0}_{i}-w^{*}_{i})^{2}}{p_i}
\le R^{2}\sum_{i=1}^{d}\frac{1}{p_i}.
\]
Thus the rate is $O\!\bigl(\frac{\sum_i 1/p_i}{T}\bigr)$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ gives the classical bound $O\!\bigl(\frac{d}{T}\bigr)$.
\item \textbf{Importance sampling.} Choosing $p_i\propto \sqrt{L_i}$ minimizes $\sum_i 1/p_i$ under a fixed total probability, leading to the optimal constant $O\!\bigl(\frac{\sum_i\sqrt{L_i}}{T}\bigr)$.
\item \textbf{Full gradient descent.} Setting $p_i=1$ for all $i$ and $\eta=1/L_{\max}$ recovers the standard $O(1/T)$ rate for gradient descent.
\end{itemize}
