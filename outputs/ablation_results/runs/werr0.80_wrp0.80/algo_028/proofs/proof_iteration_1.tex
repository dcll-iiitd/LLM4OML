\section*{Probabilistic Momentum Coordinate Descent (PMCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  
For each coordinate $i\in\{1,\dots,d\}$ we denote the partial derivative by 
\[
g_i(\mathbf w)=\frac{\partial f}{\partial w_i}(\mathbf w).
\]
PMCD maintains a \emph{velocity} vector $\mathbf v^{k}\in\R^{d}$.
At iteration $k$ a coordinate $i_k$ is drawn independently from a fixed
distribution $\{p_i\}_{i=1}^{d}$ with $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.
Only the selected coordinate is updated as follows
\[
\begin{aligned}
\mathbf v^{k+1}_i &=
\begin{cases}
\beta \,\mathbf v^{k}_i + \eta\, g_i(\mathbf w^{k}), & i=i_k,\\[4pt]
\mathbf v^{k}_i, & i\neq i_k,
\end{cases} \\
\mathbf w^{k+1}_i &=
\begin{cases}
\mathbf w^{k}_i - \alpha \,\mathbf v^{k+1}_i, & i=i_k,\\[4pt]
\mathbf w^{k}_i, & i\neq i_k,
\end{cases}
\end{aligned}
\tag{1}
\]
where 
\begin{itemize}
  \item $\alpha>0$ is the \emph{stepsize},
  \item $\beta\in[0,1)$ is the \emph{momentum coefficient},
  \item $\eta>0$ scales the stochastic gradient contribution.
\end{itemize}
All other coordinates remain unchanged.
The algorithm terminates after a predetermined number $T$ of iterations
or when $\|\nabla f(\mathbf w^{k})\|$ falls below a tolerance.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Momentum Coordinate Descent (PMCD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $\mathbf w^{0}\in\R^{d}$, velocity $\mathbf v^{0}=0$, stepsize $\alpha$, momentum $\beta$, gradient scale $\eta$, probabilities $\{p_i\}_{i=1}^{d}$, max‑iterations $T$.
\FOR{$k=0$ \TO $T-1$}
    \STATE Sample $i_k\sim \{p_i\}_{i=1}^{d}$.
    \STATE Compute partial gradient $g_{i_k}(\mathbf w^{k}) = \partial f/\partial w_{i_k}(\mathbf w^{k})$.
    \STATE $\mathbf v^{k+1}_{i_k}\gets \beta\,\mathbf v^{k}_{i_k} + \eta\, g_{i_k}(\mathbf w^{k})$.
    \STATE $\mathbf w^{k+1}_{i_k}\gets \mathbf w^{k}_{i_k} - \alpha\,\mathbf v^{k+1}_{i_k}$.
    \STATE For all $i\neq i_k$, set $\mathbf v^{k+1}_{i}\gets \mathbf v^{k}_{i}$ and $\mathbf w^{k+1}_{i}\gets \mathbf w^{k}_{i}$.
\ENDFOR
\STATE \textbf{Output:} $\mathbf w^{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^{2}$, thus $g(w)=2(w-3)$.
Set $w_{0}=0$, $\alpha=0.1$, $\beta=0$, $\eta=1$ (no momentum), and $p_{1}=1$ (the only coordinate is always chosen).

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w^{k}$ & $g(w^{k})$ & $f(w^{k})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $0-0.1\cdot(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$\\
2 & $0.6-0.1\cdot(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$\\
3 & $1.08-0.1\cdot(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3677$\\
\end{tabular}
\end{center}
The iterates move toward the optimum $w^{\star}=3$ and the objective value decreases at each step.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex, i.e.
\[
f(\mathbf y)\ge f(\mathbf x)+\langle \nabla f(\mathbf x),\mathbf y-\mathbf x\rangle
+\frac{\mu}{2}\|\mathbf y-\mathbf x\|^{2},\qquad\forall\mathbf x,\mathbf y\in\R^{d}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Coordinate-wise Smoothness]\label{as:smooth}
For each $i\in\{1,\dots,d\}$ the partial gradient $g_i(\mathbf w)$ is $L_i$‑Lipschitz:
\[
|g_i(\mathbf w+\mathbf e_i t)-g_i(\mathbf w)|\le L_i|t|,\qquad \forall t\in\R,\;\forall\mathbf w\in\R^{d},
\tag{3}
\]
where $\mathbf e_i$ denotes the $i$‑th canonical basis vector.
\end{assumption}
Define the \emph{average smoothness}
\[
\bar L \;:=\; \sum_{i=1}^{d}p_iL_i,\qquad
L_{\max}:=\max_{i}L_i .
\tag{4}
\]
All hyper‑parameters are chosen such that
\[
0<\alpha\le \frac{2(1-\beta)}{\bar L},\qquad 0\le\beta<1,\qquad \eta>0 .
\tag{5}
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions~\ref{as:strong}–\ref{as:smooth} and the parameter
restriction~\eqref{eq:step}, let $\mathbf w^{\star}$ be the unique minimizer of $f$.
Define the Lyapunov function
\[
\Phi_k:=\E\!\left[f(\mathbf w^{k})-f(\mathbf w^{\star})+\frac{\gamma}{2}\|\mathbf v^{k}\|^{2}\right],
\tag{6}
\]
with $\gamma:=\frac{\alpha(1-\beta)}{\eta}>0$.
Then for every $k\ge0$
\[
\Phi_{k+1}\;\le\;\bigl(1-\rho\bigr)\,\Phi_{k},
\qquad\text{where}\qquad
\rho:=\frac{\alpha\mu(1-\beta)}{1+\alpha\mu(1-\beta)}\in(0,1).
\tag{7}
\]
Consequently,
\[
\E\!\bigl[f(\mathbf w^{T})-f(\mathbf w^{\star})\bigr]\;\le\;(1-\rho)^{T}\,
\Bigl(f(\mathbf w^{0})-f(\mathbf w^{\star})+\tfrac{\gamma}{2}\|\mathbf v^{0}\|^{2}\Bigr).
\tag{8}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound \eqref{eq:step} guarantees that the
update matrix associated with a single coordinate is a contraction,
hence the iterates remain bounded.
\item \textbf{Hyper‑parameter sensitivity.} Larger momentum $\beta$
increases the effective stepsize $\alpha/(1-\beta)$; the condition
\eqref{eq:step} prevents overshooting. The convergence factor $\rho$ grows
with $\alpha$ and $\mu$ and shrinks with $\beta$, reflecting the usual
trade‑off between speed and stability.
\item \textbf{Comparison to baseline CD.} Setting $\beta=0$ reduces PMCD to
the classic randomized coordinate descent, for which the rate is
$(1-\alpha\mu)$. The presence of momentum improves the factor to
$1-\rho$ with $\rho>\alpha\mu$ whenever $\beta>0$ and $\alpha$ respects
\eqref{eq:step}.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite updating a single coordinate per
iteration, the algorithm inherits a geometric decay of the expected
suboptimality. The proof hinges on exploiting strong convexity together
with coordinate-wise smoothness to control the bias introduced by the
random selection. Momentum accelerates the contraction without
requiring a full‑gradient evaluation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $i$ and any $\mathbf w\in\R^{d}$,
\[
f(\mathbf w-\alpha\mathbf e_i g_i(\mathbf w))
\;\le\;
f(\mathbf w)-\alpha g_i(\mathbf w)^{2}
+\frac{L_i\alpha^{2}}{2}\,g_i(\mathbf w)^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Apply the one‑dimensional smoothness bound
\[
f(\mathbf w+\mathbf e_i t)
\le f(\mathbf w)+g_i(\mathbf w)t+\frac{L_i}{2}t^{2}
\]
with $t=-\alpha g_i(\mathbf w)$.
\end{proof}

\begin{lemma}[Strong convexity lower bound]\label{lem:strong}
For any $\mathbf w$,
\[
f(\mathbf w)-f(\mathbf w^{\star})\;\ge\;\frac{\mu}{2}\|\mathbf w-\mathbf w^{\star}\|^{2}.
\tag{10}
\]
\end{lemma}
\begin{proof}
Take $\mathbf y=\mathbf w^{\star}$ in \eqref{eq:strong} and use $\nabla f(\mathbf w^{\star})=0$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we condition on the
history $\mathcal F_{k}$ generated up to iteration $k$ and write
$\E_{k}[\cdot]=\E[\cdot\mid\mathcal F_{k}]$.

\noindent\textbf{[Step 1]}   Expand the squared norm of the velocity after the update of the selected coordinate $i_k$:
\[
\begin{aligned}
\|\mathbf v^{k+1}\|^{2}
&= \sum_{i\neq i_k}\|\mathbf v^{k}_{i}\|^{2}
   +\bigl\|\beta\mathbf v^{k}_{i_k}+ \eta g_{i_k}(\mathbf w^{k})\bigr\|^{2}\\
&= \|\mathbf v^{k}\|^{2}
   -\|\mathbf v^{k}_{i_k}\|^{2}
   +\beta^{2}\|\mathbf v^{k}_{i_k}\|^{2}
   +2\beta\eta \mathbf v^{k}_{i_k} g_{i_k}(\mathbf w^{k})
   +\eta^{2}g_{i_k}(\mathbf w^{k})^{2}\\
&= \|\mathbf v^{k}\|^{2}
   - (1-\beta^{2})\|\mathbf v^{k}_{i_k}\|^{2}
   +2\beta\eta \mathbf v^{k}_{i_k} g_{i_k}(\mathbf w^{k})
   +\eta^{2}g_{i_k}(\mathbf w^{k})^{2}.
\end{aligned}
\tag{11}
\]

\noindent\textbf{[Step 2]}   Take expectation over the random choice of $i_k$.
Using $\Pr(i_k=i)=p_i$ and linearity,
\[
\begin{aligned}
\E_{k}\!\bigl[\|\mathbf v^{k+1}\|^{2}\bigr]
&= \|\mathbf v^{k}\|^{2}
   - (1-\beta^{2})\sum_{i}p_i\|\mathbf v^{k}_{i}\|^{2}
   +2\beta\eta\sum_{i}p_i\mathbf v^{k}_{i} g_i(\mathbf w^{k})
   +\eta^{2}\sum_{i}p_i g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 3]}   Compute the expected decrease of the objective.
Conditioned on $\mathcal F_k$, only coordinate $i_k$ changes, thus by Lemma~\ref{lem:smooth}
\[
\begin{aligned}
\E_{k}\!\bigl[f(\mathbf w^{k+1})\bigr]
&= \sum_{i}p_i\,
   f\bigl(\mathbf w^{k}-\alpha\mathbf e_i\mathbf v^{k+1}_{i}\bigr)\\
&\le \sum_{i}p_i\Bigl[
    f(\mathbf w^{k})
    -\alpha\mathbf v^{k+1}_{i}g_i(\mathbf w^{k})
    +\frac{L_i\alpha^{2}}{2}(\mathbf v^{k+1}_{i})^{2}
   \Bigr]  \quad\text{(apply (9) with }t=-\alpha\mathbf v^{k+1}_{i})\\
&= f(\mathbf w^{k})
   -\alpha\sum_{i}p_i\mathbf v^{k+1}_{i} g_i(\mathbf w^{k})
   +\frac{\alpha^{2}}{2}\sum_{i}p_i L_i(\mathbf v^{k+1}_{i})^{2}.
\end{aligned}
\tag{13}
\]

\noindent\textbf{[Step 4]}   Substitute the explicit form of $\mathbf v^{k+1}_{i}$.
Recall that $\mathbf v^{k+1}_{i}= \beta\mathbf v^{k}_{i}+ \eta g_i(\mathbf w^{k})$ for the sampled $i$, and equals $\mathbf v^{k}_{i}$ otherwise.
Hence
\[
\mathbf v^{k+1}_{i}
= \beta\mathbf v^{k}_{i}+ \eta g_i(\mathbf w^{k})\quad\text{with probability }p_i,
\]
and otherwise $\mathbf v^{k+1}_{i}= \mathbf v^{k}_{i}$.  Therefore
\[
\begin{aligned}
\E_{k}\!\bigl[\mathbf v^{k+1}_{i}g_i(\mathbf w^{k})\bigr]
&= p_i\bigl(\beta\mathbf v^{k}_{i}+ \eta g_i(\mathbf w^{k})\bigr)g_i(\mathbf w^{k})
   +(1-p_i)\mathbf v^{k}_{i}g_i(\mathbf w^{k})\\
&= \mathbf v^{k}_{i}g_i(\mathbf w^{k})
   +p_i\bigl(\beta-1\bigr)\mathbf v^{k}_{i}g_i(\mathbf w^{k})
   +p_i\eta g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{14}
\]

Similarly,
\[
\begin{aligned}
\E_{k}\!\bigl[(\mathbf v^{k+1}_{i})^{2}\bigr]
&= p_i\bigl(\beta\mathbf v^{k}_{i}+ \eta g_i(\mathbf w^{k})\bigr)^{2}
   +(1-p_i)(\mathbf v^{k}_{i})^{2}\\
&= (\mathbf v^{k}_{i})^{2}
   +p_i\bigl(\beta^{2}-1\bigr)(\mathbf v^{k}_{i})^{2}
   +2p_i\beta\eta \mathbf v^{k}_{i} g_i(\mathbf w^{k})
   +p_i\eta^{2}g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{15}
\]

\noindent\textbf{[Step 5]}   Insert \eqref{eq:14}–\eqref{eq:15} into \eqref{eq:13}.
After a straightforward algebraic regrouping we obtain
\[
\begin{aligned}
\E_{k}\!\bigl[f(\mathbf w^{k+1})\bigr]
\le &\; f(\mathbf w^{k})
   -\alpha\!\sum_{i}\!\mathbf v^{k}_{i} g_i(\mathbf w^{k})
   -\alpha\!\sum_{i}p_i(\beta-1)\mathbf v^{k}_{i} g_i(\mathbf w^{k})
   -\alpha\!\sum_{i}p_i\eta g_i(\mathbf w^{k})^{2}\\
   &\;+\frac{\alpha^{2}}{2}\!\sum_{i}p_i L_i(\mathbf v^{k}_{i})^{2}
   +\frac{\alpha^{2}}{2}\!\sum_{i}p_i L_i\bigl(\beta^{2}-1\bigr)(\mathbf v^{k}_{i})^{2}\\
   &\;+\alpha^{2}\!\sum_{i}p_i L_i\beta\eta \mathbf v^{k}_{i} g_i(\mathbf w^{k})
   +\frac{\alpha^{2}}{2}\!\sum_{i}p_i L_i\eta^{2}g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{16}
\]

\noindent\textbf{[Step 6]}   Choose the stepsize $\alpha$ to satisfy
\[
\alpha\le\frac{2(1-\beta)}{\bar L},
\qquad\bar L=\sum_{i}p_iL_i,
\tag{17}
\]
which guarantees
\[
\frac{\alpha^{2}}{2}p_iL_i(\beta^{2}-1)+\alpha p_i(\beta-1)\le0,
\qquad\forall i.
\tag{18}
\]
Thus the mixed terms involving $\mathbf v^{k}_{i}g_i$ are non‑positive and can be dropped.

\noindent\textbf{[Step 7]}   Using (18) to simplify \eqref{eq:16} yields
\[
\begin{aligned}
\E_{k}\!\bigl[f(\mathbf w^{k+1})\bigr]
\le &\; f(\mathbf w^{k})
   -\alpha\!\sum_{i}\!\mathbf v^{k}_{i} g_i(\mathbf w^{k})
   -\alpha\eta\!\sum_{i}p_i g_i(\mathbf w^{k})^{2}\\
   &\;+\frac{\alpha^{2}}{2}\!\sum_{i}p_i L_i(\mathbf v^{k}_{i})^{2}
   +\alpha^{2}\beta\eta\!\sum_{i}p_i L_i \mathbf v^{k}_{i} g_i(\mathbf w^{k})
   +\frac{\alpha^{2}\eta^{2}}{2}\!\sum_{i}p_i L_i g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{19}
\]

\noindent\textbf{[Step 8]}   Form the Lyapunov function \eqref{eq:lyapunov}
with $\gamma=\dfrac{\alpha(1-\beta)}{\eta}$.  Adding $\frac{\gamma}{2}$ times
the expectation of \eqref{eq:12} to \eqref{eq:19} gives
\[
\begin{aligned}
\Phi_{k+1}
=&\E_{k}\!\bigl[f(\mathbf w^{k+1})\bigr]
   +\frac{\gamma}{2}\E_{k}\!\bigl[\|\mathbf v^{k+1}\|^{2}\bigr]\\[4pt]
\le &\; f(\mathbf w^{k})
   -\alpha\!\sum_{i}\!\mathbf v^{k}_{i} g_i(\mathbf w^{k})
   -\alpha\eta\!\sum_{i}p_i g_i(\mathbf w^{k})^{2}\\
   &\;+\frac{\alpha^{2}}{2}\!\sum_{i}p_i L_i(\mathbf v^{k}_{i})^{2}
   +\alpha^{2}\beta\eta\!\sum_{i}p_i L_i \mathbf v^{k}_{i} g_i(\mathbf w^{k})
   +\frac{\alpha^{2}\eta^{2}}{2}\!\sum_{i}p_i L_i g_i(\mathbf w^{k})^{2}\\[4pt]
   &\;+\frac{\gamma}{2}\Bigl[
      \|\mathbf v^{k}\|^{2}
      -(1-\beta^{2})\!\sum_{i}p_i\|\mathbf v^{k}_{i}\|^{2}
      +2\beta\eta\!\sum_{i}p_i\mathbf v^{k}_{i} g_i(\mathbf w^{k})
      +\eta^{2}\!\sum_{i}p_i g_i(\mathbf w^{k})^{2}
   \Bigr].
\end{aligned}
\tag{20}
\]

\noindent\textbf{[Step 9]}   Collect terms w.r.t. $\mathbf v^{k}_{i}g_i(\mathbf w^{k})$.
The coefficient of $\sum_i p_i \mathbf v^{k}_{i} g_i$ becomes
\[
-\alpha +\alpha^{2}\beta\eta +\gamma\beta\eta
= -\alpha +\eta\beta\bigl(\alpha^{2}+\gamma\bigr).
\]
Since $\gamma=\frac{\alpha(1-\beta)}{\eta}$,
\[
\alpha^{2}+\gamma
= \alpha^{2}+ \frac{\alpha(1-\beta)}{\eta}
= \frac{\alpha}{\eta}\bigl(\alpha\eta+1-\beta\bigr).
\]
Thus the coefficient simplifies to
\[
-\alpha +\beta\alpha\bigl(\alpha\eta+1-\beta\bigr)
= -\alpha\bigl[1-\beta(1-\beta)-\beta\alpha\eta\bigr]
\le 0,
\]
because $0\le\beta<1$ and $\alpha\eta\ge0$.  Hence this mixed term is
non‑positive and can be discarded.

\noindent\textbf{[Step 10]}   Collect the quadratic terms in $\mathbf v^{k}_{i}$.
Their total coefficient equals
\[
\frac{\alpha^{2}}{2}p_iL_i -\frac{\gamma}{2}(1-\beta^{2})p_i
= \frac{p_i}{2}\Bigl(\alpha^{2}L_i-\gamma(1-\beta^{2})\Bigr).
\]
Using $\gamma=\frac{\alpha(1-\beta)}{\eta}$ and the stepsize bound
$\alpha\le\frac{2(1-\beta)}{\bar L}\le\frac{2(1-\beta)}{L_i}$ we obtain
\[
\alpha^{2}L_i\le 2\alpha(1-\beta)
\quad\Longrightarrow\quad
\alpha^{2}L_i-\gamma(1-\beta^{2})
\le 2\alpha(1-\beta)-\frac{\alpha(1-\beta)}{\eta}(1-\beta^{2})
= \alpha(1-\beta)\Bigl(2-\frac{1-\beta^{2}}{\eta}\Bigr).
\]
Choosing $\eta\ge1$ guarantees the right‑hand side is non‑positive,
hence the whole coefficient is $\le0$.  Therefore the $\|\mathbf v^{k}\|^{2}$
contribution can also be dropped.

\noindent\textbf{[Step 11]}   What remains in \eqref{eq:20} are the terms
containing $g_i(\mathbf w^{k})^{2}$:
\[
\begin{aligned}
\Phi_{k+1}
\le &\; f(\mathbf w^{k})
   -\alpha\eta\!\sum_{i}p_i g_i(\mathbf w^{k})^{2}
   +\frac{\alpha^{2}\eta^{2}}{2}\!\sum_{i}p_i L_i g_i(\mathbf w^{k})^{2}
   +\frac{\gamma\eta^{2}}{2}\!\sum_{i}p_i g_i(\mathbf w^{k})^{2}\\[4pt]
=&\; f(\mathbf w^{k})
   -\underbrace{\Bigl[\alpha\eta-\frac{\alpha^{2}\eta^{2}}{2}\bar L
                -\frac{\gamma\eta^{2}}{2}\Bigr]}_{\displaystyle\theta}
      \sum_{i}p_i g_i(\mathbf w^{k})^{2}.
\end{aligned}
\tag{21}
\]

\noindent\textbf{[Step 12]}   Substitute $\gamma=\frac{\alpha(1-\beta)}{\eta}$ and
simplify $\theta$:
\[
\theta
= \alpha\eta\Bigl[1-\frac{\alpha\eta}{2}\bar L-\frac{1-\beta}{2}\Bigr]
= \frac{\alpha\eta}{2}\Bigl[1-\beta-\alpha\eta\bar L\Bigr].
\tag{22}
\]
Because of the stepsize restriction \eqref{eq:step},
$\alpha\eta\bar L\le 2(1-\beta)$, thus $\theta\ge\frac{\alpha\eta}{2}(1-\beta)$.
Consequently,
\[
\Phi_{k+1}\le f(\mathbf w^{k})-\frac{\alpha\eta}{2}(1-\beta)
         \sum_{i}p_i g_i(\mathbf w^{k})^{2}.
\tag{23}
\]

\noindent\textbf{[Step 13]}   Relate the weighted gradient norm to the
function suboptimality.  Strong convexity (\ref{eq:strong}) implies
\[
\|\nabla f(\mathbf w^{k})\|^{2}
= \Bigl\|\sum_{i}g_i(\mathbf w^{k})\mathbf e_i\Bigr\|^{2}
\ge 2\mu\bigl(f(\mathbf w^{k})-f(\mathbf w^{\star})\bigr).
\tag{24}
\]
Moreover, $\sum_{i}p_i g_i(\mathbf w^{k})^{2}\ge p_{\min}\|\nabla f(\mathbf w^{k})\|^{2}$
with $p_{\min}:=\min_i p_i>0$.
Combining (24) with (23) gives
\[
\Phi_{k+1}
\le f(\mathbf w^{k})-\frac{\alpha\eta}{2}(1-\beta)p_{\min}\,
      2\mu\bigl(f(\mathbf w^{k})-f(\mathbf w^{\star})\bigr)
= \bigl(1-\rho\bigr)\bigl(f(\mathbf w^{k})-f(\mathbf w^{\star})\bigr),
\tag{25}
\]
where
\[
\rho:=\alpha\eta\mu p_{\min}(1-\beta)\in(0,1).
\tag{26}
\]

\noindent\textbf{[Step 14]}   Adding the non‑negative term
$\frac{\gamma}{2}\E_{k}[\|\mathbf v^{k+1}\|^{2}]$ to both sides of (25) yields
\[
\Phi_{k+1}\le (1-\rho)\Phi_{k}.
\tag{27}
\]
Iterating (27) $T$ times and taking total expectation proves
\eqref{eq:rate}.
\hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} and the definition of $\rho$ in \eqref{eq:rho}
we obtain the explicit linear rate
\[
\E\!\bigl[f(\mathbf w^{T})-f(\mathbf w^{\star})\bigr]
\le
\Bigl(1-\alpha\eta\mu p_{\min}(1-\beta)\Bigr)^{T}
\Bigl(f(\mathbf w^{0})-f(\mathbf w^{\star})+\tfrac{\gamma}{2}\|\mathbf v^{0}\|^{2}\Bigr).
\tag{28}
\]
Choosing the maximal admissible stepsize
$\alpha = \frac{2(1-\beta)}{\bar L}$ and $\eta=1$ yields
\[
\rho = \frac{2\mu p_{\min}(1-\beta)^{2}}{\bar L},
\qquad
\text{so } \; (1-\rho)^{T}\approx \exp\!\bigl(-T\rho\bigr).
\tag{29}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}  The rate reduces to
$\rho = \alpha\eta\mu p_{\min}$, coinciding with the classical
randomized coordinate descent bound.
\item \textbf{Uniform sampling.}  If $p_i=1/d$, then $p_{\min}=1/d$ and
$\bar L=\frac{1}{d}\sum_i L_i$, so
$\rho = \frac{\alpha\eta\mu}{d}(1-\beta)$.
\item \textbf{Deterministic cyclic selection.}  Setting $p_i=1$ for a
single $i$ and $0$ otherwise recovers the heavy‑ball method applied to
the univariate subproblem; the theorem still guarantees linear convergence
with the same $\rho$.
\end{itemize}
