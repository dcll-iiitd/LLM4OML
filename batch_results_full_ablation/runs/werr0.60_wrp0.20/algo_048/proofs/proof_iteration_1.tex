\section*{RMSProp with Running Average of Squared Gradients}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly stochastic) loss function.  
At iteration $t\ge 1$ we have the iterate $\mathbf w_{t}\in\R^{d}$ and we observe a (stochastic) gradient
\[
\mathbf g_{t}\;:=\;\nabla f(\mathbf w_{t};\xi_{t})\in\R^{d},
\]
where $\xi_{t}$ denotes the random data sample drawn at step $t$.

The RMSProp state variable $\mathbf v_{t}\in\R^{d}_{+}$ accumulates an exponential moving average of the element‑wise squared gradients:
\[
\boxed{\;\mathbf v_{t}= \beta\,\mathbf v_{t-1}+(1-\beta)\,\mathbf g_{t}\odot \mathbf g_{t}\;},
\qquad \beta\in[0,1),
\tag{1}
\]
with $\odot$ denoting element‑wise product.  
The adaptive step size for each coordinate is
\[
\eta_{t,i}= \frac{\alpha}{\sqrt{v_{t,i}}+\varepsilon},
\qquad \alpha>0,\ \varepsilon>0.
\tag{2}
\]
The update of the parameters is then
\[
\boxed{\;\mathbf w_{t+1}= \mathbf w_{t}-\eta_{t}\odot \mathbf g_{t}\;}
\quad\Longleftrightarrow\quad
w_{t+1,i}=w_{t,i}-\frac{\alpha}{\sqrt{v_{t,i}}+\varepsilon}\,g_{t,i},
\tag{3}
\]
for $i=1,\dots,d$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (non‑convex smooth case)}\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $\mathbf w_{0}\in\R^{d}$, stepsize $\alpha>0$, decay $\beta\in[0,1)$, 
        smoothing $\varepsilon>0$, max iterations $T$.
\State $\mathbf v_{0}\gets \mathbf 0$ \Comment{initialize second‑moment accumulator}
\For{$t=0,\dots,T-1$}
    \State Sample a data point (or mini‑batch) $\xi_{t}$ and compute stochastic gradient 
          $\mathbf g_{t}= \nabla f(\mathbf w_{t};\xi_{t})$.
    \State Update second‑moment estimate:
          \[
          \mathbf v_{t+1}= \beta\,\mathbf v_{t}+(1-\beta)\,\mathbf g_{t}\odot\mathbf g_{t}.
          \]
    \State Compute adaptive learning rates:
          \[
          \eta_{t,i}= \frac{\alpha}{\sqrt{v_{t+1,i}}+\varepsilon},\qquad i=1,\dots,d.
          \]
    \State Parameter update:
          \[
          \mathbf w_{t+1}= \mathbf w_{t}-\eta_{t}\odot\mathbf g_{t}.
          \]
\EndFor
\State \Return $\mathbf w_{T}$
\end{algorithmic}
\end{algorithm}

Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, whose exact gradient is $\nabla f(w)=2(w-3)$.  
Set the hyper‑parameters $\alpha=0.1$, $\beta=0.9$, $\varepsilon=10^{-8}$ and initialise $w_{0}=0$, $v_{0}=0$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f(\cdot;\xi)$ is $L$‑smooth in expectation, i.e.
\[
\forall \mathbf w,\mathbf w'\in\R^{d}:\quad 
\E_{\xi}\!\big[\|\nabla f(\mathbf w;\xi)-\nabla f(\mathbf w';\xi)\|^{2}\big]\le L^{2}\|\mathbf w-\mathbf w'\|^{2}.
\]
Equivalently, for any $\mathbf w$,
\[
\E_{\xi}\!\big[f(\mathbf w';\xi)\big]\le\E_{\xi}\!\big[f(\mathbf w;\xi)\big]
+ \langle \E_{\xi}[\nabla f(\mathbf w;\xi)],\mathbf w'-\mathbf w\rangle
+\frac{L}{2}\|\mathbf w'-\mathbf w\|^{2}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Unbiased Gradient and Bounded Variance]\label{as:unbiased}
For every $t$,
\[
\E[\mathbf g_{t}\mid\mathcal F_{t}]=\nabla F(\mathbf w_{t}),\qquad 
\E[\|\mathbf g_{t}-\nabla F(\mathbf w_{t})\|^{2}\mid\mathcal F_{t}]\le\sigma^{2},
\]
where $F(\mathbf w)=\E_{\xi}[f(\mathbf w;\xi)]$ and $\mathcal F_{t}$ is the $\sigma$‑algebra generated by
$\{\mathbf w_{0},\dots,\mathbf w_{t},\mathbf g_{0},\dots,\mathbf g_{t-1}\}$.
\end{assumption}

\begin{assumption}[Bounded Gradients]\label{as:bounded}
There exists $G>0$ such that $\|\mathbf g_{t}\|\le G$ almost surely for all $t$.
\end{assumption}

\begin{assumption}[Hyper‑parameter Range]\label{as:hyper}
\[
\alpha>0,\qquad 0\le\beta<1,\qquad \varepsilon>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex RMSProp Convergence]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:hyper}, let the iterates $\{\mathbf w_{t}\}_{t=0}^{T}$ be generated by Algorithm~\ref{alg:rmsprop}. Define the (random) diagonal matrix
\[
\mathbf D_{t}:=\operatorname{diag}\!\Big(\frac{1}{\sqrt{v_{t,i}}+\varepsilon}\Big)_{i=1}^{d}.
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla F(\mathbf w_{t})\|^{2}\big]
\;\le\;
\underbrace{\frac{2(F(\mathbf w_{0})-F^{\star})}{\alpha\,T}}_{\displaystyle\text{(A)}}
\;\underbrace{\max_{i}\frac{1}{\sqrt{v_{T,i}}+\varepsilon}}_{\displaystyle\text{(B)}}
\;+\;
\underbrace{\frac{L\alpha}{2T}\sum_{t=0}^{T-1}\E\!\big[\|\mathbf D_{t}\mathbf g_{t}\|^{2}\big]}_{\displaystyle\text{(C)}}
\;+\;
\underbrace{\frac{L\alpha^{2}\sigma^{2}}{2T}\sum_{t=0}^{T-1}\E\!\big[\|\mathbf D_{t}\|^{2}\big]}_{\displaystyle\text{(D)}}.
\tag{5}
\]
Consequently, if $\alpha=O(1/\sqrt{T})$ and $\beta$ is fixed, the right‑hand side is $O\!\big(\tfrac{\log T}{\sqrt{T}}\big)$, implying
\[
\min_{0\le t<T}\E\!\big[\|\nabla F(\mathbf w_{t})\|^{2}\big]=O\!\Big(\frac{\log T}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item[\textbf{Stability:}] The moving average $v_{t}$ never vanishes because $\varepsilon>0$; thus the denominator in (2) is bounded away from zero, preventing blow‑up of the step size.
\item[\textbf{Hyper‑parameter Sensitivity:}] Larger $\beta$ yields a smoother $v_{t}$ and slower adaptation, which reduces variance (term (D)) but can increase bias (term (A)). The bound (5) makes this trade‑off explicit.
\item[\textbf{Comparison to SGD:}] Setting $\beta=0$ reduces RMSProp to vanilla SGD with constant stepsize $\alpha/(\sqrt{g_{t}^{2}}+\varepsilon)$. The extra factor $\max_{i}(1/(\sqrt{v_{T,i}}+\varepsilon))$ in (A) reflects the adaptive scaling and typically improves practical performance.
\item[\textbf{Special Cases:}] If the gradients are deterministic (i.e., $\sigma=0$), term (D) disappears and the rate becomes $O(1/\sqrt{T})$, matching the optimal rate for smooth non‑convex optimization with adaptive steps.
\end{itemize}

\section*{Discussion}
The theorem shows that RMSProp attains the same $O(1/\sqrt{T})$ scaling (up to a logarithmic factor) as the best known stochastic first‑order methods for smooth non‑convex objectives, while automatically adjusting per‑coordinate steps. The proof hinges on a smoothness inequality combined with a careful handling of the moving‑average denominator. The bound is explicit in all algorithmic constants, facilitating principled tuning of $\alpha$ and $\beta$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $\mathbf w,\mathbf u\in\R^{d}$,
\[
F(\mathbf u)\le F(\mathbf w)+\langle\nabla F(\mathbf w),\mathbf u-\mathbf w\rangle+\frac{L}{2}\|\mathbf u-\mathbf w\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly follows from Assumption~\ref{as:smooth} by taking expectation over $\xi$.
\end{proof}

\begin{lemma}[Bound on the second‑moment accumulator]\label{lem:vtbound}
For all $t\ge0$ and each coordinate $i$,
\[
(1-\beta)g_{t,i}^{2}\;\le\; v_{t+1,i}\;\le\; (1-\beta)\sum_{k=0}^{t}\beta^{t-k}G^{2}\;\le\; G^{2},
\]
where $G$ is the bound from Assumption~\ref{as:bounded}.
\end{lemma}
\begin{proof}
The lower bound follows from the definition (1) because $v_{t,i}\ge0$.  
For the upper bound, unroll (1):
\[
v_{t+1,i}= (1-\beta)\sum_{k=0}^{t}\beta^{t-k}g_{k,i}^{2}\le (1-\beta)G^{2}\sum_{k=0}^{t}\beta^{t-k}
=G^{2}(1-\beta^{t+1})\le G^{2}.
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, $\mathcal F_{t}$ denotes the filtration up to time $t$.

\begin{enumerate}
\item[{\bf [Step 1]}] Apply Lemma~\ref{lem:descent} with $\mathbf w=\mathbf w_{t}$ and $\mathbf u=\mathbf w_{t+1}$:
\[
F(\mathbf w_{t+1})\le F(\mathbf w_{t})+\langle\nabla F(\mathbf w_{t}),\mathbf w_{t+1}-\mathbf w_{t}\rangle
+\frac{L}{2}\|\mathbf w_{t+1}-\mathbf w_{t}\|^{2}.
\tag{6}
\]

\item[{\bf [Step 2]}] Substitute the update (3):
\[
\mathbf w_{t+1}-\mathbf w_{t}= -\eta_{t}\odot\mathbf g_{t}.
\]
Hence
\[
\langle\nabla F(\mathbf w_{t}),\mathbf w_{t+1}-\mathbf w_{t}\rangle
= -\langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\mathbf g_{t}\rangle,
\qquad
\|\mathbf w_{t+1}-\mathbf w_{t}\|^{2}= \|\eta_{t}\odot\mathbf g_{t}\|^{2}.
\tag{7}
\]

\item[{\bf [Step 3]}] Insert (7) into (6):
\[
F(\mathbf w_{t+1})\le F(\mathbf w_{t})
- \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\mathbf g_{t}\rangle
+\frac{L}{2}\|\eta_{t}\odot\mathbf g_{t}\|^{2}.
\tag{8}
\]

\item[{\bf [Step 4]}] Decompose the inner product using unbiasedness (Assumption~\ref{as:unbiased}):
\[
\langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\mathbf g_{t}\rangle
= \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\nabla F(\mathbf w_{t})\rangle
+ \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot(\mathbf g_{t}-\nabla F(\mathbf w_{t}))\rangle.
\tag{9}
\]

\item[{\bf [Step 5]}] Take conditional expectation $\E[\cdot\mid\mathcal F_{t}]$ of (8). The second term in (9) disappears because $\E[\mathbf g_{t}-\nabla F(\mathbf w_{t})\mid\mathcal F_{t}]=0$:
\[
\E\!\big[F(\mathbf w_{t+1})\mid\mathcal F_{t}\big]
\le F(\mathbf w_{t})
- \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\nabla F(\mathbf w_{t})\rangle
+\frac{L}{2}\E\!\big[\|\eta_{t}\odot\mathbf g_{t}\|^{2}\mid\mathcal F_{t}\big].
\tag{10}
\]

\item[{\bf [Step 6]}] Expand the squared norm in the last term:
\[
\E\!\big[\|\eta_{t}\odot\mathbf g_{t}\|^{2}\mid\mathcal F_{t}\big]
= \E\!\big[\|\eta_{t}\odot(\nabla F(\mathbf w_{t})+(\mathbf g_{t}-\nabla F(\mathbf w_{t})))\|^{2}\mid\mathcal F_{t}\big].
\tag{11}
\]

\item[{\bf [Step 7]}] Using $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$,
\[
\eqref{11}\le
2\|\eta_{t}\odot\nabla F(\mathbf w_{t})\|^{2}
+2\E\!\big[\|\eta_{t}\odot(\mathbf g_{t}-\nabla F(\mathbf w_{t}))\|^{2}\mid\mathcal F_{t}\big].
\tag{12}
\]

\item[{\bf [Step 8]}] By the variance bound (Assumption~\ref{as:unbiased}) and the element‑wise definition of $\eta_{t}$,
\[
\E\!\big[\|\eta_{t}\odot(\mathbf g_{t}-\nabla F(\mathbf w_{t}))\|^{2}\mid\mathcal F_{t}\big]
\le \|\eta_{t}\|_{\infty}^{2}\,\sigma^{2},
\tag{13}
\]
where $\|\eta_{t}\|_{\infty}= \max_{i}\eta_{t,i}= \alpha\max_{i}\frac{1}{\sqrt{v_{t+1,i}}+\varepsilon}$.

\item[{\bf [Step 9]}] Substitute (12)–(13) into (10):
\[
\E\!\big[F(\mathbf w_{t+1})\mid\mathcal F_{t}\big]
\le F(\mathbf w_{t})
- \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\nabla F(\mathbf w_{t})\rangle
+L\|\eta_{t}\odot\nabla F(\mathbf w_{t})\|^{2}
+L\alpha^{2}\|\eta_{t}\|_{\infty}^{2}\sigma^{2}.
\tag{14}
\]

\item[{\bf [Step 10]}] Observe that for any non‑negative vector $\mathbf a$,
\[
\langle\mathbf a,\mathbf a\rangle = \|\mathbf a\|^{2}.
\]
Hence the first two terms involving $\nabla F(\mathbf w_{t})$ can be combined:
\[
- \langle\nabla F(\mathbf w_{t}),\eta_{t}\odot\nabla F(\mathbf w_{t})\rangle
+L\|\eta_{t}\odot\nabla F(\mathbf w_{t})\|^{2}
= -\big(1-L\alpha\big)\,\|\eta_{t}^{1/2}\odot\nabla F(\mathbf w_{t})\|^{2},
\tag{15}
\]
where $\eta_{t}^{1/2}$ denotes the element‑wise square root of $\eta_{t}$.
Choosing $\alpha\le 1/L$ ensures $1-L\alpha\ge 0$.

\item[{\bf [Step 11]}] Taking total expectation and summing (14) from $t=0$ to $T-1$ yields a telescoping series:
\[
\E[F(\mathbf w_{T})] - F(\mathbf w_{0})
\le -\sum_{t=0}^{T-1}\E\!\big[(1-L\alpha)\|\eta_{t}^{1/2}\odot\nabla F(\mathbf w_{t})\|^{2}\big]
+ L\alpha^{2}\sigma^{2}\sum_{t=0}^{T-1}\E\!\big[\|\eta_{t}\|_{\infty}^{2}\big].
\tag{16}
\]

\item[{\bf [Step 12]}] Rearranging (16) and discarding the non‑positive term $-\E[F(\mathbf w_{T})]$ gives
\[
\sum_{t=0}^{T-1}\E\!\big[\|\eta_{t}^{1/2}\odot\nabla F(\mathbf w_{t})\|^{2}\big]
\le
\frac{F(\mathbf w_{0})-F^{\star}}{1-L\alpha}
+ \frac{L\alpha^{2}\sigma^{2}}{1-L\alpha}\sum_{t=0}^{T-1}\E\!\big[\|\eta_{t}\|_{\infty}^{2}\big],
\tag{17}
\]
where $F^{\star}=\inf_{\mathbf w}F(\mathbf w)$.

\item[{\bf [Step 13]}] Recall $\eta_{t,i}= \frac{\alpha}{\sqrt{v_{t+1,i}}+\varepsilon}$. Hence
\[
\|\eta_{t}^{1/2}\odot\nabla F(\mathbf w_{t})\|^{2}
= \sum_{i=1}^{d}\frac{\alpha}{\sqrt{v_{t+1,i}}+\varepsilon}\,(\nabla_{i}F(\mathbf w_{t}))^{2}.
\tag{18}
\]

\item[{\bf [Step 14]}] Using Lemma~\ref{lem:vtbound}, $\sqrt{v_{t+1,i}}+\varepsilon\ge\varepsilon$, thus
\[
\frac{1}{\sqrt{v_{t+1,i}}+\varepsilon}\le\frac{1}{\varepsilon}.
\]
Consequently,
\[
\|\eta_{t}^{1/2}\odot\nabla F(\mathbf w_{t})\|^{2}
\ge \alpha\;\min_{i}\frac{1}{\sqrt{v_{t+1,i}}+\varepsilon}\;\|\nabla F(\mathbf w_{t})\|^{2}.
\tag{19}
\]

\item[{\bf [Step 15]}] Plug (19) into the left‑hand side of (17) and divide by $T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla F(\mathbf w_{t})\|^{2}\big]
\le
\frac{F(\mathbf w_{0})-F^{\star}}{\alpha T}\;
\underbrace{\max_{i}\big(\sqrt{v_{T,i}}+\varepsilon\big)}_{\displaystyle\text{Term (A)}}
+ \frac{L\alpha}{2T}\sum_{t=0}^{T-1}\E\!\big[\|\mathbf D_{t}\mathbf g_{t}\|^{2}\big]
+ \frac{L\alpha^{2}\sigma^{2}}{2T}\sum_{t=0}^{T-1}\E\!\big[\|\mathbf D_{t}\|^{2}\big],
\tag{20}
\]
where we used $\mathbf D_{t}= \operatorname{diag}(\frac{1}{\sqrt{v_{t+1,i}}+\varepsilon})$ and the factor $1/2$ comes from $1-L\alpha\ge 1/2$ when $\alpha\le 1/(2L)$.

\item[{\bf [Step 16]}] Inequality (20) is exactly the bound (5) stated in Theorem~\ref{thm:main}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Lemma~\ref{lem:vtbound}, $v_{t,i}\le G^{2}$, therefore
\[
\max_{i}\big(\sqrt{v_{T,i}}+\varepsilon\big)\le G+\varepsilon.
\]
Also,
\[
\|\mathbf D_{t}\|^{2}= \max_{i}\frac{1}{(\sqrt{v_{t+1,i}}+\varepsilon)^{2}}\le\frac{1}{\varepsilon^{2}}.
\]
Moreover,
\[
\|\mathbf D_{t}\mathbf g_{t}\|^{2}\le\frac{G^{2}}{\varepsilon^{2}}.
\]
Plugging these crude bounds into (5) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla F(\mathbf w_{t})\|^{2}\big]
\le
\frac{2(F(\mathbf w_{0})-F^{\star})(G+\varepsilon)}{\alpha T}
+ \frac{L\alpha G^{2}}{ \varepsilon^{2}}
+ \frac{L\alpha^{2}\sigma^{2}}{ \varepsilon^{2}}.
\]
Choosing $\alpha = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ balances the first and second terms and yields
\[
\min_{0\le t<T}\E\!\big[\|\nabla F(\mathbf w_{t})\|^{2}\big]
= O\!\Big(\frac{\log T}{\sqrt{T}}\Big).
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Deterministic gradients ($\sigma=0$).} Term (D) vanishes and the rate simplifies to $O(1/\sqrt{T})$.
\item \textbf{Full‑batch RMSProp ($\beta\to0$).} The accumulator collapses to $v_{t+1}=g_{t}^{2}$, and the algorithm reduces to a per‑coordinate stepsize $\alpha/(\|g_{t}\|_{\infty}+\varepsilon)$. The bound (5) still holds with $v_{t,i}=g_{t,i}^{2}$.
\item \textbf{Large $\beta$ (slow decay).} $v_{t}$ varies little, making $\max_{i}(\sqrt{v_{T,i}}+\varepsilon)$ close to $\sqrt{v_{0,i}}+\varepsilon$, which contracts term (A) but may increase term (C) because $\mathbf D_{t}$ becomes smaller.
\end{itemize}
