\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimisation problem  
\[
\min_{x\in\R^{d}}\; F(x)\;:=\;f(x)+g(x),
\]
where  

\begin{itemize}[noitemsep]
\item $f:\R^{d}\to\R$ is convex and differentiable with $L$‑Lipschitz continuous gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d};
\]
\item $g:\R^{d}\to\R\cup\{+\infty\}$ is convex, possibly non‑smooth, and its proximal operator
\[
\prox_{\eta g}(u):=\arg\min_{z\in\R^{d}}\Big\{ g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Big\}
\]
is easy to compute for any stepsize $\eta>0$.
\end{itemize}

Given a stepsize $\eta\in(0,1/L]$, the Proximal Gradient Method generates the sequence $\{x_k\}$ by  
\[
\boxed{\,x_{k+1}= \prox_{\eta g}\bigl(x_k-\eta\nabla f(x_k)\bigr)\,},\qquad k=0,1,\dots
\tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^{d}$, stepsize $\eta\in(0,1/L]$, maximal iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_k\gets \nabla f(x_k)$.
    \State Take gradient step $y_k\gets x_k-\eta g_k$.
    \State Apply proximal map $x_{k+1}\gets \prox_{\eta g}(y_k)$.
\EndFor
\State \textbf{output} $x_K$ (or the best iterate seen).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimise $f(w)=(w-3)^2$ with $g\equiv0$ (hence $\prox_{\eta g}= \operatorname{id}$).  
Take $w_0=0$, stepsize $\eta=0.1$, and perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta\nabla f(w_k)$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values: $F(w_0)=9$, $F(w_1)=5.76$, $F(w_2)=3.6864$, $F(w_3)=2.3660$, showing monotone decrease toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:lip}
$f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{as:conv}
$g$ is proper, closed, convex (possibly non‑smooth). Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Existence of a minimiser]\label{as:opt}
The set of optimal solutions $\mathcal{X}^\star:=\arg\min_{x}F(x)$ is non‑empty; pick any $x^\star\in\mathcal{X}^\star$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
$\eta\in(0,1/L]$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Under Assumptions \ref{as:lip}–\ref{as:step}, the iterates $\{x_k\}$ generated by \eqref{1} satisfy for all $k\ge1$
\[
F(x_k)-F(x^\star)\;\le\;\frac{\|x_0-x^\star\|^{2}}{2\eta k}.
\tag{2}
\]
Consequently $F(x_k)\to F(x^\star)$ with rate $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[noitemsep]
\item \textbf{Stability.} The update \eqref{1} is a non‑expansive mapping when $\eta\le 1/L$, which guarantees that the distance to the solution set never increases.
\item \textbf{Hyperparameter Sensitivity.} Choosing $\eta$ too large ($\eta>1/L$) may break the descent property; any $\eta\in(0,1/L]$ yields the $O(1/k)$ guarantee.
\item \textbf{Comparison.} When $g\equiv0$, \eqref{1} reduces to standard gradient descent with the classical $O(1/k)$ rate. When $g$ is the $\ell_1$ norm, the method becomes the Iterative Soft‑Thresholding Algorithm (ISTA) with the same rate.
\end{enumerate}

\section*{Discussion}
The proof rests on two elementary facts:
\begin{itemize}[noitemsep]
\item \emph{Descent Lemma} for $f$ (smoothness).
\item \emph{Non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{3}
\]
Combining these yields a recursive inequality on $\|x_{k+1}-x^\star\|^{2}$, from which \eqref{2} follows by telescoping.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; omitted for brevity. \hfill $\square$
\end{proof}
\begin{lemma}[Moreau’s decomposition of the proximal map]\label{lem:prox_opt}
For any $u\in\R^{d}$ and $\eta>0$,
\[
\prox_{\eta g}(u)\;\in\;\arg\min_{z}\Big\{ g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Big\}
\quad\Longleftrightarrow\quad
\frac{1}{\eta}\bigl(u-\prox_{\eta g}(u)\bigr)\in\partial g\bigl(\prox_{\eta g}(u)\bigr).
\tag{5}
\]
\end{lemma}
\begin{proof}
First‑order optimality condition of the proximal subproblem. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^\star\in\mathcal{X}^\star$ be arbitrary but fixed. Define the gradient step
\[
y_k:=x_k-\eta\nabla f(x_k).
\tag{6}
\]
By definition of the algorithm,
\[
x_{k+1}= \prox_{\eta g}(y_k).
\tag{7}
\]

\begin{enumerate}[label={[\textbf{Step \arabic*}]}]
\item \textbf{Expand the squared distance.}
\[
\|x_{k+1}-x^\star\|^{2}
= \|\prox_{\eta g}(y_k)-\prox_{\eta g}(x^\star-\eta\nabla f(x^\star))\|^{2}
\tag{8}
\]
where we used that $x^\star$ is optimal, hence $0\in\nabla f(x^\star)+\partial g(x^\star)$, i.e. $x^\star=\prox_{\eta g}(x^\star-\eta\nabla f(x^\star))$ (by Lemma~\ref{lem:prox_opt}).

\item \textbf{Apply non‑expansiveness (3).}
\[
\|x_{k+1}-x^\star\|^{2}\le \|y_k-(x^\star-\eta\nabla f(x^\star))\|^{2}
= \|x_k-\eta\nabla f(x_k)-x^\star+\eta\nabla f(x^\star)\|^{2}.
\tag{9}
\]

\item \textbf{Expand the RHS.}
\[
\begin{aligned}
\|x_k-x^\star\|^{2}
&-2\eta\langle\nabla f(x_k)-\nabla f(x^\star),\,x_k-x^\star\rangle
+\eta^{2}\|\nabla f(x_k)-\nabla f(x^\star)\|^{2}.
\end{aligned}
\tag{10}
\]

\item \textbf{Use convexity of $f$.}
Convexity gives
\[
\langle\nabla f(x_k)-\nabla f(x^\star),\,x_k-x^\star\rangle\ge f(x_k)-f(x^\star).
\tag{11}
\]

\item \textbf{Use Lipschitz bound on the gradient.}
From $L$‑Lipschitzness,
\[
\|\nabla f(x_k)-\nabla f(x^\star)\|^{2}\le L^{2}\|x_k-x^\star\|^{2}.
\tag{12}
\]

\item \textbf{Combine (9)–(12).}
\[
\|x_{k+1}-x^\star\|^{2}
\le \|x_k-x^\star\|^{2}
-2\eta\bigl(f(x_k)-f(x^\star)\bigr)
+\eta^{2}L^{2}\|x_k-x^\star\|^{2}.
\tag{13}
\]

\item \textbf{Collect terms.}
Since $\eta\le 1/L$, we have $\eta^{2}L^{2}\le\eta L\le 1$, thus
\[
\|x_{k+1}-x^\star\|^{2}
\le \bigl(1-\eta L\bigr)\|x_k-x^\star\|^{2}
-2\eta\bigl(f(x_k)-f(x^\star)\bigr).
\tag{14}
\]

\item \textbf{Add $g$ terms using optimality of $x^\star$.}
Because $0\in\nabla f(x^\star)+\partial g(x^\star)$, we have the subgradient inequality
\[
g(x_{k+1})-g(x^\star)\le \langle s_{k+1},\,x_{k+1}-x^\star\rangle,
\quad s_{k+1}\in\partial g(x_{k+1}).
\tag{15}
\]
From Lemma~\ref{lem:prox_opt} with $u=y_k$, we obtain
\[
\frac{1}{\eta}\bigl(y_k-x_{k+1}\bigr)\in\partial g(x_{k+1}),
\]
hence we can set $s_{k+1}= \frac{1}{\eta}(y_k-x_{k+1})$.
Plugging into (15) yields
\[
g(x_{k+1})-g(x^\star)\le \frac{1}{\eta}\langle y_k-x_{k+1},\,x_{k+1}-x^\star\rangle.
\tag{16}
\]

\item \textbf{Rewrite the inner product.}
Using $y_k=x_k-\eta\nabla f(x_k)$,
\[
\begin{aligned}
\langle y_k-x_{k+1},\,x_{k+1}-x^\star\rangle
&=\langle x_k-x_{k+1},\,x_{k+1}-x^\star\rangle
-\eta\langle\nabla f(x_k),\,x_{k+1}-x^\star\rangle.
\end{aligned}
\tag{17}
\]

\item \textbf{Bound the first term.}
By the identity $2\langle a,b\rangle =\|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$,
\[
2\langle x_k-x_{k+1},\,x_{k+1}-x^\star\rangle
= \|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}-\|x_k-x_{k+1}\|^{2}.
\tag{18}
\]

\item \textbf{Combine (16)–(18).}
Dividing (18) by $2\eta$ and inserting into (16) gives
\[
\begin{aligned}
g(x_{k+1})-g(x^\star)
\le &\frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}-\|x_k-x_{k+1}\|^{2}\Bigr)\\
&-\langle\nabla f(x_k),\,x_{k+1}-x^\star\rangle .
\end{aligned}
\tag{19}
\]

\item \textbf{Add $f$ to both sides.}
Add $f(x_{k+1})-f(x^\star)$ to (19). Using convexity of $f$,
\[
f(x_{k+1})-f(x^\star)\le \langle\nabla f(x_k),\,x_{k+1}-x^\star\rangle
+\frac{L}{2}\|x_{k+1}-x_k\|^{2},
\tag{20}
\]
which follows from Lemma~\ref{lem:descent} with $x=x_k$, $y=x_{k+1}$.

\item \textbf{Cancel the gradient inner product.}
Adding (20) to (19) eliminates $\langle\nabla f(x_k),x_{k+1}-x^\star\rangle$:
\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&\le \frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}-\|x_k-x_{k+1}\|^{2}\Bigr)\\
&\qquad +\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\end{aligned}
\tag{21}
\]

\item \textbf{Use the stepsize condition $\eta\le 1/L$.}
Since $\eta\le 1/L$, we have $L/2\le 1/(2\eta)$. Hence the two terms involving $\|x_{k+1}-x_k\|^{2}$ combine to a non‑positive quantity:
\[
-\frac{1}{2\eta}\|x_k-x_{k+1}\|^{2}+ \frac{L}{2}\|x_{k+1}-x_k\|^{2}\le 0.
\tag{22}
\]

\item \textbf{Derive the recursive inequality.}
From (21) and (22),
\[
F(x_{k+1})-F(x^\star)\le \frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}\Bigr).
\tag{23}
\]

\item \textbf{Telescoping sum.}
Sum (23) over $k=0,\dots, K-1$:
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^\star)\bigr)
\le \frac{1}{2\eta}\bigl(\|x_0-x^\star\|^{2}-\|x_{K}-x^\star\|^{2}\bigr)
\le \frac{\|x_0-x^\star\|^{2}}{2\eta}.
\tag{24}
\]

\item \textbf{Extract the bound for a single iterate.}
Since each term on the left of (24) is non‑negative,
\[
K\bigl(F(x_{K})-F(x^\star)\bigr)\le \frac{\|x_0-x^\star\|^{2}}{2\eta},
\]
which is equivalent to the rate (2):
\[
F(x_{K})-F(x^\star)\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}.
\tag{25}
\]
\end{enumerate}

Thus Theorem~\ref{thm:rate} is proved. \hfill $\square$

\section*{Rate Derivation (With Constants)}
The constant in \eqref{2} is explicitly $\frac{1}{2\eta}$.
If we choose the maximal admissible stepsize $\eta=1/L$, the bound becomes
\[
F(x_k)-F(x^\star)\le \frac{L\|x_0-x^\star\|^{2}}{2k}.
\tag{26}
\]

\section*{Special Cases}
\begin{itemize}[noitemsep]
\item \textbf{Pure gradient descent} ($g\equiv0$): $\prox_{\eta g}$ is the identity, and the proof reduces to the classical $O(1/k)$ result for smooth convex $f$.
\item \textbf{Lasso (elastic net)}: $g(x)=\lambda\|x\|_{1}$, $\prox_{\eta g}$ is soft‑thresholding. The same $O(1/k)$ rate holds without any modification.
\item \textbf{Strongly convex $F$}: If $F$ is $\mu$‑strongly convex, a refined analysis yields a linear rate $O\bigl((1-\eta\mu)^{k}\bigr)$.
\end{itemize}
