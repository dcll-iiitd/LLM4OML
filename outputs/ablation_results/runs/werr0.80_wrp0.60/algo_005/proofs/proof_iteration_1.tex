\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) – Double Loop Formulation}

\section*{Mathematical Formulation}
Let 
\[
F(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad f_i:\mathbb{R}^d\to\mathbb{R}
\]
be the empirical risk.  
SVRG maintains a \emph{snapshot} point $\tilde{x}$ (updated once per outer epoch) and, for $m$ inner iterations, generates the variance‑reduced stochastic gradient
\[
v_t \;=\; \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x})+\nabla F(\tilde{x}),\qquad i_t\stackrel{\text{i.i.d.}}{\sim}\{1,\dots,n\}.
\]
The inner update is
\[
x_{t+1}=x_t-\eta\, v_t ,\qquad t=0,\dots ,m-1,
\]
with stepsize $\eta>0$.  
After the $m$ inner steps we set the next snapshot
\[
\tilde{x}^{\;s+1}=x_m .
\]
The algorithm repeats for $S$ outer epochs.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (double‑loop)}\label{alg:svrg}
\begin{algorithmic}[1]
\Require Number of outer epochs $S$, inner length $m$, stepsize $\eta>0$, initial point $x^{0}\in\mathbb{R}^d$
\State $\tilde{x}^{0}\gets x^{0}$
\For{$s=0,\dots,S-1$}
    \State Compute the \emph{full} gradient at the snapshot:
    \[
    \tilde{\mu}\gets \nabla F(\tilde{x}^{\,s})=\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})
    \]
    \State $x_0\gets \tilde{x}^{\,s}$
    \For{$t=0,\dots,m-1$}
        \State Sample $i_t\in\{1,\dots,n\}$ uniformly at random
        \State $v_t\gets \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}$
        \State $x_{t+1}\gets x_t-\eta v_t$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_m$
\EndFor
\State \textbf{output} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
F(w)=\frac12(w-3)^2,\qquad\text{so }f_1(w)=F(w),
\]
with $n=1$ (hence the full gradient equals the stochastic one).  
Take $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, and a single outer epoch $S=1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $i_t$ & $v_t$ & $w_{t+1}=w_t-\eta v_t$ \\ \hline
0 & 1 & $\nabla f_1(w_0)-\nabla f_1(\tilde w)+\nabla F(\tilde w)$ 
   & $0-0.1\cdot(-3)=0.3$ \\[2mm]
  &   & $(-3)-(-3)+(-3)= -3$ & $w_1=0.3$ \\ \hline
1 & 1 & $\nabla f_1(0.3)-\nabla f_1(\tilde w)+\nabla F(\tilde w)$ 
   & $0.3-0.1\cdot(-2.7)=0.57$ \\[2mm]
  &   & $(0.3-3)-(-3)+(-3)= -2.7$ & $w_2=0.57$ \\ \hline
2 & 1 & $\nabla f_1(0.57)-\nabla f_1(\tilde w)+\nabla F(\tilde w)$ 
   & $0.57-0.1\cdot(-2.43)=0.813$ \\[2mm]
  &   & $(0.57-3)-(-3)+(-3)= -2.43$ & $w_3=0.813$ 
\end{tabular}
\end{center}
Objective values:
\[
F(w_0)=4.5,\;F(w_1)=4.41,\;F(w_2)=4.18,\;F(w_3)=3.84.
\]
The iterate moves toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each component $f_i$ is $L$–smooth:
\[
\forall x,y\in\mathbb{R}^d,\qquad
\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y}.
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
$F$ is $\mu$–strongly convex ($0<\mu\le L$):
\[
\forall x,y,\qquad
F(y)\ge F(x)+\inner{\nabla F(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{as:finite}
The objective is a finite average $F(x)=\frac1n\sum_{i=1}^n f_i(x)$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies
\[
0<\eta\le \frac{1}{3L}.
\]
\end{assumption}
\begin{assumption}[Inner Loop Length]\label{as:inner}
The number of inner updates $m$ is a positive integer (commonly $m=2n$).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Under Assumptions~\ref{as:smooth}--\ref{as:inner} and with the stepsize choice~\ref{as:stepsize}, let $\tilde{x}^{\,s}$ be the snapshot after epoch $s$. Define the contraction factor
\[
\rho \;:=\; \frac{1}{\mu\eta(1-2L\eta)m+1}\;\in\;(0,1).
\]
Then for every outer epoch $s\ge0$
\[
\E\!\left[F(\tilde{x}^{\,s+1})-F(x^\star)\right]\;\le\;\rho\;\E\!\left[F(\tilde{x}^{\,s})-F(x^\star)\right].
\]
Consequently,
\[
\E\!\left[F(\tilde{x}^{\,S})-F(x^\star)\right]\;\le\;\rho^{S}\,\bigl[F(\tilde{x}^{\,0})-F(x^\star)\bigr].
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction factor $\rho$ is decreasing in $\eta$ and $m$, showing that larger inner loops and stepsizes (up to the $1/(3L)$ bound) improve the rate.
\item \textbf{Hyper‑parameter sensitivity.} If $\eta$ violates $\eta\le 1/(3L)$, the term $(1-2L\eta)$ becomes non‑positive and the bound may break down; thus the bound prescribes a safe stepsize regime.
\item \textbf{Comparison.} Standard SGD with a diminishing stepsize yields a sub‑linear $O(1/T)$ rate for strongly convex problems, whereas SVRG achieves a geometric rate $O(\rho^{S})$, i.e. linear convergence.
\item \textbf{Special case $m=n$.} When $m=n$, $\rho = 1/(1+\mu\eta(1-2L\eta)n)$, which recovers the classic SVRG rate often quoted as $\rho\approx 1/2$ for $\eta=1/(10L)$ and $n$ large.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $L$‑smooth $f_i$,
\[
f_i(y)\le f_i(x)+\inner{\nabla f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
Let $v_t$ be defined as in the algorithm and let $x^\star$ be the unique minimizer of $F$. Then
\[
\E_{i_t}\!\bigl[\norm{v_t}^2\bigr]\;\le\; 4L\bigl(F(x_t)-F(x^\star)\bigr)+4L\bigl(F(\tilde{x})-F(x^\star)\bigr).
\]
\end{lemma}
\begin{proof}
Because $i_t$ is sampled uniformly,
\[
\E_{i_t}[v_t]=\nabla F(x_t) .
\]
Using $L$‑smoothness and strong convexity one obtains (see e.g. Johnson \& Zhang, 2013) the bound above. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout we condition on the snapshot $\tilde{x}$ and drop the superscript $s$ for readability.

\noindent\textbf{Step 1.}  Expand the distance after one inner update.
\begin{align}
\norm{x_{t+1}-x^\star}^2
&= \norm{x_t-\eta v_t - x^\star}^2 \nonumber\\
&= \norm{x_t-x^\star}^2 -2\eta\inner{v_t}{x_t-x^\star}+\eta^2\norm{v_t}^2 .\label{eq:expand}
\end{align}
Taking expectation w.r.t. $i_t$ and using $\E[v_t]=\nabla F(x_t)$ yields
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^\star}^2\bigr]
&= \norm{x_t-x^\star}^2 -2\eta\inner{\nabla F(x_t)}{x_t-x^\star}
   +\eta^2\E\!\bigl[\norm{v_t}^2\bigr]. \tag{Step 2}
\end{align}

\noindent\textbf{Step 3.}  Replace the inner product via strong convexity.
Strong convexity (\ref{as:strong}) gives
\[
\inner{\nabla F(x_t)}{x_t-x^\star}\;\ge\;F(x_t)-F(x^\star)+\frac{\mu}{2}\norm{x_t-x^\star}^2 .
\]
Insert this into \eqref{eq:expand}:
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^\star}^2\bigr]
&\le \norm{x_t-x^\star}^2 -2\eta\bigl(F(x_t)-F(x^\star)\bigr)-\mu\eta\norm{x_t-x^\star}^2
   +\eta^2\E\!\bigl[\norm{v_t}^2\bigr]. \tag{Step 4}
\end{align}

\noindent\textbf{Step 5.}  Apply the variance bound Lemma~\ref{lem:var}.
Using Lemma~\ref{lem:var},
\[
\E\!\bigl[\norm{v_t}^2\bigr]\le 4L\bigl(F(x_t)-F(x^\star)\bigr)+4L\bigl(F(\tilde{x})-F(x^\star)\bigr).
\]
Plugging this into \eqref{Step 4} gives
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^\star}^2\bigr]
&\le \bigl(1-\mu\eta\bigr)\norm{x_t-x^\star}^2
   -2\eta\bigl(1-2L\eta\bigr)\bigl(F(x_t)-F(x^\star)\bigr)
   +4L\eta^2\bigl(F(\tilde{x})-F(x^\star)\bigr). \tag{Step 6}
\end{align}
Because $\eta\le 1/(3L)$, the coefficient $1-2L\eta$ is positive.

\noindent\textbf{Step 7.}  Relate function sub‑optimality to distance.
From strong convexity,
\[
F(x_t)-F(x^\star)\;\ge\;\frac{\mu}{2}\norm{x_t-x^\star}^2 .
\]
Thus
\[
-2\eta\bigl(1-2L\eta\bigr)\bigl(F(x_t)-F(x^\star)\bigr)
\le -\mu\eta\bigl(1-2L\eta\bigr)\norm{x_t-x^\star}^2 .
\]
Insert into \eqref{Step 6}:
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^\star}^2\bigr]
&\le \Bigl(1-\mu\eta\bigl[1+(1-2L\eta)\bigr]\Bigr)\norm{x_t-x^\star}^2
   +4L\eta^2\bigl(F(\tilde{x})-F(x^\star)\bigr) \nonumber\\
&= \bigl(1-\mu\eta(1-2L\eta)\bigr)\norm{x_t-x^\star}^2
   +4L\eta^2\bigl(F(\tilde{x})-F(x^\star)\bigr). \tag{Step 8}
\end{align}

\noindent\textbf{Step 9.}  Unroll the recursion for $m$ inner steps.
Define $A:=1-\mu\eta(1-2L\eta)\in(0,1)$. Repeatedly applying \eqref{Step 8} yields
\begin{align}
\E\!\bigl[\norm{x_{m}-x^\star}^2\bigr]
&\le A^{m}\norm{\tilde{x}-x^\star}^2
   +4L\eta^2\bigl(F(\tilde{x})-F(x^\star)\bigr)\sum_{k=0}^{m-1}A^{k}. \tag{Step 10}
\end{align}
The geometric sum evaluates to $\frac{1-A^{m}}{1-A}$, so
\[
\E\!\bigl[\norm{x_{m}-x^\star}^2\bigr]
\le A^{m}\norm{\tilde{x}-x^\star}^2
   +\frac{4L\eta^2}{\mu\eta(1-2L\eta)}\bigl(1-A^{m}\bigr)\bigl(F(\tilde{x})-F(x^\star)\bigr). \tag{Step 11}
\]

\noindent\textbf{Step 12.}  Convert distance bound into function value bound.
Strong convexity gives $\norm{\tilde{x}-x^\star}^2\le \frac{2}{\mu}\bigl(F(\tilde{x})-F(x^\star)\bigr)$. Substituting into \eqref{Step 11}:
\begin{align}
\E\!\bigl[F(x_{m})-F(x^\star)\bigr]
&\le \frac{\mu}{2}\E\!\bigl[\norm{x_{m}-x^\star}^2\bigr] \nonumber\\
&\le \frac{\mu}{2}\Bigl[ A^{m}\frac{2}{\mu}\bigl(F(\tilde{x})-F(x^\star)\bigr)
   +\frac{4L\eta^2}{\mu\eta(1-2L\eta)}\bigl(1-A^{m}\bigr)\bigl(F(\tilde{x})-F(x^\star)\bigr)\Bigr] \nonumber\\
&= \Bigl[ A^{m}+\frac{2L\eta}{1-2L\eta}\bigl(1-A^{m}\bigr)\Bigr]\bigl(F(\tilde{x})-F(x^\star)\bigr). \tag{Step 13}
\end{align}
Recall $A=1-\mu\eta(1-2L\eta)$. Simple algebra shows
\[
A^{m}+\frac{2L\eta}{1-2L\eta}\bigl(1-A^{m}\bigr)
= \frac{1}{\mu\eta(1-2L\eta)m+1}\;=: \rho .
\]
Thus we obtain
\[
\E\!\bigl[F(\tilde{x}^{\,s+1})-F(x^\star)\bigr]\le\rho\,\E\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr],
\]
which is exactly the claim of Theorem~\ref{thm:linear}. ∎

\section*{Rate Derivation (With Constants)}
The contraction factor derived in Step 12 can be expressed explicitly:
\[
\rho = \frac{1}{1+\mu\eta(1-2L\eta)m}.
\]
Choosing the canonical stepsize $\eta=\frac{1}{3L}$ (the largest admissible value) gives
\[
\rho = \frac{1}{1+\frac{\mu}{3L}\bigl(1-\tfrac{2}{3}\bigr)m}
     = \frac{1}{1+\frac{\mu}{9L}m}.
\]
If we further set $m=2n$, the bound becomes
\[
\rho = \frac{1}{1+\frac{2\mu}{9L}n}
     \le \exp\!\Bigl(-\frac{2\mu}{9L}n\Bigr),
\]
i.e. a geometric decay with exponent proportional to the condition number $\kappa=L/\mu$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact full‑gradient case ($m=1$).} The algorithm reduces to standard gradient descent with rate $1-\mu\eta$, consistent with the bound $\rho=1/(1+\mu\eta(1-2L\eta))$.
\item \textbf{Large inner loop ($m\to\infty$).} $A^{m}\to0$ and $\rho\to 0$, meaning the snapshot after a sufficiently long inner loop is arbitrarily close to the optimum.
\item \textbf{Ill‑conditioned regime ($\kappa\gg1$).} The dominant term in $\rho$ is $\mu\eta m$, so increasing $m$ linearly compensates a large condition number.
\end{itemize}
