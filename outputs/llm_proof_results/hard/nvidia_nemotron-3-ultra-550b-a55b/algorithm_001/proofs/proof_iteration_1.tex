\section*{Algorithm Name}
Proximal Gradient Descent (Forward–Backward Splitting)

\section*{Mathematical Formulation}
Consider the composite optimization problem
\[
\min_{x \in \mathbb{R}^d} \; F(x) \coloneqq f(x) + g(x),
\]
where
\begin{itemize}
    \item $f: \mathbb{R}^d \to \mathbb{R}$ is convex and differentiable with $L$-Lipschitz continuous gradient ($\|\nabla f(x) - \nabla f(y)\| \le L \|x-y\|$ for all $x,y$).
    \item $g: \mathbb{R}^d \to \mathbb{R} \cup \{+\infty\}$ is proper, closed, convex (possibly non‑smooth).
\end{itemize}
The \emph{proximal operator} of $\eta g$ ($\eta > 0$) is defined as
\[
\operatorname{prox}_{\eta g}(v) \coloneqq \arg\min_{x \in \mathbb{R}^d} \left\{ g(x) + \frac{1}{2\eta}\|x - v\|^2 \right\}.
\]
The proximal gradient iteration with step size $\eta > 0$ is
\[
x_{k+1} = \operatorname{prox}_{\eta g}\bigl( x_k - \eta \nabla f(x_k) \bigr), \qquad k = 0,1,2,\dots
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0 \in \operatorname{dom} g$, step size $\eta > 0$, max iterations $K$, tolerance $\varepsilon > 0$.
\FOR{$k = 0$ to $K-1$}
    \STATE Compute gradient $g_k = \nabla f(x_k)$.
    \STATE Compute gradient step $v_k = x_k - \eta g_k$.
    \STATE Compute proximal update $x_{k+1} = \operatorname{prox}_{\eta g}(v_k)$.
    \IF{$\|x_{k+1} - x_k\| < \varepsilon$} \textbf{break} \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let $f(w) = (w-3)^2$, $g(w) = 0$ (so $\operatorname{prox}_{\eta g}(v)=v$), $w_0 = 0$, $\eta = 0.1$.
Then $\nabla f(w) = 2(w-3)$.
\begin{center}
\begin{tabular}{c c c c c c}
\toprule
$k$ & $w_k$ & $\nabla f(w_k)$ & $v_k = w_k - 0.1 \nabla f(w_k)$ & $w_{k+1}=v_k$ & $f(w_{k+1})$ \\
\midrule
0 & 0.000 & $-6.000$ & $0.000 - 0.1(-6.000) = 0.600$ & 0.600 & $(0.6-3)^2 = 5.76$ \\

After 3 iterations, $w_3 \approx 1.464$, $f(w_3) \approx 2.359$. The method converges to the minimizer $w^*=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{assump:smooth}
$f$ is convex and its gradient is $L$-Lipschitz continuous:
\[
\|\nabla f(x) - \nabla f(y)\| \le L \|x-y\| \quad \forall x,y \in \mathbb{R}^d.
\]
Equivalently, for all $x,y$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2}\|y-x\|^2. \tag{Descent Lemma}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{assump:convex_g}
$g$ is proper, closed, and convex.
\end{assumption}

\begin{assumption}[Step Size]\label{assump:step}
The step size satisfies $0 < \eta \le 1/L$.
\end{assumption}

\begin{assumption}[Existence of Minimizer]\label{assump:opt}
The solution set $X^* = \arg\min F$ is non‑empty; let $x^* \in X^*$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear Convergence Rate]\label{thm:rate}
Under Assumptions \ref{assump:smooth}--\ref{assump:opt}, the sequence $\{x_k\}$ generated by Proximal Gradient Descent satisfies
\[
F(x_k) - F(x^*) \le \frac{\|x_0 - x^*\|^2}{2\eta k}, \qquad \forall k \ge 1.
\]
In particular, with the maximal step size $\eta = 1/L$,
\[
F(x_k) - F(x^*) \le \frac{L \|x_0 - x^*\|^2}{2k}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{property}[Monotonicity]
$F(x_{k+1}) \le F(x_k)$ for all $k$ (the objective values are non‑increasing).
\end{property}
\begin{property}[Fixed‑Point Characterization]
$x^*$ is a minimizer of $F$ iff $x^* = \operatorname{prox}_{\eta g}(x^* - \eta \nabla f(x^*))$.
\end{property}
\begin{property}[Hyperparameter Sensitivity]
The rate depends linearly on $L$ and inversely on $\eta$. Choosing $\eta > 1/L$ may break convergence; $\eta \ll 1/L$ slows convergence proportionally.
\end{property}
\begin{property}[Comparison to Gradient Descent]
When $g \equiv 0$, the method reduces to Gradient Descent with the same $O(1/k)$ rate. For non‑smooth $g$ (e.g., $\ell_1$ regularization), it handles the non‑smooth term exactly via the proximal operator.
\end{property}

\section*{Discussion}
Proximal Gradient Descent is a cornerstone algorithm for composite convex optimization. Its $O(1/k)$ rate is optimal among first‑order methods for this problem class. Accelerated variants (FISTA) achieve $O(1/k^2)$. The proof below relies crucially on the firm non‑expansiveness of the proximal operator, which encodes the optimality condition of the proximal subproblem.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Proximal Operator]
For a proper closed convex function $h$ and $\eta > 0$,
\[
\operatorname{prox}_{\eta h}(v) = \arg\min_{x} \left\{ h(x) + \frac{1}{2\eta}\|x-v\|^2 \right\}.
\]
\end{definition}

\begin{lemma}[Firm Non‑expansiveness]\label{lem:firm}
The proximal operator is firmly non‑expansive: for any $u,v \in \mathbb{R}^d$,
\[
\|\operatorname{prox}_{\eta g}(u) - \operatorname{prox}_{\eta g}(v)\|^2 \le \langle u-v,\; \operatorname{prox}_{\eta g}(u) - \operatorname{prox}_{\eta g}(v) \rangle.
\]
\end{lemma}

\begin{lemma}[Descent Lemma for $L$-smooth $f$]\label{lem:descent}
Under Assumption \ref{assump:smooth}, for all $x,y$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{lemma}

\begin{lemma}[Convexity Inequalities]\label{lem:convex}
For convex $f$ and $g$,
\begin{align}
f(y) &\ge f(x) + \langle \nabla f(x), y-x \rangle, \label{eq:cvx_f} \\
g(y) &\ge g(x) + \langle s, y-x \rangle \quad \forall s \in \partial g(x). \label{eq:cvx_g}
\end{align}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}

We prove Theorem \ref{thm:rate}. Let $x^*$ be any minimizer of $F$.

\paragraph{Step 1: Optimality condition of the proximal update.} [Step 1]
By definition, $x_{k+1} = \operatorname{prox}_{\eta g}(x_k - \eta \nabla f(x_k))$ is the unique minimizer of
\[
x \mapsto g(x) + \frac{1}{2\eta}\|x - (x_k - \eta \nabla f(x_k))\|^2.
\]
The first‑order optimality condition gives
\[
0 \in \partial g(x_{k+1}) + \frac{1}{\eta}\bigl(x_{k+1} - x_k + \eta \nabla f(x_k)\bigr).
\]
Hence there exists a subgradient $s_{k+1} \in \partial g(x_{k+1})$ such that
\[
x_k - x_{k+1} = \eta \bigl( \nabla f(x_k) + s_{k+1} \bigr). \tag{1}
\]

\paragraph{Step 2: Convexity bound for $g$.} [Step 2]
Using Lemma \ref{lem:convex} (\ref{eq:cvx_g}) with $x = x_{k+1}$, $y = x^*$, and $s = s_{k+1}$:
\[
g(x^*) \ge g(x_{k+1}) + \langle s_{k+1}, x^* - x_{k+1} \rangle.
\]
Substitute $s_{k+1}$ from (1):
\[
s_{k+1} = \frac{x_k - x_{k+1}}{\eta} - \nabla f(x_k).
\]
Thus
\[
g(x_{k+1}) \le g(x^*) + \langle \nabla f(x_k), x^* - x_{k+1} \rangle + \frac{1}{\eta}\langle x_k - x_{k+1}, x_{k+1} - x^* \rangle. \tag{2}
\]

\paragraph{Step 3: Smoothness bound for $f$.} [Step 3]
Apply Lemma \ref{lem:descent} with $x = x_k$, $y = x_{k+1}$:
\[
f(x_{k+1}) \le f(x_k) + \langle \nabla f(x_k), x_{k+1} - x_k \rangle + \frac{L}{2}\|x_{k+1} - x_k\|^2. \tag{3}
\]

\paragraph{Step 4: Convexity bound for $f$ relating $x_k$ and $x^*$.} [Step 4]
From Lemma \ref{lem:convex} (\ref{eq:cvx_f}) with $x = x_k$, $y = x^*$:
\[
f(x^*) \ge f(x_k) + \langle \nabla f(x_k), x^* - x_k \rangle
\;\Longrightarrow\;
f(x_k) + \langle \nabla f(x_k), x^* - x_k \rangle \le f(x^*). \tag{4}
\]

\paragraph{Step 5: Combine (2) and (3) to bound $F(x_{k+1})$.} [Step 5]
Add (2) and (3):
\begin{align*}
F(x_{k+1}) &\le f(x_k) + g(x^*) + \langle \nabla f(x_k), x_{k+1} - x_k + x^* - x_{k+1} \rangle \\
&\quad + \frac{L}{2}\|x_{k+1} - x_k\|^2 + \frac{1}{\eta}\langle x_k - x_{k+1}, x_{k+1} - x^* \rangle \\
&= f(x_k) + g(x^*) + \langle \nabla f(x_k), x^* - x_k \rangle + \frac{L}{2}\|x_{k+1} - x_k\|^2 \\
&\quad + \frac{1}{\eta}\langle x_k - x_{k+1}, x_{k+1} - x^* \rangle.
\end{align*}
Now use (4) to replace $f(x_k) + \langle \nabla f(x_k), x^* - x_k \rangle$ with $f(x^*)$:
\[
F(x_{k+1}) \le F(x^*) + \frac{L}{2}\|x_{k+1} - x_k\|^2 + \frac{1}{\eta}\langle x_k - x_{k+1}, x_{k+1} - x^* \rangle. \tag{5}
\]

\paragraph{Step 6: Rewrite the inner product using the polarization identity.} [Step 6]
For any vectors $a,b,c$,
\[
\langle a-b, b-c \rangle = \frac{1}{2}\bigl( \|a-c\|^2 - \|a-b\|^2 - \|b-c\|^2 \bigr).
\]
With $a = x_k$, $b = x_{k+1}$, $c = x^*$:
\[
\langle x_k - x_{k+1}, x_{k+1} - x^* \rangle = \frac{1}{2}\bigl( \|x_k - x^*\|^2 - \|x_{k+1} - x^*\|^2 - \|x_k - x_{k+1}\|^2 \bigr). \tag{6}
\]

\paragraph{Step 7: Substitute (6) into (5).} [Step 7]
\begin{align*}
F(x_{k+1}) &\le F(x^*) + \frac{L}{2}\|x_{k+1} - x_k\|^2 \\
&\quad + \frac{1}{2\eta}\bigl( \|x_k - x^*\|^2 - \|x_{k+1} - x^*\|^2 - \|x_k - x_{k+1}\|^2 \bigr) \\
&= F(x^*) + \frac{1}{2\eta}\bigl( \|x_k - x^*\|^2 - \|x_{k+1} - x^*\|^2 \bigr) + \Bigl(\frac{L}{2} - \frac{1}{2\eta}\Bigr)\|x_{k+1} - x_k\|^2. \tag{7}
\end{align*}

\paragraph{Step 8: Use the step‑size condition $\eta \le 1/L$.} [Step 8]
Since $\eta \le 1/L$, we have $\frac{L}{2} - \frac{1}{2\eta} \le 0$. Therefore the last term in (7) is non‑positive, and we can drop it to obtain the key inequality
\[
F(x_{k+1}) \le F(x^*) + \frac{1}{2\eta}\bigl( \|x_k - x^*\|^2 - \|x_{k+1} - x^*\|^2 \bigr). \tag{8}
\]

\paragraph{Step 9: Monotonicity of the objective values.} [Step 9]
Setting $x^* = x_k$ in (7) (which is valid because (7) holds for any $y$ in place of $x^*$) gives
\[
F(x_{k+1}) \le F(x_k) + \Bigl(\frac{L}{2} - \frac{1}{2\eta}\Bigr)\|x_{k+1} - x_k\|^2 \le F(x_k),
\]
so $\{F(x_k)\}$ is non‑increasing.

\paragraph{Step 10: Telescoping sum.} [Step 10]
Sum (8) for $k = 0,1,\dots,N-1$:
\[
\sum_{k=0}^{N-1} \bigl( F(x_{k+1}) - F(x^*) \bigr) \le \frac{1}{2\eta}\bigl( \|x_0 - x^*\|^2 - \|x_N - x^*\|^2 \bigr) \le \frac{\|x_0 - x^*\|^2}{2\eta}.
\]
Because $F(x_{k+1}) \ge F(x^*)$ and the sequence is non‑increasing, $F(x_N) \le F(x_{k+1})$ for all $k \le N-1$. Hence
\[
N \bigl( F(x_N) - F(x^*) \bigr) \le \sum_{k=0}^{N-1} \bigl( F(x_{k+1}) - F(x^*) \bigr) \le \frac{\|x_0 - x^*\|^2}{2\eta}.
\]

\paragraph{Step 11: Final rate.} [Step 11]
Dividing by $N$ yields
\[
F(x_N) - F(x^*) \le \frac{\|x_0 - x^*\|^2}{2\eta N}, \qquad \forall N \ge 1.
\]
With $\eta = 1/L$, this becomes $F(x_N) - F(x^*) \le \frac{L\|x_0 - x^*\|^2}{2N}$. $\quad\square$

\section*{Rate Derivation (With Constants)}
The proof above gives the explicit constant:
\[
C = \frac{\|x_0 - x^*\|^2}{2\eta}.
\]
For the maximal step size $\eta = 1/L$,
\[
C = \frac{L \|x_0 - x^*\|^2}{2}.
\]
If a smaller step size $\eta = c/L$ ($0 < c \le 1$) is used, the constant increases by a factor $1/c$:
\[
F(x_k) - F(x^*) \le \frac{L \|x_0 - x^*\|^2}{2c k}.
\]

\section*{Special Cases}

\begin{itemize}
    \item \textbf{Gradient Descent ($g \equiv 0$):} $\operatorname{prox}_{\eta g}(v) = v$. The algorithm reduces to $x_{k+1} = x_k - \eta \nabla f(x_k)$, and the same $O(1/k)$ rate holds.
    \item \textbf{ISTA ($g(x) = \lambda \|x\|_1$):} The proximal operator is soft‑thresholding:
    \[
    [\operatorname{prox}_{\eta \lambda \|\cdot\|_1}(v)]_i = \operatorname{sign}(v_i) \max\{|v_i| - \eta\lambda, 0\}.
    \]
    The convergence rate remains $O(1/k)$.
    \item \textbf{Projected Gradient Descent ($g = \iota_C$, indicator of a closed convex set $C$):} $\operatorname{prox}_{\eta g}(v) = \Pi_C(v)$ (Euclidean projection onto $C$). The rate is again $O(1/k)$.
    \item \textbf{Strongly Convex $f$:} If $f$ is $\mu$-strongly convex, a linear rate $O((1-\eta\mu)^k)$ can be shown with a slightly modified proof.
\end{itemize}
