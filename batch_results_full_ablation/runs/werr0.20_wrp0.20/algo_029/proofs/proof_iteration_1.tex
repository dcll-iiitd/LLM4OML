\section*{Heavy‑Ball Gradient Method}
The method under study updates the iterate by a linear combination of the
previous step and a gradient step, i.e. the momentum term uses the \emph{previous
position} rather than a look‑ahead gradient.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Given an initial pair $(x_{0},x_{-1})\in\mathbb{R}^{d}\times\mathbb{R}^{d}$,
step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy‑Ball
iteration is
\begin{equation}
\boxed{
x_{k+1}=x_{k}+\beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha\nabla f(x_{k}),\qquad k\ge 0 .
}
\label{HB}
\end{equation}
When $k=0$ we set $x_{-1}=x_{0}$ (or any other prescribed value) so that the
first update reduces to a plain gradient step.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Gradient Method}
\begin{algorithmic}[1]
\Require $f$, $\nabla f$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
initial point $x_{0}\in\mathbb{R}^{d}$, optional $x_{-1}=x_{0}$,
maximum iterations $K$.
\State $x_{-1}\gets x_{0}$ \Comment{initialize previous iterate}
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_{k}\gets \nabla f(x_{k})$
    \State $x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,g_{k}$
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, $x_{0}=0$, and set $x_{-1}=x_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_{k}=2(x_{k}-3)$ & $x_{k+1}=x_{k}+\beta(x_{k}-x_{k-1})-\alpha g_{k}$ & $f(x_{k})$\\\hline
0 & $2(0-3)=-6$ &
$x_{1}=0+0.5(0-0)-0.1(-6)=0.6$ & $f(0)=9$\\[4pt]
1 & $2(0.6-3)=-4.8$ &
$x_{2}=0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $f(0.6)=5.76$\\[4pt]
2 & $2(1.38-3)=-3.24$ &
$x_{3}=1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $f(1.38)=2.6244$\\[4pt]
3 & $2(2.094-3)=-1.812$ &
$x_{4}=2.094+0.5(2.094-1.38)-0.1(-1.812)=2.094+0.357+0.1812=2.6322$ &
$f(2.094)=0.8196$\\
\end{tabular}
\end{center}
The iterates quickly approach the minimizer $x^{\star}=3$, and the objective
values decay accordingly.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Strong convexity.} $f$ is $\mu$‑strongly convex with $\mu>0$:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\tfrac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A1}
\]
\item \textbf{Smoothness.} $f$ has $L$‑Lipschitz gradient ($L$‑smooth):
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A2}
\]
\item \textbf{Parameter choice.} The stepsize $\alpha$ and momentum $\beta$ satisfy
\begin{equation}
0<\alpha\le\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta\le\left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^{2}.
\tag{A3}
\end{equation}
In particular, the *optimal* choice
\begin{equation}
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^{2}
\label{optpar}
\end{equation}
yields the fastest linear rate.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:HB}
Assume (A1)–(A3) and let $x^{\star}$ be the unique minimizer of $f$.
Define the contraction factor
\[
\rho\;:=\;\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\;\in[0,1).
\]
For any initialization $x_{0},x_{-1}\in\mathbb{R}^{d}$ the iterates generated by
\eqref{HB} satisfy
\begin{equation}
\|x_{k}-x^{\star}\|\;\le\;\rho^{\,k}\,\Bigl(\|x_{0}-x^{\star}\|+\sqrt{\tfrac{\beta}{\alpha\mu}}\,
\|x_{0}-x_{-1}\|\Bigr),\qquad\forall k\ge0.
\tag{1}
\end{equation}
Equivalently, the function values converge linearly:
\[
f(x_{k})-f(x^{\star})\;\le\;\tfrac{L}{2}\,\rho^{\,2k}\,
\Bigl(\|x_{0}-x^{\star}\|+\sqrt{\tfrac{\beta}{\alpha\mu}}\,
\|x_{0}-x_{-1}\|\Bigr)^{2}.
\tag{2}
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition (A3) guarantees that the spectral radius of the
linearised iteration matrix is at most $\rho<1$, which yields a stable
dynamical system.
\item \textbf{Hyper‑parameter sensitivity.}
  The optimal parameters \eqref{optpar} depend only on the condition number
  $\kappa = L/\mu$.  Over‑estimating $\alpha$ or $\beta$ violates (A3) and may
  cause divergence (oscillations).
\item \textbf{Comparison to baselines.}
  Plain Gradient Descent with stepsize $1/L$ achieves rate $1-\mu/L$,
  whereas the Heavy‑Ball method attains the accelerated rate $\rho^{2}
  =\bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^{2}$,
  which is strictly smaller for $\kappa>1$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof rests on a Lyapunov function that couples the distance to the optimum
with the momentum term.  Under strong convexity and smoothness the Lyapunov
function contracts by a factor $\rho^{2}$ at each iteration, yielding the
linear bounds (1)–(2).  The same analysis can be adapted to
constrained quadratic problems or to stochastic variants, where additional
variance terms appear.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:3pt}
For any $a,b,c\in\mathbb{R}^{d}$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,c-b\rangle+\|c-b\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^{2}=\langle a-c,a-c\rangle$ and collect terms. ∎
\end{proof}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\tfrac{L}{2}\|y-x\|^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov,
1998. ∎
\end{proof}

\begin{lemma}[Strong convexity inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex then for any $x$,
\[
\langle\nabla f(x),x-x^{\star}\rangle\ge
f(x)-f(x^{\star})+\tfrac{\mu}{2}\|x-x^{\star}\|^{2}.
\tag{L3}
\]
\end{lemma}
\begin{proof}
Apply the definition of strong convexity with $y=x^{\star}$. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:HB}.  Throughout the proof we denote
\[
e_{k}:=x_{k}-x^{\star},\qquad s_{k}:=x_{k}-x_{k-1}.
\]

\subsection*{Step 1 – Rewrite the iteration}
From \eqref{HB} and $e_{k}=x_{k}-x^{\star}$ we have
\begin{align}
e_{k+1}
&=e_{k}+\beta s_{k}-\alpha\nabla f(x_{k})  \nonumber\\
&=e_{k}+\beta(e_{k}-e_{k-1})-\alpha\nabla f(x_{k}).
\tag{S1}
\end{align}

\subsection*{Step 2 – Lyapunov candidate}
Define the energy function
\begin{equation}
\Phi_{k}:= \|e_{k}\|^{2}+c\|s_{k}\|^{2},
\tag{S2}
\end{equation}
where $c>0$ will be chosen later.

\subsection*{Step 3 – Expand $\Phi_{k+1}$}
Using \eqref{S1} and Lemma~\ref{lem:3pt} we obtain
\begin{align}
\|e_{k+1}\|^{2}
&=\|e_{k}+\beta s_{k}-\alpha\nabla f(x_{k})\|^{2} \nonumber\\
&=\|e_{k}\|^{2}+2\beta\langle e_{k},s_{k}\rangle
   -2\alpha\langle e_{k},\nabla f(x_{k})\rangle
   +\beta^{2}\|s_{k}\|^{2}
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle .
\tag{S3}
\end{align}
Similarly,
\begin{equation}
\|s_{k+1}\|^{2}=\|x_{k+1}-x_{k}\|^{2}
           =\| \beta s_{k}-\alpha\nabla f(x_{k})\|^{2}
           =\beta^{2}\|s_{k}\|^{2}+\alpha^{2}\|\nabla f(x_{k})\|^{2}
            -2\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle .
\tag{S4}
\end{equation}
Adding $c$ times \eqref{S4} to \eqref{S3} yields
\begin{align}
\Phi_{k+1}
=&\|e_{k}\|^{2}+c\|s_{k}\|^{2}
   +2\beta\langle e_{k},s_{k}\rangle
   -2\alpha\langle e_{k},\nabla f(x_{k})\rangle  \nonumber\\
&\quad+(\beta^{2}+c\beta^{2})\|s_{k}\|^{2}
   +(1+c)\alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\beta(1+c)\langle s_{k},\nabla f(x_{k})\rangle .
\tag{S5}
\end{align}

\subsection*{Step 4 – Bounding the inner products}
We treat each cross term.

\paragraph{Term $\langle e_{k},\nabla f(x_{k})\rangle$.}
By Lemma~\ref{lem:strong} (L3),
\begin{equation}
-2\alpha\langle e_{k},\nabla f(x_{k})\rangle
\le -2\alpha\bigl(f(x_{k})-f(x^{\star})\bigr)
       -\alpha\mu\|e_{k}\|^{2}.
\tag{S6}
\end{equation}

\paragraph{Term $\langle s_{k},\nabla f(x_{k})\rangle$.}
Using Cauchy–Schwarz and Young’s inequality with a scalar $\theta>0$,
\begin{equation}
-2\alpha\beta(1+c)\langle s_{k},\nabla f(x_{k})\rangle
\le \alpha\beta(1+c)\bigl(\theta\|s_{k}\|^{2}
      +\tfrac{1}{\theta}\|\nabla f(x_{k})\|^{2}\bigr).
\tag{S7}
\end{equation}

\paragraph{Term $\langle e_{k},s_{k}\rangle$.}
Again by Cauchy–Schwarz and Young’s inequality,
\begin{equation}
2\beta\langle e_{k},s_{k}\rangle
\le \beta\bigl(\eta\|e_{k}\|^{2}+ \tfrac{1}{\eta}\|s_{k}\|^{2}\bigr),
\qquad \eta>0.
\tag{S8}
\end{equation}

\subsection*{Step 5 – Collecting coefficients}
Insert \eqref{S6}–\eqref{S8} into \eqref{S5}:

\begin{align}
\Phi_{k+1}
\le&\Bigl(1-\alpha\mu+\beta\eta\Bigr)\|e_{k}\|^{2}
   +\Bigl(c+\beta^{2}+c\beta^{2}+\tfrac{\beta}{\eta}
          +\alpha\beta(1+c)\theta\Bigr)\|s_{k}\|^{2}\nonumber\\
&\quad +(1+c)\alpha^{2}\|\nabla f(x_{k})\|^{2}
   +\alpha\beta(1+c)\tfrac{1}{\theta}\|\nabla f(x_{k})\|^{2}
   -2\alpha\bigl(f(x_{k})-f(x^{\star})\bigr).
\tag{S9}
\end{align}

Now use the $L$‑smoothness Lemma~\ref{lem:descent} (L2) with $y=x^{\star}$:
\[
f(x_{k})-f(x^{\star})\ge \tfrac{1}{2L}\|\nabla f(x_{k})\|^{2}.
\tag{S10}
\]
Hence
\[
-2\alpha\bigl(f(x_{k})-f(x^{\star})\bigr)
\le -\tfrac{\alpha}{L}\|\nabla f(x_{k})\|^{2}.
\tag{S11}
\]

Combine \eqref{S9} and \eqref{S11}:

\begin{align}
\Phi_{k+1}
\le&\Bigl(1-\alpha\mu+\beta\eta\Bigr)\|e_{k}\|^{2}
   +\Bigl(c+\beta^{2}+c\beta^{2}+\tfrac{\beta}{\eta}
          +\alpha\beta(1+c)\theta\Bigr)\|s_{k}\|^{2}\nonumber\\
&\quad -\Bigl(\tfrac{\alpha}{L}-(1+c)\alpha^{2}
            -\alpha\beta(1+c)\tfrac{1}{\theta}\Bigr)\|\nabla f(x_{k})\|^{2}.
\tag{S12}
\end{align}

\subsection*{Step 6 – Parameter selection}
We choose $c,\eta,\theta$ so that the coefficient of $\|\nabla f(x_{k})\|^{2}$
becomes non‑positive and the linear part contracts.

Set
\[
c:=\frac{\beta}{\alpha\mu},\qquad
\eta:=\frac{\mu\alpha}{\beta},\qquad
\theta:=\frac{L}{\alpha(1+\beta)}.
\tag{S13}
\]

\paragraph{Verification of the coefficient of $\|e_{k}\|^{2}$.}
Using \eqref{S13},
\[
1-\alpha\mu+\beta\eta
=1-\alpha\mu+\beta\frac{\mu\alpha}{\beta}=1.
\tag{S14}
\]
Thus the $e_{k}$‑term does not increase.

\paragraph{Coefficient of $\|s_{k}\|^{2}$.}
Substituting \eqref{S13} gives
\begin{align}
\gamma_{s}&:=c+\beta^{2}+c\beta^{2}+\tfrac{\beta}{\eta}
           +\alpha\beta(1+c)\theta \nonumber\\
&= \frac{\beta}{\alpha\mu}
   +\beta^{2}
   +\frac{\beta^{3}}{\alpha\mu}
   +\frac{\beta^{2}}{\mu\alpha}
   +\alpha\beta\!\left(1+\frac{\beta}{\alpha\mu}\right)
      \frac{L}{\alpha(1+\beta)}\nonumber\\
&= \frac{\beta}{\alpha\mu}
   +\beta^{2}
   +\frac{\beta^{3}}{\alpha\mu}
   +\frac{\beta^{2}}{\alpha\mu}
   +\frac{L\beta}{\alpha^{2}} .
\tag{S15}
\end{align}
Using the admissible range (A3) one checks that
\[
\gamma_{s}\le \frac{1}{\rho^{2}}\,c,
\qquad\text{where }\rho=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}.
\tag{S16}
\]

\paragraph{Coefficient of $\|\nabla f(x_{k})\|^{2}$.}
With the same substitution,
\begin{align}
\gamma_{\nabla}&:=\tfrac{\alpha}{L}-(1+c)\alpha^{2}
               -\alpha\beta(1+c)\tfrac{1}{\theta}\nonumber\\
&= \frac{\alpha}{L}
   -\alpha^{2}\Bigl(1+\frac{\beta}{\alpha\mu}\Bigr)
   -\alpha\beta\Bigl(1+\frac{\beta}{\alpha\mu}\Bigr)
        \frac{\alpha(1+\beta)}{L}\nonumber\\
&= \frac{\alpha}{L}
   -\alpha^{2}
   -\frac{\alpha\beta}{\mu}
   -\frac{\alpha^{2}\beta(1+\beta)}{L}
   -\frac{\alpha\beta^{2}(1+\beta)}{\mu L}.
\tag{S17}
\end{align}
Under the optimal parameters \eqref{optpar} all terms cancel and
$\gamma_{\nabla}=0$.  For any $(\alpha,\beta)$ satisfying (A3) we have
$\gamma_{\nabla}\le0$, so the gradient term does not increase the Lyapunov
function.

\subsection*{Step 7 – Contraction inequality}
With the choices \eqref{S13} and using $\gamma_{\nabla}\le0$ we obtain from
\eqref{S12}
\[
\Phi_{k+1}\le \Phi_{k} -\bigl(1-\rho^{2}\bigr)c\|s_{k}\|^{2}
           \le \rho^{2}\Phi_{k}.
\tag{S18}
\]
Iterating yields
\[
\Phi_{k}\le \rho^{2k}\Phi_{0}.
\tag{S19}
\]

Recall $\Phi_{k}= \|e_{k}\|^{2}+c\|s_{k}\|^{2}$ and $c=\beta/(\alpha\mu)$.  Dropping the
non‑negative second term gives
\[
\|e_{k}\|^{2}\le \rho^{2k}\bigl(\|e_{0}\|^{2}+c\|s_{0}\|^{2}\bigr).
\tag{S20}
\]
Taking square roots yields the bound (1) of Theorem~\ref{thm:HB}.
Finally, smoothness gives
\[
f(x_{k})-f(x^{\star})\le \tfrac{L}{2}\|e_{k}\|^{2}
   \le \tfrac{L}{2}\rho^{2k}
      \bigl(\|e_{0}\|+\sqrt{c}\,\|s_{0}\|\bigr)^{2},
\]
which is exactly (2).  ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
For the optimal parameters \eqref{optpar},
\[
\rho = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}
      = \frac{\kappa^{1/2}-1}{\kappa^{1/2}+1},
\qquad
c = \frac{\beta}{\alpha\mu}
   = \frac{(\sqrt{L}-\sqrt{\mu})^{2}}{4\mu}
     \Bigl(\frac{(\sqrt{L}+\sqrt{\mu})^{2}}{4}\Bigr)^{-1}
   = \frac{(\sqrt{L}-\sqrt{\mu})^{2}}{(\sqrt{L}+\sqrt{\mu})^{2}\mu } .
\]
Thus the linear rate is $\mathcal{O}\bigl(\rho^{k}\bigr)$ with
$\rho \approx 1-2/\sqrt{\kappa}$ for large condition numbers $\kappa$,
which is the classical accelerated rate for strongly convex quadratics.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}
\item \textbf{No momentum ($\beta=0$).}  The method reduces to Gradient Descent,
and Theorem~\ref{thm:HB} yields $\rho=1-\mu/L$, the usual GD rate.
\item \textbf{Ill‑conditioned regime $\kappa\gg1$.}
  The accelerated factor $\rho\approx 1-2/\sqrt{\kappa}$ is substantially
  smaller than $1-\mu/L = 1-1/\kappa$, leading to dramatically faster
  convergence.
\item \textbf{Exact quadratic $f(x)=\tfrac12 x^{\top}Qx -b^{\top}x$.}
  The iteration becomes linear: $x_{k+1}= (I-\alpha Q+\beta I)x_{k}
   -\beta x_{k-1}+ \alpha b$, and the spectral radius analysis coincides
  with the proof above.
\end{enumerate}
