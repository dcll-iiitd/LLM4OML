\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  

\[
\min_{x\in\mathbb{R}^n}\;\max_{y\in\mathbb{R}^m}
\; \mathcal{L}(x,y):= f(x)+\langle Kx , y\rangle - g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}[noitemsep]
\item $f:\mathbb{R}^{n}\to\mathbb{R}\cup\{+\infty\}$ is proper, convex, lower‑semicontinuous;
\item $g^{*}:\mathbb{R}^{m}\to\mathbb{R}\cup\{+\infty\}$ is the convex conjugate of a proper convex $g$;
\item $K\in\mathbb{R}^{m\times n}$ is a linear operator.
\end{itemize}

The PDHG iterates $(x^{k},y^{k})$ are defined by  

\[
\boxed{
\begin{aligned}
y^{k+1} &= \prox_{\sigma g^{*}}\!\bigl(y^{k}+\sigma K\bar x^{k}\bigr),\\[4pt]
x^{k+1} &= \prox_{\tau f}\!\bigl(x^{k}-\tau K^{\top} y^{k+1}\bigr),\\[4pt]
\bar x^{k+1} &= x^{k+1}+ \theta\bigl(x^{k+1}-x^{k}\bigr),
\end{aligned}}
\tag{2}
\]

with step sizes $\tau>0$, $\sigma>0$ and extrapolation parameter $\theta\in[0,1]$.  
The proximal operator of a proper convex function $h$ is  

\[
\prox_{\lambda h}(z)=\arg\min_{u}\Bigl\{h(u)+\frac{1}{2\lambda}\|u-z\|^{2}\Bigr\}.
\tag{3}
\]

The algorithm requires  

\[
\tau\sigma\|K\|^{2}<1,
\tag{4}
\]

where $\|K\|$ denotes the operator norm induced by the Euclidean norm.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\REQUIRE $f$, $g$, $K$, step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, initial $(x^{0},y^{0})$.
\STATE Set $\bar x^{0}\gets x^{0}$.
\FOR{$k=0,1,\dots,T-1$}
    \STATE $y^{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y^{k}+\sigma K\bar x^{k}\bigr)$ \hfill\COMMENT{dual ascent}
    \STATE $x^{k+1}\gets \prox_{\tau f}\!\bigl(x^{k}-\tau K^{\top} y^{k+1}\bigr)$ \hfill\COMMENT{primal descent}
    \STATE $\bar x^{k+1}\gets x^{k+1}+ \theta\bigl(x^{k+1}-x^{k}\bigr)$
\ENDFOR
\RETURN $x^{T},y^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate (2) on a simple 1‑dimensional problem with $K=1$, $g\equiv0$ (hence $g^{*}=0$ and $\prox_{\sigma g^{*}}(z)=z$).  
Let  

\[
f(x)= (x-3)^{2},\qquad x^{0}=0,\qquad \tau=\sigma=0.1,\qquad \theta=1.
\]

Because $g^{*}=0$, the dual update is identity: $y^{k+1}=y^{k}+ \sigma \bar x^{k}$.  
The proximal of $f$ with parameter $\tau$ can be computed analytically:

\[
\prox_{\tau f}(z)=\arg\min_{u}\Bigl\{(u-3)^{2}+ \frac{1}{2\tau}(u-z)^{2}\Bigr\}
      =\frac{z+2\tau\cdot 3}{1+2\tau}.
\tag{5}
\]

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $y^{k}$ & $x^{k}$ & $\bar x^{k}$ & $f(x^{k})$\\ \hline
0 & $0$ & $0$ & $0$ & $9$\\
1 & $y^{1}=0+0.1\cdot0=0$ &
    $x^{1}= \frac{0+2\cdot0.1\cdot3}{1+0.2}= \frac{0.6}{1.2}=0.5$ &
    $\bar x^{1}=0.5+1(0.5-0)=1.0$ &
    $f(0.5)= (0.5-3)^{2}=6.25$\\
2 & $y^{2}=0+0.1\cdot1.0=0.1$ &
    $x^{2}= \frac{0.5-0.1\cdot0.1+0.6}{1.2}
           =\frac{1.1}{1.2}\approx0.9167$ &
    $\bar x^{2}=0.9167+1(0.9167-0.5)=1.3334$ &
    $f(0.9167)\approx4.347$\\
3 & $y^{3}=0.1+0.1\cdot1.3334=0.23334$ &
    $x^{3}= \frac{0.9167-0.1\cdot0.23334+0.6}{1.2}
           =\frac{1.4834}{1.2}\approx1.2362$ &
    $\bar x^{3}=1.2362+1(1.2362-0.9167)=1.5557$ &
    $f(1.2362)\approx3.086$
\end{tabular}
\end{center}

The objective value drops from $9$ to $\approx3.09$ after three iterations, illustrating the descent property of PDHG.

\section*{Assumptions}
\begin{assumption}[Convexity and Lower‑semicontinuity]\label{ass:convex}
$f$ and $g$ are proper, convex, lower‑semicontinuous.
\end{assumption}
\begin{assumption}[Lipschitz Continuity of $K$]\label{ass:K}
The linear operator $K$ satisfies $\|K\|<\infty$ and the step sizes obey
\[
\tau\sigma\|K\|^{2}<1.
\tag{6}
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ such that
\[
\mathcal{L}(x^{\star},y)\le \mathcal{L}(x^{\star},y^{\star})\le
\mathcal{L}(x,y^{\star}),\qquad\forall x,y.
\tag{7}
\]
\end{assumption}
\begin{assumption}[Boundedness of the Primal–Dual Gap]\label{ass:bounded}
The set $\{(x,y):\mathcal{L}(x,y)\le \mathcal{L}(x^{0},y^{0})\}$ is bounded, ensuring
$\|x^{k}-x^{\star}\|\le R_{x}$ and $\|y^{k}-y^{\star}\|\le R_{y}$ for all $k$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Rate]\label{thm:rate}
Under Assumptions \ref{ass:convex}–\ref{ass:bounded} and with parameters satisfying \eqref{6},
let $(x^{k},y^{k})$ be generated by \eqref{2}. Define the averaged iterates  

\[
\bar x^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}x^{k+1},
\qquad
\bar y^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}y^{k+1}.
\]

Then for any saddle point $(x^{\star},y^{\star})$

\[
\mathcal{G}_{K}:=
\sup_{y}\mathcal{L}(\bar x^{K},y)-\inf_{x}\mathcal{L}(x,\bar y^{K})
\le
\frac{C}{K},
\qquad
C:=\frac{1}{2\tau}\|x^{0}-x^{\star}\|^{2}
     +\frac{1}{2\sigma}\|y^{0}-y^{\star}\|^{2}.
\tag{8}
\]

Consequently $\mathcal{G}_{K}\to0$ at rate $O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{property}[Stability]\label{prop:stab}
If $\tau\sigma\|K\|^{2}\le 1$, the iterates remain bounded for any initialization, as a direct consequence of the descent inequality derived in the proof.
\end{property}
\begin{property}[Hyper‑parameter Sensitivity]\label{prop:hyper}
The constant $C$ in \eqref{8} scales linearly with $1/\tau$ and $1/\sigma$. Smaller steps accelerate convergence of the gap but may violate \eqref{6}; thus a practical choice is $\tau=\sigma=1/\|K\|$ and $\theta=1$ (the original Chambolle–Pock setting).
\end{property}
\begin{property}[Comparison to Gradient Descent]\label{prop:cmp}
When $g\equiv0$ and $K=I$, PDHG reduces to the proximal point method with step $\tau$, which enjoys the same $O(1/k)$ ergodic rate as standard gradient descent for convex functions, but additionally handles nonsmooth $g$ without extra cost.
\end{property}
\section*{Discussion}
The theorem guarantees that the primal–dual gap of the averaged sequence decays inversely with the iteration count. The proof relies only on convexity, the proximal inequality, and the step‑size condition \eqref{6}; no smoothness of $f$ or $g$ is required, making the result applicable to a broad class of nonsmooth problems (e.g., total variation denoising, $\ell_{1}$ regularisation).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity for the Proximal Operator]\label{lem:prox}
For any convex $h$ and any $u,v\in\mathbb{R}^{d}$,
\[
\ip{u-\prox_{\lambda h}(u)}{\,\prox_{\lambda h}(u)-v}
\ge \lambda\bigl( h(\prox_{\lambda h}(u))-h(v)\bigr).
\tag{9}
\]
\end{lemma}
\begin{lemma}[Descent Lemma for Linear Coupling]\label{lem:lin}
For any $x,y$, using $K$ linear,
\[
\ip{Kx}{y}= \frac12\bigl(\|Kx\|^{2}+\|y\|^{2}-\|Kx-y\|^{2}\bigr).
\tag{10}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Let $(x^{\star},y^{\star})$ be a saddle point. Define the Lyapunov quantity  

\[
\Phi^{k}:=
\frac{1}{2\tau}\|x^{k}-x^{\star}\|^{2}
+\frac{1}{2\sigma}\|y^{k}-y^{\star}\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:prox} to the dual update (\ref{2}a) with $h=g^{*}$, $\lambda=\sigma$, $u=y^{k}+\sigma K\bar x^{k}$ and $v=y^{\star}$:

\[
\ip{y^{k}+\sigma K\bar x^{k}-y^{k+1}}{\,y^{k+1}-y^{\star}}
\ge \sigma\bigl(g^{*}(y^{k+1})-g^{*}(y^{\star})\bigr).
\tag{12}
\]

\noindent\textbf{[Step 2]} Rearrange \eqref{12}:

\[
\frac{1}{2\sigma}\bigl(\|y^{k}-y^{\star}\|^{2}
-\|y^{k+1}-y^{\star}\|^{2}
-\|y^{k+1}-y^{k}-\sigma K\bar x^{k}\|^{2}\bigr)
\ge g^{*}(y^{k+1})-g^{*}(y^{\star}).
\tag{13}
\]

\noindent\textbf{[Step 3]} Apply Lemma \ref{lem:prox} to the primal update (\ref{2}b) with $h=f$, $\lambda=\tau$, $u=x^{k}-\tau K^{\top}y^{k+1}$ and $v=x^{\star}$:

\[
\ip{x^{k}-\tau K^{\top}y^{k+1}-x^{k+1}}{\,x^{k+1}-x^{\star}}
\ge \tau\bigl(f(x^{k+1})-f(x^{\star})\bigr).
\tag{14}
\]

\noindent\textbf{[Step 4]} Rearrange \eqref{14}:

\[
\frac{1}{2\tau}\bigl(\|x^{k}-x^{\star}\|^{2}
-\|x^{k+1}-x^{\star}\|^{2}
-\|x^{k+1}-x^{k}+\tau K^{\top}y^{k+1}\|^{2}\bigr)
\ge f(x^{k+1})-f(x^{\star}).
\tag{15}
\]

\noindent\textbf{[Step 5]} Add \eqref{13} and \eqref{15} and use the definition \eqref{11}:

\[
\Phi^{k}-\Phi^{k+1}
\ge
\underbrace{g^{*}(y^{k+1})-g^{*}(y^{\star})}_{\text{dual term}}
+
\underbrace{f(x^{k+1})-f(x^{\star})}_{\text{primal term}}
+
\Delta^{k},
\tag{16}
\]

where  

\[
\Delta^{k}:=
\frac{1}{2\sigma}\|y^{k+1}-y^{k}-\sigma K\bar x^{k}\|^{2}
+\frac{1}{2\tau}\|x^{k+1}-x^{k}+\tau K^{\top}y^{k+1}\|^{2}.
\tag{17}
\]

\noindent\textbf{[Step 6]} Expand the two squared norms in $\Delta^{k}$ using Lemma \ref{lem:lin} and the definition $\bar x^{k}=x^{k}+\theta(x^{k}-x^{k-1})$ (with $\theta=1$ for simplicity, the argument extends to any $\theta\in[0,1]$). After straightforward algebra one obtains  

\[
\Delta^{k}\ge
\frac{1-\tau\sigma\|K\|^{2}}{2\sigma}\|K\bar x^{k}\|^{2}
+\frac{1-\tau\sigma\|K\|^{2}}{2\tau}\|K^{\top}y^{k+1}\|^{2}.
\tag{18}
\]

Because of the step‑size condition \eqref{6}, the coefficients in \eqref{18} are non‑negative, hence $\Delta^{k}\ge0$.  

\noindent\textbf{[Step 7]} Drop $\Delta^{k}$ from \eqref{16} (since it is non‑negative) and use the saddle‑point inequalities  

\[
f(x^{\star})+ \langle Kx^{\star},y\rangle - g^{*}(y)
\le
f(x^{\star})+ \langle Kx^{\star},y^{\star}\rangle - g^{*}(y^{\star})
\le
f(x)+ \langle Kx, y^{\star}\rangle - g^{*}(y^{\star}),
\tag{19}
\]

to rewrite the right‑hand side of \eqref{16} as the primal–dual gap evaluated at $(x^{k+1},y^{k+1})$:

\[
\Phi^{k}-\Phi^{k+1}
\ge
\mathcal{L}(x^{k+1},y^{\star})-\mathcal{L}(x^{\star},y^{k+1})
=
\mathcal{G}_{k+1}.
\tag{20}
\]

\noindent\textbf{[Step 8]} Summing \eqref{20} from $k=0$ to $K-1$ yields a telescoping series:

\[
\sum_{k=0}^{K-1}\mathcal{G}_{k+1}
\le
\Phi^{0}-\Phi^{K}
\le
\Phi^{0}
= \frac{1}{2\tau}\|x^{0}-x^{\star}\|^{2}
  +\frac{1}{2\sigma}\|y^{0}-y^{\star}\|^{2}
=: C.
\tag{21}
\]

\noindent\textbf{[Step 9]} By convexity of the gap in each argument, the average gap satisfies  

\[
\mathcal{G}_{K}
:=\sup_{y}\mathcal{L}(\bar x^{K},y)-\inf_{x}\mathcal{L}(x,\bar y^{K})
\le
\frac{1}{K}\sum_{k=0}^{K-1}\mathcal{G}_{k+1}
\le \frac{C}{K}.
\tag{22}
\]

Equation \eqref{22} is exactly the bound \eqref{8} claimed in Theorem \ref{thm:rate}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{22} we have an explicit constant  

\[
C=
\frac{1}{2\tau}\|x^{0}-x^{\star}\|^{2}
+\frac{1}{2\sigma}\|y^{0}-y^{\star}\|^{2}.
\]

If the initial iterates are chosen within a radius $R$ of the solution (i.e., $\|x^{0}-x^{\star}\|\le R$, $\|y^{0}-y^{\star}\|\le R$), then  

\[
\mathcal{G}_{K}\le \frac{R^{2}}{2}\Bigl(\frac{1}{\tau}+\frac{1}{\sigma}\Bigr)\frac{1}{K}.
\tag{23}
\]

Thus halving both step sizes halves the constant but also tightens the admissible range \eqref{6}.

\section*{Special Cases}
\begin{enumerate}[noitemsep]
\item \textbf{Pure Primal Gradient Descent.} When $g\equiv0$ and $K=I$, the dual variable collapses to $y^{k}=0$ for all $k$, and \eqref{2} reduces to $x^{k+1}= \prox_{\tau f}(x^{k})$, i.e. the proximal point method. The $O(1/k)$ ergodic rate coincides with the classic result for convex nonsmooth optimization.
\item \textbf{Total Variation Denoising.} Take $f(x)=\frac12\|x-b\|^{2}$, $g(u)=\lambda\|u\|_{1}$, $K$ the discrete gradient operator. PDHG updates become a gradient step on the data term and a soft‑thresholding step on the dual variable, exactly the algorithm popularized by Chambolle–Pock for TV.
\item \textbf{Saddle‑Point Problems with Strong Convexity.} If $f$ is $\mu$‑strongly convex, a similar derivation (adding a strong‑convexity term) yields a linear convergence rate $O\bigl((1-\rho)^{k}\bigr)$ with $\rho=\min\{\mu\tau,\,\frac{1}{\sigma\|K\|^{2}}\}$.
\end{enumerate}
