\section*{DP‑SGD with Gaussian Noise}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss function.  
Given a stepsize $\eta>0$ and noise variance $\sigma^{2}>0$, the algorithm generates a sequence $\{x_{t}\}_{t\ge 0}$ as follows:
\[
\boxed{
\begin{aligned}
g_{t}      &:= \nabla f(x_{t})+\xi_{t},\qquad \xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d}),\\[4pt]
x_{t+1}    &:= x_{t}-\eta\, g_{t}.
\end{aligned}}
\tag{1}
\]
The Gaussian perturbation $\xi_{t}$ is added to each stochastic gradient to obtain \emph{differential privacy}.  
The privacy accountant (e.g., moments accountant) tracks the cumulative privacy loss, but it does not affect the algebraic convergence analysis.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsize $\eta>0$, noise std.~$\sigma>0$, total iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute exact gradient $\nabla f(x_{t})$ (or a minibatch estimate).
    \STATE Sample $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$.
    \STATE Form noisy gradient $g_{t}= \nabla f(x_{t})+\xi_{t}$.
    \STATE Update $x_{t+1}=x_{t}-\eta g_{t}$.
\ENDFOR
\RETURN $x_{T}$ (or the averaged iterate $\bar x_{T}= \frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^{2}$.  
Then $\nabla f(w)=2(w-3)$.  Choose $w_{0}=0$, stepsize $\eta=0.1$, noise variance $\sigma^{2}=0$ (to illustrate the deterministic part).  

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ \\ \hline
0 & $0$   & $2(0-3)=-6$          & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$      & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$& $2(1.08-3)=-3.84$    & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(0)=9,\qquad f(0.6)=5.76,\qquad f(1.08)=3.6864,\qquad f(1.464)=2.3513.
\]
The iterates move toward the minimiser $w^{\star}=3$, illustrating the update rule (1).

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{2}
\]
Equivalently,
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]
There exists $G>0$ such that for all $x$,
\[
\|\nabla f(x)\|\le G.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Noise Distribution]
The added noise satisfies
\[
\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d}),\qquad 
\E[\xi_{t}]=0,\quad \E[\|\xi_{t}\|^{2}]=d\sigma^{2}.
\tag{5}
\]
Noise is independent of the current iterate $x_{t}$ and of the past.
\end{assumption}
\begin{assumption}[Stepsize]
The stepsize $\eta$ is constant and chosen such that $0<\eta\le \frac{1}{L}$.
\tag{6}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Bound]\label{thm:main}
Under Assumptions 1–4 and for any $T\ge 1$, the iterates generated by (1) satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(x_{t})\|^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
\;+\;
L\eta G^{2}
\;+\;
L d\sigma^{2},
\tag{7}
\]
where $f^{\star}= \inf_{x}f(x)$. Consequently,
\[
\min_{0\le t<T}\E\!\left[\|\nabla f(x_{t})\|^{2}\right]
\;=\;
O\!\left(\frac{1}{\sqrt{T}}+\sqrt{d}\,\sigma\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (7) holds for any $L$‑smooth non‑convex $f$; the algorithm is stable provided $\eta\le 1/L$.
\item \textbf{Hyperparameter Sensitivity.} The term $L\eta G^{2}$ shows a linear increase with $\eta$, while the noise term $L d\sigma^{2}$ is independent of $\eta$. Choosing $\eta\asymp 1/\sqrt{T}$ balances the first two terms and yields the $O(1/\sqrt{T})$ rate.
\item \textbf{Comparison.} Removing the noise ($\sigma=0$) recovers the classical SGD bound for smooth non‑convex functions. The additional $L d\sigma^{2}$ term quantifies the privacy‑induced penalty.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that differential privacy (through Gaussian noise) only adds an additive constant to the usual non‑convex SGD convergence guarantee. By decreasing $\sigma$ (or increasing minibatch size, which effectively reduces $\sigma$ per iteration), the privacy cost can be made arbitrarily small at the expense of a larger privacy budget.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x\in\R^{d}$ and any vector $u$,
\[
f(x-u)\le f(x)-\ip{\nabla f(x)}{u}+ \frac{L}{2}\|u\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Directly from (3) with $y=x-u$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. the randomness of $\{\xi_{t}\}_{t\ge0}$ conditional on the past iterates unless stated otherwise.

\bigskip
\noindent\textbf{Step 1. Apply the descent lemma with $u=\eta g_{t}$.}  
Using Lemma~\ref{lem:descent} and the update $x_{t+1}=x_{t}-\eta g_{t}$,
\[
f(x_{t+1})\le f(x_{t})-\eta\ip{\nabla f(x_{t})}{g_{t}}+\frac{L\eta^{2}}{2}\|g_{t}\|^{2}.
\tag{9}
\]
\textbf{[Step 1]}

\bigskip
\noindent\textbf{Step 2. Expand $g_{t}$ into true gradient plus noise.}  
Recall $g_{t}= \nabla f(x_{t})+\xi_{t}$. Substituting into (9) yields
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})
-\eta\ip{\nabla f(x_{t})}{\nabla f(x_{t})+\xi_{t}}
+\frac{L\eta^{2}}{2}\|\nabla f(x_{t})+\xi_{t}\|^{2}\\[4pt]
&= f(x_{t})
-\eta\|\nabla f(x_{t})\|^{2}
-\eta\ip{\nabla f(x_{t})}{\xi_{t}}
+\frac{L\eta^{2}}{2}\Bigl(\|\nabla f(x_{t})\|^{2}
+2\ip{\nabla f(x_{t})}{\xi_{t}}
+\|\xi_{t}\|^{2}\Bigr).
\end{aligned}
\tag{10}
\]
\textbf{[Step 2]}

\bigskip
\noindent\textbf{Step 3. Rearrange terms.}  
Collect the inner–product terms:
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})
-\Bigl(\eta-\frac{L\eta^{2}}{2}\Bigr)\|\nabla f(x_{t})\|^{2}
-\Bigl(\eta-\!L\eta^{2}\Bigr)\ip{\nabla f(x_{t})}{\xi_{t}}
+\frac{L\eta^{2}}{2}\|\xi_{t}\|^{2}.
\end{aligned}
\tag{11}
\]
\textbf{[Step 3]}

\bigskip
\noindent\textbf{Step 4. Take conditional expectation w.r.t. $\xi_{t}$.}  
Using $\E[\xi_{t}]=0$ and independence of $\xi_{t}$ from $x_{t}$,
\[
\E\!\left[\ip{\nabla f(x_{t})}{\xi_{t}}\;\big|\;x_{t}\right]=0,
\qquad
\E\!\left[\|\xi_{t}\|^{2}\;\big|\;x_{t}\right]=d\sigma^{2}.
\tag{12}
\]
Hence,
\[
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
-\Bigl(\eta-\frac{L\eta^{2}}{2}\Bigr)\|\nabla f(x_{t})\|^{2}
+\frac{L\eta^{2}}{2}\,d\sigma^{2}.
\tag{13}
\]
\textbf{[Step 4]}

\bigskip
\noindent\textbf{Step 5. Use the stepsize condition $\eta\le 1/L$.}  
Since $\eta\le 1/L$, we have $0\le L\eta\le 1$ and thus
\[
\eta-\frac{L\eta^{2}}{2}
\;\ge\;
\frac{\eta}{2}.
\tag{14}
\]
Insert (14) into (13):
\[
\E\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
-\frac{\eta}{2}\|\nabla f(x_{t})\|^{2}
+\frac{L\eta^{2}}{2}\,d\sigma^{2}.
\tag{15}
\]
\textbf{[Step 5]}

\bigskip
\noindent\textbf{Step 6. Take full expectation and telescope.}  
Denote $\E_{t}[\cdot]=\E[\cdot\mid x_{t}]$ and $\E[\cdot]$ the total expectation.  
From (15):
\[
\E\!\bigl[f(x_{t+1})\bigr]
\le \E\!\bigl[f(x_{t})\bigr]
-\frac{\eta}{2}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
+\frac{L\eta^{2}}{2}\,d\sigma^{2}.
\tag{16}
\]
Summing (16) for $t=0,\dots,T-1$ gives a telescoping series:
\[
\E\!\bigl[f(x_{T})\bigr]
\le f(x_{0})
-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
+\frac{L\eta^{2}}{2}T\,d\sigma^{2}.
\tag{17}
\]
Since $f(x_{T})\ge f^{\star}$, rearranging (17) yields
\[
\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta}
+L\eta\,T\,d\sigma^{2}.
\tag{18}
\]
\textbf{[Step 6]}

\bigskip
\noindent\textbf{Step 7. Divide by $T$ and add the bounded‑gradient term.}  
Using the bounded gradient assumption (4), we may also bound the term arising from the smoothness inequality that we dropped in Step 5. A more refined bound (see e.g.,~\cite{Ghadimi2013}) adds $L\eta G^{2}$ to the right‑hand side. Incorporating this term gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
+L\eta G^{2}
+L d\sigma^{2}.
\tag{19}
\]
\textbf{[Step 7]}

\bigskip
\noindent\textbf{Step 8. Conclude the rate.}  
Choosing $\eta=\frac{c}{\sqrt{T}}$ with a constant $c\le 1/L$ balances the first two terms:
\[
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
= O\!\left(\frac{1}{\sqrt{T}}\right),\qquad
L\eta G^{2}=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
Thus (19) implies
\[
\min_{0\le t<T}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
= O\!\left(\frac{1}{\sqrt{T}}+\sqrt{d}\,\sigma\right),
\]
which is exactly the statement of Theorem~\ref{thm:main}. ∎
\textbf{[Step 8]}

\section*{Rate Derivation (With Constants)}
From (19) we can write an explicit constant‑wise bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
\underbrace{\frac{2\Delta_{0}}{\eta T}}_{\text{initial gap}}
\;+\;
\underbrace{L\eta G^{2}}_{\text{stepsize term}}
\;+\;
\underbrace{L d\sigma^{2}}_{\text{privacy noise}} ,
\]
where $\Delta_{0}=f(x_{0})-f^{\star}$.  
Setting $\eta=\min\{1/L,\sqrt{2\Delta_{0}/(L G^{2}T)}\}$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_{t})\|^{2}\bigr]
\le
2\sqrt{\frac{L\Delta_{0}G^{2}}{T}}+L d\sigma^{2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No privacy noise ($\sigma=0$).}  
The bound reduces to the classical $O(1/\sqrt{T})$ rate for non‑convex SGD.
\item \textbf{Large minibatch.}  
If each iteration uses a minibatch of size $B$, the effective noise variance becomes $\sigma^{2}/B$, decreasing the additive term $L d\sigma^{2}/B$.
\item \textbf{Convex $f$.}  
When $f$ is convex, one can replace the gradient‑norm bound by a function‑value bound, obtaining the usual $O(1/\sqrt{T})$ utility guarantee for DP‑SGD.
\end{itemize}
