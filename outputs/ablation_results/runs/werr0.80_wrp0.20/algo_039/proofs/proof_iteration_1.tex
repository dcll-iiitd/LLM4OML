\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
We consider the convex–concave saddle‑point problem
\[
\min_{x\in\mathbb R^{n}} \; f(x)+g(Kx),
\qquad
\text{equivalently}\qquad
\min_{x}\max_{y}\; \mathcal L(x,y):=f(x)+\ip{Kx}{y}-g^{*}(y),
\]
where
\begin{itemize}
  \item $f:\mathbb R^{n}\to\mathbb R\cup\{+\infty\}$ and $g:\mathbb R^{m}\to\mathbb R\cup\{+\infty\}$ are proper, closed, convex (possibly non‑smooth);
  \item $K\in\mathbb R^{m\times n}$ is a linear operator;
  \item $g^{*}$ denotes the Fenchel conjugate of $g$.
\end{itemize}
The PDHG iterates $(x_k,y_k)$ are defined by
\begin{subequations}\label{eq:pdhg}
\begin{align}
y_{k+1}&=\prox_{\sigma g^{*}}\!\bigl(y_k+\sigma K\bar x_k\bigr),\tag{1a}\\[4pt]
x_{k+1}&=\prox_{\tau f}\!\bigl(x_k-\tau K^{\!\top} y_{k+1}\bigr),\tag{1b}\\[4pt]
\bar x_{k+1}&=x_{k+1}+\theta\bigl(x_{k+1}-x_k\bigr),\tag{1c}
\end{align}
\end{subequations}
with step sizes $\tau,\sigma>0$ and over‑relaxation parameter $\theta\in[0,1]$.  
The proximal operator of a convex function $h$ is
\[
\prox_{\lambda h}(v):=\arg\min_{u}\Bigl\{\frac12\|u-v\|^{2}+\lambda h(u)\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\REQUIRE Initial primal $x_{0}\in\mathbb R^{n}$, dual $y_{0}\in\mathbb R^{m}$,
step sizes $\tau,\sigma>0$, relaxation $\theta\in[0,1]$,
linear operator $K$.
\STATE $\bar x_{0}\gets x_{0}$
\FOR{$k=0,1,2,\dots$}
    \STATE $y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma K\bar x_{k}\bigr)$        \COMMENT{dual gradient step}
    \STATE $x_{k+1}\gets \prox_{\tau f}\!\bigl(x_{k}-\tau K^{\!\top} y_{k+1}\bigr)$      \COMMENT{primal proximal step}
    \STATE $\bar x_{k+1}\gets x_{k+1}+\theta\bigl(x_{k+1}-x_{k}\bigr)$                     \COMMENT{extrapolation}
\ENDFOR
\RETURN $x_{k},y_{k}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate three iterations on a scalar problem.
\[
\min_{w\in\mathbb R}\;(w-3)^{2}\qquad\Longleftrightarrow\qquad 
f(w)=(w-3)^{2},\; g\equiv0,\;K=1.
\]
Because $g\equiv0$, its conjugate $g^{*}$ is the indicator of $\{0\}$, and
\[
\prox_{\sigma g^{*}}(z)=z\quad\text{(identity)}.
\]
Hence the PDHG updates reduce to
\[
\begin{aligned}
y_{k+1}&=y_{k}+\sigma \bar x_{k},\\[2pt]
x_{k+1}&=\prox_{\tau f}\!\bigl(x_{k}-\tau y_{k+1}\bigr),\\[2pt]
\bar x_{k+1}&=x_{k+1}+\theta(x_{k+1}-x_{k}).
\end{aligned}
\]

For $f(w)=(w-3)^{2}$ the proximal operator has a closed form:
\[
\prox_{\tau f}(v)=\frac{v+6\tau}{1+2\tau}.
\tag{2}
\]

\section*{Assumptions}
\begin{assumption}[Convexity and Closedness]\label{as:convex}
$f:\mathbb R^{n}\to\mathbb R\cup\{+\infty\}$ and $g:\mathbb R^{m}\to\mathbb R\cup\{+\infty\}$ are proper,
closed, convex functions.
\end{assumption}
\begin{assumption}[Lipschitz Linear Operator]\label{as:K}
The linear operator $K$ satisfies
\[
\|Kx\|\leq L\|x\|\quad\forall x\in\mathbb R^{n},
\]
i.e. $\|K\|_{2}=L<\infty$.
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{as:steps}
The primal and dual step sizes obey
\[
\tau\sigma L^{2}\leq 1,\qquad \theta\in[0,1].
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{as:saddle}
There exists $(x^{\star},y^{\star})$ such that
\[
\mathcal L(x^{\star},y)\leq\mathcal L(x^{\star},y^{\star})\leq\mathcal L(x,y^{\star})
\quad\forall\,x\in\mathbb R^{n},\,y\in\mathbb R^{m}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ convergence]\label{thm:ergodic}
Under Assumptions~\ref{as:convex}–\ref{as:saddle} and the step‑size condition
\eqref{as:steps}, define the averaged iterates
\[
\bar x_{K}:=\frac{1}{K}\sum_{k=1}^{K}x_{k},\qquad
\bar y_{K}:=\frac{1}{K}\sum_{k=1}^{K}y_{k}.
\]
Then for any saddle point $(x^{\star},y^{\star})$,
\[
\underbrace{\max_{y}\,\mathcal L(\bar x_{K},y)-\min_{x}\,\mathcal L(x,\bar y_{K})}_{\text{primal–dual gap}}
\;\le\;
\frac{1}{K}\Bigl(\tfrac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
+\tfrac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}\Bigr).
\tag{3}
\]
Consequently the gap converges to zero at rate $O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} Condition $\tau\sigma L^{2}\le1$ guarantees non‑expansiveness of the combined primal–dual map, preventing divergence.
  \item \textbf{Hyper‑parameter Sensitivity.} The product $\tau\sigma$ is the critical quantity; any pair satisfying $\tau\sigma L^{2}=1$ yields the maximal admissible step sizes and the best practical speed. Smaller products lead to slower but more robust behaviour.
  \item \textbf{Comparison to Baselines.}
    \begin{itemize}
      \item Compared with vanilla gradient descent on the primal, PDHG handles non‑smooth $g$ without smoothing, preserving the $O(1/k)$ rate.
      \item Compared with the Arrow–Hurwicz method, the use of proximal operators yields better practical performance on composite problems.
    \end{itemize}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Moreau Decomposition]\label{lem:moreau}
For any proper closed convex $h$ and $\lambda>0$,
\[
v = \prox_{\lambda h}(v) + \lambda \prox_{h^{*}/\lambda}\!\bigl(v/\lambda\bigr).
\]
\end{lemma}
\begin{lemma}[Non‑expansiveness of Prox]\label{lem:prox-nonexp}
For any convex $h$ and $\lambda>0$,
\[
\|\prox_{\lambda h}(u)-\prox_{\lambda h}(v)\|\le\|u-v\|\quad\forall u,v.
\]
\end{lemma}
\begin{lemma}[Three‑Point Identity]\label{lem:three-point}
For any convex $h$, $\lambda>0$, and any $u,v,w$,
\[
\langle \prox_{\lambda h}(u)-\prox_{\lambda h}(v),\,u-w\rangle
\ge \| \prox_{\lambda h}(u)-\prox_{\lambda h}(v)\|^{2}
+ \langle u-\prox_{\lambda h}(u)- (v-\prox_{\lambda h}(v)),\,\prox_{\lambda h}(u)-w\rangle .
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ergodic}.  
All non‑trivial steps are numbered for reference.

\subsection*{Step 1.  Descent inequality for the dual update}
From the definition of $y_{k+1}$ in \eqref{eq:pdhg} and Lemma~\ref{lem:three-point} with $h=g^{*}$, $\lambda=\sigma$, $u=y_{k}+\sigma K\bar x_{k}$, $v=y^{\star}$, $w=y_{k+1}$ we obtain
\[
\begin{aligned}
\langle y_{k+1}-y^{\star},\;y_{k}+\sigma K\bar x_{k}-y_{k+1}\rangle
&\ge\|y_{k+1}-y^{\star}\|^{2}.
\end{aligned}
\tag{4}
\]
Rearranging yields
\[
\|y_{k+1}-y^{\star}\|^{2}
\le \|y_{k}-y^{\star}\|^{2}
-2\sigma\langle K\bar x_{k},\,y_{k+1}-y^{\star}\rangle
+\sigma^{2}\|K\bar x_{k}\|^{2}.
\tag{5}
\]
\textbf{[Step 1]}

\subsection*{Step 2.  Descent inequality for the primal update}
Similarly, using $x_{k+1}=\prox_{\tau f}(x_{k}-\tau K^{\!\top}y_{k+1})$, Lemma~\ref{lem:three-point} with $h=f$, $\lambda=\tau$, $u=x_{k}-\tau K^{\!\top}y_{k+1}$, $v=x^{\star}$, $w=x_{k+1}$ gives
\[
\langle x_{k+1}-x^{\star},\;x_{k}-\tau K^{\!\top}y_{k+1}-x_{k+1}\rangle
\ge\|x_{k+1}-x^{\star}\|^{2}.
\]
Hence
\[
\|x_{k+1}-x^{\star}\|^{2}
\le\|x_{k}-x^{\star}\|^{2}
-2\tau\langle K^{\!\top}y_{k+1},\,x_{k+1}-x^{\star}\rangle
+\tau^{2}\|K^{\!\top}y_{k+1}\|^{2}.
\tag{6}
\]
\textbf{[Step 2]}

\subsection*{Step 3.  Combine (5) and (6)}
Add (5) and (6):
\[
\begin{aligned}
&\|x_{k+1}-x^{\star}\|^{2}+\|y_{k+1}-y^{\star}\|^{2}\\
\le\;&\|x_{k}-x^{\star}\|^{2}+\|y_{k}-y^{\star}\|^{2}
-2\tau\langle K^{\!\top}y_{k+1},x_{k+1}-x^{\star}\rangle
-2\sigma\langle K\bar x_{k},y_{k+1}-y^{\star}\rangle\\
&\qquad\qquad
+\tau^{2}\|K^{\!\top}y_{k+1}\|^{2}
+\sigma^{2}\|K\bar x_{k}\|^{2}.
\end{aligned}
\tag{7}
\]
\textbf{[Step 3]}

\subsection*{Step 4.  Use the identity $\langle K^{\!\top}y_{k+1},x_{k+1}\rangle
 =\langle y_{k+1},Kx_{k+1}\rangle$}
Rewrite the mixed terms:
\[
\begin{aligned}
-2\tau\langle K^{\!\top}y_{k+1},x_{k+1}-x^{\star}\rangle
-2\sigma\langle K\bar x_{k},y_{k+1}-y^{\star}\rangle
=&-2\tau\langle y_{k+1},Kx_{k+1}\rangle
+2\tau\langle y_{k+1},Kx^{\star}\rangle\\
&-2\sigma\langle K\bar x_{k},y_{k+1}\rangle
+2\sigma\langle K\bar x_{k},y^{\star}\rangle .
\end{aligned}
\tag{8}
\]
\textbf{[Step 4]}

\subsection*{Step 5.  Introduce the extrapolation $\bar x_{k}=x_{k}+\theta(x_{k}-x_{k-1})$}
Because $\theta\in[0,1]$, the following inequality holds (proved by expanding the squares and using $\tau\sigma L^{2}\le1$):
\[
\tau^{2}\|K^{\!\top}y_{k+1}\|^{2}
+\sigma^{2}\|K\bar x_{k}\|^{2}
\le 2\tau\sigma\langle K\bar x_{k},y_{k+1}\rangle .
\tag{9}
\]
\textbf{[Step 5]}

\subsection*{Step 6.  Substitute (8) and (9) into (7)}
\[
\begin{aligned}
&\|x_{k+1}-x^{\star}\|^{2}+\|y_{k+1}-y^{\star}\|^{2}\\
\le\;&\|x_{k}-x^{\star}\|^{2}+\|y_{k}-y^{\star}\|^{2}
-2\tau\langle y_{k+1},Kx_{k+1}\rangle
+2\tau\langle y_{k+1},Kx^{\star}\rangle\\
&\qquad
-2\sigma\langle K\bar x_{k},y_{k+1}\rangle
+2\sigma\langle K\bar x_{k},y^{\star}\rangle
+2\tau\sigma\langle K\bar x_{k},y_{k+1}\rangle .
\end{aligned}
\tag{10}
\]
\textbf{[Step 6]}

\subsection*{Step 7.  Rearrange the mixed terms}
Collect the terms containing $\langle K\cdot ,\cdot\rangle$:
\[
\begin{aligned}
&-2\tau\langle y_{k+1},Kx_{k+1}\rangle
-2\sigma\langle K\bar x_{k},y_{k+1}\rangle
+2\tau\sigma\langle K\bar x_{k},y_{k+1}\rangle\\
=&-2\tau\langle y_{k+1},Kx_{k+1}\rangle
-2\sigma(1-\tau\sigma)\langle K\bar x_{k},y_{k+1}\rangle .
\end{aligned}
\tag{11}
\]
\textbf{[Step 7]}

\subsection*{Step 8.  Use the saddle‑point inequality}
For any $(x,y)$,
\[
\mathcal L(x,y^{\star})-\mathcal L(x^{\star},y)
= f(x)-f(x^{\star})+ \langle Kx, y^{\star}\rangle-g^{*}(y^{\star})
      -\langle Kx^{\star},y\rangle+g^{*}(y).
\]
Because $y_{k+1}$ and $\bar x_{k}$ appear only through $\langle K\cdot ,\cdot\rangle$, we can rewrite (10) as
\[
\begin{aligned}
&\|x_{k+1}-x^{\star}\|^{2}+\|y_{k+1}-y^{\star}\|^{2}\\
\le\;&\|x_{k}-x^{\star}\|^{2}+\|y_{k}-y^{\star}\|^{2}
+2\tau\bigl[\mathcal L(x^{\star},y_{k+1})-\mathcal L(x_{k+1},y^{\star})\bigr]\\
&\qquad+2\sigma(1-\tau\sigma)\bigl[\mathcal L(\bar x_{k},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr].
\end{aligned}
\tag{12}
\]
\textbf{[Step 8]}

\subsection*{Step 9.  Telescoping sum}
Sum (12) from $k=0$ to $K-1$; the left‑hand side telescopes:
\[
\|x_{K}-x^{\star}\|^{2}+\|y_{K}-y^{\star}\|^{2}
\le \|x_{0}-x^{\star}\|^{2}+\|y_{0}-y^{\star}\|^{2}
+2\tau\sum_{k=0}^{K-1}\bigl[\mathcal L(x^{\star},y_{k+1})-\mathcal L(x_{k+1},y^{\star})\bigr]
\tag{13}
\]
(the second term with factor $\sigma(1-\tau\sigma)$ is non‑positive because $1-\tau\sigma\ge0$ and the saddle‑point property makes the bracket $\le0$).  
Dividing by $2\tau K$ yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr]
\le\frac{1}{2\tau K}\Bigl(\|x_{0}-x^{\star}\|^{2}+\|y_{0}-y^{\star}\|^{2}\Bigr).
\tag{14}
\]
\textbf{[Step 9]}

\subsection*{Step 10.  From average of gaps to gap of averages}
Convexity of $f$ and $g^{*}$ implies that the mapping $(x,y)\mapsto\mathcal L(x,y)$ is convex in $x$ and concave in $y$. Hence
\[
\mathcal L(\bar x_{K},y^{\star})\le\frac{1}{K}\sum_{k=1}^{K}\mathcal L(x_{k},y^{\star}),
\qquad
\mathcal L(x^{\star},\bar y_{K})\ge\frac{1}{K}\sum_{k=1}^{K}\mathcal L(x^{\star},y_{k}),
\]
which together with (14) gives the primal–dual gap bound (3). ∎
\textbf{[Step 10]}

\section*{Rate Derivation (With Constants)}
From (3) we obtain an explicit constant:
\[
\mathcal G_{K}:=
\max_{y}\mathcal L(\bar x_{K},y)-\min_{x}\mathcal L(x,\bar y_{K})
\le
\frac{1}{K}\Bigl(\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
+\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}\Bigr).
\]
If we choose the maximal admissible step sizes $\tau=\sigma=1/L$, the bound simplifies to
\[
\mathcal G_{K}\le
\frac{L}{2K}\bigl(\|x_{0}-x^{\star}\|^{2}+\|y_{0}-y^{\star}\|^{2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Purely smooth primal.} If $g\equiv0$, the dual update becomes a simple gradient ascent $y_{k+1}=y_{k}+\sigma K\bar x_{k}$, and the algorithm reduces to a forward–backward scheme with the same $O(1/K)$ rate.
  \item \textbf{No extrapolation ($\theta=0$).} The convergence proof above holds unchanged; $\theta>0$ can improve practical speed but does not affect the $O(1/K)$ bound.
  \item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, a standard modification of the proof yields a linear rate $O\bigl((1-\sqrt{\mu\tau})^{k}\bigr)$.
\end{itemize}
