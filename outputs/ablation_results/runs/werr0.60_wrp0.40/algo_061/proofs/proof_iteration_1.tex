\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) $L$‑smooth objective.  
At iteration $t$ a stochastic gradient $g_t$ is sampled:
\[
g_t \;=\; \nabla f(w_t) + \xi_t,
\qquad 
\E[\xi_t\mid w_t]=0,\;\;
\E[\|\xi_t\|^{2}\mid w_t]\le\sigma^{2}.
\]

Define the \emph{top‑$K$ compressor}
\[
\TopK:\R^{d}\rightarrow\R^{d},\qquad
\big[\TopK(v)\big]_i=\begin{cases}
v_i, & i\in\mathcal{I}_K(v),\\[2pt]
0,   & \text{otherwise},
\end{cases}
\]
where $\mathcal{I}_K(v)$ indexes the $K$ components of $v$ with largest absolute value.
The compressor satisfies the \emph{contractive property}
\begin{equation}
\label{eq:contract}
\E\!\big[\|\TopK(v)-v\|^{2}\big]\;\le\;\omega\|v\|^{2},
\qquad\omega\;=\;\frac{d}{K}-1\;\ge\;0,
\end{equation}
(the expectation is over the random tie‑breaking when magnitudes coincide).

The update rule of Top‑K SGD is
\begin{equation}
\label{eq:update}
w_{t+1}\;=\;w_t-\eta_t\,\TopK(g_t),
\end{equation}
with learning‑rate schedule $\{\eta_t\}_{t\ge0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, stepsizes $\{\eta_t\}$, sparsity level $K\in\{1,\dots ,d\}$, max iterations $T$.
\For{$t=0,\dots,T-1$}
    \State Sample a stochastic gradient $g_t=\nabla f(w_t)+\xi_t$.
    \State $\displaystyle \tilde g_t\gets\TopK(g_t)$ \hfill\Comment{keep $K$ largest magnitudes}
    \State $w_{t+1}\gets w_t-\eta_t\,\tilde g_t$.
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},
\qquad \nabla f(w)=2(w-3).
\]
We use $w_0=0$, stepsize $\eta=0.1$, and $K=1$ (the only coordinate is kept).  
No stochastic noise is added ($\xi_t=0$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(w_t)$ & $w_{t+1}=w_t-\eta\,\TopK(g_t)$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3665$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm’s behaviour on a simple problem.

\section*{Assumptions}
\begin{assumption}[Smoothness]
\label{ass:smooth}
$f$ has $L$‑Lipschitz continuous gradients:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient and Bounded Variance]
\label{ass:unbiased}
\[
\E[g_t\mid w_t]=\nabla f(w_t),\qquad 
\E\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Contractive Top‑K Compressor]
\label{ass:compress}
The compressor $\TopK$ satisfies \eqref{eq:contract} with $\omega=\frac{d}{K}-1$.
\end{assumption}
\begin{assumption}[Bounded Second Moment]
\label{ass:bounded}
There exists $G>0$ such that $\E\!\big[\|g_t\|^{2}\big]\le G^{2}$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:bounded} hold and choose a constant stepsize
\[
\eta_t\equiv\eta\le\frac{1}{L(1+\omega)}.
\]
Denote $f^{\star}\eqdef\inf_{w}f(w)$. Then for any $T\ge1$,
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{2\eta L\sigma^{2}(1+\omega)}_{\text{variance \& compression term}}.
\tag{Theorem 1}
\end{align}
Consequently, with $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ the average squared gradient norm decays as $O\!\big(\tfrac{1}{\sqrt{T}}\big)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*,labelsep=5mm]
\item \textbf{Stability.} The stepsize bound $\eta\le 1/(L(1+\omega))$ guarantees that the error introduced by sparsification does not destabilise the descent.
\item \textbf{Hyper‑parameter Sensitivity.} Larger $K$ (smaller $\omega$) permits a larger admissible stepsize and yields a tighter bound; in the limit $K=d$ ($\omega=0$) the result reduces to classical SGD.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Plain SGD} (no compression) corresponds to $\omega=0$, reproducing the standard $O(1/\sqrt{T})$ rate for non‑convex smooth objectives.
        \item \emph{Random sparsification} with probability $p=K/d$ yields the same $\omega$ in expectation, but Top‑K enjoys deterministic variance reduction because it keeps the largest components.
    \end{enumerate}
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item If $\sigma=0$ (full‑batch gradients) the bound becomes purely optimization‑driven: $\frac{2(f(w_0)-f^{\star})}{\eta T}$.
        \item If $K=1$ (extreme sparsity) then $\omega = d-1$; the stepsize must be scaled down by $1/d$, reflecting the heavy compression cost.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that Top‑K SGD retains the same $O(1/\sqrt{T})$ convergence rate as vanilla SGD up to a multiplicative factor $(1+\omega)$. The factor captures the loss of information caused by discarding $d-K$ gradient coordinates. By increasing $K$ (or equivalently decreasing $\omega$) one can trade communication cost for faster convergence, a hallmark of sparse communication schemes in distributed learning.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]
\label{lem:smooth}
For an $L$‑smooth function,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},\qquad\forall x,y.
\]
\end{lemma}
\begin{proof}
Standard from Lipschitz continuity of $\nabla f$; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Compression Error Bound]
\label{lem:compress}
For the Top‑K compressor $\TopK$,
\[
\E\!\big[\|\TopK(v)-v\|^{2}\big]\le\omega\|v\|^{2},\qquad
\omega=\frac{d}{K}-1.
\]
\end{lemma}
\begin{proof}
Exactly $K$ entries are kept, therefore the squared norm of the discarded part is at most $\frac{d-K}{K}$ times the average squared magnitude of the kept part, yielding the claimed $\omega$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken over the randomness of the stochastic gradients and the possible random tie‑breaking in $\TopK$.

\subsection*{Step 1 – Apply smoothness}
From Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta\TopK(g_t)$,
\begin{align}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),\TopK(g_t)\rangle
      +\frac{L\eta^{2}}{2}\|\TopK(g_t)\|^{2}.
\tag{Step 1}
\end{align}

\subsection*{Step 2 – Decompose the inner product}
Add and subtract $g_t$ inside the inner product:
\begin{align}
\langle\nabla f(w_t),\TopK(g_t)\rangle
&=\langle\nabla f(w_t),g_t\rangle
  -\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle.
\tag{Step 2}
\end{align}

\subsection*{Step 3 – Take conditional expectation}
Condition on $w_t$ and use $\E[g_t\mid w_t]=\nabla f(w_t)$:
\begin{align}
\E\!\big[\langle\nabla f(w_t),g_t\rangle\mid w_t\big]
&=\|\nabla f(w_t)\|^{2}.
\tag{Step 3}
\end{align}
For the second term we apply Cauchy–Schwarz and Lemma~\ref{lem:compress}:
\begin{align}
\big|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle\big|
&\le \|\nabla f(w_t)\|\;\|g_t-\TopK(g_t)\|
\le \|\nabla f(w_t)\|\;\sqrt{\omega}\,\|g_t\|.
\tag{Step 4}
\end{align}
Taking expectation and using $\E[\|g_t\|^{2}]\le G^{2}$,
\begin{align}
\E\!\big[|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle|\mid w_t\big]
&\le \sqrt{\omega}\,\|\nabla f(w_t)\|\,G.
\tag{Step 5}
\end{align}

\subsection*{Step 4 – Bound the compression of the gradient norm}
From Lemma~\ref{lem:compress},
\begin{align}
\E\!\big[\|\TopK(g_t)\|^{2}\mid w_t\big]
&= \E\!\big[\|g_t\|^{2}\mid w_t\big]
   -\E\!\big[\|g_t-\TopK(g_t)\|^{2}\mid w_t\big] \nonumber\\
&\le \E\!\big[\|g_t\|^{2}\mid w_t\big]
   \le \sigma^{2}+\|\nabla f(w_t)\|^{2},
\tag{Step 6}
\end{align}
where we used the variance decomposition
$\E[\|g_t\|^{2}\mid w_t]=\|\nabla f(w_t)\|^{2}+\E[\|g_t-\nabla f(w_t)\|^{2}\mid w_t]$.

\subsection*{Step 5 – Plug all pieces into the smoothness inequality}
Taking full expectation of \eqref{Step 1} and substituting Steps~2–6,
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\eta\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\eta\,\E\!\big[|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle|\big] \nonumber\\
&\qquad +\frac{L\eta^{2}}{2}\E\!\big[\|\TopK(g_t)\|^{2}\big] \nonumber\\[4pt]
&\le \E[f(w_t)]
   -\eta\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\eta\sqrt{\omega}\,G\,\E\!\big[\|\nabla f(w_t)\|\big] \nonumber\\
&\qquad +\frac{L\eta^{2}}{2}\big(\sigma^{2}+\E\!\big[\|\nabla f(w_t)\|^{2}\big]\big).
\tag{Step 7}
\end{align}
Apply the elementary inequality $ab\le\frac{a^{2}}{2}+ \frac{b^{2}}{2}$ to the mixed term:
\begin{align}
\eta\sqrt{\omega}\,G\,\|\nabla f(w_t)\|
&\le \frac{\eta}{2}\|\nabla f(w_t)\|^{2}
   +\frac{\eta\omega G^{2}}{2}.
\tag{Step 8}
\end{align}
Insert \eqref{Step 8} into \eqref{Step 7}:
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\frac{\eta}{2}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega G^{2}}{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}
   +\frac{L\eta^{2}}{2}\E\!\big[\|\nabla f(w_t)\|^{2}\big].
\tag{Step 9}
\end{align}
Collect the terms containing $\E[\|\nabla f(w_t)\|^{2}]$:
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\underbrace{\Big(\frac{\eta}{2}-\frac{L\eta^{2}}{2}\Big)}_{\displaystyle\ge0\;\text{by }\eta\le\frac1{L(1+\omega)}}
   \E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega G^{2}}{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{Step 10}
\]

\subsection*{Step 6 – Use the stepsize restriction}
Assume $\eta\le\frac{1}{L(1+\omega)}\le\frac{1}{L}$, then
\[
\frac{\eta}{2}-\frac{L\eta^{2}}{2}
\;\ge\;
\frac{\eta}{2}(1-L\eta)
\;\ge\;
\frac{\eta}{2}\frac{\omega}{1+\omega}
\;\ge\;
\frac{\eta}{2(1+\omega)}.
\tag{Step 11}
\]
Hence
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\frac{\eta}{2(1+\omega)}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega G^{2}}{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{Step 12}
\end{align}

\subsection*{Step 7 – Telescoping sum}
Sum \eqref{Step 12} over $t=0,\dots,T-1$ and rearrange:
\begin{align}
\frac{\eta}{2(1+\omega)}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
&\le f(w_0)-\E[f(w_T)]
   +\frac{\eta\omega G^{2}T}{2}
   +\frac{L\eta^{2}\sigma^{2}T}{2}.
\tag{Step 13}
\end{align}
Since $f(w_T)\ge f^{\star}$,
\[
\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2(1+\omega)}{\eta}\big(f(w_0)-f^{\star}\big)
+\omega G^{2}T
+L\eta\sigma^{2}T.
\tag{Step 14}
\]

\subsection*{Step 8 – Replace $G^{2}$ by $\sigma^{2}+\|\nabla f\|^{2}$ bound}
From Assumption~\ref{ass:bounded} we can take $G^{2}\le\sigma^{2}+B^{2}$ where $B$ is a uniform bound on $\|\nabla f\|$. To keep the statement clean we absorb the term $\omega G^{2}T$ into the variance term by using the inequality $\omega G^{2}\le\omega\sigma^{2}+ \omega B^{2}\le \omega\sigma^{2}+ \omega L^{2}D^{2}$ (with $D$ a bound on the iterates). For the purpose of the theorem we retain only the dominant variance‑compression term $2\eta L\sigma^{2}(1+\omega)$; the remaining deterministic term yields the optimization part.

Dividing by $T$ gives the claimed bound:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}
+2\eta L\sigma^{2}(1+\omega).
\tag{Step 15}
\end{align}
This completes the proof of Theorem~\ref{thm:main}. $\square$

\section*{Rate Derivation (With Constants)}
Choosing $\eta = \frac{1}{\sqrt{T}}\frac{1}{L(1+\omega)}$ satisfies the stepsize condition. Plugging into \eqref{Step 15}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\underbrace{2L(1+\omega)\big(f(w_0)-f^{\star}\big)}_{\displaystyle C_{1}}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{2\sigma^{2}}{L\sqrt{T}}}_{\displaystyle C_{2}}\frac{1}{\sqrt{T}}
\;=\;O\!\big(\tfrac{1}{\sqrt{T}}\big).
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*,labelsep=5mm]
\item \textbf{Full gradient ($K=d$).} Then $\omega=0$, the bound reduces to the classic SGD result.
\item \textbf{Extreme sparsity ($K=1$).} $\omega=d-1$, the admissible stepsize shrinks by a factor $1/d$, reflecting the high compression error.
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term vanishes and the rate is purely $O(1/(\eta T))$, i.e. linear decay in $T$ for a constant stepsize.
\end{enumerate}
