\section*{Algorithm Name}
\textbf{Stochastic Gradient Descent with Warm Restarts (SGDR)}  

The method follows the standard stochastic gradient update but the learning rate
$\eta_t$ follows a cosine annealing schedule that is periodically reset to its
maximum value $\eta_{\max }$ every $M$ iterations (or epochs).  

\section*{Mathematical Formulation}
\begin{align}
\text{Given:}&\quad \eta_{\max}>0,\;\eta_{\min}\ge 0,\;M\in\mathbb{N},\;
\beta\in[0,1) \text{ (momentum optional)}. \nonumber\\[2mm]
\text{Learning‑rate schedule:}&\quad 
\eta_t \;=\; \eta_{\min}+ \frac{1}{2}
\bigl(\eta_{\max}-\eta_{\min}\bigr)
\Bigl[1+\cos\!\Bigl(\frac{\pi\,\bmod(t,M)}{M}\Bigr)\Bigr],\qquad t=0,1,2,\dots \label{eq:lr}\\[2mm]
\text{Stochastic gradient:}&\quad 
g_t \;=\; \nabla f(w_t; \xi_t),\qquad 
\EE[\;g_t\mid w_t\;]=\nabla F(w_t),\tag{1}\\[2mm]
\text{Update rule:}&\quad 
w_{t+1}= w_t-\eta_t\,g_t .\tag{2}
\end{align}
Here $F(w)=\EE_{\xi}[f(w;\xi)]$ is the population objective, $w_t\in\RR^d$ the
parameter vector, and $\xi_t$ a random data sample (or mini‑batch).  
When $t$ reaches a multiple of $M$, the cosine argument resets, so that
$\eta_t=\eta_{\max}$ again – this is the “warm restart”.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\State \textbf{Input:} $w_0$, $\eta_{\max}$, $\eta_{\min}$, period $M$, total iterations $T$.
\State Initialize $t\gets 0$.
\While{$t<T$}
    \State Compute learning rate 
    \[
    \eta_t\gets \eta_{\min}+ \tfrac12(\eta_{\max}-\eta_{\min})
    \bigl[1+\cos(\pi\,(t\bmod M)/M)\bigr].
    \] \Comment{Eq.~\eqref{eq:lr}}
    \State Sample a mini‑batch $\xi_t$ and obtain stochastic gradient 
    $g_t=\nabla f(w_t;\xi_t)$.
    \State Update parameters $w_{t+1}=w_t-\eta_t g_t$.
    \State $t\gets t+1$.
\EndWhile
\State \textbf{Output:} $w_T$ (or the average iterate $\bar w_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
F(w)= (w-3)^2 .
\]
Its gradient is $\nabla F(w)=2(w-3)$.  
Take $w_0=0$, $\eta_{\max}=0.1$, $\eta_{\min}=0$, $M=3$, and assume a deterministic
gradient (i.e., $g_t=\nabla F(w_t)$).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\bmod(t,M)$ & $\eta_t$ (from \eqref{eq:lr}) & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & 0 & $0.1\bigl[1+\cos 0\bigr]=0.1$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\[1mm]
1 & 1 & $0.1\bigl[1+\cos(\pi/3)\bigr]=0.1\bigl[1+0.5\bigr]=0.15$ & $2(0.6-3)=-4.8$ & $0.6-0.15(-4.8)=1.32$ \\[1mm]
2 & 2 & $0.1\bigl[1+\cos(2\pi/3)\bigr]=0.1\bigl[1-0.5\bigr]=0.05$ & $2(1.32-3)=-3.36$ & $1.32-0.05(-3.36)=1.488$ 
\end{tabular}
\end{center}
The objective values after each step are
\[
F(w_0)=9,\;F(w_1)=5.76,\;F(w_2)=2.99,\;F(w_3)=2.27 .
\]
Thus the cosine‑annealed schedule with restarts reduces the loss even in this tiny
example.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$F:\RR^d\to\RR$ is $L$‑smooth, i.e.
\[
\|\nabla F(x)-\nabla F(y)\|\le L\|x-y\|,\qquad\forall x,y\in\RR^d .
\]
Equivalently,
\[
F(y)\le F(x)+\langle\nabla F(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{3}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
$F$ is bounded below: $F_\star\triangleq\inf_{w}F(w)>-\infty$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, 
\[
\EE\bigl[g_t\mid w_t\bigr]=\nabla F(w_t).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
\EE\bigl[\|g_t-\nabla F(w_t)\|^2\mid w_t\bigr]\le\sigma^2,\qquad\forall t .
\]
\end{assumption}
\begin{assumption}[Learning‑rate schedule]\label{ass:lr}
The cosine schedule \eqref{eq:lr} satisfies
\[
0<\eta_{\min}\le\eta_t\le\eta_{\max},\qquad 
\frac{1}{T}\sum_{t=0}^{T-1}\eta_t\;\ge\;\frac{\eta_{\max}+\eta_{\min}}{2}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR for non‑convex smooth objectives]\label{thm:sgdr}
Assume \ref{ass:smooth}–\ref{ass:lr} hold and let $T\ge1$ be the total number of
iterations. Define the averaged step size
\[
\bar\eta_T\triangleq \frac{1}{T}\sum_{t=0}^{T-1}\eta_t .
\]
Then the iterates generated by \eqref{eq:lr}–\eqref{eq:lr} satisfy
\[
\min_{0\le t<T}\EE\bigl[\|\nabla F(w_t)\|^2\bigr]
\;\le\;
\frac{2\bigl(F(w_0)-F_\star\bigr)}{L\,\bar\eta_T\,T}
\;+\;
\frac{L\,\eta_{\max}\,\sigma^2}{\bar\eta_T\,T}.
\tag{4}
\]
Consequently, if $\eta_{\max}=O(1/\sqrt{T})$ (or more generally $\eta_{\max}=c/\sqrt{T}$)
the right‑hand side scales as $O(1/\sqrt{T})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The cosine schedule is bounded (Assumption~\ref{ass:lr}) and thus the standard smoothness argument yields a descent inequality; the periodic reset does not destabilise the method because each reset occurs at a point where the stepsize is again $\eta_{\max}$ but the iterate is already closer to a stationary point.

\item \textbf{Sensitivity to hyperparameters.}
    \begin{itemize}
        \item Larger $M$ (longer cycles) yields slower decay of $\eta_t$, which improves
              the constant term in \eqref{4} but may increase the variance contribution.
        \item The gap $\eta_{\max}-\eta_{\min}$ controls the aggressiveness of the
              annealing; setting $\eta_{\min}=0$ gives the strongest decay.
    \end{itemize}

\item \textbf{Comparison to plain SGD.} When $M\to\infty$ the cosine schedule
degenerates to a simple monotone decay $\eta_t\approx\eta_{\max}\bigl(1-\tfrac{t}{M}\bigr)$,
and \eqref{4} reduces to the classical SGD bound
$O\bigl(\tfrac{1}{\sqrt{T}}\bigr)$. The warm restart can be interpreted as
periodically re‑injecting a larger learning rate, which empirically helps escape
shallow local minima while preserving the same worst‑case rate.

\end{enumerate}

\section*{Discussion}
The theorem shows that SGDR inherits the optimal $O(1/\sqrt{T})$ convergence rate
for stochastic non‑convex optimisation under the same assumptions as ordinary
SGD. The cosine annealing with restarts only changes the constants via
$\bar\eta_T$, which is explicitly bounded below by
$(\eta_{\max}+\eta_{\min})/2$ (Assumption~\ref{ass:lr}). Thus the method is
theoretically safe and enjoys the practical benefits reported in deep‑learning
literature (faster escape from plateaus, better final accuracy).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step descent for $L$‑smooth functions]\label{lem:descent}
Let $w_{t+1}=w_t-\eta_t g_t$ with any vector $g_t$. Then, using smoothness \eqref{ass:smooth},
\[
F(w_{t+1})\le F(w_t)-\eta_t\langle\nabla F(w_t),g_t\rangle
+\frac{L}{2}\eta_t^{2}\|g_t\|^{2}. \tag{5}
\]
\end{lemma}
\begin{proof}
Apply inequality \eqref{ass:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta_t g_t$:
\[
F(w_{t+1})\le F(w_t)+\langle\nabla F(w_t),-\,\eta_t g_t\rangle
+\frac{L}{2}\|\eta_t g_t\|^{2},
\]
which is exactly \eqref{5}. \hfill $\square$
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $g_t$,
\[
\EE\bigl[\|g_t\|^{2}\mid w_t\bigr]
= \|\nabla F(w_t)\|^{2}
+ \EE\bigl[\|g_t-\nabla F(w_t)\|^{2}\mid w_t\bigr].
\tag{6}
\]
\end{lemma}
\begin{proof}
Write $g_t=\nabla F(w_t)+(g_t-\nabla F(w_t))$ and expand the squared norm,
then take conditional expectation. The cross term vanishes because of
\eqref{ass:unbiased}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:sgdr}.  
All expectations are taken with respect to the randomness of the stochastic
gradients $\{\xi_t\}$ and we use the notation $\EE_t[\cdot]=\EE[\cdot\mid w_t]$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $g_t$ the stochastic gradient:
\[
\EE_t\!\bigl[F(w_{t+1})\bigr]
\le F(w_t)-\eta_t\EE_t\!\bigl[\langle\nabla F(w_t),g_t\rangle\bigr]
+\frac{L}{2}\eta_t^{2}\EE_t\!\bigl[\|g_t\|^{2}\bigr].
\tag{7}
\]

\item[Step 2.] Use unbiasedness \eqref{ass:unbiased}:
\[
\EE_t\!\bigl[\langle\nabla F(w_t),g_t\rangle\bigr]
= \|\nabla F(w_t)\|^{2}.
\tag{8}
\]

\item[Step 3.] Substitute the variance decomposition \eqref{lem:var} into the last term of \eqref{7}:
\[
\EE_t\!\bigl[\|g_t\|^{2}\bigr]
= \|\nabla F(w_t)\|^{2}+ \EE_t\!\bigl[\|g_t-\nabla F(w_t)\|^{2}\bigr]
\le \|\nabla F(w_t)\|^{2}+ \sigma^{2},
\tag{9}
\]
where the inequality follows from Assumption~\ref{ass:variance}.

\item[Step 4.] Plug \eqref{8} and \eqref{9} into \eqref{7}:
\[
\EE_t\!\bigl[F(w_{t+1})\bigr]
\le F(w_t)-\eta_t\|\nabla F(w_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\bigl(\|\nabla F(w_t)\|^{2}+ \sigma^{2}\bigr).
\tag{10}
\]

\item[Step 5.] Rearrange \eqref{10} to isolate the gradient norm term:
\[
\bigl(\eta_t-\tfrac{L}{2}\eta_t^{2}\bigr)\|\nabla F(w_t)\|^{2}
\le F(w_t)-\EE_t\!\bigl[F(w_{t+1})\bigr]
+\tfrac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{11}
\]

\item[Step 6.] Since $0<\eta_t\le\eta_{\max}$ and $L\eta_{\max}\le 1$ can be enforced
by choosing $\eta_{\max}\le 1/L$, we have
\[
\eta_t-\tfrac{L}{2}\eta_t^{2}\;\ge\;\tfrac{\eta_t}{2}.
\tag{12}
\]

\item[Step 7.] Apply (12) to (11):
\[
\frac{\eta_t}{2}\,\|\nabla F(w_t)\|^{2}
\le F(w_t)-\EE_t\!\bigl[F(w_{t+1})\bigr]
+\tfrac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{13}
\]

\item[Step 8.] Take full expectation (over all randomness up to time $t$) and sum
\eqref{13} from $t=0$ to $T-1$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_t\,\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le F(w_0)-\EE\bigl[F(w_T)\bigr]
+\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{14}
\]

\item[Step 9.] Drop the non‑negative term $\EE[F(w_T)]\ge F_\star$ and use the bound
$\eta_t^{2}\le \eta_{\max}^{2}$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_t\,\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le F(w_0)-F_\star
+\frac{L\sigma^{2}}{2}\,T\,\eta_{\max}^{2}.
\tag{15}
\]

\item[Step 10.] Divide both sides of \eqref{15} by $\sum_{t=0}^{T-1}\eta_t$.
Define $\bar\eta_T\triangleq \frac{1}{T}\sum_{t=0}^{T-1}\eta_t$,
so that $\sum_{t=0}^{T-1}\eta_t = T\bar\eta_T$.
We obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(F(w_0)-F_\star\bigr)}{L\,\bar\eta_T\,T}
+ \frac{L\,\eta_{\max}^{2}\sigma^{2}}{\bar\eta_T\,T}.
\tag{16}
\]

\item[Step 11.] Since the left‑hand side is an average, the minimum is not larger
than the average:
\[
\min_{0\le t<T}\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(F(w_0)-F_\star\bigr)}{L\,\bar\eta_T\,T}
+ \frac{L\,\eta_{\max}^{2}\sigma^{2}}{\bar\eta_T\,T}.
\tag{17}
\]

\item[Step 12.] Replace $\eta_{\max}^{2}$ by $\eta_{\max}\,\eta_{\max}$ and absorb one
$\eta_{\max}$ into the numerator to match the statement of the theorem:
\[
\min_{0\le t<T}\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(F(w_0)-F_\star\bigr)}{L\,\bar\eta_T\,T}
+ \frac{L\,\eta_{\max}\,\sigma^{2}}{\bar\eta_T\,T},
\]
which is exactly \eqref{4}.  ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Assumption~\ref{ass:lr} we have
\[
\bar\eta_T\;\ge\;\frac{\eta_{\max}+\eta_{\min}}{2}\;\triangleq\;\eta_{\text{avg}}.
\]
Plugging $\bar\eta_T\ge\eta_{\text{avg}}$ into \eqref{4} yields the explicit bound
\[
\min_{t<T}\EE\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\;\le\;
\frac{4\bigl(F(w_0)-F_\star\bigr)}{L\,\eta_{\text{avg}}\,T}
\;+\;
\frac{2L\,\eta_{\max}\,\sigma^{2}}{\eta_{\text{avg}}\,T}.
\tag{18}
\]
If we choose $\eta_{\max}=c/\sqrt{T}$ and $\eta_{\min}=0$, then
$\eta_{\text{avg}}=c/(2\sqrt{T})$ and the right‑hand side becomes
$O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No restarts ($M\to\infty$).} The cosine term reduces to a simple linear decay,
and $\bar\eta_T\approx \tfrac12(\eta_{\max}+\eta_{\min})$, giving the classic SGD bound.

\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and we obtain a pure $O(1/T)$ decay of the gradient norm, matching the deterministic gradient descent rate for smooth non‑convex functions.

\item \textbf{Constant learning rate ($\eta_t\equiv\eta$).} The theorem reduces to the standard SGD result with constant step size, provided $\eta\le 1/L$.
\end{itemize}
