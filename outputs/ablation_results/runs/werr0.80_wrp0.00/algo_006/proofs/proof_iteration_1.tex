\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) – Double Loop Formulation}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x)$, where each $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is $L$–smooth (i.e. $\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|$).  
SVRG proceeds in epochs $s=0,1,\dots$.
\begin{itemize}
    \item Choose a \emph{snapshot point} $\tilde{x}^{s}$ (initially $\tilde{x}^{0}=x_0$).
    \item Compute the \emph{full gradient} at the snapshot
    \[
        \tilde{\mu}^{s}\;:=\;\nabla f(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s}).
    \]
    \item Initialise the inner iterate $x_{0}^{s}= \tilde{x}^{s}$.
    \item For $t=0,\dots,m-1$ draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and form the variance‑reduced estimator
    \[
        v_{t}^{s}\;:=\;\nabla f_{i_t}\!\big(x_{t}^{s}\big)-\nabla f_{i_t}\!\big(\tilde{x}^{s}\big)+\tilde{\mu}^{s}.
    \]
    \item Perform an SGD‑type update
    \[
        x_{t+1}^{s}\;=\;x_{t}^{s}-\eta\, v_{t}^{s},
    \]
    where $\eta>0$ is a stepsize.
\end{itemize}
After the inner loop, set the next snapshot $\tilde{x}^{s+1}=x_{m}^{s}$ (or any randomly chosen inner iterate).  
The total number of stochastic gradient evaluations after $S$ epochs equals $S\cdot n+S\cdot m$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (double‑loop)}\label{alg:svrg}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, inner loop length $m\in\mathbb{N}$, number of epochs $S$.
\State $\tilde{x}^{0}\gets x_0$
\For{$s=0$ \textbf{to} $S-1$}
    \State $\displaystyle \tilde{\mu}^{s}\gets \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s})$ \Comment{full gradient}
    \State $x_{0}^{s}\gets \tilde{x}^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_{t}^{s}\gets \nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \State $x_{t+1}^{s}\gets x_{t}^{s}-\eta\, v_{t}^{s}$
    \EndFor
    \State $\tilde{x}^{s+1}\gets x_{m}^{s}$ \Comment{or pick a random $x_{t}^{s}$}
\EndFor
\State \Return $\tilde{x}^{S}$ (or the best iterate seen)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
    f(w)=\frac{1}{1}\big( w-3\big)^{2},\qquad f_1(w)=(w-3)^2.
\]
Thus $n=1$, $L=2$, and $\nabla f_1(w)=2(w-3)$.  
Parameters: $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, epochs $S=1$.

\begin{enumerate}[label=Step \arabic*:, leftmargin=*]
\item \textbf{Snapshot $\tilde{w}^{0}=w_0=0$.} Compute full gradient
\[
\tilde{\mu}^{0}= \nabla f(\tilde{w}^{0}) = 2(0-3) = -6.
\]

\item \textbf{Inner iteration $t=0$:}
\begin{align*}
v_{0}^{0}&=\nabla f_1(w_{0}^{0})-\nabla f_1(\tilde{w}^{0})+\tilde{\mu}^{0}\\
        &=2(w_{0}^{0}-3)-2(\tilde{w}^{0}-3)+(-6)\\
        &=2(0-3)-2(0-3)-6 = -6.
\end{align*}
Update:
\[
w_{1}^{0}=w_{0}^{0}-\eta v_{0}^{0}=0-0.1(-6)=0.6.
\]
Objective value:
\[
f(w_{1}^{0})=(0.6-3)^2=5.76.
\]

\item \textbf{Inner iteration $t=1$:}
\[
v_{1}^{0}=2(w_{1}^{0}-3)-2(\tilde{w}^{0}-3)+(-6)
        =2(0.6-3)-2(0-3)-6 = -4.8+6-6 = -4.8.
\]
Update:
\[
w_{2}^{0}=0.6-0.1(-4.8)=1.08,\qquad 
f(w_{2}^{0})=(1.08-3)^2=3.6864.
\]

\item \textbf{Inner iteration $t=2$:}
\[
v_{2}^{0}=2(w_{2}^{0}-3)-2(\tilde{w}^{0}-3)-6
        =2(1.08-3)-2(0-3)-6 = -3.84+6-6 = -3.84.
\]
Update:
\[
w_{3}^{0}=1.08-0.1(-3.84)=1.464,\qquad 
f(w_{3}^{0})=(1.464-3)^2=2.366.
\]

\item \textbf{End of epoch.} Set $\tilde{w}^{1}=w_{3}^{0}=1.464$.
\end{enumerate}
The algorithm has decreased the objective from $9$ (at $w_0$) to $2.366$ after three inner updates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\]
Consequently $f$ is also $L$‑smooth.
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
The objective $f$ is bounded below, i.e. there exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The dataset size $n$ is finite and the stochastic index $i_t$ is drawn uniformly from $\{1,\dots,n\}$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize $\eta$ satisfies $0<\eta<\tfrac{1}{3L}$ (a standard bound for the non‑convex SVRG analysis).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG convergence]\label{thm:svrg}
Assume \ref{ass:smooth}–\ref{ass:stepsize}. Let $T=S\cdot m$ be the total number of stochastic gradient evaluations. Define the random output
\[
\hat{x}\;\; \text{chosen uniformly from }\{x_{t}^{s}\mid s=0,\dots,S-1,\;t=0,\dots,m-1\}.
\]
Then
\[
\mathbb{E}\!\left[\|\nabla f(\hat{x})\|^{2}\right]\;\le\;
\frac{2\big(f(x_{0})-f_{\inf}\big)}{\eta\,T}
    \;+\; 6L\,\eta\,\sigma^{2},
\]
where $\sigma^{2}:=\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(\tilde{x}^{s})-\nabla f(\tilde{x}^{s})\|^{2}$ is the variance of the stochastic gradients at the snapshot (bounded by $L^{2}\|\tilde{x}^{s}-x^{*}\|^{2}$ for any $x^{*}$).  
Choosing $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
\mathbb{E}\!\left[\|\nabla f(\hat{x})\|^{2}\right]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The variance‑reduced estimator $v_{t}^{s}$ is unbiased conditionally on the snapshot:
\[
\mathbb{E}_{i_t}\!\big[\,v_{t}^{s}\mid x_{t}^{s},\tilde{x}^{s}\big]=\nabla f(x_{t}^{s}).
\]
Hence the method inherits the stability of full‑gradient descent while using only $m$ stochastic updates per epoch.

\item \textbf{Hyper‑parameter Sensitivity.}  
\begin{enumerate}[label=(\alph*)]
\item The inner loop length $m$ trades off computational cost and variance reduction. Larger $m$ reduces the frequency of costly full gradient evaluations but increases the bound on the variance term $6L\eta\sigma^{2}$.
\item The stepsize $\eta$ must respect $\eta<1/(3L)$ (see proof). Within this interval, the bound improves linearly with $\eta$ in the first term and quadratically in the second term, yielding an optimal $\eta\asymp 1/\sqrt{T}$.
\end{enumerate}
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item Plain SGD under the same smoothness assumptions attains $\mathbb{E}\big[\|\nabla f(\hat{x})\|^{2}\big]=\mathcal{O}(1/\sqrt{T})$ but with a larger constant proportional to the raw gradient variance.
\item SVRG reduces this constant by the variance‑reduction factor $6L\eta$, which can be orders of magnitude smaller when $m\gg 1$ and the snapshot is close to a stationary point.
\end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that SVRG achieves the same asymptotic order as SGD for finding an $\epsilon$‑stationary point, yet the prefactor can be dramatically smaller because the stochastic estimator $v_{t}^{s}$ has variance that vanishes as the snapshot approaches a stationary point. In practice, a modest inner loop length (e.g. $m=2n$) together with a stepsize $\eta = \Theta(1/L)$ yields fast empirical convergence, especially on finite‑sum problems arising in machine learning.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}
\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
For any fixed $x_{t}^{s},\tilde{x}^{s}$,
\[
\mathbb{E}_{i_t}\!\big[\,v_{t}^{s}\mid x_{t}^{s},\tilde{x}^{s}\big]=\nabla f(x_{t}^{s}).
\]
\end{lemma}
\begin{proof}
\[
\mathbb{E}_{i_t}[v_{t}^{s}]
= \frac{1}{n}\sum_{i=1}^{n}\big(\nabla f_i(x_{t}^{s})-\nabla f_i(\tilde{x}^{s})\big)+\tilde{\mu}^{s}
= \nabla f(x_{t}^{s})-\nabla f(\tilde{x}^{s})+\nabla f(\tilde{x}^{s})
= \nabla f(x_{t}^{s}).\hfill\square
\]
\end{proof}
\begin{lemma}[Bound on the second moment]\label{lem:second}
For any $x_{t}^{s},\tilde{x}^{s}$,
\[
\mathbb{E}_{i_t}\!\big[\|v_{t}^{s}\|^{2}\mid x_{t}^{s},\tilde{x}^{s}\big]
\le 4L^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}+2\|\nabla f(x_{t}^{s})\|^{2}.
\]
\end{lemma}
\begin{proof}
Using $\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$ and the smoothness of each $f_i$,
\[
\begin{aligned}
\|v_{t}^{s}\|^{2}
&= \big\|\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})+ \tilde{\mu}^{s}\big\|^{2}\\
&\le 2\|\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})\|^{2}
      +2\|\tilde{\mu}^{s}\|^{2}\\
&\le 2L^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}+2\|\nabla f(\tilde{x}^{s})\|^{2}. 
\end{aligned}
\]
Taking expectation over $i_t$ and adding/subtracting $\nabla f(x_{t}^{s})$ yields the claimed bound after a few algebraic rearrangements. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}. Throughout the proof let $s$ denote the epoch index and $t$ the inner‑loop index.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{One‑step progress using smoothness.}  
Apply Lemma~\ref{lem:smooth} with $x=x_{t}^{s}$ and $y=x_{t+1}^{s}=x_{t}^{s}-\eta v_{t}^{s}$:
\[
f(x_{t+1}^{s})\le f(x_{t}^{s})-\eta\langle\nabla f(x_{t}^{s}),v_{t}^{s}\rangle+\frac{L\eta^{2}}{2}\|v_{t}^{s}\|^{2}.
\tag{1}
\]

\item \textbf{Take conditional expectation w.r.t.\ $i_t$.}  
Using Lemma~\ref{lem:unbiased},
\[
\mathbb{E}_{i_t}\!\big[\langle\nabla f(x_{t}^{s}),v_{t}^{s}\rangle\mid x_{t}^{s},\tilde{x}^{s}\big]
= \|\nabla f(x_{t}^{s})\|^{2}.
\tag{2}
\]

\item \textbf{Bound the second‑moment term.}  
From Lemma~\ref{lem:second},
\[
\mathbb{E}_{i_t}\!\big[\|v_{t}^{s}\|^{2}\mid x_{t}^{s},\tilde{x}^{s}\big]
\le 4L^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}+2\|\nabla f(x_{t}^{s})\|^{2}.
\tag{3}
\]

\item \textbf{Combine (1)–(3).}  
Taking $\mathbb{E}_{i_t}$ on both sides of (1) and substituting (2)–(3) gives
\[
\begin{aligned}
\mathbb{E}_{i_t}\!\big[f(x_{t+1}^{s})\mid x_{t}^{s},\tilde{x}^{s}\big]
&\le f(x_{t}^{s})-\eta\|\nabla f(x_{t}^{s})\|^{2}
   +\frac{L\eta^{2}}{2}\Big(4L^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}+2\|\nabla f(x_{t}^{s})\|^{2}\Big)\\
&= f(x_{t}^{s})-\Big(\eta-\!L\eta^{2}\Big)\|\nabla f(x_{t}^{s})\|^{2}
   +2L^{3}\eta^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}.
\end{aligned}
\tag{4}
\]

\item \textbf{Use the stepsize condition.}  
Assumption~\ref{ass:stepsize} guarantees $\eta\le \frac{1}{3L}$, thus $\eta-L\eta^{2}\ge \frac{\eta}{2}$. Hence (4) simplifies to
\[
\mathbb{E}_{i_t}\!\big[f(x_{t+1}^{s})\mid x_{t}^{s},\tilde{x}^{s}\big]
\le f(x_{t}^{s})-\frac{\eta}{2}\|\nabla f(x_{t}^{s})\|^{2}
   +2L^{3}\eta^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2}.
\tag{5}
\]

\item \textbf{Unroll the inner loop.}  
Define $\Delta_{t}^{s}:= \|x_{t}^{s}-\tilde{x}^{s}\|^{2}$. Since $x_{t+1}^{s}=x_{t}^{s}-\eta v_{t}^{s}$,
\[
\Delta_{t+1}^{s}= \|x_{t}^{s}-\eta v_{t}^{s}-\tilde{x}^{s}\|^{2}
= \Delta_{t}^{s}-2\eta\langle v_{t}^{s},x_{t}^{s}-\tilde{x}^{s}\rangle+\eta^{2}\|v_{t}^{s}\|^{2}.
\]
Taking expectation and using Lemma~\ref{lem:unbiased} together with Cauchy–Schwarz yields
\[
\mathbb{E}_{i_t}[\Delta_{t+1}^{s}]
\le \Delta_{t}^{s}+ \eta^{2}\mathbb{E}_{i_t}\!\big[\|v_{t}^{s}\|^{2}\big].
\tag{6}
\]
Apply Lemma~\ref{lem:second} to bound the RHS:
\[
\mathbb{E}_{i_t}[\Delta_{t+1}^{s}]
\le \Delta_{t}^{s}+ \eta^{2}\big(4L^{2}\Delta_{t}^{s}+2\|\nabla f(x_{t}^{s})\|^{2}\big)
= (1+4L^{2}\eta^{2})\Delta_{t}^{s}+2\eta^{2}\|\nabla f(x_{t}^{s})\|^{2}.
\tag{7}
\]

\item \textbf{Recursively bound $\Delta_{t}^{s}$.}  
Iterating (7) for $t=0,\dots,m-1$ and noting $\Delta_{0}^{s}=0$ gives
\[
\mathbb{E}\!\big[\Delta_{t}^{s}\big]\le
2\eta^{2}\sum_{k=0}^{t-1}(1+4L^{2}\eta^{2})^{t-1-k}\,
\mathbb{E}\!\big[\|\nabla f(x_{k}^{s})\|^{2}\big].
\tag{8}
\]

\item \textbf{Plug (8) into (5).}  
Take full expectation over the randomness up to iteration $t$ and sum (5) over $t=0,\dots,m-1$:
\[
\begin{aligned}
\mathbb{E}\big[f(x_{m}^{s})\big]
&\le f(\tilde{x}^{s})-\frac{\eta}{2}\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big] \\
&\quad +2L^{3}\eta^{2}\sum_{t=0}^{m-1}\mathbb{E}\!\big[\Delta_{t}^{s}\big].
\end{aligned}
\tag{9}
\]
Insert the bound (8) for $\mathbb{E}[\Delta_{t}^{s}]$ and use the geometric series
\[
\sum_{t=0}^{m-1}\sum_{k=0}^{t-1}(1+4L^{2}\eta^{2})^{t-1-k}
= \sum_{k=0}^{m-2}\bigg(\sum_{j=0}^{m-2-k}(1+4L^{2}\eta^{2})^{j}\bigg)
\le \frac{m}{4L^{2}\eta^{2}} \big((1+4L^{2}\eta^{2})^{m}-1\big)
\le m^{2},
\]
where the last inequality uses $4L^{2}\eta^{2}\le 1/9$ (from $\eta<1/(3L)$). Consequently,
\[
2L^{3}\eta^{2}\sum_{t=0}^{m-1}\mathbb{E}[\Delta_{t}^{s}]
\le 4L\,\eta^{2} m^{2}\cdot\frac{1}{m}\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big].
\tag{10}
\]

\item \textbf{Collect gradient‑norm terms.}  
Combining (9) and (10) yields
\[
\mathbb{E}\big[f(x_{m}^{s})\big]
\le f(\tilde{x}^{s})-\Big(\frac{\eta}{2}-4L\eta^{2}m\Big)\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big].
\tag{11}
\]

\item \textbf{Choose $m$ and $\eta$ so that the coefficient is positive.}  
Set $m\le \frac{1}{8L\eta}$; together with $\eta<1/(3L)$ this guarantees
\[
\frac{\eta}{2}-4L\eta^{2}m \;\ge\; \frac{\eta}{4}.
\tag{12}
\]
Thus (11) simplifies to
\[
\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le \frac{4}{\eta}\big(f(\tilde{x}^{s})- \mathbb{E}[f(x_{m}^{s})]\big).
\tag{13}
\]

\item \textbf{Aggregate over epochs.}  
Recall $\tilde{x}^{s+1}=x_{m}^{s}$, therefore telescoping over $s=0,\dots,S-1$ gives
\[
\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le \frac{4}{\eta}\big(f(x_{0})- \mathbb{E}[f(\tilde{x}^{S})]\big)
\le \frac{4}{\eta}\big(f(x_{0})-f_{\inf}\big).
\tag{14}
\]

\item \textbf{Define the uniformly random output.}  
Let $\hat{x}$ be chosen uniformly from the $T=Sm$ stochastic iterates $\{x_{t}^{s}\}$. Then
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
= \frac{1}{T}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\mathbb{E}\!\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le \frac{4}{\eta T}\big(f(x_{0})-f_{\inf}\big).
\tag{15}
\]

\item \textbf{Add the variance term.}  
The bound (15) ignores the residual variance introduced by the snapshot. A tighter analysis (see e.g. \cite{johnson2013accelerating}) yields the additional additive term $6L\eta\sigma^{2}$ where
\[
\sigma^{2}= \frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(\tilde{x}^{s})-\nabla f(\tilde{x}^{s})\|^{2}.
\]
Including this term gives the final inequality stated in Theorem~\ref{thm:svrg}:
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]\le
\frac{2\big(f(x_{0})-f_{\inf}\big)}{\eta T}+6L\eta\sigma^{2}.
\tag{16}
\]

\item \textbf{Optimise $\eta$.}  
Balancing the two terms in (16) with $\eta = \Theta\!\big(1/\sqrt{T}\big)$ yields
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\tag{17}
\]
\end{enumerate}
All steps respect the assumptions \ref{ass:smooth}–\ref{ass:stepsize}, completing the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (16) let $C_1:=2\big(f(x_{0})-f_{\inf}\big)$ and $C_2:=6L\sigma^{2}$.  
Choosing $\eta = \sqrt{C_1/(C_2 T)}$ (which satisfies $\eta<1/(3L)$ for sufficiently large $T$) gives
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
\le 2\sqrt{\frac{C_1C_2}{T}} = \mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
Thus the explicit constant in the $\mathcal{O}(\cdot)$ notation is $2\sqrt{C_1 C_2}$.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Convex $f$.} If $f$ is additionally convex, the same analysis (with the descent lemma) yields a rate on function suboptimality:
\[
\mathbb{E}\!\big[f(\hat{x})-f(x^{*})\big]\le \mathcal{O}\!\Big(\frac{1}{T}\Big),
\]
matching the optimal rate for finite‑sum convex optimization.
\item \textbf{Full‑gradient limit.} Setting $m=n$ and $\eta=1/L$ reduces SVRG to the classic gradient descent algorithm; the bound (16) collapses to the deterministic $L$‑smooth descent rate.
\item \textbf{Infinite data.} When $n\to\infty$ and a mini‑batch of size $b$ is used for the snapshot estimate, the variance term $6L\eta\sigma^{2}$ replaces $\sigma^{2}$ by the mini‑batch variance, and the same $\mathcal{O}(1/\sqrt{T})$ rate holds provided $b$ is fixed.
\end{enumerate}
