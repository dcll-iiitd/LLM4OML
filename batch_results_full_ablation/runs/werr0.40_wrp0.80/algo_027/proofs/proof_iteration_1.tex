\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
Probabilistic Coordinate Descent (PCD)

\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be differentiable.  
Denote by $\nabla_i f(w)$ the $i$‑th partial derivative and by $e_i\in\mathbb{R}^d$ the $i$‑th canonical basis vector.  

\begin{itemize}
    \item \textbf{Probability distribution.} A fixed probability vector $p=(p_1,\dots ,p_d)$ satisfies $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.
    \item \textbf{Stochastic gradient estimator.} At iteration $k$, draw an index $i_k\in\{1,\dots ,d\}$ according to $p$.  
    Define
    \[
        g_k \;:=\; \frac{1}{p_{i_k}}\,\nabla_{i_k} f(w_k)\,e_{i_k}\in\mathbb{R}^d .
    \]
    Then $\mathbb{E}[g_k\mid w_k]=\nabla f(w_k)$ (unbiasedness).
    \item \textbf{Update rule.} With stepsize $\eta_k>0$,
    \[
        w_{k+1}= w_k - \eta_k\, g_k .
    \]
\end{itemize}

\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent (PCD)}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\mathbb{R}^d$, stepsize schedule $\{\eta_k\}_{k\ge0}$, probability vector $p$ with $p_i>0$, $ \sum_i p_i =1$.
\FOR{$k=0,1,2,\dots,T-1$}
    \STATE Sample $i_k\sim p$.
    \STATE Compute partial gradient $g_{k,i_k}:=\nabla_{i_k}f(w_k)$.
    \STATE Form unbiased estimator $g_k:=\frac{1}{p_{i_k}}\,g_{k,i_k}\,e_{i_k}$.
    \STATE Update $w_{k+1}\gets w_k-\eta_k g_k$.
\ENDFOR
\RETURN $w_T$ (or the averaged iterate $\bar w_T:=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the one‑dimensional quadratic
\[
    f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Here $d=1$, $p_1=1$, and the algorithm reduces to ordinary gradient descent.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)$ & $w_{k+1}=w_k-\eta \nabla f(w_k)$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.366$.
The iterates move toward the minimiser $w^\star=3$ as expected.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{as:smooth}
For each coordinate $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\mathbb{R}^d$ and $\alpha\in\mathbb{R}$,
\[
    \bigl|\nabla_i f(w+\alpha e_i)-\nabla_i f(w)\bigr|\le L_i|\alpha|.
\]
Equivalently,
\[
    f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^2 .
\]
\end{assumption}

\begin{assumption}[Bounded variance of stochastic estimator]\label{as:variance}
Define the stochastic estimator $g_k$ as above. There exists $\sigma^2\ge0$ such that for all $k$,
\[
    \mathbb{E}\!\left[\bigl\|g_k-\nabla f(w_k)\bigr\|^2\mid w_k\right]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsizes are constant, $\eta_k\equiv\eta$, and satisfy
\[
    0<\eta\le \min_{i}\frac{p_i}{L_i}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:stepsize}. Let $w^\star$ be any point (not necessarily a global minimum). After $T$ iterations of PCD,
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
    \;\le\;
    \frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{\eta T}
    \;+\;
    \eta\,\underbrace{\Bigl(\sum_{i=1}^d\frac{L_i}{p_i}\Bigr)}_{=:L_{\text{eff}}}
    \;+\;
    \sigma^2 .
\]
In particular, choosing $\eta=\sqrt{\dfrac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{L_{\text{eff}}\,T}}$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
    =\mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}[leftmargin=*]
    \item \textbf{Unbiasedness.} By construction $\mathbb{E}[g_k\mid w_k]=\nabla f(w_k)$, which is crucial for the descent argument.
    \item \textbf{Stability w.r.t. probabilities.} The factor $1/p_i$ appears both in the estimator and in the smoothness constant $L_{\text{eff}}$, showing a trade‑off: very small $p_i$ lead to large variance but allow larger steps on rarely‑updated coordinates.
    \item \textbf{Comparison to uniform coordinate descent.} If $p_i=1/d$ for all $i$, then $L_{\text{eff}} = d\cdot \overline{L}$ with $\overline{L}=\frac{1}{d}\sum_i L_i$, reproducing the classic $O(d^{1/2}T^{-1/2})$ rate.
    \item \textbf{Non‑convexity.} No convexity is required; the theorem guarantees convergence to a point with small gradient norm in expectation.
\end{itemize}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The result mirrors the standard analysis of stochastic gradient methods but incorporates the coordinate‑sampling distribution explicitly. By weighting the estimator with $1/p_i$, the expected update recovers the full gradient, allowing the proof to follow the usual descent lemma. The effective smoothness constant $L_{\text{eff}}$ quantifies how the sampling probabilities affect the achievable rate.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]\label{lem:descent}
Under Assumption \ref{as:smooth}, for any $w\in\mathbb{R}^d$, any $i\in\{1,\dots ,d\}$ and any $\alpha\in\mathbb{R}$,
\[
    f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^2 .
\]
\end{lemma}
\begin{proof}
Directly from the definition of $L_i$‑smoothness (see \eqref{as:smooth}). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the randomness of the index $i_k$ and any other stochasticity; conditional expectations are denoted $\mathbb{E}_k[\cdot]=\mathbb{E}[\cdot\mid w_k]$.

\begin{enumerate}
    \item[Step 1.] \textbf{Apply Lemma \ref{lem:descent}} with $\alpha=-\eta\,\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)$ for the sampled coordinate $i_k$:
    \begin{align}
        f(w_{k+1}) 
        &= f\!\bigl(w_k-\eta\,\tfrac{1}{p_{i_k}}\nabla_{i_k}f(w_k)e_{i_k}\bigr) \nonumber\\
        &\le f(w_k) -\eta\,\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)\,\nabla_{i_k}f(w_k)
           +\frac{L_{i_k}}{2}\Bigl(\eta\,\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)\Bigr)^2 . \label{eq:descent_one}
    \end{align}
    [Step 1]

    \item[Step 2.] \textbf{Rewrite the RHS using the stochastic estimator $g_k$.}
    Recall $g_k=\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)e_{i_k}$, hence $\langle g_k,\nabla f(w_k)\rangle=\frac{1}{p_{i_k}}(\nabla_{i_k}f(w_k))^2$ and $\|g_k\|^2=\frac{1}{p_{i_k}^2}(\nabla_{i_k}f(w_k))^2$.  
    Then \eqref{eq:descent_one} becomes
    \begin{equation}
        f(w_{k+1}) \le f(w_k) -\eta\langle g_k,\nabla f(w_k)\rangle
        +\frac{\eta^2}{2}\sum_{i=1}^{d} L_i\;p_i\,\|g_k\|^2 . \label{eq:descent_two}
    \end{equation}
    The equality $\sum_i L_i p_i\|g_k\|^2 = L_{i_k}\frac{1}{p_{i_k}^2}(\nabla_{i_k}f(w_k))^2$ follows because $\|g_k\|^2$ is zero on all coordinates except $i_k$. \hfill[Step 2]

    \item[Step 3.] \textbf{Take conditional expectation $\mathbb{E}_k[\cdot]$.}
    Using unbiasedness $\mathbb{E}_k[g_k]=\nabla f(w_k)$ and the definition of $L_{\text{eff}}:=\sum_{i=1}^{d}\frac{L_i}{p_i}$,
    \begin{align}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        &\le f(w_k) -\eta \|\nabla f(w_k)\|^2
           +\frac{\eta^2}{2}\,\underbrace{\mathbb{E}_k\!\Bigl[\sum_{i=1}^{d} L_i p_i \|g_k\|^2\Bigr]}_{\displaystyle =\sum_i L_i p_i \frac{1}{p_i^2}(\nabla_i f(w_k))^2 } \nonumber\\
        &= f(w_k) -\eta \|\nabla f(w_k)\|^2
           +\frac{\eta^2}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}(\nabla_i f(w_k))^2 . \label{eq:exp_descent}
    \end{align}
    \hfill[Step 3]

    \item[Step 4.] \textbf{Upper‑bound the last term by $L_{\text{eff}}\|\nabla f(w_k)\|^2$.}
    Since each term $\frac{L_i}{p_i}(\nabla_i f(w_k))^2\le \bigl(\max_i\frac{L_i}{p_i}\bigr)(\nabla_i f(w_k))^2$, we have
    \[
        \sum_{i=1}^{d}\frac{L_i}{p_i}(\nabla_i f(w_k))^2 \le L_{\text{eff}}\;\|\nabla f(w_k)\|^2 .
    \]
    Substituting into \eqref{eq:exp_descent} yields
    \begin{equation}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k) -\Bigl(\eta-\frac{\eta^2 L_{\text{eff}}}{2}\Bigr)\|\nabla f(w_k)\|^2 . \label{eq:descent_final}
    \end{equation}
    \hfill[Step 4]

    \item[Step 5.] \textbf{Choose $\eta$ satisfying Assumption \ref{as:stepsize}.}
    By $\eta\le \frac{p_i}{L_i}$ for every $i$, we have $\eta L_{\text{eff}}\le 1$. Consequently,
    \[
        \eta-\frac{\eta^2 L_{\text{eff}}}{2}\ge \frac{\eta}{2}.
    \]
    Plugging this lower bound into \eqref{eq:descent_final} gives
    \begin{equation}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k) -\frac{\eta}{2}\,\|\nabla f(w_k)\|^2 . \label{eq:descent_simplified}
    \end{equation}
    \hfill[Step 5]

    \item[Step 6.] \textbf{Take full expectation and telescope.}
    Let $\mathbb{E}$ denote expectation over all randomness up to iteration $k$. Iterating \eqref{eq:descent_simplified} and summing for $k=0,\dots ,T-1$,
    \begin{align}
        \mathbb{E}[f(w_T)] 
        &\le \mathbb{E}[f(w_0)] -\frac{\eta}{2}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr] .
    \end{align}
    Rearranging,
    \begin{equation}
        \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
        \le \frac{2\bigl(\mathbb{E}[f(w_0)]- \mathbb{E}[f(w_T)]\bigr)}{\eta T}
        \le \frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{\eta T} . \label{eq:rate_no_variance}
    \end{equation}
    \hfill[Step 6]

    \item[Step 7.] \textbf{Incorporate variance term.}
    If the variance bound \ref{as:variance} holds, a standard decomposition
    \[
        \|\nabla f(w_k)\|^2
        =\|\mathbb{E}_k[g_k]\|^2
        \le \mathbb{E}_k\!\bigl[\|g_k\|^2\bigr] 
        \le \|\nabla f(w_k)\|^2 + \sigma^2
    \]
    implies that the right‑hand side of \eqref{eq:descent_final} acquires an additive $\frac{\eta^2\sigma^2}{2}$ term. Carrying this extra term through Steps 5–6 adds $\sigma^2$ to the final bound, yielding the statement of Theorem \ref{thm:main}. \hfill[Step 7]
\end{enumerate}

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
From \eqref{eq:rate_no_variance} and the choice
\[
    \eta = \sqrt{\frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{L_{\text{eff}}\,T}},
\]
the first term becomes $\frac{2(\mathbb{E}[f(w_0)]-f^\star)}{\eta T}= \sqrt{\frac{2L_{\text{eff}}(\mathbb{E}[f(w_0)]-f^\star)}{T}}$.
The second term $\eta L_{\text{eff}}/2$ equals the same quantity, so both contribute a constant factor $2$. Hence
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
    \le 2\sqrt{\frac{2L_{\text{eff}}(\mathbb{E}[f(w_0)]-f^\star)}{T}} + \sigma^2
    =\mathcal{O}\!\bigl(T^{-1/2}\bigr) + \sigma^2 .
\]

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}
    \item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$. Then $L_{\text{eff}} = d\cdot \overline{L}$ with $\overline{L} = \frac{1}{d}\sum_i L_i$. The bound matches the classical $O\bigl(\sqrt{d/T}\bigr)$ rate for uniform coordinate descent.
    \item \textbf{Importance sampling.} Choosing $p_i \propto \sqrt{L_i}$ minimises $L_{\text{eff}} = (\sum_i \sqrt{L_i})^2$, which can be substantially smaller than the uniform case when the $L_i$ are heterogeneous.
    \item \textbf{Deterministic cyclic coordinate descent.} If $p_i=1$ for a single $i$ and $0$ for others (i.e., always update the same coordinate), the algorithm reduces to a one‑dimensional gradient descent with stepsize $\eta\le p_i/L_i = 1/L_i$, and the bound simplifies to the standard smooth‑non‑convex rate $\mathcal{O}(1/(\eta T))$.
\end{enumerate}

\section*{Conclusion}
\addcontentsline{toc}{section}{Conclusion}
We presented a fully detailed analysis of Probabilistic Coordinate Descent for non‑convex, coordinate‑wise smooth objectives. The unbiased estimator $g_k$ weighted by $1/p_{i_k}$ allows the classic stochastic‑gradient proof technique to go through unchanged, while the effective smoothness constant $L_{\text{eff}}=\sum_i L_i/p_i$ captures the influence of the sampling distribution. The resulting $\mathcal{O}(T^{-1/2})$ rate matches the best known guarantees for stochastic first‑order methods in the non‑convex setting.
