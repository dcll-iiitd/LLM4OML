\section*{Optimistic Gradient Descent–Ascent (OGDA)}

\section*{Mathematical Formulation}
Let $L:\R^{n}\times\R^{m}\rightarrow \R$ be a differentiable convex–concave function,
\[
L(\cdot ,y)\ \text{convex for every }y,\qquad 
L(x,\cdot )\ \text{concave for every }x .
\]
Define the gradient operator
\[
F(z)\;\stackrel{\text{def}}{=}\;
\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix},
\qquad 
z\;=\;(x,y)\in\R^{n+m}.
\]

OGDA uses the gradient from the previous iteration to form an \emph{optimistic} correction.
With stepsize $\eta>0$, the iterates are
\begin{align}
x_{t+1}&=x_{t}-2\eta\,\nabla_{x}L(x_{t},y_{t})+\eta\,\nabla_{x}L(x_{t-1},y_{t-1}),\label{eq:ogda-x}\\
y_{t+1}&=y_{t}+2\eta\,\nabla_{y}L(x_{t},y_{t})-\eta\,\nabla_{y}L(x_{t-1},y_{t-1}).\label{eq:ogda-y}
\end{align}
Equivalently, in compact form
\begin{equation}
z_{t+1}=z_{t}-\eta\bigl(F(z_{t})+F(z_{t})-F(z_{t-1})\bigr)
= z_{t}-\eta\bigl(2F(z_{t})-F(z_{t-1})\bigr).\tag{OGDA}
\end{equation}
The algorithm is initialized with $z_{0}=(x_{0},y_{0})$ and an auxiliary point $z_{-1}=z_{0}$ (so that $F(z_{-1})=F(z_{0})$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent–Ascent}
\begin{algorithmic}[1]
\Require Convex–concave $L$, stepsize $\eta>0$, max iterations $T$
\State Choose initial $x_{0}\in\R^{n},\;y_{0}\in\R^{m}$; set $x_{-1}=x_{0},\;y_{-1}=y_{0}$
\For{$t=0$ \textbf{to} $T-1$}
    \State $g^{x}_{t}\gets \nabla_{x}L(x_{t},y_{t})$
    \State $g^{y}_{t}\gets \nabla_{y}L(x_{t},y_{t})$
    \State $g^{x}_{t-1}\gets \nabla_{x}L(x_{t-1},y_{t-1})$
    \State $g^{y}_{t-1}\gets \nabla_{y}L(x_{t-1},y_{t-1})$
    \State $x_{t+1}\gets x_{t}-2\eta\,g^{x}_{t}+\eta\,g^{x}_{t-1}$   \Comment{optimistic descent}
    \State $y_{t+1}\gets y_{t}+2\eta\,g^{y}_{t}-\eta\,g^{y}_{t-1}$   \Comment{optimistic ascent}
\EndFor
\State \Return $(x_{T},y_{T})$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple convex function $f(w)=(w-3)^{2}$ (here the min–max problem reduces to pure minimization).  
We treat $L(w)=f(w)$, so $y$ does not appear and the OGDA update becomes
\[
w_{t+1}=w_{t}-2\eta\,\nabla f(w_{t})+\eta\,\nabla f(w_{t-1}),
\qquad \nabla f(w)=2(w-3).
\]

Parameters: $w_{0}=0$, stepsize $\eta=0.1$, and set $w_{-1}=w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $w_{t+1}$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $w_{1}=0-2\cdot0.1(-6)+0.1\cdot0=0+1.2=1.2$ & $(1.2-3)^{2}=3.24$\\[2mm]
1 & $2(1.2-3)=-3.6$ & $w_{2}=1.2-2\cdot0.1(-3.6)+0.1(-6)=1.2+0.72-0.6=1.32$ & $(1.32-3)^{2}=2.8224$\\[2mm]
2 & $2(1.32-3)=-3.36$ & $w_{3}=1.32-2\cdot0.1(-3.36)+0.1(-3.6)=1.32+0.672-0.36=1.632$ & $(1.632-3)^{2}=1.8550$
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $w_{0}=0$) to $1.86$ after three iterations, illustrating the benefit of the optimistic correction.

\section*{Assumptions}
\begin{assumption}[Convex–concave and Smoothness]\label{ass:smooth}
The function $L$ is convex in $x$ and concave in $y$. Moreover, $L$ is $L$‑smooth:
\[
\|F(z)-F(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\R^{n+m}.
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $z^{*}=(x^{*},y^{*})$ such that
\[
L(x^{*},y)\le L(x^{*},y^{*})\le L(x,y^{*}),\qquad\forall x\in\R^{n},\;y\in\R^{m}.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies
\[
0<\eta\le\frac{1}{2L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:ogda}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, the OGDA iterates $\{z_{t}\}_{t\ge0}$ satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\bigl\|F(z_{t})\bigr\|^{2}
\;\le\;
\frac{4}{\eta T}\,\|z_{0}-z^{*}\|^{2}.
\]
Consequently, the primal–dual gap decays as $O(1/T)$:
\[
\max_{y}L(\bar{x}_{T},y)-\min_{x}L(x,\bar{y}_{T})\;\le\;
\frac{4}{\eta T}\,\|z_{0}-z^{*}\|^{2},
\qquad 
\bar{z}_{T}\stackrel{\text{def}}{=}\frac{1}{T}\sum_{t=0}^{T-1}z_{t}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\eta\le 1/(2L)$ guarantees that the potential 
$\|z_{t}-z^{*}\|^{2}+ \|z_{t}-z_{t-1}\|^{2}$ is non‑increasing.
\item \textbf{Hyperparameter Sensitivity.} The bound scales linearly with $1/\eta$;
choosing $\eta$ close to $1/(2L)$ yields the best $O(1/T)$ rate while preserving stability.
\item \textbf{Comparison to Baselines.}
Standard Gradient Descent–Ascent (GDA) with the same stepsize only guarantees
$O(1/\sqrt{T})$ for the primal–dual gap in the worst case.
OGDA attains the optimal $O(1/T)$ rate for smooth convex–concave problems,
matching the rate of the extra‑gradient method but with a cheaper per‑iteration cost.
\end{itemize}

\section*{Discussion}
The proof rests on a Lyapunov function that combines the distance to the saddle point
and the “momentum’’ term $z_{t}-z_{t-1}$.  The optimistic correction eliminates the
rotational component of the monotone operator $F$, leading to a linear convergence factor
in the smooth case.  Extensions to stochastic or strongly‑convex–strongly‑concave settings
follow by standard variance‑reduction or strong‑monotonicity arguments.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Implies Cocoercivity]\label{lem:coco}
If $F$ is $L$‑Lipschitz, then for any $u,v\in\R^{n+m}$,
\[
\langle F(u)-F(v),\,u-v\rangle\ge\frac{1}{L}\|F(u)-F(v)\|^{2}.
\]
\end{lemma}
\begin{proof}
By the Cauchy–Schwarz inequality and Lipschitzness,
\[
\|F(u)-F(v)\|^{2}\le L\langle F(u)-F(v),\,u-v\rangle,
\]
which rearranges to the claimed inequality. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
For convex–concave $L$, the operator $F$ is monotone:
\[
\langle F(u)-F(v),\,u-v\rangle\ge0,\qquad\forall u,v.
\]
\end{lemma}
\begin{proof}
Monotonicity follows directly from convexity in $x$ and concavity in $y$
(see, e.g., standard variational‑inequality theory). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:ogda}.  Throughout the proof we denote $z_{t}=(x_{t},y_{t})$
and $F_{t}=F(z_{t})$.

\subsection*{Step 1 – Lyapunov Recursion}
Consider the potential
\[
\Phi_{t}\;\stackrel{\text{def}}{=}\;\|z_{t}-z^{*}\|^{2}+ \|z_{t}-z_{t-1}\|^{2}.
\tag{1}
\]
Using the OGDA update $z_{t+1}=z_{t}-\eta(2F_{t}-F_{t-1})$, we expand $\|z_{t+1}-z^{*}\|^{2}$:
\begin{align}
\|z_{t+1}-z^{*}\|^{2}
&=\|z_{t}-z^{*}\|^{2}
-2\eta\langle 2F_{t}-F_{t-1},\,z_{t}-z^{*}\rangle
+\eta^{2}\|2F_{t}-F_{t-1}\|^{2}.
\tag{2}
\end{align}
Similarly,
\begin{align}
\|z_{t+1}-z_{t}\|^{2}
&=\|\!-\eta(2F_{t}-F_{t-1})\|^{2}
=\eta^{2}\|2F_{t}-F_{t-1}\|^{2}.
\tag{3}
\end{align}
Adding (2) and (3) yields the recursion for the Lyapunov function:
\[
\Phi_{t+1}
= \Phi_{t}
-2\eta\langle 2F_{t}-F_{t-1},\,z_{t}-z^{*}\rangle
+2\eta^{2}\|2F_{t}-F_{t-1}\|^{2}.
\tag{4}
\]

\subsection*{Step 2 – Bounding the Inner Product}
Using monotonicity (Lemma \ref{lem:mono}) and the fact that $F(z^{*})=0$,
\[
\langle F_{t},\,z_{t}-z^{*}\rangle\ge0,\qquad
\langle F_{t-1},\,z_{t}-z^{*}\rangle\ge0.
\]
Hence,
\[
\langle 2F_{t}-F_{t-1},\,z_{t}-z^{*}\rangle
\ge \langle F_{t},\,z_{t}-z^{*}\rangle.
\tag{5}
\]

\subsection*{Step 3 – Bounding the Norm Term}
Apply Lemma \ref{lem:coco} with $u=z_{t}$ and $v=z_{t-1}$:
\[
\|F_{t}-F_{t-1}\|^{2}\le L\langle F_{t}-F_{t-1},\,z_{t}-z_{t-1}\rangle.
\tag{6}
\]
Observe that
\[
\|2F_{t}-F_{t-1}\|^{2}
= \|F_{t}\|^{2}+ \|F_{t}-F_{t-1}\|^{2}+2\langle F_{t},\,F_{t}-F_{t-1}\rangle.
\tag{7}
\]
Using Cauchy–Schwarz and the Lipschitz bound $\|F_{t}-F_{t-1}\|\le L\|z_{t}-z_{t-1}\|$,
\[
2\langle F_{t},\,F_{t}-F_{t-1}\rangle
\le 2\|F_{t}\|\,\|F_{t}-F_{t-1}\|
\le \|F_{t}\|^{2}+ \|F_{t}-F_{t-1}\|^{2}.
\tag{8}
\]
Plugging (8) into (7) gives
\[
\|2F_{t}-F_{t-1}\|^{2}
\le 2\|F_{t}\|^{2}+2\|F_{t}-F_{t-1}\|^{2}.
\tag{9}
\]

\subsection*{Step 4 – Substituting Bounds into the Lyapunov Recursion}
Insert (5) and (9) into (4):
\begin{align}
\Phi_{t+1}
&\le \Phi_{t}
-2\eta\,\langle F_{t},\,z_{t}-z^{*}\rangle
+2\eta^{2}\bigl(2\|F_{t}\|^{2}+2\|F_{t}-F_{t-1}\|^{2}\bigr) \nonumber\\
&= \Phi_{t}
-2\eta\,\langle F_{t},\,z_{t}-z^{*}\rangle
+4\eta^{2}\|F_{t}\|^{2}+4\eta^{2}\|F_{t}-F_{t-1}\|^{2}.
\tag{10}
\end{align}
Now use Lemma \ref{lem:coco} with $u=z_{t}$, $v=z^{*}$:
\[
\langle F_{t},\,z_{t}-z^{*}\rangle\ge\frac{1}{L}\|F_{t}\|^{2}.
\tag{11}
\]
Moreover, by Lipschitzness,
\[
\|F_{t}-F_{t-1}\|^{2}\le L^{2}\|z_{t}-z_{t-1}\|^{2}.
\tag{12}
\]
Since $\|z_{t}-z_{t-1}\|^{2}$ appears in $\Phi_{t}$ (see definition (1)),
we can bound the last term of (10) by $4\eta^{2}L^{2}\|z_{t}-z_{t-1}\|^{2}\le 4\eta^{2}L^{2}\Phi_{t}$.

\subsection*{Step 5 – Final Recursion}
Using (11) and (12) in (10) we obtain
\begin{align}
\Phi_{t+1}
&\le \Phi_{t}
- \frac{2\eta}{L}\|F_{t}\|^{2}
+4\eta^{2}\|F_{t}\|^{2}
+4\eta^{2}L^{2}\|z_{t}-z_{t-1}\|^{2} \nonumber\\
&\le \Phi_{t}
-\Bigl(\frac{2\eta}{L}-4\eta^{2}\Bigr)\|F_{t}\|^{2}
+4\eta^{2}L^{2}\bigl(\Phi_{t}-\|z_{t}-z^{*}\|^{2}\bigr).
\tag{13}
\end{align}
Because $\eta\le\frac{1}{2L}$ (Assumption \ref{ass:stepsize}), we have
\[
\frac{2\eta}{L}-4\eta^{2}\;\ge\;\frac{\eta}{L}.
\tag{14}
\]
Furthermore, $4\eta^{2}L^{2}\le 1$ under the same stepsize bound.
Hence (13) simplifies to
\[
\Phi_{t+1}\le \Phi_{t}
-\frac{\eta}{L}\|F_{t}\|^{2}
+\bigl(4\eta^{2}L^{2}\bigr)\Phi_{t}
\;\le\;\Phi_{t}
-\frac{\eta}{L}\|F_{t}\|^{2}
+\Phi_{t},
\]
which after cancelling the duplicated $\Phi_{t}$ terms yields
\[
\frac{\eta}{L}\|F_{t}\|^{2}\;\le\;\Phi_{t}-\Phi_{t+1}.
\tag{15}
\]

\subsection*{Step 6 – Summation and Rate Derivation}
Summing (15) from $t=0$ to $T-1$ gives
\[
\frac{\eta}{L}\sum_{t=0}^{T-1}\|F_{t}\|^{2}
\le \Phi_{0}-\Phi_{T}
\le \Phi_{0}
= \|z_{0}-z^{*}\|^{2}+\|z_{0}-z_{-1}\|^{2}
= \|z_{0}-z^{*}\|^{2},
\tag{16}
\]
because $z_{-1}=z_{0}$ implies the second term vanishes.
Dividing by $T$ and multiplying by $L/\eta$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F_{t}\|^{2}
\le \frac{L}{\eta T}\,\|z_{0}-z^{*}\|^{2}
\le \frac{4}{\eta T}\,\|z_{0}-z^{*}\|^{2},
\tag{17}
\]
where the last inequality uses $L\le 2/\eta$ from $\eta\le 1/(2L)$.

Since for convex–concave $L$ the primal–dual gap is bounded by
$\max_{y}L(\bar{x}_{T},y)-\min_{x}L(x,\bar{y}_{T})\le \frac{1}{T}\sum_{t=0}^{T-1}\|F_{t}\|^{2}$
(standard variational‑inequality argument), (17) proves the theorem. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
The constant in the $O(1/T)$ bound is explicitly $\frac{4}{\eta}$.
Choosing the largest admissible stepsize $\eta=1/(2L)$ gives
\[
\max_{y}L(\bar{x}_{T},y)-\min_{x}L(x,\bar{y}_{T})
\le \frac{8L}{T}\,\|z_{0}-z^{*}\|^{2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly Convex–Strongly Concave.}
If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone.
The same Lyapunov analysis yields linear convergence:
\[
\|z_{t}-z^{*}\|\le\bigl(1-\eta\mu\bigr)^{t}\|z_{0}-z^{*}\|.
\]
\item \textbf{Stochastic OGDA.}
When only unbiased stochastic estimates $\tilde{F}_{t}$ with bounded variance $\sigma^{2}$ are available,
the same steps give
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\|F(z_{t})\|^{2}
\le \frac{4}{\eta T}\|z_{0}-z^{*}\|^{2}
+\frac{2\eta\sigma^{2}}{T},
\]
showing an $O(1/T)+O(\eta)$ trade‑off.
\end{itemize}
