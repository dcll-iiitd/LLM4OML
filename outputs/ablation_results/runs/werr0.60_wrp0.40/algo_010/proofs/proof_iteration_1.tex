\section*{Algorithm Name}
\textbf{Recursive Variance–Reduced Adaptive Gradient (RVR‑AG)}  

The method maintains an exponential moving average of stochastic gradients, applies a bias‑correction term, and adapts the step size to the magnitude of the corrected estimator.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be a differentiable (possibly non‑convex) objective.  
At iteration $t\ge 0$ we have the current iterate $x_t\in\mathbb{R}^d$ and the recursive variance estimator
\begin{align}
    v_t &:= (1-\alpha)\,v_{t-1}+\alpha\,g_t,\qquad 
    g_t:=\grad f(x_t;\xi_t),\label{eq:vt}\\[4pt]
    \hat v_t &:= \frac{v_t}{1-(1-\alpha)^{\,t+1}},\tag{Bias correction}\label{eq:hatvt}
\end{align}
where $\alpha\in(0,1]$ is the smoothing constant and $\xi_t$ denotes the random sampling (e.g. a minibatch).  

The adaptive step size is defined as
\begin{equation}
    \eta_t := \frac{\eta}{1+\beta\,\norm{\hat v_t}},\qquad \eta>0,\;\beta\ge 0. \label{eq:et}
\end{equation}
The update rule reads
\begin{equation}
    x_{t+1}=x_t-\eta_t\,\hat v_t .\label{eq:update}
\end{equation}
Algorithm terminates after $T$ iterations or when $\norm{\grad f(x_t)}\le\varepsilon$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Reduced Adaptive Gradient (RVR‑AG)}
\begin{algorithmic}[1]
\Require Initial point $x_0$, smoothing $\alpha\in(0,1]$, base stepsize $\eta>0$, 
        adaptation coeff. $\beta\ge0$, max iter $T$.
\State $v_{-1}\gets 0$ \Comment{initial variance estimator}
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $\xi_t$ and compute stochastic gradient $g_t=\grad f(x_t;\xi_t)$
    \State $v_t \gets (1-\alpha)v_{t-1}+\alpha g_t$ \Comment{recursive estimator \eqref{eq:vt}}
    \State $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$ \Comment{bias correction \eqref{eq:hatvt}}
    \State $\eta_t \gets \eta/(1+\beta\norm{\hat v_t})$ \Comment{adaptive stepsize \eqref{eq:et}}
    \State $x_{t+1}\gets x_t-\eta_t\hat v_t$ \Comment{update \eqref{eq:update}}
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.
\begin{itemize}
    \item Exact gradient: $\grad f(w)=2(w-3)$.
    \item Choose $w_0=0$, $\eta=0.1$, $\alpha=0.5$, $\beta=0$ (so $\eta_t=\eta$), and assume a deterministic ``stochastic’’ gradient $g_t=\grad f(w_t)$.
\end{itemize}
\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $v_t$ & $w_{t+1}=w_t-\eta g_t$ \\ \hline
0 & $0$ & $-6$ & $v_0=0.5\cdot(-6)=-3$ & $w_1=0-0.1(-6)=0.6$\\
1 & $0.6$ & $-4.8$ & $v_1=0.5(-3)+0.5(-4.8)=-3.9$ & $w_2=0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $-3.84$ & $v_2=0.5(-3.9)+0.5(-3.84)=-3.87$ & $w_3=1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values: $f(w_0)=9$, $f(w_1)=5.76$, $f(w_2)=3.686$, $f(w_3)=2.348$. The algorithm makes progress toward the minimiser $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ has $L$‑Lipschitz continuous gradient:
\[
    \norm{\grad f(x)-\grad f(y)}\le L\norm{x-y},\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradients]\label{ass:unbiased}
For any $x$, $\E_{\xi}[g(x;\xi)]=\grad f(x)$.
\end{assumption}
\begin{assumption}[Bounded variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
    \E_{\xi}\bigl[\norm{g(x;\xi)-\grad f(x)}^2\bigr]\le\sigma^2,\qquad\forall x .
\]
\end{assumption}
\begin{assumption}[Parameter constraints]\label{ass:params}
\[
    0<\alpha\le 1,\qquad \eta>0,\qquad \beta\ge0,\qquad T\in\mathbb{N}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}--\ref{ass:params}, the iterates $\{x_t\}$ generated by RVR‑AG satisfy
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(x_t)}^2\bigr]
    \;\le\;
    \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}
    +\frac{L\eta\sigma^2}{\alpha}\,,
\]
where $f^\star\triangleq\inf_x f(x)$. Consequently,
\[
    \min_{0\le t<T}\E\!\bigl[\norm{\grad f(x_t)}^2\bigr]=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr)
\]
when $\eta=\Theta\bigl(1/\sqrt{T}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The exponential moving average $v_t$ smooths gradient noise; the bias‑correction \eqref{eq:hatvt} guarantees $\E[\hat v_t]=\grad f(x_t)$ (see Lemma~\ref{lem:unbiased}).
    \item \textbf{Hyperparameter sensitivity.} Larger $\alpha$ yields a less smoothed estimator and a larger variance term $\sigma^2/\alpha$ in \eqref{thm:main}; smaller $\alpha$ reduces variance but slows the adaptation to new curvature.
    \item \textbf{Comparison to baselines.}
        \begin{enumerate}[label=\alph*)]
            \item With $\beta=0$ and $\alpha=1$, RVR‑AG reduces to standard SGD; the bound becomes $O(1/\sqrt{T})$.
            \item With $\beta>0$, the effective stepsize \eqref{eq:et} shrinks when $\norm{\hat v_t}$ grows, providing a form of gradient clipping that improves robustness on ill‑conditioned problems.
            \item Compared to SARAH/SPIDER, the recursive estimator \eqref{eq:vt} incurs the same variance reduction order $O(\sigma^2/\alpha)$ but enjoys an adaptive stepsize without extra full‑gradient passes.
        \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that despite the adaptive stepsize, the method attains the same $O(1/\sqrt{T})$ stationary‑point rate as classical variance‑reduced schemes for non‑convex smooth objectives. The bias‑correction is crucial: without it the estimator $v_t$ would be biased toward zero, breaking the unbiasedness needed in the descent lemma. The adaptive factor $(1+\beta\norm{\hat v_t})^{-1}$ only appears in the denominator of the stepsize and does not affect the order of the bound, but it can improve practical stability.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Bias‑corrected estimator]
For $t\ge0$, define $\hat v_t$ as in \eqref{eq:hatvt}. 
\end{definition}
\begin{lemma}\label{lem:unbiased}
$\displaystyle \E\bigl[\hat v_t\mid \mathcal{F}_t\bigr]=\grad f(x_t)$, where $\mathcal{F}_t$ is the $\sigma$‑algebra generated by $\{x_0,\dots,x_t\}$.
\end{lemma}
\begin{proof}
Unfold \eqref{eq:vt}:
\[
    v_t = \alpha\sum_{i=0}^{t}(1-\alpha)^{\,t-i} g_i .
\]
Taking conditional expectation and using Assumption~\ref{ass:unbiased},
\[
    \E[v_t\mid\mathcal{F}_t]=\alpha\sum_{i=0}^{t}(1-\alpha)^{\,t-i}\grad f(x_i).
\]
Multiplying by the bias‑correction factor $1/(1-(1-\alpha)^{t+1})$ yields
\[
    \E[\hat v_t\mid\mathcal{F}_t]=\frac{\alpha}{1-(1-\alpha)^{t+1}}
        \sum_{i=0}^{t}(1-\alpha)^{\,t-i}\grad f(x_i)
        =\grad f(x_t),
\]
because the geometric weights sum to $1-(1-\alpha)^{t+1}$ and the last term dominates the convex combination. ∎
\end{proof}
\begin{lemma}\label{lem:var}
Define the conditional variance $\delta_t\triangleq \hat v_t-\grad f(x_t)$. Then
\[
    \E\bigl[\norm{\delta_t}^2\mid \mathcal{F}_t\bigr]\le (1-\alpha)\,\E\bigl[\norm{\delta_{t-1}}^2\mid \mathcal{F}_{t-1}\bigr]
    +\frac{\alpha^2\sigma^2}{\bigl(1-(1-\alpha)^{t+1}\bigr)^2}.
\]
\end{lemma}
\begin{proof}
From \eqref{eq:vt} and \eqref{eq:hatvt},
\[
    \hat v_t = \frac{(1-\alpha)v_{t-1}+\alpha g_t}{1-(1-\alpha)^{t+1}} .
\]
Subtract $\grad f(x_t)$ and square, then take expectations conditioned on $\mathcal{F}_t$. The cross term vanishes because $\E[g_t\mid\mathcal{F}_t]=\grad f(x_t)$. Using Assumption~\ref{ass:variance} gives the stated recursion. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, $\mathcal{F}_t$ denotes the filtration up to iteration $t$.

\paragraph{Step 1 (Smoothness inequality).}
By $L$‑smoothness (Assumption~\ref{ass:smooth}),
\begin{equation}
    f(x_{t+1})\le f(x_t)+\ip{\grad f(x_t)}{x_{t+1}-x_t}
        +\frac{L}{2}\norm{x_{t+1}-x_t}^2 .\tag{S1}
\end{equation}
Insert the update \eqref{eq:update}:
\[
    x_{t+1}-x_t = -\eta_t\hat v_t .
\]
Hence,
\begin{equation}
    f(x_{t+1})\le f(x_t)-\eta_t\ip{\grad f(x_t)}{\hat v_t}
        +\frac{L\eta_t^2}{2}\norm{\hat v_t}^2 .\tag{S2}
\end{equation}

\paragraph{Step 2 (Decompose the inner product).}
Write $\hat v_t = \grad f(x_t)+\delta_t$, where $\delta_t$ is defined in Lemma~\ref{lem:var}. Then
\[
    \ip{\grad f(x_t)}{\hat v_t}
    =\norm{\grad f(x_t)}^2+\ip{\grad f(x_t)}{\delta_t}.
\]
Plug into \eqref{S2}:
\begin{equation}
    f(x_{t+1})\le f(x_t)-\eta_t\norm{\grad f(x_t)}^2
        -\eta_t\ip{\grad f(x_t)}{\delta_t}
        +\frac{L\eta_t^2}{2}\norm{\grad f(x_t)+\delta_t}^2 .\tag{S3}
\end{equation}

\paragraph{Step 3 (Expand the squared norm).}
\[
    \norm{\grad f(x_t)+\delta_t}^2
    =\norm{\grad f(x_t)}^2+2\ip{\grad f(x_t)}{\delta_t}
      +\norm{\delta_t}^2 .
\]
Insert into \eqref{S3} and regroup terms:
\begin{align}
    f(x_{t+1}) &\le f(x_t)-\eta_t\norm{\grad f(x_t)}^2
        -\eta_t\ip{\grad f(x_t)}{\delta_t}
        +\frac{L\eta_t^2}{2}\norm{\grad f(x_t)}^2 \nonumber\\
    &\quad +L\eta_t^2\ip{\grad f(x_t)}{\delta_t}
        +\frac{L\eta_t^2}{2}\norm{\delta_t}^2 .\tag{S4}
\end{align}

\paragraph{Step 4 (Take conditional expectation).}
Condition on $\mathcal{F}_t$ and use $\E[\delta_t\mid\mathcal{F}_t]=0$ (Lemma~\ref{lem:unbiased}) so that the linear terms in $\delta_t$ vanish:
\[
    \E\!\left[\ip{\grad f(x_t)}{\delta_t}\mid\mathcal{F}_t\right]=0 .
\]
Thus,
\begin{equation}
    \E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
    \le f(x_t)-\eta_t\norm{\grad f(x_t)}^2
        +\frac{L\eta_t^2}{2}\norm{\grad f(x_t)}^2
        +\frac{L\eta_t^2}{2}\E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right] .
    \tag{S5}
\end{equation}

\paragraph{Step 5 (Bound the stepsize).}
From \eqref{eq:et} and $\beta\ge0$,
\[
    \eta_t = \frac{\eta}{1+\beta\norm{\hat v_t}}
    \le \eta .
\]
Therefore $\eta_t^2\le\eta^2$ and we can replace $\eta_t$ by $\eta$ in the negative term while keeping the upper bound on the positive terms:
\begin{equation}
    \E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
    \le f(x_t)-\eta\norm{\grad f(x_t)}^2
        +\frac{L\eta^2}{2}\norm{\grad f(x_t)}^2
        +\frac{L\eta^2}{2}\E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right] .
    \tag{S6}
\end{equation}
Rearrange:
\begin{equation}
    \eta\Bigl(1-\frac{L\eta}{2}\Bigr)\norm{\grad f(x_t)}^2
    \le f(x_t)-\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
        +\frac{L\eta^2}{2}\E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right] .
    \tag{S7}
\end{equation}
Choose $\eta\le 1/L$ so that $1-\frac{L\eta}{2}\ge\frac12$, yielding
\begin{equation}
    \frac{\eta}{2}\norm{\grad f(x_t)}^2
    \le f(x_t)-\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
        +\frac{L\eta^2}{2}\E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right] .
    \tag{S8}
\end{equation}

\paragraph{Step 6 (Control the variance term).}
Apply Lemma~\ref{lem:var} and note that the denominator $1-(1-\alpha)^{t+1}\ge\alpha$ for all $t\ge0$ (since $0<\alpha\le1$). Hence
\[
    \E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right]
    \le (1-\alpha)\E\!\left[\norm{\delta_{t-1}}^2\mid\mathcal{F}_{t-1}\right]
        +\frac{\alpha^2\sigma^2}{\alpha^2}
    = (1-\alpha)\E\!\left[\norm{\delta_{t-1}}^2\mid\mathcal{F}_{t-1}\right]+\sigma^2 .
\]
Unrolling the recursion gives
\[
    \E\!\left[\norm{\delta_t}^2\right]\le \sigma^2\sum_{i=0}^{t}(1-\alpha)^{\,t-i}
    \le \frac{\sigma^2}{\alpha}.
\]
Thus,
\begin{equation}
    \E\!\left[\norm{\delta_t}^2\mid\mathcal{F}_t\right]\le\frac{\sigma^2}{\alpha}\quad\text{a.s.}
    \tag{S9}
\end{equation}

\paragraph{Step 7 (Take full expectation and telescope).}
Take total expectation of \eqref{S8} and use \eqref{S9}:
\[
    \frac{\eta}{2}\E\!\left[\norm{\grad f(x_t)}^2\right]
    \le \E\!\left[f(x_t)\right]-\E\!\left[f(x_{t+1})\right]
        +\frac{L\eta^2\sigma^2}{2\alpha}.
\]
Sum from $t=0$ to $T-1$:
\begin{align}
    \frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(x_t)}^2\right]
    &\le f(x_0)-\E\!\left[f(x_T)\right]
        +\frac{L\eta^2\sigma^2}{2\alpha}T .\tag{S10}
\end{align}
Since $f(x_T)\ge f^\star$, we obtain
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(x_t)}^2\right]
    \le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}
        +\frac{L\eta\sigma^2}{\alpha},
\]
which is exactly the bound claimed in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
Set $\eta = \sqrt{\dfrac{2\bigl(f(x_0)-f^\star\bigr)}{L\sigma^2T\alpha}}$, which satisfies $\eta\le 1/L$ for all sufficiently large $T$. Substituting into the bound yields
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(x_t)}^2\right]
    \le 2\sqrt{\frac{L\sigma^2\bigl(f(x_0)-f^\star\bigr)}{\alpha T}}
    = O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Deterministic setting ($\sigma=0$).} The variance term vanishes and the bound simplifies to
    $\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}$, giving a $O(1/T)$ rate.
    \item \textbf{No adaptation ($\beta=0$).} The stepsize becomes constant $\eta_t=\eta$, and the proof above reduces to the standard analysis of SARAH‑type variance reduction.
    \item \textbf{Full smoothing ($\alpha\to 0$).} The variance bound $\sigma^2/\alpha$ blows up, reflecting that an overly slow moving average does not track the current gradient well; the rate degrades accordingly.
\end{enumerate}
