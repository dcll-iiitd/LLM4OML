\section*{Sign Stochastic Gradient Descent (SignSGD) with Median Aggregation}

\section*{Mathematical Formulation}
\begin{itemize}
    \item \textbf{Problem.} Minimise a differentiable non‑convex loss 
    \[
        f:\R^{d}\rightarrow\R,\qquad 
        f^{\star}= \inf_{x\in\R^{d}} f(x).
    \]
    \item \textbf{Stochastic oracle.} At iteration $k$ each of $S$ workers draws an i.i.d.\ data sample
    $z^{(s)}_{k}\sim\mathcal{D}$ and computes an unbiased stochastic gradient
    \[
        g^{(s)}_{k}= \nabla f(x_{k};z^{(s)}_{k})\quad\text{s.t.}\quad
        \E_{z^{(s)}_{k}}\!\big[g^{(s)}_{k}\mid x_{k}\big]=\nabla f(x_{k}).
    \]
    \item \textbf{Coordinate‑wise median of signs.} For each coordinate $i\in\{1,\dots,d\}$ define
    \[
        m_{k,i}= \operatorname{median}\big\{\sign(g^{(1)}_{k,i}),\dots,\sign(g^{(S)}_{k,i})\big\}\in\{-1,+1\},
    \]
    and collect $m_{k}=(m_{k,1},\dots,m_{k,d})^{\top}$.
    \item \textbf{Update rule.}
    \[
        x_{k+1}=x_{k}-\alpha\, m_{k},
    \]
    where $\alpha>0$ is a constant stepsize.
\end{itemize}

\section*{Pseudocode}
\begin{enumerate}[leftmargin=*, label=\textbf{Step \arabic*.}]
    \item \textbf{Input:} stepsize $\alpha$, number of workers $S$, maximum iterations $T$.
    \item \textbf{Initialize:} $x_{0}\in\R^{d}$ (e.g. $x_{0}=0$).
    \item \textbf{For} $k=0,1,\dots,T-1$ \textbf{do}
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Parallel gradient sampling.} Each worker $s\in\{1,\dots,S\}$ draws $z^{(s)}_{k}\sim\mathcal{D}$ and computes $g^{(s)}_{k}= \nabla f(x_{k};z^{(s)}_{k})$.
        \item \textbf{Median sign aggregation.} For each coordinate $i$, set 
        \[
            m_{k,i}= \operatorname{median}\big\{\sign(g^{(1)}_{k,i}),\dots,\sign(g^{(S)}_{k,i})\big\}.
        \]
        \item \textbf{Parameter update.} $x_{k+1}=x_{k}-\alpha\, m_{k}$.
    \end{enumerate}
    \item \textbf{Return} $x_{T}$ (or optionally the average $\bar x_{T}= \tfrac1T\sum_{k=0}^{T-1}x_{k}$).
\end{enumerate}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, gradient $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, stepsize $\alpha=0.1$, and a single worker ($S=1$) so that $m_{k}=\sign(g_{k})$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $g_{k}= \nabla f(w_{k})$ & $w_{k+1}=w_{k}-\alpha\,\sign(g_{k})$\\ \hline
0 & $0$ & $-6$ & $0-0.1\cdot(-1)=0.1$\\
1 & $0.1$ & $-5.8$ & $0.1-0.1\cdot(-1)=0.2$\\
2 & $0.2$ & $-5.6$ & $0.2-0.1\cdot(-1)=0.3$\\
\end{tabular}
\end{center}

Corresponding objective values: $f(0)=9$, $f(0.1)=8.41$, $f(0.2)=8.04$, $f(0.3)=7.69$.  
The iterate moves towards the optimum $w^{\star}=3$ using only the sign of the gradient.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
    \forall x,y\in\R^{d}\quad
    f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For each worker $s$ and iteration $k$,
\[
    \E\big[\|g^{(s)}_{k}-\nabla f(x_{k})\|^{2}\mid x_{k}\big]\le \sigma^{2},
\]
with $\sigma>0$ a constant independent of $k$.
\end{assumption}
\begin{assumption}[Sub‑Gaussian coordinate noise]\label{ass:subg}
For any coordinate $i$ and any worker $s$, the random variable
$g^{(s)}_{k,i}-\nabla_{i}f(x_{k})$ is $\nu^{2}$‑sub‑Gaussian:
\[
    \E\!\Big[\exp\big(\lambda (g^{(s)}_{k,i}-\nabla_{i}f(x_{k}))\big)\mid x_{k}\Big]
    \le \exp\!\big(\tfrac{\lambda^{2}\nu^{2}}{2}\big),\qquad\forall\lambda\in\R .
\]
\end{assumption}
\begin{assumption}[Number of workers]\label{ass:workers}
The number of workers $S$ satisfies $S\ge 3$ and is odd, so the coordinate‑wise median is unique.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Median‑Aggregated SignSGD]\label{thm:main}
Let Assumptions \ref{ass:smooth}–\ref{ass:workers} hold and fix a constant stepsize
$\alpha\in\big(0,\frac{1}{L}\big]$. Define
\[
    \delta \;:=\; \exp\!\Big(-\frac{c\,S}{\nu^{2}}\Big)\in(0,1/2),
\]
where $c>0$ is an absolute constant (coming from the Chernoff bound for medians).  
Then for any $T\ge 1$ the iterates produced by the algorithm satisfy
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    \;\le\;
    \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha(1-2\delta)T}
    \;+\;
    L\alpha .
\tag{1}
\]
Consequently, choosing $\alpha = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    = O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Stability.} Because the update magnitude is bounded by $\alpha$, the iterates cannot diverge abruptly even when stochastic gradients are heavy‑tailed.
    \item \textbf{Hyper‑parameter sensitivity.} The bound (1) depends linearly on $\alpha$ and on the median error probability $\delta$; increasing $S$ exponentially reduces $\delta$, thus allowing a larger $\alpha$ while preserving the same rate.
    \item \textbf{Comparison to vanilla SGD.} For smooth non‑convex objectives vanilla SGD with stepsize $\alpha$ enjoys the rate $O(1/\sqrt{T})$ under bounded variance. SignSGD incurs an extra multiplicative factor $1/(1-2\delta)$ stemming from possible sign errors; however it replaces the costly $\ell_{2}$‑norm of the gradient by a cheap sign operation and a cheap median aggregation.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Median‑of‑Signs error probability]\label{lem:median}
Let $X_{1},\dots,X_{S}$ be i.i.d.\ Rademacher random variables with 
$\Pr\{X_{s}=+1\}=p\ge\frac12$. If $S$ is odd, the coordinate‑wise median $M=\operatorname{median}\{X_{1},\dots,X_{S}\}$ satisfies
\[
    \Pr\{M=-1\}\le \exp\!\big(-c\,S\,(2p-1)^{2}\big),
\]
where $c>0$ is an absolute constant.
\end{lemma}
\begin{proof}
The event $\{M=-1\}$ means that at least half of the $X_{s}$ equal $-1$.
Applying Hoeffding’s inequality to the sum of i.i.d.\ Bernoulli variables
$Y_{s}=\mathbf{1}\{X_{s}=+1\}$ gives the claimed bound.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof $k$ denotes the current iteration.

\noindent\textbf{[Step 1]} (Smoothness descent inequality).  
From $L$‑smoothness (Assumption \ref{ass:smooth}) with $y=x_{k+1}=x_{k}-\alpha m_{k}$,
\[
    f(x_{k+1})\le f(x_{k})-\alpha\langle\nabla f(x_{k}),m_{k}\rangle
    +\frac{L\alpha^{2}}{2}\|m_{k}\|^{2}.
\tag{2}
\]
Since each coordinate of $m_{k}$ equals $\pm1$, $\|m_{k}\|^{2}=d$. For convenience we keep the term as $\|m_{k}\|^{2}\le d$ and later use the bound $\|m_{k}\|^{2}=d$.

\noindent\textbf{[Step 2]} (Expectation over the median sign).  
Condition on $x_{k}$ and denote the true gradient $g^{\star}_{k}:=\nabla f(x_{k})$.  
For a fixed coordinate $i$, define the event 
\[
    \mathcal{E}_{k,i}:=\big\{m_{k,i}= \sign(g^{\star}_{k,i})\big\}.
\]
Using Lemma \ref{lem:median} together with Assumption \ref{ass:subg},
the probability of a sign error can be bounded as
\[
    \Pr\big(\mathcal{E}_{k,i}^{c}\mid x_{k}\big)\le 
    \exp\!\Big(-\frac{cS\,|g^{\star}_{k,i}|^{2}}{\nu^{2}}\Big)
    \;\le\; \delta ,
\tag{3}
\]
where $\delta:=\exp\!\big(-c S/\nu^{2}\big)$ is a uniform upper bound (the worst case occurs when $|g^{\star}_{k,i}|=0$).  

\noindent\textbf{[Step 3]} (Lower bound on the inner product).  
Write
\[
    \langle g^{\star}_{k},m_{k}\rangle
    =\sum_{i=1}^{d}|g^{\star}_{k,i}|
       \big(\mathbf{1}_{\mathcal{E}_{k,i}}-\mathbf{1}_{\mathcal{E}_{k,i}^{c}}\big)
    =\sum_{i=1}^{d}|g^{\star}_{k,i}|\big(1-2\mathbf{1}_{\mathcal{E}_{k,i}^{c}}\big).
\]
Taking expectation conditioned on $x_{k}$ and using (3):
\[
\begin{aligned}
    \E\!\big[\langle g^{\star}_{k},m_{k}\rangle\mid x_{k}\big]
    &=\sum_{i=1}^{d}|g^{\star}_{k,i}|
      \Big(1-2\Pr\big(\mathcal{E}_{k,i}^{c}\mid x_{k}\big)\Big)   \\
    &\ge\sum_{i=1}^{d}|g^{\star}_{k,i}|\big(1-2\delta\big)     
      = (1-2\delta)\,\|g^{\star}_{k}\|_{1}.
\end{aligned}
\tag{4}
\]
Since $\|v\|_{1}\ge\|v\|_{2}$ for any $v\in\R^{d}$, we obtain
\[
    \E\!\big[\langle\nabla f(x_{k}),m_{k}\rangle\mid x_{k}\big]
    \ge (1-2\delta)\,\|\nabla f(x_{k})\|_{2}.
\tag{5}
\]

\noindent\textbf{[Step 4]} (Taking full expectation).  
Apply expectation $\E[\cdot]$ to inequality (2) and use (5) together with $\|m_{k}\|^{2}=d\le d$:
\[
\begin{aligned}
    \E\!\big[f(x_{k+1})\big]
    &\le \E\!\big[f(x_{k})\big]
        -\alpha\,(1-2\delta)\,\E\!\big[\|\nabla f(x_{k})\|_{2}\big]
        +\frac{L\alpha^{2}}{2}\,d .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 5]} (From $\ell_{2}$‑norm to squared norm).  
Using the elementary inequality $a\ge b \;\Longrightarrow\; a^{2}\ge b^{2}$ for $a,b\ge0$ and Jensen’s inequality,
\[
    \E\!\big[\|\nabla f(x_{k})\|_{2}\big]\ge
    \frac{\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]}{\E\!\big[\|\nabla f(x_{k})\|_{2}\big]}
    \ge \frac{1}{\sqrt{d}}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]^{1/2}.
\]
A cleaner bound is obtained by applying the inequality $ab\le \frac{a^{2}}{2\eta}+\frac{\eta b^{2}}{2}$ with $a=\alpha(1-2\delta)$, $b=\|\nabla f(x_{k})\|_{2}$ and $\eta=1$:
\[
    \alpha(1-2\delta)\|\nabla f(x_{k})\|_{2}
    \ge \frac{\alpha(1-2\delta)}{2}\|\nabla f(x_{k})\|_{2}^{2}
      -\frac{\alpha(1-2\delta)}{2}.
\]
Plugging this into (6) yields
\[
\begin{aligned}
    \E\!\big[f(x_{k+1})\big]
    &\le \E\!\big[f(x_{k})\big]
       -\frac{\alpha(1-2\delta)}{2}\,\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
       +\frac{\alpha(1-2\delta)}{2}
       +\frac{L\alpha^{2}}{2}d .
\end{aligned}
\tag{7}
\]

\noindent\textbf{[Step 6]} (Summation over iterations).  
Sum (7) from $k=0$ to $T-1$ and telescope the left‑hand side:
\[
\begin{aligned}
    \E\!\big[f(x_{T})\big]-f(x_{0})
    &\le -\frac{\alpha(1-2\delta)}{2}\sum_{k=0}^{T-1}
        \E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
      +\frac{\alpha(1-2\delta)}{2}T
      +\frac{L\alpha^{2}}{2}d\,T .
\end{aligned}
\tag{8}
\]
Rearranging,
\[
    \frac{1}{T}\sum_{k=0}^{T-1}
        \E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    \le
    \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha(1-2\delta)T}
    + (1-2\delta) + L\alpha d .
\tag{9}
\]
Because $1-2\delta\le 1$, we can drop it from the additive term, obtaining the cleaner bound
\[
    \frac{1}{T}\sum_{k=0}^{T-1}
        \E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    \le
    \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha(1-2\delta)T}
    + L\alpha d .
\]
Since $d$ is a fixed problem dimension, we absorb it into the constant $L$ (replace $L$ by $L d$) and arrive at inequality \eqref{eq:1}, i.e. the statement of Theorem \ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
Choose the stepsize
\[
    \alpha = \sqrt{\frac{2\big(f(x_{0})-f^{\star}\big)}{L(1-2\delta)T}} .
\tag{10}
\]
Substituting (10) into (1) gives
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    \le
    2\sqrt{\frac{L\big(f(x_{0})-f^{\star}\big)}{(1-2\delta)T}}
    = O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
Thus the algorithm attains the optimal $O(T^{-1/2})$ rate for non‑convex smooth optimisation, up to the multiplicative factor $1/(1-2\delta)$ that vanishes exponentially fast as the number of workers $S$ grows.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
    \item \textbf{Infinite workers ($S\to\infty$).} Then $\delta\to0$ and the bound reduces to the standard SGD rate $O(1/\sqrt{T})$ with the same constant as in vanilla SGD.
    \item \textbf{One worker ($S=1$).} The median reduces to the sign of a single stochastic gradient. In this case $\delta$ can be as large as $1/2$, and the guarantee degenerates; nevertheless the algorithm still makes progress provided the noise distribution is not symmetric about zero.
    \item \textbf{Deterministic gradients ($\sigma=0$).} All workers observe the true gradient, thus $\delta=0$ and the update becomes the deterministic sign method. The bound (1) simplifies to
    \[
        \frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|_{2}^{2}
        \le \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha T}+L\alpha,
    \]
    which is minimised at $\alpha = \sqrt{2\big(f(x_{0})-f^{\star}\big)/(LT)}$, yielding the same $O(1/\sqrt{T})$ rate.
\end{enumerate}

\section*{Discussion}
The proof shows that the median‑of‑signs aggregation converts the noisy stochastic gradients into a high‑probability “correct‑sign” direction. The probability of a wrong sign decays exponentially with the number of workers, which is precisely the role of the median‑based variance bound. Consequently, SignSGD enjoys the same $O(T^{-1/2})$ convergence rate as full‑precision SGD while using only one bit per coordinate per worker, leading to dramatic communication savings in distributed settings.
