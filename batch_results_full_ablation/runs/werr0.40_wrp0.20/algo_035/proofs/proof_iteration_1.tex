\section*{Clipped Gradient Momentum (CGM) Algorithm}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Given a clipping threshold $C>0$, a stepsize $\eta>0$, and a momentum
parameter $\beta\in[0,1)$, the CGM iterates are defined as follows:

\begin{align}
    g_t &\;=\; \frac{\nabla f(x_t)}{\displaystyle \max\Bigl\{1,\,
        \frac{\norm{\nabla f(x_t)}}{C}\Bigr\}} \label{eq:clip}\\[4pt]
    m_t &\;=\; \beta\, m_{t-1} + (1-\beta)\, g_t ,\qquad m_{-1}=0 \label{eq:momentum}\\[4pt]
    x_{t+1} &\;=\; x_t - \eta\, m_t . \label{eq:update}
\end{align}

The clipping operator in \eqref{eq:clip} guarantees $\norm{g_t}\le C$ for all $t$.  
When $C=\infty$ the algorithm reduces to classical momentum SGD.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Gradient Momentum (CGM)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, momentum $\beta\in[0,1)$, clipping $C>0$
\STATE \textbf{Initialize:} $x_0\in\mathbb{R}^d$, $m_{-1}=0$
\FOR{$t=0,1,\dots,T-1$}
    \STATE Compute full gradient $g^{\text{raw}}_t=\nabla f(x_t)$
    \STATE \textbf{Clip:} 
    \[
        g_t \leftarrow \frac{g^{\text{raw}}_t}{\max\{1,\norm{g^{\text{raw}}_t}/C\}}
    \]
    \STATE \textbf{Momentum update:} $m_t \leftarrow \beta\,m_{t-1}+(1-\beta)g_t$
    \STATE \textbf{Parameter update:} $x_{t+1}\leftarrow x_t-\eta\,m_t$
\ENDFOR
\STATE \textbf{Output:} $\{x_t\}_{t=0}^T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^2$.  
Hence $\nabla f(w)=2(w-3)$.  Choose $\eta=0.1$, $C=1$, $\beta=0.5$, and $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\nabla f(w_t)$ & $g_t$ (clipped) & $m_t$ & $w_{t+1}$ \\ \hline
0 & $2(0-3)=-6$ & $g_0=\frac{-6}{\max\{1,6/1\}}=-1$ & $m_0=0.5\cdot0+0.5\cdot(-1)=-0.5$ & $w_1=0-0.1(-0.5)=0.05$ \\[4pt]
1 & $2(0.05-3)=-5.9$ & $g_1=-1$ & $m_1=0.5(-0.5)+0.5(-1)=-0.75$ & $w_2=0.05-0.1(-0.75)=0.125$ \\[4pt]
2 & $2(0.125-3)=-5.75$ & $g_2=-1$ & $m_2=0.5(-0.75)+0.5(-1)=-0.875$ & $w_3=0.125-0.1(-0.875)=0.2125$
\end{tabular}
\end{center}

The objective values are $f(w_0)=9$, $f(w_1)=8.7025$, $f(w_2)=8.3825$, $f(w_3)=8.0366$, showing a monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$–smooth, i.e.
\[
    f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2,
    \qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:bound}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Clipping Threshold]\label{as:clip}
The clipping constant $C$ is chosen so that $C>0$ and all iterates satisfy $\norm{g_t}\le C$ by construction of \eqref{eq:clip}.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
The stepsize $\eta$ is constant and satisfies $0<\eta\le \frac{1}{L}$.
\end{assumption}
\begin{assumption}[Momentum]\label{as:beta}
Momentum parameter $\beta\in[0,1)$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationary‑point convergence of CGM]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:beta}, let $\{x_t\}_{t=0}^T$ be the iterates generated by CGM.
Define the averaged gradient norm
\[
    \bar{G}_T \;=\; \frac{1}{T}\sum_{t=0}^{T-1}\norm{\nabla f(x_t)}^2 .
\]
Then for any $T\ge 1$,
\begin{align}
    \bar{G}_T 
    &\le 
    \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)C\,T}
    \;+\;
    \frac{L C^2\eta}{1-\beta}.
    \label{eq:rate}
\end{align}
In particular, choosing $\eta = \Theta\!\bigl(\frac{1}{\sqrt{T}}\bigr)$ yields
\[
    \bar{G}_T = \mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Stability via clipping.} Since $\norm{g_t}\le C$, the momentum term $m_t$ is uniformly bounded:
    \[
        \norm{m_t}\le \frac{1-\beta}{1-\beta}\,C = C .
    \]
    This prevents exploding updates even when $\norm{\nabla f(x_t)}$ is large.
    \item \textbf{Hyperparameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ weakens clipping, making the bound in \eqref{eq:rate} approach the classical momentum SGD rate.
        \item Smaller $C$ tightens the $LC^2\eta/(1-\beta)$ term but may increase the first term due to a smaller denominator $C$.
        \item $\beta$ close to $1$ reduces the variance term $LC^2\eta/(1-\beta)$ but inflates the first term through $1/(1-\beta)$.
    \end{itemize}
    \item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item \textbf{Momentum SGD (no clipping).} Setting $C=\infty$ eliminates the clipping factor, and \eqref{eq:rate} reduces to the standard $\mathcal{O}(1/\sqrt{T})$ bound for non‑convex smooth objectives.
        \item \textbf{Plain SGD with clipping (no momentum).} Taking $\beta=0$ yields the same bound as \cite{Ghadimi2013} but with the additional $C$ factor that can improve the constant.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that clipping does not deteriorate the classical $\mathcal{O}(1/\sqrt{T})$ rate for non‑convex smooth problems; it merely introduces a controllable constant $C$ that can be used to trade off between robustness (small $C$) and speed (large $C$). Momentum interacts with clipping only through the factor $1/(1-\beta)$, identical to the un‑clipped case, confirming that the momentum mechanism remains effective.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
    f(y) \le f(x) + \langle\nabla f(x),y-x\rangle + \frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \hfill $\square$
\end{proof}
\begin{lemma}[Bound on momentum vector]\label{lem:momentum}
For the recursion $m_t=\beta m_{t-1}+(1-\beta)g_t$ with $m_{-1}=0$ and $\norm{g_t}\le C$, we have
\[
    \norm{m_t}\le C,\qquad\forall t\ge0 .
\]
\end{lemma}
\begin{proof}
We prove by induction.
\begin{itemize}
    \item Base $t=0$: $m_0=(1-\beta)g_0$, thus $\norm{m_0}\le (1-\beta)C\le C$.
    \item Inductive step: assume $\norm{m_{t-1}}\le C$. Then
    \[
        \norm{m_t}= \norm{\beta m_{t-1}+(1-\beta)g_t}
        \le \beta\norm{m_{t-1}}+(1-\beta)\norm{g_t}
        \le \beta C+(1-\beta)C = C .
    \]
\end{itemize}
\hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  All expectations are trivial because we consider deterministic gradients; nevertheless we keep the notation $\E[\cdot]$ for consistency with stochastic extensions.

\begin{enumerate}
    \item[Step 1.] \textbf{Apply the descent lemma} \eqref{lem:descent} with $x=x_t$, $y=x_{t+1}=x_t-\eta m_t$:
    \begin{align}
        f(x_{t+1})
        &\le f(x_t) - \eta\langle\nabla f(x_t),m_t\rangle
           + \frac{L\eta^2}{2}\norm{m_t}^2 .
        \label{eq:descent1}
    \end{align}
    \item[Step 2.] \textbf{Rewrite $m_t$ using its definition}.  From \eqref{eq:momentum},
    \[
        m_t = (1-\beta)g_t + \beta m_{t-1}.
    \]
    Substitute into the inner product term of \eqref{eq:descent1}:
    \begin{align}
        \langle\nabla f(x_t),m_t\rangle
        &= (1-\beta)\langle\nabla f(x_t),g_t\rangle
           + \beta\langle\nabla f(x_t),m_{t-1}\rangle .
        \label{eq:inner}
    \end{align}
    \item[Step 3.] \textbf{Express $g_t$ via the clipping operator}.  
    By definition \eqref{eq:clip},
    \[
        g_t = \frac{\nabla f(x_t)}{\max\{1,\norm{\nabla f(x_t)}/C\}} .
    \]
    Hence
    \begin{align}
        \langle\nabla f(x_t),g_t\rangle
        &= \frac{\norm{\nabla f(x_t)}^2}{\max\{1,\norm{\nabla f(x_t)}/C\}} .
        \label{eq:clipinner}
    \end{align}
    Observe that for any $a\ge0$,
    \[
        \frac{a^2}{\max\{1,a/C\}} \ge a\,\min\{a,C\}
        \ge a\,\bigl(a - (a-C)_+\bigr) = a^2 - a\,(a-C)_+ .
    \]
    In particular,
    \begin{align}
        \langle\nabla f(x_t),g_t\rangle
        \ge \norm{\nabla f(x_t)}^2 - \norm{\nabla f(x_t)}\,
               \bigl(\norm{\nabla f(x_t)}-C\bigr)_+ .
        \label{eq:clipbound}
    \end{align}
    Since $0\le (\norm{\nabla f(x_t)}-C)_+ \le \norm{\nabla f(x_t)}$, we obtain the crude but useful bound
    \begin{align}
        \langle\nabla f(x_t),g_t\rangle \ge
        \norm{\nabla f(x_t)}^2 - C\,\norm{\nabla f(x_t)} .
        \label{eq:clipbound2}
    \end{align}
    \item[Step 4.] \textbf{Upper bound the momentum norm}.  
    By Lemma \ref{lem:momentum} and the stepsize restriction $\eta\le 1/L$, we have
    \begin{align}
        \norm{m_t}^2 \le C^2 .
        \label{eq:mtbound}
    \end{align}
    \item[Step 5.] \textbf{Combine \eqref{eq:descent1}–\eqref{eq:mtbound}}.  
    Using \eqref{eq:inner} and discarding the non‑negative term $\beta\langle\nabla f(x_t),m_{t-1}\rangle$ (which only makes the RHS larger), we get
    \begin{align}
        f(x_{t+1})
        &\le f(x_t) - \eta(1-\beta)\langle\nabla f(x_t),g_t\rangle
            + \frac{L\eta^2}{2}C^2 .
        \label{eq:descent2}
    \end{align}
    Insert the bound \eqref{eq:clipbound2}:
    \begin{align}
        f(x_{t+1})
        &\le f(x_t) - \eta(1-\beta)\bigl(\norm{\nabla f(x_t)}^2 - C\norm{\nabla f(x_t)}\bigr)
            + \frac{L\eta^2}{2}C^2 .
        \label{eq:descent3}
    \end{align}
    \item[Step 6.] \textbf{Complete the square for the linear term in $\norm{\nabla f(x_t)}$}.  
    For any $a\ge0$ and any $\alpha>0$,
    $a - \alpha a \le \frac{1}{2\alpha}a^2 + \frac{\alpha}{2}$.
    Applying with $a=\norm{\nabla f(x_t)}$ and $\alpha = \frac{\eta(1-\beta)C}{2}$ gives
    \begin{align}
        C\norm{\nabla f(x_t)}
        \le \frac{1}{2\eta(1-\beta)}\norm{\nabla f(x_t)}^2
           + \frac{\eta(1-\beta)C^2}{2} .
        \label{eq:square}
    \end{align}
    Substitute \eqref{eq:square} into \eqref{eq:descent3}:
    \begin{align}
        f(x_{t+1})
        &\le f(x_t) - \eta(1-\beta)\norm{\nabla f(x_t)}^2
            + \frac{1}{2}\norm{\nabla f(x_t)}^2
            + \frac{\eta(1-\beta)C^2}{2}
            + \frac{L\eta^2}{2}C^2 .
        \label{eq:descent4}
    \end{align}
    Rearranging terms,
    \begin{align}
        \frac{\eta(1-\beta)}{2}\norm{\nabla f(x_t)}^2
        &\le f(x_t)-f(x_{t+1}) 
            + \frac{\eta(1-\beta)C^2}{2}
            + \frac{L\eta^2}{2}C^2 .
        \label{eq:telescoping}
    \end{align}
    \item[Step 7.] \textbf{Sum over $t=0,\dots,T-1$}.  
    Summing \eqref{eq:telescoping} and using telescoping on the first difference yields
    \begin{align}
        \frac{\eta(1-\beta)}{2}\sum_{t=0}^{T-1}\norm{\nabla f(x_t)}^2
        &\le f(x_0)-f(x_T) 
            + \frac{T\,\eta(1-\beta)C^2}{2}
            + \frac{T\,L\eta^2 C^2}{2} .
        \label{eq:sum}
    \end{align}
    Since $f(x_T)\ge f_{\inf}$, we can replace $-f(x_T)$ by $-f_{\inf}$ and obtain
    \begin{align}
        \sum_{t=0}^{T-1}\norm{\nabla f(x_t)}^2
        &\le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)}
            + T C^2
            + T L\eta C^2 .
        \label{eq:bound}
    \end{align}
    Divide by $T$ and use $\eta\le 1/L$ to simplify the last term:
    \[
        \frac{1}{T}\sum_{t=0}^{T-1}\norm{\nabla f(x_t)}^2
        \le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)C\,T}
           + \frac{L C^2\eta}{1-\beta}.
    \]
    This is exactly the bound \eqref{eq:rate}. \hfill $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{eq:rate} we can select a stepsize that balances the two terms.
Let
\[
    \eta = \sqrt{\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{L C^2 (1-\beta) T}} .
\]
Because $\eta\le 1/L$ for $T$ large enough, the condition of Assumption \ref{as:step} holds.
Plugging this $\eta$ into \eqref{eq:rate} gives
\[
    \bar{G}_T \le 2\sqrt{\frac{L C^2\bigl(f(x_0)-f_{\inf}\bigr)}{(1-\beta)T}}
    = \mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
    \item \textbf{No clipping ($C=\infty$).}  
    The clipping factor becomes $1$, $g_t=\nabla f(x_t)$, and Lemma \ref{lem:momentum} still holds with $C=\infty$.  The bound \eqref{eq:rate} reduces to the classic non‑convex momentum SGD rate $\mathcal{O}(1/\sqrt{T})$.
    \item \textbf{No momentum ($\beta=0$).}  
    The algorithm coincides with clipped SGD.  The bound simplifies to
    \[
        \bar{G}_T \le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta C T}
            + L C^2\eta .
    \]
    This matches the known result for SGD with gradient clipping (see e.g., \cite{Ghadimi2013}).
    \item \textbf{Deterministic quadratic.}  
    For $f(w)=(w-a)^2$, the gradient norm is linear in $w-a$, and clipping forces $g_t$ to be either the true gradient (if $|2(w-a)|\le C$) or a constant $C$ in magnitude.  The recursion can be solved in closed form, confirming the $1/T$ decrease of $f(x_t)-f_{\inf}$ when $\eta$ is chosen as $1/L$.
\end{enumerate}
