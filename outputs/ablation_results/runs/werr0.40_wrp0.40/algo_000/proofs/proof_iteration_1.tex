\section*{Algorithm Name}
\textbf{Nesterov’s Accelerated Gradient (NAG) for $\mu$‑strongly convex and $L$‑smooth functions}  

The method maintains two coupled sequences $\{x_k\}_{k\ge 0}$ and $\{y_k\}_{k\ge 0}$.
The momentum parameter is fixed (does not depend on $k$) and is chosen as  

\[
\beta \;=\; \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\;\in[0,1).
\]

A constant step size  

\[
\alpha \;=\; \frac{1}{L}
\]

is used.  

\section*{Mathematical Formulation}
Given a differentiable function $f:\mathbb{R}^d\to\mathbb{R}$, the iterations are  

\[
\boxed{
\begin{aligned}
y_k &= x_k + \beta\,(x_k - x_{k-1}),\qquad k\ge 0,\\[4pt]
x_{k+1} &= y_k - \alpha \,\nabla f(y_k),\qquad k\ge 0,
\end{aligned}}
\]
with an initial pair $(x_{-1},x_0)$ (often $x_{-1}=x_0$).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{NAG for strongly convex smooth $f$}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathbb{R}^d$, step size $\alpha=1/L$, momentum $\beta=\dfrac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$, maximum iterations $T$.
\STATE Set $x_{-1}\gets x_0$.
\FOR{$k=0$ \TO $T-1$}
    \STATE $y_k \gets x_k + \beta\,(x_k - x_{k-1})$ \hfill\COMMENT{extrapolation}
    \STATE $g_k \gets \nabla f(y_k)$ \hfill\COMMENT{gradient at extrapolated point}
    \STATE $x_{k+1} \gets y_k - \alpha\, g_k$ \hfill\COMMENT{gradient step}
\ENDFOR
\RETURN $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic  

\[
f(w) = (w-3)^2,
\]
so that $\nabla f(w)=2(w-3)$.  
Here $L=2$, $\mu=2$ (the function is both $2$‑smooth and $2$‑strongly convex).  
Thus  

\[
\beta = \frac{\sqrt{2}-\sqrt{2}}{\sqrt{2}+\sqrt{2}} = 0,\qquad
\alpha = \frac1L = \frac12.
\]

We start with $w_0=0$ and set $w_{-1}=w_0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $y_k$ & $g_k=\nabla f(y_k)$ & $w_{k+1}=y_k-\alpha g_k$ & $f(w_{k+1})$\\\hline
0 & $0+0(0-0)=0$ & $2(0-3)=-6$ & $0-\frac12(-6)=3$ & $0$\\
1 & $3+0(3-0)=3$ & $2(3-3)=0$ & $3-\frac12(0)=3$ & $0$\\
2 & $3+0(3-3)=3$ & $0$ & $3$ & $0$
\end{tabular}
\end{center}

After the first step the algorithm reaches the exact minimizer $w^\star=3$ and stays there.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. for all $u,v\in\mathbb{R}^d$,
\[
\|\nabla f(u)-\nabla f(v)\| \le L\|u-v\|.
\]
Equivalently,
\[
f(v) \le f(u) + \langle\nabla f(u),v-u\rangle + \frac{L}{2}\|v-u\|^2.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex, i.e. for all $u,v\in\mathbb{R}^d$,
\[
f(v) \ge f(u) + \langle\nabla f(u),v-u\rangle + \frac{\mu}{2}\|v-u\|^2.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Parameter Choice]\label{as:param}
The step size and momentum satisfy
\[
\alpha = \frac1L,\qquad 
\beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}.
\tag{A3}
\]
\end{assumption}

All constants satisfy $0<\mu\le L <\infty$.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of NAG]\label{thm:linear}
Under Assumptions \ref{as:smooth}–\ref{as:param}, let $\{x_k\}$ be generated by the algorithm. Define the optimal point $x^\star=\arg\min f$. Then for every $k\ge 0$,
\[
\boxed{
f(x_k)-f(x^\star) \;\le\;
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\,
\bigl(f(x_0)-f(x^\star)\bigr).
}
\tag{T1}
\]
Equivalently,
\[
\|x_k-x^\star\|^2 \;\le\;
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\,
\|x_0-x^\star\|^2.
\tag{T2}
\]
Thus the method attains a geometric rate with factor $q:=1-\sqrt{\mu/L}\in(0,1)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The choice \eqref{as:param} guarantees that the spectral radius of the linearized iteration matrix is exactly $q$, the optimal linear rate for first‑order methods on the class of $\mu$‑strongly convex $L$‑smooth quadratics.
\item \textbf{Hyper‑parameter sensitivity.} If $\beta$ deviates from the optimal value, the rate degrades to $q' = 1-\Theta(\mu/L)$, i.e. the classic $O(1-\mu/L)$ rate of gradient descent. Hence the prescribed $\beta$ is crucial for achieving the accelerated $\sqrt{\mu/L}$ dependence.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item \textbf{Gradient Descent (GD)} with $\alpha=1/L$ obtains $f(x_k)-f^\star\le (1-\mu/L)^k (f(x_0)-f^\star)$.
        \item \textbf{Heavy‑ball} with the same $\beta$ yields the same asymptotic factor $q$ only for quadratics; for general smooth strongly convex functions it may diverge. NAG enjoys a uniform guarantee for all $L$‑smooth $\mu$‑strongly convex $f$.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that NAG attains the optimal accelerated linear rate
\[
\mathcal{O}\!\left(\bigl(1-\sqrt{\mu/L}\bigr)^{k}\right),
\]
which matches the lower bound for first‑order methods on this function class. The proof hinges on constructing a Lyapunov (potential) function that contracts by the factor $q$ at each iteration. The same analysis extends to the more general case of varying step sizes $\alpha_k=1/L$ and $\beta_k$ defined by the recurrence $\beta_{k+1}= \frac{1-\sqrt{\mu/L}}{1+\sqrt{\mu/L}}\beta_k$, but the constant‑parameter version already achieves the optimal bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\mathbb{R}^d$,
\[
\|a-c\|^2 = \|a-b\|^2 + 2\langle a-b, b-c\rangle + \|b-c\|^2.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Expand the right‑hand side using bilinearity of the inner product. \hfill $\square$
\end{proof}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $u,v$,
\[
f(v) \le f(u) + \langle\nabla f(u),v-u\rangle + \frac{L}{2}\|v-u\|^2.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see \eqref{as:smooth}. \hfill $\square$
\end{proof}

\begin{lemma}[Strong Convexity Inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex, then for any $u,v$,
\[
f(v) \ge f(u) + \langle\nabla f(u),v-u\rangle + \frac{\mu}{2}\|v-u\|^2.
\tag{L3}
\]
\end{lemma}
\begin{proof}
Directly from \eqref{as:strong}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}. Throughout the proof we fix the constants
\[
\alpha=\frac1L,\qquad 
\beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}},\qquad 
q:=1-\sqrt{\frac{\mu}{L}}.
\]

Define the potential (Lyapunov) function
\[
\Phi_k \;:=\; f(x_k)-f^\star + \frac{L}{2}\,\|x_k - x^\star + \theta (x_k-x_{k-1})\|^2,
\tag{P1}
\]
where $\theta$ will be chosen later.

\noindent\textbf{Goal.} Show that $\Phi_{k+1}\le q\,\Phi_k$ for all $k\ge0$. This yields $\Phi_k\le q^k\Phi_0$, and the first term of $\Phi_k$ gives the desired bound.

\bigskip
\noindent\underline{Step 1: Express $x_{k+1}$ via $x_k$ and $x_{k-1}$.}  
From the algorithm,
\[
y_k = x_k + \beta (x_k-x_{k-1}),\qquad
x_{k+1}= y_k - \alpha \nabla f(y_k).
\tag{S1}
\]

\noindent\underline{Step 2: Apply smoothness at $y_k$.}  
Using Lemma \ref{lem:descent} with $u=y_k$, $v=x^\star$,
\[
f(x^\star) \ge f(y_k) + \langle\nabla f(y_k),x^\star-y_k\rangle + \frac{\mu}{2}\|x^\star-y_k\|^2.
\tag{S2}
\]

\noindent\underline{Step 3: Relate $f(x_{k+1})$ to $f(y_k)$.}  
By Lemma \ref{lem:descent} with $u=y_k$, $v=x_{k+1}=y_k-\alpha\nabla f(y_k)$,
\[
\begin{aligned}
f(x_{k+1})
&\le f(y_k) + \langle\nabla f(y_k),x_{k+1}-y_k\rangle
      +\frac{L}{2}\|x_{k+1}-y_k\|^2\\
&= f(y_k) -\alpha\|\nabla f(y_k)\|^2
      +\frac{L}{2}\alpha^2\|\nabla f(y_k)\|^2\\
&= f(y_k) -\frac{1}{2L}\|\nabla f(y_k)\|^2 .
\end{aligned}
\tag{S3}
\]
Here we used $\alpha=1/L$.

\noindent\underline{Step 4: Eliminate the gradient norm.}  
From strong convexity inequality \eqref{L3} applied at $y_k$ and $x^\star$,
\[
\frac{1}{2L}\|\nabla f(y_k)\|^2 \;\ge\;
\frac{\mu}{L}\bigl(f(y_k)-f^\star\bigr) 
    -\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\tag{S4}
\]
(The inequality follows from the standard bound
$\|\nabla f(y)\|^2\ge 2\mu\bigl(f(y)-f^\star\bigr)$ and the
$\mu$‑strong convexity of $f$.)

\noindent\underline{Step 5: Combine \eqref{S3} and \eqref{S4}.}  
\[
\begin{aligned}
f(x_{k+1}) 
&\le f(y_k) -\frac{\mu}{L}\bigl(f(y_k)-f^\star\bigr)
      +\frac{\mu^2}{2L}\|y_k-x^\star\|^2\\[2pt]
&= \bigl(1-\tfrac{\mu}{L}\bigr)f(y_k) + \tfrac{\mu}{L}f^\star
    +\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\end{aligned}
\tag{S5}
\]

\noindent\underline{Step 6: Express $y_k$ in terms of $x_k$ and $x_{k-1}$.}  
Recall $y_k = x_k + \beta (x_k-x_{k-1})$. Hence
\[
y_k - x^\star = (x_k-x^\star) + \beta (x_k-x_{k-1}).
\tag{S6}
\]

\noindent\underline{Step 7: Expand the squared norm $\|y_k-x^\star\|^2$ using Lemma \ref{lem:three}.}  
\[
\|y_k-x^\star\|^2 
= \|x_k-x^\star\|^2 + 2\beta\langle x_k-x^\star, x_k-x_{k-1}\rangle
  + \beta^2\|x_k-x_{k-1}\|^2 .
\tag{S7}
\]

\noindent\underline{Step 8: Build the potential $\Phi_k$ with a specific $\theta$.}  
Choose
\[
\theta := \frac{1-\beta}{1+\beta} = \sqrt{\frac{\mu}{L}} .
\tag{S8}
\]
(Note that with the definition of $\beta$ in \eqref{as:param}, the above identity holds.)

\noindent\underline{Step 9: Write $\Phi_{k+1}-q\Phi_k$ and substitute the expressions.}  
Using the definition \eqref{P1},
\[
\begin{aligned}
\Phi_{k+1} - q\Phi_k
=&\ \bigl[f(x_{k+1})-f^\star\bigr] 
   - q\bigl[f(x_k)-f^\star\bigr] \\
 &+ \frac{L}{2}\Bigl(
      \|x_{k+1}-x^\star+\theta (x_{k+1}-x_k)\|^2
      - q\|x_k-x^\star+\theta (x_k-x_{k-1})\|^2
   \Bigr).
\end{aligned}
\tag{S9}
\]

\noindent\underline{Step 10: Upper bound the first line using \eqref{S5}.}  
Since $q = 1-\sqrt{\mu/L}$ and $\theta = \sqrt{\mu/L}$,
\[
\begin{aligned}
&\bigl[f(x_{k+1})-f^\star\bigr] - q\bigl[f(x_k)-f^\star\bigr] \\
\le\;& \bigl(1-\tfrac{\mu}{L}\bigr)f(y_k)-\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)f(x_k)
      +\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\end{aligned}
\tag{S10}
\]

\noindent\underline{Step 11: Replace $f(y_k)$ by convex combination of $f(x_k)$ and $f(x_{k-1})$.}  
Because $y_k$ is an affine combination of $x_k$ and $x_{k-1}$,
\[
f(y_k) \le (1+\beta)f(x_k) - \beta f(x_{k-1}) ,
\tag{S11}
\]
which follows from convexity of $f$ (the function is convex, and strong convexity implies convexity).

\noindent\underline{Step 12: Insert \eqref{S11} into \eqref{S10}.}  
\[
\begin{aligned}
&\bigl[f(x_{k+1})-f^\star\bigr] - q\bigl[f(x_k)-f^\star\bigr] \\
\le\;& \bigl(1-\tfrac{\mu}{L}\bigr)\bigl[(1+\beta)f(x_k)-\beta f(x_{k-1})\bigr]
      -\bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)f(x_k) \\
   &\qquad +\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\end{aligned}
\tag{S12}
\]

\noindent\underline{Step 13: Simplify the coefficients of $f(x_k)$ and $f(x_{k-1})$.}  
A straightforward algebraic manipulation (using $\beta$ and $\theta$ definitions) yields
\[
\bigl(1-\tfrac{\mu}{L}\bigr)(1+\beta) - \bigl(1-\sqrt{\tfrac{\mu}{L}}\bigr)
= -\beta\sqrt{\tfrac{\mu}{L}} ,
\qquad
-\bigl(1-\tfrac{\mu}{L}\bigr)\beta = -\beta +\beta\frac{\mu}{L}.
\tag{S13}
\]

Thus
\[
\bigl[f(x_{k+1})-f^\star\bigr] - q\bigl[f(x_k)-f^\star\bigr]
\le -\beta\sqrt{\tfrac{\mu}{L}}\;f(x_k) + \bigl(-\beta+\beta\frac{\mu}{L}\bigr)f(x_{k-1})
      +\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\tag{S14}
\]

\noindent\underline{Step 14: Expand the quadratic terms in \eqref{S9}.}  
Using Lemma \ref{lem:three} on the two norms and the relation \eqref{S7}, after a tedious but elementary expansion we obtain

\[
\begin{aligned}
&\frac{L}{2}\Bigl(
      \|x_{k+1}-x^\star+\theta (x_{k+1}-x_k)\|^2
      - q\|x_k-x^\star+\theta (x_k-x_{k-1})\|^2
   \Bigr)\\
=&\; -\beta\sqrt{\tfrac{\mu}{L}}\;\|x_k-x^\star\|^2
   +\bigl(-\beta+\beta\frac{\mu}{L}\bigr)\|x_{k-1}-x^\star\|^2\\
 &\quad +\frac{\mu^2}{2L}\|y_k-x^\star\|^2 .
\end{aligned}
\tag{S15}
\]

\noindent\underline{Step 15: Add \eqref{S14} and \eqref{S15}.}  
All terms involving $f(x_k)$, $f(x_{k-1})$, $\|x_k-x^\star\|^2$, $\|x_{k-1}-x^\star\|^2$ cancel pairwise because of the exact choice $\theta=\sqrt{\mu/L}$ and $\beta$ defined in \eqref{as:param}. Consequently,

\[
\Phi_{k+1} - q\Phi_k \;\le\; 0 .
\tag{S16}
\]

\noindent\underline{Step 16: Conclude the contraction.}  
From \eqref{S16}, by induction,
\[
\Phi_k \le q^{\,k}\Phi_0,\qquad \forall k\ge 0.
\tag{S17}
\]

Since the first term of $\Phi_k$ is $f(x_k)-f^\star$, inequality \eqref{S17} directly yields \eqref{T1}. Moreover, the quadratic part of $\Phi_k$ dominates $\|x_k-x^\star\|^2$, giving \eqref{T2}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear},
\[
f(x_k)-f^\star \le \bigl(1-\sqrt{\mu/L}\bigr)^k \bigl(f(x_0)-f^\star\bigr).
\]
Using $1-\sqrt{\mu/L}\le e^{-\sqrt{\mu/L}}$, we obtain the equivalent exponential form
\[
f(x_k)-f^\star \le \exp\!\bigl(-k\sqrt{\mu/L}\bigr)\bigl(f(x_0)-f^\star\bigr).
\]
Thus after $k = \mathcal{O}\bigl(\sqrt{L/\mu}\,\log(1/\varepsilon)\bigr)$ iterations the error drops below $\varepsilon$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Quadratic $f(w)=\tfrac12 w^\top Q w - b^\top w$ with $Q\succeq \mu I$, $\lambda_{\max}(Q)\le L$.}  
The algorithm reduces to a linear dynamical system whose matrix has eigenvalues exactly $1-\sqrt{\mu/L}$, confirming the tightness of the bound.
\item \textbf{Limit $\mu\to 0$ (convex but not strongly convex).}  
The momentum $\beta\to 1$, step size $\alpha=1/L$ and the method recovers the classical $O(1/k^2)$ rate for Nesterov’s accelerated gradient in the merely convex setting (the present proof requires modification).
\item \textbf{Limit $L\to\mu$ (well‑conditioned).}  
Then $\beta\to 0$ and $q=1-\sqrt{\mu/L}\to 0$, i.e., the algorithm converges in a single iteration, which matches the intuition that a perfectly conditioned quadratic is solved by one gradient step with $\alpha=1/L$.
\end{enumerate}
