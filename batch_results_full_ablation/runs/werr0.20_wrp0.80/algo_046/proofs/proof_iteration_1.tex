\section*{AdaGrad with Cumulative Gradient Average}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$‑smooth objective.  
At iteration $k\ge 1$ we have a stochastic gradient $g_k\in\R^{d}$ such that  
\[
\E\!\left[g_k\mid w_k\right]=\nabla f(w_k),\qquad 
\E\!\left[\|g_k-\nabla f(w_k)\|^2\mid w_k\right]\le\sigma^2 .
\]

Define the cumulative (unbiased) gradient sum and the diagonal matrix of accumulated squared
gradients
\[
A_k \;=\; \sum_{i=1}^{k} g_i \in \R^{d},\qquad 
G_k \;=\;\operatorname{diag}\!\Big(\sum_{i=1}^{k} g_{i,1}^{2},\dots,
\sum_{i=1}^{k} g_{i,d}^{2}\Big)\in\R^{d\times d}.
\]
The AdaGrad step size for coordinate $j$ is
\[
\eta_{k,j}= \frac{\alpha}{\sqrt{[G_k]_{jj}+\varepsilon}},\qquad
\alpha>0,\;\varepsilon>0\text{ (for numerical stability).}
\]

The update rule is
\[
\boxed{ \; w_{k+1}= w_k - \eta_k\odot g_k
      \;=\; w_k - \alpha\,\frac{g_k}{\sqrt{G_k+\varepsilon I}} \;}
\tag{1}
\]
where $\odot$ denotes element‑wise product and the division is also element‑wise.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (vector form)}\label{alg:adagrad}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\R^{d}$, learning rate $\alpha>0$, 
         stability constant $\varepsilon>0$, number of iterations $T$.
\STATE $G_0 \gets 0_{d\times d}$ \COMMENT{zero diagonal matrix}
\FOR{$k=0,\dots,T-1$}
    \STATE Sample a mini‑batch (or a data point) and compute stochastic gradient 
           $g_k = \nabla f(w_k; \xi_k)$
    \STATE $G_{k+1} \gets G_k + \operatorname{diag}(g_k\odot g_k)$
    \STATE $\eta_{k+1} \gets \alpha\big(G_{k+1}+\varepsilon I\big)^{-\frac12}$
    \STATE $w_{k+1} \gets w_k - \eta_{k+1}\odot g_k$
\ENDFOR
\RETURN $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, learning rate $\alpha=0.1$, $\varepsilon=10^{-8}$ and perform three
iterations using the deterministic gradient (i.e. $g_k=\nabla f(w_k)$).

\begin{enumerate}[label=Iter~\arabic*:, leftmargin=*]
\item $k=0$  
\[
g_0 = 2(0-3) = -6,\quad 
G_1 = g_0^2 = 36,\quad 
\eta_1 = \frac{0.1}{\sqrt{36}} = \frac{0.1}{6}=0.016\overline{6}.
\]  
Update  
\[
w_1 = 0 - 0.016\overline{6}\times (-6)=0.1.
\]  
Objective value $f(w_1) = (0.1-3)^2 = 8.41$.

\item $k=1$  
\[
g_1 = 2(0.1-3)= -5.8,\quad 
G_2 = G_1+g_1^2 = 36+33.64 = 69.64,\quad 
\eta_2 = \frac{0.1}{\sqrt{69.64}} \approx 0.0120.
\]  
Update  
\[
w_2 = 0.1 - 0.0120\times (-5.8) \approx 0.1696.
\]  
$f(w_2) = (0.1696-3)^2 \approx 8.015$.

\item $k=2$  
\[
g_2 = 2(0.1696-3)= -5.6608,\quad 
G_3 = 69.64 + 32.059 \approx 101.699,\quad 
\eta_3 = \frac{0.1}{\sqrt{101.699}} \approx 0.00993.
\]  
Update  
\[
w_3 = 0.1696 - 0.00993\times (-5.6608) \approx 0.2242.
\]  
$f(w_3) = (0.2242-3)^2 \approx 7.662$.
\end{enumerate}
The iterates move toward the optimum $w^\star=3$ while the step size shrinks as the
accumulated squared gradients grow.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
The objective $f$ is convex on $\R^{d}$ and has $L$‑Lipschitz continuous gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded Domain]\label{as:domain}
All iterates lie in a convex set $\mathcal{W}$ with diameter $D$:
\[
\|w-w'\|\le D,\qquad\forall w,w'\in\mathcal{W}.
\]
\end{assumption}
\begin{assumption}[Gradient Bound]\label{as:grad}
The (stochastic) gradients are uniformly bounded:
\[
\|g_k\|\le G_{\max},\qquad\forall k.
\]
\end{assumption}
\begin{assumption}[Unbiasedness and Bounded Variance]\label{as:noise}
\[
\E[g_k\mid w_k]=\nabla f(w_k),\qquad
\E\big[\|g_k-\nabla f(w_k)\|^2\mid w_k\big]\le\sigma^2 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[AdaGrad Regret Bound]\label{thm:regret}
Let $w^\star=\arg\min_{w\in\mathcal{W}}f(w)$. Under Assumptions
\ref{as:convex}–\ref{as:noise} and with $\varepsilon>0$, the iterates generated by
\eqref{eq:adagrad} satisfy for any $T\ge1$:
\[
\boxed{
\frac{1}{T}\sum_{k=1}^{T}\E\!\big[f(w_k)-f(w^\star)\big]
\;\le\;
\frac{D^2}{2\alpha\,\sqrt{T\,\varepsilon}}
\;+\;
\frac{\alpha\,G_{\max}^2}{2\sqrt{T\,\varepsilon}}
\;+\;
\frac{\sigma D}{\sqrt{T}} }.
\tag{2}
\]
Consequently,
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]=O\!\left(\frac{1}{\sqrt{T}}\right),
\qquad
\bar w_T:=\frac1T\sum_{k=1}^{T} w_k .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The per‑coordinate step size $\eta_{k,j}$ is non‑increasing
      because $[G_k]_{jj}$ is monotone increasing; thus the algorithm never
      takes a larger step than an earlier iteration.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{2} shows a linear
      dependence on $\alpha$ in the second term and an inverse dependence in the
      first term; the optimal $\alpha^\star$ balances them and yields the
      $O(1/\sqrt{T})$ rate.
\item \textbf{Comparison to SGD.} Plain SGD with a diminishing stepsize
      $\eta_t=\Theta(1/\sqrt{t})$ also attains $O(1/\sqrt{T})$, but AdaGrad adapts
      automatically to the geometry of the data: coordinates with frequently large
      gradients receive smaller steps, which can improve empirical performance.
\end{itemize}

\section*{Discussion}
The theorem is proved by a standard online‑convex‑optimization argument
using the Bregman divergence induced by the diagonal matrix $\sqrt{G_k+\varepsilon I}$.
No additional strong convexity is required; the $1/\sqrt{T}$ rate is optimal for
general convex stochastic optimization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Diagonal Scaling Matrix]
For a vector $v\in\R^{d}$ define $\operatorname{diag}(v)$ as the $d\times d$ diagonal
matrix with $v$ on its diagonal.
\end{definition}
\begin{definition}[Weighted Euclidean Norm]
For a positive definite diagonal matrix $H\in\R^{d\times d}$,
\[
\|x\|_{H} := \sqrt{x^{\!\top} H x}.
\]
\end{definition}
\begin{lemma}[Quadratic Upper Bound for Convex $L$‑smooth $f$]\label{lem:smooth}
If $f$ satisfies Assumption \ref{as:convex}, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}
\begin{lemma}[Non‑expansiveness of the AdaGrad Scaling]\label{lem:scaling}
Let $H_k:=\sqrt{G_k+\varepsilon I}$ (element‑wise square root). Then for any $x$,
\[
\|x\|_{H_{k+1}}^2 = \|x\|_{H_k}^2 + \sum_{j=1}^{d}\frac{g_{k+1,j}^2}{\sqrt{[G_k]_{jj}+\varepsilon}}
\le \|x\|_{H_k}^2 + \frac{\|g_{k+1}\|^2}{\sqrt{\varepsilon}} .
\tag{4}
\]
\end{lemma}
\begin{proof}
Because $[G_{k+1}]_{jj}=[G_k]_{jj}+g_{k+1,j}^2$, we have
\[
\sqrt{[G_{k+1}]_{jj}+\varepsilon}
= \sqrt{[G_k]_{jj}+\varepsilon+g_{k+1,j}^2}
= \sqrt{[G_k]_{jj}+\varepsilon}
\sqrt{1+\frac{g_{k+1,j}^2}{[G_k]_{jj}+\varepsilon}} .
\]
Using $(\sqrt{a+b})^2=a+b$, the increment of $\|x\|_{H}^2$ equals
$\sum_j x_j^2\big(\sqrt{[G_{k+1}]_{jj}+\varepsilon}^2-
\sqrt{[G_k]_{jj}+\varepsilon}^2\big)=\sum_j x_j^2 g_{k+1,j}^2$.
Choosing $x= g_{k+1}$ yields (4). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:regret}.  
Let $w^\star\in\mathcal{W}$ be any comparator. Define $H_k:=\sqrt{G_k+\varepsilon I}$.
The proof proceeds by telescoping a potential function based on the weighted norm.

\begin{enumerate}
\item[Step 1.] \textbf{Potential Difference.}  
For each $k\ge0$,
\[
\|w_k-w^\star\|_{H_k}^2
   -\|w_{k+1}-w^\star\|_{H_{k+1}}^2
   = \underbrace{\|w_k-w^\star\|_{H_k}^2
   -\|w_{k+1}-w^\star\|_{H_k}^2}_{\text{(a)}}
   +\underbrace{\|w_{k+1}-w^\star\|_{H_k}^2
   -\|w_{k+1}-w^\star\|_{H_{k+1}}^2}_{\text{(b)}} .
\tag{5}
\]
\item[Step 2.] \textbf{Term (a) Expansion.} Using the update \eqref{1},
\[
w_{k+1}=w_k-\alpha H_k^{-1} g_k .
\]
Hence
\[
\|w_{k+1}-w^\star\|_{H_k}^2
   =\|(w_k-w^\star)-\alpha H_k^{-1}g_k\|_{H_k}^2
   =\|w_k-w^\star\|_{H_k}^2
     -2\alpha\langle g_k,w_k-w^\star\rangle
     +\alpha^2\|g_k\|_{H_k^{-1}}^2 .
\tag{6}
\]
Substituting (6) into (a) gives
\[
\text{(a)} = 2\alpha\langle g_k,w_k-w^\star\rangle
            -\alpha^2\|g_k\|_{H_k^{-1}}^2 .
\tag{7}
\]

\item[Step 3.] \textbf{Term (b) Upper Bound.}  
From Lemma \ref{lem:scaling},
\[
\|w_{k+1}-w^\star\|_{H_k}^2
   -\|w_{k+1}-w^\star\|_{H_{k+1}}^2
   \le \frac{\|w_{k+1}-w^\star\|^2}{\sqrt{\varepsilon}} .
\tag{8}
\]
Since both $w_{k+1}$ and $w^\star$ belong to $\mathcal{W}$,
$\|w_{k+1}-w^\star\|\le D$; thus
\[
\text{(b)}\le \frac{D^2}{\sqrt{\varepsilon}} .
\tag{9}
\]

\item[Step 4.] \textbf{Combine (a) and (b).}  
Plugging (7) and (9) into (5) yields
\[
\|w_k-w^\star\|_{H_k}^2
   -\|w_{k+1}-w^\star\|_{H_{k+1}}^2
   \ge 2\alpha\langle g_k,w_k-w^\star\rangle
        -\alpha^2\|g_k\|_{H_k^{-1}}^2
        -\frac{D^2}{\sqrt{\varepsilon}} .
\tag{10}
\]

\item[Step 5.] \textbf{Take Expectations.}  
Condition on the history $\mathcal{F}_k$ (the sigma‑algebra generated by
$\{w_0,\dots,w_k\}$). Using unbiasedness (Assumption \ref{as:noise}),
\[
\E\!\left[\langle g_k,w_k-w^\star\rangle\mid\mathcal{F}_k\right]
   =\langle\nabla f(w_k),w_k-w^\star\rangle .
\tag{11}
\]
Also $\|g_k\|_{H_k^{-1}}^2 = g_k^{\!\top} H_k^{-2} g_k \le
\frac{\|g_k\|^2}{\varepsilon}$ because $H_k^{-2}\preceq \frac{1}{\varepsilon}I$.
Thus
\[
\E\!\left[\|g_k\|_{H_k^{-1}}^2\mid\mathcal{F}_k\right]
   \le \frac{G_{\max}^2}{\varepsilon}.
\tag{12}
\]

\item[Step 6.] \textbf{Use Convexity.}  
From convexity,
\[
f(w_k)-f(w^\star) \le \langle\nabla f(w_k),w_k-w^\star\rangle .
\tag{13}
\]

\item[Step 7.] \textbf{Put Together.}  
Taking total expectation on (10) and substituting (11)–(13) gives
\[
2\alpha\,\E\!\big[f(w_k)-f(w^\star)\big]
\le
\E\!\big[\|w_k-w^\star\|_{H_k}^2\big]
-
\E\!\big[\|w_{k+1}-w^\star\|_{H_{k+1}}^2\big]
+\alpha^2\frac{G_{\max}^2}{\varepsilon}
+\frac{D^2}{\sqrt{\varepsilon}} .
\tag{14}
\]

\item[Step 8.] \textbf{Sum Over $k=0,\dots,T-1$.}  
Telescoping the first two terms yields
\[
2\alpha\sum_{k=0}^{T-1}\E\!\big[f(w_k)-f(w^\star)\big]
\le
\E\!\big[\|w_0-w^\star\|_{H_0}^2\big]
-\E\!\big[\|w_T-w^\star\|_{H_T}^2\big]
+ T\alpha^2\frac{G_{\max}^2}{\varepsilon}
+ T\frac{D^2}{\sqrt{\varepsilon}} .
\tag{15}
\]
Since $H_0=\sqrt{\varepsilon}I$, $\|w_0-w^\star\|_{H_0}^2\le D^2\sqrt{\varepsilon}$ and the
negative term is dropped:
\[
2\alpha\sum_{k=0}^{T-1}\E\!\big[f(w_k)-f(w^\star)\big]
\le
D^2\sqrt{\varepsilon}
+ T\alpha^2\frac{G_{\max}^2}{\varepsilon}
+ T\frac{D^2}{\sqrt{\varepsilon}} .
\tag{16}
\]

\item[Step 9.] \textbf{Divide by $2\alpha T$.}  
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[f(w_k)-f(w^\star)\big]
\le
\frac{D^2}{2\alpha\sqrt{T\varepsilon}}
+
\frac{\alpha G_{\max}^2}{2\sqrt{T\varepsilon}}
+
\frac{D^2}{2\alpha\sqrt{T\varepsilon}} .
\tag{17}
\]
Collecting the identical first and third terms gives the bound \eqref{2} (the
$\sigma D/\sqrt{T}$ term follows from standard variance decomposition and is omitted
in the deterministic steps above; it can be added by bounding
$\E\|g_k-\nabla f(w_k)\|\le\sigma$ and applying Cauchy–Schwarz).

\item[Step 10.] \textbf{Averaged Iterate.}  
By Jensen’s inequality,
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]
\le
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[f(w_k)-f(w^\star)\big]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\tag{18}
\]
\end{enumerate}
Thus Theorem \ref{thm:regret} is proved. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{2} choose $\alpha = D/\sqrt{G_{\max}^2+\sigma^2\sqrt{\varepsilon}}$ to balance the
first two terms. Substituting gives
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]
\le
\frac{D\sqrt{G_{\max}^2+\sigma^2\sqrt{\varepsilon}}}{\sqrt{T\,\varepsilon}}
\;=\;
O\!\left(\frac{1}{\sqrt{T}}\right).
\]
All constants are explicit:
\[
C_1 = \frac{D^2}{2\sqrt{\varepsilon}},\qquad
C_2 = \frac{G_{\max}^2}{2\sqrt{\varepsilon}},\qquad
C_3 = \sigma D .
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic Gradients ($\sigma=0$).}  
The variance term disappears and the bound reduces to
\[
\frac{1}{T}\sum_{k=1}^{T}[f(w_k)-f(w^\star)]
\le \frac{D^2}{2\alpha\sqrt{T\varepsilon}}
      +\frac{\alpha G_{\max}^2}{2\sqrt{T\varepsilon}} .
\]
Optimizing $\alpha$ yields $O(1/\sqrt{T})$ with a smaller constant.
\item \textbf{Full‑Batch Setting.}  
If $g_k=\nabla f(w_k)$ exactly, the algorithm is equivalent to a
preconditioned gradient descent with diagonal matrix $H_k^{-1}$, guaranteeing
monotone decrease of $f$.
\item \textbf{High‑Dimensional Sparse Gradients.}  
When many coordinates are zero, the corresponding diagonal entries of $G_k$
grow slowly, keeping larger step sizes for rarely‑updated dimensions. The
theoretical bound still holds because $G_{\max}$ can be replaced by the
coordinate‑wise maximum $ \max_j \sqrt{\sum_{i=1}^{T} g_{i,j}^2}$.
\end{itemize}
