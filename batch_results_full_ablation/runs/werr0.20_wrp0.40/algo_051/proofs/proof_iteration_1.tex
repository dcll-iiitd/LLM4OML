\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics with Decreasing Temperature (SGLD‑DT)}  

The method adds isotropic Gaussian noise whose variance decays with the iteration index, thereby interpolating between stochastic gradient descent and Langevin diffusion.

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the objective function.  
For $k=0,1,2,\dots$ the iterate $x_{k}\in\R^{d}$ is updated by  

\[
\boxed{
x_{k+1}=x_{k}-\alpha_{k}\,\nabla f(x_{k})+\sqrt{2\alpha_{k}\,\tau_{k}}\;\xi_{k}
}
\tag{1}
\]

where  

\begin{itemize}
  \item $\alpha_{k}>0$ is the stepsize (learning rate) at iteration $k$,
  \item $\tau_{k}>0$ is the temperature (noise variance) at iteration $k$,
  \item $\xi_{k}\sim\mathcal N(0,I_{d})$ are i.i.d. standard Gaussian vectors independent of $x_{k}$,
  \item the temperature schedule is \(\displaystyle \tau_{k}= \frac{\tau_{0}}{(k+1)^{\beta}}\) with $\tau_{0}>0$ and $\beta\in(0,1]$.
\end{itemize}

\section*{Pseudocode}
\begin{verbatim}
Input: initial point x0, stepsize schedule {αk}, 
       temperature schedule {τk}, maxIter K
for k = 0,...,K-1 do
    gk = ∇f(xk)                     # exact gradient
    ξk ~ N(0, I_d)                  # draw Gaussian noise
    xk+1 = xk - αk * gk + sqrt(2*αk*τk) * ξk
end for
return xK
\end{verbatim}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^{2}$, so $\nabla f(w)=2(w-3)$.  
Choose $\alpha_{k}=0.1$ (constant), $\tau_{0}=1$, $\beta=1$ (so $\tau_{k}=1/(k+1)$) and start from $w_{0}=0$.

\[
\begin{array}{c|c|c|c}
k & \nabla f(w_{k}) & \sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k} & w_{k+1}=w_{k}-\alpha_{k}\nabla f(w_{k})
      +\sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}\\ \hline
0 & 2(0-3)=-6 & \sqrt{0.2\cdot1}\,\xi_{0}\approx0.447\,\xi_{0} &
0-0.1(-6)+0.447\xi_{0}=0.6+0.447\xi_{0}\\[2mm]
1 & 2(w_{1}-3) & \sqrt{0.2\cdot\frac12}\,\xi_{1}\approx0.316\,\xi_{1} &
w_{1}-0.1\cdot2(w_{1}-3)+0.316\xi_{1}\\[2mm]
2 & 2(w_{2}-3) & \sqrt{0.2\cdot\frac13}\,\xi_{2}\approx0.258\,\xi_{2} &
w_{2}-0.1\cdot2(w_{2}-3)+0.258\xi_{2}
\end{array}
\]

If we draw $\xi_{0}=0$, $\xi_{1}=0$, $\xi_{2}=0$ (deterministic illustration), the iterates are  

\[
w_{1}=0.6,\qquad 
w_{2}=0.6-0.1\cdot2(0.6-3)=0.6-0.1\cdot(-4.8)=1.08,\qquad
w_{3}=1.08-0.1\cdot2(1.08-3)=1.08-0.1\cdot(-3.84)=1.464 .
\]

Corresponding objective values: $f(w_{1})=5.76$, $f(w_{2})=3.6864$, $f(w_{3})=2.3529$ (monotonically decreasing).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. $\forall x,y\in\R^{d}$  
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_{\inf}>-\infty$ such that $\forall x,\; f(x)\ge f_{\inf}$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The deterministic stepsizes satisfy  
\[
\sum_{k=0}^{\infty}\alpha_{k}= \infty,\qquad 
\sum_{k=0}^{\infty}\alpha_{k}^{2}<\infty .
\tag{3}
\]
A typical choice is $\alpha_{k}=c/(k+1)^{a}$ with $a\in(0.5,1]$.
\end{assumption}
\begin{assumption}[Temperature decay]\label{ass:temp}
Temperature obeys $\tau_{k}= \tau_{0}/(k+1)^{\beta}$ with $\beta\in(0,1]$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:temp} and the update \eqref{1}, let $x_{k}$ be the generated iterates. Then for any $K\ge1$,
\[
\frac{1}{\sum_{k=0}^{K-1}\alpha_{k}}\,
\sum_{k=0}^{K-1}\alpha_{k}\,\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\sum_{k=0}^{K-1}\alpha_{k}}
\;+\;
\frac{L d\,\tau_{0}}{(1-\beta)}\,
\frac{\sum_{k=0}^{K-1}\alpha_{k}(k+1)^{-\beta}}{\sum_{k=0}^{K-1}\alpha_{k}} .
\tag{4}
\]
In particular, if $\alpha_{k}=c/(k+1)^{a}$ with $a\in(0.5,1]$, then  
\[
\min_{0\le k\le K-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
= O\!\left( K^{-\,\min\{1-a,\beta\}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} The term $L\alpha_{k}^{2}\norm{\nabla f(x_{k})}^{2}$ in the descent inequality is summable by \eqref{3}, guaranteeing that the noise does not cause divergence.
  \item \textbf{Hyper‑parameter Sensitivity.} Larger $c$ (stepsize constant) accelerates early progress but increases the $L\alpha_{k}^{2}$ error term; the temperature decay exponent $\beta$ trades off exploration (large $\beta$ quickly reduces noise) against asymptotic bias.
  \item \textbf{Comparison to Baselines.}
    \begin{itemize}
      \item With $\tau_{k}\equiv0$, \eqref{1} reduces to vanilla GD and \eqref{4} recovers the classic $O(1/\sum\alpha_{k})$ gradient bound.
      \item With constant temperature $\tau_{k}\equiv\tau_{0}$, the second term in \eqref{4} does not vanish, leading to a non‑zero asymptotic bias (the stationary distribution of the Langevin diffusion). The decreasing schedule removes this bias asymptotically.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the non‑convex landscape, the average squared gradient norm converges to zero at a sub‑linear rate determined by the slower of the stepsize decay $a$ and the temperature decay $\beta$. The proof follows the classical descent lemma for smooth functions, augmented with an explicit treatment of the stochastic perturbation term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any vector $u$,
\[
f(x+u)\le f(x)+\langle\nabla f(x),u\rangle+\frac{L}{2}\norm{u}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Direct consequence of the integral form of the gradient and Cauchy–Schwarz; see standard textbooks. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the randomness of $\{\xi_{k}\}$ conditioned on the past.

\medskip
\noindent\textbf{Step 1. Apply the descent lemma to the update \eqref{1}.}
\begin{align}
f(x_{k+1})
&\le f(x_{k})+\Big\langle \nabla f(x_{k}),\,
      -\alpha_{k}\nabla f(x_{k})+\sqrt{2\alpha_{k}\tau_{k}}\,\xi_{k}\Big\rangle
      +\frac{L}{2}\norm{-\alpha_{k}\nabla f(x_{k})+\sqrt{2\alpha_{k}\tau_{k}}\,\xi_{k}}^{2}
      \nonumber\\
&= f(x_{k}) -\alpha_{k}\norm{\nabla f(x_{k})}^{2}
   +\sqrt{2\alpha_{k}\tau_{k}}\;\langle \nabla f(x_{k}),\xi_{k}\rangle
   +\frac{L}{2}\Big(\alpha_{k}^{2}\norm{\nabla f(x_{k})}^{2}
   +2\alpha_{k}\tau_{k}\norm{\xi_{k}}^{2}
   -2\alpha_{k}^{3/2}\sqrt{2\tau_{k}}\;\langle \nabla f(x_{k}),\xi_{k}\rangle\Big)
   \nonumber\\
&= f(x_{k}) -\alpha_{k}\Bigl(1-\frac{L\alpha_{k}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
   +\sqrt{2\alpha_{k}\tau_{k}}\Bigl(1-L\alpha_{k}^{1/2}\Bigr)\langle \nabla f(x_{k}),\xi_{k}\rangle
   +L\alpha_{k}\tau_{k}\norm{\xi_{k}}^{2}.
   \tag{6}\label{eq:step1}
\end{align}
\noindent\textbf{[Step 1]} – all algebraic expansions are written explicitly; cross‑terms are kept for later expectation.

\medskip
\noindent\textbf{Step 2. Take conditional expectation $\E[\cdot\mid x_{k}]$.}
Since $\xi_{k}\sim\mathcal N(0,I_{d})$ is independent of $x_{k}$ and has zero mean,
\[
\E\bigl[\langle \nabla f(x_{k}),\xi_{k}\rangle\mid x_{k}\bigr]=0,
\qquad 
\E\bigl[\norm{\xi_{k}}^{2}\mid x_{k}\bigr]=d .
\]
Applying this to \eqref{eq:step1} yields
\begin{align}
\E\!\left[f(x_{k+1})\mid x_{k}\right]
&\le f(x_{k}) -\alpha_{k}\Bigl(1-\frac{L\alpha_{k}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
   +L\alpha_{k}\tau_{k}\,d .
   \tag{7}\label{eq:step2}
\end{align}
\noindent\textbf{[Step 2]} – uses zero‑mean of Gaussian and its second moment.

\medskip
\noindent\textbf{Step 3. Unconditional expectation and telescoping sum.}
Taking full expectation on both sides of \eqref{eq:step2} and rearranging,
\[
\alpha_{k}\Bigl(1-\frac{L\alpha_{k}}{2}\Bigr)\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le \E\!\left[f(x_{k})-f(x_{k+1})\right] + L\alpha_{k}\tau_{k}d .
\tag{8}\label{eq:step3}
\]
Summing \eqref{eq:step3} over $k=0,\dots,K-1$ gives
\begin{align}
\sum_{k=0}^{K-1}\alpha_{k}\Bigl(1-\frac{L\alpha_{k}}{2}\Bigr)
\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le f(x_{0})- \E\!\left[f(x_{K})\right] 
   + L d \sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
   \tag{9}\label{eq:step4}
\end{align}
Since $f(x_{K})\ge f_{\inf}$, we may replace $- \E[f(x_{K})]$ by $-f_{\inf}$.

\medskip
\noindent\textbf{Step 4. Lower bound the coefficient.}
Because of \eqref{3}, there exists $k_{0}$ such that for all $k\ge k_{0}$, $L\alpha_{k}\le 1$. Hence for all $k$,
\[
1-\frac{L\alpha_{k}}{2}\ge \frac12 .
\tag{10}\label{eq:coeff}
\]
Applying \eqref{eq:coeff} to \eqref{eq:step4},
\[
\frac12\sum_{k=0}^{K-1}\alpha_{k}\,
\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le f(x_{0})-f_{\inf}+ L d \sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{11}\label{eq:step5}
\]

\medskip
\noindent\textbf{Step 5. Divide by $\sum_{k=0}^{K-1}\alpha_{k}$.}
\[
\frac{1}{\sum_{k=0}^{K-1}\alpha_{k}}
\sum_{k=0}^{K-1}\alpha_{k}\,
\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\sum_{k=0}^{K-1}\alpha_{k}}
   +\frac{2L d}{\sum_{k=0}^{K-1}\alpha_{k}}
      \sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{12}\label{eq:step6}
\]
Substituting the explicit temperature schedule $\tau_{k}= \tau_{0}(k+1)^{-\beta}$ yields the bound \eqref{4} of Theorem \ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
Assume $\alpha_{k}=c/(k+1)^{a}$ with $a\in(0.5,1]$.
\[
\sum_{k=0}^{K-1}\alpha_{k}
= c\sum_{k=1}^{K}\frac{1}{k^{a}}
\asymp 
\begin{cases}
\displaystyle \frac{c}{1-a}\,K^{1-a}, & a<1,\\[1mm]
c\log K, & a=1 .
\end{cases}
\tag{13}
\]
Similarly,
\[
\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}
= c\tau_{0}\sum_{k=1}^{K}\frac{1}{k^{a+\beta}}
\asymp 
\begin{cases}
\displaystyle \frac{c\tau_{0}}{1-a-\beta}\,K^{1-a-\beta},
   & a+\beta<1,\\[1mm]
c\tau_{0}\log K, & a+\beta=1,\\[1mm]
O(1), & a+\beta>1 .
\end{cases}
\tag{14}
\]
Plugging (13)–(14) into \eqref{eq:step6} gives the explicit rates stated in Theorem \ref{thm:main}.

\section*{Special Cases}
\begin{enumerate}
  \item \textbf{Pure Gradient Descent ($\tau_{0}=0$).} The second term in \eqref{4} vanishes and we recover the classic $O\!\bigl(1/\sum\alpha_{k}\bigr)$ bound.
  \item \textbf{Constant Temperature ($\beta=0$).} The term $\sum\alpha_{k}\tau_{k}$ behaves like $\tau_{0}\sum\alpha_{k}$, leading to a non‑vanishing asymptotic bias of order $O(\tau_{0})$.
  \item \textbf{Fast Decay ($\beta> a$).} The noise contribution becomes $O(1/\sum\alpha_{k})$, i.e. it does not affect the dominant rate; the algorithm behaves similarly to GD asymptotically.
\end{enumerate}
