\section*{Algorithm Name}
\textbf{Trust‑Region Gradient Descent (TR‑GD)}  

The method iteratively builds a quadratic model of a convex, $L$‑smooth objective
$f:\R^{n}\!\to\!\R$ around the current iterate $x_k$ and solves the model
approximately inside a Euclidean ball of radius $\Delta_k$ (the trust‑region
radius).  The step is accepted or rejected according to the ratio of actual
reduction to predicted reduction, and the radius is updated adaptively.

\section*{Mathematical Formulation}
\begin{align}
\text{Given }&x_0\in\R^{n},\;\Delta_0>0,\;\eta_1\in(0,1),\;\eta_2\in(0,1),\;
\gamma_{\text{inc}}>1,\;\gamma_{\text{dec}}\in(0,1). \nonumber\\[4pt]
\text{For }k=0,1,2,\dots&\nonumber\\
\qquad g_k &:= \nabla f(x_k),\qquad 
B_k := L I_n \;\;(\text{Lipschitz constant of }\nabla f). \label{eq:grad}\\[4pt]
\qquad\text{Model: } 
m_k(p) &:= f(x_k)+g_k^{\top}p+\frac12 p^{\top}B_k p,
\qquad p\in\R^{n}. \label{eq:model}\\[4pt]
\qquad\text{Approximate solution of }
\min_{p}\; m_k(p) \;\text{s.t.}\; \norm{p}\le \Delta_k 
\text{ is the Cauchy point} \nonumber\\
\qquad p_k &:= -\alpha_k g_k,\qquad
\alpha_k:=\min\!\Big\{\frac{1}{L},\frac{\Delta_k}{\norm{g_k}}\Big\}. \label{eq:cauchy}\\[4pt]
\qquad\text{Actual reduction: } 
\operatorname{ared}_k := f(x_k)-f(x_k+p_k). \label{eq:ared}\\[4pt]
\qquad\text{Predicted reduction: } 
\operatorname{pred}_k := m_k(0)-m_k(p_k). \label{eq:pred}\\[4pt]
\qquad\text{Ratio: } 
\rho_k := \frac{\operatorname{ared}_k}{\operatorname{pred}_k}. \label{eq:rho}\\[4pt]
\qquad\text{Update iterate: } 
x_{k+1} := 
\begin{cases}
x_k+p_k, & \rho_k\ge \eta_1,\\[2pt]
x_k, & \text{otherwise.}
\end{cases} \label{eq:update}\\[4pt]
\qquad\text{Update radius: } 
\Delta_{k+1} :=
\begin{cases}
\min\{\gamma_{\text{inc}}\Delta_k,\;\Delta_{\max}\}, & \rho_k\ge \eta_2,\\[2pt]
\gamma_{\text{dec}}\Delta_k, & \rho_k<\eta_1,\\[2pt]
\Delta_k, & \text{otherwise.}
\end{cases} \label{eq:radius}
\end{align}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust‑Region Gradient Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0$, $\Delta_0$, $L$, $\eta_1,\eta_2$, $\gamma_{\text{inc}}$, $\gamma_{\text{dec}}$, $\Delta_{\max}$.
\FOR{$k=0,1,2,\dots$}
    \STATE $g_k\gets \nabla f(x_k)$
    \STATE $\alpha_k\gets \min\{1/L,\;\Delta_k/\|g_k\|\}$
    \STATE $p_k\gets -\alpha_k g_k$ \COMMENT{Cauchy point}
    \STATE $\text{ared}_k\gets f(x_k)-f(x_k+p_k)$
    \STATE $\text{pred}_k\gets -g_k^{\top}p_k-\tfrac12 p_k^{\top}B_k p_k$
    \STATE $\rho_k\gets \text{ared}_k/\text{pred}_k$
    \IF{$\rho_k\ge \eta_1$}
        \STATE $x_{k+1}\gets x_k+p_k$
    \ELSE
        \STATE $x_{k+1}\gets x_k$
    \ENDIF
    \IF{$\rho_k\ge \eta_2$}
        \STATE $\Delta_{k+1}\gets \min\{\gamma_{\text{inc}}\Delta_k,\;\Delta_{\max}\}$
    \ELSIF{$\rho_k<\eta_1$}
        \STATE $\Delta_{k+1}\gets \gamma_{\text{dec}}\Delta_k$
    \ELSE
        \STATE $\Delta_{k+1}\gets \Delta_k$
    \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional convex quadratic
\[
f(w)=(w-3)^2 .
\]
Its gradient and Hessian are $g(w)=2(w-3)$ and $L=2$ (since $f''(w)=2$).  
Set $w_0=0$, initial radius $\Delta_0=1$, and parameters
$\eta_1=0.1$, $\eta_2=0.75$, $\gamma_{\text{inc}}=2$, $\gamma_{\text{dec}}=0.5$.
We perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $\alpha_k=\min\{1/L,\Delta_k/|g_k|\}$ & $w_{k+1}$\\ \hline
0 & $0$ & $-6$ & $\min\{0.5,\;1/6\}=1/6$ & $0+(-\alpha_0 g_0)=0+(-\frac16)(-6)=1$\\
1 & $1$ & $-4$ & $\min\{0.5,\;1/4\}=0.25$ & $1+(-0.25)(-4)=2$\\
2 & $2$ & $-2$ & $\min\{0.5,\;1/2\}=0.5$ & $2+(-0.5)(-2)=3$\\
\end{tabular}
\end{center}
After the third iteration $w_3=3$, which is the exact minimiser; the objective
values $f(w_k)$ drop from $9$ to $4$ to $1$ and finally to $0$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
The objective $f:\R^{n}\!\to\!\R$ is convex and $L$‑smooth, i.e.
\[
f(y)\le f(x)+\nabla f(x)^{\top}(y-x)+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{n}.
\]
\end{assumption}
\begin{assumption}[Bounded Level Set]\label{as:bounded}
For the initial point $x_0$ the level set
$\mathcal{L}:=\{x\in\R^{n}\mid f(x)\le f(x_0)\}$ is bounded, and we denote
$D:=\sup_{x,y\in\mathcal{L}}\|x-y\|<\infty$.
\end{assumption}
\begin{assumption}[Algorithmic Parameters]\label{as:param}
The constants satisfy
\[
0<\eta_1<\eta_2<1,\qquad
\gamma_{\text{inc}}>1,\qquad
0<\gamma_{\text{dec}}<1,\qquad
\Delta_{\max}\ge \Delta_0>0 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Let Assumptions \ref{as:convex}–\ref{as:param} hold.  Denote
\[
g_k:=\nabla f(x_k),\qquad 
\Delta_k\text{ the trust‑region radius at iteration }k .
\]
For any $T\ge 1$ the iterates produced by TR‑GD satisfy
\[
\min_{0\le k\le T}\|g_k\|^{2}\;\le\;
\frac{2L\bigl(f(x_0)-f^{\star}\bigr)}{\eta_1\,\min\{1/L,\Delta_0/D\}}\;\frac{1}{T+1},
\tag{T1}
\]
where $f^{\star}:=\inf_{x}f(x)$.  Consequently,
$\|g_k\|\to0$ and $f(x_k)\to f^{\star}$ as $k\to\infty$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because the model uses the exact Lipschitz constant
$L$, the predicted reduction $\operatorname{pred}_k$ is always a lower bound on
the actual reduction when the step is accepted; this prevents divergence.
\item \textbf{Hyper‑parameter sensitivity.}
The convergence bound (T1) depends linearly on $\eta_1^{-1}$ and on the
minimum of $1/L$ and $\Delta_0/D$.  Larger $\eta_1$ (more stringent acceptance)
or a larger initial radius $\Delta_0$ improve the constant but do not affect the
$1/T$ rate.
\item \textbf{Comparison to plain Gradient Descent.}
For convex $L$‑smooth problems plain GD with fixed step $1/L$ yields
$f(x_T)-f^{\star}=O(1/T)$.  TR‑GD attains the same $O(1/T)$ rate while
automatically adapting the step length via $\Delta_k$, which can be advantageous
when $L$ is unknown or when the objective is ill‑conditioned.
\end{itemize}

\section*{Discussion}
The theorem shows that, under the standard convex‑smooth hypothesis, the
trust‑region scheme enjoys a worst‑case $O(1/T)$ decrease of the squared
gradient norm, identical to the optimal rate for first‑order methods on this
class.  The proof follows the classical trust‑region analysis but is spelled
out in full detail below, with every algebraic manipulation labelled.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Predicted reduction of the Cauchy point]\label{def:pred}
For $p_k$ defined in \eqref{eq:cauchy},
\[
\operatorname{pred}_k
= -g_k^{\top}p_k -\frac12 p_k^{\top}B_k p_k
= \alpha_k\|g_k\|^{2}-\frac{L}{2}\alpha_k^{2}\|g_k\|^{2}
= \frac{\alpha_k}{2}\|g_k\|^{2}\bigl(2-L\alpha_k\bigr).
\]
\end{definition}
\begin{lemma}[Lower bound on $\operatorname{pred}_k$]\label{lem:predlb}
For any $k$,
\[
\operatorname{pred}_k\;\ge\;
\frac12\min\!\Big\{\frac{1}{L},\frac{\Delta_k}{\|g_k\|}\Big\}\,\|g_k\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
From Definition \ref{def:pred} and the fact that $0\le L\alpha_k\le1$,
\[
\operatorname{pred}_k
= \frac{\alpha_k}{2}\|g_k\|^{2}(2-L\alpha_k)
\ge \frac{\alpha_k}{2}\|g_k\|^{2}.
\]
Because $\alpha_k=\min\{1/L,\Delta_k/\|g_k\| \}$, the claim follows. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  Throughout the proof all expectations are
deterministic because the algorithm is fully deterministic under the given
assumptions.

\noindent\textbf{Notation.}  $f^{\star}:=\inf_{x}f(x)$, $g_k:=\nabla f(x_k)$,
$\Delta_k$ as in \eqref{eq:radius}, and $\rho_k$ as in \eqref{eq:rho}.

\bigskip
\noindent\textbf{Step 1.  Relating actual and predicted reduction.}\\
\noindent By definition of $\rho_k$,
\begin{align}
\operatorname{ared}_k
= \rho_k\,\operatorname{pred}_k .
\tag{S1}
\end{align}
If the step is accepted ($\rho_k\ge\eta_1$) we have
\[
f(x_{k+1}) = f(x_k)-\operatorname{ared}_k
= f(x_k)-\rho_k\operatorname{pred}_k
\le f(x_k)-\eta_1\operatorname{pred}_k .
\tag{S2}
\]

\noindent\textbf{Step 2.  Lower bound on predicted reduction using Lemma \ref{lem:predlb}.}\\
\begin{align}
\operatorname{pred}_k
\ge \frac12\min\!\Big\{\frac{1}{L},\frac{\Delta_k}{\|g_k\|}\Big\}\,\|g_k\|^{2}
\ge \frac{1}{2L}\,\|g_k\|^{2}\;\min\Big\{1,\frac{L\Delta_k}{\|g_k\|}\Big\}.
\tag{S3}
\end{align}
Since the level set $\mathcal{L}$ is bounded by $D$ (Assumption \ref{as:bounded}),
$\|g_k\|\le L D$ (by $L$‑smoothness).  Hence $L\Delta_k/\|g_k\|\ge
\Delta_k/D$.  Consequently,
\[
\operatorname{pred}_k\;\ge\;\frac{1}{2L}\,\|g_k\|^{2}\,\min\!\Big\{1,\frac{\Delta_k}{D}\Big\}.
\tag{S4}
\]

\noindent\textbf{Step 3.  Descent inequality for accepted steps.}\\
Combining \eqref{S2} and \eqref{S4},
\[
f(x_{k+1})\le f(x_k)-\frac{\eta_1}{2L}\,\|g_k\|^{2}\,
\min\!\Big\{1,\frac{\Delta_k}{D}\Big\}.
\tag{S5}
\]

\noindent\textbf{Step 4.  Bounding the radius from below.}\\
If a step is rejected ($\rho_k<\eta_1$) the radius is multiplied by
$\gamma_{\text{dec}}<1$; otherwise it is either kept or multiplied by
$\gamma_{\text{inc}}>1$ only when $\rho_k\ge\eta_2$.  Since $\eta_2>\eta_1$, a
rejection never leads to an increase.  Therefore,
\[
\Delta_{k+1}\ge \gamma_{\text{dec}}\Delta_k\quad\text{whenever }\rho_k<\eta_1.
\tag{S6}
\]
If the step is accepted, $\Delta_{k+1}\ge\Delta_k$ (case $\rho_k\ge\eta_2$) or
$\Delta_{k+1}=\Delta_k$ (case $\eta_1\le\rho_k<\eta_2$).  Hence
\[
\Delta_{k+1}\ge \min\{\gamma_{\text{dec}},1\}\,\Delta_k
= \gamma_{\text{dec}}\Delta_k .
\tag{S7}
\]
Iterating \eqref{S7} yields for all $k$,
\[
\Delta_k\ge \gamma_{\text{dec}}^{k}\Delta_0 .
\tag{S8}
\]
Because $\gamma_{\text{dec}}^{k}\ge \gamma_{\text{dec}}^{T}$ for $k\le T$,
\[
\min_{0\le k\le T}\Delta_k\;\ge\;
\gamma_{\text{dec}}^{T}\Delta_0.
\tag{S9}
\]

\noindent\textbf{Step 5.  Summation of descent inequalities.}\\
Let $\mathcal{A}:=\{k\mid\rho_k\ge\eta_1\}$ be the set of accepted indices.
Summing \eqref{S5} over $k\in\mathcal{A}\cap\{0,\dots,T\}$ gives
\[
\sum_{k\in\mathcal{A}\cap\{0,\dots,T\}}
\|g_k\|^{2}\,
\min\!\Big\{1,\frac{\Delta_k}{D}\Big\}
\;\le\;
\frac{2L}{\eta_1}\bigl(f(x_0)-f^{\star}\bigr).
\tag{S10}
\]

\noindent\textbf{Step 6.  Using the lower bound on $\Delta_k$.}\\
From \eqref{S9}, for every $k\le T$,
\[
\min\!\Big\{1,\frac{\Delta_k}{D}\Big\}
\;\ge\;
\min\!\Big\{1,\frac{\gamma_{\text{dec}}^{T}\Delta_0}{D}\Big\}
\;=\;
c_T,
\]
where we denote
\[
c_T:=\min\Big\{1,\frac{\gamma_{\text{dec}}^{T}\Delta_0}{D}\Big\}.
\tag{S11}
\]
Insert $c_T$ into \eqref{S10}:
\[
c_T\sum_{k\in\mathcal{A}\cap\{0,\dots,T\}}\|g_k\|^{2}
\;\le\;
\frac{2L}{\eta_1}\bigl(f(x_0)-f^{\star}\bigr).
\tag{S12}
\]

\noindent\textbf{Step 7.  Relating the minimum gradient norm to the sum.}\\
Since the sum of non‑negative numbers bounds the product of the number of
terms and their minimum,
\[
\sum_{k\in\mathcal{A}\cap\{0,\dots,T\}}\|g_k\|^{2}
\;\ge\;
|\mathcal{A}\cap\{0,\dots,T\}|\;
\min_{0\le k\le T}\|g_k\|^{2}.
\tag{S13}
\]
Moreover $|\mathcal{A}\cap\{0,\dots,T\}|\ge 1$ because at least the first
iteration is accepted (the Cauchy point always yields a reduction for a convex
quadratic).  Combining \eqref{S12}–\eqref{S13} we obtain
\[
\min_{0\le k\le T}\|g_k\|^{2}
\;\le\;
\frac{2L}{\eta_1\,c_T}\,
\frac{f(x_0)-f^{\star}}{|\mathcal{A}\cap\{0,\dots,T\}|}
\;\le\;
\frac{2L}{\eta_1\,c_T}\,
\frac{f(x_0)-f^{\star}}{T+1}.
\tag{S14}
\]

\noindent\textbf{Step 8.  Simplifying $c_T$.}\\
Since $c_T\ge\min\{1,\Delta_0/D\}$ (the factor $\gamma_{\text{dec}}^{T}\le1$),
\[
c_T\ge \min\Big\{1,\frac{\Delta_0}{D}\Big\}.
\tag{S15}
\]
Insert \eqref{S15} into \eqref{S14} to obtain the bound stated in
Theorem \ref{thm:rate}:
\[
\boxed{
\displaystyle
\min_{0\le k\le T}\|g_k\|^{2}
\;\le\;
\frac{2L\bigl(f(x_0)-f^{\star}\bigr)}
{\eta_1\,\min\{1,\Delta_0/D\}}\;
\frac{1}{T+1}
}
\tag{S16}
\]
which is exactly inequality (T1).  Because the right‑hand side tends to zero,
$\|g_k\|\to0$ and, by convexity and $L$‑smoothness, $f(x_k)\to f^{\star}$.
∎

\section*{Rate Derivation (With Constants)}
The constant appearing in \eqref{S16} can be rewritten as
\[
C:=\frac{2L\bigl(f(x_0)-f^{\star}\bigr)}
{\eta_1\,\min\{1,\Delta_0/D\}}.
\]
Thus the algorithm enjoys the explicit $O(1/T)$ rate
\[
\min_{0\le k\le T}\| \nabla f(x_k)\| \;\le\; \sqrt{C/(T+1)} .
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact line search.}  If the subproblem is solved exactly,
$p_k = -\frac{1}{L}g_k$, then $\alpha_k=1/L$ and $\Delta_k$ can be kept
constant.  The bound simplifies to $C = \frac{2L\bigl(f(x_0)-f^{\star}\bigr)}{\eta_1}$.
\item \textbf{Large initial radius.}  When $\Delta_0\ge D$, the factor
$\min\{1,\Delta_0/D\}=1$ and the constant does not depend on $\Delta_0$.
\item \textbf{Strongly convex $f$.}  If $f$ is $\mu$‑strongly convex,
then $\|g_k\|\ge\mu\|x_k-x^{\star}\|$ and the $O(1/T)$ bound on the gradient
implies an $O(1/T)$ bound on the distance to the optimum.
\end{itemize}
