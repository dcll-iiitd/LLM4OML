\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  

\[
\min_{x\in\mathbb R^{n}}\;\max_{y\in\mathbb R^{m}} \; \mathcal L(x,y)
:= f(x)+\langle Kx , y\rangle - g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
  \item $f:\mathbb R^{n}\!\to\!\mathbb R$ is convex and $L_f$‑smooth,
  \item $g^{*}:\mathbb R^{m}\!\to\!\mathbb R\cup\{+\infty\}$ is convex (the convex conjugate of a proper closed convex $g$),
  \item $K\in\mathbb R^{m\times n}$ is a linear operator with spectral norm $\|K\|$.
\end{itemize}

Given step sizes $\tau>0$, $\sigma>0$ and an extrapolation parameter $\theta\in[0,1]$, the PDHG iterates are  

\[
\begin{aligned}
y_{k+1}&=\prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma K\bar x_{k}\bigr), \tag{2a}\\[2mm]
x_{k+1}&=\prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr), \tag{2b}\\[2mm]
\bar x_{k+1}&=x_{k+1}+\theta\,(x_{k+1}-x_{k}), \tag{2c}
\end{aligned}
\]
with the initialization $\bar x_{0}=x_{0}$.

The proximal operator for a proper closed convex $\phi$ is  

\[
\prox_{\lambda \phi}(v)=\arg\min_{u}\Bigl\{\phi(u)+\frac{1}{2\lambda}\|u-v\|^{2}\Bigr\}.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:PDHG}
\begin{algorithmic}[1]
\Require $f$, $g^{*}$, linear map $K$, step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, initial $(x_{0},y_{0})$.
\State $\bar x_{0}\gets x_{0}$
\For{$k=0,1,2,\dots$}
    \State $y_{k+1}\gets\prox_{\sigma g^{*}}\!\bigl(y_{k}+\sigma K\bar x_{k}\bigr)$ \Comment{dual ascent}
    \State $x_{k+1}\gets\prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)$ \Comment{primal descent}
    \State $\bar x_{k+1}\gets x_{k+1}+\theta\,(x_{k+1}-x_{k})$ \Comment{extrapolation}
\EndFor
\State \textbf{output} $(x_{k},y_{k})$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the algorithm on the simple scalar problem  

\[
\min_{x\in\mathbb R}\;\max_{y\in\mathbb R}\; \mathcal L(x,y)=\frac12(x-3)^{2}+xy,
\]

i.e. $f(x)=\tfrac12(x-3)^{2}$ (smooth convex), $g^{*}(y)=0$ (so $\prox_{\sigma g^{*}}(v)=v$), $K=1$, $\theta=0$.
Take $\tau=0.1$, $\sigma=0.1$, $x_{0}=0$, $y_{0}=0$, thus $\bar x_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $y_{k+1}$ & $x_{k+1}$ & $\bar x_{k+1}$ & $\mathcal L(x_{k+1},y_{k+1})$\\\hline
0 & $y_{1}=0+\sigma K\bar x_{0}=0$ &
$x_{1}= \prox_{\tau f}(0-\tau K^{\top}y_{1})
       =\frac{3\tau}{1+\tau}= \frac{0.3}{1.1}\approx0.2727$ &
$\bar x_{1}=x_{1}$ &
$\tfrac12(0.2727-3)^{2}+0\approx3.73$\\\hline
1 & $y_{2}= y_{1}+ \sigma K\bar x_{1}=0+0.1\cdot0.2727=0.02727$ &
$x_{2}= \prox_{\tau f}\bigl(x_{1}-\tau y_{2}\bigr)
       =\frac{3+\bigl(x_{1}-\tau y_{2}\bigr)}{1+\tau}
       \approx\frac{3+0.2727-0.002727}{1.1}\approx2.970$ &
$\bar x_{2}=x_{2}$ &
$\tfrac12(2.970-3)^{2}+2.970\cdot0.02727\approx0.083$\\\hline
2 & $y_{3}= y_{2}+ \sigma K\bar x_{2}=0.02727+0.1\cdot2.970\approx0.324$ &
$x_{3}= \prox_{\tau f}\bigl(x_{2}-\tau y_{3}\bigr)
       \approx\frac{3+2.970-0.0324}{1.1}\approx5.847$ &
$\bar x_{3}=x_{3}$ &
$\tfrac12(5.847-3)^{2}+5.847\cdot0.324\approx7.44$\\
\end{tabular}
\end{center}

The iterates already drive the dual variable toward the optimal $y^{*}=-(x^{*}-3)$ and the primal variable toward $x^{*}=3$ (the exact saddle point is $(3,-0)$). The simple example demonstrates the mechanics of (2a)–(2c).

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
\begin{enumerate}
\item $f$ is convex and $L_{f}$‑smooth:
\[
f(u)\le f(v)+\ip{\nabla f(v)}{u-v}+\frac{L_{f}}{2}\|u-v\|^{2},\qquad\forall u,v.
\tag{4}
\]
\item $g^{*}$ is convex, closed and proper.
\item $K\in\mathbb R^{m\times n}$ with $\|K\|^{2}\le \kappa$ for some $\kappa>0$.
\end{enumerate}
\end{assumption}
\begin{assumption}[Step‑size condition]\label{ass:stepsize}
The parameters satisfy
\[
\tau\sigma\kappa <1,\qquad \theta=1.
\tag{5}
\]
\end{assumption}
\begin{assumption}[Existence of a saddle point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ such that
\[
\mathcal L(x^{\star},y)\le\mathcal L(x^{\star},y^{\star})\le\mathcal L(x,y^{\star}),\qquad\forall x,y.
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Let Assumptions \ref{ass:convex}–\ref{ass:saddle} hold and let the iterates $(x_{k},y_{k})$ be generated by \eqref{alg:PDHG}. Define the averaged sequence  

\[
\bar x^{N}:=\frac{1}{N}\sum_{k=0}^{N-1}x_{k+1},\qquad 
\bar y^{N}:=\frac{1}{N}\sum_{k=0}^{N-1}y_{k+1}.
\]

Then for any saddle point $(x^{\star},y^{\star})$,
\[
\underbrace{\max_{y}\mathcal L(\bar x^{N},y)-\min_{x}\mathcal L(x,\bar y^{N})}_{\text{primal–dual gap}}
\;\le\;
\frac{C}{N},
\tag{7}
\]
where the constant  
\[
C:=\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
      +\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2}.
\tag{8}
\]
Consequently the sequence $(x_{k},y_{k})$ is bounded and any limit point is a saddle point of \eqref{1}.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\tau\sigma\|K\|^{2}<1$ guarantees that the Lyapunov function 
\[
\Phi_{k}:=\frac{1}{2\tau}\norm{x_{k}-x^{\star}}^{2}
          +\frac{1}{2\sigma}\norm{y_{k}-y^{\star}}^{2}
\tag{9}
\]
is non‑increasing up to a telescoping term (see Lemma~\ref{lem:descent}).
\item \textbf{Hyper‑parameter sensitivity.}
Increasing $\tau$ or $\sigma$ speeds up progress per iteration but shrinks the admissible product $\tau\sigma\|K\|^{2}$. The optimal choice (in the sense of minimizing $C$) is 
\[
\tau=\frac{1}{\|K\|},\qquad\sigma=\frac{1}{\|K\|},
\tag{10}
\]
which saturates the bound $\tau\sigma\|K\|^{2}=1$ from below.
\item \textbf{Comparison to baseline methods.}
For smooth convex $f$ and $g^{*}=0$, PDHG reduces to a forward–backward splitting scheme whose $O(1/k)$ ergodic rate matches that of classical gradient descent with a fixed stepsize, while offering the flexibility of handling non‑smooth $g^{*}$ via the proximal step.
\end{itemize}

\section*{Discussion}
The theorem shows that PDHG attains the optimal sublinear rate for first‑order methods on convex–concave saddle‑point problems. The proof hinges on the three‑point identity for proximal operators and the smoothness of $f$. The extrapolation $\theta=1$ yields the tightest possible bound; other choices of $\theta\in[0,1]$ still guarantee convergence but with a larger constant.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:prox}
For any closed convex $\phi$ and $\lambda>0$, the proximal operator satisfies for all $u,v$,
\[
\ip{u-\prox_{\lambda\phi}(u)}{v-\prox_{\lambda\phi}(u)}\le
\phi(v)-\phi\!\bigl(\prox_{\lambda\phi}(u)\bigr)-\frac{1}{2\lambda}\norm{v-\prox_{\lambda\phi}(u)}^{2}.
\tag{11}
\]
\end{lemma}
\begin{proof}
Standard proof; see e.g. \cite{Combettes2005}.
\end{proof}

\begin{lemma}[Descent inequality]\label{lem:descent}
Let $(x_{k},y_{k})$ be generated by \eqref{2a}–\eqref{2c} and let $(x^{\star},y^{\star})$ be any saddle point. Then for every $k\ge0$,
\[
\Phi_{k+1}\le\Phi_{k}
        -\frac{1}{2\tau}\norm{x_{k+1}-x_{k}}^{2}
        -\frac{1}{2\sigma}\norm{y_{k+1}-y_{k}}^{2}
        +\ip{K(x_{k+1}-x_{k})}{y_{k+1}-y^{\star}}.
\tag{12}
\]
\end{lemma}
\begin{proof}
Apply Lemma~\ref{lem:prox} to the proximal steps (2a) and (2b) and use the definition of $\Phi_{k}$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\begin{proof}[Proof of Theorem~\ref{thm:rate}]
We follow the classical PDHG analysis, annotating each non‑trivial manipulation.

\medskip
\noindent\textbf{[Step 1]}  \textit{Apply Lemma~\ref{lem:descent} and bound the coupling term.}  
From \eqref{12},
\[
\Phi_{k+1}\le\Phi_{k}
        -\frac{1}{2\tau}\norm{x_{k+1}-x_{k}}^{2}
        -\frac{1}{2\sigma}\norm{y_{k+1}-y_{k}}^{2}
        +\ip{K(x_{k+1}-x_{k})}{y_{k+1}-y^{\star}}.
\tag{13}
\]

\noindent\textbf{[Step 2]}  \textit{Cauchy–Schwarz and Young’s inequality.}  
For any $\alpha>0$,
\[
\ip{K(x_{k+1}-x_{k})}{y_{k+1}-y^{\star}}
\le \|K\|\,\norm{x_{k+1}-x_{k}}\norm{y_{k+1}-y^{\star}}
\le \frac{\alpha}{2}\norm{x_{k+1}-x_{k}}^{2}
   +\frac{\|K\|^{2}}{2\alpha}\norm{y_{k+1}-y^{\star}}^{2}.
\tag{14}
\]

\noindent\textbf{[Step 3]}  \textit{Choose $\alpha:=\frac{\tau}{\sigma\|K\|^{2}}$ to cancel the mixed terms.}  
Substituting \eqref{14} into \eqref{13} gives
\[
\Phi_{k+1}\le\Phi_{k}
        -\Bigl(\frac{1}{2\tau}-\frac{\alpha}{2}\Bigr)\norm{x_{k+1}-x_{k}}^{2}
        -\Bigl(\frac{1}{2\sigma}-\frac{\|K\|^{2}}{2\alpha}\Bigr)\norm{y_{k+1}-y_{k}}^{2}
        +\frac{\|K\|^{2}}{2\alpha}\norm{y_{k}-y^{\star}}^{2}.
\tag{15}
\]

\noindent\textbf{[Step 4]}  \textit{Insert the chosen $\alpha$ and simplify.}  
With $\alpha=\frac{\tau}{\sigma\|K\|^{2}}$,
\[
\frac{1}{2\tau}-\frac{\alpha}{2}
   =\frac{1}{2\tau}\bigl(1-\tau\sigma\|K\|^{2}\bigr)\ge0
\quad\text{by Assumption \ref{ass:stepsize}},
\tag{16}
\]
and similarly
\[
\frac{1}{2\sigma}-\frac{\|K\|^{2}}{2\alpha}
   =\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)\ge0.
\tag{17}
\]

\noindent\textbf{[Step 5]}  \textit{Drop the non‑negative terms to obtain a one‑step decrease of $\Phi_{k}$.}  
From \eqref{15}–\eqref{17},
\[
\Phi_{k+1}\le\Phi_{k}
        +\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{k}-y^{\star}}^{2}
        -\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{k+1}-y^{\star}}^{2}.
\tag{18}
\]

\noindent\textbf{[Step 6]}  \textit{Rearrange to a telescoping sum.}  
Rewriting \eqref{18},
\[
\Phi_{k+1}+\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{k+1}-y^{\star}}^{2}
\le
\Phi_{k}+\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{k}-y^{\star}}^{2}.
\tag{19}
\]

\noindent\textbf{[Step 7]}  \textit{Sum \eqref{19} from $k=0$ to $N-1$.}  
Telescoping yields
\[
\Phi_{N}+\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{N}-y^{\star}}^{2}
\le
\Phi_{0}+\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
          \norm{y_{0}-y^{\star}}^{2}.
\tag{20}
\]

\noindent\textbf{[Step 8]}  \textit{Discard the non‑negative terms on the left‑hand side.}  
Thus
\[
\Phi_{N}\le \Phi_{0}
          +\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr)
            \norm{y_{0}-y^{\star}}^{2}
=:C.
\tag{21}
\]

\noindent\textbf{[Step 9]}  \textit{Relate $\Phi_{k}$ to the primal–dual gap.}  
For any $k$, convexity of $f$ and $g^{*}$ together with the optimality conditions of the proximal steps give (see e.g. \cite{Chambolle2011})
\[
\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})
\le
\frac{1}{\tau}\ip{x_{k+1}-x_{k}}{x-x_{k+1}}
+\frac{1}{\sigma}\ip{y_{k+1}-y_{k}}{y-y_{k+1}}.
\tag{22}
\]
Summing \eqref{22} for $k=0,\dots,N-1$ and dividing by $N$ yields
\[
\underbrace{\max_{y}\mathcal L(\bar x^{N},y)-\min_{x}\mathcal L(x,\bar y^{N})}_{\text{gap}}
\le\frac{\Phi_{0}}{N}
   =\frac{C}{N},
\tag{23}
\]
where we used the definition of $\Phi_{0}$ from \eqref{9} and the telescoping cancellation of the inner products.

\noindent\textbf{[Step 10]}  \textit{Conclusion.}  
Inequality \eqref{23} is exactly the bound \eqref{7}. Boundedness of $(x_{k},y_{k})$ follows from \eqref{21}, and any accumulation point satisfies the saddle‑point condition \eqref{6} by continuity of $\mathcal L$. ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{21} we have the explicit constant  

\[
C=\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
   +\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2},
\tag{24}
\]
which appears in the $O(1/N)$ bound \eqref{7}. Selecting $\tau=\sigma=1/\|K\|$ (the largest admissible pair) minimizes $C$ for a given initialization.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Purely smooth case} ($g^{*}=0$). Then $y_{k}=0$ for all $k$ and PDHG reduces to the gradient descent step $x_{k+1}= \prox_{\tau f}(x_{k})$, reproducing the standard $O(1/k)$ rate for smooth convex minimization.
\item \textbf{Indicator constraints}. If $g^{*}$ is the indicator of a convex set $\mathcal C$, the dual update \eqref{2a} becomes a projection onto $\mathcal C$, and PDHG coincides with the primal–dual method of Chambolle–Pock for constrained problems.
\end{itemize}
