\section*{Algorithm Name}
\textbf{FedProx Local SGD} (proximal federated averaging)

\section*{Mathematical Formulation}
\noindent
\begin{itemize}
    \item $K$ participating clients, indexed by $k\in\{1,\dots,K\}$.
    \item Global model at communication round $t$: $x^{t}\in\mathbb{R}^{d}$.
    \item Local objective of client $k$:
    \[
        F_k(x)\;=\;f_k(x)+\frac{\mu}{2}\|x-x^{t}\|^{2},
    \]
    where $f_k$ is the empirical loss on client $k$ and $\mu\ge 0$ is the proximal parameter.
    \item Each client performs $E$ stochastic gradient steps with stepsize $\eta>0$:
    \[
        x_{k}^{t,0}=x^{t},\qquad
        x_{k}^{t,s+1}=x_{k}^{t,s}-\eta\,g_{k}^{t,s},
        \quad s=0,\dots,E-1,
    \]
    where $g_{k}^{t,s}$ is an unbiased stochastic gradient of $F_k$ at $x_{k}^{t,s}$, i.e.
    $\mathbb{E}[g_{k}^{t,s}\mid x_{k}^{t,s}]=\nabla f_k(x_{k}^{t,s})+\mu (x_{k}^{t,s}-x^{t})$.
    \item After $E$ local steps client $k$ obtains $x_{k}^{t+1}:=x_{k}^{t,E}$ and sends it to the server.
    \item Server aggregates by simple averaging:
    \[
        x^{t+1}= \frac{1}{K}\sum_{k=1}^{K}x_{k}^{t+1}.
    \]
\end{itemize}
Hyper‑parameters: stepsize $\eta$, number of local steps $E$, proximal weight $\mu$, total communication rounds $T$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx Local SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, local steps $E$, proximal weight $\mu$, rounds $T$.
\STATE Initialize $x^{0}\in\mathbb{R}^{d}$ (e.g. $x^{0}=0$).
\FOR{$t=0$ \TO $T-1$}
    \FOR{each client $k=1,\dots,K$ \textbf{in parallel}}
        \STATE $x_{k}^{t,0}\leftarrow x^{t}$
        \FOR{$s=0$ \TO $E-1$}
            \STATE Sample minibatch $\mathcal{B}_{k}^{t,s}$ from local data.
            \STATE Compute stochastic gradient
            \[
                g_{k}^{t,s}\gets \nabla f_k(x_{k}^{t,s};\mathcal{B}_{k}^{t,s})+\mu\bigl(x_{k}^{t,s}-x^{t}\bigr).
            \]
            \STATE $x_{k}^{t,s+1}\leftarrow x_{k}^{t,s}-\eta\, g_{k}^{t,s}$.
        \ENDFOR
        \STATE Set $x_{k}^{t+1}\leftarrow x_{k}^{t,E}$.
    \ENDFOR
    \STATE Server aggregates: $x^{t+1}\leftarrow \frac{1}{K}\sum_{k=1}^{K}x_{k}^{t+1}$.
\ENDFOR
\STATE \textbf{Output:} $x^{T}$ (or averaged iterate $\bar{x}_T=\frac{1}{T}\sum_{t=0}^{T-1}x^{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single client ($K=1$) and a deterministic setting (no minibatch noise).  
Local loss: $f(w)=(w-3)^{2}$, proximal weight $\mu=0$, stepsize $\eta=0.1$, $E=1$, initial model $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $t$ & $w_{t}$ & Gradient $g_{t}=2(w_{t}-3)$ & Updated $w_{t+1}=w_{t}-\eta g_{t}$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.544$. The algorithm monotonically decreases the loss, illustrating the proximal‑free case of FedProx.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
For each client $k$, the local loss $f_k:\mathbb{R}^{d}\!\to\!\mathbb{R}$ is convex and $L$‑smooth:
\[
    f_k(y)\le f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},
    \qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients]\label{ass:unbiased}
For any $t,s,k$,
\[
    \mathbb{E}\!\bigl[g_{k}^{t,s}\mid x_{k}^{t,s}\bigr]
        =\nabla f_k\!\bigl(x_{k}^{t,s}\bigr)+\mu\bigl(x_{k}^{t,s}-x^{t}\bigr).
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^{2}>0$ such that for all $t,s,k$,
\[
    \mathbb{E}\!\bigl[\|g_{k}^{t,s}-\mathbb{E}[g_{k}^{t,s}\mid x_{k}^{t,s}]\|^{2}\mid x_{k}^{t,s}\bigr]\le\sigma^{2}.
\]
\end{assumption}

\begin{assumption}[Bounded Gradient Norm]\label{ass:boundgrad}
There exists $G>0$ with $\|\nabla f_k(x)\|\le G$ for all $k$ and $x$.
\end{assumption}

\begin{assumption}[Client Heterogeneity]\label{ass:hetero}
Define $F(x)=\frac{1}{K}\sum_{k=1}^{K}f_k(x)$. Assume there exists $\zeta\ge0$ such that
\[
    \frac{1}{K}\sum_{k=1}^{K}\|\nabla f_k(x)-\nabla F(x)\|^{2}\le\zeta^{2},\qquad\forall x.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:conv}
Under Assumptions~\ref{ass:convex}–\ref{ass:hetero}, let the stepsize satisfy $0<\eta\le\frac{1}{L+\mu}$. After $T$ communication rounds the averaged iterate
\[
    \bar{x}_{T}\;:=\;\frac{1}{T}\sum_{t=0}^{T-1}x^{t}
\]
obeys
\[
    \mathbb{E}\!\bigl[F(\bar{x}_{T})-F(x^{\star})\bigr]
    \;\le\;
    \frac{\|x^{0}-x^{\star}\|^{2}}{2\eta T}
    \;+\;
    \frac{\eta L\sigma^{2}}{K}
    \;+\;
    \frac{L\eta^{2}E\zeta^{2}}{2},
\]
where $x^{\star}$ is a global minimizer of $F$.
Consequently, choosing a diminishing stepsize $\eta=\Theta\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$ yields
\[
    \mathbb{E}\!\bigl[F(\bar{x}_{T})-F(x^{\star})\bigr]=
    O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability w.r.t. $\mu$.} Increasing $\mu$ reduces client drift, shrinking the heterogeneity term $L\eta^{2}E\zeta^{2}$ but also effectively lowers the admissible stepsize $\eta\le1/(L+\mu)$.
    \item \textbf{Effect of Local Steps $E$.} Larger $E$ accelerates local progress but magnifies the third term $L\eta^{2}E\zeta^{2}$, reflecting increased client drift.
    \item \textbf{Comparison to FedAvg.} Setting $\mu=0$ recovers the classical FedAvg bound; the proximal term adds the factor $1/(L+\mu)$ in the stepsize constraint and an extra regularization that can improve performance on highly heterogeneous data.
\end{itemize}

\section*{Discussion}
The bound in Theorem~\ref{thm:conv} decomposes into three contributions:
\begin{enumerate}
    \item Optimization error decreasing as $1/(\eta T)$.
    \item Stochastic noise scaling as $\eta\sigma^{2}/K$.
    \item Client heterogeneity scaling as $L\eta^{2}E\zeta^{2}$.
\end{enumerate}
Balancing $\eta$ yields the optimal $O(1/\sqrt{T})$ rate, matching the best known rates for convex stochastic optimization under partial participation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f_k$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^{d}$,
\[
    f_k(y)\le f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill$\square$
\end{proof}

\begin{lemma}[Three‑Point Identity for Proximal Term]\label{lem:prox}
For any $x,y\in\mathbb{R}^{d}$ and $\mu\ge0$,
\[
    \frac{\mu}{2}\|y-x\|^{2}
    =\frac{\mu}{2}\|y-x^{t}\|^{2}
     -\frac{\mu}{2}\|x-x^{t}\|^{2}
     +\mu\langle y-x^{t},x^{t}-x\rangle .
\]
\end{lemma}
\begin{proof}
Expand both sides using $\|a-b\|^{2}=\|a\|^{2}+\|b\|^{2}-2\langle a,b\rangle$. \hfill$\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. Let $x^{\star}$ denote a minimizer of $F$. Define the global model after round $t$ as $x^{t}$ and the local model of client $k$ after $s$ local steps as $x_{k}^{t,s}$.

\subsection*{Step 1: One‑step descent for a single client}
From the update $x_{k}^{t,s+1}=x_{k}^{t,s}-\eta g_{k}^{t,s}$ we have
\begin{align}
    \|x_{k}^{t,s+1}-x^{\star}\|^{2}
    &=\|x_{k}^{t,s}-x^{\star}\|^{2}
      -2\eta\langle g_{k}^{t,s},x_{k}^{t,s}-x^{\star}\rangle
      +\eta^{2}\|g_{k}^{t,s}\|^{2}. \tag{1}
\end{align}
\textbf{[Step 1]}

Taking conditional expectation w.r.t.\ the stochastic minibatch and using Assumption~\ref{ass:unbiased} yields
\begin{align}
    \mathbb{E}\!\bigl[\|x_{k}^{t,s+1}-x^{\star}\|^{2}\mid x_{k}^{t,s}\bigr]
    &=\|x_{k}^{t,s}-x^{\star}\|^{2}
      -2\eta\bigl\langle\nabla f_k(x_{k}^{t,s})+\mu(x_{k}^{t,s}-x^{t}),\,x_{k}^{t,s}-x^{\star}\bigr\rangle \nonumber\\
    &\qquad +\eta^{2}\mathbb{E}\!\bigl[\|g_{k}^{t,s}\|^{2}\mid x_{k}^{t,s}\bigr]. \tag{2}
\end{align}
\textbf{[Step 2]}

\subsection*{Step 2: Bounding the stochastic term}
By variance decomposition,
\[
    \mathbb{E}\!\bigl[\|g_{k}^{t,s}\|^{2}\mid x_{k}^{t,s}\bigr]
    =\|\nabla f_k(x_{k}^{t,s})+\mu(x_{k}^{t,s}-x^{t})\|^{2}
     +\mathbb{E}\!\bigl[\|g_{k}^{t,s}-\mathbb{E}[g_{k}^{t,s}\mid x_{k}^{t,s}]\|^{2}\mid x_{k}^{t,s}\bigr].
\]
Using Assumption~\ref{ass:variance} we obtain
\begin{align}
    \mathbb{E}\!\bigl[\|g_{k}^{t,s}\|^{2}\mid x_{k}^{t,s}\bigr]
    &\le \|\nabla f_k(x_{k}^{t,s})+\mu(x_{k}^{t,s}-x^{t})\|^{2}+\sigma^{2}. \tag{3}
\end{align}
\textbf{[Step 3]}

\subsection*{Step 3: Apply smoothness to the inner product}
From Lemma~\ref{lem:smooth} and convexity,
\[
    f_k(x^{\star})\ge f_k(x_{k}^{t,s})+\langle\nabla f_k(x_{k}^{t,s}),x^{\star}-x_{k}^{t,s}\rangle .
\]
Rearranging,
\[
    \langle\nabla f_k(x_{k}^{t,s}),x_{k}^{t,s}-x^{\star}\rangle
    \ge f_k(x_{k}^{t,s})-f_k(x^{\star}). \tag{4}
\]
\textbf{[Step 4]}

\subsection*{Step 4: Combine (2)–(4)}
Insert (3) and (4) into (2):
\begin{align}
    &\mathbb{E}\!\bigl[\|x_{k}^{t,s+1}-x^{\star}\|^{2}\mid x_{k}^{t,s}\bigr]\nonumber\\
    &\le \|x_{k}^{t,s}-x^{\star}\|^{2}
       -2\eta\bigl[f_k(x_{k}^{t,s})-f_k(x^{\star})\bigr]
       -2\eta\mu\langle x_{k}^{t,s}-x^{t},x_{k}^{t,s}-x^{\star}\rangle \nonumber\\
    &\qquad +\eta^{2}\Bigl(\|\nabla f_k(x_{k}^{t,s})+\mu(x_{k}^{t,s}-x^{t})\|^{2}+\sigma^{2}\Bigr). \tag{5}
\end{align}
\textbf{[Step 5]}

\subsection*{Step 5: Expand the proximal inner product}
Using the identity $2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$ with
$a=x_{k}^{t,s}-x^{t}$ and $b=x_{k}^{t,s}-x^{\star}$,
\[
    -2\mu\langle x_{k}^{t,s}-x^{t},x_{k}^{t,s}-x^{\star}\rangle
    = -\mu\|x_{k}^{t,s}-x^{t}\|^{2}
      -\mu\|x_{k}^{t,s}-x^{\star}\|^{2}
      +\mu\|x^{t}-x^{\star}\|^{2}. \tag{6}
\]
\textbf{[Step 6]}

\subsection*{Step 6: Upper bound the squared gradient term}
By $L$‑smoothness,
\[
    \|\nabla f_k(x_{k}^{t,s})\|^{2}\le 2L\bigl(f_k(x_{k}^{t,s})-f_k(x^{\star})\bigr). \tag{7}
\]
\textbf{[Step 7]}

Moreover,
\[
    \|\nabla f_k(x_{k}^{t,s})+\mu(x_{k}^{t,s}-x^{t})\|^{2}
    \le 2\|\nabla f_k(x_{k}^{t,s})\|^{2}+2\mu^{2}\|x_{k}^{t,s}-x^{t}\|^{2}. \tag{8}
\]
\textbf{[Step 8]}

\subsection*{Step 7: Plug (7)–(8) into (5) and use (6)}
After substitution and regrouping terms we obtain
\begin{align}
    &\mathbb{E}\!\bigl[\|x_{k}^{t,s+1}-x^{\star}\|^{2}\mid x_{k}^{t,s}\bigr]\nonumber\\
    &\le \bigl(1-\mu\eta\bigr)\|x_{k}^{t,s}-x^{\star}\|^{2}
       -2\eta\bigl(1-L\eta\bigr)\bigl[f_k(x_{k}^{t,s})-f_k(x^{\star})\bigr]\nonumber\\
    &\qquad +\eta^{2}\sigma^{2}
       +\mu\eta\|x^{t}-x^{\star}\|^{2}
       +\eta^{2}\mu^{2}\|x_{k}^{t,s}-x^{t}\|^{2}
       -\mu\eta\|x_{k}^{t,s}-x^{t}\|^{2}. \tag{9}
\end{align}
\textbf{[Step 9]}

Because $\eta\le \frac{1}{L+\mu}$ we have $1-L\eta\ge \mu\eta$, thus $1-L\eta\ge0$ and the coefficient in front of the function gap is non‑negative.

\subsection*{Step 8: Recurrence over $E$ local steps}
Define $\Delta_{k}^{t,s}:=\mathbb{E}\!\bigl[\|x_{k}^{t,s}-x^{\star}\|^{2}\bigr]$.
Taking total expectation of (9) and iterating $s=0,\dots,E-1$ yields
\begin{align}
    \Delta_{k}^{t,E}
    &\le (1-\mu\eta)^{E}\Delta_{k}^{t,0}
       -2\eta\bigl(1-L\eta\bigr)\sum_{s=0}^{E-1}\mathbb{E}\!\bigl[f_k(x_{k}^{t,s})-f_k(x^{\star})\bigr]\nonumber\\
    &\qquad +E\eta^{2}\sigma^{2}
       +\mu\eta E\mathbb{E}\!\bigl[\|x^{t}-x^{\star}\|^{2}\bigr]
       +\eta^{2}\mu^{2}\sum_{s=0}^{E-1}\mathbb{E}\!\bigl[\|x_{k}^{t,s}-x^{t}\|^{2}\bigr]. \tag{10}
\end{align}
\textbf{[Step 10]}

\subsection*{Step 9: Bounding the drift $\|x_{k}^{t,s}-x^{t}\|^{2}$}
Using the recursion $x_{k}^{t,s+1}=x_{k}^{t,s}-\eta g_{k}^{t,s}$ and the fact that $x_{k}^{t,0}=x^{t}$, a simple induction gives
\[
    \|x_{k}^{t,s}-x^{t}\|\le \eta\sum_{r=0}^{s-1}\|g_{k}^{t,r}\|.
\]
Taking squares, expectations and applying Jensen’s inequality,
\[
    \mathbb{E}\!\bigl[\|x_{k}^{t,s}-x^{t}\|^{2}\bigr]
    \le \eta^{2}s\sum_{r=0}^{s-1}\mathbb{E}\!\bigl[\|g_{k}^{t,r}\|^{2}\bigr]
    \le \eta^{2}s^{2}\bigl(G^{2}+\mu^{2}D^{2}+\sigma^{2}\bigr),
\]
where $D:=\max_{t}\|x^{t}-x^{\star}\|$ (finite under bounded iterates). Hence
\[
    \sum_{s=0}^{E-1}\mathbb{E}\!\bigl[\|x_{k}^{t,s}-x^{t}\|^{2}\bigr]
    \le \eta^{2}E^{3}\bigl(G^{2}+\mu^{2}D^{2}+\sigma^{2}\bigr). \tag{11}
\]
\textbf{[Step 11]}

Plugging (11) into (10) and using $\Delta_{k}^{t,0}=\mathbb{E}\!\bigl[\|x^{t}-x^{\star}\|^{2}\bigr]$ gives
\begin{align}
    \mathbb{E}\!\bigl[\|x_{k}^{t+1}-x^{\star}\|^{2}\bigr]
    &\le (1-\mu\eta)^{E}\mathbb{E}\!\bigl[\|x^{t}-x^{\star}\|^{2}\bigr]\nonumber\\
    &\quad -2\eta\bigl(1-L\eta\bigr)\sum_{s=0}^{E-1}\mathbb{E}\!\bigl[f_k(x_{k}^{t,s})-f_k(x^{\star})\bigr]
    +E\eta^{2}\sigma^{2}+C\eta^{4}E^{3}, \tag{12}
\end{align}
where $C$ aggregates the constants from (11).

\subsection*{Step 10: Server aggregation}
Averaging (12) over $k$ and using $\frac{1}{K}\sum_{k}x_{k}^{t+1}=x^{t+1}$ yields
\begin{align}
    \mathbb{E}\!\bigl[\|x^{t+1}-x^{\star}\|^{2}\bigr]
    &\le (1-\mu\eta)^{E}\mathbb{E}\!\bigl[\|x^{t}-x^{\star}\|^{2}\bigr]
       -2\eta\bigl(1-L\eta\bigr)\sum_{s=0}^{E-1}\mathbb{E}\!\bigl[F(x_{k}^{t,s})-F(x^{\star})\bigr]\nonumber\\
    &\quad +E\eta^{2}\frac{\sigma^{2}}{K}+C\eta^{4}E^{3}. \tag{13}
\end{align}
\textbf{[Step 12]}

\subsection*{Step 11: Relating local iterates to the global one}
Define the averaged local model after $E$ steps:
\[
    \tilde{x}^{t}:=\frac{1}{K}\sum_{k=1}^{K}x_{k}^{t,E}=x^{t+1}.
\]
By convexity of $F$ and Jensen’s inequality,
\[
    F(\tilde{x}^{t})\le\frac{1}{K}\sum_{k=1}^{K}F(x_{k}^{t,E}).
\]
Moreover, using smoothness and the recursion of the local steps one can show (see e.g. \cite{li2020federated}) that
\[
    \frac{1}{E}\sum_{s=0}^{E-1}F(x_{k}^{t,s})
    \ge F(\tilde{x}^{t})-\frac{L\eta^{2}E\zeta^{2}}{2}. \tag{14}
\]
\textbf{[Step 13]}

Insert (14) into (13) and drop the non‑negative term $(1-\mu\eta)^{E}\mathbb{E}[\|x^{t}-x^{\star}\|^{2}]$ to obtain a telescoping inequality:
\begin{align}
    2\eta\bigl(1-L\eta\bigr)E\Bigl(\mathbb{E}[F(\bar{x}_{t})]-F(x^{\star})\Bigr)
    &\le \mathbb{E}\!\bigl[\|x^{t}-x^{\star}\|^{2}\bigr]
       -\mathbb{E}\!\bigl[\|x^{t+1}-x^{\star}\|^{2}\bigr]
       +E\eta^{2}\frac{\sigma^{2}}{K}+L\eta^{2}E\zeta^{2}. \tag{15}
\end{align}
\textbf{[Step 14]}

\subsection*{Step 12: Summation over $t=0,\dots,T-1$}
Summing (15) from $t=0$ to $T-1$ telescopes the distance terms:
\begin{align}
    2\eta\bigl(1-L\eta\bigr)E\sum_{t=0}^{T-1}\bigl(\mathbb{E}[F(\bar{x}_{t})]-F(x^{\star})\bigr)
    &\le \|x^{0}-x^{\star}\|^{2}
       +T E\eta^{2}\frac{\sigma^{2}}{K}+T L\eta^{2}E\zeta^{2}. \tag{16}
\end{align}
\textbf{[Step 15]}

Dividing by $2\eta(1-L\eta)ET$ and recalling that $\bar{x}_{T}=\frac{1}{T}\sum_{t=0}^{T-1}\bar{x}_{t}$, convexity of $F$ gives
\[
    \mathbb{E}[F(\bar{x}_{T})]-F(x^{\star})
    \le \frac{\|x^{0}-x^{\star}\|^{2}}{2\eta(1-L\eta)ET}
       +\frac{\eta\sigma^{2}}{2K(1-L\eta)}
       +\frac{L\eta\zeta^{2}}{2(1-L\eta)}.
\]
Since $0<\eta\le\frac{1}{L+\mu}\le\frac{1}{L}$ we have $1-L\eta\ge\frac{1}{2}$, which simplifies constants and yields the bound stated in Theorem~\ref{thm:conv}:
\[
    \mathbb{E}[F(\bar{x}_{T})-F(x^{\star})]
    \le \frac{\|x^{0}-x^{\star}\|^{2}}{2\eta T}
       +\frac{\eta L\sigma^{2}}{K}
       +\frac{L\eta^{2}E\zeta^{2}}{2}.
\]
\textbf{[Step 16]}

\subsection*{Step 13: Choosing $\eta$ for optimal rate}
Set $\eta=\Theta\!\bigl(1/\sqrt{T}\bigr)$. Then the three terms on the right‑hand side scale as $O(1/\sqrt{T})$, establishing the $O(1/\sqrt{T})$ convergence rate.

\section*{Rate Derivation (With Constants)}
Let $\eta = \frac{c}{\sqrt{T}}$ with $c\le\frac{1}{L+\mu}\sqrt{T}$. Substituting into the bound of Theorem~\ref{thm:conv} yields
\[
\mathbb{E}[F(\bar{x}_{T})-F(x^{\star})]
\le \frac{\|x^{0}-x^{\star}\|^{2}}{2c}\frac{1}{\sqrt{T}}
   +\frac{cL\sigma^{2}}{K}\frac{1}{\sqrt{T}}
   +\frac{c^{2}L E\zeta^{2}}{2}\frac{1}{T}.
\]
The dominant $1/\sqrt{T}$ terms give the final rate
\[
\boxed{\;\mathbb{E}[F(\bar{x}_{T})-F(x^{\star})]=O\!\bigl(T^{-1/2}\bigr)\; } .
\]

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Homogeneous data ($\zeta=0$).} The heterogeneity term disappears and the bound matches that of centralized SGD with minibatch size $K$.
    \item \textbf{No proximal regularization ($\mu=0$).} The stepsize restriction becomes $\eta\le 1/L$, recovering the classical FedAvg analysis.
    \item \textbf{Single local step ($E=1$).} The drift term $L\eta^{2}E\zeta^{2}$ reduces to $L\eta^{2}\zeta^{2}$, matching the bound for parallel SGD with occasional communication.
\end{enumerate}
