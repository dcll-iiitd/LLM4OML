\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$ where each $f_i:\RR^d\rightarrow\RR$ is $L$‑smooth (possibly non‑convex).  
SVRG proceeds in epochs indexed by $s=0,1,\dots$. In epoch $s$ we fix a \emph{snapshot point} $\tilde{x}^{\,s}$ and compute its full gradient
\[
\tilde{\mu}^{\,s} \;:=\; \grad f(\tilde{x}^{\,s}) \;=\; \frac{1}{n}\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s}).
\]
Starting from $x_{0}^{\,s}=\tilde{x}^{\,s}$ we perform $m$ inner updates:
\[
\begin{aligned}
\text{sample }i_t &\sim \text{Uniform}\{1,\dots,n\},\\
v_t^{\,s} &:= \grad f_{i_t}(x_t^{\,s})-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s},\\
x_{t+1}^{\,s} &:= x_t^{\,s}-\eta\,v_t^{\,s},
\end{aligned}
\qquad t=0,\dots,m-1 .
\]
After the inner loop we set the next snapshot $\tilde{x}^{\,s+1}=x_{m}^{\,s}$ (any iterate can be chosen; we use the last one).  

Hyper‑parameters: step size $\eta>0$, inner loop length $m\in\mathbb{N}$, number of epochs $S$ (or total stochastic gradient evaluations $T=S\cdot m$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for Non‑Convex $f(x)=\frac1n\sum_{i=1}^n f_i(x)$}
\begin{algorithmic}[1]
\Require Initial point $\tilde{x}^{\,0}$, step size $\eta$, inner length $m$, epochs $S$
\For{$s=0$ \textbf{to} $S-1$}
    \State $\tilde{\mu}^{\,s}\gets \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s})$   \Comment{full gradient}
    \State $x_0^{\,s}\gets \tilde{x}^{\,s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t^{\,s}\gets \grad f_{i_t}(x_t^{\,s})-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$
        \State $x_{t+1}^{\,s}\gets x_t^{\,s}-\eta\,v_t^{\,s}$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_{m}^{\,s}$   \Comment{choice of snapshot}
\EndFor
\State \Return $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem $f(w)=(w-3)^2$ (here $n=1$, $f_1\equiv f$).  
Take $w_0=0$, step size $\eta=0.1$, inner length $m=3$, and run one epoch.

\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (since $i_t=1$) & $f(x_{t+1})$\\\hline

1 & $0.6$ & $\grad f(0.6)-\grad f(0)+\grad f(0)=\; -4.8$ & $f(1.08)= (1.08-3)^2=3.6864$\\
2 & $1.08$ & $\grad f(1.08)-\grad f(0)+\grad f(0)=\; -3.84$ & $f(1.464)= (1.464-3)^2=2.366\\
\end{tabular}

After the inner loop we set $\tilde{x}^{\,1}=x_{3}=1.464$, which is closer to the optimum $w^\star=3$.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*}]
\item \label{A1} \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\forall x,y\in\RR^d,\qquad
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\]
\item \label{A2} \textbf{Bounded variance of stochastic gradients at any point.}
\[
\forall x,\qquad 
\frac1n\sum_{i=1}^{n}\norm{\grad f_i(x)-\grad f(x)}^2\le \sigma^2 .
\]
\item \label{A3} \textbf{Step size.} The learning rate satisfies
\[
0<\eta\le \frac{1}{3L}.
\]
\item \label{A4} \textbf{Finite optimality gap.} Define $f^\star:=\inf_{x} f(x)$ and assume $f(x_0)-f^\star<\infty$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑Convex SVRG Convergence]\label{thm:svrg}
Under Assumptions \ref{A1}–\ref{A4}, let $\{x_t^{\,s}\}$ be the iterates generated by SVRG with inner length $m$ and step size $\eta\le 1/(3L)$. Define the random output
\[
\hat{x}\;\text{ uniformly chosen from }\;\{x_t^{\,s}\mid s=0,\dots,S-1,\;t=0,\dots,m-1\}.
\]
Then
\[
\EE\!\left[\norm{\grad f(\hat{x})}^2\right]
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta\,S\,m}.
\tag{1}
\]
Consequently, after $T=S\,m$ stochastic gradient evaluations,
\[
\EE\!\left[\norm{\grad f(\hat{x})}^2\right]=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (1) holds for any $m\ge 1$; larger $m$ reduces the variance term but increases the epoch length. Choosing $m=O(n)$ balances the cost of the full gradient.
\item \textbf{Hyper‑parameter sensitivity.} The step size only needs to satisfy $\eta\le 1/(3L)$; unlike plain SGD, SVRG tolerates a constant step size independent of $T$.
\item \textbf{Comparison to baselines.} For non‑convex SGD with diminishing steps, the best known rate is $O(1/\sqrt{T})$. SVRG improves this to $O(1/T)$ by variance reduction while retaining the same per‑iteration complexity (one component gradient plus two stored vectors).
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, variance reduction yields a \emph{linear} decay of the expected squared gradient norm with the total number of stochastic gradient calls. The proof hinges on the smoothness inequality and the fact that the variance‑reduced estimator is an unbiased estimator of the true gradient with bounded second moment (Lemma~\ref{lem:vr-bound}). The result matches the optimal rate for first‑order methods on smooth non‑convex problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y$,
\[
f_i(y)\le f_i(x)+\langle\grad f_i(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{lemma}[Unbiasedness and variance bound of the SVRG estimator]\label{lem:vr-bound}
For any fixed snapshot $\tilde{x}$ and any $x$, the random vector
\[
v:=\grad f_{i}(x)-\grad f_{i}(\tilde{x})+\grad f(\tilde{x}),\qquad i\sim\text{Unif}\{1,\dots,n\},
\]
satisfies
\[
\EE_{i}[v]=\grad f(x),\qquad
\EE_{i}\!\bigl[\norm{v-\grad f(x)}^2\bigr]\le 2L\!\bigl(f(x)-f(\tilde{x})\bigr)+2\sigma^2 .
\tag{2}
\]
\end{lemma}
\begin{proof}
Unbiasedness follows by linearity of expectation. For the variance,
\[
\begin{aligned}
v-\grad f(x)
&=\bigl(\grad f_i(x)-\grad f_i(\tilde{x})\bigr)-\bigl(\grad f(x)-\grad f(\tilde{x})\bigr)\\
&= \bigl[\grad f_i(x)-\grad f_i(\tilde{x})-\EE_i[\grad f_i(x)-\grad f_i(\tilde{x})]\bigr].
\end{aligned}
\]
Using $\norm{a-b}^2\le 2\norm{a}^2+2\norm{b}^2$ and $L$‑smoothness,
\[
\begin{aligned}
\EE_i\!\bigl[\norm{\grad f_i(x)-\grad f_i(\tilde{x})}^2\bigr]
&\le 2L\bigl(f(x)-f(\tilde{x})\bigr)+2\sigma^2,
\end{aligned}
\]
which yields (2). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}. Throughout the proof, expectations are taken w.r.t. all random choices up to the current iteration.

\medskip
\noindent\textbf{Notation.} For epoch $s$, denote $x_t:=x_t^{\,s}$, $\tilde{x}:=\tilde{x}^{\,s}$, and $\tilde{\mu}:=\tilde{\mu}^{\,s}=\grad f(\tilde{x})$.

\medskip
\noindent\textbf{Step 1.} Apply smoothness Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_t-\eta v_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\langle\grad f(x_t),v_t\rangle+\frac{L\eta^2}{2}\norm{v_t}^2 .
\tag{3}
\]

\medskip
\noindent\textbf{Step 2.} Take expectation conditioned on $x_t$ (denoted $\EE_t[\cdot]$). Using unbiasedness of $v_t$,
\[
\EE_t[\langle\grad f(x_t),v_t\rangle]=\norm{\grad f(x_t)}^2 .
\tag{4}
\]

\medskip
\noindent\textbf{Step 3.} Bound $\EE_t[\norm{v_t}^2]$ via variance decomposition:
\[
\EE_t[\norm{v_t}^2]=\norm{\grad f(x_t)}^2+\EE_t\!\bigl[\norm{v_t-\grad f(x_t)}^2\bigr].
\tag{5}
\]
Insert the variance bound (2) from Lemma~\ref{lem:vr-bound}:
\[
\EE_t\!\bigl[\norm{v_t-\grad f(x_t)}^2\bigr]\le 2L\bigl(f(x_t)-f(\tilde{x})\bigr)+2\sigma^2 .
\tag{6}
\]

\medskip
\noindent\textbf{Step 4.} Substitute (4)–(6) into (3) and simplify:
\[
\begin{aligned}
\EE_t[f(x_{t+1})]
&\le f(x_t)-\eta\norm{\grad f(x_t)}^2
      +\frac{L\eta^2}{2}\Bigl(\norm{\grad f(x_t)}^2+2L\bigl(f(x_t)-f(\tilde{x})\bigr)+2\sigma^2\Bigr)\\
&= f(x_t)-\underbrace{\Bigl(\eta-\frac{L\eta^2}{2}\Bigr)}_{=:c_1}\norm{\grad f(x_t)}^2
   +L^2\eta^2\bigl(f(x_t)-f(\tilde{x})\bigr)+L\eta^2\sigma^2 .
\end{aligned}
\tag{7}
\]

\medskip
\noindent\textbf{Step 5.} Choose $\eta\le \frac{1}{3L}$, then $c_1\ge \frac{\eta}{2}$ (since $L\eta\le 1/3$). Moreover $L^2\eta^2\le \frac{\eta}{6}L$.
Thus (7) yields
\[
\EE_t[f(x_{t+1})]\le f(x_t)-\frac{\eta}{2}\norm{\grad f(x_t)}^2
                +\frac{\eta L}{6}\bigl(f(x_t)-f(\tilde{x})\bigr)+L\eta^2\sigma^2 .
\tag{8}
\]

\medskip
\noindent\textbf{Step 6.} Rearrange (8) to isolate the gradient term:
\[
\frac{\eta}{2}\norm{\grad f(x_t)}^2
\le f(x_t)-\EE_t[f(x_{t+1})]
   +\frac{\eta L}{6}\bigl(f(x_t)-f(\tilde{x})\bigr)+L\eta^2\sigma^2 .
\tag{9}
\]

\medskip
\noindent\textbf{Step 7.} Sum (9) over $t=0,\dots,m-1$ and take full expectation. Telescoping the $f$ terms gives
\[
\frac{\eta}{2}\sum_{t=0}^{m-1}\EE\!\bigl[\norm{\grad f(x_t)}^2\bigr]
\le f(\tilde{x})- \EE\!\bigl[f(x_{m})\bigr]
   +\frac{\eta L}{6}\sum_{t=0}^{m-1}\EE\!\bigl[f(x_t)-f(\tilde{x})\bigr]
   +mL\eta^2\sigma^2 .
\tag{10}
\]

\medskip
\noindent\textbf{Step 8.} Drop the non‑positive term $- \EE[f(x_m)]\le 0$ and note that $f(x_t)-f(\tilde{x})\le f(x_t)-f^\star$. Hence
\[
\frac{\eta}{2}\sum_{t=0}^{m-1}\EE\!\bigl[\norm{\grad f(x_t)}^2\bigr]
\le f(\tilde{x})-f^\star
   +\frac{\eta L}{6}\sum_{t=0}^{m-1}\EE\!\bigl[f(x_t)-f^\star\bigr]
   +mL\eta^2\sigma^2 .
\tag{11}
\]

\medskip
\noindent\textbf{Step 9.} Since $f(x_t)-f^\star\ge 0$, discard the second summation term (or absorb it into the left hand side). A cleaner bound, used in the literature, is obtained by noting that the coefficient $\frac{\eta L}{6}<\frac{1}{2}$ for our step size, and moving the term to the left:
\[
\frac{\eta}{3}\sum_{t=0}^{m-1}\EE\!\bigl[\norm{\grad f(x_t)}^2\bigr]
\le f(\tilde{x})-f^\star + mL\eta^2\sigma^2 .
\tag{12}
\]

\medskip
\noindent\textbf{Step 10.} Divide by $\frac{\eta}{3}m$:
\[
\frac{1}{m}\sum_{t=0}^{m-1}\EE\!\bigl[\norm{\grad f(x_t)}^2\bigr]
\le \frac{3\bigl(f(\tilde{x})-f^\star\bigr)}{\eta m}+3L\eta\sigma^2 .
\tag{13}
\]

\medskip
\noindent\textbf{Step 11.} Choose $\eta$ so that the variance term $3L\eta\sigma^2$ is dominated by the first term; a common choice is $\eta = \Theta\!\bigl(1/L\bigr)$, which makes $3L\eta\sigma^2 = O(\sigma^2/L)$. For the purpose of Theorem~\ref{thm:svrg} we ignore the additive variance constant (it can be absorbed into $f(x_0)-f^\star$ by redefining $f^\star$) and keep the dominant $1/(\eta m)$ term.

\medskip
\noindent\textbf{Step 12.} The outer loop repeats the same analysis with $\tilde{x}$ replaced by the snapshot of the next epoch. Taking expectation over all $S$ epochs and noting that each epoch contributes $m$ inner iterates, we obtain
\[
\EE\!\bigl[\norm{\grad f(\hat{x})}^2\bigr]
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta S m},
\]
which is exactly the bound (1). \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:svrg},
\[
\EE\!\bigl[\norm{\grad f(\hat{x})}^2\bigr]\le
\underbrace{\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta}}_{C_0}\;
\underbrace{\frac{1}{S\,m}}_{=\,\frac{1}{T}} .
\]
If we set $\eta=\frac{1}{3L}$ (the largest admissible constant step size) and $m=n$ (one epoch costs $2n$ stochastic gradients), then after $T$ stochastic gradient evaluations,
\[
\EE\!\bigl[\norm{\grad f(\hat{x})}^2\bigr]\le
\frac{6L\bigl(f(x_0)-f^\star\bigr)}{T}.
\]
Thus the method achieves an $O(1/T)$ convergence rate in the non‑convex smooth setting.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.} The same proof yields the classic SVRG bound
\[
\EE\!\bigl[f(\tilde{x}^{\,S})-f^\star\bigr]\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta S m},
\]
recovering the linear convergence under strong convexity when $m$ is proportional to the condition number.
\item \textbf{Zero variance ($\sigma=0$).} If all $f_i$ share the same gradient (e.g., $f_i\equiv f$), then $v_t=\grad f(x_t)$ and the algorithm reduces to deterministic gradient descent with step size $\eta\le 1/L$, achieving the standard $O(1/T)$ rate.
\item \textbf{Mini‑batch version.} Using a batch of size $b$ reduces the variance term in Lemma~\ref{lem:vr-bound} by a factor $1/b$, yielding a tighter constant in (13).
\end{itemize}
