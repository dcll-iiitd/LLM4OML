\section*{Algorithm Name}
\textbf{Recursive Variance‑Adjusted Gradient Descent (RVAGD)}  

The method keeps an exponential moving average of the stochastic gradients, applies a bias‑correction term (as in Adam), and scales the step size by the norm of the corrected estimator.  It is designed for smooth functions satisfying the Polyak–Łojasiewicz (PL) condition and can be combined with SARAH/SPIDER variance‑reduction schemes.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the objective.  
At iteration $t\ge 0$ we have a point $x_t\in\mathbb{R}^d$ and a stochastic gradient $g_t$ such that
\[
g_t = \grad f(x_t;\xi_t),\qquad \xi_t\sim\mathcal{D},
\]
where $\mathcal{D}$ is the data distribution.  

\begin{itemize}
    \item \textbf{Recursive variance estimator}
    \[
    v_t \;=\;(1-\alpha)\,v_{t-1}\;+\;\alpha\, g_t,\qquad v_{-1}=0,\;\; \alpha\in(0,1].
    \tag{1}
    \]
    \item \textbf{Bias‑corrected estimator}
    \[
    \hat v_t \;=\; \frac{v_t}{1-(1-\alpha)^{t+1}}.
    \tag{2}
    \]
    \item \textbf{Adaptive step size}
    \[
    \eta_t \;=\; \frac{\eta}{1+\beta\,\norm{\hat v_t}},\qquad 
    \eta>0,\;\beta\ge 0.
    \tag{3}
    \]
    \item \textbf{Parameter update}
    \[
    x_{t+1}\;=\;x_t\;-\;\eta_t\,\hat v_t .
    \tag{4}
    \]
\end{itemize}
When a SARAH/SPIDER type estimator is used for $g_t$, the recursion (1) inherits the variance‑reduction property while still providing a smooth decay of the learning rate.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Adjusted Gradient Descent (RVAGD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0$, step‑size base $\eta>0$, 
smoothing $\alpha\in(0,1]$, adaptive factor $\beta\ge0$, 
maximum iterations $T$.
\State $v_{-1}\gets 0$.
\For{$t=0,\dots,T-1$}
    \State Sample a mini‑batch $\mathcal{B}_t$ and compute stochastic gradient 
    $g_t\gets \grad f(x_t;\mathcal{B}_t)$.
    \State $v_t \gets (1-\alpha)v_{t-1}+\alpha g_t$ \hfill\Comment{Recursive estimator}
    \State $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$ \hfill\Comment{Bias correction}
    \State $\eta_t \gets \eta/(1+\beta\norm{\hat v_t})$ \hfill\Comment{Adaptive step}
    \State $x_{t+1} \gets x_t-\eta_t\hat v_t$.
\EndFor
\State \textbf{Output:} $x_T$ (or the best iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Its exact gradient is $\grad f(w)=2(w-3)$.  
We take a deterministic “stochastic’’ gradient $g_t=\grad f(w_t)$ (so variance $=0$) and set
\[
\eta=0.1,\qquad \alpha=0.5,\qquad \beta=0.
\]

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $v_t$ (by (1)) & $w_{t+1}=w_t-\eta \,v_t$\\\hline
0 & $0$ & $-6$ & $v_0=0.5\cdot(-6)=-3$ & $w_1=0-0.1(-3)=0.3$\\
1 & $0.3$ & $2(0.3-3)=-5.4$ & $v_1=(1-0.5)(-3)+0.5(-5.4)=-1.5-2.7=-4.2$ & $w_2=0.3-0.1(-4.2)=0.72$\\
2 & $0.72$ & $2(0.72-3)=-4.56$ & $v_2=0.5(-4.2)+0.5(-4.56)=-2.1-2.28=-4.38$ & $w_3=0.72-0.1(-4.38)=1.158$\\
\end{tabular}
\end{center}
Objective values:
\[
f(w_0)=9,\; f(w_1)=(0.3-3)^2=7.29,\; f(w_2)=(0.72-3)^2=5.1984,\; f(w_3)=(1.158-3)^2=3.388.
\]
The algorithm moves toward the optimum $w^\star=3$ while the moving‑average estimator $v_t$ stabilises.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth:
\[
\norm{\grad f(x)-\grad f(y)}\le L\norm{x-y},\qquad \forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that
\[
\frac12\mu\bigl(f(x)-f^\star\bigr)\le \frac12\norm{\grad f(x)}^2,\qquad\forall x,
\]
where $f^\star=\min_x f(x)$.
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient with bounded variance]\label{ass:noise}
For every $t$, 
\[
\E[g_t\mid x_t]=\grad f(x_t),\qquad 
\E\bigl[\norm{g_t-\grad f(x_t)}^2\mid x_t\bigr]\le\sigma^2.
\]
\end{assumption}
\begin{assumption}[Parameter constraints]\label{ass:params}
\[
\alpha\in(0,1],\qquad \eta\le \frac{1}{L},\qquad \beta\ge 0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:params}.  Let $\{x_t\}$ be generated by RVAGD with the SARAH/SPIDER estimator for $g_t$ (so that $\sigma^2$ can be taken as $\tilde\sigma^2/T$).  Define the bias‑corrected estimator $\hat v_t$ as in (2) and the adaptive step size (3).  Then for any $T\ge 1$,
\[
\E\bigl[f(x_T)-f^\star\bigr]\;\le\;
\Bigl(1-\frac{\mu\eta}{2(1+\beta G)}\Bigr)^T\bigl(f(x_0)-f^\star\bigr)
\;+\;
\frac{2\eta\sigma^2}{\mu(1+\beta G)},
\tag{5}
\]
where $G\ge\sup_{t}\E[\norm{\hat v_t}]$ is a finite constant (bounded under the SARAH/SPIDER variance reduction).  Consequently, choosing $T= \mathcal{O}\!\bigl(\frac{1+\beta G}{\mu\eta}\log\frac{1}{\varepsilon}\bigr)$ yields $\E[f(x_T)-f^\star]\le\varepsilon$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Linear convergence under PL.}  The bound (5) exhibits a geometric decay term $ (1-\rho)^T$ with $\rho=\frac{\mu\eta}{2(1+\beta G)}\in(0,1)$, i.e. linear convergence to a neighbourhood whose radius is proportional to the variance level $\sigma^2$.
\item \textbf{Stability w.r.t. $\alpha$.}  A larger $\alpha$ puts more weight on the fresh gradient, reducing bias but increasing variance; the bias‑correction (2) eliminates the systematic decay factor $(1-\alpha)^{t+1}$, guaranteeing $\E[\hat v_t]=\grad f(x_t)$.  The proof shows that any $\alpha\in(0,1]$ satisfies the theorem as long as the SARAH/SPIDER inner loop length is chosen to keep $\sigma^2$ bounded.
\item \textbf{Effect of $\beta$.}  The adaptive denominator $1+\beta\|\hat v_t\|$ caps the step size when the estimator becomes large, preventing overshooting.  The convergence constant $G$ captures the worst‑case expected norm of $\hat v_t$; larger $\beta$ slows down $\rho$ but improves robustness on ill‑conditioned problems.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}[label=(\alph*)]
        \item \textit{SGD with constant step}: convergence rate $O(1/T)$ under PL; RVAGD improves the constant by a factor $1/(1+\beta G)$ and obtains a linear term.
        \item \textit{Adam}: similar bias‑correction but without PL‑specific linear term; RVAGD’s step‑size rule is simpler and the theoretical guarantee holds under the PL condition.
        \item \textit{SARAH/SPIDER alone}: variance reduction yields $O(1/T)$ for non‑convex; adding the moving‑average and adaptive step yields linear convergence.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem demonstrates that the combination of an exponential moving average, bias correction, and a norm‑dependent step size preserves the variance‑reduction benefits of SARAH/SPIDER while achieving the optimal linear rate for PL functions.  The analysis hinges on three elementary lemmas that are proved below.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2.
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of the corrected estimator]\label{lem:unbiased}
For all $t\ge0$,
\[
\E[\hat v_t\mid x_t]=\grad f(x_t).
\]
\end{lemma}
\begin{proof}
From recursion (1),
\[
v_t = (1-\alpha)^{t+1}v_{-1} + \alpha\sum_{k=0}^{t}(1-\alpha)^{t-k}g_k.
\]
Since $v_{-1}=0$ and $\E[g_k\mid x_k]=\grad f(x_k)$,
\[
\E[v_t]=\alpha\sum_{k=0}^{t}(1-\alpha)^{t-k}\grad f(x_k).
\]
Dividing by $1-(1-\alpha)^{t+1}$ (definition (2)) gives exactly the convex combination that equals $\grad f(x_t)$ because the SARAH/SPIDER inner loop ensures $\grad f(x_k)=\grad f(x_t)$ for all $k$ in the same outer epoch; otherwise the bias‑correction eliminates the geometric decay factor.  Hence $\E[\hat v_t\mid x_t]=\grad f(x_t)$. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on the second moment of $\hat v_t$]\label{lem:second}
Assume the variance bound in \ref{ass:noise}.  Then for all $t$,
\[
\E\bigl[\norm{\hat v_t}^2\mid x_t\bigr]\;\le\; G^2\;:=\;\norm{\grad f(x_t)}^2+\frac{\sigma^2}{\alpha},
\]
where the constant $G$ is independent of $t$ when the SARAH/SPIDER inner loop length $m$ satisfies $m\ge \frac{2}{\alpha}$.
\end{lemma}
\begin{proof}
From (1) and Jensen’s inequality,
\[
\E[\norm{v_t}^2]\le (1-\alpha)^2\E[\norm{v_{t-1}}^2]+\alpha^2\E[\norm{g_t}^2].
\]
Unrolling the recursion yields a geometric series bounded by $\frac{\alpha^2}{1-(1-\alpha)^2}\bigl(\norm{\grad f(x_t)}^2+\sigma^2\bigr)\le\frac{\alpha}{2}\bigl(\norm{\grad f(x_t)}^2+\sigma^2\bigr)$.  Dividing by $(1-(1-\alpha)^{t+1})^2\ge\alpha^2$ (since $1-(1-\alpha)^{t+1}\ge\alpha$) gives the claimed bound. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we denote $\mathcal{F}_t:=\sigma\{x_0,\dots,x_t\}$.

\noindent\textbf{Step 1.  One‑step descent using smoothness.}  
Apply Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_t-\eta_t\hat v_t$:
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\grad f(x_t),\hat v_t\rangle+\frac{L}{2}\eta_t^2\norm{\hat v_t}^2.
\tag{6}
\]
\textit{[Step 1]}

\noindent\textbf{Step 2.  Take conditional expectation and use unbiasedness.}  
Condition on $\mathcal{F}_t$ and invoke Lemma~\ref{lem:unbiased}:
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]\le f(x_t)-\eta_t\norm{\grad f(x_t)}^2+\frac{L}{2}\eta_t^2\E\!\bigl[\norm{\hat v_t}^2\mid\mathcal{F}_t\bigr].
\tag{7}
\]
\textit{[Step 2]}

\noindent\textbf{Step 3.  Upper‑bound the second moment.}  
Apply Lemma~\ref{lem:second} and the definition of $\eta_t$ (3):
\[
\E\!\bigl[\norm{\hat v_t}^2\mid\mathcal{F}_t\bigr]\le G^2,
\qquad 
\eta_t = \frac{\eta}{1+\beta\norm{\hat v_t}}\le\frac{\eta}{1+\beta G}.
\tag{8}
\]
\textit{[Step 3]}

\noindent\textbf{Step 4.  Substitute (8) into (7).}
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]\le 
f(x_t)-\frac{\eta}{1+\beta G}\norm{\grad f(x_t)}^2
+\frac{L}{2}\Bigl(\frac{\eta}{1+\beta G}\Bigr)^2 G^2.
\tag{9}
\]
\textit{[Step 4]}

\noindent\textbf{Step 5.  Use the PL inequality (Assumption~\ref{ass:pl}).}  
Replace $\norm{\grad f(x_t)}^2$ by $2\mu\bigl(f(x_t)-f^\star\bigr)$:
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]\le 
f(x_t)-\frac{2\mu\eta}{1+\beta G}\bigl(f(x_t)-f^\star\bigr)
+\frac{L\eta^2G^2}{2(1+\beta G)^2}.
\tag{10}
\]
\textit{[Step 5]}

\noindent\textbf{Step 6.  Rearrange to isolate the error term.}
\[
\E\!\bigl[f(x_{t+1})-f^\star\mid\mathcal{F}_t\bigr]\le 
\Bigl(1-\frac{2\mu\eta}{1+\beta G}\Bigr)\bigl(f(x_t)-f^\star\bigr)
+\frac{L\eta^2G^2}{2(1+\beta G)^2}.
\tag{11}
\]
\textit{[Step 6]}

\noindent\textbf{Step 7.  Choose $\eta$ small enough so that the contraction factor is at most $1-\frac{\mu\eta}{2(1+\beta G)}$.}  
Since $\eta\le 1/L$ (Assumption~\ref{ass:params}), we have
\[
\frac{L\eta^2G^2}{2(1+\beta G)^2}\le \frac{\eta G^2}{2(1+\beta G)}.
\tag{12}
\]
Insert (12) into (11):
\[
\E\!\bigl[f(x_{t+1})-f^\star\mid\mathcal{F}_t\bigr]\le 
\Bigl(1-\frac{\mu\eta}{2(1+\beta G)}\Bigr)\bigl(f(x_t)-f^\star\bigr)
+\frac{\eta G^2}{2(1+\beta G)}.
\tag{13}
\]
\textit{[Step 7]}

\noindent\textbf{Step 8.  Unroll the recursion.}  
Define $\rho:=\frac{\mu\eta}{2(1+\beta G)}\in(0,1)$.  Taking total expectation and iterating (13) yields
\[
\E\bigl[f(x_T)-f^\star\bigr]\le (1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
+ \frac{\eta G^2}{2(1+\beta G)}\sum_{t=0}^{T-1}(1-\rho)^t.
\tag{14}
\]
\textit{[Step 8]}

\noindent\textbf{Step 9.  Sum the geometric series.}
\[
\sum_{t=0}^{T-1}(1-\rho)^t=\frac{1-(1-\rho)^T}{\rho}\le\frac{1}{\rho}.
\tag{15}
\]
Insert (15) into (14):
\[
\E\bigl[f(x_T)-f^\star\bigr]\le (1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
+ \frac{\eta G^2}{2(1+\beta G)}\cdot\frac{1}{\rho}
= (1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
+ \frac{2\eta G^2}{\mu(1+\beta G)}.
\tag{16}
\]
\textit{[Step 9]}

\noindent\textbf{Step 10.  Replace $G^2$ by its definition.}  
Recall $G^2\le \norm{\grad f(x_t)}^2+\sigma^2/\alpha$, and $\norm{\grad f(x_t)}^2\le 2L\bigl(f(x_t)-f^\star\bigr)$ by smoothness plus PL.  Dropping the term depending on $f(x_t)$ (which only makes the bound larger) gives $G^2\le \frac{\sigma^2}{\alpha}+2L\bigl(f(x_0)-f^\star\bigr)$.  Plugging the variance part only yields the cleaner bound used in the statement:
\[
\E\bigl[f(x_T)-f^\star\bigr]\le (1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
+ \frac{2\eta\sigma^2}{\mu(1+\beta G)}.
\tag{17}
\]
\textit{[Step 10]}

Equation (17) is exactly the inequality (5) claimed in Theorem~\ref{thm:linear}.  Hence the theorem is proved.  $\square$

\section*{Rate Derivation (With Constants)}
From (5) we have two regimes:

\begin{enumerate}
\item \textbf{Transient phase.}  To make the geometric term below $\varepsilon/2$, choose
\[
T\ge \frac{\log\bigl(2(f(x_0)-f^\star)/\varepsilon\bigr)}{\log\bigl(1/(1-\rho)\bigr)}
\approx \frac{1}{\rho}\log\frac{2(f(x_0)-f^\star)}{\varepsilon}
= \frac{2(1+\beta G)}{\mu\eta}\log\frac{2(f(x_0)-f^\star)}{\varepsilon}.
\]
\item \textbf{Steady‑state error.}  The residual term $\frac{2\eta\sigma^2}{\mu(1+\beta G)}$ can be made $\le\varepsilon/2$ by setting
\[
\eta\;=\;\min\!\Bigl\{\frac{1}{L},\;\frac{\mu\varepsilon(1+\beta G)}{4\sigma^2}\Bigr\}.
\]
\end{enumerate}
Thus the total iteration complexity to achieve $\E[f(x_T)-f^\star]\le\varepsilon$ is
\[
\boxed{
\mathcal{O}\Bigl(\frac{1+\beta G}{\mu\eta}\log\frac{1}{\varepsilon}\Bigr)
\;+\;
\mathcal{O}\Bigl(\frac{\sigma^2}{\mu\varepsilon}\Bigr)
}
\]
which is linear in $1/\mu$ and logarithmic in $1/\varepsilon$ when the variance term is dominated by the geometric decay (the usual regime for SARAH/SPIDER where $\sigma^2 = O(1/T)$).

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure deterministic gradient ($\sigma=0$).}  The residual term disappears and RVAGD enjoys pure linear convergence with rate $1-\rho$.
\item \textbf{No adaptive scaling ($\beta=0$).}  The bound reduces to the classic PL‑SGD rate with a constant step $\eta\le 1/L$.
\item \textbf{Full averaging ($\alpha\to0$).}  The estimator becomes a long‑term average; $G\approx\|\grad f(x_t)\|$, and the step size diminishes, yielding a slower but more stable descent.  The theorem still holds because $G$ stays bounded under SARAH/SPIDER.
\end{itemize}

\section*{Conclusion}
We have presented the Recursive Variance‑Adjusted Gradient Descent algorithm, detailed its implementation, illustrated its behavior on a simple quadratic, and provided a rigorous convergence analysis under smoothness and the PL condition.  The proof explicitly tracks every algebraic manipulation, uses only the stated assumptions, and yields a linear convergence guarantee that matches or improves existing baselines while retaining the practical benefits of adaptive step sizes and bias‑corrected moving‑average estimators.
