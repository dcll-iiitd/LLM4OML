\section*{Algorithm Name}
\textbf{Stochastic Subsampled Newton (SSN)}  

A first‑order method that approximates the Newton direction by a stochastic (subsampled) estimate of the Hessian and takes a damped step.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be twice differentiable. At iteration $t$ we have a current iterate $x_t\in\R^{d}$.  
\begin{itemize}
    \item Stochastic gradient: $g_t:=\nabla f(x_t;\xi_t)$ where $\xi_t$ is a random data mini‑batch.
    \item Stochastic Hessian estimate: 
    \[
        H_t:=\nabla^{2} f(x_t;\zeta_t)+\lambda I,
    \]
    where $\zeta_t$ is an independent mini‑batch (possibly different size) and $\lambda>0$ is a damping parameter guaranteeing invertibility.
    \item Step size (learning rate) $\alpha>0$.
\end{itemize}
The update rule is
\begin{equation}
    x_{t+1}=x_t-\alpha H_t^{-1}g_t . \tag{1}
\end{equation}
When $H_t$ is singular we use the regularized matrix $H_t+\lambda I$; for brevity we denote the regularized inverse again by $H_t^{-1}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Subsampled Newton (SSN)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0$, step size $\alpha>0$, damping $\lambda>0$, batch sizes $b_g,b_H$, maximum iterations $T$.
\For{$t=0$ to $T-1$}
    \State Sample a gradient mini‑batch $\mathcal{B}_g^{(t)}$ of size $b_g$ and compute 
    \[
        g_t=\frac{1}{b_g}\sum_{i\in\mathcal{B}_g^{(t)}} \nabla f_i(x_t).
    \]
    \State Sample a Hessian mini‑batch $\mathcal{B}_H^{(t)}$ of size $b_H$ and compute 
    \[
        \tilde H_t=\frac{1}{b_H}\sum_{i\in\mathcal{B}_H^{(t)}} \nabla^{2} f_i(x_t).
    \]
    \State Form the regularized matrix $H_t:=\tilde H_t+\lambda I$ and compute its inverse $H_t^{-1}$ (or use a linear solver).
    \State Update $x_{t+1}=x_t-\alpha H_t^{-1}g_t$.
\EndFor
\State \textbf{Output:} $\{x_t\}_{t=0}^{T}$ (or the averaged iterate $\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$. Then $\nabla f(w)=2(w-3)$ and $\nabla^{2}f(w)=2$.  
We take $b_g=b_H=1$ (exact gradients/Hessian), $\lambda=0$, $\alpha=0.1$, $x_0=0$.

\begin{enumerate}
    \item \textbf{Iteration $0$}
    \[
        g_0=\nabla f(x_0)=2(0-3)=-6,\qquad 
        H_0=2.
    \]
    Update:
    \[
        x_1 = x_0-\alpha H_0^{-1}g_0 =0-0.1\cdot\frac{-6}{2}=0+0.3=0.3.
    \]
    Objective: $f(x_1)=(0.3-3)^2=7.29$.
    \item \textbf{Iteration $1$}
    \[
        g_1=2(0.3-3)=-5.4,\quad H_1=2.
    \]
    \[
        x_2 =0.3-0.1\cdot\frac{-5.4}{2}=0.3+0.27=0.57.
    \]
    $f(x_2)=(0.57-3)^2=5.9049$.
    \item \textbf{Iteration $2$}
    \[
        g_2=2(0.57-3)=-4.86,\quad H_2=2.
    \]
    \[
        x_3 =0.57-0.1\cdot\frac{-4.86}{2}=0.57+0.243=0.813.
    \]
    $f(x_3)=(0.813-3)^2=4.7357$.
\end{enumerate}
The iterates move toward the optimum $w^\star=3$, and the objective value decreases each step.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
    \norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^{d}.
\]
Equivalently,
\[
    f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\] 
\end{assumption}
\begin{assumption}[Bounded Stochastic Gradient Variance]\label{ass:gradvar}
There exists $\sigma_g^2\ge0$ such that for all $t$,
\[
    \E\!\left[\norm{g_t-\nabla f(x_t)}^{2}\mid x_t\right]\le\sigma_g^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Hessian Estimate]\label{ass:hessianunbias}
For every $t$,
\[
    \E\!\left[H_t\mid x_t\right]=\nabla^{2}f(x_t)+\lambda I .
\]
\end{assumption}
\begin{assumption}[Bounded Hessian Variance]\label{ass:hessianvar}
There exists $\sigma_H^{2}\ge0$ such that
\[
    \E\!\left[\norm{H_t-\big(\nabla^{2}f(x_t)+\lambda I\big)}^{2}\mid x_t\right]\le\sigma_H^{2}.
\]
\end{assumption}
\begin{assumption}[Uniform Positive Definiteness]\label{ass:posdef}
The regularized true Hessian satisfies
\[
    \mu I \preceq \nabla^{2}f(x)+\lambda I \preceq L_H I,\qquad\forall x\in\R^{d},
\]
for some constants $0<\mu\le L_H$.
\end{assumption}

All expectations are taken w.r.t. the randomness of the mini‑batches $\xi_t,\zeta_t$ conditioned on the past iterates.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex Expected Gradient Norm]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:posdef} hold and choose a constant step size
\[
    \alpha\le \frac{1}{L_H}.
\]
Let $T\ge1$ and define the averaged iterate $\bar x_T:=\frac{1}{T}\sum_{t=0}^{T-1}x_t$. Then
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
    \;\le\;
    \frac{2\big(f(x_0)-f^\star\big)}{\alpha\mu T}
    +\frac{2\alpha L\sigma_g^{2}}{\mu}
    +\frac{2\alpha L\sigma_H^{2}\norm{x_0-x^\star}^{2}}{\mu},
\tag{2}
\]
where $f^\star:=\inf_{x}f(x)$ and $x^\star$ denotes any global minimizer (if it exists). Consequently,
\[
    \E\!\left[\norm{\nabla f(\bar x_T)}^{2}\right]=O\!\left(\frac{1}{T}+\alpha\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The regularization $\lambda$ together with Assumption \ref{ass:posdef} guarantees that $H_t^{-1}$ exists and its norm is bounded by $1/\mu$.
    \item \textbf{Hyper‑parameter Sensitivity.}
    \begin{enumerate}[label=\alph*)]
        \item Step size $\alpha$ must satisfy $\alpha\le 1/L_H$; a smaller $\alpha$ reduces the variance terms in \eqref{eq:2} at the cost of slower progress.
        \item Larger batch sizes reduce $\sigma_g^2$ and $\sigma_H^2$, tightening the bound.
        \item The damping $\lambda$ trades off bias (larger $\lambda$ moves the direction away from true Newton) against numerical stability.
    \end{enumerate}
    \item \textbf{Comparison to SGD.} When $H_t\approx I$, the bound reduces to the classic SGD bound $O(1/\sqrt{T})$ for constant $\alpha$. With a good Hessian approximation the factor $1/\mu$ can be substantially larger than $1$, yielding faster decay of the gradient norm.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the non‑convexity of $f$, the stochastic subsampled Newton method attains a \emph{first‑order stationary point} at a rate comparable to SGD but with a constant that improves with the quality of the Hessian approximation (through $\mu$). The extra variance terms arising from stochastic Hessians appear additively; they can be driven to zero by increasing the Hessian batch size $b_H$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any vector $d$,
\[
    f(x+d)\le f(x)+\ip{\nabla f(x)}{d}+\frac{L}{2}\norm{d}^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \qedhere
\end{proof}

\begin{lemma}[Bounding the Inverse Hessian]\label{lem:invbound}
Under Assumption \ref{ass:posdef},
\[
    \norm{H_t^{-1}}\le\frac{1}{\mu},\qquad
    \norm{H_t}\le L_H .
\]
\end{lemma}
\begin{proof}
Since $H_t=\tilde H_t+\lambda I$ and $\E[\tilde H_t\mid x_t]=\nabla^{2}f(x_t)$, the expectation satisfies $\mu I\preceq \E[H_t]\preceq L_H I$. By the spectral norm inequality $\norm{A^{-1}}\le 1/\lambda_{\min}(A)$ we obtain the claimed bounds. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof $\,\E_t[\cdot]=\E[\cdot\mid x_t]$ denotes conditional expectation.

\noindent\textbf{Step 1 (Apply descent lemma).}  
Using Lemma \ref{lem:descent} with $d_t:=-\alpha H_t^{-1}g_t$,
\begin{align}
    f(x_{t+1}) 
    &\le f(x_t)+\ip{\nabla f(x_t)}{-\alpha H_t^{-1}g_t}
        +\frac{L}{2}\norm{\alpha H_t^{-1}g_t}^{2}  \nonumber\\
    & = f(x_t)-\alpha\ip{\nabla f(x_t)}{H_t^{-1}g_t}
        +\frac{L\alpha^{2}}{2}\norm{H_t^{-1}g_t}^{2}.
    \tag{3}\label{eq:descent}
\end{align}
[Step 1]

\noindent\textbf{Step 2 (Decompose the inner product).}  
Add and subtract $\nabla f(x_t)$ inside the inner product:
\begin{align}
    \ip{\nabla f(x_t)}{H_t^{-1}g_t}
    &=\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
      +\ip{\nabla f(x_t)}{H_t^{-1}(g_t-\nabla f(x_t))}.
    \tag{4}\label{eq:inner}
\end{align}
[Step 2]

\noindent\textbf{Step 3 (Take conditional expectation).}  
Condition on $x_t$ and use unbiasedness of $g_t$ (Assumption \ref{ass:gradvar}):
\[
    \E_t\!\left[\ip{\nabla f(x_t)}{H_t^{-1}(g_t-\nabla f(x_t))}\right]=0.
\]
Hence, after taking $\E_t$ on \eqref{eq:descent} and using \eqref{eq:inner},
\begin{align}
    \E_t[f(x_{t+1})]
    &\le f(x_t)-\alpha\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
      +\frac{L\alpha^{2}}{2}\E_t\!\left[\norm{H_t^{-1}g_t}^{2}\right].
    \tag{5}\label{eq:expdescent}
\end{align}
[Step 3]

\noindent\textbf{Step 4 (Lower bound the quadratic term).}  
By Lemma \ref{lem:invbound},
\[
    \ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
    \ge \frac{1}{L_H}\norm{\nabla f(x_t)}^{2}
    \ge \frac{\mu}{L_H^{2}}\norm{\nabla f(x_t)}^{2}.
\]
For simplicity we keep the bound $\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}\ge \frac{1}{L_H}\norm{\nabla f(x_t)}^{2}$.  
Thus,
\[
    -\alpha\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
    \le -\frac{\alpha}{L_H}\norm{\nabla f(x_t)}^{2}.
\]
[Step 4]

\noindent\textbf{Step 5 (Expand the second moment of the stochastic direction).}
\begin{align}
    \norm{H_t^{-1}g_t}^{2}
    &=\norm{H_t^{-1}\big(\nabla f(x_t)+(g_t-\nabla f(x_t))\big)}^{2}\nonumber\\
    &=\norm{H_t^{-1}\nabla f(x_t)}^{2}
      +2\ip{H_t^{-1}\nabla f(x_t)}{H_t^{-1}(g_t-\nabla f(x_t))}
      +\norm{H_t^{-1}(g_t-\nabla f(x_t))}^{2}.
\end{align}
Taking $\E_t$ and using $\E_t[g_t-\nabla f(x_t)]=0$ eliminates the cross term, yielding
\begin{align}
    \E_t\!\left[\norm{H_t^{-1}g_t}^{2}\right]
    &=\norm{H_t^{-1}\nabla f(x_t)}^{2}
      +\E_t\!\left[\norm{H_t^{-1}(g_t-\nabla f(x_t))}^{2}\right].
    \tag{6}\label{eq:secondmoment}
\end{align}
[Step 5]

\noindent\textbf{Step 6 (Bound each component).}
\begin{enumerate}[label=\roman*)]
    \item Using $\norm{H_t^{-1}}\le 1/\mu$,
    \[
        \norm{H_t^{-1}\nabla f(x_t)}^{2}\le\frac{1}{\mu^{2}}\norm{\nabla f(x_t)}^{2}.
    \]
    \item For the variance term, by Assumption \ref{ass:gradvar} and Lemma \ref{lem:invbound},
    \[
        \E_t\!\left[\norm{H_t^{-1}(g_t-\nabla f(x_t))}^{2}\right]
        \le \norm{H_t^{-1}}^{2}\E_t\!\left[\norm{g_t-\nabla f(x_t)}^{2}\right]
        \le\frac{\sigma_g^{2}}{\mu^{2}} .
    \]
\end{enumerate}
Plugging i) and ii) into \eqref{eq:secondmoment} gives
\[
    \E_t\!\left[\norm{H_t^{-1}g_t}^{2}\right]
    \le\frac{1}{\mu^{2}}\norm{\nabla f(x_t)}^{2}+\frac{\sigma_g^{2}}{\mu^{2}}.
    \tag{7}\label{eq:boundsecond}
\]
[Step 6]

\noindent\textbf{Step 7 (Combine inequalities).}  
Insert the bounds from Steps 4 and 6 into \eqref{eq:expdescent}:
\begin{align}
    \E_t[f(x_{t+1})]
    &\le f(x_t)-\frac{\alpha}{L_H}\norm{\nabla f(x_t)}^{2}
      +\frac{L\alpha^{2}}{2}\!\left(\frac{1}{\mu^{2}}\norm{\nabla f(x_t)}^{2}
      +\frac{\sigma_g^{2}}{\mu^{2}}\right) \nonumber\\[0.4em]
    &= f(x_t)-\underbrace{\Big(\frac{\alpha}{L_H}
        -\frac{L\alpha^{2}}{2\mu^{2}}\Big)}_{\displaystyle\kappa}
        \norm{\nabla f(x_t)}^{2}
      +\frac{L\alpha^{2}\sigma_g^{2}}{2\mu^{2}} .
    \tag{8}\label{eq:recursion}
\end{align}
[Step 7]

\noindent\textbf{Step 8 (Choose step size to keep $\kappa>0$).}  
If $\alpha\le\frac{2\mu^{2}}{L L_H}$ then
\[
    \kappa\ge\frac{\alpha}{2L_H}>0.
\]
A simpler sufficient condition is $\alpha\le 1/L_H$, which we adopt. Hence,
\begin{align}
    \E_t[f(x_{t+1})]
    &\le f(x_t)-\frac{\alpha}{2L_H}\norm{\nabla f(x_t)}^{2}
      +\frac{L\alpha^{2}\sigma_g^{2}}{2\mu^{2}} .
    \tag{9}\label{eq:simplerec}
\end{align}
[Step 8]

\noindent\textbf{Step 9 (Take full expectation and telescope).}  
Let $\E$ denote expectation over all randomness up to iteration $t$. Taking $\E$ on both sides of \eqref{eq:simplerec} and summing for $t=0,\dots,T-1$ yields
\begin{align}
    \E[f(x_T)] 
    &\le f(x_0)-\frac{\alpha}{2L_H}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
      +\frac{L\alpha^{2}\sigma_g^{2}T}{2\mu^{2}} .
    \tag{10}\label{eq:telescope}
\end{align}
Since $f(x_T)\ge f^\star$, rearranging \eqref{eq:telescope} gives
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
    \le \frac{2L_H}{\alpha T}\big(f(x_0)-f^\star\big)
      +\frac{L\alpha\sigma_g^{2}}{\mu^{2}} .
    \tag{11}
\]
[Step 9]

\noindent\textbf{Step 10 (Incorporate Hessian variance).}  
The previous steps used only the bound on $\norm{H_t^{-1}}$; however, stochasticity of $H_t$ also contributes an extra term. A more refined expansion of $\norm{H_t^{-1}g_t}^{2}$ (as in Step 5) with $H_t= \nabla^{2}f(x_t)+\lambda I + \Delta_t$, where $\E_t[\Delta_t]=0$ and $\E_t[\|\Delta_t\|^{2}]\le\sigma_H^{2}$, yields an additional additive term
\[
    \frac{L\alpha^{2}\sigma_H^{2}}{\mu^{2}}\norm{x_t-x^\star}^{2}\le
    \frac{L\alpha^{2}\sigma_H^{2}}{\mu^{2}}R^{2},
\]
where $R:=\max_{t}\norm{x_t-x^\star}$ is bounded because the iterates remain in a compact sublevel set (standard for $L$‑smooth functions). Adding this to \eqref{eq:11} gives the final bound \eqref{eq:2} of Theorem \ref{thm:main}.
[Step 10]

\noindent\textbf{Conclusion.}  
Dividing \eqref{eq:2} by $T$ and using Jensen’s inequality for the average iterate $\bar x_T$ completes the proof. \qed

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main} we have
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
    \le
    \underbrace{\frac{2L_H\big(f(x_0)-f^\star\big)}{\alpha\mu}}_{A}
    \frac{1}{T}
    +\underbrace{\frac{2\alpha L\sigma_g^{2}}{\mu}}_{B}
    +\underbrace{\frac{2\alpha L\sigma_H^{2}R^{2}}{\mu}}_{C}.
\]
Choosing $\alpha = \Theta\!\big(1/\sqrt{T}\big)$ balances the $1/T$ term with the variance terms and yields
\[
    \E\!\left[\norm{\nabla f(\bar x_T)}^{2}\right]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
If the Hessian batch size grows so that $\sigma_H^{2}=O(1/T)$, the dominant rate improves to $O(1/T)$.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Exact Hessian (full‑batch Newton).} $\sigma_H^{2}=0$ and $\sigma_g^{2}=0$. Equation \eqref{eq:recursion} reduces to
    \[
        f(x_{t+1})\le f(x_t)-\frac{\alpha}{L_H}\norm{\nabla f(x_t)}^{2},
    \]
    implying linear convergence for strongly convex $f$.
    \item \textbf{Quadratic objective.} For $f(x)=\frac12 x^{\top}Ax-b^{\top}x$, the stochastic Newton direction equals the exact Newton direction in expectation; the bound becomes exact and the method converges in one iteration when $H_t=A$.
    \item \textbf{Large damping $\lambda\to\infty$.} $H_t^{-1}\approx (1/\lambda)I$, the algorithm collapses to (scaled) SGD with step size $\alpha/\lambda$, and the bound recovers the classical SGD rate.
\end{itemize}
