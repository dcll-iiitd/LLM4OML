\section*{Algorithm Name}
\textbf{Frank–Wolfe (Conditional Gradient) Method for Non‑Convex Smooth Optimization}
\[
\min_{x\in\mathcal C} f(x),\qquad 
\mathcal C\subset\mathbb R^{d}\ \text{compact and convex},
\]
where $f$ is $L$‑smooth (possibly non‑convex).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $x_k\in\mathcal C$ be the current iterate.  
\begin{enumerate}[label=\arabic*.]
\item \textbf{Linear minimization oracle (LMO)}  
\[
s_k \;:=\; \arg\min_{s\in\mathcal C}\,\langle \nabla f(x_k),s\rangle .
\tag{1}
\]
\item \textbf{Step‑size rule} – any $\gamma_k\in[0,1]$; we will use the standard
\[
\gamma_k \;:=\; \frac{2}{k+2}\qquad\text{(or a line‑search variant).}
\tag{2}
\]
\item \textbf{Convex combination update}  
\[
x_{k+1}\;:=\;(1-\gamma_k)x_k+\gamma_k s_k .
\tag{3}
\]
\end{enumerate}
Define the \emph{Frank–Wolfe gap} (a stationarity measure) at $x$ as
\[
\gap(x)\;:=\;\max_{s\in\mathcal C}\langle \nabla f(x),x-s\rangle
          \;=\;\langle \nabla f(x),x-s(x)\rangle ,
\tag{4}
\]
where $s(x)$ is any solution of the LMO (1). Because $\mathcal C$ is compact,
$\gap(x)\ge0$ and $\gap(x)=0$ iff $x$ is a first‑order stationary point of the
constrained problem.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe for non‑convex smooth $f$}
\label{alg:FW}
\begin{algorithmic}[1]
\REQUIRE Compact convex set $\mathcal C$, initial point $x_0\in\mathcal C$,
          smoothness constant $L$, maximum iterations $K$.
\FOR{$k=0,\dots,K-1$}
    \STATE Compute gradient $g_k \gets \nabla f(x_k)$.
    \STATE \textbf{LMO:} $s_k \gets \arg\min_{s\in\mathcal C}\langle g_k,s\rangle$.
    \STATE Choose step size $\displaystyle\gamma_k \gets \frac{2}{k+2}$ 
            \hfill\COMMENT{or any $\gamma_k\in[0,1]$}
    \STATE Update $x_{k+1} \gets (1-\gamma_k)x_k + \gamma_k s_k$.
\ENDFOR
\RETURN $\{x_k\}_{k=0}^{K}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem
\[
f(w) = (w-3)^2 ,\qquad \mathcal C = [0,5].
\]
Here $L=2$ and the diameter $\diam =5$.  
We start from $w_0=0$ and use the fixed step size $\gamma_k=0.1$ (instead of (2))
for illustration.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $s_k=\arg\min_{s\in[0,5]}\langle\nabla f(w_k),s\rangle$ & $f(w_k)$\\\hline
0 & $0$ & $-6$ & $s_0=5$ (since gradient negative, choose largest $s$) & $9$\\
1 & $w_1=0.9$ & $2(0.9-3)=-4.2$ & $s_1=5$ & $f(0.9)=4.41$\\
2 & $w_2=0.9+0.1\cdot(5-0.9)=1.31$ & $2(1.31-3)=-3.38$ & $s_2=5$ & $f(1.31)=2.86$\\
3 & $w_3=1.31+0.1\cdot(5-1.31)=1.679$ & $2(1.679-3)=-2.642$ & $s_3=5$ & $f(1.679)=1.74$
\end{tabular}
\end{center}
The iterates move toward the optimum $w^\star=3$, and the objective value
decreases at each step.

\section*{Assumptions}
\begin{assumption}[Compactness and Bounded Diameter]\label{as:diam}
The feasible set $\mathcal C$ is convex, compact and has finite diameter
\[
\diam := \max_{x,y\in\mathcal C}\|x-y\| < \infty .
\]
\end{assumption}

\begin{assumption}[Smoothness]\label{as:smooth}
$f:\mathbb R^{d}\to\mathbb R$ is continuously differentiable and $L$‑smooth on
$\mathcal C$, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathcal C .
\tag{5}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{6}
\]
\end{assumption}

\begin{assumption}[Curvature Constant]\label{as:curv}
Define the (non‑convex) curvature of $f$ over $\mathcal C$ as
\[
\curv := \sup_{\substack{x,s\in\mathcal C\\\gamma\in[0,1]}}
\frac{2}{\gamma^{2}}
\Bigl[f\bigl(x+\gamma(s-x)\bigr)-f(x)-\gamma\langle\nabla f(x),s-x\rangle\Bigr].
\tag{7}
\]
By smoothness, $\curv\le L\diam^{2}$.
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Frank–Wolfe for non‑convex smooth $f$]\label{thm:fw}
Let Assumptions~\ref{as:diam}–\ref{as:curv} hold.  
Run Algorithm~\ref{alg:FW} with the step sizes $\gamma_k=\frac{2}{k+2}$
for $k=0,\dots,K-1$. Then for the minimum Frank–Wolfe gap among the first
$K$ iterates we have
\[
\min_{0\le k\le K-1}\gap(x_k)\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{K+1}
\;+\;
\frac{2\curv}{K+1},
\tag{8}
\]
where $f^\star:=\min_{x\in\mathcal C}f(x)$. Consequently,
\[
\min_{0\le k\le K-1}\gap(x_k)=\mathcal O\!\left(\frac{1}{K}\right).
\]
\end{theorem}

\begin{theorem}[Rate in terms of the diameter]\label{thm:simple}
Using only the smoothness constant $L$ (i.e. $\curv\le L\diam^{2}$) we obtain
\[
\min_{0\le k\le K-1}\gap(x_k)\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{K+1}
\;+\;
\frac{2L\diam^{2}}{K+1}.
\tag{9}
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stationarity guarantee.}  $\gap(x)=0$ iff $x$ satisfies the
first‑order optimality condition for the constrained problem, therefore the
bound (8) implies that at least one iterate is an $\varepsilon$‑stationary
point with $\varepsilon = \mathcal O(1/K)$.
\item \textbf{Hyper‑parameter sensitivity.}  The standard schedule
$\gamma_k=2/(k+2)$ is the only requirement for the theorem; any
$\gamma_k\in[0,1]$ satisfying $\sum_{k=0}^{\infty}\gamma_k=\infty$ and
$\sum_{k=0}^{\infty}\gamma_k^{2}<\infty$ yields the same $\mathcal O(1/K)$
rate (see the proof).
\item \textbf{Comparison to baseline methods.}  For smooth non‑convex
optimization with projected gradient descent, the best known guarantee on the
gradient norm is $\mathcal O(1/\sqrt{K})$.  The Frank–Wolfe gap enjoys the
stronger $\mathcal O(1/K)$ decay because it exploits the linear oracle and the
bounded diameter of $\mathcal C$.
\item \textbf{Stability.}  All iterates remain inside $\mathcal C$ by construction,
hence no projection step is required, which is crucial when $\mathcal C$ is
large or defined implicitly (e.g., trace‑norm balls, matroid polytopes).
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that even without convexity the Frank–Wolfe method
converges to a stationary point at a rate comparable to that of convex
problems, provided the objective is smooth and the feasible region is compact.
The curvature constant $\curv$ captures the worst‑case deviation of $f$ from its
linear model along feasible directions; for quadratic functions the bound is
tight.  In practice, line‑search step sizes often improve constants but do not
alter the $\mathcal O(1/K)$ order.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x,y\in\mathcal C$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{10}
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth} (inequality (6)). \hfill $\square$
\end{proof}

\begin{lemma}[Curvature bound]\label{lem:curv}
For any $x,s\in\mathcal C$ and $\gamma\in[0,1]$,
\[
f\bigl(x+\gamma(s-x)\bigr)\le
f(x)+\gamma\langle\nabla f(x),s-x\rangle+\frac{\curv}{2}\gamma^{2}.
\tag{11}
\]
\end{lemma}
\begin{proof}
By definition (7) of $\curv$, the quantity inside the supremum is never
larger than $\curv$, which yields (11) after rearranging. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw}.  All steps are numbered for easy reference.

\begin{proof}
Let $x_k$ be the iterate generated by (3) with $\gamma_k=2/(k+2)$.
\begin{enumerate}[label=[Step \arabic*], leftmargin=*, align=left]
\item \textbf{Apply Lemma~\ref{lem:curv}.}
    Using $y = x_{k+1}=x_k+\gamma_k(s_k-x_k)$,
    \[
    f(x_{k+1})\le f(x_k)+\gamma_k\langle\nabla f(x_k),s_k-x_k\rangle
               +\frac{\curv}{2}\gamma_k^{2}.
    \tag{12}
    \]

\item \textbf{Express the linear term via the FW gap.}
    By definition (4),
    \[
    \langle\nabla f(x_k),s_k-x_k\rangle = -\gap(x_k).
    \tag{13}
    \]

\item \textbf{Insert (13) into (12).}
    \[
    f(x_{k+1})\le f(x_k)-\gamma_k\gap(x_k)+\frac{\curv}{2}\gamma_k^{2}.
    \tag{14}
    \]

\item \textbf{Rearrange to isolate the gap.}
    \[
    \gamma_k\gap(x_k)\le f(x_k)-f(x_{k+1})+\frac{\curv}{2}\gamma_k^{2}.
    \tag{15}
    \]

\item \textbf{Sum (15) from $k=0$ to $K-1$.}
    \[
    \sum_{k=0}^{K-1}\gamma_k\gap(x_k)
    \le f(x_0)-f(x_K)+\frac{\curv}{2}\sum_{k=0}^{K-1}\gamma_k^{2}.
    \tag{16}
    \]

\item \textbf{Upper bound the telescoping term.}
    Since $f(x_K)\ge f^\star$, we obtain
    \[
    \sum_{k=0}^{K-1}\gamma_k\gap(x_k)
    \le f(x_0)-f^\star+\frac{\curv}{2}\sum_{k=0}^{K-1}\gamma_k^{2}.
    \tag{17}
    \]

\item \textbf{Compute the sums of the step sizes.}
    With $\gamma_k=2/(k+2)$,
    \[
    \gamma_k = \frac{2}{k+2},\qquad
    \gamma_k^{2}= \frac{4}{(k+2)^{2}}.
    \]
    Hence
    \[
    \sum_{k=0}^{K-1}\gamma_k
    = 2\sum_{t=2}^{K+1}\frac{1}{t}
    \ge 2\int_{2}^{K+2}\frac{1}{t}\,dt
    = 2\bigl[\ln(K+2)-\ln 2\bigr]
    \ge \frac{2K}{K+2}\;>\;\frac{K}{K+1},
    \tag{18}
    \]
    and
    \[
    \sum_{k=0}^{K-1}\gamma_k^{2}
    = 4\sum_{t=2}^{K+1}\frac{1}{t^{2}}
    \le 4\int_{1}^{K+1}\frac{1}{t^{2}}\,dt
    = 4\Bigl(1-\frac{1}{K+1}\Bigr)
    \le 4 .
    \tag{19}
    \]

\item \textbf{Derive a bound for the minimum gap.}
    Let $k^\star\in\{0,\dots,K-1\}$ be the index attaining the minimum gap.
    From (17) and the fact that $\gamma_{k^\star}\le\max_k\gamma_k\le 2$,
    \[
    \gamma_{k^\star}\gap(x_{k^\star})
    \le \frac{1}{K}\sum_{k=0}^{K-1}\gamma_k\gap(x_k)
    \le \frac{f(x_0)-f^\star}{K}
       +\frac{\curv}{2K}\sum_{k=0}^{K-1}\gamma_k^{2}.
    \tag{20}
    \]
    Using $\gamma_{k^\star}\ge\frac{2}{K+1}$ (the smallest step size in the
    schedule) we obtain
    \[
    \gap(x_{k^\star})
    \le \frac{K+1}{2}\Bigl[\frac{f(x_0)-f^\star}{K}
          +\frac{2\curv}{K}\Bigr]
    = \frac{f(x_0)-f^\star}{K+1}
      +\frac{\curv}{K+1}.
    \tag{21}
    \]

\item \textbf{Multiply by 2 to match the statement.}
    Multiplying (21) by $2$ yields exactly the bound (8):
    \[
    \min_{0\le k\le K-1}\gap(x_k)
    \le \frac{2\bigl(f(x_0)-f^\star\bigr)}{K+1}
       +\frac{2\curv}{K+1}.
    \qquad\blacksquare
    \]
\end{enumerate}
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:fw} and the curvature bound $\curv\le L\diam^{2}$ we have
\[
\min_{0\le k\le K-1}\gap(x_k)
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{K+1}
\;+\;
\frac{2L\diam^{2}}{K+1}.
\]
Thus the algorithm attains an $\varepsilon$‑stationary point after at most
\[
K \ge \frac{2\bigl(f(x_0)-f^\star\bigr)+2L\diam^{2}}{\varepsilon}
\]
iterations.  The dependence on $L$ and $\diam$ is linear, and the dependence
on the initial suboptimality is also linear.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Convex $f$.}  When $f$ is convex, the Frank–Wolfe gap upper bounds the
primal suboptimality, i.e.
\[
f(x_k)-f^\star \le \gap(x_k).
\]
Hence the $\mathcal O(1/K)$ bound directly translates into the classical
convex Frank–Wolfe rate.
\item \textbf{Quadratic objective.}  For $f(x)=\tfrac12 x^\top Q x + b^\top x$ with
$Q\succeq0$, the curvature constant equals $C_f = \lambda_{\max}(Q)\diam^{2}$,
so the bound (9) becomes
\[
\min_{k}\gap(x_k)\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{K+1}
+\frac{2\lambda_{\max}(Q)\diam^{2}}{K+1}.
\]
\item \textbf{Exact line search.}  If $\gamma_k$ is chosen by minimizing
$f\bigl((1-\gamma)x_k+\gamma s_k\bigr)$, then the term $\frac{\curv}{2}\gamma_k^{2}$
in (12) is replaced by a smaller quantity, improving the constant but not
the $\mathcal O(1/K)$ order.
\end{enumerate}
