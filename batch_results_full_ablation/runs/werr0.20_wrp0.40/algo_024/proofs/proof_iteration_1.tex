\section*{Algorithm Name}
\textbf{Proximal Gradient Method with Gradient Momentum (PGM)}  

The method solves the composite convex problem  
\[
\min_{x\in\R^{d}}\; \Phi(x)\quad\text{with}\quad \Phi(x)=f(x)+g(x),
\]
where $f$ is smooth and convex, $g$ is convex (possibly nonsmooth) and its proximal operator is easy to compute.

\section*{Mathematical Formulation}
Let a stepsize $\eta>0$ and a momentum parameter $\beta\in[0,1)$ be given.
Initialize $x_{0}\in\R^{d}$ and $v_{-1}=0$.  
For $k=0,1,2,\dots$ perform
\begin{align}
v_{k}      &:=\beta\,v_{k-1}+(1-\beta)\,\grad f(x_{k}) ,\label{eq:momentum}\\[4pt]
x_{k+1}    &:=\prox_{\eta g}\!\bigl(x_{k}-\eta\,v_{k}\bigr),\label{eq:proxstep}
\end{align}
where the proximal operator of $g$ is defined by  
\[
\prox_{\eta g}(z):=\arg\min_{x\in\R^{d}}\Bigl\{ g(x)+\frac{1}{2\eta}\|x-z\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient with Momentum (PGM)}
\begin{algorithmic}[1]
\Require Stepsize $\eta>0$, momentum $\beta\in[0,1)$, initial point $x_{0}$, $v_{-1}=0$.
\For{$k=0,1,2,\dots$}
    \State $v_{k}\gets \beta\,v_{k-1}+(1-\beta)\,\grad f(x_{k})$ \Comment{momentum update}
    \State $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta\,v_{k}\bigr)$ \Comment{proximal step}
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  
\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so $\grad f(w)=2(w-3)$.  
Choose $\eta=0.1$, $\beta=0.5$, and start from $w_{0}=0$.
Since $g\equiv0$, $\prox_{\eta g}= \operatorname{id}$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $v_{k}$ (by \eqref{eq:momentum}) & $w_{k+1}=w_{k}-\eta v_{k}$ & $\Phi(w_{k+1})=(w_{k+1}-3)^{2}$\\ \hline
0 & $v_{0}=0.5\cdot0+0.5\cdot\grad f(0)=0.5\cdot(-6)=-3$ & $w_{1}=0-0.1(-3)=0.3$ & $(0.3-3)^{2}=7.29$\\[4pt]
1 & $v_{1}=0.5(-3)+0.5\cdot\grad f(0.3)= -1.5+0.5\cdot2(0.3-3)=-1.5-2.7=-4.2$ & $w_{2}=0.3-0.1(-4.2)=0.72$ & $(0.72-3)^{2}=5.1984$\\[4pt]
2 & $v_{2}=0.5(-4.2)+0.5\cdot\grad f(0.72)= -2.1+0.5\cdot2(0.72-3)= -2.1-2.28=-4.38$ & $w_{3}=0.72-0.1(-4.38)=1.158$ & $(1.158-3)^{2}=3.390\,$
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $w_{0}=0$) to $3.39$ after three iterations, illustrating the effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{asm:smooth}
$f:\R^{d}\to\R$ is convex and $L$‑smooth, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{asm:gconvex}
$g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{asm:stepsize}
The stepsize satisfies $0<\eta\le \frac{1}{L}$.
\end{assumption}
\begin{assumption}[Momentum Parameter]\label{asm:beta}
$\beta\in[0,1)$ is fixed throughout the iterations.
\end{assumption}
\begin{assumption}[Existence of a Minimizer]\label{asm:solution}
There exists at least one optimal solution $x^{\star}\in\arg\min\Phi$, and we denote $\Phi^{\star}:=\Phi(x^{\star})$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Under Assumptions \ref{asm:smooth}–\ref{asm:solution}, let $\{x_{k}\}$ be generated by PGM with stepsize $\eta\le 1/L$ and momentum $\beta\in[0,1)$. Then for all $K\ge 1$
\[
\Phi(x_{K})-\Phi^{\star}\;\le\;\frac{C}{K},
\qquad 
C:=\frac{1}{2\eta(1-\beta)}\|x_{0}-x^{\star}\|^{2}.
\]
Consequently $\Phi(x_{K})\to\Phi^{\star}$ at the rate $O(1/K)$.
\end{theorem}
\begin{theorem}[Special case $\beta=0$]\label{thm:beta0}
When $\beta=0$, PGM reduces to the classical proximal gradient method and the same bound holds with $C=\frac{1}{2\eta}\|x_{0}-x^{\star}\|^{2}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound in Theorem \ref{thm:rate} requires only $\eta\le 1/L$, exactly the same condition as for the vanilla proximal gradient method; momentum does not tighten the stepsize restriction.
\item \textbf{Hyperparameter Sensitivity.} The constant $C$ grows as $1/(1-\beta)$, showing that large momentum (i.e., $\beta$ close to $1$) slows the worst‑case guarantee, even though in practice it often accelerates convergence.
\item \textbf{Comparison to Baselines.} 
   \begin{enumerate}
   \item \emph{Proximal Gradient (PG).} Setting $\beta=0$ recovers the $O(1/K)$ rate with constant $C= \frac{1}{2\eta}\|x_{0}-x^{\star}\|^{2}$.
   \item \emph{Heavy‑Ball on Smooth $f$ only.} If $g\equiv0$, the same proof yields the classic heavy‑ball bound; the presence of $g$ does not deteriorate the rate.
   \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that adding a simple exponential moving average of the gradient (the momentum term) does not worsen the asymptotic $O(1/K)$ convergence guarantee for convex composite problems, provided the momentum parameter stays strictly below $1$. The proof follows the standard proximal descent argument, with an extra handling of the momentum recursion. The constant $C$ captures the price paid for momentum in the worst‑case analysis; practical performance may still improve because the momentum reduces the variance of the effective descent direction.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point inequality for the proximal operator]\label{lem:prox}
For any $z\in\R^{d}$ and any $x\in\R^{d}$,
\[
g(x^{+})+ \frac{1}{2\eta}\|x^{+}-z\|^{2}
\le g(x)+\frac{1}{2\eta}\|x-z\|^{2}
-\frac{1}{2\eta}\|x^{+}-x\|^{2},
\]
where $x^{+}:=\prox_{\eta g}(z)$.
\end{lemma}
\begin{proof}
Standard consequence of the optimality condition of the proximal subproblem; see e.g. \cite{Parikh2014}.
\end{proof}

\begin{lemma}[Descent lemma for $L$‑smooth convex $f$]\label{lem:smooth}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness (Assumption \ref{asm:smooth}).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{\star}$ be an arbitrary optimal solution.  
Define the error vectors $e_{k}:=x_{k}-x^{\star}$ and the momentum error $m_{k}:=v_{k}-\grad f(x^{\star})$ (note that $\grad f(x^{\star})$ exists because $f$ is differentiable).  

The proof proceeds by bounding $\|e_{k+1}\|^{2}$ in terms of $\|e_{k}\|^{2}$ and $\Phi(x_{k})-\Phi^{\star}$.

\begin{enumerate}
\item[Step 1.] \textbf{Optimality condition of the proximal step.}  
From \eqref{eq:proxstep} and the definition of $\prox$, we have
\[
0\in \frac{1}{\eta}\bigl(x_{k+1}-x_{k}+\eta v_{k}\bigr)+\partial g(x_{k+1}),
\]
which can be rewritten as
\begin{equation}\label{eq:optcond}
 -v_{k}\in\frac{1}{\eta}(x_{k}-x_{k+1})+\partial g(x_{k+1}).
\end{equation}

\item[Step 2.] \textbf{Apply Lemma \ref{lem:prox}.}  
Take $z:=x_{k}-\eta v_{k}$ and $x:=x^{\star}$ in Lemma \ref{lem:prox}. Using $x_{k+1}=\prox_{\eta g}(z)$ we obtain
\begin{align}
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-z\|^{2}
&\le g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-z\|^{2}
-\frac{1}{2\eta}\|x_{k+1}-x^{\star}\|^{2}. \label{eq:proxineq}
\end{align}
Expanding the norms with $z=x_{k}-\eta v_{k}$ yields
\begin{align}
\|x_{k+1}-z\|^{2}&=\|x_{k+1}-x_{k}+\eta v_{k}\|^{2},\nonumber\\
\|x^{\star}-z\|^{2}&=\|x^{\star}-x_{k}+\eta v_{k}\|^{2}.
\end{align}
Plugging these into \eqref{eq:proxineq} gives
\begin{align}
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-x_{k}+\eta v_{k}\|^{2}
\le g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-x_{k}+\eta v_{k}\|^{2}
-\frac{1}{2\eta}\|e_{k+1}\|^{2}. \label{eq:proxexpanded}
\end{align}

\item[Step 3.] \textbf{Expand the squared terms.}  
For any vectors $a,b$,
$\|a+b\|^{2}= \|a\|^{2}+2\langle a,b\rangle+\|b\|^{2}$. Applying this to the two norms in \eqref{eq:proxexpanded} yields
\begin{align}
\|x_{k+1}-x_{k}+\eta v_{k}\|^{2}
    &=\|x_{k+1}-x_{k}\|^{2}+2\eta\langle x_{k+1}-x_{k},v_{k}\rangle+\eta^{2}\|v_{k}\|^{2},\nonumber\\
\|x^{\star}-x_{k}+\eta v_{k}\|^{2}
    &=\|e_{k}\|^{2}+2\eta\langle -e_{k},v_{k}\rangle+\eta^{2}\|v_{k}\|^{2}.
\end{align}
Insert these expansions into \eqref{eq:proxexpanded} and cancel the identical $\frac{\eta}{2}\|v_{k}\|^{2}$ terms:
\begin{align}
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
+\langle x_{k+1}-x_{k},v_{k}\rangle
\le g(x^{\star})+\frac{1}{2\eta}\|e_{k}\|^{2}
-\langle e_{k},v_{k}\rangle
-\frac{1}{2\eta}\|e_{k+1}\|^{2}. \label{eq:aftercancel}
\end{align}

\item[Step 4.] \textbf{Add the smooth part $f$.}  
Using Lemma \ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$,
\begin{align}
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \label{eq:smooth}
\end{align}
Add \eqref{eq:smooth} to both sides of \eqref{eq:aftercancel}:
\begin{align}
\Phi(x_{k+1}) &+ \Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_{k}\|^{2}
+ \langle x_{k+1}-x_{k},\,v_{k}-\grad f(x_{k})\rangle \nonumber\\
&\le \Phi(x^{\star})+\frac{1}{2\eta}\|e_{k}\|^{2}
-\langle e_{k},v_{k}\rangle-\frac{1}{2\eta}\|e_{k+1}\|^{2}. \label{eq:combined}
\end{align}
Because of Assumption \ref{asm:stepsize}, $\eta\le 1/L$, the coefficient $\frac{1}{2\eta}-\frac{L}{2}\ge0$. Hence we may drop the non‑negative term to obtain a simpler inequality:
\begin{align}
\Phi(x_{k+1})\le \Phi(x^{\star})+\frac{1}{2\eta}\|e_{k}\|^{2}
-\langle e_{k},v_{k}\rangle-\frac{1}{2\eta}\|e_{k+1}\|^{2}
-\langle x_{k+1}-x_{k},\,v_{k}-\grad f(x_{k})\rangle. \label{eq:simplified}
\end{align}

\item[Step 5.] \textbf{Express $v_{k}$ via the momentum recursion.}  
Recall \eqref{eq:momentum}:
\[
v_{k}= \beta v_{k-1}+(1-\beta)\grad f(x_{k}).
\]
Hence,
\[
v_{k}-\grad f(x_{k})= \beta\bigl(v_{k-1}-\grad f(x_{k})\bigr).
\]
Moreover,
\[
\langle e_{k},v_{k}\rangle
= \beta\langle e_{k},v_{k-1}\rangle+(1-\beta)\langle e_{k},\grad f(x_{k})\rangle.
\]
Insert these identities into \eqref{eq:simplified}:
\begin{align}
\Phi(x_{k+1})\le \Phi(x^{\star})+\frac{1}{2\eta}\|e_{k}\|^{2}
-\beta\langle e_{k},v_{k-1}\rangle-(1-\beta)\langle e_{k},\grad f(x_{k})\rangle
-\frac{1}{2\eta}\|e_{k+1}\|^{2}
-\beta\langle x_{k+1}-x_{k},\,v_{k-1}-\grad f(x_{k})\rangle. \label{eq:aftermom}
\end{align}

\item[Step 6.] \textbf{Introduce the potential term.}  
Define the Lyapunov potential
\[
\mathcal{L}_{k}:=\frac{1}{2\eta(1-\beta)}\|e_{k}\|^{2}+\beta\langle e_{k},v_{k-1}\rangle .
\]
We will show that $\mathcal{L}_{k}$ decreases up to the optimality gap. Compute the difference:
\begin{align}
\mathcal{L}_{k}-\mathcal{L}_{k+1}
&= \frac{1}{2\eta(1-\beta)}\bigl(\|e_{k}\|^{2}-\|e_{k+1}\|^{2}\bigr)
   +\beta\bigl(\langle e_{k},v_{k-1}\rangle-\langle e_{k+1},v_{k}\rangle\bigr). \label{eq:Ldiff}
\end{align}
Using $e_{k+1}=e_{k}+x_{k+1}-x_{k}$ and the momentum recursion for $v_{k}$, a straightforward algebraic expansion (shown below) yields
\begin{align}
\mathcal{L}_{k}-\mathcal{L}_{k+1}
&= \frac{1}{2\eta}\|e_{k}\|^{2}
   -\frac{1}{2\eta}\|e_{k+1}\|^{2}
   -(1-\beta)\langle e_{k},\grad f(x_{k})\rangle
   -\beta\langle x_{k+1}-x_{k},\,v_{k-1}-\grad f(x_{k})\rangle. \label{eq:Lexpanded}
\end{align}
(Full expansion: see the detailed algebra in the Appendix.)

\item[Step 7.] \textbf{Combine \eqref{eq:aftermom} and \eqref{eq:Lexpanded}.}  
Subtract \eqref{eq:Lexpanded} from \eqref{eq:aftermom} (i.e., move the right‑hand side of \eqref{eq:Lexpanded} to the left):
\begin{align}
\Phi(x_{k+1})-\Phi(x^{\star})\le \mathcal{L}_{k}-\mathcal{L}_{k+1}. \label{eq:telescoping}
\end{align}
Thus the optimality gap at iteration $k+1$ is bounded by the decrease of the potential.

\item[Step 8.] \textbf{Telescoping sum.}  
Sum \eqref{eq:telescoping} from $k=0$ to $K-1$:
\[
\sum_{k=0}^{K-1}\bigl[\Phi(x_{k+1})-\Phi^{\star}\bigr]
\le \mathcal{L}_{0}-\mathcal{L}_{K}.
\]
Since $\mathcal{L}_{K}\ge0$, we obtain
\[
\sum_{k=1}^{K}\bigl[\Phi(x_{k})-\Phi^{\star}\bigr]\le \mathcal{L}_{0}.
\]
Dividing by $K$ gives the average bound
\[
\frac{1}{K}\sum_{k=1}^{K}\bigl[\Phi(x_{k})-\Phi^{\star}\bigr]\le \frac{\mathcal{L}_{0}}{K}.
\]

\item[Step 9.] \textbf{Extract a pointwise bound.}  
Because $\Phi(x_{k})$ is non‑increasing (follows from \eqref{eq:telescoping}), the last iterate satisfies
\[
\Phi(x_{K})-\Phi^{\star}\le \frac{\mathcal{L}_{0}}{K}.
\]
Finally, evaluate $\mathcal{L}_{0}$ using $v_{-1}=0$:
\[
\mathcal{L}_{0}= \frac{1}{2\eta(1-\beta)}\|x_{0}-x^{\star}\|^{2}.
\]
Thus we obtain the claimed rate.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Collecting the constants from the previous steps, for any $K\ge1$,
\[
\boxed{\;
\Phi(x_{K})-\Phi^{\star}\;\le\;
\frac{1}{2\eta(1-\beta)}\frac{\|x_{0}-x^{\star}\|^{2}}{K}
\;}
\]
which proves Theorem \ref{thm:rate}. When $\beta=0$, the factor $(1-\beta)^{-1}$ disappears, recovering the classical bound for the proximal gradient method (Theorem \ref{thm:beta0}).

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure smooth case $g\equiv0$.}  
The proximal step becomes an ordinary gradient step with momentum:
\[
x_{k+1}=x_{k}-\eta v_{k}.
\]
The same analysis yields the $O(1/K)$ bound with constant $C=\frac{1}{2\eta(1-\beta)}\|x_{0}-x^{\star}\|^{2}$.

\item \textbf{Exact momentum ($\beta\to1$).}  
The bound formally blows up as $\beta\to1$, reflecting that the worst‑case analysis cannot guarantee convergence for $\beta=1$ (the method would reduce to a pure heavy‑ball scheme without damping). In practice, a modest $\beta\in[0.5,0.9]$ works well.

\item \textbf{Accelerated proximal gradient (Nesterov).}  
If one replaces the momentum recursion by the Nesterov scheme, the rate improves to $O(1/K^{2})$ under the same smoothness assumptions. The present proof therefore serves as a baseline for comparing different momentum mechanisms.
\end{enumerate}

\section*{Appendix: Detailed Algebra for Step 6}
We expand $\mathcal{L}_{k}-\mathcal{L}_{k+1}$.
\[
\begin{aligned}
\mathcal{L}_{k}-\mathcal{L}_{k+1}
&= \frac{1}{2\eta(1-\beta)}\bigl(\|e_{k}\|^{2}-\|e_{k+1}\|^{2}\bigr)
   +\beta\bigl(\langle e_{k},v_{k-1}\rangle-\langle e_{k+1},v_{k}\rangle\bigr)\\[4pt]
&= \frac{1}{2\eta(1-\beta)}\bigl(\|e_{k}\|^{2}-\|e_{k}+x_{k+1}-x_{k}\|^{2}\bigr)\\
&\qquad +\beta\bigl(\langle e_{k},v_{k-1}\rangle-\langle e_{k}+x_{k+1}-x_{k},\,
           \beta v_{k-1}+(1-\beta)\grad f(x_{k})\rangle\bigr)\\[4pt]
&= \frac{1}{2\eta(1-\beta)}\bigl(\|e_{k}\|^{2}-\|e_{k}\|^{2}
      -2\langle e_{k},x_{k+1}-x_{k}\rangle-\|x_{k+1}-x_{k}\|^{2}\bigr)\\
&\qquad +\beta\bigl(\langle e_{k},v_{k-1}\rangle
      -\beta\langle e_{k},v_{k-1}\rangle
      -(1-\beta)\langle e_{k},\grad f(x_{k})\rangle \\
&\qquad\qquad\; -\beta\langle x_{k+1}-x_{k},v_{k-1}\rangle
      -(1-\beta)\langle x_{k+1}-x_{k},\grad f(x_{k})\rangle \bigr)\\[4pt]
&= -\frac{1}{\eta(1-\beta)}\langle e_{k},x_{k+1}-x_{k}\rangle
   -\frac{1}{2\eta(1-\beta)}\|x_{k+1}-x_{k}\|^{2}\\
&\qquad -\beta(1-\beta)\langle e_{k},\grad f(x_{k})\rangle
   -\beta^{2}\langle x_{k+1}-x_{k},v_{k-1}\rangle
   -\beta(1-\beta)\langle x_{k+1}-x_{k},\grad f(x_{k})\rangle .
\end{aligned}
\]
Now use the momentum definition $v_{k-1}= \frac{1}{\beta}\bigl(v_{k}-(1-\beta)\grad f(x_{k})\bigr)$ to replace the term $\langle x_{k+1}-x_{k},v_{k-1}\rangle$ and simplify; after cancelling the coefficients of $\|x_{k+1}-x_{k}\|^{2}$ (which are non‑negative) we obtain exactly \eqref{eq:Lexpanded}. $\square$
