\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a differentiable (possibly non‑convex) loss.  
At iteration $t$ we draw a random mini‑batch $\xi_t$ and compute an (unbiased) stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t)\in\mathbb{R}^{d}.
\]
Define the \emph{sparsification operator}
\[
\TopK{K}{g}\;=\; \begin{cases}
g_i & \text{if }|g_i|\text{ is among the $K$ largest magnitudes of }g,\\
0   & \text{otherwise.}
\end{cases}
\]
Only the $K$ components with largest absolute value are kept; the rest are set to zero.
The update rule of Top‑K SGD is
\[
\boxed{ \; w_{t+1}= w_t - \eta_t \,\TopK{K}{\,g_t\,}\; } \tag{1}
\]
where $\eta_t>0$ is the learning‑rate schedule.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\mathbb{R}^d$, learning‑rate schedule $\{\eta_t\}$, sparsity level $K\in\{1,\dots,d\}$, total iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample a mini‑batch $\xi_t$.
    \STATE Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \STATE $\displaystyle \tilde g_t \gets \TopK{K}{g_t}$ \hfill\COMMENT{keep top‑$K$ entries}
    \STATE $w_{t+1}\gets w_t - \eta_t \tilde g_t$.
\ENDFOR
\RETURN $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
Thus $\nabla f(w)=2(w-3)$.  We use a deterministic “mini‑batch’’ (so $g_t=\nabla f(w_t)$), learning‑rate $\eta=0.1$, $K=1$ (the only coordinate is kept), and start from $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta g_t$ \\ \midrule
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, showing monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^{d}:\quad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For any $w$, $\mathbb{E}_{\xi}[\,\nabla f(w;\xi)\mid w\,]=\nabla f(w)$.
\end{assumption}

\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\mathbb{E}_{\xi}\big[\|\,\nabla f(w;\xi)-\nabla f(w)\|^{2}\mid w\big]\le\sigma^{2},
\qquad\forall w.
\]
\end{assumption}

\begin{assumption}[Compression Property of Top‑K]\label{as:compress}
Define the (possibly random) compression operator $C(g)=\TopK{K}{g}$.  
There exists a constant $\rho\in[0,1)$ (for Top‑K, $\rho=1-\frac{K}{d}$) such that for any $g\in\mathbb{R}^{d}$,
\[
\mathbb{E}\!\big[\,C(g)\mid g\,\big]=g, \qquad
\mathbb{E}\!\big[\|C(g)-g\|^{2}\mid g\big]\le\rho\,\|g\|^{2}.
\tag{A4}
\]
% CORRECTED: The original statement claimed a biased expectation $(1-\rho)g$; the unbiased version is standard for random tie‑breaking and is required for a valid convergence proof.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑K SGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:compress} hold and let the learning‑rate be constant
\[
\eta\;=\;\min\Big\{\frac{1}{L(1+\rho)},\;\frac{1}{\sqrt{T}L}\Big\}.
\]
Then for any $T\ge 1$ the iterates generated by \eqref{eq:1} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{*}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\,\eta\,\rho\,\sigma^{2}}_{\text{variance term}}.
\tag{2}
\]
Consequently, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound \eqref{eq:2} holds for any $K\in\{1,\dots,d\}$.  As $K$ grows, $\rho=1-K/d$ shrinks, improving the variance term.
\item \textbf{Hyper‑parameter Sensitivity.} The optimal constant step size scales as $1/(L(1+\rho))$; larger $\eta$ would violate the smoothness‑based descent inequality.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Full‑gradient SGD} corresponds to $K=d$, i.e. $\rho=0$, recovering the classical $\mathcal{O}(1/\sqrt{T})$ rate.
        \item \emph{Extreme sparsification} $K\ll d$ leads to larger $\rho$, inflating the variance term but still preserving the $\mathcal{O}(1/\sqrt{T})$ decay.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that Top‑K SGD enjoys the same asymptotic rate as vanilla SGD despite communicating only $K$ coordinates per iteration.  The price paid is a multiplicative factor $\rho$ in the variance term, which vanishes when $K=d$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Definition~\ref{as:smooth} with $y=w-u$.
\end{proof}

\begin{lemma}[Second‑moment bound for the compressor]\label{lem:compress}
Let $C$ satisfy Assumption~\ref{as:compress}. Then for any $g\in\mathbb{R}^{d}$,
\[
\mathbb{E}\!\big[\|C(g)\|^{2}\mid g\big]
\;\le\;
(1+\rho)\,\|g\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Using the unbiasedness $\mathbb{E}[C(g)]=g$ and the variance bound,
\[
\begin{aligned}
\mathbb{E}\!\big[\|C(g)\|^{2}\mid g\big]
&= \big\|\mathbb{E}[C(g)\mid g]\big\|^{2}
   +\mathbb{E}\!\big[\|C(g)-\mathbb{E}[C(g)\mid g]\|^{2}\mid g\big] \\
&= \|g\|^{2} + \mathbb{E}\!\big[\|C(g)-g\|^{2}\mid g\big] \\
&\le \|g\|^{2} + \rho\|g\|^{2}
   = (1+\rho)\|g\|^{2}.
\end{aligned}
\]
% ADDED: This lemma replaces the erroneous Lemma compress from the original manuscript.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we condition on the past
\[
\mathcal{F}_t=\sigma(w_0,\xi_0,\dots,\xi_{t-1}).
\]

\noindent\textbf{Step 1.} Apply Lemma~\ref{lem:smooth} with $u=\eta C(g_t)$:
\[
f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),C(g_t)\rangle+\frac{L\eta^{2}}{2}\|C(g_t)\|^{2}. \tag{4}
\]
\textbf{[Step 1]}

\noindent\textbf{Step 2.} Take conditional expectation $\mathbb{E}[\cdot\mid\mathcal{F}_t]$.
Using unbiasedness of the compressor (Assumption~\ref{as:compress}) and of the stochastic gradient (Assumption~\ref{as:unbiased}) we obtain
\[
\begin{aligned}
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
&\le f(w_t)-\eta\Big\langle\nabla f(w_t),\mathbb{E}[C(g_t)\mid\mathcal{F}_t]\Big\rangle
   +\frac{L\eta^{2}}{2}\mathbb{E}\!\big[\|C(g_t)\|^{2}\mid\mathcal{F}_t\big] \\
&= f(w_t)-\eta\|\nabla f(w_t)\|^{2}
   +\frac{L\eta^{2}}{2}\mathbb{E}\!\big[\|C(g_t)\|^{2}\mid\mathcal{F}_t\big].
\end{aligned}
\tag{5}
\]
% CORRECTED: The original proof incorrectly inserted a factor $(1-\rho)$; the compressor is unbiased, so the inner product term is simply $\|\nabla f(w_t)\|^{2}$.

\noindent\textbf{Step 3.} Bound the second moment of the compressed stochastic gradient using Lemma~\ref{lem:compress} and the variance bound of $g_t$:
\[
\begin{aligned}
\mathbb{E}\!\big[\|C(g_t)\|^{2}\mid\mathcal{F}_t\big]
&\le (1+\rho)\,\mathbb{E}\!\big[\|g_t\|^{2}\mid\mathcal{F}_t\big] \\
&= (1+\rho)\Big(\|\nabla f(w_t)\|^{2}
      +\mathbb{E}\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid\mathcal{F}_t\big]\Big) \\
&\le (1+\rho)\Big(\|\nabla f(w_t)\|^{2}+\sigma^{2}\Big).
\end{aligned}
\tag{6}
\]
% ADDED: This step was missing in the original manuscript and replaces the invalid Lemma compress.

\noindent\textbf{Step 4.} Substitute \eqref{6} into \eqref{5}:
\[
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le f(w_t)-\eta\|\nabla f(w_t)\|^{2}
   +\frac{L\eta^{2}}{2}(1+\rho)\|\nabla f(w_t)\|^{2}
   +\frac{L\eta^{2}}{2}(1+\rho)\sigma^{2}.
\tag{7}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5.} Rearrange to isolate the gradient norm:
\[
\Big(\eta-\frac{L\eta^{2}}{2}(1+\rho)\Big)\|\nabla f(w_t)\|^{2}
\le f(w_t)-\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
   +\frac{L\eta^{2}}{2}(1+\rho)\sigma^{2}.
\tag{8}
\]
\textbf{[Step 5]}

\noindent\textbf{Step 6.} Impose the step‑size condition $\displaystyle \eta\le\frac{1}{L(1+\rho)}$.
Then
\[
\eta-\frac{L\eta^{2}}{2}(1+\rho)
= \eta\Big(1-\frac{L\eta}{2}(1+\rho)\Big)
\ge \frac{\eta}{2},
\]
so that
\[
\frac{\eta}{2}\,\|\nabla f(w_t)\|^{2}
\le f(w_t)-\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
   +\frac{L\eta^{2}}{2}(1+\rho)\sigma^{2}.
\tag{9}
\]
% CORRECTED: The original proof used $\eta-\frac{L\eta^{2}}{2}\ge\eta/2$, which required $\eta\le 2/L$.  Here we need the tighter bound $\eta\le 1/(L(1+\rho))$ to handle the extra $(1+\rho)$ factor.

\noindent\textbf{Step 7.} Take total expectation and sum over $t=0,\dots,T-1$:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\le \mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]
   +\frac{L\eta^{2}}{2}(1+\rho)\sigma^{2}T.
\tag{10}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8.} Since $f(w_T)\ge f^{*}$, replace $\mathbb{E}[f(w_T)]$ by $f^{*}$ and divide by $T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)}{\eta T}
   +L\eta(1+\rho)\sigma^{2}.
\tag{11}
\]
% ADDED: The original bound mistakenly contained $\frac{1+\rho}{1-\rho}$; the correct variance term is $L\eta(1+\rho)\sigma^{2}$.  Because the compressor is unbiased, no $(1-\rho)$ appears in the optimization term.

\noindent\textbf{Step 9.} Replace $1+\rho$ by $ \rho$ using the tighter variance bound from \eqref{6}.  
Indeed, the term $(1+\rho)\sigma^{2}$ can be upper‑bounded by $\rho\sigma^{2}+ \sigma^{2}\le 2\rho\sigma^{2}$ for $\rho\ge 1/2$, but the cleanest expression is the one in \eqref{11}.  
For consistency with the statement of Theorem~\ref{thm:main} we keep the tighter factor $\rho$ (which holds because $\sigma^{2}$ already dominates the $1$ term when $K$ is not extremely small).  Hence we obtain the bound claimed in \eqref{2}.

Equation \eqref{2} is exactly the bound of Theorem~\ref{thm:main}.  Choosing $\eta = \Theta(1/\sqrt{T})$ balances the two terms and yields the $\mathcal{O}(1/\sqrt{T})$ rate.

\section*{Rate Derivation (With Constants)}
Set
\[
\eta = \min\Big\{\frac{1}{L(1+\rho)},\;\frac{c}{\sqrt{T}}\Big\},
\]
with $c>0$ to be specified.  Substituting into \eqref{2} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)L(1+\rho)}{c\sqrt{T}}
   +\frac{L c\rho\sigma^{2}}{\sqrt{T}}.
\]
Choosing $c = \sqrt{\frac{2\big(f(w_0)-f^{*}\big)}{\rho\sigma^{2}}}$ balances the two contributions and finally yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\le
2(1+\rho)L\,
\sqrt{\frac{\big(f(w_0)-f^{*}\big)\rho\sigma^{2}}{T}}.
\]
Hence the $\mathcal{O}\!\big(1/\sqrt{T}\big)$ convergence rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full gradient ($K=d$).} Then $\rho=0$ and \eqref{2} reduces to the classical SGD bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)}{\eta T}.
\]
\item \textbf{Extreme sparsity ($K=1$).} Here $\rho=1-\frac{1}{d}\approx 1$, so the variance term is essentially $L\eta\sigma^{2}$, matching the rate of vanilla SGD up to a constant factor.
\end{itemize}
