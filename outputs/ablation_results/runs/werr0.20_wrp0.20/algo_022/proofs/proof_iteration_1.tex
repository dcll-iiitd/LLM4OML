\section*{Proximal Gradient Descent (PGD) for a Non‑Convex Smooth $f$}

\section*{Mathematical Formulation}
We consider the composite optimization problem  

\[
\min_{x\in\mathbb{R}^d}\; \Phi(x)\stackrel{\text{def}}{=} f(x)+g(x),
\]

where  

\begin{itemize}[noitemsep]
    \item $f:\mathbb{R}^d\rightarrow\mathbb{R}$ is \emph{continuously differentiable} and has $L$‑Lipschitz continuous gradient,
          \[
          \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^d .
          \]
    \item $g:\mathbb{R}^d\rightarrow\mathbb{R}\cup\{+\infty\}$ is proper, closed and \emph{convex}.  
    \item $\operatorname{prox}_{\eta g}$ denotes the proximal operator of $g$,
          \[
          \operatorname{prox}_{\eta g}(v)\stackrel{\text{def}}{=}\arg\min_{u}\Big\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
          \]
\end{itemize}

For a stepsize $\eta>0$, the \textbf{proximal gradient update} reads  

\[
\boxed{\,x_{k+1}= \operatorname{prox}_{\eta g}\!\big(x_k-\eta\nabla f(x_k)\big)\,},\qquad k=0,1,\dots
\tag{PGD}
\]

Define the \emph{gradient mapping}  

\[
\mathcal{G}_{\eta}(x)\stackrel{\text{def}}{=}\frac{1}{\eta}\Big(x-\operatorname{prox}_{\eta g}\!\big(x-\eta\nabla f(x)\big)\Big).
\tag{1}
\]

When $g\equiv 0$, $\mathcal{G}_{\eta}(x)=\nabla f(x)$. The norm $\|\mathcal{G}_{\eta}(x)\|$ is a natural stationarity measure for the non‑convex problem.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Descent (PGD)}\label{alg:pgd}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\mathbb{R}^{d}$, stepsize $\eta\in(0,1/L]$, maximal iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute stochastic (here deterministic) gradient $g_k\gets\nabla f(x_k)$.
    \STATE $x_{k+1}\gets \operatorname{prox}_{\eta g}\!\big(x_k-\eta g_k\big)$.
\ENDFOR
\RETURN $x_{K}$
\end{algorithmic}
\end{algorithm}

Termination criteria (not shown) may be based on $\|\mathcal{G}_{\eta}(x_k)\|$ falling below a tolerance or on a maximum runtime.

\section*{Dry Run Example}
We minimise the simple scalar function $f(w)=(w-3)^{2}$ and take $g\equiv0$ (hence $\operatorname{prox}_{\eta g}= \operatorname{id}$).  
Parameters: $w_{0}=0$, stepsize $\eta=0.1$, $K=3$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta\nabla f(w_k)$ \\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}

Corresponding objective values:  

\[
\Phi(w_0)=9,\;\Phi(w_1)=5.76,\;\Phi(w_2)=4.147,\;\Phi(w_3)=3.202.
\]

The iterates move toward the minimiser $w^{\star}=3$, illustrating the descent property of \eqref{PGD}.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is differentiable and its gradient is $L$‑Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{ass:convex}
$g$ is proper, closed and convex, so $\operatorname{prox}_{\eta g}$ is single‑valued and firmly non‑expansive.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Bounded Below]\label{ass:lower}
$\Phi$ is bounded from below: $\Phi^{\inf}\stackrel{\text{def}}{=}\inf_{x}\Phi(x)>-\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:lower} and with a stepsize $\eta\in(0,1/L]$, the iterates generated by \eqref{PGD} satisfy for every $K\ge 1$  

\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_k)\|^{2}
\;\le\;
\frac{2\big(\Phi(x_0)-\Phi^{\inf}\big)}{\eta K}.
\tag{2}
\]

Consequently,
\[
\min_{0\le k\le K-1}\|\mathcal{G}_{\eta}(x_k)\|^{2}
\;\le\;
\frac{2\big(\Phi(x_0)-\Phi^{\inf}\big)}{\eta K}
=O\!\left(\frac{1}{K}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
    \item \textbf{Descent.} The method yields a monotone decrease of the composite objective:
          \[
          \Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
          \tag{3}
          \]
    \item \textbf{Stationarity.} Because $\|\mathcal{G}_{\eta}(x_k)\|\to 0$ at the rate $O(1/\sqrt{k})$, any limit point $\bar x$ of $\{x_k\}$ satisfies the first‑order optimality condition 
          $0\in\nabla f(\bar x)+\partial g(\bar x)$.
    \item \textbf{Hyperparameter Sensitivity.} The bound \eqref{2} is inversely proportional to $\eta$; a larger stepsize (up to $1/L$) yields a tighter bound but may violate the descent inequality if $\eta>1/L$.
    \item \textbf{Comparison.} When $g\equiv0$, $\mathcal{G}_{\eta}(x)=\nabla f(x)$ and Theorem~\ref{thm:main} reduces to the classic $O(1/K)$ guarantee for gradient descent on smooth non‑convex functions. Adding a convex regulariser $g$ preserves the same rate thanks to the non‑expansiveness of the proximal operator.
\end{itemize}

\section*{Discussion}
The result shows that even without convexity of $f$, the proximal gradient method enjoys the same $O(1/K)$ stationarity decay as ordinary gradient descent. The proof hinges on two elementary facts:
\begin{enumerate}[noitemsep]
    \item $L$‑smoothness of $f$ gives a quadratic upper bound.
    \item The proximal operator is \emph{firmly non‑expansive}, i.e. 
          \[
          \|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\|^{2}
          \le \langle u-v,\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\rangle .
          \tag{4}
          \]
\end{enumerate}
These two ingredients lead to the clean descent inequality \eqref{3}, from which the $O(1/K)$ bound follows by telescoping.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient and the Lipschitz condition. \hfill $\square$
\end{proof}

\begin{lemma}[Firm non‑expansiveness of $\operatorname{prox}_{\eta g}$]\label{lem:firm}
For any $u,v\in\mathbb{R}^{d}$,
\[
\|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\|^{2}
\le\langle u-v,\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\rangle .
\tag{6}
\]
\end{lemma}
\begin{proof}
Since $g$ is convex, its proximal operator is the resolvent of a monotone operator, which is firmly non‑expansive; see e.g. \cite{Parikh2014}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we fix a stepsize $\eta\in(0,1/L]$ and write $x_{k+1}= \operatorname{prox}_{\eta g}(x_k-\eta\nabla f(x_k))$.

\begin{enumerate}[label={[\textbf{Step \arabic*.}]},leftmargin=*,align=left]
\item \textbf{Notation.} Define the auxiliary point 
\[
y_k\stackrel{\text{def}}{=}x_k-\eta\nabla f(x_k).
\tag{7}
\]
Then $x_{k+1}= \operatorname{prox}_{\eta g}(y_k)$.

\item \textbf{Apply Lemma~\ref{lem:descent}} with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{8}
\]

\item \textbf{Rewrite the inner product.} From the definition of $y_k$,
\[
x_{k+1}-x_k = \big(\operatorname{prox}_{\eta g}(y_k)-y_k\big)+\big(y_k-x_k\big)
               = \big(\operatorname{prox}_{\eta g}(y_k)-y_k\big)-\eta\nabla f(x_k).
\tag{9}
\]
Hence
\[
\langle\nabla f(x_k),x_{k+1}-x_k\rangle
= -\eta\|\nabla f(x_k)\|^{2}
  +\langle\nabla f(x_k),\operatorname{prox}_{\eta g}(y_k)-y_k\rangle .
\tag{10}
\]

\item \textbf{Use the definition of the gradient mapping.} By \eqref{1},
\[
\mathcal{G}_{\eta}(x_k)=\frac{1}{\eta}\big(x_k-\operatorname{prox}_{\eta g}(y_k)\big)
\quad\Longrightarrow\quad
\operatorname{prox}_{\eta g}(y_k)=x_k-\eta\mathcal{G}_{\eta}(x_k).
\tag{11}
\]

Insert \eqref{11} into \eqref{9}:
\[
x_{k+1}-x_k = -\eta\big(\nabla f(x_k)+\mathcal{G}_{\eta}(x_k)\big).
\tag{12}
\]

Consequently,
\[
\|x_{k+1}-x_k\|^{2}= \eta^{2}\big\|\nabla f(x_k)+\mathcal{G}_{\eta}(x_k)\big\|^{2}
                 =\eta^{2}\big(\|\nabla f(x_k)\|^{2}+2\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle+\|\mathcal{G}_{\eta}(x_k)\|^{2}\big).
\tag{13}
\]

\item \textbf{Plug (10)–(13) into (8).} After straightforward algebra,
\[
\begin{aligned}
f(x_{k+1}) &\le f(x_k)
            -\eta\|\nabla f(x_k)\|^{2}
            +\langle\nabla f(x_k),\operatorname{prox}_{\eta g}(y_k)-y_k\rangle \\
           &\quad+\frac{L}{2}\eta^{2}\Big(\|\nabla f(x_k)\|^{2}
           +2\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle
           +\|\mathcal{G}_{\eta}(x_k)\|^{2}\Big).
\end{aligned}
\tag{14}
\]

\item \textbf{Express $\operatorname{prox}_{\eta g}(y_k)-y_k$ via $\mathcal{G}_{\eta}(x_k)$.} Using \eqref{11} and $y_k=x_k-\eta\nabla f(x_k)$,
\[
\operatorname{prox}_{\eta g}(y_k)-y_k
= \big(x_k-\eta\mathcal{G}_{\eta}(x_k)\big)-\big(x_k-\eta\nabla f(x_k)\big)
= \eta\big(\nabla f(x_k)-\mathcal{G}_{\eta}(x_k)\big).
\tag{15}
\]

Thus
\[
\langle\nabla f(x_k),\operatorname{prox}_{\eta g}(y_k)-y_k\rangle
= \eta\big(\|\nabla f(x_k)\|^{2}
          -\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle\big).
\tag{16}
\]

\item \textbf{Insert (16) into (14).} After cancelation of the $-\eta\|\nabla f(x_k)\|^{2}$ term,
\[
\begin{aligned}
f(x_{k+1}) &\le f(x_k)
            -\eta\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle   \\
           &\quad+\frac{L}{2}\eta^{2}\Big(\|\nabla f(x_k)\|^{2}
           +2\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle
           +\|\mathcal{G}_{\eta}(x_k)\|^{2}\Big).
\end{aligned}
\tag{17}
\]

\item \textbf{Combine terms containing $\langle\nabla f,\mathcal{G}_{\eta}\rangle$.}  
Factor the coefficient of $\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle$:
\[
-\eta + L\eta^{2}= -\eta\big(1-L\eta\big).
\tag{18}
\]

Similarly, the coefficient of $\|\nabla f(x_k)\|^{2}$ is $\frac{L}{2}\eta^{2}$.

Hence
\[
f(x_{k+1})\le f(x_k)
            -\eta\big(1-L\eta\big)\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle
            +\frac{L}{2}\eta^{2}\|\nabla f(x_k)\|^{2}
            +\frac{L}{2}\eta^{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
\tag{19}
\]

\item \textbf{Add $g(x_{k+1})$ to both sides.} By the definition of the proximal step (optimality condition of \eqref{11}),
\[
\frac{1}{\eta}\big(x_k-\eta\nabla f(x_k)-x_{k+1}\big)\in\partial g(x_{k+1}),
\]
which implies (see e.g. \cite{Parikh2014})
\[
g(x_{k+1})\le g(x_k)-\langle\mathcal{G}_{\eta}(x_k),x_k-x_{k+1}\rangle-\frac{1}{2\eta}\|x_k-x_{k+1}\|^{2}.
\tag{20}
\]

Using \eqref{12}, $x_k-x_{k+1}= \eta\big(\nabla f(x_k)+\mathcal{G}_{\eta}(x_k)\big)$, and after simplifying,
\[
g(x_{k+1})\le g(x_k)-\eta\langle\nabla f(x_k),\mathcal{G}_{\eta}(x_k)\rangle
            -\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
\tag{21}
\]

\item \textbf{Sum (19) and (21).} The terms $-\eta\langle\nabla f,\mathcal{G}_{\eta}\rangle$ cancel, yielding the composite descent inequality
\[
\Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}
               +\frac{L}{2}\eta^{2}\Big(\|\nabla f(x_k)\|^{2}
               +\|\mathcal{G}_{\eta}(x_k)\|^{2}\Big).
\tag{22}
\]

\item \textbf{Use the stepsize bound $\eta\le 1/L$.} Then $L\eta^{2}\le\eta$, and consequently
\[
\frac{L}{2}\eta^{2}\|\nabla f(x_k)\|^{2}\le\frac{\eta}{2}\|\nabla f(x_k)\|^{2},
\qquad
\frac{L}{2}\eta^{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}\le\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
\tag{23}
\]

Insert (23) into (22):
\[
\Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}
               +\frac{\eta}{2}\|\nabla f(x_k)\|^{2}
               +\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
\tag{24}
\]

The two $\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}$ terms cancel, leaving

\[
\boxed{\Phi(x_{k+1})\le \Phi(x_k)-\frac{\eta}{2}\|\mathcal{G}_{\eta}(x_k)\|^{2}}.
\tag{25}
\]

This is exactly inequality \eqref{3}.

\item \textbf{Telescoping sum.} Sum \eqref{25} for $k=0,\dots,K-1$:
\[
\Phi(x_{K})\le \Phi(x_{0})-\frac{\eta}{2}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_k)\|^{2}.
\tag{26}
\]

Rearrange and use $\Phi(x_{K})\ge\Phi^{\inf}$ (Assumption~\ref{ass:lower}):

\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\mathcal{G}_{\eta}(x_k)\|^{2}
\le \frac{2\big(\Phi(x_{0})-\Phi^{\inf}\big)}{\eta K}.
\tag{27}
\]

Equation \eqref{27} is precisely the bound \eqref{2} claimed in Theorem~\ref{thm:main}. \hfill $\square$

\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{27} we obtain the explicit constant
\[
C\;=\;\frac{2\big(\Phi(x_{0})-\Phi^{\inf}\big)}{\eta},
\]
so that for any $K\ge1$,
\[
\min_{0\le k\le K-1}\|\mathcal{G}_{\eta}(x_k)\|^{2}\le\frac{C}{K}.
\]
If $\Phi$ is bounded below by $0$ and $x_0$ is chosen such that $\Phi(x_0)\le B$, then $C\le 2B/\eta$.

\section*{Special Cases}
\begin{enumerate}[label={\arabic*.},leftmargin=*]
    \item \textbf{Pure gradient descent ($g\equiv0$).} Then $\mathcal{G}_{\eta}(x)=\nabla f(x)$ and \eqref{25} reduces to the classical smooth‑non‑convex descent lemma.
    \item \textbf{Indicator regulariser.} If $g=\iota_{\mathcal{C}}$ is the indicator of a closed convex set $\mathcal{C}$, $\operatorname{prox}_{\eta g}$ becomes the Euclidean projection onto $\mathcal{C}$. The same $O(1/K)$ bound holds for projected gradient descent.
    \item \textbf{Strongly convex $g$.} When $g$ is $\mu$‑strongly convex, the proximal map is contractive with factor $1/(1+\eta\mu)$. One can sharpen \eqref{25} to obtain linear convergence under additional Polyak–Łojasiewicz conditions on $f$.
\end{enumerate}
