\section*{Algorithm Name}
\textbf{Mirror Descent (MD) with a strongly convex prox‑function.}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be a convex differentiable objective and let
$h:\mathbb{R}^{d}\to\mathbb{R}$ be a $1$‑strongly convex function w.r.t. a norm $\|\cdot\|$.
The Bregman divergence generated by $h$ is
\[
\Breg(x,y)=h(x)-h(y)-\inner{\nabla h(y)}{x-y}.
\]

Given a stepsize sequence $\{\eta_k\}_{k\ge 0}$, the Mirror Descent update is
\begin{equation}
\label{eq:md}
x_{k+1}= \arg\min_{x\in\mathbb{R}^{d}}
\Big\{ \inner{\nabla f(x_k)}{x}+ \frac{1}{\eta_k}\,\Breg(x,x_k) \Big\}.
\end{equation}
When $h(x)=\tfrac12\|x\|_2^2$, $\Breg$ reduces to the Euclidean squared distance and \eqref{eq:md}
coincides with standard (projected) gradient descent.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, stepsize schedule $\{\eta_k\}_{k\ge0}$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\nabla f(x_k)$.
    \STATE $x_{k+1}\gets \arg\min\limits_{x}\Big\{ \inner{g_k}{x}+ \frac{1}{\eta_k}\,\Breg(x,x_k) \Big\}$.
    \IF{stopping criterion satisfied}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (last iterate) or the averaged iterate $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Choose the Euclidean prox $h(w)=\tfrac12 w^{2}$, thus $\Breg(u,v)=\tfrac12 (u-v)^2$.
Set $w_0=0$ and a constant stepsize $\eta_k=\eta=0.1$.

\begin{enumerate}[label=\textbf{Iteration \arabic*:}, leftmargin=*]
\item $k=0$:
\[
g_0=\nabla f(w_0)=2(0-3)=-6.
\]
The update solves
\[
w_{1}= \arg\min_{w}\Big\{-6w+\frac1{0.1}\tfrac12(w-0)^2\Big\}
      =\arg\min_{w}\Big\{-6w+5w^{2}\Big\}.
\]
Setting derivative $-6+10w=0$ gives $w_{1}=0.6$.
Objective value: $f(w_1)=(0.6-3)^2=5.76$.

\item $k=1$:
\[
g_1=2(0.6-3)=-4.8.
\]
\[
w_{2}= \arg\min_{w}\Big\{-4.8 w+5 (w-0.6)^2\Big\}
      \;\Longrightarrow\; -4.8+10(w-0.6)=0\;\Longrightarrow\; w_{2}=1.08.
\]
$f(w_2)=(1.08-3)^2=3.6864$.

\item $k=2$:
\[
g_2=2(1.08-3)=-3.84.
\]
\[
w_{3}= \arg\min_{w}\Big\{-3.84 w+5 (w-1.08)^2\Big\}
      \;\Longrightarrow\; -3.84+10(w-1.08)=0\;\Longrightarrow\; w_{3}=1.464.
\]
$f(w_3)=(1.464-3)^2=2.366\; .
\]
\end{enumerate}
The iterates move monotonically toward the minimiser $w^{\star}=3$ and the objective decreases.

\section*{Assumptions}
\begin{assumption}[Convexity]
$f$ is convex on $\mathbb{R}^{d}$:
\[
f(y)\ge f(x)+\inner{\nabla f(x)}{y-x},\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Relative Smoothness]
There exists $L>0$ such that for all $x,y$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+L\,\Breg(y,x).
\]
\end{assumption}
\begin{assumption}[Strong Convexity of the Prox‑function]
$h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$:
\[
\Breg(x,y)\ge \tfrac12\|x-y\|^{2},\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Stepsize]
The stepsizes satisfy $0<\eta_k\le \frac{1}{L}$ for all $k$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Descent]\label{thm:md}
Under Assumptions 1–4 and for any $T\ge1$, the average iterate
\[
\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k
\]
satisfies
\[
f(\bar x_T)-f(x^{\star})\;\le\;\frac{1}{T}\sum_{k=0}^{T-1}\frac{\Breg(x^{\star},x_k)}{\eta_k}
\;\le\;\frac{\Breg(x^{\star},x_0)}{T\,\eta},
\]
where $\eta:=\min_k\eta_k$ and $x^{\star}\in\arg\min f$.
Consequently, with a constant stepsize $\eta_k=\eta\le 1/L$,
\[
f(\bar x_T)-f(x^{\star})=O\!\left(\frac{1}{T}\right).
\]
If, in addition, $f$ is $\mu$‑strongly convex w.r.t. $\Breg$, then
\[
\Breg(x_{k},x^{\star})\le \bigl(1-\eta\mu\bigr)^{k}\,\Breg(x_0,x^{\star}),
\]
i.e. a linear (geometric) rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update \eqref{eq:md} is a proximal step; the strong convexity of $h$ guarantees a unique minimiser and prevents divergence even for large gradients.
\item \textbf{Hyperparameter Sensitivity.} The stepsize must respect $\eta_k\le 1/L$ (relative smoothness constant). Over‑stepping destroys the descent property; under‑stepping slows the $O(1/T)$ rate but never harms convergence.
\item \textbf{Comparison to Gradient Descent.} When $h$ is Euclidean, Theorem~\ref{thm:md} recovers the classic $O(1/T)$ rate of gradient descent for convex smooth functions. For non‑Euclidean geometries (e.g., KL‑divergence), MD can exploit problem structure and achieve the same rate with respect to the natural Bregman distance.
\item \textbf{Strongly Convex Case.} The linear rate matches that of accelerated methods up to a constant factor; no acceleration is claimed because the analysis follows the basic MD recursion.
\end{itemize}

\section*{Discussion}
The convergence proof hinges on the three‑point identity of Bregman divergences and the relative smoothness condition, both of which are intrinsic to the mirror‑descent framework. The result is optimal for first‑order methods on the class of convex, relatively smooth functions. Extensions (e.g., stochastic gradients, adaptive stepsizes) follow the same template with additional martingale concentration arguments.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $x,y,z\in\mathbb{R}^{d}$,
\[
\Breg(x,z)=\Breg(x,y)+\Breg(y,z)+\inner{\nabla h(y)-\nabla h(z)}{x-y}.
\]
\end{lemma}
\begin{proof}
Direct expansion of the definition of $\Breg$ yields the identity. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Relative smoothness inequality]\label{lem:smooth}
If $f$ satisfies Assumption~2, then for any $x,y$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+L\,\Breg(y,x).
\]
\end{lemma}
\begin{proof}
Assumption~2 is precisely the inequality above. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:md}. Throughout the proof let $x^{\star}\in\arg\min f$.

\noindent\textbf{Step 1 (Optimality condition of the MD subproblem).}
From \eqref{eq:md}, $x_{k+1}$ satisfies the first‑order optimality condition
\begin{equation}
\label{eq:opt}
\inner{\nabla f(x_k)}{x-x_{k+1}}+\frac{1}{\eta_k}\inner{\nabla h(x_{k+1})-\nabla h(x_k)}{x-x_{k+1}}\ge0,
\qquad\forall x.
\end{equation}
[Step 1]

\noindent\textbf{Step 2 (Choose $x=x^{\star}$).}
Plug $x=x^{\star}$ into \eqref{eq:opt} and rearrange:
\begin{equation}
\label{eq:opt_star}
\inner{\nabla f(x_k)}{x^{\star}-x_{k+1}}
\ge
\frac{1}{\eta_k}\inner{\nabla h(x_{k+1})-\nabla h(x_k)}{x_{k+1}-x^{\star}}.
\end{equation}
[Step 2]

\noindent\textbf{Step 3 (Apply three‑point identity).}
Using Lemma~\ref{lem:three} with $(x,y,z)=(x^{\star},x_{k+1},x_k)$,
\[
\Breg(x^{\star},x_k)=\Breg(x^{\star},x_{k+1})+\Breg(x_{k+1},x_k)
+\inner{\nabla h(x_{k+1})-\nabla h(x_k)}{x^{\star}-x_{k+1}}.
\]
Rearrange to isolate the inner product:
\[
\inner{\nabla h(x_{k+1})-\nabla h(x_k)}{x_{k+1}-x^{\star}}
= \Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})-\Breg(x_{k+1},x_k).
\]
[Step 3]

\noindent\textbf{Step 4 (Insert into \eqref{eq:opt_star}).}
Multiplying the identity from Step 3 by $1/\eta_k$ and substituting into \eqref{eq:opt_star} yields
\[
\inner{\nabla f(x_k)}{x^{\star}-x_{k+1}}
\ge
\frac{1}{\eta_k}\Bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})-\Breg(x_{k+1},x_k)\Bigr].
\]
[Step 4]

\noindent\textbf{Step 5 (Convexity of $f$).}
Convexity (Assumption~1) gives
\[
f(x_k)-f(x^{\star})\le \inner{\nabla f(x_k)}{x_k-x^{\star}}.
\]
Equivalently,
\[
\inner{\nabla f(x_k)}{x^{\star}-x_{k}} \le f(x^{\star})-f(x_k).
\]
[Step 5]

\noindent\textbf{Step 6 (Combine Steps 4 and 5).}
Add $\inner{\nabla f(x_k)}{x_{k+1}-x_k}$ to both sides of the inequality in Step 4:
\[
\inner{\nabla f(x_k)}{x^{\star}-x_k}
\ge
\frac{1}{\eta_k}\Bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})-\Breg(x_{k+1},x_k)\Bigr]
+\inner{\nabla f(x_k)}{x_{k+1}-x_k}.
\]
Now replace the left‑hand side using Step 5:
\[
f(x^{\star})-f(x_k)
\ge
\frac{1}{\eta_k}\Bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})-\Breg(x_{k+1},x_k)\Bigr]
+\inner{\nabla f(x_k)}{x_{k+1}-x_k}.
\]
[Step 6]

\noindent\textbf{Step 7 (Relative smoothness to bound the last inner product).}
From Lemma~\ref{lem:smooth} with $x=x_k$, $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_k)+\inner{\nabla f(x_k)}{x_{k+1}-x_k}+L\,\Breg(x_{k+1},x_k).
\]
Rearrange:
\[
\inner{\nabla f(x_k)}{x_{k+1}-x_k}\ge f(x_{k+1})-f(x_k)-L\,\Breg(x_{k+1},x_k).
\]
[Step 7]

\noindent\textbf{Step 8 (Plug Step 7 into Step 6).}
\[
f(x^{\star})-f(x_k)
\ge
\frac{1}{\eta_k}\Bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})-\Breg(x_{k+1},x_k)\Bigr]
+f(x_{k+1})-f(x_k)-L\,\Breg(x_{k+1},x_k).
\]
Cancel $-f(x_k)$ on both sides:
\[
f(x^{\star})\ge f(x_{k+1})+\frac{1}{\eta_k}\bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})\bigr]
-\Bigl(\frac{1}{\eta_k}+L\Bigr)\Breg(x_{k+1},x_k).
\]
[Step 8]

\noindent\textbf{Step 9 (Choose stepsize $\eta_k\le 1/L$).}
Since $\eta_k\le 1/L$, we have $\frac{1}{\eta_k}+L\le \frac{2}{\eta_k}$. Dropping the non‑negative term $-\frac{2}{\eta_k}\Breg(x_{k+1},x_k)$ yields the simpler inequality
\[
f(x_{k+1})-f(x^{\star})\le \frac{1}{\eta_k}\Bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})\Bigr].
\]
[Step 9]

\noindent\textbf{Step 10 (Telescoping sum).}
Sum the inequality of Step 9 over $k=0,\dots,T-1$:
\[
\sum_{k=0}^{T-1}\bigl[f(x_{k+1})-f(x^{\star})\bigr]
\le
\sum_{k=0}^{T-1}\frac{1}{\eta_k}\bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})\bigr].
\]
The right‑hand side telescopes:
\[
\sum_{k=0}^{T-1}\frac{1}{\eta_k}\bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})\bigr]
\le
\frac{1}{\eta}\,\Breg(x^{\star},x_0),
\]
where $\eta:=\min_{k}\eta_k$.
[Step 10]

\noindent\textbf{Step 11 (From sum to average).}
Divide both sides of Step 10 by $T$ and use convexity of $f$ to replace the average of function values by the function at the average iterate $\bar x_T$ (Jensen’s inequality):
\[
f(\bar x_T)-f(x^{\star})\le\frac{1}{T}\sum_{k=0}^{T-1}\bigl[f(x_{k+1})-f(x^{\star})\bigr]
\le\frac{\Breg(x^{\star},x_0)}{T\,\eta}.
\]
[Step 11]

\noindent\textbf{Step 12 (Strongly convex case).}
If $f$ is $\mu$‑strongly convex w.r.t. $\Breg$, i.e.
\[
f(x)-f(x^{\star})\ge \mu\,\Breg(x,x^{\star}),\qquad\forall x,
\]
then Step 9 becomes
\[
\mu\,\Breg(x_{k+1},x^{\star})\le\frac{1}{\eta_k}\bigl[\Breg(x^{\star},x_k)-\Breg(x^{\star},x_{k+1})\bigr].
\]
Rearranging yields
\[
\Breg(x_{k+1},x^{\star})\le\bigl(1-\eta_k\mu\bigr)\Breg(x_k,x^{\star}).
\]
With constant $\eta_k=\eta\le1/L$, recursion gives
\[
\Breg(x_{k},x^{\star})\le(1-\eta\mu)^{k}\,\Breg(x_0,x^{\star}),
\]
which is a linear convergence rate. \hfill\qed
[Step 12]

\section*{Rate Derivation (With Constants)}
From Step 11 we have the explicit bound
\[
f(\bar x_T)-f(x^{\star})\le \frac{1}{\eta\,T}\,\Breg(x^{\star},x_0).
\]
If $h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$, then $\Breg(x^{\star},x_0)\ge \tfrac12\|x^{\star}-x_0\|^{2}$, so a concrete constant is
\[
f(\bar x_T)-f(x^{\star})\le \frac{2\,\|x^{\star}-x_0\|^{2}}{\eta\,T}.
\]
For the linear rate (strongly convex case) the constant factor is the initial Bregman distance multiplied by $(1-\eta\mu)^{k}$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Euclidean setting.} $h(x)=\tfrac12\|x\|_{2}^{2}$ gives $\Breg(x,y)=\tfrac12\|x-y\|_{2}^{2}$. The update reduces to
\[
x_{k+1}=x_k-\eta_k\nabla f(x_k),
\]
and Theorem~\ref{thm:md} recovers the classical $O(1/T)$ rate for gradient descent.
\item \textbf{Entropy prox.} On the simplex $\Delta^{d}$, choose $h(x)=\sum_{i}x_i\log x_i$ (negative entropy). Then $\Breg$ is the Kullback–Leibler divergence and MD becomes the exponentiated gradient algorithm with the same $O(1/T)$ guarantee.
\item \textbf{Relative smoothness with non‑Euclidean norm.} If $f$ is $L$‑smooth relative to $h$ (e.g., logistic loss relative to the entropy prox), the analysis above holds unchanged; the stepsize bound $\eta\le 1/L$ is expressed in the relative smoothness constant.
\end{enumerate}
