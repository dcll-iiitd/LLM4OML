\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimisation problem  

\[
\min_{x\in\R^{d}} \; F(x)\;:=\;f(x)+g(x),
\]

where  

\begin{itemize}
  \item $f:\R^{d}\rightarrow\R$ is \emph{convex} and $L$-\emph{smooth}, i.e.  
        \[
        \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
        \]
  \item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).  
  \item $\eta>0$ is a stepsize (also called learning rate).  
\end{itemize}

The \textbf{proximal operator} of $g$ with parameter $\eta$ is  

\[
\prox_{\eta g}(v)\;:=\;\arg\min_{u\in\R^{d}}\Big\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]

The algorithm generates a sequence $\{x_{k}\}_{k\ge0}$ by  

\[
\boxed{\,x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr)\,},\qquad k=0,1,2,\dots
\tag{1}
\]

The proximal operator is \emph{non‑expansive}:  

\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}. \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta\in(0,2/L)$, initial point $x_{0}\in\R^{d}$, maximal iter $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute gradient $g_{k}\gets\nabla f(x_{k})$.
    \STATE Take a proximal step 
    \[
    x_{k+1}\gets\prox_{\eta g}\bigl(x_{k}-\eta g_{k}\bigr).
    \]
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or the average $\bar x_{K}=\frac1K\sum_{k=1}^{K}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Take a scalar problem ($d=1$) with  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so that $\prox_{\eta g}(v)=v$.  The gradient is $\nabla f(w)=2(w-3)$ and $L=2$.
Choose $\eta=0.1\in(0,2/L=1)$ and $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
3 & $1.464$ & $-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}

Objective values $f(w_{k})$ decrease: $9,\,5.1984,\,3.1456,\,2.0410,\dots$ confirming convergence toward the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f$ is convex and $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{3}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g$ is proper, closed and convex; its sub‑differential $\partial g$ is non‑empty at any finite point.
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies 
\[
0<\eta<\frac{2}{L}. \tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{1} under Assumptions \ref{as:smooth}–\ref{as:stepsize}.  
Denote $x^{\star}\in\arg\min F$ and $D:=\|x_{0}-x^{\star}\|$. Then for all $K\ge1$
\[
F(x_{K})-F(x^{\star})\;\le\;\frac{D^{2}}{2\eta K}. \tag{5}
\]
Consequently $F(x_{K})\to F(x^{\star})$ with a sub‑linear rate $O(1/K)$. 
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} Because of non‑expansiveness \eqref{2}, the iterates cannot diverge faster than the gradient step alone; the bound (5) holds for any $\eta\in(0,2/L)$.
  \item \textbf{Hyperparameter sensitivity.} The constant $1/(2\eta)$ in \eqref{5} shows a trade‑off: larger $\eta$ (up to $2/L$) yields a smaller prefactor, but if $\eta\ge 2/L$ the descent inequality (see proof) fails.
  \item \textbf{Comparison.} When $g\equiv0$, \eqref{1} reduces to classical gradient descent and Theorem \ref{thm:rate} recovers the standard $O(1/K)$ bound. For non‑smooth $g$, the same rate is obtained without any additional variance or smoothing assumptions.
\end{itemize}

\section*{Discussion}
The proof hinges on two elementary facts: (i) the descent lemma for $L$‑smooth $f$ (inequality \eqref{3}) and (ii) the non‑expansiveness of the proximal map \eqref{2}. By combining them with the optimality condition of the proximal step we obtain a one‑step inequality that telescopes when summed over iterations, leading directly to the $O(1/K)$ guarantee. No stochasticity or strong convexity is required; stronger assumptions (e.g., strong convexity) would improve the rate to linear convergence, but the present analysis already captures the essential behaviour of the proximal gradient method.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Optimality condition of the proximal operator]\label{lem:prox-opt}
For any $v\in\R^{d}$ and $\eta>0$, $u=\prox_{\eta g}(v)$ iff  

\[
0\in \partial g(u)+\frac1\eta (u-v). \tag{6}
\]
\end{lemma}
\begin{proof}
By definition $u$ minimises $g(u)+\frac1{2\eta}\|u-v\|^{2}$. The sub‑gradient optimality condition yields \eqref{6}. \hfill $\square$
\end{proof}

\begin{lemma}[Descent lemma for $L$‑smooth convex $f$]\label{lem:descent}
For all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{7}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout let $x^{\star}\in\arg\min F$.

\noindent\textbf{Step 1 (Proximal optimality).}  
From Lemma \ref{lem:prox-opt} with $v = x_{k}-\eta\nabla f(x_{k})$ we have  

\[
0\in \partial g(x_{k+1})+\frac1\eta\bigl(x_{k+1}-x_{k}+\eta\nabla f(x_{k})\bigr). \tag{8}
\]

\noindent\textbf{Step 2 (Variational inequality).}  
Take any $x\in\R^{d}$ and inner‑product \eqref{8} with $x_{k+1}-x$:

\[
\big\langle \partial g(x_{k+1}),\,x_{k+1}-x\big\rangle
\le -\frac1\eta\big\langle x_{k+1}-x_{k}+\eta\nabla f(x_{k}),\,x_{k+1}-x\big\rangle. \tag{9}
\]

\noindent\textbf{Step 3 (Convexity of $g$).}  
By convexity of $g$,
\[
g(x_{k+1})-g(x)\le\big\langle \partial g(x_{k+1}),\,x_{k+1}-x\big\rangle. \tag{10}
\]

\noindent\textbf{Step 4 (Combine 9 and 10).}  
Insert \eqref{9} into \eqref{10}:

\[
g(x_{k+1})-g(x)\le -\frac1\eta\big\langle x_{k+1}-x_{k},\,x_{k+1}-x\big\rangle
               -\langle \nabla f(x_{k}),\,x_{k+1}-x\big\rangle. \tag{11}
\]

\noindent\textbf{Step 5 (Expand the quadratic term).}  
Recall the identity $2\langle a,b\rangle=\|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$. With $a=x_{k+1}-x_{k}$ and $b=x_{k+1}-x$ we obtain  

\[
2\langle x_{k+1}-x_{k},\,x_{k+1}-x\rangle
= \|x_{k+1}-x_{k}\|^{2}+\|x_{k+1}-x\|^{2}-\|x_{k}-x\|^{2}. \tag{12}
\]

Dividing \eqref{12} by $2\eta$ and substituting into \eqref{11} gives  

\[
\begin{aligned}
g(x_{k+1})-g(x)
\le &-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
     -\frac{1}{2\eta}\|x_{k+1}-x\|^{2}
     +\frac{1}{2\eta}\|x_{k}-x\|^{2}\\
   &-\langle \nabla f(x_{k}),\,x_{k+1}-x\rangle. \tag{13}
\end{aligned}
\]

\noindent\textbf{Step 6 (Add the smooth part).}  
Apply Lemma \ref{lem:descent} with $y=x_{k+1}$, $x=x_{k}$:

\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
            +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \tag{14}
\]

Add \eqref{13} and \eqref{14} (the left‑hand sides give $F(x_{k+1})-F(x)$) and rearrange:

\[
\begin{aligned}
F(x_{k+1})-F(x) \le &\; f(x_{k})-f(x) +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle\\
&\;-\langle \nabla f(x_{k}),x_{k+1}-x\rangle
   -\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
   -\frac{1}{2\eta}\|x_{k+1}-x\|^{2}
   +\frac{1}{2\eta}\|x_{k}-x\|^{2}
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{15}
\]

\noindent\textbf{Step 7 (Cancel linear terms).}  
Observe that  

\[
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle-\langle\nabla f(x_{k}),x_{k+1}-x\rangle
= -\langle\nabla f(x_{k}),x_{k}-x\rangle. \tag{16}
\]

Insert \eqref{16} into \eqref{15}:

\[
\begin{aligned}
F(x_{k+1})-F(x) \le &\; f(x_{k})-f(x)-\langle\nabla f(x_{k}),x_{k}-x\rangle\\
&\;-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
   -\frac{1}{2\eta}\|x_{k+1}-x\|^{2}
   +\frac{1}{2\eta}\|x_{k}-x\|^{2}
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{17}
\]

\noindent\textbf{Step 8 (Convexity of $f$).}  
Convexity of $f$ yields  

\[
f(x_{k})-f(x)\le\langle\nabla f(x_{k}),x_{k}-x\rangle. \tag{18}
\]

Applying \eqref{18} to the first two terms of \eqref{17} eliminates them, leaving

\[
F(x_{k+1})-F(x)\le
-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
-\frac{1}{2\eta}\|x_{k+1}-x\|^{2}
+\frac{1}{2\eta}\|x_{k}-x\|^{2}
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \tag{19}
\]

\noindent\textbf{Step 9 (Collect the $\|x_{k+1}-x_{k}\|^{2}$ terms).}  
Since $0<\eta<2/L$ we have $1-\frac{L\eta}{2}>0$. Rewrite \eqref{19} as  

\[
F(x_{k+1})-F(x)\le
\frac{1}{2\eta}\Bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}\Bigr)
-\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_{k}\|^{2}. \tag{20}
\]

\noindent\textbf{Step 10 (Drop the non‑negative term).}  
Because the coefficient $\frac{1}{2\eta}-\frac{L}{2}>0$, the second term on the right‑hand side of \eqref{20} is non‑negative and can be discarded to obtain a cleaner inequality:

\[
F(x_{k+1})-F(x)\le
\frac{1}{2\eta}\Bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}\Bigr). \tag{21}
\]

\noindent\textbf{Step 11 (Set $x=x^{\star}$ and telescope).}  
Let $x=x^{\star}$. Summing \eqref{21} over $k=0,\dots,K-1$ yields  

\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\frac{1}{2\eta}\Bigl(\|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}\Bigr)
\le\frac{D^{2}}{2\eta}, \tag{22}
\]
where $D:=\|x_{0}-x^{\star}\|$.

\noindent\textbf{Step 12 (Extract the worst‑case bound).}  
Since the left‑hand side of \eqref{22} is a sum of $K$ non‑negative terms,
\[
K\bigl(F(x_{K})-F(x^{\star})\bigr)\le\frac{D^{2}}{2\eta},
\]
which is exactly inequality \eqref{5}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:rate} we have the explicit constant  

\[
C(\eta)=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta},
\]
so that for any iteration $K$,
\[
F(x_{K})-F(x^{\star})\le\frac{C(\eta)}{K}.
\]
The bound is decreasing in $\eta$, confirming the intuition that a larger stepsize (up to $2/L$) accelerates convergence.

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Pure gradient descent ($g\equiv0$).} $\prox_{\eta g}$ reduces to the identity, and the proof collapses to the classical $O(1/K)$ rate for smooth convex functions.
  \item \textbf{Indicator of a convex set $C$ ($g=\iota_{C}$).} $\prox_{\eta g}$ becomes Euclidean projection onto $C$, and the algorithm is the projected gradient method with the same $O(1/K)$ guarantee.
  \item \textbf{Strongly convex $f+g$.} If $F$ is $\mu$‑strongly convex, inequality \eqref{21} can be strengthened to a linear recursion, yielding $F(x_{k})-F(x^{\star})\le\bigl(1-\eta\mu\bigr)^{k}D^{2}/(2\eta)$.
\end{itemize}
