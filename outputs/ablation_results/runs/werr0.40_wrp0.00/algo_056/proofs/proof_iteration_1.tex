\section*{Random Block Coordinate Descent (RBCD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\to\RR$ be a convex differentiable function.  
The variable $w\in\RR^{d}$ is partitioned into $B\ge 1$ (non‑overlapping) blocks
\[
w=\bigl(w^{(1)},w^{(2)},\dots ,w^{(B)}\bigr),\qquad 
w^{(i)}\in\RR^{d_i},\; \sum_{i=1}^{B}d_i=d .
\]
For block $i$ we denote by 
\[
U_i\in\RR^{d\times d_i}
\]
the column‑selection matrix such that $U_iU_i^{\top}w = w^{(i)}$ (i.e. $U_i^{\top}w$ extracts the $i$‑th block).  
The block gradient is $\grad{i}f(w)=U_i^{\top}\nabla f(w)\in\RR^{d_i}$.

At iteration $t$ a block $i_t\in\{1,\dots ,B\}$ is drawn independently
according to a probability distribution $p=(p_1,\dots ,p_B)$; in the
standard uniform case $p_i=1/B$.
Given a stepsize sequence $(\eta_t)_{t\ge0}$ the update reads
\begin{equation}
\boxed{
w_{t+1}=w_t-\eta_t\,U_{i_t}\,\grad{i_t}f(w_t)
}\tag{1}
\end{equation}
Only the selected block is modified; all other blocks remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\RR^{d}$, stepsize rule $\{\eta_t\}_{t\ge0}$,
          block probabilities $p_i>0$, $ \sum_i p_i=1$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample $i_t\sim p$ independently.
    \STATE Compute block gradient $\grad{i_t}f(w_t)=U_{i_t}^{\top}\nabla f(w_t)$.
    \STATE Update the selected block
          \[
          w_{t+1}=w_t-\eta_t\,U_{i_t}\,\grad{i_t}f(w_t).
          \]
\ENDFOR
\RETURN $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^2$ (single block, $B=1$).  
Take $w_0=0$, constant stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)=2(w_t-3)$ & $f(w_t)$\\\hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.1\cdot(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1\cdot(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1\cdot(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3649$
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the effect of the
random (here deterministic) block update.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex: for all $w,w'\in\RR^{d}$,
\[
f(w')\ge f(w)+\langle\nabla f(w),w'-w\rangle .
\]
\end{assumption}
\begin{assumption}[Block‑coordinate $L_i$‑smoothness]\label{ass:smooth}
For each block $i$ there exists $L_i>0$ such that for all $w\in\RR^{d}$ and all
$h\in\RR^{d_i}$,
\[
f\bigl(w+U_i h\bigr)\le f(w)+\langle\grad{i}f(w),h\rangle
          +\frac{L_i}{2}\norm{h}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded distance to optimum]\label{ass:bounded}
Let $w^{\star}\in\arg\min f$ and define
\[
D^{2}:=\norm{w_0-w^{\star}}^{2}<\infty .
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize is constant and satisfies
\[
0<\eta\le \frac{1}{\max_{i}L_i}\; .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of RBCD]\label{thm:main}
Under Assumptions \ref{ass:convex}–\ref{ass:stepsize}, let $\{w_t\}$ be the
iterates generated by \eqref{1} with uniform sampling
$p_i=1/B$. Then for any $T\ge1$,
\[
\EE\!\bigl[f(w_T)-f(w^{\star})\bigr]
\;\le\;
\frac{B\,L_{\max}\,D^{2}}{2\,T},
\qquad L_{\max}:=\max_{i}L_i .
\]
Equivalently, the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$
satisfies the same bound.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Rate.} The bound is $O(1/T)$, matching the optimal rate for
first‑order methods on convex smooth functions.
\item \textbf{Stability.} The requirement $\eta\le 1/L_{\max}$ guarantees that the
expected descent per iteration is non‑negative.
\item \textbf{Hyperparameter sensitivity.}
    \begin{itemize}
        \item Larger $\eta$ (up to $1/L_{\max}$) speeds up convergence but may violate
              the descent guarantee.
        \item Non‑uniform sampling can be incorporated; the bound changes by a factor
              $\frac{1}{p_{\min}}$ where $p_{\min}=\min_i p_i$.
    \end{itemize}
\item \textbf{Comparison to full‑gradient GD.}
    Full‑gradient GD with stepsize $1/L_{\max}$ yields
    $f(w_T)-f(w^{\star})\le \frac{L_{\max}D^{2}}{2T}$.
    RBCD incurs an extra factor $B$ because only one block is updated per
    iteration; however the per‑iteration cost is reduced by roughly $1/B$.
\end{itemize}

\section*{Discussion}
The theorem shows that random block coordinate descent attains the same
asymptotic $1/T$ decay as classical gradient descent while exploiting
the block structure to lower computational cost.  Extensions to
non‑uniform sampling, accelerated variants, or strongly convex objectives
lead to improved constants or linear rates, respectively.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Block‑smoothness inequality]\label{lem:smooth}
For any block $i$ and any $w\in\RR^{d}$,
\[
f\bigl(w-U_i\grad{i}f(w)\bigr)
\;\le\;
f(w)-\frac{1}{2L_i}\norm{\grad{i}f(w)}^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption \ref{ass:smooth} with $h=-\grad{i}f(w)$:
\[
f\bigl(w-U_i\grad{i}f(w)\bigr)
\le f(w)-\norm{\grad{i}f(w)}^{2}
      +\frac{L_i}{2}\norm{\grad{i}f(w)}^{2}
= f(w)-\Bigl(1-\frac{L_i}{2}\Bigr)\norm{\grad{i}f(w)}^{2}.
\]
Since $\eta\le 1/L_{\max}\le 1/L_i$, the factor $1-\frac{L_i}{2}\ge
\frac{1}{2L_i}$, yielding the claimed inequality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof expectations are taken
with respect to the random choice of the block $i_t$ conditioned on the past
iterates.

\noindent\textbf{Notation.} Let $w^{\star}$ be a minimizer of $f$.
Define $L_{\max}:=\max_i L_i$ and $p_i=1/B$ for uniform sampling.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step descent inequality.}  
Using Lemma \ref{lem:smooth} with stepsize $\eta\le 1/L_{\max}$,
\[
f(w_{t+1})
= f\!\bigl(w_t-\eta\,U_{i_t}\grad{i_t}f(w_t)\bigr)
\le f(w_t)-\frac{\eta}{2}\norm{\grad{i_t}f(w_t)}^{2}.
\tag{2}
\]

\item[Step 2.] \textbf{Take expectation over $i_t$.}  
Because $i_t$ is independent of $w_t$,
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\frac{\eta}{2}\sum_{i=1}^{B}p_i\norm{\grad{i}f(w_t)}^{2}.
\tag{3}
\]

\item[Step 3.] \textbf{Relate block gradient norm to full gradient norm.}  
Since $p_i=1/B$ and $\norm{\nabla f(w)}^{2}
   =\sum_{i=1}^{B}\norm{\grad{i}f(w)}^{2}$,
\[
\sum_{i=1}^{B}p_i\norm{\grad{i}f(w_t)}^{2}
= \frac1B\sum_{i=1}^{B}\norm{\grad{i}f(w_t)}^{2}
= \frac1B\norm{\nabla f(w_t)}^{2}.
\tag{4}
\]

\item[Step 4.] \textbf{Combine (3) and (4).}
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\frac{\eta}{2B}\norm{\nabla f(w_t)}^{2}.
\tag{5}
\]

\item[Step 5.] \textbf{Use convexity to lower‑bound $\norm{\nabla f(w_t)}^{2}$.}  
Convexity \eqref{ass:convex} gives
\[
f(w_t)-f(w^{\star})\le\langle\nabla f(w_t),w_t-w^{\star}\rangle
\le \norm{\nabla f(w_t)}\norm{w_t-w^{\star}}.
\tag{6}
\]
Thus
\[
\norm{\nabla f(w_t)}^{2}
\ge \frac{\bigl(f(w_t)-f(w^{\star})\bigr)^{2}}{\norm{w_t-w^{\star}}^{2}}.
\tag{7}
\]

\item[Step 6.] \textbf{Bound the denominator by $D^{2}$.}  
From Assumption \ref{ass:bounded}, $\norm{w_t-w^{\star}}^{2}\le D^{2}$ for all $t$,
because the update never increases the distance when $\eta\le 1/L_{\max}$
(see Lemma \ref{lem:smooth} with $h=0$). Hence
\[
\frac{1}{\norm{w_t-w^{\star}}^{2}}\ge\frac{1}{D^{2}}.
\tag{8}
\]

\item[Step 7.] \textbf{Insert (7)–(8) into (5).}
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\frac{\eta}{2B D^{2}}\bigl(f(w_t)-f(w^{\star})\bigr)^{2}.
\tag{9}
\]

\item[Step 8.] \textbf{Define the error sequence $e_t:=\EE[f(w_t)]-f(w^{\star})$.}  
Taking total expectation in (9) yields
\[
e_{t+1}\le e_t-\frac{\eta}{2B D^{2}}\,\EE\!\bigl[(f(w_t)-f(w^{\star}))^{2}\bigr].
\tag{10}
\]

\item[Step 9.] \textbf{Apply the inequality $a^{2}\ge a^{2}$ and Jensen.}  
Since $x\mapsto x^{2}$ is convex,
\[
\EE\!\bigl[(f(w_t)-f(w^{\star}))^{2}\bigr]\ge e_t^{2}.
\tag{11}
\]

\item[Step 10.] \textbf{Obtain a scalar recursion.}
\[
e_{t+1}\le e_t-\frac{\eta}{2B D^{2}}\,e_t^{2}.
\tag{12}
\]

\item[Step 11.] \textbf{Reciprocal transformation.}  
If $e_t>0$, divide both sides of (12) by $e_{t+1}e_t$:
\[
\frac{1}{e_{t+1}}-\frac{1}{e_t}\ge \frac{\eta}{2B D^{2}}.
\tag{13}
\]

\item[Step 12.] \textbf{Sum (13) from $t=0$ to $T-1$.}
\[
\frac{1}{e_T}-\frac{1}{e_0}\ge \frac{\eta T}{2B D^{2}}.
\tag{14}
\]

\item[Step 13.] \textbf{Upper‑bound $1/e_0$ by a non‑negative term.}  
Since $e_0\le f(w_0)-f(w^{\star})\le \frac{L_{\max}}{2}D^{2}$ by smoothness,
\[
\frac{1}{e_0}\ge 0.
\tag{15}
\]

\item[Step 14.] \textbf{Solve for $e_T$.}  
From (14)–(15),
\[
\frac{1}{e_T}\ge \frac{\eta T}{2B D^{2}}
\;\Longrightarrow\;
e_T\le \frac{2B D^{2}}{\eta T}.
\tag{16}
\]

\item[Step 15.] \textbf{Insert the maximal admissible stepsize.}  
Choosing $\eta=1/L_{\max}$ (largest step that satisfies Assumption
\ref{ass:stepsize}) gives
\[
e_T\le \frac{2B L_{\max} D^{2}}{T}\cdot\frac12
     =\frac{B L_{\max} D^{2}}{2T}.
\tag{17}
\]
Thus
\[
\EE\!\bigl[f(w_T)-f(w^{\star})\bigr]\le\frac{B L_{\max} D^{2}}{2T},
\]
which is exactly the claim of Theorem \ref{thm:main}.  $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The bound \eqref{17} can be rewritten as
\[
\boxed{\;
\EE\!\bigl[f(w_T)-f(w^{\star})\bigr]
\le \underbrace{\frac{B\,L_{\max}\,D^{2}}{2}}_{\displaystyle C}
\cdot\frac{1}{T}
\;}
\]
where the constant $C$ depends linearly on the number of blocks $B$,
the worst‑case block smoothness $L_{\max}$, and the initial distance $D$.

If the blocks have heterogeneous smoothness constants,
the bound can be refined to
\[
\EE\!\bigl[f(w_T)-f(w^{\star})\bigr]
\le \frac{D^{2}}{2T}\sum_{i=1}^{B}p_i^{-1}L_i .
\]

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Strongly convex $f$.}  
If $f$ is $\mu$‑strongly convex, the same analysis yields a linear rate
\[
\EE\!\bigl[f(w_T)-f(w^{\star})\bigr]
\le \bigl(1-\tfrac{\eta\mu}{B}\bigr)^{T}\bigl(f(w_0)-f(w^{\star})\bigr).
\]

\item \textbf{Non‑uniform sampling.}  
With probabilities $p_i$, the recursion (12) becomes
\[
e_{t+1}\le e_t-\frac{\eta}{2D^{2}}\Bigl(\sum_{i}p_i^{-1}\norm{\grad{i}f(w_t)}^{2}\Bigr) .
\]
The final bound contains $\displaystyle \frac{1}{p_{\min}}$ where
$p_{\min}=\min_i p_i$.

\item \textbf{Accelerated RBCD.}  
Nesterov‑type extrapolation applied to each block improves the constant
to $O\!\bigl(\frac{B\sqrt{L_{\max}}}{\sqrt{T}}\bigr)$ for smooth convex problems.
\end{enumerate}
