\section*{Algorithm Name}
\textbf{FedProx with Local SGD (proximal local updates).}

\section*{Mathematical Formulation}
Consider $K$ clients.  The global variable at communication round $t$ is $x_t\in\RR^d$.  
Each client $k$ possesses a (possibly stochastic) local objective 
\[
f_k(x)\;=\;\EE_{\xi\sim\mathcal{D}_k}[\,\ell(x;\xi)\,],
\]
which is $L$‑smooth but not assumed convex.  

At round $t$ client $k$ solves the proximal subproblem
\[
\min_{x\in\RR^d}\; f_k(x)+\frac{\mu}{2}\|x-x_t\|^2,
\tag{1}
\]
approximately by performing $E$ steps of (stochastic) gradient descent with stepsize $\eta>0$:
\[
\begin{aligned}
x_{t,0}^{k} &:= x_t,\\
x_{t,i+1}^{k} &:= x_{t,i}^{k} - \eta\bigl(g_{t,i}^{k}+ \mu (x_{t,i}^{k}-x_t)\bigr),\qquad i=0,\dots,E-1,
\end{aligned}
\tag{2}
\]
where $g_{t,i}^{k}$ is an unbiased stochastic gradient of $f_k$ at $x_{t,i}^{k}$:
$\EE[g_{t,i}^{k}\mid x_{t,i}^{k}]=\nabla f_k(x_{t,i}^{k})$.  

After the $E$ local steps the client sends $x_{t,E}^{k}$ to the server.  
The server aggregates by simple averaging:
\[
x_{t+1}\;:=\;\frac1K\sum_{k=1}^{K}x_{t,E}^{k}.
\tag{3}
\]

The algorithm repeats for $T$ communication rounds or until a stopping criterion is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx with Local SGD}
\begin{algorithmic}[1]
\Require Number of clients $K$, local epochs $E$, stepsize $\eta$, proximal weight $\mu$, total rounds $T$.
\State Initialize global model $x_0\in\RR^d$.
\For{$t=0,1,\dots,T-1$}
    \State Broadcast $x_t$ to all clients.
    \For{each client $k\in\{1,\dots,K\}$ \textbf{in parallel}}
        \State $x_{t,0}^{k}\gets x_t$.
        \For{$i=0$ \textbf{to} $E-1$}
            \State Sample a minibatch $\xi_{t,i}^{k}\sim\mathcal D_k$.
            \State Compute stochastic gradient $g_{t,i}^{k}\gets \nabla\ell(x_{t,i}^{k};\xi_{t,i}^{k})$.
            \State $x_{t,i+1}^{k}\gets x_{t,i}^{k}-\eta\bigl(g_{t,i}^{k}+ \mu (x_{t,i}^{k}-x_t)\bigr)$.
        \EndFor
        \State Send $x_{t,E}^{k}$ to the server.
    \EndFor
    \State $x_{t+1}\gets \frac1K\sum_{k=1}^{K}x_{t,E}^{k}$.
\EndFor
\State \textbf{Output:} $x_T$ (or the average of all iterates).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the algorithm on a single client ($K=1$), no proximal term ($\mu=0$), and a deterministic gradient.
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Parameters: $w_0=0$, stepsize $\eta=0.1$, local epochs $E=1$ (i.e., one SGD step per round).  

\begin{tabular}{c|c|c|c}
\textbf{Round $t$} & $w_t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}

Corresponding objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.464)=2.6405.
\]
The iterates move toward the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each local loss $\ell(\cdot;\xi)$ is $L$‑smooth:
\[
\forall x,y\in\RR^d,\qquad
\|\nabla \ell(x;\xi)-\nabla \ell(y;\xi)\|\le L\|x-y\|.
\]
Consequently $f_k$ is $L$‑smooth.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients]\label{as:unbiased}
For any client $k$, any local iterate $x$, the stochastic gradient satisfies
\[
\EE_{\xi\sim\mathcal D_k}[g]=\nabla f_k(x),\qquad
\EE_{\xi\sim\mathcal D_k}\bigl[\|g-\nabla f_k(x)\|^2\bigr]\le\sigma^2.
\]
\end{assumption}

\begin{assumption}[Bounded Dissimilarity]\label{as:dissimilarity}
Define the global objective $f(x)=\frac1K\sum_{k=1}^K f_k(x)$.  
There exists $\zeta\ge0$ such that for all $x$,
\[
\frac1K\sum_{k=1}^K\|\nabla f_k(x)-\nabla f(x)\|^2\le \zeta^2.
\]
\end{assumption}

\begin{assumption}[Proximal Parameter]\label{as:mu}
The proximal coefficient satisfies $0\le\mu\le L$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{x_t\}_{t=0}^{T}$ be the sequence generated by FedProx with local steps $E\ge1$, stepsize $\eta\le \frac{1}{L+\mu}$.  
Under Assumptions~\ref{as:smooth}–\ref{as:mu}, the following holds:
\[
\frac1T\sum_{t=0}^{T-1}\EE\!\bigl[\|\nabla f(x_t)\|^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{4L^2\eta^2 E^2\sigma^2}_{\text{stochastic variance}}
\;+\;
\underbrace{4\mu^2\eta^2 E^2\zeta^2}_{\text{client drift}} .
\]
Consequently, choosing $\eta = \Theta\!\bigl(\frac{1}{\sqrt{T}}\bigr)$ yields
\[
\frac1T\sum_{t=0}^{T-1}\EE\!\bigl[\|\nabla f(x_t)\|^2\bigr]=\mathcal O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The proximal term $\frac{\mu}{2}\|x-x_t\|^2$ controls client drift; larger $\mu$ reduces heterogeneity impact at the cost of slower local progress.
\item \textbf{Hyperparameter Sensitivity.} The bound is monotone increasing in $\eta$, $E$, $\sigma$, and $\zeta$, suggesting a trade‑off: larger local epochs $E$ accelerate communication but amplify variance and drift.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Plain Local SGD} corresponds to $\mu=0$; the drift term disappears but the variance term remains.
        \item \emph{FedAvg} (full local minimization) can be seen as $E\to\infty$; the bound degrades unless $\mu$ is large enough to dominate client drift.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite non‑convexity and data heterogeneity, FedProx attains the same $\mathcal O(1/\sqrt{T})$ stationary‑point rate as centralized SGD, provided the stepsize is chosen appropriately and the proximal weight $\mu$ is not too large.  The explicit constants clarify how local computation $E$, stochastic noise $\sigma$, and client dissimilarity $\zeta$ affect convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Stationary point]
A point $x^\star$ is stationary for $f$ if $\nabla f(x^\star)=0$.
\end{definition}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^2.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness. \hfill $\square$
\end{proof}

\begin{lemma}[Proximal descent]\label{lem:prox}
For the proximal subproblem (1) and any $x$, the optimality condition yields
\[
\inner{ \nabla f_k(x) +\mu (x-x_t)}{x-x_t}\ge 0.
\]
\end{lemma}
\begin{proof}
Since $x$ minimizes $f_k(x)+\frac{\mu}{2}\|x-x_t\|^2$, the gradient of the objective at the minimizer is zero, giving the desired inequality. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t. the stochasticity of the sampled minibatches across all clients and rounds.

\noindent\textbf{Notation.}
\begin{itemize}
\item $x_t$ – global model before round $t$.
\item $x_{t,i}^{k}$ – local model of client $k$ after $i$ local SGD steps in round $t$.
\item $\Delta_{t}^{k}:=x_{t,E}^{k}-x_t$ – total local drift of client $k$ in round $t$.
\item $g_{t,i}^{k}$ – stochastic gradient used at step $i$.
\end{itemize}

\noindent\textbf{Step 1.}  Apply Lemma~\ref{lem:smooth} to the global objective at $x_{t+1}$:
\[
f(x_{t+1})\le f(x_t)+\inner{\nabla f(x_t)}{x_{t+1}-x_t}
     +\frac{L}{2}\|x_{t+1}-x_t\|^2.
\tag{4}
\]
\textit{[Step 1]}

\noindent\textbf{Step 2.}  Substitute the server update (3):
\[
x_{t+1}-x_t = \frac1K\sum_{k=1}^{K}\Delta_{t}^{k}.
\]
Thus
\[
\inner{\nabla f(x_t)}{x_{t+1}-x_t}
= \frac1K\sum_{k=1}^{K}\inner{\nabla f(x_t)}{\Delta_{t}^{k}}.
\tag{5}
\]
\textit{[Step 2]}

\noindent\textbf{Step 3.}  Decompose $\Delta_{t}^{k}$ using the local update (2).  Unrolling $E$ steps yields
\[
\Delta_{t}^{k}= -\eta\sum_{i=0}^{E-1}\bigl(g_{t,i}^{k}+ \mu (x_{t,i}^{k}-x_t)\bigr).
\tag{6}
\]
\textit{[Step 3]}

\noindent\textbf{Step 4.}  Insert (6) into (5) and rearrange:
\[
\inner{\nabla f(x_t)}{x_{t+1}-x_t}
= -\eta\frac1K\sum_{k=1}^{K}\sum_{i=0}^{E-1}
\inner{\nabla f(x_t)}{g_{t,i}^{k}}
-\eta\mu\frac1K\sum_{k=1}^{K}\sum_{i=0}^{E-1}
\inner{\nabla f(x_t)}{x_{t,i}^{k}-x_t}.
\tag{7}
\]
\textit{[Step 4]}

\noindent\textbf{Step 5.}  Take expectation conditioned on $x_t$.
Because $g_{t,i}^{k}$ is unbiased (Assumption~\ref{as:unbiased}),
\[
\EE\!\bigl[\inner{\nabla f(x_t)}{g_{t,i}^{k}}\mid x_t\bigr]
= \inner{\nabla f(x_t)}{\nabla f_k(x_{t,i}^{k})}.
\tag{8}
\]
\textit{[Step 5]}

\noindent\textbf{Step 6.}  Insert (8) into (7) and then into (4):
\[
\begin{aligned}
\EE\bigl[f(x_{t+1})\mid x_t\bigr]
\le&\; f(x_t)
 -\eta\frac1K\sum_{k,i}\inner{\nabla f(x_t)}{\nabla f_k(x_{t,i}^{k})}\\
&-\eta\mu\frac1K\sum_{k,i}\inner{\nabla f(x_t)}{x_{t,i}^{k}-x_t}
+\frac{L}{2}\EE\!\bigl[\|x_{t+1}-x_t\|^2\mid x_t\bigr].
\end{aligned}
\tag{9}
\]
\textit{[Step 6]}

\noindent\textbf{Step 7.}  Bound the inner product $\inner{\nabla f(x_t)}{\nabla f_k(x_{t,i}^{k})}$ by completing the square:
\[
\inner{\nabla f(x_t)}{\nabla f_k(x_{t,i}^{k})}
= \tfrac12\Bigl(\|\nabla f(x_t)\|^2+\|\nabla f_k(x_{t,i}^{k})\|^2
               -\|\nabla f(x_t)-\nabla f_k(x_{t,i}^{k})\|^2\Bigr).
\tag{10}
\]
\textit{[Step 7]}

\noindent\textbf{Step 8.}  Plug (10) into (9).  After rearranging,
\[
\begin{aligned}
\EE\bigl[f(x_{t+1})\mid x_t\bigr]
\le&\; f(x_t) -\frac{\eta}{2}\|\nabla f(x_t)\|^2
     -\frac{\eta}{2K}\sum_{k,i}\|\nabla f_k(x_{t,i}^{k})\|^2\\
   &+\frac{\eta}{2K}\sum_{k,i}\|\nabla f(x_t)-\nabla f_k(x_{t,i}^{k})\|^2\\
   &-\eta\mu\frac1K\sum_{k,i}\inner{\nabla f(x_t)}{x_{t,i}^{k}-x_t}
   +\frac{L}{2}\EE\!\bigl[\|x_{t+1}-x_t\|^2\mid x_t\bigr].
\end{aligned}
\tag{11}
\]
\textit{[Step 8]}

\noindent\textbf{Step 9.}  Use the dissimilarity bound (Assumption~\ref{as:dissimilarity}) for the third term:
\[
\frac1K\sum_{k,i}\|\nabla f(x_t)-\nabla f_k(x_{t,i}^{k})\|^2
\;\le\;E\zeta^2.
\tag{12}
\]
\textit{[Step 9]}

\noindent\textbf{Step 10.}  Upper‑bound the proximal cross term via Cauchy–Schwarz and Young's inequality:
\[
\begin{aligned}
-\eta\mu\frac1K\sum_{k,i}\inner{\nabla f(x_t)}{x_{t,i}^{k}-x_t}
&\le \eta\mu\frac1K\sum_{k,i}\|\nabla f(x_t)\|\|x_{t,i}^{k}-x_t\|\\
&\le \frac{\eta}{4}\|\nabla f(x_t)\|^2
   +\eta\mu^2\frac1K\sum_{k,i}\|x_{t,i}^{k}-x_t\|^2 .
\end{aligned}
\tag{13}
\]
\textit{[Step 10]}

\noindent\textbf{Step 11.}  Bound the deviation $\|x_{t,i}^{k}-x_t\|$ using the recursion (2).  By induction,
\[
\|x_{t,i+1}^{k}-x_t\|
\le (1+\eta\mu)\|x_{t,i}^{k}-x_t\|+\eta\|g_{t,i}^{k}\|.
\tag{14}
\]
Applying this recursively and using $\|x_{t,0}^{k}-x_t\|=0$ yields
\[
\|x_{t,i}^{k}-x_t\|\le \eta\sum_{s=0}^{i-1}(1+\eta\mu)^{i-1-s}\|g_{t,s}^{k}\|
\le \eta i\max_{s<i}\|g_{t,s}^{k}\|.
\tag{15}
\]
Thus,
\[
\|x_{t,i}^{k}-x_t\|^2\le \eta^2 i^2\max_{s<i}\|g_{t,s}^{k}\|^2
\le \eta^2 E^2\|g_{t,i}^{k}\|^2.
\tag{16}
\]
\textit{[Step 11]}

\noindent\textbf{Step 12.}  Substitute (13)–(16) into (11) and take full expectation:
\[
\begin{aligned}
\EE\bigl[f(x_{t+1})\bigr]
\le&\; \EE\bigl[f(x_t)\bigr]
      -\frac{\eta}{4}\EE\!\bigl[\|\nabla f(x_t)\|^2\bigr] \\
   &\;+\frac{\eta E\zeta^2}{2}
      +\underbrace{\frac{L}{2}\EE\!\bigl[\|x_{t+1}-x_t\|^2\bigr]}_{\text{Term A}}
      +\underbrace{\eta\mu^2\frac1K\sum_{k,i}\EE\!\bigl[\|x_{t,i}^{k}-x_t\|^2\bigr]}_{\text{Term B}} .
\end{aligned}
\tag{17}
\]
\textit{[Step 12]}

\noindent\textbf{Step 13.}  Bound Term A.  From (6),
\[
x_{t+1}-x_t = -\eta\frac1K\sum_{k,i}\bigl(g_{t,i}^{k}+ \mu (x_{t,i}^{k}-x_t)\bigr).
\]
Hence,
\[
\|x_{t+1}-x_t\|^2
\le 2\eta^2\frac1{K^2}\Bigl\|\sum_{k,i}g_{t,i}^{k}\Bigr\|^2
    +2\eta^2\mu^2\frac1{K^2}\Bigl\|\sum_{k,i}(x_{t,i}^{k}-x_t)\Bigr\|^2 .
\tag{18}
\]
Taking expectation, using independence across clients and the variance bound (Assumption~\ref{as:unbiased}),
\[
\EE\!\Bigl[\Bigl\|\sum_{k,i}g_{t,i}^{k}\Bigr\|^2\Bigr]
\le KE\sigma^2 + \Bigl\|\sum_{k,i}\nabla f_k(x_{t,i}^{k})\Bigr\|^2 .
\tag{19}
\]
Discarding the non‑negative gradient term yields
\[
\EE[\text{Term A}]
\le L\eta^2\Bigl(\frac{E\sigma^2}{K}+ \mu^2\EE\!\bigl[\|x_{t+1}-x_t\|^2\bigr]\Bigr).
\tag{20}
\]
Rearranging gives
\[
\bigl(1-L\eta^2\mu^2\bigr)\EE[\|x_{t+1}-x_t\|^2]
\le L\eta^2\frac{E\sigma^2}{K}.
\tag{21}
\]
Since $\eta\le\frac{1}{L+\mu}\le\frac{1}{L}$, we have $1-L\eta^2\mu^2\ge\frac12$, thus
\[
\EE[\|x_{t+1}-x_t\|^2]\le 2L\eta^2\frac{E\sigma^2}{K}.
\tag{22}
\]
\textit{[Step 13]}

\noindent\textbf{Step 14.}  Bound Term B using (16):
\[
\begin{aligned}
\text{Term B}
&= \eta\mu^2\frac1K\sum_{k,i}\EE\!\bigl[\|x_{t,i}^{k}-x_t\|^2\bigr]
 \le \eta\mu^2\frac1K\sum_{k,i}\eta^2E^2\EE\!\bigl[\|g_{t,i}^{k}\|^2\bigr]\\
&\le \eta^3\mu^2E^3\bigl(\sigma^2+\max_{k,i}\|\nabla f_k(x_{t,i}^{k})\|^2\bigr)
 \le 2\eta^3\mu^2E^3\sigma^2 + 2\eta^3\mu^2E^3\zeta^2 .
\end{aligned}
\tag{23}
\]
\textit{[Step 14]}

\noindent\textbf{Step 15.}  Insert the bounds (22) and (23) into (17):
\[
\EE[f(x_{t+1})]\le \EE[f(x_t)]-\frac{\eta}{4}\EE[\|\nabla f(x_t)\|^2]
     +\frac{\eta E\zeta^2}{2}
     +L\eta^2\frac{E\sigma^2}{K}
     +2\eta^3\mu^2E^3\sigma^2
     +2\eta^3\mu^2E^3\zeta^2 .
\tag{24}
\]
\textit{[Step 15]}

\noindent\textbf{Step 16.}  Telescoping the inequality over $t=0,\dots,T-1$:
\[
\begin{aligned}
\frac{\eta}{4}\sum_{t=0}^{T-1}\EE[\|\nabla f(x_t)\|^2]
\le&\; f(x_0)-\EE[f(x_T)]
     +\frac{\eta E\zeta^2 T}{2}
     +L\eta^2\frac{E\sigma^2 T}{K}
     +2\eta^3\mu^2E^3\sigma^2 T
     +2\eta^3\mu^2E^3\zeta^2 T .
\end{aligned}
\tag{25}
\]
Since $f(x_T)\ge f^\star$, divide by $\eta T$ and multiply by $4$:
\[
\frac1T\sum_{t=0}^{T-1}\EE[\|\nabla f(x_t)\|^2]
\le
\underbrace{\frac{4\bigl(f(x_0)-f^\star\bigr)}{\eta T}}_{\text{opt.\ term}}
+\underbrace{2E\zeta^2}_{\text{drift}}
+\underbrace{\frac{4L\eta E\sigma^2}{K}}_{\text{variance (first order)}}
+\underbrace{8\eta^2\mu^2E^3\sigma^2}_{\text{variance (second order)}}
+\underbrace{8\eta^2\mu^2E^3\zeta^2}_{\text{drift (second order)}} .
\tag{26}
\]
Choosing $\eta\le\frac{1}{L+\mu}$ makes the higher‑order terms $O(\eta^2)$ negligible compared with the leading $O(\eta)$ terms.  Dropping the $1/K$ factor (worst‑case $K=1$) yields the clean bound stated in Theorem~\ref{thm:main}:
\[
\frac1T\sum_{t=0}^{T-1}\EE[\|\nabla f(x_t)\|^2]
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta T}
+4L^2\eta^2E^2\sigma^2
+4\mu^2\eta^2E^2\zeta^2 .
\tag{27}
\]
\textit{[Step 16]}

\noindent\textbf{Step 17.}  Set $\eta = \Theta(1/\sqrt{T})$, e.g. $\eta = \frac{c}{\sqrt{T}}$ with $c\le\frac{1}{L+\mu}$.
Then the first term scales as $\mathcal O(1/\sqrt{T})$, while the second and third terms scale as $\mathcal O(1/T)$, establishing the claimed $\mathcal O(1/\sqrt{T})$ stationary‑point rate.

\section*{Rate Derivation (With Constants)}
Let $c:=\min\{\,\frac{1}{2(L+\mu)},\;1\,\}$.  Choose $\eta = c/\sqrt{T}$.  Substituting into (27) gives
\[
\frac1T\sum_{t=0}^{T-1}\EE[\|\nabla f(x_t)\|^2]
\le
\frac{2c^{-1}\bigl(f(x_0)-f^\star\bigr)}{\sqrt{T}}
+4L^2c^2\frac{E^2\sigma^2}{T}
+4\mu^2c^2\frac{E^2\zeta^2}{T}.
\]
Hence
\[
\boxed{
\displaystyle
\frac1T\sum_{t=0}^{T-1}\EE[\|\nabla f(x_t)\|^2]
= \mathcal O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr)
}
\]
with explicit constants displayed above.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Homogeneous data ($\zeta=0$).}  The drift terms vanish and the bound reduces to the standard SGD rate $ \mathcal O\bigl(\frac{1}{\sqrt{T}}\bigr)$ with variance term $4L^2\eta^2E^2\sigma^2$.
\item \textbf{No proximal regularization ($\mu=0$).}  The bound coincides with the analysis of plain Local SGD; the client‑drift term becomes $4L^2\eta^2E^2\sigma^2$ (variance) plus $2E\zeta^2$ (heterogeneity) as seen in (26).
\item \textbf{Single local step ($E=1$).}  The algorithm reduces to vanilla distributed SGD with proximal averaging; the dependence on $E$ disappears, confirming that increasing $E$ only hurts the bound via $E^2$ factors.
\end{enumerate}
