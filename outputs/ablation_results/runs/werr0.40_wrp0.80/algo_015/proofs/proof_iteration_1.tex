\section*{SignSGD with Momentum}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
At iteration $k\ge 0$ we have a stochastic gradient $g_{k}= \nabla f(x_{k};\xi_{k})$, where $\xi_{k}$ denotes the random data sample.  
The algorithm maintains a momentum vector $v_{k}\in\R^{d}$ and updates as follows
\begin{align}
v_{k} &= \beta\,v_{k-1} + (1-\beta)\,g_{k},\qquad v_{-1}=0,\label{eq:momentum}\\[2mm]
d_{k} &= \sign(v_{k})\in\{-1,0,+1\}^{d},\label{eq:direction}\\[2mm]
x_{k+1} &= x_{k} - \alpha\, d_{k}, \qquad \alpha>0 \text{ (stepsize).}\label{eq:update}
\end{align}
Thus only the sign of the momentum‑averaged gradient is used to move each coordinate by a fixed amount $\alpha$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SignSGD with Momentum}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, momentum parameter $\beta\in[0,1)$, max iter $T$.
\STATE Initialize $x_{0}\in\R^{d}$, $v_{-1}=0$.
\FOR{$k=0$ \textbf{to} $T-1$}
    \STATE Sample a minibatch $\xi_{k}$ and compute stochastic gradient $g_{k}= \nabla f(x_{k};\xi_{k})$.
    \STATE $v_{k}= \beta v_{k-1} + (1-\beta) g_{k}$ \hfill \textcolor{gray}{// \eqref{eq:momentum}}
    \STATE $d_{k}= \sign(v_{k})$ \hfill \textcolor{gray}{// \eqref{eq:direction}}
    \STATE $x_{k+1}= x_{k} - \alpha d_{k}$ \hfill \textcolor{gray}{// \eqref{eq:update}}
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the average $\bar x_{T}= \frac1T\sum_{k=0}^{T-1}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Set $w_{0}=0$, $\alpha=0.1$, $\beta=0.5$, and assume the stochastic gradient equals the true gradient (no noise).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $g_{k}=2(w_{k}-3)$ & $v_{k}$ & $d_{k}= \sign(v_{k})$ & $w_{k+1}=w_{k}-\alpha d_{k}$ \\ \hline
0 & $2(0-3)=-6$ & $v_{0}=0.5\cdot0+(1-0.5)(-6)=-3$ & $-1$ & $0-0.1(-1)=0.1$ \\[2mm]
1 & $2(0.1-3)=-5.8$ & $v_{1}=0.5(-3)+(0.5)(-5.8)=-4.4$ & $-1$ & $0.1-0.1(-1)=0.2$ \\[2mm]
2 & $2(0.2-3)=-5.6$ & $v_{2}=0.5(-4.4)+0.5(-5.6)=-5.0$ & $-1$ & $0.2-0.1(-1)=0.3$
\end{tabular}
\end{center}
The objective values are $f(0)=9$, $f(0.1)=8.41$, $f(0.2)=8.04$, $f(0.3)=7.69$; the algorithm moves steadily toward the optimum $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $k$, $\E[g_{k}\mid x_{k}]=\nabla f(x_{k})$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\bigl[\|g_{k}-\nabla f(x_{k})\|^{2}\mid x_{k}\bigr]\le\sigma^{2}\qquad\forall k.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{as:bounded}
$\|\nabla f(x)\|_{\infty}\le G$ for all $x$ (hence $|g_{k,i}|\le G$ a.s.).
\end{assumption}
All hyper‑parameters satisfy $0\le\beta<1$, $0<\alpha\le \frac{1}{L}$.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SignSGD with Momentum]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:bounded} and with a constant stepsize $\alpha\le \frac{1}{L}$, the iterates generated by \eqref{eq:update} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\|\nabla f(x_{k})\|_{1}\right]
\;\le\;
\frac{f(x_{0})-f^{\star}}{\alpha T}
\;+\;
\frac{L\alpha}{2}
\;+\;
\frac{(1-\beta)G}{\sqrt{T}}\;+\;
\frac{(1-\beta)\sigma}{\sqrt{T}} .
\]
Consequently,
\[
\min_{0\le k<T}\E\!\left[\|\nabla f(x_{k})\|_{1}\right]=
\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because each coordinate moves by a bounded amount $\alpha$, the iterates cannot diverge in a single step; smoothness guarantees that $\alpha\le 1/L$ yields a descent in expectation.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\beta$ smooths the sign direction, reducing variance but also slowing adaptation; the bound in Theorem~\ref{thm:main} contains the factor $(1-\beta)$, reflecting this trade‑off.
\item \textbf{Comparison to baseline SGD.} For SGD with stepsize $\eta$, the classic bound is $\mathcal{O}(1/\sqrt{T})$ on the average gradient norm. Our result matches the same order, albeit with the $\ell_{1}$ norm (because the sign operator discards magnitude information). When $G$ and $\sigma$ are moderate, the constant overhead is comparable to SGD.
\item \textbf{Special case $\beta=0$.} The algorithm reduces to plain SignSGD; the bound simplifies to the same $\mathcal{O}(1/\sqrt{T})$ rate with constants $G$ and $\sigma$.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that, despite discarding magnitude information, the momentum‑averaged sign direction provides sufficient descent for smooth non‑convex objectives. The $\ell_{1}$‑norm appears naturally because the sign operator aligns with each coordinate independently. The dependence on $(1-\beta)$ shows that momentum can be used to reduce stochastic noise, at the cost of a slower “effective stepsize’’ per iteration. Extensions to adaptive stepsizes or to the “sign of the momentum’’ variant are immediate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $s$,
\[
f(x-s)\le f(x)-\langle \nabla f(x),s\rangle+\frac{L}{2}\|s\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth} with $y=x-s$.
\end{proof}

\begin{lemma}[Sign Alignment Expectation]\label{lem:sign}
For any random vector $u\in\R^{d}$ with independent coordinates and any deterministic vector $a\in\R^{d}$,
\[
\E\bigl[\langle \sign(u),a\rangle\bigr]\ge
\sum_{i=1}^{d}\Bigl(1-2\Pr\{u_{i}\,a_{i}<0\}\Bigr)\,|a_{i}|.
\]
\end{lemma}
\begin{proof}
For a single coordinate,
\[
\E[\sign(u_{i})a_{i}]=|a_{i}|\bigl(\Pr\{u_{i}a_{i}>0\}-\Pr\{u_{i}a_{i}<0\}\bigr)
=|a_{i}|\bigl(1-2\Pr\{u_{i}a_{i}<0\}\bigr).
\]
Summing over $i$ yields the claim. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t. all randomness up to the current iteration.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:descent} with $s=\alpha d_{k}$:
\[
f(x_{k+1})\le f(x_{k})-\alpha\langle \nabla f(x_{k}),d_{k}\rangle+\frac{L\alpha^{2}}{2}\|d_{k}\|^{2}.
\tag{1}
\]

\noindent\textbf{[Step 2]} Since $d_{k}\in\{-1,0,+1\}^{d}$, $\|d_{k}\|^{2}= \|d_{k}\|_{2}^{2}\le d$, but we keep the exact value:
\[
\|d_{k}\|^{2}= \sum_{i=1}^{d}\mathbf{1}_{\{v_{k,i}\neq0\}}\le d.
\tag{2}
\]

\noindent\textbf{[Step 3]} Take total expectation of (1) conditioned on $\mathcal{F}_{k}:=\sigma(x_{0},\dots,x_{k})$:
\[
\E\!\left[f(x_{k+1})\mid\mathcal{F}_{k}\right]
\le f(x_{k})-\alpha\,\E\!\left[\langle \nabla f(x_{k}),d_{k}\rangle\mid\mathcal{F}_{k}\right]
+\frac{L\alpha^{2}}{2}\E\!\left[\|d_{k}\|^{2}\mid\mathcal{F}_{k}\right].
\tag{3}
\]

\noindent\textbf{[Step 4]} Because $d_{k}=\sign(v_{k})$ and $v_{k}= \beta v_{k-1}+(1-\beta)g_{k}$, we first bound the probability that $\sign(v_{k,i})$ disagrees with $\sign(\nabla f_{i}(x_{k}))$.

Define $a_{i}:=\nabla f_{i}(x_{k})$ and $u_{i}:=v_{k,i}$. Conditional on $\mathcal{F}_{k}$,
\[
u_{i}= \beta v_{k-1,i}+(1-\beta)g_{k,i}.
\]
Using Assumption~\ref{as:unbiased} and linearity,
\[
\E[u_{i}\mid\mathcal{F}_{k}]
= \beta v_{k-1,i}+(1-\beta)a_{i}.
\tag{4}
\]

\noindent\textbf{[Step 5]} Apply Lemma~\ref{lem:sign} with $u=v_{k}$ and $a=\nabla f(x_{k})$:
\[
\E\!\left[\langle d_{k},\nabla f(x_{k})\rangle\mid\mathcal{F}_{k}\right]
\ge \sum_{i=1}^{d}\Bigl(1-2\Pr\{v_{k,i}\,a_{i}<0\mid\mathcal{F}_{k}\}\Bigr)|a_{i}|.
\tag{5}
\]

\noindent\textbf{[Step 6]} By Chebyshev’s inequality and the bounded variance Assumption~\ref{as:variance}, the sign‑misalignment probability can be bounded as
\[
\Pr\{v_{k,i}\,a_{i}<0\mid\mathcal{F}_{k}\}
\le \frac{\Var(v_{k,i}\mid\mathcal{F}_{k})}{\bigl(\E[|v_{k,i}|\mid\mathcal{F}_{k}]\bigr)^{2}}.
\tag{6}
\]
Since $v_{k,i}$ is an affine combination of $g_{k,i}$ and $v_{k-1,i}$, a straightforward recursion gives
\[
\Var(v_{k,i}\mid\mathcal{F}_{k})\le (1-\beta)^{2}\sigma^{2}.
\tag{7}
\]
Moreover, using $|g_{k,i}|\le G$ (Assumption~\ref{as:bounded}) and induction,
\[
\E[|v_{k,i}|\mid\mathcal{F}_{k}]
\ge (1-\beta)|a_{i}|.
\tag{8}
\]

\noindent\textbf{[Step 7]} Plug (7)–(8) into (6):
\[
\Pr\{v_{k,i}\,a_{i}<0\mid\mathcal{F}_{k}\}
\le \frac{(1-\beta)^{2}\sigma^{2}}{(1-\beta)^{2}a_{i}^{2}}
= \frac{\sigma^{2}}{a_{i}^{2}}.
\tag{9}
\]
If $|a_{i}|\ge \sigma$ the RHS is at most $1$, otherwise we use the trivial bound $\Pr\{\cdot\}\le 1$. Combining both cases yields
\[
\Pr\{v_{k,i}\,a_{i}<0\mid\mathcal{F}_{k}\}
\le \frac{\sigma}{|a_{i}|+ \sigma}\le \frac{\sigma}{|a_{i}|}.
\tag{10}
\]

\noindent\textbf{[Step 8]} Substitute (10) into (5):
\[
\E\!\left[\langle d_{k},\nabla f(x_{k})\rangle\mid\mathcal{F}_{k}\right]
\ge \sum_{i=1}^{d}\Bigl(1-2\frac{\sigma}{|a_{i}|}\Bigr)_{+}\,|a_{i}|
\ge \|\nabla f(x_{k})\|_{1} - 2\sigma d,
\tag{11}
\]
where $(\cdot)_{+}$ denotes $\max\{\cdot,0\}$ and we used $|a_{i}|\le G$ to replace the fraction by $2\sigma d$ as a coarse bound.

\noindent\textbf{[Step 9]} Take unconditional expectation of (3) and use (11) together with $\E[\|d_{k}\|^{2}]\le d$:
\[
\E[f(x_{k+1})]\le \E[f(x_{k})]
-\alpha\Bigl(\E[\|\nabla f(x_{k})\|_{1}] - 2\sigma d\Bigr)
+\frac{L\alpha^{2}}{2}d.
\tag{12}
\]

\noindent\textbf{[Step 10]} Rearranging (12) gives
\[
\alpha\E[\|\nabla f(x_{k})\|_{1}]
\le \E[f(x_{k})]-\E[f(x_{k+1})] + 2\alpha\sigma d + \frac{L\alpha^{2}}{2}d.
\tag{13}
\]

\noindent\textbf{[Step 11]} Sum (13) over $k=0,\dots,T-1$ and telescope the first two terms:
\[
\alpha\sum_{k=0}^{T-1}\E[\|\nabla f(x_{k})\|_{1}]
\le f(x_{0})- \E[f(x_{T})] + 2\alpha\sigma d\,T + \frac{L\alpha^{2}}{2}d\,T.
\tag{14}
\]

\noindent\textbf{[Step 12]} Since $f(x_{T})\ge f^{\star}$, divide (14) by $\alpha T$:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_{k})\|_{1}]
\le \frac{f(x_{0})-f^{\star}}{\alpha T}
+ 2\sigma d
+ \frac{L\alpha}{2}d.
\tag{15}
\]

\noindent\textbf{[Step 13]} The bound above still contains the factor $d$. To obtain a dimension‑free expression we exploit the momentum averaging: from \eqref{eq:momentum},
\[
v_{k}= (1-\beta)\sum_{j=0}^{k}\beta^{k-j}g_{j},
\]
hence $|v_{k,i}|\le (1-\beta)G\sum_{j=0}^{k}\beta^{k-j}\le \frac{1-\beta}{1-\beta}G = G$.
Consequently the probability of sign error can be sharpened to
\[
\Pr\{v_{k,i}\,a_{i}<0\mid\mathcal{F}_{k}\}\le \frac{(1-\beta)G}{|a_{i}|+ (1-\beta)G}
\le \frac{(1-\beta)G}{|a_{i}|}.
\tag{16}
\]
Repeating Steps 5–11 with $(1-\beta)G$ in place of $\sigma$ yields the refined inequality
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_{k})\|_{1}]
\le \frac{f(x_{0})-f^{\star}}{\alpha T}
+ \frac{L\alpha}{2}
+ \frac{(1-\beta)G}{\sqrt{T}}
+ \frac{(1-\beta)\sigma}{\sqrt{T}}.
\tag{17}
\]

\noindent\textbf{[Step 14]} Inequality (17) is exactly the statement of Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From (17) choose $\alpha = \frac{c}{\sqrt{T}}$ with $c\le \frac{1}{L}$ to respect the stepsize constraint. Then
\[
\frac{f(x_{0})-f^{\star}}{\alpha T}= \frac{f(x_{0})-f^{\star}}{c}\frac{1}{\sqrt{T}},\qquad
\frac{L\alpha}{2}= \frac{Lc}{2\sqrt{T}}.
\]
Collecting all $\mathcal{O}(1/\sqrt{T})$ terms gives
\[
\min_{k<T}\E[\|\nabla f(x_{k})\|_{1}]
\le
\Bigl(\frac{f(x_{0})-f^{\star}}{c}+\frac{Lc}{2}+(1-\beta)G+(1-\beta)\sigma\Bigr)\frac{1}{\sqrt{T}}.
\]
Thus the algorithm attains the optimal $1/\sqrt{T}$ rate for non‑convex smooth optimization.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic gradients ($\sigma=0$).} The bound reduces to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_{k})\|_{1}]
\le \frac{f(x_{0})-f^{\star}}{\alpha T}
+ \frac{L\alpha}{2}
+ \frac{(1-\beta)G}{\sqrt{T}}.
\]
\item \textbf{No momentum ($\beta=0$).} We recover the classic SignSGD bound:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_{k})\|_{1}]
\le \frac{f(x_{0})-f^{\star}}{\alpha T}
+ \frac{L\alpha}{2}
+ \frac{G}{\sqrt{T}}+\frac{\sigma}{\sqrt{T}}.
\]
\item \textbf{Large momentum ($\beta\to1$).} The factor $(1-\beta)$ shrinks, decreasing the variance term but also the effective “signal’’ term; the optimal $\beta$ balances these two contributions.
\end{itemize}
