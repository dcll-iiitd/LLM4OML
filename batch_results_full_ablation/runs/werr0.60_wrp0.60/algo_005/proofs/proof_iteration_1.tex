\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) with Double‑Loop Structure}

\section*{Mathematical Formulation}
Let the finite‑sum objective be  
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\R^{d},
\]
where each component $f_i$ is $L$‑smooth and $\mu$‑strongly convex.  
SVRG proceeds in epochs $s=0,1,2,\dots$.
\begin{enumerate}[label=\arabic*)]
\item \textbf{Full‑gradient snapshot.} At the beginning of epoch $s$ choose a reference point 
\[
\tilde{x}^{\,s}= 
\begin{cases}
x_{0}, & s=0,\\[2mm]
x_{m}^{\,s-1}, & s\ge 1,
\end{cases}
\]
and compute the exact gradient
\[
\tilde{\mu}^{\,s}= \nabla f(\tilde{x}^{\,s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s}).
\]

\item \textbf{Inner loop (variance‑reduced SGD).} Initialise $x_{0}^{\,s}= \tilde{x}^{\,s}$.  
For $t=0,\dots,m-1$ draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and form the estimator
\[
v_{t}^{\,s}= \nabla f_{i_t}\!\left(x_{t}^{\,s}\right)-\nabla f_{i_t}\!\left(\tilde{x}^{\,s}\right)+\tilde{\mu}^{\,s}.
\]
Update
\[
x_{t+1}^{\,s}= x_{t}^{\,s}-\eta\,v_{t}^{\,s},
\]
where $\eta>0$ is a constant stepsize.

\item \textbf{Epoch transition.} Set the next reference point $\tilde{x}^{\,s+1}=x_{m}^{\,s}$.
\end{enumerate}
The algorithm terminates after $S$ epochs or when a prescribed tolerance is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (double‑loop)}\label{alg:svrg}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsize $\eta>0$, inner‑loop length $m$, epochs $S$.
\STATE $\tilde{x}^{\,0}\gets x_{0}$
\FOR{epoch $s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{\,s}\gets \nabla f(\tilde{x}^{\,s})$
    \STATE $x_{0}^{\,s}\gets \tilde{x}^{\,s}$
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \STATE $v_{t}^{\,s}\gets \nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$
        \STATE $x_{t+1}^{\,s}\gets x_{t}^{\,s}-\eta\,v_{t}^{\,s}$
    \ENDFOR
    \STATE $\tilde{x}^{\,s+1}\gets x_{m}^{\,s}$
\ENDFOR
\RETURN $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem $n=1$ with 
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, stepsize $\eta=0.1$, inner‑loop length $m=3$, and run one epoch.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_{t}$ & $v_{t}$ (SVRG estimator) & $f(x_{t})$\\ \hline
0 & $0$ & $2(0-3)-2(0-3)+2(0-3)=2(0-3)=-6$ & $9$\\
1 & $x_{1}=0-0.1(-6)=0.6$ & $2(0.6-3)-2(0-3)+(-6)=2(-2.4)+6-6=-4.8$ & $5.76$\\
2 & $x_{2}=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)-2(0-3)+(-6)=2(-1.92)+6-6=-3.84$ & $3.6864$\\
3 & $x_{3}=1.08-0.1(-3.84)=1.464$ & (not needed) & $2.3510$\\
\end{tabular}
\end{center}
After three inner updates the reference point becomes $\tilde{x}^{\,1}=1.464$, already much closer to the optimum $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
Each $f_i$ (hence $f$) is $\mu$‑strongly convex:
\[
f_i(y)\ge f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{as:finite}
The objective is a finite average $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$ with $n\ge 1$.
\end{assumption}
All constants satisfy $0<\mu\le L$ and the stepsize obeys
\[
0<\eta\le\frac{1}{3L}. \tag{A1}
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Let Assumptions~\ref{as:smooth}–\ref{as:finite} hold and choose stepsize $\eta$ satisfying (A1).  
Define the contraction factor
\[
\rho\;:=\;\bigl(1-\eta\mu\bigr)^{m}+ \frac{\eta L^{2}}{\mu\bigl(1-\eta L\bigr)}.
\]
If the inner‑loop length $m$ and $\eta$ are selected such that $\rho<1$, then for the sequence of snapshot points $\{\tilde{x}^{\,s}\}$ produced by Algorithm~\ref{alg:svrg} we have
\[
\E\bigl[\|\tilde{x}^{\,s+1}-x^{*}\|^{2}\bigr]\;\le\;\rho\;\E\bigl[\|\tilde{x}^{\,s}-x^{*}\|^{2}\bigr],\qquad\forall s\ge0,
\]
where $x^{*}$ is the unique minimiser of $f$. Consequently,
\[
\E\bigl[f(\tilde{x}^{\,s})-f(x^{*})\bigr]\le \frac{L}{2}\rho^{s}\,\|\tilde{x}^{\,0}-x^{*}\|^{2}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The recursion in Theorem~\ref{thm:linear} is a deterministic linear contraction, guaranteeing geometric decay irrespective of stochastic sampling noise.
\item \textbf{Hyper‑parameter sensitivity.} The factor $\rho$ decreases when $\eta$ is increased up to $1/(3L)$ and when $m$ is enlarged; a common practical choice is $\eta=1/(10L)$ and $m=2n$, yielding $\rho\le 1/2$.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}[label=(\alph*)]
    \item \textit{Plain SGD} with diminishing stepsizes attains $O(1/T)$ sub‑linear rates for strongly convex problems.
    \item \textit{SVRG} achieves a \emph{linear} rate with a per‑epoch cost of $n+m$ gradient evaluations, i.e. $O((n+m)\log(1/\varepsilon))$ to reach $\varepsilon$ accuracy.
    \end{enumerate}
\item \textbf{Special case $m\to\infty$.} When $m$ grows without bound, the first term $(1-\eta\mu)^{m}\to0$ and $\rho\to \frac{\eta L^{2}}{\mu(1-\eta L)}$, reproducing the rate of the deterministic gradient method with stepsize $\eta$.
\end{itemize}

\section*{Discussion}
The theorem underscores that, under strong convexity and smoothness, the variance‑reduced estimator eliminates the stochastic error after each epoch, leaving only a deterministic contraction. The double‑loop design isolates the expensive full‑gradient computation to the outer loop while keeping the inner stochastic updates cheap, which explains the practical efficiency of SVRG on large‑scale finite‑sum problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness]\label{lem:unbiased}
For any $t$ and epoch $s$, conditional on $x_{t}^{\,s}$,
\[
\E_{i_t}[\,v_{t}^{\,s}\mid x_{t}^{\,s}\,]=\nabla f(x_{t}^{\,s}).
\]
\end{lemma}
\begin{proof}
\[
\E_{i_t}[v_{t}^{\,s}]
=\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_{t}^{\,s})-\nabla f_i(\tilde{x}^{\,s})\bigr)+\tilde{\mu}^{\,s}
= \nabla f(x_{t}^{\,s})-\nabla f(\tilde{x}^{\,s})+\nabla f(\tilde{x}^{\,s})=\nabla f(x_{t}^{\,s}).
\] 
\end{proof}

\begin{lemma}[Variance bound]\label{lem:var}
For any $t$ and epoch $s$,
\[
\E_{i_t}\!\bigl[\|v_{t}^{\,s}\|^{2}\mid x_{t}^{\,s}\bigr]
\le 4L\bigl(f(x_{t}^{\,s})-f(x^{*})\bigr)+4L\bigl(f(\tilde{x}^{\,s})-f(x^{*})\bigr).
\]
\end{lemma}
\begin{proof}
Using smoothness \eqref{as:smooth} and the identity
\[
\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2},
\]
we obtain
\[
\begin{aligned}
\|v_{t}^{\,s}\|^{2}
&=\bigl\|\nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(x^{*})-\bigl(\nabla f_{i_t}(\tilde{x}^{\,s})-\nabla f_{i_t}(x^{*})\bigr)+\tilde{\mu}^{\,s}\bigr\|^{2}\\
&\le 2\bigl\|\nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(x^{*})\bigr\|^{2}
   +2\bigl\|\nabla f_{i_t}(\tilde{x}^{\,s})-\nabla f_{i_t}(x^{*})-\tilde{\mu}^{\,s}\bigr\|^{2}.
\end{aligned}
\]
Taking expectation over $i_t$ and applying $L$‑smoothness gives
\[
\E\bigl[\|v_{t}^{\,s}\|^{2}\bigr]\le 2L^{2}\E\bigl[\|x_{t}^{\,s}-x^{*}\|^{2}\bigr]+2L^{2}\E\bigl[\|\tilde{x}^{\,s}-x^{*}\|^{2}\bigr].
\]
Strong convexity $f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}$ implies
\[
\frac{\mu}{2}\|x-x^{*}\|^{2}\le f(x)-f(x^{*}),
\]
hence $\|x-x^{*}\|^{2}\le\frac{2}{\mu}\bigl(f(x)-f(x^{*})\bigr)$. Substituting and using $L\ge\mu$ yields the claimed bound. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout, expectations are taken over all random choices up to the current iteration.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step recursion.}  
From the inner update $x_{t+1}=x_{t}-\eta v_{t}$,
\[
\|x_{t+1}-x^{*}\|^{2}
= \|x_{t}-x^{*}\|^{2}
-2\eta\langle v_{t},x_{t}-x^{*}\rangle
+\eta^{2}\|v_{t}\|^{2}. \tag{1}
\]

\item[Step 2.] \textbf{Take conditional expectation.}  
Using Lemma~\ref{lem:unbiased} and the fact that $\E[\langle v_{t},x_{t}-x^{*}\rangle]=\langle\nabla f(x_{t}),x_{t}-x^{*}\rangle$, we obtain
\[
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\mid x_{t}\bigr]
= \|x_{t}-x^{*}\|^{2}
-2\eta\langle\nabla f(x_{t}),x_{t}-x^{*}\rangle
+\eta^{2}\E\!\bigl[\|v_{t}\|^{2}\mid x_{t}\bigr]. \tag{2}
\]

\item[Step 3.] \textbf{Apply strong convexity.}  
From Assumption~\ref{as:strong},
\[
\langle\nabla f(x_{t}),x_{t}-x^{*}\rangle\ge f(x_{t})-f(x^{*})+\frac{\mu}{2}\|x_{t}-x^{*}\|^{2}. \tag{3}
\]

\item[Step 4.] \textbf{Insert variance bound (Lemma~\ref{lem:var}).}  
Using (3) and Lemma~\ref{lem:var} in (2) gives
\[
\begin{aligned}
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\mid x_{t}\bigr]
&\le \|x_{t}-x^{*}\|^{2}
-2\eta\!\left[f(x_{t})-f(x^{*})+\frac{\mu}{2}\|x_{t}-x^{*}\|^{2}\right] \\
&\quad +\eta^{2}\!\bigl[4L(f(x_{t})-f(x^{*})) +4L(f(\tilde{x})-f(x^{*}))\bigr].
\end{aligned} \tag{4}
\]

\item[Step 5.] \textbf{Collect like terms.}  
Rearrange (4):
\[
\begin{aligned}
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\mid x_{t}\bigr]
&\le \bigl(1-\eta\mu\bigr)\|x_{t}-x^{*}\|^{2}
+\bigl(-2\eta+4\eta^{2}L\bigr)\bigl[f(x_{t})-f(x^{*})\bigr] \\
&\quad +4\eta^{2}L\bigl[f(\tilde{x})-f(x^{*})\bigr].
\end{aligned} \tag{5}
\]

\item[Step 6.] \textbf{Choose stepsize to make the coefficient of $[f(x_{t})-f(x^{*})]$ non‑positive.}  
Condition (A1) guarantees $0<\eta\le\frac{1}{3L}$, thus
\[
-2\eta+4\eta^{2}L\le -2\eta+4\eta\cdot\frac{1}{3}= -\frac{2}{3}\eta\le0.
\]
Hence we can drop the non‑positive term:
\[
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\mid x_{t}\bigr]
\le (1-\eta\mu)\|x_{t}-x^{*}\|^{2}
+4\eta^{2}L\bigl[f(\tilde{x})-f(x^{*})\bigr]. \tag{6}
\]

\item[Step 7.] \textbf{Unroll over the inner loop.}  
Iterating (6) for $t=0,\dots,m-1$ and taking full expectation yields
\[
\E\!\bigl[\|x_{m}-x^{*}\|^{2}\bigr]
\le (1-\eta\mu)^{m}\,\E\!\bigl[\|\tilde{x}-x^{*}\|^{2}\bigr]
+4\eta^{2}L\sum_{t=0}^{m-1}(1-\eta\mu)^{t}\,\E\!\bigl[f(\tilde{x})-f(x^{*})\bigr].
\tag{7}
\]

\item[Step 8.] \textbf{Sum the geometric series.}  
Since $\sum_{t=0}^{m-1}(1-\eta\mu)^{t}= \frac{1-(1-\eta\mu)^{m}}{\eta\mu}$, (7) becomes
\[
\E\!\bigl[\|x_{m}-x^{*}\|^{2}\bigr]
\le (1-\eta\mu)^{m}\,\E\!\bigl[\|\tilde{x}-x^{*}\|^{2}\bigr]
+\frac{4\eta L}{\mu}\bigl[1-(1-\eta\mu)^{m}\bigr]\E\!\bigl[f(\tilde{x})-f(x^{*})\bigr]. \tag{8}
\]

\item[Step 9.] \textbf{Relate function suboptimality to distance.}  
Strong convexity gives $f(\tilde{x})-f(x^{*})\le \frac{L}{2}\|\tilde{x}-x^{*}\|^{2}$. Substituting into (8):
\[
\E\!\bigl[\|x_{m}-x^{*}\|^{2}\bigr]
\le \Bigl[(1-\eta\mu)^{m}+ \frac{2\eta L^{2}}{\mu}\bigl[1-(1-\eta\mu)^{m}\bigr]\Bigr]\E\!\bigl[\|\tilde{x}-x^{*}\|^{2}\bigr]. \tag{9}
\]

\item[Step 10.] \textbf{Define the contraction factor.}  
Let
\[
\rho:= (1-\eta\mu)^{m}+ \frac{2\eta L^{2}}{\mu}\bigl[1-(1-\eta\mu)^{m}\bigr]
= (1-\eta\mu)^{m}+ \frac{\eta L^{2}}{\mu(1-\eta L)}\bigl[1-(1-\eta\mu)^{m}\bigr]}.
\]
Because $\eta\le 1/(3L)<1/L$, the denominator $1-\eta L$ is positive and the second expression coincides with the one stated in Theorem~\ref{thm:linear}.  
Thus
\[
\E\!\bigl[\|\tilde{x}^{\,s+1}-x^{*}\|^{2}\bigr]
=\E\!\bigl[\|x_{m}^{\,s}-x^{*}\|^{2}\bigr]\le\rho\,\E\!\bigl[\|\tilde{x}^{\,s}-x^{*}\|^{2}\bigr].
\] 
This completes the proof of the linear recursion.

\item[Step 11.] \textbf{Derive the function‑value rate.}  
Applying strong convexity again,
\[
f(\tilde{x}^{\,s})-f(x^{*})\le \frac{L}{2}\|\tilde{x}^{\,s}-x^{*}\|^{2}.
\]
Iterating the contraction $\|\tilde{x}^{\,s}-x^{*}\|^{2}\le\rho^{s}\|\tilde{x}^{\,0}-x^{*}\|^{2}$ yields
\[
\E\bigl[f(\tilde{x}^{\,s})-f(x^{*})\bigr]\le \frac{L}{2}\rho^{s}\|\tilde{x}^{\,0}-x^{*}\|^{2},
\]
which is the claimed geometric rate. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
For the common choice $\eta=\frac{1}{10L}$ and $m=2n$, one verifies
\[
(1-\eta\mu)^{m}\le e^{-\eta\mu m}=e^{-\frac{2n\mu}{10L}}
\le e^{-0.2\,\kappa^{-1}n},
\]
where $\kappa:=L/\mu$ is the condition number. Moreover,
\[
\frac{\eta L^{2}}{\mu(1-\eta L)}\le \frac{(1/10L)L^{2}}{\mu(1-1/10)} =\frac{1}{9}\frac{L}{\mu}= \frac{1}{9}\kappa .
\]
Choosing $n\ge \kappa$ ensures $\rho\le \tfrac12$. Hence after $S$ epochs,
\[
\E[f(\tilde{x}^{\,S})-f(x^{*})]\le \frac{L}{2}\Bigl(\tfrac12\Bigr)^{S}\|\tilde{x}^{\,0}-x^{*}\|^{2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Quadratic case.} If each $f_i(x)=\frac12 x^{\top}A_i x-b_i^{\top}x$, then $\nabla f_i(x)=A_i x-b_i$. The SVRG estimator becomes an exact affine combination, and the recursion (Step 1) reduces to a linear system whose spectral radius equals $\rho$.
\item \textbf{Infinite‑sum (online) limit.} When $n\to\infty$ and a full gradient is approximated by a large mini‑batch, the same analysis holds with $\tilde{\mu}$ replaced by an unbiased estimator; the contraction factor acquires an additional variance term that vanishes as the mini‑batch size grows.
\end{itemize}
