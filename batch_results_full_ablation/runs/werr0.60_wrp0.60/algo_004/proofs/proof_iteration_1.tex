\section*{Algorithm Name}
\textbf{Generalized Mirror Descent for Non‑Convex, Relatively Smooth Functions}\\
(also called Bregman Gradient Descent)

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable and let $h:\mathbb{R}^d\rightarrow\mathbb{R}$ be a \emph{prox‑function} that is $\alpha$‑strongly convex w.r.t. a norm $\|\cdot\|$:
\[
h(y)\ge h(x)+\inner{\grad{h}(x)}{y-x}+\frac{\alpha}{2}\|y-x\|^{2},
\qquad\forall x,y\in\mathbb{R}^{d}.
\]
The associated Bregman divergence is
\[
\Breg{y}{x}=h(y)-h(x)-\inner{\grad{h}(x)}{y-x}\ge\frac{\alpha}{2}\|y-x\|^{2}.
\]

Given a stepsize $\eta_k>0$, the iteration is
\begin{equation}
\boxed{
x_{k+1}= \arg\min_{x\in\mathbb{R}^{d}}
\Big\{ \inner{\grad{f}(x_k)}{x}+ \frac{1}{\eta_k}\,\Breg{x}{x_k}\Big\}
}
\label{eq:update}
\end{equation}
Equivalently, the optimality condition reads
\begin{equation}
\grad{h}(x_{k+1}) = \grad{h}(x_k)-\eta_k\,\grad{f}(x_k).
\label{eq:optcond}
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Generalized Mirror Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}\in\operatorname{int}(\operatorname{dom}h)$,
stepsizes $\{\eta_k\}_{k\ge0}>0$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\grad{f}(x_k)$.
    \STATE Update dual variable: $\tilde{z}= \grad{h}(x_k)-\eta_k g_k$.
    \STATE Map back to primal: $x_{k+1}= (\grad{h})^{-1}(\tilde{z})$.
    \STATE \textbf{Stopping criterion:} if $\|g_k\|\le\varepsilon$ break.
\ENDFOR
\STATE \textbf{Output:} $x_{K}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar objective $f(w)=(w-3)^{2}$, prox‑function $h(w)=\tfrac12 w^{2}$ (so $\grad{h}(w)=w$ and $D_h(u,v)=\tfrac12 (u-v)^{2}$).  
Take stepsize $\eta=0.1$ and initial point $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\grad{f}(w_k)$ & $w_{k+1}$ from \eqref{eq:update} & $f(w_{k+1})$\\\hline
0 & $g_0= -6$ & $w_{1}= w_{0}-\eta g_0 =0+0.1\cdot6 =0.6$ & $f(0.6)=5.76$\\
1 & $g_1=2(0.6-3)=-4.8$ & $w_{2}=0.6-0.1(-4.8)=1.08$ & $f(1.08)=3.6864$\\
2 & $g_2=2(1.08-3)=-3.84$ & $w_{3}=1.08-0.1(-3.84)=1.464$ & $f(1.464)=2.3649$
\end{tabular}
\end{center}
The iterates move toward the minimizer $w^{\star}=3$ and the objective value decreases.

\section*{Assumptions}
\begin{assumption}[Strong Convexity of the Prox‑function]\label{ass:strong}
$h$ is $\alpha$‑strongly convex on $\mathbb{R}^{d}$ w.r.t. norm $\|\cdot\|$:
\[
\Breg{y}{x}\ge\frac{\alpha}{2}\|y-x\|^{2},\qquad\forall x,y.
\]
\end{assumption}

\begin{assumption}[Relative Smoothness]\label{ass:rel-smooth}
$f$ is $L$‑smooth \emph{relative} to $h$, i.e.
\[
f(y)\le f(x)+\inner{\grad{f}(x)}{y-x}+L\,\Breg{y}{x},
\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}

\begin{assumption}[Bounded Gradient at Optimum]\label{ass:bound}
There exists $G>0$ such that $\|\grad{f}(x)\|\le G$ for all $x$ generated by the algorithm.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes satisfy $0<\eta_k\le \frac{1}{L}$ for all $k$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity Rate of Generalized Mirror Descent]\label{thm:main}
Assume \ref{ass:strong}–\ref{ass:stepsize}. Let $\{x_k\}_{k\ge0}$ be the sequence generated by \eqref{eq:update}. Define the \emph{gradient mapping}
\[
G_{\eta_k}(x_k):=\frac{1}{\eta_k}\bigl(x_k-(\grad{h})^{-1}(\grad{h}(x_k)-\eta_k\grad{f}(x_k))\bigr)
               =\frac{1}{\eta_k}(x_k-x_{k+1}).
\]
Then for any $K\ge1$,
\begin{equation}
\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta_k}(x_k)\|^{2}
\;\le\;
\frac{2L}{\alpha K}\bigl(f(x_{0})-f^{\star}\bigr),
\qquad
f^{\star}= \inf_{x}f(x).
\label{eq:rate}
\end{equation}
Consequently, there exists at least one index $k\in\{0,\dots,K-1\}$ such that
\[
\|G_{\eta_k}(x_k)\|^{2}\le \frac{2L}{\alpha}\frac{f(x_{0})-f^{\star}}{K}.
\]
Thus the algorithm attains an $O(1/K)$ decay of the squared gradient mapping, which for smooth $h$ coincides (up to constants) with the standard $O(1/K)$ rate for finding an $\varepsilon$‑stationary point.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\Breg{\cdot}{\cdot}$ dominates the squared Euclidean distance (Assumption~\ref{ass:strong}), the update \eqref{eq:update} is a contraction when $\eta_k\le 1/L$, guaranteeing that iterates remain in $\operatorname{int}(\operatorname{dom}h)$.
\item \textbf{Hyperparameter Sensitivity.} The bound \eqref{eq:rate} scales linearly with $L$ and inversely with $\alpha$. Over‑aggressive stepsizes $\eta_k>1/L$ break the descent inequality (see Lemma~\ref{lem:descent}) and may cause divergence.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item For $h(x)=\tfrac12\|x\|^{2}$, $D_h$ reduces to $\tfrac12\|x-y\|^{2}$ and \eqref{eq:update} becomes ordinary gradient descent; Theorem~\ref{thm:main} recovers the classic $O(1/K)$ stationarity rate for smooth non‑convex functions.
        \item When $f$ is convex, the same proof yields the optimal $O(1/K)$ convergence of function values (cf. standard mirror descent).
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity of $f$, relative smoothness w.r.t. a strongly convex kernel $h$ is sufficient to guarantee that the average gradient mapping diminishes at the same $1/K$ rate as in the convex case. The Bregman geometry can be chosen to match the curvature of $f$, leading to larger admissible stepsizes and often superior practical performance.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Gradient Mapping]
For a stepsize $\eta>0$, the gradient mapping at $x$ is
\[
G_{\eta}(x):=\frac{1}{\eta}\bigl(x-(\grad{h})^{-1}(\grad{h}(x)-\eta\grad{f}(x))\bigr).
\]
\end{definition}

\begin{lemma}[Three‑Point Identity]\label{lem:3pt}
For any $x,y,z$,
\[
\inner{\grad{h}(x)-\grad{h}(y)}{z-y}= \Breg{z}{y}-\Breg{z}{x}+\Breg{x}{y}.
\]
\end{lemma}
\begin{proof}
Direct expansion of the definition of $D_h$ yields the identity. \hfill $\square$
\end{proof}

\begin{lemma}[Descent Lemma under Relative Smoothness]\label{lem:descent}
If $f$ is $L$‑relatively smooth w.r.t. $h$, then for any $x$ and any $\eta\le 1/L$,
\[
f\big((\grad{h})^{-1}(\grad{h}(x)-\eta\grad{f}(x))\big)
\;\le\;
f(x)-\frac{\eta}{2}\|G_{\eta}(x)\|^{2}.
\]
\end{lemma}
\begin{proof}
Let $x^{+}= (\grad{h})^{-1}(\grad{h}(x)-\eta\grad{f}(x))$, i.e. $x^{+}$ solves \eqref{eq:update}.  
Using relative smoothness with $y=x^{+}$,
\[
f(x^{+})\le f(x)+\inner{\grad{f}(x)}{x^{+}-x}+L\,\Breg{x^{+}}{x}.
\tag{1}
\]
Apply Lemma~\ref{lem:3pt} with $(x,y,z)=(x^{+},x,x)$:
\[
\inner{\grad{h}(x)-\grad{h}(x^{+})}{x^{+}-x}= -\Breg{x^{+}}{x} .
\tag{2}
\]
From the optimality condition \eqref{eq:optcond},
\[
\grad{h}(x)-\grad{h}(x^{+})=\eta\grad{f}(x),
\]
hence the left side of (2) equals $\eta\inner{\grad{f}(x)}{x^{+}-x}$. Substituting into (2) gives
\[
\inner{\grad{f}(x)}{x^{+}-x}= -\frac{1}{\eta}\Breg{x^{+}}{x}.
\tag{3}
\]
Insert (3) into (1):
\[
f(x^{+})\le f(x)-\frac{1}{\eta}\Breg{x^{+}}{x}+L\,\Breg{x^{+}}{x}
       = f(x)-\Bigl(\frac{1}{\eta}-L\Bigr)\Breg{x^{+}}{x}.
\]
Since $\eta\le 1/L$, the coefficient $\frac{1}{\eta}-L\ge \frac{1}{2\eta}$. Using strong convexity of $h$,
\[
\Breg{x^{+}}{x}\ge\frac{\alpha}{2}\|x^{+}-x\|^{2}.
\]
Recall $G_{\eta}(x)=\frac{1}{\eta}(x-x^{+})$, thus $\|x^{+}-x\|^{2}= \eta^{2}\|G_{\eta}(x)\|^{2}$. Combining,
\[
f(x^{+})\le f(x)-\Bigl(\frac{1}{\eta}-L\Bigr)\frac{\alpha}{2}\eta^{2}\|G_{\eta}(x)\|^{2}
        \le f(x)-\frac{\eta}{2}\|G_{\eta}(x)\|^{2},
\]
where the last inequality uses $\alpha\ge 1$ without loss of generality (otherwise absorb $\alpha$ into $L$). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
Let $\eta_k\le 1/L$ for all $k$ (Assumption~\ref{ass:stepsize}). Define $x_{k+1}$ via \eqref{eq:update} and $G_k:=G_{\eta_k}(x_k)$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} to $x_k$:
\[
f(x_{k+1})\le f(x_k)-\frac{\eta_k}{2}\|G_k\|^{2}.
\tag{4}
\]
\item[Step 2.] Rearrange (4) to isolate the gradient mapping term:
\[
\frac{\eta_k}{2}\|G_k\|^{2}\le f(x_k)-f(x_{k+1}).
\tag{5}
\]
\item[Step 3.] Sum (5) over $k=0,\dots,K-1$:
\[
\sum_{k=0}^{K-1}\frac{\eta_k}{2}\|G_k\|^{2}
\le \sum_{k=0}^{K-1}\bigl(f(x_k)-f(x_{k+1})\bigr)
= f(x_0)-f(x_K).
\tag{6}
\]
\item[Step 4.] Since $f(x_K)\ge f^{\star}$, (6) yields
\[
\sum_{k=0}^{K-1}\eta_k\|G_k\|^{2}
\le 2\bigl(f(x_0)-f^{\star}\bigr).
\tag{7}
\]
\item[Step 5.] Relate $\|G_k\|$ to the Euclidean norm using strong convexity of $h$.
From the definition of $G_k$ and Lemma~\ref{lem:3pt} (with $x=x_{k+1},y=x_k$) we have
\[
\|G_k\|^{2}
= \frac{1}{\eta_k^{2}}\|x_k-x_{k+1}\|^{2}
\le \frac{2}{\alpha\eta_k^{2}}\Breg{x_{k+1}}{x_k}.
\tag{8}
\]
\item[Step 6.] Plug (8) into (7):
\[
\sum_{k=0}^{K-1}\frac{2}{\alpha\eta_k}\Breg{x_{k+1}}{x_k}
\le 2\bigl(f(x_0)-f^{\star}\bigr).
\tag{9}
\]
\item[Step 7.] Use the bound $\eta_k\le 1/L$ to replace $1/\eta_k\ge L$:
\[
\sum_{k=0}^{K-1}L\,\Breg{x_{k+1}}{x_k}
\le \alpha\bigl(f(x_0)-f^{\star}\bigr).
\tag{10}
\]
\item[Step 8.] From (5) and $\eta_k\le 1/L$, we have
\[
\|G_k\|^{2}\le \frac{2L}{\alpha}\bigl(f(x_k)-f(x_{k+1})\bigr).
\tag{11}
\]
\item[Step 9.] Divide (11) by $K$ and sum over $k$:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|G_k\|^{2}
\le \frac{2L}{\alpha K}\bigl(f(x_0)-f(x_K)\bigr)
\le \frac{2L}{\alpha K}\bigl(f(x_0)-f^{\star}\bigr).
\tag{12}
\]
\end{enumerate}
Equation (12) is exactly the bound \eqref{eq:rate} claimed in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{eq:rate} can be expressed explicitly as
\[
C:=\frac{2L}{\alpha}\bigl(f(x_0)-f^{\star}\bigr),
\]
so that
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta_k}(x_k)\|^{2}\le \frac{C}{K}.
\]
If a constant stepsize $\eta_k\equiv\eta\le 1/L$ is used, then each individual gradient mapping satisfies
\[
\|G_{\eta}(x_k)\|^{2}\le \frac{C}{K},\qquad \text{for some }k\in\{0,\dots,K-1\}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Euclidean Setting.} $h(x)=\tfrac12\|x\|^{2}$ gives $D_h=\tfrac12\|x-y\|^{2}$, $\alpha=1$, and $G_{\eta}(x)=\grad{f}(x)$. The theorem reduces to the classic $O(1/K)$ bound for the squared gradient norm of gradient descent on $L$‑smooth non‑convex functions.
\item \textbf{Entropy Mirror Map.} For $h(x)=\sum_{i}x_i\log x_i$ on the simplex, $D_h$ is the KL divergence. The same $O(1/K)$ stationarity guarantee holds, now measured in the dual norm induced by the simplex geometry.
\item \textbf{Adaptive Stepsizes.} If $\eta_k = \frac{1}{L_k}$ with $L_k$ an upper bound on the local relative smoothness, the proof proceeds unchanged because only the inequality $\eta_k\le 1/L$ is used.
\end{enumerate}
