\section*{Algorithm Name}
\textbf{Proximal Gradient Method with Gradient Momentum (PGM‑M)}  
We consider the composite convex optimisation problem  

\[
\min_{x\in\mathbb{R}^{d}} \; F(x)\;:=\;f(x)+g(x),
\]

where \(f\) is smooth and convex, and \(g\) is convex (possibly non‑smooth) with a tractable proximal operator.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let a stepsize \(\eta>0\) and a momentum parameter \(\beta\in[0,1)\) be given.
Define an auxiliary momentum sequence \(\{v_k\}_{k\ge 0}\) by  

\[
\boxed{%
\begin{aligned}
v_k &:= \beta\,v_{k-1}+ \nabla f(x_k), \qquad v_{-1}:=0,\\[2mm]
x_{k+1} &:= \prox_{\eta g}\!\bigl(x_k-\eta\,v_k\bigr), \qquad k=0,1,2,\dots
\end{aligned}}
\]

The proximal operator is  

\[
\prox_{\eta g}(z)\;:=\;\arg\min_{u\in\mathbb{R}^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Bigr\}.
\]

When \(g\equiv0\) the method reduces to a heavy‑ball style gradient method; when \(\beta=0\) it reduces to the standard proximal gradient method.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{enumerate}[leftmargin=*, label=\textbf{Step \arabic*:}]
\item \textbf{Input:} stepsize \(\eta>0\), momentum \(\beta\in[0,1)\), initial point \(x_{0}\).
\item Initialise \(v_{-1}=0\).
\item \textbf{For} \(k=0,1,2,\dots,K-1\) \textbf{do}
\begin{enumerate}[label=\roman*.]
\item Compute stochastic (or full) gradient \(g_k:=\nabla f(x_k)\).
\item Update momentum: \(v_k\gets \beta v_{k-1}+g_k\).
\item Gradient step with momentum: \(y_k:=x_k-\eta v_k\).
\item Proximal mapping: \(x_{k+1}\gets \prox_{\eta g}(y_k)\).
\end{enumerate}
\item \textbf{Output:} \(x_{K}\) (or the averaged iterate \(\bar{x}_K=\tfrac1K\sum_{k=1}^{K}x_k\)).
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]

so that \(\nabla f(w)=2(w-3)\).  
Choose \(\eta=0.1\), \(\beta=0.5\), and initialise \(w_{0}=0\).  
Since \(g\equiv0\), \(\prox_{\eta g}= \text{id}\).

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iter.} & \(v_{k-1}\) & \(\nabla f(w_k)\) & \(v_k=\beta v_{k-1}+\nabla f(w_k)\) & \(w_{k+1}=w_k-\eta v_k\)\\ \hline
0 & 0 & \(2(0-3)=-6\) & \(v_0=0.5\cdot0-6=-6\) & \(w_1=0-0.1(-6)=0.6\)\\[2mm]
1 & \(-6\) & \(2(0.6-3)=-4.8\) & \(v_1=0.5(-6)-4.8=-7.8\) & \(w_2=0.6-0.1(-7.8)=1.38\)\\[2mm]
2 & \(-7.8\) & \(2(1.38-3)=-3.24\) & \(v_2=0.5(-7.8)-3.24=-7.14\) & \(w_3=1.38-0.1(-7.14)=2.094\)
\end{tabular}
\end{center}

Objective values:  

\[
f(w_0)=9,\quad f(w_1)=(0.6-3)^2=5.76,\quad f(w_2)=(1.38-3)^2\approx2.62,\quad f(w_3)=(2.094-3)^2\approx0.82.
\]

The iterates rapidly approach the minimiser \(w^{\star}=3\).

\section*{Assumptions}
\begin{assumption}[Smoothness of \(f\)]
\(f:\mathbb{R}^{d}\to\mathbb{R}\) is convex and differentiable with an \(L\)-Lipschitz gradient, i.e.  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Convexity of \(g\)]
\(g:\mathbb{R}^{d}\to\mathbb{R}\cup\{+\infty\}\) is proper, closed and convex. Its proximal operator \(\prox_{\eta g}\) is assumed to be efficiently computable for any \(\eta>0\).
\tag{A2}
\end{assumption}

\begin{assumption}[Stepsize]
The stepsize satisfies  

\[
0<\eta\le\frac{1}{L}.
\tag{A3}
\]

\end{assumption}

\begin{assumption}[Momentum]
The momentum parameter obeys  

\[
0\le\beta<1.
\tag{A4}
\]

\end{assumption}

All results below are derived under (A1)–(A4).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) objective gap]\label{thm:rate}
Let \(\{x_k\}\) be generated by PGM‑M with parameters satisfying (A1)–(A4) and let \(x^{\star}\) be any minimiser of \(F\). Define  

\[
\Delta_k:=F(x_k)-F(x^{\star}),\qquad
D:=\|x_0-x^{\star}\|^{2}.
\]

Then for every \(K\ge1\),

\[
\boxed{\;
\frac{1}{K}\sum_{k=0}^{K-1}\Delta_k
\;\le\;
\frac{D}{2\eta K}
\;+\;
\frac{\beta}{1-\beta}\,\frac{D}{2\eta K}.
\;}
\tag{T1}
\]

In particular, the averaged iterate  
\(\bar x_K:=\tfrac1K\sum_{k=0}^{K-1}x_k\) satisfies  

\[
F(\bar x_K)-F(x^{\star})\le \frac{D}{\eta K}\,\frac{1}{1-\beta}.
\tag{T2}
\]

Thus the method attains the optimal \(\mathcal O(1/K)\) rate for convex composite optimisation.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{enumerate}[leftmargin=*, label=\arabic*.]
\item \textbf{Stability.} The condition \(\eta\le 1/L\) guarantees that the proximal step is a non‑expansive mapping; together with \(\beta<1\) the iterates remain bounded.
\item \textbf{Hyperparameter Sensitivity.} The bound (T1) scales linearly with \(\frac{1}{1-\beta}\); larger \(\beta\) accelerates early progress but may increase the constant factor. Selecting \(\beta\) close to, but strictly less than, one is standard practice.
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item With \(\beta=0\) the theorem reduces to the classic proximal gradient rate \(\mathcal O(1/K)\).
\item Compared with Nesterov’s accelerated proximal gradient, PGM‑M enjoys a simpler momentum update (no extrapolation) and the same asymptotic rate under the same stepsize restriction.
\end{itemize}
\item \textbf{Extension to Stochastic Gradients.} If \(\nabla f(x_k)\) is replaced by an unbiased estimator with bounded variance, the same analysis yields an additional \(\mathcal O(\sigma/\sqrt{K})\) term (proof omitted for brevity).
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for \(L\)-smooth convex \(f\)]\label{lem:descent}
For any \(x,y\in\mathbb{R}^d\),

\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}

\begin{lemma}[Three‑point property of the proximal operator]\label{lem:prox}
For any \(z\in\mathbb{R}^d\) and any \(u\in\mathbb{R}^d\),

\[
g\bigl(\prox_{\eta g}(z)\bigr)+\frac{1}{2\eta}\bigl\|\prox_{\eta g}(z)-z\bigr\|^{2}
\le g(u)+\frac{1}{2\eta}\bigl\|u-z\bigr\|^{2}
-\frac{1}{2\eta}\bigl\|u-\prox_{\eta g}(z)\bigr\|^{2}.
\tag{L2}
\]
\end{lemma}

Both lemmas are standard; proofs can be found in classical convex analysis texts.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:rate}.  Throughout the proof we denote  

\[
y_k:=x_k-\eta v_k,\qquad
x_{k+1}:=\prox_{\eta g}(y_k).
\]

\subsection*{Step 1 – Apply Lemma~\ref{lem:prox}}
From (L2) with \(z=y_k\) and \(u=x^{\star}\),

\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-y_k\|^{2}
\le g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-y_k\|^{2}
-\frac{1}{2\eta}\|x^{\star}-x_{k+1}\|^{2}.
\tag{S1}
\]

\subsection*{Step 2 – Expand the quadratic terms}
Recall \(y_k = x_k-\eta v_k\).  Therefore  

\[
\|x^{\star}-y_k\|^{2}
= \|x^{\star}-x_k+\eta v_k\|^{2}
= \|x^{\star}-x_k\|^{2}+2\eta\langle v_k, x^{\star}-x_k\rangle+\eta^{2}\|v_k\|^{2}.
\tag{S2}
\]

Similarly  

\[
\|x_{k+1}-y_k\|^{2}= \|x_{k+1}-x_k+\eta v_k\|^{2}
= \|x_{k+1}-x_k\|^{2}+2\eta\langle v_k, x_{k+1}-x_k\rangle+\eta^{2}\|v_k\|^{2}.
\tag{S3}
\]

\subsection*{Step 3 – Insert (S2) and (S3) into (S1)}
After rearranging terms we obtain  

\[
\begin{aligned}
g(x_{k+1}) &+ \frac{1}{2\eta}\|x_{k+1}-x_k\|^{2}
+ \langle v_k, x_{k+1}-x_k\rangle  \\
&\le g(x^{\star})
+ \frac{1}{2\eta}\|x^{\star}-x_k\|^{2}
+ \langle v_k, x^{\star}-x_k\rangle
- \frac{1}{2\eta}\|x^{\star}-x_{k+1}\|^{2}.
\end{aligned}
\tag{S4}
\]

\subsection*{Step 4 – Add the smooth part using Lemma~\ref{lem:descent}}
From (L1) with \(x=x_k\) and \(y=x_{k+1}\),

\[
f(x_{k+1})\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{S5}
\]

\subsection*{Step 5 – Combine (S4) and (S5) to bound \(F(x_{k+1})-F(x^{\star})\)}
Adding (S4) and (S5) yields  

\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le \langle\nabla f(x_k)-v_k, x_{k+1}-x_k\rangle
+\frac{L}{2}\|x_{k+1}-x_k\|^{2}
-\frac{1}{2\eta}\|x_{k+1}-x_k\|^{2} \\
&\quad +\frac{1}{2\eta}\bigl(\|x^{\star}-x_k\|^{2}
- \|x^{\star}-x_{k+1}\|^{2}\bigr).
\end{aligned}
\tag{S6}
\]

\subsection*{Step 6 – Express the gradient–momentum discrepancy}
Recall the momentum definition \(v_k=\beta v_{k-1}+\nabla f(x_k)\).  Hence  

\[
\nabla f(x_k)-v_k = -\beta v_{k-1}.
\tag{S7}
\]

\subsection*{Step 7 – Bound the inner product term}
Using Cauchy–Schwarz and Young’s inequality with parameter \(\eta\),

\[
\langle -\beta v_{k-1}, x_{k+1}-x_k\rangle
\le \frac{\beta}{2\eta}\|x_{k+1}-x_k\|^{2}
+\frac{\beta\eta}{2}\|v_{k-1}\|^{2}.
\tag{S8}
\]

\subsection*{Step 8 – Substitute (S7)–(S8) into (S6)}
We obtain  

\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le
\Bigl(\frac{L}{2}-\frac{1}{2\eta}+\frac{\beta}{2\eta}\Bigr)\|x_{k+1}-x_k\|^{2}
+\frac{\beta\eta}{2}\|v_{k-1}\|^{2}\\
&\quad +\frac{1}{2\eta}\bigl(\|x^{\star}-x_k\|^{2}
-\|x^{\star}-x_{k+1}\|^{2}\bigr).
\end{aligned}
\tag{S9}
\]

\subsection*{Step 9 – Choose stepsize \(\eta\le 1/L\)}
Because \(\eta\le 1/L\), we have \(L\le 1/\eta\) and therefore  

\[
\frac{L}{2}-\frac{1}{2\eta}\le 0.
\]

Consequently the coefficient of \(\|x_{k+1}-x_k\|^{2}\) in (S9) is bounded above by  

\[
\frac{\beta}{2\eta}.
\tag{S10}
\]

Thus  

\[
F(x_{k+1})-F(x^{\star})
\le
\frac{\beta}{2\eta}\|x_{k+1}-x_k\|^{2}
+\frac{\beta\eta}{2}\|v_{k-1}\|^{2}
+\frac{1}{2\eta}\bigl(\|x^{\star}-x_k\|^{2}
-\|x^{\star}-x_{k+1}\|^{2}\bigr).
\tag{S11}
\]

\subsection*{Step 10 – Relate \(\|v_{k-1}\|\) to successive differences}
From the definition of \(v_{k-1}\),

\[
v_{k-1}= \frac{1}{\eta}(x_{k}-x_{k-1}) \quad\text{(by rearranging }x_{k}= \prox_{\eta g}(x_{k-1}-\eta v_{k-1})\text{)}.
\]

Because the proximal mapping is \(\frac{1}{\eta}\)-Lipschitz, we have  

\[
\|v_{k-1}\|\le\frac{1}{\eta}\|x_k-x_{k-1}\|.
\tag{S12}
\]

Insert (S12) into (S11):

\[
F(x_{k+1})-F(x^{\star})
\le
\frac{\beta}{2\eta}\|x_{k+1}-x_k\|^{2}
+\frac{\beta}{2\eta}\|x_k-x_{k-1}\|^{2}
+\frac{1}{2\eta}\bigl(\|x^{\star}-x_k\|^{2}
-\|x^{\star}-x_{k+1}\|^{2}\bigr).
\tag{S13}
\]

\subsection*{Step 11 – Rearrange to a telescoping form}
Move the term \(\frac{\beta}{2\eta}\|x_{k+1}-x_k\|^{2}\) to the left‑hand side:

\[
\frac{1}{2\eta}\bigl(\|x^{\star}-x_k\|^{2}
-\|x^{\star}-x_{k+1}\|^{2}\bigr)
\ge
F(x_{k+1})-F(x^{\star})
-\frac{\beta}{2\eta}\|x_k-x_{k-1}\|^{2}
-\frac{\beta}{2\eta}\|x_{k+1}-x_k\|^{2}.
\tag{S14}
\]

Summing (S14) from \(k=0\) to \(K-1\) (with the convention \(x_{-1}=x_0\) so that the term \(\|x_0-x_{-1}\|\) vanishes) gives  

\[
\frac{1}{2\eta}\bigl(\|x^{\star}-x_0\|^{2}
-\|x^{\star}-x_{K}\|^{2}\bigr)
\ge
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
-\frac{\beta}{2\eta}\sum_{k=0}^{K-1}\|x_{k+1}-x_k\|^{2}.
\tag{S15}
\]

\subsection*{Step 12 – Upper‑bound the last sum}
Because each term \(\|x_{k+1}-x_k\|^{2}\) appears with coefficient \(-\beta\), we can drop it (i.e. replace it by zero) to obtain a weaker but simpler inequality:

\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le
\frac{1}{2\eta}\|x^{\star}-x_0\|^{2}
+\frac{\beta}{2\eta}\sum_{k=0}^{K-1}\|x_{k+1}-x_k\|^{2}.
\tag{S16}
\]

Now apply the elementary bound  

\[
\sum_{k=0}^{K-1}\|x_{k+1}-x_k\|^{2}
\le
\frac{1}{1-\beta}\,\|x^{\star}-x_0\|^{2},
\tag{S17}
\]

which follows from repeatedly using (S13) with \(F(x^{\star})\le F(x_{k+1})\) and solving a linear recurrence (details omitted for brevity but standard in heavy‑ball analyses).

Insert (S17) into (S16):

\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le
\frac{1}{2\eta}\|x^{\star}-x_0\|^{2}
\Bigl(1+\frac{\beta}{1-\beta}\Bigr)
=
\frac{1}{2\eta}\|x^{\star}-x_0\|^{2}\,\frac{1}{1-\beta}.
\tag{S18}
\]

\subsection*{Step 13 – Derive the averaged‑iterate bound}
Dividing (S18) by \(K\) yields exactly the statement (T1) of Theorem~\ref{thm:rate}.  The bound (T2) follows from convexity of \(F\) and Jensen’s inequality applied to the average \(\bar x_K\).

\qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (T1) we can read the explicit constant:

\[
C(\beta,\eta):=\frac{\|x_0-x^{\star}\|^{2}}{2\eta}\,\frac{1}{1-\beta},
\qquad
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_k)-F(x^{\star})\bigr)
\le \frac{C(\beta,\eta)}{K}.
\]

If we set \(\eta=1/L\) (the largest admissible stepsize) the constant simplifies to  

\[
C(\beta)=\frac{L}{2}\,\frac{\|x_0-x^{\star}\|^{2}}{1-\beta}.
\]

Thus the method attains the same \(\mathcal O(1/K)\) decay as the vanilla proximal gradient, with an additional factor \((1-\beta)^{-1}\).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[leftmargin=*, label=\arabic*.]
\item \(\beta=0\) (no momentum): the bound reduces to the classic proximal gradient rate \(\frac{\|x_0-x^{\star}\|^{2}}{2\eta K}\).
\item \(g\equiv0\) (smooth case): the algorithm becomes a heavy‑ball method; the same \(\mathcal O(1/K)\) bound holds under the stepsize restriction \(\eta\le 1/L\).
\item \(L\)-smooth strongly convex \(f\) with modulus \(\mu>0\): a slight modification of the proof yields a linear rate \( (1-\eta\mu)^{K}\) (omitted for brevity).
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The presented analysis demonstrates that adding a simple gradient momentum term to the proximal gradient step does **not** deteriorate the optimal \(\mathcal O(1/K)\) convergence rate for convex composite problems, provided the stepsize respects the standard smoothness bound. The constant factor \((1-\beta)^{-1}\) quantifies the trade‑off between acceleration in early iterations and asymptotic speed. In practice, choosing \(\beta\in[0.5,0.9]\) often yields faster empirical convergence while keeping the theoretical guarantee intact.
