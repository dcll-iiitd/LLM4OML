\section*{Algorithm Name}
Stochastic Proximal Gradient Method (Prox-SGD) for convex and smooth objectives

\section*{Mathematical Formulation}
Problem: minimize a composite convex objective
\[
F(w) \triangleq f(w) + h(w), \quad w \in \R^d,
\]
where:
\begin{itemize}
\item $f:\R^d \to \R$ is convex and $L$-smooth (i.e., $\nabla f$ is $L$-Lipschitz).
\item $h:\R^d \to \R \cup \{+\infty\}$ is proper, closed, convex (possibly non-smooth) with a tractable proximal operator.
\end{itemize}
At iteration $t$, the algorithm samples a random data point $z_t$ and computes a stochastic gradient $g_t$ satisfying $\E[g_t \mid w_t] = \nabla f(w_t)$.
Given a step size $\eta_t>0$, the update is
\[
w_{t+1} = \prox_{\eta_t h}\!\left(w_t - \eta_t g_t\right),
\]
where the proximal operator is
\[
\prox_{\eta h}(u) \triangleq \arg\min_{w \in \R^d}\left\{h(w) + \frac{1}{2\eta}\|w-u\|^2\right\}.
\]
Hyperparameters: total iterations $T$, step-size schedule $\{\eta_t\}_{t=0}^{T-1}$ (with $\eta_t \le 1/L$ in the theory), and optionally an output averaging rule $\bar w_T \triangleq \frac{1}{T}\sum_{t=1}^T w_t$.

\section*{Pseudocode}
\begin{itemize}
\item Input: initial $w_0 \in \R^d$, number of iterations $T$, step sizes $\{\eta_t\}_{t=0}^{T-1}$.
\item For $t=0,1,\dots,T-1$:
\begin{itemize}
\item Sample data $z_t$ and compute stochastic gradient $g_t = \nabla f(w_t; z_t)$.
\item Update $w_{t+1} = \prox_{\eta_t h}(w_t - \eta_t g_t)$.
\end{itemize}
\item Output: either $w_T$ or the average $\bar w_T = \frac{1}{T}\sum_{t=1}^T w_t$.
\end{itemize}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^2$ (one-dimensional), $h \equiv 0$, $w_0=0$, $\eta=0.1$, and perform $3$ iterations. Since $h\equiv 0$, $\prox_{\eta h}$ is the identity, so the update is $w_{t+1}=w_t-\eta \nabla f(w_t)$ with $\nabla f(w)=2(w-3)$.
\begin{itemize}
\item Iteration $t=0$:
\[
g_0 = \nabla f(0) = 2(0-3) = -6,\quad
w_1 = 0 - 0.1(-6) = 0.6,\quad
f(w_1)=(0.6-3)^2 = 2.4^2 = 5.76.
\]
\item Iteration $t=1$:
\[
g_1 = \nabla f(0.6) = 2(0.6-3) = -4.8,\quad
w_2 = 0.6 - 0.1(-4.8) = 1.08,\quad
f(w_2)=(1.08-3)^2 = 1.92^2 = 3.6864.
\]
\item Iteration $t=2$:
\[
g_2 = \nabla f(1.08) = 2(1.08-3) = -3.84,\quad
w_3 = 1.08 - 0.1(-3.84) = 1.464,\quad
f(w_3)=(1.464-3)^2 = 1.536^2 = 2.359296.
\]
\end{itemize}

\section*{Assumptions}
We analyze under the following standard assumptions:
\begin{itemize}
\item[A1] $f$ is convex and $L$-smooth: for all $x,y$, $\|\nabla f(x)-\nabla f(y)\| \le L\|x-y\|$.
\item[A2] $h$ is proper, closed, convex, and $\prox_{\eta h}$ is well-defined for all $\eta>0$.
\item[A3] Unbiased stochastic gradients with bounded variance: for all $t$,
\[
\E[g_t \mid w_t] = \nabla f(w_t), \qquad \E\big[\|g_t - \nabla f(w_t)\|^2 \mid w_t\big] \le \sigma^2.
\]
\item[A4] Step sizes satisfy $0<\eta_t \le 1/L$.
\item[A5] Existence of minimizers: $w^* \in \arg\min_w F(w)$ exists.
\end{itemize}

\section*{Main Convergence Theorem}
\begin{theorem}
Suppose A1--A5 hold. Let $\{w_t\}_{t=0}^T$ be generated by
$w_{t+1} = \prox_{\eta h}\!\left(w_t - \eta g_t\right)$ with a constant step size $\eta = c/\sqrt{T}$ for some $c \in (0,1/L]$. Define the average iterate $\bar w_T \triangleq \frac{1}{T}\sum_{t=1}^T w_t$. Then, for all $T$ such that $\sqrt{T} > cL$,
\[
\E\big[F(\bar w_T) - F(w^*)\big]
\;\le\; \frac{\|w_0 - w^*\|^2}{2c\sqrt{T}} \;+\; \frac{\sigma^2 c}{2(\sqrt{T} - cL)}.
\]
In particular, $\E\big[F(\bar w_T) - F(w^*)\big] = O\!\left(\frac{1}{\sqrt{T}}\right)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item Stability: With $\eta \le 1/L$, the per-iteration descent inequality holds with a nonpositive coefficient on the step displacement $\|w_{t+1}-w_t\|^2$, ensuring stable updates in expectation.
\item Hyperparameter sensitivity: The bound scales as $\frac{\|w_0-w^*\|^2}{c\sqrt{T}} + \frac{\sigma^2 c}{\sqrt{T}}$. Balancing the two terms suggests an optimal $c$ on the order of $\|w_0-w^*\|/\sigma$ (subject to $c \le 1/L$).
\item Comparison to baselines:
\begin{itemize}
\item If $h\equiv 0$, the method reduces to SGD and recovers the $O(1/\sqrt{T})$ rate for convex smooth objectives.
\item If gradients are deterministic ($\sigma=0$), using $\eta=1/L$ recovers the $O(1/T)$ rate for the averaged objective values (Proximal Gradient Descent regime).
\end{itemize}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
We use the following standard tools.

\begin{lemma}[Optimality of proximal step and three-point inequality]
Let $u \in \R^d$, $\eta>0$, and $w_+ \triangleq \prox_{\eta h}(u)$. Then there exists $s_+ \in \partial h(w_+)$ such that
\[
0 \in s_+ + \frac{1}{\eta}(w_+ - u),
\]
and for all $w \in \R^d$,
\[
\langle s_+ + \tfrac{1}{\eta}(w_+ - u),\, w_+ - w\rangle \le \frac{1}{2\eta}\Big(\|w-u\|^2 - \|w-w_+\|^2 - \|w_+ - u\|^2\Big).
\]
\end{lemma}
\noindent Proof.
By definition of $\prox_{\eta h}$, $w_+$ minimizes $\phi(w) \triangleq h(w) + \frac{1}{2\eta}\|w-u\|^2$. Hence $0 \in \partial \phi(w_+) = \partial h(w_+) + \frac{1}{\eta}(w_+ - u)$, proving the optimality condition. [Step 1]
Pick $s_+ \in \partial h(w_+)$ such that $s_+ + \frac{1}{\eta}(w_+ - u) = 0$. For any $w$, convexity of $h$ implies $h(w) \ge h(w_+) + \langle s_+, w - w_+\rangle$. [Step 2]
Thus
\[
\phi(w) - \phi(w_+) = h(w) - h(w_+) + \frac{1}{2\eta}\Big(\|w-u\|^2 - \|w_+ - u\|^2\Big) \ge \langle s_+, w - w_+\rangle + \frac{1}{2\eta}\Big(\|w-u\|^2 - \|w_+ - u\|^2\Big). \quad \text{[Step 3]}
\]
Since $w_+$ minimizes $\phi$, $\phi(w) - \phi(w_+)\ge 0$, thus
\[
0 \le \langle s_+, w - w_+\rangle + \frac{1}{2\eta}\Big(\|w-u\|^2 - \|w_+ - u\|^2\Big). \quad \text{[Step 4]}
\]
Rearranging and using $\langle a, b\rangle = \frac{1}{2}\big(\|a\|^2 + \|b\|^2 - \|a-b\|^2\big)$ with $a=\frac{1}{\sqrt{\eta}}(w_+-u)$ and $b=\frac{1}{\sqrt{\eta}}(w-w_+)$ yields
\[
\langle s_+ + \tfrac{1}{\eta}(w_+ - u),\, w_+ - w\rangle \le \frac{1}{2\eta}\Big(\|w-u\|^2 - \|w-w_+\|^2 - \|w_+ - u\|^2\Big). \quad \text{[Step 5]}
\]
\hfill $\square$

\begin{lemma}[Descent lemma for $L$-smooth $f$]
If $f$ is $L$-smooth, then for all $x,y \in \R^d$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x\rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{lemma}
\noindent Proof.
By $L$-Lipschitzness of $\nabla f$, for any $x,y$ and $\tau \in [0,1]$,
$\|\nabla f(x+\tau(y-x)) - \nabla f(x)\| \le L\tau \|y-x\|$. [Step 6]
Integrate the derivative of $f$ along the segment from $x$ to $y$:
\[
f(y) - f(x) = \int_{0}^{1} \langle \nabla f(x+\tau(y-x)),\, y-x\rangle \, d\tau. \quad \text{[Step 7]}
\]
Add and subtract $\nabla f(x)$ inside the inner product and bound via Cauchy--Schwarz:
\[
\langle \nabla f(x+\tau(y-x)), y-x\rangle 
= \langle \nabla f(x), y-x\rangle + \langle \nabla f(x+\tau(y-x))-\nabla f(x), y-x\rangle \quad \text{[Step 8]}
\]
\[
\le \langle \nabla f(x), y-x\rangle + \|\nabla f(x+\tau(y-x))-\nabla f(x)\|\,\|y-x\| \quad \text{[Step 9]}
\]
\[
\le \langle \nabla f(x), y-x\rangle + L\tau \|y-x\|^2. \quad \text{[Step 10]}
\]
Integrating $\tau$ from $0$ to $1$ gives
\[
f(y) - f(x) \le \langle \nabla f(x), y-x\rangle + \frac{L}{2}\|y-x\|^2. \quad \text{[Step 11]}
\]
\hfill $\square$

\begin{lemma}[Young's inequality]
For any $a,b \in \R^d$ and any $\alpha>0$, $\langle a,b\rangle \le \frac{1}{2\alpha}\|a\|^2 + \frac{\alpha}{2}\|b\|^2$.
\end{lemma}
\noindent Proof.
Consider $0 \le \|a - \alpha b\|^2 = \|a\|^2 - 2\alpha \langle a,b\rangle + \alpha^2\|b\|^2$. [Step 12]
Rearrange to obtain $2\alpha \langle a,b\rangle \le \|a\|^2 + \alpha^2\|b\|^2$. [Step 13]
Divide by $2\alpha$ to conclude $\langle a,b\rangle \le \frac{1}{2\alpha}\|a\|^2 + \frac{\alpha}{2}\|b\|^2$. [Step 14]
\hfill $\square$

\section*{Main Proof (Step-by-Step Derivation)}
We prove a one-step progress inequality and then sum it to obtain the rate.

Let $d_t \triangleq w_{t+1}-w_t$. By the proximal update
$w_{t+1} = \prox_{\eta_t h}(w_t - \eta_t g_t)$, Lemma 1 with $u = w_t - \eta_t g_t$ yields an $s_{t+1}\in \partial h(w_{t+1})$ such that
\[
\langle g_t + s_{t+1},\, w_{t+1} - w\rangle \le \frac{1}{2\eta_t}\Big(\|w - w_t\|^2 - \|w - w_{t+1}\|^2 - \|d_t\|^2\Big), \quad \forall w. \quad \text{[Step 15]}
\]
Choose $w = w^*$ to get
\[
\langle g_t + s_{t+1},\, w_{t+1} - w^*\rangle \le \frac{1}{2\eta_t}\Big(\|w^* - w_t\|^2 - \|w^* - w_{t+1}\|^2 - \|d_t\|^2\Big). \quad \text{[Step 16]}
\]
By Lemma 2 (descent lemma) and convexity of $f$, we have
\[
f(w_{t+1}) - f(w^*) \le \langle \nabla f(w_t),\, w_{t+1} - w^*\rangle + \frac{L}{2}\|d_t\|^2. \quad \text{[Step 17]}
\]
Also, by convexity of $h$ and $s_{t+1} \in \partial h(w_{t+1})$,
\[
h(w_{t+1}) - h(w^*) \le \langle s_{t+1},\, w_{t+1} - w^*\rangle. \quad \text{[Step 18]}
\]
Summing [Step 17] and [Step 18] gives
\[
F(w_{t+1}) - F(w^*) \le \langle \nabla f(w_t) + s_{t+1},\, w_{t+1} - w^*\rangle + \frac{L}{2}\|d_t\|^2. \quad \text{[Step 19]}
\]
Insert and subtract $g_t$ inside the inner product:
\[
\langle \nabla f(w_t) + s_{t+1}, w_{t+1} - w^*\rangle 
= \langle g_t + s_{t+1}, w_{t+1} - w^*\rangle + \langle \nabla f(w_t) - g_t, w_{t+1} - w^*\rangle. \quad \text{[Step 20]}
\]
Apply [Step 16] in [Step 19] and use [Step 20]:
\begin{align*}
F(w_{t+1}) - F(w^*) 
&\le \frac{1}{2\eta_t}\Big(\|w^* - w_t\|^2 - \|w^* - w_{t+1}\|^2 - \|d_t\|^2\Big) + \frac{L}{2}\|d_t\|^2 \\
&\qquad + \langle \nabla f(w_t) - g_t,\, w_{t+1} - w^*\rangle. \quad \text{[Step 21]}
\end{align*}
Rearrange the $\|d_t\|^2$ terms:
\[
\frac{L}{2}\|d_t\|^2 - \frac{1}{2\eta_t}\|d_t\|^2 = \left(\frac{L}{2} - \frac{1}{2\eta_t}\right)\|d_t\|^2. \quad \text{[Step 22]}
\]
Decompose $w_{t+1} - w^* = d_t + (w_t - w^*)$ in the noise inner product:
\[
\langle \nabla f(w_t) - g_t,\, w_{t+1} - w^*\rangle 
= \langle \nabla f(w_t) - g_t,\, d_t\rangle + \langle \nabla f(w_t) - g_t,\, w_t - w^*\rangle. \quad \text{[Step 23]}
\]
Combine [Step 21]--[Step 23]:
\begin{align*}
F(w_{t+1}) - F(w^*)
&\le \frac{1}{2\eta_t}\Big(\|w^* - w_t\|^2 - \|w^* - w_{t+1}\|^2\Big) + \left(\frac{L}{2} - \frac{1}{2\eta_t}\right)\|d_t\|^2 \\
&\qquad + \langle \nabla f(w_t) - g_t,\, d_t\rangle + \langle \nabla f(w_t) - g_t,\, w_t - w^*\rangle. \quad \text{[Step 24]}
\end{align*}
Take conditional expectation $\E_t[\cdot] \triangleq \E[\cdot \mid w_t]$ and use $\E_t[g_t] = \nabla f(w_t)$ so that $\E_t[\langle \nabla f(w_t) - g_t,\, w_t - w^*\rangle]=0$. [Step 25]
Apply Young's inequality (Lemma 3) with $\alpha_t \triangleq \frac{1}{\eta_t} - L \ge 0$ (by A4) to the remaining inner product term:
\[
\langle \nabla f(w_t) - g_t,\, d_t\rangle \le \frac{1}{2\alpha_t}\|\nabla f(w_t) - g_t\|^2 + \frac{\alpha_t}{2}\|d_t\|^2. \quad \text{[Step 26]}
\]
Plug [Step 26] into the conditional expectation of [Step 24] and use A3:
\begin{align*}
\E_t\!\left[F(w_{t+1}) - F(w^*)\right]
&\le \frac{1}{2\eta_t}\Big(\|w^* - w_t\|^2 - \E_t\|w^* - w_{t+1}\|^2\Big) \\
&\quad + \left(\frac{L}{2} - \frac{1}{2\eta_t} + \frac{\alpha_t}{2}\right)\E_t\|d_t\|^2 + \frac{1}{2\alpha_t}\E_t\|\nabla f(w_t) - g_t\|^2 \\
&\le \frac{1}{2\eta_t}\Big(\|w^* - w_t\|^2 - \E_t\|w^* - w_{t+1}\|^2\Big) + \frac{\sigma^2}{2\alpha_t}, \quad \text{[Step 27]}
\end{align*}
where we used $\frac{L}{2} - \frac{1}{2\eta_t} + \frac{\alpha_t}{2} = 0$ by the definition of $\alpha_t$, making the $\E_t\|d_t\|^2$ term vanish. [Step 28]

Now take full expectation and sum from $t=0$ to $T-1$:
\begin{align*}
\sum_{t=0}^{T-1} \E\!\left[F(w_{t+1}) - F(w^*)\right]
&\le \sum_{t=0}^{T-1} \frac{1}{2\eta_t}\Big(\E\|w^* - w_t\|^2 - \E\|w^* - w_{t+1}\|^2\Big) + \sum_{t=0}^{T-1} \frac{\sigma^2}{2\alpha_t}. \quad \text{[Step 29]}
\end{align*}
For a constant step size $\eta_t \equiv \eta$, the first sum telescopes:
\[
\sum_{t=0}^{T-1} \frac{1}{2\eta}\Big(\E\|w^* - w_t\|^2 - \E\|w^* - w_{t+1}\|^2\Big) = \frac{1}{2\eta}\Big(\E\|w^* - w_0\|^2 - \E\|w^* - w_T\|^2\Big) \le \frac{\|w_0 - w^*\|^2}{2\eta}. \quad \text{[Step 30]}
\]
Thus, with $\alpha \triangleq \frac{1}{\eta} - L$ and $\eta \le 1/L$,
\[
\sum_{t=0}^{T-1} \E\!\left[F(w_{t+1}) - F(w^*)\right] \le \frac{\|w_0 - w^*\|^2}{2\eta} + \frac{T\sigma^2}{2\alpha}. \quad \text{[Step 31]}
\]
By convexity of $F$,
\[
F(\bar w_T) \le \frac{1}{T}\sum_{t=1}^T F(w_t) \quad \Rightarrow \quad \E\!\left[F(\bar w_T) - F(w^*)\right] \le \frac{1}{T}\sum_{t=0}^{T-1} \E\!\left[F(w_{t+1}) - F(w^*)\right]. \quad \text{[Step 32]}
\]
Combining [Step 31] and [Step 32] gives, for constant $\eta$,
\[
\E\!\left[F(\bar w_T) - F(w^*)\right] \le \frac{\|w_0 - w^*\|^2}{2\eta T} + \frac{\sigma^2}{2\alpha}. \quad \text{[Step 33]}
\]

\section*{Rate Derivation (With Constants)}
Pick $\eta = \frac{c}{\sqrt{T}}$ with $0 < c \le 1/L$. Then $\alpha = \frac{1}{\eta} - L = \frac{\sqrt{T}}{c} - L$, which is positive when $\sqrt{T} > cL$. [Step 34]
Substitute $\eta$ and $\alpha$ into [Step 33]:
\[
\E\!\left[F(\bar w_T) - F(w^*)\right] \le \frac{\|w_0 - w^*\|^2}{2(c/\sqrt{T})T} + \frac{\sigma^2}{2(\sqrt{T}/c - L)}
= \frac{\|w_0 - w^*\|^2}{2c\sqrt{T}} + \frac{\sigma^2 c}{2(\sqrt{T} - cL)}. \quad \text{[Step 35]}
\]
This proves the theorem and exhibits the explicit $O(1/\sqrt{T})$ rate with constants.

\section*{Special Cases}
\begin{itemize}
\item No regularization ($h\equiv 0$): Then $\prox_{\eta h}$ is the identity, and the method reduces to standard SGD. The proof above remains valid and yields the same $O(1/\sqrt{T})$ convergence for the averaged iterate.
\item Deterministic gradients ($\sigma=0$): Take $g_t \equiv \nabla f(w_t)$ and $\eta = 1/L$. In [Step 27], the noise term vanishes exactly, and with $\eta=1/L$ the $\|d_t\|^2$ coefficient is zero. Summing [Step 29]--[Step 30] gives
\[
\sum_{t=0}^{T-1} \big(F(w_{t+1}) - F(w^*)\big) \le \frac{L}{2}\|w_0 - w^*\|^2,
\]
hence
\[
F(\bar w_T) - F(w^*) \le \frac{L}{2T}\|w_0 - w^*\|^2,
\]
which is the standard $O(1/T)$ rate for proximal gradient descent on smooth convex objectives.
\end{itemize}

\section*{Discussion}
The analysis shows that applying a proximal step after a stochastic gradient update preserves the $O(1/\sqrt{T})$ convergence rate for convex smooth objectives under standard unbiasedness and bounded-variance assumptions. The requirement $\eta \le 1/L$ ensures that the potentially harmful displacement term $\|w_{t+1}-w_t\|^2$ is neutralized in expectation. The explicit constants reveal a trade-off in the step-size scale $c$: larger $c$ accelerates the deterministic contraction term but amplifies the stochastic variance term, suggesting a balance guided by the problem-dependent ratio $\|w_0-w^*\|/\sigma$, subject to the stability constraint $c \le 1/L$.
