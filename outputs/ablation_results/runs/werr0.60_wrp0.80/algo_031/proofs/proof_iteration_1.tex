\section*{Algorithm Name}
\textbf{Frank–Wolfe (Conditional Gradient) Method for Convex Smooth Minimization}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$-smooth function and let $C\subset\R^{d}$ be a non‑empty compact convex set (the feasible domain).  
The goal is
\[
\min_{x\in C} f(x).
\]

At iteration $k\ge 0$ the algorithm computes the \emph{linear minimization oracle} (LMO)
\[
s_{k}\;=\;\arg\min_{s\in C}\;\langle \nabla f(x_{k}),\,s\rangle .
\tag{1}
\]
Given a stepsize $\gamma_{k}\in[0,1]$, the next iterate is a convex combination of the current point and the LMO solution:
\[
x_{k+1}\;=\;(1-\gamma_{k})\,x_{k}\;+\;\gamma_{k}\,s_{k}.
\tag{2}
\]

Typical choices for $\gamma_{k}$:
\begin{itemize}
\item \emph{Exact line search}: $\displaystyle\gamma_{k}\in\arg\min_{\gamma\in[0,1]} f\big((1-\gamma)x_{k}+\gamma s_{k}\big)$,
\item \emph{Closed‑form schedule}: $\displaystyle\gamma_{k}= \frac{2}{k+2}$ (the classical FW schedule).
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe}
\begin{algorithmic}[1]
\Require Convex smooth $f$, feasible set $C$, initial $x_{0}\in C$, stepsize rule $\{\gamma_{k}\}$, max iterations $K$
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$
    \State \textbf{LMO:} $s_{k}\gets\arg\min_{s\in C}\langle g_{k},s\rangle$
    \State Choose stepsize $\gamma_{k}\in[0,1]$ (e.g., $\gamma_{k}=2/(k+2)$)
    \State Update $x_{k+1}\gets (1-\gamma_{k})x_{k}+\gamma_{k}s_{k}$
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\textbf{Problem data}
\[
f(w)=(w-3)^{2},\qquad C=[0,5],\qquad w_{0}=0,\qquad\gamma_{k}=0.1\;\;(\text{fixed}).
\]

\begin{enumerate}[label=Iteration \arabic*:,leftmargin=*]
\item $k=0$  
\[
\nabla f(w_{0})=2(w_{0}-3)=-6.
\]
LMO:
\[
s_{0}=\arg\min_{s\in[0,5]}\{-6s\}=5\quad(\text{since }-6s\text{ is decreasing}).
\]
Update:
\[
w_{1}= (1-0.1)w_{0}+0.1\,s_{0}=0.5.
\]
Objective:
\[
f(w_{1})=(0.5-3)^{2}=6.25.
\]

\item $k=1$  
\[
\nabla f(w_{1})=2(w_{1}-3)=2(0.5-3)=-5.
\]
LMO:
\[
s_{1}=5\;(\text{again the minimizer of }-5s\text{ on }[0,5]).
\]
Update:
\[
w_{2}=0.9\cdot0.5+0.1\cdot5=0.95.
\]
Objective:
\[
f(w_{2})=(0.95-3)^{2}=4.2025.
\]

\item $k=2$  
\[
\nabla f(w_{2})=2(0.95-3)=-4.1.
\]
LMO:
\[
s_{2}=5.
\]
Update:
\[
w_{3}=0.9\cdot0.95+0.1\cdot5=1.355.
\]
Objective:
\[
f(w_{3})=(1.355-3)^{2}=2.702\;(\text{approximately}).
\]
\end{enumerate}
The iterates move towards the optimal point $w^{*}=3$ while remaining in $C$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convexsmooth}
The objective $f:C\to\R$ satisfies:
\begin{enumerate}[label=(\alph*)]
\item $f$ is convex: $\forall x,y\in C,\;\;f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle$.
\item $f$ is $L$‑smooth on $C$: $\forall x,y\in C$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{enumerate}
\end{assumption}

\begin{assumption}[Compact Feasible Set]\label{as:compact}
$C\subset\R^{d}$ is convex, compact and has finite diameter
\[
D:=\max_{x,y\in C}\|x-y\|<\infty .
\tag{4}
\]
\end{assumption}

\begin{assumption}[Curvature Constant]\label{as:curvature}
Define the (global) curvature of $f$ w.r.t.\ $C$ as
\[
C_{f}\;:=\;\sup_{\substack{x,s\in C\\ \gamma\in[0,1]}}
\frac{2}{\gamma^{2}}\Bigl[f\big(x+\gamma(s-x)\big)-f(x)-\gamma\langle\nabla f(x),s-x\rangle\Bigr].
\tag{5}
\]
Under Assumptions~\ref{as:convexsmooth} and~\ref{as:compact}, $C_{f}\le L D^{2}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal Gap $O(1/k)$]\label{thm:fw-rate}
Let $\{x_{k}\}$ be generated by the Frank–Wolfe method with stepsizes
\[
\gamma_{k}= \frac{2}{k+2},\qquad k\ge0.
\]
Under Assumptions~\ref{as:convexsmooth}–\ref{as:curvature}, the primal gap satisfies
\[
f(x_{k})-f(x^{\star})\;\le\;\frac{2C_{f}}{k+2},\qquad\forall k\ge0,
\tag{6}
\]
where $x^{\star}\in\arg\min_{x\in C}f(x)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Since each iterate is a convex combination of two points in $C$, all iterates remain in $C$, guaranteeing boundedness.
\item \textbf{Step‑size sensitivity.} The classical schedule $\gamma_{k}=2/(k+2)$ yields the optimal $O(1/k)$ rate without line search. Using exact line search can improve the constant but not the asymptotic order.
\item \textbf{Comparison to projected gradient.} Frank–Wolfe avoids costly Euclidean projections; its rate matches that of projected gradient for smooth convex problems but with cheaper per‑iteration cost when the LMO is cheap.
\item \textbf{Special case $C$ a polytope.} The curvature $C_{f}$ can be bounded by $L\,\diam(C)^{2}$, making (6) explicit in $L$ and the geometry of $C$.
\end{itemize}

\section*{Discussion}
The theorem shows that the Frank–Wolfe method enjoys a sublinear convergence rate identical to the optimal rate for first‑order methods on smooth convex problems, while maintaining iterates in the feasible set without projection. The proof hinges on the curvature constant (5) which captures how far $f$ deviates from its linear approximation over $C$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth on $C$, then for any $x,y\in C$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness; see (3). \qedhere
\end{proof}

\begin{lemma}[Curvature Upper Bound]\label{lem:curvbound}
Under Assumptions~\ref{as:convexsmooth} and~\ref{as:compact},
\[
C_{f}\;\le\;L D^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
From (7) with $y=x+\gamma(s-x)$ and $\|s-x\|\le D$, we obtain
\[
f(x+\gamma(s-x))\le f(x)+\gamma\langle\nabla f(x),s-x\rangle+\frac{L}{2}\gamma^{2}D^{2}.
\]
Rearranging yields the bound (8). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw-rate}. Let $x^{\star}\in\arg\min_{x\in C}f(x)$. Throughout the proof we denote the primal gap $h_{k}:=f(x_{k})-f(x^{\star})$.

\begin{enumerate}[label=[Step \arabic*],leftmargin=*]
\item \textbf{Linear approximation error.} By convexity,
\[
f(x^{\star})\ge f(x_{k})+\langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle.
\tag{9}
\]
Rearranging gives
\[
\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\ge h_{k}.
\tag{10}
\]

\item \textbf{Definition of the FW gap.} The LMO defines the \emph{Frank–Wolfe gap}
\[
\gap_{k}:=\langle\nabla f(x_{k}),x_{k}-s_{k}\rangle
          =\max_{s\in C}\langle\nabla f(x_{k}),x_{k}-s\rangle .
\tag{11}
\]
Because $x^{\star}\in C$, we have $\gap_{k}\ge\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\ge h_{k}$ by (10). Hence
\[
\gap_{k}\ge h_{k}.
\tag{12}
\]

\item \textbf{One‑step progress via curvature.} From the definition of $C_{f}$ (5) with $x=x_{k}$, $s=s_{k}$ and $\gamma=\gamma_{k}$,
\[
f\big(x_{k}+\gamma_{k}(s_{k}-x_{k})\big)
\le f(x_{k})+\gamma_{k}\langle\nabla f(x_{k}),s_{k}-x_{k}\rangle
     +\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{13}
\]
Since $x_{k+1}=(1-\gamma_{k})x_{k}+\gamma_{k}s_{k}=x_{k}+\gamma_{k}(s_{k}-x_{k})$, (13) becomes
\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}\gap_{k}+\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{14}
\]

\item \textbf{Relating gap to primal gap.} Using (12) we replace $\gap_{k}$ by $h_{k}$ in (14):
\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}h_{k}+\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{15}
\]

\item \textbf{Recursion for the primal gap.} Subtract $f(x^{\star})$ from both sides of (15) and recall $h_{k}=f(x_{k})-f(x^{\star})$:
\[
h_{k+1}\le (1-\gamma_{k})h_{k}+\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{16}
\]

\item \textbf{Choice of stepsize.} Insert the classical schedule $\gamma_{k}=2/(k+2)$.
Then $1-\gamma_{k}= \frac{k}{k+2}$ and $\gamma_{k}^{2}= \frac{4}{(k+2)^{2}}$.
Equation (16) yields
\[
h_{k+1}\le \frac{k}{k+2}h_{k}+\frac{2C_{f}}{(k+2)^{2}}.
\tag{17}
\]

\item \textbf{Induction hypothesis.} We prove by induction that
\[
h_{k}\le \frac{2C_{f}}{k+2},\qquad\forall k\ge0.
\tag{18}
\]

\item \textbf{Base case $k=0$.} From convexity $h_{0}=f(x_{0})-f(x^{\star})\le C_{f}$ (by definition of curvature with $\gamma=1$). Hence
\[
h_{0}\le C_{f}\le \frac{2C_{f}}{2},
\]
so (18) holds for $k=0$.

\item \textbf{Inductive step.} Assume (18) holds for some $k\ge0$. Plug the bound into (17):
\[
h_{k+1}\le \frac{k}{k+2}\cdot\frac{2C_{f}}{k+2}
          +\frac{2C_{f}}{(k+2)^{2}}
        =\frac{2C_{f}}{(k+2)^{2}}\Bigl(k+1\Bigr)
        =\frac{2C_{f}}{(k+1)+2}.
\tag{19}
\]
Thus (18) holds for $k+1$.

\item \textbf{Conclusion.} By induction, (18) is true for all $k$, which is exactly the statement (6) of Theorem~\ref{thm:fw-rate}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Lemma~\ref{lem:curvbound} we have $C_{f}\le L D^{2}$. Substituting into (6) gives an explicit rate:
\[
f(x_{k})-f(x^{\star})\;\le\;\frac{2L D^{2}}{k+2}.
\tag{20}
\]
Hence the convergence constant depends linearly on the smoothness constant $L$ and quadratically on the diameter of $C$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact line search.} If $\gamma_{k}$ is chosen by exact line search, inequality (14) holds with the optimal $\gamma_{k}$, leading to
\[
h_{k+1}\le h_{k}-\frac{\gap_{k}^{2}}{2C_{f}}
\le h_{k}-\frac{h_{k}^{2}}{2C_{f}},
\]
which gives a \emph{linear} convergence bound for strongly convex $f$.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, the curvature can be related to $\mu$, and one obtains
\[
h_{k}\le \Bigl(1-\frac{\mu}{L}\Bigr)^{k}h_{0},
\]
i.e., a geometric rate.
\item \textbf{Polyhedral $C$.} When $C$ is a polytope, the LMO can be solved by a simple vertex selection, and the algorithm reduces to a sparse representation of the solution (the iterates are convex combinations of at most $k+1$ vertices).
\end{enumerate}
