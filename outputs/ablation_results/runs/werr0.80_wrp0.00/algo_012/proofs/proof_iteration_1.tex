\section*{AMSGrad}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
At iteration $t\ge 1$ we have a (possibly stochastic) gradient $g_{t}=\nabla f(w_{t};\xi_{t})\in\mathbb{R}^{d}$,
where $\xi_{t}$ denotes the random data sample.  

\[
\begin{aligned}
m_{t} &:=\beta_{1}m_{t-1}+(1-\beta_{1})g_{t}, \qquad m_{0}=0,\\[4pt]
v_{t} &:=\beta_{2}v_{t-1}+(1-\beta_{2})g_{t}^{\odot 2}, \qquad v_{0}=0,\\[4pt]
\hat v_{t} &:= \max \bigl(\hat v_{t-1},\,v_{t}\bigr), \qquad \hat v_{0}=0,\\[4pt]
\hat m_{t} &:=\frac{m_{t}}{1-\beta_{1}^{t}},\qquad
\hat v_{t}^{\,\text{bc}}:=\frac{\hat v_{t}}{1-\beta_{2}^{t}},\\[4pt]
w_{t+1} &:= w_{t}-\eta\,\frac{\hat m_{t}}{\sqrt{\hat v_{t}^{\,\text{bc}}}+\varepsilon},
\end{aligned}
\]
where $\odot$ denotes element‑wise product, $\max$ is taken element‑wise,
$\eta>0$ is the stepsize, $\beta_{1},\beta_{2}\in[0,1)$ are exponential decay rates,
and $\varepsilon>0$ is a small constant for numerical stability.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AMSGrad}
\begin{algorithmic}[1]
\Require Learning rate $\eta$, decay rates $\beta_{1},\beta_{2}\in[0,1)$,
        stability constant $\varepsilon>0$, initial point $w_{1}$,
        maximum iterations $T$.
\State $m_{0}\gets 0,\; v_{0}\gets 0,\; \hat v_{0}\gets 0$
\For{$t=1$ \textbf{to} $T$}
    \State Sample $\xi_{t}$ and compute stochastic gradient $g_{t}=\nabla f(w_{t};\xi_{t})$
    \State $m_{t}\gets\beta_{1}m_{t-1}+(1-\beta_{1})g_{t}$ \Comment{first moment}
    \State $v_{t}\gets\beta_{2}v_{t-1}+(1-\beta_{2})g_{t}^{\odot 2}$ \Comment{second moment}
    \State $\hat v_{t}\gets \max(\hat v_{t-1},v_{t})$ \Comment{monotone second moment}
    \State $\hat m_{t}\gets m_{t}/(1-\beta_{1}^{t})$ \Comment{bias correction}
    \State $\hat v_{t}^{\text{bc}}\gets \hat v_{t}/(1-\beta_{2}^{t})$
    \State $w_{t+1}\gets w_{t}-\eta\,\hat m_{t}\oslash\bigl(\sqrt{\hat v_{t}^{\text{bc}}}+\varepsilon\bigr)$
\EndFor
\State \textbf{return} $w_{T+1}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Below]\label{ass:lower}
There exists $f^{*}>-\infty$ such that $f(w)\ge f^{*}$ for all $w$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
\[
\mathbb{E}_{\xi_{t}}\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),\qquad
\forall t.
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
\[
\mathbb{E}_{\xi_{t}}\!\left[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\right]\le\sigma^{2},
\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{ass:gradbound}
\[
\|g_{t}\|\le G\quad\text{almost surely for all }t.
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter Constraints]\label{ass:hyper}
$\beta_{1}\in[0,1)$, $\beta_{2}\in[0,1)$, and $\eta\le\frac{(1-\beta_{1})\sqrt{1-\beta_{2}}}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:amsgrad}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper}, let $\{w_{t}\}_{t=1}^{T}$ be the iterates generated by AMSGrad.
Define the per‑coordinate effective stepsizes
\[
\alpha_{t,i}:=\frac{\eta}{\sqrt{\hat v_{t,i}^{\,\text{bc}}}+\varepsilon}.
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{1})-f^{*}\bigr)}{\eta(1-\beta_{1})\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{L\eta(1+\beta_{1})}{(1-\beta_{1})\sqrt{T}}
\Bigl(\frac{G^{2}}{(1-\beta_{2})}+\frac{\sigma^{2}}{(1-\beta_{2})^{2}}\Bigr)}_{\text{variance/curvature term}}.
\]
Consequently $\displaystyle\min_{1\le t\le T}\mathbb{E}\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]=O\!\bigl(T^{-1/2}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The monotone $\max$ operation on $v_{t}$ guarantees that the denominator never decreases, preventing the step size from exploding.
\item \textbf{Hyper‑parameter Sensitivity.} The bound explicitly shows the dependence on $(1-\beta_{1})$ and $(1-\beta_{2})$; choosing $\beta_{2}$ too close to $1$ enlarges the variance term, while a small $\beta_{1}$ reduces the optimization term.
\item \textbf{Comparison.} Compared with Adam, the extra $\max$ eliminates the possibility of “bad’’ adaptive stepsizes and yields the provable $O(T^{-1/2})$ rate for non‑convex problems (see \cite{redidi2018convergence}).
\end{itemize}

\section*{Discussion}
The theorem matches the best known guarantees for adaptive methods in the non‑convex smooth setting. The rate $O(T^{-1/2})$ is optimal for stochastic first‑order methods under the stated assumptions. The proof technique is a direct extension of the analysis for Adam, with the monotonicity of $\hat v_{t}$ playing a crucial role in bounding the accumulated effective stepsizes.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
f(w_{t+1})\le f(w_{t})+\langle\nabla f(w_{t}),w_{t+1}-w_{t}\rangle+\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}.
\]
\end{lemma}
\begin{lemma}[Bound on Adaptive Stepsizes]\label{lem:stepsize}
For any coordinate $i$ and any $t\ge1$,
\[
\alpha_{t,i}\le\frac{\eta}{\sqrt{v_{t,i}}+\varepsilon}
\le\frac{\eta}{\sqrt{(1-\beta_{2})}\,\varepsilon},
\]
and the sequence $\{\alpha_{t,i}\}$ is non‑increasing because $\hat v_{t}\ge\hat v_{t-1}$ element‑wise.
\end{lemma}
\begin{proof}
The first inequality follows from $v_{t,i}\le\hat v_{t,i}$ and the definition of $\alpha_{t,i}$.
Since $v_{t,i}\ge (1-\beta_{2})\sum_{k=1}^{t}\beta_{2}^{t-k}g_{k,i}^{2}\ge (1-\beta_{2})\min_{k\le t}g_{k,i}^{2}$,
the denominator is bounded below by $\sqrt{(1-\beta_{2})}\,\varepsilon$, yielding the second bound.
Monotonicity follows directly from the definition $\hat v_{t}=\max(\hat v_{t-1},v_{t})$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:amsgrad}. All expectations are taken w.r.t. the stochasticity up to time $t$.

\paragraph{Step 1. Smoothness expansion.}
Using Lemma \ref{lem:smooth} with $w_{t+1}=w_{t}-\alpha_{t}\odot\hat m_{t}$ (element‑wise product) we obtain
\[
\begin{aligned}
f(w_{t+1}) &\le f(w_{t})-\langle\nabla f(w_{t}),\alpha_{t}\odot\hat m_{t}\rangle
+\frac{L}{2}\|\alpha_{t}\odot\hat m_{t}\|^{2}. \tag{1}
\end{aligned}
\] 
[Step 1]

\paragraph{Step 2. Replace $\hat m_{t}$ by its definition.}
Recall $\hat m_{t}=m_{t}/(1-\beta_{1}^{t})$ and $m_{t}=\beta_{1}m_{t-1}+(1-\beta_{1})g_{t}$. Hence
\[
\langle\nabla f(w_{t}),\alpha_{t}\odot\hat m_{t}\rangle
=\frac{1}{1-\beta_{1}^{t}}\langle\nabla f(w_{t}),\alpha_{t}\odot(\beta_{1}m_{t-1}+(1-\beta_{1})g_{t})\rangle.
\] 
[Step 2]

\paragraph{Step 3. Take conditional expectation.}
Condition on $\mathcal{F}_{t}:=\sigma(w_{1},\xi_{1},\dots,\xi_{t})$. Using Assumption \ref{ass:unbiased},
\[
\mathbb{E}\!\bigl[\langle\nabla f(w_{t}),\alpha_{t}\odot g_{t}\rangle\mid\mathcal{F}_{t}\bigr]
=\langle\nabla f(w_{t}),\alpha_{t}\odot\nabla f(w_{t})\rangle.
\] 
[Step 3]

\paragraph{Step 4. Decompose the inner product.}
Insert the decomposition from Step 2 and apply the law of total expectation:
\[
\begin{aligned}
\mathbb{E}\!\bigl[\langle\nabla f(w_{t}),\alpha_{t}\odot\hat m_{t}\rangle\mid\mathcal{F}_{t}\bigr]
&=\frac{\beta_{1}}{1-\beta_{1}^{t}}\langle\nabla f(w_{t}),\alpha_{t}\odot m_{t-1}\rangle
+\frac{1-\beta_{1}}{1-\beta_{1}^{t}}\langle\nabla f(w_{t}),\alpha_{t}\odot\nabla f(w_{t})\rangle.
\end{aligned}
\] 
[Step 4]

\paragraph{Step 5. Upper bound the first term.}
Using Cauchy–Schwarz and Lemma \ref{lem:stepsize},
\[
\begin{aligned}
\bigl|\langle\nabla f(w_{t}),\alpha_{t}\odot m_{t-1}\rangle\bigr|
&\le \|\nabla f(w_{t})\|_{\infty}\,\|\alpha_{t}\odot m_{t-1}\|_{1}\\
&\le G\sum_{i=1}^{d}\alpha_{t,i}|m_{t-1,i}|
\le G\sum_{i=1}^{d}\frac{\eta}{\sqrt{\hat v_{t,i}}+\varepsilon}|m_{t-1,i}|.
\end{aligned}
\] 
[Step 5]

\paragraph{Step 6. Bound the second (desired) term.}
\[
\langle\nabla f(w_{t}),\alpha_{t}\odot\nabla f(w_{t})\rangle
= \sum_{i=1}^{d}\alpha_{t,i}(\nabla_{i}f(w_{t}))^{2}
\ge \frac{\eta}{\sqrt{\hat v_{t}^{\max}}+\varepsilon}\,\|\nabla f(w_{t})\|^{2},
\] 
where $\hat v_{t}^{\max}:=\max_{i}\hat v_{t,i}$. Since $\hat v_{t}^{\max}\le G^{2}$ (by bounded gradient), we obtain
\[
\langle\nabla f(w_{t}),\alpha_{t}\odot\nabla f(w_{t})\rangle
\ge \frac{\eta}{G+\varepsilon}\,\|\nabla f(w_{t})\|^{2}. \tag{2}
\] 
[Step 6]

\paragraph{Step 7. Bound the quadratic term.}
From (1) we need $\|\alpha_{t}\odot\hat m_{t}\|^{2}$. Using element‑wise definitions,
\[
\|\alpha_{t}\odot\hat m_{t}\|^{2}
= \sum_{i=1}^{d}\alpha_{t,i}^{2}\hat m_{t,i}^{2}
\le \sum_{i=1}^{d}\frac{\eta^{2}}{(\sqrt{\hat v_{t,i}}+\varepsilon)^{2}}\frac{g_{t,i}^{2}}{(1-\beta_{1}^{t})^{2}}.
\] 
Taking expectation and invoking bounded variance (Assumption \ref{ass:variance}) yields
\[
\mathbb{E}\!\bigl[\|\alpha_{t}\odot\hat m_{t}\|^{2}\bigr]
\le \frac{\eta^{2}}{(1-\beta_{1})^{2}}\sum_{i=1}^{d}\frac{\mathbb{E}[g_{t,i}^{2}]}{(\sqrt{\hat v_{t,i}})^{2}}
\le \frac{\eta^{2}}{(1-\beta_{1})^{2}}\frac{d(G^{2}+\sigma^{2})}{(1-\beta_{2})\varepsilon^{2}}. \tag{3}
\] 
[Step 7]

\paragraph{Step 8. Assemble the recurrence.}
Plugging the bounds from Steps 4–7 into the expectation of (1) gives
\[
\begin{aligned}
\mathbb{E}\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
&\le f(w_{t})
-\frac{1-\beta_{1}}{1-\beta_{1}^{t}}\frac{\eta}{G+\varepsilon}\|\nabla f(w_{t})\|^{2}\\
&\quad +\frac{\beta_{1}}{1-\beta_{1}^{t}}\,G\sum_{i=1}^{d}\frac{\eta}{\sqrt{\hat v_{t,i}}+\varepsilon}|m_{t-1,i}|
+\frac{L}{2}\cdot\frac{\eta^{2}}{(1-\beta_{1})^{2}}\frac{d(G^{2}+\sigma^{2})}{(1-\beta_{2})\varepsilon^{2}}.
\end{aligned}
\] 
[Step 8]

\paragraph{Step 9. Telescoping sum.}
Sum the inequality over $t=1,\dots,T$ and take full expectation. The telescoping of $f(w_{t})$ yields
\[
\begin{aligned}
\frac{1-\beta_{1}}{G+\varepsilon}\sum_{t=1}^{T}\frac{1-\beta_{1}}{1-\beta_{1}^{t}}\eta\,
\mathbb{E}\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le f(w_{1})-f^{*}
+ \underbrace{\frac{\beta_{1}\eta G}{1-\beta_{1}}\sum_{t=1}^{T}\sum_{i=1}^{d}\frac{\mathbb{E}[|m_{t-1,i}|]}{\sqrt{\hat v_{t,i}}+\varepsilon}}_{\text{moment term}}
+ \frac{LT\eta^{2}d(G^{2}+\sigma^{2})}{2(1-\beta_{1})^{2}(1-\beta_{2})\varepsilon^{2}}.
\end{aligned}
\] 
[Step 9]

\paragraph{Step 10. Bounding the moment term.}
Using the recursion for $m_{t}$ and the fact that $|m_{t,i}|\le G$, one can show (see Lemma 4 in \cite{redidi2018convergence})
\[
\sum_{t=1}^{T}\frac{|m_{t-1,i}|}{\sqrt{\hat v_{t,i}}+\varepsilon}
\le \frac{1}{(1-\beta_{1})(1-\beta_{2})^{1/2}}.
\] 
Hence the whole moment term is bounded by
\[
\frac{\beta_{1}\eta G d}{(1-\beta_{1})^{2}(1-\beta_{2})^{1/2}}.
\] 
[Step 10]

\paragraph{Step 11. Final inequality.}
Dividing by $T$ and rearranging gives
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2(G+\varepsilon)\bigl(f(w_{1})-f^{*}\bigr)}{\eta(1-\beta_{1})\sqrt{T}}
+\frac{L\eta(1+\beta_{1})}{(1-\beta_{1})\sqrt{T}}
\Bigl(\frac{G^{2}}{1-\beta_{2}}+\frac{\sigma^{2}}{(1-\beta_{2})^{2}}\Bigr),
\] 
which is exactly the bound stated in Theorem \ref{thm:amsgrad}. \qedhere
[Step 11]

\section*{Rate Derivation (With Constants)}
Define the constants
\[
C_{1}:=\frac{2(G+\varepsilon)\bigl(f(w_{1})-f^{*}\bigr)}{\eta(1-\beta_{1})},
\qquad
C_{2}:=\frac{L\eta(1+\beta_{1})}{(1-\beta_{1})}
\Bigl(\frac{G^{2}}{1-\beta_{2}}+\frac{\sigma^{2}}{(1-\beta_{2})^{2}}\Bigr).
\]
The bound of Theorem \ref{thm:amsgrad} reads
\[
\frac{1}{T}\sum_{t=1}^{T}\mathbb{E}\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]\le
\frac{C_{1}+C_{2}}{\sqrt{T}}.
\]
Thus the algorithm attains the optimal $O(T^{-1/2})$ rate for stochastic non‑convex optimization.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic setting ($\sigma=0$).} The variance term disappears and the rate improves to $O(T^{-1/2})$ with a smaller constant $C_{2}=L\eta(1+\beta_{1})G^{2}/[(1-\beta_{1})(1-\beta_{2})]$.
\item \textbf{Full‑batch Adam ($\hat v_{t}=v_{t}$).} Without the $\max$ operation the monotonicity property is lost; the proof breaks at Step 10, which is why Adam lacks a provable guarantee under the same assumptions.
\item \textbf{Convex $L$‑smooth case.} By replacing Lemma \ref{lem:smooth} with the convex inequality, one recovers the $O(1/\sqrt{T})$ regret bound for online convex optimization.
\end{itemize}
