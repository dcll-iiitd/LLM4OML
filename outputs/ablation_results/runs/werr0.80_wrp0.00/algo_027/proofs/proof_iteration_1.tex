\section*{Probabilistic Coordinate Descent (PCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable and bounded below by $f^{\star}$.  
At iteration $t$ a coordinate $i_{t}\in\{1,\dots ,d\}$ is drawn independently
according to a fixed probability vector $p=(p_{1},\dots ,p_{d})$, where
$p_{i}>0$ and $\sum_{i=1}^{d}p_{i}=1$.

Denote by $\be_{i}\in\R^{d}$ the $i$‑th canonical basis vector and by 
$g_{i}(w)=\partial_{i}f(w)$ the $i$‑th partial derivative.
The update is
\[
\boxed{
w_{t+1}=w_{t}-\frac{\eta}{p_{i_{t}}}\, g_{i_{t}}(w_{t})\,\be_{i_{t}}
}
\tag{1}
\]
where $\eta>0$ is a (global) stepsize.  
The factor $1/p_{i_{t}}$ makes the stochastic direction an unbiased estimator
of the full gradient:
\[
\E\!\left[\frac{1}{p_{i_{t}}}g_{i_{t}}(w_{t})\be_{i_{t}}\mid w_{t}\right]
= \sum_{i=1}^{d}p_{i}\frac{1}{p_{i}}g_{i}(w_{t})\be_{i}
= \grad f(w_{t}).                                      \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsize $\eta>0$, 
probability vector $p\in\Delta^{d}$.
\FOR{$t=0,1,\dots ,T-1$}
    \STATE Sample $i_{t}\sim p$.
    \STATE Compute partial derivative $g_{i_{t}}(w_{t})=\partial_{i_{t}}f(w_{t})$.
    \STATE $w_{t+1}\gets w_{t}-\dfrac{\eta}{p_{i_{t}}}\,g_{i_{t}}(w_{t})\,\be_{i_{t}}$.
\ENDFOR
\STATE \textbf{Return} $w_{T}$ (or the best iterate among $\{w_{t}\}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional function $f(w)=(w-3)^{2}$, initial point $w_{0}=0$,
stepsize $\eta=0.1$, and $d=1$ so that $p_{1}=1$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $t=0$:  
    $g_{1}(w_{0})=\partial_{1}f(0)=2(0-3)=-6$.  
    Update: $w_{1}=0-\dfrac{0.1}{1}(-6)=0+0.6=0.6$.  
    $f(w_{1})=(0.6-3)^{2}=5.76$.

\item $t=1$:  
    $g_{1}(w_{1})=2(0.6-3)=-4.8$.  
    $w_{2}=0.6-\dfrac{0.1}{1}(-4.8)=0.6+0.48=1.08$.  
    $f(w_{2})=(1.08-3)^{2}=3.6864$.

\item $t=2$:  
    $g_{1}(w_{2})=2(1.08-3)=-3.84$.  
    $w_{3}=1.08-\dfrac{0.1}{1}(-3.84)=1.08+0.384=1.464$.  
    $f(w_{3})=(1.464-3)^{2}=2.3690$.
\end{enumerate}
The objective value decreases at each step, illustrating the algorithm.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L_{i}$‑smoothness]\label{ass:smooth}
For every $i\in\{1,\dots ,d\}$ there exists $L_{i}>0$ such that
\[
\bigl|\,\partial_{i}f(x+\alpha\be_{i})-\partial_{i}f(x)\,\bigr|
\le L_{i}\,|\alpha|\qquad\forall x\in\R^{d},\;\forall\alpha\in\R .
\tag{3}
\]
Equivalently,
\[
f(x+\alpha\be_{i})\le f(x)+\partial_{i}f(x)\,\alpha+\frac{L_{i}}{2}\alpha^{2}.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Lower boundedness]\label{ass:lb}
There exists $f^{\star}>-\infty$ such that $f(w)\ge f^{\star}$ for all $w\in\R^{d}$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies $0<\eta\le \displaystyle\frac{1}{\max_{i}L_{i}/p_{i}}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{w_{t}\}$ be generated by the PCD update \eqref{1} under
Assumptions \ref{ass:smooth}–\ref{ass:stepsize}.
Define the random gradient norm $G_{t}\coloneqq\|\grad f(w_{t})\|^{2}
=\sum_{i=1}^{d}(\partial_{i}f(w_{t}))^{2}$.
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[G_{t}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\,T}.
\tag{5}
\]
Consequently,
\[
\min_{0\le t<T}\E\!\bigl[\|\grad f(w_{t})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\,T}.
\tag{6}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Rate.} The bound \eqref{5} shows an $\mathcal O(1/T)$ decay of the
expected squared gradient norm, which is the standard rate for
non‑convex smooth optimization with an unbiased stochastic gradient.
\item \textbf{Effect of sampling probabilities.}
The quantity $\max_{i}L_{i}/p_{i}$ governs the admissible stepsize
(Assumption \ref{ass:stepsize}) and thus directly influences the constant in
\eqref{5}. Choosing $p_{i}\propto L_{i}$ equalises the terms $L_{i}/p_{i}$
and yields the largest possible stepsize.
\item \textbf{Stability.}
Because the update uses the unbiased estimator \eqref{2} and the smoothness
inequality \eqref{4}, the algorithm is guaranteed to be stable as long as
$\eta$ respects Assumption \ref{ass:stepsize}.
\item \textbf{Comparison.}
When $p_{i}=1/d$ (uniform sampling) the bound reduces to the classic
randomized coordinate descent rate with stepsize $\eta\le d/\max_{i}L_{i}$.
Non‑uniform sampling can dramatically improve the constant when the $L_{i}$
vary widely.
\end{itemize}

\section*{Discussion}
The theorem treats the non‑convex case; no convexity is required.
If additionally $f$ is $L$‑smooth (i.e. $L_{i}\le L$ for all $i$) and uniform
sampling is used, the bound coincides with the standard SGD result for
coordinate‑wise stochastic gradients. The proof below follows the same
template as the classical analysis of SGD, with the only extra technical
step being the weighting by $1/p_{i}$ in the update.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
Under Assumption \ref{ass:smooth}, for any $w\in\R^{d}$ and any scalar
$\alpha\in\R$,
\[
f(w+\alpha\be_{i})\le f(w)+\partial_{i}f(w)\,\alpha
+\frac{L_{i}}{2}\alpha^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Directly from \eqref{4}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken with respect to
the random coordinate $i_{t}$ conditioned on the history up to iteration $t$,
denoted $\mathcal F_{t}$.

\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with
$\alpha=-\dfrac{\eta}{p_{i_{t}}}\partial_{i_{t}}f(w_{t})$:
\[
f(w_{t+1})\le f(w_{t})
-\frac{\eta}{p_{i_{t}}}\bigl(\partial_{i_{t}}f(w_{t})\bigr)^{2}
+\frac{L_{i_{t}}}{2}\Bigl(\frac{\eta}{p_{i_{t}}}\partial_{i_{t}}f(w_{t})\Bigr)^{2}.
\tag{8}
\]

\noindent\textbf{[Step 2]} Rearrange \eqref{8}:
\[
f(w_{t+1})\le f(w_{t})
-\frac{\eta}{p_{i_{t}}}\bigl(\partial_{i_{t}}f(w_{t})\bigr)^{2}
\Bigl[1-\frac{\eta L_{i_{t}}}{2p_{i_{t}}}\Bigr].
\tag{9}
\]

\noindent\textbf{[Step 3]} Take conditional expectation $\E[\cdot\mid\mathcal F_{t}]$.
Using $\Prob(i_{t}=i)=p_{i}$,
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal F_{t}\bigr]
\le f(w_{t})-\eta\sum_{i=1}^{d}
\bigl(\partial_{i}f(w_{t})\bigr)^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_{i}}{p_{i}}
\bigl(\partial_{i}f(w_{t})\bigr)^{2}.
\tag{10}
\]

\noindent\textbf{[Step 4]} Define the gradient norm
$G_{t}\coloneqq\|\grad f(w_{t})\|^{2}
= \sum_{i=1}^{d}(\partial_{i}f(w_{t}))^{2}$.
Introduce the constant
\[
\Lambda\;\coloneqq\;\max_{i}\frac{L_{i}}{p_{i}}.
\tag{11}
\]
Since each term $\frac{L_{i}}{p_{i}}\le\Lambda$, inequality \eqref{10} yields
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal F_{t}\bigr]
\le f(w_{t})-\eta G_{t}
+\frac{\eta^{2}\Lambda}{2}G_{t}
= f(w_{t})-\Bigl(\eta-\frac{\eta^{2}\Lambda}{2}\Bigr)G_{t}.
\tag{12}
\]

\noindent\textbf{[Step 5]} By Assumption \ref{ass:stepsize},
$0<\eta\le 1/\Lambda$, thus $\eta-\frac{\eta^{2}\Lambda}{2}\ge\frac{\eta}{2}$.
Hence,
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal F_{t}\bigr]
\le f(w_{t})-\frac{\eta}{2}G_{t}.
\tag{13}
\]

\noindent\textbf{[Step 6]} Take total expectation and sum over $t=0,\dots ,T-1$:
\[
\E\!\bigl[f(w_{T})\bigr]
\le f(w_{0})-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[G_{t}\bigr].
\tag{14}
\]

\noindent\textbf{[Step 7]} Using the lower bound $f(w_{T})\ge f^{\star}$ from
Assumption \ref{ass:lb},
\[
f^{\star}\le f(w_{0})-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[G_{t}\bigr]
\;\Longrightarrow\;
\sum_{t=0}^{T-1}\E\!\bigl[G_{t}\bigr]
\le\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta}.
\tag{15}
\]

\noindent\textbf{[Step 8]} Divide both sides of \eqref{15} by $T$ and obtain
the average bound \eqref{5}.  The bound on the minimum follows from the
observation that $\min_{t}a_{t}\le\frac{1}{T}\sum_{t}a_{t}$ for any
non‑negative sequence $\{a_{t}\}$, yielding \eqref{6}.  \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main} and the definition \eqref{11},
choosing the maximal admissible stepsize
\[
\eta^{\star}=\frac{1}{\Lambda}
=\frac{1}{\displaystyle\max_{i}\frac{L_{i}}{p_{i}}}
\tag{16}
\]
gives the explicit rate
\[
\min_{0\le t<T}\E\!\bigl[\|\grad f(w_{t})\|^{2}\bigr]
\;\le\;
\frac{2\Lambda\bigl(f(w_{0})-f^{\star}\bigr)}{T}.
\tag{17}
\]
If the probabilities are chosen as $p_{i}=L_{i}/\sum_{j}L_{j}$,
then $\Lambda=\sum_{j}L_{j}$ and the bound becomes
\[
\min_{t}\E\!\bigl[\|\grad f(w_{t})\|^{2}\bigr]
\le \frac{2\bigl(\sum_{j}L_{j}\bigr)\bigl(f(w_{0})-f^{\star}\bigr)}{T},
\]
which is optimal up to the constant factor for coordinate‑wise smooth
functions.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_{i}=1/d$.
    Then $\Lambda = d\max_{i}L_{i}$ and the admissible stepsize is
    $\eta\le 1/(d\max_{i}L_{i})$, recovering the classic
    $O(d\max_{i}L_{i}/T)$ bound.
\item \textbf{Exact coordinate smoothness.}
    If all $L_{i}$ are equal to a common $L$, any distribution satisfies
    $\Lambda = L/\min_{i}p_{i}$.  The best choice is $p_{i}=1/d$, which yields
    $\Lambda = dL$.
\item \textbf{Convex case.}
    When $f$ is additionally convex, the same analysis gives a
    $O(1/T)$ convergence of the function values
    $f(\bar w_{T})-f^{\star}$, where $\bar w_{T}=\frac{1}{T}\sum_{t}w_{t}$.
\end{itemize}
