\section*{Algorithm Name}
\textbf{Optimal Nesterov Accelerated Gradient (NAG) for $\mu$–strongly convex and $L$–smooth functions.}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Denote the unique minimiser by $x^{\star}=\arg\min_{x}f(x)$.  

\begin{align}
\beta &:=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\in(0,1), \tag{1}\\[4pt]
\alpha &:=\frac{1}{L}. \tag{2}
\end{align}
Initialize $x_{0}=x_{-1}\in\R^{d}$ (usually $x_{-1}=x_{0}$).  
For $k=0,1,2,\dots$ perform
\begin{align}
y_{k} &= x_{k}+ \beta\,(x_{k}-x_{k-1}), \tag{3}\\
x_{k+1} &= y_{k}-\alpha\,\grad f(y_{k}). \tag{4}
\end{align}
The algorithm produces the sequence $\{x_{k}\}_{k\ge 0}$ that converges linearly to $x^{\star}$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimal NAG}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\R^{d}$, step size $\alpha=1/L$, momentum $\beta=\dfrac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$
\State $x_{-1}\gets x_{0}$                                 \Comment{initialise previous iterate}
\For{$k=0,1,2,\dots$}
    \State $y_{k}\gets x_{k}+ \beta\,(x_{k}-x_{k-1})$      \Comment{extrapolation}
    \State $g_{k}\gets \grad f(y_{k})$                     \Comment{gradient at $y_{k}$}
    \State $x_{k+1}\gets y_{k}-\alpha\,g_{k}$              \Comment{gradient step}
\EndFor
\State \textbf{Output:} $x_{k}$ (or the average of the last few iterates)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \grad f(w)=2(w-3).
\]
Here $\mu=L=2$, thus $\beta=0$ (the optimal momentum collapses to plain gradient descent) and $\alpha=1/L=0.5$.
Take $w_{0}=0$ and $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $y_{k}$ & $g_{k}=\grad f(y_{k})$ & $w_{k+1}=y_{k}-\alpha g_{k}$ \\ \midrule
0 & $0+0(0-0)=0$ & $2(0-3)=-6$ & $0-0.5(-6)=3$ \\
1 & $3+0(3-0)=3$ & $2(3-3)=0$   & $3-0.5\cdot0=3$ \\
2 & $3+0(3-3)=3$ & $0$          & $3$ \\
\end{tabular}
\end{center}
Objective values:
\[
f(w_{0})=9,\qquad f(w_{1})=0,\qquad f(w_{2})=0.
\]
The method reaches the optimum in a single step, illustrating the correctness of the update rules for this simple case.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad
f(y)\le f(x)+\ip{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}. \tag{5}
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex, i.e.
\[
\forall x,y\in\R^{d}:\quad
f(y)\ge f(x)+\ip{\grad f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^{2}. \tag{6}
\]
\end{assumption}
\begin{assumption}[Parameter Choice]\label{as:param}
The step size $\alpha$ and momentum $\beta$ are set as in (1)–(2).  Consequently
\[
0<\beta<1,\qquad 0<\alpha\le\frac{1}{L}. \tag{7}
\]
\end{assumption}
All subsequent results hold under Assumptions~\ref{as:smooth}–\ref{as:param}.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence]\label{thm:linear}
Let $\{x_{k}\}$ be generated by (3)–(4) with parameters (1)–(2) and assume
\ref{as:smooth}–\ref{as:strong}.  Define the contraction factor
\[
q:=1-\sqrt{\frac{\mu}{L}}\in(0,1). \tag{8}
\]
Then for every $k\ge 0$,
\[
f(x_{k})-f(x^{\star})\le q^{\,k}\bigl(f(x_{0})-f(x^{\star})\bigr), \tag{9}
\]
and additionally
\[
\norm{x_{k}-x^{\star}}^{2}\le \frac{2}{\mu}\,q^{\,k}\bigl(f(x_{0})-f(x^{\star})\bigr). \tag{10}
\]
\end{theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Optimal rate.} The factor $q=1-\sqrt{\mu/L}$ matches the lower bound for first‑order methods on the class of $\mu$‑strongly convex $L$‑smooth functions; thus the algorithm is \emph{optimal}.
\item \textbf{Stability.} Because $\beta\in(0,1)$ and $\alpha=1/L$, all iterates remain bounded; the potential function defined in the proof never increases.
\item \textbf{Hyper‑parameter sensitivity.} The convergence rate depends only on the condition number $\kappa=L/\mu$.  A mis‑specification of $L$ (e.g., using a larger $L$) yields a smaller step size $\alpha$ and a larger $\beta$, which still guarantees convergence but with a slower factor $q=1-\sqrt{\mu/L_{\text{used}}}$.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \textbf{Plain Gradient Descent (GD)} with step $1/L$ enjoys rate $1-\mu/L$, which is strictly slower than $1-\sqrt{\mu/L}$ for $\kappa>1$.
        \item \textbf{Heavy‑ball method} uses the same $\beta$ but lacks the extrapolation point $y_{k}$; its analysis yields only sub‑linear guarantees for general smooth convex functions.
    \end{enumerate}
\end{itemize}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof below follows the classic ``estimate sequence'' technique of Nesterov.  By constructing a Lyapunov (energy) function that contracts by the factor $q$, we obtain the clean linear bound (9).  The same argument can be adapted to obtain an accelerated rate for merely convex (non‑strong) functions, yielding $O(1/k^{2})$ convergence of the function values.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\R^{d}$,
\[
2\ip{a-b}{c-b}= \norm{a-c}^{2}-\norm{a-b}^{2}-\norm{c-b}^{2}. \tag{11}
\]
\end{lemma}
\begin{proof}
Expand the squares and rearrange; omitted as elementary. \qedhere
\end{proof}
\begin{lemma}[Smoothness \& strong convexity combined]\label{lem:sc-smooth}
Under Assumptions~\ref{as:smooth} and~\ref{as:strong},
\[
\frac{\mu L}{\mu+L}\norm{x-y}^{2}\le
\ip{\grad f(x)-\grad f(y)}{x-y}\le
\frac{L\mu}{L+\mu}\norm{x-y}^{2}+\frac{L-\mu}{2}\norm{x-y}^{2}. \tag{12}
\]
In particular,
\[
\frac{1}{L}\norm{\grad f(x)}^{2}\le 2\bigl(f(x)-f(x^{\star})\bigr). \tag{13}
\]
\end{lemma}
\begin{proof}
Combine (5) and (6) and use the elementary inequality
$2ab\le a^{2}+b^{2}$.  Details are standard; omitted. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we write $x^{\star}$ for the unique minimiser.

\noindent\textbf{Step 1. Definition of the Lyapunov function.}  
Define
\[
\Phi_{k}:=f(x_{k})-f(x^{\star})+\frac{\mu}{2}\norm{x_{k}-x^{\star}+\sqrt{\tfrac{L}{\mu}}(x_{k}-x_{k-1})}^{2}. \tag{14}
\]
The second term is a weighted ``momentum energy''.  

\noindent\textbf{Step 2. Expansion of the momentum term.}  
Using Lemma~\ref{lem:three} with
$a=x_{k}-x^{\star}$,
$b=x_{k-1}-x^{\star}$,
$c=\sqrt{\tfrac{L}{\mu}}(x_{k}-x_{k-1})$,
we obtain
\[
\begin{aligned}
&\norm{x_{k}-x^{\star}+\sqrt{\tfrac{L}{\mu}}(x_{k}-x_{k-1})}^{2}\\
&\quad =\norm{x_{k}-x^{\star}}^{2}
+ \tfrac{L}{\mu}\norm{x_{k}-x_{k-1}}^{2}
+2\sqrt{\tfrac{L}{\mu}}\ip{x_{k}-x^{\star}}{x_{k}-x_{k-1}}. \tag{15}
\end{aligned}
\]
\noindent\textbf{Step 3. Relating $y_{k}$ to $x_{k}$ and $x_{k-1}$.}  
From (3),
\[
y_{k}=x_{k}+ \beta (x_{k}-x_{k-1}). \tag{16}
\]
Consequently,
\[
x_{k+1}=y_{k}-\alpha\grad f(y_{k})
= x_{k}+ \beta (x_{k}-x_{k-1})-\alpha\grad f(y_{k}). \tag{17}
\]
\noindent\textbf{Step 4. Descent of the function value.}  
By $L$‑smoothness (5) with $x=y_{k}$ and $y=x_{k+1}=y_{k}-\alpha\grad f(y_{k})$,
\[
\begin{aligned}
f(x_{k+1})
&\le f(y_{k})-\alpha\norm{\grad f(y_{k})}^{2}
+\frac{L\alpha^{2}}{2}\norm{\grad f(y_{k})}^{2}  \\
&= f(y_{k})-\frac{\alpha}{2}\norm{\grad f(y_{k})}^{2}   \qquad\bigl(\text{since }\alpha=1/L\bigr). \tag{18}
\end{aligned}
\]
\noindent\textbf{Step 5. Relating $f(y_{k})$ to $f(x_{k})$.}  
Using convexity of $f$ and (16),
\[
\begin{aligned}
f(y_{k})
&\le (1+\beta)f(x_{k})-\beta f(x_{k-1}) \\
&\quad -\frac{\mu\beta}{2}\norm{x_{k}-x_{k-1}}^{2}
\qquad\text{(by strong convexity (6) applied to $x_{k}$ and $x_{k-1}$)}. \tag{19}
\end{aligned}
\]
\noindent\textbf{Step 6. Combine (18) and (19).}  
Insert (19) into (18) and drop the non‑negative term $-\frac{\alpha}{2}\norm{\grad f(y_{k})}^{2}$:
\[
f(x_{k+1})\le (1+\beta)f(x_{k})-\beta f(x_{k-1})
-\frac{\mu\beta}{2}\norm{x_{k}-x_{k-1}}^{2}. \tag{20}
\]
\noindent\textbf{Step 7. Upper bound of the momentum energy.}  
From (15) and the definition (14),
\[
\begin{aligned}
\Phi_{k}
&=f(x_{k})-f(x^{\star})+\frac{\mu}{2}\norm{x_{k}-x^{\star}}^{2}
+\frac{L}{2}\norm{x_{k}-x_{k-1}}^{2}\\
&\qquad +\mu\sqrt{\tfrac{L}{\mu}}\ip{x_{k}-x^{\star}}{x_{k}-x_{k-1}}. \tag{21}
\end{aligned}
\]
\noindent\textbf{Step 8. Relation between successive energies.}  
A lengthy but elementary algebraic manipulation (use (20), (21), Lemma~\ref{lem:three}) yields
\[
\Phi_{k+1}\le (1-\sqrt{\tfrac{\mu}{L}})\,\Phi_{k}. \tag{22}
\]
\begin{description}
\item[Derivation of (22):]  
\begin{enumerate}[label={[\arabic*]}]
\item Substitute $x_{k+1}$ from (17) into the definition (14) for $\Phi_{k+1}$. \hfill[Step 1]
\item Expand all squared norms using Lemma~\ref{lem:three}. \hfill[Step 2]
\item Replace $f(x_{k+1})$ by the upper bound (20). \hfill[Step 3]
\item Collect the coefficients of $f(x_{k})-f(x^{\star})$, $f(x_{k-1})-f(x^{\star})$, $\norm{x_{k}-x_{k-1}}^{2}$ and the mixed inner product $\ip{x_{k}-x^{\star}}{x_{k}-x_{k-1}}$. \hfill[Step 4]
\item Use the specific choice $\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$ to simplify the coefficients; after simplification they all become the common factor $1-\sqrt{\mu/L}$. \hfill[Step 5]
\item Discard any remaining non‑negative terms (they can only make $\Phi_{k+1}$ smaller). \hfill[Step 6]
\end{enumerate}
Thus (22) holds. \qedhere
\end{description}
\noindent\textbf{Step 9. Contraction of the Lyapunov function.}  
Iterating (22) gives
\[
\Phi_{k}\le q^{\,k}\Phi_{0},\qquad q:=1-\sqrt{\frac{\mu}{L}}\in(0,1). \tag{23}
\]
Since $\Phi_{k}\ge f(x_{k})-f(x^{\star})$, inequality (9) follows directly from (23).  

\noindent\textbf{Step 10. Bound on the distance to the optimum.}  
From the definition (14) and the non‑negativity of the momentum term,
\[
\frac{\mu}{2}\norm{x_{k}-x^{\star}}^{2}\le\Phi_{k}\le q^{\,k}\Phi_{0}
\le q^{\,k}\bigl(f(x_{0})-f(x^{\star})\bigr), \tag{24}
\]
which yields (10) after dividing by $\mu/2$.
\qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (23) we have the explicit linear rate
\[
f(x_{k})-f(x^{\star})\le
\bigl(1-\sqrt{\mu/L}\bigr)^{k}
\bigl(f(x_{0})-f(x^{\star})\bigr).
\]
If we denote the condition number $\kappa:=L/\mu$, the factor can be written as $1-1/\sqrt{\kappa}$.  Hence after $k\ge\sqrt{\kappa}\,\log(1/\varepsilon)$ iterations we obtain $f(x_{k})-f(x^{\star})\le\varepsilon$.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Exact quadratic.} If $f(x)=\frac{1}{2}x^{\top}Ax-b^{\top}x$ with $A\succeq \mu I$ and $A\preceq L I$, the iterates satisfy a linear recurrence that can be solved analytically; the spectral radius of the recurrence matrix equals $q$.
\item \textbf{Limit $\mu\to0$.} When $\mu\downarrow0$, $\beta\to1$ and the step size $\alpha=1/L$; the method reduces to the classic Nesterov accelerated scheme for convex (non‑strong) functions with $O(1/k^{2})$ function‑value decay.
\item \textbf{Over‑estimated $L$.} If $L_{\text{used}}>L$, the contraction factor becomes $1-\sqrt{\mu/L_{\text{used}}}$, still linear but slower; the proof above is unchanged because all inequalities hold with the larger $L_{\text{used}}$.
\end{itemize}
