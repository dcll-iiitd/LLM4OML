\section*{Algorithm Name}
\textbf{Variance‑Reduced Stochastic Gradient Descent with Outer Full‑Gradient Loop (SVRG‑PL)}  

The method consists of an outer loop that computes the exact gradient of the finite‑sum objective at a reference point and an inner loop that performs $m$ stochastic updates using the SVRG estimator.

\section*{Mathematical Formulation}
Consider the finite‑sum problem
\[
\min_{x\in\mathbb{R}^d}\; f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each $f_i$ is $L$‑smooth.  
Let $x^{(s)}$ denote the reference point at outer iteration $s$ and $\tilde{\mu}^{(s)}:=\nabla f(x^{(s)})$ its full gradient.  

For inner iteration $t=0,\dots,m-1$ we keep a copy $x_t^{(s)}$ and draw an index $i_t$ uniformly from $\{1,\dots,n\}$.  
The variance‑reduced estimator is
\[
v_t^{(s)} \;:=\; \nabla f_{i_t}\!\bigl(x_t^{(s)}\bigr)-\nabla f_{i_t}\!\bigl(x^{(s)}\bigr)+\tilde{\mu}^{(s)} .
\]
The update rule is
\[
x_{t+1}^{(s)} \;=\; x_t^{(s)} - \eta\, v_t^{(s)} ,
\]
with stepsize $\eta>0$.  
After $m$ inner steps we set $x^{(s+1)}:=x_m^{(s)}$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG‑PL}
\begin{algorithmic}[1]
\Require Number of outer epochs $S$, inner length $m$, stepsize $\eta>0$, initial point $x^{(0)}$.
\For{$s=0$ \textbf{to} $S-1$}
    \State $\tilde{\mu}^{(s)}\gets \nabla f\bigl(x^{(s)}\bigr)=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i\bigl(x^{(s)}\bigr)$ \Comment{full gradient}
    \State $x_0^{(s)}\gets x^{(s)}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim \text{Unif}\{1,\dots,n\}$
        \State $v_t^{(s)}\gets \nabla f_{i_t}\!\bigl(x_t^{(s)}\bigr)-\nabla f_{i_t}\!\bigl(x^{(s)}\bigr)+\tilde{\mu}^{(s)}$
        \State $x_{t+1}^{(s)}\gets x_t^{(s)}-\eta\, v_t^{(s)}$
    \EndFor
    \State $x^{(s+1)}\gets x_m^{(s)}$
\EndFor
\State \textbf{return} $x^{(S)}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$; here $n=1$, $f_1=f$.  
Parameters: $w_0=0$, learning rate $\eta=0.1$, inner length $m=3$, outer epochs $S=1$.

\begin{enumerate}[label=\textbf{Step \arabic*.}]
\item Compute full gradient at $w^{(0)}=0$:
\[
\tilde{\mu}^{(0)}=\nabla f(0)=2(0-3)=-6 .
\]
\item Initialise inner iterate $w_0^{(0)}=0$.
\item \textbf{Inner iteration $t=0$}: sample $i_0=1$.
\[
v_0^{(0)}=\nabla f(w_0^{(0)})-\nabla f(w^{(0)})+\tilde{\mu}^{(0)}
          =(-6)-(-6)+(-6)=-6 .
\]
Update $w_1^{(0)}=0-0.1(-6)=0.6$.
Objective $f(0.6)=(0.6-3)^2=5.76$.
\item \textbf{Inner iteration $t=1$}:
\[
v_1^{(0)}=\nabla f(0.6)-\nabla f(0)+\tilde{\mu}^{(0)}
        =2(0.6-3)-(-6)-6 = -4.8-(-6)-6 = -4.8+6-6 = -4.8 .
\]
Update $w_2^{(0)}=0.6-0.1(-4.8)=1.08$.
Objective $f(1.08)=(1.08-3)^2=3.6864$.
\item \textbf{Inner iteration $t=2$}:
\[
v_2^{(0)}=\nabla f(1.08)-\nabla f(0)+\tilde{\mu}^{(0)}
        =2(1.08-3)-(-6)-6 = -3.84+6-6 = -3.84 .
\]
Update $w_3^{(0)}=1.08-0.1(-3.84)=1.464$.
Objective $f(1.464)=(1.464-3)^2=2.366\text{ (approx)}$.
\item Set $w^{(1)}=w_3^{(0)}=1.464$.
\end{enumerate}
The algorithm makes progress toward the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad \forall x,y\in\mathbb{R}^d .
\]
Consequently $f$ is $L$‑smooth.
\end{assumption}

\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:PL}
There exists $\mu>0$ such that for all $x$,
\[
\frac{1}{2}\|\nabla f(x)\|^2 \;\ge\; \mu\bigl(f(x)-f^\star\bigr),
\]
where $f^\star:=\min_x f(x)$.
\end{assumption}

\begin{assumption}[Finite sum]\label{ass:finite}
The objective is a finite average $f(x)=\frac1n\sum_{i=1}^n f_i(x)$ with $n\ge 1$.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The learning rate satisfies
\[
0<\eta\le \frac{1}{3L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence under PL]\label{thm:linear}
Let Assumptions \ref{ass:smooth}–\ref{ass:stepsize} hold.  
Choose the inner loop length
\[
m \;\ge\; \frac{2}{\eta\mu}\, .
\]
Then the sequence $\{x^{(s)}\}$ generated by SVRG‑PL satisfies
\[
\E\!\bigl[f(x^{(s)})-f^\star\bigr]\;\le\;\rho^{\,s}\,\bigl(f(x^{(0)})-f^\star\bigr),
\qquad\text{with}\qquad 
\rho:=\frac{1}{1+\eta\mu m/2}\;<1 .
\]
Hence the method enjoys a geometric (linear) rate in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound $\eta\le 1/(3L)$ guarantees that the variance‑reduced estimator is an unbiased contraction of the true gradient (see Lemma~\ref{lem:variance}).
\item \textbf{Hyper‑parameter sensitivity.} Increasing $m$ improves the contraction factor $\rho$ but also raises per‑epoch cost; the optimal trade‑off is $m\approx\Theta(L/\mu)$, matching classical SVRG analysis.
\item \textbf{Comparison.} Under the same $L$ and $\mu$, plain SGD with diminishing stepsize yields $O(1/T)$ sub‑linear decay, whereas SVRG‑PL attains linear decay with the same computational budget per epoch.
\end{enumerate}

\section*{Discussion}
The theorem shows that the PL condition, weaker than strong convexity, is sufficient for linear convergence of SVRG. The proof exploits smoothness to control the variance of the estimator and the PL inequality to relate gradient norm to sub‑optimality. The result extends the classical SVRG analysis to non‑strongly convex but PL‑satisfying objectives.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{proof}
Standard from the definition of $L$‑smoothness. \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the SVRG estimator]\label{lem:unbiased}
For any inner iterate $x_t^{(s)}$,
\[
\E_{i_t}\bigl[v_t^{(s)}\mid x_t^{(s)}\bigr]=\nabla f\!\bigl(x_t^{(s)}\bigr).
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly sampled index $i_t$,
\[
\E_{i_t}\!\bigl[\nabla f_{i_t}(x_t^{(s)})\bigr]=\nabla f(x_t^{(s)}),\qquad
\E_{i_t}\!\bigl[\nabla f_{i_t}(x^{(s)})\bigr]=\nabla f(x^{(s)}),
\]
hence the claim follows. \qedhere
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance}
For any $x_t^{(s)}$,
\[
\E_{i_t}\!\bigl[\|v_t^{(s)}-\nabla f(x_t^{(s)})\|^2\mid x_t^{(s)}\bigr]
\;\le\; 2L\bigl(f(x_t^{(s)})-f^\star+f(x^{(s)})-f^\star\bigr).
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the identity
\[
v_t^{(s)}-\nabla f(x_t^{(s)})=
\bigl(\nabla f_{i_t}(x_t^{(s)})-\nabla f_{i_t}(x^{(s)})\bigr)
-\bigl(\nabla f(x_t^{(s)})-\nabla f(x^{(s)})\bigr),
\]
we obtain
\[
\|v_t^{(s)}-\nabla f(x_t^{(s)})\|^2
\le 2\|\nabla f_{i_t}(x_t^{(s)})-\nabla f_{i_t}(x^{(s)})\|^2
   +2\|\nabla f(x_t^{(s)})-\nabla f(x^{(s)})\|^2 .
\]
Taking expectation and applying smoothness yields the bound. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  
All expectations are taken w.r.t. the randomness of the inner indices $\{i_t\}$ conditioned on the outer iterate $x^{(s)}$.

\begin{enumerate}[label=[Step \arabic*]]
\item \textbf{One‑step recursion.}  
From the update $x_{t+1}=x_t-\eta v_t$ and expanding the squared distance to the optimum $x^\star$,
\[
\|x_{t+1}-x^\star\|^2
= \|x_t-x^\star\|^2-2\eta\langle v_t,x_t-x^\star\rangle+\eta^2\|v_t\|^2 .
\tag{1}
\]

\item \textbf{Take conditional expectation.}  
Using Lemma~\ref{lem:unbiased} and the fact that $\E[\|v_t\|^2]=\|\nabla f(x_t)\|^2+\E\|v_t-\nabla f(x_t)\|^2$, we obtain
\[
\E\bigl[\|x_{t+1}-x^\star\|^2\mid x_t\bigr]
= \|x_t-x^\star\|^2-2\eta\langle \nabla f(x_t),x_t-x^\star\rangle
   +\eta^2\|\nabla f(x_t)\|^2+\eta^2\E\|v_t-\nabla f(x_t)\|^2 .
\tag{2}
\]

\item \textbf{Apply smoothness inequality.}  
From Lemma~\ref{lem:smooth} with $y=x^\star$ (optimality $\nabla f(x^\star)=0$) we have
\[
f(x_t)-f^\star\le \langle \nabla f(x_t),x_t-x^\star\rangle-\frac{1}{2L}\|\nabla f(x_t)\|^2 .
\tag{3}
\]
Rearranging,
\[
\langle \nabla f(x_t),x_t-x^\star\rangle
\ge f(x_t)-f^\star+\frac{1}{2L}\|\nabla f(x_t)\|^2 .
\tag{4}
\]

\item \textbf{Insert (4) into (2).}  
\[
\begin{aligned}
\E\bigl[\|x_{t+1}-x^\star\|^2\mid x_t\bigr]
&\le \|x_t-x^\star\|^2-2\eta\Bigl(f(x_t)-f^\star\Bigr)
   -\frac{\eta}{L}\|\nabla f(x_t)\|^2   \\
&\quad +\eta^2\|\nabla f(x_t)\|^2
   +\eta^2\E\|v_t-\nabla f(x_t)\|^2 .
\end{aligned}
\tag{5}
\]

\item \textbf{Bound the variance term.}  
Using Lemma~\ref{lem:variance} and Assumption~\ref{ass:stepsize} ($\eta\le 1/(3L)$) we get
\[
\eta^2\E\|v_t-\nabla f(x_t)\|^2
\le 2\eta^2L\bigl(f(x_t)-f^\star+f(x^{(s)})-f^\star\bigr)
\le \frac{2\eta}{3}\bigl(f(x_t)-f^\star\bigr)
   +\frac{2\eta}{3}\bigl(f(x^{(s)})-f^\star\bigr) .
\tag{6}
\]

\item \textbf{Collect terms in (5).}  
Combine the coefficients of $f(x_t)-f^\star$:
\[
-2\eta + \frac{2\eta}{3}= -\frac{4\eta}{3}.
\]
Also, the $\|\nabla f(x_t)\|^2$ terms satisfy $-\frac{\eta}{L}+\eta^2\le 0$ because $\eta\le 1/L$. Hence
\[
\E\bigl[\|x_{t+1}-x^\star\|^2\mid x_t\bigr]
\le \|x_t-x^\star\|^2-\frac{4\eta}{3}\bigl(f(x_t)-f^\star\bigr)
   +\frac{2\eta}{3}\bigl(f(x^{(s)})-f^\star\bigr) .
\tag{7}
\]

\item \textbf{Unroll over $t=0,\dots,m-1$.}  
Take full expectation and sum (7) for $t=0$ to $m-1$:
\[
\begin{aligned}
\E\|x_{m}^{(s)}-x^\star\|^2
&\le \|x^{(s)}-x^\star\|^2
   -\frac{4\eta}{3}\sum_{t=0}^{m-1}\E\bigl[f(x_t^{(s)})-f^\star\bigr]
   +\frac{2\eta m}{3}\bigl(f(x^{(s)})-f^\star\bigr) .
\end{aligned}
\tag{8}
\]

\item \textbf{Relate the average inner function value to the outer one.}  
Since $f$ is convex, $f\bigl(\frac1m\sum_{t}x_t^{(s)}\bigr)\le \frac1m\sum_t f(x_t^{(s)})$.  
Define the outer iterate $x^{(s+1)}:=x_m^{(s)}$ and drop the non‑negative distance term:
\[
\frac{4\eta}{3}\sum_{t=0}^{m-1}\E\bigl[f(x_t^{(s)})-f^\star\bigr]
\ge \frac{4\eta m}{3}\E\bigl[f(x^{(s+1)})-f^\star\bigr].
\tag{9}
\]

\item \textbf{Insert (9) into (8) and rearrange.}
\[
\begin{aligned}
\frac{4\eta m}{3}\E\bigl[f(x^{(s+1)})-f^\star\bigr]
&\le \|x^{(s)}-x^\star\|^2
   +\frac{2\eta m}{3}\bigl(f(x^{(s)})-f^\star\bigr) .
\end{aligned}
\tag{10}
\]

\item \textbf{Use the PL inequality.}  
From Assumption~\ref{ass:PL},
\[
\|x^{(s)}-x^\star\|^2\le \frac{2}{\mu}\bigl(f(x^{(s)})-f^\star\bigr) .
\tag{11}
\]

\item \textbf{Combine (10) and (11).}
\[
\frac{4\eta m}{3}\E\bigl[f(x^{(s+1)})-f^\star\bigr]
\le \Bigl(\frac{2}{\mu}+\frac{2\eta m}{3}\Bigr)\bigl(f(x^{(s)})-f^\star\bigr) .
\tag{12}
\]

\item \textbf{Choose $m$ to obtain a contraction.}  
If $m\ge \frac{2}{\eta\mu}$ then $\frac{2}{\mu}\le \frac{\eta m}{1}$. Hence
\[
\frac{2}{\mu}+\frac{2\eta m}{3}\le \frac{4\eta m}{3}.
\]
Plugging into (12) yields
\[
\E\bigl[f(x^{(s+1)})-f^\star\bigr]
\le \underbrace{\frac{1}{1+\eta\mu m/2}}_{\displaystyle \rho}\,\bigl(f(x^{(s)})-f^\star\bigr) .
\tag{13}
\]

\item \textbf{Iterate the recursion.}  
Applying (13) recursively over $s=0,\dots,S-1$ gives
\[
\E\bigl[f(x^{(S)})-f^\star\bigr]\le \rho^{\,S}\bigl(f(x^{(0)})-f^\star\bigr) .
\tag{14}
\]

\end{enumerate}
Equation (14) is exactly the statement of Theorem~\ref{thm:linear}. \qed

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear}, with $\eta=1/(3L)$ and $m= \lceil 6L/\mu\rceil$ we obtain
\[
\rho = \frac{1}{1+\frac{\eta\mu m}{2}}
      \le \frac{1}{1+\frac{1}{6}}
      = \frac{6}{7}\approx0.857 .
\]
Thus after $S$ outer epochs the sub‑optimality decays as $(6/7)^{S}$, i.e. linear convergence with factor $0.857$ per epoch.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Strongly convex $f$.}  
If $f$ is $\mu$‑strongly convex, the PL condition holds with the same $\mu$, so the same linear rate applies. Moreover, one can set $m=O(L/\mu)$ to achieve the optimal complexity $O((n+L/\mu)\log(1/\epsilon))$.
\item \textbf{Non‑convex but PL.}  
Even when $f$ is non‑convex, the PL inequality guarantees that every stationary point is globally optimal; the theorem therefore yields global linear convergence.
\item \textbf{Limit $\eta\to 0$.}  
When $\eta$ is taken arbitrarily small, the contraction factor $\rho\approx 1-\frac{\eta\mu m}{2}$, and the bound recovers the classical gradient descent rate $1-\eta\mu$ after $m$ inner steps.
\end{enumerate}
