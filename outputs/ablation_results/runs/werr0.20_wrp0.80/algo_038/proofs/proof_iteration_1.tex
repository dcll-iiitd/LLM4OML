\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle–point problem  

\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\;\mathcal{L}(x,y)\;:=\;f(x)+\langle Kx , y\rangle-g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
    \item $f:\mathbb{R}^{n}\rightarrow\mathbb{R}\cup\{+\infty\}$ is proper, closed and convex;
    \item $g^{*}:\mathbb{R}^{m}\rightarrow\mathbb{R}\cup\{+\infty\}$ is the convex conjugate of a convex function $g$;
    \item $K\in\mathbb{R}^{m\times n}$ is a linear operator;
    \item The primal optimal solution set is $\mathcal{X}^{\star}$ and the dual optimal set is $\mathcal{Y}^{\star}$.
\end{itemize}

The PDHG iterates are defined by a primal step (proximal) and a dual step (gradient ascent) together with an extrapolation of the primal variable:

\[
\boxed{
\begin{aligned}
y^{k+1} &:= \prox_{\sigma g^{*}}\!\bigl(y^{k}+\sigma K\tilde{x}^{k}\bigr), \\
x^{k+1} &:= \prox_{\tau f}\!\bigl(x^{k}-\tau K^{\top}y^{k+1}\bigr),\\
\tilde{x}^{k+1} &:= x^{k+1}+ \theta\bigl(x^{k+1}-x^{k}\bigr),
\end{aligned}}
\tag{2}
\]

with step sizes $\tau>0$, $\sigma>0$ and extrapolation parameter $\theta\in[0,1]$.  
In the most common choice $\theta=1$ (full extrapolation) and the stepsizes must satisfy  

\[
\tau\sigma\|K\|^{2}<1,
\tag{3}
\]

where $\|K\|$ denotes the spectral norm of $K$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\Require $f$, $g^{*}$, linear operator $K$, stepsizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, initial $x^{0},y^{0}$.
\State $\tilde{x}^{0}\gets x^{0}$.
\For{$k=0,1,2,\dots$}
    \State $y^{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y^{k}+\sigma K\tilde{x}^{k}\bigr)$ \hfill\Comment{dual ascent}
    \State $x^{k+1}\gets \prox_{\tau f}\!\bigl(x^{k}-\tau K^{\top}y^{k+1}\bigr)$ \hfill\Comment{primal descent}
    \State $\tilde{x}^{k+1}\gets x^{k+1}+ \theta\bigl(x^{k+1}-x^{k}\bigr)$ \hfill\Comment{extrapolation}
\EndFor
\State \textbf{output} $(x^{k},y^{k})$ or ergodic averages $\bar{x}^{K}=\frac1K\sum_{k=1}^{K}x^{k}$, $\bar{y}^{K}=\frac1K\sum_{k=1}^{K}y^{k}$.
\end{algorithmic}
\end{algorithm}

\[

\prox_{\tau f}(z)=\frac{z+ \tau 3}{1+\tau } .

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
\begin{enumerate}
    \item $f$ is convex and $L_{f}$‑smooth, i.e. $\|\nabla f(x)-\nabla f(x')\|\le L_{f}\|x-x'\|$ for all $x,x'$.
    \item $g^{*}$ is convex (no smoothness required).
    \item $K$ is a linear operator with spectral norm $\|K\|$.
\end{enumerate}
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{as:saddle}
There exists $(x^{\star},y^{\star})$ such that  

\[
\mathcal{L}(x^{\star},y)\le\mathcal{L}(x^{\star},y^{\star})\le\mathcal{L}(x,y^{\star}),\qquad\forall x,y.
\tag{4}
\]
\end{assumption}
\begin{assumption}[Stepsize Condition]\label{as:steps}
Choose $\tau,\sigma>0$ and $\theta\in[0,1]$ satisfying  

\[
\tau\sigma\|K\|^{2}\le\frac{1-\theta}{1+\theta}.
\tag{5}
\]

In particular, with $\theta=1$ we need $\tau\sigma\|K\|^{2}<1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Convergence]\label{thm:ergodic}
Let Assumptions \ref{as:convex}–\ref{as:steps} hold and denote the ergodic averages  

\[
\bar{x}^{K}:=\frac{1}{K}\sum_{k=1}^{K}x^{k},\qquad 
\bar{y}^{K}:=\frac{1}{K}\sum_{k=1}^{K}y^{k}.
\]

For any saddle point $(x^{\star},y^{\star})$ we have  

\[
\gap\bigl(\bar{x}^{K},\bar{y}^{K}\bigr)
:=\max_{y}\mathcal{L}(\bar{x}^{K},y)-\min_{x}\mathcal{L}(x,\bar{y}^{K})
\le \frac{1}{2K}\Bigl(
\frac{\|x^{0}-x^{\star}\|^{2}}{\tau}
+\frac{\|y^{0}-y^{\star}\|^{2}}{\sigma}
\Bigr).
\tag{6}
\]

Consequently, $\gap(\bar{x}^{K},\bar{y}^{K})=O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The condition \eqref{as:steps} guarantees that the Lyapunov function  
    $V^{k}:=\frac{1}{2\tau}\|x^{k}-x^{\star}\|^{2}+\frac{1}{2\sigma}\|y^{k}-y^{\star}\|^{2}$  
    is non‑increasing up to a telescoping term (see the proof).
    \item \textbf{Hyperparameter Sensitivity.} Larger steps $\tau,\sigma$ accelerate convergence until the bound \eqref{as:steps} is violated; the optimal choice (when $\|K\|$ is known) is $\tau=\sigma=1/\|K\|$ for $\theta=0$.
    \item \textbf{Comparison to Baselines.} Compared with vanilla gradient descent on the primal alone, PDHG attains the same $O(1/K)$ rate without requiring $f$ to be strongly convex, and it handles the non‑smooth term $g^{*}$ via a cheap proximal map.
\end{itemize}

\section*{Discussion}
The theorem shows that PDHG enjoys an \emph{ergodic} convergence rate identical to that of the classical primal‑only gradient method for smooth convex problems, while simultaneously coping with a possibly non‑smooth dual term. The proof is constructive: it builds a Lyapunov sequence that telescopes, and the extrapolation parameter $\theta$ can be tuned to improve practical performance without affecting the $O(1/K)$ guarantee.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three–point identity for proximal operator]\label{lem:prox}
For any convex $h$ and $\lambda>0$, let $u^{+}=\prox_{\lambda h}(u)$. Then for all $v$,
\[
\langle u-u^{+},v-u^{+}\rangle \ge \lambda\bigl(h(v)-h(u^{+})\bigr).
\tag{7}
\]
\end{lemma}
\begin{proof}
By definition of $\prox$, $u^{+}$ minimizes $h(z)+\frac{1}{2\lambda}\|z-u\|^{2}$, thus the first‑order optimality condition gives
\[
0\in \partial h(u^{+})+\frac{1}{\lambda}(u^{+}-u),
\]
which is equivalent to \eqref{lem:prox}. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Descent lemma for $L_{f}$‑smooth $f$]\label{lem:descent}
For $L_{f}$‑smooth $f$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L_{f}}{2}\|v-u\|^{2},
\qquad\forall u,v.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard consequence of the Lipschitz continuity of $\nabla f$. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:ergodic}. Throughout the proof we fix an arbitrary saddle point $(x^{\star},y^{\star})$.

\noindent\textbf{Step 1.} \emph{Bounding the dual update.}  
Apply Lemma \ref{lem:prox} to the dual step $y^{k+1}=\prox_{\sigma g^{*}}(y^{k}+\sigma K\tilde{x}^{k})$ with $v=y^{\star}$:
\[
\langle y^{k}+ \sigma K\tilde{x}^{k} - y^{k+1},\, y^{\star}-y^{k+1}\rangle
\ge \sigma\bigl(g^{*}(y^{\star})-g^{*}(y^{k+1})\bigr).
\tag{9}
\]
Rearranging,
\[
\langle K\tilde{x}^{k},y^{\star}-y^{k+1}\rangle
\ge \frac{1}{\sigma}\langle y^{k+1}-y^{k},y^{\star}-y^{k+1}\rangle
+ g^{*}(y^{\star})-g^{*}(y^{k+1}).
\tag{10}
\]

\noindent\textbf{Step 2.} \emph{Bounding the primal update.}  
Apply Lemma \ref{lem:prox} to the primal step $x^{k+1}=\prox_{\tau f}(x^{k}-\tau K^{\top}y^{k+1})$ with $v=x^{\star}$:
\[
\langle x^{k}-\tau K^{\top}y^{k+1}-x^{k+1},\,x^{\star}-x^{k+1}\rangle
\ge \tau\bigl(f(x^{\star})-f(x^{k+1})\bigr).
\tag{11}
\]
Equivalently,
\[
\langle -K^{\top}y^{k+1},x^{\star}-x^{k+1}\rangle
\ge \frac{1}{\tau}\langle x^{k+1}-x^{k},x^{\star}-x^{k+1}\rangle
+ f(x^{\star})-f(x^{k+1}).
\tag{12}
\]

\noindent\textbf{Step 3.} \emph{Combine the two inequalities.}  
Add \eqref{10} and \eqref{12} and use $\langle K\tilde{x}^{k},y^{\star}-y^{k+1}\rangle
= \langle \tilde{x}^{k},K^{\top}(y^{\star}-y^{k+1})\rangle$:
\[
\begin{aligned}
&\langle \tilde{x}^{k},K^{\top}(y^{\star}-y^{k+1})\rangle
-\langle K^{\top}y^{k+1},x^{\star}-x^{k+1}\rangle  \\
&\qquad\ge 
\frac{1}{\sigma}\langle y^{k+1}-y^{k},y^{\star}-y^{k+1}\rangle
+\frac{1}{\tau}\langle x^{k+1}-x^{k},x^{\star}-x^{k+1}\rangle
+ g^{*}(y^{\star})-g^{*}(y^{k+1})
+ f(x^{\star})-f(x^{k+1}).
\end{aligned}
\tag{13}
\]

\noindent\textbf{Step 4.} \emph{Rewrite the left‑hand side.}  
Notice that
\[
\begin{aligned}
\langle \tilde{x}^{k},K^{\top}(y^{\star}-y^{k+1})\rangle
-\langle K^{\top}y^{k+1},x^{\star}-x^{k+1}\rangle
&= \langle K\tilde{x}^{k},y^{\star}-y^{k+1}\rangle
+\langle Kx^{k+1},y^{k+1}-y^{\star}\rangle\\
&= \langle K(\tilde{x}^{k}-x^{k+1}),y^{\star}-y^{k+1}\rangle.
\end{aligned}
\tag{14}
\]

\noindent\textbf{Step 5.} \emph{Introduce the extrapolation.}  
Recall $\tilde{x}^{k}=x^{k}+\theta (x^{k}-x^{k-1})$.  For $\theta=1$ (the standard PDHG) we have  
$\tilde{x}^{k}=2x^{k}-x^{k-1}$ and consequently  

\[
\tilde{x}^{k}-x^{k+1}= (x^{k}-x^{k+1})+(x^{k}-x^{k-1}).
\tag{15}
\]

\noindent\textbf{Step 6.} \emph{Bound the coupling term using Cauchy–Schwarz and Young’s inequality.}  
For any $\alpha>0$,
\[
\langle K(\tilde{x}^{k}-x^{k+1}),y^{\star}-y^{k+1}\rangle
\le \frac{\alpha}{2}\|\tilde{x}^{k}-x^{k+1}\|^{2}
+\frac{1}{2\alpha}\|K^{\top}(y^{\star}-y^{k+1})\|^{2}.
\tag{16}
\]

Choosing $\alpha = \frac{1}{\tau}$ and using $\|K^{\top}v\|\le\|K\|\|v\|$ yields  

\[
\langle K(\tilde{x}^{k}-x^{k+1}),y^{\star}-y^{k+1}\rangle
\le \frac{1}{2\tau}\|\tilde{x}^{k}-x^{k+1}\|^{2}
+\frac{\|K\|^{2}\tau}{2}\|y^{\star}-y^{k+1}\|^{2}.
\tag{17}
\]

\noindent\textbf{Step 7.} \emph{Insert (17) into (13).}  After rearranging terms we obtain a one‑step inequality for the Lyapunov function $V^{k}$:

\[
\begin{aligned}
V^{k+1} - V^{k}
&\le -\frac{1}{2\tau}\|x^{k+1}-x^{k}\|^{2}
-\frac{1}{2\sigma}\|y^{k+1}-y^{k}\|^{2}
+\frac{\|K\|^{2}\tau}{2}\|y^{k+1}-y^{\star}\|^{2}
\\
&\qquad +\frac{\|K\|^{2}\sigma}{2}\|x^{k+1}-x^{\star}\|^{2}
\;-\;\bigl(\mathcal{L}(x^{k+1},y^{\star})-\mathcal{L}(x^{\star},y^{k+1})\bigr).
\end{aligned}
\tag{18}
\]

\noindent\textbf{Step 8.} \emph{Use the stepsize condition.}  
Assumption \ref{as:steps} with $\theta=1$ gives $\tau\sigma\|K\|^{2}<1$, i.e.

\[
\frac{\|K\|^{2}\tau}{2}\le\frac{1}{2\sigma}\bigl(1-\tau\sigma\|K\|^{2}\bigr),\qquad
\frac{\|K\|^{2}\sigma}{2}\le\frac{1}{2\tau}\bigl(1-\tau\sigma\|K\|^{2}\bigr).
\tag{19}
\]

Plugging \eqref{19} into \eqref{18} cancels the positive terms and yields

\[
V^{k+1} - V^{k}
\le -\bigl(\mathcal{L}(x^{k+1},y^{\star})-\mathcal{L}(x^{\star},y^{k+1})\bigr).
\tag{20}
\]

\noindent\textbf{Step 9.} \emph{Telescoping sum.}  
Summing \eqref{20} from $k=0$ to $K-1$ gives

\[
\sum_{k=0}^{K-1}\bigl(\mathcal{L}(x^{k+1},y^{\star})-\mathcal{L}(x^{\star},y^{k+1})\bigr)
\le V^{0}-V^{K}
\le V^{0},
\tag{21}
\]
because $V^{K}\ge0$.

\noindent\textbf{Step 10.} \emph{Pass to ergodic averages.}  
Divide \eqref{21} by $K$ and use convex‑concave structure of $\mathcal{L}$:

\[
\mathcal{L}(\bar{x}^{K},y^{\star})-\mathcal{L}(x^{\star},\bar{y}^{K})
\le\frac{V^{0}}{K}
= \frac{1}{2K}\Bigl(\frac{\|x^{0}-x^{\star}\|^{2}}{\tau}
+\frac{\|y^{0}-y^{\star}\|^{2}}{\sigma}\Bigr).
\tag{22}
\]

The left‑hand side is exactly the primal–dual gap $\gap(\bar{x}^{K},\bar{y}^{K})$, which proves \eqref{6}. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
From \eqref{22} we read off the explicit constant  

\[
C:=\frac12\Bigl(\frac{\|x^{0}-x^{\star}\|^{2}}{\tau}
+\frac{\|y^{0}-y^{\star}\|^{2}}{\sigma}\Bigr),
\]
so that  

\[
\gap(\bar{x}^{K},\bar{y}^{K})\le \frac{C}{K},\qquad K\ge1.
\tag{23}
\]

If the optimal primal solution $x^{\star}$ is known to lie in a bounded set of diameter $D_{x}$ and similarly $\|y^{0}-y^{\star}\|\le D_{y}$, then  

\[
C\le \frac12\Bigl(\frac{D_{x}^{2}}{\tau}+\frac{D_{y}^{2}}{\sigma}\Bigr).
\tag{24}
\]

Thus the rate can be made tighter by scaling $\tau$ and $\sigma$ proportionally to $D_{x}$ and $D_{y}$ while respecting \eqref{as:steps}.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No dual regularizer ($g^{*}=0$).} The dual proximal reduces to an explicit gradient ascent step, i.e. $y^{k+1}=y^{k}+\sigma K\tilde{x}^{k}$. The same proof yields the $O(1/K)$ bound with $g^{*}\equiv0$.
    \item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, one can replace the Lyapunov function by a weighted sum and obtain a linear rate $O((1-\sqrt{\mu\tau})^{K})$ (see e.g. Chambolle–Pock 2011). The present analysis recovers the $O(1/K)$ bound as a special case when $\mu=0$.
    \item \textbf{Choice $\theta=0$.} The extrapolation disappears and the stepsize condition simplifies to $\tau\sigma\|K\|^{2}\le1$. The same proof goes through with $\tilde{x}^{k}=x^{k}$, yielding the identical $O(1/K)$ bound but often with slower practical performance.
\end{itemize}
