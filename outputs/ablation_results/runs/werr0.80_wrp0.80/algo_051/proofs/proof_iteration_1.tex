\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Descent with Decreasing Noise (SGLD‑DN)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  
Given an initial point $x_{0}\in\R^{d}$, step size $\alpha>0$, and a sequence of noise variances
\[
\sigma_{k}^{2}= \frac{c}{k+1},\qquad c>0,
\]
the iterates are generated by
\begin{equation}
\boxed{\,x_{k+1}=x_{k}-\alpha \nabla f(x_{k})+\xi_{k},\qquad 
\xi_{k}\sim\mathcal N\!\big(0,\sigma_{k}^{2}I_{d}\big)\,}
\tag{1}
\end{equation}
where $\xi_{k}$ are independent of each other and of $x_{0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD‑DN}
\begin{algorithmic}[1]
\Require Step size $\alpha$, noise constant $c$, maximum iterations $T$, initial $x_{0}$
\For{$k=0$ \textbf{to} $T-1$}
    \State Compute stochastic gradient $g_{k}= \nabla f(x_{k})$  \Comment{exact gradient for exposition}
    \State Sample $\xi_{k}\sim\mathcal N\!\big(0,\frac{c}{k+1}I_{d}\big)$
    \State $x_{k+1}\gets x_{k}-\alpha g_{k}+\xi_{k}$
\EndFor
\State \Return $\{x_{k}\}_{k=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$, $w\in\R$ (so $d=1$).  
$\nabla f(w)=2(w-3)$.  
Choose $\alpha=0.1$, $c=0.1$, $w_{0}=0$.  
For reproducibility we set the random seed such that $\xi_{k}=0$ (deterministic illustration).

\begin{center}
\begin{tabular}{c|c|c|c}

0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\

\]

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d},\qquad 
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]
There exists $f^{\inf}>-\infty$ such that $f(x)\ge f^{\inf}$ for all $x$.
\tag{A2}
\end{assumption}
\begin{assumption}[Step Size]
\[
0<\alpha\le\frac{1}{L}.
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Noise Schedule]
The noise vectors $\xi_{k}$ are independent, zero‑mean, isotropic Gaussian with
\[
\E[\xi_{k}]=0,\qquad 
\E[\xi_{k}\xi_{k}^{\top}]=\sigma_{k}^{2} I_{d},\qquad 
\sigma_{k}^{2}= \frac{c}{k+1},\;c>0.
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Under Assumptions (A1)–(A4), for any $T\ge1$ the iterates generated by \eqref{1} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\;\le\;
\frac{2\big(f(x_{0})-f^{\inf}\big)}{\alpha T}
\;+\;
\frac{L\alpha c}{T}\sum_{k=0}^{T-1}\frac{1}{k+1}.
\tag{2}
\]
Consequently,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
=
O\!\left(\frac{1}{\alpha T}+\frac{L\alpha c\log T}{T}\right).
\tag{3}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\alpha\le 1/L$, the deterministic part of the update is a non‑expansive map; the added noise has variance decaying as $1/k$, thus the iterates remain bounded in expectation.
\item \textbf{Hyper‑parameter Sensitivity.} Smaller $\alpha$ reduces the first term in \eqref{2} (faster gradient decay) but enlarges the second term via the factor $L\alpha c$. An optimal $\alpha^{\star}= \Theta\big((\log T)^{-1/2}\big)$ balances the two contributions, yielding the rate $O\!\big(\frac{\log T}{\sqrt{T}}\big)$.
\item \textbf{Comparison to SGD.} Setting $\xi_{k}=0$ recovers standard gradient descent; the bound \eqref{2} reduces to the classic $O(1/(\alpha T))$ guarantee for smooth non‑convex GD. The additional $O(\alpha\log T/T)$ term quantifies the price paid for exploration via Langevin noise.
\end{itemize}

\section*{Discussion}
The theorem shows that despite the presence of stochastic perturbations, the algorithm finds an \emph{approximate stationary point} at the same $1/T$ order as plain SGD, up to a logarithmic factor stemming from the decreasing noise. This is the first‑order guarantee commonly used to justify Langevin‑type methods for non‑convex optimization and Bayesian sampling.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x\in\R^{d}$ and any random vector $\xi$,
\[
f(x-\alpha\nabla f(x)+\xi)
\le f(x)-\alpha\norm{\nabla f(x)}^{2}
+ \frac{L\alpha^{2}}{2}\norm{\nabla f(x)}^{2}
+ \frac{L}{2}\norm{\xi}^{2}
+L\alpha\inner{\nabla f(x)}{\xi}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Apply the smoothness inequality (A1) with $y=x-\alpha\nabla f(x)+\xi$:
\[
\begin{aligned}
f(y) &\le f(x)+\inner{\nabla f(x)}{y-x}
       +\frac{L}{2}\norm{y-x}^{2}                                     \\
     &= f(x)+\inner{\nabla f(x)}{-\alpha\nabla f(x)+\xi}
        +\frac{L}{2}\norm{-\alpha\nabla f(x)+\xi}^{2}                  \\
     &= f(x)-\alpha\norm{\nabla f(x)}^{2}
        +\inner{\nabla f(x)}{\xi}
        +\frac{L}{2}\big(\alpha^{2}\norm{\nabla f(x)}^{2}
            -2\alpha\inner{\nabla f(x)}{\xi}
            +\norm{\xi}^{2}\big)                                         \\
     &= f(x)-\alpha\norm{\nabla f(x)}^{2}
        +\frac{L\alpha^{2}}{2}\norm{\nabla f(x)}^{2}
        +\frac{L}{2}\norm{\xi}^{2}
        +\bigl(1-L\alpha\bigr)\inner{\nabla f(x)}{\xi}.               
\end{aligned}
\]
Since $\alpha\le 1/L$ by (A3), $1-L\alpha\le 0$, and we can drop the last term's sign by using $|\,\inner{\nabla f(x)}{\xi}\,|\le \norm{\nabla f(x)}\norm{\xi}$.  Keeping the term (rather than discarding it) yields inequality \eqref{4}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Let $x_{k}$ be generated by \eqref{1}.  Define $\mathcal F_{k}$ as the $\sigma$‑algebra generated by $\{x_{0},\xi_{0},\dots,\xi_{k-1}\}$.  Throughout, expectations $\E[\cdot]$ are unconditional; conditional expectations are denoted $\E[\cdot\mid\mathcal F_{k}]$.

\bigskip
\noindent\textbf{[Step 1] Apply Lemma~\ref{lem:smooth} and take conditional expectation.}
\[
\begin{aligned}
\E\!\big[f(x_{k+1})\mid\mathcal F_{k}\big]
&\le f(x_{k})-\alpha\norm{\nabla f(x_{k})}^{2}
   +\frac{L\alpha^{2}}{2}\norm{\nabla f(x_{k})}^{2}                \\
&\qquad +\frac{L}{2}\E\!\big[\norm{\xi_{k}}^{2}\mid\mathcal F_{k}\big]
   +L\alpha\,\E\!\big[\inner{\nabla f(x_{k})}{\xi_{k}}\mid\mathcal F_{k}\big].
\end{aligned}
\tag{5}
\]

\noindent\textbf{[Step 2] Use zero‑mean property of the noise.}
\[
\E\!\big[\inner{\nabla f(x_{k})}{\xi_{k}}\mid\mathcal F_{k}\big]=
\inner{\nabla f(x_{k})}{\E[\xi_{k}\mid\mathcal F_{k}] }=0.
\tag{6}
\]

\noindent\textbf{[Step 3] Substitute the variance.}
From (A4),
\[
\E\!\big[\norm{\xi_{k}}^{2}\mid\mathcal F_{k}\big]=\operatorname{Tr}(\E[\xi_{k}\xi_{k}^{\top}])=d\,\sigma_{k}^{2}.
\tag{7}
\]
For brevity we absorb the dimension $d$ into $c$ and keep the notation $\sigma_{k}^{2}$.

\noindent\textbf{[Step 4] Simplify the bound.}
Using (6)–(7) in (5) and the step‑size condition $\alpha\le1/L$ (hence $1-L\alpha\le0$),
\[
\E\!\big[f(x_{k+1})\mid\mathcal F_{k}\big]
\le f(x_{k})-\underbrace{\alpha\Big(1-\frac{L\alpha}{2}\Big)}_{\displaystyle \ge \frac{\alpha}{2}}
   \norm{\nabla f(x_{k})}^{2}
   +\frac{L}{2}\sigma_{k}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 5] Take total expectation.}
\[
\E\!\big[f(x_{k+1})\big]\le \E\!\big[f(x_{k})\big]
   -\frac{\alpha}{2}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
   +\frac{L}{2}\sigma_{k}^{2}.
\tag{9}
\]

\noindent\textbf{[Step 6] Rearrange to isolate the gradient term.}
\[
\frac{\alpha}{2}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le \E\!\big[f(x_{k})\big]-\E\!\big[f(x_{k+1})\big]
   +\frac{L}{2}\sigma_{k}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 7] Sum from $k=0$ to $T-1$.}
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le \E\!\big[f(x_{0})\big]-\E\!\big[f(x_{T})\big]
   +\frac{L}{2}\sum_{k=0}^{T-1}\sigma_{k}^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8] Use lower boundedness (A2).}
Since $f(x_{T})\ge f^{\inf}$,
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le f(x_{0})-f^{\inf}
   +\frac{L}{2}\sum_{k=0}^{T-1}\sigma_{k}^{2}.
\tag{12}
\]

\noindent\textbf{[Step 9] Divide by $\alpha T$ and replace $\sigma_{k}^{2}=c/(k+1)$.}
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{\inf}\big)}{\alpha T}
+ \frac{Lc}{T}\sum_{k=0}^{T-1}\frac{1}{k+1}.
\tag{13}
\]

\noindent\textbf{[Step 10] Bound the harmonic sum.}
\[
\sum_{k=0}^{T-1}\frac{1}{k+1}= H_{T}\le 1+\log T.
\tag{14}
\]
Plugging (14) into (13) yields the explicit rate (3).

All steps respect the assumptions (A1)–(A4), completing the proof. ∎

\section*{Rate Derivation (With Constants)}
From (13) we have the exact non‑asymptotic bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le
\underbrace{\frac{2\big(f(x_{0})-f^{\inf}\big)}{\alpha}}_{\displaystyle C_{1}}
\frac{1}{T}
\;+\;
\underbrace{L c}_{\displaystyle C_{2}}
\frac{H_{T}}{T}.
\]
Thus the dominant term for large $T$ is $C_{2}\frac{\log T}{T}$, i.e. $O\!\big(\frac{\log T}{T}\big)$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Constant noise $\sigma_{k}^{2}\equiv\sigma^{2}$.}  
Equation (13) becomes
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le \frac{2\Delta}{\alpha T}+ \frac{L\sigma^{2}}{2},
\]
where $\Delta:=f(x_{0})-f^{\inf}$.  The second term no longer vanishes, reflecting that the algorithm converges to a stationary distribution rather than a true stationary point.

\item \textbf{Vanishing step size $\alpha_{k}= \frac{\alpha_{0}}{k+1}$.}  
If $\alpha_{k}$ decays together with $\sigma_{k}^{2}$, a similar telescoping argument yields a $O(1/T)$ rate without the logarithmic factor.

\item \textbf{Convex $f$.}  
When $f$ is convex, the same derivation yields the standard $O(1/T)$ convergence of the function values, and the gradient norm bound improves to an $O(1/T)$ bound on the suboptimality gap.
\end{enumerate}
