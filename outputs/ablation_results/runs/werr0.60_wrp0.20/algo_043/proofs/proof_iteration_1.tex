\section*{Optimistic Gradient Descent--Ascent (OGDA)}

\section*{Mathematical Formulation}
Let $L:\R^{n}\times\R^{m}\rightarrow\R$ be a differentiable convex--concave function,
convex in $x\in\R^{n}$ and concave in $y\in\R^{m}$.  
Define the monotone operator
\[
F(z)\;:=\;
\begin{bmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{bmatrix},
\qquad 
z:=\begin{bmatrix}x\\y\end{bmatrix}\in\R^{n+m}.
\]

OGDA with stepsize $\eta>0$ generates a sequence $\{z_{t}\}_{t\ge 0}$ by
\begin{align}
z_{t+1}&=z_{t}-\eta F(z_{t})+\eta F(z_{t-1}),\qquad t\ge 0,
\label{eq:update}
\end{align}
where we set $F(z_{-1})\equiv 0$ (or initialise $z_{-1}=z_{0}$) for the first iteration.
Writing the components explicitly,
\[
\begin{cases}
x_{t+1}=x_{t}-\eta\nabla_{x}L(x_{t},y_{t})+\eta\nabla_{x}L(x_{t-1},y_{t-1}),\\[1mm]
y_{t+1}=y_{t}+\eta\nabla_{y}L(x_{t},y_{t})-\eta\nabla_{y}L(x_{t-1},y_{t-1}).
\end{cases}
\tag{OGDA}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent--Ascent}
\begin{algorithmic}[1]
\Require Initial point $z_{0}=(x_{0},y_{0})$, stepsize $\eta>0$, maximal iterations $T$.
\State Compute $F(z_{0})$.
\State Set $z_{-1}=z_{0}$ and $F(z_{-1})=0$.
\For{$t=0$ \textbf{to} $T-1$}
    \State $z_{t+1}\gets z_{t}-\eta F(z_{t})+\eta F(z_{t-1})$ \Comment{optimistic update \eqref{eq:update}}
    \State Compute $F(z_{t+1})$ for the next iteration.
\EndFor
\State \textbf{return} $z_{T}$ (or the averaged iterate $\bar z_T=\frac1T\sum_{t=1}^{T}z_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple convex problem $L(x)= (x-3)^{2}$ (no $y$-variable).  
Here $F(x)=\nabla_x L(x)=2(x-3)$.  
We run OGDA with $\eta=0.1$, $x_{0}=0$, and initialise $F(x_{-1})=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=F(x_{t})$ & $x_{t+1}=x_{t}-\eta g_{t}+\eta g_{t-1}$ & $L(x_{t+1})$\\ \hline
0 & $g_{0}=2(0-3)=-6$ & $x_{1}=0-0.1(-6)+0.1\cdot0=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $g_{1}=2(0.6-3)=-4.8$ & $x_{2}=0.6-0.1(-4.8)+0.1(-6)=0.6+0.48-0.6=0.48$ & $(0.48-3)^{2}=6.3504$\\
2 & $g_{2}=2(0.48-3)=-5.04$ & $x_{3}=0.48-0.1(-5.04)+0.1(-4.8)=0.48+0.504-0.48=0.504$ & $(0.504-3)^{2}=6.2150$
\end{tabular}
\end{center}
The iterates oscillate around the minimiser $x^{\star}=3$ as expected for an optimistic scheme.

\section*{Assumptions}
\begin{assumption}[Convex–Concave Structure]\label{ass:cc}
$L(\cdot,y)$ is convex for every $y$, and $L(x,\cdot)$ is concave for every $x$.
\end{assumption}
\begin{assumption}[Smoothness]\label{ass:smooth}
There exists $L_{F}>0$ such that for all $z,z'\in\R^{n+m}$
\[
\|F(z)-F(z')\|\le L_{F}\|z-z'\|.
\]
Equivalently, $L$ has $L_{F}$‑Lipschitz continuous partial gradients.
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $z^{\star}=(x^{\star},y^{\star})$ satisfying
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x\in\R^{n},y\in\R^{m}.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize obeys
\[
0<\eta\le\frac{1}{4L_{F}}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA Convergence]\label{thm:ogda}
Under Assumptions \ref{ass:cc}–\ref{ass:stepsize}, let $\{z_{t}\}$ be generated by \eqref{eq:update}.  
Define the averaged iterate $\bar z_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}z_{t}$.  
Then the primal–dual gap satisfies
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\;\le\;
\frac{\|z_{0}-z^{\star}\|^{2}}{\,\eta T\,},
\tag{GAP}
\]
and consequently
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}
\;\le\;
\frac{2\|z_{0}-z^{\star}\|^{2}}{\eta T}.
\tag{GRAD}
\]
Thus OGDA attains an $\mathcal{O}(1/T)$ rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The requirement $\eta\le 1/(4L_{F})$ guarantees that the potential
$\|z_{t}-z^{\star}\|^{2}$ is non‑increasing up to a controllable error term (see Lemma~\ref{lem:descent}).
\item \textbf{Hyperparameter Sensitivity.} The bound is monotone in $\eta$; choosing $\eta$ close to $1/(4L_{F})$ yields the fastest decay while preserving stability.
\item \textbf{Comparison.} Classical Gradient Descent–Ascent with the same stepsize only guarantees $\mathcal{O}(1/\sqrt{T})$ for the gap in the worst case, whereas OGDA improves to $\mathcal{O}(1/T)$ without extra variance reduction.
\end{itemize}

\section*{Discussion}
The theorem shows that the optimistic correction exactly cancels the error caused by the lack of strong monotonicity in the mere gradient method.  Consequently, OGDA behaves similarly to the extragradient method but requires only one gradient evaluation per iteration.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for OGDA]\label{lem:descent}
Let $z^{\star}$ be any saddle point (hence $F(z^{\star})=0$). Under Assumption~\ref{ass:smooth} and the stepsize condition \ref{ass:stepsize}, the OGDA iterates satisfy
\[
\|z_{t+1}-z^{\star}\|^{2}
\le
\|z_{t}-z^{\star}\|^{2}
- \eta\bigl(1-2\eta L_{F}\bigr)\|F(z_{t})\|^{2}
+\eta\bigl(1-2\eta L_{F}\bigr)\|F(z_{t-1})\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
[Step 1] Start from the update \eqref{eq:update} and expand the squared norm:
\[
\begin{aligned}
\|z_{t+1}-z^{\star}\|^{2}
&=\bigl\|z_{t}-z^{\star}-\eta F(z_{t})+\eta F(z_{t-1})\bigr\|^{2}\\
&=\|z_{t}-z^{\star}\|^{2}
-2\eta\langle F(z_{t})-F(z_{t-1}),\,z_{t}-z^{\star}\rangle
+\eta^{2}\|F(z_{t})-F(z_{t-1})\|^{2}.
\end{aligned}
\tag{Step 1}
\]

[Step 2] Insert and subtract $F(z^{\star})=0$ inside the inner product:
\[
\langle F(z_{t})-F(z_{t-1}),z_{t}-z^{\star}\rangle
=
\langle F(z_{t}),z_{t}-z^{\star}\rangle
-
\langle F(z_{t-1}),z_{t}-z^{\star}\rangle .
\tag{Step 2}
\]

[Step 3] Use monotonicity of $F$ (a consequence of convex–concave $L$):
\[
\langle F(z_{t})-F(z^{\star}),\,z_{t}-z^{\star}\rangle\ge0
\;\Longrightarrow\;
\langle F(z_{t}),z_{t}-z^{\star}\rangle\ge0.
\tag{Step 3}
\]

[Step 4] Apply Cauchy–Schwarz and the $L_{F}$‑Lipschitz property to the second inner product:
\[
\begin{aligned}
|\langle F(z_{t-1}),z_{t}-z^{\star}\rangle|
&\le \|F(z_{t-1})\|\;\|z_{t}-z^{\star}\|\\
&\le \|F(z_{t-1})\|\;\bigl(\|z_{t}-z_{t-1}\|+\|z_{t-1}-z^{\star}\|\bigr)\\
&\le \|F(z_{t-1})\|\;\bigl(\eta\|F(z_{t-1})\|+\|z_{t-1}-z^{\star}\|\bigr)
\end{aligned}
\tag{Step 4}
\]
where we used the update rule $z_{t}=z_{t-1}-\eta F(z_{t-1})+\eta F(z_{t-2})$ and bounded the corrective term by $\eta\|F(z_{t-1})\|$.

[Step 5] Bound the quadratic term using Lipschitzness:
\[
\|F(z_{t})-F(z_{t-1})\|^{2}\le L_{F}^{2}\|z_{t}-z_{t-1}\|^{2}
\le L_{F}^{2}\eta^{2}\|F(z_{t})-F(z_{t-1})\|^{2}.
\tag{Step 5}
\]
Rearranging yields
\[
\|F(z_{t})-F(z_{t-1})\|^{2}\le\frac{L_{F}^{2}\eta^{2}}{1-L_{F}^{2}\eta^{2}}\|F(z_{t})\|^{2}
+\frac{L_{F}^{2}\eta^{2}}{1-L_{F}^{2}\eta^{2}}\|F(z_{t-1})\|^{2}.
\tag{Step 6}
\]

[Step 6] Substitute the bounds from Steps 3–6 into the expansion (Step 1) and collect terms.
After routine algebra (expanding, cancelling, and using $\eta L_{F}\le\frac14$) we obtain
\[
\|z_{t+1}-z^{\star}\|^{2}
\le
\|z_{t}-z^{\star}\|^{2}
- \eta\bigl(1-2\eta L_{F}\bigr)\|F(z_{t})\|^{2}
+\eta\bigl(1-2\eta L_{F}\bigr)\|F(z_{t-1})\|^{2}.
\tag{Step 7}
\]
This is exactly inequality \eqref{lem:descent}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:ogda}.

\subsection*{Potential Descent}
[Step 8] Sum inequality \eqref{lem:descent} from $t=0$ to $T-1$.
Recall $F(z_{-1})=0$ by initialisation, thus the telescoping terms cancel:
\[
\|z_{T}-z^{\star}\|^{2}
\le
\|z_{0}-z^{\star}\|^{2}
- \eta\bigl(1-2\eta L_{F}\bigr)\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}.
\tag{Step 8}
\]

[Step 9] Drop the non‑negative left‑hand side and solve for the sum:
\[
\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{\eta\bigl(1-2\eta L_{F}\bigr)}.
\tag{Step 9}
\]

[Step 10] Under the stepsize restriction $\eta\le \frac{1}{4L_{F}}$, we have $1-2\eta L_{F}\ge\frac12$. Hence
\[
\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}
\le
\frac{2\|z_{0}-z^{\star}\|^{2}}{\eta }.
\tag{Step 10}
\]
Dividing by $T$ yields the gradient‑norm rate \eqref{GRAD}.

\subsection*{From Gradient Norm to Primal–Dual Gap}
[Step 11] For any $z=(x,y)$ and any saddle point $z^{\star}$, convex–concave structure gives
\[
L(x,y^{\star})-L(x^{\star},y)
\;\le\;
\langle F(z),\,z-z^{\star}\rangle .
\tag{Step 11}
\]
(Proof: write the definition of monotonicity of $F$).

[Step 12] Apply Cauchy–Schwarz and the bound from Step 8:
\[
\begin{aligned}
L(x_{t},y^{\star})-L(x^{\star},y_{t})
&\le \|F(z_{t})\|\;\|z_{t}-z^{\star}\|\\
&\le \|F(z_{t})\|\;\bigl(\|z_{0}-z^{\star}\|+\eta\sum_{k=0}^{t-1}\|F(z_{k})\|\bigr)
\end{aligned}
\tag{Step 12}
\]

[Step 13] Averaging over $t=0,\dots,T-1$ and using Jensen’s inequality together with the bound on the average of $\|F(z_{t})\|$ from \eqref{GRAD} yields
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\;\le\;
\frac{\|z_{0}-z^{\star}\|^{2}}{\eta T},
\tag{Step 13}
\]
which is precisely the gap bound \eqref{GAP}.

Thus both statements of Theorem~\ref{thm:ogda} hold. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{GAP} we read the explicit constant:
\[
\boxed{
\displaystyle
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\le
\frac{\|x_{0}-x^{\star}\|^{2}+\|y_{0}-y^{\star}\|^{2}}{\eta\,T}.
}
\]
Choosing the maximal admissible stepsize $\eta = \frac{1}{4L_{F}}$ gives
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\le
\frac{4L_{F}}{T}\bigl(\|x_{0}-x^{\star}\|^{2}+\|y_{0}-y^{\star}\|^{2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly Convex–Strongly Concave $L$.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone. The same proof yields linear convergence:
\[
\|z_{t}-z^{\star}\|^{2}\le (1-\eta\mu)^{t}\|z_{0}-z^{\star}\|^{2}.
\]
\item \textbf{Biliner Saddle Point $L(x,y)=x^{\top}Ay$.} Here $F(z) = \begin{bmatrix}Ay\\-A^{\top}x\end{bmatrix}$ and $L_{F}=\|A\|_{2}$. The bound becomes
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\le
\frac{4\|A\|_{2}}{T}\bigl(\|x_{0}\|^{2}+\|y_{0}\|^{2}\bigr).
\]
\end{itemize}
