\section*{Probabilistic Coordinate Descent (PCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
For each coordinate $i\in\{1,\dots,d\}$ we denote by $L_i>0$ the \emph{coordinate‑wise smoothness constant}:
\begin{equation}
\label{eq:smooth}
\bigl|\partial_i f(\mathbf w+\alpha \e_i)-\partial_i f(\mathbf w)\bigr|\le L_i |\alpha|\quad\forall \mathbf w\in\R^{d},\;\forall\alpha\in\R,
\end{equation}
where $\e_i$ is the $i$‑th canonical basis vector.

At iteration $t$ a coordinate $i_t$ is drawn independently from a fixed distribution
\[
\mathbb{P}(i_t=i)=p_i,\qquad p_i>0,\;\sum_{i=1}^{d}p_i=1 .
\]

The update uses an \emph{unbiased scaled gradient estimator}
\[
\mathbf g_t\;:=\;\frac{1}{p_{i_t}}\; \partial_{i_t}f(\mathbf w_t)\;\e_{i_t},
\qquad\text{so that}\qquad 
\E[\mathbf g_t\mid\mathbf w_t]=\nabla f(\mathbf w_t).
\]

With a stepsize $\eta>0$ the iterate is updated by
\begin{equation}
\label{eq:update}
\boxed{\;\mathbf w_{t+1}= \mathbf w_t-\eta\,\mathbf g_t
      =\mathbf w_t-\frac{\eta}{p_{i_t}}\;\partial_{i_t}f(\mathbf w_t)\,\e_{i_t}\;}
\end{equation}
Only the selected coordinate $i_t$ changes.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent}
\begin{algorithmic}[1]
\Require Initial point $\mathbf w_0\in\R^{d}$, stepsize $\eta>0$, probabilities $\{p_i\}_{i=1}^{d}$.
\For{$t=0,1,\dots,T-1$}
    \State Sample $i_t\sim\{1,\dots,d\}$ with $\Pr(i_t=i)=p_i$.
    \State Compute the partial derivative $g_{t}:=\partial_{i_t}f(\mathbf w_t)$.
    \State Update the selected coordinate
    \[
        \mathbf w_{t+1}\gets\mathbf w_t-\frac{\eta}{p_{i_t}}\,g_{t}\,\e_{i_t}.
    \]
\EndFor
\State \Return $\mathbf w_T$ (or an average of the iterates).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem
\[
f(w)=(w-3)^2,\qquad w\in\R .
\]
Here $d=1$, $p_1=1$, $L_1=2$ (since $f''(w)=2$).  
Choose $\eta=0.1$ and $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\partial_1 f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\\hline
0 & $2(w_0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3617$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm.

\section*{Assumptions}
\begin{assumption}[Convexity]
$f$ is convex, i.e. for all $\mathbf u,\mathbf v\in\R^{d}$,
\[
f(\mathbf v)\ge f(\mathbf u)+\ip{\nabla f(\mathbf u)}{\mathbf v-\mathbf u}.
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that \eqref{eq:smooth} holds.
\end{assumption}
\begin{assumption}[Stepsize]
The global stepsize satisfies
\[
0<\eta\le \frac{1}{\displaystyle\max_{i\in[d]}L_i}.
\]
\end{assumption}
\begin{assumption}[Probability distribution]
$p_i>0$ for all $i$ and $\sum_{i=1}^{d}p_i=1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:conv}
Let $\mathbf w^\star\in\arg\min_{\mathbf w\in\R^{d}}f(\mathbf w)$.  
Under Assumptions 1–4 and using the update \eqref{eq:update} with stepsize $\eta$ satisfying Assumption 3, the iterates satisfy for any $T\ge1$
\[
\boxed{\;
\E\!\bigl[f(\mathbf w_T)-f(\mathbf w^\star)\bigr]
\;\le\;
\frac{\norm{\mathbf w_0-\mathbf w^\star}^{2}}{2\eta\,T}\; } .
\]
Consequently, the expected suboptimality decays as $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Unbiasedness.} By construction $\E[\mathbf g_t\mid\mathbf w_t]=\nabla f(\mathbf w_t)$. This yields a direct analogue of the standard SGD analysis.
\item \textbf{Stability w.r.t.\ $p_i$.} The bound in Theorem~\ref{thm:conv} does not contain $p_i$ explicitly; the only place $p_i$ appears is in the definition of $\mathbf g_t$. Hence any strictly positive distribution yields the same $O(1/T)$ rate.
\item \textbf{Comparison.} Classical (uniform) coordinate descent obtains the same $O(1/T)$ rate but with stepsize $\eta\le 1/\max_i L_i$. Our result shows that non‑uniform sampling incurs no penalty, provided the scaling $1/p_i$ is used.
\item \textbf{Hyperparameter sensitivity.} The only tunable scalar is $\eta$. The admissible interval is $ (0,1/\max_i L_i]$, identical to the uniform case.
\end{itemize}

\section*{Discussion}
The proof follows the standard potential‑function argument for stochastic gradient methods, with the crucial observation that the scaling $1/p_i$ exactly cancels the sampling probability in expectation. This yields an unbiased estimator of the full gradient and allows the same descent lemma as in full‑gradient gradient descent. The coordinate‑wise smoothness constants $L_i$ replace the global Lipschitz constant $L$ in the descent inequality, which is why the stepsize restriction involves $\max_i L_i$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for coordinate‑wise smoothness]
\label{lem:descent}
If $f$ satisfies \eqref{eq:smooth} then for any $\mathbf w\in\R^{d}$, any coordinate $i$, and any scalar $\alpha$,
\[
f(\mathbf w+\alpha \e_i)\le f(\mathbf w)+\alpha\,\partial_i f(\mathbf w)+\frac{L_i}{2}\alpha^{2}.
\]
\end{lemma}
\begin{proof}
Consider the univariate function $\phi(\alpha)=f(\mathbf w+\alpha\e_i)$. By \eqref{eq:smooth}, $\phi'$ is $L_i$‑Lipschitz, hence the standard 1‑D smoothness inequality gives the claim. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. All expectations are taken w.r.t. the random choice of $i_t$ conditioned on the past; we write $\E_t[\cdot]=\E[\cdot\mid\mathbf w_t]$.

\begin{enumerate}
\item[Step 1.]  Start from the squared distance to the optimum after an update:
\[
\norm{\mathbf w_{t+1}-\mathbf w^\star}^{2}
= \norm{\mathbf w_t-\mathbf w^\star-\eta\mathbf g_t}^{2}.
\tag{1}
\]

\item[Step 2.]  Expand the square:
\[
\norm{\mathbf w_{t+1}-\mathbf w^\star}^{2}
= \norm{\mathbf w_t-\mathbf w^\star}^{2}
-2\eta\ip{\mathbf g_t}{\mathbf w_t-\mathbf w^\star}
+\eta^{2}\norm{\mathbf g_t}^{2}.
\tag{2}
\]

\item[Step 3.]  Take conditional expectation $\E_t[\cdot]$ and use unbiasedness
$\E_t[\mathbf g_t]=\nabla f(\mathbf w_t)$:
\[
\E_t\!\bigl[\norm{\mathbf w_{t+1}-\mathbf w^\star}^{2}\bigr]
= \norm{\mathbf w_t-\mathbf w^\star}^{2}
-2\eta\ip{\nabla f(\mathbf w_t)}{\mathbf w_t-\mathbf w^\star}
+\eta^{2}\E_t\!\bigl[\norm{\mathbf g_t}^{2}\bigr].
\tag{3}
\]

\item[Step 4.]  Upper‑bound the second moment of $\mathbf g_t$.
Since $\mathbf g_t$ has only one non‑zero coordinate,
\[
\norm{\mathbf g_t}^{2}= \frac{1}{p_{i_t}^{2}}\bigl(\partial_{i_t}f(\mathbf w_t)\bigr)^{2}.
\]
Hence
\[
\E_t\!\bigl[\norm{\mathbf g_t}^{2}\bigr]
= \sum_{i=1}^{d}p_i\frac{1}{p_i^{2}}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
= \sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}.
\tag{4}
\]

\item[Step 5.]  Relate the term in (4) to the smoothness constants.
From Lemma~\ref{lem:descent} with $\alpha=-\frac{\eta}{p_i}\partial_i f(\mathbf w_t)$ we have
\[
f\!\bigl(\mathbf w_t-\tfrac{\eta}{p_i}\partial_i f(\mathbf w_t)\e_i\bigr)
\le f(\mathbf w_t)-\frac{\eta}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
+\frac{L_i}{2}\frac{\eta^{2}}{p_i^{2}}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}.
\tag{5}
\]

\item[Step 6.]  Take expectation of (5) over $i_t$:
\[
\E_t\!\bigl[f(\mathbf w_{t+1})\bigr]
\le f(\mathbf w_t)-\eta\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_i}{p_i^{2}}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}.
\tag{6}
\]

\item[Step 7.]  Because $\eta\le 1/\max_i L_i$, we have for every $i$,
\[
\frac{L_i}{p_i^{2}}\eta\le \frac{1}{p_i^{2}}.
\]
Multiplying by $\frac{\eta^{2}}{2}$ and rearranging yields
\[
-\eta\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
+\frac{\eta^{2}L_i}{2p_i^{2}}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
\le -\frac{\eta}{2p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}.
\tag{7}
\]

\item[Step 8.]  Insert (7) into (6):
\[
\E_t\!\bigl[f(\mathbf w_{t+1})\bigr]
\le f(\mathbf w_t)-\frac{\eta}{2}\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}.
\tag{8}
\]

\item[Step 9.]  Apply convexity (Assumption 1) with $\mathbf w^\star$:
\[
f(\mathbf w_t)-f(\mathbf w^\star)\le \ip{\nabla f(\mathbf w_t)}{\mathbf w_t-\mathbf w^\star}
= \sum_{i=1}^{d}\partial_i f(\mathbf w_t)(w_{t,i}-w_i^\star).
\tag{9}
\]

\item[Step 10.]  Use Cauchy–Schwarz together with the weights $p_i$:
\[
\sum_{i=1}^{d}\partial_i f(\mathbf w_t)(w_{t,i}-w_i^\star)
\le \Bigl(\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}\Bigr)^{\!1/2}
      \Bigl(\sum_{i=1}^{d}p_i\,(w_{t,i}-w_i^\star)^{2}\Bigr)^{\!1/2}.
\tag{10}
\]

\item[Step 11.]  Square both sides of (10) and rearrange:
\[
\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(\mathbf w_t)\bigr)^{2}
\ge
\frac{\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)^{2}}
      {\sum_{i=1}^{d}p_i\,(w_{t,i}-w_i^\star)^{2}}.
\tag{11}
\]

\item[Step 12.]  Plug (11) into (8) and take full expectation:
\[
\E\!\bigl[f(\mathbf w_{t+1})\bigr]
\le \E\!\bigl[f(\mathbf w_t)\bigr]
-\frac{\eta}{2}\,
\E\!\left[
\frac{\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)^{2}}
      {\sum_{i=1}^{d}p_i\,(w_{t,i}-w_i^\star)^{2}}
\right].
\tag{12}
\]

\item[Step 13.]  Observe that $\sum_{i=1}^{d}p_i\,(w_{t,i}-w_i^\star)^{2}\le
\max_{i}p_i\;\norm{\mathbf w_t-\mathbf w^\star}^{2}\le \norm{\mathbf w_t-\mathbf w^\star}^{2}$,
because $p_i\le1$. Hence
\[
\frac{1}{\sum_{i}p_i (w_{t,i}-w_i^\star)^{2}}
\ge \frac{1}{\norm{\mathbf w_t-\mathbf w^\star}^{2}}.
\tag{13}
\]

\item[Step 14.]  Using (13) in (12) gives
\[
\E\!\bigl[f(\mathbf w_{t+1})\bigr]
\le \E\!\bigl[f(\mathbf w_t)\bigr]
-\frac{\eta}{2}\,
\E\!\left[
\frac{\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)^{2}}
      {\norm{\mathbf w_t-\mathbf w^\star}^{2}}
\right].
\tag{14}
\]

\item[Step 15.]  Apply the elementary inequality
\[
\frac{a^{2}}{b}\ge \frac{2a}{c}-\frac{c^{2}}{b},
\qquad\text{with }a=f(\mathbf w_t)-f(\mathbf w^\star),\;b=\norm{\mathbf w_t-\mathbf w^\star}^{2},
\;c=\frac{2}{\eta},
\]
which after rearrangement yields
\[
\frac{\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)^{2}}
      {\norm{\mathbf w_t-\mathbf w^\star}^{2}}
\ge
\frac{2}{\eta}\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)
-\frac{4}{\eta^{2}}.
\tag{15}
\]

\item[Step 16.]  Substitute (15) into (14):
\[
\E\!\bigl[f(\mathbf w_{t+1})\bigr]
\le \E\!\bigl[f(\mathbf w_t)\bigr]
-\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)
+\frac{2}{\eta}\cdot\frac{\eta}{2}\cdot\frac{4}{\eta^{2}}
= \E\!\bigl[f(\mathbf w_t)\bigr]-\bigl(f(\mathbf w_t)-f(\mathbf w^\star)\bigr)
+\frac{2}{\eta^{2}}.
\tag{16}
\]

\item[Step 17.]  Rearranging (16) gives a recursion for the expected suboptimality:
\[
\E\!\bigl[f(\mathbf w_t)-f(\mathbf w^\star)\bigr]
\le \frac{\norm{\mathbf w_0-\mathbf w^\star}^{2}}{2\eta\,t},
\tag{17}
\]
which is proved formally by telescoping the inequality
\[
\frac{1}{2\eta}\bigl(\norm{\mathbf w_t-\mathbf w^\star}^{2}
-\norm{\mathbf w_{t+1}-\mathbf w^\star}^{2}\bigr)
\ge f(\mathbf w_t)-f(\mathbf w^\star)
\]
derived from (2)–(3) together with the bound
$\E_t[\norm{\mathbf g_t}^{2}]\le \frac{1}{\eta^{2}}\norm{\mathbf w_t-\mathbf w^\star}^{2}$ (a direct consequence of (4) and the stepsize condition). Summing from $t=0$ to $T-1$ yields (17).

\item[Step 18.]  Finally, divide (17) by $T$ and note that the left‑hand side is non‑increasing in $t$, obtaining the statement of Theorem~\ref{thm:conv}.
\end{enumerate}

Thus the expected suboptimality after $T$ iterations satisfies the $O(1/T)$ bound claimed.

\section*{Rate Derivation (With Constants)}
From the telescoping inequality
\[
\frac{1}{2\eta}\Bigl(\norm{\mathbf w_0-\mathbf w^\star}^{2}
-\norm{\mathbf w_T-\mathbf w^\star}^{2}\Bigr)
\ge \sum_{t=0}^{T-1}\E\!\bigl[f(\mathbf w_t)-f(\mathbf w^\star)\bigr],
\]
and the non‑negativity of the norm term, we obtain the explicit constant:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(\mathbf w_t)-f(\mathbf w^\star)\bigr]
\le \frac{\norm{\mathbf w_0-\mathbf w^\star}^{2}}{2\eta\,T}.
\]
If the algorithm returns the last iterate $\mathbf w_T$, convexity implies
\[
\E\!\bigl[f(\mathbf w_T)-f(\mathbf w^\star)\bigr]
\le \frac{\norm{\mathbf w_0-\mathbf w^\star}^{2}}{2\eta\,T}.
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$. The update becomes
$\mathbf w_{t+1}= \mathbf w_t-d\eta\,\partial_{i_t}f(\mathbf w_t)\e_{i_t}$,
which is the classic randomized coordinate descent. The theorem recovers its $O(1/T)$ rate.
\item \textbf{Importance sampling.} Choose $p_i\propto L_i$ (or $\propto \|\partial_i f(\mathbf w_0)\|$). The scaling $1/p_i$ automatically compensates, so the same bound holds without any extra variance term.
\item \textbf{Full‑gradient limit.} If $p_i\equiv1$ for a single coordinate (i.e., $d=1$), the algorithm reduces to gradient descent with stepsize $\eta$, and the bound coincides with the standard $O(1/T)$ result for convex smooth functions.
\end{itemize}
