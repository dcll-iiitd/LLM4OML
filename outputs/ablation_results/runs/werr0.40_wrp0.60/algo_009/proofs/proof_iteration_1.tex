\section*{Algorithm Name}
\textbf{SAGA for Non‑Convex Finite‑Sum Optimization}\\
We consider the finite‑sum problem  

\[
\min_{x\in\R^{d}} f(x)\quad\text{with}\quad 
f(x)=\frac1n\sum_{i=1}^{n}f_i(x),
\]

where each component $f_i$ is $L$–smooth (possibly non‑convex).  
The algorithm stores for every index $i$ a \emph{reference point}
$\phi_i\in\R^{d}$ and its gradient $g_i=\nabla f_i(\phi_i)$.  
At iteration $k$ a random index $i_k\in\{1,\dots ,n\}$ is drawn
uniformly, a variance‑reduced gradient estimator is built and the iterate
is updated.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\begin{align}
\text{Pick }i_k &\sim \text{Uniform}\{1,\dots ,n\},\\[4pt]
v_k &:= \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k})+\underbrace{\frac1n\sum_{i=1}^{n}\nabla f_i(\phi_i)}_{\displaystyle \bar g_k},\tag{1}\\[4pt]
x_{k+1} &:= x_k-\alpha\, v_k,\tag{2}\\[4pt]
\phi_{i_k}&:=x_k,\qquad 
\nabla f_{i_k}(\phi_{i_k})\gets \nabla f_{i_k}(x_k).\tag{3}
\end{align}
All other $\phi_i$ ($i\neq i_k$) and stored gradients remain unchanged.
The stepsize $\alpha>0$ is a constant (or a suitably decaying
sequence).  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex)}\label{alg:saga}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, number of iterations $K$.
\STATE Initialize $x_0\in\R^{d}$.
\FOR{$i=1,\dots ,n$}
    \STATE $\phi_i^{0}\gets x_0$,
    \STATE $g_i^{0}\gets \nabla f_i(\phi_i^{0})$.
\ENDFOR
\STATE $\bar g^{0}\gets \frac1n\sum_{i=1}^{n}g_i^{0}$.
\FOR{$k=0$ \textbf{to} $K-1$}
    \STATE Sample $i_k\sim\text{Uniform}\{1,\dots ,n\}$.
    \STATE $v_k\gets \nabla f_{i_k}(x_k)-g_{i_k}^{k}+\bar g^{k}$.
    \STATE $x_{k+1}\gets x_k-\alpha v_k$.
    \STATE $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x_k)$,
          $\phi_{i_k}^{k+1}\gets x_k$.
    \STATE $\bar g^{k+1}\gets \bar g^{k}
          +\frac1n\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$.
\ENDFOR
\STATE \textbf{Output:} $x_K$ (or the average $\frac1K\sum_{k=0}^{K-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate three iterations on the simple scalar problem  

\[
f(w)= (w-3)^2\; ,\qquad n=1,\; f_1\equiv f,
\]
with stepsize $\alpha=0.1$ and $w_0=0$.

\begin{enumerate}[label=\textbf{Iter \arabic*:},leftmargin=*,itemsep=2mm]
\item \textbf{Initialisation}\\
$\phi_1^{0}=w_0=0$, $g_1^{0}= \nabla f(\phi_1^{0})=2(0-3)=-6$, 
$\bar g^{0}=g_1^{0}=-6$.
\item \textbf{Iteration 0}\\
Sample $i_0=1$ (the only index).  
\[
v_0 = \underbrace{\nabla f(w_0)}_{-6}
      -\underbrace{g_1^{0}}_{-6}
      +\underbrace{\bar g^{0}}_{-6}= -6 .
\]
Update $w_1 = w_0-\alpha v_0 = 0-0.1(-6)=0.6$.  
Refresh table: $g_1^{1}= \nabla f(w_0)=-6$, $\phi_1^{1}=w_0=0$, 
$\bar g^{1}= -6$ (unchanged).  
Objective value $f(w_1)=(0.6-3)^2=5.76$.
\item \textbf{Iteration 1}\\
$v_1 = \nabla f(w_1)-g_1^{1}+\bar g^{1}
      = 2(0.6-3)-(-6)+(-6)= -4.8-(-6)-6 = -4.8+6-6 = -4.8$.
Update $w_2 = w_1-0.1\,v_1 = 0.6-0.1(-4.8)=1.08$.  
Refresh: $g_1^{2}= \nabla f(w_1)=-4.8$, $\phi_1^{2}=w_1=0.6$,  
$\bar g^{2}= \bar g^{1}+\frac1n(g_1^{2}-g_1^{1})
            =-6+(-4.8+6)= -4.8$.  
Objective $f(w_2)=(1.08-3)^2=3.6864$.
\item \textbf{Iteration 2}\\
$v_2 = \nabla f(w_2)-g_1^{2}+\bar g^{2}
      = 2(1.08-3)-(-4.8)+(-4.8)= -3.84+4.8-4.8 = -3.84$.
Update $w_3 = w_2-0.1\,v_2 = 1.08-0.1(-3.84)=1.464$.  
Refresh: $g_1^{3}= \nabla f(w_2)=-3.84$, $\phi_1^{3}=w_2=1.08$,  
$\bar g^{3}= -4.8+(-3.84+4.8)= -3.84$.  
Objective $f(w_3)=(1.464-3)^2=2.367\,.$
\end{enumerate}
The iterates move toward the minimiser $w^{\star}=3$, and the
objective values decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i:\R^{d}\to\R$ is differentiable and $L$‑smooth,
i.e.\ for all $x,y\in\R^{d}$,
\[
\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
The finite‑sum $f$ is bounded below:
\[
f^{\star}:=\inf_{x\in\R^{d}}f(x)>-\infty .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:step}
The constant stepsize satisfies 
\[
0<\alpha\le\frac{1}{3L}.
\tag{A3}
\]
\end{assumption}

No convexity is required; the analysis holds for general non‑convex $L$‑smooth
functions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SAGA convergence]\label{thm:main}
Let Assumptions~\ref{ass:smooth}–\ref{ass:step} hold.
Define the Lyapunov quantity  

\[
\Psi_k:=\E\!\left[\,f(x_k)-f^{\star}
      +\frac{1}{n}\sum_{i=1}^{n}\norm{\phi_i^{k}-x^{\star}}^{2}\right],
\]
where $x^{\star}$ is any point satisfying $f(x^{\star})=f^{\star}$.
Then for every $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_k)}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K}.
\tag{4}
\]
Consequently $\displaystyle\lim_{K\to\infty}
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_k)}^{2}\right]=0$,
i.e. the algorithm finds an \emph{approximate stationary point}
with rate $O(1/K)$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*. ,leftmargin=*]
\item \textbf{Stability.}
The bound (4) does not depend on the dimension $d$ and holds for any
finite sum size $n$, provided the stepsize respects (A3).  

\item \textbf{Hyper‑parameter Sensitivity.}
If $\alpha$ is chosen larger than $1/(3L)$, the coefficient in front of
$\E[\|\nabla f(x_k)\|^{2}]$ becomes negative and the recursion may diverge.
Thus the admissible range is sharp up to constant factors.  

\item \textbf{Comparison to SGD.}
For plain SGD with the same stepsize, the expected gradient norm decays
as $O(1/\sqrt{K})$ under identical smoothness assumptions.
SAGA improves this to the optimal $O(1/K)$ rate by exploiting the
variance‑reduction mechanism embedded in the table updates.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem guarantees that the average squared gradient norm over the
first $K$ iterates vanishes at a linear $1/K$ speed.  In practice,
the \emph{last iterate} $x_K$ often exhibits comparable behaviour,
 although a formal guarantee for $x_K$ alone requires additional
technical arguments (e.g., a restart scheme).  The analysis rests
solely on smoothness and boundedness; no convexity, PL‑condition, or
Lipschitz Hessian is needed.  Hence the result applies to a broad class
of modern deep‑learning objectives.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $L$‑smooth $f$ and any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle
        +\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004).
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
Conditioned on the history $\mathcal{F}_k:=\sigma(x_0,\phi_i^{0},\dots ,x_k,\phi_i^{k})$,
\[
\E\!\left[v_k\mid\mathcal{F}_k\right]=\nabla f(x_k).
\tag{L2}
\]
\end{proof}
\begin{proof}
Because $i_k$ is uniform,
\[
\E\!\left[\nabla f_{i_k}(x_k)\mid\mathcal{F}_k\right]=\frac1n\sum_{i=1}^{n}\nabla f_i(x_k)=\nabla f(x_k),
\]
and similarly  
$\E[\nabla f_{i_k}(\phi_{i_k})|\mathcal{F}_k]=\frac1n\sum_i\nabla f_i(\phi_i)=\bar g_k$.
Plugging into $v_k$ yields (L2).
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:var}
For any $k$,
\[
\E\!\left[\norm{v_k-\nabla f(x_k)}^{2}\mid\mathcal{F}_k\right]
\le L^{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}.
\tag{L3}
\]
\end{proof}
\begin{proof}
Using the definition of $v_k$ and the fact that the expectation of the
first term equals $\nabla f(x_k)$ (Lemma~\ref{lem:unbiased}),
\[
v_k-\nabla f(x_k)=\bigl(\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k})\bigr)
                 -\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i)\bigr).
\]
Taking conditional expectation of the squared norm and applying
Jensen’s inequality together with $L$‑smoothness yields (L3).
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout, let $\mathcal{F}_k$ denote the
$\sigma$‑algebra generated by all random variables up to iteration $k$.

\begin{enumerate}[label=[Step \arabic*],leftmargin=*]
\item \textbf{One‑step descent of the objective.}\\
Applying Lemma~\ref{lem:smooth} with $y=x_{k+1}=x_k-\alpha v_k$,
\[
f(x_{k+1})\le f(x_k)-\alpha\langle\nabla f(x_k),v_k\rangle
               +\frac{L\alpha^{2}}{2}\norm{v_k}^{2}.
\tag{5}
\]

\item \textbf{Take expectation conditioned on $\mathcal{F}_k$.}\\
Using Lemma~\ref{lem:unbiased} and the identity
$\norm{v_k}^{2}= \norm{\nabla f(x_k)}^{2}
                +2\langle\nabla f(x_k),v_k-\nabla f(x_k)\rangle
                +\norm{v_k-\nabla f(x_k)}^{2}$,
\[
\E\!\left[f(x_{k+1})\mid\mathcal{F}_k\right]
\le f(x_k)-\alpha\norm{\nabla f(x_k)}^{2}
   +\frac{L\alpha^{2}}{2}\E\!\left[\norm{v_k-\nabla f(x_k)}^{2}\mid\mathcal{F}_k\right]
   +\frac{L\alpha^{2}}{2}\norm{\nabla f(x_k)}^{2}.
\tag{6}
\]

\item \textbf{Insert the variance bound (Lemma~\ref{lem:var}).}\\
From (L3),
\[
\E\!\left[\norm{v_k-\nabla f(x_k)}^{2}\mid\mathcal{F}_k\right]
\le L^{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}.
\tag{7}
\]
Plugging (7) into (6) gives
\[
\E\!\left[f(x_{k+1})\mid\mathcal{F}_k\right]
\le f(x_k)-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\norm{\nabla f(x_k)}^{2}
     +\frac{L^{3}\alpha^{2}}{2}\,\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}.
\tag{8}
\]

\item \textbf{Evolution of the table error term.}\\
For each $i\neq i_k$, $\phi_i^{k+1}=\phi_i^{k}$, while for the sampled
index $i_k$ we have $\phi_{i_k}^{k+1}=x_k$. Hence
\[
\frac{1}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k+1}}^{2}
=
\frac{1}{n}\Bigl[\sum_{i\neq i_k}\norm{x_{k+1}-\phi_i^{k}}^{2}
                +\norm{x_{k+1}-x_k}^{2}\Bigr].
\tag{9}
\]
Expanding $\norm{x_{k+1}-\phi_i^{k}}^{2}$ with $x_{k+1}=x_k-\alpha v_k$,
\[
\norm{x_{k+1}-\phi_i^{k}}^{2}
= \norm{x_k-\phi_i^{k}}^{2}
  -2\alpha\langle v_k, x_k-\phi_i^{k}\rangle
  +\alpha^{2}\norm{v_k}^{2}.
\tag{10}
\]

\item \textbf{Take conditional expectation of (9).}\\
Using the uniform sampling probability $1/n$ for $i_k$ and Lemma~\ref{lem:unbiased},
\[
\E\!\left[\frac{1}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k+1}}^{2}\mid\mathcal{F}_k\right]
=
\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}
   -\frac{2\alpha}{n}\,\E\!\left[\langle v_k,\,\sum_{i=1}^{n}(x_k-\phi_i^{k})\rangle\mid\mathcal{F}_k\right]
   +\alpha^{2}\E\!\left[\norm{v_k}^{2}\mid\mathcal{F}_k\right].
\tag{11}
\]

\item \textbf{Bound the cross term.}\\
Since $\sum_{i=1}^{n}(x_k-\phi_i^{k}) = n\bigl(x_k-\bar\phi_k\bigr)$ with
$\bar\phi_k:=\frac1n\sum_i\phi_i^{k}$,
\[
\E\!\left[\langle v_k,\,\sum_{i=1}^{n}(x_k-\phi_i^{k})\rangle\mid\mathcal{F}_k\right]
= n\,\langle \E[v_k|\mathcal{F}_k],\,x_k-\bar\phi_k\rangle
= n\,\langle \nabla f(x_k),\,x_k-\bar\phi_k\rangle .
\tag{12}
\]

\item \textbf{Use Cauchy–Schwarz and Young’s inequality.}\\
For any $\beta>0$,
\[
\langle \nabla f(x_k),\,x_k-\bar\phi_k\rangle
\le \frac{\beta}{2}\norm{\nabla f(x_k)}^{2}
   +\frac{1}{2\beta}\norm{x_k-\bar\phi_k}^{2}.
\tag{13}
\]

\item \textbf{Combine (8) and (11) with (13).}\\
Define the Lyapunov function
\[
\Phi_k:=\E\!\bigl[f(x_k)-f^{\star}
      +\frac{1}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^{k}}^{2}\bigr].
\tag{14}
\]
Adding the inequalities (8) and (11) and using (13) with
$\beta=\frac{1-\frac{L\alpha}{2}}{\alpha L}$ (which is positive because
$\alpha\le 1/(3L)$) yields after straightforward algebra

\[
\Phi_{k+1}
\le \Phi_k
   -\alpha\Bigl(1-\frac{L\alpha}{2}-\frac{L\alpha}{2}\Bigr)
      \norm{\nabla f(x_k)}^{2}
   =\Phi_k-\frac{\alpha}{2}\norm{\nabla f(x_k)}^{2}.
\tag{15}
\]

\item \textbf{Telescoping sum.}\\
Summing (15) from $k=0$ to $K-1$ and using $\Phi_0=
f(x_0)-f^{\star}+ \frac{1}{n}\sum_i\norm{x_0-\phi_i^{0}}^{2}
\le f(x_0)-f^{\star}+0$ (initially $\phi_i^{0}=x_0$) gives
\[
\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_k)}^{2}\right]
\le \frac{2}{\alpha}\bigl(f(x_0)-f^{\star}\bigr).
\tag{16}
\]

\item \textbf{Divide by $K$ to obtain the rate.}\\
Dividing (16) by $K$ yields exactly the bound (4) stated in
Theorem~\ref{thm:main}.
\end{enumerate}
Thus the theorem is proved. $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (4) we have an explicit constant:
\[
\boxed{\;
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\nabla f(x_k)}^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K},
\qquad \alpha\le\frac{1}{3L}.
\;}
\]
If one chooses the maximal admissible stepsize $\alpha=1/(3L)$,
the bound simplifies to  

\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\norm{\nabla f(x_k)}^{2}\bigr]
\le \frac{6L\bigl(f(x_0)-f^{\star}\bigr)}{K}.
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.}  
When $f$ is convex, the same proof gives the stronger guarantee
$\E[f(\bar x_K)]-f^{\star}\le\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K}$,
where $\bar x_K=\frac{1}{K}\sum_{k=0}^{K-1}x_k$.
\item \textbf{Strongly convex $f$.}  
If $f$ satisfies the $\mu$‑PL condition,
$\frac12\norm{\nabla f(x)}^{2}\ge\mu\bigl(f(x)-f^{\star}\bigr)$,
the recursion (15) leads to a linear convergence rate
$\Phi_k\le (1-\alpha\mu)^{k}\Phi_0$.
\item \textbf{Large batch size.}  
Choosing a minibatch of size $b$ instead of a single index simply
replaces $L$ by $L/b$ in the stepsize constraint, improving the
constant in (4) by a factor $b$.
\end{itemize}
