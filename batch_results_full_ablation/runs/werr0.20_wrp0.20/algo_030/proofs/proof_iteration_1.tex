\section*{Heavy‑Ball Method}

The Heavy‑Ball (HB) method is a first‑order optimization algorithm that augments the usual gradient descent step with a \emph{momentum} term that uses the previous displacement.  It is especially popular for smooth non‑convex objectives.

\section*{Mathematical Formulation}

Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Given an initial pair $(x_{0},x_{-1})\in\mathbb{R}^{d}\times\mathbb{R}^{d}$, step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the iterates are defined by  

\[
\boxed{%
x_{k+1}=x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k}),\qquad k=0,1,2,\dots
}
\tag{HB}
\]

When $x_{-1}=x_{0}$ the algorithm starts from a pure gradient step.

\section*{Pseudocode}

\begin{algorithm}[H]
\caption{Heavy‑Ball (HB) Method}
\begin{algorithmic}[1]
\Require Objective $f$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
        initial point $x_{0}$, optional previous point $x_{-1}=x_{0}$,
        maximum iterations $K$ (or tolerance).
\State $x_{-1}\gets x_{0}$ \Comment{initialize previous iterate}
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_{k}\gets \nabla f(x_{k})$
    \State $x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,g_{k}$
    \If{$\|g_{k}\|\leq\varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

Consider the one‑dimensional quadratic  

\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]

Choose $\alpha=0.1$, $\beta=0.5$, $w_{0}=0$, and initialise $w_{-1}=w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_{k}= \nabla f(w_{k})$ & $w_{k+1}=w_{k}+\beta(w_{k}-w_{k-1})-\alpha g_{k}$ & $f(w_{k+1})$\\\hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^{2}=2.6244$\\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^{2}=0.8196$
\end{tabular}
\end{center}

The objective value decreases from $9$ (at $w_{0}=0$) to $0.8196$ after three iterations, illustrating the accelerating effect of momentum.

\section*{Assumptions}

\begin{enumerate}[label={\bf A\arabic*}.]
\item \label{A1} \textbf{Smoothness.} $f$ is $L$‑smooth, i.e. for all $x,y\in\mathbb{R}^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{A1}
\]
Equivalently (Descent Lemma),
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1$'$}
\]

\item \label{A2} \textbf{Lower boundedness.} $f$ admits a finite infimum $f_{\inf}>-\infty$ and we denote $f_{\star}= \inf_{x}f(x)$.

\item \label{A3} \textbf{Parameter range.} The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0\le \beta<1,\qquad 0<\alpha\le\frac{1-\beta}{L}.
\tag{A3}
\]

\item \label{A4} \textbf{Initialization.} The first two iterates are bounded, e.g. $x_{-1}=x_{0}$, which guarantees that the sequence $\{x_{k}\}$ is well defined.
\end{enumerate}

All results below are derived under \ref{A1}–\ref{A4} only; no convexity is required.

\section*{Main Convergence Theorem}

\begin{theorem}[Average Gradient Norm Decay]\label{thm:main}
Let $\{x_{k}\}$ be generated by \eqref{HB} under Assumptions \ref{A1}–\ref{A3}.  
Define the Lyapunov function  

\[
\Phi_{k}\;:=\; f(x_{k})-\!f_{\star}+\frac{\beta}{2\alpha}\,\|x_{k}-x_{k-1}\|^{2}.
\tag{T1}
\]

Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\;\le\;
\frac{2\bigl(f(x_{0})-f_{\star}\bigr)}{\alpha\,(1-\beta)\,T}.
\tag{T2}
\]
Consequently $\displaystyle\lim_{T\to\infty}\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}=0$,
i.e. the algorithm finds an \emph{approximate stationary point} at the rate $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}

\begin{enumerate}[label={\bf P\arabic*}.]
\item \textbf{Stability.} Condition \eqref{A3} guarantees that the Lyapunov sequence $\{\Phi_{k}\}$ is non‑increasing, preventing divergence even in the presence of non‑convex curvature.

\item \textbf{Hyper‑parameter sensitivity.} The admissible stepsize shrinks linearly with $\beta$: larger momentum requires a smaller $\alpha$ to preserve \eqref{A3}.  When $\beta=0$ the bound reduces to the classic gradient‑descent rate $O(1/(\alpha T))$.

\item \textbf{Comparison to baselines.} Compared with plain gradient descent ($\beta=0$) the HB method enjoys the same $O(1/T)$ guarantee but often exhibits a much smaller constant in practice because the momentum term reduces the “oscillation’’ around narrow valleys.

\item \textbf{Special case – Polyak–Łojasiewicz (PL) condition.} If, in addition to \ref{A1}–\ref{A3}, $f$ satisfies the PL inequality
\[
\frac{1}{2}\|\nabla f(x)\|^{2}\ge\mu\bigl(f(x)-f_{\star}\bigr),\qquad \mu>0,
\]
then the same Lyapunov analysis yields a linear convergence bound
\[
f(x_{k})-f_{\star}\le\bigl(1-\alpha\mu(1-\beta)\bigr)^{k}\bigl(f(x_{0})-f_{\star}\bigr).
\]
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \qedhere
\end{proof}

\begin{lemma}[Quadratic expansion]\label{lem:exp}
For any vectors $a,b\in\mathbb{R}^{d}$ and scalar $\gamma>0$,
\[
\|a+b\|^{2}= \|a\|^{2}+2\langle a,b\rangle+\|b\|^{2}.
\]
\end{lemma}
\begin{proof}
Direct consequence of the inner‑product definition. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof we keep the iteration index $k$ generic and suppress it when the meaning is clear.

\bigskip
\noindent\textbf{Step 1.  Apply the descent lemma to the HB update.}  
Using Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$ gives
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})+\big\langle\nabla f(x_{k}),x_{k+1}-x_{k}\big\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{1}
\]

\noindent\textbf{Step 2.  Substitute the HB update.}  
From \eqref{HB},
\[
x_{k+1}-x_{k}= \beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k}).
\tag{2}
\]

Insert (2) into (1):
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})+\big\langle\nabla f(x_{k}),
      \beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k})\big\rangle\\
&\qquad+\frac{L}{2}\Big\|\beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k})\Big\|^{2}.
\end{aligned}
\tag{3}
\]

\noindent\textbf{Step 3.  Expand the inner product and the squared norm.}  
Using Lemma~\ref{lem:exp} repeatedly,
\[
\big\langle\nabla f(x_{k}),\beta\,(x_{k}-x_{k-1})\big\rangle
   =\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle,
\]
and
\[
\Big\|\beta\,(x_{k}-x_{k-1})-\alpha\,\nabla f(x_{k})\Big\|^{2}
 =\beta^{2}\|x_{k}-x_{k-1}\|^{2}
   -2\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}.
\tag{4}
\]

Plugging (4) into (3) yields
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
   +\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   -\alpha\|\nabla f(x_{k})\|^{2}\\
&\quad+\frac{L}{2}\Big(\beta^{2}\|x_{k}-x_{k-1}\|^{2}
   -2\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}\Big).
\end{aligned}
\tag{5}
\]

\noindent\textbf{Step 4.  Rearrange terms.}  
Collect coefficients of $\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle$ and $\|\nabla f(x_{k})\|^{2}$:
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
   +\Big(\beta-\alpha\beta L\Big)\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle\\
&\quad+\Big(-\alpha+\tfrac{L\alpha^{2}}{2}\Big)\|\nabla f(x_{k})\|^{2}
   +\tfrac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{6}
\]

\noindent\textbf{Step 5.  Use the stepsize condition \eqref{A3}.}  
Because $\alpha\le\frac{1-\beta}{L}$ we have $L\alpha\le 1-\beta$, which implies  

\[
\beta-\alpha\beta L = \beta\bigl(1-L\alpha\bigr)\ge\beta^{2}.
\tag{7}
\]

Moreover,
\[
-\alpha+\frac{L\alpha^{2}}{2}
= -\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)
\le -\frac{\alpha}{2},
\qquad\text{since }L\alpha\le 1-\beta<1.
\tag{8}
\]

Insert (7)–(8) into (6):
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
   +\beta^{2}\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   -\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}
   +\frac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{9}
\]

\noindent\textbf{Step 6.  Couple the inequality with the momentum term.}  
Consider the Lyapunov quantity defined in (T1):
\[
\Phi_{k}=f(x_{k})-f_{\star}+\frac{\beta}{2\alpha}\|x_{k}-x_{k-1}\|^{2}.
\]

Compute the difference $\Phi_{k+1}-\Phi_{k}$:

\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&= \bigl[f(x_{k+1})-f(x_{k})\bigr]
   +\frac{\beta}{2\alpha}\bigl(\|x_{k+1}-x_{k}\|^{2}
                               -\|x_{k}-x_{k-1}\|^{2}\bigr).
\end{aligned}
\tag{10}
\]

Using (2), $\;x_{k+1}-x_{k}= \beta\,(x_{k}-x_{k-1})-\alpha\nabla f(x_{k})$, we obtain

\[
\|x_{k+1}-x_{k}\|^{2}
= \beta^{2}\|x_{k}-x_{k-1}\|^{2}
   -2\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}.
\tag{11}
\]

Insert (9) for $f(x_{k+1})-f(x_{k})$ and (11) into (10). After cancelling the $\beta^{2}\|x_{k}-x_{k-1}\|^{2}$ terms we obtain

\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&\le
   -\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}
   +\beta^{2}\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle\\
&\quad -\frac{\beta}{\alpha}\alpha\beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   +\frac{\beta}{2\alpha}\alpha^{2}\|\nabla f(x_{k})\|^{2}\\
&= -\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}
   +\underbrace{\bigl(\beta^{2}-\beta^{2}\bigr)}_{=0}
     \langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   +\frac{\beta\alpha}{2}\|\nabla f(x_{k})\|^{2}.
\end{aligned}
\tag{12}
\]

Thus
\[
\Phi_{k+1}-\Phi_{k}
\le -\frac{\alpha}{2}(1-\beta)\,\|\nabla f(x_{k})\|^{2}.
\tag{13}
\]

\noindent\textbf{Step 7.  Telescoping sum.}  
Summing (13) from $k=0$ to $T-1$ yields

\[
\Phi_{T}-\Phi_{0}
\le -\frac{\alpha}{2}(1-\beta)\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}.
\tag{14}
\]

Since $\Phi_{T}\ge f_{\star}-f_{\star}=0$, we have $\Phi_{T}\ge0$, and $\Phi_{0}=f(x_{0})-f_{\star}+\frac{\beta}{2\alpha}\|x_{0}-x_{-1}\|^{2}\le f(x_{0})-f_{\star}$ (the second term vanishes if $x_{-1}=x_{0}$).  Hence

\[
0-\Phi_{0}\le -\frac{\alpha}{2}(1-\beta)\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2},
\]
or equivalently

\[
\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\le \frac{2\bigl(f(x_{0})-f_{\star}\bigr)}{\alpha(1-\beta)}.
\tag{15}
\]

Dividing both sides of (15) by $T$ gives the desired average‑gradient bound (T2).  \qed

\section*{Rate Derivation (With Constants)}

From Theorem \ref{thm:main} we have the explicit constant

\[
C:=\frac{2\bigl(f(x_{0})-f_{\star}\bigr)}{\alpha(1-\beta)}.
\]

Therefore for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}\le\frac{C}{T}=O\!\left(\frac{1}{T}\right).
\]

If additionally the PL inequality holds with constant $\mu$, the same Lyapunov analysis (replace the bound on $\|\nabla f(x_{k})\|^{2}$ by $\mu\bigl(f(x_{k})-f_{\star}\bigr)$) yields

\[
f(x_{k})-f_{\star}\le\bigl(1-\alpha\mu(1-\beta)\bigr)^{k}\bigl(f(x_{0})-f_{\star}\bigr),
\]
i.e. a linear rate with contraction factor $1-\alpha\mu(1-\beta)\in(0,1)$.

\section*{Special Cases}

\begin{itemize}
\item \textbf{Pure gradient descent} ($\beta=0$): the Lyapunov function reduces to $f(x_{k})-f_{\star}$ and (T2) becomes the classic $O(1/(\alpha T))$ bound.
\item \textbf{Heavy‑ball with $\beta\to1^{-}$}: condition \eqref{A3} forces $\alpha\to0$, so the guarantee degenerates to the same $O(1/T)$ rate but with a vanishing stepsize, reflecting the well‑known instability of large momentum on non‑convex landscapes.
\item \textbf{Quadratic case} $f(x)=\frac{1}{2}x^{\top}Qx$: the HB iteration is a linear dynamical system; spectral analysis shows that the optimal $\beta$ and $\alpha$ achieve a convergence factor $ \bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^{2}$ where $\kappa$ is the condition number of $Q$.  This matches the classical optimal rate for the heavy‑ball method on strongly convex quadratics.
\end{itemize}

\section*{Discussion}

The Heavy‑Ball method, despite its simplicity, enjoys a rigorous $O(1/T)$ guarantee for smooth non‑convex objectives under a modest stepsize restriction.  The proof hinges on a carefully crafted Lyapunov function that couples function values with the kinetic energy $\|x_{k}-x_{k-1}\|^{2}$.  The analysis shows precisely why the momentum parameter must be paired with a reduced stepsize: the term $(1-\beta)$ appears both in the admissible stepsize bound and in the denominator of the convergence constant.  When additional curvature information (e.g. PL condition) is available, the same machinery upgrades to a linear rate, explaining the empirical acceleration observed on many machine‑learning loss surfaces.
