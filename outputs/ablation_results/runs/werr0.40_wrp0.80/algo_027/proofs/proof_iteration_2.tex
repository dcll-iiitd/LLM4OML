\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
Probabilistic Coordinate Descent (PCD)

\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be differentiable.  
Denote by $\nabla_i f(w)$ the $i$‑th partial derivative and by $e_i\in\mathbb{R}^d$ the $i$‑th canonical basis vector.  

\begin{itemize}
    \item \textbf{Probability distribution.} A fixed probability vector $p=(p_1,\dots ,p_d)$ satisfies $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.
    \item \textbf{Stochastic gradient estimator.} At iteration $k$, draw an index $i_k\in\{1,\dots ,d\}$ according to $p$.  
    Define
    \[
        g_k \;:=\; \frac{1}{p_{i_k}}\,\nabla_{i_k} f(w_k)\,e_{i_k}\in\mathbb{R}^d .
    \]
    Then $\mathbb{E}[g_k\mid w_k]=\nabla f(w_k)$ (unbiasedness).
    \item \textbf{Update rule.} With stepsize $\eta_k>0$,
    \[
        w_{k+1}= w_k - \eta_k\, g_k .
    \]
\end{itemize}

\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent (PCD)}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\mathbb{R}^d$, stepsize schedule $\{\eta_k\}_{k\ge0}$, probability vector $p$ with $p_i>0$, $ \sum_i p_i =1$.
\FOR{$k=0,1,2,\dots,T-1$}
    \STATE Sample $i_k\sim p$.
    \STATE Compute partial gradient $g_{k,i_k}:=\nabla_{i_k}f(w_k)$.
    \STATE Form unbiased estimator $g_k:=\frac{1}{p_{i_k}}\,g_{k,i_k}\,e_{i_k}$.
    \STATE Update $w_{k+1}\gets w_k-\eta_k g_k$.
\ENDFOR
\RETURN $w_T$ (or the averaged iterate $\bar w_T:=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\addcontentsline{toc}{section}{Dry Run Example}

Here $d=1$, $p_1=1$, and the algorithm reduces to ordinary gradient descent.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{as:smooth}
For each coordinate $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\mathbb{R}^d$ and $\alpha\in\mathbb{R}$,
\[
    \bigl|\nabla_i f(w+\alpha e_i)-\nabla_i f(w)\bigr|\le L_i|\alpha|.
\]
Equivalently,
\[
    f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^2 .
\]
\end{assumption}

\begin{assumption}[Bounded variance of stochastic estimator]\label{as:variance}
Define the stochastic estimator $g_k$ as above. There exists $\sigma^2\ge0$ such that for all $k$,
\[
    \mathbb{E}\!\left[\bigl\|g_k-\nabla f(w_k)\bigr\|^2\mid w_k\right]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsizes are constant, $\eta_k\equiv\eta$, and satisfy
\[
    0<\eta\le \min_{i}\frac{p_i}{L_i}\;=\;\frac{1}{\displaystyle\max_{i}\frac{L_i}{p_i}}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:stepsize}. Let $w^\star$ be any point (not necessarily a global minimum). After $T$ iterations of PCD,
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
    \;\le\;
    \frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{\eta T}
    \;+\;
    \sigma^2 .
\]
In particular, choosing
\[
    \eta = \sqrt{\frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{L_{\max}\,T}},
    \qquad
    L_{\max}:=\max_{i}\frac{L_i}{p_i},
\]
yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^2\bigr]
    =\mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}[leftmargin=*]
    \item \textbf{Unbiasedness.} By construction $\mathbb{E}[g_k\mid w_k]=\nabla f(w_k)$, which is crucial for the descent argument.
    \item \textbf{Stability w.r.t. probabilities.} The factor $1/p_i$ appears both in the estimator and in the smoothness constant $L_{\max}$, showing a trade‑off: very small $p_i$ lead to large variance but allow larger steps on rarely‑updated coordinates.
    \item \textbf{Comparison to uniform coordinate descent.} If $p_i=1/d$ for all $i$, then $L_{\max}=d\;\max_i L_i$, reproducing the classic $O(\sqrt{d/T})$ rate.
    \item \textbf{Non‑convexity.} No convexity is required; the theorem guarantees convergence to a point with small gradient norm in expectation.
\end{itemize}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The result mirrors the standard analysis of stochastic gradient methods but incorporates the coordinate‑sampling distribution explicitly. By weighting the estimator with $1/p_i$, the expected update recovers the full gradient, allowing the proof to follow the usual descent lemma. The maximal smoothness constant $L_{\max}$ quantifies how the sampling probabilities affect the achievable rate.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]\label{lem:descent}
Under Assumption \ref{as:smooth}, for any $w\in\mathbb{R}^d$, any $i\in\{1,\dots ,d\}$ and any $\alpha\in\mathbb{R}$,
\[
    f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^2 .
\]
\end{lemma}
\begin{proof}
Directly from the definition of $L_i$‑smoothness. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:variance}
For the estimator $g_k$ defined in the algorithm,
\[
    \mathbb{E}_k\!\bigl[\|g_k\|^2\bigr]
    =\|\nabla f(w_k)\|^2+\mathbb{E}_k\!\bigl[\|g_k-\nabla f(w_k)\|^2\bigr]
    \le \|\nabla f(w_k)\|^2+\sigma^2 .
\]
\end{lemma}
\begin{proof}
The identity $\|a\|^2=\|b\|^2+2\langle b,a-b\rangle+\|a-b\|^2$ together with $\mathbb{E}_k[g_k]=\nabla f(w_k)$ gives the first equality; the inequality follows from Assumption \ref{as:variance}. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the randomness of the index $i_k$ and any other stochasticity; conditional expectations are denoted $\mathbb{E}_k[\cdot]=\mathbb{E}[\cdot\mid w_k]$.

\begin{enumerate}
    \item[Step 1.] \textbf{Apply Lemma \ref{lem:descent}} with $\alpha=-\eta\,\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)$ for the sampled coordinate $i_k$:
    \begin{align}
        f(w_{k+1})
        &= f\!\bigl(w_k-\eta\,\tfrac{1}{p_{i_k}}\nabla_{i_k}f(w_k)e_{i_k}\bigr) \nonumber\\
        &\le f(w_k)
           -\eta\,\frac{1}{p_{i_k}}\bigl(\nabla_{i_k}f(w_k)\bigr)^2
           +\frac{L_{i_k}}{2}\Bigl(\eta\,\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)\Bigr)^2 .
        \label{eq:descent_one}
    \end{align}
    % CORRECTED: the quadratic term now contains $L_{i_k}$ only (no extra $p_{i_k}$).

    \item[Step 2.] \textbf{Rewrite the RHS using the stochastic estimator $g_k$.}
    Recall $g_k=\frac{1}{p_{i_k}}\nabla_{i_k}f(w_k)e_{i_k}$, hence
    \[
        \langle g_k,\nabla f(w_k)\rangle
        =\frac{1}{p_{i_k}}\bigl(\nabla_{i_k}f(w_k)\bigr)^2,
        \qquad
        \|g_k\|^2
        =\frac{1}{p_{i_k}^2}\bigl(\nabla_{i_k}f(w_k)\bigr)^2 .
    \]
    Substituting these identities into \eqref{eq:descent_one} yields
    \begin{equation}
        f(w_{k+1})
        \le f(w_k)
           -\eta\langle g_k,\nabla f(w_k)\rangle
           +\frac{\eta^{2}}{2}\,L_{i_k}\,\|g_k\|^{2}.
        \label{eq:descent_two}
    \end{equation}
    % CORRECTED: the extra factor $p_{i_k}$ present in the original proof has been removed.

    \item[Step 3.] \textbf{Take conditional expectation $\mathbb{E}_k[\cdot]$.}
    Using the sampling distribution of $i_k$,
    \[
        \mathbb{E}_k\!\bigl[L_{i_k}\|g_k\|^{2}\bigr]
        =\sum_{i=1}^{d}p_i L_i\frac{1}{p_i^{2}}\bigl(\nabla_i f(w_k)\bigr)^{2}
        =\sum_{i=1}^{d}\frac{L_i}{p_i}\bigl(\nabla_i f(w_k)\bigr)^{2}.
    \]
    Moreover $\mathbb{E}_k[g_k]=\nabla f(w_k)$. Hence
    \begin{align}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        &\le f(w_k)
           -\eta\|\nabla f(w_k)\|^{2}
           +\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}\bigl(\nabla_i f(w_k)\bigr)^{2}.
        \label{eq:exp_descent}
    \end{align}
    % ADDED: explicit computation of the conditional expectation.

    \item[Step 4.] \textbf{Introduce the maximal smoothness constant.}
    Define
    \[
        L_{\max}:=\max_{i}\frac{L_i}{p_i}.
    \]
    Since each term $\frac{L_i}{p_i}(\nabla_i f)^2\le L_{\max}(\nabla_i f)^2$, we obtain
    \[
        \sum_{i=1}^{d}\frac{L_i}{p_i}(\nabla_i f)^2
        \le L_{\max}\sum_{i=1}^{d}(\nabla_i f)^2
        = L_{\max}\,\|\nabla f\|^{2}.
    \]
    Substituting into \eqref{eq:exp_descent} gives
    \begin{equation}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k)
           -\Bigl(\eta-\frac{\eta^{2}L_{\max}}{2}\Bigr)\|\nabla f(w_k)\|^{2}.
        \label{eq:descent_final}
    \end{equation}
    % CORRECTED: the bound now uses $L_{\max}$, not the erroneous $L_{\text{eff}}$.

    \item[Step 5.] \textbf{Use the stepsize condition.}
    By Assumption \ref{as:stepsize},
    \[
        \eta\le\frac{1}{L_{\max}}\quad\Longrightarrow\quad
        \eta-\frac{\eta^{2}L_{\max}}{2}\ge\frac{\eta}{2}.
    \]
    Hence
    \begin{equation}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k)-\frac{\eta}{2}\,\|\nabla f(w_k)\|^{2}.
        \label{eq:descent_simplified}
    \end{equation}
    % ADDED: explicit derivation of the inequality $\eta-\eta^{2}L_{\max}/2\ge\eta/2$.

    \item[Step 6.] \textbf{Incorporate the variance bound.}
    From Lemma \ref{lem:variance},
    \[
        \mathbb{E}_k\!\bigl[\|g_k\|^{2}\bigr]
        \le \|\nabla f(w_k)\|^{2}+\sigma^{2}.
    \]
    Returning to \eqref{eq:descent_two} and using the bound above instead of the exact equality yields
    \[
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k)
           -\eta\|\nabla f(w_k)\|^{2}
           +\frac{\eta^{2}}{2}L_{\max}\bigl(\|\nabla f(w_k)\|^{2}+\sigma^{2}\bigr).
    \]
    Applying the same steps as in Steps 4–5 we obtain the *variance‑aware* descent inequality
    \begin{equation}
        \mathbb{E}_k\!\bigl[f(w_{k+1})\bigr]
        \le f(w_k)-\frac{\eta}{2}\,\|\nabla f(w_k)\|^{2}
           +\frac{\eta}{2}\,\sigma^{2}.
        \label{eq:descent_variance}
    \end{equation}
    % ADDED: a dedicated variance step that leads to the additive $\sigma^{2}$ term.

    \item[Step 7.] \textbf{Take full expectation and telescope.}
    Taking total expectation in \eqref{eq:descent_variance} and summing for $k=0,\dots,T-1$ gives
    \[
        \mathbb{E}[f(w_T)]
        \le \mathbb{E}[f(w_0)]
           -\frac{\eta}{2}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
           +\frac{\eta T}{2}\sigma^{2}.
    \]
    Rearranging,
    \[
        \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
        \le \frac{2\bigl(\mathbb{E}[f(w_0)]- \mathbb{E}[f(w_T)]\bigr)}{\eta T}
           +\sigma^{2}
        \le \frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{\eta T}
           +\sigma^{2}.
    \]
    This is exactly the bound claimed in Theorem \ref{thm:main}. \hfill\qedsymbol
\end{enumerate}

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
Choosing
\[
    \eta = \sqrt{\frac{2\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{L_{\max}\,T}}
\]
satisfies the stepsize condition $\eta\le 1/L_{\max}$ and yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
    \le
    2\sqrt{\frac{2L_{\max}\bigl(\mathbb{E}[f(w_0)]-f^\star\bigr)}{T}}
    \;+\;\sigma^{2}
    =\mathcal{O}\!\bigl(T^{-1/2}\bigr)+\sigma^{2}.
\]

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}
    \item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$. Then $L_{\max}=d\;\max_i L_i$, and the bound matches the classical $O(\sqrt{d/T})$ rate for uniform coordinate descent.
    \item \textbf{Importance sampling.} Choosing $p_i\propto\sqrt{L_i}$ minimises $L_{\max}= \max_i\frac{L_i}{p_i}$ and can lead to a substantial reduction compared with uniform sampling when the $L_i$ are heterogeneous.
    \item \textbf{Deterministic cyclic coordinate descent.} If $p_i=1$ for a single $i$ and $0$ for the others, the algorithm reduces to a one‑dimensional gradient descent with stepsize $\eta\le 1/L_i$, and the bound simplifies to the standard $\mathcal{O}(1/(\eta T))$ rate.
\end{enumerate}

\section*{Conclusion}
\addcontentsline{toc}{section}{Conclusion}
We presented a fully rigorous analysis of Probabilistic Coordinate Descent for non‑convex, coordinate‑wise smooth objectives. The unbiased estimator $g_k$ weighted by $1/p_{i_k}$ allows the classic stochastic‑gradient proof technique to go through, while the maximal smoothness constant $L_{\max}=\max_i L_i/p_i$ captures the influence of the sampling distribution. The resulting $\mathcal{O}(T^{-1/2})$ rate matches the best known guarantees for stochastic first‑order methods in the non‑convex setting.
