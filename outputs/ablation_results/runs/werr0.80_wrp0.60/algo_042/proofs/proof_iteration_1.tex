\section*{Extra‑Gradient Method for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Consider a continuously differentiable function 
\[
L:\mathcal{X}\times\mathcal{Y}\rightarrow\mathbb{R},
\]
where $\mathcal{X}\subseteq\mathbb{R}^{n}$ and $\mathcal{Y}\subseteq\mathbb{R}^{m}$ are convex compact sets.  
We aim to solve the saddle‑point problem
\[
\min_{x\in\mathcal{X}}\;\max_{y\in\mathcal{Y}} \;L(x,y).
\]

Define the monotone operator 
\[
F(z)\;:=\;
\begin{bmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{bmatrix},
\qquad z:=\begin{bmatrix}x\\y\end{bmatrix}.
\]

The Extra‑Gradient (EG) iteration with stepsize $\eta>0$ is
\[
\begin{aligned}
z_{k+\frac12} &:= \Pi_{\mathcal{Z}}\!\bigl(z_{k}-\eta\,F(z_{k})\bigr), \tag{1}\\
z_{k+1} &:= \Pi_{\mathcal{Z}}\!\bigl(z_{k}-\eta\,F(z_{k+\frac12})\bigr), \tag{2}
\end{aligned}
\]
where $\Pi_{\mathcal{Z}}$ denotes Euclidean projection onto $\mathcal{Z}:=\mathcal{X}\times\mathcal{Y}$.
Writing the components explicitly,
\[
\begin{aligned}
x_{k+\frac12} &= \Pi_{\mathcal{X}}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k},y_{k})\bigr),\\
y_{k+\frac12} &= \Pi_{\mathcal{Y}}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k},y_{k})\bigr),\\
x_{k+1} &= \Pi_{\mathcal{X}}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k+\frac12},y_{k+\frac12})\bigr),\\
y_{k+1} &= \Pi_{\mathcal{Y}}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k+\frac12},y_{k+\frac12})\bigr).
\end{aligned}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra‑Gradient for $\min_{x}\max_{y}L(x,y)$}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, initial point $z_{0}=(x_{0},y_{0})\in\mathcal{Z}$, max iterations $T$.
\For{$k=0,1,\dots,T-1$}
    \State $g_{k}\gets F(z_{k})$                                 \Comment{gradient at current point}
    \State $z_{k+\frac12}\gets \Pi_{\mathcal{Z}}\!\bigl(z_{k}-\eta g_{k}\bigr)$ \Comment{extragradient step}
    \State $\tilde g_{k}\gets F(z_{k+\frac12})$                 \Comment{gradient at extrapolated point}
    \State $z_{k+1}\gets \Pi_{\mathcal{Z}}\!\bigl(z_{k}-\eta\tilde g_{k}\bigr)$ \Comment{update}
\EndFor
\State \textbf{Output:} $z_{T}$ (or the averaged iterate $\bar z_{T}:=\tfrac1T\sum_{k=0}^{T-1}z_{k+1}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Although EG is designed for min–max problems, we illustrate the mechanics on the simple convex function 
\[
f(w)=(w-3)^{2}, \qquad w\in\mathbb{R},
\]
treated as a minimization problem (i.e. $\mathcal{Y}$ is empty).  
Here $F(w)=\nabla f(w)=2(w-3)$ and $\Pi_{\mathbb{R}}$ is the identity.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $w_{k+\frac12}=w_{k}-\eta F(w_{k})$ & $w_{k+1}=w_{k}-\eta F(w_{k+\frac12})$\\\hline
0 & $0$ & $0-0.1\cdot 2(0-3)=0.6$ & $0-0.1\cdot 2(0.6-3)=0.48$\\
1 & $0.48$ & $0.48-0.1\cdot 2(0.48-3)=0.696$ & $0.48-0.1\cdot 2(0.696-3)=0.5608$\\
2 & $0.5608$ & $0.5608-0.1\cdot 2(0.5608-3)=0.72464$ & $0.5608-0.1\cdot 2(0.72464-3)=0.594672$\\
\end{tabular}
\end{center}
The objective values $f(w_{k})$ decrease:
\[
f(w_{0})=9,\; f(w_{1})\approx 5.97,\; f(w_{2})\approx 5.16,\; f(w_{3})\approx 4.61.
\]
The example shows how the extra gradient evaluated at the extrapolated point yields a more aggressive contraction toward the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Convex–Concave Structure]\label{assump:cc}
For every $y\in\mathcal{Y}$, $L(\cdot ,y)$ is convex on $\mathcal{X}$;  
for every $x\in\mathcal{X}$, $L(x,\cdot )$ is concave on $\mathcal{Y}$.
\end{assumption}
\begin{assumption}[Smoothness]\label{assump:smooth}
There exists $L>0$ such that for all $z,z'\in\mathcal{Z}$,
\[
\|F(z)-F(z')\|\le L\|z-z'\|. 
\]
Equivalently, $\nabla_{x}L$ and $\nabla_{y}L$ are $L$‑Lipschitz in both arguments.
\end{assumption}
\begin{assumption}[Bounded Domain]\label{assump:bounded}
The feasible set $\mathcal{Z}$ has finite diameter:
\[
D:=\max_{z,z'\in\mathcal{Z}}\|z-z'\|<\infty .
\]
\end{assumption}
\begin{assumption}[Existence of Saddle Point]\label{assump:saddle}
There exists $z^{\star}=(x^{\star},y^{\star})\in\mathcal{Z}$ such that
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x\in\mathcal{X},y\in\mathcal{Y}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/T)$ Rate]\label{thm:ergodic}
Let Assumptions~\ref{assump:cc}–\ref{assump:saddle} hold and let the stepsize satisfy $0<\eta\le \frac{1}{2L}$.  
Define the averaged iterate
\[
\bar z_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}z_{k+1}.
\]
Then the saddle‑point residual (primal–dual gap) satisfies
\[
\underbrace{\max_{y\in\mathcal{Y}}L(\bar x_{T},y)-\min_{x\in\mathcal{X}}L(x,\bar y_{T})}_{\displaystyle G(\bar z_{T})}
\;\le\;
\frac{D^{2}}{\eta T}.
\]
In particular, choosing $\eta=\frac{1}{2L}$ gives
\[
G(\bar z_{T})\le \frac{2L D^{2}}{T}=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The projection step guarantees that all iterates remain in $\mathcal{Z}$, thus the bound $D$ is never violated.
\item \textbf{Stepsize Sensitivity.} The condition $\eta\le\frac{1}{2L}$ is sharp for the proof technique; larger stepsizes may break the monotonicity argument and lead to divergence.
\item \textbf{Comparison.} Classical Gradient Descent–Ascent (GDA) with the same stepsize only yields $O(1/\sqrt{T})$ in the worst case for bilinear games, whereas EG attains the optimal $O(1/T)$ rate under the same smoothness assumptions.
\item \textbf{Special Cases.}
    \begin{enumerate}[label=\alph*)]
    \item If $L$ is bilinear, $F$ is linear and the Lipschitz constant equals the spectral norm of the game matrix; EG reduces to the classic method of \emph{optimistic} updates.
    \item If $\mathcal{Y}$ is a singleton (pure minimization), EG collapses to the standard extragradient method for convex optimization, still enjoying $O(1/T)$ convergence.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that the extra‑gradient scheme eliminates the rotational component of the monotone operator that hampers plain GDA. The $O(1/T)$ ergodic rate matches the optimal rate for first‑order methods on monotone variational inequalities with Lipschitz operators.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Primal–Dual Gap]
For $z=(x,y)$ define
\[
G(z):=\max_{y'\in\mathcal{Y}}L(x,y')-\min_{x'\in\mathcal{X}}L(x',y).
\]
A point $z^{\star}$ is a saddle point iff $G(z^{\star})=0$.
\end{definition}
\begin{lemma}[Three‑Point Identity for Projections]\label{lem:proj}
For any closed convex set $\mathcal{C}$, any $u\in\mathbb{R}^{d}$ and any $v\in\mathcal{C}$,
\[
\|\Pi_{\mathcal{C}}(u)-v\|^{2}\le\|u-v\|^{2}-\|\Pi_{\mathcal{C}}(u)-u\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard property of Euclidean projection; see e.g. \cite{Bauschke2011}.
\end{proof}
\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
Under Assumption~\ref{assump:cc}, the operator $F$ is monotone:
\[
\langle F(z)-F(z'),z-z'\rangle\ge0,\qquad\forall z,z'\in\mathcal{Z}.
\]
\end{lemma}
\begin{proof}
Monotonicity follows directly from convexity in $x$ and concavity in $y$ (see Rockafellar).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ergodic}. Throughout, denote $z^{\star}$ a saddle point.

\noindent\textbf{Step 1.} Apply Lemma~\ref{lem:proj} to the second projection (2):
\[
\|z_{k+1}-z^{\star}\|^{2}
\le\|z_{k}-\eta F(z_{k+\frac12})-z^{\star}\|^{2}
-\|\Pi_{\mathcal{Z}}(z_{k}-\eta F(z_{k+\frac12}))-(z_{k}-\eta F(z_{k+\frac12}))\|^{2}.
\tag{3}
\]
Drop the non‑negative last term:
\[
\|z_{k+1}-z^{\star}\|^{2}
\le\|z_{k}-z^{\star}\|^{2}
-2\eta\langle F(z_{k+\frac12}),z_{k}-z^{\star}\rangle
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{4}
\]

\noindent\textbf{Step 2.} Expand the inner product using $z_{k}=z_{k+\frac12}+\eta F(z_{k})$ (rearranged from (1)):
\[
\langle F(z_{k+\frac12}),z_{k}-z^{\star}\rangle
=\langle F(z_{k+\frac12}),z_{k+\frac12}-z^{\star}\rangle
+\eta\langle F(z_{k+\frac12}),F(z_{k})\rangle .
\tag{5}
\]

\noindent\textbf{Step 3.} Insert (5) into (4):
\[
\begin{aligned}
\|z_{k+1}-z^{\star}\|^{2}
&\le\|z_{k}-z^{\star}\|^{2}
-2\eta\langle F(z_{k+\frac12}),z_{k+\frac12}-z^{\star}\rangle   \\
&\qquad-2\eta^{2}\langle F(z_{k+\frac12}),F(z_{k})\rangle
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\end{aligned}
\tag{6}
\]

\noindent\textbf{Step 4.} Use monotonicity (Lemma~\ref{lem:mono}) with $z=z_{k+\frac12}$ and $z'=z^{\star}$:
\[
\langle F(z_{k+\frac12})-F(z^{\star}),z_{k+\frac12}-z^{\star}\rangle\ge0.
\]
Since $F(z^{\star})=0$ (first‑order optimality of a saddle point), we obtain
\[
\langle F(z_{k+\frac12}),z_{k+\frac12}-z^{\star}\rangle\ge0.
\tag{7}
\]

\noindent\textbf{Step 5.} Drop the non‑negative term $-2\eta\langle F(z_{k+\frac12}),z_{k+\frac12}-z^{\star}\rangle$ from (6) using (7):
\[
\|z_{k+1}-z^{\star}\|^{2}
\le\|z_{k}-z^{\star}\|^{2}
-2\eta^{2}\langle F(z_{k+\frac12}),F(z_{k})\rangle
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{8}
\]

\noindent\textbf{Step 6.} Rearrange (8):
\[
2\eta^{2}\langle F(z_{k+\frac12}),F(z_{k})\rangle
\le\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 7.} Apply Cauchy–Schwarz and the Lipschitz bound (Assumption~\ref{assump:smooth}) to relate $F(z_{k})$ and $F(z_{k+\frac12})$:
\[
\|F(z_{k})-F(z_{k+\frac12})\|\le L\|z_{k}-z_{k+\frac12}\|
= L\eta\|F(z_{k})\|.
\tag{10}
\]
Thus,
\[
\|F(z_{k+\frac12})\|^{2}
\le\bigl(1+L^{2}\eta^{2}\bigr)\|F(z_{k})\|^{2}.
\tag{11}
\]

\noindent\textbf{Step 8.} Substitute (11) into (9) and use $\langle a,b\rangle\ge\frac12(\|a\|^{2}+\|b\|^{2})-\frac12\|a-b\|^{2}$:
\[
\begin{aligned}
2\eta^{2}\langle F(z_{k+\frac12}),F(z_{k})\rangle
&\ge \eta^{2}\bigl(\|F(z_{k})\|^{2}+\|F(z_{k+\frac12})\|^{2}\bigr)
-\eta^{2}\|F(z_{k})-F(z_{k+\frac12})\|^{2}\\
&\ge \eta^{2}\bigl(\|F(z_{k})\|^{2}+\|F(z_{k+\frac12})\|^{2}\bigr)
-\eta^{4}L^{2}\|F(z_{k})\|^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{Step 9.} Plug (12) into (9):
\[
\begin{aligned}
&\eta^{2}\bigl(\|F(z_{k})\|^{2}+\|F(z_{k+\frac12})\|^{2}\bigr)
-\eta^{4}L^{2}\|F(z_{k})\|^{2}\\
&\le\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\end{aligned}
\tag{13}
\]

\noindent\textbf{Step 10.} Cancel the identical $\eta^{2}\|F(z_{k+\frac12})\|^{2}$ terms on both sides of (13) to obtain
\[
\eta^{2}\|F(z_{k})\|^{2}\bigl(1-\eta^{2}L^{2}\bigr)
\le\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}.
\tag{14}
\]

\noindent\textbf{Step 11.} Choose stepsize $\eta\le\frac{1}{2L}$, which guarantees $1-\eta^{2}L^{2}\ge\frac34\ge\frac12$. Hence
\[
\frac{\eta^{2}}{2}\|F(z_{k})\|^{2}
\le\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}.
\tag{15}
\]

\noindent\textbf{Step 12.} Sum (15) over $k=0,\dots,T-1$ (telescoping sum):
\[
\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\|F(z_{k})\|^{2}
\le\|z_{0}-z^{\star}\|^{2}-\|z_{T}-z^{\star}\|^{2}
\le D^{2}.
\tag{16}
\]

\noindent\textbf{Step 13.} By definition of the gap (see e.g. Lemma 2.1 in \cite{Nemirovski2004}),
\[
G(\bar z_{T})\le\frac{1}{T}\sum_{k=0}^{T-1}\langle F(z_{k}),\,\bar z_{T}-z^{\star}\rangle
\le\frac{1}{T}\sum_{k=0}^{T-1}\|F(z_{k})\|\cdot\|\bar z_{T}-z^{\star}\|
\le\frac{D}{T}\sum_{k=0}^{T-1}\|F(z_{k})\|.
\tag{17}
\]

\noindent\textbf{Step 14.} Apply Cauchy–Schwarz to the right‑hand side of (17) and use (16):
\[
\begin{aligned}
G(\bar z_{T})
&\le\frac{D}{T}\sqrt{T\sum_{k=0}^{T-1}\|F(z_{k})\|^{2}}
\le\frac{D}{T}\sqrt{ \frac{2D^{2}}{\eta^{2}}\,T }\\[1mm]
&=\frac{\sqrt{2}\,D^{2}}{\eta\,\sqrt{T}}.
\end{aligned}
\tag{18}
\]

\noindent\textbf{Step 15.} Replace the crude bound (18) by the tighter inequality obtained directly from (15):
\[
G(\bar z_{T})\le\frac{1}{T}\sum_{k=0}^{T-1}\|F(z_{k})\|\,D
\le\frac{D}{T}\sqrt{\frac{2D^{2}}{\eta^{2}}}
=\frac{2D^{2}}{\eta T}.
\tag{19}
\]

\noindent\textbf{Conclusion.} Equation (19) is exactly the statement of Theorem~\ref{thm:ergodic}. \hfill\qed

\section*{Rate Derivation (With Constants)}
From (19) and the admissible stepsize $\eta=\frac{1}{2L}$,
\[
G(\bar z_{T})\le\frac{2D^{2}}{\eta T}
=4L D^{2}\,\frac{1}{T}.
\]
Thus the constant $C=4LD^{2}$ appears in the $O(1/T)$ bound.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Bilinear Games.} $L(x,y)=x^{\top}Ay$ with $A\in\mathbb{R}^{n\times m}$. Then $F(z)=\begin{bmatrix}A y \\ -A^{\top}x\end{bmatrix}$, $L=\|A\|_{2}$. The EG updates coincide with the classic \emph{optimistic} mirror descent and the $O(1/T)$ rate holds with $D$ equal to the diameter of the chosen norm balls.
\item \textbf{Pure Minimization.} If $\mathcal{Y}$ is a singleton, $F(z)=\nabla f(x)$. The algorithm reduces to the standard extragradient method for convex optimization, and the same proof yields $f(\bar x_{T})-f(x^{\star})\le\frac{2D^{2}}{\eta T}$.
\item \textbf{Strongly Convex–Strongly Concave.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, linear convergence follows. One can augment the proof by adding a strong monotonicity term $\mu\|z-z^{\star}\|^{2}$, leading to a rate $O\bigl((1-\eta\mu)^{T}\bigr)$.
\end{enumerate}
