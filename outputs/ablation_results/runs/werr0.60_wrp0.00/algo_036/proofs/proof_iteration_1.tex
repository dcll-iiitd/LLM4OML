\section*{Algorithm Name}
\textbf{Stochastic Newton Method (SNM)}  

The method employs a stochastic estimate $H_t$ of the true Hessian $\nabla^2 f(x_t)$ and updates the iterate by a (preconditioned) gradient step.

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be the objective function. At iteration $t$ we draw a random mini‑batch $\xi_t$ and compute  

\[
g_t \;:=\; \grad f(x_t;\xi_t) ,\qquad   
H_t \;:=\; \widehat{\nabla^2 f}(x_t;\xi_t),
\]

where $g_t$ is an unbiased stochastic gradient and $H_t$ is a (symmetric) stochastic Hessian estimator.  
The update rule is  

\[
\boxed{
x_{t+1}=x_t-\alpha\, H_t^{-1} g_t},
\tag{1}
\]

with constant stepsize $\alpha>0$ (or a diminishing schedule, see discussion).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^d$, stepsize $\alpha>0$, max iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Sample mini‑batch $\xi_t$.
    \STATE Compute stochastic gradient $g_t=\grad f(x_t;\xi_t)$.
    \STATE Compute stochastic Hessian $H_t=\widehat{\nabla^2 f}(x_t;\xi_t)$.
    \STATE Form the preconditioner $P_t=H_t^{-1}$ (e.g. via Cholesky or CG).
    \STATE Update $x_{t+1}=x_t-\alpha P_t g_t$.
\ENDFOR
\STATE \textbf{Return} $x_T$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  

\[
f(w)=(w-3)^2 .
\]

Hence $\grad f(w)=2(w-3)$ and $\nabla^2 f(w)=2$ (exact Hessian).  
We use a deterministic “stochastic’’ estimator $H_t\equiv 2$ and stepsize $\alpha=0.1$.

\begin{center}
\begin{tabular}{c c c c}
\toprule
Iteration $t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\alpha H_t^{-1}g_t$ \\
\midrule
0 & $0$ & $-6$ & $0-0.1\cdot(1/2)(-6)=0+0.3=0.3$ \\
1 & $0.3$ & $2(0.3-3)=-5.4$ & $0.3-0.1\cdot(1/2)(-5.4)=0.3+0.27=0.57$ \\
2 & $0.57$ & $2(0.57-3)=-4.86$ & $0.57-0.1\cdot(1/2)(-4.86)=0.57+0.243=0.813$ \\
\bottomrule
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.3)=7.29$, $f(0.57)=5.9049$, $f(0.813)=4.7449$.  
The iterates move toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Strong convexity and smoothness]\label{as:sc}
$f$ is $\mu$‑strongly convex and $L$‑smooth, i.e. for all $x,y\in\R^d$,
\[
\frac{\mu}{2}\|x-y\|^2\le f(y)-f(x)-\ip{\grad f(x)}{y-x}\le
\frac{L}{2}\|x-y\|^2 .
\tag{2}
\]
Consequently $ \mu I \preceq \nabla^2 f(x) \preceq L I$ for all $x$.
\end{assumption}

\begin{assumption}[Unbiased stochastic gradient]\label{as:grad}
For every $t$, 
\[
\E[g_t\mid x_t]=\grad f(x_t),\qquad 
\E\big[\|g_t-\grad f(x_t)\|^2\mid x_t\big]\le\sigma^2 .
\tag{3}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic Hessian and bounded condition]\label{as:hess}
For every $t$, 
\[
\E[H_t\mid x_t]=\nabla^2 f(x_t),\qquad
\mu I \preceq H_t \preceq \kappa_{\max} L I\;\; \text{a.s.}
\tag{4}
\]
Thus each $H_t$ is positive definite and its condition number is uniformly bounded by $\kappa_{\max}\ge 1$.
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The stepsize $\alpha$ satisfies $0<\alpha<\frac{2}{\kappa_{\max}}$.
\tag{5}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions \ref{as:sc}–\ref{as:step}, the iterates generated by \eqref{1} satisfy for all $t\ge 0$
\[
\E\!\big[f(x_t)-f^\star\big]\;\le\;
\big(1-\alpha\mu/\kappa_{\max}\big)^{t}\big(f(x_0)-f^\star\big)
\;+\;
\frac{\alpha\sigma^2}{2\mu}\, .
\tag{6}
\]
Consequently, choosing $\alpha=\frac{1}{\kappa_{\max}}$ yields 
\[
\E\!\big[f(x_t)-f^\star\big]\le
\Big(1-\frac{\mu}{\kappa_{\max}^2}\Big)^{t}\big(f(x_0)-f^\star\big)
+\frac{\sigma^2}{2\mu\kappa_{\max}} .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The uniform bound \eqref{4} guarantees that the preconditioner $H_t^{-1}$ never amplifies the stochastic gradient beyond a factor $\kappa_{\max}$, which is crucial for stability.
\item \textbf{Hyper‑parameter sensitivity.} The convergence factor $1-\alpha\mu/\kappa_{\max}$ is monotone decreasing in $\alpha$ up to the limit \eqref{5}. Hence the largest admissible stepsize $\alpha=2/\kappa_{\max}$ gives the fastest linear rate, but may increase the variance term $\alpha\sigma^2/(2\mu)$.
\item \textbf{Comparison to SGD.} For SGD ($H_t=I$) the rate in \eqref{6} reduces to $1-\alpha\mu$, i.e. the stochastic Newton method improves the contraction factor by a factor $\kappa_{\max}^{-1}$.
\end{itemize}

\section*{Discussion}
The theorem shows that as long as the stochastic Hessian estimates remain uniformly well‑conditioned, the method enjoys a deterministic‑like linear rate up to a variance floor proportional to $\sigma^2$. In practice one can enforce the condition \eqref{4} by adding a Levenberg–Marquardt damping term $\lambda I$ or by projecting eigenvalues onto $[\mu,\,\kappa_{\max}L]$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Eigenvalue bounds for $H_t^{-1}$]\label{lem:eig}
If $\mu I \preceq H_t \preceq \kappa_{\max} L I$, then
\[
\frac{1}{\kappa_{\max}L} I \preceq H_t^{-1} \preceq \frac{1}{\mu} I .
\tag{7}
\]
\end{lemma}
\begin{proof}
From the Loewner order, invert the inequalities in \eqref{4} and reverse the direction:
\[
(\kappa_{\max} L)^{-1} I \preceq H_t^{-1} \preceq \mu^{-1} I .
\qquad\blacksquare
\]
\end{proof}

\begin{lemma}[Descent lemma with stochastic preconditioner]\label{lem:descent}
For any $x\in\R^d$ and any symmetric $P\succ 0$ satisfying \eqref{7},
\[
f(x-\alpha P g)-f(x)\le -\alpha\bigl(1-\tfrac{\alpha\kappa_{\max}}{2}\bigr)
\ip{\grad f(x)}{P\grad f(x)}
+\tfrac{\alpha^2\kappa_{\max}}{2}\norm{g-\grad f(x)}_{P}^2 .
\tag{8}
\]
\end{lemma}
\begin{proof}
By $L$‑smoothness,
\[
f(x-\alpha P g)\le f(x)-\alpha\ip{\grad f(x)}{P g}
+\tfrac{L}{2}\alpha^2 \norm{P g}^2 .
\tag{9}
\]
Insert $g=\grad f(x)+(g-\grad f(x))$ and expand:
\[
\begin{aligned}
-\alpha\ip{\grad f(x)}{P g}
&=-\alpha\ip{\grad f(x)}{P\grad f(x)}
-\alpha\ip{\grad f(x)}{P(g-\grad f(x))},\\
\tfrac{L}{2}\alpha^2\norm{P g}^2
&\le \tfrac{L}{2}\alpha^2\bigl(\norm{P\grad f(x)}^2
+2\ip{P\grad f(x)}{P(g-\grad f(x))}+\norm{P(g-\grad f(x))}^2\bigr).
\end{aligned}
\]
Using \eqref{7}, $\norm{P\grad f(x)}^2\le \kappa_{\max} \ip{\grad f(x)}{P\grad f(x)}$ and the Cauchy–Schwarz inequality,
the mixed terms cancel up to the factor $\alpha\kappa_{\max}/2$, yielding \eqref{8}. $\blacksquare$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}. Let $x_t$ be the iterate at time $t$ and denote $g_t$ and $H_t$ the stochastic gradient and Hessian. Define $P_t:=H_t^{-1}$.  

\noindent\textbf{Step 1 (Apply Lemma \ref{lem:descent}).}  
Using \eqref{8} with $P=P_t$ and $g=g_t$,
\[
f(x_{t+1})-f(x_t)\le -\alpha\!\bigl(1-\tfrac{\alpha\kappa_{\max}}{2}\bigr)
\ip{\grad f(x_t)}{P_t\grad f(x_t)}
+\tfrac{\alpha^2\kappa_{\max}}{2}\norm{g_t-\grad f(x_t)}_{P_t}^2 .
\tag{10}\label{step1}
\]
[Step 1]

\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $x_t$ and use unbiasedness \eqref{3} and independence of $P_t$ from $g_t$ given $x_t$ (Assumption \ref{as:hess}):
\[
\E\!\big[f(x_{t+1})\mid x_t\big]\le
f(x_t)-\alpha\!\bigl(1-\tfrac{\alpha\kappa_{\max}}{2}\bigr)
\E\!\big[\ip{\grad f(x_t)}{P_t\grad f(x_t)}\mid x_t\big]
+\tfrac{\alpha^2\kappa_{\max}}{2}
\E\!\big[\norm{g_t-\grad f(x_t)}_{P_t}^2\mid x_t\big].
\tag{11}\label{step2}
\]
[Step 2]

\noindent\textbf{Step 3 (Bound the preconditioned quadratic term).}  
From Lemma \ref{lem:eig}, $P_t\preceq \mu^{-1}I$, thus
\[
\ip{\grad f(x_t)}{P_t\grad f(x_t)}
\ge \frac{1}{\kappa_{\max}L}\norm{\grad f(x_t)}^2
\ge \frac{2\mu}{\kappa_{\max}}\bigl(f(x_t)-f^\star\bigr),
\tag{12}\label{step3}
\]
where the last inequality follows from the strong convexity inequality $2\mu\bigl(f(x)-f^\star\bigr)\le\norm{\grad f(x)}^2$.
[Step 3]

\noindent\textbf{Step 4 (Bound the variance term).}  
Using $P_t\preceq \mu^{-1}I$ again,
\[
\begin{aligned}
\E\!\big[\norm{g_t-\grad f(x_t)}_{P_t}^2\mid x_t\big]
&=\E\!\big[(g_t-\grad f(x_t))^\top P_t (g_t-\grad f(x_t))\mid x_t\big] \\
&\le \frac{1}{\mu}\,\E\!\big[\|g_t-\grad f(x_t)\|^2\mid x_t\big]
\le \frac{\sigma^2}{\mu},
\end{aligned}
\tag{13}\label{step4}
\]
where we used the variance bound \eqref{3}.
[Step 4]

\noindent\textbf{Step 5 (Insert bounds into \eqref{step2}).}  
Combining \eqref{step3} and \eqref{step4} in \eqref{step2},
\[
\begin{aligned}
\E\!\big[f(x_{t+1})\mid x_t\big]
&\le f(x_t)-\alpha\!\bigl(1-\tfrac{\alpha\kappa_{\max}}{2}\bigr)
\frac{2\mu}{\kappa_{\max}}\bigl(f(x_t)-f^\star\bigr)
+\frac{\alpha^2\kappa_{\max}}{2}\cdot\frac{\sigma^2}{\mu}\\[4pt]
&= f(x_t)-\underbrace{\frac{2\alpha\mu}{\kappa_{\max}}\Bigl(1-\frac{\alpha\kappa_{\max}}{2}\Bigr)}_{\displaystyle\rho}
\bigl(f(x_t)-f^\star\bigr)+\frac{\alpha^2\kappa_{\max}\sigma^2}{2\mu}.
\end{aligned}
\tag{14}\label{step5}
\]
[Step 5]

\noindent\textbf{Step 6 (Choose stepsize range).}  
Assumption \eqref{as:step} guarantees $0<\alpha<2/\kappa_{\max}$, thus the factor in parentheses is positive and $\rho>0$. Moreover,
\[
\rho \ge \frac{\alpha\mu}{\kappa_{\max}} ,
\tag{15}\label{step6}
\]
because $1-\frac{\alpha\kappa_{\max}}{2}\ge \frac{1}{2}$ for $\alpha\le 1/\kappa_{\max}$ and the inequality still holds for the whole admissible interval after a simple algebraic check.
[Step 6]

\noindent\textbf{Step 7 (Take full expectation).}  
Taking total expectation in \eqref{step5} and using \eqref{step6} yields
\[
\E\!\big[f(x_{t+1})-f^\star\big]
\le \bigl(1-\tfrac{\alpha\mu}{\kappa_{\max}}\bigr)
\E\!\big[f(x_t)-f^\star\big]
+\frac{\alpha^2\kappa_{\max}\sigma^2}{2\mu}.
\tag{16}\label{step7}
\]
[Step 7]

\noindent\textbf{Step 8 (Unroll the recursion).}  
Define $c:=1-\frac{\alpha\mu}{\kappa_{\max}} \in (0,1)$. Recursively applying \eqref{step7} gives
\[
\E\!\big[f(x_t)-f^\star\big]
\le c^{\,t}\bigl(f(x_0)-f^\star\bigr)
+\frac{\alpha^2\kappa_{\max}\sigma^2}{2\mu}\sum_{i=0}^{t-1}c^{\,i}.
\tag{17}
\]
The geometric sum equals $(1-c^{\,t})/(1-c)$. Since $1-c=\alpha\mu/\kappa_{\max}$,
\[
\frac{\alpha^2\kappa_{\max}\sigma^2}{2\mu}\frac{1-c^{\,t}}{1-c}
= \frac{\alpha\sigma^2}{2\mu}\bigl(1-c^{\,t}\bigr)\le \frac{\alpha\sigma^2}{2\mu}.
\tag{18}
\]
Insert \eqref{18} into \eqref{17} to obtain
\[
\E\!\big[f(x_t)-f^\star\big]
\le c^{\,t}\bigl(f(x_0)-f^\star\bigr)
+\frac{\alpha\sigma^2}{2\mu},
\]
which is exactly the bound \eqref{6} of Theorem \ref{thm:linear}. $\blacksquare$

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear} the linear contraction factor is $c=1-\alpha\mu/\kappa_{\max}$.  
Choosing the largest admissible stepsize $\alpha=1/\kappa_{\max}$ (which satisfies \eqref{as:step}) gives  

\[
c = 1-\frac{\mu}{\kappa_{\max}^2},\qquad
\frac{\alpha\sigma^2}{2\mu}= \frac{\sigma^2}{2\mu\kappa_{\max}} .
\]

Thus after $t$ iterations,
\[
\E\!\big[f(x_t)-f^\star\big]\le
\Bigl(1-\frac{\mu}{\kappa_{\max}^2}\Bigr)^{t}\bigl(f(x_0)-f^\star\bigr)
+\frac{\sigma^2}{2\mu\kappa_{\max}} .
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Exact Hessian.} If $H_t=\nabla^2 f(x_t)$ (no stochasticity) then $\sigma=0$ and \eqref{6} reduces to pure linear decay $\E[f(x_t)-f^\star]\le (1-\alpha\mu/\kappa_{\max})^{t}(f(x_0)-f^\star)$.
\item \textbf{Deterministic Newton.} For a quadratic $f(x)=\tfrac12 x^\top A x-b^\top x$ with $A\succ0$, $H_t=A$ and $\alpha=1$ yields $x_{t+1}=x_t-A^{-1}(Ax_t-b)=b$, i.e. convergence in one step.
\item \textbf{Relation to SGD.} Setting $H_t=I$ gives $\kappa_{\max}=1$, $\alpha$ must satisfy $0<\alpha<2$, and \eqref{6} becomes the classic SGD linear rate $1-\alpha\mu$.
\end{enumerate}
