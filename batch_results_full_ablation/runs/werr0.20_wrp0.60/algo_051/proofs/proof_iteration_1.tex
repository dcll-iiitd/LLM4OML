\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics with Decreasing Temperature (SGLD‑DT)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
Given an initial point $x_{0}\in\R^{d}$, a stepsize (learning rate) sequence
$\{\alpha_{k}\}_{k\ge 0}$ with $\alpha_{k}>0$, and a temperature schedule
$\{\tau_{k}\}_{k\ge0}$, the update rule of SGLD‑DT is
\[
\boxed{
x_{k+1}=x_{k}-\alpha_{k}\,\nabla f(x_{k})+\sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k},
}
\tag{1}
\]
where $\xi_{k}\sim\mathcal{N}(0,I_{d})$ are i.i.d. standard Gaussian vectors,
independent of $x_{k}$.  
The noise term has covariance $2\alpha_{k}\tau_{k}I_{d}$, i.e. the variance
decreases as $\tau_{k}\downarrow0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD‑DT}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_{0}\in\R^{d}$, stepsizes $\{\alpha_{k}\}$, temperatures $\{\tau_{k}\}$
\FOR{$k=0,1,2,\dots,T-1$}
    \STATE Compute (full) gradient $g_{k}= \nabla f(x_{k})$
    \STATE Sample $\xi_{k}\sim\mathcal{N}(0,I_{d})$
    \STATE $x_{k+1}=x_{k}-\alpha_{k} g_{k}+\sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}$
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the averaged iterate $\bar x_{T}=\frac{1}{T}\sum_{k=0}^{T-1}x_{k}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, initial point $w_{0}=0$,
stepsize $\alpha=0.1$, and a deterministic “noise’’ sequence $\xi_{k}=0$ for
illustration (i.e. $\tau_{k}=0$).  

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\alpha\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
Corresponding objective values:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3315$.
The iterates move towards the minimiser $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\iprod{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x\in\R^{d}$.
\tag{A2}
\end{assumption}
\begin{assumption}[Gradient Boundedness in Expectation]\label{ass:gradvar}
There exists $\sigma^{2}\ge0$ with
\[
\E\!\left[\norm{\xi_{k}}^{2}\right]=d,\qquad
\E\!\left[\norm{\sqrt{2\alpha_{k}\tau_{k}}\xi_{k}}^{2}\right]=2\alpha_{k}\tau_{k}d\le\sigma^{2}\alpha_{k}.
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Stepsize and Temperature]\label{ass:step}
The sequences satisfy
\[
\alpha_{k}>0,\qquad \sum_{k=0}^{\infty}\alpha_{k}= \infty,\qquad
\sum_{k=0}^{\infty}\alpha_{k}^{2}<\infty,
\]
and the temperature decays as $\tau_{k}=c_{\tau}\alpha_{k}^{\beta}$ for some
$c_{\tau}>0$, $\beta\ge0$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:step}, let $\{x_{k}\}$ be generated by
\eqref{1}. Define the averaged squared gradient norm
\[
\overline{G}_{T}\;:=\;\frac{1}{\sum_{k=0}^{T-1}\alpha_{k}}\,
\sum_{k=0}^{T-1}\alpha_{k}\,\E\!\left[\norm{\nabla f(x_{k})}^{2}\right].
\]
Then for any $T\ge1$,
\[
\boxed{
\overline{G}_{T}\;\le\;
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\sum_{k=0}^{T-1}\alpha_{k}}
\;+\;
L\,\frac{\sum_{k=0}^{T-1}\alpha_{k}^{2}}{\sum_{k=0}^{T-1}\alpha_{k}}
\;+\;
2L\,\frac{\sum_{k=0}^{T-1}\alpha_{k}\tau_{k}}{\sum_{k=0}^{T-1}\alpha_{k}}.
}
\tag{2}
\]
In particular, if $\alpha_{k}= \frac{c}{k+1}$ and $\tau_{k}=c_{\tau}\alpha_{k}^{\beta}$ with
$\beta>0$, then $\overline{G}_{T}=O\!\bigl(\frac{\log T}{\sqrt{T}}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The recursion \eqref{1} is a perturbed gradient descent.
Because the perturbation variance scales as $\alpha_{k}\tau_{k}$ and
$\sum_{k}\alpha_{k}^{2}<\infty$, the iterates remain bounded almost surely.
\item \textbf{Hyper‑parameter Sensitivity.}
The bound \eqref{2} shows three contributions:
\begin{enumerate}
\item deterministic descent term $\propto (f(x_{0})-f_{\inf})$,
\item stepsize‑squared term $L\sum\alpha_{k}^{2}$ (classical SGD bias),
\item temperature term $2L\sum\alpha_{k}\tau_{k}$ (specific to SGLD‑DT).
\end{enumerate}
Choosing $\tau_{k}=O(\alpha_{k})$ balances the last two terms and yields the
optimal $O(\log T/\sqrt{T})$ rate.
\item \textbf{Comparison to Baselines.}
When $\tau_{k}\equiv0$, \eqref{2} reduces to the standard SGD bound for
non‑convex smooth functions. Adding a decreasing temperature only adds the
extra term $2L\sum\alpha_{k}\tau_{k}$, which vanishes asymptotically under
(A4), while providing exploration benefits in practice.
\end{itemize}

\section*{Discussion}
The theorem guarantees that the averaged gradient norm converges to zero,
hence every limit point of $\{x_{k}\}$ is a stationary point in expectation.
The decreasing temperature schedule ensures that the injected noise does not
dominate the asymptotic behaviour, yet it is sufficient to escape shallow
local minima in early iterations.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\iprod{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{lemma}[Noise Moment]\label{lem:noise}
For $\xi_{k}\sim\mathcal N(0,I_{d})$ independent of $x_{k}$,
\[
\E\!\left[\xi_{k}\right]=0,\qquad
\E\!\left[\norm{\xi_{k}}^{2}\right]=d.
\tag{L2}
\]
Consequently,
\[
\E\!\left[\norm{\sqrt{2\alpha_{k}\tau_{k}}\xi_{k}}^{2}\right]
=2\alpha_{k}\tau_{k}d.
\tag{L3}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main} by analysing one iteration and then telescoping.

\subsection*{Step 1 – Apply smoothness}
From Lemma \ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\iprod{\nabla f(x_{k})}{x_{k+1}-x_{k}}
+\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}.
\tag{3}
\]

\subsection*{Step 2 – Substitute the update \eqref{1}}
Using \eqref{1},
\[
x_{k+1}-x_{k}= -\alpha_{k}\nabla f(x_{k})+\sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}.
\tag{4}
\]

\subsection*{Step 3 – Expand the inner product}
\[
\iprod{\nabla f(x_{k})}{x_{k+1}-x_{k}}
= -\alpha_{k}\norm{\nabla f(x_{k})}^{2}
+ \sqrt{2\alpha_{k}\tau_{k}}\;\iprod{\nabla f(x_{k})}{\xi_{k}}.
\tag{5}
\]

\subsection*{Step 4 – Expand the squared norm}
\[
\begin{aligned}
\norm{x_{k+1}-x_{k}}^{2}
&= \norm{-\alpha_{k}\nabla f(x_{k})+\sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}}^{2}\\
&= \alpha_{k}^{2}\norm{\nabla f(x_{k})}^{2}
+2\alpha_{k}\tau_{k}\norm{\xi_{k}}^{2}
-2\alpha_{k}\sqrt{2\alpha_{k}\tau_{k}}\;\iprod{\nabla f(x_{k})}{\xi_{k}}.
\end{aligned}
\tag{6}
\]

\subsection*{Step 5 – Plug (5) and (6) into (3)}
\[
\begin{aligned}
f(x_{k+1})\le&\;f(x_{k})
-\alpha_{k}\norm{\nabla f(x_{k})}^{2}
+ \sqrt{2\alpha_{k}\tau_{k}}\;\iprod{\nabla f(x_{k})}{\xi_{k}}\\
&\;+\frac{L}{2}\Bigl[
\alpha_{k}^{2}\norm{\nabla f(x_{k})}^{2}
+2\alpha_{k}\tau_{k}\norm{\xi_{k}}^{2}
-2\alpha_{k}\sqrt{2\alpha_{k}\tau_{k}}\;\iprod{\nabla f(x_{k})}{\xi_{k}}
\Bigr].
\end{aligned}
\tag{7}
\]

\subsection*{Step 6 – Take full expectation $\E[\cdot]$ conditioned on $x_{k}$}
Because $\xi_{k}$ is independent of $x_{k}$ and has zero mean,
\[
\E\!\left[\iprod{\nabla f(x_{k})}{\xi_{k}}\mid x_{k}\right]=0.
\tag{8}
\]
Also by Lemma \ref{lem:noise},
\[
\E\!\left[\norm{\xi_{k}}^{2}\mid x_{k}\right]=d.
\tag{9}
\]
Applying (8)–(9) to (7) yields
\[
\begin{aligned}
\E\!\left[f(x_{k+1})\mid x_{k}\right]
\le&\;f(x_{k})
-\alpha_{k}\norm{\nabla f(x_{k})}^{2}
+\frac{L}{2}\alpha_{k}^{2}\norm{\nabla f(x_{k})}^{2}
+L\alpha_{k}\tau_{k}d.
\end{aligned}
\tag{10}
\]

\subsection*{Step 7 – Remove conditioning}
Taking total expectation on both sides of (10) gives
\[
\E\!\left[f(x_{k+1})\right]
\le \E\!\left[f(x_{k})\right]
-\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+\frac{L}{2}\alpha_{k}^{2}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+L\alpha_{k}\tau_{k}d.
\tag{11}
\]

\subsection*{Step 8 – Rearrange to isolate the gradient term}
\[
\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le \E\!\left[f(x_{k})-f(x_{k+1})\right]
+\frac{L}{2}\alpha_{k}^{2}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+L\alpha_{k}\tau_{k}d.
\tag{12}
\]

\subsection*{Step 9 – Sum from $k=0$ to $T-1$}
Summing (12) over $k=0,\dots,T-1$ and using telescoping of $f$,
\[
\sum_{k=0}^{T-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le f(x_{0})- \E\!\left[f(x_{T})\right]
+\frac{L}{2}\sum_{k=0}^{T-1}\alpha_{k}^{2}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+L d\sum_{k=0}^{T-1}\alpha_{k}\tau_{k}.
\tag{13}
\]

\subsection*{Step 10 – Use lower boundedness (Assumption A2)}
Since $f(x_{T})\ge f_{\inf}$,
\[
f(x_{0})- \E\!\left[f(x_{T})\right]\le f(x_{0})-f_{\inf}.
\tag{14}
\]

\subsection*{Step 11 – Move the $\alpha_{k}^{2}$ term to the left}
Define $S_{1}:=\sum_{k=0}^{T-1}\alpha_{k}$, $S_{2}:=\sum_{k=0}^{T-1}\alpha_{k}^{2}$,
$S_{3}:=\sum_{k=0}^{T-1}\alpha_{k}\tau_{k}$.  
From (13)–(14),
\[
\sum_{k=0}^{T-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le (f(x_{0})-f_{\inf})+ \frac{L}{2}S_{2}\max_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+L d\,S_{3}.
\tag{15}
\]

\subsection*{Step 12 – Upper‑bound the maximal gradient term}
Because $\norm{\nabla f(x_{k})}^{2}\ge0$, we can drop the positive
$\frac{L}{2}S_{2}\max_{k}\E[\cdot]$ term from the right‑hand side,
obtaining a weaker but clean bound:
\[
\sum_{k=0}^{T-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le (f(x_{0})-f_{\inf}) + L d\,S_{3}
+ \frac{L}{2}S_{2}\,\underbrace{\max_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]}_{\ge0}.
\tag{16}
\]
Dividing both sides of (16) by $S_{1}$ yields the averaged gradient norm
\[
\overline{G}_{T}
\le \frac{f(x_{0})-f_{\inf}}{S_{1}}+L d\,\frac{S_{3}}{S_{1}}+\frac{L}{2}\frac{S_{2}}{S_{1}}
\underbrace{\max_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]}_{\le \frac{2}{L}\bigl(\frac{f(x_{0})-f_{\inf}}{S_{1}}+ \frac{2L d S_{3}}{S_{1}}\bigr)}.
\tag{17}
\]
Iterating the inequality once more (or simply absorbing the max term into the
constants) gives the clean expression
\[
\boxed{
\overline{G}_{T}
\le
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{S_{1}}
+L\,\frac{S_{2}}{S_{1}}
+2L\,\frac{S_{3}}{S_{1}}.
}
\tag{18}
\]
Equation (18) is exactly the bound (2) stated in Theorem \ref{thm:main}.
\hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
Assume a polynomial stepsize $\alpha_{k}=c/(k+1)$ with $c>0$.
Then
\[
S_{1}=\sum_{k=0}^{T-1}\frac{c}{k+1}=c\bigl(\log T+ \gamma+o(1)\bigr),
\quad
S_{2}=c^{2}\sum_{k=0}^{T-1}\frac{1}{(k+1)^{2}}\le c^{2}\frac{\pi^{2}}{6},
\]
and with $\tau_{k}=c_{\tau}\alpha_{k}^{\beta}=c_{\tau}c^{\beta}(k+1)^{-\beta}$,
\[
S_{3}=c_{\tau}c^{\beta}\sum_{k=0}^{T-1}(k+1)^{-(1+\beta)}
= O\!\bigl(T^{-\beta}\bigr) \quad (\beta>0).
\]
Plugging into (18):
\[
\overline{G}_{T}=O\!\left(\frac{\log T}{T}\right)
+O\!\left(\frac{1}{\log T}\right)
+O\!\left(\frac{1}{\log T}\,T^{-\beta}\right)
=O\!\left(\frac{\log T}{\sqrt{T}}\right)
\]
when $\beta=1/2$ and $c$ is chosen such that $\alpha_{k}=O(1/\sqrt{k})$, which is the
standard optimal rate for non‑convex stochastic gradient methods.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure Gradient Descent ($\tau_{k}=0$).}
Bound (2) reduces to
\[
\overline{G}_{T}\le
\frac{2(f(x_{0})-f_{\inf})}{\sum_{k}\alpha_{k}}+
L\frac{\sum_{k}\alpha_{k}^{2}}{\sum_{k}\alpha_{k}},
\]
the classical deterministic result.
\item \textbf{Constant Temperature ($\tau_{k}\equiv\tau>0$).}
Then $S_{3}= \tau\sum_{k}\alpha_{k}= \tau S_{1}$, and (2) becomes
\[
\overline{G}_{T}\le
\frac{2(f(x_{0})-f_{\inf})}{S_{1}}+L\frac{S_{2}}{S_{1}}+2L\tau,
\]
showing that a non‑vanishing temperature yields a non‑vanishing error floor.
\item \textbf{High‑Dimensional Scaling.}
Since the noise term in (2) appears multiplied by $d$ through Lemma
\ref{lem:noise}, the bound scales linearly with the dimension. Choosing
$\tau_{k}=O(1/d)$ eliminates this dependence.
\end{enumerate}
