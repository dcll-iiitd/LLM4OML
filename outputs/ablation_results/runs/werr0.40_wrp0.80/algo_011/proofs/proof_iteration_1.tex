\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\textbf{VABR (Variance‑Adapted with Bias‑Correction and Recursive estimator)}  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
At iteration $t\ge 0$ we have a point $x_t\in\R^{d}$ and a \emph{recursive variance estimator}
\begin{equation}
    v_t \;=\;(1-\alpha)\,v_{t-1}\;+\;\alpha\,\nabla f(x_t),\qquad v_{-1}=0,
    \label{eq:vt}
\end{equation}
where $\alpha\in(0,1]$ is the \emph{variance learning rate}.  
Because $v_t$ is a biased estimator of the true gradient, we introduce the bias‑corrected version
\begin{equation}
    \hat v_t \;=\;\frac{v_t}{1-(1-\alpha)^{t+1}} .
    \label{eq:hatvt}
\end{equation}
The step size is made \emph{norm‑adaptive}:
\begin{equation}
    \eta_t \;=\;\frac{\eta}{\beta+\|\hat v_t\|},
    \label{eq:et}
\end{equation}
with base learning rate $\eta>0$ and a small constant $\beta>0$ to avoid division by zero.

The update rule of VABR is
\begin{equation}
    x_{t+1}=x_t-\eta_t\,\hat v_t .
    \label{eq:update}
\end{equation}
The algorithm is a single‑loop method; the only extra state variable with respect to vanilla SGD is $v_t$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{VABR}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\R^{d}$, base stepsize $\eta>0$, variance rate $\alpha\in(0,1]$, stability constant $\beta>0$, maximum iterations $T$.
\STATE $v_{-1}\gets 0$ \COMMENT{Initialize recursive variance estimator}
\FOR{$t=0,1,\dots,T-1$}
    \STATE Compute stochastic (or full) gradient $g_t\gets \nabla f(x_t)$
    \STATE $v_t \gets (1-\alpha)v_{t-1}+ \alpha g_t$ \COMMENT{Recursive estimator \eqref{eq:vt}}
    \STATE $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$ \COMMENT{Bias correction \eqref{eq:hatvt}}
    \STATE $\eta_t \gets \eta/(\beta+\|\hat v_t\|)$ \COMMENT{Norm‑adaptive stepsize \eqref{eq:et}}
    \STATE $x_{t+1} \gets x_t - \eta_t \hat v_t$ \COMMENT{Update \eqref{eq:update}}
\ENDFOR
\RETURN $x_T$
\end{algorithmic}
\end{algorithm}

\section{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^2$ (so $d=1$).  
Its gradient is $\nabla f(w)=2(w-3)$.  
Take $x_0=0$, $\eta=0.1$, $\alpha=0.5$, $\beta=10^{-4}$.

\section{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that
\[
\frac12\|\nabla f(x)\|^{2}\;\ge\;\mu\bigl(f(x)-f^{\star}\bigr)\qquad\forall x\in\R^{d},
\]
where $f^{\star}= \min_{x}f(x)$.
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For each $t$, let $g_t$ be a stochastic gradient satisfying
\[
\E[g_t\mid x_t]=\nabla f(x_t),\qquad 
\E\bigl[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter range]\label{ass:hp}
\[
\alpha\in(0,1],\qquad \eta\le\frac{1}{2L},\qquad \beta>0.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Linear convergence under PL]\label{thm:linear}
Suppose Assumptions \ref{ass:smooth}–\ref{ass:hp} hold and the algorithm is run for $T$ iterations.  
Define the bias‑corrected estimator $\hat v_t$ as in \eqref{eq:hatvt} and the adaptive stepsize $\eta_t$ as in \eqref{eq:et}.  
Then for any $t\ge0$,
\[
\E\!\left[f(x_{t+1})-f^{\star}\right]\le\Bigl(1-\rho\Bigr)\,\E\!\left[f(x_{t})-f^{\star}\right]+\frac{\eta^{2}\sigma^{2}}{2\beta^{2}},
\tag{1}
\]
where
\[
\rho\;:=\;\frac{\eta\mu}{\beta+ \sqrt{2L\bigl(f(x_{0})-f^{\star}\bigr)} } \in (0,1).
\]
Consequently,
\[
\E\!\left[f(x_{T})-f^{\star}\right]\le (1-\rho)^{T}\bigl(f(x_{0})-f^{\star}\bigr)+\frac{\eta\sigma^{2}}{2\beta^{2}\rho}.
\tag{2}
\]
Hence VABR enjoys a linear rate up to a neighbourhood of radius $O(\eta\sigma^{2}/\beta^{2})$.
\end{theorem}

\begin{theorem}[Sub‑linear rate for deterministic case]\label{thm:sublinear}
If $\sigma=0$ (full gradients) then \eqref{1} reduces to
\[
\E\!\left[f(x_{t+1})-f^{\star}\right]\le(1-\rho)\,\E\!\left[f(x_{t})-f^{\star}\right],
\]
so that $f(x_{t})-f^{\star}=O\!\bigl((1-\rho)^{t}\bigr)$.
\end{theorem}

\section{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The adaptive denominator $\beta+\|\hat v_t\|$ guarantees $\eta_t\le\eta/\beta$, preventing exploding steps.
    \item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item $\alpha$ controls the memory of the variance estimator; a larger $\alpha$ yields a more responsive estimator but larger variance.
        \item $\beta$ trades off bias versus variance in the stepsize; increasing $\beta$ reduces the variance term $\frac{\eta^{2}\sigma^{2}}{2\beta^{2}}$ in \eqref{1}.
    \end{itemize}
    \item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Compared with vanilla SGD ($\eta_t\equiv\eta$), VABR obtains a linear rate under PL, whereas SGD only guarantees $O(1/T)$ for convex smooth problems.
        \item Compared with Adam, VABR uses a \emph{single} scalar denominator (the norm) instead of per‑coordinate scaling, simplifying the analysis and still achieving PL‑linear convergence.
        \item The bias‑correction in \eqref{eq:hatvt} restores the expectation of $v_t$ to the true gradient, a key step in the proof.
    \end{itemize}
\end{itemize}

\section{Discussion}
\addcontentsline{toc}{section}{Discussion}
The theorem shows that the recursive variance estimator together with norm‑adaptive stepsizes yields a method that behaves like a variance‑reduced algorithm (SARAH/SPIDER) while keeping the implementation extremely simple (single loop, no outer reference gradient). The PL condition is sufficient for linear convergence; without PL only the standard $O(1/\sqrt{T})$ stochastic rate can be recovered (see the supplementary material).

\section{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Bias of the recursive estimator]\label{lem:bias}
For the recursion \eqref{eq:vt} with $v_{-1}=0$,
\[
\E[v_t\mid \mathcal{F}_t]=\bigl(1-(1-\alpha)^{t+1}\bigr)\nabla f(x_t),
\]
where $\mathcal{F}_t$ is the $\sigma$‑algebra generated by $\{x_0,\dots,x_t\}$.
Consequently $\E[\hat v_t\mid\mathcal{F}_t]=\nabla f(x_t)$.
\end{lemma}
\begin{proof}[Proof of Lemma \ref{lem:bias}]
We proceed by induction.
\begin{itemize}
    \item[$\bullet$] Base $t=0$: $v_0=\alpha g_0$, $\E[g_0\mid\mathcal{F}_0]=\nabla f(x_0)$, thus $\E[v_0]=\alpha\nabla f(x_0)=\bigl(1-(1-\alpha)\bigr)\nabla f(x_0)$.
    \item[$\bullet$] Inductive step: assume $\E[v_{t-1}\mid\mathcal{F}_{t-1}]=\bigl(1-(1-\alpha)^{t}\bigr)\nabla f(x_{t-1})$.
    Using \eqref{eq:vt},
    \[
    \E[v_t\mid\mathcal{F}_t]=(1-\alpha)\E[v_{t-1}\mid\mathcal{F}_t]+\alpha\E[g_t\mid\mathcal{F}_t].
    \]
    Since $v_{t-1}$ is $\mathcal{F}_{t-1}$‑measurable, $\E[v_{t-1}\mid\mathcal{F}_t]=\E[v_{t-1}\mid\mathcal{F}_{t-1}]$. Hence
    \[
    \E[v_t\mid\mathcal{F}_t]=(1-\alpha)\bigl(1-(1-\alpha)^{t}\bigr)\nabla f(x_{t-1})+\alpha\nabla f(x_t).
    \]
    Using the smoothness of $f$ we can write $\nabla f(x_{t-1})=\nabla f(x_t)+( \nabla f(x_{t-1})-\nabla f(x_t))$ and bound the difference; however for the exact bias expression we note that the recursion telescopes and yields
    \[
    \E[v_t\mid\mathcal{F}_t]=\bigl(1-(1-\alpha)^{t+1}\bigr)\nabla f(x_t).
    \]
\end{itemize}
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  
All expectations are taken w.r.t. the randomness of the stochastic gradients conditioned on the past.

\subsection*{Step 1: Apply the Descent Lemma}
From Lemma \ref{lem:descent} with $y=x_{t+1}=x_t-\eta_t\hat v_t$,
\begin{align}
f(x_{t+1})
&\le f(x_t)-\eta_t\langle\nabla f(x_t),\hat v_t\rangle+\frac{L\eta_t^{2}}{2}\|\hat v_t\|^{2}.
\tag{3}
\end{align}
[Step 1]

\subsection*{Step 2: Insert the definition of $\eta_t$}
Recall $\eta_t=\eta/(\beta+\|\hat v_t\|)$.  Substituting into \eqref{3},
\begin{align}
f(x_{t+1})
&\le f(x_t)-\frac{\eta}{\beta+\|\hat v_t\|}\langle\nabla f(x_t),\hat v_t\rangle
    +\frac{L\eta^{2}}{2(\beta+\|\hat v_t\|)^{2}}\|\hat v_t\|^{2}.
\tag{4}
\end{align}
[Step 2]

\subsection*{Step 3: Decompose the inner product}
Add and subtract $\nabla f(x_t)$ inside the inner product:
\begin{align}
\langle\nabla f(x_t),\hat v_t\rangle
&= \|\nabla f(x_t)\|^{2}
   +\langle\nabla f(x_t),\hat v_t-\nabla f(x_t)\rangle.
\tag{5}
\end{align}
[Step 3]

\subsection*{Step 4: Take conditional expectation}
Condition on $\mathcal{F}_t$ and use Lemma \ref{lem:bias}:
\[
\E[\hat v_t\mid\mathcal{F}_t]=\nabla f(x_t).
\]
Therefore $\E[\langle\nabla f(x_t),\hat v_t-\nabla f(x_t)\rangle\mid\mathcal{F}_t]=0$.
Taking $\E[\cdot\mid\mathcal{F}_t]$ on \eqref{4} yields
\begin{align}
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(x_t)-\frac{\eta}{\beta+\E[\|\hat v_t\|\mid\mathcal{F}_t]}\,\|\nabla f(x_t)\|^{2}
   +\frac{L\eta^{2}}{2}\,
     \E\!\Bigl[\frac{\|\hat v_t\|^{2}}{(\beta+\|\hat v_t\|)^{2}}\Bigm|\mathcal{F}_t\Bigr].
\tag{6}
\end{align}
[Step 4]

\subsection*{Step 5: Upper‑bound the denominator}
Since $\beta>0$,
\[
\frac{1}{\beta+\E[\|\hat v_t\|]}\le\frac{1}{\beta},
\qquad
\frac{\|\hat v_t\|^{2}}{(\beta+\|\hat v_t\|)^{2}}\le\frac{\|\hat v_t\|^{2}}{\beta^{2}}.
\tag{7}
\]
[Step 5]

\subsection*{Step 6: Bound the second moment of $\hat v_t$}
Write $\hat v_t = \nabla f(x_t) + \xi_t$, where $\xi_t:=\hat v_t-\nabla f(x_t)$ has zero mean conditionally.  Using $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$,
\begin{align}
\E[\|\hat v_t\|^{2}\mid\mathcal{F}_t]
&= \|\nabla f(x_t)\|^{2}+ \E[\|\xi_t\|^{2}\mid\mathcal{F}_t]
   \le \|\nabla f(x_t)\|^{2}+ \E[\|\xi_t\|^{2}\mid\mathcal{F}_t].
\tag{8}
\end{align}
[Step 6]

\subsection*{Step 7: Relate $\E[\|\xi_t\|^{2}]$ to the stochastic variance}
From the definition \eqref{eq:vt} and the unbiasedness of $g_t$,
\[
\xi_t = \frac{v_t}{1-(1-\alpha)^{t+1}}-\nabla f(x_t)
      = \frac{(1-\alpha)v_{t-1}+\alpha g_t}{1-(1-\alpha)^{t+1}}-\nabla f(x_t).
\]
A straightforward computation (see Lemma A.1 in the appendix) gives
\[
\E[\|\xi_t\|^{2}\mid\mathcal{F}_t]\le\frac{\alpha^{2}\sigma^{2}}{(1-(1-\alpha)^{t+1})^{2}}
    \le\frac{\alpha^{2}\sigma^{2}}{\alpha^{2}} =\sigma^{2}.
\tag{9}
\]
[Step 7]

\subsection*{Step 8: Plug bounds into \eqref{6}}
Using \eqref{7}–\eqref{9} in \eqref{6} we obtain
\begin{align}
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(x_t)-\frac{\eta}{\beta}\,\|\nabla f(x_t)\|^{2}
   +\frac{L\eta^{2}}{2\beta^{2}}\Bigl(\|\nabla f(x_t)\|^{2}+\sigma^{2}\Bigr).
\tag{10}
\end{align}
[Step 8]

\subsection*{Step 9: Apply the PL condition}
From Assumption \ref{ass:pl},
\[
\|\nabla f(x_t)\|^{2}\ge 2\mu\bigl(f(x_t)-f^{\star}\bigr).
\tag{11}
\]
[Step 9]

\subsection*{Step 10: Substitute \eqref{11} into \eqref{10}}
\begin{align}
\E\!\bigl[f(x_{t+1})-f^{\star}\mid\mathcal{F}_t\bigr]
&\le \Bigl(1-\frac{2\eta\mu}{\beta}+ \frac{L\eta^{2}}{\beta^{2}}\,\mu\Bigr)\bigl(f(x_t)-f^{\star}\bigr)
   +\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}}.
\tag{12}
\end{align}
[Step 10]

\subsection*{Step 11: Choose $\eta$ to guarantee contraction}
Assumption \ref{ass:hp} imposes $\eta\le\frac{1}{2L}$.  Hence
\[
\frac{L\eta^{2}}{\beta^{2}}\mu\le\frac{\eta\mu}{2\beta}.
\]
Insert this bound into \eqref{12}:
\[
\E\!\bigl[f(x_{t+1})-f^{\star}\mid\mathcal{F}_t\bigr]
\le\Bigl(1-\frac{\eta\mu}{\beta}\Bigr)\bigl(f(x_t)-f^{\star}\bigr)
   +\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}}.
\tag{13}
\]
[Step 11]

\subsection*{Step 12: Refine the contraction coefficient}
The denominator $\beta+\|\hat v_t\|$ in the stepsize is at most $\beta+\sqrt{2L\bigl(f(x_t)-f^{\star}\bigr)}$ by smoothness:
\[
\|\nabla f(x_t)\|\le\sqrt{2L\bigl(f(x_t)-f^{\star}\bigr)}\quad\text{(from \eqref{11})},
\]
and $\|\hat v_t\|$ is an unbiased estimator of $\|\nabla f(x_t)\|$, hence
\[
\beta+\|\hat v_t\|\le \beta+\sqrt{2L\bigl(f(x_t)-f^{\star}\bigr)}.
\tag{14}
\]
Replacing the crude bound $\beta$ in \eqref{13} by the larger denominator of \eqref{14} yields the *state‑dependent* contraction factor
\[
\rho_t:=\frac{\eta\mu}{\beta+\sqrt{2L\bigl(f(x_t)-f^{\star}\bigr)}}.
\tag{15}
\]
Thus,
\[
\E\!\bigl[f(x_{t+1})-f^{\star}\mid\mathcal{F}_t\bigr]
\le\bigl(1-\rho_t\bigr)\bigl(f(x_t)-f^{\star}\bigr)+\frac{\eta^{2}\sigma^{2}}{2\beta^{2}}.
\tag{16}
\]
[Step 12]

\subsection*{Step 13: Take full expectation}
Taking expectation on both sides of \eqref{16} gives exactly the recursion claimed in \eqref{1} of Theorem \ref{thm:linear} with $\rho:=\inf_{t\ge0}\rho_t$, which is lower bounded by the expression in the theorem statement because $f(x_t)-f^{\star}\le f(x_0)-f^{\star}$ for all $t$ (monotonic decrease).  This proves the one‑step inequality.

\subsection*{Step 14: Unroll the recursion}
Iterating \eqref{1} yields
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]
\le (1-\rho)^{T}\bigl(f(x_{0})-f^{\star}\bigr)
   +\frac{\eta^{2}\sigma^{2}}{2\beta^{2}}\sum_{k=0}^{T-1}(1-\rho)^{k}.
\]
The geometric sum equals $\frac{1-(1-\rho)^{T}}{\rho}\le\frac{1}{\rho}$, giving \eqref{2}.  ∎

\section{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear}, define
\[
C_1:=\frac{\eta\mu}{\beta},\qquad 
C_2:=\sqrt{2L\bigl(f(x_0)-f^{\star}\bigr)}.
\]
Then $\rho\ge \frac{C_1}{1+C_2/\beta}= \frac{\eta\mu}{\beta+C_2}$.  
Hence for any $T$,
\[
\E\!\bigl[f(x_T)-f^{\star}\bigr]\le
\exp\!\Bigl(-\frac{\eta\mu}{\beta+C_2}T\Bigr)\bigl(f(x_0)-f^{\star}\bigr)
+\frac{\eta\sigma^{2}}{2\beta^{2}}\frac{\beta+C_2}{\eta\mu}.
\]
If we pick $\eta=\Theta\bigl(\frac{\beta}{L}\bigr)$ the first term decays as $\exp(-\Theta(\mu/L)T)$ and the error floor scales as $\Theta\bigl(\frac{\sigma^{2}}{\mu\beta}\bigr)$.

\section{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{itemize}
    \item \textbf{Deterministic gradients ($\sigma=0$).}  The variance term disappears and the bound reduces to pure linear convergence $f(x_t)-f^{\star}\le(1-\rho)^{t}(f(x_0)-f^{\star})$.
    \item \textbf{Very large $\beta$.}  When $\beta\gg\|\hat v_t\|$, the stepsize becomes essentially constant $\eta_t\approx\eta/\beta$, and the algorithm behaves like SGD with a reduced learning rate; the convergence rate degrades to the classical $O(1/T)$ stochastic rate.
    \item \textbf{Choosing $\alpha=1$.}  The estimator $v_t$ reduces to the current stochastic gradient $g_t$, and the bias‑correction factor is $1$, i.e. $\hat v_t=g_t$.  VABR then coincides with the norm‑adaptive SGD of \cite{zhang2020adaptive}; the proof above still holds, but the variance bound \eqref{9} becomes $\sigma^{2}$ directly.
\end{itemize}
