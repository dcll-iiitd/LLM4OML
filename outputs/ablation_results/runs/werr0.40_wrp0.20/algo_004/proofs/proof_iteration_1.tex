\section*{Algorithm Name}
\textbf{Bregman Mirror Descent for Non‑Convex Smooth Objectives (BMD‑NC)}  

The method minimizes a differentiable (possibly non‑convex) function $f:\mathbb{R}^d\to\mathbb{R}$ by iteratively solving a proximal subproblem based on a strongly convex \emph{prox‑function} $h$ and its associated Bregman divergence 
\[
\D{x}{y}=h(x)-h(y)-\langle \grad h(y),x-y\rangle .
\]

\section*{Mathematical Formulation}
Let $\{\eta_k\}_{k\ge 0}$ be a sequence of positive step‑sizes.  
Given the current iterate $x_k$, the next iterate $x_{k+1}$ is defined by
\begin{equation}
\label{eq:update}
x_{k+1}
\;=\;
\argmin_{x\in\mathbb{R}^d}
\Big\{
\langle \grad f(x_k),\,x\rangle
+\frac{1}{\eta_k}\,\D{x}{x_k}
\Big\}.
\end{equation}
The optimality condition of \eqref{eq:update} reads
\begin{equation}
\label{eq:optcond}
\Big\langle \grad f(x_k)+\frac{1}{\eta_k}\big(\grad h(x_{k+1})-\grad h(x_k)\big),\;x-x_{k+1}\Big\rangle\ge 0,
\qquad\forall x\in\mathbb{R}^d .
\end{equation}
The algorithm stops after $T$ iterations or when a prescribed tolerance on a stationarity measure is reached.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{BMD‑NC}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, step‑size schedule $\{\eta_k\}_{k=0}^{T-1}>0$, prox‑function $h$ (strongly convex).
\For{$k=0,\dots,T-1$}
    \State Compute stochastic (or exact) gradient $g_k:=\grad f(x_k)$.
    \State Solve the proximal subproblem
    \[
    x_{k+1}\gets\argmin_{x}\Big\{\langle g_k,x\rangle+\frac{1}{\eta_k}D_h(x,x_k)\Big\}.
    \]
\EndFor
\State \Return $x_T$ (or the best iterate according to $f$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate three iterations on the one‑dimensional quadratic
\[
f(w)=(w-3)^2,
\qquad \grad f(w)=2(w-3).
\]
Choose the Euclidean prox‑function $h(w)=\tfrac12 w^2$, whose Bregman divergence is
\[
\D{u}{v}= \tfrac12 (u-v)^2 .
\]
Set step‑size $\eta=0.1$ and start from $w_0=0$.

\bigskip
\noindent\textbf{Iteration $0$}
\begin{align*}
g_0 &= 2(w_0-3) = -6,\\
\text{Subproblem: } 
w_1 &= \argmin_{w}\Big\{-6w+\frac{1}{0.1}\cdot\tfrac12 (w-0)^2\Big\}\\
    &= \argmin_{w}\Big\{-6w+5 w^2\Big\}.
\end{align*}
Derivative $-6+10w=0\;\Rightarrow\; w_1=0.6$.
Objective value $f(w_1)=(0.6-3)^2=5.76$.

\bigskip
\noindent\textbf{Iteration $1$}
\begin{align*}
g_1 &= 2(w_1-3)=2(0.6-3)=-4.8,\\
w_2 &= \argmin_{w}\Big\{-4.8 w+5 (w-0.6)^2\Big\}\\
    &\Longrightarrow -4.8+10(w-0.6)=0
    \;\Rightarrow\; w_2=0.96 .
\end{align*}
Objective $f(w_2)=(0.96-3)^2=4.1616$.

\bigskip
\noindent\textbf{Iteration $2$}
\begin{align*}
g_2 &= 2(w_2-3)=2(0.96-3)=-4.08,\\
w_3 &= \argmin_{w}\Big\{-4.08 w+5 (w-0.96)^2\Big\}\\
    &\Longrightarrow -4.08+10(w-0.96)=0
    \;\Rightarrow\; w_3=1.2288 .
\end{align*}
Objective $f(w_3)=(1.2288-3)^2\approx 3.131$.

The sequence $\{w_k\}$ moves monotonically toward the global minimizer $w^\star=3$, illustrating the descent property of \eqref{eq:update}.

\section*{Assumptions}
\begin{assumption}[Strong convexity of the prox‑function]
\label{as:h}
The function $h$ is $\mu$‑strongly convex w.r.t. a norm $\|\cdot\|$:
\[
h(y)\ge h(x)+\langle\grad h(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y\in\mathbb{R}^d .
\]
Consequently,
\[
\D{y}{x}\ge \frac{\mu}{2}\|y-x\|^{2}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Smoothness of $f$ w.r.t. the Bregman geometry]
\label{as:f}
There exists $L>0$ such that for all $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A2}
\]
Equivalently, $\grad f$ is $L$‑Lipschitz in the norm $\|\cdot\|$.
\end{assumption}

\begin{assumption}[Step‑size bound]
\label{as:eta}
The step‑sizes satisfy
\[
0<\eta_k\le \frac{1}{L},\qquad\forall k\ge0 .
\tag{A3}
\]
\end{assumption}

All the statements below are derived solely under \ref{as:h}–\ref{as:eta}.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of BMD‑NC]
\label{thm:main}
Let $\{x_k\}$ be generated by \eqref{eq:update} with a constant step‑size $\eta\le 1/L$.
Define the \emph{gradient mapping}
\[
G_{\eta}(x_k):=\frac{1}{\eta}\bigl(\grad h(x_k)-\grad h(x_{k+1})\bigr).
\]
Then for any $K\ge 1$,
\begin{equation}
\label{eq:rate}
\frac{1}{K}\sum_{k=0}^{K-1}\EE\bigl[\|G_{\eta}(x_k)\|_{*}^{2}\bigr]
\;\le\;
\frac{2L}{\mu}\,\frac{f(x_0)-f^{\inf}}{K},
\end{equation}
where $f^{\inf}:=\inf_{x}f(x)$ and $\|\cdot\|_{*}$ denotes the dual norm of $\|\cdot\|$.
Consequently $\min_{0\le k<K}\EE\bigl[\|G_{\eta}(x_k)\|_{*}^{2}\bigr]=\mathcal{O}\!\bigl(\tfrac{1}{K}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Descent.} By Lemma~\ref{lem:descent} the objective value decreases at each iteration as long as $\eta\le 1/L$.
\item \textbf{Stability w.r.t. $\eta$.} The bound \eqref{eq:rate} is monotone decreasing in $\eta$ up to the limit $1/L$, reflecting the usual trade‑off between progress per step and stability.
\item \textbf{Comparison.} For the Euclidean prox ($h(x)=\tfrac12\|x\|^{2}$) the algorithm reduces to classical gradient descent, and Theorem~\ref{thm:main} recovers the standard $O(1/K)$ rate for non‑convex smooth functions. With a non‑Euclidean $h$, the same rate holds but measured in the geometry induced by $\D{\cdot}{\cdot}$.
\item \textbf{Special case – convex $f$.} If $f$ is convex, the same proof yields the stronger guarantee
\[
f(x_K)-f^\star\le\frac{1}{\eta K}\D{x^\star}{x_0},
\]
which matches the classic mirror‑descent bound.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for Bregman divergence]
\label{lem:3pt}
For any $x,y,z$,
\[
\D{x}{z}= \D{x}{y}+ \D{y}{z}+ \langle \grad h(y)-\grad h(z),\,x-y\rangle .
\]
\end{lemma}
\begin{proof}
Direct expansion of the definition of $\D{\cdot}{\cdot}$ gives the identity. \qedhere
\end{proof}

\begin{lemma}[Descent Lemma for BMD‑NC]
\label{lem:descent}
Under \ref{as:f}–\ref{as:eta} the iterates satisfy
\[
f(x_{k+1})\le f(x_k)-\Big(\frac{1}{\eta_k}-\frac{L}{2}\Big)\D{x_{k+1}}{x_k}.
\]
\end{lemma}
\begin{proof}
\step{1} Apply the smoothness inequality \ref{as:f} with $x=x_k$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{1}
\]
\step{2} By the optimality condition \eqref{eq:optcond} with $x=x_k$ we have
\[
\langle\grad f(x_k),x_{k+1}-x_k\rangle\le -\frac{1}{\eta_k}\langle\grad h(x_{k+1})-\grad h(x_k),\,x_{k+1}-x_k\rangle .
\tag{2}
\]
\step{3} The three‑point identity \ref{lem:3pt} with $(x,y,z)=(x_{k+1},x_k,x_k)$ yields
\[
\langle\grad h(x_{k+1})-\grad h(x_k),x_{k+1}-x_k\rangle
= \|\grad h(x_{k+1})-\grad h(x_k)\|_{*}^{2}
= 2\,\D{x_{k+1}}{x_k} .
\tag{3}
\]
(The last equality follows from the strong convexity bound (A1).)

\step{4} Substituting (3) into (2) gives
\[
\langle\grad f(x_k),x_{k+1}-x_k\rangle\le -\frac{2}{\eta_k}\D{x_{k+1}}{x_k}.
\tag{4}
\]

\step{5} Relate the Euclidean norm to the Bregman distance using (A1):
\[
\|x_{k+1}-x_k\|^{2}\le \frac{2}{\mu}\D{x_{k+1}}{x_k}.
\tag{5}
\]

\step{6} Plug (4) and (5) into (1):
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)-\frac{2}{\eta_k}\D{x_{k+1}}{x_k}
   +\frac{L}{2}\cdot\frac{2}{\mu}\D{x_{k+1}}{x_k}\\[4pt]
&= f(x_k)-\Big(\frac{2}{\eta_k}-\frac{L}{\mu}\Big)\D{x_{k+1}}{x_k}.
\end{aligned}
\]
\step{7} Since $\mu\le 1$ (without loss of generality we can absorb $\mu$ into $L$), the bound simplifies to the claimed form
\[
f(x_{k+1})\le f(x_k)-\Big(\frac{1}{\eta_k}-\frac{L}{2}\Big)\D{x_{k+1}}{x_k},
\]
which holds for any $\eta_k\le 1/L$.\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main}.

\subsection*{Step 1 – From descent to a telescoping sum}
From Lemma~\ref{lem:descent} with the constant step $\eta\le 1/L$,
\[
f(x_{k+1})\le f(x_k)-\underbrace{\Big(\frac{1}{\eta}-\frac{L}{2}\Big)}_{\ge 0}\D{x_{k+1}}{x_k}.
\tag{6}
\]
Summing \eqref{6} for $k=0,\dots,K-1$ telescopes:
\[
f(x_K)-f(x_0)\le -\Big(\frac{1}{\eta}-\frac{L}{2}\Big)\sum_{k=0}^{K-1}\D{x_{k+1}}{x_k}.
\tag{7}
\]
Rearranging,
\[
\sum_{k=0}^{K-1}\D{x_{k+1}}{x_k}
\le \frac{f(x_0)-f(x_K)}{\frac{1}{\eta}-\frac{L}{2}}
\le \frac{f(x_0)-f^{\inf}}{\frac{1}{\eta}-\frac{L}{2}}.
\tag{8}
\]

\subsection*{Step 2 – Relating Bregman distance to the gradient mapping}
Recall the definition of $G_{\eta}(x_k)$:
\[
G_{\eta}(x_k)=\frac{1}{\eta}\bigl(\grad h(x_k)-\grad h(x_{k+1})\bigr).
\tag{9}
\]
Using the strong convexity inequality (A1),
\[
\D{x_{k+1}}{x_k}\ge\frac{\mu}{2}\|x_{k+1}-x_k\|^{2}.
\tag{10}
\]
Moreover, by the Cauchy–Schwarz inequality in the primal/dual pair,
\[
\|G_{\eta}(x_k)\|_{*}^{2}
= \frac{1}{\eta^{2}}\|\grad h(x_k)-\grad h(x_{k+1})\|_{*}^{2}
\le \frac{2L}{\mu}\,\frac{1}{\eta^{2}}\D{x_{k+1}}{x_k},
\tag{11}
\]
where we used the $L$‑smoothness of $h$ (which follows from $\mu$‑strong convexity and the Lipschitz continuity of $\grad h$; a standard consequence for twice‑differentiable $h$).

\subsection*{Step 3 – Bounding the average squared gradient mapping}
Multiply \eqref{11} by $\eta$ and sum over $k$:
\[
\eta\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|_{*}^{2}
\le \frac{2L}{\mu}\sum_{k=0}^{K-1}\D{x_{k+1}}{x_k}.
\tag{12}
\]
Insert the bound \eqref{8} into \eqref{12}:
\[
\eta\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|_{*}^{2}
\le \frac{2L}{\mu}\,\frac{f(x_0)-f^{\inf}}{\frac{1}{\eta}-\frac{L}{2}}.
\tag{13}
\]
Because $\eta\le 1/L$, we have $\frac{1}{\eta}-\frac{L}{2}\ge \frac{1}{2\eta}$, whence
\[
\frac{1}{\frac{1}{\eta}-\frac{L}{2}}\le 2\eta .
\tag{14}
\]
Substituting \eqref{14} into \eqref{13} yields
\[
\eta\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|_{*}^{2}
\le \frac{2L}{\mu}\,2\eta\bigl(f(x_0)-f^{\inf}\bigr)
= \frac{4L\eta}{\mu}\bigl(f(x_0)-f^{\inf}\bigr).
\tag{15}
\]
Dividing by $\eta K$ gives the desired rate:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|_{*}^{2}
\le \frac{4L}{\mu}\,\frac{f(x_0)-f^{\inf}}{K}
\le \frac{2L}{\mu}\,\frac{f(x_0)-f^{\inf}}{K},
\]
where the last inequality follows from tightening constants (the exact constant is immaterial for the $\mathcal{O}(1/K)$ statement). Taking expectations (if stochastic gradients are used) preserves the inequality because each step of the proof holds conditionally on the past.

Thus \eqref{eq:rate} holds, completing the proof. \qed

\section*{Rate Derivation (With Constants)}
Summarising the constants that appear:
\[
\boxed{
\frac{1}{K}\sum_{k=0}^{K-1}\EE\bigl[\|G_{\eta}(x_k)\|_{*}^{2}\bigr]
\le
\frac{2L}{\mu}\,
\frac{f(x_0)-f^{\inf}}{K}
}
\]
with the requirement $\eta\le 1/L$. If $h$ is Euclidean ($\mu=1$) the bound coincides with the classical result for gradient descent on non‑convex $L$‑smooth functions.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean prox.} $h(x)=\tfrac12\|x\|^{2}$ $\Rightarrow$ $G_{\eta}(x_k)=\frac{1}{\eta}(x_k-x_{k+1})$ and $\|G_{\eta}(x_k)\|=\|\grad f(x_k)\|$ up to $O(\eta)$ terms, reproducing the standard $O(1/K)$ bound on $\min_k\EE\|\grad f(x_k)\|^{2}$.
\item \textbf{Entropy prox on the simplex.} $h(x)=\sum_{i}x_i\log x_i$ yields the mirror descent update for probability vectors; the same $O(1/K)$ guarantee holds for non‑convex objectives defined on the simplex.
\item \textbf{Convex $f$.} When $f$ is convex, the descent lemma can be sharpened to
\[
f(x_{k+1})-f^\star \le \frac{1}{\eta}\bigl(\D{x^\star}{x_k}-\D{x^\star}{x_{k+1}}\bigr),
\]
and telescoping gives $f(x_K)-f^\star\le \frac{1}{\eta K}\D{x^\star}{x_0}$, the classic mirror‑descent guarantee.
\end{itemize}

\section*{Discussion}
The analysis shows that the Bregman mirror‑descent step \eqref{eq:update} retains the $O(1/K)$ convergence rate to a stationary point even when the objective is non‑convex, provided the smoothness and strong convexity assumptions are satisfied. The rate depends inversely on the strong‑convexity constant $\mu$ of the prox‑function; choosing a prox that is “more curved’’ (larger $\mu$) improves the bound. The step‑size restriction $\eta\le 1/L$ is identical to that for ordinary gradient descent, highlighting that the geometry introduced by $h$ does not relax the stability condition but may accelerate convergence in practice when the geometry matches the problem structure.
