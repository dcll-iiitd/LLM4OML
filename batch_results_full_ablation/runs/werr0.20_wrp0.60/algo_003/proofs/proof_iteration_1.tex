\section*{Algorithm Name}
\textbf{Mirror Descent (MD)} – a first‑order method that uses a Bregman divergence generated by a strongly convex prox‑function $h$ as the distance‑generating function.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable objective and let $h:\R^{d}\rightarrow\R$ be a $1$‑strongly convex function w.r.t. a norm $\|\cdot\|$ (i.e.\ $h$ satisfies
\[
h(y)\ge h(x)+\langle\nabla h(x),y-x\rangle+\frac{1}{2}\|y-x\|^{2}\quad\forall x,y\in\R^{d}.
\])
Define the Bregman divergence associated with $h$:
\[
D_{h}(x,y)=h(x)-h(y)-\langle\nabla h(y),x-y\rangle .
\]

Given a stepsize sequence $\{\eta_{k}\}_{k\ge0}$, Mirror Descent generates the iterates
\begin{equation}
\boxed{
x_{k+1}= \arg\min_{x\in\R^{d}}\Big\{\langle \grad f(x_{k}),\,x\rangle
+\frac{1}{\eta_{k}}\,D_{h}(x,x_{k})\Big\}
}
\label{eq:update}
\end{equation}
Equivalently, using the optimality condition of the minimisation,
\[
\nabla h(x_{k+1}) = \nabla h(x_{k}) - \eta_{k}\,\grad f(x_{k}).
\tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}\in\R^{d}$, stepsizes $\{\eta_{k}\}_{k\ge0}$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_{k}= \grad f(x_{k})$.
    \STATE Update dual variable: $\tilde{y}_{k}= \nabla h(x_{k})-\eta_{k}g_{k}$.
    \STATE Map back to primal: $x_{k+1}= (\nabla h)^{-1}(\tilde{y}_{k})$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^{2},\qquad w\in\R,
\]
with prox‑function $h(w)=\frac{1}{2}w^{2}$ (so $D_{h}(u,v)=\frac12(u-v)^{2}$) and constant stepsize $\eta_{k}=0.1$.
Initialize $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_{k}= \grad f(w_{k})$ & $w_{k+1}= w_{k}-\eta_{k} g_{k}$ & $f(w_{k+1})$\\\hline
0 & $g_{0}=2(w_{0}-3)=-6$ & $w_{1}=0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $g_{1}=2(0.6-3)=-4.8$ & $w_{2}=0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $g_{2}=2(1.08-3)=-3.84$ & $w_{3}=1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3665$
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the descent property of \eqref{eq:update}.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The objective $f$ is convex and $L$‑smooth w.r.t. the norm $\|\cdot\|$, i.e.
\[
\|\grad f(x)-\grad f(y)\|\_{*}\le L\|x-y\|\quad\forall x,y\in\R^{d},
\]
where $\|\cdot\|\_{*}$ denotes the dual norm.
\end{assumption}

\begin{assumption}[Strong Convexity of $h$]\label{ass:hstrong}
The prox‑function $h$ is $\alpha$‑strongly convex w.r.t. $\|\cdot\|$:
\[
h(y)\ge h(x)+\langle\nabla h(x),y-x\rangle+\frac{\alpha}{2}\|y-x\|^{2}\quad\forall x,y.
\]
Consequently, the Bregman divergence satisfies
\[
D_{h}(y,x)\ge\frac{\alpha}{2}\|y-x\|^{2}.
\tag{2}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes are constant $\eta_{k}\equiv\eta$ with
\[
0<\eta\le \frac{\alpha}{L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Descent]\label{thm:md}
Under Assumptions \ref{ass:convex}–\ref{ass:stepsize}, let $x^{\star}\in\arg\min f$ and define the Bregman distance to the optimum $V_{0}=D_{h}(x^{\star},x_{0})$. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(x_{k})-f(x^{\star})\bigr)
\;\le\;
\frac{V_{0}}{\eta T}.
\tag{3}
\]
In particular, choosing $\eta=\alpha/L$ yields the $O(1/T)$ rate:
\[
f(\bar{x}_{T})-f(x^{\star})\le \frac{L\,V_{0}}{\alpha\,T},
\qquad 
\bar{x}_{T}\triangleq\frac{1}{T}\sum_{k=0}^{T-1}x_{k}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update \eqref{eq:update} is a proximal step in the dual space; strong convexity of $h$ guarantees that the iterates remain bounded as long as $V_{0}<\infty$.
\item \textbf{Hyperparameter Sensitivity.} The rate (3) depends linearly on $1/\eta$. The admissible range \eqref{ass:stepsize} prevents overshooting; the optimal choice $\eta^{\star}=\alpha/L$ balances the smoothness constant $L$ and the curvature $\alpha$ of $h$.
\item \textbf{Comparison to Gradient Descent.} When $h(x)=\frac12\|x\|_{2}^{2}$, $D_{h}$ reduces to the Euclidean squared distance and \eqref{eq:update} becomes ordinary gradient descent. The theorem recovers the classic $O(1/T)$ convergence for smooth convex functions.
\item \textbf{Extension to Stochastic Gradients.} If $g_{k}$ is an unbiased estimator of $\grad f(x_{k})$ with bounded variance, the same analysis yields an $O(1/\sqrt{T})$ bound on the expected suboptimality, analogously to SGD.
\end{itemize}

\section*{Discussion}
The Mirror Descent framework allows the algorithm to exploit geometry of the feasible set via a non‑Euclidean distance. When the feasible domain is a simplex, for instance, choosing $h$ as the negative entropy leads to the exponentiated gradient algorithm, which enjoys the same $O(1/T)$ guarantee while staying in the simplex automatically.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:threepoint}
For any $x,y,z\in\R^{d}$,
\[
\langle \nabla h(x)-\nabla h(y),\,z-y\rangle
= D_{h}(z,y)-D_{h}(z,x)-D_{h}(x,y).
\tag{4}
\]
\end{lemma}
\begin{proof}
By definition of $D_{h}$,
\[
\begin{aligned}
D_{h}(z,y)&=h(z)-h(y)-\langle\nabla h(y),z-y\rangle,\\
D_{h}(z,x)&=h(z)-h(x)-\langle\nabla h(x),z-x\rangle,\\
D_{h}(x,y)&=h(x)-h(y)-\langle\nabla h(y),x-y\rangle.
\end{aligned}
\]
Subtracting the last two from the first and rearranging yields \eqref{lem:threepoint}. ∎
\end{proof}

\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for all $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness via integration of the gradient bound. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:md}. Throughout the proof we fix an optimal point $x^{\star}\in\arg\min f$.

\noindent\textbf{[Step 1]} (Optimality condition of the MD subproblem).  
From \eqref{eq:update}, the first‑order optimality condition gives
\[
\big\langle \grad f(x_{k})+\frac{1}{\eta}\big(\nabla h(x_{k+1})-\nabla h(x_{k})\big),\,x-x_{k+1}\big\rangle\ge0,
\qquad\forall x\in\R^{d}.
\tag{6}
\]

\noindent\textbf{[Step 2]} (Plug $x=x^{\star}$).  
Setting $x=x^{\star}$ in \eqref{Step 2} yields
\[
\langle\grad f(x_{k}),\,x_{k+1}-x^{\star}\rangle
\le\frac{1}{\eta}\langle\nabla h(x_{k})-\nabla h(x_{k+1}),\,x_{k+1}-x^{\star}\rangle .
\tag{7}
\]

\noindent\textbf{[Step 3]} (Apply three‑point identity).  
Using Lemma \ref{lem:threepoint} with $(x,y,z)=(x_{k},x_{k+1},x^{\star})$,
\[
\langle\nabla h(x_{k})-\nabla h(x_{k+1}),\,x_{k+1}-x^{\star}\rangle
= D_{h}(x^{\star},x_{k+1})-D_{h}(x^{\star},x_{k})-D_{h}(x_{k},x_{k+1}).
\tag{8}
\]

\noindent\textbf{[Step 4]} (Combine (7) and (8)).  
Substituting \eqref{Step 4} into \eqref{Step 2} gives
\[
\langle\grad f(x_{k}),\,x_{k+1}-x^{\star}\rangle
\le\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{k+1})-D_{h}(x^{\star},x_{k})-D_{h}(x_{k},x_{k+1})\Big).
\tag{9}
\]

\noindent\textbf{[Step 5]} (Convexity of $f$).  
Convexity implies
\[
f(x_{k})-f(x^{\star})\le\langle\grad f(x_{k}),\,x_{k}-x^{\star}\rangle.
\tag{10}
\]

\noindent\textbf{[Step 6]} (Rewrite the inner product).  
Add and subtract $x_{k+1}$ inside the RHS of \eqref{Step 5}:
\[
\langle\grad f(x_{k}),\,x_{k}-x^{\star}\rangle
= \langle\grad f(x_{k}),\,x_{k}-x_{k+1}\rangle
+ \langle\grad f(x_{k}),\,x_{k+1}-x^{\star}\rangle .
\tag{11}
\]

\noindent\textbf{[Step 7]} (Bound the first term using smoothness).  
From Lemma \ref{lem:smooth} with $y=x_{k+1}$ and $x=x_{k}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\]
Rearranging,
\[
\langle\grad f(x_{k}),\,x_{k}-x_{k+1}\rangle
\le f(x_{k})-f(x_{k+1})+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 8]} (Insert (9)–(12) into (10)).  
Combining \eqref{Step 5}, \eqref{Step 6}, \eqref{Step 7} and \eqref{Step 4} yields
\[
\begin{aligned}
f(x_{k})-f(x^{\star})
&\le f(x_{k})-f(x_{k+1})+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&\quad+\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{k+1})-D_{h}(x^{\star},x_{k})-D_{h}(x_{k},x_{k+1})\Big).
\end{aligned}
\tag{13}
\]

\noindent\textbf{[Step 9]} (Relate Bregman divergence to norm).  
By strong convexity of $h$ (Assumption \ref{ass:hstrong}), we have
\[
D_{h}(x_{k},x_{k+1})\ge\frac{\alpha}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{14}
\]

\noindent\textbf{[Step 10]} (Apply (14) to (13)}.  
Replace $D_{h}(x_{k},x_{k+1})$ using \eqref{Step 10}:
\[
\begin{aligned}
f(x_{k})-f(x^{\star})
&\le f(x_{k})-f(x_{k+1})
+\Big(\frac{L}{2}-\frac{\alpha}{2\eta}\Big)\|x_{k+1}-x_{k}\|^{2}\\
&\quad+\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{k+1})-D_{h}(x^{\star},x_{k})\Big).
\end{aligned}
\tag{15}
\]

\noindent\textbf{[Step 11]} (Choose stepsize).  
Assumption \ref{ass:stepsize} guarantees $ \eta\le\alpha/L$, i.e. $\frac{L}{2}-\frac{\alpha}{2\eta}\le0$. Hence the second term in \eqref{Step 11} is non‑positive and can be dropped:
\[
f(x_{k})-f(x^{\star})\le f(x_{k})-f(x_{k+1})+\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{k+1})-D_{h}(x^{\star},x_{k})\Big).
\tag{16}
\]

\noindent\textbf{[Step 12]} (Telescoping sum).  
Rearrange \eqref{Step 12}:
\[
f(x_{k})-f(x^{\star})\le\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{k})-D_{h}(x^{\star},x_{k+1})\Big).
\tag{17}
\]
Summing \eqref{Step 12} from $k=0$ to $T-1$ telescopes the Bregman terms:
\[
\sum_{k=0}^{T-1}\bigl(f(x_{k})-f(x^{\star})\bigr)
\le\frac{1}{\eta}\Big( D_{h}(x^{\star},x_{0})-D_{h}(x^{\star},x_{T})\Big)
\le\frac{V_{0}}{\eta},
\tag{18}
\]
where $V_{0}=D_{h}(x^{\star},x_{0})$ and the last inequality uses non‑negativity of Bregman divergence.

\noindent\textbf{[Step 13]} (Average iterate bound).  
Dividing \eqref{Step 13} by $T$ gives the claimed average guarantee \eqref{Step 3}:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(x_{k})-f(x^{\star})\bigr)\le\frac{V_{0}}{\eta T}.
\tag{19}
\]

\noindent\textbf{[Step 14]} (From average to a single point).  
Since $f$ is convex,
\[
f(\bar{x}_{T})\le\frac{1}{T}\sum_{k=0}^{T-1}f(x_{k}),
\qquad\bar{x}_{T}=\frac{1}{T}\sum_{k=0}^{T-1}x_{k}.
\]
Combining with \eqref{Step 13} yields
\[
f(\bar{x}_{T})-f(x^{\star})\le\frac{V_{0}}{\eta T}.
\tag{20}
\]

\noindent\textbf{[Step 15]} (Optimal stepsize).  
Choosing $\eta=\alpha/L$ (the largest admissible stepsize) gives the explicit rate
\[
f(\bar{x}_{T})-f(x^{\star})\le\frac{L\,V_{0}}{\alpha\,T},
\]
which completes the proof. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{Step 15} we have the bound
\[
\boxed{ \; f(\bar{x}_{T})-f(x^{\star})\le \frac{L}{\alpha}\,\frac{D_{h}(x^{\star},x_{0})}{T}\; }.
\]
The constant $L/\alpha$ reflects the ratio between the smoothness of $f$ and the curvature of the prox‑function $h$. For the Euclidean case $h(x)=\frac12\|x\|_{2}^{2}$, $\alpha=1$ and we recover the classical $L D_{h}(x^{\star},x_{0})/T$ bound.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Euclidean Mirror Descent.} $h(x)=\frac12\|x\|_{2}^{2}$ gives $D_{h}(u,v)=\frac12\|u-v\|_{2}^{2}$ and \eqref{eq:update} reduces to gradient descent. The theorem coincides with the standard $O(1/T)$ rate for smooth convex functions.
\item \textbf{Entropy Mirror Descent.} On the probability simplex $\Delta^{d}$, let $h(x)=\sum_{i}x_{i}\log x_{i}$ (negative entropy). Then $D_{h}$ is the Kullback‑Leibler divergence and the update becomes the exponentiated gradient algorithm. The same $O(1/T)$ guarantee holds with $V_{0}=D_{\mathrm{KL}}(x^{\star}\|x_{0})$.
\item \textbf{Stochastic Gradients.} If $g_{k}$ satisfies $\EE[g_{k}\mid x_{k}]=\grad f(x_{k})$ and $\EE\|g_{k}-\grad f(x_{k})\|_{*}^{2}\le\sigma^{2}$, the same derivation up to \eqref{Step 6} holds in expectation and leads to
\[
\EE\bigl[f(\bar{x}_{T})-f(x^{\star})\bigr]\le\frac{V_{0}}{\eta T}+\frac{\eta\sigma^{2}}{2},
\]
optimizing $\eta\propto 1/\sqrt{T}$ gives the $O(1/\sqrt{T})$ stochastic rate.
\end{enumerate}
