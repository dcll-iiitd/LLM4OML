\section*{Algorithm Name}
\textbf{Proximal Accelerated Gradient Method (PAGM)}  
The method solves the composite convex problem  

\[
\min_{x\in\R^{d}}\; \Phi(x)\stackrel{\mathrm{def}}{=}f(x)+g(x),
\]

where $f$ is smooth (Lipschitz‑continuous gradient) and $g$ is convex, possibly non‑smooth, but with an efficiently computable proximal operator.

\section*{Mathematical Formulation}
Let $\eta>0$ be a stepsize such that $\eta\le \frac1L$, where $L$ is the Lipschitz constant of $\nabla f$.  
Define sequences $\{x_k\}_{k\ge 0}$, $\{y_k\}_{k\ge 0}$ and scalars $\{t_k\}_{k\ge 0}$ as follows:

\[
\begin{aligned}
t_0 &:=1, \qquad  t_{k+1}:=\frac{1+\sqrt{1+4t_k^{2}}}{2},\\[4pt]
\beta_k &:=\frac{t_k-1}{t_{k+1}},\\[4pt]
y_k &:=x_k+\beta_k\,(x_k-x_{k-1})\qquad (k\ge 1),\qquad y_0:=x_0,\\[4pt]
x_{k+1} &:=\prox_{\eta g}\!\bigl(y_k-\eta \nabla f(y_k)\bigr),\qquad k\ge 0 .
\end{aligned}
\tag{1}
\]

The proximal operator is defined by  

\[
\prox_{\eta g}(v):=\arg\min_{u\in\R^{d}}\Bigl\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Accelerated Gradient Method}
\begin{algorithmic}[1]
\Require Initial points $x_{-1}=x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$, maximal iteration $K$.
\State $t_{0}\gets 1$
\For{$k=0,\dots,K-1$}
    \State $t_{k+1}\gets\frac{1+\sqrt{1+4t_{k}^{2}}}{2}$
    \State $\beta_{k}\gets\frac{t_{k}-1}{t_{k+1}}$
    \State $y_{k}\gets x_{k}+\beta_{k}(x_{k}-x_{k-1})$
    \State $x_{k+1}\gets\prox_{\eta g}\bigl(y_{k}-\eta\nabla f(y_{k})\bigr)$
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]

so $\Phi(w)=f(w)$.  
We have $\nabla f(w)=2(w-3)$ and $L=2$. Choose $\eta=0.1\le 1/L$, $x_{-1}=x_{0}=0$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
 $k$ & $t_k$ & $\beta_k$ & $y_k$ & $x_{k+1}$ \\ \hline
0 & $1$ & $0$ & $0$ & $0-\eta\,\nabla f(0)=0-0.1\cdot(-6)=0.6$ \\[4pt]
1 & $\frac{1+\sqrt{5}}{2}\approx1.618$ & $\frac{1-1}{1.618}=0$ & $0.6$ & $0.6-\eta\nabla f(0.6)=0.6-0.1\cdot(-4.8)=1.08$ \\[4pt]
2 & $\frac{1+\sqrt{1+4(1.618)^{2}}}{2}\approx2.193$ & $\frac{1.618-1}{2.193}\approx0.282$ & 
$1.08+0.282\,(1.08-0.6)=1.225$ &
$1.225-\eta\nabla f(1.225)=1.225-0.1\cdot(-3.55)=1.580$ 
\end{tabular}
\end{center}

Objective values: $\Phi(0.6)=5.76$, $\Phi(1.08)=3.69$, $\Phi(1.58)=2.04$, showing rapid decrease toward the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f:\R^{d}\to\R$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\| \le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is single‑valued for any $\eta>0$.
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}

\begin{assumption}[Existence of a minimizer]\label{as:opt}
The set of minimizers 
\[
\mathcal{X}^{\star}:=\arg\min_{x\in\R^{d}}\Phi(x)
\]
is non‑empty; denote $x^{\star}\in\mathcal{X}^{\star}$ arbitrarily.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Optimal $O(1/k^{2})$ rate]\label{thm:rate}
Under Assumptions \ref{as:smooth}–\ref{as:opt} and using the iterates defined in \eqref{1}, for every $k\ge 1$ we have  

\[
\Phi(x_{k})- \Phi(x^{\star})\;\le\;
\frac{2\|x_{0}-x^{\star}\|^{2}}{\eta\, (k+1)^{2}} .
\tag{2}
\]

Consequently $\Phi(x_{k})\to\Phi(x^{\star})$ and the convergence is $O(1/k^{2})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The recursion for $t_k$ guarantees $t_k\ge \frac{k+1}{2}$, which is crucial for the $1/k^{2}$ decay.
\item \textbf{Hyperparameter sensitivity:} The stepsize bound $\eta\le 1/L$ is tight; choosing $\eta$ larger may break the descent inequality used in the proof.
\item \textbf{Comparison:} Compared with the plain proximal gradient method (which yields $O(1/k)$), PAGM attains the optimal accelerated rate for the composite convex class.
\end{itemize}

\section*{Discussion}
The theorem shows that the momentum term $\beta_k$ (equivalently the sequence $\{t_k\}$) injects a “look‑ahead’’ effect that mimics Nesterov’s acceleration while preserving the proximal structure necessary for non‑smooth $g$. The bound \eqref{2} is deterministic; no stochasticity is required.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $u,v\in\R^{d}$,
\[
f(u)\le f(v)+\langle \nabla f(v),u-v\rangle+\frac{L}{2}\|u-v\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Three–point inequality for proximal operator]\label{lem:prox}
For any $z\in\R^{d}$ and any $u\in\R^{d}$,
\[
g\bigl(\prox_{\eta g}(z)\bigr)+\frac{1}{2\eta}\bigl\|\prox_{\eta g}(z)-z\bigr\|^{2}
\le g(u)+\frac{1}{2\eta}\|u-z\|^{2}
-\frac{1}{2\eta}\bigl\|\prox_{\eta g}(z)-u\bigr\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
From the optimality condition of the proximal subproblem.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  
All non‑trivial steps are numbered \textbf{[Step N]}.

\medskip
\noindent\textbf{Notation.} Write $x^{\star}\in\mathcal{X}^{\star}$, define  

\[
s_{k}:=t_{k}\bigl(x_{k}-x^{\star}\bigr)-\bigl(t_{k}-1\bigr)\bigl(x_{k-1}-x^{\star}\bigr),\qquad k\ge 1.
\tag{5}
\]

\noindent\textbf{Goal.} Show that $\|s_{k}\|^{2}$ is non‑increasing up to a controllable error term, which yields \eqref{2} after telescoping.

\medskip
\noindent\textbf{Step 1.}  From the definition of $y_k$ and $\beta_k$ we have  

\[
y_k-x^{\star}=x_k-x^{\star}+\beta_k\bigl(x_k-x_{k-1}\bigr)
= \frac{t_{k+1}}{t_k}\bigl(x_k-x^{\star}\bigr)-\frac{t_{k+1}-t_k}{t_k}\bigl(x_{k-1}-x^{\star}\bigr).
\tag{6}
\]

\noindent\textbf{Step 2.}  Multiply both sides of \eqref{6} by $t_k$ and rearrange:

\[
t_k(y_k-x^{\star}) = t_{k+1}(x_k-x^{\star})-(t_{k+1}-t_k)(x_{k-1}-x^{\star})
= s_k .
\tag{7}
\]

Thus $s_k = t_k (y_k-x^{\star})$.

\medskip
\noindent\textbf{Step 3.}  Apply Lemma \ref{lem:prox} with $z = y_k-\eta\nabla f(y_k)$ and $u=x^{\star}$:

\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-z\|^{2}
\le g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-z\|^{2}
-\frac{1}{2\eta}\|x_{k+1}-x^{\star}\|^{2}.
\tag{8}
\]

\noindent\textbf{Step 4.}  Using $z = y_k-\eta\nabla f(y_k)$, expand the two squared norms:

\[
\|x^{\star}-z\|^{2}= \|x^{\star}-y_k+\eta\nabla f(y_k)\|^{2}
= \|x^{\star}-y_k\|^{2}+2\eta\langle x^{\star}-y_k,\nabla f(y_k)\rangle+\eta^{2}\|\nabla f(y_k)\|^{2},
\tag{9}
\]

\[
\|x_{k+1}-z\|^{2}= \|x_{k+1}-y_k+\eta\nabla f(y_k)\|^{2}
= \|x_{k+1}-y_k\|^{2}+2\eta\langle x_{k+1}-y_k,\nabla f(y_k)\rangle+\eta^{2}\|\nabla f(y_k)\|^{2}.
\tag{10}
\]

Insert \eqref{9}–\eqref{10} into \eqref{8}; the $\eta^{2}\|\nabla f(y_k)\|^{2}$ terms cancel:

\[
\begin{aligned}
& g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-y_k\|^{2}
+ \langle x_{k+1}-y_k,\nabla f(y_k)\rangle \\
\le\;& g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-y_k\|^{2}
+ \langle x^{\star}-y_k,\nabla f(y_k)\rangle
-\frac{1}{2\eta}\|x_{k+1}-x^{\star}\|^{2}.
\end{aligned}
\tag{11}
\]

\noindent\textbf{Step 5.}  Add $f(y_k)$ to both sides and use the convexity of $f$:

\[
f(y_k)+\langle \nabla f(y_k),x_{k+1}-y_k\rangle \le f(x_{k+1}) \quad\text{(convexity)}.
\tag{12}
\]

Similarly  

\[
f(y_k)+\langle \nabla f(y_k),x^{\star}-y_k\rangle \le f(x^{\star}).
\tag{13}
\]

\noindent\textbf{Step 6.}  Combine \eqref{11} with \eqref{12}–\eqref{13} and recall $\Phi=\!f+g$:

\[
\Phi(x_{k+1})\le \Phi(x^{\star})
+ \frac{1}{2\eta}\bigl(\|x^{\star}-y_k\|^{2}-\|x_{k+1}-y_k\|^{2}\bigr)
-\frac{1}{2\eta}\|x_{k+1}-x^{\star}\|^{2}.
\tag{14}
\]

\noindent\textbf{Step 7.}  Multiply \eqref{14} by $t_{k+1}^{2}$ and use $s_{k+1}=t_{k+1}(y_{k+1}-x^{\star})$ (by \eqref{7} with index $k+1$).  After algebraic rearrangement we obtain

\[
t_{k+1}^{2}\bigl(\Phi(x_{k+1})-\Phi(x^{\star})\bigr)
\le \frac{1}{2\eta}\bigl(\|s_{k}\|^{2}-\|s_{k+1}\|^{2}\bigr).
\tag{15}
\]

\noindent\textbf{Justification of (15).}  
From \eqref{7}, $y_k-x^{\star}=s_k/t_k$, therefore  

\[
\|x^{\star}-y_k\|^{2}= \frac{\|s_k\|^{2}}{t_k^{2}},\qquad
\|x_{k+1}-y_k\|^{2}= \frac{\|s_{k+1}\|^{2}}{t_{k+1}^{2}}
\]
(using the definition of $x_{k+1}$ and the same relation for $k+1$). Substituting these into \eqref{14} and multiplying by $t_{k+1}^{2}$ yields \eqref{15}. \hfill\qedsymbol

\medskip
\noindent\textbf{Step 8.}  Telescoping \eqref{15} for $k=0,\dots, N-1$ gives  

\[
\sum_{k=0}^{N-1} t_{k+1}^{2}\bigl(\Phi(x_{k+1})-\Phi(x^{\star})\bigr)
\le \frac{1}{2\eta}\bigl(\|s_{0}\|^{2}-\|s_{N}\|^{2}\bigr)
\le \frac{1}{2\eta}\|s_{0}\|^{2}.
\tag{16}
\]

Since $t_{k}\ge \frac{k+1}{2}$ (proved by induction from the recursion of $t_k$), we have $t_{k+1}^{2}\ge \frac{(k+2)^{2}}{4}$. Hence each term in the left-hand side of \eqref{16} satisfies  

\[
\Phi(x_{k+1})-\Phi(x^{\star})\le
\frac{2}{\eta (k+2)^{2}}\,\|s_{0}\|^{2}.
\tag{17}
\]

\noindent\textbf{Step 9.}  Compute $s_{0}$ from \eqref{5}.  By definition $t_{0}=1$, $x_{-1}=x_{0}$, thus  

\[
s_{0}=t_{0}(x_{0}-x^{\star})-(t_{0}-1)(x_{-1}-x^{\star})
= x_{0}-x^{\star}.
\tag{18}
\]

Insert \eqref{18} into \eqref{17} and rename $k+1\leftarrow k$ to obtain the final bound

\[
\Phi(x_{k})-\Phi(x^{\star})\le
\frac{2\|x_{0}-x^{\star}\|^{2}}{\eta\, (k+1)^{2}},\qquad k\ge 1,
\]
which is exactly \eqref{2}.  $\square$

\section*{Rate Derivation (With Constants)}
The constant $C:=\dfrac{2\|x_{0}-x^{\star}\|^{2}}{\eta}$ in \eqref{2} depends only on the initial distance and the stepsize.  Choosing the maximal admissible stepsize $\eta=1/L$ yields the tightest bound  

\[
\Phi(x_{k})-\Phi(x^{\star})\le
\frac{2L\|x_{0}-x^{\star}\|^{2}}{(k+1)^{2}}.
\tag{19}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure smooth case ($g\equiv0$).}  The algorithm reduces to Nesterov’s accelerated gradient method; the same $O(1/k^{2})$ bound holds.
\item \textbf{Indicator $g=\iota_{\mathcal{C}}$ (projection onto a convex set).}  The proximal step becomes a Euclidean projection, yielding the accelerated projected gradient method.
\item \textbf{Strongly convex $f+g$ with parameter $\mu>0$.}  By adding a standard restart or using a decreasing stepsize, one can improve the rate to $O\bigl((1-\sqrt{\mu/L})^{k}\bigr)$; the proof follows from the same potential function with an extra $\mu$ term.
\end{itemize}
