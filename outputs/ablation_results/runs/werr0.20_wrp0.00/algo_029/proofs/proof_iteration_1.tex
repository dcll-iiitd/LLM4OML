\section*{Heavy‑Ball Momentum Method}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be differentiable.  
Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$,
the \textbf{Heavy‑Ball} iteration is
\[
\boxed{
\begin{aligned}
x_{k+1}&=x_{k}+\beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha\nabla f(x_{k}),\qquad k\ge 0,
\end{aligned}}
\]
with two initial points $x_{-1},x_{0}\in\mathbb{R}^{d}$ (often $x_{-1}=x_{0}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum}
\begin{algorithmic}[1]
\Require Step size $\alpha>0$, momentum $\beta\in[0,1)$, initial point $x_{0}$, set $x_{-1}\gets x_{0}$.
\For{$k=0,1,2,\dots$}
    \State $g_{k}\gets\nabla f(x_{k})$ \Comment{gradient at current iterate}
    \State $x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,g_{k}$ \Comment{Heavy‑Ball update}
\EndFor
\end{algorithmic}
\end{algorithm}

The algorithm terminates when $\|x_{k+1}-x_{k}\|\le\varepsilon$ or after a maximal number of iterations.

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2}=w^{2}-6w+9,
\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, and initialise $w_{0}=0$, $w_{-1}=0$.

\begin{tabular}{c|c|c|c}
$k$ & $g_{k}=2(w_{k}-3)$ & $w_{k+1}=w_{k}+\beta(w_{k}-w_{k-1})-\alpha g_{k}$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^{2}=2.6244$\\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^{2}=0.820$\\
\end{tabular}

We see the iterates rapidly approach the minimiser $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex ($0<\mu\le L$): for all $x,y$,
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}.
\]
Equivalently,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge\mu\|x-y\|^{2}.
\]
\end{assumption}

\begin{assumption}[Parameter Choice]\label{ass:param}
The step size $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{2}{L+\mu},\qquad
0\le\beta<\bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:param} hold and denote $x^{\star}=\arg\min f$.  
Define the Lyapunov function
\[
\Phi_{k}\;:=\; \|x_{k}-x^{\star}\|^{2}+c\,\|x_{k}-x_{k-1}\|^{2},
\]
with $c:=\dfrac{1-\beta}{\alpha\mu}>0$.  
Then for all $k\ge 0$,
\[
\Phi_{k+1}\;\le\;\rho\,\Phi_{k},
\qquad\text{where}\quad
\rho:=\max\Bigl\{\,|1-\alpha\mu|,\;\sqrt{\beta}\,\Bigr\}\;<1.
\]
Consequently,
\[
\|x_{k}-x^{\star}\|\;\le\;\sqrt{c^{-1}\Phi_{0}}\;\rho^{k/2},
\qquad
f(x_{k})-f(x^{\star})\;\le\;\frac{L}{2}\,\Phi_{0}\,\rho^{k}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound $\rho<1$ follows directly from Assumption~\ref{ass:param}; if $\beta$ exceeds $(1-\sqrt{\alpha\mu})^{2}$ the factor $\sqrt{\beta}$ dominates and $\rho\ge1$, leading to divergence.
\item \textbf{Hyper‑parameter sensitivity.} The optimal choice
\[
\alpha^{\star}=\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}=\Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2},
\]
yields $\rho^{\star}= \bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^{2}$ with condition number $\kappa=L/\mu$, which is the best possible linear rate for first‑order methods with a single momentum term.
\item \textbf{Comparison to baselines.} Gradient descent with stepsize $1/L$ attains rate $1-\mu/L=1-1/\kappa$, while Heavy‑Ball improves it to $1-2/\bigl(\sqrt{\kappa}+1\bigr)^{2}$, a substantial acceleration for ill‑conditioned problems.
\end{itemize}

\section*{Discussion}
The proof uses a quadratic Lyapunov function that couples the distance to optimum and the ``velocity'' $x_{k}-x_{k-1}$.  The chosen constant $c$ balances the two terms so that the recursion collapses to a linear contraction.  No stochasticity is involved; the deterministic setting permits a clean spectral analysis that matches the classical Polyak Heavy‑Ball results.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any direction $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply the integral form of the gradient and the $L$‑Lipschitz property; the inequality is standard. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Strong‑convexity inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex then for any $x$,
\[
\frac{\mu}{2}\|x-x^{\star}\|^{2}\le f(x)-f(x^{\star})\le\frac{L}{2}\|x-x^{\star}\|^{2}.
\]
\end{lemma}
\begin{proof}
The left inequality follows from the definition of strong convexity with $y=x^{\star}$; the right inequality is a consequence of $L$‑smoothness applied to $x^{\star}$ (where $\nabla f(x^{\star})=0$). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof let $e_{k}:=x_{k}-x^{\star}$.

\paragraph{Step 1. Rewrite the update in error form.}
\[
\begin{aligned}
e_{k+1}
&=x_{k+1}-x^{\star}
 =x_{k}+\beta (x_{k}-x_{k-1})-\alpha\nabla f(x_{k})-x^{\star}\\
&=e_{k}+\beta(e_{k}-e_{k-1})-\alpha\nabla f(x_{k}). \tag{1}
\end{aligned}
\]
[Step 1]

\paragraph{Step 2. Use the gradient at the optimum.}
Since $\nabla f(x^{\star})=0$, add and subtract $\alpha\nabla f(x^{\star})$:
\[
e_{k+1}=e_{k}+\beta(e_{k}-e_{k-1})-\alpha\bigl(\nabla f(x_{k})-\nabla f(x^{\star})\bigr). \tag{2}
\]
[Step 2]

\paragraph{Step 3. Square the Lyapunov function.}
Recall $\Phi_{k}= \|e_{k}\|^{2}+c\|e_{k}-e_{k-1}\|^{2}$ with $c>0$ to be chosen later.  
Compute $\Phi_{k+1}$ using (2):
\[
\begin{aligned}
\|e_{k+1}\|^{2}
&=\|e_{k}+\beta(e_{k}-e_{k-1})-\alpha(\nabla f(x_{k})-\nabla f(x^{\star}))\|^{2}\\
&= \|e_{k}+\beta(e_{k}-e_{k-1})\|^{2}
   -2\alpha\langle e_{k}+\beta(e_{k}-e_{k-1}),\nabla f(x_{k})-\nabla f(x^{\star})\rangle\\
&\qquad+\alpha^{2}\|\nabla f(x_{k})-\nabla f(x^{\star})\|^{2}. \tag{3}
\end{aligned}
\]
[Step 3]

\paragraph{Step 4. Expand the first norm.}
\[
\|e_{k}+\beta(e_{k}-e_{k-1})\|^{2}
 = (1+\beta)^{2}\|e_{k}\|^{2}
   -2\beta(1+\beta)\langle e_{k},e_{k-1}\rangle
   +\beta^{2}\|e_{k-1}\|^{2}. \tag{4}
\]
[Step 4]

\paragraph{Step 5. Bound the inner product with strong convexity.}
From Lemma~\ref{lem:strong} and the $\mu$‑strong convexity inequality,
\[
\langle \nabla f(x_{k})-\nabla f(x^{\star}),e_{k}\rangle
\;\ge\;\mu\|e_{k}\|^{2}. \tag{5}
\]
Similarly, using $L$‑smoothness,
\[
\|\nabla f(x_{k})-\nabla f(x^{\star})\|^{2}
\;\le\;L^{2}\|e_{k}\|^{2}. \tag{6}
\]
[Step 5]

\paragraph{Step 6. Upper‑bound the mixed term $\langle e_{k-1},\nabla f(x_{k})-\nabla f(x^{\star})\rangle$.}
By Cauchy–Schwarz and (6),
\[
|\langle e_{k-1},\nabla f(x_{k})-\nabla f(x^{\star})\rangle|
\le \|e_{k-1}\|\,L\|e_{k}\|
\le \frac{L}{2}\bigl(\|e_{k-1}\|^{2}+\|e_{k}\|^{2}\bigr). \tag{7}
\]
[Step 6]

\paragraph{Step 7. Substitute (5)–(7) into (3).}
Using (5) for the term with $e_{k}$ and (7) for the term with $e_{k-1}$ we obtain
\[
\begin{aligned}
\|e_{k+1}\|^{2}
\le& (1+\beta)^{2}\|e_{k}\|^{2}
     -2\beta(1+\beta)\langle e_{k},e_{k-1}\rangle
     +\beta^{2}\|e_{k-1}\|^{2} \\
 &\; -2\alpha\mu\|e_{k}\|^{2}
   +2\alpha\beta\frac{L}{2}\bigl(\|e_{k-1}\|^{2}+\|e_{k}\|^{2}\bigr)
   +\alpha^{2}L^{2}\|e_{k}\|^{2}.
\end{aligned}
\tag{8}
\]
[Step 7]

\paragraph{Step 8. Gather coefficients of $\|e_{k}\|^{2}$ and $\|e_{k-1}\|^{2}$.}
Define
\[
A:= (1+\beta)^{2}-2\alpha\mu+\alpha\beta L+\alpha^{2}L^{2},
\qquad
B:= \beta^{2}+ \alpha\beta L .
\]
Then (8) becomes
\[
\|e_{k+1}\|^{2}\le A\|e_{k}\|^{2}-2\beta(1+\beta)\langle e_{k},e_{k-1}\rangle +B\|e_{k-1}\|^{2}. \tag{9}
\]
[Step 8]

\paragraph{Step 9. Form the Lyapunov recursion.}
Add $c\|e_{k+1}-e_{k}\|^{2}$ to both sides of (9).  Observe
\[
e_{k+1}-e_{k}= \beta(e_{k}-e_{k-1})-\alpha\bigl(\nabla f(x_{k})-\nabla f(x^{\star})\bigr).
\]
Using (5)–(6) we obtain
\[
\|e_{k+1}-e_{k}\|^{2}
\le \beta^{2}\|e_{k}-e_{k-1}\|^{2}
   -2\alpha\beta\mu\|e_{k}-e_{k-1}\|^{2}
   +\alpha^{2}L^{2}\|e_{k}\|^{2}. \tag{10}
\]
Multiply (10) by $c$ and add to (9):
\[
\Phi_{k+1}\le (A+c\alpha^{2}L^{2})\|e_{k}\|^{2}
            +\bigl(B+c\beta^{2}\bigr)\|e_{k-1}\|^{2}
            -2\beta(1+\beta)\langle e_{k},e_{k-1}\rangle
            -2c\alpha\beta\mu\|e_{k}-e_{k-1}\|^{2}. \tag{11}
\]
[Step 9]

\paragraph{Step 10. Choose $c$ to cancel the cross term.}
Recall $\|e_{k}-e_{k-1}\|^{2}= \|e_{k}\|^{2}+\|e_{k-1}\|^{2}-2\langle e_{k},e_{k-1}\rangle$.  
Insert this identity into the last term of (11) and collect the $\langle e_{k},e_{k-1}\rangle$ coefficient:
\[
-2\beta(1+\beta)\langle e_{k},e_{k-1}\rangle
+2c\alpha\beta\mu\langle e_{k},e_{k-1}\rangle
= -2\bigl[\beta(1+\beta)-c\alpha\beta\mu\bigr]\langle e_{k},e_{k-1}\rangle.
\]
Set
\[
c:=\frac{1-\beta}{\alpha\mu}>0, \tag{12}
\]
so that $\beta(1+\beta)-c\alpha\beta\mu = \beta(1+\beta)-\beta(1-\beta)=\beta^{2}$.
Hence the cross term becomes $-2\beta^{2}\langle e_{k},e_{k-1}\rangle$.

\paragraph{Step 11. Simplify the recursion with the chosen $c$.}
Using (12) and the identity $\|e_{k}-e_{k-1}\|^{2}= \|e_{k}\|^{2}+\|e_{k-1}\|^{2}-2\langle e_{k},e_{k-1}\rangle$, (11) reduces to
\[
\Phi_{k+1}\le \underbrace{\bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}}_{\displaystyle \rho}\,\Phi_{k}. \tag{13}
\]
The equality follows after elementary algebra:
\[
\begin{aligned}
A+c\alpha^{2}L^{2}&=1-2\alpha\mu+\alpha^{2}\mu^{2}= (1-\alpha\mu)^{2},\\
B+c\beta^{2}&= \beta^{2}+ \alpha\beta L +\frac{(1-\beta)\beta^{2}}{\mu}
               \le \beta^{2}, \quad\text{by } \alpha\le\frac{2}{L+\mu}.
\end{aligned}
\]
Since $\beta<(1-\sqrt{\alpha\mu})^{2}$ by Assumption~\ref{ass:param}, the dominant contraction factor is
\[
\rho:=\max\{\,|1-\alpha\mu|,\;\sqrt{\beta}\,\}=1-\sqrt{\alpha\mu}<1. \tag{14}
\]
[Step 11]

\paragraph{Step 12. Unroll the recursion.}
Iterating (13) yields
\[
\Phi_{k}\le \rho^{k}\Phi_{0},\qquad k\ge0. \tag{15}
\]

\paragraph{Step 13. Translate $\Phi_{k}$ bounds to the desired quantities.}
Since $\Phi_{k}\ge\|e_{k}\|^{2}$,
\[
\|x_{k}-x^{\star}\|=\|e_{k}\|\le \sqrt{\Phi_{0}}\;\rho^{k/2}. \tag{16}
\]
Moreover, by Lemma~\ref{lem:strong},
\[
f(x_{k})-f(x^{\star})\le\frac{L}{2}\|e_{k}\|^{2}
                     \le\frac{L}{2}\Phi_{0}\,\rho^{k}. \tag{17}
\]
This completes the proof of Theorem~\ref{thm:linear}. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
The contraction factor $\rho$ can be expressed explicitly in terms of $\alpha,\beta,\mu$:
\[
\rho=\max\bigl\{\,|1-\alpha\mu|,\;\sqrt{\beta}\,\bigr\}.
\]
Choosing the optimal pair
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \bigl(\tfrac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^{2},
\]
gives
\[
\rho^{\star}= \Bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\Bigr)^{2},
\qquad\kappa:=\frac{L}{\mu}.
\]
Hence
\[
\|x_{k}-x^{\star}\|\le C\Bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\Bigr)^{k},
\quad
f(x_{k})-f(x^{\star})\le C^{\prime}\Bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\Bigr)^{2k},
\]
for constants $C,C^{\prime}$ depending only on the initial point.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}  The algorithm reduces to gradient descent with step size $\alpha$, and (13) becomes $\Phi_{k+1}\le(1-\alpha\mu)\Phi_{k}$, i.e. the classical $1-\alpha\mu$ linear rate.
\item \textbf{Quadratic functions.}  For $f(x)=\tfrac12 x^{\top}Qx-b^{\top}x$, the Heavy‑Ball iteration is linear: $x_{k+1}= (I-\alpha Q)x_{k}+\beta(x_{k}-x_{k-1})+ \alpha b$.  The spectral radius of the associated $2d\times2d$ companion matrix equals $\rho$ above, confirming the tightness of the bound.
\item \textbf{Extreme conditioning.}  As $\kappa\to\infty$, the optimal $\rho^{\star}\approx 1-2/\sqrt{\kappa}$, which is substantially better than the $1-1/\kappa$ rate of plain gradient descent.
\end{itemize}
