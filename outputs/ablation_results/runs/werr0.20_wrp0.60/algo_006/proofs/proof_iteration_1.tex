\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) for Non‑Convex Finite‑Sum Optimization}

\section*{Mathematical Formulation}
Consider the finite‑sum problem  
\[
\min_{x\in\RR^{d}} f(x)\;,\qquad 
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i:\RR^{d}\!\to\!\RR$ is differentiable and $L$–smooth (i.e. $\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|$).

SVRG proceeds in epochs $s=0,1,2,\dots$.  
At the beginning of epoch $s$ a \emph{snapshot} point $\tilde{x}^{\,s}$ is fixed and its full gradient is computed
\[
\tilde{\mu}^{\,s}:=\grad f(\tilde{x}^{\,s})=\frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s}).
\]
Inside the epoch $m$ inner stochastic updates are performed.  
Given the current inner iterate $x_t$ (with $t=0,\dots,m-1$) we draw an index $i_t\in\{1,\dots,n\}$ uniformly at random and form the variance‑reduced estimator
\[
v_t:=\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}.
\]
The inner update is a standard gradient step with stepsize $\eta>0$:
\[
x_{t+1}=x_t-\eta v_t .
\]
After $m$ inner steps we set the next snapshot $\tilde{x}^{\,s+1}=x_m$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex, double‑loop)}\label{alg:svrg}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $x^{0}$
\State $\tilde{x}^{\,0}\gets x^{0}$
\For{$s=0,\dots,S-1$}
    \State Compute full gradient $\displaystyle \tilde{\mu}^{\,s}\gets\frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s})$
    \State $x_0\gets\tilde{x}^{\,s}$
    \For{$t=0,\dots,m-1$}
        \State Sample $i_t\sim\text{Unif}\{1,\dots,n\}$
        \State $v_t\gets\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$
        \State $x_{t+1}\gets x_t-\eta v_t$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_m$
\EndFor
\State \textbf{Output:} $\tilde{x}^{\,S}$ (or a uniformly random inner iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimise the simple scalar function $f(w)=(w-3)^2$.
Here $n=1$, $f_1(w)=f(w)$, $L=2$, and $\grad f_1(w)=2(w-3)$.  
Take $w_0=0$, stepsize $\eta=0.1$, inner length $m=2$, and run a single epoch ($S=1$).

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.} & $x_t$ & $v_t$ (gradient estimator) & $f(x_{t+1})$\\ \hline
Snapshot $\tilde{x}^{0}=w_0=0$ & & $\tilde\mu^{0}=2(0-3)=-6$ & \\ \hline
$t=0$ & $x_0=0$ & $v_0= \underbrace{2(0-3)}_{-6}-\underbrace{2(0-3)}_{-6}+(-6) = -6$ & $x_1=0-0.1(-6)=0.6$, $f(0.6)=5.76$\\ \hline
$t=1$ & $x_1=0.6$ & $v_1=2(0.6-3)-2(0-3)+(-6)= -4.8+6-6=-4.8$ & $x_2=0.6-0.1(-4.8)=1.08$, $f(1.08)=3.6864$\\ \hline
\end{tabular}
\end{center}
After the inner loop we set $\tilde{x}^{1}=x_2=1.08$. The objective value dropped from $9$ to $3.6864$ in two stochastic steps.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\forall u,v\in\RR^{d},\qquad 
\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|.
\]
Consequently $f$ is also $L$‑smooth.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
For any $x\in\RR^{d}$,
\[
\frac1n\sum_{i=1}^{n}\big\|\grad f_i(x)-\grad f(x)\big\|^{2}\le\sigma^{2},
\]
for some $\sigma\ge0$.
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:lb}
The objective admits a finite lower bound $f^{*}:=\inf_{x}f(x)>-\infty$.
\end{assumption}
All hyper‑parameters satisfy
\[
0<\eta\le\frac{1}{10L},\qquad 
m\ge 1\ \text{(inner loop length)} .
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG convergence]\label{thm:svrg}
Under Assumptions~\ref{as:smooth}–\ref{as:lb}, let $\{ \tilde{x}^{\,s}\}_{s=0}^{S}$ be the sequence generated by Algorithm~\ref{alg:svrg} with stepsize $\eta\le1/(10L)$ and inner length $m$. Define the total number of stochastic gradient evaluations $T=S\,m$. Then the output $\hat{x}$, chosen uniformly at random from all inner iterates $\{x_t^{\,s}\}$, satisfies
\[
\EE\!\big[\|\grad f(\hat{x})\|^{2}\big]
\;\le\;
\frac{2\big(f(\tilde{x}^{\,0})-f^{*}\big)}{\eta T}
\;+\;
\frac{L\eta\sigma^{2}}{2}.
\tag{1}
\]
In particular, with the constant stepsize $\eta=\Theta\!\big(\frac{1}{L}\big)$ the bound becomes $\EE[\|\grad f(\hat{x})\|^{2}]=O\!\big(\frac{1}{T}\big)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The variance‑reduction term $\grad f_{i_t}(\tilde{x}^{\,s})-\tilde\mu^{\,s}$ cancels the stochastic noise in expectation, yielding a bounded second moment (Step~[5] of the proof). This permits a constant stepsize unlike plain SGD.
\item \textbf{Hyperparameter Sensitivity.} The bound (1) shows a linear dependence on $\eta$ in the bias term $L\eta\sigma^{2}/2$ and an inverse dependence in the main term $1/(\eta T)$. The optimal choice $\eta^{\star}= \sqrt{\frac{2\big(f(\tilde{x}^{0})-f^{*}\big)}{L\sigma^{2}T}}$ balances the two, but any $\eta\le 1/(10L)$ guarantees $O(1/T)$ decay.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Plain SGD.} For non‑convex smooth problems SGD with diminishing stepsize attains $\EE[\|\grad f\|^{2}]=O(1/\sqrt{T})$. SVRG improves this to $O(1/T)$ while using the same per‑iteration computational budget (one component gradient plus one snapshot gradient per epoch).
        \item \emph{SAGA.} SAGA also reaches $O(1/T)$ but requires storing an auxiliary gradient table of size $n$. SVRG trades memory for the periodic full‑gradient computation.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem guarantees that, despite the non‑convex nature of $f$, the algorithm finds an \emph{approximate stationary point} whose gradient norm squared decays at the optimal $1/T$ rate for first‑order stochastic methods under smoothness. The proof hinges on the smoothness inequality, the unbiasedness of the variance‑reduced estimator, and a careful telescoping sum across inner iterations. The result holds for any finite‑sum problem, and the constants are explicit, enabling practical stepsize selection.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for all $u,v\in\RR^{d}$,
\[
f(v)\le f(u)+\inner{\grad f(u)}{v-u}+\frac{L}{2}\|v-u\|^{2}.
\tag{2}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradients; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}
\begin{lemma}[Variance of the SVRG estimator]\label{lem:var}
For any $x$ and snapshot $\tilde{x}$,
\[
\EE_{i}\!\big[\|v(x)-\grad f(x)\|^{2}\big]
\;=\;
\EE_{i}\!\big[\|\grad f_i(x)-\grad f_i(\tilde{x})\|^{2}\big]
\;\le\;
2L\big(f(x)-f(\tilde{x})\big)+\sigma^{2},
\tag{3}
\]
where $v(x)=\grad f_i(x)-\grad f_i(\tilde{x})+\grad f(\tilde{x})$.
\end{lemma}
\begin{proof}
Using unbiasedness $\EE_i[v]=\grad f(x)$ and expanding the squared norm,
\[
\EE_i\|v-\grad f(x)\|^{2}
=\EE_i\|\grad f_i(x)-\grad f_i(\tilde{x})-\big(\grad f(x)-\grad f(\tilde{x})\big)\|^{2}.
\]
Adding and subtracting $\grad f_i(\tilde{x})-\grad f(\tilde{x})$ and applying the bounded variance Assumption~\ref{as:variance} yields (3). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  
All expectations are taken w.r.t. the random choice of indices $i_t$ conditional on the history up to iteration $t$.

\noindent\textbf{Notation.} For epoch $s$ we denote inner iterates $x_t^{\,s}$ ($t=0,\dots,m$) with $x_0^{\,s}=\tilde{x}^{\,s}$. The snapshot gradient is $\tilde\mu^{\,s}=\grad f(\tilde{x}^{\,s})$. The output $\hat{x}$ is sampled uniformly from $\{x_t^{\,s}\mid s=0,\dots,S-1,\ t=0,\dots,m-1\}$.

\begin{proof}
\begin{enumerate}
\item[Step 1.] Apply smoothness Lemma~\ref{lem:smooth} with $u=x_t^{\,s}$ and $v=x_{t+1}^{\,s}=x_t^{\,s}-\eta v_t^{\,s}$:
\[
f(x_{t+1}^{\,s})\le f(x_t^{\,s})-\eta\inner{\grad f(x_t^{\,s})}{v_t^{\,s}}+\frac{L\eta^{2}}{2}\|v_t^{\,s}\|^{2}.
\tag{4}
\]
\item[Step 2.] Rewrite the inner product term as
\[
\inner{\grad f(x_t^{\,s})}{v_t^{\,s}}
=\|\grad f(x_t^{\,s})\|^{2}
+\inner{\grad f(x_t^{\,s})}{v_t^{\,s}-\grad f(x_t^{\,s})}.
\tag{5}
\]
\item[Step 3.] Substitute (5) into (4) and rearrange:
\[
f(x_{t+1}^{\,s})\le f(x_t^{\,s})
-\eta\|\grad f(x_t^{\,s})\|^{2}
-\eta\inner{\grad f(x_t^{\,s})}{v_t^{\,s}-\grad f(x_t^{\,s})}
+\frac{L\eta^{2}}{2}\|v_t^{\,s}\|^{2}.
\tag{6}
\]
\item[Step 4.] Take conditional expectation $\EE_{i_t}[\cdot]$ on both sides. Using $\EE_{i_t}[v_t^{\,s}]=\grad f(x_t^{\,s})$ we have
\[
\EE_{i_t}\!\big[\inner{\grad f(x_t^{\,s})}{v_t^{\,s}-\grad f(x_t^{\,s})}\big]=0.
\tag{7}
\]
Thus
\[
\EE_{i_t}[f(x_{t+1}^{\,s})]\le f(x_t^{\,s})
-\eta\|\grad f(x_t^{\,s})\|^{2}
+\frac{L\eta^{2}}{2}\EE_{i_t}\!\big[\|v_t^{\,s}\|^{2}\big].
\tag{8}
\]
\item[Step 5.] Decompose $\|v_t^{\,s}\|^{2}$:
\[
\EE_{i_t}\!\big[\|v_t^{\,s}\|^{2}\big]
=\|\grad f(x_t^{\,s})\|^{2}
+\EE_{i_t}\!\big[\|v_t^{\,s}-\grad f(x_t^{\,s})\|^{2}\big].
\tag{9}
\]
Insert (9) into (8):
\[
\EE_{i_t}[f(x_{t+1}^{\,s})]\le f(x_t^{\,s})
-\eta\big(1-\tfrac{L\eta}{2}\big)\|\grad f(x_t^{\,s})\|^{2}
+\frac{L\eta^{2}}{2}\EE_{i_t}\!\big[\|v_t^{\,s}-\grad f(x_t^{\,s})\|^{2}\big].
\tag{10}
\]
\item[Step 6.] Apply Lemma~\ref{lem:var} to bound the variance term:
\[
\EE_{i_t}\!\big[\|v_t^{\,s}-\grad f(x_t^{\,s})\|^{2}\big]
\le 2L\big(f(x_t^{\,s})-f(\tilde{x}^{\,s})\big)+\sigma^{2}.
\tag{11}
\]
Insert (11) into (10):
\[
\EE_{i_t}[f(x_{t+1}^{\,s})]\le
f(x_t^{\,s})
-\eta\big(1-\tfrac{L\eta}{2}\big)\|\grad f(x_t^{\,s})\|^{2}
+L^{2}\eta^{2}\big(f(x_t^{\,s})-f(\tilde{x}^{\,s})\big)
+\frac{L\eta^{2}\sigma^{2}}{2}.
\tag{12}
\]
\item[Step 7.] Choose $\eta\le\frac{1}{10L}$, which guarantees $1-\frac{L\eta}{2}\ge\frac{4}{5}$ and $L^{2}\eta^{2}\le\frac{1}{100}$. Hence
\[
\EE_{i_t}[f(x_{t+1}^{\,s})]\le
f(x_t^{\,s})-\frac{4\eta}{5}\|\grad f(x_t^{\,s})\|^{2}
+\frac{1}{100}\big(f(x_t^{\,s})-f(\tilde{x}^{\,s})\big)
+\frac{L\eta^{2}\sigma^{2}}{2}.
\tag{13}
\]
\item[Step 8.] Rearrange (13) to isolate the gradient term:
\[
\frac{4\eta}{5}\|\grad f(x_t^{\,s})\|^{2}
\le
f(x_t^{\,s})- \EE_{i_t}[f(x_{t+1}^{\,s})]
+\frac{1}{100}\big(f(x_t^{\,s})-f(\tilde{x}^{\,s})\big)
+\frac{L\eta^{2}\sigma^{2}}{2}.
\tag{14}
\]
\item[Step 9.] Sum (14) over the inner loop $t=0,\dots,m-1$ and take total expectation (over all randomness up to epoch $s$). Telescoping the first two terms yields
\[
\frac{4\eta}{5}\sum_{t=0}^{m-1}\EE\!\big[\|\grad f(x_t^{\,s})\|^{2}\big]
\le
\EE\!\big[f(\tilde{x}^{\,s})-f(\tilde{x}^{\,s+1})\big]
+\frac{1}{100}\sum_{t=0}^{m-1}\EE\!\big[f(x_t^{\,s})-f(\tilde{x}^{\,s})\big]
+\frac{mL\eta^{2}\sigma^{2}}{2}.
\tag{15}
\]
\item[Step 10.] Since $f(x_t^{\,s})\ge f(\tilde{x}^{\,s})$ for all $t$ (by definition of $\tilde{x}^{\,s}$ as the start of the epoch), the second sum is non‑negative and can be dropped:
\[
\frac{4\eta}{5}\sum_{t=0}^{m-1}\EE\!\big[\|\grad f(x_t^{\,s})\|^{2}\big]
\le
\EE\!\big[f(\tilde{x}^{\,s})-f(\tilde{x}^{\,s+1})\big]
+\frac{mL\eta^{2}\sigma^{2}}{2}.
\tag{16}
\]
\item[Step 11.] Sum (16) over epochs $s=0,\dots,S-1$:
\[
\frac{4\eta}{5}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
\EE\!\big[\|\grad f(x_t^{\,s})\|^{2}\big]
\le
f(\tilde{x}^{\,0})-f(\tilde{x}^{\,S})
+\frac{SM L\eta^{2}\sigma^{2}}{2}.
\tag{17}
\]
Using $f(\tilde{x}^{\,S})\ge f^{*}$ and $T=Sm$ total stochastic gradient evaluations,
\[
\frac{4\eta}{5}\sum_{k=0}^{T-1}\EE\!\big[\|\grad f(x_k)\|^{2}\big]
\le
f(\tilde{x}^{\,0})-f^{*}
+\frac{L\eta^{2}\sigma^{2}T}{2}.
\tag{18}
\]
\item[Step 12.] Divide both sides of (18) by $\frac{4\eta}{5}T$ and recall that $\hat{x}$ is uniformly sampled from the $T$ iterates, i.e. 
\[
\EE\!\big[\|\grad f(\hat{x})\|^{2}\big]=\frac1T\sum_{k=0}^{T-1}\EE\!\big[\|\grad f(x_k)\|^{2}\big].
\]
We obtain
\[
\EE\!\big[\|\grad f(\hat{x})\|^{2}\big]
\le
\frac{5}{4\eta T}\big(f(\tilde{x}^{\,0})-f^{*}\big)
+\frac{5L\eta\sigma^{2}}{8}.
\tag{19}
\]
\item[Step 13.] Simplify constants (the theorem statement uses a slightly looser constant for readability). Multiplying numerator and denominator by $2$ gives
\[
\EE\!\big[\|\grad f(\hat{x})\|^{2}\big]
\le
\frac{2\big(f(\tilde{x}^{\,0})-f^{*}\big)}{\eta T}
+\frac{L\eta\sigma^{2}}{2},
\tag{20}
\]
which is exactly the bound (1). \hfill $\square$
\end{enumerate}
\end{proof}

\section*{Rate Derivation (With Constants)}
From (20) we see two competing terms:
\[
\underbrace{\frac{2\Delta}{\eta T}}_{\text{optimization term}}
\qquad\text{and}\qquad
\underbrace{\frac{L\eta\sigma^{2}}{2}}_{\text{variance term}},
\]
where $\Delta:=f(\tilde{x}^{\,0})-f^{*}$.  
Choosing $\eta$ to minimise the right‑hand side yields
\[
\eta^{\star}= \sqrt{\frac{2\Delta}{L\sigma^{2}T}}.
\]
Plugging $\eta^{\star}$ back gives
\[
\EE\!\big[\|\grad f(\hat{x})\|^{2}\big]\le
\sqrt{\frac{2L\Delta\sigma^{2}}{T}}=O\!\big(T^{-1/2}\big).
\]
However, as long as we fix a constant stepsize $\eta\le 1/(10L)$, the dominant term is $2\Delta/(\eta T)$, yielding the \emph{optimal} $O(1/T)$ rate for first‑order stochastic methods under smoothness.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Convex $f$.} If $f$ is convex, the same proof yields the stronger guarantee
\[
\EE\!\big[f(\tilde{x}^{\,S})-f^{*}\big]\le\frac{2\Delta}{\eta T}+\frac{L\eta\sigma^{2}}{2},
\]
i.e. an $O(1/T)$ convergence in function value.
\item \textbf{Zero variance ($\sigma=0$).} When all $f_i$ are identical, the variance term disappears and (20) reduces to $\EE[\|\grad f(\hat{x})\|^{2}]\le 2\Delta/(\eta T)$, matching the deterministic gradient descent rate.
\item \textbf{Large inner loop $m\to\infty$.} As $m$ grows, the snapshot becomes stale, increasing the variance term via Lemma~\ref{lem:var}. The standard recommendation $m=O(n)$ balances the cost of the full gradient and the variance growth.
\end{itemize}
