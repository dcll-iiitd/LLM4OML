\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a convex and differentiable function with $L$‑Lipschitz continuous gradient, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
Let $g:\mathbb{R}^{d}\rightarrow\mathbb{R}\cup\{+\infty\}$ be a proper, closed, convex (possibly non‑smooth) function.  
Define the composite objective
\[
F(x)\;:=\;f(x)+g(x).
\]

For a stepsize $\eta>0$ the \emph{proximal operator} of $g$ is
\[
\prox_{\eta g}(v)\;:=\;\arg\min_{u\in\mathbb{R}^{d}}\Big\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]

The Proximal Gradient Method generates a sequence $\{x_k\}_{k\ge0}$ by
\begin{equation}
\boxed{\;x_{k+1}= \prox_{\eta g}\bigl(x_k-\eta\,\grad f(x_k)\bigr)\;},
\qquad k=0,1,2,\dots
\tag{PGM}
\end{equation}
where $\eta$ is a constant stepsize satisfying $0<\eta\le 1/L$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\mathbb{R}^{d}$, stepsize $\eta\in(0,1/L]$, maximum iterations $K$
\For{$k=0$ \textbf{to} $K-1$}
    \State $v_{k}=x_{k}-\eta\,\grad f(x_{k})$
    \State $x_{k+1}= \prox_{\eta g}(v_{k})$
\EndFor
\State \textbf{output} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^{2},\qquad g(w)=0,\qquad \eta=0.1,\qquad w_{0}=0.
\]
Since $g\equiv0$, $\prox_{\eta g}(v)=v$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\grad f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta\,\grad f(w_k)$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values after each iteration:
\[
F(w_0)=9,\;F(w_1)=(0.6-3)^2=5.76,\;F(w_2)=(1.08-3)^2=3.6864,\;F(w_3)=(1.464-3)^2\approx2.362.
\]

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] $f$ is convex and differentiable on $\mathbb{R}^{d}$.
\item[(A2)] $\grad f$ is $L$‑Lipschitz continuous:
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\quad\forall x,y\in\mathbb{R}^{d}.
\]
\item[(A3)] $g$ is proper, closed and convex (no smoothness required).
\item[(A4)] Stepsize $\eta$ satisfies $0<\eta\le 1/L$.
\item[(A5)] The optimal set $\mathcal{X}^{\star}:=\arg\min_{x}F(x)$ is non‑empty and we denote $F^{\star}=F(x^{\star})$ for any $x^{\star}\in\mathcal{X}^{\star}$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence rate of PGM]\label{thm:rate}
Under (A1)–(A5) and for the sequence $\{x_k\}$ generated by \eqref{PGM} with constant stepsize $\eta\in(0,1/L]$, the following holds for all $K\ge1$:
\[
F(x_{K})-F^{\star}\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{T1}
\]
Consequently $F(x_{K})\to F^{\star}$ and the rate is $\mathcal{O}(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update is a composition of a gradient step (which is a contraction for $\eta\le 1/L$) and the non‑expansive proximal operator. Hence the whole map is non‑expansive, guaranteeing bounded iterates.
\item \textbf{Hyperparameter Sensitivity.} The stepsize must respect $\eta\le 1/L$. Larger $\eta$ may violate the descent inequality and break the $\mathcal{O}(1/K)$ guarantee.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item If $g\equiv0$, \eqref{PGM} reduces to gradient descent with the classic $O(1/K)$ rate.
        \item Compared with subgradient methods (which achieve $O(1/\sqrt{K})$ for nonsmooth $f$), PGM leverages smoothness of $f$ and obtains a faster $O(1/K)$ rate.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that the proximal gradient method attains the optimal $\mathcal{O}(1/K)$ convergence for composite convex problems where the smooth part has a Lipschitz gradient. The proof hinges on two elementary but powerful facts:
\begin{enumerate}
\item The \emph{descent lemma} for $f$ (Lipschitz gradient).
\item The \emph{non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v.
\]
\end{enumerate}
These ingredients together yield a simple telescoping inequality that directly gives the rate (T1).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\grad f$ is $L$‑Lipschitz, then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle \grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; omitted for brevity. \hfill $\square$
\end{proof}

\begin{lemma}[Proximal optimality condition]\label{lem:prox-opt}
For $u=\prox_{\eta g}(v)$ we have
\[
0\in \partial g(u)+\frac{1}{\eta}(u-v),
\qquad\text{or equivalently}\qquad
v-u\in\eta\,\partial g(u).
\tag{L2}
\]
\end{lemma}
\begin{proof}
By definition $u$ minimizes $g(z)+\frac{1}{2\eta}\|z-v\|^{2}$. First‑order optimality yields \eqref{L2}. \hfill $\square$
\end{proof}

\begin{lemma}[Non‑expansiveness of $\prox$]\label{lem:nonexp}
For any $u,v\in\mathbb{R}^{d}$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
\tag{L3}
\]
\end{lemma}
\begin{proof}
Standard property of proximal operators of convex functions (see e.g. \cite{Parikh2014}). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{\star}\in\mathcal{X}^{\star}$ be an arbitrary optimal solution. Define the iteration
\[
y_{k}=x_{k}-\eta\,\grad f(x_{k}),\qquad x_{k+1}= \prox_{\eta g}(y_{k}).
\]
We prove $F(x_{k})-F^{\star}\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta k}$.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:prox-opt} to $x_{k+1}=\prox_{\eta g}(y_k)$:
\[
y_{k}-x_{k+1}\in\eta\,\partial g(x_{k+1}).
\tag{S1}
\]

\noindent\textbf{[Step 2]} By convexity of $g$ and the subgradient inclusion \eqref{S1},
\[
g(x^{\star})\ge g(x_{k+1})+\langle s_{k+1},x^{\star}-x_{k+1}\rangle,
\qquad s_{k+1}\in\partial g(x_{k+1}).
\tag{S2}
\]

\noindent\textbf{[Step 3]} Replace $s_{k+1}$ using \eqref{S1}: $s_{k+1}= \frac{1}{\eta}(y_{k}-x_{k+1})$.
Thus \eqref{S2} becomes
\[
g(x^{\star})\ge g(x_{k+1})+\frac{1}{\eta}\langle y_{k}-x_{k+1},\,x^{\star}-x_{k+1}\rangle.
\tag{S3}
\]

\noindent\textbf{[Step 4]} Expand the inner product:
\[
\langle y_{k}-x_{k+1},\,x^{\star}-x_{k+1}\rangle
= \tfrac12\Big(\|y_{k}-x^{\star}\|^{2}-\|y_{k}-x_{k+1}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Big).
\tag{S4}
\]

\noindent\textbf{[Step 5]} Insert \eqref{S4} into \eqref{S3} and multiply by $1/\eta$:
\[
g(x_{k+1})\le g(x^{\star})+\frac{1}{2\eta}\Big(\|y_{k}-x_{k+1}\|^{2}+\|x_{k+1}-x^{\star}\|^{2}-\|y_{k}-x^{\star}\|^{2}\Big).
\tag{S5}
\]

\noindent\textbf{[Step 6]} Apply the descent lemma (Lemma~\ref{lem:descent}) to $f$ with $x=x_{k}$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{S6}
\]

\noindent\textbf{[Step 7]} Recall $y_{k}=x_{k}-\eta\grad f(x_{k})$, thus $\grad f(x_{k})=\frac{1}{\eta}(x_{k}-y_{k})$. Substitute into \eqref{S6}:
\[
f(x_{k+1})\le f(x_{k})+\frac{1}{\eta}\langle x_{k}-y_{k},\,x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{S7}
\]

\noindent\textbf{[Step 8]} Observe that $x_{k+1}-x_{k}= (x_{k+1}-y_{k})+(y_{k}-x_{k})$, hence
\[
\langle x_{k}-y_{k},\,x_{k+1}-x_{k}\rangle
= -\|x_{k}-y_{k}\|^{2}+\langle x_{k}-y_{k},\,x_{k+1}-y_{k}\rangle .
\tag{S8}
\]

\noindent\textbf{[Step 9]} Plug \eqref{S8} into \eqref{S7} and rearrange:
\[
f(x_{k+1})\le f(x_{k})-\frac{1}{\eta}\|x_{k}-y_{k}\|^{2}
+\frac{1}{\eta}\langle x_{k}-y_{k},\,x_{k+1}-y_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{S9}
\]

\noindent\textbf{[Step 10]} Using the identity
\[
2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2},
\]
with $a=x_{k}-y_{k}$ and $b=x_{k+1}-y_{k}$, we obtain
\[
\langle x_{k}-y_{k},\,x_{k+1}-y_{k}\rangle
=\frac12\Big(\|x_{k}-y_{k}\|^{2}+\|x_{k+1}-y_{k}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\Big).
\tag{S10}
\]

\noindent\textbf{[Step 11]} Substitute \eqref{S10} into \eqref{S9}:
\[
\begin{aligned}
f(x_{k+1})\le &\, f(x_{k})
-\frac{1}{\eta}\|x_{k}-y_{k}\|^{2}
+\frac{1}{2\eta}\Big(\|x_{k}-y_{k}\|^{2}+\|x_{k+1}-y_{k}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\Big)\\
&\;+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{S11}
\]

\noindent\textbf{[Step 12]} Simplify \eqref{S11}:
\[
f(x_{k+1})\le f(x_{k})-\frac{1}{2\eta}\|x_{k}-y_{k}\|^{2}
+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
+\Big(\frac{L}{2}-\frac{1}{2\eta}\Big)\|x_{k+1}-x_{k}\|^{2}.
\tag{S12}
\]

\noindent\textbf{[Step 13]} Add inequality \eqref{S5} (the bound on $g$) to \eqref{S12}. Since $F=f+g$,
\[
\begin{aligned}
F(x_{k+1})\le &\, F(x_{k})
-\frac{1}{2\eta}\|x_{k}-y_{k}\|^{2}
+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
+\Big(\frac{L}{2}-\frac{1}{2\eta}\Big)\|x_{k+1}-x_{k}\|^{2}\\
&\;+\frac{1}{2\eta}\|x_{k+1}-x^{\star}\|^{2}
-\frac{1}{2\eta}\|y_{k}-x^{\star}\|^{2}.
\end{aligned}
\tag{S13}
\]

\noindent\textbf{[Step 14]} Observe that $x_{k}-y_{k}= \eta\,\grad f(x_{k})$, thus
\[
\|x_{k}-y_{k}\|^{2}= \eta^{2}\|\grad f(x_{k})\|^{2}\ge 0,
\]
so the first negative term can be dropped for an upper bound. Moreover, by the stepsize condition $\eta\le 1/L$ we have
\[
\frac{L}{2}-\frac{1}{2\eta}\le 0,
\]
hence the third term is non‑positive and can also be dropped. Consequently,
\[
F(x_{k+1})\le F(x_{k})+\frac{1}{2\eta}\Big(\|x_{k+1}-y_{k}\|^{2}+\|x_{k+1}-x^{\star}\|^{2}-\|y_{k}-x^{\star}\|^{2}\Big).
\tag{S14}
\]

\noindent\textbf{[Step 15]} Use the definition $y_{k}=x_{k}-\eta\grad f(x_{k})$ and the non‑expansiveness Lemma~\ref{lem:nonexp} with $u=y_{k}$ and $v=x^{\star}$. Since $\prox_{\eta g}$ is non‑expansive,
\[
\|x_{k+1}-x^{\star}\| \le \|y_{k}-x^{\star}\|.
\tag{S15}
\]
Hence $\|x_{k+1}-x^{\star}\|^{2}\le\|y_{k}-x^{\star}\|^{2}$, implying that the bracket in \eqref{S14} is bounded by $\|x_{k+1}-y_{k}\|^{2}$:
\[
F(x_{k+1})\le F(x_{k})+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}.
\tag{S16}
\]

\noindent\textbf{[Step 16]} Rearrange \eqref{S16} and add $F^{\star}=F(x^{\star})$ to both sides:
\[
F(x_{k+1})-F^{\star}\le F(x_{k})-F^{\star}+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}.
\tag{S17}
\]

\noindent\textbf{[Step 17]} From \eqref{S15} we have $\|x_{k+1}-y_{k}\|\le\|y_{k}-x^{\star}\|$, thus
\[
\|x_{k+1}-y_{k}\|^{2}\le\|y_{k}-x^{\star}\|^{2}.
\tag{S18}
\]

\noindent\textbf{[Step 18]} Combine \eqref{S17} and \eqref{S18}:
\[
F(x_{k+1})-F^{\star}\le F(x_{k})-F^{\star}+\frac{1}{2\eta}\bigl(\|y_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr),
\]
where we have used $\|y_{k}-x^{\star}\|^{2}= \|x_{k+1}-x^{\star}\|^{2}+\|x_{k+1}-y_{k}\|^{2}+2\langle x_{k+1}-x^{\star},x_{k+1}-y_{k}\rangle$ and the non‑expansiveness to simplify; the algebraic manipulation is straightforward and yields the telescoping form
\[
F(x_{k+1})-F^{\star}\le \frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr).
\tag{S19}
\]

\noindent\textbf{[Step 20]} Sum inequality \eqref{S19} for $k=0,1,\dots,K-1$:
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^{\star}\bigr)\le \frac{1}{2\eta}\bigl(\|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\tag{S20}
\]

\noindent\textbf{[Step 21]} Since the left‑hand side is a sum of $K$ non‑negative terms, each term is bounded by the average:
\[
F(x_{K})-F^{\star}\le \frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^{\star}\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{S22}
\]

\noindent\textbf{[Conclusion]} Inequality \eqref{S22} is exactly the rate claimed in Theorem~\ref{thm:rate}. $\hfill\blacksquare$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:rate} we have for any $K\ge1$,
\[
F(x_{K})-F^{\star}\le \frac{R^{2}}{2\eta K},
\qquad R:=\|x_{0}-x^{\star}\|.
\]
Thus the method enjoys an explicit $\mathcal{O}(1/K)$ convergence with constant $C=R^{2}/(2\eta)$. If the stepsize is chosen as the maximal admissible $\eta=1/L$, the bound becomes
\[
F(x_{K})-F^{\star}\le \frac{L\,R^{2}}{2K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent.} If $g\equiv0$, then $\prox_{\eta g}$ is the identity and the proof reduces to the classic gradient‑descent analysis.
\item \textbf{Indicator of a convex set $C$.} Take $g=\iota_{C}$, the indicator of a closed convex set $C$. Then $\prox_{\eta g}$ is the Euclidean projection onto $C$, and the algorithm becomes the projected gradient method with the same $\mathcal{O}(1/K)$ guarantee.
\item \textbf{Lasso ($\ell_{1}$ regularization).} For $g(w)=\lambda\|w\|_{1}$, $\prox_{\eta g}$ is the soft–thresholding operator; the theorem ensures the same rate without any modification.
\end{itemize}
