\section*{Proximal Gradient Method (PGM)}

The algorithm solves the composite convex problem  

\[
\min_{x\in\R^{d}} \; F(x)\;:=\;f(x)+g(x),
\]

where $f$ is convex and $L$‑smooth (differentiable with $L$‑Lipschitz gradient) and $g$ is proper, closed and convex (possibly nonsmooth).

\section*{Mathematical Formulation}

Given a stepsize $\eta>0$, the iterates are defined by  

\[
\boxed{\;
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta \nabla f(x_{k})\bigr),\qquad k=0,1,2,\dots
\;}
\tag{1}
\]

The proximal operator of $g$ with parameter $\eta$ is  

\[
\prox_{\eta g}(z)\;:=\;\arg\min_{u\in\R^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Bigr\}.
\tag{2}
\]

\section*{Pseudocode}

\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\R^{d}$, max iterations $K$.
\FOR{$k=0$ to $K-1$}
    \STATE Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \STATE Take a gradient step $y_{k}\gets x_{k}-\eta g_{k}$.
    \STATE Apply proximal map $x_{k+1}\gets \prox_{\eta g}(y_{k})$.
\ENDFOR
\STATE \textbf{Output:} $x_{K}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

Consider a scalar problem with  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so that $\prox_{\eta g}(z)=z$ (identity).  
Take $w_{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\ 
\end{tabular}
\end{center}

The objective values are  

\[
F(w_{0})=9,\;F(w_{1})=(0.6-3)^{2}=5.76,\;F(w_{2})=(1.08-3)^{2}=3.6864,\;F(w_{3})=(1.464-3)^{2}=2.3677.
\]

The values decrease, illustrating the descent property of (1).

\section*{Assumptions}

\begin{enumerate}
\item[(A1)] \textbf{Convexity.} $f$ and $g$ are convex on $\R^{d}$.
\item[(A2)] \textbf{$L$‑smoothness of $f$.} For all $x,y\in\R^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{3}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{4}
\]
\item[(A3)] \textbf{Stepsize.} The constant stepsize satisfies  
\[
0<\eta\le\frac{1}{L}.
\tag{5}
\]
\item[(A4)] \textbf{Existence of a minimiser.} The solution set
\[
\mathcal{X}^{\star}:=\arg\min_{x\in\R^{d}}F(x)
\]
is non‑empty; pick any $x^{\star}\in\mathcal{X}^{\star}$.
\end{enumerate}

\section*{Main Convergence Theorem}

\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Under (A1)–(A4) and the iteration (1) with stepsize (5), the sequence $\{x_{k}\}$ satisfies for all $k\ge1$  

\[
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,k}.
\tag{6}
\]

Consequently, $F(x_{k})\to F(x^{\star})$ and $x_{k}$ converges to a minimiser.
\end{theorem}

\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Stability.} The iteration (1) is a non‑expansive mapping composed with a firmly non‑expansive proximal operator; thus the generated sequence is bounded whenever a minimiser exists.
\item \textbf{Hyperparameter Sensitivity.} The bound (6) deteriorates linearly with $1/\eta$, but the requirement $\eta\le1/L$ guarantees the descent inequality used in the proof. Choosing $\eta$ close to $1/L$ yields the smallest constant.
\item \textbf{Comparison to Gradient Descent.} When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and (6) coincides with the classic $O(1/k)$ rate for gradient descent on smooth convex $f$.
\item \textbf{Extension.} If $f$ is $\mu$‑strongly convex, the same proof yields a linear rate $F(x_{k})-F(x^{\star})\le (1-\eta\mu)^{k}\|x_{0}-x^{\star}\|^{2}/(2\eta)$.
\end{itemize}

\section*{Discussion}

The theorem follows directly from the non‑expansiveness of $\prox_{\eta g}$ and the smoothness of $f$. No stochasticity or variance terms appear, making the analysis deterministic and elementary. The result is optimal for first‑order methods on the class of convex $L$‑smooth composite problems.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Non‑expansiveness of proximal operator]\label{lem:nonexp}
For any $z,z'\in\R^{d}$ and any $\eta>0$,
\[
\bigl\| \prox_{\eta g}(z)-\prox_{\eta g}(z') \bigr\|\le \|z-z'\|.
\tag{7}
\]
\end{lemma}
\begin{proof}
Standard property of the proximal map; it follows from the firm non‑expansiveness of the resolvent of a monotone operator. \(\square\)
\end{proof}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Direct consequence of (3). \(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}

Let $x_{k+1}$ be defined by (1). Denote $y_{k}=x_{k}-\eta\nabla f(x_{k})$ so that $x_{k+1}=\prox_{\eta g}(y_{k})$.

\noindent\textbf{[Step 1]} Optimality condition of the proximal map.  
By definition (2),

\[
x_{k+1}\in\arg\min_{u}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-y_{k}\|^{2}\Bigr\}.
\]

Therefore, for any $u\in\R^{d}$,

\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
\;\le\;
g(u)+\frac{1}{2\eta}\|u-y_{k}\|^{2}.
\tag{9}
\]

\noindent\textbf{[Step 2]} Choose $u=x^{\star}$ (a minimiser). Plugging $u=x^{\star}$ into (9) yields

\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
\;\le\;
g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-y_{k}\|^{2}.
\tag{10}
\]

\noindent\textbf{[Step 3]} Expand the right‑hand side of (10).  
Recall $y_{k}=x_{k}-\eta\nabla f(x_{k})$, thus

\[
\begin{aligned}
\|x^{\star}-y_{k}\|^{2}
&=\|x^{\star}-x_{k}+\eta\nabla f(x_{k})\|^{2}\\
&=\|x^{\star}-x_{k}\|^{2}+2\eta\langle x^{\star}-x_{k},\nabla f(x_{k})\rangle+\eta^{2}\|\nabla f(x_{k})\|^{2}.
\end{aligned}
\tag{11}
\]

\noindent\textbf{[Step 4]} Insert (11) into (10) and rearrange:

\[
\begin{aligned}
g(x_{k+1})+ \frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
&\le g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-x_{k}\|^{2}\\
&\quad+ \langle x^{\star}-x_{k},\nabla f(x_{k})\rangle
+\frac{\eta}{2}\|\nabla f(x_{k})\|^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 5]} Apply the Descent Lemma (Lemma~\ref{lem:descent}) with $x=x_{k}$ and $y=x_{k+1}$:

\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{13}
\]

\noindent\textbf{[Step 6]} Add (12) and (13).  
Since $F(\cdot)=f(\cdot)+g(\cdot)$, we obtain

\[
\begin{aligned}
F(x_{k+1}) &\le f(x_{k})+g(x^{\star})+\frac{1}{2\eta}\|x^{\star}-x_{k}\|^{2}\\
&\quad+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&\quad+\langle x^{\star}-x_{k},\nabla f(x_{k})\rangle
+\frac{\eta}{2}\|\nabla f(x_{k})\|^{2}
-\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}.
\end{aligned}
\tag{14}
\]

\noindent\textbf{[Step 7]} Simplify the inner products.  
Observe that

\[
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\langle x^{\star}-x_{k},\nabla f(x_{k})\rangle
= \langle\nabla f(x_{k}),x_{k+1}+x^{\star}-2x_{k}\rangle.
\tag{15}
\]

But $y_{k}=x_{k}-\eta\nabla f(x_{k})$ implies  

\[
x_{k+1}-y_{k}=x_{k+1}-x_{k}+\eta\nabla f(x_{k}).
\tag{16}
\]

Hence  

\[
\|x_{k+1}-y_{k}\|^{2}
= \|x_{k+1}-x_{k}\|^{2}+2\eta\langle x_{k+1}-x_{k},\nabla f(x_{k})\rangle+\eta^{2}\|\nabla f(x_{k})\|^{2}.
\tag{17}
\]

\noindent\textbf{[Step 8]} Substitute (17) into the last term of (14) and cancel the mixed inner product terms. After straightforward algebra we obtain

\[
\begin{aligned}
F(x_{k+1})
&\le g(x^{\star})+f(x_{k})+\frac{1}{2\eta}\|x^{\star}-x_{k}\|^{2}\\
&\quad -\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
+\frac{L-\frac{1}{\eta}}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{18}
\]

\noindent\textbf{[Step 9]} Use the stepsize condition (5), i.e. $\eta\le 1/L$, which implies $L-\frac{1}{\eta}\le0$. Therefore the last term in (18) is non‑positive and can be dropped, yielding the fundamental descent inequality

\[
F(x_{k+1})\le F(x^{\star})+\frac{1}{2\eta}\bigl(\|x^{\star}-x_{k}\|^{2}-\|x^{\star}-x_{k+1}\|^{2}\bigr).
\tag{19}
\]

\noindent\textbf{[Step 10]} Rearranging (19) gives  

\[
F(x_{k+1})-F(x^{\star})\le \frac{1}{2\eta}\Bigl(\|x^{\star}-x_{k}\|^{2}-\|x^{\star}-x_{k+1}\|^{2}\Bigr).
\tag{20}
\]

\noindent\textbf{[Step 11]} Sum (20) for $k=0,\dots,K-1$ (telescoping sum):

\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \frac{1}{2\eta}\bigl(\|x^{\star}-x_{0}\|^{2}-\|x^{\star}-x_{K}\|^{2}\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\tag{21}
\]

\noindent\textbf{[Step 12]} Since each term on the left is non‑negative, the average satisfies  

\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{22}
\]

In particular, for the last iterate $x_{K}$ we have (by monotonicity of the bound)

\[
F(x_{K})-F(x^{\star})\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K},
\tag{23}
\]

which is exactly the claim (6) of Theorem~\ref{thm:rate}. \(\square\)

\section*{Rate Derivation (With Constants)}

From (23) we can read off the explicit constant:

\[
C:=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\]

Thus the convergence rate is  

\[
F(x_{k})-F(x^{\star})\le \frac{C}{k},\qquad k\ge1.
\]

If $f$ is additionally $\mu$‑strongly convex, the same steps with the stronger inequality  

\[
\langle\nabla f(x_{k})- \nabla f(x^{\star}),x_{k}-x^{\star}\rangle\ge \mu\|x_{k}-x^{\star}\|^{2}
\]

lead to  

\[
\|x_{k+1}-x^{\star}\|^{2}\le (1-\eta\mu)\|x_{k}-x^{\star}\|^{2},
\]

hence a linear rate $O\bigl((1-\eta\mu)^{k}\bigr)$.

\section*{Special Cases}

\begin{itemize}
\item \textbf{Pure Gradient Descent ($g\equiv0$).} The proximal step reduces to identity, and (19) becomes the classic descent lemma for gradient descent.
\item \textbf{Indicator function $g=\iota_{C}$ (projection onto convex set $C$).} Then $\prox_{\eta g}$ is the Euclidean projection $\Pi_{C}$, and the algorithm is the projected gradient method with the same $O(1/k)$ guarantee.
\item \textbf{Lasso ($g(x)=\lambda\|x\|_{1}$).} The proximal map is the soft‑thresholding operator; the bound (6) applies unchanged, giving the standard rate for proximal gradient applied to $\ell_{1}$‑regularised least squares.
\end{itemize}
