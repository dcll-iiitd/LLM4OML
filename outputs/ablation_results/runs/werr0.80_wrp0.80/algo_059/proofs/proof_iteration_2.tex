\section*{Algorithm Name}
\textbf{Quantized Stochastic Gradient Descent (Q‑SGD)}  

The method is identical to vanilla SGD except that each worker transmits a \emph{randomly quantized} version of its stochastic gradient. The quantizer $\Q:\mathbb{R}^d\to\mathbb{R}^d$ satisfies $\E[\Q(g)]=g$ (unbiasedness) and a bounded second‑moment property.

--------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the (possibly non‑convex) loss we wish to minimize.
At iteration $t$ we draw an i.i.d.\ data sample $\xi_t$ and compute a stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t).
\]
Instead of sending $g_t$, we send its quantized version
\[
\tilde g_t \;:=\; \Q(g_t),\qquad \E[\tilde g_t\mid w_t]=g_t .
\]
The central parameter server (or the local copy) updates
\[
\boxed{\,w_{t+1}=w_t-\eta_t\,\tilde g_t\,},
\]
where $\eta_t>0$ is the learning‑rate schedule.

--------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, stepsizes $\{\eta_t\}_{t\ge 0}$, quantizer $\Q$
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample a data point $\xi_t$ (or a minibatch)  
    \STATE Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$
    \STATE Quantize: $\tilde g_t=\Q(g_t)$   \hfill\Comment{Unbiased: $\E[\tilde g_t|w_t]=g_t$}
    \STATE Update parameters: $w_{t+1}=w_t-\eta_t\tilde g_t$
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3).
\]
We use a *stochastic* rounding quantizer: for any real $g$,
\[
\Q(g)=
\begin{cases}
\operatorname{round}(g,0.1)      &\text{with probability }p,\\[4pt]
\operatorname{round}(g,0.1)+0.1 &\text{with probability }1-p,
\end{cases}
\]
where $p$ is chosen so that $\E[\Q(g)]=g$ (standard stochastic rounding).  
Parameters: $w_0=0$, constant stepsize $\eta=0.1$, run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $\tilde g_t=\Q(g_t)$ & $w_{t+1}=w_t-\eta\tilde g_t$ \\ \hline
0 & $0$   & $-6$   & $-6.0$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$& $-3.84$& $-3.8$ & $1.08-0.1(-3.8)=1.46$ \\
\end{tabular}
\end{center}

Objective values:
\[
f(w_0)=9,\; f(w_1)=5.76,\; f(w_2)=3.69,\; f(w_3)=2.38.
\]
The iterates move towards the optimum $w^\star=3$, illustrating the effect of quantized updates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For any $w$, the stochastic gradient satisfies
\[
\E_{\xi}[\,\nabla f(w;\xi)\mid w\,]=\nabla f(w),\qquad
\E_{\xi}\!\big[\norm{\nabla f(w;\xi)}^2\mid w\big]\le G^2 .
\]
\end{assumption}
\begin{assumption}[Quantizer]\label{ass:quant}
The random quantizer $\Q:\mathbb{R}^d\to\mathbb{R}^d$ fulfills
\begin{enumerate}[label=(\roman*)]
\item \textbf{Unbiasedness:} $\E[\Q(g)\mid g]=g$ for any $g\in\mathbb{R}^d$.
\item \textbf{Bounded variance:} there exists $\omega\ge0$ such that
\[
\E\!\big[\norm{\Q(g)-g}^2\mid g\big]\le\omega\,\norm{g}^2 .
\]
\end{enumerate}
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The learning rates are deterministic, non‑increasing and satisfy
\[
\eta_t\le \frac{1}{L(1+\omega)}\qquad\forall t .
\]
\end{assumption}

--------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of Q‑SGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let $w_T$ be the iterate produced by Q‑SGD after $T$ steps and define the averaged gradient norm
\[
\Phi_T\;:=\;\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\norm{\nabla f(w_t)}^2\big].
\]
If a constant stepsize $\eta_t\equiv\eta\le\frac{1}{L(1+\omega)}$ is used, then
\[
\Phi_T\;\le\;
\frac{f(w_0)-f^\star}{\eta T}
\;+\;
\frac{L\eta}{2}\,(1+\omega)\,G^2 .
\tag{1}
\]
Equivalently,
\[
\Phi_T\;\le\;
\frac{f(w_0)-f^\star}{\eta T}
\;+\;
\frac{L\eta}{2}G^2
\;+\;
\frac{L\eta}{2}\,\omega\,G^2 .
\tag{1$'$}
\]
Consequently, choosing $\eta=\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\Phi_T = O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

--------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (1) holds for any $L$‑smooth non‑convex $f$, showing that quantization does not destroy the descent property as long as $\eta\le1/[L(1+\omega)]$.
\item \textbf{Effect of quantization.} The extra term $\frac{L\eta}{2}\,\omega\,G^2$ is linear in $\omega$ (the relative variance of the quantizer). When $\omega=0$ (exact gradients) the theorem reduces to the classical SGD bound.
\item \textbf{Hyper‑parameter sensitivity.} The stepsize must respect the stricter bound $\eta\le 1/(L(1+\omega))$; larger $\omega$ forces a smaller admissible learning rate.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \emph{Plain SGD} (no quantization): $\omega=0$, thus the quantization term disappears.
        \item \emph{Deterministic rounding} is not unbiased; a stochastic rounding scheme satisfies the assumptions.
    \end{enumerate}
\end{itemize}

--------------------------------------------------------------------

\section*{Discussion}
The theorem demonstrates that quantized communication—common in distributed deep learning—preserves the $O(1/\sqrt{T})$ convergence rate for smooth non‑convex objectives, up to an additive term proportional to the quantizer variance $\omega$. By decreasing the stepsize at the customary $1/\sqrt{T}$ schedule, the quantization term vanishes asymptotically, guaranteeing that the average gradient norm converges to zero.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function $f$,
\[
f(w_{t+1})\le f(w_t)+\inner{\nabla f(w_t)}{w_{t+1}-w_t}
+\frac{L}{2}\norm{w_{t+1}-w_t}^2 .
\]
\end{lemma}
\begin{proof}
Directly from Definition \ref{ass:smooth} with $x=w_t$, $y=w_{t+1}$. \qedhere
\end{proof}

\begin{lemma}[Quantizer second‑moment bound]\label{lem:quant}
For any $g\in\mathbb{R}^d$,
\[
\E\big[\norm{\Q(g)}^2\mid g\big]\le (1+\omega)\norm{g}^2 .
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\E\!\big[\norm{\Q(g)}^2\mid g\big]
&=\E\!\big[\norm{g+(\Q(g)-g)}^2\mid g\big] \\
&=\norm{g}^2+2\inner{g}{\E[\Q(g)-g\mid g]}+\E\!\big[\norm{\Q(g)-g}^2\mid g\big].
\end{aligned}
\]
Unbiasedness makes the middle term zero; applying Assumption \ref{ass:quant}(ii) yields the claim. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken over the randomness of the stochastic gradient and the quantizer, conditioned on the past iterates.

\medskip
\noindent\textbf{Step 1 (Smoothness applied).}  
From Lemma \ref{lem:smooth} and the update $w_{t+1}=w_t-\eta\tilde g_t$,
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_t)+\inner{\nabla f(w_t)}{-\eta\tilde g_t}
+\frac{L}{2}\norm{\eta\tilde g_t}^2 \\
&= f(w_t)-\eta\inner{\nabla f(w_t)}{\tilde g_t}
+\frac{L\eta^2}{2}\norm{\tilde g_t}^2 .
\tag{2}
\end{aligned}
\] 
[Step 1]

\medskip
\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $w_t$ and use the unbiasedness of both the stochastic gradient and the quantizer:
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid w_t\big]
&\le f(w_t)-\eta\norm{\nabla f(w_t)}^2
+\frac{L\eta^2}{2}\E\!\big[\norm{\tilde g_t}^2\mid w_t\big] .
\end{aligned}
\tag{3}
\] 
[Step 2]

\medskip
\noindent\textbf{Step 3 (Bound the second moment of the quantized gradient).}  
Using Lemma \ref{lem:quant} and the bounded second moment of the stochastic gradient (Assumption \ref{ass:unbiased}),
\[
\begin{aligned}
\E\!\big[\norm{\tilde g_t}^2\mid w_t\big]
&\le (1+\omega)\E\!\big[\norm{g_t}^2\mid w_t\big] \\
&\le (1+\omega)G^2 .
\end{aligned}
\tag{4}
\] 
[Step 3]

\medskip
\noindent\textbf{Step 4 (Insert (4) into (3)).}
\[
\E\!\big[f(w_{t+1})\mid w_t\big]
\le f(w_t)-\eta\norm{\nabla f(w_t)}^2
+\frac{L\eta^2}{2}(1+\omega)G^2 .
\tag{5}
\] 
[Step 4]

\medskip
\noindent\textbf{Step 5 (Take full expectation and sum).}  
Denote $\Phi_T:=\frac{1}{T}\sum_{t=0}^{T-1}\E[\norm{\nabla f(w_t)}^2]$. Summing (5) over $t=0,\dots,T-1$ and telescoping the left‑hand side gives
\[
\begin{aligned}
\E[f(w_T)]-f(w_0)
&\le -\eta\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
+\frac{L\eta^2}{2}(1+\omega)G^2 T .
\end{aligned}
\tag{6}
\] 
[Step 5]

\medskip
\noindent\textbf{Step 6 (Rearrange).}
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le f(w_0)-\E[f(w_T)]+ \frac{L\eta^2}{2}(1+\omega)G^2 T .
\tag{7}
\] 
Since $f(w_T)\ge f^\star$, we replace $\E[f(w_T)]$ by $f^\star$:
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le f(w_0)-f^\star+ \frac{L\eta^2}{2}(1+\omega)G^2 T .
\tag{8}
\] 
[Step 6]

\medskip
\noindent\textbf{Step 7 (Divide by $T\eta$).}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^2\big]
\le \frac{f(w_0)-f^\star}{\eta T}
+ \frac{L\eta}{2}(1+\omega)G^2 .
\tag{9}
\] 
[Step 7]

\medskip
\noindent\textbf{Step 8 (Correct handling of the $\omega$ term).}  
The right‑hand side of (9) is already the tight bound required in (1).  
If one wishes to separate the contribution of quantization, we simply split the factor $(1+\omega)$:
\[
\frac{L\eta}{2}(1+\omega)G^2
= \frac{L\eta}{2}G^2 \;+\; \frac{L\eta}{2}\,\omega\,G^2 .
\tag{10}
\]
No further manipulation is needed.  
The stepsize condition $\eta\le 1/[L(1+\omega)]$ guarantees $L\eta\le 1$, which is used only to ensure that the term $\frac{L\eta}{2}G^2$ does not dominate the bound; it does **not** allow us to replace (10) by $2L\omega\eta G^2$, a step that was incorrect in the original manuscript.  
Thus the final bound is exactly (1) (or equivalently (1$'$)). \qedhere
[Step 8]

--------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
Choose a constant stepsize
\[
\eta = \min\!\Big\{\frac{1}{L(1+\omega)},\;\sqrt{\frac{2\big(f(w_0)-f^\star\big)}{L(1+\omega)G^2 T}}\Big\}.
\]
Substituting this $\eta$ into (1) yields
\[
\Phi_T \;\le\; O\!\Big(\frac{1}{\sqrt{T}}\Big) \;+\; O\!\big(\omega\,\tfrac{1}{\sqrt{T}}\big)
= O\!\Big(\frac{1}{\sqrt{T}}\Big),
\]
so the $O(1/\sqrt{T})$ rate is retained even in the presence of quantization.

--------------------------------------------------------------------

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact gradients} ($\omega=0$).  The bound reduces to the standard SGD guarantee
\[
\Phi_T\le\frac{f(w_0)-f^\star}{\eta T}+\frac{L\eta}{2}G^2,
\]
and with $\eta\asymp1/\sqrt{T}$ we obtain $\Phi_T=O(1/\sqrt{T})$.
\item \textbf{Uniform stochastic quantizer} with $k$ bits per coordinate satisfies $\omega = \frac{d}{2^{2k}}$ (see Alistarh \emph{et al.}, 2017). Plugging this $\omega$ into (1) quantifies precisely how many bits are needed to keep the extra term negligible.
\item \textbf{Increasing minibatch size}. If a minibatch of size $B$ is used, the stochastic‑gradient variance scales as $G^2/B$, and the bound (1) becomes
\[
\Phi_T\le\frac{f(w_0)-f^\star}{\eta T}+ \frac{L\eta}{2}(1+\omega)\frac{G^2}{B},
\]
showing a trade‑off between communication (via $\omega$) and statistical variance (via $B$).
\end{enumerate}

--------------------------------------------------------------------

\section*{Conclusion}
We have presented a fully detailed and mathematically rigorous derivation of the convergence guarantee for Quantized Stochastic Gradient Descent on smooth non‑convex objectives. The proof proceeds from first principles, carefully expanding every term and annotating each non‑trivial manipulation with explicit step numbers. Under standard smoothness, unbiasedness, and bounded‑variance assumptions on both the stochastic gradient and the quantizer, Q‑SGD achieves the same $O(1/\sqrt{T})$ decay of the average gradient norm as vanilla SGD, with an additional term proportional to the quantizer variance $\omega$ that vanishes when the stepsize is chosen appropriately or when $\omega$ is small.
