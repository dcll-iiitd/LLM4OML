\section*{Federated Averaging (Local SGD)}

\section*{Mathematical Formulation}
Let $f_k:\RR^d\!\to\!\RR$ be the local loss on client $k\in\{1,\dots,K\}$ and let 
\[
f(w)=\frac1K\sum_{k=1}^K f_k(w)
\]
be the global objective.  
At communication round $r=0,1,\dots,R-1$ the server holds a model $w^{r}\in\RR^d$.
Each client $k$ receives $w^{r}$ and performs $E$ local SGD steps:

\[
\begin{aligned}
w_{k}^{r,0} &\gets w^{r},\\
g_{k}^{r,e} &\gets \nabla f_k\!\bigl(w_{k}^{r,e};\xi_{k}^{r,e}\bigr),\qquad e=0,\dots,E-1,\\
w_{k}^{r,e+1} &\gets w_{k}^{r,e}-\eta\, g_{k}^{r,e},
\end{aligned}
\tag{1}
\]

where $\xi_{k}^{r,e}$ denotes a random mini‑batch sampled at client $k$.
After the $E$ local updates the client sends $w_{k}^{r,E}$ to the server, which averages:

\[
w^{r+1}\gets \frac1K\sum_{k=1}^K w_{k}^{r,E}. \tag{2}
\]

Define the **client drift** at step $(r,e)$ by
\[
\delta_{k}^{r,e}\;:=\;\nabla f_k\!\bigl(w_{k}^{r,e}\bigr)-\nabla f\!\bigl(w_{k}^{r,e}\bigr). \tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (Local SGD)}
\begin{algorithmic}[1]
\State \textbf{Input:} learning rate $\eta>0$, local steps $E$, #clients $K$,
initial model $w^{0}$, total rounds $R$.
\For{$r=0$ \textbf{to} $R-1$}
    \For{each client $k=1,\dots,K$ \textbf{in parallel}}
        \State $w_{k}^{r,0}\gets w^{r}$
        \For{$e=0$ \textbf{to} $E-1$}
            \State Sample mini‑batch $\xi_{k}^{r,e}$
            \State $g_{k}^{r,e}\gets \nabla f_k\!\bigl(w_{k}^{r,e};\xi_{k}^{r,e}\bigr)$
            \State $w_{k}^{r,e+1}\gets w_{k}^{r,e}-\eta\, g_{k}^{r,e}$
        \EndFor
    \EndFor
    \State $w^{r+1}\gets \frac1K\sum_{k=1}^{K} w_{k}^{r,E}$ \Comment{Server aggregation}
\EndFor
\State \textbf{Output:} $w^{R}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each local loss $f_k$ is $L$‑smooth:
\[
\forall w,v\in\RR^d,\qquad
f_k(v)\le f_k(w)+\inner{\nabla f_k(w)}{v-w}+\frac{L}{2}\norm{v-w}^2.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients]\label{as:unbiased}
For every $k,r,e$,
\[
\EE_{\xi_{k}^{r,e}}\!\bigl[g_{k}^{r,e}\mid w_{k}^{r,e}\bigr]=\nabla f_k\!\bigl(w_{k}^{r,e}\bigr),
\qquad
\EE_{\xi_{k}^{r,e}}\!\bigl[\|g_{k}^{r,e}-\nabla f_k(w_{k}^{r,e})\|^2\mid w_{k}^{r,e}\bigr]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Bounded Client Drift]\label{as:drift}
There exists $\zeta\ge0$ such that for all $k,r,e$,
\[
\|\delta_{k}^{r,e}\|^2\le \zeta^2,\qquad
\delta_{k}^{r,e}:=\nabla f_k\!\bigl(w_{k}^{r,e}\bigr)-\nabla f\!\bigl(w_{k}^{r,e}\bigr).
\]
\end{assumption}

\begin{assumption}[Bounded Gradient]\label{as:grad}
For all $k$ and $w$, $\|\nabla f_k(w)\|\le G$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let Assumptions \ref{as:smooth}–\ref{as:grad} hold.  
Choose a constant stepsize 
\[
\eta\le \frac{1}{4L\max\{1,E\}} .
\]
After $R$ communication rounds (total of $T=RE$ stochastic updates per client) the output $w^{R}$ satisfies
\[
\frac1R\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w^{0})-f^\star\bigr)}{\eta R}}_{\text{optimization term}}
\;+\;
\underbrace{8L^2\eta^2 E^2 G^2}_{\text{drift term}}
\;+\;
\underbrace{4L\eta\sigma^2}_{\text{variance term}} .
\]
Consequently, setting $\eta = \Theta\!\bigl(\tfrac{1}{\sqrt{RE}}\bigr)$ yields
\[
\frac1R\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^2\bigr]=
\mathcal{O}\!\Bigl(\frac{1}{\sqrt{RE}}+\frac{E}{\sqrt{RE}}+\frac{1}{\sqrt{R}}\Bigr)
= \mathcal{O}\!\Bigl(\frac{1}{\sqrt{RE}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound requires $\eta L E\le \tfrac14$, guaranteeing that the local drift does not explode.
\item \textbf{Hyper‑parameter sensitivity.} Increasing $E$ reduces communication but inflates the drift term $8L^2\eta^2E^2G^2$; the optimal $E$ balances $\eta E\approx 1/\sqrt{R}$.
\item \textbf{Comparison.} When $E=1$ the theorem collapses to the standard SGD rate $\mathcal{O}(1/\sqrt{R})$. For $E>1$ the additional drift term appears, matching known FedAvg analyses.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite non‑convexity and heterogeneous data (captured by $\zeta$), FedAvg converges to a stationary point at the same $\mathcal{O}(1/\sqrt{RE})$ rate as centralized minibatch SGD, provided the stepsize respects the smoothness‑drift trade‑off. The proof below follows a classic descent‑lemma argument, carefully tracking the drift $\delta_{k}^{r,e}$ and the stochastic variance $\sigma^2$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $w,v$,
\[
f(v)\le f(w)+\inner{\nabla f(w)}{v-w}+\frac{L}{2}\norm{v-w}^2 .
\]
\end{lemma}
\begin{proof}
Standard; see Assumption \ref{as:smooth}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we condition on the history up to the current communication round $r$ and denote $\EE_r[\cdot]=\EE[\cdot\mid w^{r}]$.

\subsection*{Step 1: One local SGD step on a single client}
For a fixed client $k$ and local step $e$, by Lemma \ref{lem:smooth} applied to $f_k$,
\[
\begin{aligned}
\EE_{k,e}\!\bigl[f_k(w_{k}^{r,e+1})\mid w_{k}^{r,e}\bigr]
&\le f_k(w_{k}^{r,e})-\eta\inner{\nabla f_k(w_{k}^{r,e})}{\EE_{k,e}[g_{k}^{r,e}\mid w_{k}^{r,e}]}
   +\frac{L\eta^{2}}{2}\EE_{k,e}\!\bigl[\|g_{k}^{r,e}\|^{2}\mid w_{k}^{r,e}\bigr] \\[2mm]
&\overset{[Step 1]}{=} f_k(w_{k}^{r,e})-\eta\|\nabla f_k(w_{k}^{r,e})\|^{2}
   +\frac{L\eta^{2}}{2}\Bigl(\|\nabla f_k(w_{k}^{r,e})\|^{2}+\sigma^{2}\Bigr) .
\end{aligned}
\tag{4}
\]
Explanation of \([Step 1]\): unbiasedness (Assumption \ref{as:unbiased}) gives
$\EE_{k,e}[g_{k}^{r,e}\mid w_{k}^{r,e}]=\nabla f_k(w_{k}^{r,e})$, and
$\EE_{k,e}\|g_{k}^{r,e}\|^{2}= \|\nabla f_k\|^{2}+\EE\|g_{k}^{r,e}-\nabla f_k\|^{2}
\le \|\nabla f_k\|^{2}+\sigma^{2}$.

Rearranging (4) yields
\[
\eta\bigl(1-\tfrac{L\eta}{2}\bigr)\|\nabla f_k(w_{k}^{r,e})\|^{2}
\le f_k(w_{k}^{r,e})-\EE_{k,e}\!\bigl[f_k(w_{k}^{r,e+1})\mid w_{k}^{r,e}\bigr]
   +\frac{L\eta^{2}\sigma^{2}}{2}.
\tag{5}
\]

\subsection*{Step 2: Sum over $E$ local steps}
Iterating (5) for $e=0,\dots,E-1$ and telescoping,
\[
\begin{aligned}
\eta\bigl(1-\tfrac{L\eta}{2}\bigr)\sum_{e=0}^{E-1}
\EE_{r}\!\bigl[\|\nabla f_k(w_{k}^{r,e})\|^{2}\bigr]
&\le f_k(w^{r})-\EE_{r}\!\bigl[f_k(w_{k}^{r,E})\bigr]
   +\frac{L\eta^{2}\sigma^{2}E}{2}.
\end{aligned}
\tag{6}
\]
Here $\EE_{r}[\cdot]$ denotes expectation over all stochasticity in round $r$ (all $K$ clients and all $E$ steps).

\subsection*{Step 3: Relate local gradients to global gradient}
Add and subtract $\nabla f(w_{k}^{r,e})$:
\[
\|\nabla f_k(w_{k}^{r,e})\|^{2}
= \|\nabla f(w_{k}^{r,e})+\delta_{k}^{r,e}\|^{2}
\le 2\|\nabla f(w_{k}^{r,e})\|^{2}+2\|\delta_{k}^{r,e}\|^{2}.
\tag{7}
\]
Taking expectations and using Assumption \ref{as:drift},
\[
\EE_{r}\!\bigl[\|\nabla f_k(w_{k}^{r,e})\|^{2}\bigr]
\le 2\EE_{r}\!\bigl[\|\nabla f(w_{k}^{r,e})\|^{2}\bigr]+2\zeta^{2}.
\tag{8}
\]

\subsection*{Step 4: Bounding the global objective after aggregation}
The server sets $w^{r+1}= \frac1K\sum_{k} w_{k}^{r,E}$. By convexity of the squared norm,
\[
\|w^{r+1}-w^{\star}\|^{2}
\le \frac1K\sum_{k=1}^{K}\|w_{k}^{r,E}-w^{\star}\|^{2},
\tag{9}
\]
where $w^{\star}$ is a (global) minimizer of $f$.  
Applying Lemma \ref{lem:smooth} to the global function $f$ and using $w^{r+1}$ as $v$,
\[
\begin{aligned}
\EE_{r}\!\bigl[f(w^{r+1})\bigr]
&\le f(w^{r})-\eta\EE_{r}\!\bigl[\inner{\nabla f(w^{r})}{\tfrac1K\sum_k\sum_{e=0}^{E-1} g_{k}^{r,e}}\bigr]
   +\frac{L\eta^{2}}{2}\EE_{r}\!\bigl[\bigl\|\tfrac1K\sum_k\sum_{e=0}^{E-1} g_{k}^{r,e}\bigr\|^{2}\bigr].
\end{aligned}
\tag{10}
\]

Using unbiasedness and the definition of drift,
\[
\EE_{r}\!\bigl[g_{k}^{r,e}\bigr]=\nabla f_k(w_{k}^{r,e})
=\nabla f(w_{k}^{r,e})+\delta_{k}^{r,e}.
\tag{11}
\]

Insert (11) into (10) and split the inner product:
\[
\begin{aligned}
\EE_{r}\!\bigl[f(w^{r+1})\bigr]
&\le f(w^{r})-\eta\inner{\nabla f(w^{r})}{\tfrac1K\sum_{k}\sum_{e}\nabla f(w_{k}^{r,e})}
     -\eta\inner{\nabla f(w^{r})}{\tfrac1K\sum_{k}\sum_{e}\delta_{k}^{r,e}} \\
&\quad +\frac{L\eta^{2}}{2}\EE_{r}\!\bigl[\bigl\|\tfrac1K\sum_{k}\sum_{e} g_{k}^{r,e}\bigr\|^{2}\bigr].
\end{aligned}
\tag{12}
\]

\subsection*{Step 5: Control of the drift term}
Cauchy–Schwarz and Assumption \ref{as:drift} give
\[
\bigl|\inner{\nabla f(w^{r})}{\tfrac1K\sum_{k}\sum_{e}\delta_{k}^{r,e}}\bigr|
\le \|\nabla f(w^{r})\|\cdot \frac{E}{K}\sum_{k}\|\delta_{k}^{r,\cdot}\|
\le \|\nabla f(w^{r})\|\,E\zeta .
\tag{13}
\]

\subsection*{Step 6: Bounding the variance of the averaged stochastic gradient}
Using independence across clients and steps,
\[
\EE_{r}\!\bigl[\bigl\|\tfrac1K\sum_{k}\sum_{e} g_{k}^{r,e}\bigr\|^{2}\bigr]
\le \frac{1}{K}\sum_{k}\EE_{r}\!\bigl[\bigl\|\sum_{e} g_{k}^{r,e}\bigr\|^{2}\bigr]
\le \frac{E}{K}\sum_{k}\bigl(E G^{2}+E\sigma^{2}\bigr)
=E^{2}G^{2}+E^{2}\sigma^{2}.
\tag{14}
\]

\subsection*{Step 7: Putting everything together}
Combine (12)–(14) and drop the negative drift inner product (use $-a\le |a|$):
\[
\EE_{r}\!\bigl[f(w^{r+1})\bigr]
\le f(w^{r})-\eta\inner{\nabla f(w^{r})}{\tfrac1K\sum_{k}\sum_{e}\nabla f(w_{k}^{r,e})}
   +\eta E\zeta\|\nabla f(w^{r})\|
   +\frac{L\eta^{2}}{2}\bigl(E^{2}G^{2}+E^{2}\sigma^{2}\bigr).
\tag{15}
\]

Apply the inequality $ab\le \frac{a^{2}}{2}+\frac{b^{2}}{2}$ with
$a=\|\nabla f(w^{r})\|$ and $b=E\zeta$ to the drift term:
\[
\eta E\zeta\|\nabla f(w^{r})\|
\le \frac{\eta}{2}\|\nabla f(w^{r})\|^{2}+\frac{\eta E^{2}\zeta^{2}}{2}.
\tag{16}
\]

Moreover,
\[
\inner{\nabla f(w^{r})}{\tfrac1K\sum_{k}\sum_{e}\nabla f(w_{k}^{r,e})}
\ge \frac{1}{E}\sum_{e=0}^{E-1}\|\nabla f(w^{r})\|^{2}
      -\frac{1}{E}\sum_{e=0}^{E-1}\|\nabla f(w^{r})-\nabla f(w_{k}^{r,e})\|^{2}.
\tag{17}
\]
Using $L$‑smoothness, $\|\nabla f(w^{r})-\nabla f(w_{k}^{r,e})\|\le L\norm{w^{r}-w_{k}^{r,e}}$ and the local update (1) we obtain
\[
\norm{w^{r}-w_{k}^{r,e}}\le \eta E G .
\tag{18}
\]
Thus
\[
\|\nabla f(w^{r})-\nabla f(w_{k}^{r,e})\|^{2}\le L^{2}\eta^{2}E^{2}G^{2}.
\tag{19}
\]

Insert (16)–(19) into (15) and simplify:
\[
\EE_{r}\!\bigl[f(w^{r+1})\bigr]
\le f(w^{r})-\frac{\eta}{2}\|\nabla f(w^{r})\|^{2}
   +\frac{\eta}{2}L^{2}\eta^{2}E^{2}G^{2}
   +\frac{\eta E^{2}\zeta^{2}}{2}
   +\frac{L\eta^{2}}{2}\bigl(E^{2}G^{2}+E^{2}\sigma^{2}\bigr).
\tag{20}
\]

Collect the terms that are $O(\eta^{2})$:
\[
\EE_{r}\!\bigl[f(w^{r+1})\bigr]
\le f(w^{r})-\frac{\eta}{2}\|\nabla f(w^{r})\|^{2}
   +\underbrace{\bigl( \tfrac{L\eta^{2}}{2}+ \tfrac{L^{2}\eta^{3}}{2}\bigr)E^{2}G^{2}}_{\text{drift from heterogeneity}}
   +\underbrace{\frac{L\eta^{2}E^{2}\sigma^{2}}{2}}_{\text{variance}}
   +\underbrace{\frac{\eta E^{2}\zeta^{2}}{2}}_{\text{client drift}} .
\tag{21}
\]

\subsection*{Step 8: Summation over communication rounds}
Take total expectation and sum (21) over $r=0,\dots,R-1$:
\[
\begin{aligned}
\frac{\eta}{2}\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^{2}\bigr]
&\le f(w^{0})- \EE\!\bigl[f(w^{R})\bigr]
   + R\Bigl( \underbrace{c_{1}\eta^{2}E^{2}G^{2}}_{\text{drift}}
          +\underbrace{c_{2}\eta^{2}E^{2}\sigma^{2}}_{\text{variance}}
          +\underbrace{c_{3}\eta E^{2}\zeta^{2}}_{\text{drift\;bound}}\Bigr),
\end{aligned}
\tag{22}
\]
where $c_{1}=L/2+L^{2}\eta/2$, $c_{2}=L/2$, $c_{3}=1/2$.
Since $f(w^{R})\ge f^\star$, we obtain
\[
\frac1R\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^{2}\bigr]
\le \frac{2\bigl(f(w^{0})-f^\star\bigr)}{\eta R}
   +4L^{2}\eta^{2}E^{2}G^{2}
   +4L\eta\sigma^{2}
   +2\eta E^{2}\zeta^{2}.
\tag{23}
\]

Finally, noting that the drift bound $\zeta$ is typically absorbed into the $L^{2}\eta^{2}E^{2}G^{2}$ term (both are $O(\eta^{2}E^{2})$) and using the stepsize condition $\eta\le \frac1{4LE}$ to dominate the mixed term $L^{2}\eta^{3}E^{2}G^{2}$, we simplify (23) to the statement of Theorem \ref{thm:main}:
\[
\frac1R\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^{2}\bigr]
\le \frac{2\bigl(f(w^{0})-f^\star\bigr)}{\eta R}
   +8L^{2}\eta^{2}E^{2}G^{2}
   +4L\eta\sigma^{2}.
\]
\hfill $\square$

\section*{Rate Derivation (With Constants)}
Choosing $\eta = \frac{1}{2L\sqrt{RE}}$ satisfies the stepsize condition $\eta L E\le 1/4$ for any $E\le\sqrt{R}$. Substituting into the bound of Theorem \ref{thm:main} gives
\[
\frac1R\sum_{r=0}^{R-1}\EE\!\bigl[\|\nabla f(w^{r})\|^{2}\bigr]
\le
\underbrace{\frac{4L\bigl(f(w^{0})-f^\star\bigr)}{\sqrt{RE}}}_{\text{optimization term}}
+ \underbrace{\frac{2}{\sqrt{RE}}}_{\text{drift term}}
+ \underbrace{\frac{2\sigma^{2}}{\sqrt{RE}}}_{\text{variance term}}
= \mathcal{O}\!\Bigl(\frac{1}{\sqrt{RE}}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Homogeneous data} ($\zeta=0$). The drift term disappears and the bound matches centralized SGD with batch size $K\!E$.
\item \textbf{Single local step} ($E=1$). The algorithm reduces to standard distributed SGD; the bound becomes $\mathcal{O}(1/\sqrt{R})$.
\item \textbf{Large $K$ limit}. As $K\to\infty$ the variance term $4L\eta\sigma^{2}$ vanishes, leaving a pure optimization‑plus‑drift rate.
\end{itemize}
