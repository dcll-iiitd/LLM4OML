\section*{AdaGrad (Scalar Version)}

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be a convex and $L$-smooth objective.  
Denote by $g_k:=\nabla f(w_k)$ the (possibly stochastic) gradient at iteration $k$.
Define the accumulated squared‐gradient
\[
G_k \;:=\; \sum_{i=1}^{k} \|g_i\|^{2}\;\in\R_{\ge 0},
\qquad G_0:=0 .
\]
Given a base learning rate $\eta>0$ and a tiny constant $\epsilon>0$ for numerical stability, the AdaGrad update is
\[
\boxed{%
w_{k+1}\;=\;w_k\;-\;\frac{\eta}{\sqrt{G_k+\epsilon}}\;g_k
}
\tag{1}
\]
The step size at iteration $k$ is therefore $\alpha_k:=\eta/ \sqrt{G_k+\epsilon}$, which monotonically decreases as more gradient information is accumulated.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (Scalar)}\label{alg:adagrad}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^d$, base step size $\eta>0$, stability constant $\epsilon>0$, number of iterations $T$.
\State $G_0 \gets 0$
\For{$k=0,\dots,T-1$}
    \State $g_k \gets \nabla f(w_k)$ \Comment{or stochastic estimate}
    \State $G_{k+1} \gets G_k + \|g_k\|^{2}$
    \State $w_{k+1} \gets w_k - \dfrac{\eta}{\sqrt{G_{k+1}+\epsilon}}\,g_k$
\EndFor
\State \textbf{Output:} $w_T$ (or the average $\bar w_T:=\frac1T\sum_{k=1}^T w_k$)
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f:\R^d\to\R$ is convex, i.e.
\[
f(y)\ge f(x)+\langle \nabla f(x),y-x\rangle,\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Bounded Gradients]\label{ass:bound}
There exists $G_{\max}>0$ such that $\|g_k\|\le G_{\max}$ for all $k$ (deterministic) or $\E[\|g_k\|^{2}]\le G_{\max}^{2}$ (stochastic).
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradients]\label{ass:unbiased}
If $g_k$ is stochastic, then $\E[g_k\mid w_k]=\nabla f(w_k)$.
\end{assumption}
All constants $\eta,\epsilon$ are positive, and the feasible set $\mathcal{W}\subset\R^d$ is assumed to have bounded diameter:
\[
D:=\max_{u,v\in\mathcal{W}}\|u-v\|<\infty .
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Regret bound for AdaGrad]\label{thm:regret}
Under Assumptions \ref{ass:convex}--\ref{ass:bound} and with $\epsilon>0$, the iterates generated by \eqref{alg:adagrad} satisfy for any $w^{\star}\in\mathcal{W}$,
\[
\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\;\le\;
\frac{D^{2}}{2\eta}\sqrt{G_T+\epsilon}
\;+\;
\frac{\eta}{2}\sum_{k=0}^{T-1}\frac{\|g_k\|^{2}}{\sqrt{G_k+\epsilon}} .
\tag{2}
\]
In particular, if the gradients are uniformly bounded by $G_{\max}$, then
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\;\le\;
\frac{D\,G_{\max}}{\eta\sqrt{T}} \;+\; \frac{\eta G_{\max}^{2}}{2\sqrt{T}} .
\tag{3}
\]
Choosing $\eta = D/G_{\max}$ yields the optimal $O(DG_{\max}/\sqrt{T})$ rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The denominator $\sqrt{G_k+\epsilon}$ grows monotonically, guaranteeing that the effective step size never explodes.
\item \textbf{Hyper‑parameter Sensitivity.} Only the base learning rate $\eta$ needs to be tuned; the adaptivity automatically scales the step size according to the observed gradient magnitudes.
\item \textbf{Comparison to SGD.} For constant step size $\eta$, SGD attains $O(1/\sqrt{T})$ regret with a constant factor depending on $\eta$. AdaGrad replaces this constant by $D\,G_{\max}/\eta+\eta G_{\max}^{2}/2$, which can be minimized analytically, yielding a tighter bound without manual decay scheduling.
\item \textbf{Special Cases.} If all gradients are orthogonal (e.g., in sparse settings), $G_T$ grows slowly and the bound improves to $O(D\sqrt{\log T}/\sqrt{T})$.
\end{itemize}

\section*{Discussion}
The theorem shows that AdaGrad automatically adapts to the geometry of the data through $G_k$. The bound holds for both deterministic and unbiased stochastic gradients, and no explicit learning‑rate decay schedule is required. The $O(1/\sqrt{T})$ rate matches the optimal rate for convex, smooth optimization, confirming that the adaptive scheme does not sacrifice statistical efficiency.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three–point inequality for convex functions]\label{lem:convex}
For any convex $f$ and any $u,v\in\R^d$,
\[
f(u)-f(v)\le \langle \nabla f(u),u-v\rangle .
\]
\end{lemma}
\begin{proof}
Directly from Definition \ref{ass:convex}. \hfill $\square$
\end{proof}
\begin{lemma}[Bound on the telescoping sum]\label{lem:telescoping}
For any sequence $\{a_k\}_{k\ge0}$ with $a_k\ge0$,
\[
\sum_{k=0}^{T-1}\frac{a_k}{\sqrt{A_k+\epsilon}}
\;\le\;
2\bigl(\sqrt{A_T+\epsilon}-\sqrt{\epsilon}\bigr),
\]
where $A_k:=\sum_{i=0}^{k-1} a_i$.
\end{lemma}
\begin{proof}
Observe that
\[
\sqrt{A_{k+1}+\epsilon}-\sqrt{A_k+\epsilon}
=\frac{A_{k+1}-A_k}{\sqrt{A_{k+1}+\epsilon}+\sqrt{A_k+\epsilon}}
=\frac{a_k}{\sqrt{A_{k+1}+\epsilon}+\sqrt{A_k+\epsilon}}
\ge \frac{a_k}{2\sqrt{A_{k+1}+\epsilon}} .
\]
Rearranging gives $a_k/\sqrt{A_{k+1}+\epsilon}\le 2\bigl(\sqrt{A_{k+1}+\epsilon}-\sqrt{A_k+\epsilon}\bigr)$. Summing over $k$ yields the claim. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:regret}. Throughout the proof we denote $w^{\star}\in\mathcal{W}$ as an arbitrary comparator.

\begin{proof}
\begin{enumerate}
\item[Step 1.] \textbf{Start from the squared distance recursion.}
\[
\|w_{k+1}-w^{\star}\|^{2}
= \Bigl\|w_k -\frac{\eta}{\sqrt{G_{k+1}+\epsilon}}g_k - w^{\star}\Bigr\|^{2}
\]
Expanding the norm,
\[
= \|w_k-w^{\star}\|^{2}
-2\frac{\eta}{\sqrt{G_{k+1}+\epsilon}}\langle g_k, w_k-w^{\star}\rangle
+\frac{\eta^{2}}{G_{k+1}+\epsilon}\|g_k\|^{2}.
\tag{4}
\]

\item[Step 2.] \textbf{Apply Lemma \ref{lem:convex} to replace the inner product.}
From convexity,
\[
\langle g_k, w_k-w^{\star}\rangle
\ge f(w_k)-f(w^{\star}).
\tag{5}
\]

\item[Step 3.] \textbf{Insert (5) into (4).}
\[
\|w_{k+1}-w^{\star}\|^{2}
\le \|w_k-w^{\star}\|^{2}
-2\frac{\eta}{\sqrt{G_{k+1}+\epsilon}}\bigl(f(w_k)-f(w^{\star})\bigr)
+\frac{\eta^{2}}{G_{k+1}+\epsilon}\|g_k\|^{2}.
\tag{6}
\]

\item[Step 4.] \textbf{Rearrange (6) to isolate the regret term.}
\[
2\frac{\eta}{\sqrt{G_{k+1}+\epsilon}}\bigl(f(w_k)-f(w^{\star})\bigr)
\le
\|w_k-w^{\star}\|^{2}-\|w_{k+1}-w^{\star}\|^{2}
+\frac{\eta^{2}}{G_{k+1}+\epsilon}\|g_k\|^{2}.
\tag{7}
\]

\item[Step 5.] \textbf{Sum (7) over $k=0,\dots,T-1$.}
The telescoping sum on the right-hand side yields
\[
2\eta\sum_{k=0}^{T-1}\frac{f(w_k)-f(w^{\star})}{\sqrt{G_{k+1}+\epsilon}}
\le
\|w_0-w^{\star}\|^{2}
+\eta^{2}\sum_{k=0}^{T-1}\frac{\|g_k\|^{2}}{G_{k+1}+\epsilon}.
\tag{8}
\]

\item[Step 6.] \textbf{Upper‑bound the distance term.}
Since $w_0,w^{\star}\in\mathcal{W}$ and $\operatorname{diam}(\mathcal{W})=D$, we have $\|w_0-w^{\star}\|\le D$, thus
\[
\|w_0-w^{\star}\|^{2}\le D^{2}.
\tag{9}
\]

\item[Step 7.] \textbf{Bound the second sum using Lemma \ref{lem:telescoping}.}
Notice that $a_k:=\|g_k\|^{2}$ and $A_{k+1}=G_{k+1}$. Lemma \ref{lem:telescoping} gives
\[
\sum_{k=0}^{T-1}\frac{\|g_k\|^{2}}{\sqrt{G_{k+1}+\epsilon}}
\le
2\bigl(\sqrt{G_T+\epsilon}-\sqrt{\epsilon}\bigr)
\le
2\sqrt{G_T+\epsilon}.
\tag{10}
\]
Multiplying by $\eta/2$ (as appears in (8)) yields
\[
\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\frac{\|g_k\|^{2}}{G_{k+1}+\epsilon}
\le
\eta\sqrt{G_T+\epsilon}.
\tag{11}
\]

\item[Step 8.] \textbf{Combine (8), (9) and (11).}
Dividing (8) by $2\eta$ and using (9)–(11) we obtain
\[
\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\le
\frac{D^{2}}{2\eta}\sqrt{G_T+\epsilon}
+\frac{\eta}{2}\sum_{k=0}^{T-1}\frac{\|g_k\|^{2}}{\sqrt{G_k+\epsilon}} .
\]
This is exactly inequality \eqref{2}, establishing the first part of the theorem. \hfill $\square$

\item[Step 9.] \textbf{Derive the simplified $O(1/\sqrt{T})$ rate.}
Assume $\|g_k\|\le G_{\max}$ for all $k$. Then $G_T\le TG_{\max}^{2}$, and
\[
\sqrt{G_T+\epsilon}\le G_{\max}\sqrt{T}+ \sqrt{\epsilon}\le 2G_{\max}\sqrt{T}
\quad (\text{for }T\ge1).
\]
Moreover,
\[
\frac{\|g_k\|^{2}}{\sqrt{G_k+\epsilon}}\le
\frac{G_{\max}^{2}}{\sqrt{k\,G_{\max}^{2}}}= \frac{G_{\max}}{\sqrt{k}} .
\]
Summing over $k$ gives $\sum_{k=0}^{T-1}\frac{G_{\max}}{\sqrt{k}}\le 2G_{\max}\sqrt{T}$. Plugging both bounds into \eqref{2} yields
\[
\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\le
\frac{D^{2}}{2\eta}\,2G_{\max}\sqrt{T}
+\frac{\eta}{2}\,2G_{\max}\sqrt{T}
=
\Bigl(\frac{D^{2}G_{\max}}{\eta}+\eta G_{\max}\Bigr)\sqrt{T}.
\]
Dividing by $T$ gives the average regret bound \eqref{3}. \hfill $\square$
\end{enumerate}
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{3}, the average suboptimality satisfies
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\le
\frac{DG_{\max}}{\eta\sqrt{T}}+\frac{\eta G_{\max}^{2}}{2\sqrt{T}} .
\]
Choosing $\eta^{\star}=D/G_{\max}$ balances the two terms and yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^{\star})\bigr)
\le
\frac{DG_{\max}}{\sqrt{T}} .
\]
Thus AdaGrad attains the optimal $O(1/\sqrt{T})$ rate for convex smooth problems with an explicit constant $DG_{\max}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Sparse gradients.} If at each iteration only a subset of coordinates is non‑zero, then each coordinate accumulates far fewer squared gradients, making the corresponding denominator grow slowly. The bound improves to $O\bigl(D\sqrt{\sum_{j=1}^{d} \log(1+G_{T}^{(j)})}/\sqrt{T}\bigr)$, matching the classic per‑coordinate AdaGrad analysis.
\item \textbf{Deterministic gradients.} When $g_k=\nabla f(w_k)$, the expectation operator disappears and the same deterministic bound holds.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, a refined analysis (using the strong convexity inequality) yields an $O\bigl((\log T)/T\bigr)$ rate, analogous to the standard AdaGrad result for strongly convex objectives.
\end{itemize}
