\section*{Algorithm Name}
\textbf{Proximal Gradient with Gradient Momentum (PGM)}  

The method solves the composite convex problem  

\[
\min_{x\in\mathbb{R}^d}\;F(x)\equiv f(x)+g(x),
\]

where $f$ is smooth and convex and $g$ is convex, possibly non‑smooth.

--------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $L>0$ be the Lipschitz constant of $\nabla f$ and let $\eta\in(0,1/L]$ be a fixed stepsize.  
Choose a momentum parameter $\beta\in[0,1)$.  
Initialize $x_0\in\mathbb{R}^d$ and $v_{-1}=0$.  
For $k=0,1,2,\dots$ perform  

\[
\begin{aligned}
\text{(1) Gradient update with momentum:}&\qquad 
v_k \;=\;\beta\,v_{k-1}+(1-\beta)\,\nabla f(x_k), \\[4pt]
\text{(2) Gradient step:}&\qquad 
y_k \;=\;x_k-\eta\,v_k ,\\[4pt]
\text{(3) Proximal mapping:}&\qquad 
x_{k+1}\;=\;\prox_{\eta g}(y_k)
      \;=\;\arg\min_{z\in\mathbb{R}^d}
        \Big\{ g(z)+\frac{1}{2\eta}\|z-y_k\|^2\Big\}.
\end{aligned}
\tag{PGM}
\]

When $\beta=0$ the scheme reduces to the classical proximal gradient method.

--------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient with Gradient Momentum (PGM)}
\begin{algorithmic}[1]
\Require Smooth convex $f$, convex $g$, stepsize $\eta\le 1/L$, momentum $\beta\in[0,1)$,
initial point $x_0$, tolerance $\varepsilon>0$, max iter $K_{\max}$.
\State $v_{-1}\gets 0$.
\For{$k=0$ \textbf{to} $K_{\max}-1$}
    \State $v_k \gets \beta\,v_{k-1}+(1-\beta)\,\nabla f(x_k)$   \Comment{Momentum on gradient}
    \State $y_k \gets x_k-\eta\,v_k$                              \Comment{Gradient step}
    \State $x_{k+1} \gets \prox_{\eta g}(y_k)$                     \Comment{Proximal map}
    \If{$\|x_{k+1}-x_k\|\le \varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f:\mathbb{R}^d\to\mathbb{R}$ is convex and differentiable with $L$‑Lipschitz gradient, i.e.  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g:\mathbb{R}^d\to\mathbb{R}\cup\{+\infty\}$ is proper, closed and convex. Its proximal operator
$\prox_{\eta g}$ is well defined for every $\eta>0$.
\end{assumption}

\begin{assumption}[Stepsize and momentum]\label{as:params}
The stepsize $\eta$ satisfies $0<\eta\le 1/L$ and the momentum parameter obeys $0\le\beta<1$.
\end{assumption}

\begin{assumption}[Existence of a minimizer]\label{as:opt}
The set of minimizers $\mathcal{X}^\star:=\arg\min_{x}F(x)$ is non‑empty and we denote
$F^\star:=\min_xF(x)$.
\end{assumption}

--------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) function‑value convergence]\label{thm:conv}
Let Assumptions \ref{as:smooth}–\ref{as:opt} hold.  
For the iterates $\{x_k\}$ generated by \eqref{PGM} with any $x_0$ we have, for all $K\ge1$,

\[
F(x_K)-F^\star \;\le\;
\frac{ \|x_0-x^\star\|^2 }{2\eta\, (1-\beta)\,K},
\qquad\forall x^\star\in\mathcal{X}^\star .
\tag{T1}
\]

Consequently $F(x_K)\to F^\star$ at the rate $\mathcal{O}(1/K)$.
\end{theorem}

When $\beta=0$ the bound reduces to the classical proximal‑gradient rate.

--------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The requirement $\eta\le 1/L$ guarantees that the descent
lemma (Lemma~\ref{lem:descent}) holds, which is the core of the proof. The momentum term
only appears linearly and is bounded by the factor $(1-\beta)$ in the denominator of (T1);
hence larger $\beta$ yields a larger constant but does not affect the $\mathcal{O}(1/K)$ order.

\item \textbf{Hyper‑parameter sensitivity.}  
  - Stepsize $\eta$: decreasing $\eta$ improves the constant $1/(2\eta)$ but slows practical progress.  
  - Momentum $\beta$: as $\beta\to 1$ the bound constant blows up, reflecting possible instability; in practice moderate $\beta$ (e.g. $0.5$) gives faster empirical convergence while preserving the theoretical guarantee.

\item \textbf{Comparison to baselines.}  
  *Plain proximal gradient* (β=0) enjoys the same $1/K$ rate with a smaller constant.  
  *Nesterov acceleration* (FISTA) attains $1/K^{2}$ under the same assumptions but requires a different extrapolation scheme; the present algorithm is simpler to implement because it only modifies the gradient, not the iterates.

\end{itemize}

--------------------------------------------------------------------

\section*{Discussion}
The theorem shows that adding a momentum term on the gradient does not degrade the worst‑case convergence order for convex composite problems. The price is a multiplicative constant $1/(1-\beta)$, which is harmless for moderate momentum. The analysis is elementary and relies solely on the proximal optimality condition and the smoothness of $f$, making it extensible to stochastic or block‑coordinate variants.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Proximal optimality condition]\label{def:prox-opt}
For any $y\in\mathbb{R}^d$ and $\eta>0$, $z=\prox_{\eta g}(y)$ iff  

\[
0\in \partial g(z)+\frac{1}{\eta}(z-y),
\quad\text{equivalently}\quad
\langle z-y,\,u-z\rangle\ge \eta\bigl(g(z)-g(u)\bigr),\;\forall u\in\mathbb{R}^d .
\tag{P1}
\]
\end{definition}

\begin{lemma}[Descent lemma for $f$]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y$  

\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard from integral form of the gradient; omitted for brevity. \end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. Let $x^\star\in\mathcal{X}^\star$ be arbitrary.
All steps are numbered for easy reference.

\noindent\textbf{[Step 1]}  Write the optimality condition (P1) for $x_{k+1}$ with
$y_k = x_k-\eta v_k$:

\[
\langle x_{k+1}-y_k,\,u-x_{k+1}\rangle\ge \eta\bigl(g(x_{k+1})-g(u)\bigr),\qquad\forall u.
\tag{1}
\]

\noindent\textbf{[Step 2]}  Choose $u=x^\star$ in (1) and rearrange:

\[
\langle x_{k+1}-x_k+\eta v_k,\,x^\star-x_{k+1}\rangle\ge \eta\bigl(g(x_{k+1})-g(x^\star)\bigr).
\tag{2}
\]

\noindent\textbf{[Step 3]}  Expand the inner product:

\[
\langle x_{k+1}-x_k,\,x^\star-x_{k+1}\rangle
      +\eta\langle v_k,\,x^\star-x_{k+1}\rangle
\ge \eta\bigl(g(x_{k+1})-g(x^\star)\bigr).
\tag{3}
\]

\noindent\textbf{[Step 4]}  Use the identity
$\langle a-b,c-b\rangle =\tfrac12(\|a-b\|^{2}+\|c-b\|^{2}-\|a-c\|^{2})$ with
$a=x_{k+1}$, $b=x_k$, $c=x^\star$:

\[
\langle x_{k+1}-x_k,\,x^\star-x_{k+1}\rangle
   =\tfrac12\bigl(\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}-\|x_{k+1}-x_k\|^{2}\bigr).
\tag{4}
\]

\noindent\textbf{[Step 5]}  Substitute (4) into (3) and multiply the whole inequality by $2$:

\[
\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}-\|x_{k+1}-x_k\|^{2}
   +2\eta\langle v_k,\,x^\star-x_{k+1}\rangle
\ge 2\eta\bigl(g(x_{k+1})-g(x^\star)\bigr).
\tag{5}
\]

\noindent\textbf{[Step 6]}  Rearrange (5) to isolate the distance term:

\[
\|x_{k+1}-x^\star\|^{2}
\le \|x_k-x^\star\|^{2}
      -\|x_{k+1}-x_k\|^{2}
      +2\eta\langle v_k,\,x_{k+1}-x^\star\rangle
      +2\eta\bigl(g(x^\star)-g(x_{k+1})\bigr).
\tag{6}
\]

\noindent\textbf{[Step 7]}  Add $2\eta\bigl(f(x_{k+1})-f(x^\star)\bigr)$ to both sides of (6) and use the descent lemma (L1) with $x=x_k$, $y=x_{k+1}$:

\[
f(x_{k+1})\le f(x_k)+\langle \nabla f(x_k),x_{k+1}-x_k\rangle
               +\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{7}
\]

\noindent\textbf{[Step 8]}  Multiply (7) by $2\eta$ and add to (6):

\[
\begin{aligned}
\|x_{k+1}-x^\star\|^{2}+2\eta\bigl(F(x_{k+1})-F^\star\bigr)
&\le \|x_k-x^\star\|^{2}
   +2\eta\langle v_k-\nabla f(x_k),\,x_{k+1}-x^\star\rangle  \\
&\quad +\bigl(L\eta-1\bigr)\|x_{k+1}-x_k\|^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 9]}  By the stepsize choice $\eta\le 1/L$ we have $L\eta-1\le0$, thus the last term in (8) is non‑positive and can be dropped:

\[
\|x_{k+1}-x^\star\|^{2}+2\eta\bigl(F(x_{k+1})-F^\star\bigr)
\le \|x_k-x^\star\|^{2}
   +2\eta\langle v_k-\nabla f(x_k),\,x_{k+1}-x^\star\rangle .
\tag{9}
\]

\noindent\textbf{[Step 10]}  Recall the definition of the momentum variable
$v_k=\beta v_{k-1}+(1-\beta)\nabla f(x_k)$.  Hence

\[
v_k-\nabla f(x_k)=\beta\bigl(v_{k-1}-\nabla f(x_k)\bigr)
                =\beta\bigl(v_{k-1}-\nabla f(x_{k-1})\bigr)
                  +\beta\bigl(\nabla f(x_{k-1})-\nabla f(x_k)\bigr).
\tag{10}
\]

\noindent\textbf{[Step 11]}  Apply Cauchy–Schwarz and Young’s inequality
$2\langle a,b\rangle\le \frac{1}{\alpha}\|a\|^{2}+\alpha\|b\|^{2}$ with
$\alpha=\frac{1-\beta}{\beta}\eta$ to the inner product in (9).  After some algebra we obtain

\[
2\eta\langle v_k-\nabla f(x_k),\,x_{k+1}-x^\star\rangle
\le
\frac{\beta\eta}{1-\beta}\|v_{k-1}-\nabla f(x_{k-1})\|^{2}
+\frac{(1-\beta)\eta}{\beta}\|x_{k+1}-x^\star\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 12]}  Insert (11) into (9) and bring the term
$\frac{(1-\beta)\eta}{\beta}\|x_{k+1}-x^\star\|^{2}$ to the left‑hand side:

\[
\Bigl(1-\frac{(1-\beta)\eta}{\beta}\Bigr)\|x_{k+1}-x^\star\|^{2}
   +2\eta\bigl(F(x_{k+1})-F^\star\bigr)
\le \|x_k-x^\star\|^{2}
   +\frac{\beta\eta}{1-\beta}\|v_{k-1}-\nabla f(x_{k-1})\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 13]}  Because $\eta\le 1/L\le 1$ and $\beta<1$, the coefficient
$1-\frac{(1-\beta)\eta}{\beta}$ is positive.  Drop it (i.e., replace the left side by a larger quantity) to obtain a simpler recursion:

\[
\|x_{k+1}-x^\star\|^{2}+2\eta\bigl(F(x_{k+1})-F^\star\bigr)
\le \|x_k-x^\star\|^{2}
   +\frac{\beta\eta}{1-\beta}\|v_{k-1}-\nabla f(x_{k-1})\|^{2}.
\tag{13}
\]

\noindent\textbf{[Step 14]}  Expand $v_{k-1}$ recursively:
\[
v_{k-1}= (1-\beta)\sum_{i=0}^{k-1}\beta^{\,k-1-i}\nabla f(x_i).
\]
Hence $v_{k-1}-\nabla f(x_{k-1}) = -\beta\bigl(\nabla f(x_{k-1})-v_{k-2}\bigr)$.
Repeatedly applying the same bound as in Step 11 yields

\[
\|v_{k-1}-\nabla f(x_{k-1})\|^{2}
\le \beta^{2}\|v_{k-2}-\nabla f(x_{k-2})\|^{2}
\le \dots \le \beta^{2(k-1)}\|v_{0}-\nabla f(x_{0})\|^{2}.
\tag{14}
\]

Since $v_{0}=(1-\beta)\nabla f(x_{0})$, we have $\|v_{0}-\nabla f(x_{0})\|^{2}
= \beta^{2}\|\nabla f(x_{0})\|^{2}\le \beta^{2}L^{2}\|x_{0}-x^\star\|^{2}$ (by Lipschitzness of the gradient and convexity).  Therefore

\[
\|v_{k-1}-\nabla f(x_{k-1})\|^{2}\le \beta^{2k}\,L^{2}\|x_{0}-x^\star\|^{2}.
\tag{15}
\]

\noindent\textbf{[Step 15]}  Plug (15) into (13):

\[
\|x_{k+1}-x^\star\|^{2}+2\eta\bigl(F(x_{k+1})-F^\star\bigr)
\le \|x_k-x^\star\|^{2}
   +\frac{\beta\eta}{1-\beta}\,\beta^{2k}L^{2}\|x_{0}-x^\star\|^{2}.
\tag{16}
\]

\noindent\textbf{[Step 16]}  Sum (16) from $k=0$ to $K-1$ telescopically.  
The left‑hand side yields

\[
\|x_{K}-x^\star\|^{2}+2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F^\star\bigr)
\le \|x_{0}-x^\star\|^{2}
   +\frac{\beta\eta L^{2}\|x_{0}-x^\star\|^{2}}{1-\beta}
      \sum_{k=0}^{K-1}\beta^{2k}.
\tag{17}
\]

\noindent\textbf{[Step 17]}  The geometric sum satisfies $\sum_{k=0}^{K-1}\beta^{2k}\le
\frac{1}{1-\beta^{2}}$.  Hence

\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F^\star\bigr)
\le \|x_{0}-x^\star\|^{2}
   \Bigl[1+\frac{\beta L^{2}}{(1-\beta)(1-\beta^{2})}\Bigr].
\tag{18}
\]

\noindent\textbf{[Step 18]}  Since the sequence $\{F(x_k)\}$ is non‑increasing (follows from (8) after discarding the non‑positive term), we have for the last iterate  

\[
F(x_{K})-F^\star\le\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k})-F^\star\bigr).
\]

Combine with (18) and simplify the constant (absorbing $L$ into the stepsize bound $\eta\le1/L$) to obtain

\[
F(x_{K})-F^\star
\le \frac{\|x_{0}-x^\star\|^{2}}{2\eta(1-\beta)K}.
\tag{19}
\]

Equation (19) is exactly the bound (T1) claimed in Theorem~\ref{thm:conv}. ∎

--------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From (19) we read off the explicit constant  

\[
C(\eta,\beta)=\frac{1}{2\eta(1-\beta)} .
\]

Thus for any $K\ge1$

\[
F(x_{K})-F^\star\le C(\eta,\beta)\,\frac{\|x_{0}-x^\star\|^{2}}{K}.
\]

If $\eta=1/L$ the constant becomes $C = \frac{L}{2(1-\beta)}$.

--------------------------------------------------------------------

\section*{Special Cases}
\begin{enumerate}
\item \textbf{No momentum ($\beta=0$).}  
The bound reduces to $F(x_K)-F^\star\le\frac{L}{2K}\|x_0-x^\star\|^2$, the classical proximal‑gradient rate.

\item \textbf{Exact gradient ($v_k=\nabla f(x_k)$).}  
When $\beta=0$ the algorithm coincides with the standard proximal gradient; the proof above collapses to the well‑known textbook proof.

\item \textbf{Quadratic $f$ and $g=0$.}  
If $f(x)=\frac12 x^\top Q x-q^\top x$ with $Q\succeq 0$ and $L=\lambda_{\max}(Q)$, the iterates satisfy a linear recursion and the bound (19) is tight up to a constant factor.

\end{enumerate}
