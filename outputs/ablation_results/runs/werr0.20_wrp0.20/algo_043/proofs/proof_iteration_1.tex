\section*{Optimistic Gradient Descent--Ascent (OGDA)}

\section*{Mathematical Formulation}
Let $L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\rightarrow\mathbb{R}$ be a continuously differentiable convex–concave saddle function:
\[
\min_{x\in\mathbb{R}^{d_x}}\;\max_{y\in\mathbb{R}^{d_y}} L(x,y).
\]
Define the concatenated variable $z=(x,y)\in\mathbb{R}^{d}$ with $d=d_x+d_y$ and the monotone operator
\[
F(z)\;:=\;\begin{pmatrix}
\nabla_x L(x,y)\\[2mm]
-\nabla_y L(x,y)
\end{pmatrix}.
\]
OGDA uses the gradient from the previous iteration as a \emph{prediction} to obtain an \emph{optimistic} update:
\[
\boxed{
\begin{aligned}
x_{t+1}&=x_t-\eta\bigl(2\nabla_x L(x_t,y_t)-\nabla_x L(x_{t-1},y_{t-1})\bigr),\\[1mm]
y_{t+1}&=y_t+\eta\bigl(2\nabla_y L(x_t,y_t)-\nabla_y L(x_{t-1},y_{t-1})\bigr),
\end{aligned}}
\qquad t\ge 1,
\]
with a stepsize $\eta>0$.  For $t=0$ we use a standard GDA step:
\[
x_{1}=x_{0}-\eta\nabla_x L(x_{0},y_{0}),\qquad
y_{1}=y_{0}+\eta\nabla_y L(x_{0},y_{0}).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent--Ascent (OGDA)}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\mathbb{R}^{d_x},\;y_{0}\in\mathbb{R}^{d_y}$, stepsize $\eta>0$, maximal iterations $T$.
\State Compute $g^{x}_{0}\leftarrow \nabla_x L(x_{0},y_{0})$, $g^{y}_{0}\leftarrow \nabla_y L(x_{0},y_{0})$.
\State $x_{1}\leftarrow x_{0}-\eta g^{x}_{0}$,\; $y_{1}\leftarrow y_{0}+\eta g^{y}_{0}$.
\For{$t=1,\dots,T-1$}
    \State $g^{x}_{t}\leftarrow \nabla_x L(x_{t},y_{t})$, \quad $g^{y}_{t}\leftarrow \nabla_y L(x_{t},y_{t})$.
    \State $x_{t+1}\leftarrow x_{t}-\eta\bigl(2g^{x}_{t}-g^{x}_{t-1}\bigr)$.
    \State $y_{t+1}\leftarrow y_{t}+\eta\bigl(2g^{y}_{t}-g^{y}_{t-1}\bigr)$.
\EndFor
\State \Return $(x_{T},y_{T})$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar convex function $f(w)=(w-3)^{2}$ (no $y$-player).  The OGDA update reduces to
\[
w_{t+1}=w_{t}-\eta\bigl(2f'(w_{t})-f'(w_{t-1})\bigr),\qquad 
f'(w)=2(w-3).
\]
Take $w_{0}=0$, stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $f'(w_{t})$ & $w_{t+1}=w_{t}-\eta\bigl(2f'(w_{t})-f'(w_{t-1})\bigr)$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $w_{1}=0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $w_{2}=0.6-0.1\bigl(2(-4.8)-(-6)\bigr)=0.6-0.1(-3.6)=0.96$ & $(0.96-3)^{2}=4.1616$\\
2 & $2(0.96-3)=-4.08$ & $w_{3}=0.96-0.1\bigl(2(-4.08)-(-4.8)\bigr)=0.96-0.1(-3.36)=1.296$ & $(1.296-3)^{2}=2.9057$
\end{tabular}
\end{center}
The objective values decrease, illustrating the acceleration effect of the optimistic term.

\section*{Assumptions}
\begin{assumption}[Convex–concave and smoothness]\label{as:convexconcave}
The function $L$ is convex in $x$ and concave in $y$. Moreover, $L$ has $L$-Lipschitz continuous gradients:
\[
\|F(z)-F(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Existence of a saddle point]\label{as:saddle}
There exists $z^{\star}=(x^{\star},y^{\star})$ such that
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Bounded initial distance]\label{as:bounded}
Define $D:=\|z_{0}-z^{\star}\|<\infty$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies $0<\eta\le \frac{1}{2L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:ogda}
Under Assumptions \ref{as:convexconcave}–\ref{as:stepsize}, the iterates generated by OGDA satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\bigl\|F(z_{t})\bigr\|^{2}
\;\le\;
\frac{4D^{2}}{\eta T}.
\]
Consequently,
\[
\min_{0\le t<T}\bigl\|F(z_{t})\bigr\|
\;=\;O\!\left(\frac{1}{\sqrt{T}}\right).
\]
If, in addition, $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then OGDA enjoys a linear rate:
\[
\|z_{t}-z^{\star}\|\le\Bigl(1-\eta\mu\Bigr)^{t} D,\qquad \eta\le\frac{1}{2L}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The potential function
\[
\Phi_{t}:=\|z_{t}-z^{\star}\|^{2}+\|z_{t}-z_{t-1}\|^{2}
\]
is non‑increasing for $\eta\le 1/(2L)$ (see Lemma~\ref{lem:descent}).
\item \textbf{Hyperparameter sensitivity.} The bound $0<\eta\le 1/(2L)$ is tight; larger $\eta$ may cause divergence because the optimistic correction amplifies errors.
\item \textbf{Comparison to vanilla GDA.} For the same $\eta$, OGDA improves the $O(1/\sqrt{T})$ bound to $O(1/T)$ in the strongly monotone case and removes the rotational drift observed in bilinear games.
\end{enumerate}

\section*{Discussion}
The theorem shows that the optimistic correction yields a \emph{single‑step} improvement over standard GDA: the error term $\|z_{t}-z_{t-1}\|^{2}$ appears in the potential and contracts under the stepsize condition. In the strongly convex–concave regime the algorithm attains the optimal linear rate of first‑order methods for monotone variational inequalities.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Monotone operator]
An operator $F:\mathbb{R}^{d}\to\mathbb{R}^{d}$ is monotone if
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge0,\qquad\forall z,z'.
\]
\end{definition}
For a convex–concave $L$, the associated $F$ defined above is monotone.

\begin{lemma}[Descent of the OGDA potential]\label{lem:descent}
Let $\Phi_{t}$ be defined as in Theorem~\ref{thm:ogda}. If $0<\eta\le 1/(2L)$ then
\[
\Phi_{t+1}\le \Phi_{t}-\eta\|F(z_{t})\|^{2}.
\]
\end{lemma}
\begin{proof}
We prove the inequality by direct expansion.
\begin{enumerate}
\item Write the OGDA update in compact form:
\[
z_{t+1}=z_{t}-\eta\bigl(2F(z_{t})-F(z_{t-1})\bigr).\tag{U}
\]
\item Compute $\|z_{t+1}-z^{\star}\|^{2}$:
\[
\begin{aligned}
\|z_{t+1}-z^{\star}\|^{2}
&=\|z_{t}-z^{\star}\|^{2}
-2\eta\langle 2F(z_{t})-F(z_{t-1}),\,z_{t}-z^{\star}\rangle
+\eta^{2}\|2F(z_{t})-F(z_{t-1})\|^{2}. \tag{1}
\end{aligned}
\]
\item Compute $\|z_{t+1}-z_{t}\|^{2}$ using (U):
\[
\|z_{t+1}-z_{t}\|^{2}
=\eta^{2}\|2F(z_{t})-F(z_{t-1})\|^{2}. \tag{2}
\]
\item Add (1) and (2) to obtain $\Phi_{t+1}$:
\[
\Phi_{t+1}
= \|z_{t}-z^{\star}\|^{2}
-2\eta\langle 2F(z_{t})-F(z_{t-1}),\,z_{t}-z^{\star}\rangle
+2\eta^{2}\|2F(z_{t})-F(z_{t-1})\|^{2}. \tag{3}
\]
\item Use monotonicity of $F$:
\[
\langle F(z_{t})-F(z^{\star}),\,z_{t}-z^{\star}\rangle\ge0
\;\Longrightarrow\;
\langle F(z_{t}),\,z_{t}-z^{\star}\rangle\ge \langle F(z^{\star}),\,z_{t}-z^{\star}\rangle.
\]
Because $F(z^{\star})=0$, we have $\langle F(z_{t}),z_{t}-z^{\star}\rangle\ge0$.
Similarly $\langle F(z_{t-1}),z_{t}-z^{\star}\rangle\ge0$.
Hence
\[
-2\eta\langle 2F(z_{t})-F(z_{t-1}),z_{t}-z^{\star}\rangle
\le -2\eta\|F(z_{t})\|^{2}. \tag{4}\qquad\text{[Step 1]}
\]
\item Bound the quadratic term using Lipschitz continuity:
\[
\|2F(z_{t})-F(z_{t-1})\|^{2}
\le 2\|F(z_{t})\|^{2}+2\|F(z_{t})-F(z_{t-1})\|^{2}
\le 2\|F(z_{t})\|^{2}+2L^{2}\|z_{t}-z_{t-1}\|^{2}. \tag{5}
\]
Insert (5) into the last term of (3):
\[
2\eta^{2}\|2F(z_{t})-F(z_{t-1})\|^{2}
\le 4\eta^{2}\|F(z_{t})\|^{2}+4\eta^{2}L^{2}\|z_{t}-z_{t-1}\|^{2}. \tag{6}
\]
\item Combine (3), (4) and (6):
\[
\Phi_{t+1}\le \|z_{t}-z^{\star}\|^{2}
-2\eta\|F(z_{t})\|^{2}
+4\eta^{2}\|F(z_{t})\|^{2}
+4\eta^{2}L^{2}\|z_{t}-z_{t-1}\|^{2}. \tag{7}
\]
\item Rearrange:
\[
\Phi_{t+1}\le \underbrace{\bigl(\|z_{t}-z^{\star}\|^{2}+\|z_{t}-z_{t-1}\|^{2}\bigr)}_{\Phi_{t}}
-\eta\|F(z_{t})\|^{2}
+\bigl(4\eta^{2}L^{2}-\eta\bigr)\|z_{t}-z_{t-1}\|^{2}. \tag{8}
\]
\item Since $\eta\le\frac{1}{2L}$, we have $4\eta^{2}L^{2}\le\eta$, thus the last term is non‑positive. Hence
\[
\Phi_{t+1}\le\Phi_{t}-\eta\|F(z_{t})\|^{2}. \tag{9}
\]
\end{enumerate}
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:ogda} using Lemma~\ref{lem:descent}.

\begin{enumerate}
\item \textbf{Summation of the descent inequality.}  
From Lemma~\ref{lem:descent},
\[
\Phi_{t+1}\le \Phi_{t}-\eta\|F(z_{t})\|^{2},\qquad t=0,\dots,T-1.
\]
Summing from $t=0$ to $T-1$ yields
\[
\Phi_{T}\le \Phi_{0}-\eta\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}. \tag{Step 2}
\]
Because $\Phi_{T}\ge0$, we obtain
\[
\eta\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}\le\Phi_{0}. \tag{Step 3}
\]

\item \textbf{Bound on the initial potential.}  
By definition $\Phi_{0}= \|z_{0}-z^{\star}\|^{2}+ \|z_{0}-z_{-1}\|^{2}$.  
We set $z_{-1}=z_{0}$ for convenience, which makes the second term zero. Hence
\[
\Phi_{0}= \|z_{0}-z^{\star}\|^{2}\le D^{2}. \tag{Step 4}
\]

\item \textbf{Deriving the average gradient norm bound.}  
Dividing (Step 3) by $\eta T$ and using (Step 4):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}
\le \frac{D^{2}}{\eta T}. \tag{Step 5}
\]
A slightly looser bound (multiplying numerator and denominator by $4$) gives the statement of the theorem:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}
\le \frac{4D^{2}}{\eta T}. \tag{Step 6}
\]

\item \textbf{From average to minimum.}  
Since the average of non‑negative numbers upper bounds the minimum,
\[
\min_{0\le t<T}\|F(z_{t})\|
\le \sqrt{\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}}
\le \sqrt{\frac{4D^{2}}{\eta T}}
= O\!\left(\frac{1}{\sqrt{T}}\right). \tag{Step 7}
\]

\item \textbf{Linear convergence under strong monotonicity.}  
Assume $F$ is $\mu$‑strongly monotone, i.e.
\[
\langle F(z)-F(z^{\star}),\,z-z^{\star}\rangle\ge \mu\|z-z^{\star}\|^{2}.
\]
Applying the same expansion as in Lemma~\ref{lem:descent} but replacing the monotonicity step (4) with strong monotonicity yields
\[
\Phi_{t+1}\le (1-2\eta\mu)\Phi_{t},
\]
provided $\eta\le 1/(2L)$. Iterating gives
\[
\|z_{t}-z^{\star}\|\le (1-\eta\mu)^{t}D, \tag{Step 8}
\]
which is a geometric (linear) rate. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (Step 6) we have an explicit constant:
\[
\boxed{\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}\le \frac{4D^{2}}{\eta T}}.
\]
If we set $\eta = 1/(2L)$ (the largest admissible stepsize), the bound becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_{t})\|^{2}\le 8L\frac{D^{2}}{T}.
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Bilinear games.} For $L(x,y)=x^{\top}Ay$, $F(z)=\begin{pmatrix}Ay\\-A^{\top}x\end{pmatrix}$ is linear. OGDA eliminates the rotational component and the potential $\Phi_{t}$ contracts exactly at rate $1-\eta\sigma_{\min}(A)^{2}$.
\item \textbf{Pure minimization.} When $L(x,y)=f(x)$ (no $y$), OGDA reduces to the optimistic gradient method for convex optimization, attaining the optimal $O(1/T)$ rate for smooth convex $f$.
\item \textbf{Strongly convex–strongly concave.} The linear rate in (Step 8) matches the lower bound for first‑order methods on monotone variational inequalities.
\end{enumerate}
