\section*{Trust Region Method (TRM)}

\section*{Mathematical Formulation}
Let $f:\R^{n}\to\R$ be a (possibly non‑convex) twice continuously differentiable objective.  
At iteration $k$ we are given a current point $x_{k}\in\R^{n}$ and a trust‑region radius $\Delta_{k}>0$.  
Define the (quadratic) model
\[
m_{k}(s)\;:=\;f(x_{k})+g_{k}^{\top}s+\tfrac12 s^{\top}B_{k}s,
\qquad 
g_{k}:=\nabla f(x_{k}),\; B_{k}\approx\nabla^{2}f(x_{k}).
\]
The subproblem is
\[
\min_{s\in\R^{n}}\; m_{k}(s)\quad\text{s.t.}\quad \norm{s}\le \Delta_{k}. \tag{TR}
\]
Let $s_{k}$ be an \emph{approximate} solution of (TR) satisfying the \emph{Cauchy decrease condition}
\[
m_{k}(0)-m_{k}(s_{k})\;\ge\;\kappa_{\rm C}\,
\bigl(m_{k}(0)-m_{k}(s_{k}^{\rm C})\bigr),\qquad 0<\kappa_{\rm C}\le 1,
\]
where $s_{k}^{\rm C}$ is the exact Cauchy step (the minimizer of $m_k$ along $-g_k$ under the radius).

Define the \emph{actual reduction}
\[
\text{ared}_{k}:=f(x_{k})-f(x_{k}+s_{k}),
\]
and the \emph{predicted reduction}
\[
\text{pred}_{k}:=m_{k}(0)-m_{k}(s_{k}).
\]
The \emph{reduction ratio} is
\[
\rho_{k}:=\frac{\text{ared}_{k}}{\text{pred}_{k}}.
\]
Given a \emph{acceptance parameter} $\eta\in(0,1)$ we set
\[
x_{k+1}= 
\begin{cases}
x_{k}+s_{k}, & \text{if }\rho_{k}\ge\eta,\\[4pt]
x_{k}, & \text{otherwise.}
\end{cases}
\]
The trust‑region radius is updated by
\[
\Delta_{k+1}= 
\begin{cases}
\gamma_{\rm inc}\,\Delta_{k}, & \text{if }\rho_{k}\ge \eta_{1},\\[4pt]
\Delta_{k}, & \text{if }\eta\le\rho_{k}<\eta_{1},\\[4pt]
\gamma_{\rm dec}\,\Delta_{k}, & \text{if }\rho_{k}<\eta,
\end{cases}
\]
with constants $0<\gamma_{\rm dec}<1<\gamma_{\rm inc}$ and $0<\eta<\eta_{1}<1$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust Region Method}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\R^{n}$, $\Delta_{0}>0$, parameters $\eta,\eta_{1},\gamma_{\rm inc},\gamma_{\rm dec},\kappa_{\rm C}\in(0,1]$.
\State $k\gets0$.
\While{termination criteria not met}
    \State $g_{k}\gets\nabla f(x_{k})$, $B_{k}\gets\nabla^{2}f(x_{k})$ (or a suitable approximation).
    \State Compute an approximate solution $s_{k}$ of (TR) satisfying the Cauchy decrease condition.
    \State $\text{ared}_{k}\gets f(x_{k})-f(x_{k}+s_{k})$,
          $\text{pred}_{k}\gets m_{k}(0)-m_{k}(s_{k})$.
    \State $\rho_{k}\gets \text{ared}_{k}/\text{pred}_{k}$.
    \If{$\rho_{k}\ge\eta$}
        \State $x_{k+1}\gets x_{k}+s_{k}$ \Comment{accept step}
    \Else
        \State $x_{k+1}\gets x_{k}$ \Comment{reject step}
    \EndIf
    \If{$\rho_{k}\ge\eta_{1}$}
        \State $\Delta_{k+1}\gets \gamma_{\rm inc}\,\Delta_{k}$.
    \ElsIf{$\rho_{k}<\eta$}
        \State $\Delta_{k+1}\gets \gamma_{\rm dec}\,\Delta_{k}$.
    \Else
        \State $\Delta_{k+1}\gets \Delta_{k}$.
    \EndIf
    \State $k\gets k+1$.
\EndWhile
\State \textbf{Output:} $x_{k}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^{2},\qquad w\in\R.
\]
Here $g(w)=\nabla f(w)=2(w-3)$ and $B(w)=\nabla^{2}f(w)=2$ (exact Hessian).  
Choose
\[
w_{0}=0,\qquad \Delta_{0}=1,\qquad
\eta=0.1,\;\eta_{1}=0.75,\;
\gamma_{\rm inc}=2,\;\gamma_{\rm dec}=0.5,\;
\kappa_{\rm C}=1.
\]

Since the problem is one‑dimensional, the exact solution of (TR) is
\[
s_{k}^{\star}= \operatorname*{arg\,min}_{|s|\le\Delta_{k}}
\bigl\{ g_{k}s+\tfrac12 B_{k}s^{2}\bigr\}
= -\operatorname{clip}\!\bigl(\tfrac{g_{k}}{B_{k}},\;\Delta_{k}\bigr),
\]
where $\operatorname{clip}(a,\Delta)=\max\{-\Delta,\min\{a,\Delta\}\}$.
Because $\kappa_{\rm C}=1$ we take $s_{k}=s_{k}^{\star}$.

\begin{center}
\begin{tabular}{c|c|c|c|c|c}
$k$ & $w_{k}$ & $g_{k}=2(w_{k}-3)$ & $s_{k}$ & $\rho_{k}$ & $w_{k+1}$ \\
\hline
0 & $0$ & $-6$ & $ \operatorname{clip}(3,1)=1$ & 
\(\displaystyle\frac{f(0)-f(1)}{m_{0}(0)-m_{0}(1)}=
\frac{9-4}{9- ( -6\cdot1+\tfrac12\cdot2\cdot1^{2})}= \frac{5}{5}=1\) &
$1$ \\
1 & $1$ & $-4$ & $\operatorname{clip}(2,2)=2$ (since $\Delta_{1}=2$ after increase) &
\(\displaystyle\frac{f(1)-f(3)}{m_{1}(0)-m_{1}(2)}=
\frac{4-0}{4- ( -4\cdot2+\tfrac12\cdot2\cdot2^{2})}= \frac{4}{4}=1\) &
$3$ \\
2 & $3$ & $0$ & $s_{2}=0$ (gradient zero) &
$\rho_{2}=1$ (both numerator and denominator zero, define $\rho_{2}=1$) &
$3$ \\
\end{tabular}
\end{center}
After three iterations the algorithm has reached the global minimiser $w^{\star}=3$ and the objective values are $f(0)=9$, $f(1)=4$, $f(3)=0$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is twice continuously differentiable and its gradient is $L$‑Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{n}.
\]
\end{assumption}
\begin{assumption}[Bounded Hessian Approximation]\label{ass:hessian}
For all $k$, the matrix $B_{k}$ satisfies
\[
\|B_{k}\|\le M,\qquad\text{and}\qquad 
\|B_{k}-\nabla^{2}f(x_{k})\|\le \delta_{k},\quad\delta_{k}\ge0,\;
\sum_{k=0}^{\infty}\delta_{k}<\infty .
\]
\end{assumption}
\begin{assumption}[Cauchy Decrease]\label{ass:cauchy}
The step $s_{k}$ returned by the subproblem solver fulfills
\[
m_{k}(0)-m_{k}(s_{k})\;\ge\;\kappa_{\rm C}\,
\bigl(m_{k}(0)-m_{k}(s_{k}^{\rm C})\bigr),\qquad 0<\kappa_{\rm C}\le1,
\]
where $s_{k}^{\rm C}= -\frac{g_{k}^{\top}g_{k}}{g_{k}^{\top}B_{k}g_{k}}\,g_{k}$ if $\|s_{k}^{\rm C}\|\le\Delta_{k}$, otherwise $s_{k}^{\rm C}=-\Delta_{k}\frac{g_{k}}{\|g_{k}\|}$.
\end{assumption}
\begin{assumption}[Parameter Bounds]\label{ass:param}
The algorithmic parameters satisfy
\[
0<\eta<\eta_{1}<1,\qquad
0<\gamma_{\rm dec}<1<\gamma_{\rm inc},\qquad
\Delta_{\min}>0,\;\Delta_{\max}<\infty .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global Convergence and Worst‑Case Complexity]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:param} hold and that the subproblem solver satisfies Assumption~\ref{ass:cauchy}.  
Let $\{x_{k}\}$ be the sequence generated by the Trust Region Method. Then
\begin{enumerate}[label=(\alph*)]
\item\label{itm:limit} $\displaystyle\lim_{k\to\infty}\|\nabla f(x_{k})\|=0$; i.e. every accumulation point of $\{x_{k}\}$ is a first‑order stationary point.
\item\label{itm:complexity} For any tolerance $\varepsilon>0$, let
\[
\mathcal{K}(\varepsilon):=\Bigl\{k\ge0:\|\nabla f(x_{k})\|>\varepsilon\Bigr\}.
\]
Then, under the choice $\Delta_{0}\le\Delta_{\max}$,
\[
|\mathcal{K}(\varepsilon)|\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\kappa_{\rm C}\,\eta\,(1-\eta)}\;
\frac{L}{\varepsilon^{2}} \;+\; C,
\]
where $C$ is a constant independent of $\varepsilon$.
Consequently, to obtain $\|\nabla f(x_{k})\|\le\varepsilon$ it suffices to perform
\[
\mathcal{O}\!\bigl(\varepsilon^{-2}\bigr)
\]
iterations.
\end{enumerate}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The trust‑region radius adapts to the local model quality; large $\rho_{k}$ enlarges $\Delta_{k}$, while poor agreement shrinks it, guaranteeing that steps are never taken when the model is unreliable.
\item \textbf{Hyper‑parameter sensitivity.} The constants $\eta,\eta_{1},\gamma_{\rm inc},\gamma_{\rm dec}$ appear only in the multiplicative factor of the complexity bound; standard choices $(\eta,\eta_{1})=(0.1,0.75)$, $(\gamma_{\rm inc},\gamma_{\rm dec})=(2,0.5)$ satisfy all assumptions and yield the same $\mathcal{O}(\varepsilon^{-2})$ rate.
\item \textbf{Comparison to baseline methods.}
    \begin{enumerate}[label=\roman*.]
    \item \emph{Gradient descent with fixed stepsize} obtains the same $\mathcal{O}(\varepsilon^{-2})$ bound only under a stepsize $\eta\le 1/L$, whereas TRM does not require a priori knowledge of $L$.
    \item \emph{Cubic regularization} achieves $\mathcal{O}(\varepsilon^{-3/2})$ for non‑convex problems but needs a more expensive subproblem; TRM trades a slightly worse rate for a cheap quadratic subproblem.
    \end{enumerate}
\end{itemize}
\section*{Discussion}
The theorem shows that, despite the possible non‑convexity of $f$, the trust‑region framework is guaranteed to drive the gradient norm to zero.  The $\varepsilon^{-2}$ iteration complexity matches the optimal rate for first‑order methods on smooth non‑convex problems, while the adaptive radius provides robustness to ill‑conditioned curvature that fixed‑step methods lack.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Model Decrease Lower Bound]\label{lem:model}
Under Assumptions~\ref{ass:smooth} and~\ref{ass:cauchy},
\[
m_{k}(0)-m_{k}(s_{k})\;\ge\;
\frac{\kappa_{\rm C}}{2}\,\min\!\Bigl\{\frac{\|g_{k}\|^{2}}{M},\;\|g_{k}\|\Delta_{k}\Bigr\}.
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:model}]
The Cauchy step $s_{k}^{\rm C}$ satisfies (see Nocedal \& Wright, \S4.2)
\[
m_{k}(0)-m_{k}(s_{k}^{\rm C})=
\frac{1}{2}\|g_{k}\|\min\!\Bigl\{\frac{\|g_{k}\|}{\|B_{k}g_{k}\|},\;\Delta_{k}\Bigr\}
\ge
\frac{1}{2}\|g_{k}\|\min\!\Bigl\{\frac{\|g_{k}\|}{M},\;\Delta_{k}\Bigr\}.
\]
Multiplying by $\kappa_{\rm C}$ yields the claimed bound because $s_{k}$ fulfills the Cauchy decrease condition. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Actual vs. Predicted Reduction]\label{lem:ratio}
If $\|B_{k}-\nabla^{2}f(x_{k})\|\le\delta_{k}$ then for any $s$ with $\|s\|\le\Delta_{k}$
\[
|\,\text{ared}_{k}-\text{pred}_{k}\,|
\;\le\;
\frac{L}{2}\|s\|^{2}+\frac{\delta_{k}}{2}\|s\|^{2}.
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:ratio}]
By the Taylor expansion with $L$‑Lipschitz gradient,
\[
f(x_{k}+s)=f(x_{k})+g_{k}^{\top}s+\tfrac12 s^{\top}\nabla^{2}f(x_{k})s
+\underbrace{R}_{\text{remainder}},
\]
where $|R|\le \frac{L}{6}\|s\|^{3}\le\frac{L}{2}\|s\|^{2}\Delta_{k}$ because $\|s\|\le\Delta_{k}\le1$ (we may enforce $\Delta_{\max}\le 1$ without loss of generality).  
Subtracting the model $m_{k}(s)$ gives
\[
\text{ared}_{k}-\text{pred}_{k}
= \tfrac12 s^{\top}\bigl(\nabla^{2}f(x_{k})-B_{k}\bigr)s+R.
\]
Taking absolute values and using $\|\nabla^{2}f(x_{k})-B_{k}\|\le\delta_{k}$ yields the bound. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  The proof proceeds in four logical blocks; each non‑trivial manipulation is labelled.

\subsection*{Step 1: Lower bound on predicted reduction}
From Lemma~\ref{lem:model} and the definition of the Cauchy step,
\[
\text{pred}_{k}=m_{k}(0)-m_{k}(s_{k})
\;\ge\;
\frac{\kappa_{\rm C}}{2}\,\min\!\Bigl\{\frac{\|g_{k}\|^{2}}{M},\;\|g_{k}\|\Delta_{k}\Bigr\}.
\tag{1}\label{eq:predlb}
\]

\subsection*{Step 2: Upper bound on the discrepancy between actual and predicted reduction}
Using Lemma~\ref{lem:ratio} with $s=s_{k}$,
\[
|\text{ared}_{k}-\text{pred}_{k}|
\le\frac{L+\delta_{k}}{2}\|s_{k}\|^{2}
\le\frac{L+\delta_{k}}{2}\Delta_{k}^{2}.
\tag{2}\label{eq:errub}
\]

\subsection*{Step 3: Lower bound on the reduction ratio for sufficiently large $\|g_{k}\|$}
Assume $\|g_{k}\|>\varepsilon$ for a prescribed $\varepsilon>0$.  
We consider two cases.

\paragraph{Case A: $\|g_{k}\|\Delta_{k}\le\frac{\|g_{k}\|^{2}}{M}$.}
Then from \eqref{eq:predlb},
\[
\text{pred}_{k}\ge \frac{\kappa_{\rm C}}{2}\,\|g_{k}\|\Delta_{k}.
\]
Using \eqref{eq:errub},
\[
\rho_{k}
=\frac{\text{ared}_{k}}{\text{pred}_{k}}
\ge
1-\frac{|\text{ared}_{k}-\text{pred}_{k}|}{\text{pred}_{k}}
\ge
1-\frac{\frac{L+\delta_{k}}{2}\Delta_{k}^{2}}{\frac{\kappa_{\rm C}}{2}\|g_{k}\|\Delta_{k}}
=
1-\frac{(L+\delta_{k})\Delta_{k}}{\kappa_{\rm C}\|g_{k}\|}.
\tag{3}\label{eq:rhoA}
\]

\paragraph{Case B: $\|g_{k}\|\Delta_{k}>\frac{\|g_{k}\|^{2}}{M}$.}
Now
\[
\text{pred}_{k}\ge \frac{\kappa_{\rm C}}{2}\,\frac{\|g_{k}\|^{2}}{M},
\]
and consequently
\[
\rho_{k}
\ge
1-\frac{\frac{L+\delta_{k}}{2}\Delta_{k}^{2}}{\frac{\kappa_{\rm C}}{2}\frac{\|g_{k}\|^{2}}{M}}
=
1-\frac{(L+\delta_{k})M\Delta_{k}^{2}}{\kappa_{\rm C}\|g_{k}\|^{2}}.
\tag{4}\label{eq:rhoB}
\]

\subsection*{Step 4: Choice of minimal radius guaranteeing acceptance}
Define a minimal radius
\[
\Delta_{\min}
:=\min\Bigl\{\frac{\kappa_{\rm C}\,\eta\,\varepsilon}{2(L+ \bar\delta)},\;
\sqrt{\frac{\kappa_{\rm C}\,\eta\,\varepsilon^{2}}{2M(L+\bar\delta)}}\Bigr\},
\tag{5}\label{eq:deltamin}
\]
where $\bar\delta:=\sup_{k}\delta_{k}<\infty$ (finite because $\sum\delta_{k}<\infty$).  

If $\Delta_{k}\ge\Delta_{\min}$ and $\|g_{k}\|>\varepsilon$, then from \eqref{eq:rhoA}–\eqref{eq:rhoB} we obtain $\rho_{k}\ge\eta$. Hence the step is \emph{accepted} and the radius is either kept or enlarged (according to the algorithm).

\subsection*{Step 5: Sufficient decrease when a step is accepted}
When $\rho_{k}\ge\eta$, the actual reduction satisfies
\[
f(x_{k})-f(x_{k+1})
= \text{ared}_{k}
\ge \eta\,\text{pred}_{k}
\stackrel{(1)}{\ge}
\frac{\eta\kappa_{\rm C}}{2}\,
\min\!\Bigl\{\frac{\|g_{k}\|^{2}}{M},\;\|g_{k}\|\Delta_{k}\Bigr\}.
\tag{6}\label{eq:suffdec}
\]

\subsection*{Step 6: Bounding the number of “large‑gradient” iterations}
Let $\mathcal{K}(\varepsilon)$ be as in Theorem~\ref{thm:main}.\,
Partition $\mathcal{K}(\varepsilon)$ into two subsets:
\[
\mathcal{K}_{1}:=\{k\in\mathcal{K}(\varepsilon):\Delta_{k}<\Delta_{\min}\},\qquad
\mathcal{K}_{2}:=\{k\in\mathcal{K}(\varepsilon):\Delta_{k}\ge\Delta_{\min}\}.
\]

\paragraph{Bound on $|\mathcal{K}_{1}|$.}
Each time $\Delta_{k}<\Delta_{\min}$ and a step is \emph{rejected} ($\rho_{k}<\eta$), the algorithm multiplies the radius by $\gamma_{\rm dec}<1$. Hence after at most
\[
\frac{\log(\Delta_{\min}/\Delta_{0})}{\log\gamma_{\rm dec}}
\]
rejections the radius becomes $\le\Delta_{\min}$. Since a rejection cannot increase $\Delta_{k}$, the total number of iterations belonging to $\mathcal{K}_{1}$ is bounded by a constant $C_{1}$ independent of $\varepsilon$.

\paragraph{Bound on $|\mathcal{K}_{2}|$.}
For $k\in\mathcal{K}_{2}$ the step is accepted, and by \eqref{eq:suffdec}
\[
f(x_{k})-f(x_{k+1})
\ge
\frac{\eta\kappa_{\rm C}}{2M}\,\|g_{k}\|^{2}
\ge
\frac{\eta\kappa_{\rm C}}{2M}\,\varepsilon^{2}.
\]
Summing over all $k\in\mathcal{K}_{2}$ yields
\[
f(x_{0})-f^{\star}
\;\ge\;
\sum_{k\in\mathcal{K}_{2}}\bigl(f(x_{k})-f(x_{k+1})\bigr)
\;\ge\;
|\mathcal{K}_{2}|\;\frac{\eta\kappa_{\rm C}}{2M}\,\varepsilon^{2}.
\]
Thus
\[
|\mathcal{K}_{2}|\;\le\;
\frac{2M\bigl(f(x_{0})-f^{\star}\bigr)}{\eta\kappa_{\rm C}}\,
\varepsilon^{-2}.
\tag{7}\label{eq:k2bound}
\]

\subsection*{Step 7: Assemble the total bound}
Combining the two subsets,
\[
|\mathcal{K}(\varepsilon)|
=|\mathcal{K}_{1}|+|\mathcal{K}_{2}|
\le C_{1}+ \frac{2M\bigl(f(x_{0})-f^{\star}\bigr)}{\eta\kappa_{\rm C}}\,
\varepsilon^{-2}.
\]
Since $M\le L$ (by Assumption~\ref{ass:hessian}) we may replace $M$ with $L$ and absorb $C_{1}$ into a generic constant $C$, obtaining the bound stated in Theorem~\ref{thm:main}\ref{itm:complexity}.  The bound on $|\mathcal{K}(\varepsilon)|$ directly implies the iteration complexity $\mathcal{O}(\varepsilon^{-2})$.

\subsection*{Step 8: Convergence of the gradient norm}
Because $\sum_{k=0}^{\infty}\bigl(f(x_{k})-f(x_{k+1})\bigr)$ converges (the objective is bounded below), the sufficient‑decrease inequality \eqref{eq:suffdec} forces $\|g_{k}\|\to0$ along the subsequence of accepted steps.  The radius update rule guarantees that the distance between successive iterates is bounded by $\Delta_{k}$, which itself cannot stay bounded away from zero indefinitely without violating the acceptance condition (Step 4).  Hence every limit point of $\{x_{k}\}$ satisfies $\nabla f(x^{\star})=0$, establishing part \ref{itm:limit} of the theorem. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
Collecting the constants appearing in the bound \eqref{eq:k2bound} we obtain an explicit rate:
\[
|\mathcal{K}(\varepsilon)|
\le
\underbrace{\frac{2L\bigl(f(x_{0})-f^{\star}\bigr)}{\kappa_{\rm C}\,\eta\,(1-\eta)}}_{\displaystyle C_{0}}
\;\varepsilon^{-2}
\;+\;
C_{1},
\]
where $C_{1}$ stems from the maximal number of consecutive radius reductions before reaching $\Delta_{\min}$ (see Step 6).  The term $C_{0}$ dominates for small $\varepsilon$, confirming the $\mathcal{O}(\varepsilon^{-2})$ worst‑case iteration complexity.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Convex $f$.} When $f$ is convex, every stationary point is a global minimiser, so the algorithm converges to $x^{\star}$ with the same $\varepsilon^{-2}$ rate.  Moreover, if $f$ is additionally $L$‑smooth and strongly convex with modulus $\mu$, the bound can be sharpened to a linear rate by exploiting the strong‑convexity inequality.
\item \textbf{Exact subproblem solution.} If $s_{k}=s_{k}^{\star}$ (the exact minimiser of (TR)), then $\kappa_{\rm C}=1$ and the constants improve by a factor of $2$ in the complexity bound.
\item \textbf{Quadratic models with exact Hessian.} When $B_{k}=\nabla^{2}f(x_{k})$ and $L=0$ (i.e., $f$ is a quadratic), the discrepancy term \eqref{eq:errub} vanishes, $\rho_{k}=1$ for all $k$, and the algorithm reduces to the classical dogleg method with finite‑step convergence in at most $n$ iterations.
\end{enumerate}
