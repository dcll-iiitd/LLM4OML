\section*{Heavy‑Ball Method (Momentum with Previous Gradient)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy‑Ball iteration is
\[
\boxed{
x_{k+1}=x_k+\beta\bigl(x_k-x_{k-1}\bigr)-\alpha\nabla f(x_k),\qquad k\ge 0,
}
\]
with two initial points $x_{-1},x_{0}\in\mathbb{R}^d$ (often $x_{-1}=x_0$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Method}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\mathbb{R}^d$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
                 max iterations $K$, tolerance $\varepsilon>0$.
\State Set $x_{-1}\gets x_{0}$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $x_{k+1} \gets x_k + \beta (x_k-x_{k-1}) - \alpha g_k$
    \If{$\norm{g_k}\le \varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{Output:} $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^2$.  
Then $\nabla f(w)=2(w-3)$. Choose $\alpha=0.1$, $\beta=0.5$, and start with $w_{-1}=w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\nabla f(w_k)$ & $w_{k+1}=w_k+\beta(w_k-w_{k-1})-\alpha g_k$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ &
$0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ &
$(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ &
$0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ &
$(1.38-3)^2=2.6244$\\
2 & $2(1.38-3)=-3.24$ &
$1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ &

\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the acceleration effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex ($0<\mu\le L$), i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2 .
\]
Equivalently,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge \mu\|x-y\|^2 .
\]
\end{assumption}

\begin{assumption}[Parameter Range]\label{ass:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\begin{equation}\label{eq:param}
0<\alpha<\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta<\frac{(1-\sqrt{\alpha\mu})^{2}}{1-\alpha\mu}.
\end{equation}
These bounds guarantee that the linear dynamical system induced by the Heavy‑Ball method is contractive (see Lemma~\ref{lem:contract}).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:linear}
Let $f$ satisfy Assumptions~\ref{ass:smooth}–\ref{ass:strong} and let $\alpha,\beta$ obey \eqref{eq:param}.  
Denote the unique minimizer by $x^{\star}$. Then for the iterates generated by the Heavy‑Ball method,
\[
\boxed{
\norm{x_k-x^{\star}}\le C\,\rho^{\,k},\qquad k\ge 0,
}
\]
where
\[
\rho:=\max\Bigl\{\bigl|1-\sqrt{\alpha\mu}\bigr|,\;\bigl|1-\sqrt{\alpha L}\bigr|\Bigr\}<1,
\]
and $C>0$ depends only on the initial points $x_{-1},x_0$.
Consequently,
\[
f(x_k)-f(x^{\star})\le \frac{L}{2}C^{2}\rho^{\,2k},
\]
i.e. the method enjoys a geometric (linear) rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The Lyapunov function
\[
\mathcal{V}_k:=\norm{x_k-x^{\star}}^{2}+c\,\norm{x_k-x_{k-1}}^{2},
\]
with $c=\frac{1-\alpha\mu}{\beta}$ (well‑defined under \eqref{eq:param}) is non‑increasing, establishing stability of the iterates.
\item \textbf{Hyperparameter Sensitivity.} The admissible region \eqref{eq:param} shrinks as the condition number $\kappa=L/\mu$ grows. Choosing $\alpha=\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}}$ and $\beta=\frac{(1-\sqrt{\alpha\mu})^{2}}{1-\alpha\mu}$ yields the optimal contraction factor $\rho=1-\sqrt{\mu/L}=1-1/\sqrt{\kappa}$, matching the lower bound for first‑order methods on strongly convex quadratics.
\item \textbf{Comparison to Gradient Descent.} Gradient descent with stepsize $1/L$ achieves $\rho_{\text{GD}}=1-\mu/L=1-1/\kappa$, which is slower than the Heavy‑Ball factor $1-1/\sqrt{\kappa}$ for ill‑conditioned problems.
\item \textbf{Relation to Nesterov’s Accelerated Gradient.} Both methods share the same optimal linear rate for quadratics, yet Heavy‑Ball uses the previous *position* rather than a look‑ahead gradient; the analysis above reflects this structural difference.
\end{itemize}

\section*{Discussion}
The theorem shows that under the classical smooth‑strongly‑convex setting the Heavy‑Ball method converges *linearly* with a rate that improves over plain gradient descent, especially when the condition number $\kappa$ is large. The proof relies on a carefully chosen Lyapunov function and the spectral analysis of the induced $2\times2$ linear recursion. The parameter constraints are essential: violating \eqref{eq:param} can lead to divergence or oscillatory behaviour, a phenomenon often observed empirically.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic bounds]\label{lem:quad}
For any $x\in\mathbb{R}^d$,
\[
\mu\|x-x^{\star}\|^{2}\le \langle\nabla f(x),x-x^{\star}\rangle\le L\|x-x^{\star}\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply strong convexity and smoothness with $y=x^{\star}$ (where $\nabla f(x^{\star})=0$). \hfill $\square$
\end{proof}

\begin{lemma}[Contractivity of the linear system]\label{lem:contract}
Define the error vector $e_k:=\begin{bmatrix}x_k-x^{\star}\\ x_{k-1}-x^{\star}\end{bmatrix}$.  
Under \eqref{eq:param} there exists a norm $\|\cdot\|_{P}$ (induced by a positive definite matrix $P$) such that
\[
\|e_{k+1}\|_{P}\le\rho\,\|e_{k}\|_{P},\qquad \rho<1.
\]
\end{lemma}
\begin{proof}
From the iteration,
\[
e_{k+1}=A\,e_{k},\qquad 
A:=\begin{bmatrix} 1+\beta-\alpha H & -\beta I\\ I & 0\end{bmatrix},
\]
where $H:=\nabla^{2}f(\xi)$ for some $\xi$ on the line segment $[x_k,x^{\star}]$. By Assumptions~\ref{ass:smooth}–\ref{ass:strong}, $H$ satisfies $\mu I\preceq H\preceq L I$. Spectral radius analysis of $A$ over the interval $[\mu,L]$ yields the bound $\rho$ stated in Theorem~\ref{thm:linear}. A standard Lyapunov argument (see e.g. Polyak 1964) constructs $P\succ0$ such that $A^{\!\top}PA-P\preceq -\delta I$ for some $\delta>0$, which is equivalent to $\|e_{k+1}\|_{P}^{2}\le(1-\delta)\|e_{k}\|_{P}^{2}$. The explicit constants lead to the $\rho$ given in the theorem. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout, let $x^{\star}$ denote the unique minimizer of $f$.

\noindent\textbf{Notation.} Set $e_k:=x_k-x^{\star}$.  

\noindent\textbf{Step 1 (Error recursion).}  
From the algorithm,
\[
x_{k+1}=x_k+\beta(x_k-x_{k-1})-\alpha\nabla f(x_k).
\]
Subtract $x^{\star}$ and use $\nabla f(x^{\star})=0$:
\[
e_{k+1}=e_k+\beta(e_k-e_{k-1})-\alpha\nabla f(x_k). \tag{1}
\]
\textit{[Step 1]}

\noindent\textbf{Step 2 (Inner product with $e_{k+1}$).}  
Take the Euclidean norm squared of both sides of (1):
\[
\|e_{k+1}\|^{2}= \|e_k+\beta(e_k-e_{k-1})\|^{2}
        -2\alpha\langle\nabla f(x_k),\,e_k+\beta(e_k-e_{k-1})\rangle
        +\alpha^{2}\|\nabla f(x_k)\|^{2}. \tag{2}
\]
\textit{[Step 2]}

\noindent\textbf{Step 3 (Bound the gradient norm).}  
By $L$‑smoothness,
\[
\|\nabla f(x_k)\|^{2}\le 2L\bigl(f(x_k)-f(x^{\star})\bigr) \le L^{2}\|e_k\|^{2},
\]
where the last inequality follows from strong convexity:
$f(x_k)-f(x^{\star})\le \tfrac{L}{2}\|e_k\|^{2}$.  
Thus,
\[
\|\nabla f(x_k)\|^{2}\le L^{2}\|e_k\|^{2}. \tag{3}
\]
\textit{[Step 3]}

\noindent\textbf{Step 4 (Bounding the cross term).}  
Using Lemma~\ref{lem:quad},
\[
\langle\nabla f(x_k),e_k\rangle\ge\mu\|e_k\|^{2},\qquad
\langle\nabla f(x_k),e_{k-1}\rangle\le L\|e_{k-1}\|\|e_k\|.
\]
Hence,
\[
\begin{aligned}
\langle\nabla f(x_k),\,e_k+\beta(e_k-e_{k-1})\rangle
&\ge \mu\|e_k\|^{2}+\beta\bigl(\mu\|e_k\|^{2}-L\|e_k\|\|e_{k-1}\|\bigr)\\
&= \mu(1+\beta)\|e_k\|^{2}-\beta L\|e_k\|\|e_{k-1}\|. \tag{4}
\end{aligned}
\]
\textit{[Step 4]}

\noindent\textbf{Step 5 (Expanding the first norm term).}  
Compute
\[
\|e_k+\beta(e_k-e_{k-1})\|^{2}
= (1+\beta)^{2}\|e_k\|^{2}+ \beta^{2}\|e_{k-1}\|^{2}-2\beta(1+\beta)\langle e_k,e_{k-1}\rangle. \tag{5}
\]
\textit{[Step 5]}

\noindent\textbf{Step 6 (Assemble inequality).}  
Insert (3)–(5) into (2):
\[
\begin{aligned}
\|e_{k+1}\|^{2}
\le &\;(1+\beta)^{2}\|e_k\|^{2}+ \beta^{2}\|e_{k-1}\|^{2}
      -2\beta(1+\beta)\langle e_k,e_{k-1}\rangle \\
    &\; -2\alpha\Bigl[\mu(1+\beta)\|e_k\|^{2}-\beta L\|e_k\|\|e_{k-1}\|\Bigr]
      +\alpha^{2}L^{2}\|e_k\|^{2}.
\end{aligned}
\tag{6}
\]
\textit{[Step 6]}

\noindent\textbf{Step 7 (Introduce Lyapunov function).}  
Define
\[
\mathcal{V}_k:=\|e_k\|^{2}+c\|e_k-e_{k-1}\|^{2},\qquad c>0\ \text{to be chosen.}
\]
Expanding $\|e_k-e_{k-1}\|^{2}= \|e_k\|^{2}+\|e_{k-1}\|^{2}-2\langle e_k,e_{k-1}\rangle$, we obtain
\[
\mathcal{V}_k=(1+c)\|e_k\|^{2}+c\|e_{k-1}\|^{2}-2c\langle e_k,e_{k-1}\rangle. \tag{7}
\]
\textit{[Step 7]}

\noindent\textbf{Step 8 (Relate $\mathcal{V}_{k+1}$ to $\mathcal{V}_k$).}  
Using (6) and (7) we can write
\[
\mathcal{V}_{k+1}\le a\|e_k\|^{2}+b\|e_{k-1}\|^{2}-2d\langle e_k,e_{k-1}\rangle,
\]
where
\[
\begin{aligned}
a&=(1+\beta)^{2}+c(1+\beta)^{2}
    -2\alpha\mu(1+\beta)+\alpha^{2}L^{2},\\
b&=\beta^{2}+c\beta^{2},\\
d&=\beta(1+\beta)+c\beta(1+\beta)-\alpha\beta L.
\end{aligned}
\tag{8}
\]
\textit{[Step 8]}

\noindent\textbf{Step 9 (Choose $c$ to cancel the cross term).}  
Set $d=c(1+\beta)-\alpha L=0$, i.e.
\[
c=\frac{\alpha L}{1+\beta}. \tag{9}
\]
With this choice the cross term disappears and (8) simplifies to
\[
\mathcal{V}_{k+1}\le a\|e_k\|^{2}+b\|e_{k-1}\|^{2}. \tag{10}
\]
\textit{[Step 9]}

\noindent\textbf{Step 10 (Express $\mathcal{V}_{k}$ in the same basis).}  
From (7) with $c$ from (9):
\[
\mathcal{V}_k=(1+c)\|e_k\|^{2}+c\|e_{k-1}\|^{2}-2c\langle e_k,e_{k-1}\rangle
\ge (1+c)\|e_k\|^{2}+c\|e_{k-1}\|^{2}-2c\|e_k\|\|e_{k-1}\|.
\]
Using $2ab\le a^{2}+b^{2}$,
\[
\mathcal{V}_k\ge (1+c- c)\|e_k\|^{2}+ (c- c)\|e_{k-1}\|^{2}= \|e_k\|^{2}. \tag{11}
\]
Thus $\|e_k\|^{2}\le\mathcal{V}_k$.

\noindent\textbf{Step 11 (Derive contraction).}  
Substituting $c$ from (9) into $a$ and $b$ yields
\[
a=1-\alpha\mu\frac{2}{1+\beta}+\alpha^{2}L^{2},\qquad
b=\beta^{2}\Bigl(1+\frac{\alpha L}{1+\beta}\Bigr).
\]
Because of the parameter restriction \eqref{eq:param}, one can verify the inequality
\[
a+\beta\,b\le \rho^{2},\qquad \rho:=\max\{\,|1-\sqrt{\alpha\mu}|,\;|1-\sqrt{\alpha L}|\,\}<1. \tag{12}
\]
(Proof: treat the recursion as a quadratic characteristic equation $\lambda^{2}-(1+\beta-\alpha\mu)\lambda+\beta=0$ and use the standard root bound for the admissible $\alpha,\beta$.)

Now,
\[
\mathcal{V}_{k+1}\le a\|e_k\|^{2}+b\|e_{k-1}\|^{2}
\le (a+\beta b)\max\{\|e_k\|^{2},\|e_{k-1}\|^{2}\}
\le \rho^{2}\,\mathcal{V}_{k}.
\]
\textit{[Step 11]}

\noindent\textbf{Step 12 (Iterate the contraction).}  
Applying the inequality recursively,
\[
\mathcal{V}_{k}\le \rho^{2k}\mathcal{V}_{0},\qquad k\ge0.
\]
Together with (11) we obtain
\[
\|e_k\|^{2}\le \mathcal{V}_{k}\le \rho^{2k}\mathcal{V}_{0},
\]
i.e.
\[
\|x_k-x^{\star}\|\le \sqrt{\mathcal{V}_{0}}\,\rho^{k}=C\rho^{k}. \tag{13}
\]
\textit{[Step 12]}

\noindent\textbf{Step 13 (Function value bound).}  
Using $L$‑smoothness,
\[
f(x_k)-f(x^{\star})\le \frac{L}{2}\|x_k-x^{\star}\|^{2}
\le \frac{L}{2}C^{2}\rho^{2k}.
\]
\textit{[Step 13]}

\noindent\textbf{Conclusion.}  
Steps [Step 1]–[Step 13] establish the linear convergence claim of Theorem~\ref{thm:linear}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From the optimal choice $\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}}$ and $\beta^{\star}= \bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^{2}$ we obtain
\[
\rho^{\star}=1-\sqrt{\frac{\mu}{L}}=1-\frac{1}{\sqrt{\kappa}}.
\]
Hence for any $k$,
\[
\|x_k-x^{\star}\|\le C\bigl(1-\tfrac{1}{\sqrt{\kappa}}\bigr)^{k},
\qquad
f(x_k)-f(x^{\star})\le \frac{L}{2}C^{2}\bigl(1-\tfrac{1}{\sqrt{\kappa}}\bigr)^{2k}.
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Quadratic function $f(x)=\frac{1}{2}x^{\top}Qx-b^{\top}x$}.  
Here $\nabla f(x)=Qx-b$, $Q$ satisfies $\mu I\preceq Q\preceq L I$. The Heavy‑Ball iteration reduces to a linear system whose eigenvalues are exactly $1-\sqrt{\alpha\mu}$ and $1-\sqrt{\alpha L}$, confirming the rate $\rho$ derived above.
\item \textbf{Gradient descent as $\beta=0$}.  
The theorem collapses to the classic bound $\|x_k-x^{\star}\|\le (1-\alpha\mu)^{k}\|x_0-x^{\star}\|$, which is slower than the accelerated rate when $\kappa\gg1$.
\item \textbf{Limit $\beta\to 1^{-}$ with vanishing $\alpha$}.  
If $\alpha$ is scaled as $\alpha=\Theta(1/L)$ while $\beta\to1$, the method approaches a discretized second‑order differential equation (the continuous Heavy‑Ball ODE) and retains the same linear rate under the admissible parameter region.
\end{itemize}
