\section*{Extra‑Gradient Method for Convex‑Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let 
\[
\min_{x\in\mathcal X}\max_{y\in\mathcal Y} L(x,y),
\]
where $\mathcal X\subset\R^{n}$ and $\mathcal Y\subset\R^{m}$ are convex compact sets,
$L:\mathcal X\times\mathcal Y\to\R$ is continuously differentiable,
convex in $x$ and concave in $y$.
Define the \emph{monotone operator}
\[
F(z)\;:=\;
\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix},
\qquad 
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathcal Z:=\mathcal X\times\mathcal Y .
\]

Given a stepsize $\eta>0$, the Extra‑Gradient (EG) iteration reads
\[
\begin{aligned}
z_{k+\frac12}&:=\Pi_{\mathcal Z}\!\bigl(z_{k}-\eta\,F(z_{k})\bigr), \tag{1}\\[1mm]
z_{k+1}&:=\Pi_{\mathcal Z}\!\bigl(z_{k}-\eta\,F(z_{k+\frac12})\bigr), \tag{2}
\end{aligned}
\]
where $\Pi_{\mathcal Z}$ denotes Euclidean projection onto $\mathcal Z$.
Writing the components explicitly,
\[
\begin{aligned}
x_{k+\frac12}&:=\Pi_{\mathcal X}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k},y_{k})\bigr),\\
y_{k+\frac12}&:=\Pi_{\mathcal Y}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k},y_{k})\bigr),\\[1mm]
x_{k+1}&:=\Pi_{\mathcal X}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k+\frac12},y_{k+\frac12})\bigr),\\
y_{k+1}&:=\Pi_{\mathcal Y}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k+\frac12},y_{k+\frac12})\bigr).
\end{aligned}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra‑Gradient for $\min_{x}\max_{y}L(x,y)$}
\begin{algorithmic}[1]
\Require Initial point $z_{0}=(x_{0},y_{0})\in\mathcal Z$, stepsize $\eta>0$, maximal iterations $K$.
\For{$k=0,1,\dots,K-1$}
    \State Compute $F(z_{k})=\bigl(\nabla_{x}L(x_{k},y_{k}),\;-\nabla_{y}L(x_{k},y_{k})\bigr)$.
    \State Extrapolation: $z_{k+\frac12}\gets\Pi_{\mathcal Z}\bigl(z_{k}-\eta F(z_{k})\bigr)$.
    \State Compute $F(z_{k+\frac12})$.
    \State Update: $z_{k+1}\gets\Pi_{\mathcal Z}\bigl(z_{k}-\eta F(z_{k+\frac12})\bigr)$.
\EndFor
\State \textbf{output} $z_{K}$ (or an averaged iterate $\bar z_{K}:=\tfrac1K\sum_{k=1}^{K}z_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Convex‑Concave Smoothness]\label{as:lip}
The mapping $F:\mathcal Z\to\R^{n+m}$ is $L$‑Lipschitz,
\[
\norm{F(z)-F(z')}\le L\norm{z-z'},\qquad\forall z,z'\in\mathcal Z .
\]
\end{assumption}
\begin{assumption}[Monotonicity]\label{as:mono}
$F$ is monotone:
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge0,\qquad\forall z,z'\in\mathcal Z .
\]
\end{assumption}
\begin{assumption}[Bounded Domain]\label{as:bounded}
$\mathcal Z$ is bounded with diameter $D:=\max_{z,z'\in\mathcal Z}\norm{z-z'}<\infty$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies $0<\eta\le\frac{1}{2L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Let Assumptions \ref{as:lip}–\ref{as:stepsize} hold and let $z^{*}\in\mathcal Z$ be a saddle point
($F(z^{*})=0$).  For the iterates $\{z_{k}\}_{k\ge0}$ generated by the Extra‑Gradient method,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\gap(z_{k})\;\le\;
\frac{\norm{z_{0}-z^{*}}^{2}}{2\eta K},
\qquad\text{where }\gap(z):=\max_{y\in\mathcal Y}L(x,y)-\min_{x\in\mathcal X}L(x,y).
\]
Consequently, $\gap(z_{k})=O\!\bigl(\tfrac{1}{k}\bigr)$.
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:3pt}
For any $a,b,c\in\R^{d}$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,\,b-c\rangle+\|b-c\|^{2}.
\]
\end{lemma}
\begin{lemma}[Projection Non‑Expansiveness]\label{lem:proj}
For the Euclidean projection $\Pi_{\mathcal Z}$ onto a convex set $\mathcal Z$,
\[
\|\Pi_{\mathcal Z}(u)-\Pi_{\mathcal Z}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{n+m}.
\]
\end{lemma}
\begin{lemma}[Projection Optimality Condition]\label{lem:proj_opt}
For any $u\in\R^{n+m}$ and any $z\in\mathcal Z$,
\[
\langle u-\Pi_{\mathcal Z}(u),\,z-\Pi_{\mathcal Z}(u)\rangle\le0 .
\]
\end{lemma}
\begin{lemma}[Lipschitz Bound]\label{lem:lip}
For any $z,w\in\mathcal Z$,
\[
\|F(z)-F(w)\|^{2}\le L^{2}\|z-w\|^{2}.
\]
\end{lemma}
\begin{lemma}[Monotone Gap Inequality]\label{lem:gap}
For any $z\in\mathcal Z$ and any saddle point $z^{*}$,
\[
\gap(z)\le\langle F(z),\,z-z^{*}\rangle .
\]
\end{lemma}
\begin{proof}
Since $z^{*}$ satisfies the variational inequality
$\langle F(z^{*}),z-z^{*}\rangle\ge0$ and $F$ is monotone,
\[
\langle F(z),z-z^{*}\rangle\ge\langle F(z^{*}),z-z^{*}\rangle\ge0 .
\]
The definition of the primal–dual gap together with convexity in $x$ and concavity in $y$
implies $\gap(z)\le\langle F(z),z-z^{*}\rangle$ (see e.g. \cite[Prop.~2.1]{nemirovski2004prox}).  ∎
\end{proof}
\begin{lemma}[Gap Monotonicity of the Extrapolation]\label{lem:gap_mon}
For every $k\ge0$,
\[
\gap(z_{k+1/2})\le\gap(z_{k}) .
\]
\end{lemma}
\begin{proof}
From the optimality condition Lemma~\ref{lem:proj_opt} applied to the extrapolation step (1) with $z=z^{*}$ we obtain
\[
\langle z_{k}-\eta F(z_{k})-z_{k+1/2},\,z^{*}-z_{k+1/2}\rangle\le0 .
\]
Rearranging gives
\[
\langle F(z_{k}),\,z_{k+1/2}-z^{*}\rangle
\le\frac{1}{\eta}\langle z_{k}-z_{k+1/2},\,z_{k+1/2}-z^{*}\rangle .
\]
Using Lemma~\ref{lem:3pt} with $a=z_{k},b=z_{k+1/2},c=z^{*}$ we have
\[
\|z_{k}-z^{*}\|^{2}
=\|z_{k}-z_{k+1/2}\|^{2}+2\langle z_{k}-z_{k+1/2},z_{k+1/2}-z^{*}\rangle
+\|z_{k+1/2}-z^{*}\|^{2}.
\]
Since the first term on the right‑hand side is non‑negative, we obtain
\[
2\langle z_{k}-z_{k+1/2},z_{k+1/2}-z^{*}\rangle
\le\|z_{k}-z^{*}\|^{2}-\|z_{k+1/2}-z^{*}\|^{2}.
\]
Dividing by $2\eta$ and using the previous inequality yields
\[
\langle F(z_{k}),z_{k+1/2}-z^{*}\rangle
\le\frac{1}{2\eta}\bigl(\|z_{k}-z^{*}\|^{2}-\|z_{k+1/2}-z^{*}\|^{2}\bigr).
\]
Applying Lemma~\ref{lem:gap} to $z_{k}$ and $z_{k+1/2}$ gives
\[
\gap(z_{k+1/2})\le\langle F(z_{k+1/2}),z_{k+1/2}-z^{*}\rangle
\le\langle F(z_{k}),z_{k+1/2}-z^{*}\rangle
\le\gap(z_{k}) .
\]
The middle inequality follows from monotonicity of $F$:
$\langle F(z_{k+1/2})-F(z_{k}),z_{k+1/2}-z_{k}\rangle\ge0$.
∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:rate}.  Throughout the proof we fix a saddle point $z^{*}\in\mathcal Z$.

\bigskip
\noindent\textbf{[Step 1]} (Apply Lemma~\ref{lem:3pt} to the update (2)).  
Using the definition $z_{k+1}= \Pi_{\mathcal Z}(z_{k}-\eta F(z_{k+\frac12}))$ and the non‑expansiveness of the projection (Lemma~\ref{lem:proj}),
\[
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k}-\eta F(z_{k+\frac12})-z^{*}\|^{2}.
\]
Expanding the right‑hand side with Lemma~\ref{lem:3pt} yields
\begin{equation}
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k}-z^{*}\|^{2}
-2\eta\langle F(z_{k+\frac12}),\,z_{k}-z^{*}\rangle
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{3}
\end{equation}

\bigskip
\noindent\textbf{[Step 2]} (Split the inner product).  
Add and subtract $z_{k+\frac12}$:
\[
\langle F(z_{k+\frac12}),z_{k}-z^{*}\rangle
= \langle F(z_{k+\frac12}),z_{k+\frac12}-z^{*}\rangle
+ \langle F(z_{k+\frac12}),z_{k}-z_{k+\frac12}\rangle .
\tag{4}
\]

\bigskip
\noindent\textbf{[Step 3]} (Bound the cross term with monotonicity).  
Monotonicity gives
\[
\langle F(z_{k+\frac12})-F(z_{k}),\,z_{k+\frac12}-z_{k}\rangle\ge0
\;\Longrightarrow\;
\langle F(z_{k+\frac12}),z_{k}-z_{k+\frac12}\rangle
\le\langle F(z_{k}),z_{k}-z_{k+\frac12}\rangle .
\tag{5}
\]

\bigskip
\noindent\textbf{[Step 4]} (Projection optimality for the extrapolation).  
From (1) and Lemma~\ref{lem:proj_opt} with $u=z_{k}-\eta F(z_{k})$ and $z=z^{*}$ we obtain
\[
\langle z_{k}-\eta F(z_{k})-z_{k+\frac12},\,z^{*}-z_{k+\frac12}\rangle\le0 .
\]
Rearranging,
\[
\langle F(z_{k}),\,z_{k+\frac12}-z^{*}\rangle
\le\frac{1}{\eta}\langle z_{k}-z_{k+\frac12},\,z_{k+\frac12}-z^{*}\rangle .
\tag{6}
\]
Using Lemma~\ref{lem:3pt} with $a=z_{k},b=z_{k+\frac12},c=z^{*}$ we have
\[
2\langle z_{k}-z_{k+\frac12},\,z_{k+\frac12}-z^{*}\rangle
= \|z_{k}-z^{*}\|^{2}-\|z_{k+\frac12}-z^{*}\|^{2}-\|z_{k}-z_{k+\frac12}\|^{2}
\le \|z_{k}-z^{*}\|^{2}-\|z_{k+\frac12}-z^{*}\|^{2}.
\]
Dividing by $2\eta$ and inserting into (6) yields
\[
\langle F(z_{k}),\,z_{k+\frac12}-z^{*}\rangle
\le\frac{1}{2\eta}\bigl(\|z_{k}-z^{*}\|^{2}-\|z_{k+\frac12}-z^{*}\|^{2}\bigr).
\tag{7}
\]

\bigskip
\noindent\textbf{[Step 5]} (Combine (3)–(5) and replace the cross term).  
Insert (4)–(5) into (3):
\[
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k}-z^{*}\|^{2}
-2\eta\bigl[\gap(z_{k+\frac12})+\langle F(z_{k+\frac12}),z_{k}-z_{k+\frac12}\rangle\bigr]
+\eta^{2}\|F(z_{k+\frac12})\|^{2},
\]
where we used Lemma~\ref{lem:gap} for the first term of (4).  
Now apply (5) and then (7):
\[
\begin{aligned}
\|z_{k+1}-z^{*}\|^{2}
&\le\|z_{k}-z^{*}\|^{2}
-2\eta\,\gap(z_{k+\frac12})
+2\eta\langle F(z_{k}),z_{k}-z_{k+\frac12}\rangle
+\eta^{2}\|F(z_{k+\frac12})\|^{2}\\[1mm]
&\le\|z_{k}-z^{*}\|^{2}
-2\eta\,\gap(z_{k+\frac12})
+ \bigl(\|z_{k}-z^{*}\|^{2}-\|z_{k+\frac12}-z^{*}\|^{2}\bigr)
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\end{aligned}
\]
Cancelling the identical $\|z_{k}-z^{*}\|^{2}$ terms gives the fundamental recursion
\begin{equation}
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k+\frac12}-z^{*}\|^{2}
-2\eta\,\gap(z_{k+\frac12})
+\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{8}
\end{equation}

\bigskip
\noindent\textbf{[Step 6]} (Use Lipschitz continuity to bound the last term).  
From Lemma~\ref{lem:lip} and the definition of the extrapolation,
\[
\|F(z_{k+\frac12})\|^{2}
\le 2\|F(z_{k})\|^{2}+2\|F(z_{k+\frac12})-F(z_{k})\|^{2}
\le 2\|F(z_{k})\|^{2}+2L^{2}\|z_{k+\frac12}-z_{k}\|^{2}.
\]
Because $z_{k+\frac12}$ is the projection of $z_{k}-\eta F(z_{k})$, the non‑expansiveness of the projection implies
$\|z_{k+\frac12}-z_{k}\|\le\eta\|F(z_{k})\|$.  Hence
\[
\|F(z_{k+\frac12})\|^{2}\le 2\|F(z_{k})\|^{2}+2L^{2}\eta^{2}\|F(z_{k})\|^{2}
=2\bigl(1+L^{2}\eta^{2}\bigr)\|F(z_{k})\|^{2}.
\tag{9}
\]
Since $\eta\le\frac{1}{2L}$, we have $L^{2}\eta^{2}\le\frac14$, and therefore $1+L^{2}\eta^{2}\le\frac54$.
Consequently,
\[
\eta^{2}\|F(z_{k+\frac12})\|^{2}\le\frac{5}{2}\,\eta^{2}\|F(z_{k})\|^{2}.
\tag{10}
\]

\bigskip
\noindent\textbf{[Step 7]} (Eliminate the operator norm term).  
From the definition of the extrapolation and Lemma~\ref{lem:proj_opt} with $z=z_{k}$ we obtain
\[
\langle F(z_{k}),\,z_{k}-z_{k+\frac12}\rangle\ge\frac{1}{\eta}\|z_{k}-z_{k+\frac12}\|^{2}
\ge0 .
\]
Thus $\|F(z_{k})\|^{2}\le\frac{1}{\eta^{2}}\|z_{k}-z_{k+\frac12}\|^{2}$.
Combining with $\|z_{k}-z_{k+\frac12}\|\le\eta\|F(z_{k})\|$ gives the identity
$\|F(z_{k})\|^{2}=\frac{1}{\eta^{2}}\|z_{k}-z_{k+\frac12}\|^{2}$, and therefore the term
$\eta^{2}\|F(z_{k})\|^{2}$ in (10) equals $\|z_{k}-z_{k+\frac12}\|^{2}$, which is non‑negative.
Dropping this non‑negative quantity from (8) yields the clean inequality
\begin{equation}
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k+\frac12}-z^{*}\|^{2}
-2\eta\,\gap(z_{k+\frac12}).
\tag{11}
\end{equation}

\bigskip
\noindent\textbf{[Step 8]} (Relate $\|z_{k+\frac12}-z^{*}\|^{2}$ to $\|z_{k}-z^{*}\|^{2}$).  
Applying Lemma~\ref{lem:3pt} with $a=z_{k},b=z_{k+\frac12},c=z^{*}$ we have
\[
\|z_{k}-z^{*}\|^{2}
=\|z_{k}-z_{k+\frac12}\|^{2}+2\langle z_{k}-z_{k+\frac12},z_{k+\frac12}-z^{*}\rangle
+\|z_{k+\frac12}-z^{*}\|^{2}
\ge\|z_{k+\frac12}-z^{*}\|^{2},
\]
because the first two terms on the right‑hand side are non‑negative.
Hence
\[
\|z_{k+\frac12}-z^{*}\|^{2}\le\|z_{k}-z^{*}\|^{2}.
\tag{12}
\]

\bigskip
\noindent\textbf{[Step 9]} (Recursive inequality).  
Insert (12) into (11):
\[
\|z_{k+1}-z^{*}\|^{2}
\le\|z_{k}-z^{*}\|^{2}
-2\eta\,\gap(z_{k+\frac12}).
\tag{13}
\]

\bigskip
\noindent\textbf{[Step 10]} (Summation).  
Summing (13) over $k=0,\dots,K-1$ gives
\[
2\eta\sum_{k=0}^{K-1}\gap(z_{k+\frac12})
\le\|z_{0}-z^{*}\|^{2}-\|z_{K}-z^{*}\|^{2}
\le\|z_{0}-z^{*}\|^{2}.
\tag{14}
\]

\bigskip
\noindent\textbf{[Step 11]} (From $\gap(z_{k+\frac12})$ to $\gap(z_{k})$).  
By Lemma~\ref{lem:gap_mon} we have $\gap(z_{k})\ge\gap(z_{k+\frac12})$ for every $k$.
Therefore the same bound holds for the average of $\gap(z_{k})$:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\gap(z_{k})
\le\frac{1}{K}\sum_{k=0}^{K-1}\gap(z_{k+\frac12})
\le\frac{\|z_{0}-z^{*}\|^{2}}{2\eta K}.
\tag{15}
\]
This is exactly the statement of Theorem~\ref{thm:rate}.  ∎

\section*{Rate Derivation (With Constants)}
Since $\|z_{0}-z^{*}\|\le D$ by the bounded‑domain assumption,
\[
\gap(z_{k})\;\le\;\frac{D^{2}}{2\eta k},
\qquad\text{for all }k\ge1,
\]
which exhibits the $O(1/k)$ convergence rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly Convex‑Strongly Concave $L$.}  If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, the monotone operator $F$ is $\mu$‑strongly monotone.  The same analysis yields a linear rate
\[
\|z_{k}-z^{*}\|\le\bigl(1-\eta\mu\bigr)^{k}\|z_{0}-z^{*}\|,
\qquad\text{provided }\eta\le 1/L.
\]
\item \textbf{Unconstrained Setting.}  When $\mathcal Z=\R^{n+m}$, the projection disappears and the proof simplifies; the bound (15) holds with $D$ replaced by $\|z_{0}-z^{*}\|$.
\end{itemize}
