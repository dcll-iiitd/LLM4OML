\section*{Nesterov's Accelerated Gradient Descent}
\section*{Mathematical Formulation}
Let $f: \mathbb{R}^n \to \mathbb{R}$ be a convex and smooth function. Nesterov's Accelerated Gradient Descent maintains two sequences $\{x_k\}$ and $\{y_k\}$, initialized with $x_0 = x_{-1}$. The update rules at iteration $k \geq 0$ are:
\begin{align*}
y_k &= x_k + \beta_k (x_k - x_{k-1}) \\
x_{k+1} &= y_k - \alpha \nabla f(y_k)
\end{align*}
where $\alpha > 0$ is the learning rate (step size) and the momentum scheduling parameter is explicitly defined as $\beta_k = \frac{k-1}{k+2}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Nesterov's Accelerated Gradient Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0$, step size $\alpha > 0$, number of iterations $K$
\STATE $x_{-1} \leftarrow x_0$
\FOR{$k = 0, 1, 2, \dots, K-1$}
\STATE $\beta_k \leftarrow \frac{k-1}{k+2}$
\STATE $y_k \leftarrow x_k + \beta_k (x_k - x_{k-1})$
\STATE $x_{k+1} \leftarrow y_k - \alpha \nabla f(y_k)$
\ENDFOR
\ENSURE $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Apply to $f(w) = (w-3)^2$ with $w_0 = 0$, $\alpha = 0.1$, for 3 iterations. The gradient is $\nabla f(w) = 2(w-3)$.

\textbf{Iteration 0:} ($k=0$)
\begin{itemize}
\item $\beta_0 = \frac{0-1}{0+2} = -0.5$
\item $y_0 = x_0 + \beta_0(x_0 - x_{-1}) = 0 - 0.5(0 - 0) = 0$
\item $\nabla f(y_0) = 2(0 - 3) = -6$
\item $x_1 = y_0 - \alpha \nabla f(y_0) = 0 - 0.1(-6) = 0.6$
\item $f(x_1) = (0.6 - 3)^2 = 5.76$
\end{itemize}

\textbf{Iteration 1:} ($k=1$)
\begin{itemize}
\item $\beta_1 = \frac{1-1}{1+2} = 0$
\item $y_1 = x_1 + \beta_1(x_1 - x_0) = 0.6 + 0(0.6 - 0) = 0.6$
\item $\nabla f(y_1) = 2(0.6 - 3) = -4.8$
\item $x_2 = y_1 - \alpha \nabla f(y_1) = 0.6 - 0.1(-4.8) = 1.08$
\item $f(x_2) = (1.08 - 3)^2 = 3.6864$
\end{itemize}

\textbf{Iteration 2:} ($k=2$)
\begin{itemize}
\item $\beta_2 = \frac{2-1}{2+2} = 0.25$
\item $y_2 = x_2 + \beta_2(x_2 - x_1) = 1.08 + 0.25(1.08 - 0.6) = 1.2$
\item $\nabla f(y_2) = 2(1.2 - 3) = -3.6$
\item $x_3 = y_2 - \alpha \nabla f(y_2) = 1.2 - 0.1(-3.6) = 1.56$
\item $f(x_3) = (1.56 - 3)^2 = 2.0736$
\end{itemize}

\section*{Assumptions}
\begin{definition}
A function $f: \mathbb{R}^n \to \mathbb{R}$ is convex if for all $x, y \in \mathbb{R}^n$ and $\lambda \in [0,1]$:
$$f(\lambda x + (1-\lambda)y) \leq \lambda f(x) + (1-\lambda) f(y)$$
\end{definition}

\begin{definition}
A differentiable function $f$ is $L$-smooth if for all $x, y \in \mathbb{R}^n$:
$$\|\nabla f(x) - \nabla f(y)\| \leq L \|x - y\|$$
\end{definition}

\textbf{Formal Assumptions:}
\begin{enumerate}
\item $f$ is convex and differentiable.
\item $f$ is $L$-smooth for some $L > 0$.
\item The step size is fixed at $\alpha = \frac{1}{L}$.
\item The optimal value $f^* = \min f(x)$ is attained at some $x^*$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
Let $\{x_k\}$ be the sequence generated by Nesterov's Accelerated Gradient Descent with $\alpha = \frac{1}{L}$ and $\beta_k = \frac{k-1}{k+2}$. For any $K \geq 1$,
$$f(x_K) - f^* \leq \frac{2L \|x_0 - x^*\|^2}{(K+1)^2}$$
\end{theorem}

\section*{Theoretical Properties}
\textbf{Stability:} The algorithm is stable provided $\alpha \leq \frac{1}{L}$. The momentum term does not introduce divergence because the specific scheduling $\beta_k$ is carefully balanced with the step size.

\textbf{Hyperparameter Sensitivity:} The convergence rate heavily relies on the exact scheduling of $\beta_k$. If $\beta_k$ grows too fast, the algorithm may overshoot; if too slow, it fails to achieve acceleration.

\textbf{Comparison to Baselines:} Standard Gradient Descent achieves $f(x_K) - f^* = O(1/K)$. Nesterov's method achieves $O(1/K^2)$, which is provably optimal among first-order methods for this class of functions.

\section*{Discussion}
The $O(1/K^2)$ rate is a landmark result in optimization, representing a theoretical speedup from $O(1/\epsilon)$ to $O(1/\sqrt{\epsilon})$ iterations to reach $\epsilon$-accuracy. The proof relies on an estimate sequence framework, which constructs a sequence of functions that bound $f(x_k)$ and converge to $f^*$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]
For an $L$-smooth function $f$, for any $x, y \in \mathbb{R}^n$:
$$f(y) \leq f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2} \|y - x\|^2$$
\end{lemma}
\begin{proof}
This is a standard property of $L$-smooth functions derived from the descent lemma and Cauchy-Schwarz.
\end{proof}

\begin{lemma}[Convexity Inequality]
For a convex function $f$, for any $x, y \in \mathbb{R}^n$:
$$f(y) \geq f(x) + \langle \nabla f(x), y - x \rangle$$
\end{lemma}
\begin{proof}
This is the first-order characterization of convexity.
\end{proof}

Now we need to relate the RHS to $V_k$.
We know $y_k = x_k + \frac{k-1}{k+2}(x_k - x_{k-1})$.

$L\|y_k - x^*\|^2 = L\|v_k - x^*\|^2 + 2L\langle v_k - x^*, y_k - v_k \rangle + L\|y_k - v_k\|^2$.
We also know $x_{k+1} - v_k = (y_k - v_k) - \frac{1}{L}\nabla f(y_k)$.
So $\frac{1}{L}\nabla f(y_k) = (y_k - v_k) - (x_{k+1} - v_k)$.
