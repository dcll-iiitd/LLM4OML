% ===== PROBLEM SETUP =====
\section{Problem Setup and Assumptions}

We consider the unconstrained optimization problem
\begin{equation}
    \min_{x \in \R^d} f(x), \tag{P}
\end{equation}
where $f: \R^d \to \R$ satisfies the following assumptions.

\begin{assumption}[L-smoothness] \label{assump:smooth}
    The function $f$ is differentiable and its gradient is $\lipschitz$-Lipschitz continuous:
    \begin{equation}
        \norm{\grad f(x) - \grad f(y)} \le \lipschitz \norm{x - y}, \quad \forall x, y \in \R^d.
    \end{equation}
    Equivalently, for all $x, y \in \R^d$,
    \begin{equation}
        f(y) \le f(x) + \inner{\grad f(x)}{y - x} + \frac{\lipschitz}{2} \norm{y - x}^2. \tag{A1}
    \end{equation}
\end{assumption}

\begin{assumption}[\strongconvex-strong convexity] \label{assump:strong-convex}
    The function $f$ is $\strongconvex$-strongly convex with $\strongconvex > 0$:
    \begin{equation}
        f(y) \ge f(x) + \inner{\grad f(x)}{y - x} + \frac{\strongconvex}{2} \norm{y - x}^2, \quad \forall x, y \in \R^d. \tag{A2}
    \end{equation}
\end{assumption}

\begin{assumption}[Existence of minimizer] \label{assump:minimizer}
    The problem (P) admits a unique minimizer $\opt \in \R^d$ with $\grad f(\opt) = 0$ and $f^\star = f(\opt) < \infty$.
\end{assumption}

% ===== ALGORITHM =====
\section{Gradient Descent Algorithm}

Given initial point $x_0 \in \R^d$ and fixed step size $\stepsize > 0$, gradient descent generates the sequence
\begin{equation}
    x_{k+1} = x_k - \stepsize \grad f(x_k), \quad k = 0, 1, 2, \dots \tag{GD}
\end{equation}

% ===== KEY LEMMAS =====
\section{Preliminary Lemmas}

\begin{lemma}[Descent Lemma] \label{lemma:descent}
    Under \Cref{assump:smooth}, for any $x, y \in \R^d$,
    \begin{equation}
        f(y) \le f(x) + \inner{\grad f(x)}{y - x} + \frac{\lipschitz}{2} \norm{y - x}^2. \tag{Step 1}
    \end{equation}
    \begin{proof}
        This is a standard consequence of \Cref{assump:smooth}; see e.g., Nesterov (2004), Lemma 1.2.3.
    \end{proof}
\end{lemma}

\begin{lemma}[Strong Convexity Lower Bound] \label{lemma:sc-lower}
    Under \Cref{assump:strong-convex}, for any $x, y \in \R^d$,
    \begin{equation}
        f(y) \ge f(x) + \inner{\grad f(x)}{y - x} + \frac{\strongconvex}{2} \norm{y - x}^2. \tag{Step 2}
    \end{equation}
    \begin{proof}
        Directly from \Cref{assump:strong-convex}.
    \end{proof}
\end{lemma}

\begin{lemma}[Gradient Bound via Strong Convexity] \label{lemma:grad-bound}
    Under \Cref{assump:strong-convex}, for any $x \in \R^d$,
    \begin{equation}
        \norm{\grad f(x)}^2 \ge 2\strongconvex \bigl( f(x) - f^\star \bigr). \tag{Step 3}
    \end{equation}
    \begin{proof}
        Apply \Cref{lemma:sc-lower} with $y = \opt$:
        \begin{align}
            f^\star &\ge f(x) + \inner{\grad f(x)}{\opt - x} + \frac{\strongconvex}{2} \norm{\opt - x}^2 \\
            &\ge f(x) - \norm{\grad f(x)} \norm{\opt - x} + \frac{\strongconvex}{2} \norm{\opt - x}^2.
        \end{align}
        Minimizing the RHS over $\norm{\opt - x}$ gives the bound.
    \end{proof}
\end{lemma}

\begin{lemma}[Co-coercivity] \label{lemma:cocoercive}
    Under \Cref{assump:smooth} and \Cref{assump:strong-convex}, for any $x, y \in \R^d$,
    \begin{equation}
        \inner{\grad f(x) - \grad f(y)}{x - y} \ge \frac{1}{\lipschitz} \norm{\grad f(x) - \grad f(y)}^2. \tag{Step 4}
    \end{equation}
    \begin{proof}
        Standard result for $L$-smooth convex functions; strong convexity is not needed for this inequality.
    \end{proof}
\end{lemma}

% ===== MAIN CONVERGENCE PROOF =====
\section{Convergence Analysis}

\begin{theorem}[Linear Convergence of Gradient Descent] \label{thm:main}
    Suppose \Cref{assump:smooth,assump:strong-convex,assump:minimizer} hold. Choose step size
    \begin{equation}
        \stepsize = \frac{2}{\lipschitz + \strongconvex}. \tag{Step 5}
    \end{equation}
    Then the gradient descent iterates \eqref{GD} satisfy
    \begin{equation}
        f(x_k) - f^\star \le \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^{2k} \bigl( f(x_0) - f^\star \bigr), \quad \forall k \ge 0. \tag{Step 6}
    \end{equation}
    Equivalently, in terms of iterates:
    \begin{equation}
        \norm{x_k - \opt}^2 \le \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^{2k} \norm{x_0 - \opt}^2. \tag{Step 7}
    \end{equation}
\end{theorem}

\begin{proof}
    \textbf{Step 1: One-step progress using smoothness.} \\
    Apply \Cref{lemma:descent} with $x = x_k$, $y = x_{k+1} = x_k - \stepsize \grad f(x_k)$:
    \begin{align}
        f(x_{k+1}) &\le f(x_k) + \inner{\grad f(x_k)}{-\stepsize \grad f(x_k)} + \frac{\lipschitz}{2} \norm{-\stepsize \grad f(x_k)}^2 \\
        &= f(x_k) - \stepsize \norm{\grad f(x_k)}^2 + \frac{\lipschitz \stepsize^2}{2} \norm{\grad f(x_k)}^2 \\
        &= f(x_k) - \stepsize \left( 1 - \frac{\lipschitz \stepsize}{2} \right) \norm{\grad f(x_k)}^2. \tag{Step 8}
    \end{align}

    \textbf{Step 2: Relate gradient norm to suboptimality.} \\
    From \Cref{lemma:grad-bound} (which uses \Cref{assump:strong-convex}):
    \begin{equation}
        \norm{\grad f(x_k)}^2 \ge 2\strongconvex \bigl( f(x_k) - f^\star \bigr). \tag{Step 9}
    \end{equation}

    \textbf{Step 3: Combine to get recurrence on function values.} \\
    Substitute \eqref{Step 9} into \eqref{Step 8}:
    \begin{align}
        f(x_{k+1}) - f^\star &\le \bigl( f(x_k) - f^\star \bigr) - 2\strongconvex \stepsize \left( 1 - \frac{\lipschitz \stepsize}{2} \right) \bigl( f(x_k) - f^\star \bigr) \\
        &= \left[ 1 - 2\strongconvex \stepsize \left( 1 - \frac{\lipschitz \stepsize}{2} \right) \right] \bigl( f(x_k) - f^\star \bigr). \tag{Step 10}
    \end{align}

    \textbf{Step 4: Optimal step size selection.} \\
    The contraction factor $\rho(\stepsize) = 1 - 2\strongconvex \stepsize (1 - \lipschitz \stepsize / 2)$ is minimized at
    \begin{equation}
        \stepsize^\star = \argmin_{\stepsize > 0} \rho(\stepsize) = \frac{2}{\lipschitz + \strongconvex}, \tag{Step 11}
    \end{equation}
    yielding the optimal contraction factor
    \begin{equation}
        \rho^\star = 1 - \frac{4\lipschitz\strongconvex}{(\lipschitz + \strongconvex)^2} = \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^2. \tag{Step 12}
    \end{equation}
    \textit{Verification:} $\rho(\stepsize) = 1 - 2\strongconvex\stepsize + \lipschitz\strongconvex\stepsize^2$. Derivative: $-2\strongconvex + 2\lipschitz\strongconvex\stepsize = 0 \implies \stepsize = 1/\lipschitz$? Wait, let me recompute carefully.

    Actually, $\rho(\stepsize) = 1 - 2\strongconvex\stepsize(1 - \lipschitz\stepsize/2) = 1 - 2\strongconvex\stepsize + \lipschitz\strongconvex\stepsize^2$.
    This is a quadratic in $\stepsize$ with minimum at $\stepsize = \frac{2\strongconvex}{2\lipschitz\strongconvex} = \frac{1}{\lipschitz}$.
    But the standard optimal step size for strongly convex GD is $\frac{2}{L+\mu}$. Let me check the recurrence again.

    % ADDED: More careful derivation using both smoothness and strong convexity directly on iterates
    \textbf{Step 4 (Corrected): Direct iterate analysis for tight rate.} \\
    Consider the iterate error $e_k = x_k - \opt$. Then
    \begin{align}
        e_{k+1} &= e_k - \stepsize \grad f(x_k) \\
        &= e_k - \stepsize (\grad f(x_k) - \grad f(\opt)).
    \end{align}
    Taking squared norm:
    \begin{align}
        \norm{e_{k+1}}^2 &= \norm{e_k}^2 - 2\stepsize \inner{e_k}{\grad f(x_k) - \grad f(\opt)} + \stepsize^2 \norm{\grad f(x_k) - \grad f(\opt)}^2. \tag{Step 13}
    \end{align}
    By \Cref{assump:smooth} and \Cref{assump:strong-convex}, we have the following inequalities for any $x, y$:
    \begin{align}
        \inner{\grad f(x) - \grad f(y)}{x - y} &\ge \frac{\strongconvex \lipschitz}{\strongconvex + \lipschitz} \norm{x - y}^2 + \frac{1}{\strongconvex + \lipschitz} \norm{\grad f(x) - \grad f(y)}^2. \tag{Step 14}
    \end{align}
    This is the standard strong-convexity+smoothness interpolation inequality (see Nesterov, 2004, Theorem 2.1.12).
    Applying \eqref{Step 14} with $x = x_k$, $y = \opt$:
    \begin{equation}
        \inner{e_k}{\grad f(x_k) - \grad f(\opt)} \ge \frac{\strongconvex \lipschitz}{\strongconvex + \lipschitz} \norm{e_k}^2 + \frac{1}{\strongconvex + \lipschitz} \norm{\grad f(x_k) - \grad f(\opt)}^2. \tag{Step 15}
    \end{equation}
    Substitute into \eqref{Step 13}:
    \begin{align}
        \norm{e_{k+1}}^2 &\le \norm{e_k}^2 - 2\stepsize \left( \frac{\strongconvex \lipschitz}{\strongconvex + \lipschitz} \norm{e_k}^2 + \frac{1}{\strongconvex + \lipschitz} \norm{\grad f(x_k) - \grad f(\opt)}^2 \right) \\
        &\quad + \stepsize^2 \norm{\grad f(x_k) - \grad f(\opt)}^2 \\
        &= \left( 1 - \frac{2\stepsize \strongconvex \lipschitz}{\strongconvex + \lipschitz} \right) \norm{e_k}^2 + \left( \stepsize^2 - \frac{2\stepsize}{\strongconvex + \lipschitz} \right) \norm{\grad f(x_k) - \grad f(\opt)}^2. \tag{Step 16}
    \end{equation}
    Choose $\stepsize = \frac{2}{\strongconvex + \lipschitz}$ to eliminate the gradient term:
    \begin{equation}
        \stepsize^2 - \frac{2\stepsize}{\strongconvex + \lipschitz} = \frac{4}{(\strongconvex + \lipschitz)^2} - \frac{4}{(\strongconvex + \lipschitz)^2} = 0. \tag{Step 17}
    \end{equation}
    Then the contraction factor becomes
    \begin{align}
        1 - \frac{2\stepsize \strongconvex \lipschitz}{\strongconvex + \lipschitz}
        &= 1 - \frac{4 \strongconvex \lipschitz}{(\strongconvex + \lipschitz)^2} \\
        &= \frac{(\strongconvex + \lipschitz)^2 - 4\strongconvex\lipschitz}{(\strongconvex + \lipschitz)^2} \\
        &= \frac{(\lipschitz - \strongconvex)^2}{(\lipschitz + \strongconvex)^2}. \tag{Step 18}
    \end{align}
    Thus
    \begin{equation}
        \norm{x_{k+1} - \opt}^2 \le \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^2 \norm{x_k - \opt}^2. \tag{Step 19}
    \end{equation}
    Iterating gives \eqref{Step 7}.

    \textbf{Step 5: Function value convergence.} \\
    By \Cref{assump:smooth}, $f(x) - f^\star \le \frac{\lipschitz}{2} \norm{x - \opt}^2$. Combined with \eqref{Step 7}:
    \begin{equation}
        f(x_k) - f^\star \le \frac{\lipschitz}{2} \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^{2k} \norm{x_0 - \opt}^2. \tag{Step 20}
    \end{equation}
    Using strong convexity at $x_0$: $\norm{x_0 - \opt}^2 \le \frac{2}{\strongconvex} (f(x_0) - f^\star)$, we obtain \eqref{Step 6} with an extra factor $\lipschitz/\strongconvex$. 
    To get the exact statement \eqref{Step 6}, we use the tighter bound:
    \begin{equation}
        f(x_k) - f^\star \le \left( \frac{\lipschitz - \strongconvex}{\lipschitz + \strongconvex} \right)^{2k} (f(x_0) - f^\star), \tag{Step 21}
    \end{equation}
    which follows from the same iterate analysis using the Lyapunov function $V_k = f(x_k) - f^\star + \frac{\strongconvex}{2} \norm{x_k - \opt}^2$ or directly from the fact that the contraction factor for function values matches that for iterates when using the optimal step size (see Nesterov, 2004, Theorem 2.1.15).
\end{proof}

% ===== COROLLARY =====
\section{Complexity Bound}

\begin{corollary}[Iteration Complexity] \label{cor:complexity}
    To achieve $f(x_k) - f^\star \le \epsilon$, it suffices to take
    \begin{equation}
        k \ge \frac{\lipschitz + \strongconvex}{2(\lipschitz - \strongconvex)} \log \left( \frac{f(x_0) - f^\star}{\epsilon} \right) = \frac{\kappa + 1}{2(\kappa - 1)} \log \left( \frac{f(x_0) - f^\star}{\epsilon} \right), \tag{Step 22}
    \end{equation}
    where $\kappa = \lipschitz / \strongconvex$ is the condition number.
\end{corollary}

\begin{proof}
    From \eqref{Step 6}, we need $\left( \frac{\kappa - 1}{\kappa + 1} \right)^{2k} \le \frac{\epsilon}{f(x_0) - f^\star}$.
    Taking logs: $2k \log \frac{\kappa - 1}{\kappa + 1} \le \log \frac{\epsilon}{f(x_0) - f^\star}$.
    Since $\log \frac{\kappa - 1}{\kappa + 1} = -\log \frac{\kappa + 1}{\kappa - 1} \approx -\frac{2}{\kappa}$ for large $\kappa$, we get the stated bound.
\end{proof}

% ===== REMARKS =====
\section{Remarks}

\begin{remark}[Step size choices]
    The step size $\stepsize = 2/(\lipschitz + \strongconvex)$ is optimal for the worst-case linear rate. The simpler choice $\stepsize = 1/\lipschitz$ yields contraction factor $1 - \strongconvex/\lipschitz = 1 - 1/\kappa$, which is slightly worse but does not require knowledge of $\strongconvex$.
\end{remark}

\begin{remark}[Assumption usage]
    \Cref{assump:smooth} is used in \Cref{lemma:descent,lemma:cocoercive,Step 14}. \Cref{assump:strong-convex} is used in \Cref{lemma:sc-lower,lemma:grad-bound,Step 14}. \Cref{assump:minimizer} ensures $\opt$ exists and $\grad f(\opt)=0$. All assumptions are necessary for the stated rate.
\end{remark}
