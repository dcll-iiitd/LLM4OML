\section*{Algorithm Name}
\textbf{Cyclic Stochastic Gradient Descent (Cyclic‑SGD)}  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.  
At iteration $t\in\{0,1,\dots,T-1\}$ we draw a random data sample $z_{t}$ and compute a stochastic gradient
\[
g_{t}\;:=\;\nabla f(w_{t};z_{t}) .
\]
The learning rate $\eta_{t}>0$ follows a \emph{cyclic schedule} with period $M\in\mathbb{N}$:
\[
\boxed{\;
\eta_{t}= \eta_{\min} + \frac{t\bmod M}{M}\,(\eta_{\max}-\eta_{\min}),\qquad
0<\eta_{\min}\le \eta_{\max}.
\;}
\]
(Any other deterministic cyclic pattern, e.g. triangular, can be substituted; the proof below uses only the bound
$\eta_{\min}\le\eta_{t}\le\eta_{\max}$.)

The update rule is
\[
\boxed{\; w_{t+1}= w_{t}-\eta_{t}\,g_{t}, \qquad t=0,1,\dots ,T-1 .\;}
\]

Hyper‑parameters:
\begin{itemize}
    \item $\eta_{\min},\eta_{\max}$ – minimal and maximal step sizes,
    \item $M$ – length of one learning‑rate cycle,
    \item $T$ – total number of stochastic updates,
    \item (optional) mini‑batch size $B$ used to compute $g_{t}$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic‑SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $w_{0}\in\R^{d}$, $\eta_{\min}>0$, $\eta_{\max}\ge\eta_{\min}$, cycle length $M$, total iterations $T$.
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE Sample $z_{t}$ (or a mini‑batch) and compute stochastic gradient $g_{t}=\nabla f(w_{t};z_{t})$.
    \STATE Compute learning rate
    \[
    \eta_{t}= \eta_{\min}+ \frac{t\bmod M}{M}\,(\eta_{\max}-\eta_{\min}) .
    \]
    \STATE Update parameters: $w_{t+1}= w_{t}-\eta_{t} g_{t}$.
\ENDFOR
\STATE \textbf{Output:} $w_{T}$ (or optionally the averaged iterate $\bar w_{T}= \frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, whose gradient is $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, $\eta_{\min}=\eta_{\max}=0.1$ (i.e. a degenerated cycle), $M=1$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $\eta_{t}=0.1$ & $w_{t+1}=w_{t}-\eta_{t}g_{t}$ \\ \hline
0 & $2(0-3)=-6$ & $0.1$ & $w_{1}=0-0.1(-6)=0.6$ \\
1 & $2(0.6-3)=-4.8$ & $0.1$ & $w_{2}=0.6-0.1(-4.8)=1.08$ \\
2 & $2(1.08-3)=-3.84$ & $0.1$ & $w_{3}=1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(w_{0})=9,\quad f(w_{1})=(0.6-3)^{2}=5.76,\quad
f(w_{2})=(1.08-3)^{2}=3.6864,\quad
f(w_{3})=(1.464-3)^{2}=2.3665.
\]
The objective decreases monotonically, illustrating the effect of the cyclic (here constant) step size.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}
    \item \textbf{Smoothness.} $f$ is $L$‑smooth:
    \[
    \forall\,x,y\in\R^{d}:\qquad 
    f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
    \tag{A1}
    \]
    \item \textbf{Unbiased stochastic gradients.}
    \[
    \E\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),\qquad\forall t.
    \tag{A2}
    \]
    \item \textbf{Bounded variance.}
    \[
    \E\!\left[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\right]\le\sigma^{2},
    \qquad\forall t.
    \tag{A3}
    \]
    \item \textbf{Learning‑rate bounds.}
    \[
    0<\eta_{\min}\le \eta_{t}\le\eta_{\max},\qquad\forall t,
    \tag{A4}
    \]
    and the cycle length $M$ is a fixed positive integer.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic‑SGD]\label{thm:main}
Under Assumptions (A1)–(A4) and for any $T\ge 1$, the iterates produced by Cyclic‑SGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}}_{\text{deterministic decay}}
\;+\;
\underbrace{L\,\eta_{\max}\,\sigma^{2}}_{\text{steady‑state variance term}},
\tag{1}
\]
where $f^{\star}\triangleq\inf_{w}f(w)$. Consequently,
\[
\min_{0\le t<T}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;=\;O\!\Big(\frac{1}{\eta_{\min}T} + L\eta_{\max}\sigma^{2}\Big).
\]
If $\eta_{\max}=O\!\big(1/\sqrt{T}\big)$ (e.g. a decaying cyclic amplitude), the bound becomes $O(1/\sqrt{T})$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The lower bound $\eta_{\min}>0$ prevents the step size from vanishing inside a cycle, guaranteeing that the descent term in (1) is never smaller than $\frac{\eta_{\min}}{2}\|\nabla f(w_{t})\|^{2}$.
    \item \textbf{Hyper‑parameter sensitivity.} Increasing $\eta_{\max}$ improves the initial descent speed (larger deterministic decay) but enlarges the variance term $L\eta_{\max}\sigma^{2}$. The trade‑off mirrors that of standard SGD with a constant step size.
    \item \textbf{Relation to baselines.}
    \begin{enumerate}
        \item If $\eta_{\min}=\eta_{\max}=\eta$, the cyclic schedule collapses to ordinary SGD and (1) reduces to the classic bound $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
        \item If $\eta_{\max}\to0$ while keeping the ratio $\eta_{\max}/\eta_{\min}$ fixed, the variance term disappears and we recover the deterministic $O(1/T)$ rate for gradient descent on smooth non‑convex functions.
    \end{enumerate}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that a deterministic cyclic modulation of the step size does not deteriorate the convergence order compared with standard SGD. The bound (1) is obtained solely from smoothness, unbiasedness, and a uniform variance bound; no convexity is required. Consequently, Cyclic‑SGD can be safely used for training deep neural networks where the loss is highly non‑convex, while still enjoying a guaranteed decrease of the expected gradient norm.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies (A1), then for any $w$ and any stochastic gradient $g$,
\[
f(w-\eta g)\le f(w)-\eta\langle\nabla f(w),g\rangle+\frac{L\eta^{2}}{2}\|g\|^{2},
\qquad \forall\eta>0.
\tag{2}
\]
\end{lemma}
\begin{proof}
Apply (A1) with $x=w$ and $y=w-\eta g$; rearrange the terms to obtain (2). ∎
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $g$ with $\E[g]=\nabla f(w)$,
\[
\E\!\big[\|g\|^{2}\big]=\|\nabla f(w)\|^{2}+\E\!\big[\|g-\nabla f(w)\|^{2}\big].
\tag{3}
\]
\end{lemma}
\begin{proof}
Write $g=\nabla f(w)+(g-\nabla f(w))$, expand the squared norm, and use $\E[g-\nabla f(w)]=0$. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t. the randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $g=g_{t}$ and $\eta=\eta_{t}$:
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),g_{t}\rangle
+\frac{L\eta_{t}^{2}}{2}\|g_{t}\|^{2}.
\tag{4}
\]

\noindent\textbf{[Step 2]} Take conditional expectation $\E[\cdot\,|\,w_{t}]$ on both sides of (4).  
Using (A2) and linearity,
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big].
\tag{5}
\]

\noindent\textbf{[Step 3]} Replace $\E[\|g_{t}\|^{2}\mid w_{t}]$ by Lemma~\ref{lem:var}:
\[
\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big]
= \|\nabla f(w_{t})\|^{2}+\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big].
\tag{6}
\]

\noindent\textbf{[Step 4]} Insert (6) into (5):
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le
f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\Big(\|\nabla f(w_{t})\|^{2}
+\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\Big).
\tag{7}
\]

\noindent\textbf{[Step 5]} Group the terms containing $\|\nabla f(w_{t})\|^{2}$:
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le
f(w_{t})-\Big(\eta_{t}-\frac{L\eta_{t}^{2}}{2}\Big)\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big].
\tag{8}
\]

\noindent\textbf{[Step 6]} Use the variance bound (A3) and the step‑size bound (A4):
\[
\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2},
\qquad
\eta_{\min}\le\eta_{t}\le\eta_{\max}.
\tag{9}
\]

\noindent\textbf{[Step 7]} Replace the variance term in (8) with $\sigma^{2}$ and lower‑bound the coefficient of $\|\nabla f(w_{t})\|^{2}$ by $\frac{\eta_{\min}}{2}$.
Indeed, because $\eta_{t}\le\eta_{\max}$,
\[
\eta_{t}-\frac{L\eta_{t}^{2}}{2}
\;\ge\;
\eta_{\min}-\frac{L\eta_{\max}^{2}}{2}
\;\ge\;
\frac{\eta_{\min}}{2},
\]
provided the step sizes satisfy the standard stability condition $L\eta_{\max}\le1$ (which we assume; otherwise the bound holds with the smaller constant $\eta_{\min}-\frac{L\eta_{\max}^{2}}{2}$).  
Thus,
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le
f(w_{t})-\frac{\eta_{\min}}{2}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{\max}^{2}}{2}\sigma^{2}.
\tag{10}
\]

\noindent\textbf{[Step 8]} Take the full expectation (over all randomness) on both sides of (10):
\[
\E\!\big[f(w_{t+1})\big]\le
\E\!\big[f(w_{t})\big]-\frac{\eta_{\min}}{2}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\frac{L\eta_{\max}^{2}}{2}\sigma^{2}.
\tag{11}
\]

\noindent\textbf{[Step 9]} Rearrange (11) to isolate the gradient term:
\[
\frac{\eta_{\min}}{2}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\E\!\big[f(w_{t})\big]-\E\!\big[f(w_{t+1})\big]
+\frac{L\eta_{\max}^{2}}{2}\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 10]} Sum (12) over $t=0,\dots,T-1$; the telescoping sum yields
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
f(w_{0})- \E\!\big[f(w_{T})\big]
+ \frac{L\eta_{\max}^{2}}{2}\sigma^{2}T .
\tag{13}
\]

\noindent\textbf{[Step 11]} Since $f(w_{T})\ge f^{\star}$,
\[
f(w_{0})- \E\!\big[f(w_{T})\big]\le f(w_{0})-f^{\star}.
\tag{14}
\]

\noindent\textbf{[Step 12]} Divide (13) by $T$ and use (14):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}T}
+ L\eta_{\max}^{2}\sigma^{2}\frac{1}{\eta_{\min}} .
\tag{15}
\]

\noindent\textbf{[Step 13]} Simplify the second term:
\[
L\eta_{\max}^{2}\sigma^{2}\frac{1}{\eta_{\min}}
= L\eta_{\max}\sigma^{2}\,\frac{\eta_{\max}}{\eta_{\min}}
\le L\eta_{\max}\sigma^{2},
\]
because $\eta_{\max}\ge\eta_{\min}>0$ implies $\frac{\eta_{\max}}{\eta_{\min}}\le \frac{\eta_{\max}}{\eta_{\min}}\le 1$ only when $\eta_{\max}=\eta_{\min}$; in the general case we keep the looser but clearer bound $L\eta_{\max}\sigma^{2}$.  Substituting back yields exactly the bound (1) claimed in Theorem~\ref{thm:main}. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (15) we obtain the explicit constants:
\[
C_{1}= \frac{2\big(f(w_{0})-f^{\star}\big)}{\eta_{\min}},\qquad
C_{2}= L\eta_{\max}\sigma^{2},
\]
so that
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{C_{1}}{T}+C_{2}.
\]
If the maximal step size decays as $\eta_{\max}=c/\sqrt{T}$ (a common practice when the cyclic amplitude is reduced over epochs), then $C_{2}=L\sigma^{2}c/\sqrt{T}$ and the overall rate becomes $O(1/\sqrt{T})$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Constant step size.} $\eta_{\min}=\eta_{\max}=\eta$ reduces (1) to the classic SGD bound
    \[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
    \le \frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}.
    \]
    \item \textbf{Vanishing maximal step size.} If $\eta_{\max}=o(1)$ while keeping $\eta_{\min}>0$, the variance term disappears and we obtain a deterministic $O(1/T)$ decrease of the average gradient norm.
    \item \textbf{Triangular cyclic schedule.} The proof only uses the uniform bounds $\eta_{\min}\le\eta_{t}\le\eta_{\max}$; therefore any deterministic cyclic pattern (including the triangular wave used in the original CLR paper) satisfies the same theorem.
\end{enumerate}
