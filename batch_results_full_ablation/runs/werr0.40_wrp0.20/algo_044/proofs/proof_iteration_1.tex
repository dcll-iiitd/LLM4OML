\section*{Algorithm Name}
\textbf{DP--SGD with Gaussian Mechanism}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be the empirical risk
\[
f(w)=\frac{1}{n}\sum_{i=1}^{n} \ell(w;z_i),
\]
where $\ell$ is convex and $L$--smooth in $w$, and $\{z_i\}_{i=1}^n$ is the dataset.
At iteration $t\in\{0,1,\dots,T-1\}$ we sample a minibatch $\mathcal{B}_t$ of size $b$ uniformly without replacement,
compute the (clipped) stochastic gradient
\[
\tilde{g}_t=\frac{1}{b}\sum_{i\in\mathcal{B}_t}\operatorname{clip}\big(\nabla\ell(w_t;z_i),C\big),
\]
add Gaussian noise
\[
g_t=\tilde{g}_t + \xi_t,\qquad \xi_t\sim\mathcal{N}(0,\sigma^2\mathbf{I}_d),
\]
and update
\[
w_{t+1}= w_t - \eta_t g_t,
\]
where $\eta_t>0$ is the learning rate.  
The privacy loss is tracked by the moments accountant (or Rényi DP), yielding a total $(\varepsilon,\delta)$ budget after $T$ steps.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP--SGD with Gaussian Noise}
\begin{algorithmic}[1]
\STATE \textbf{Input:} data $\{z_i\}_{i=1}^n$, initial $w_0$, learning rates $\{\eta_t\}$, clipping norm $C$, noise scale $\sigma$, batch size $b$, total iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample minibatch $\mathcal{B}_t\subset\{1,\dots,n\}$, $|\mathcal{B}_t|=b$.
    \STATE Compute per‑example gradients $g_i=\nabla\ell(w_t;z_i)$ for $i\in\mathcal{B}_t$.
    \STATE Clip: $\hat{g}_i=\frac{g_i}{\max\{1,\|g_i\|_2/C\}}$.
    \STATE Average: $\tilde{g}_t=\frac{1}{b}\sum_{i\in\mathcal{B}_t}\hat{g}_i$.
    \STATE Sample noise $\xi_t\sim\mathcal{N}(0,\sigma^2\mathbf{I}_d)$.
    \STATE Form noisy gradient $g_t=\tilde{g}_t+\xi_t$.
    \STATE Update: $w_{t+1}=w_t-\eta_t g_t$.
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or averaged iterate $\bar w_T=\frac{1}{T}\sum_{t=1}^{T}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3),\qquad L=2.
\]
Set $w_0=0$, learning rate $\eta=0.1$, clipping $C=10$ (no effect), noise variance $\sigma^2=0$ (to illustrate the deterministic part), and batch size $b=1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$ \\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$ \\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$ \\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3689$
\end{tabular}
\end{center}
The iterates move toward the optimum $w^\star=3$, and the objective value decreases each step.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity.} $f$ is convex:
\[
f(\theta w+(1-\theta)w')\le \theta f(w)+(1-\theta)f(w'),\qquad\forall w,w'\in\R^d,\ \theta\in[0,1].
\]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\|\nabla f(w)-\nabla f(w')\|_2\le L\|w-w'\|_2,\qquad\forall w,w'.
\]
\item \textbf{Gradient clipping bound.} For every example $i$ and any $w$, $\|\operatorname{clip}(\nabla\ell(w;z_i),C)\|_2\le C$.
\item \textbf{Noise.} $\xi_t\sim\mathcal{N}(0,\sigma^2\mathbf{I}_d)$ independent across $t$.
\item \textbf{Learning rates.} $\eta_t=\eta$ constant with $0<\eta\le \frac{1}{L}$.
\item \textbf{Finite variance of stochastic gradient.} For the clipped gradient,
\[
\E_{\mathcal{B}_t}\big[\|\tilde{g}_t-\nabla f(w_t)\|_2^2\big]\le \frac{C^2}{b}.
\]
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of DP--SGD]\label{thm:main}
Under the assumptions above, for any $T\ge1$ the averaged iterate
\[
\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1} w_t
\]
satisfies
\[
\E\!\left[f(\bar w_T)-f(w^\star)\right]\le
\frac{\|w_0-w^\star\|_2^2}{2\eta T}
+\frac{\eta C^2}{2b}
+\frac{\eta d\sigma^2}{2},
\]
where $w^\star$ is a minimizer of $f$.
Consequently, choosing $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
\E\!\left[f(\bar w_T)-f(w^\star)\right]=O\!\Big(\frac{1}{\sqrt{T}}\Big)
+O\!\big(\eta d\sigma^2\big).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The clipping operator makes the per‑iteration update $L$‑Lipschitz in $w_t$, guaranteeing bounded sensitivity and thus finite privacy loss.
\item \textbf{Hyperparameter Sensitivity.}
    \begin{itemize}
    \item Larger batch size $b$ reduces the stochastic variance term $\frac{C^2}{b}$.
    \item Larger noise scale $\sigma$ improves privacy (smaller $\varepsilon$) but inflates the term $\frac{\eta d\sigma^2}{2}$, slowing convergence.
    \item The stepsize $\eta$ trades off bias (first term) against variance (second and third terms); optimal $\eta\propto 1/\sqrt{T}$ balances them.
    \end{itemize}
\item \textbf{Comparison to non‑private SGD.} Removing the noise ($\sigma=0$) recovers the classical bound $\frac{\|w_0-w^\star\|^2}{2\eta T}+ \frac{\eta C^2}{2b}$, i.e. $O(1/\sqrt{T})$ for $\eta\propto 1/\sqrt{T}$.
\item \textbf{Privacy Guarantee.} By the moments accountant, the sequence $\{w_t\}_{t=0}^{T-1}$ is $(\varepsilon,\delta)$‑DP with
\[
\varepsilon = O\!\Big(\frac{q\sqrt{T\log(1/\delta)}}{\sigma}\Big),
\qquad q:=\frac{b}{n}.
\]
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w,w'$
\[
f(w')\le f(w)+\langle\nabla f(w),w'-w\rangle+\frac{L}{2}\|w'-w\|_2^2 .
\]
\end{lemma}
\begin{lemma}[Unbiased noisy gradient]\label{lem:unbiased}
For any $t$,
\[
\E[g_t\mid w_t]=\nabla f(w_t),
\qquad
\E\big[\|g_t-\nabla f(w_t)\|_2^2\mid w_t\big]
\le \frac{C^2}{b}+d\sigma^2 .
\]
\end{lemma}
\begin{proof}
The clipping ensures $\|\hat g_i\|\le C$, thus $\|\tilde g_t\|\le C$. Hence
\[
\E[\tilde g_t\mid w_t]=\nabla f(w_t),\qquad
\E\|\tilde g_t-\nabla f(w_t)\|_2^2\le\frac{C^2}{b}.
\]
Since $\xi_t$ is independent zero‑mean Gaussian with covariance $\sigma^2\mathbf I_d$, we add $d\sigma^2$ to the variance bound. The expectation of $g_t$ remains $\nabla f(w_t)$. \qedhere
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken over the randomness of minibatch sampling and Gaussian noise.

\bigskip
\noindent\textbf{[Step 1]}  Start from the squared distance recursion using the update $w_{t+1}=w_t-\eta g_t$:
\[
\|w_{t+1}-w^\star\|_2^2
= \|w_t-w^\star\|_2^2-2\eta\langle g_t,w_t-w^\star\rangle+\eta^2\|g_t\|_2^2 .
\tag{1}
\]

\noindent\textbf{[Step 2]}  Take conditional expectation w.r.t.\ $\mathcal{F}_t:=\sigma(w_0,\dots,w_t)$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\left[\|w_{t+1}-w^\star\|_2^2\mid\mathcal{F}_t\right]
= \|w_t-w^\star\|_2^2-2\eta\langle \nabla f(w_t),w_t-w^\star\rangle
+\eta^2\E\!\left[\|g_t\|_2^2\mid\mathcal{F}_t\right].
\tag{2}
\]

\noindent\textbf{[Step 3]}  Expand $\E[\|g_t\|_2^2\mid\mathcal{F}_t]$:
\[
\E[\|g_t\|_2^2\mid\mathcal{F}_t]
= \|\nabla f(w_t)\|_2^2
+ \E\!\left[\|g_t-\nabla f(w_t)\|_2^2\mid\mathcal{F}_t\right]
\le \|\nabla f(w_t)\|_2^2+\frac{C^2}{b}+d\sigma^2 .
\tag{3}
\]

\noindent\textbf{[Step 4]}  Plug (3) into (2):
\[
\E\!\left[\|w_{t+1}-w^\star\|_2^2\mid\mathcal{F}_t\right]
\le \|w_t-w^\star\|_2^2-2\eta\langle \nabla f(w_t),w_t-w^\star\rangle
+\eta^2\|\nabla f(w_t)\|_2^2+\eta^2\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{4}
\]

\noindent\textbf{[Step 5]}  Use the identity $-2\langle a,b\rangle+\|a\|_2^2 = \|b-a\|_2^2-\|b\|_2^2$ with $a=\eta\nabla f(w_t)$ and $b=w_t-w^\star$:
\[
-2\eta\langle \nabla f(w_t),w_t-w^\star\rangle+\eta^2\|\nabla f(w_t)\|_2^2
= \|w_t-w^\star-\eta\nabla f(w_t)\|_2^2-\|w_t-w^\star\|_2^2 .
\tag{5}
\]

\noindent\textbf{[Step 6]}  Substitute (5) into (4):
\[
\E\!\left[\|w_{t+1}-w^\star\|_2^2\mid\mathcal{F}_t\right]
\le \|w_t-w^\star-\eta\nabla f(w_t)\|_2^2
+\eta^2\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{6}
\]

\noindent\textbf{[Step 7]}  Apply Lemma~\ref{lem:smooth} with $w'=w_t-\eta\nabla f(w_t)$:
\[
f(w_t-\eta\nabla f(w_t))\le f(w_t)-\eta\|\nabla f(w_t)\|_2^2+\frac{L\eta^2}{2}\|\nabla f(w_t)\|_2^2.
\]
Since $\eta\le 1/L$, we have $1-\eta L/2\ge 1/2$, yielding
\[
\|w_t-w^\star-\eta\nabla f(w_t)\|_2^2
\le \|w_t-w^\star\|_2^2-2\eta\bigl(f(w_t)-f(w^\star)\bigr).
\tag{7}
\]

\noindent\textbf{[Step 8]}  Combine (6) and (7):
\[
\E\!\left[\|w_{t+1}-w^\star\|_2^2\mid\mathcal{F}_t\right]
\le \|w_t-w^\star\|_2^2-2\eta\bigl(f(w_t)-f(w^\star)\bigr)
+\eta^2\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{8}
\]

\noindent\textbf{[Step 9]}  Take total expectation and rearrange:
\[
2\eta\,\E\!\big[f(w_t)-f(w^\star)\big]
\le \E\!\big[\|w_t-w^\star\|_2^2\big]-\E\!\big[\|w_{t+1}-w^\star\|_2^2\big]
+\eta^2\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{9}
\]

\noindent\textbf{[Step 10]}  Sum (9) over $t=0,\dots,T-1$ telescopically:
\[
2\eta\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^\star)\big]
\le \|w_0-w^\star\|_2^2-\E\!\big[\|w_T-w^\star\|_2^2\big]
+T\eta^2\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{10}
\]

\noindent\textbf{[Step 11]}  Drop the non‑negative term $-\E[\|w_T-w^\star\|^2]$ and divide by $2\eta T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^\star)\big]
\le \frac{\|w_0-w^\star\|_2^2}{2\eta T}
+\frac{\eta}{2}\!\left(\frac{C^2}{b}+d\sigma^2\right).
\tag{11}
\]

\noindent\textbf{[Step 12]}  By convexity, $f(\bar w_T)\le\frac{1}{T}\sum_{t=0}^{T-1}f(w_t)$. Taking expectation yields the desired bound:
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]
\le\frac{\|w_0-w^\star\|_2^2}{2\eta T}
+\frac{\eta C^2}{2b}
+\frac{\eta d\sigma^2}{2}.
\tag{12}
\]

This completes the proof of Theorem~\ref{thm:main}. \qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
Choosing $\eta = \frac{\|w_0-w^\star\|_2}{\sqrt{T\left(\frac{C^2}{b}+d\sigma^2\right)}}$ (which satisfies $\eta\le 1/L$ for sufficiently large $T$) gives
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]
\le
\frac{\|w_0-w^\star\|_2\sqrt{\frac{C^2}{b}+d\sigma^2}}{\sqrt{T}} .
\]
Hence the convergence rate is $O\!\big(\tfrac{1}{\sqrt{T}}\big)$ with an explicit constant depending on the clipping bound $C$, batch size $b$, dimension $d$, and noise scale $\sigma$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{No noise ($\sigma=0$).} The bound reduces to the classical SGD result.
\item \textbf{Full‑batch ($b=n$).} The stochastic variance term $\frac{C^2}{b}$ disappears, leaving only the privacy‑induced term $\frac{\eta d\sigma^2}{2}$.
\item \textbf{High privacy ($\sigma\gg C/\sqrt{b}$).} The term $\frac{\eta d\sigma^2}{2}$ dominates, and the optimal stepsize scales as $\eta\propto 1/(\sigma\sqrt{T})$, giving a slower $O\big(\sigma/\sqrt{T}\big)$ decay.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof follows the standard deterministic SGD analysis, with two additional variance contributions:
the minibatch stochasticity $\frac{C^2}{b}$ and the Gaussian mechanism $d\sigma^2$.
Both appear linearly in the stepsize $\eta$, so the practitioner can trade privacy (large $\sigma$) against utility by adjusting $\eta$ and $b$.
The bound is tight up to constant factors and matches known lower bounds for convex DP optimization.
