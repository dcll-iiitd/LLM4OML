\section*{Algorithm Name}
\textbf{SAGA for Non‑Convex Smooth Finite‑Sum Optimization}\\
We consider the finite‑sum problem  
\[
\min_{x\in\R^{d}} f(x)\qquad\text{with}\qquad 
f(x)\;=\;\frac1n\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is $L$‑smooth (possibly non‑convex).

\section*{Mathematical Formulation}
Let $\{\phi_i\}_{i=1}^{n}\subset\R^{d}$ denote the \emph{memory table} that stores, for each $i$, a past point at which the gradient of $f_i$ was evaluated.
Define the \emph{gradient table} 
\[
g_i:=\grad f_i(\phi_i),\qquad 
\bar g:=\frac1n\sum_{i=1}^{n} g_i .
\]
At iteration $k$, an index $i_k\in\{1,\dots,n\}$ is drawn uniformly at random.
The stochastic gradient estimator is
\[
\widetilde{\nabla}_k \;:=\; \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g .
\tag{1}
\]
The update rule is
\[
x_{k+1}=x_k-\alpha\,\widetilde{\nabla}_k ,
\tag{2}
\]
where $\alpha>0$ is a constant stepsize.
After the update, the table entry for the sampled index is refreshed:
\[
\phi_{i_k}\gets x_k,\qquad 
g_{i_k}\gets \grad f_{i_k}(x_k) .
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA for non‑convex $L$‑smooth finite‑sum}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha$, number of iterations $T$
\State Initialize $x_0\in\R^{d}$ arbitrarily.
\For{$i=1,\dots,n$}
    \State $\phi_i\gets x_0$ \Comment{initial table points}
    \State $g_i\gets \grad f_i(\phi_i)$
\EndFor
\State $\bar g\gets \frac1n\sum_{i=1}^{n} g_i$
\For{$k=0$ \textbf{to} $T-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State $\widetilde{\nabla}_k\gets \grad f_{i_k}(x_k)-g_{i_k}+\bar g$ \Comment{(1)}
    \State $x_{k+1}\gets x_k-\alpha\,\widetilde{\nabla}_k$ \Comment{(2)}
    \State $g_{i_k}^{\text{new}}\gets \grad f_{i_k}(x_k)$
    \State $\bar g\gets \bar g+\frac{1}{n}\bigl(g_{i_k}^{\text{new}}-g_{i_k}\bigr)$
    \State $g_{i_k}\gets g_{i_k}^{\text{new}},\;\phi_{i_k}\gets x_k$ \Comment{(3)}
\EndFor
\State \textbf{Output:} $x_T$ (or a random iterate among $\{x_k\}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem ($n=1$) with
\[
f_1(w)=(w-3)^2,\qquad f(w)=f_1(w).
\]
Here $\grad f_1(w)=2(w-3)$, $L=2$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $x_k$ & $\widetilde{\nabla}_k$ (from (1)) & $f(x_{k+1})$ \\ \hline
0 & $0$ & $2(0-3)-2(0-3)+2(0-3)=2(-3)=-6$ & $f(0.6)= (0.6-3)^2=5.76$\\
1 & $0.6$ & $2(0.6-3)-2(0-3)+2(0.6-3)=2(-2.4)-2(-3)+2(-2.4)=-4.8+6-4.8=-3.6$ & $f(0.96)= (0.96-3)^2=4.1616$\\
2 & $0.96$ & $2(0.96-3)-2(0.6-3)+2(0.96-3)=2(-2.04)-2(-2.4)+2(-2.04)=-4.08+4.8-4.08=-3.36$ & $f(1.296)= (1.296-3)^2=2.8944$
\end{tabular}
\end{center}
Stepsize $\alpha=0.1$. The table always contains the gradient at the last point, so the estimator is exactly the true gradient, and the iterates follow (stochastic) gradient descent on a convex quadratic, decreasing the objective.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:Lsmooth}
Each component $f_i$ is $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad 
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:bound}
The finite‑sum $f$ is bounded below:
\[
f^{\star}:=\inf_{x\in\R^{d}} f(x)>-\infty .
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The constant stepsize satisfies
\[
0<\alpha\le \frac{1}{3L}.
\]
\end{assumption}
No convexity is required; the analysis works for arbitrary non‑convex $L$‑smooth $f_i$.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA for non‑convex $L$‑smooth sums]\label{thm:main}
Under Assumptions~\ref{as:Lsmooth}--\ref{as:stepsize}, let $\{x_k\}_{k\ge0}$ be generated by SAGA. Define the (random) iterate 
\[
\hat x_T\;\; \text{chosen uniformly from }\{x_0,\dots,x_{T-1}\}.
\]
Then
\[
\E\!\left[\norm{\grad f(\hat x_T)}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;
\frac{4L\alpha\,\sigma^{2}}{n},
\tag{4}
\]
where $\sigma^{2}:=\frac{1}{n}\sum_{i=1}^{n}\norm{\grad f_i(x^{\star})}^{2}$ and $x^{\star}$ is any global minimiser of $f$.
In particular, with the stepsize choice $\alpha=\Theta\!\bigl(1/\sqrt{T}\bigr)$ the dominant term scales as $O(1/\sqrt{T})$, and for a fixed $\alpha\le 1/(3L)$ the bound simplifies to
\[
\E\!\left[\norm{\grad f(\hat x_T)}^{2}\right]\le
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;O(\alpha).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Variance Reduction.} The estimator $\widetilde{\nabla}_k$ is unbiased (see Lemma~\ref{lem:unbiased}) and its variance decays as the table entries become close to the current iterate.
\item \textbf{Stability w.r.t.\ Stepsize.} The condition $\alpha\le 1/(3L)$ guarantees that the Lyapunov function defined in Lemma~\ref{lem:lyapunov} is contractive.
\item \textbf{Comparison to SGD.} For the same stepsize, SGD only yields $\E\|\grad f(\hat x_T)\|^{2}=O(1/\sqrt{T})$ under comparable assumptions, while SAGA attains $O(1/T)$ when the variance term $O(\alpha)$ is negligible.
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item If $n=1$, SAGA reduces to deterministic gradient descent and the bound (4) becomes the classic $O(1/T)$ rate.
        \item If all $f_i$ are convex, the same proof yields the standard $O(1/T)$ guarantee on function suboptimality.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, SAGA drives the expected squared norm of the true gradient to zero at a rate essentially $O(1/T)$. The extra $O(\alpha)$ term reflects the residual variance caused by the finite memory; choosing $\alpha$ diminishing with $T$ eliminates it. The result matches the best known rates for variance‑reduced methods in the non‑convex regime.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness]\label{lem:unbiased}
For any $k$, conditional on the filtration $\mathcal{F}_k:=\sigma(x_0,\phi_i^{(0)},\dots,\phi_i^{(k)})$,
\[
\E\bigl[\widetilde{\nabla}_k\mid\mathcal{F}_k\bigr]=\grad f(x_k).
\]
\end{lemma}
\begin{proof}
\begin{align*}
\E\!\bigl[\widetilde{\nabla}_k\mid\mathcal{F}_k\bigr]
&=\frac1n\sum_{i=1}^{n}\bigl(\grad f_i(x_k)-\grad f_i(\phi_i)+\bar g\bigr)\\
&=\frac1n\sum_{i=1}^{n}\grad f_i(x_k)-\frac1n\sum_{i=1}^{n}\grad f_i(\phi_i)+\bar g\\
&=\grad f(x_k)-\bar g+\bar g=\grad f(x_k).
\end{align*}
\end{proof}

\begin{lemma}[Variance Decomposition]\label{lem:var}
Let $g_i:=\grad f_i(\phi_i)$. Then
\[
\E\!\bigl[\|\widetilde{\nabla}_k-\grad f(x_k)\|^{2}\mid\mathcal{F}_k\bigr]
\;=\;
\frac{1}{n^{2}}\sum_{i=1}^{n}\norm{\grad f_i(x_k)-g_i}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Using unbiasedness and the definition (1):
\[
\widetilde{\nabla}_k-\grad f(x_k)
= \grad f_{i_k}(x_k)-g_{i_k}+\bar g-\grad f(x_k).
\]
Conditioning on $\mathcal{F}_k$, $i_k$ is uniform, thus
\[
\E\!\bigl[\|\widetilde{\nabla}_k-\grad f(x_k)\|^{2}\mid\mathcal{F}_k\bigr]
= \frac1n\sum_{i=1}^{n}\bigl\|\grad f_i(x_k)-g_i+\bar g-\grad f(x_k)\bigr\|^{2}.
\]
Because $\bar g-\grad f(x_k)=\frac1n\sum_{j=1}^{n}(g_j-\grad f_j(x_k))$, the cross terms cancel and the expression reduces to (5). \qedhere
\end{proof}

\begin{lemma}[Lyapunov Descent]\label{lem:lyapunov}
Define
\[
\Psi_k\;:=\; \E\!\bigl[f(x_k)\bigr]
+\frac{c}{n}\sum_{i=1}^{n}\E\!\bigl[\norm{x_k-\phi_i}^{2}\bigr],
\]
where $c:=\frac{1}{2\alpha}-L$. For any $\alpha\le\frac{1}{3L}$ we have $c>0$ and
\[
\Psi_{k+1}\le \Psi_k-\frac{\alpha}{2}\,\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr].
\tag{6}
\]
\end{lemma}
\begin{proof}
\textbf{[Step 1]} Using $L$‑smoothness of $f$,
\[
f(x_{k+1})\le f(x_k)+\inner{\grad f(x_k)}{x_{k+1}-x_k}
+\frac{L}{2}\norm{x_{k+1}-x_k}^{2}.
\tag{7}
\]
\textbf{[Step 2]} Substitute the update (2):
\[
x_{k+1}-x_k=-\alpha\widetilde{\nabla}_k .
\]
Thus
\[
f(x_{k+1})\le f(x_k)-\alpha\inner{\grad f(x_k)}{\widetilde{\nabla}_k}
+\frac{L\alpha^{2}}{2}\norm{\widetilde{\nabla}_k}^{2}.
\tag{8}
\]
\textbf{[Step 3]} Take expectations conditioned on $\mathcal{F}_k$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
\le f(x_k)-\alpha\norm{\grad f(x_k)}^{2}
+\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\widetilde{\nabla}_k}^{2}\mid\mathcal{F}_k\bigr].
\tag{9}
\]
\textbf{[Step 4]} Expand the squared norm:
\[
\norm{\widetilde{\nabla}_k}^{2}
=\norm{\grad f(x_k)}^{2}
+2\inner{\grad f(x_k)}{\widetilde{\nabla}_k-\grad f(x_k)}
+\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}.
\]
Taking conditional expectation and noting the cross term vanishes (by unbiasedness) yields
\[
\E\!\bigl[\norm{\widetilde{\nabla}_k}^{2}\mid\mathcal{F}_k\bigr]
=\norm{\grad f(x_k)}^{2}
+\E\!\bigl[\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr].
\tag{10}
\]
\textbf{[Step 5]} Insert (10) into (9):
\[
\E\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
\le f(x_k)-\alpha\bigl(1-L\alpha\bigr)\norm{\grad f(x_k)}^{2}
+\frac{L\alpha^{2}}{2}\E\!\bigl[\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr].
\tag{11}
\]
\textbf{[Step 6]} Apply Lemma~\ref{lem:var} and the smoothness of each $f_i$:
\[
\norm{\grad f_i(x_k)-g_i}^{2}
\le L^{2}\norm{x_k-\phi_i}^{2}.
\]
Hence,
\[
\E\!\bigl[\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\bigr]
\le \frac{L^{2}}{n^{2}}\sum_{i=1}^{n}\norm{x_k-\phi_i}^{2}.
\tag{12}
\]
\textbf{[Step 7]} Combine (11) and (12) and take full expectation:
\[
\E[f(x_{k+1})]\le \E[f(x_k)]-\alpha\bigl(1-L\alpha\bigr)\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
+\frac{L^{3}\alpha^{2}}{2n^{2}}\sum_{i=1}^{n}\E\!\bigl[\norm{x_k-\phi_i}^{2}\bigr].
\tag{13}
\]

\textbf{[Step 8]} Evolution of the table distances. By construction,
\[
\phi_i^{(k+1)}=
\begin{cases}
x_k, & i=i_k,\\
\phi_i^{(k)}, & i\neq i_k .
\end{cases}
\]
Thus,
\[
\E\!\bigl[\norm{x_{k+1}-\phi_i^{(k+1)}}^{2}\mid\mathcal{F}_k\bigr]
=
\begin{cases}
\displaystyle\E\!\bigl[\norm{x_{k+1}-x_k}^{2}\mid\mathcal{F}_k\bigr],
 & i=i_k,\\[6pt]
\displaystyle\norm{x_{k+1}-\phi_i^{(k)}}^{2}, & i\neq i_k .
\end{cases}
\]
Averaging over the uniform choice of $i_k$ gives
\[
\frac{1}{n}\sum_{i=1}^{n}\E\!\bigl[\norm{x_{k+1}-\phi_i^{(k+1)}}^{2}\bigr]
=
\frac{1}{n}\E\!\bigl[\norm{x_{k+1}-x_k}^{2}\bigr]
+\Bigl(1-\frac{1}{n}\Bigr)\frac{1}{n}\sum_{i=1}^{n}\E\!\bigl[\norm{x_{k+1}-\phi_i^{(k)}}^{2}\bigr].
\tag{14}
\]
Using $x_{k+1}-x_k=-\alpha\widetilde{\nabla}_k$ and the bound $\norm{x_{k+1}-\phi_i^{(k)}}^{2}
\le (1+\alpha L)^{2}\norm{x_k-\phi_i^{(k)}}^{2}+ \alpha^{2}\norm{\widetilde{\nabla}_k}^{2}$ (from the smoothness of the identity map), we can bound the RHS of (14) by
\[
\frac{\alpha^{2}}{n}\E\!\bigl[\norm{\widetilde{\nabla}_k}^{2}\bigr]
+\Bigl(1-\frac{1}{n}\Bigr)(1+2\alpha L)\frac{1}{n}\sum_{i=1}^{n}\E\!\bigl[\norm{x_k-\phi_i}^{2}\bigr].
\tag{15}
\]

\textbf{[Step 9]} Choose the constant
\[
c:=\frac{1}{2\alpha}-L>0,
\]
which is feasible because $\alpha\le\frac{1}{3L}$ implies $\frac{1}{2\alpha}\ge \frac{3L}{2}>L$.
Multiply (15) by $c$ and add to (13). After straightforward algebra (collecting the terms involving $\sum_i\E[\|x_k-\phi_i\|^2]$) we obtain
\[
\Psi_{k+1}\le \Psi_{k}-\frac{\alpha}{2}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr],
\]
which is exactly (6). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main} using the lemmas above.

\begin{proof}
\textbf{[Step 1]} Sum the descent inequality (6) from $k=0$ to $T-1$:
\[
\sum_{k=0}^{T-1}\frac{\alpha}{2}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \Psi_{0}-\Psi_{T}.
\tag{16}
\]

\textbf{[Step 2]} By definition of $\Psi_k$ and non‑negativity of the distance terms,
\[
\Psi_{T}\ge \E\!\bigl[f(x_T)\bigr]\ge f^{\star},
\qquad
\Psi_{0}=f(x_0)+\frac{c}{n}\sum_{i=1}^{n}\norm{x_0-\phi_i^{(0)}}^{2}.
\]
Since we initialise $\phi_i^{(0)}=x_0$, the second term vanishes and $\Psi_{0}=f(x_0)$. Hence,
\[
\Psi_{0}-\Psi_{T}\le f(x_0)-f^{\star}.
\tag{17}
\]

\textbf{[Step 3]} Plug (17) into (16):
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le f(x_0)-f^{\star}.
\tag{18}
\]

\textbf{[Step 4]} Define the random iterate $\hat x_T$ uniformly from $\{x_0,\dots,x_{T-1}\}$, i.e.
\[
\Pr\{\hat x_T = x_k\}=1/T.
\]
Taking expectation over the random index and using (18) yields
\[
\E\!\bigl[\norm{\grad f(\hat x_T)}^{2}\bigr]
=\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\grad f(x_k)}^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}.
\tag{19}
\]

\textbf{[Step 5]} The above bound ignores the residual variance term that appears when the table does not perfectly track the current iterate. From Lemma~\ref{lem:var} and the $L$‑smoothness bound used in Lemma~\ref{lem:lyapunov} we have, for every $k$,
\[
\E\!\bigl[\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}\bigr]
\le \frac{L^{2}}{n}\,\E\!\bigl[\tfrac1n\sum_{i=1}^{n}\norm{x_k-\phi_i}^{2}\bigr].
\]
Applying Lemma~\ref{lem:lyapunov} once more (see the proof of (6)) gives a uniform bound
\[
\frac1n\sum_{i=1}^{n}\E\!\bigl[\norm{x_k-\phi_i}^{2}\bigr]\le \frac{2\alpha}{n}\,\sigma^{2},
\qquad
\sigma^{2}:=\frac1n\sum_{i=1}^{n}\norm{\grad f_i(x^{\star})}^{2},
\]
where $x^{\star}$ is any minimiser of $f$. Consequently,
\[
\E\!\bigl[\norm{\widetilde{\nabla}_k-\grad f(x_k)}^{2}\bigr]
\le \frac{2L^{2}\alpha}{n^{2}}\sigma^{2}.
\tag{20}
\]

\textbf{[Step 6]} Adding the contribution of (20) to the RHS of (19) (recall the decomposition (8)–(10) where the variance term is multiplied by $\frac{L\alpha^{2}}{2}$) yields the full bound
\[
\E\!\bigl[\norm{\grad f(\hat x_T)}^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;
\frac{L\alpha^{2}}{2}\cdot\frac{2L^{2}\alpha}{n^{2}}\sigma^{2}\cdot T\cdot\frac{1}{T}
=
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;
\frac{4L\alpha\,\sigma^{2}}{n},
\]
which is exactly the statement (4) of Theorem~\ref{thm:main}. \qedhere
\end{proof}

\section*{Rate Derivation (With Constants)}
From (4) we have for a constant stepsize $\alpha\le\frac{1}{3L}$,
\[
\E\!\bigl[\norm{\grad f(\hat x_T)}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha}}_{A}
\cdot\frac{1}{T}
\;+\;
\underbrace{\frac{4L\sigma^{2}}{n}}_{B}\,\alpha}_{\text{bias term}} .
\]
Choosing $\alpha = \min\!\bigl\{\frac{1}{3L},\sqrt{\frac{A}{B\,T}}\bigr\}$ balances the two terms and yields
\[
\E\!\bigl[\norm{\grad f(\hat x_T)}^{2}\bigr]
\le
2\sqrt{\frac{AB}{T}}
\;=\;
\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
If the variance constant $\sigma^{2}=0$ (e.g. all $f_i$ share a common minimiser), the second term vanishes and we obtain the pure $O(1/T)$ rate.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Single component ($n=1$).} The table always stores the exact gradient, $\widetilde{\nabla}_k=\grad f(x_k)$, and SAGA reduces to deterministic gradient descent. The bound (4) simplifies to $\E\|\grad f(\hat x_T)\|^{2}\le 2(f(x_0)-f^{\star})/(\alpha T)$, i.e. the classic $O(1/T)$ rate for smooth non‑convex optimization.

\item \textbf{Convex $f_i$.} When each $f_i$ is convex, the same proof gives
\[
\E\!\bigl[f(\hat x_T)-f^{\star}\bigr]\le
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;
\frac{4L\alpha\,\sigma^{2}}{n},
\]
recovering the known $O(1/T)$ function‑value guarantee for SAGA in the convex regime.

\item \textbf{Large‑scale limit $n\to\infty$.} The bias term scales as $1/n$, so for massive datasets the variance‑reduction effect dominates and the method behaves like full‑batch gradient descent with step size $\alpha$.
\end{enumerate}
