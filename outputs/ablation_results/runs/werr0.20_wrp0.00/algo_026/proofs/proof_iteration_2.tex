\section*{Algorithm Name}
\textbf{Stochastic Coordinate Descent with Importance Sampling (SCD-IS)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be convex and coordinate‑wise $L_i$‑smooth, i.e.
\[
\big|\partial_i f(x+\alpha e_i)-\partial_i f(x)\big|\le L_i|\alpha|,\qquad\forall x\in\R^{d},\;\forall\alpha\in\R,\;i\in\{1,\dots,d\}.
\]
A probability vector $p=(p_1,\dots,p_d)$ with $p_i>0$ and $\sum_i p_i=1$ is fixed before the run.
At iteration $t$ we draw a coordinate $i_t\sim p$ and define the stochastic gradient
\[
g_t \;:=\; \frac{1}{p_{i_t}}\;\partial_{i_t} f(w_t)\;e_{i_t},
\tag{1}
\]
where $e_i$ denotes the $i$‑th canonical basis vector.  
The update reads
\[
w_{t+1}=w_t-\eta_t\,g_t
\;=\;
w_t-\eta_t\,\frac{1}{p_{i_t}}\partial_{i_t}f(w_t)\,e_{i_t},
\tag{2}
\]
with a stepsize schedule $\{\eta_t\}_{t\ge0}$ (constant or diminishing).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Coordinate Descent with Importance Sampling}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, stepsizes $\{\eta_t\}$, probability vector $p$.
\For{$t=0,1,\dots,T-1$}
    \State Sample $i_t\in\{1,\dots,d\}$ with $\Pr(i_t=i)=p_i$.
    \State Compute partial derivative $g_{t,i_t}= \partial_{i_t}f(w_t)$.
    \State Form importance‑scaled gradient 
          $g_t=\dfrac{1}{p_{i_t}}\,g_{t,i_t}\,e_{i_t}$ \Comment{see (1)}.
    \State Update $w_{t+1}=w_t-\eta_t g_t$ \Comment{see (2)}.
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\tfrac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^2$, thus $d=1$, $L_1=2$, $p_1=1$.
Take $w_0=0$, constant stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t$ (partial derivative) & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\ \hline
0 & $\partial f(w_0)=2(w_0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\

\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the method on a simple problem.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Convexity.} $f$ is convex:
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x},\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\item \textbf{Coordinate‑wise smoothness.} For each $i$ there exists $L_i>0$ such that
\[
\big|\partial_i f(x+\alpha e_i)-\partial_i f(x)\big|\le L_i|\alpha|,\qquad\forall x\in\R^{d},\;\forall\alpha\in\R.
\tag{A2}
\]
\item \textbf{Probability vector.} $p_i>0$, $\sum_i p_i=1$.
\tag{A3}
\item \textbf{Stepsize.} The stepsize is constant $\eta_t\equiv\eta$ with
\[
0<\eta\le\frac{p_{\min}}{\displaystyle\max_i L_i},
\qquad p_{\min}:=\min_i p_i.
\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality rate]\label{thm:main}
Under Assumptions A1–A4, let $w^*$ be a minimizer of $f$. For the iterate
$\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$ produced by the algorithm,
\[
\E\!\big[f(\bar w_T)-f(w^*)\big]\;\le\;
\frac{\norm{w_0-w^*}^2}{2\eta T}.
\tag{3}
\]
Consequently the method attains an $O(1/T)$ convergence rate in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Unbiasedness.} By construction of $g_t$,
\[
\E[g_t\mid w_t]=\sum_{i=1}^dp_i\frac{1}{p_i}\partial_i f(w_t)e_i=\nabla f(w_t).
\tag{P1}
\]
\item \textbf{Variance bound.} Using the smoothness constants,
\[
\E\!\big[\|g_t\|^2\mid w_t\big]\le \sum_{i=1}^d\frac{1}{p_i}\big(\partial_i f(w_t)\big)^2
\le \frac{1}{p_{\min}}\|\nabla f(w_t)\|^2,
\tag{P2}
\]
where $p_{\min}:=\min_i p_i$.
\item \textbf{Comparison.} If $p_i\propto L_i$, the bound (3) matches the optimal rate of uniform coordinate descent up to a constant factor, while allowing cheaper updates for inexpensive coordinates.
\end{itemize}

\section*{Discussion}
The theorem shows that importance sampling does not deteriorate the $O(1/T)$ rate provided the stepsize respects the largest smoothness constant scaled by $p_{\min}$. The factor $1/p_i$ in the stochastic gradient exactly compensates the non‑uniform sampling, guaranteeing unbiasedness (P1). The analysis follows the classical proof for SGD, specialized to a gradient that is sparse and coordinate‑wise.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for convex functions]\label{lem:three}
For any $x,y\in\R^{d}$ and any subgradient $g\in\partial f(x)$,
\[
f(y)\ge f(x)+\ip{g}{y-x}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from convexity (A1). \hfill $\square$
\end{proof}

\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For each coordinate $i$ and any $x\in\R^{d}$, $\alpha\in\R$,
\[
f(x+\alpha e_i)\le f(x)+\alpha\partial_i f(x)+\frac{L_i}{2}\alpha^2.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Integrate the Lipschitz bound (A2) on the partial derivative. \hfill $\square$
\end{proof}

\begin{lemma}[Coordinate‑gradient bound]\label{lem:gradbound}
For a convex $L_i$‑smooth function $f$ and any $x\in\R^{d}$,
\[
\big(\partial_i f(x)\big)^2\;\le\;2L_i\big(f(x)-f(w^*)\big),\qquad i=1,\dots,d,
\]
where $w^*$ is a global minimizer of $f$.
\end{lemma}
\begin{proof}
Fix $i$ and consider the one‑dimensional function $\phi(\alpha)=f(x+\alpha e_i)$.  
By Lemma~\ref{lem:smooth} with $\alpha=-\frac{1}{L_i}\partial_i f(x)$ we obtain
\[
f\!\Big(x-\frac{1}{L_i}\partial_i f(x)e_i\Big)
\le f(x)-\frac{1}{L_i}\big(\partial_i f(x)\big)^2
      +\frac{L_i}{2}\Big(\frac{1}{L_i}\partial_i f(x)\Big)^2
= f(x)-\frac{1}{2L_i}\big(\partial_i f(x)\big)^2 .
\]
Since $w^*$ minimizes $f$, convexity implies
\[
f(w^*)\;\ge\;f\!\Big(x-\frac{1}{L_i}\partial_i f(x)e_i\Big).
\]
Combining the two inequalities yields
\[
f(w^*)\ge f(x)-\frac{1}{2L_i}\big(\partial_i f(x)\big)^2,
\]
which is equivalent to the claimed bound. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken with respect to the random draws
$\{i_t\}_{t\ge0}$.

\noindent\textbf{Step 1.} Expand the squared distance after one update.
\[
\begin{aligned}
\norm{w_{t+1}-w^*}^2
&=\norm{w_t-\eta g_t-w^*}^2 \\
&=\norm{w_t-w^*}^2-2\eta\ip{g_t}{w_t-w^*}+\eta^2\norm{g_t}^2 .
\end{aligned}
\tag{S1}
\]

\noindent\textbf{Step 2.} Take conditional expectation w.r.t.\ $i_t$ (i.e., given $w_t$) and use unbiasedness (P1).
\[
\begin{aligned}
\E\!\big[\norm{w_{t+1}-w^*}^2\mid w_t\big]
&=\norm{w_t-w^*}^2-2\eta\ip{\E[g_t\mid w_t]}{w_t-w^*}
      +\eta^2\E\!\big[\norm{g_t}^2\mid w_t\big] \\
&=\norm{w_t-w^*}^2-2\eta\ip{\nabla f(w_t)}{w_t-w^*}
      +\eta^2\E\!\big[\norm{g_t}^2\mid w_t\big].
\end{aligned}
\tag{S2}
\]

\noindent\textbf{Step 3.} Apply Lemma~\ref{lem:three} (convexity) to lower‑bound the inner product.
\[
\ip{\nabla f(w_t)}{w_t-w^*}\;\ge\; f(w_t)-f(w^*).
\tag{S3}
\]

\noindent\textbf{Step 4.} Insert (S3) into (S2):
\[
\E\!\big[\norm{w_{t+1}-w^*}^2\mid w_t\big]
\le \norm{w_t-w^*}^2-2\eta\big(f(w_t)-f(w^*)\big)
      +\eta^2\E\!\big[\norm{g_t}^2\mid w_t\big].
\tag{S4}
\]

\noindent\textbf{Step 5.} Bound the second‑moment term using Lemma~\ref{lem:gradbound}.
For each $i$,
\[
\big(\partial_i f(w_t)\big)^2\le 2L_i\big(f(w_t)-f(w^*)\big)
\quad\text{(Lemma~\ref{lem:gradbound})}.
\tag{S5}
\]
Summing the weighted squares and using $p_i\ge p_{\min}$ gives
\[
\begin{aligned}
\E\!\big[\norm{g_t}^2\mid w_t\big]
&= \sum_{i=1}^d\frac{1}{p_i}\big(\partial_i f(w_t)\big)^2
   \le \sum_{i=1}^d\frac{1}{p_{\min}}\,2L_i\big(f(w_t)-f(w^*)\big) \\
&= \frac{2\bar L}{p_{\min}}\big(f(w_t)-f(w^*)\big),
\end{aligned}
\tag{S6}
\]
where $\displaystyle\bar L:=\max_i L_i$.

\noindent\textbf{Step 6.} Plug (S6) into (S4):
\[
\E\!\big[\norm{w_{t+1}-w^*}^2\mid w_t\big]
\le \norm{w_t-w^*}^2
   -2\eta\big(f(w_t)-f(w^*)\big)
   +2\eta^2\frac{\bar L}{p_{\min}}\big(f(w_t)-f(w^*)\big).
\tag{S7}
\]

\noindent\textbf{Step 7.} Choose a stepsize satisfying (A4), i.e. $\displaystyle\eta\le\frac{p_{\min}}{\bar L}$.
Under this condition,
\[
2\eta^2\frac{\bar L}{p_{\min}}\;\le\;2\eta,
\tag{S8}
\]
so the last two terms in (S7) combine to a non‑positive quantity. Hence
\[
\E\!\big[\norm{w_{t+1}-w^*}^2\mid w_t\big]
\le \norm{w_t-w^*}^2-2\eta\big(f(w_t)-f(w^*)\big).
\tag{S9}
\]

\noindent\textbf{Step 8.} Take total expectation and rearrange:
\[
2\eta\,\E\!\big[f(w_t)-f(w^*)\big]\le
\E\!\big[\norm{w_t-w^*}^2\big]-\E\!\big[\norm{w_{t+1}-w^*}^2\big].
\tag{S10}
\]

\noindent\textbf{Step 9.} Sum (S10) over $t=0,\dots,T-1$ (telescoping sum):
\[
2\eta\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^*)\big]
\le \norm{w_0-w^*}^2-\E\!\big[\norm{w_T-w^*}^2\big]
\le \norm{w_0-w^*}^2.
\tag{S11}
\]

\noindent\textbf{Step 10.} Divide by $2\eta T$ and use convexity of $f$ to move the average inside:
\[
\E\!\big[f(\bar w_T)-f(w^*)\big]
\le\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^*)\big]
\le\frac{\norm{w_0-w^*}^2}{2\eta T}.
\tag{S12}
\]
This is exactly the bound (3) claimed in Theorem~\ref{thm:main}. \hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From (S12) we read off the explicit constant:
\[
C:=\frac{\norm{w_0-w^*}^2}{2\eta},
\qquad
\E\!\big[f(\bar w_T)-f(w^*)\big]\le \frac{C}{T}.
\]
If we set the maximal admissible stepsize $\eta = p_{\min}/\bar L$, then
$C = \frac{\bar L}{2p_{\min}}\norm{w_0-w^*}^2$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$. The importance factor $1/p_i=d$ recovers the standard coordinate descent estimator.
\item \textbf{Optimal importance sampling.} Choosing $p_i\propto L_i$ minimizes the variance bound (P2) and leads to the smallest possible constant $C$.
\item \textbf{One‑dimensional case.} The algorithm reduces to classical (stochastic) gradient descent with the usual $O(1/T)$ rate.
\end{itemize}
