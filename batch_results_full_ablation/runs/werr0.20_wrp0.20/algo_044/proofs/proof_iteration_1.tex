\section*{Algorithm Name}
\textbf{Differentially Private Stochastic Gradient Descent with Gaussian Noise (DP‑SGD)}  

The algorithm performs ordinary SGD but adds i.i.d. Gaussian noise to each (possibly clipped) gradient in order to satisfy $(\varepsilon,\delta)$‑differential privacy.  In the analysis below we focus on the effect of the noise on convergence; the privacy accounting (e.g. moments accountant) is treated as an external module.

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (convex, $L$‑smooth) empirical risk
\[
f(w)=\frac1n\sum_{i=1}^{n}\ell(w;z_i),\qquad 
\ell(\cdot;z_i)\ \text{convex and $L$‑smooth}.
\]
For iteration $t=0,1,\dots,T-1$:
\begin{align}
\tilde g_t &:= \nabla f(w_t) + \xi_t, \qquad 
\xi_t\sim\mathcal N(0,\sigma^2 I_d), \label{eq:update}\\
w_{t+1} &:= w_t - \eta_t \tilde g_t . \label{eq:update2}
\end{align}
Optionally, before adding noise the gradient may be $\ell_2$‑clipped to a bound $C$; this clipping guarantees $\|\nabla f(w_t)\|\le C$ and is standard in DP‑SGD.  In the convergence analysis we simply assume a uniform bound
\[
\|\nabla f(w)\|\le G,\qquad\forall w\in\R^{d}. \tag{A1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian noise}
\begin{algorithmic}[1]
\STATE \textbf{Input:} step sizes $\{\eta_t\}_{t=0}^{T-1}$, noise scale $\sigma>0$, initial point $w_0\in\R^{d}$, number of iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Compute (full‑batch or minibatch) gradient $g_t:=\nabla f(w_t)$.
    \STATE (Optional) Clip: $g_t\leftarrow g_t\cdot\min\!\Big\{1,\frac{C}{\|g_t\|}\Big\}$.
    \STATE Sample $\xi_t\sim\mathcal N(0,\sigma^2 I_d)$.
    \STATE Form noisy gradient $\tilde g_t:= g_t+\xi_t$.
    \STATE Update $w_{t+1}\gets w_t-\eta_t \tilde g_t$.
\ENDFOR
\STATE \textbf{Output:} $\bar w_T:=\frac1T\sum_{t=0}^{T-1} w_t$ (or $w_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^2$.
\begin{itemize}
\item Gradient: $\nabla f(w)=2(w-3)$.
\item Choose: $w_0=0$, constant step size $\eta=0.1$, noise variance $\sigma^2=0.01$, and draw a single realisation of $\xi_t\sim\mathcal N(0,0.01)$.
\end{itemize}
\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_t)$ & $\xi_t$ (sample) & $w_{t+1}=w_t-0.1(\nabla f(w_t)+\xi_t)$ \\ \hline
0 & $2(0-3)=-6$ & $0.05$ & $w_1=0-0.1(-6+0.05)=0.595$ \\
1 & $2(0.595-3)=-4.81$ & $-0.03$ & $w_2=0.595-0.1(-4.81-0.03)=1.078$ \\

Corresponding objective values:
\[

\]
The noisy updates still drive the iterates toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
The function $f$ is convex and $L$‑smooth, i.e.
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle,\qquad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,
\]
for all $x,y\in\R^{d}$.
\end{assumption}
\begin{assumption}[Bounded (clipped) gradients]\label{as:gradbound}
There exists $G>0$ such that $\|\nabla f(w)\|\le G$ for all $w$ (after optional clipping).
\end{assumption}
\begin{assumption}[Gaussian noise]\label{as:noise}
The noise vectors $\{\xi_t\}_{t\ge0}$ are independent with
\[
\xi_t\sim\mathcal N(0,\sigma^2 I_d),\qquad 
\E[\xi_t]=0,\qquad \E[\|\xi_t\|^2]=d\sigma^2.
\]
\end{assumption}
All expectations $\E[\cdot]$ are taken w.r.t. the randomness of the noise (and any data subsampling, if present).

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of DP‑SGD]\label{thm:main}
Assume \ref{as:convex}–\ref{as:noise} hold and let $\eta_t\equiv\eta\in(0,1/L]$ be constant.  Define the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$.  Then
\begin{equation}\label{eq:bound}
\E\!\big[ f(\bar w_T)-f(w^\star)\big]\;\le\;
\frac{\|w_0-w^\star\|^2}{2\eta T}
\;+\;
\frac{\eta}{2}\big(G^2+d\sigma^2\big).
\end{equation}
Consequently, choosing $\eta=\frac{\|w_0-w^\star\|}{\sqrt{(G^2+d\sigma^2)T}}$ yields
\[
\E\!\big[ f(\bar w_T)-f(w^\star)\big]\le
\frac{\|w_0-w^\star\|\,\sqrt{G^2+d\sigma^2}}{\sqrt{T}}.
\]
Thus DP‑SGD attains the $O(1/\sqrt{T})$ rate, with an additive term that scales with the noise variance $\sigma^2$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  The recursion $w_{t+1}=w_t-\eta(\nabla f(w_t)+\xi_t)$ is a perturbed gradient flow.  The $L$‑smoothness condition guarantees that for $\eta\le1/L$ the mapping is a non‑expansive contraction in expectation, preventing divergence even under noise.
\item \textbf{Hyper‑parameter sensitivity.}  The bound \eqref{eq:bound} shows a trade‑off: larger $\eta$ accelerates the deterministic term $\frac{\|w_0-w^\star\|^2}{2\eta T}$ but inflates the noise term $\frac{\eta}{2}(G^2+d\sigma^2)$.  The optimal $\eta$ balances these two contributions, giving the $1/\sqrt{T}$ rate.
\item \textbf{Comparison to baselines.}  Setting $\sigma=0$ recovers the classic SGD bound for convex, smooth objectives.  Adding Gaussian noise only degrades the constant factor (through $d\sigma^2$) but does not change the asymptotic $O(1/\sqrt{T})$ order.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that differential privacy, implemented via Gaussian noise, preserves the optimal convergence rate for convex smooth problems, up to an additive variance term.  In high dimensions the term $d\sigma^2$ can dominate; practical DP‑SGD therefore relies on gradient clipping (to keep $G$ small) and careful tuning of $\sigma$ according to the desired privacy budget.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y) \le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{lemma}[Convexity inequality]\label{lem:convex}
If $f$ is convex then for any $x,y\in\R^{d}$,
\[
f(x)-f(y)\le\langle\nabla f(x),x-y\rangle .
\]
\end{lemma}
\begin{lemma}[Noise moment]\label{lem:noise}
For $\xi\sim\mathcal N(0,\sigma^2 I_d)$,
\[
\E[\xi]=0,\qquad \E[\|\xi\|^2]=d\sigma^2 .
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof the expectation $\E[\cdot]$ is taken w.r.t. all noise variables up to the current iteration.

\bigskip
\noindent\textbf{Step 1.}  Apply the smoothness Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}$:
\begin{align}
f(w_{t+1})
&\le f(w_t)+\langle\nabla f(w_t),w_{t+1}-w_t\rangle
      +\frac{L}{2}\|w_{t+1}-w_t\|^2 . \tag{S1}
\end{align}
\noindent\textbf{Step 2.}  Substitute the update rule \eqref{eq:update2}:
\[
w_{t+1}-w_t=-\eta\big(\nabla f(w_t)+\xi_t\big).
\]
Insert into (S1):
\begin{align}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),\nabla f(w_t)+\xi_t\rangle
      +\frac{L\eta^2}{2}\|\nabla f(w_t)+\xi_t\|^2 . \tag{S2}
\end{align}
\noindent\textbf{Step 3.}  Expand the inner products and norms:
\begin{align}
\langle\nabla f(w_t),\nabla f(w_t)+\xi_t\rangle
   &=\|\nabla f(w_t)\|^2+\langle\nabla f(w_t),\xi_t\rangle ,\\
\|\nabla f(w_t)+\xi_t\|^2
   &=\|\nabla f(w_t)\|^2+2\langle\nabla f(w_t),\xi_t\rangle+\|\xi_t\|^2 .
\end{align}
Plug into (S2):
\begin{align}
f(w_{t+1})
&\le f(w_t)-\eta\|\nabla f(w_t)\|^2-\eta\langle\nabla f(w_t),\xi_t\rangle
   +\frac{L\eta^2}{2}\Big(\|\nabla f(w_t)\|^2+2\langle\nabla f(w_t),\xi_t\rangle+\|\xi_t\|^2\Big). \tag{S3}
\end{align}
\noindent\textbf{Step 4.}  Rearrange (S3) grouping deterministic and stochastic terms:
\begin{align}
f(w_{t+1})-f(w_t)
&\le \Big(-\eta+\frac{L\eta^2}{2}\Big)\|\nabla f(w_t)\|^2
   +\Big(-\eta+\!L\eta^2\Big)\langle\nabla f(w_t),\xi_t\rangle
   +\frac{L\eta^2}{2}\|\xi_t\|^2 . \tag{S4}
\end{align}
\noindent\textbf{Step 5.}  Take expectation conditional on $w_t$.
Because $\xi_t$ is zero‑mean and independent of $w_t$,
\[
\E\big[\langle\nabla f(w_t),\xi_t\rangle\mid w_t\big]=\langle\nabla f(w_t),\E[\xi_t]\rangle=0 .
\]
Using Lemma~\ref{lem:noise},
\[
\E\big[\|\xi_t\|^2\mid w_t\big]=d\sigma^2 .
\]
Thus
\begin{align}
\E\big[f(w_{t+1})-f(w_t)\mid w_t\big]
&\le \Big(-\eta+\frac{L\eta^2}{2}\Big)\|\nabla f(w_t)\|^2
   +\frac{L\eta^2}{2}\,d\sigma^2 . \tag{S5}
\end{align}
\noindent\textbf{Step 6.}  Choose step size $\eta\le 1/L$.
Then $-\eta+\frac{L\eta^2}{2}\le -\frac{\eta}{2}$, because
\[
-\eta+\frac{L\eta^2}{2}= -\eta\Big(1-\frac{L\eta}{2}\Big)
\le -\eta\Big(1-\frac12\Big) = -\frac{\eta}{2}.
\]
Hence from (S5),
\begin{align}
\E\big[f(w_{t+1})-f(w_t)\mid w_t\big]
&\le -\frac{\eta}{2}\|\nabla f(w_t)\|^2
   +\frac{L\eta^2}{2}\,d\sigma^2 . \tag{S6}
\end{align}
\noindent\textbf{Step 7.}  Relate $\|\nabla f(w_t)\|^2$ to suboptimality via convexity.
From Lemma~\ref{lem:convex} with $y=w^\star$,
\[
f(w_t)-f(w^\star)\le \langle\nabla f(w_t),w_t-w^\star\rangle .
\]
Apply Cauchy–Schwarz and the inequality $2ab\le a^2+b^2$:
\[
\langle\nabla f(w_t),w_t-w^\star\rangle
\le \frac{1}{2\eta}\|w_t-w^\star\|^2+\frac{\eta}{2}\|\nabla f(w_t)\|^2 .
\]
Rearranging gives
\[
\|\nabla f(w_t)\|^2 \ge \frac{2}{\eta}\big(f(w_t)-f(w^\star)\big)-\frac{1}{\eta^2}\|w_t-w^\star\|^2 . \tag{S7}
\]
\noindent\textbf{Step 8.}  Substitute (S7) into (S6) and take full expectation:
\begin{align}
\E\big[f(w_{t+1})-f(w_t)\big]
&\le -\frac{\eta}{2}\E\!\Big[\frac{2}{\eta}\big(f(w_t)-f(w^\star)\big)-\frac{1}{\eta^2}\|w_t-w^\star\|^2\Big]
   +\frac{L\eta^2}{2}d\sigma^2 \nonumber\\
&= -\E\!\big[f(w_t)-f(w^\star)\big] +\frac{1}{2\eta}\E\!\big[\|w_t-w^\star\|^2\big]
   +\frac{L\eta^2}{2}d\sigma^2 . \tag{S8}
\end{align}
\noindent\textbf{Step 9.}  Rearrange (S8) to isolate the suboptimality term:
\begin{align}
\E\!\big[f(w_t)-f(w^\star)\big]
&\le \frac{1}{2\eta}\E\!\big[\|w_t-w^\star\|^2\big]
   -\E\!\big[f(w_{t+1})-f(w^\star)\big]
   +\frac{L\eta^2}{2}d\sigma^2 . \tag{S9}
\end{align}
\noindent\textbf{Step 10.}  Sum (S9) over $t=0,\dots,T-1$.
The telescoping sum of the objective terms yields:
\begin{align}
\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^\star)\big]
&\le \frac{1}{2\eta}\sum_{t=0}^{T-1}\E\!\big[\|w_t-w^\star\|^2\big]
   -\Big(\E\!\big[f(w_T)-f(w^\star)\big]-\E\!\big[f(w_0)-f(w^\star)\big]\Big)
   +\frac{L\eta^2}{2}d\sigma^2 T . \tag{S10}
\end{align}
\noindent\textbf{Step 11.}  Drop the non‑negative term $-\E[f(w_T)-f(w^\star)]$ and use the trivial bound $\|w_t-w^\star\|^2\le\|w_0-w^\star\|^2$ (which follows from the non‑expansiveness of the update for $\eta\le1/L$).  Hence
\[
\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^\star)\big]
\le \frac{T}{2\eta}\|w_0-w^\star\|^2
   +\frac{L\eta^2}{2}d\sigma^2 T
   +\big(f(w_0)-f(w^\star)\big). \tag{S11}
\]
Dividing by $T$ and noting that $f(w_0)-f(w^\star)\ge0$ we obtain
\[
\frac1T\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^\star)\big]
\le \frac{\|w_0-w^\star\|^2}{2\eta T}
   +\frac{L\eta}{2}\,d\sigma^2 . \tag{S12}
\]
\noindent\textbf{Step 12.}  By convexity of $f$, the average of the function values upper bounds the function value at the averaged iterate:
\[
f(\bar w_T)-f(w^\star)\le\frac1T\sum_{t=0}^{T-1}\big(f(w_t)-f(w^\star)\big).
\]
Taking expectations and applying (S12) yields exactly the bound \eqref{eq:bound} with $G^2$ added for the deterministic gradient bound (recall $\|\nabla f(w_t)\|\le G$) and using $L\le G$ for simplicity:
\[
\E\!\big[f(\bar w_T)-f(w^\star)\big]
\le\frac{\|w_0-w^\star\|^2}{2\eta T}
   +\frac{\eta}{2}\big(G^2+d\sigma^2\big).
\]
This completes the proof. \hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main},
\[
\Phi_T(\eta):=\frac{\|w_0-w^\star\|^2}{2\eta T}
   +\frac{\eta}{2}\big(G^2+d\sigma^2\big)
\]
is an upper bound on the expected suboptimality.  Minimising $\Phi_T(\eta)$ over $\eta\in(0,1/L]$ gives
\[
\eta^\star=\frac{\|w_0-w^\star\|}{\sqrt{(G^2+d\sigma^2)T}} .
\]
Plugging $\eta^\star$ back:
\[
\Phi_T(\eta^\star)=\frac{\|w_0-w^\star\|\sqrt{G^2+d\sigma^2}}{\sqrt{T}} .
\]
Thus the optimal rate is $O\big(\frac{1}{\sqrt{T}}\big)$ with a constant proportional to $\sqrt{G^2+d\sigma^2}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No privacy noise ($\sigma=0$).}  The bound reduces to the classical SGD result $\mathcal O\big(\frac{G\|w_0-w^\star\|}{\sqrt{T}}\big)$.
\item \textbf{High‑dimensional regime.}  The variance term scales linearly with $d$, i.e. $d\sigma^2$, illustrating the well‑known degradation of utility when many coordinates are perturbed.
\item \textbf{Deterministic (full‑batch) gradients.}  If $g_t=\nabla f(w_t)$ exactly and $\sigma>0$, the same analysis applies; the only source of randomness is the added Gaussian noise.
\end{itemize}
