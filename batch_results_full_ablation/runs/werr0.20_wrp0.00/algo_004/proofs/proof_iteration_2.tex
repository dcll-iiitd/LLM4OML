\section*{Algorithm Name}
\textbf{Relative (Mirror) Gradient Descent (RGD)}  

The method is a Bregman‑proximal gradient step where the objective $f$ is assumed to be \emph{smooth relative to} a strongly convex distance‑generating function $h$.

\section*{Mathematical Formulation}
Let $h:\mathbb R^{d}\to\mathbb R$ be differentiable and $\mu$‑strongly convex w.r.t. a norm $\|\cdot\|$, i.e.  
\[
h(y)\ge h(x)+\langle \grad h(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2},
\qquad\forall x,y.
\]
Its Bregman divergence is  
\[
\Breg (y,x)=h(y)-h(x)-\langle \grad h(x),y-x\rangle .
\]

Given a stepsize $\eta>0$, the iteration reads
\begin{equation}
\boxed{
x_{k+1}= \argmin_{x\in\mathbb R^{d}}
\Big\{ \langle \grad f(x_k),\,x\rangle+\frac{1}{\eta}\,\Breg (x,x_k) \Big\}
}\tag{1}
\end{equation}
The optimality condition of (1) is
\begin{equation}
\grad h(x_{k+1}) = \grad h(x_k)-\eta \grad f(x_k). \tag{2}
\end{equation}
Define the \emph{gradient mapping} (relative to $h$) as
\begin{equation}
G_{\eta}(x_k):=\frac{1}{\eta}\bigl(x_k-\prox_{\eta h}\bigl(x_k-\eta\grad f(x_k)\bigr)\bigr)
               =\frac{1}{\eta}\bigl(x_k-x_{k+1}\bigr). \tag{3}
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Relative Gradient Descent (RGD)}\label{alg:RGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\mathbb R^{d}$, stepsize $\eta\in(0,1/L]$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
\STATE Compute gradient $g_k=\grad f(x_k)$.
\STATE $x_{k+1}\gets\argmin_{x}\Big\{\langle g_k,x\rangle+\frac{1}{\eta}\Breg(x,x_k)\Big\}$.
\IF{termination criterion satisfied (e.g. $\|G_{\eta}(x_k)\|\le\varepsilon$)} 
\STATE \textbf{break}
\ENDIF
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or the best iterate).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Relative Smoothness]\label{ass:smooth}
There exists $L>0$ such that for all $x,y\in\mathbb R^{d}$
\[
f(y)\le f(x)+\langle \grad f(x),y-x\rangle+L\,\Breg (y,x).
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity of the Prox‑function]\label{ass:strong}
The distance‑generating function $h$ is $\mu$‑strongly convex ($\mu>0$) w.r.t. $\|\cdot\|$, i.e.
\[
\Breg (y,x)\ge\frac{\mu}{2}\|y-x\|^{2},\qquad\forall x,y.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Bounded Below]\label{ass:bound}
The objective $f$ is bounded below: $f^\star:=\inf_{x}f(x)>-\infty$.
\tag{A3}
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies $0<\eta\le \frac{1}{L}$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of RGD for Non‑Convex $f$]\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}, let $\{x_k\}$ be generated by \eqref{alg:RGD} with a \emph{constant} stepsize $\eta\in(0,1/L]$. Define the gradient mapping \eqref{3}. Then for any integer $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\norm{G_{\eta}(x_k)}^{2}
\;\le\;
\frac{2L}{\mu}\,\frac{f(x_0)-f^\star}{K}.
\tag{4}
\]
Consequently,
\[
\min_{0\le k\le K-1}\norm{G_{\eta}(x_k)}^{2}
\;\le\;
\frac{2L}{\mu}\,\frac{f(x_0)-f^\star}{K}.
\]
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity for Bregman Divergence]\label{lem:3pt}
For any $x,y,z$,
\[
\Breg (x,z)-\Breg (y,z)-\Breg (x,y)=\langle \grad h(x)-\grad h(y),\,z-y\rangle .
\tag{5}
\]
\end{lemma}
\begin{proof}
By definition,
\[
\Breg (x,z)=h(x)-h(z)-\langle \grad h(z),x-z\rangle,
\]
\[
\Breg (y,z)=h(y)-h(z)-\langle \grad h(z),y-z\rangle,
\]
\[
\Breg (x,y)=h(x)-h(y)-\langle \grad h(y),x-y\rangle .
\]
Subtracting the second and third from the first and rearranging yields \eqref{5}. ∎
\end{proof}

\begin{lemma}[Strong convexity $\Rightarrow$ lower bound on Bregman]\label{lem:strong}
If $h$ satisfies Assumption~\ref{ass:strong}, then for all $x,y$
\[
\Breg (y,x)\ge\frac{\mu}{2}\|y-x\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Directly from the definition of $\mu$‑strong convexity of $h$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All steps retain the original numbering for easy reference.

\bigskip
\noindent\textbf{[Step 1]} \emph{Relative smoothness inequality.}  
From Assumption~\ref{ass:smooth} with $x=x_k$, $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_k)+\langle \grad f(x_k),x_{k+1}-x_k\rangle+L\,\Breg (x_{k+1},x_k).
\tag{7}
\]

\noindent\textbf{[Step 2]} \emph{Optimality condition of the subproblem.}  
The definition \eqref{1} yields the first‑order optimality condition \eqref{2}:
\[
\grad h(x_{k+1}) = \grad h(x_k)-\eta \grad f(x_k).
\tag{8}
\]

\noindent\textbf{[Step 3]} \emph{Express the linear term via Bregman divergences.}  
Take the inner product of \eqref{8} with $x_{k+1}-x_k$:
\[
\langle \grad h(x_{k+1})-\grad h(x_k),x_{k+1}-x_k\rangle
   = -\eta \langle \grad f(x_k),x_{k+1}-x_k\rangle .
\tag{9}
\]
Hence
\[
\langle \grad f(x_k),x_{k+1}-x_k\rangle
   = -\frac{1}{\eta}\langle \grad h(x_{k+1})-\grad h(x_k),x_{k+1}-x_k\rangle .
\tag{10}
\]

\noindent\textbf{[Step 4]} \emph{Apply the three‑point identity \eqref{5}.}  
Set $(x,y,z)=(x_{k+1},x_k,x_k)$ in \eqref{5}:
\[
\Breg (x_{k+1},x_k)-\Breg (x_k,x_k)-\Breg (x_{k+1},x_k)
   =\langle \grad h(x_{k+1})-\grad h(x_k),x_k-x_k\rangle=0 .
\]
A more useful choice is $(x,y,z)=(x_{k+1},x_k,x_k)$ which gives
\[
\langle \grad h(x_{k+1})-\grad h(x_k),x_{k+1}-x_k\rangle
   =\Breg (x_{k+1},x_k)+\Breg (x_k,x_{k+1}) .
\tag{11}
\]
% CORRECTED: the sign is positive; the previous version had a sign error.

\noindent\textbf{[Step 5]} \emph{Insert \eqref{11} into \eqref{10}.}  
Using \eqref{11} in \eqref{10} we obtain
\[
\langle \grad f(x_k),x_{k+1}-x_k\rangle
   = -\frac{1}{\eta}\bigl(\Breg (x_{k+1},x_k)+\Breg (x_k,x_{k+1})\bigr).
\tag{12}
\]

\noindent\textbf{[Step 6]} \emph{Plug \eqref{12} into the smoothness inequality \eqref{7}.}  
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)
   -\frac{1}{\eta}\bigl(\Breg (x_{k+1},x_k)+\Breg (x_k,x_{k+1})\bigr)
   +L\,\Breg (x_{k+1},x_k)\\[1ex]
&= f(x_k)
   -\Bigl(\frac{1}{\eta}-L\Bigr)\Breg (x_{k+1},x_k)
   -\frac{1}{\eta}\Breg (x_k,x_{k+1}) .
\end{aligned}
\tag{13}
\]

\noindent\textbf{[Step 7]} \emph{Discard the non‑positive term.}  
Since $\eta\le 1/L$ (Assumption~\ref{ass:stepsize}), we have $\frac{1}{\eta}-L\ge0$, therefore the first Bregman term in \eqref{13} is non‑positive and can be dropped:
\[
f(x_{k+1})\le f(x_k)-\frac{1}{\eta}\Breg (x_k,x_{k+1}).
\tag{14}
\]

\noindent\textbf{[Step 8]} \emph{Use strong convexity of $h$.}  
By Lemma~\ref{lem:strong} (Assumption~\ref{ass:strong}),
\[
\Breg (x_k,x_{k+1})\ge\frac{\mu}{2}\|x_{k+1}-x_k\|^{2}.
\tag{15}
\]
Insert \eqref{15} into \eqref{14}:
\[
f(x_{k+1})\le f(x_k)-\frac{\mu}{2\eta}\|x_{k+1}-x_k\|^{2}.
\tag{16}
\]

\noindent\textbf{[Step 9]} \emph{Relate the distance to the gradient mapping.}  
From definition \eqref{3},
\[
\|G_{\eta}(x_k)\|^{2}
   =\frac{1}{\eta^{2}}\|x_k-x_{k+1}\|^{2}
   =\frac{1}{\eta^{2}}\|x_{k+1}-x_k\|^{2}.
\tag{17}
\]
Rearranging \eqref{16} and using \eqref{17} yields
\[
\|G_{\eta}(x_k)\|^{2}
   \le \frac{2}{\mu\eta}\bigl(f(x_k)-f(x_{k+1})\bigr).
\tag{18}
\]

\noindent\textbf{[Step 10]} \emph{Sum over $k=0,\dots,K-1$ and invoke boundedness below (A3).}  
Summing \eqref{18} gives
\[
\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|^{2}
   \le \frac{2}{\mu\eta}\sum_{k=0}^{K-1}\bigl(f(x_k)-f(x_{k+1})\bigr)
   = \frac{2}{\mu\eta}\bigl(f(x_0)-f(x_K)\bigr)
   \le \frac{2}{\mu\eta}\bigl(f(x_0)-f^\star\bigr),
\tag{19}
\]
where the last inequality uses Assumption~\ref{ass:bound} ($f^\star\le f(x_K)$).

\noindent\textbf{[Step 11]} \emph{Divide by $K$ and replace $\eta$ by its maximal admissible value $1/L$.}  
Because $\eta\le 1/L$, we have $1/\eta\le L$. Hence
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta}(x_k)\|^{2}
   \le \frac{2L}{\mu}\,\frac{f(x_0)-f^\star}{K}.
\tag{20}
\]
Inequality \eqref{20} is exactly the bound \eqref{4} stated in Theorem~\ref{thm:main}. The claim about the minimum follows from the fact that the average upper‑bounds the minimum of the non‑negative sequence.

\noindent\textbf{[Step 12]} \emph{Conclusion.}  
Thus the algorithm generates a sequence whose gradient‑mapping norm decays at the rate $O(1/K)$, completing the proof. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{4} is $C=\dfrac{2L}{\mu}\bigl(f(x_0)-f^\star\bigr)$.  
Choosing the largest admissible stepsize $\eta=1/L$ minimises this constant.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Euclidean setting.} $h(x)=\tfrac12\|x\|^{2}$ gives $\mu=1$, $\Breg (y,x)=\tfrac12\|y-x\|^{2}$, and $G_{\eta}(x)=\grad f(x)$. Theorem~\ref{thm:main} reduces to the classic $O(1/K)$ bound for gradient descent on $L$‑smooth non‑convex functions.
\item \textbf{KL‑divergence.} For $h(x)=\sum_{i}x_i\log x_i -x_i$, $\mu=1$ on the simplex and $\Breg$ becomes the Kullback–Leibler divergence. The theorem then applies to problems such as Poisson regression where ordinary Lipschitz smoothness fails but relative smoothness holds.
\item \textbf{Adaptive stepsizes.} If one uses a locally estimated $L_k$ and sets $\eta_k=1/L_k$, the same proof works provided $\eta_k\le 1/L$ for each $k$; the bound then involves $\max_k L_k$.
\end{enumerate}
