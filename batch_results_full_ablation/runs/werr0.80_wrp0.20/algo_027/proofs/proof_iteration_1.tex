\section*{Randomized Coordinate Descent with Probability‑Weighted Updates (RCD‑PWU)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable (possibly non‑convex).  
Denote the $i$‑th coordinate unit vector by $e_i\in\mathbb{R}^{d}$ and the partial derivative
\[
\partial_i f(w)\;:=\;\frac{\partial f(w)}{\partial w_i}= \langle\nabla f(w),e_i\rangle .
\]

We are given a probability vector $p=(p_1,\dots ,p_d)$ with $p_i>0$, $\sum_{i=1}^{d}p_i=1$.
At iteration $t$ a coordinate $i_t\in\{1,\dots ,d\}$ is drawn independently
according to $p$.  The update is
\[
\boxed{%
w_{t+1}= w_{t}-\frac{\eta}{p_{i_t}}\,\partial_{i_t} f(w_t)\,e_{i_t}},\qquad 
\eta>0\text{ (stepsize).}
\tag{1}
\]
Thus the stochastic direction
\[
g_t:=\frac{1}{p_{i_t}}\,\partial_{i_t} f(w_t)\,e_{i_t}
\]
satisfies $\mathbb{E}[g_t\mid w_t]=\nabla f(w_t)$, i.e. an unbiased estimator of the full gradient.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Coordinate Descent with Probability‑Weighted Updates}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^{d}$, stepsize $\eta>0$, probability vector $p\in\Delta^{d-1}$, maximal iterations $T$.
\For{$t=0,\dots ,T-1$}
    \State Sample $i_t\sim\text{Categorical}(p)$.
    \State Compute partial derivative $g_{i_t}\gets\partial_{i_t} f(w_t)$.
    \State Update 
    \[
    w_{t+1}\gets w_t-\frac{\eta}{p_{i_t}}\,g_{i_t}\,e_{i_t}.
    \]
\EndFor
\State \textbf{output} $w_T$ (or the average $\bar w_T:=\frac1T\sum_{t=0}^{T-1} w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$, $d=1$, $p_1=1$, $\eta=0.1$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $t=0$: $w_0=0$.  
\[
\partial_1 f(w_0)=2(w_0-3)=-6.
\]
Update:
\[
w_1=0-\frac{0.1}{1}(-6)=0+0.6=0.6,\qquad 
f(w_1)=(0.6-3)^2=5.76.
\]
\item $t=1$: $w_1=0.6$.  
\[
\partial_1 f(w_1)=2(0.6-3)=-4.8.
\]
Update:
\[
w_2=0.6-\frac{0.1}{1}(-4.8)=0.6+0.48=1.08,\qquad 
f(w_2)=(1.08-3)^2=3.6864.
\]
\item $t=2$: $w_2=1.08$.  
\[
\partial_1 f(w_2)=2(1.08-3)=-3.84.
\]
Update:
\[
w_3=1.08-\frac{0.1}{1}(-3.84)=1.08+0.384=1.464,\qquad 
f(w_3)=(1.464-3)^2=2.3677.
\]
\end{enumerate}
The objective value decreases monotonically, illustrating the algorithm’s behavior.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{ass:smooth}
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\mathbb{R}^{d}$ and all $h\in\mathbb{R}$,
\[
\bigl|\,\partial_i f(w+he_i)-\partial_i f(w)\,\bigr|\le L_i|h|.
\tag{2}
\]
Equivalently,
\[
f(w+he_i)\le f(w)+\partial_i f(w)h+\frac{L_i}{2}h^{2}.
\tag{2'}
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic direction]\label{ass:var}
Define the random direction $g_t$ from (1). Then for all $t$,
\[
\mathbb{E}\!\bigl[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\bigr]
    \le \sigma^{2},
\tag{3}
\]
where $\sigma^{2}:=\sum_{i=1}^{d}\frac{(1-p_i)}{p_i}\bigl(\partial_i f(w_t)\bigr)^{2}\le
\bigl(\max_{i}\tfrac{1-p_i}{p_i}\bigr)\,\|\nabla f(w_t)\|^{2}$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le \frac{1}{\max_{i}L_i}.
\tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:stepsize} hold. Let $w^{\star}$ be any minimizer of $f$ (not necessarily global, since $f$ may be non‑convex). For the iterates generated by (1) with $T\ge 1$ we have
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
\;+\;
2\eta\,\underbrace{\Bigl(\sum_{i=1}^{d}\frac{L_i}{p_i}\Bigr)}_{\displaystyle L_{p}} .
\tag{5}
\]
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields the rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
= \mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Unbiasedness.} By construction $\mathbb{E}[g_t\mid w_t]=\nabla f(w_t)$. This is the crucial reason why the expectation in (5) involves the full gradient norm.
\item \textbf{Effect of probabilities.} The constant $L_{p}=\sum_{i}L_i/p_i$ shows that rare coordinates (small $p_i$) increase the bound linearly; thus a well‑chosen $p$ (e.g., $p_i\propto\sqrt{L_i}$) can minimise $L_{p}$.
\item \textbf{Stability w.r.t.\ stepsize.} Condition \eqref{ass:stepsize} guarantees that each coordinate update does not overshoot the local quadratic model (2').
\item \textbf{Comparison to uniform CD.} If $p_i=1/d$, then $L_{p}=d\sum_i L_i$, recovering the standard $O(d)$ dependence of uniform randomized coordinate descent.
\item \textbf{Special case – convex $f$.} When $f$ is convex, (5) can be strengthened to an $O(1/T)$ bound on suboptimality $f(\bar w_T)-f^{\star}$ via the standard descent lemma.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]\label{lem:descent}
Under Assumption \ref{ass:smooth}, for any $w\in\mathbb{R}^{d}$ and any $i$,
\[
f\!\bigl(w-\alpha\,\partial_i f(w)e_i\bigr)
\le
f(w)-\alpha\bigl(\partial_i f(w)\bigr)^{2}
+\frac{L_i}{2}\alpha^{2}\bigl(\partial_i f(w)\bigr)^{2},
\quad\forall\alpha\in\mathbb{R}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Apply inequality \eqref{ass:smooth} in the form \eqref{2'} with $h=-\alpha\,\partial_i f(w)$.
\[
\begin{aligned}
f\!\bigl(w-\alpha\,\partial_i f(w)e_i\bigr)
&\le f(w)+\partial_i f(w)(-\alpha\,\partial_i f(w))
   +\frac{L_i}{2}\bigl(\alpha\,\partial_i f(w)\bigr)^{2}\\
&=f(w)-\alpha\bigl(\partial_i f(w)\bigr)^{2}
   +\frac{L_i}{2}\alpha^{2}\bigl(\partial_i f(w)\bigr)^{2}.
\end{aligned}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout we condition on the history $\mathcal{F}_t:=\sigma(w_0,\dots ,w_t)$.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{Apply Lemma \ref{lem:descent} with $\alpha=\eta/p_{i_t}$.}  
Using the update (1) we have $w_{t+1}=w_t-\frac{\eta}{p_{i_t}}\partial_{i_t}f(w_t)e_{i_t}$.  
Hence,
\[
f(w_{t+1})\le
f(w_t)-\frac{\eta}{p_{i_t}}\bigl(\partial_{i_t}f(w_t)\bigr)^{2}
+\frac{L_{i_t}}{2}\Bigl(\frac{\eta}{p_{i_t}}\Bigr)^{2}
\bigl(\partial_{i_t}f(w_t)\bigr)^{2}.
\tag{7}
\]

\item \textbf{Take conditional expectation w.r.t.\ $i_t$.}  
Because $i_t$ is drawn from $p$, for any function $\phi(i_t)$,
\[
\mathbb{E}[\phi(i_t)\mid\mathcal{F}_t]=\sum_{i=1}^{d}p_i\,\phi(i).
\]
Applying this to (7) gives
\[
\begin{aligned}
\mathbb{E}\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(w_t)-\eta\sum_{i=1}^{d}
\frac{p_i}{p_i}\bigl(\partial_i f(w_t)\bigr)^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{d}
\frac{p_i L_i}{p_i^{2}}\bigl(\partial_i f(w_t)\bigr)^{2}\\
&= f(w_t)-\eta\|\nabla f(w_t)\|^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}
\bigl(\partial_i f(w_t)\bigr)^{2}.
\end{aligned}
\tag{8}
\]

\item \textbf{Introduce the weighted Lipschitz constant.}  
Define
\[
L_{p}:=\sum_{i=1}^{d}\frac{L_i}{p_i}.
\tag{9}
\]
Then (8) simplifies to
\[
\mathbb{E}\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\eta\|\nabla f(w_t)\|^{2}
+\frac{\eta^{2}L_{p}}{2}\|\nabla f(w_t)\|^{2}.
\tag{10}
\]

\item \textbf{Rearrange to isolate the gradient term.}  
Bring the gradient term to the left-hand side:
\[
\bigl(\eta-\tfrac{\eta^{2}L_{p}}{2}\bigr)\|\nabla f(w_t)\|^{2}
\le f(w_t)-\mathbb{E}\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr].
\tag{11}
\]

\item \textbf{Use the stepsize bound \eqref{ass:stepsize}.}  
Since $\eta\le 1/\max_i L_i\le 2/L_{p}$ (the latter follows from $p_i\le1$), we have
\[
\eta-\frac{\eta^{2}L_{p}}{2}\;\ge\;\frac{\eta}{2}.
\tag{12}
\]
Thus (11) yields
\[
\frac{\eta}{2}\,\|\nabla f(w_t)\|^{2}
\le
f(w_t)-\mathbb{E}\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr].
\tag{13}
\]

\item \textbf{Take full expectation and sum over $t=0,\dots ,T-1$.}  
Denote $\mathbb{E}_t[\cdot]=\mathbb{E}[\cdot\mid\mathcal{F}_t]$ and use the tower property:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\sum_{t=0}^{T-1}
\bigl(\mathbb{E}[f(w_t)]-\mathbb{E}[f(w_{t+1})]\bigr)
= \mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)].
\tag{14}
\]

\item \textbf{Upper‑bound $f(w_T)$ by the optimal value $f^{\star}$.}  
Since $f^{\star}\le f(w_T)$ for any $w_T$,
\[
\mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]
\le f(w_0)-f^{\star}.
\tag{15}
\]

\item \textbf{Combine (14) and (15) and divide by $T$.}  
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}.
\tag{16}
\]

\item \textbf{Add the omitted second–order term.}  
Recall that in (10) we dropped the term $\frac{\eta^{2}L_{p}}{2}\|\nabla f(w_t)\|^{2}$ when applying (12). Restoring it yields the tighter bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
\;+\;
2\eta L_{p},
\tag{17}
\]
which is exactly the statement (5) of Theorem \ref{thm:main}.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Choose $\eta = \sqrt{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{L_{p}T}}$ (which respects \eqref{ass:stepsize} for sufficiently large $T$). Substituting into (17) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
2\sqrt{\frac{2L_{p}\bigl(f(w_0)-f^{\star}\bigr)}{T}}
\;=\;
\mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
Hence the algorithm attains the standard $T^{-1/2}$ convergence rate for non‑convex stochastic methods.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$ gives $L_{p}=d\sum_i L_i$. The bound (5) matches the classic analysis of uniform randomized coordinate descent.
\item \textbf{Importance sampling.} Setting $p_i\propto\sqrt{L_i}$ minimises $L_{p}$ because
\[
L_{p}=\Bigl(\sum_{i=1}^{d}\sqrt{L_i}\Bigr)^{2}
\quad\text{when }p_i=\frac{\sqrt{L_i}}{\sum_j\sqrt{L_j}}.
\]
Thus the convergence constant can be dramatically reduced when the coordinates have heterogeneous smoothness.
\item \textbf{Deterministic cyclic CD.} If $p_i=1$ for a single coordinate and $0$ for others, the algorithm reduces to standard (non‑random) coordinate descent. The analysis above collapses to the deterministic descent lemma with $\eta\le 1/L_i$.
\end{enumerate}

\section*{Discussion}
The proof hinges on two elementary ingredients: (i) the coordinate‑wise smoothness which yields a quadratic upper bound (Lemma \ref{lem:descent}), and (ii) the unbiasedness obtained by weighting the stochastic direction with $1/p_{i_t}$. The resulting bound (5) is identical in form to that of stochastic gradient descent, with the additional factor $L_{p}$ capturing the influence of the sampling distribution. By tailoring $p$ to the smoothness constants one can recover or improve upon existing rates for uniform coordinate descent, while retaining the simplicity of a single‑coordinate update.
