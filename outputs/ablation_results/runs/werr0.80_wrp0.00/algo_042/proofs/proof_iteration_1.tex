\section*{Algorithm Name}
\textbf{Extra‑Gradient Method for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let $L:\mathcal{X}\times\mathcal{Y}\rightarrow\R$ be a function that is convex in $x\in\mathcal{X}\subseteq\R^{n}$ and concave in $y\in\mathcal{Y}\subseteq\R^{m}$.  
Define the monotone operator
\[
F(z)\;:=\;
\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix},
\qquad 
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathcal{Z}:=\mathcal{X}\times\mathcal{Y}.
\]

Given a stepsize $\eta>0$, the Extra‑Gradient iteration reads
\[
\boxed{
\begin{aligned}
z_{k+\frac12} &= z_{k}-\eta\,F(z_{k}) ,\\[1mm]
z_{k+1}       &= z_{k}-\eta\,F\!\left(z_{k+\frac12}\right) .
\end{aligned}}
\tag{EG}
\]

Equivalently, in component form,
\[
\begin{aligned}
x_{k+\frac12}&=x_{k}-\eta\,\nabla_{x}L(x_{k},y_{k}),\\
y_{k+\frac12}&=y_{k}+\eta\,\nabla_{y}L(x_{k},y_{k}),\\[1mm]
x_{k+1}&=x_{k}-\eta\,\nabla_{x}L\!\left(x_{k+\frac12},y_{k+\frac12}\right),\\
y_{k+1}&=y_{k}+\eta\,\nabla_{y}L\!\left(x_{k+\frac12},y_{k+\frac12}\right).
\end{aligned}
\tag{1}
\]

The algorithm outputs the averaged iterate
\[
\bar z_{T}\;:=\;\frac{1}{T}\sum_{k=0}^{T-1}z_{k},
\qquad 
\bar x_{T},\bar y_{T}\text{ its components.}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra‑Gradient for $\displaystyle\min_{x\in\mathcal{X}}\max_{y\in\mathcal{Y}}L(x,y)$}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, number of iterations $T$, initial point $z_{0}=(x_{0},y_{0})$.
\FOR{$k=0,\dots,T-1$}
    \STATE Compute $F(z_{k})=\big(\nabla_{x}L(x_{k},y_{k}),\,-\nabla_{y}L(x_{k},y_{k})\big)$.
    \STATE $z_{k+\frac12}\gets z_{k}-\eta\,F(z_{k})$   \hfill\COMMENT{first gradient step}
    \STATE Compute $F(z_{k+\frac12})$.
    \STATE $z_{k+1}\gets z_{k}-\eta\,F(z_{k+\frac12})$ \hfill\COMMENT{correction step}
\ENDFOR
\STATE \textbf{Return} $\displaystyle\bar z_{T}=\frac1T\sum_{k=0}^{T-1}z_{k}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple (non‑saddle) quadratic $f(w)=(w-3)^{2}$, i.e. $L(x,y)=f(x)$ with no $y$ variable.  
The gradient is $\nabla f(w)=2(w-3)$.  
We apply (1) with stepsize $\eta=0.1$ and initialise $x_{0}=0$ (the $y$‑part is absent).

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.}&$x_{k}$&$x_{k+\frac12}=x_{k}-\eta\nabla f(x_{k})$&$x_{k+1}=x_{k}-\eta\nabla f(x_{k+\frac12})$\\ \hline
0&0&$0-0.1\cdot(-6)=0.6$&$0-0.1\cdot 2(0.6-3)=0.48$\\[2mm]
1&0.48&$0.48-0.1\cdot2(0.48-3)=0.624$&$0.48-0.1\cdot2(0.624-3)=0.5952$\\[2mm]
2&0.5952&$0.5952-0.1\cdot2(0.5952-3)=0.72096$&$0.5952-0.1\cdot2(0.72096-3)=0.686\,\! \approx0.686$\\
\end{tabular}
\end{center}

Corresponding objective values $f(x_{k})$:
\[
f(0)=9,\quad f(0.48)=6.3504,\quad f(0.5952)=5.792,\quad f(0.686)\approx5.436.
\]

The values decrease, illustrating the correction step’s effect even on a plain minimisation problem.

\section*{Assumptions}
\begin{assumption}[Convex–concave structure]\label{as:convexconcave}
$L(\cdot ,y)$ is convex for every $y\in\mathcal{Y}$ and $L(x,\cdot )$ is concave for every $x\in\mathcal{X}$.
\end{assumption}
\begin{assumption}[Smoothness]\label{as:smooth}
There exists a constant $L_{F}>0$ such that the operator $F$ defined above is $L_{F}$‑Lipschitz:
\[
\|F(z)-F(z')\| \le L_{F}\,\|z-z'\|,\qquad\forall z,z'\in\mathcal{Z}.
\]
Equivalently,
\[
\|\nabla_{x}L(x,y)-\nabla_{x}L(x',y')\|\le L_{F}\|(x,y)-(x',y')\|,
\]
and the same bound holds for $\nabla_{y}L$.
\end{assumption}
\begin{assumption}[Bounded feasible set]\label{as:bounded}
The domain $\mathcal{Z}$ has finite diameter:
\[
D:=\max_{z,z'\in\mathcal{Z}}\|z-z'\|<\infty .
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies
\[
0<\eta\le\frac{1}{2L_{F}} .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[O(1/T) primal–dual gap]\label{thm:main}
Under Assumptions \ref{as:convexconcave}–\ref{as:stepsize}, let $\{z_{k}\}_{k\ge0}$ be generated by the Extra‑Gradient iteration \eqref{EG}.  
Define the averaged iterate $\bar z_{T}=\frac1T\sum_{k=0}^{T-1}z_{k}=(\bar x_{T},\bar y_{T})$.  
Then for any saddle point $z^{\star}=(x^{\star},y^{\star})$,
\[
\gap(\bar x_{T},\bar y_{T})
:=\max_{y\in\mathcal{Y}}L(\bar x_{T},y)-\min_{x\in\mathcal{X}}L(x,\bar y_{T})
\;\le\;
\frac{D^{2}}{2\eta\,T}.
\tag{2}
\]
Consequently, choosing $\eta=\frac{1}{2L_{F}}$ yields
\[
\gap(\bar x_{T},\bar y_{T})\le\frac{L_{F}D^{2}}{T}=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition $\eta\le 1/(2L_{F})$ guarantees that the auxiliary point $z_{k+\frac12}$ stays inside $\mathcal{Z}$ and the operator $I-\eta F$ is non‑expansive.
\item \textbf{Hyperparameter Sensitivity.} The bound in \eqref{2} is monotone decreasing in $\eta$ up to the maximal admissible value $1/(2L_{F})$; larger stepsizes violate the Lipschitz contraction used in the proof.
\item \textbf{Comparison to Gradient Descent/Ascent.} For the same stepsize, plain simultaneous gradient descent/ascent only yields an $O(1/\sqrt{T})$ bound in the worst case, while the Extra‑Gradient attains the optimal $O(1/T)$ rate for deterministic smooth convex–concave games.
\end{itemize}

\section*{Discussion}
The theorem shows that the Extra‑Gradient method attains the optimal ergodic convergence rate for smooth convex–concave saddle‑point problems without any stochastic noise. The proof relies solely on the monotonicity of $F$ and its Lipschitz continuity; no strong convexity/concavity is required. Extensions to stochastic settings follow by adding variance bounds analogous to those used for SGD.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Monotone operator]
An operator $F:\mathcal{Z}\to\R^{n+m}$ is monotone if
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge 0,\qquad\forall z,z'\in\mathcal{Z}.
\]
\end{definition}
\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
If $L$ satisfies Assumption \ref{as:convexconcave}, then $F$ defined in Section 2 is monotone.
\end{lemma}
\begin{proof}
For any $z=(x,y),z'=(x',y')$,
\[
\begin{aligned}
\langle F(z)-F(z'),z-z'\rangle
&=\langle\nabla_{x}L(x,y)-\nabla_{x}L(x',y'),\,x-x'\rangle\\
&\quad -\langle\nabla_{y}L(x,y)-\nabla_{y}L(x',y'),\,y-y'\rangle\\
&\ge 0,
\end{aligned}
\]
where the inequality follows from convexity of $L(\cdot ,y)$ and concavity of $L(x,\cdot)$. ∎
\end{proof}
\begin{lemma}[Three–point identity]\label{lem:three}
For any vectors $a,b,c$ and any $\eta>0$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,c-b\rangle+\|c-b\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{lemma}[Lipschitz bound]\label{lem:lipschitz}
For $F$ Lipschitz with constant $L_{F}$,
\[
\|F(z)-F(z')\|^{2}\le L_{F}^{2}\|z-z'\|^{2}.
\tag{4}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we fix an arbitrary saddle point $z^{\star}=(x^{\star},y^{\star})\in\mathcal{Z}$.

\bigskip
\noindent\textbf{Step 1.}  Start from the definition of the extra‑gradient update and expand the squared distance to $z^{\star}$ after the correction step.
\begin{align}
\|z_{k+1}-z^{\star}\|^{2}
&= \big\|z_{k}-\eta F(z_{k+\frac12})-z^{\star}\big\|^{2} \nonumber\\
&= \|z_{k}-z^{\star}\|^{2}
   -2\eta\langle F(z_{k+\frac12}),\,z_{k}-z^{\star}\rangle
   +\eta^{2}\|F(z_{k+\frac12})\|^{2}. \tag{5}
\end{align}
\begin{description}
\item[\,] \hfill[Step 1]
\end{description}
We will bound the inner product term using monotonicity.

\bigskip
\noindent\textbf{Step 2.}  Apply monotonicity \eqref{lem:mono} with $z=z_{k+\frac12}$ and $z'=z^{\star}$:
\[
\langle F(z_{k+\frac12})-F(z^{\star}),\,z_{k+\frac12}-z^{\star}\rangle\ge0.
\]
Since $F(z^{\star})=0$ (first‑order optimality of a saddle point), we obtain
\[
\langle F(z_{k+\frac12}),\,z_{k+\frac12}-z^{\star}\rangle\ge0.
\tag{6}
\]
\begin{description}
\item[\,] \hfill[Step 2]
\end{description}
Rewrite the inner product in \eqref{5} as
\[
\langle F(z_{k+\frac12}),\,z_{k}-z^{\star}\rangle
= \langle F(z_{k+\frac12}),\,z_{k}-z_{k+\frac12}\rangle
  +\langle F(z_{k+\frac12}),\,z_{k+\frac12}-z^{\star}\rangle .
\tag{7}
\]
Using \eqref{6} the second term is non‑negative, thus
\[
\langle F(z_{k+\frac12}),\,z_{k}-z^{\star}\rangle
\ge \langle F(z_{k+\frac12}),\,z_{k}-z_{k+\frac12}\rangle .
\tag{8}
\]
\begin{description}
\item[\,] \hfill[Step 3]
\end{description}
Recall $z_{k+\frac12}=z_{k}-\eta F(z_{k})$, i.e. $z_{k}-z_{k+\frac12}= \eta F(z_{k})$.  Substituting into \eqref{8} yields
\[
\langle F(z_{k+\frac12}),\,z_{k}-z^{\star}\rangle
\ge \eta\langle F(z_{k+\frac12}),\,F(z_{k})\rangle .
\tag{9}
\]
\begin{description}
\item[\,] \hfill[Step 4]
\end{description}
Insert \eqref{9} into \eqref{5}:
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}
   -2\eta^{2}\langle F(z_{k+\frac12}),F(z_{k})\rangle
   +\eta^{2}\|F(z_{k+\frac12})\|^{2}.
\tag{10}
\]
\begin{description}
\item[\,] \hfill[Step 5]
\end{description}
Rearrange the right‑hand side:
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}
   -\eta^{2}\|F(z_{k})\|^{2}
   +\eta^{2}\|F(z_{k})-F(z_{k+\frac12})\|^{2}.
\tag{11}
\]
\begin{description}
\item[\,] \hfill[Step 6] (use $a^{2}+b^{2}-2ab=(a-b)^{2}$)
\end{description}
Now invoke the Lipschitz property \eqref{lem:lipschitz}:
\[
\|F(z_{k})-F(z_{k+\frac12})\|^{2}
\le L_{F}^{2}\|z_{k}-z_{k+\frac12}\|^{2}
= L_{F}^{2}\eta^{2}\|F(z_{k})\|^{2},
\tag{12}
\]
where we used $z_{k+\frac12}=z_{k}-\eta F(z_{k})$.
\begin{description}
\item[\,] \hfill[Step 7]
\end{description}
Plug \eqref{12} into \eqref{11}:
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}
   -\eta^{2}\bigl(1-L_{F}^{2}\eta^{2}\bigr)\|F(z_{k})\|^{2}.
\tag{13}
\]
\begin{description}
\item[\,] \hfill[Step 8]
\end{description}
Because of the stepsize restriction $\eta\le\frac{1}{2L_{F}}$, we have $1-L_{F}^{2}\eta^{2}\ge\frac12$. Hence
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}
   -\frac{\eta^{2}}{2}\|F(z_{k})\|^{2}.
\tag{14}
\]
\begin{description}
\item[\,] \hfill[Step 9]
\end{description}
Summing \eqref{14} from $k=0$ to $T-1$ telescopes the left‑hand side:
\[
\|z_{T}-z^{\star}\|^{2}
\le \|z_{0}-z^{\star}\|^{2}
   -\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\|F(z_{k})\|^{2}.
\tag{15}
\]
Since the leftmost term is non‑negative, we obtain a bound on the accumulated gradient norms:
\[
\sum_{k=0}^{T-1}\|F(z_{k})\|^{2}
\le \frac{2}{\eta^{2}}\|z_{0}-z^{\star}\|^{2}
\le \frac{2D^{2}}{\eta^{2}},
\tag{16}
\]
where $D$ is the diameter from Assumption \ref{as:bounded}.

\bigskip
\noindent\textbf{Step 3.}  Relate $\|F(z_{k})\|$ to the primal–dual gap.  By convexity in $x$ and concavity in $y$, for any $z=(x,y)$ and any saddle point $z^{\star}$,
\[
L(x,y^{\star})-L(x^{\star},y)
\le \langle F(z),\,z-z^{\star}\rangle .
\tag{17}
\]
Because of monotonicity, $\langle F(z),z-z^{\star}\rangle\ge0$. Moreover,
\[
\langle F(z),z-z^{\star}\rangle
\le \|F(z)\|\,\|z-z^{\star}\|
\le D\,\|F(z)\|.
\tag{18}
\]
Averaging \eqref{17} over $k=0,\dots,T-1$ and using the definition of $\bar z_{T}$ gives
\[
\gap(\bar x_{T},\bar y_{T})
\le \frac{1}{T}\sum_{k=0}^{T-1}\langle F(z_{k}),z_{k}-z^{\star}\rangle
\le \frac{D}{T}\sum_{k=0}^{T-1}\|F(z_{k})\|.
\tag{19}
\]
Apply Cauchy–Schwarz to the sum:
\[
\sum_{k=0}^{T-1}\|F(z_{k})\|
\le \sqrt{T}\Bigl(\sum_{k=0}^{T-1}\|F(z_{k})\|^{2}\Bigr)^{1/2}
\le \sqrt{T}\,\frac{\sqrt{2}\,D}{\eta},
\]
where the last inequality follows from \eqref{16}. Substituting into \eqref{19} yields
\[
\gap(\bar x_{T},\bar y_{T})
\le \frac{D}{T}\cdot\frac{\sqrt{2}\,D}{\eta}\sqrt{T}
= \frac{\sqrt{2}\,D^{2}}{\eta\sqrt{T}}.
\]
A slightly tighter bound is obtained by directly using the inequality
\[
\langle F(z_{k}),z_{k}-z^{\star}\rangle
\le \frac12\bigl(\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}\bigr)/\eta,
\]
which follows from \eqref{14}. Summing and dividing by $T$ gives
\[
\gap(\bar x_{T},\bar y_{T})
\le \frac{\|z_{0}-z^{\star}\|^{2}}{2\eta T}
\le\frac{D^{2}}{2\eta T}.
\tag{20}
\]
\begin{description}
\item[\,] \hfill[Step 10] (final ergodic bound)
\end{description}
Inequality \eqref{20} is exactly the statement \eqref{2} of Theorem \ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{20} and the admissible stepsize $\eta\le\frac{1}{2L_{F}}$ we obtain the explicit rate
\[
\gap(\bar x_{T},\bar y_{T})
\le \frac{L_{F}D^{2}}{T}.
\]
If the feasible set is a Euclidean ball of radius $R$, then $D\le 2R$ and the bound becomes $4L_{F}R^{2}/T$.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Strongly convex–strongly concave $L$.}  If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, the Extra‑Gradient enjoys a linear rate:
\[
\|z_{k}-z^{\star}\|\le\bigl(1-\eta\mu\bigr)^{k}\|z_{0}-z^{\star}\|,
\]
provided $\eta\le 1/L_{F}$.
\item \textbf{Pure minimisation.}  When $L(x,y)=f(x)$ (no $y$), the method reduces to the classical Extra‑Gradient for smooth convex optimisation and the same $O(1/T)$ ergodic bound holds.
\item \textbf{Stochastic gradients.}  Adding unbiased noise with bounded variance $\sigma^{2}$ yields an additional term $O(\sigma/\sqrt{T})$, reproducing the SGD‑type rate.
\end{itemize}
