\section*{Algorithm Name}
Projected Gradient Descent (PGD) for smooth convex optimization over a closed, convex, and bounded constraint set.

\section*{Mathematical Formulation}
We consider the constrained optimization problem
\[
\min_{w \in \mathcal{C}} f(w),
\]
where:
- $\mathcal{C}\subset \mathbb{R}^d$ is nonempty, closed, convex, and bounded;
- $f:\mathbb{R}^d \to \mathbb{R}$ is convex and differentiable with $L$-Lipschitz gradient.

Let $P_{\mathcal{C}}(x) := \arg\min_{y\in\mathcal{C}} \|y-x\|_2^2$ denote the Euclidean projection onto $\mathcal{C}$. The PGD update with stepsize $\eta>0$ is
\[
w_{t+1} \;=\; P_{\mathcal{C}}\bigl(w_t - \eta \nabla f(w_t)\bigr).
\]

Hyperparameters:
- Stepsize $\eta \in (0,1/L]$ (required for the stated convergence guarantees).

\section*{Pseudocode}
Input: initial point $w_0 \in \mathcal{C}$, stepsize $\eta \in (0,1/L]$, iterations $T$.
\begin{enumerate}
\item For $t=0,1,2,\dots,T-1$:
\begin{enumerate}
\item Compute gradient $g_t \leftarrow \nabla f(w_t)$.
\item Gradient step: $\tilde{w}_{t+1} \leftarrow w_t - \eta g_t$.
\item Projection: $w_{t+1} \leftarrow P_{\mathcal{C}}(\tilde{w}_{t+1})$.
\end{enumerate}
\item Return $w_T$ (or an average of iterates).
\end{enumerate}

\section*{Dry Run Example}
Problem: $f(w)=(w-3)^2$ on $\mathcal{C}=\mathbb{R}$ (projection is identity), with $w_0=0$, $\eta=0.1$. Gradient: $\nabla f(w)=2(w-3)$.
\begin{itemize}
\item Iteration $t=0$:
\begin{align*}
w_0 &= 0,\quad \nabla f(w_0)=2(0-3)=-6,\\
\tilde{w}_{1} &= w_0 - \eta \nabla f(w_0) = 0 - 0.1(-6) = 0.6,\\
w_1 &= \tilde{w}_{1} = 0.6,\quad f(w_0)=9,\quad f(w_1)=(0.6-3)^2=5.76.
\end{align*}
\item Iteration $t=1$:
\begin{align*}
w_1 &= 0.6,\quad \nabla f(w_1)=2(0.6-3)=-4.8,\\
\tilde{w}_{2} &= 0.6 - 0.1(-4.8) = 1.08,\\
w_2 &= 1.08,\quad f(w_2)=(1.08-3)^2=3.6864.
\end{align*}
\item Iteration $t=2$:
\begin{align*}
w_2 &= 1.08,\quad \nabla f(w_2)=2(1.08-3)=-3.84,\\
\tilde{w}_{3} &= 1.08 - 0.1(-3.84) = 1.464,\\
w_3 &= 1.464,\quad f(w_3)=(1.464-3)^2=2.359296.
\end{align*}
\end{itemize}

\section*{Assumptions}
Let $\|\cdot\|$ denote the Euclidean norm.
\begin{enumerate}
\item Feasibility set: $\mathcal{C}\subset\mathbb{R}^d$ is nonempty, closed, convex, and bounded, with diameter
\[
D \;:=\; \sup_{x,y\in\mathcal{C}} \|x-y\| \;<\; \infty.
\]
\item Smooth convex objective: $f:\mathbb{R}^d\to\mathbb{R}$ is convex and differentiable, and $\nabla f$ is $L$-Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\;\le\; L\|x-y\|\quad\forall x,y\in\mathbb{R}^d.
\]
\item Existence of minimizer: Let $w^\star \in \arg\min_{w\in\mathcal{C}} f(w)$ (assumed nonempty).
\item Stepsize: $\eta\in (0,1/L]$.
\item Initialization: $w_0\in\mathcal{C}$.
\end{enumerate}

\section*{Main Convergence Theorem}
Define the gradient mapping $G_\eta(w):=\tfrac{1}{\eta}\left(w - P_{\mathcal{C}}\bigl(w - \eta \nabla f(w)\bigr)\right)$. For the PGD iterates $w_{t+1}=P_{\mathcal{C}}(w_t - \eta \nabla f(w_t))$ with $\eta\in(0,1/L]$:
\begin{enumerate}
\item Descent (monotonicity):
\[
f(w_{t+1}) \;\le\; f(w_t) \;-\; \left(\frac{1}{\eta}-\frac{L}{2}\right)\,\|w_{t+1}-w_t\|^2,\quad\forall t\ge 0.
\]
In particular, $f(w_{t+1})\le f(w_t)$.
\item Fundamental telescoping inequality:
\[
f(w_{t+1})-f(w^\star) \;\le\; \frac{1}{2\eta}\Bigl(\|w_t-w^\star\|^2 - \|w_{t+1}-w^\star\|^2\Bigr),\quad\forall t\ge 0.
\]
Summing yields, for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\bigl(f(w_{t+1})-f(w^\star)\bigr) \;\le\; \frac{\|w_0-w^\star\|^2}{2\eta T}
\;\le\; \frac{D^2}{2\eta T}.
\]
Since $f(w_t)$ is nonincreasing, the last iterate obeys
\[
f(w_T)-f(w^\star) \;\le\; \frac{1}{T}\sum_{t=0}^{T-1}\bigl(f(w_{t+1})-f(w^\star)\bigr)
\;\le\; \frac{\|w_0-w^\star\|^2}{2\eta T}
\;\le\; \frac{D^2}{2\eta T}.
\]
With $\eta=1/L$, this becomes $f(w_T)-f(w^\star)\le \tfrac{L D^2}{2T}$.
\item Gradient mapping bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|G_\eta(w_t)\|^2 \;\le\; \frac{2\,(f(w_0)-f(w^\star))}{(2-\eta L)\,\eta\, T}.
\]
\end{enumerate}

\section*{Theoretical Properties}
\begin{itemize}
\item Stability and feasibility: Projections maintain $w_t\in\mathcal{C}$ for all $t$, and the nonexpansiveness of $P_{\mathcal{C}}$ prevents iterates from escaping the bounded domain, ensuring numerical stability.
\item Hyperparameter sensitivity: Any $\eta\in(0,1/L]$ guarantees monotone descent. Larger $\eta$ within this range increases progress per step, with the canonical choice $\eta=1/L$ giving the sharp constant $L D^2/(2T)$ in the rate.
\item Comparison to baselines: For smooth convex problems, deterministic PGD achieves $O(1/T)$ in function value, improving over the $O(1/\sqrt{T})$ rate of subgradient or stochastic methods under comparable assumptions.
\end{itemize}

\section*{Discussion}
The above bounds are deterministic; no stochastic expectation is needed. If additional structure holds (e.g., strong convexity and interior optimal solution), linear convergence is obtained eventually (see Special Cases).

\section*{Preliminaries (Definitions and Lemmas)}
We collect standard facts used in the proof.

\paragraph{Lemma 1 (Projection optimality condition).}
Let $\mathcal{C}$ be nonempty, closed, and convex. For any $z\in\mathbb{R}^d$ and $x^+=P_{\mathcal{C}}(z)$,
\[
\langle z - x^+,\, y - x^+\rangle \;\le\; 0,\quad \forall y\in\mathcal{C}.
\]
Proof. Consider $\phi(y)=\frac12\|y-z\|^2$, which is convex and differentiable with $\nabla\phi(y)=y-z$. The point $x^+$ minimizes $\phi$ over $\mathcal{C}$. For any $y\in\mathcal{C}$ and any $\tau\in[0,1]$, $x(\tau):=x^++\tau(y-x^+)\in\mathcal{C}$. The function $\psi(\tau):=\phi(x(\tau))$ has derivative $\psi'(0)=\langle x^+-z,\, y-x^+\rangle$. Optimality of $x^+$ implies $\psi'(0)\ge 0$, hence $\langle x^+-z,\, y-x^+\rangle\ge 0$, i.e., $\langle z-x^+,\, y-x^+\rangle\le 0$. \qed

\paragraph{Lemma 2 (Cosine/three-point identity).}
For any $a,b,c\in\mathbb{R}^d$,
\[
\langle a-b,\, b-c\rangle \;=\; \frac12\Bigl(\|a-c\|^2 - \|a-b\|^2 - \|b-c\|^2\Bigr).
\]
Proof. Expand the norms: $\|a-c\|^2=\|a-b+b-c\|^2=\|a-b\|^2+\|b-c\|^2+2\langle a-b,b-c\rangle$; rearrange. \qed

\paragraph{Lemma 3 (Descent lemma).}
If $\nabla f$ is $L$-Lipschitz, then for all $x,y\in\mathbb{R}^d$,
\[
f(y)\;\le\; f(x) + \langle \nabla f(x),\, y-x\rangle + \frac{L}{2}\|y-x\|^2.
\]
Proof. Standard result from smoothness (integral form or Taylor with remainder using Lipschitz bound). \qed

\section*{Main Proof (Step-by-Step Derivation)}
Let $w_{t+1}=P_{\mathcal{C}}(w_t - \eta \nabla f(w_t))$ with $\eta\in(0,1/L]$. We prove Items (1)--(3) of the Main Convergence Theorem.

\subsection*{A. Monotonic descent}
Apply Lemma 3 with $x=w_t$ and $y=w_{t+1}$:
\begin{equation}
f(w_{t+1}) \le f(w_t) + \langle \nabla f(w_t),\, w_{t+1}-w_t\rangle + \frac{L}{2}\|w_{t+1}-w_t\|^2. \quad \text{[Step 1: smoothness]}
\label{eq:descent-raw}
\end{equation}
By Lemma 1 with $z=w_t-\eta\nabla f(w_t)$ and $y=w_t$,
\begin{align}
\langle z - w_{t+1},\, w_t - w_{t+1}\rangle \le 0 
&\iff \langle w_t - \eta \nabla f(w_t) - w_{t+1},\, w_t - w_{t+1}\rangle \le 0 \quad \text{[Step 2: substitute $z$]} \nonumber\\
&\iff -\|w_{t+1}-w_t\|^2 + \eta \langle \nabla f(w_t),\, w_t - w_{t+1}\rangle \le 0 \quad \text{[Step 3: expand inner product]} \nonumber\\
&\iff \langle \nabla f(w_t),\, w_{t+1}-w_t\rangle \le -\frac{1}{\eta}\|w_{t+1}-w_t\|^2. \quad \text{[Step 4: rearrange]}
\label{eq:proj-with-wt}
\end{align}
Insert \eqref{eq:proj-with-wt} into \eqref{eq:descent-raw}:
\begin{align*}
f(w_{t+1}) 
&\le f(w_t) - \frac{1}{\eta}\|w_{t+1}-w_t\|^2 + \frac{L}{2}\|w_{t+1}-w_t\|^2 \quad \text{[Step 5: substitute inequality]}\\
&= f(w_t) - \left(\frac{1}{\eta}-\frac{L}{2}\right)\|w_{t+1}-w_t\|^2. \quad \text{[Step 6: collect terms]}
\end{align*}
Since $\eta\in(0,1/L]$, we have $\frac{1}{\eta}-\frac{L}{2}\ge \frac{L}{2}>0$. Therefore $f(w_{t+1})\le f(w_t)$, proving Item (1). \quad \text{[Step 7: use $\eta\le 1/L$]}

\subsection*{B. Fundamental telescoping inequality}
Apply Lemma 1 with $z=w_t-\eta\nabla f(w_t)$ and $y=w^\star$:
\begin{align}
\langle z - w_{t+1},\, w^\star - w_{t+1}\rangle \le 0 
&\iff \langle w_t - \eta \nabla f(w_t) - w_{t+1},\, w^\star - w_{t+1}\rangle \le 0 \quad \text{[Step 8: substitute $z$]} \nonumber\\
&\iff \langle w_t - w_{t+1},\, w^\star - w_{t+1}\rangle \le \eta \langle \nabla f(w_t),\, w^\star - w_{t+1}\rangle. \quad \text{[Step 9: rearrange]}
\label{eq:proj-wstar}
\end{align}
Rearrange \eqref{eq:proj-wstar} to isolate the gradient term with $(w_{t+1}-w^\star)$:
\begin{align}
\langle \nabla f(w_t),\, w_{t+1} - w^\star\rangle
&\le \frac{1}{\eta}\langle w_t - w_{t+1},\, w_{t+1} - w^\star\rangle. \quad \text{[Step 10: sign flip and swap arguments]}
\label{eq:key-inner}
\end{align}
Use Lemma 2 with $a=w_t$, $b=w_{t+1}$, $c=w^\star$:
\begin{align}
\langle w_t - w_{t+1},\, w_{t+1} - w^\star\rangle
= \frac12\Bigl(\|w_t - w^\star\|^2 - \|w_t - w_{t+1}\|^2 - \|w_{t+1} - w^\star\|^2\Bigr). \quad \text{[Step 11: three-point identity]}
\label{eq:three-pt}
\end{align}
Combine \eqref{eq:key-inner} and \eqref{eq:three-pt}:
\begin{align}
\langle \nabla f(w_t),\, w_{t+1} - w^\star\rangle
&\le \frac{1}{2\eta}\Bigl(\|w_t - w^\star\|^2 - \|w_t - w_{t+1}\|^2 - \|w_{t+1} - w^\star\|^2\Bigr). \quad \text{[Step 12: substitute \eqref{eq:three-pt}]}
\label{eq:grad-wt1-wstar}
\end{align}
Apply Lemma 3 with $x=w_t$, $y=w_{t+1}$ and subtract $f(w^\star)$:
\begin{align}
f(w_{t+1}) - f(w^\star)
&\le f(w_t) - f(w^\star) + \langle \nabla f(w_t),\, w_{t+1}-w_t\rangle + \frac{L}{2}\|w_{t+1}-w_t\|^2. \quad \text{[Step 13: smoothness]}
\label{eq:descent-gap-raw}
\end{align}
Decompose the inner product $\langle \nabla f(w_t),\, w_{t+1}-w_t\rangle$:
\begin{align}
\langle \nabla f(w_t),\, w_{t+1}-w_t\rangle
&= \langle \nabla f(w_t),\, w_{t+1}-w^\star\rangle + \langle \nabla f(w_t),\, w^\star-w_t\rangle. \quad \text{[Step 14: add-and-subtract $w^\star$]}
\label{eq:split}
\end{align}
By convexity of $f$, $f(w_t)-f(w^\star)\le \langle \nabla f(w_t),\, w_t - w^\star\rangle$, equivalently
\begin{equation}
\langle \nabla f(w_t),\, w^\star-w_t\rangle \le -\bigl(f(w_t)-f(w^\star)\bigr). \quad \text{[Step 15: convexity inequality]}
\label{eq:convexity}
\end{equation}
Insert \eqref{eq:split} and \eqref{eq:convexity} into \eqref{eq:descent-gap-raw}:
\begin{align}
f(w_{t+1}) - f(w^\star)
&\le f(w_t) - f(w^\star) + \langle \nabla f(w_t),\, w_{t+1}-w^\star\rangle - \bigl(f(w_t)-f(w^\star)\bigr) + \frac{L}{2}\|w_{t+1}-w_t\|^2 \nonumber\\
&= \langle \nabla f(w_t),\, w_{t+1}-w^\star\rangle + \frac{L}{2}\|w_{t+1}-w_t\|^2. \quad \text{[Step 16: cancellation]}
\label{eq:gap-reduced}
\end{align}
Now insert \eqref{eq:grad-wt1-wstar} into \eqref{eq:gap-reduced}:
\begin{align}
f(w_{t+1}) - f(w^\star)
&\le \frac{1}{2\eta}\Bigl(\|w_t - w^\star\|^2 - \|w_t - w_{t+1}\|^2 - \|w_{t+1} - w^\star\|^2\Bigr) + \frac{L}{2}\|w_{t+1}-w_t\|^2 \nonumber\\
&= \frac{1}{2\eta}\Bigl(\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2\Bigr) - \left(\frac{1}{2\eta}-\frac{L}{2}\right)\|w_{t+1}-w_t\|^2. \quad \text{[Step 17: collect terms]}
\label{eq:telescoping-core}
\end{align}
Since $\eta\in(0,1/L]$, $\frac{1}{2\eta}-\frac{L}{2}\ge 0$, so the last term is nonpositive and can be dropped:
\begin{align}
f(w_{t+1}) - f(w^\star)
&\le \frac{1}{2\eta}\Bigl(\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2\Bigr). \quad \text{[Step 18: drop nonpositive term]}
\label{eq:telescoping-ineq}
\end{align}
This proves the fundamental telescoping inequality (Item 2, first display).

\subsection*{C. Summation, average, and last-iterate bounds}
Sum \eqref{eq:telescoping-ineq} over $t=0$ to $T-1$:
\begin{align}
\sum_{t=0}^{T-1}\bigl(f(w_{t+1})-f(w^\star)\bigr)
&\le \frac{1}{2\eta}\sum_{t=0}^{T-1}\Bigl(\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2\Bigr) \quad \text{[Step 19: sum both sides]} \nonumber\\
&= \frac{1}{2\eta}\Bigl(\|w_0 - w^\star\|^2 - \|w_T - w^\star\|^2\Bigr) \quad \text{[Step 20: telescoping]} \nonumber\\
&\le \frac{1}{2\eta}\|w_0 - w^\star\|^2 \;\le\; \frac{D^2}{2\eta}, \quad \text{[Step 21: nonnegativity of norm and diameter bound]}
\label{eq:sum-gap}
\end{align}
and divide by $T$ to obtain
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1}\bigl(f(w_{t+1})-f(w^\star)\bigr) \;\le\; \frac{\|w_0 - w^\star\|^2}{2\eta T} \;\le\; \frac{D^2}{2\eta T}. \quad \text{[Step 22: average bound]}
\label{eq:avg-gap}
\end{equation}
From Item (1), $f(w_t)$ is nonincreasing, hence
\begin{equation}
f(w_T)-f(w^\star) \;\le\; \frac{1}{T}\sum_{t=0}^{T-1}\bigl(f(w_{t+1})-f(w^\star)\bigr) \;\le\; \frac{\|w_0 - w^\star\|^2}{2\eta T} \;\le\; \frac{D^2}{2\eta T}. \quad \text{[Step 23: last-iterate $\le$ average]}
\label{eq:last-iterate}
\end{equation}
With $\eta=1/L$, \eqref{eq:last-iterate} becomes $f(w_T)-f(w^\star) \le \tfrac{L D^2}{2T}$. \quad \text{[Step 24: plug $\eta=1/L$]}

\subsection*{D. Gradient mapping bound}
From \eqref{eq:descent-raw} and \eqref{eq:proj-with-wt}, we already have
\begin{equation}
f(w_{t+1}) \;\le\; f(w_t) - \left(\frac{1}{\eta}-\frac{L}{2}\right)\|w_{t+1}-w_t\|^2. \quad \text{[Step 25: restate Item (1)]}
\label{eq:descent-strong}
\end{equation}
Sum \eqref{eq:descent-strong} from $t=0$ to $T-1$:
\begin{align}
f(w_T) \;\le\; f(w_0) - \left(\frac{1}{\eta}-\frac{L}{2}\right)\sum_{t=0}^{T-1}\|w_{t+1}-w_t\|^2
&\implies \left(\frac{1}{\eta}-\frac{L}{2}\right)\sum_{t=0}^{T-1}\|w_{t+1}-w_t\|^2 \;\le\; f(w_0)-f(w_T) \nonumber\\
&\le\; f(w_0)-f(w^\star). \quad \text{[Step 26: sum and $f(w_T)\ge f(w^\star)$]}
\label{eq:sum-delta2}
\end{align}
Recall $G_\eta(w_t)=\tfrac{1}{\eta}(w_t - w_{t+1})$, so $\|G_\eta(w_t)\|^2=\tfrac{1}{\eta^2}\|w_{t+1}-w_t\|^2$. Then from \eqref{eq:sum-delta2},
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\|G_\eta(w_t)\|^2
&= \frac{1}{\eta^2 T}\sum_{t=0}^{T-1}\|w_{t+1}-w_t\|^2 \;\le\; \frac{1}{\eta^2 T}\cdot \frac{f(w_0)-f(w^\star)}{\frac{1}{\eta}-\frac{L}{2}} \quad \text{[Step 27: divide by $T$ and by coefficient]}\\
&= \frac{1}{\eta^2 T}\cdot \frac{f(w_0)-f(w^\star)}{\frac{1}{2\eta}(2-\eta L)}
\;=\; \frac{2\,(f(w_0)-f(w^\star))}{(2-\eta L)\,\eta\, T}. \quad \text{[Step 28: algebra]}
\end{align}
This proves Item (3).

\section*{Rate Derivation (With Constants)}
From \eqref{eq:last-iterate}, for any $T\ge 1$ and any $\eta\in(0,1/L]$,
\[
f(w_T)-f(w^\star) \;\le\; \frac{\|w_0-w^\star\|^2}{2\eta T}
\;\le\; \frac{D^2}{2\eta T}.
\]
Choosing $\eta=\tfrac{1}{L}$ yields the explicit constant
\[
f(w_T)-f(w^\star) \;\le\; \frac{L\,D^2}{2\,T}.
\]
Additionally, the average squared gradient mapping satisfies
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|G_\eta(w_t)\|^2 \;\le\; \frac{2\,(f(w_0)-f(w^\star))}{(2-\eta L)\,\eta\, T},
\]
so with $\eta=\tfrac{1}{L}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|G_{1/L}(w_t)\|^2 \;\le\; \frac{2L\,(f(w_0)-f(w^\star))}{T}.
\]

\section*{Special Cases}
\paragraph{Unconstrained case.}
If $\mathcal{C}=\mathbb{R}^d$, then $P_{\mathcal{C}}$ is the identity and PGD reduces to vanilla gradient descent: $w_{t+1}=w_t-\eta\nabla f(w_t)$. The same proof (omitting projection steps) yields
\[
f(w_T)-f(w^\star) \;\le\; \frac{\|w_0-w^\star\|^2}{2\eta T}
\quad\text{and}\quad
f(w_{t+1}) \le f(w_t) - \left(\frac{1}{\eta}-\frac{L}{2}\right)\|w_{t+1}-w_t\|^2.
\]

\paragraph{Strongly convex interior optimum (eventual linear rate).}
Suppose, in addition, $f$ is $\mu$-strongly convex ($\mu>0$) and the constrained minimizer $w^\star$ lies in the interior of $\mathcal{C}$. Then there exists $r>0$ such that $\mathbb{B}(w^\star,r)\subset \mathcal{C}$. For sufficiently large $t$, $w_t\in \mathbb{B}(w^\star,r)$ and the projection is inactive ($w_{t+1}=w_t-\eta\nabla f(w_t)$). With $\eta\in (0,2/(L+\mu)]$, standard gradient descent theory gives
\[
\|w_{t+1}-w^\star\|^2 \le (1-\mu\eta)\,\|w_t-w^\star\|^2 \quad \text{[Step 29: strong convexity + smoothness GD contraction],}
\]
hence
\[
f(w_{t})-f(w^\star) \le \frac{L}{2}\|w_t-w^\star\|^2 \le \frac{L}{2}(1-\mu\eta)^{t-t_0}\|w_{t_0}-w^\star\|^2 \quad \text{for all } t\ge t_0. \quad \text{[Step 30: smoothness upper bound]}
\]
This yields an eventual linear rate once the iterates enter the interior region.
