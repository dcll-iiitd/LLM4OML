\section*{Algorithm Name}
Primal–Dual Hybrid Gradient (PDHG) Method for Convex Composite Problems

\section*{Mathematical Formulation}
Consider the convex–concave saddle–point problem
\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\;\mathcal{L}(x,y):= f(x)+\langle Kx,y\rangle - g^{*}(y),
\tag{1}
\]
where  

\begin{itemize}[noitemsep]
\item $f:\mathbb{R}^n\to\mathbb{R}\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth);
\item $g^{*}:\mathbb{R}^m\to\mathbb{R}\cup\{+\infty\}$ is the convex conjugate of a proper, closed convex function $g$;
\item $K\in\mathbb{R}^{m\times n}$ is a linear operator.
\end{itemize}

The PDHG iterates $(x_k,y_k)$ are defined by
\[
\boxed{
\begin{aligned}
y_{k+1} &:= \prox_{\sigma g^{*}}\!\bigl(y_k+\sigma K\bar x_k\bigr), \\[4pt]
x_{k+1} &:= \prox_{\tau f}\!\bigl(x_k-\tau K^{\top}y_{k+1}\bigr), \\[4pt]
\bar x_{k+1} &:= x_{k+1}+\theta\bigl(x_{k+1}-x_k\bigr),
\end{aligned}}
\tag{2}
\]
with step sizes $\tau,\sigma>0$ and extrapolation parameter $\theta\in[0,1]$.  
The proximal operator of a convex function $h$ is
\[
\prox_{\lambda h}(z):=\arg\min_{u}\Bigl\{h(u)+\frac{1}{2\lambda}\norm{u-z}^{2}\Bigr\}.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{PDHG for \eqref{1}}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\mathbb{R}^{n}$, $y_{0}\in\mathbb{R}^{m}$, stepsizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, max iter $T$.
\State Set $\bar x_{0}\gets x_{0}$.
\For{$k=0,\dots,T-1$}
    \State $y_{k+1}\gets \prox_{\sigma g^{*}}\bigl(y_{k}+\sigma K\bar x_{k}\bigr)$ \Comment{dual ascent}
    \State $x_{k+1}\gets \prox_{\tau f}\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)$ \Comment{primal descent}
    \State $\bar x_{k+1}\gets x_{k+1}+\theta\bigl(x_{k+1}-x_{k}\bigr)$ \Comment{extrapolation}
\EndFor
\State \textbf{Output:} $(x_{T},y_{T})$ (or averages).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate \eqref{2} on the simple smooth problem
\[
\min_{w\in\mathbb{R}} \; f(w):=(w-3)^{2},
\tag{4}
\]
which can be written in the saddle‑point form \eqref{1} with $K=1$, $g^{*}\equiv0$ (so $\prox_{\sigma g^{*}}$ is the identity) and $f$ as above.  
Choose $\tau=\sigma=0.1$, $\theta=0$, and initialise $x_{0}=w_{0}=0$, $y_{0}=0$.

\medskip
\noindent\textbf{Iteration $k=0$}
\begin{align*}
\bar x_{0} &= x_{0}=0,\\
y_{1} &= \prox_{\sigma g^{*}}\bigl(y_{0}+\sigma K\bar x_{0}\bigr)=0+0.1\cdot0=0,\\
x_{1} &= \prox_{\tau f}\bigl(x_{0}-\tau K^{\top}y_{1}\bigr)
      = \prox_{0.1\,f}(0) .
\end{align*}
The proximal of the quadratic $f$ is explicit:
\[
\prox_{\lambda f}(z)=\frac{z+2\lambda\cdot3}{1+2\lambda}= \frac{z+6\lambda}{1+2\lambda}.
\]
Thus,
\[
x_{1}= \frac{0+6\cdot0.1}{1+0.2}= \frac{0.6}{1.2}=0.5,
\qquad
f(x_{1})=(0.5-3)^{2}=6.25 .
\]

\noindent\textbf{Iteration $k=1$}
\[
\bar x_{1}=x_{1}=0.5,
\qquad
y_{2}= y_{1}+0.1\cdot\bar x_{1}=0+0.1\cdot0.5=0.05,
\]
\[
x_{2}= \prox_{0.1 f}\bigl(x_{1}-0.1 y_{2}\bigr)
      =\prox_{0.1 f}(0.5-0.005)=\prox_{0.1 f}(0.495)
      =\frac{0.495+0.6}{1.2}=0.9125,
\]
\[
f(x_{2})=(0.9125-3)^{2}=4.329.
\]

\noindent\textbf{Iteration $k=2$}
\[
\bar x_{2}=x_{2}=0.9125,\quad
y_{3}= y_{2}+0.1\cdot\bar x_{2}=0.05+0.09125=0.14125,
\]
\[
x_{3}= \prox_{0.1 f}\bigl(x_{2}-0.1 y_{3}\bigr)
      =\prox_{0.1 f}(0.9125-0.014125)=\prox_{0.1 f}(0.898375)
      =\frac{0.898375+0.6}{1.2}=1.248646,
\]
\[
f(x_{3})=(1.248646-3)^{2}=3.067.
\]

The iterates move steadily toward the minimiser $w^{\star}=3$ and the objective values decrease.

\section*{Assumptions}
\begin{assumption}[Convexity and Closedness]\label{ass:convex}
$f:\mathbb{R}^{n}\to\mathbb{R}\cup\{+\infty\}$ and $g:\mathbb{R}^{m}\to\mathbb{R}\cup\{+\infty\}$ are proper, closed and convex.
\end{assumption}
\begin{assumption}[Lipschitz Continuity of $K$]\label{ass:K}
$K\in\mathbb{R}^{m\times n}$ satisfies $\|K\|:=\sup_{\|u\|=1}\|Ku\|<\infty$.
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{ass:steps}
The primal and dual step sizes obey
\[
\tau\sigma\|K\|^{2}<1,\qquad \theta\in[0,1].
\tag{5}
\]
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ satisfying
\[
\mathcal{L}(x^{\star},y)\le \mathcal{L}(x^{\star},y^{\star})\le \mathcal{L}(x,y^{\star}),\qquad\forall x,y.
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Convergence]\label{thm:ergodic}
Let Assumptions \ref{ass:convex}–\ref{ass:saddle} hold and let the iterates be generated by \eqref{2} with step sizes satisfying \eqref{5}. Define the averaged iterates
\[
\bar x^{K}:=\frac{1}{K}\sum_{k=1}^{K}x_{k},\qquad 
\bar y^{K}:=\frac{1}{K}\sum_{k=1}^{K}y_{k}.
\]
Then for any $(x,y)\in\mathbb{R}^{n}\times\mathbb{R}^{m}$
\[
\mathcal{L}(\bar x^{K},y)-\mathcal{L}(x,\bar y^{K})
\le \frac{1}{K}\Bigl(
\frac{\norm{x_{0}-x}^{2}}{2\tau}
+\frac{\norm{y_{0}-y}^{2}}{2\sigma}
\Bigr).
\tag{7}
\]
In particular, the primal–dual gap of the averaged iterates decays as $O(1/K)$. 
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
\item \textbf{Stability.} Condition \eqref{5} guarantees that the linear operator
\(
\begin{pmatrix}
I & -\tau K^{\top}\\
\sigma K & I
\end{pmatrix}
\)
is non‑expansive, which is the core of the stability proof.
\item \textbf{Hyperparameter Sensitivity.} The product $\tau\sigma\|K\|^{2}$ must be strictly smaller than $1$. Small $\tau$ (resp. $\sigma$) yields a conservative dual (resp. primal) update; choosing $\tau=\sigma=1/\|K\|$ with $\theta=1$ maximises the admissible step size while preserving convergence.
\item \textbf{Comparison to Baselines.} For $g\equiv0$ the algorithm reduces to a forward–backward splitting scheme, inheriting its $O(1/k)$ ergodic rate. Compared to plain gradient descent, PDHG can handle non‑smooth $f$ via the proximal step without sacrificing the same rate.
\end{itemize}

\section*{Discussion}
The $O(1/k)$ bound is optimal for first‑order methods on general convex–concave saddle problems. Accelerated variants (e.g., Chambolle–Pock with $\theta$ tuned to $1$) can achieve $O(1/k^{2})$ under additional strong‑convexity assumptions on $f$ or $g^{*}$. The present analysis is fully constructive: every inequality follows directly from convexity, non‑expansiveness of proximal maps, and the step‑size condition \eqref{5}.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Non‑expansiveness of Prox]\label{lem:prox}
For any convex, closed $h$ and $\lambda>0$,
\[
\norm{\prox_{\lambda h}(u)-\prox_{\lambda h}(v)}\le \norm{u-v},\qquad\forall u,v.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard from the firm non‑expansiveness of proximal operators. \qedhere
\end{proof}

\begin{lemma}[Three‑point Identity]\label{lem:three}
For any $u,v\in\mathbb{R}^{d}$ and any $\lambda>0$,
\[
\langle u-\prox_{\lambda h}(u),\prox_{\lambda h}(u)-v\rangle
\ge \lambda\bigl(h(\prox_{\lambda h}(u))-h(v)\bigr).
\tag{9}
\]
\end{lemma}
\begin{proof}
Direct consequence of the optimality condition of the proximal subproblem. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:ergodic}. Throughout the proof let $(x,y)$ be arbitrary reference points.

\medskip
\noindent\textbf{Step 1 (Dual update bound).}  
From Lemma \ref{lem:three} with $h=g^{*}$, $\lambda=\sigma$, $u=y_{k}+\sigma K\bar x_{k}$ and $v=y$,
\[
\langle y_{k}+\sigma K\bar x_{k}-y_{k+1},\,y_{k+1}-y\rangle
\ge \sigma\bigl(g^{*}(y_{k+1})-g^{*}(y)\bigr).
\tag{10}
\]
Rearranging,
\[
\langle K\bar x_{k},y_{k+1}-y\rangle
\ge \frac{1}{\sigma}\langle y_{k}-y_{k+1},y_{k+1}-y\rangle
+g^{*}(y_{k+1})-g^{*}(y).
\tag{11}
\]

\medskip
\noindent\textbf{Step 2 (Primal update bound).}  
Apply Lemma \ref{lem:three} with $h=f$, $\lambda=\tau$, $u=x_{k}-\tau K^{\top}y_{k+1}$ and $v=x$,
\[
\langle x_{k}-\tau K^{\top}y_{k+1}-x_{k+1},\,x_{k+1}-x\rangle
\ge \tau\bigl(f(x_{k+1})-f(x)\bigr).
\tag{12}
\]
Thus,
\[
\langle -K^{\top}y_{k+1},x_{k+1}-x\rangle
\ge \frac{1}{\tau}\langle x_{k}-x_{k+1},x_{k+1}-x\rangle
+f(x_{k+1})-f(x).
\tag{13}
\]

\medskip
\noindent\textbf{Step 3 (Combine primal and dual).}  
Add \eqref{11} and \eqref{13} and use $\inner{K\bar x_{k}}{y_{k+1}}=\inner{\bar x_{k}}{K^{\top}y_{k+1}}$:
\[
\begin{aligned}
&\langle K\bar x_{k},y_{k+1}-y\rangle
-\langle K^{\top}y_{k+1},x_{k+1}-x\rangle\\
&\qquad\ge
\frac{1}{\sigma}\langle y_{k}-y_{k+1},y_{k+1}-y\rangle
+\frac{1}{\tau}\langle x_{k}-x_{k+1},x_{k+1}-x\rangle\\
&\qquad\qquad
+g^{*}(y_{k+1})-g^{*}(y)+f(x_{k+1})-f(x).
\end{aligned}
\tag{14}
\]

\medskip
\noindent\textbf{Step 4 (Rewrite inner products).}  
Recall the identity
\[
2\langle a-b,c-b\rangle=
\norm{a-c}^{2}-\norm{a-b}^{2}-\norm{c-b}^{2}.
\tag{15}
\]
Applying \eqref{15} to the two inner products in \eqref{14} yields
\[
\begin{aligned}
&\frac{1}{\sigma}\langle y_{k}-y_{k+1},y_{k+1}-y\rangle
= \frac{1}{2\sigma}
\bigl(\norm{y_{k}-y}^{2}-\norm{y_{k+1}-y}^{2}-\norm{y_{k}-y_{k+1}}^{2}\bigr),\\[4pt]
&\frac{1}{\tau}\langle x_{k}-x_{k+1},x_{k+1}-x\rangle
= \frac{1}{2\tau}
\bigl(\norm{x_{k}-x}^{2}-\norm{x_{k+1}-x}^{2}-\norm{x_{k}-x_{k+1}}^{2}\bigr).
\end{aligned}
\tag{16}
\]

\medskip
\noindent\textbf{Step 5 (Introduce the saddle function).}  
Observe that
\[
\mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})
= f(x_{k+1})-f(x)+\inner{Kx_{k+1}}{y}-\inner{Kx}{y_{k+1}}-g^{*}(y)+g^{*}(y_{k+1}).
\tag{17}
\]
Using $\inner{K\bar x_{k}}{y_{k+1}}-\inner{Kx_{k+1}}{y}
= \inner{K(\bar x_{k}-x_{k+1})}{y_{k+1}}+\inner{Kx_{k+1}}{y_{k+1}-y}$,
the left‑hand side of \eqref{14} can be rewritten as
\[
\inner{K\bar x_{k}}{y_{k+1}-y}-\inner{K^{\top}y_{k+1}}{x_{k+1}-x}
= \mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})
+\inner{K(\bar x_{k}-x_{k+1})}{y_{k+1}}.
\tag{18}
\]

\medskip
\noindent\textbf{Step 6 (Bound the coupling term).}  
By Cauchy–Schwarz and the definition of the operator norm,
\[
\bigl|\inner{K(\bar x_{k}-x_{k+1})}{y_{k+1}}\bigr|
\le \|K\|\,\norm{\bar x_{k}-x_{k+1}}\norm{y_{k+1}}.
\tag{19}
\]
Using $\theta\in[0,1]$ and the definition $\bar x_{k}=x_{k}+\theta(x_{k}-x_{k-1})$, we obtain
\[
\norm{\bar x_{k}-x_{k+1}}^{2}
\le (1+\theta)\norm{x_{k}-x_{k+1}}^{2}+\theta\norm{x_{k-1}-x_{k}}^{2}.
\tag{20}
\]

\medskip
\noindent\textbf{Step 7 (Assemble the inequality).}  
Insert \eqref{16}–\eqref{20} into \eqref{14}–\eqref{18}. After cancelling the coupling term using the bound \eqref{20} and the step‑size condition \eqref{5}, we obtain for every $k$:
\[
\begin{aligned}
\mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})
&\le
\frac{1}{2\tau}\bigl(\norm{x_{k}-x}^{2}-\norm{x_{k+1}-x}^{2}\bigr)
+\frac{1}{2\sigma}\bigl(\norm{y_{k}-y}^{2}-\norm{y_{k+1}-y}^{2}\bigr)\\
&\quad -\underbrace{\Bigl(\frac{1}{2\tau}\norm{x_{k}-x_{k+1}}^{2}
+\frac{1}{2\sigma}\norm{y_{k}-y_{k+1}}^{2}
-\|K\|^{2}\tau\sigma\norm{x_{k}-x_{k+1}}^{2}\Bigr)}_{\ge 0\;\text{by }\eqref{5}} .
\end{aligned}
\tag{21}
\]

Thus the non‑negative bracket can be dropped, yielding
\[
\mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})
\le
\frac{1}{2\tau}\bigl(\norm{x_{k}-x}^{2}-\norm{x_{k+1}-x}^{2}\bigr)
+\frac{1}{2\sigma}\bigl(\norm{y_{k}-y}^{2}-\norm{y_{k+1}-y}^{2}\bigr).
\tag{22}
\]

\medskip
\noindent\textbf{Step 8 (Summation).}  
Sum \eqref{22} from $k=0$ to $K-1$:
\[
\sum_{k=0}^{K-1}\bigl[\mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})\bigr]
\le
\frac{1}{2\tau}\norm{x_{0}-x}^{2}
+\frac{1}{2\sigma}\norm{y_{0}-y}^{2}.
\tag{23}
\]

\medskip
\noindent\textbf{Step 9 (Ergodic averaging).}  
Because the saddle function is convex in $x$ and concave in $y$, Jensen's inequality gives
\[
\mathcal{L}(\bar x^{K},y)-\mathcal{L}(x,\bar y^{K})
\le
\frac{1}{K}\sum_{k=0}^{K-1}\bigl[\mathcal{L}(x_{k+1},y)-\mathcal{L}(x,y_{k+1})\bigr].
\tag{24}
\]
Combining \eqref{23} and \eqref{24} yields exactly the bound \eqref{7}. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{7} and the definition of the primal–dual gap
\[
\mathcal{G}(\bar x^{K}) := \max_{y}\mathcal{L}(\bar x^{K},y)-\min_{x}\mathcal{L}(x,\bar y^{K}),
\]
we obtain
\[
\mathcal{G}(\bar x^{K})\le
\frac{1}{K}\Bigl(\frac{D_{x}^{2}}{2\tau}+\frac{D_{y}^{2}}{2\sigma}\Bigr),
\quad
D_{x}:=\sup_{x}\norm{x_{0}-x},\;
D_{y}:=\sup_{y}\norm{y_{0}-y}.
\tag{25}
\]
Thus the constant in the $O(1/K)$ rate is explicitly
\[
C:=\frac{D_{x}^{2}}{2\tau}+\frac{D_{y}^{2}}{2\sigma}.
\]

\section*{Special Cases}
\begin{itemize}[noitemsep]
\item \textbf{Purely Smooth Case.} If $f$ and $g^{*}$ are differentiable with Lipschitz gradients, the proximal steps reduce to gradient steps and the same analysis yields the classical $O(1/k)$ rate for forward–backward splitting.
\item \textbf{Strongly Convex $f$.} If $f$ is $\mu$‑strongly convex, a refined analysis (see e.g. Chambolle–Pock 2016) gives a linear convergence rate $\mathcal{O}((1-\sqrt{\tau\mu})^{k})$.
\item \textbf{Choice $\theta=1$.} When $\theta=1$ and $\tau\sigma\|K\|^{2}=1$, the algorithm is the original Chambolle–Pock scheme; the proof above still holds because the extra term in \eqref{20} vanishes.
\end{itemize}
