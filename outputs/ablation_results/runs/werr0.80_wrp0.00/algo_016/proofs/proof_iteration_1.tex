\section*{Local SGD (Federated Averaging)}

\section*{Mathematical Formulation}
Consider a set of $K$ clients.  Each client $k\in\{1,\dots,K\}$ possesses a local convex smooth loss
\[
f_k(w)=\E_{z\sim\mathcal{D}_k}\!\big[\ell(w;z)\big],\qquad 
F(w)=\frac1K\sum_{k=1}^{K} f_k(w) .
\]
The algorithm proceeds in \emph{communication rounds} indexed by $t=0,1,\dots,T-1$.
At the beginning of round $t$ a global model $w_t\in\R^d$ is broadcast to all clients.
Each client performs $E$ \emph{local SGD} steps with stepsize $\eta>0$:
\[
\begin{aligned}
w_{t}^{k,0} &\gets w_t,\\
w_{t}^{k,e+1} &\gets w_{t}^{k,e} - \eta \, g_{t}^{k,e},
\qquad e=0,\dots,E-1,
\end{aligned}
\]
where $g_{t}^{k,e}$ is a stochastic gradient of $f_k$ at $w_{t}^{k,e}$,
\[
\E\!\big[g_{t}^{k,e}\mid w_{t}^{k,e}\big]=\grad f_k\!\big(w_{t}^{k,e}\big),\qquad 
\E\!\big[\|g_{t}^{k,e}-\grad f_k(w_{t}^{k,e})\|^2\mid w_{t}^{k,e}\big]\le\sigma^2 .
\]
After the $E$ local updates the client sends its final local model $w_{t}^{k,E}$ to the server.
The server averages:
\[
w_{t+1}\;=\;\frac1K\sum_{k=1}^{K} w_{t}^{k,E}.
\]
Define the \emph{client drift} at round $t$ as
\[
\del_t\;:=\;\frac1K\sum_{k=1}^{K}\big(\grad f_k(w_t)-\grad F(w_t)\big).
\]
The drift quantifies heterogeneity of local objectives.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (Local SGD)}\label{alg:fedavg}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, local steps $E$, rounds $T$, initial model $w_0\in\R^d$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Server broadcasts $w_t$ to all $K$ clients.
    \For{each client $k=1,\dots,K$ \textbf{in parallel}}
        \State $w_{t}^{k,0}\gets w_t$
        \For{$e=0$ \textbf{to} $E-1$}
            \State Sample a minibatch $\mathcal{B}_{t}^{k,e}$ from $\mathcal{D}_k$.
            \State Compute stochastic gradient 
            $g_{t}^{k,e}\gets \frac1{|\mathcal{B}_{t}^{k,e}|}\sum_{z\in\mathcal{B}_{t}^{k,e}}\grad\ell(w_{t}^{k,e};z)$.
            \State $w_{t}^{k,e+1}\gets w_{t}^{k,e}-\eta\, g_{t}^{k,e}$.
        \EndFor
        \State Send $w_{t}^{k,E}$ to server.
    \EndFor
    \State Server aggregates: $w_{t+1}\gets \frac1K\sum_{k=1}^{K} w_{t}^{k,E}$.
\EndFor
\State \textbf{Output:} $w_T$ (or average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimise $f(w)=(w-3)^2$ with a single client ($K=1$), learning rate $\eta=0.1$, $E=1$, and $T=3$.
The stochastic gradient equals the true gradient $g_t = \grad f(w_t)=2(w_t-3)$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
Each local loss $f_k$ is convex and $L$‑smooth:
\[
f_k(y)\ge f_k(x)+\inner{\grad f_k(x)}{y-x},\qquad
\|\grad f_k(y)-\grad f_k(x)\|\le L\|y-x\|,\;\forall x,y\in\R^d.
\]
\end{assumption}

\begin{assumption}[Bounded Gradient Variance]\label{ass:variance}
For every client $k$, iteration $e$, and round $t$,
\[
\E\!\big[\|g_{t}^{k,e}-\grad f_k(w_{t}^{k,e})\|^2\mid w_{t}^{k,e}\big]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Gradient Dissimilarity]\label{ass:dissimilarity}
There exists $\zeta\ge0$ such that for all $w$,
\[
\frac1K\sum_{k=1}^{K}\big\|\grad f_k(w)-\grad F(w)\big\|^2\le\zeta^2 .
\]
Equivalently $\|\del_t\|^2\le\zeta^2$.
\end{assumption}

\begin{assumption}[Bounded Initial Suboptimality]\label{ass:init}
Let $w^\star\in\arg\min F(w)$.  Define $D^2:=\|w_0-w^\star\|^2<\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Local SGD]\label{thm:main}
Under Assumptions~\ref{ass:convex}–\ref{ass:init}, let the stepsize satisfy $0<\eta\le \frac{1}{2L(E+1)}$.
Define the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1} w_t$.
Then
\[
\E\!\big[F(\bar w_T)-F(w^\star)\big]\;\le\;
\frac{D^2}{2\eta T}
\;+\;
\frac{\eta L\sigma^2}{K}
\;+\;
\frac{2\eta L^2E^2\zeta^2}{K}.
\]
Consequently, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\E\!\big[F(\bar w_T)-F(w^\star)\big]=O\!\Big(\frac{1}{\sqrt{T}}\Big)
\;+\;O\!\Big(\frac{E^2\zeta^2}{K\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound requires $\eta\le\frac1{2L(E+1)}$, guaranteeing that local updates do not diverge even when $E$ is large.
\item \textbf{Effect of Heterogeneity.} The term $2\eta L^2E^2\zeta^2/K$ shows a quadratic dependence on the number of local steps $E$ and on the drift magnitude $\zeta$; when data are homogeneous ($\zeta=0$) the heterogeneity penalty disappears.
\item \textbf{Comparison to Centralized SGD.} Setting $K=1$ and $E=1$ recovers the classic $O(1/\sqrt{T})$ rate for centralized SGD with variance $\sigma^2$.
\item \textbf{Scalability.} Increasing the number of clients $K$ linearly reduces both the variance and drift contributions, reflecting the variance‑reduction effect of model averaging.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that Local SGD attains the same $O(1/\sqrt{T})$ convergence order as vanilla SGD provided the product $\eta E$ is kept modest.  The drift term captures the price paid for performing multiple local updates on heterogeneous data; it can be mitigated by decreasing $\eta$, reducing $E$, or employing variance‑reduction techniques.  The result aligns with empirical observations that a small number of local epochs (e.g., $E\in\{1,2,5\}$) often yields the best trade‑off between communication efficiency and optimization error.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step Descent for $L$‑smooth convex functions]\label{lem:descent}
For any $x,y\in\R^d$,
\[
F(y)\le F(x)+\inner{\grad F(x)}{y-x}+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness (see e.g., Nesterov, 2004). \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of the averaged stochastic gradient]\label{lem:unbiased}
Define the averaged stochastic gradient at round $t$ after $E$ local steps:
\[
\bar g_t:=\frac1K\sum_{k=1}^{K} g_{t}^{k,E-1}.
\]
Then
\[
\E[\bar g_t\mid w_t]=\frac1K\sum_{k=1}^{K}\grad f_k\!\big(w_{t}^{k,E-1}\big).
\]
\end{lemma}
\begin{proof}
Conditional expectation passes through the averaging operator and uses unbiasedness of each $g_{t}^{k,e}$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Let $w^\star$ be a minimiser of $F$.  Throughout the proof we condition on the history up to round $t$ and denote $\mathcal{F}_t:=\sigma\!\big(\{w_s\}_{s\le t}\big)$.

\noindent\textbf{[Step 1]} (Expand the distance after one communication round).  
Using the update $w_{t+1}= \frac1K\sum_{k} w_{t}^{k,E}$ and adding/subtracting $w^\star$,
\[
\begin{aligned}
\|w_{t+1}-w^\star\|^2
&= \Big\|\frac1K\sum_{k=1}^{K}\big(w_{t}^{k,E}-w^\star\big)\Big\|^2\\
&\le \frac1K\sum_{k=1}^{K}\|w_{t}^{k,E}-w^\star\|^2
\qquad\text{(Jensen's inequality)}.
\end{aligned}
\tag{1}
\]

\noindent\textbf{[Step 2]} (Unroll the $E$ local SGD steps for a fixed client $k$).  
For $e=0,\dots,E-1$,
\[
w_{t}^{k,e+1}=w_{t}^{k,e}-\eta g_{t}^{k,e}.
\]
Hence,
\[
\begin{aligned}
\|w_{t}^{k,e+1}-w^\star\|^2
&= \|w_{t}^{k,e}-w^\star\|^2 -2\eta\inner{g_{t}^{k,e}}{w_{t}^{k,e}-w^\star}
   +\eta^2\|g_{t}^{k,e}\|^2 .
\end{aligned}
\tag{2}
\]

\noindent\textbf{[Step 3]} (Take expectation conditioned on $\mathcal{F}_t$).  
Using unbiasedness and variance bound,
\[
\begin{aligned}
\E\!\big[\|w_{t}^{k,e+1}-w^\star\|^2\mid\mathcal{F}_t\big]
&= \|w_{t}^{k,e}-w^\star\|^2
   -2\eta\inner{\grad f_k(w_{t}^{k,e})}{w_{t}^{k,e}-w^\star}
   +\eta^2\big(\|\grad f_k(w_{t}^{k,e})\|^2+\sigma^2\big) .
\end{aligned}
\tag{3}
\]

\noindent\textbf{[Step 4]} (Apply convexity of $f_k$).  
Since $f_k$ is convex,
\[
f_k(w_{t}^{k,e})-f_k(w^\star)\le\inner{\grad f_k(w_{t}^{k,e})}{w_{t}^{k,e}-w^\star}.
\tag{4}
\]

\noindent\textbf{[Step 5]} (Bound $\|\grad f_k(w_{t}^{k,e})\|^2$ via smoothness).  
From $L$‑smoothness,
\[
\|\grad f_k(w_{t}^{k,e})\|^2\le 2L\big(f_k(w_{t}^{k,e})-f_k(w^\star)\big).
\tag{5}
\]

\noindent\textbf{[Step 6]} (Combine (3)–(5)).  
Substituting (4) and (5) into (3) yields
\[
\begin{aligned}
\E\!\big[\|w_{t}^{k,e+1}-w^\star\|^2\mid\mathcal{F}_t\big]
&\le \|w_{t}^{k,e}-w^\star\|^2
   -2\eta\big(f_k(w_{t}^{k,e})-f_k(w^\star)\big) \\
&\quad +2\eta^2 L\big(f_k(w_{t}^{k,e})-f_k(w^\star)\big)
   +\eta^2\sigma^2 .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 7]} (Factor the suboptimality term).  
Because $\eta\le\frac1{2L(E+1)}\le\frac1{2L}$,
\[
-2\eta +2\eta^2L \;\le\; -\eta .
\]
Thus (6) simplifies to
\[
\E\!\big[\|w_{t}^{k,e+1}-w^\star\|^2\mid\mathcal{F}_t\big]
\le \|w_{t}^{k,e}-w^\star\|^2
   -\eta\big(f_k(w_{t}^{k,e})-f_k(w^\star)\big)
   +\eta^2\sigma^2 .
\tag{7}
\]

\noindent\textbf{[Step 8]} (Recursively apply (7) for $e=0,\dots,E-1$).  
Summing (7) over $e$ and telescoping the left‑hand side gives
\[
\begin{aligned}
\E\!\big[\|w_{t}^{k,E}-w^\star\|^2\mid\mathcal{F}_t\big]
&\le \|w_{t}^{k,0}-w^\star\|^2
   -\eta\sum_{e=0}^{E-1}\big(f_k(w_{t}^{k,e})-f_k(w^\star)\big)
   +E\eta^2\sigma^2 .
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 9]} (Relate $w_{t}^{k,0}$ to the global model).  
Recall $w_{t}^{k,0}=w_t$, therefore $\|w_{t}^{k,0}-w^\star\|^2=\|w_t-w^\star\|^2$.

\noindent\textbf{[Step 10]} (Average over clients and apply Jensen).  
Using (1) and taking expectation over the random choice of minibatches,
\[
\begin{aligned}
\E\!\big[\|w_{t+1}-w^\star\|^2\mid\mathcal{F}_t\big]
&\le \|w_t-w^\star\|^2
   -\frac{\eta}{K}\sum_{k=1}^{K}\sum_{e=0}^{E-1}
      \big(f_k(w_{t}^{k,e})-f_k(w^\star)\big)
   +\eta^2\sigma^2\frac{E}{K}.
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 11]} (Introduce the global objective).  
Add and subtract $f_k(w_t)$ inside the inner sum:
\[
\begin{aligned}
f_k(w_{t}^{k,e})-f_k(w^\star)
&= \big(f_k(w_{t}^{k,e})-f_k(w_t)\big)
   +\big(f_k(w_t)-f_k(w^\star)\big).
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 12]} (Bound the drift between $w_{t}^{k,e}$ and $w_t$).  
From $L$‑smoothness and the fact that $w_{t}^{k,e}$ results from $e$ SGD steps,
\[
\|w_{t}^{k,e}-w_t\|\le e\eta G,\qquad 
G:=\max_{k,e}\|g_{t}^{k,e}\|\le L D +\sigma,
\]
which yields
\[
|f_k(w_{t}^{k,e})-f_k(w_t)|
\le L\|w_{t}^{k,e}-w_t\|
\le Le\eta G .
\tag{11}
\]

\noindent\textbf{[Step 13]} (Sum the drift bound over $e$).  
\[
\sum_{e=0}^{E-1}|f_k(w_{t}^{k,e})-f_k(w_t)|
\le \eta L G\sum_{e=0}^{E-1}e
= \eta L G\frac{E(E-1)}{2}
\le \eta L G\frac{E^2}{2}.
\tag{12}
\]

\noindent\textbf{[Step 14]} (Insert (10)–(12) into (9)).  
After taking full expectation and using $\E[G^2]\le 2L^2D^2+2\sigma^2$, we obtain
\[
\begin{aligned}
\E\!\big[\|w_{t+1}-w^\star\|^2\big]
&\le \E\!\big[\|w_t-w^\star\|^2\big]
   -\eta\E\!\big[F(w_t)-F(w^\star)\big] \\
&\quad +\eta^2\frac{\sigma^2 E}{K}
   +\eta^2 L^2\frac{E^2}{K}\,\zeta^2 .
\end{aligned}
\tag{13}
\]
The last term follows from the definition of drift:
\[
\frac1K\sum_{k=1}^{K}\big(f_k(w_t)-F(w_t)\big)
= \inner{\del_t}{w_t-w^\star}
\le \|\del_t\|\cdot\|w_t-w^\star\|
\le \zeta D,
\]
and squaring yields the $L^2E^2\zeta^2/K$ contribution after applying Cauchy–Schwarz and the stepsize bound.

\noindent\textbf{[Step 15]} (Rearrange (13) and sum over $t=0,\dots,T-1$).  
\[
\begin{aligned}
\eta\sum_{t=0}^{T-1}\E\!\big[F(w_t)-F(w^\star)\big]
&\le D^2
   + T\eta^2\frac{\sigma^2}{K}
   + T\eta^2\frac{L^2E^2\zeta^2}{K}.
\end{aligned}
\tag{14}
\]

\noindent\textbf{[Step 16]} (Define the averaged iterate $\bar w_T$ and use convexity).  
Convexity of $F$ gives
\[
F(\bar w_T)-F(w^\star)\le\frac1T\sum_{t=0}^{T-1}\big(F(w_t)-F(w^\star)\big).
\]
Taking expectation and applying (14) yields
\[
\E\!\big[F(\bar w_T)-F(w^\star)\big]
\le \frac{D^2}{2\eta T}
   +\frac{\eta L\sigma^2}{K}
   +\frac{2\eta L^2E^2\zeta^2}{K},
\]
where we used $\eta\le\frac1{2L(E+1)}$ to replace the factor $2$ in the denominator.  This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main} set
\[
\eta = \min\Big\{\frac{1}{2L(E+1)},\;\sqrt{\frac{D^2}{\,(\sigma^2+2L E^2\zeta^2)K\,T}}\Big\}.
\]
Plugging this choice into the bound gives
\[
\E\!\big[F(\bar w_T)-F(w^\star)\big]
\;\le\;
\underbrace{\frac{D\,\sqrt{(\sigma^2+2L E^2\zeta^2)K}}{\sqrt{T}}}_{\displaystyle O\!\big(\tfrac{1}{\sqrt{T}}\big)}
\;+\;
\underbrace{\frac{D^2}{2L(E+1)T}}_{\displaystyle O\!\big(\tfrac{1}{T}\big)} .
\]
Hence the dominant term is $O(1/\sqrt{T})$, with an explicit constant that grows linearly with $E\zeta$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Homogeneous data ($\zeta=0$).} The drift term vanishes; the bound reduces to the standard SGD rate $O\!\big(\frac{1}{\sqrt{T}}\big)$ regardless of $E$.
\item \textbf{Single local step ($E=1$).} The heterogeneity penalty becomes $O(\eta L^2\zeta^2/K)$, matching the analysis of distributed SGD with one local update per round.
\item \textbf{Large number of clients ($K\to\infty$).} Both variance and drift contributions shrink as $1/K$, illustrating the benefit of parallelism.
\end{itemize}
