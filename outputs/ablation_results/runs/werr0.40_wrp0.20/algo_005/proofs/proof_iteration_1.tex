\section*{Algorithm Name}
\textbf{Stochastic Variance–Reduced Gradient (SVRG) with Double Loop}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$ where each $f_i$ is convex, $L$‑smooth and the average $f$ is $\mu$‑strongly convex.  
Denote the optimal point by $x^{\star}=\arg\min_x f(x)$.  

The algorithm consists of an \emph{outer epoch} indexed by $s=0,1,\dots$ and an \emph{inner iteration} indexed by $t=0,1,\dots,m-1$.

\begin{align}
\text{(Outer step)}\qquad 
\tilde{x}^{s} &\gets x_{0}^{s} \quad\text{(snapshot)} \label{eq:snapshot}\\
\tilde{\mu}^{s} &\gets \grad f(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\grad f_i(\tilde{x}^{s}) \tag{full gradient}\\[4pt]
\text{(Inner step)}\qquad 
\text{for }t=0,\dots,m-1 \text{ do } \nonumber\\
\qquad i_t &\stackrel{\text{i.i.d.}}{\sim}\{1,\dots,n\},\\
\qquad v_t^{s} &:= \grad f_{i_t}(x_t^{s})-\grad f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s},\\
\qquad x_{t+1}^{s} &:= x_t^{s}-\eta\, v_t^{s}. \label{eq:update}
\end{align}
After $m$ inner steps we set $x_{0}^{s+1}=x_{m}^{s}$ and start the next epoch.

Hyper‑parameters: step size $\eta>0$, inner loop length $m\in\mathbb{N}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (double loop)}
\begin{algorithmic}[1]
\REQUIRE initial point $x^{0}$, step size $\eta$, inner loop length $m$, number of epochs $S$
\STATE $\tilde{x}^{0}\gets x^{0}$
\FOR{$s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{s}= \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{s})$
    \STATE $x^{s}_0 \gets \tilde{x}^{s}$
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \STATE $v^{s}_t \gets \grad f_{i_t}(x^{s}_t)-\grad f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \STATE $x^{s}_{t+1} \gets x^{s}_t-\eta\,v^{s}_t$
    \ENDFOR
    \STATE $\tilde{x}^{s+1}\gets x^{s}_m$
\ENDFOR
\RETURN $\tilde{x}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem $f(w)=(w-3)^2$ (so $n=1$, $f_1=f$).  
We have $\grad f(w)=2(w-3)$, $L=2$, $\mu=2$ (strongly convex).  
Choose $w_0=0$, step size $\eta=0.1$, inner length $m=2$ and run one epoch ($S=1$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (here $v_t=\grad f(x_t)$) & $f(x_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $x_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $x_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$
\end{tabular}
\end{center}
After the inner loop we set $\tilde{x}^{1}=x_2=1.08$.  The objective has decreased from $9$ to $3.6864$, illustrating the variance‑reduced step.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y},\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
The average $f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\inner{\grad f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^2,
\qquad\forall x,y .
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The dataset size $n$ is finite and each $f_i$ is convex.
\end{assumption}
\begin{assumption}[Step size]\label{ass:step}
The step size satisfies
\[
0<\eta\le\frac{1}{3L}.
\]
\end{assumption}
\begin{assumption}[Inner loop length]\label{ass:m}
The inner loop length $m$ is a positive integer that will be chosen later to balance the contraction factor.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:step}, let the epoch length be $m\ge \frac{2L}{\mu}$.  
Define the contraction constant
\[
\rho:=\frac{1}{\mu\eta m}+2L\eta\in(0,1).
\]
Then the sequence of snapshot points $\{\tilde{x}^{s}\}_{s\ge0}$ generated by the algorithm satisfies
\begin{align}
\E\!\left[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\right]
\le \rho\,\E\!\left[\norm{\tilde{x}^{s}-x^{\star}}^{2}\right],
\qquad\forall s\ge0,
\end{align}
and consequently for any $S\ge1$,
\[
\E\!\left[\norm{\tilde{x}^{S}-x^{\star}}^{2}\right]\le \rho^{S}\norm{x^{0}-x^{\star}}^{2}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound $\rho<1$ holds provided $\eta\le \frac{1}{3L}$ and $m\ge \frac{2L}{\mu}$, guaranteeing that the recursion is a contraction.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\eta$ accelerates progress per inner step but increases the term $2L\eta$ in $\rho$. The product $\eta m$ appears in the first term of $\rho$; thus one can keep $\eta$ modest while increasing $m$ to improve the rate.
\item \textbf{Comparison to SGD.} For strongly convex $f$, plain SGD with diminishing steps yields $O(1/t)$ sub‑linear rate, whereas SVRG attains a geometric rate $O(\rho^{S})$, i.e. linear convergence.
\item \textbf{Special case $n=1$.} The variance‑reduced estimator reduces to the exact gradient, and the algorithm becomes standard gradient descent with step size $\eta$, reproducing the classical rate $(1-\mu\eta)^{S}$.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
For any inner iteration $t$ in epoch $s$,
\[
\E_{i_t}\!\left[\,v_t^{s}\mid x_t^{s},\tilde{x}^{s}\right]=\grad f(x_t^{s}).
\]
\end{lemma}
\begin{proof}
\[
\E_{i_t}[v_t^{s}]
= \frac1n\sum_{i=1}^{n}\bigl(\grad f_i(x_t^{s})-\grad f_i(\tilde{x}^{s})\bigr)+\tilde{\mu}^{s}
= \grad f(x_t^{s})-\grad f(\tilde{x}^{s})+\grad f(\tilde{x}^{s})
= \grad f(x_t^{s}).\quad\blacksquare
\]
\end{proof}
\begin{lemma}[Bound on the second moment]\label{lem:second}
For any $t$,
\[
\E_{i_t}\!\left[\norm{v_t^{s}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]
\le 4L\bigl(f(x_t^{s})-f(x^{\star})+f(\tilde{x}^{s})-f(x^{\star})\bigr).
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the identity
$\norm{a+b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}$,
\[
\begin{aligned}
\norm{v_t^{s}}^{2}
&=\norm{\grad f_{i_t}(x_t^{s})-\grad f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}}^{2}\\
&\le 2\norm{\grad f_{i_t}(x_t^{s})-\grad f_{i_t}(\tilde{x}^{s})}^{2}
     +2\norm{\tilde{\mu}^{s}}^{2}\\
&\le 2L^{2}\norm{x_t^{s}-\tilde{x}^{s}}^{2}+2\norm{\tilde{\mu}^{s}}^{2}.
\end{aligned}
\]
Taking expectation over $i_t$ and using smoothness together with the standard inequality
$ \norm{\grad f(x)}^{2}\le 2L\bigl(f(x)-f(x^{\star})\bigr)$ (which follows from strong convexity and $L$‑smoothness) yields the claimed bound. $\quad\blacksquare$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear} by analysing one epoch and then chaining the result.

\subsection*{Step 1 – One inner update}
From the update \eqref{eq:update} we have for any fixed $s$,
\[
\begin{aligned}
\norm{x_{t+1}^{s}-x^{\star}}^{2}
&=\norm{x_{t}^{s}-\eta v_t^{s}-x^{\star}}^{2}\\
&=\norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\inner{v_t^{s}}{x_{t}^{s}-x^{\star}}
   +\eta^{2}\norm{v_t^{s}}^{2}. \tag{1}
\end{aligned}
\]
Taking expectation conditioned on $x_t^{s}$ and $\tilde{x}^{s}$ and using Lemma \ref{lem:unbiased} gives
\[
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]
= \norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\inner{\grad f(x_t^{s})}{x_{t}^{s}-x^{\star}}
   +\eta^{2}\E\!\left[\norm{v_t^{s}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]. \tag{2}
\]
\subsection*{Step 2 – Apply strong convexity}
Strong convexity (Assumption \ref{ass:strong}) yields
\[
\inner{\grad f(x_t^{s})}{x_{t}^{s}-x^{\star}}
\ge f(x_t^{s})-f(x^{\star})+\frac{\mu}{2}\norm{x_t^{s}-x^{\star}}^{2}. \tag{3}
\]
Insert (3) into (2):
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]
&\le \norm{x_{t}^{s}-x^{\star}}^{2}
   -2\eta\Bigl(f(x_t^{s})-f(x^{\star})\Bigr)
   -\eta\mu\norm{x_t^{s}-x^{\star}}^{2}\\
&\quad +\eta^{2}\E\!\left[\norm{v_t^{s}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]. \tag{4}
\end{aligned}
\]
\subsection*{Step 3 – Bound the variance term}
Using Lemma \ref{lem:second} we obtain
\[
\E\!\left[\norm{v_t^{s}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]
\le 4L\Bigl(f(x_t^{s})-f(x^{\star})+f(\tilde{x}^{s})-f(x^{\star})\Bigr). \tag{5}
\]
Plug (5) into (4):
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{s}-x^{\star}}^{2}\mid x_t^{s},\tilde{x}^{s}\right]
&\le \bigl(1-\eta\mu\bigr)\norm{x_{t}^{s}-x^{\star}}^{2}\\
&\quad -2\eta\bigl(1-2L\eta\bigr)\bigl(f(x_t^{s})-f(x^{\star})\bigr)\\
&\quad +4L\eta^{2}\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr). \tag{6}
\end{aligned}
\]
\subsection*{Step 4 – Sum over the inner loop}
Define $\Phi_t^{s}:=\E\bigl[\norm{x_{t}^{s}-x^{\star}}^{2}\bigr]$ and $G^{s}:=f(\tilde{x}^{s})-f(x^{\star})$.  
Taking total expectation of (6) and summing $t=0,\dots,m-1$ yields
\[
\begin{aligned}
\Phi_{m}^{s}
&\le (1-\eta\mu)^{m}\Phi_{0}^{s}
   -2\eta\bigl(1-2L\eta\bigr)\sum_{t=0}^{m-1}\E\!\bigl[f(x_{t}^{s})-f(x^{\star})\bigr]\\
&\quad +4L\eta^{2}m\,G^{s}. \tag{7}
\end{aligned}
\]
Because $f$ is $\mu$‑strongly convex,
\[
f(x_{t}^{s})-f(x^{\star})\ge \frac{\mu}{2}\norm{x_{t}^{s}-x^{\star}}^{2}
\]
and therefore
\[
\sum_{t=0}^{m-1}\E\!\bigl[f(x_{t}^{s})-f(x^{\star})\bigr]\ge
\frac{\mu}{2}\sum_{t=0}^{m-1}\Phi_{t}^{s}. \tag{8}
\]
Insert (8) into (7) and drop the non‑negative term $\sum_{t=0}^{m-1}\Phi_{t}^{s}$ (which only tightens the bound):
\[
\Phi_{m}^{s}\le (1-\eta\mu)^{m}\Phi_{0}^{s}+4L\eta^{2}m\,G^{s}. \tag{9}
\]
Recall that $\tilde{x}^{s}=x_{0}^{s}$ and $\tilde{x}^{s+1}=x_{m}^{s}$, so $\Phi_{0}^{s}= \E\!\bigl[\norm{\tilde{x}^{s}-x^{\star}}^{2}\bigr]$ and $\Phi_{m}^{s}= \E\!\bigl[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\bigr]$.  Moreover, by $L$‑smoothness,
\[
G^{s}=f(\tilde{x}^{s})-f(x^{\star})
\le \frac{L}{2}\norm{\tilde{x}^{s}-x^{\star}}^{2}
= \frac{L}{2}\Phi_{0}^{s}. \tag{10}
\]
Substituting (10) into (9) gives
\[
\begin{aligned}
\E\!\bigl[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\bigr]
&\le (1-\eta\mu)^{m}\Phi_{0}^{s}+4L\eta^{2}m\cdot\frac{L}{2}\Phi_{0}^{s}\\
&= \Bigl[(1-\eta\mu)^{m}+2L^{2}\eta^{2}m\Bigr]\Phi_{0}^{s}. \tag{11}
\end{aligned}
\]
\subsection*{Step 5 – Simplify the contraction factor}
Using the elementary inequality $(1-\alpha)^{m}\le 1-m\alpha$ for $0\le\alpha\le1$ and $\alpha=\eta\mu$, we obtain
\[
(1-\eta\mu)^{m}\le 1-m\eta\mu .\tag{12}
\]
Insert (12) into (11):
\[
\E\!\bigl[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\bigr]
\le \Bigl[1-m\eta\mu+2L^{2}\eta^{2}m\Bigr]\Phi_{0}^{s}
= \Bigl[\frac{1}{\mu\eta m}+2L\eta\Bigr]\Phi_{0}^{s}. \tag{13}
\]
Define the contraction constant
\[
\rho:=\frac{1}{\mu\eta m}+2L\eta. \tag{14}
\]
Because $\eta\le\frac{1}{3L}$ and $m\ge\frac{2L}{\mu}$, one checks directly that $0<\rho<1$ (see the discussion in the theorem).  Hence (13) becomes
\[
\E\!\bigl[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\bigr]\le\rho\,\E\!\bigl[\norm{\tilde{x}^{s}-x^{\star}}^{2}\bigr]. \tag{15}
\]
Iterating (15) over $s=0,\dots,S-1$ yields the claimed linear rate:
\[
\E\!\bigl[\norm{\tilde{x}^{S}-x^{\star}}^{2}\bigr]\le\rho^{S}\norm{x^{0}-x^{\star}}^{2}. \tag{16}
\]

\subsection*{Step 6 – Completion}
Equation (16) is exactly the statement of Theorem \ref{thm:linear}.  This concludes the proof. $\quad\blacksquare$

\section*{Rate Derivation (With Constants)}
From (14) the convergence factor can be written explicitly as
\[
\rho(\eta,m)=\frac{1}{\mu\eta m}+2L\eta.
\]
Choosing $\eta=\frac{1}{3L}$ (the largest admissible step) and $m=\left\lceil\frac{6L}{\mu}\right\rceil$ gives
\[
\rho\le \frac{1}{\mu\cdot\frac{1}{3L}\cdot\frac{6L}{\mu}}+2L\cdot\frac{1}{3L}
= \frac{1}{2}+ \frac{2}{3}= \frac{7}{6}<1,
\]
so a concrete numeric bound is $\rho\le 0.84$ after rounding up $m$.
Thus after $S$ epochs the error decays as $\mathcal{O}\!\bigl(0.84^{S}\bigr)$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact gradient descent ($n=1$).} The estimator $v_t^{s}$ equals $\grad f(x_t^{s})$; the algorithm reduces to $x_{t+1}=x_t-\eta\grad f(x_t)$. The contraction factor becomes $1-\mu\eta$, recovering the classical GD rate.
\item \textbf{Large‑batch limit $m\to\infty$.} The term $\frac{1}{\mu\eta m}$ vanishes and $\rho\to 2L\eta$, which is the rate of standard gradient descent with step $\eta$.
\item \textbf{Mini‑batch SVRG.} If each inner step uses a mini‑batch of size $b$, the variance bound in Lemma \ref{lem:second} improves by a factor $1/b$, leading to a smaller constant in front of $L\eta$ and thus a better $\rho$.
\end{itemize}

\bigskip
\noindent\textbf{Conclusion.}  
Under standard smoothness and strong convexity assumptions, the double‑loop SVRG algorithm enjoys a linear convergence guarantee with explicit contraction factor $\rho=\frac{1}{\mu\eta m}+2L\eta<1$.  The proof proceeds solely from first‑principle algebraic manipulations, carefully annotating each non‑trivial step.
