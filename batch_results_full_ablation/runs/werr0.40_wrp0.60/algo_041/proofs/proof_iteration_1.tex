\section*{Algorithm Name}
\textbf{Simultaneous Gradient Descent–Ascent (GDA) for $\displaystyle\min_{y}\max_{x}\,L(x,y)$}

\section*{Mathematical Formulation}
Let $L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\rightarrow\mathbb{R}$ be a differentiable \emph{non‑convex–concave} saddle‑function.  
Given a stepsize $\eta>0$, the GDA iterates are
\[
\boxed{
\begin{aligned}
x_{t+1}&=x_{t}+\eta\,\nabla_{x}L(x_{t},y_{t})\qquad\text{(gradient \emph{ascent} in $x$)},\\[4pt]
y_{t+1}&=y_{t}-\eta\,\nabla_{y}L(x_{t},y_{t})\qquad\text{(gradient \emph{descent} in $y$)} .
\end{aligned}}
\tag{1}
\]

All iterates are collected in the vector $z_t:=(x_t,y_t)\in\mathbb{R}^{d_x+d_y}$ and the joint gradient is
\[
\nabla L(z_t)=\begin{pmatrix}\nabla_x L(x_t,y_t)\\[2pt]\nabla_y L(x_t,y_t)\end{pmatrix}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Simultaneous GDA}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, max. iterations $T$, initial point $(x_0,y_0)$
\FOR{$t=0,\dots,T-1$}
    \STATE Compute stochastic (or exact) gradients 
    \[
    g^x_t\gets\nabla_x L(x_t,y_t),\qquad 
    g^y_t\gets\nabla_y L(x_t,y_t).
    \]
    \STATE \textbf{Ascent in $x$:} $x_{t+1}\gets x_t+\eta\,g^x_t$
    \STATE \textbf{Descent in $y$:} $y_{t+1}\gets y_t-\eta\,g^y_t$
\ENDFOR
\STATE \textbf{Return} $(x_T,y_T)$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

\]

\[

\]

\section*{Assumptions}
\begin{enumerate}[label={\bf A\arabic*:},leftmargin=*,itemsep=2pt]
\item \textbf{Smoothness.} $L$ has $L$‑Lipschitz continuous gradients:
\[
\|\nabla L(z)-\nabla L(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\mathbb{R}^{d_x+d_y}.
\tag{A1}
\]
\item \textbf{Bounded Gradient.} There exists $G>0$ such that
\[
\|\nabla L(z)\|\le G,\qquad\forall z.
\tag{A2}
\]
\item \textbf{Existence of a Stationary Point.} The set
\[
\mathcal{Z}^\star:=\{z=(x,y)\mid \nabla_x L(x,y)=0,\;\nabla_y L(x,y)=0\}
\]
is non‑empty.
\tag{A3}
\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\eta\le \frac{1}{L}.
\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:rate}
Under Assumptions A1–A4, the iterates generated by \eqref{1} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\bigl\|\nabla L(z_t)\bigr\|^{2}
\;\le\;
\frac{2\bigl(L(z_0)-L^\inf\bigr)}{\eta\,T},
\tag{2}
\]
where $L^\inf:=\inf_{x,y}L(x,y)$ (possibly $-\infty$). Consequently,
\[
\min_{0\le t<T}\bigl\|\nabla L(z_t)\bigr\|\;=\;O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\eta\le1/L$, the one‑step update is a non‑expansive map in the Euclidean norm (see Lemma~\ref{lem:descent}).
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{2} scales linearly with $1/\eta$; a larger stepsize accelerates the $1/\sqrt{T}$ decay but may violate the stability condition $ \eta\le1/L$.
\item \textbf{Comparison to Baselines.} For convex–concave $L$, the same bound reduces to the classic $O(1/T)$ rate of extragradient methods. In the non‑convex–concave regime we recover the optimal $O(1/\sqrt{T})$ rate for first‑order methods (cf. SGD).
\end{itemize}

\section*{Discussion}
The theorem guarantees that the algorithm drives the norm of the joint gradient to zero in an \emph{average} sense, which is the standard stationarity notion for non‑convex problems. No assumption of convexity in $x$ is required; only smoothness and boundedness are needed. The result matches the best known rates for deterministic first‑order methods applied to non‑convex optimization, showing that simultaneous GDA does not suffer from additional slowdown due to the saddle‑point structure.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Smoothness Inequality]
If $L$ satisfies (A1) with constant $L$, then for any $z,z'$
\[
L(z')\le L(z)+\langle\nabla L(z),z'-z\rangle+\frac{L}{2}\|z'-z\|^{2}.
\tag{3}
\]
\end{definition}
\begin{lemma}[One‑step Descent–Ascent Inequality]\label{lem:descent}
Let $z_{t+1}$ be defined by \eqref{1} with $\eta\le1/L$. Then
\[
L(z_{t+1})\le L(z_t)-\frac{\eta}{2}\bigl\|\nabla L(z_t)\bigr\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
\begin{enumerate}[label={[\arabic*]},leftmargin=*,itemsep=1pt]
\item Apply the smoothness inequality \eqref{3} with $z=z_t$ and $z'=z_{t+1}=z_t+ \eta\tilde{\nabla}L(z_t)$, where $\tilde{\nabla}L(z_t):=(\nabla_x L,-\nabla_y L)$ denotes the \emph{ascent–descent} direction. This yields
\[
L(z_{t+1})\le L(z_t)+\eta\langle\nabla L(z_t),\tilde{\nabla}L(z_t)\rangle+\frac{L\eta^{2}}{2}\|\tilde{\nabla}L(z_t)\|^{2}.
\tag{5}
\]
\item Observe that $\langle\nabla L(z_t),\tilde{\nabla}L(z_t)\rangle
= \|\nabla_x L(z_t)\|^{2}-\|\nabla_y L(z_t)\|^{2}$, while 
$\|\tilde{\nabla}L(z_t)\|^{2}= \|\nabla_x L(z_t)\|^{2}+\|\nabla_y L(z_t)\|^{2}= \|\nabla L(z_t)\|^{2}$.
\item Substitute these identities into \eqref{5}:
\[
L(z_{t+1})\le L(z_t)+\eta\bigl(\|\nabla_x L\|^{2}-\|\nabla_y L\|^{2}\bigr)
+\frac{L\eta^{2}}{2}\|\nabla L\|^{2}.
\tag{6}
\]
\item Since the two terms $\|\nabla_x L\|^{2}$ and $-\|\nabla_y L\|^{2}$ cancel partially, we rewrite \eqref{6} as
\[
L(z_{t+1})\le L(z_t)-\eta\|\nabla_y L\|^{2}
+\eta\|\nabla_x L\|^{2}
+\frac{L\eta^{2}}{2}\bigl(\|\nabla_x L\|^{2}+\|\nabla_y L\|^{2}\bigr).
\tag{7}
\]
\item Group the coefficients of $\|\nabla_x L\|^{2}$ and $\|\nabla_y L\|^{2}$:
\[
L(z_{t+1})\le L(z_t)-\Bigl[\eta-\frac{L\eta^{2}}{2}\Bigr]\|\nabla_y L\|^{2}
-\Bigl[\frac{L\eta^{2}}{2}-\eta\Bigr]\|\nabla_x L\|^{2}.
\tag{8}
\]
\item Because $\eta\le 1/L$, we have $\eta-\frac{L\eta^{2}}{2}\ge \frac{\eta}{2}$. Moreover,
$\,\frac{L\eta^{2}}{2}-\eta\le -\frac{\eta}{2}$. Hence both brackets are bounded below by $\frac{\eta}{2}$ in magnitude, and \eqref{8} simplifies to
\[
L(z_{t+1})\le L(z_t)-\frac{\eta}{2}\bigl(\|\nabla_x L\|^{2}+\|\nabla_y L\|^{2}\bigr)
= L(z_t)-\frac{\eta}{2}\|\nabla L(z_t)\|^{2}.
\tag{9}
\]
\end{enumerate}
Thus inequality \eqref{4} holds. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:rate} using Lemma~\ref{lem:descent}.

\begin{enumerate}[label={[\arabic*]},leftmargin=*,itemsep=2pt]
\item \textbf{Summation of the one‑step inequality.}  
From Lemma~\ref{lem:descent} we have for every $t\ge0$,
\[
L(z_{t+1})\le L(z_t)-\frac{\eta}{2}\|\nabla L(z_t)\|^{2}.
\tag{10}
\]
\item \textbf{Rearrange \eqref{10}.}  
Bring the gradient term to the left:
\[
\frac{\eta}{2}\|\nabla L(z_t)\|^{2}\le L(z_t)-L(z_{t+1}).
\tag{11}
\]
\item \textbf{Sum from $t=0$ to $T-1$.}  
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\|\nabla L(z_t)\|^{2}
\le \sum_{t=0}^{T-1}\bigl(L(z_t)-L(z_{t+1})\bigr)
= L(z_0)-L(z_T).
\tag{12}
\]
\item \textbf{Lower bound $L(z_T)$.}  
Since $L^\inf:=\inf_{z}L(z)\le L(z_T)$, we obtain
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\|\nabla L(z_t)\|^{2}
\le L(z_0)-L^\inf.
\tag{13}
\]
\item \textbf{Divide by $T$ and isolate the average.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla L(z_t)\|^{2}
\le \frac{2\bigl(L(z_0)-L^\inf\bigr)}{\eta\,T}.
\tag{14}
\]
This is exactly the bound \eqref{2}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Define the constant
\[
C:=2\bigl(L(z_0)-L^\inf\bigr)/\eta .
\]
Then \eqref{14} reads
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla L(z_t)\|^{2}\le \frac{C}{T}.
\]
Hence there exists at least one index $t^\star\in\{0,\dots,T-1\}$ such that
\[
\|\nabla L(z_{t^\star})\|^{2}\le \frac{C}{T},
\qquad\text{i.e.}\quad
\|\nabla L(z_{t^\star})\|\le \sqrt{\frac{C}{T}}
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{enumerate}[label={\bf S\arabic*:},leftmargin=*,itemsep=2pt]
\item \textbf{Convex–Concave $L$.}  
If $L$ is convex in $x$ and concave in $y$, the same proof yields a \emph{monotone} decrease of the primal–dual gap, leading to the faster $O(1/T)$ rate for the gap.
\item \textbf{Stochastic Gradients.}  
When $g_t$ is an unbiased estimator of the true gradient with bounded variance $\sigma^2$, an extra term $\eta\sigma^2$ appears in \eqref{10}, and the bound becomes $O\bigl((1/\sqrt{T})+\eta\sigma^2\bigr)$. Choosing $\eta\propto1/\sqrt{T}$ restores the $O(1/\sqrt{T})$ rate.
\item \textbf{Adaptive Stepsize.}  
If $\eta_t$ satisfies $\sum_{t}\eta_t=\infty$ and $\sum_{t}\eta_t^{2}<\infty$, the same telescoping argument yields $\|\nabla L(z_t)\|\to0$ almost surely.
\end{enumerate}
