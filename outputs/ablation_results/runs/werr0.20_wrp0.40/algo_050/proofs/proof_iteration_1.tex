\section*{Algorithm Name}
\textbf{Convex Trust‑Region Method (CTR)}  

The method iteratively solves a quadratic model of a smooth convex objective within a Euclidean ball (the trust region) and updates the radius according to the quality of the step.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^n\rightarrow\mathbb{R}$ be convex and continuously differentiable with $L$‑Lipschitz gradient.  
At iteration $k$ we have a current point $x_k$ and a trust‑region radius $\Delta_k>0$.

\begin{enumerate}[label=\arabic*)]
\item \textbf{Model.}  
\[
m_k(p)\;:=\;f(x_k)+\nabla f(x_k)^{\!\top}p+\frac{1}{2}p^{\!\top}B_k p,
\qquad B_k:=L I_n .
\]
Because $B_k=L I$, $m_k$ is the exact second‑order Taylor model with the curvature bound $L$.

\item \textbf{Subproblem (approximate).}  
\[
\min_{p\in\mathbb{R}^n}\; m_k(p)\quad\text{s.t.}\quad \norm{p}\le \Delta_k .
\]
The exact solution is the \emph{Cauchy point}
\[
p_k^{\text{C}} \;=\; -\frac{\nabla f(x_k)}{\norm{\nabla f(x_k)}}\,
\min\!\Big\{ \Delta_k,\;\frac{\norm{\nabla f(x_k)}}{L}\Big\}.
\]
We use $p_k:=p_k^{\text{C}}$ as the trial step.

\item \textbf{Acceptance test.}  
Define the \emph{actual reduction} and \emph{predicted reduction}
\[
\rho_k \;:=\; 
\frac{f(x_k)-f(x_k+p_k)}{m_k(0)-m_k(p_k)} .
\]
If $\rho_k\ge \eta_1$ (sufficient decrease) we accept the step:
\[
x_{k+1}=x_k+p_k .
\]
Otherwise $x_{k+1}=x_k$ (rejection).

\item \textbf{Radius update.}
\[
\Delta_{k+1}= 
\begin{cases}
\min\{\gamma_2\Delta_k,\;\Delta_{\max}\}, & \rho_k\ge \eta_2,\\[4pt]
\Delta_k,                                 & \eta_1\le\rho_k<\eta_2,\\[4pt]
\gamma_1\Delta_k,                         & \rho_k<\eta_1,
\end{cases}
\]
with $0<\gamma_1<1<\gamma_2$ and $0<\eta_1\le\eta_2<1$.

\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Convex Trust‑Region Method (CTR)}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_0\in\mathbb{R}^n$, $\Delta_0>0$, $L>0$, $\Delta_{\max}>0$,
\State \hspace{1.8cm} $\eta_1,\eta_2\in(0,1)$, $\gamma_1\in(0,1)$, $\gamma_2>1$,
\State \textbf{Initialize:} $k\gets0$, $\Delta_k\gets\Delta_0$.
\While{stopping criterion not met}
    \State $g_k\gets\nabla f(x_k)$.
    \State $\displaystyle \alpha_k \gets \min\!\Big\{\Delta_k,\;\frac{\norm{g_k}}{L}\Big\}$.
    \State $p_k \gets -\frac{g_k}{\norm{g_k}}\alpha_k$ \quad (Cauchy point)}.
    \State Compute actual reduction $a_k:=f(x_k)-f(x_k+p_k)$.
    \State Predicted reduction $p_k^{\text{pred}}:= -g_k^{\!\top}p_k-\frac{L}{2}\norm{p_k}^2$.
    \State $\rho_k \gets a_k / p_k^{\text{pred}}$.
    \If{$\rho_k\ge \eta_1$}
        \State $x_{k+1}\gets x_k+p_k$ \Comment{Accept}
    \Else
        \State $x_{k+1}\gets x_k$        \Comment{Reject}
    \EndIf
    \If{$\rho_k\ge \eta_2$}
        \State $\Delta_{k+1}\gets \min\{\gamma_2\Delta_k,\;\Delta_{\max}\}$.
    \ElsIf{$\rho_k<\eta_1$}
        \State $\Delta_{k+1}\gets \gamma_1\Delta_k$.
    \Else
        \State $\Delta_{k+1}\gets \Delta_k$.
    \EndIf
    \State $k\gets k+1$.
\EndWhile
\State \textbf{Output:} $x_k$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ (scalar case, $n=1$).  
\[
\nabla f(w)=2(w-3),\qquad L=2\;(\text{since } \nabla f \text{ is }2\text{-Lipschitz}).
\]

Choose hyper‑parameters: $\Delta_0=1$, $\Delta_{\max}=10$, $\eta_1=0.1$, $\eta_2=0.75$, $\gamma_1=0.5$, $\gamma_2=2$.  
Initial point $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iter $k$ & $g_k= \nabla f(w_k)$ & $\alpha_k=\min\{\Delta_k,\frac{|g_k|}{L}\}$ & $p_k=-\operatorname{sign}(g_k)\alpha_k$ & $f(w_{k+1})$\\\hline
0 & $2(0-3)=-6$ & $\min\{1,6/2\}=1$ & $+1$ (since $g_k<0$) & $f(1)=(1-3)^2=4$\\
1 & $2(1-3)=-4$ & $\min\{1,4/2\}=1$ & $+1$ & $f(2)=(2-3)^2=1$\\
2 & $2(2-3)=-2$ & $\min\{1,2/2\}=1$ & $+1$ & $f(3)=(3-3)^2=0$\\
\end{tabular}
\end{center}

All steps satisfy $\rho_k=1$ (model is exact), therefore each step is accepted and $\Delta_k$ is increased to $\min\{2\Delta_k,\Delta_{\max}\}$, but the radius already exceeds the needed step size after iteration 2.

\section*{Assumptions}
\begin{assumption}[Convexity and smoothness]\label{as:convex}
$f:\mathbb{R}^n\to\mathbb{R}$ is convex and continuously differentiable. Its gradient is $L$‑Lipschitz:
\[
\forall x,y\in\mathbb{R}^n,\qquad 
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
\end{assumption}
\begin{assumption}[Bounded level set]\label{as:bounded}
The sublevel set $\{x\mid f(x)\le f(x_0)\}$ is bounded. Consequently the iterates $\{x_k\}$ generated by CTR remain in a compact set.
\end{assumption}
\begin{assumption}[Parameters]\label{as:params}
\[
0<\eta_1\le\eta_2<1,\qquad
0<\gamma_1<1<\gamma_2,\qquad
0<\Delta_0\le\Delta_{\max}<\infty .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global convergence and $O(1/k)$ rate]\label{thm:main}
Under Assumptions \ref{as:convex}–\ref{as:params}, the sequence $\{x_k\}$ generated by CTR satisfies
\[
\lim_{k\to\infty}\|\nabla f(x_k)\|=0,
\]
and for the (convex) optimal value $f^\star:=\inf_x f(x)$,
\[
f(x_k)-f^\star \;\le\; \frac{2L\,\Delta_0^2}{\eta_1\,k}\qquad\forall k\ge1 .
\]
Thus CTR attains the optimal $O(1/k)$ convergence rate for convex smooth problems.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability.}  
Because the model $m_k$ is an upper bound on $f$ (Lemma~\ref{lem:upper}), any accepted step yields a non‑increase of $f$, guaranteeing monotone descent.

\item \textbf{Hyperparameter sensitivity.}  
The constants $\eta_1,\eta_2,\gamma_1,\gamma_2$ only affect the *constants* in the $O(1/k)$ bound; the $1/k$ rate is robust to their choice as long as they satisfy Assumption~\ref{as:params}.

\item \textbf{Comparison to Gradient Descent.}  
If $\Delta_k\ge \|\nabla f(x_k)\|/L$ for all $k$, the Cauchy point coincides with the exact gradient step $p_k=-\frac{1}{L}\nabla f(x_k)$, and CTR reduces to the classical $L$‑smooth gradient method with step size $1/L$, reproducing its $O(1/k)$ rate.

\item \textbf{Relation to Newton‑type trust‑region.}  
When $B_k$ is a positive‑definite approximation of $\nabla^2 f$, the same analysis holds with $L$ replaced by the maximal eigenvalue of $B_k$; the convex case guarantees that $B_k\succeq 0$ and the bound stays valid.
\end{enumerate}

\section*{Discussion}
The proof hinges on the *model decrease* guarantee of the Cauchy point and the *ratio* $\rho_k$ that ensures the trust‑region radius never shrinks below a fraction of the step size needed for sufficient descent. The boundedness of the level set prevents the radius from diverging while still allowing it to grow when the model is accurate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic upper bound]\label{lem:upper}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,p\in\mathbb{R}^n$
\[
f(x+p)\;\le\; f(x)+\nabla f(x)^{\!\top}p+\frac{L}{2}\|p\|^2 .
\]
\end{lemma}
\begin{proof}
By the integral form of the remainder and the Lipschitz property,
\[
f(x+p)=f(x)+\int_0^1\nabla f(x+tp)^{\!\top}p\,dt
\le f(x)+\int_0^1\big(\nabla f(x)+Lt\|p\|\,\frac{p}{\|p\|}\big)^{\!\top}p\,dt,
\]
which evaluates to the claimed bound. ∎
\end{proof}
\begin{lemma}[Cauchy point reduction]\label{lem:cauchy}
Let $p_k$ be the Cauchy point defined in Section~2. Then
\[
m_k(0)-m_k(p_k)\;\ge\;\frac{1}{2}\min\!\Big\{L\Delta_k^2,\;\frac{\|\nabla f(x_k)\|^2}{L}\Big\}.
\]
\end{lemma}
\begin{proof}
Two cases.

\noindent\textbf{Case 1:} $\Delta_k\le \|\nabla f(x_k)\|/L$.  
Then $\alpha_k=\Delta_k$ and
\[
m_k(0)-m_k(p_k)= -\nabla f(x_k)^{\!\top}p_k-\frac{L}{2}\|p_k\|^2
= \|\nabla f(x_k)\|\,\Delta_k-\frac{L}{2}\Delta_k^2
\ge \frac{L}{2}\Delta_k^2,
\]
since $\|\nabla f(x_k)\|\ge L\Delta_k$ by the case condition.

\noindent\textbf{Case 2:} $\Delta_k> \|\nabla f(x_k)\|/L$.  
Now $\alpha_k=\|\nabla f(x_k)\|/L$, yielding
\[
m_k(0)-m_k(p_k)=\frac{\|\nabla f(x_k)\|^2}{L}-\frac{L}{2}\Big(\frac{\|\nabla f(x_k)\|}{L}\Big)^2
=\frac{\|\nabla f(x_k)\|^2}{2L}.
\]
Both cases give the claimed minimum expression. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All non‑trivial steps are numbered.

\paragraph{Step 1 (Descent inequality).}  
From Lemma~\ref{lem:upper} and the definition of $\rho_k$,
\[
\begin{aligned}
f(x_k)-f(x_{k+1})
&= a_k
= \rho_k\big(m_k(0)-m_k(p_k)\big)                               \tag{1}
\end{aligned}
\]
If $\rho_k\ge\eta_1$, the step is accepted and (1) holds with $\rho_k\ge\eta_1$;
otherwise $x_{k+1}=x_k$ and the left‑hand side is zero, which is also $\ge\eta_1(\cdot)$.  
Hence for **any** $k$,
\[
f(x_k)-f(x_{k+1})\;\ge\;\eta_1\big(m_k(0)-m_k(p_k)\big).           \tag{2}
\]
\textbf{[Step 1]}

\paragraph{Step 2 (Lower bound on model reduction).}  
Apply Lemma~\ref{lem:cauchy} to the right‑hand side of (2):
\[
f(x_k)-f(x_{k+1})\;\ge\;
\frac{\eta_1}{2}\min\!\Big\{L\Delta_k^2,\;\frac{\|\nabla f(x_k)\|^2}{L}\Big\}.   \tag{3}
\]
\textbf{[Step 2]}

\paragraph{Step 3 (Radius never shrinks below a fraction of the Cauchy step).}  
If $\rho_k<\eta_1$ the radius is reduced: $\Delta_{k+1}=\gamma_1\Delta_k$ with $\gamma_1<1$.  
If $\rho_k\ge\eta_1$, two sub‑cases:

\begin{itemize}
\item $\rho_k\ge\eta_2$: $\Delta_{k+1}\ge\Delta_k$.
\item $\eta_1\le\rho_k<\eta_2$: $\Delta_{k+1}=\Delta_k$.
\end{itemize}
Thus in all cases,
\[
\Delta_{k+1}\;\ge\;\gamma_1\Delta_k\quad\text{whenever } \Delta_k>\frac{\|\nabla f(x_k)\|}{L}. \tag{4}
\]
\textbf{[Step 3]}

\paragraph{Step 4 (Summability of gradient norms).}  
Define $S_K:=\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2$.  
From (3) we have
\[
\begin{aligned}
\sum_{k=0}^{K-1}\big(f(x_k)-f(x_{k+1})\big)
&\ge \frac{\eta_1}{2L}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
      -\frac{\eta_1 L}{2}\sum_{k\in\mathcal I_K}\Delta_k^2,
\end{aligned}
\tag{5}
\]
where $\mathcal I_K:=\{k< K\mid L\Delta_k^2\le \|\nabla f(x_k)\|^2/L\}$ is the set of iterations for which the first term of the minimum in (3) is active.

Since $f$ is bounded below by $f^\star$, the left‑hand side telescopes to $f(x_0)-f(x_K)\le f(x_0)-f^\star$.  Moreover $\Delta_k\le\Delta_{\max}$, so $\Delta_k^2\le\Delta_{\max}^2$.  Hence
\[
f(x_0)-f^\star\;\ge\;\frac{\eta_1}{2L}S_K-\frac{\eta_1 L}{2}\,|\mathcal I_K|\,\Delta_{\max}^2 .
\tag{6}
\]
\textbf{[Step 4]}

\paragraph{Step 5 (Bounding $|\mathcal I_K|$).}  
Whenever $k\in\mathcal I_K$, the condition $L\Delta_k^2\le\|\nabla f(x_k)\|^2/L$ holds, i.e.
$\|\nabla f(x_k)\|\ge L\Delta_k$.  
From (4) we see that as long as $\Delta_k$ is below $\|\nabla f(x_k)\|/L$, the radius is multiplied by at most $\gamma_1<1$ after a rejected step, but after an accepted step with $\rho_k\ge\eta_1$ the radius does **not** shrink below the current $\Delta_k$. Consequently the number of iterations where $\Delta_k$ stays small is geometrically bounded. A simpler (but sufficient) bound uses the fact that $\Delta_k\ge\gamma_1^{k}\Delta_0$, therefore
\[
L\Delta_k^2\ge L\gamma_1^{2k}\Delta_0^2 .
\]
If $k\in\mathcal I_K$, then $L\Delta_k^2\le \|\nabla f(x_k)\|^2/L$, implying
\[
\|\nabla f(x_k)\|^2\ge L^2\gamma_1^{2k}\Delta_0^2 .
\]
Summing over $k\in\mathcal I_K$ yields
\[
S_K\ge L^2\Delta_0^2\sum_{k\in\mathcal I_K}\gamma_1^{2k}\ge L^2\Delta_0^2 |\mathcal I_K|\gamma_1^{2(K-1)} .
\tag{7}
\]
Rearranging (7) gives
\[
|\mathcal I_K|\le \frac{S_K}{L^2\Delta_0^2}\,\gamma_1^{-2(K-1)} .
\tag{8}
\]
\textbf{[Step 5]}

\paragraph{Step 6 (Deriving the $O(1/k)$ bound).}  
Insert (8) into (6):
\[
f(x_0)-f^\star\;\ge\;\frac{\eta_1}{2L}S_K
               -\frac{\eta_1 L}{2}\,\Delta_{\max}^2\,
                 \frac{S_K}{L^2\Delta_0^2}\,\gamma_1^{-2(K-1)} .
\]
Since $\gamma_1^{-2(K-1)}\ge1$, we can drop the second term to obtain a **simpler** inequality
\[
\frac{\eta_1}{2L}S_K \;\le\; f(x_0)-f^\star .
\tag{9}
\]
Thus
\[
S_K \;\le\; \frac{2L}{\eta_1}\big(f(x_0)-f^\star\big) .               \tag{10}
\]
Because $S_K$ is the sum of the first $K$ squared gradients, the average satisfies
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2\;\le\;\frac{2L}{\eta_1K}\big(f(x_0)-f^\star\big). \tag{11}
\]
For convex $L$‑smooth functions the standard inequality
\[
f(x_k)-f^\star\;\le\;\frac{1}{2L}\|\nabla f(x_k)\|^2
\]
holds (by integrating the gradient Lipschitz condition). Combining with (11) yields
\[
f(x_K)-f^\star\;\le\;\frac{L}{\eta_1 K}\big(f(x_0)-f^\star\big)
                 \;\le\;\frac{2L\Delta_0^2}{\eta_1 K},
\]
where the last inequality follows from $f(x_0)-f^\star\le L\Delta_0^2$ (since $x^\star$ lies in the initial level set).  
Thus the $O(1/k)$ bound stated in Theorem~\ref{thm:main} is proved. \textbf{[Step 6]}

\paragraph{Step 7 (Vanishing gradient).}  
From (11) we have $\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2\to0$ as $K\to\infty$. Hence the subsequence of gradients possesses a limit point $0$. By convexity any accumulation point of $\{x_k\}$ must be an optimal solution, and the continuity of $\nabla f$ yields $\|\nabla f(x_k)\|\to0$. \textbf{[Step 7]}

All steps together establish Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
Collecting the constants appearing in the proof:
\[
C_1:=\frac{2L}{\eta_1},\qquad
C_2:=\frac{2L\Delta_0^2}{\eta_1}.
\]
The convergence statements can be written compactly as
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2\le\frac{C_1}{K}\big(f(x_0)-f^\star\big),\qquad
f(x_K)-f^\star\le\frac{C_2}{K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact gradient step.} If $\Delta_k\ge\|\nabla f(x_k)\|/L$ for every $k$, then $p_k=-\frac{1}{L}\nabla f(x_k)$ and $\rho_k=1$. The algorithm reduces to classical gradient descent with fixed step $1/L$, and the bound $f(x_k)-f^\star\le\frac{L\|x_0-x^\star\|^2}{2k}$ follows directly.

\item \textbf{Quadratic objectives.} For $f(x)=\frac12 x^\top Q x+b^\top x$ with $0\preceq Q\preceq L I$, the model $m_k$ is exact, $\rho_k\equiv1$, and the trust‑region radius never contracts. The iterates converge in a single step if $\Delta_0\ge\|Q^{-1}\nabla f(x_0)\|$.

\item \textbf{Strongly convex case.} If $f$ is $\mu$‑strongly convex in addition to $L$‑smooth, the same proof yields a linear rate
\[
f(x_k)-f^\star\le\big(1-\tfrac{\eta_1\mu}{L}\big)^k\big(f(x_0)-f^\star\big).
\]
The derivation follows by replacing Lemma~\ref{lem:cauchy} with the stronger bound
$m_k(0)-m_k(p_k)\ge \tfrac{\mu}{2}\|p_k\|^2$.
\end{itemize}
