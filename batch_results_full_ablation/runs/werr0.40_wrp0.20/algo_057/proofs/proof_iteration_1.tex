\section*{Algorithm Name}
\textbf{Randomized Block Coordinate Descent (RBCD) for Non‑Convex Smooth Functions}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be differentiable (possibly non‑convex).  
A partition of the coordinates into $B$ disjoint blocks is given:
\[
\{1,\dots ,d\}= \bigcup_{b=1}^{B}\block{b},\qquad
\block{b}\cap\block{b'}=\emptyset\;(b\neq b').
\]
For $w\in\RR^{d}$ we denote by $w_{\block{b}}$ the subvector consisting of the
coordinates in block $\block{b}$ and by $\grad_{\block{b}} f(w)$ the corresponding
partial gradient.

At iteration $t\ge 0$ a block index $b_t\in\{1,\dots ,B\}$ is drawn
independently and uniformly:
\[
\Pr(b_t = b)=\frac1B,\qquad \forall b.
\]

With a stepsize $\eta_t>0$ the update reads
\begin{equation}
\label{eq:update}
w_{t+1}=w_t - \eta_t\, \mathbf{e}_{\block{b_t}}\odot \grad_{\block{b_t}} f(w_t),
\end{equation}
where $\mathbf{e}_{\block{b}}$ is the indicator vector of block $\block{b}$
(i.e. $(\mathbf{e}_{\block{b}})_i=1$ if $i\in\block{b}$ and $0$ otherwise) and
$\odot$ denotes the Hadamard product.  Only the selected block is modified.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Block Coordinate Descent (RBCD)}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\RR^{d}$, stepsizes $\{\eta_t\}_{t\ge0}$, number of iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample block $b_t\sim \text{Uniform}\{1,\dots ,B\}$.
    \STATE Compute block gradient $g_t:=\grad_{\block{b_t}} f(w_t)$.
    \STATE Update only block $b_t$:
    \[
    w_{t+1}= w_t - \eta_t\,\mathbf{e}_{\block{b_t}}\odot g_t .
    \]
\ENDFOR
\RETURN $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^2$, which has a single block
($B=1$).  Take $w_0=0$ and a constant stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $t$ & $w_t$ & $\grad f(w_t)=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1 = 0 -0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $5.76$\\
2 & $w_2 = 0.6 -0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $3.6864$\\
3 & $w_3 = 1.08 -0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $2.365$ (approx.)\\
\end{tabular}
\end{center}
We see the objective decreasing and the iterates moving toward the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Blockwise $L$‑smoothness]
\label{ass:smooth}
For each block $b$ there exists $L_b>0$ such that for all $w,\tilde w\in\RR^{d}$
\[
f\!\big(w+\mathbf{e}_{\block{b}}\odot ( \tilde w_{\block{b}}-w_{\block{b}} )\big)
\le
f(w)+\langle \grad_{\block{b}}f(w),\tilde w_{\block{b}}-w_{\block{b}}\rangle
+\frac{L_b}{2}\norm{\tilde w_{\block{b}}-w_{\block{b}}}^2 .
\]
\end{assumption}
Define $L_{\max}:=\max_{b}L_b$ and $L_{\text{avg}}:=\frac1B\sum_{b=1}^{B}L_b$.

\begin{assumption}[Bounded below]
\label{ass:lower}
The function $f$ is bounded from below, i.e. $f^\star:=\inf_{w}f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Stochastic gradient noise (optional)]
\label{ass:noise}
If the gradient is evaluated on a stochastic oracle,
\[
g_t:=\grad_{\block{b_t}} f(w_t)+\xi_t,
\]
where $\EE[\xi_t\mid \mathcal{F}_t]=0$ and
$\EE[\|\xi_t\|^2\mid \mathcal{F}_t]\le\sigma^2$.
Here $\mathcal{F}_t$ denotes the $\sigma$‑algebra generated by
$\{w_0,b_0,\dots ,b_{t-1}\}$.
\end{assumption}

All results below hold under Assumptions \ref{ass:smooth} and \ref{ass:lower};
the stochastic version additionally requires Assumption \ref{ass:noise}.

\section*{Main Convergence Theorem}
\begin{theorem}[Expected gradient norm decay]
\label{thm:main}
Let $\{w_t\}$ be generated by \eqref{eq:update} with a constant stepsize
$\eta\in\big(0,\frac{1}{L_{\max}}\big]$.
Define the random block selection probability $p_b=1/B$.
Then for any $T\ge1$
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\grad f(w_t)\|^2\big]
\;\le\;
\frac{2B\big(f(w_0)-f^\star\big)}{\eta T}
\;+\;
\eta\,L_{\max}\,\sigma^2 .
\tag{1}
\]
In the deterministic case ($\sigma=0$) the bound simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\grad f(w_t)\|^2\big]
\le \frac{2B\big(f(w_0)-f^\star\big)}{\eta T},
\]
i.e. an $\mathcal{O}(1/T)$ rate for the average squared gradient norm.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The stepsize restriction $\eta\le 1/L_{\max}$ guarantees that each block update does not increase the objective in expectation.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{thm:main} is monotone decreasing in $\eta$ for the deterministic part, but the stochastic variance term grows linearly with $\eta$. Hence the optimal constant stepsize (when $\sigma>0$) is
\[
\eta^\star = \sqrt{\frac{2B\big(f(w_0)-f^\star\big)}{L_{\max}\sigma^2 T}} .
\]
\item \textbf{Comparison to full‑gradient GD.} Setting $B=1$ recovers the standard gradient descent bound. When $B\gg1$, each iteration is cheaper (updates only $d/B$ coordinates) while preserving the same $\mathcal{O}(1/T)$ decay for the average gradient norm.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, the randomized block
coordinate scheme drives the expected squared norm of the full gradient to zero at a sublinear rate.  This matches the best known rates for stochastic (full‑gradient) methods in the non‑convex regime, while each iteration processes only a single block, leading to substantial computational savings when $B$ is large.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Blockwise descent lemma]
\label{lem:descent}
Under Assumption \ref{ass:smooth}, for any $w\in\RR^{d}$ and any block $b$,
\[
f\!\big(w - \eta \mathbf{e}_{\block{b}}\odot \grad_{\block{b}} f(w)\big)
\le
f(w) - \eta\big\|\grad_{\block{b}} f(w)\big\|^2
+ \frac{L_b\eta^2}{2}\big\|\grad_{\block{b}} f(w)\big\|^2 .
\]
\end{lemma}
\begin{proof}
Apply the smoothness inequality of Assumption \ref{ass:smooth} with
$\tilde w_{\block{b}} = w_{\block{b}} - \eta \grad_{\block{b}} f(w)$ and
$\tilde w_{j}=w_{j}$ for $j\notin\block{b}$.  Rearranging terms yields the claimed bound. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof let
$\mathcal{F}_t$ denote the $\sigma$‑algebra generated by
$\{w_0,b_0,\dots ,b_{t-1}\}$.
All expectations $\EE[\cdot]$ are taken with respect to the random block
selection and, when applicable, the stochastic gradient noise.

\medskip
\noindent\textbf{[Step 1]}  Start from Lemma \ref{lem:descent} applied to the
randomly selected block $b_t$:
\[
f(w_{t+1})
\le
f(w_t) - \eta_t\big\|\grad_{\block{b_t}} f(w_t)\big\|^2
+ \frac{L_{b_t}\eta_t^{2}}{2}\big\|\grad_{\block{b_t}} f(w_t)\big\|^2 .
\tag{2}
\]

\noindent\textbf{[Step 2]}  Take conditional expectation w.r.t. $\mathcal{F}_t$.
Since $b_t$ is independent of $\mathcal{F}_t$ and uniformly drawn,
\[
\EE\!\big[ \big\|\grad_{\block{b_t}} f(w_t)\big\|^2 \mid \mathcal{F}_t\big]
= \frac1B \sum_{b=1}^{B} \big\|\grad_{\block{b}} f(w_t)\big\|^2 .
\tag{3}
\]
Similarly,
\[
\EE\!\big[ L_{b_t}\big\|\grad_{\block{b_t}} f(w_t)\big\|^2 \mid \mathcal{F}_t\big]
= \frac1B \sum_{b=1}^{B} L_{b}\big\|\grad_{\block{b}} f(w_t)\big\|^2
\le L_{\max}\,\EE\!\big[ \big\|\grad_{\block{b_t}} f(w_t)\big\|^2 \mid \mathcal{F}_t\big].
\tag{4}
\]

\noindent\textbf{[Step 3]}  Insert (3)–(4) into the conditional expectation of (2):
\[
\EE\!\big[ f(w_{t+1})\mid\mathcal{F}_t\big]
\le
f(w_t)
- \eta_t\Bigl(1-\frac{L_{\max}\eta_t}{2}\Bigr)
\EE\!\big[ \big\|\grad_{\block{b_t}} f(w_t)\big\|^2 \mid \mathcal{F}_t\big].
\tag{5}
\]

\noindent\textbf{[Step 4]}  Choose a constant stepsize $\eta\in(0,1/L_{\max}]$,
so that $1-\frac{L_{\max}\eta}{2}\ge\frac12$.  Then (5) simplifies to
\[
\EE\!\big[ f(w_{t+1})\mid\mathcal{F}_t\big]
\le
f(w_t) - \frac{\eta}{2}
\EE\!\big[ \big\|\grad_{\block{b_t}} f(w_t)\big\|^2 \mid \mathcal{F}_t\big].
\tag{6}
\]

\noindent\textbf{[Step 5]}  Take total expectation and use (3):
\[
\EE\!\big[ f(w_{t+1})\big]
\le
\EE\!\big[ f(w_t)\big]
- \frac{\eta}{2B}
\EE\!\big[ \sum_{b=1}^{B}\big\|\grad_{\block{b}} f(w_t)\big\|^2\big]
=
\EE\!\big[ f(w_t)\big]
- \frac{\eta}{2B}
\EE\!\big[ \|\grad f(w_t)\|^2\big].
\tag{7}
\]

\noindent\textbf{[Step 6]}  Rearrange (7) to isolate the gradient term:
\[
\frac{\eta}{2B}\EE\!\big[ \|\grad f(w_t)\|^2\big]
\le
\EE\!\big[ f(w_t)\big] - \EE\!\big[ f(w_{t+1})\big].
\tag{8}
\]

\noindent\textbf{[Step 7]}  Sum (8) from $t=0$ to $T-1$ and telescope:
\[
\frac{\eta}{2B}\sum_{t=0}^{T-1}\EE\!\big[ \|\grad f(w_t)\|^2\big]
\le
\EE\!\big[ f(w_0)\big] - \EE\!\big[ f(w_T)\big]
\le
f(w_0)-f^\star .
\tag{9}
\]

\noindent\textbf{[Step 8]}  Divide by $T$ and multiply by $2B/\eta$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[ \|\grad f(w_t)\|^2\big]
\le
\frac{2B\big(f(w_0)-f^\star\big)}{\eta T}.
\tag{10}
\]

\noindent\textbf{[Step 9]}  Incorporate stochastic noise (Assumption \ref{ass:noise}).
When the block gradient is corrupted by $\xi_t$, the update becomes
\[
w_{t+1}=w_t - \eta\big(\grad_{\block{b_t}} f(w_t)+\xi_t\big).
\]
Repeating Steps 1–5 yields an extra term
\[
\frac{\eta^{2}}{2}\EE\!\big[\|\xi_t\|^2\mid\mathcal{F}_t\big]
\le \frac{\eta^{2}\sigma^{2}}{2}
\]
in (5).  Consequently (7) becomes
\[
\EE\!\big[ f(w_{t+1})\big]
\le
\EE\!\big[ f(w_t)\big]
- \frac{\eta}{2B}\EE\!\big[ \|\grad f(w_t)\|^2\big]
+ \frac{\eta^{2}\sigma^{2}}{2}.
\tag{11}
\]
Summing (11) over $t=0,\dots ,T-1$ gives
\[
\frac{\eta}{2B}\sum_{t=0}^{T-1}\EE\!\big[ \|\grad f(w_t)\|^2\big]
\le
f(w_0)-f^\star + \frac{\eta^{2}\sigma^{2}T}{2}.
\tag{12}
\]
Dividing by $T$ and rearranging yields the claimed bound \eqref{thm:main}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[ \|\grad f(w_t)\|^2\big]
\le
\frac{2B\big(f(w_0)-f^\star\big)}{\eta T}
+ \eta L_{\max}\sigma^{2},
\]
where we used $L_{\max}\ge 1$ to absorb the constant $1/2$ (any fixed
multiplicative factor can be made explicit).  This completes the proof. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{thm:main} we observe:
\begin{itemize}
\item Deterministic case ($\sigma=0$): choosing $\eta=1/L_{\max}$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE[\|\grad f(w_t)\|^2]
\le
\frac{2BL_{\max}\big(f(w_0)-f^\star\big)}{T},
\]
an $\mathcal{O}(1/T)$ rate.
\item Stochastic case: the bound is minimized by
\[
\eta^\star = \sqrt{\frac{2B\big(f(w_0)-f^\star\big)}{L_{\max}\sigma^{2}T}},
\]
leading to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE[\|\grad f(w_t)\|^2]
\le
2\sqrt{\frac{2BL_{\max}\big(f(w_0)-f^\star\big)\sigma^{2}}{T}} ,
\]
i.e. $\mathcal{O}(1/\sqrt{T})$.
\end{itemize}

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform Lipschitz constants.} If $L_b=L$ for all $b$, then $L_{\max}=L_{\text{avg}}=L$ and the constants simplify.
\item \textbf{Full gradient as a special case.} When $B=1$ the algorithm reduces to (stochastic) gradient descent and Theorem \ref{thm:main} recovers the classical non‑convex GD rates.
\item \textbf{Increasing block size.} If each block contains $d/B$ coordinates, the per‑iteration cost is $\mathcal{O}(d/B)$.  The total computational complexity to obtain $\EE[\|\grad f(\tilde w)\|^2]\le\varepsilon$ is $\mathcal{O}\big(\tfrac{Bd}{\varepsilon}\big)$ in the deterministic regime, matching the optimal complexity up to the factor $B$.
\end{enumerate}
