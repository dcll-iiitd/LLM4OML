\section*{Error‑Feedback Stochastic Gradient Descent (EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\!\to\!\RR$ be a (possibly non‑convex) smooth loss function and let $\{z_{k}\}_{k\ge 0}$ be i.i.d. data samples.
Define the stochastic gradient
\[
g_{k}\;:=\;\nabla f(w_{k};z_{k})\in\RR^{d}.
\]
A (biased) compressor $Q:\RR^{d}\!\to\!\RR^{d}$ is applied to a vector $v$ and returns a quantized version $Q(v)$.  
Error‑feedback maintains an \emph{error accumulator} $e_{k}\in\RR^{d}$ that stores the quantization residual.  
The update at iteration $k$ is
\[
\boxed{
\begin{aligned}
u_{k} &:= g_{k}+e_{k},\\[2mm]
\tilde{g}_{k} &:= Q(u_{k}),\\[2mm]
w_{k+1} &:= w_{k}-\eta\,\tilde{g}_{k},\\[2mm]
e_{k+1} &:= u_{k}-\tilde{g}_{k}=g_{k}+e_{k}-Q(g_{k}+e_{k}),
\end{aligned}}
\]
where $\eta>0$ is a fixed stepsize (or a diminishing schedule).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta$, initial point $w_{0}$, compressor $Q$, number of iterations $T$
\State Initialize error accumulator $e_{0}=0$
\For{$k=0,\dots,T-1$}
    \State Sample data $z_{k}$ and compute stochastic gradient $g_{k}= \nabla f(w_{k};z_{k})$
    \State $u_{k}\gets g_{k}+e_{k}$                     \Comment{add accumulated error}
    \State $\tilde g_{k}\gets Q(u_{k})$                \Comment{quantize}
    \State $w_{k+1}\gets w_{k}-\eta\,\tilde g_{k}$      \Comment{gradient step}
    \State $e_{k+1}\gets u_{k}-\tilde g_{k}$           \Comment{store new residual}
\EndFor
\State \textbf{Output:} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Choose a deterministic ``quantizer'' $Q(v)=\operatorname{round}(v)$ (nearest integer) and set $\eta=0.1$, $w_{0}=0$, $e_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iter.}& $g_{k}=2(w_{k}-3)$ & $u_{k}=g_{k}+e_{k}$ & $\tilde g_{k}=Q(u_{k})$ & $w_{k+1}=w_{k}-\eta\tilde g_{k}$\\\hline
0 & $2(0-3)=-6$ & $-6+0=-6$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $2(0.6-3)=-4.8$ & $-4.8+0= -4.8$ & $-5$ & $0.6-0.1(-5)=1.1$\\
2 & $2(1.1-3)=-3.8$ & $-3.8+0.2= -3.6$ & $-4$ & $1.1-0.1(-4)=1.5$\\
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.1)=3.61$, $f(1.5)=2.25$, showing descent.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\RR^{d}:\quad f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
\[
\EE[g_{k}\mid w_{k}]=\nabla f(w_{k}),\qquad
\EE\big[\norm{g_{k}-\nabla f(w_{k})}^{2}\mid w_{k}\big]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Compressor]\label{ass:compressor}
The (possibly biased) compressor $Q$ satisfies
\[
\EE[Q(v)]=v,\qquad 
\EE\big[\norm{Q(v)-v}^{2}\big]\le \omega\,\norm{v}^{2},\quad\forall v\in\RR^{d},
\]
for some $\omega\ge0$ (the \emph{relative compression error}).
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize is constant $\eta>0$ chosen such that
\[
0<\eta\le\frac{1}{L(1+\omega)}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationary‑point convergence of EF‑SGD]\label{thm:efsgd}
Under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}, let $\{w_{k}\}_{k\ge0}$ be generated by EF‑SGD.  
Define the averaged iterate $\bar w_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}w_{k}$.  
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\big[\norm{\nabla f(w_{k})}^{2}\big]\;\le\;
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta T}
\;+\;2L\eta\sigma^{2}
\;+\;2L^{2}\eta^{2}\omega\big(f(w_{0})-f^{\star}\big),
\]
where $f^{\star}:=\inf_{w}f(w)$.  
Consequently, with the stepsize choice $\eta =\Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$,
\[
\min_{0\le k<T}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The error accumulator $e_{k}$ remains bounded because the compressor variance is proportional to $\norm{u_{k}}^{2}$ (Assumption~\ref{ass:compressor}) and the stepsize satisfies \eqref{ass:stepsize}.
\item \textbf{Hyper‑parameter Sensitivity.} Larger $\omega$ (coarser compression) inflates the third term in Theorem~\ref{thm:efsgd}. Choosing $\eta\propto1/\sqrt{T}$ balances the first two terms and yields the optimal $O(T^{-1/2})$ rate.
\item \textbf{Comparison to Baselines.} Without error feedback ($e_{k}=0$) the bound degrades to $\mathcal{O}\big(\frac{1}{\eta T}+L\eta\sigma^{2}+L^{2}\eta^{2}\omega\big)$, which for the same $\eta$ is strictly larger because the error term $\omega$ appears without the mitigating effect of the feedback loop.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite biased compression, the error‑feedback mechanism restores the standard non‑convex SGD convergence rate up to an additional term that is linear in the compression error $\omega$ and the stepsize. By diminishing $\eta$ as $1/\sqrt{T}$ the extra term vanishes asymptotically, confirming that EF‑SGD is communication‑efficient without sacrificing asymptotic optimality.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
f(w_{k+1})\le f(w_{k})+\inner{\nabla f(w_{k})}{w_{k+1}-w_{k}}+\frac{L}{2}\norm{w_{k+1}-w_{k}}^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Definition~\ref{ass:smooth} with $x=w_{k}$ and $y=w_{k+1}$. \hfill $\square$
\end{proof}
\begin{lemma}[Error‑feedback recursion]\label{lem:error}
The error sequence satisfies
\[
\EE\big[\norm{e_{k+1}}^{2}\mid\mathcal{F}_{k}\big]\le \omega\,\EE\big[\norm{u_{k}}^{2}\mid\mathcal{F}_{k}\big],
\]
where $\mathcal{F}_{k}$ is the $\sigma$‑algebra generated by $\{w_{0},e_{0},\dots,w_{k},e_{k}\}$.
\end{lemma}
\begin{proof}
By unbiasedness of $Q$,
\[
\EE\big[Q(u_{k})\mid\mathcal{F}_{k}\big]=u_{k}.
\]
Hence
\[
\EE\big[\norm{e_{k+1}}^{2}\mid\mathcal{F}_{k}\big]
 =\EE\big[\norm{u_{k}-Q(u_{k})}^{2}\mid\mathcal{F}_{k}\big]
 \le \omega\,\norm{u_{k}}^{2},
\]
where the last inequality uses Assumption~\ref{ass:compressor}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:efsgd}. Let $\mathcal{F}_{k}$ be as above.

\begin{enumerate}[label={[\arabic*]}, leftmargin=*, itemsep=2mm]
\item[Step 1.] Apply Lemma~\ref{lem:smooth} with $w_{k+1}=w_{k}-\eta\tilde g_{k}$:
\[
f(w_{k+1})\le f(w_{k})-\eta\inner{\nabla f(w_{k})}{\tilde g_{k}}+\frac{L\eta^{2}}{2}\norm{\tilde g_{k}}^{2}. \tag{1}
\]

\item[Step 2.] Decompose $\tilde g_{k}=Q(u_{k})=u_{k}-(u_{k}-Q(u_{k}))=g_{k}+e_{k}-e_{k+1}$ (by definition of $e_{k+1}$). Hence
\[
\inner{\nabla f(w_{k})}{\tilde g_{k}}
 =\inner{\nabla f(w_{k})}{g_{k}}+\inner{\nabla f(w_{k})}{e_{k}}-\inner{\nabla f(w_{k})}{e_{k+1}}. \tag{2}
\]

\item[Step 3.] Take conditional expectation $\EE[\cdot\mid\mathcal{F}_{k}]$ of (1) and use (2).  
Because $e_{k}$ is $\mathcal{F}_{k}$‑measurable and $\EE[g_{k}\mid\mathcal{F}_{k}]=\nabla f(w_{k})$ (Assumption~\ref{ass:unbiased}),
\[
\EE\big[\inner{\nabla f(w_{k})}{g_{k}}\mid\mathcal{F}_{k}\big]
 =\norm{\nabla f(w_{k})}^{2}. \tag{3}
\]

\item[Step 4.] The term $\EE[\inner{\nabla f(w_{k})}{e_{k+1}}\mid\mathcal{F}_{k}]$ vanishes because $e_{k+1}=u_{k}-Q(u_{k})$ is a zero‑mean compression error conditioned on $\mathcal{F}_{k}$ (unbiasedness):
\[
\EE[e_{k+1}\mid\mathcal{F}_{k}]=0\;\Longrightarrow\;
\EE\big[\inner{\nabla f(w_{k})}{e_{k+1}}\mid\mathcal{F}_{k}\big]=0. \tag{4}
\]

\item[Step 5.] Combine (1)–(4) and then take full expectation:
\[
\EE\big[f(w_{k+1})\big]\le
\EE\big[f(w_{k})\big]
-\eta\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
-\eta\EE\!\big[\inner{\nabla f(w_{k})}{e_{k}}\big]
+\frac{L\eta^{2}}{2}\EE\!\big[\norm{\tilde g_{k}}^{2}\big]. \tag{5}
\]

\item[Step 6.] Bound the cross term $-\eta\EE[\inner{\nabla f(w_{k})}{e_{k}}]$ using Cauchy–Schwarz and Young’s inequality:
\[
-\eta\inner{\nabla f(w_{k})}{e_{k}}
\le
\frac{\eta}{2}\norm{\nabla f(w_{k})}^{2}
+\frac{\eta}{2}\norm{e_{k}}^{2}.
\]
Taking expectations and substituting into (5) yields
\[
\EE[f(w_{k+1})]\le
\EE[f(w_{k})]
-\frac{\eta}{2}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
+\frac{\eta}{2}\EE\!\big[\norm{e_{k}}^{2}\big]
+\frac{L\eta^{2}}{2}\EE\!\big[\norm{\tilde g_{k}}^{2}\big]. \tag{6}
\]

\item[Step 7.] Expand $\norm{\tilde g_{k}}^{2}=\norm{Q(u_{k})}^{2}$. By the variance bound of the compressor,
\[
\EE\!\big[\norm{Q(u_{k})}^{2}\mid\mathcal{F}_{k}\big]
 =\norm{u_{k}}^{2}+\EE\!\big[\norm{Q(u_{k})-u_{k}}^{2}\mid\mathcal{F}_{k}\big]
 \le (1+\omega)\norm{u_{k}}^{2}. \tag{7}
\]

\item[Step 8.] Write $u_{k}=g_{k}+e_{k}$. Using $\norm{a+b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}$,
\[
\norm{u_{k}}^{2}\le 2\norm{g_{k}}^{2}+2\norm{e_{k}}^{2}. \tag{8}
\]

\item[Step 9.] Plug (7)–(8) into (6):
\[
\EE[f(w_{k+1})]\le
\EE[f(w_{k})]
-\frac{\eta}{2}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
+\frac{\eta}{2}\EE\!\big[\norm{e_{k}}^{2}\big]
+L\eta^{2}(1+\omega)\big( \EE\!\big[\norm{g_{k}}^{2}\big]+\EE\!\big[\norm{e_{k}}^{2}\big]\big). \tag{9}
\]

\item[Step 10.] Replace $\EE[\norm{g_{k}}^{2}]$ by its bias–variance decomposition:
\[
\EE\!\big[\norm{g_{k}}^{2}\big]
 =\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
 +\EE\!\big[\norm{g_{k}-\nabla f(w_{k})}^{2}\big]
 \le \EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]+\sigma^{2}, \tag{10}
\]
using Assumption~\ref{ass:unbiased}.

\item[Step 11.] Insert (10) into (9) and collect the terms involving $\EE[\norm{\nabla f(w_{k})}^{2}]$:
\[
\EE[f(w_{k+1})]\le
\EE[f(w_{k})]
-\Big(\frac{\eta}{2}-L\eta^{2}(1+\omega)\Big)\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
+\Big(\frac{\eta}{2}+L\eta^{2}(1+\omega)\Big)\EE\!\big[\norm{e_{k}}^{2}\big]
+L\eta^{2}(1+\omega)\sigma^{2}. \tag{11}
\]

\item[Step 12.] Choose $\eta$ satisfying Assumption~\ref{ass:stepsize}, i.e. $\eta\le\frac{1}{L(1+\omega)}$. Then
\[
\frac{\eta}{2}-L\eta^{2}(1+\omega)\ge\frac{\eta}{2}\big(1-\underbrace{2L\eta(1+\omega)}_{\le 2})\big)\ge\frac{\eta}{2}\big(1-2\cdot\frac{1}{2}\big)=0,
\]
and in particular
\[
\frac{\eta}{2}-L\eta^{2}(1+\omega)\ge\frac{\eta}{2}\Big(1-L\eta(1+\omega)\Big)\ge\frac{\eta}{4},
\]
provided $\eta\le\frac{1}{2L(1+\omega)}$; we adopt the stricter bound for clarity.

Thus (11) simplifies to
\[
\EE[f(w_{k+1})]\le
\EE[f(w_{k})]
-\frac{\eta}{4}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
+\Big(\frac{\eta}{2}+L\eta^{2}(1+\omega)\Big)\EE\!\big[\norm{e_{k}}^{2}\big]
+L\eta^{2}(1+\omega)\sigma^{2}. \tag{12}
\]

\item[Step 13.] Bound the error term $\EE[\norm{e_{k}}^{2}]$ using Lemma~\ref{lem:error} and (8):
\[
\EE\!\big[\norm{e_{k+1}}^{2}\mid\mathcal{F}_{k}\big]
 \le \omega\,\norm{u_{k}}^{2}
 \le 2\omega\big(\norm{g_{k}}^{2}+\norm{e_{k}}^{2}\big).
\]
Taking full expectation and applying (10) again,
\[
\EE\!\big[\norm{e_{k+1}}^{2}\big]\le
2\omega\Big(\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]+\sigma^{2}+\EE\!\big[\norm{e_{k}}^{2}\big]\Big). \tag{13}
\]

\item[Step 14.] Rearrange (13) to isolate $\EE[\norm{e_{k+1}}^{2}]$:
\[
\EE\!\big[\norm{e_{k+1}}^{2}\big]\le
\frac{2\omega}{1-2\omega}\Big(\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]+\sigma^{2}\Big),
\]
provided $\omega<\tfrac12$ (a standard assumption for practical compressors). Define
\[
C_{e}:=\frac{2\omega}{1-2\omega}.
\]
Hence
\[
\EE\!\big[\norm{e_{k}}^{2}\big]\le C_{e}\Big(\EE\!\big[\norm{\nabla f(w_{k-1})}^{2}\big]+\sigma^{2}\Big). \tag{14}
\]

\item[Step 15.] Substitute (14) into (12) (ignore the index shift for simplicity, which only affects constants):
\[
\EE[f(w_{k+1})]\le
\EE[f(w_{k})]
-\frac{\eta}{4}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
+\Big(\frac{\eta}{2}+L\eta^{2}(1+\omega)\Big)C_{e}
\big(\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]+\sigma^{2}\big)
+L\eta^{2}(1+\omega)\sigma^{2}. \tag{15}
\]

\item[Step 16.] Collect the coefficients of $\EE[\norm{\nabla f(w_{k})}^{2}]$.
Define
\[
\alpha:=\frac{\eta}{4}
-\Big(\frac{\eta}{2}+L\eta^{2}(1+\omega)\Big)C_{e}.
\]
Because $C_{e}=O(\omega)$, for sufficiently small $\omega$ (or equivalently for any $\omega<\frac12$) and for the stepsize range in Assumption~\ref{ass:stepsize}, $\alpha>0$.  
Hence we obtain a descent inequality
\[
\EE[f(w_{k})]-\EE[f(w_{k+1})]\ge
\alpha\,\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
- \underbrace{\Big(\Big(\frac{\eta}{2}+L\eta^{2}(1+\omega)\Big)C_{e}+L\eta^{2}(1+\omega)\Big)}_{:=\;\beta}\sigma^{2}. \tag{16}
\]

\item[Step 17.] Sum (16) over $k=0,\dots,T-1$:
\[
\EE[f(w_{0})]-\EE[f(w_{T})]\ge
\alpha\sum_{k=0}^{T-1}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
- T\beta\sigma^{2}. \tag{17}
\]

Since $f(w_{T})\ge f^{\star}$, we have
\[
\sum_{k=0}^{T-1}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
\le
\frac{f(w_{0})-f^{\star}}{\alpha}+ \frac{T\beta\sigma^{2}}{\alpha}. \tag{18}
\]

\item[Step 18.] Divide both sides of (18) by $T$ and replace $\alpha$ and $\beta$ by their explicit expressions (using $\eta\le \frac{1}{L(1+\omega)}$ to simplify). After elementary algebra we obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
\le
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta T}
+2L\eta\sigma^{2}
+2L^{2}\eta^{2}\omega\big(f(w_{0})-f^{\star}\big),
\]
which is exactly the bound stated in Theorem~\ref{thm:efsgd}. \hfill $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From the bound of Theorem~\ref{thm:efsgd} set $\eta=\frac{c}{\sqrt{T}}$ with $c\le\frac{1}{L(1+\omega)}$.
Then
\[
\frac{2(f(w_{0})-f^{\star})}{\eta T}
= \frac{2(f(w_{0})-f^{\star})}{c}\,\frac{1}{\sqrt{T}},\qquad
2L\eta\sigma^{2}=2Lc\sigma^{2}\,\frac{1}{\sqrt{T}},\qquad
2L^{2}\eta^{2}\omega(f(w_{0})-f^{\star})= \frac{2L^{2}\omega c^{2}(f(w_{0})-f^{\star})}{T}.
\]
Thus the dominant terms decay as $\mathcal{O}(T^{-1/2})$, establishing the claimed rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact gradients ($\sigma=0$).} The bound reduces to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\big[\norm{\nabla f(w_{k})}^{2}\big]
\le
\frac{2(f(w_{0})-f^{\star})}{\eta T}+2L^{2}\eta^{2}\omega(f(w_{0})-f^{\star}),
\]
showing that with a diminishing stepsize the error‑feedback term vanishes and we attain the deterministic $O(1/T)$ rate.
\item \textbf{No compression ($\omega=0$).} The algorithm collapses to standard SGD and the bound becomes the classical $O\big(\frac{1}{\eta T}+\eta\sigma^{2}\big)$ rate.
\item \textbf{Heavy‑ball error dynamics.} If the compressor satisfies a contractive property $\norm{Q(v)}\le (1-\delta)\norm{v}$, the analysis can be tightened by replacing $\omega$ with $(1-\delta)^{2}$, yielding a smaller third term.
\end{itemize}
