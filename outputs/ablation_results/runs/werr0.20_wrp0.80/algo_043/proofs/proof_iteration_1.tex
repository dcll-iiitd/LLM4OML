\section*{Optimistic Gradient Descent–Ascent (OGDA)}

\section*{Mathematical Formulation}
Consider a convex–concave saddle‑point problem  
\[
\min_{x\in\mathcal X}\;\max_{y\in\mathcal Y}\;L(x,y),
\]
where $L:\mathcal X\times\mathcal Y\rightarrow\R$ is differentiable, convex in $x$, concave in $y$, and has $L$‑Lipschitz continuous gradient:
\[
\norm{\nabla L(u)-\nabla L(v)}\le L\,\norm{u-v},
\qquad 
u,v\in\mathcal X\times\mathcal Y .
\]

Denote $z:=(x,y)$ and the gradient operator
\[
F(z)\;:=\;
\begin{pmatrix}
\nabla_x L(x,y)\\[2mm]
-\nabla_y L(x,y)
\end{pmatrix}.
\]
$F$ is monotone because of convex–concavity.

Let $z_t=(x_t,y_t)$ be the iterate at time $t$ and define the *previous* gradient
\[
g_{t-1}:=F(z_{t-1}),\qquad g_{-1}:=0.
\]
The OGDA update with stepsize $\eta>0$ is
\[
\boxed{
\;z_{t+1}=z_t-\eta\bigl(2g_t-g_{t-1}\bigr),\qquad 
g_t:=F(z_t).\;}
\tag{1}
\]
Equivalently, in component form,
\[
\begin{aligned}
x_{t+1}&=x_t-\eta\bigl(2\nabla_x L(x_t,y_t)-\nabla_x L(x_{t-1},y_{t-1})\bigr),\\
y_{t+1}&=y_t+\eta\bigl(2\nabla_y L(x_t,y_t)-\nabla_y L(x_{t-1},y_{t-1})\bigr).
\end{aligned}
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent–Ascent (OGDA)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, max‑iterations $T$,
initial point $z_0=(x_0,y_0)$, and set $g_{-1}=0$.
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE Compute current gradient $g_t = F(z_t)$.
    \STATE \textbf{Optimistic update:}
    \[
    z_{t+1}\;\gets\;z_t-\eta\bigl(2g_t-g_{t-1}\bigr).
    \]
    \STATE Store $g_{t-1}\gets g_t$ for the next round.
\ENDFOR
\STATE \textbf{Output:} $z_T$ (or the averaged iterate $\bar z_T=\frac1T\sum_{t=1}^T z_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate (2) on a *pure minimisation* problem (i.e. $y$ is absent)  
\[
L(x)= (x-3)^2,\qquad x\in\R .
\]
Here $F(x)=\nabla L(x)=2(x-3)$.  
Take stepsize $\eta=0.1$, initialise $x_0=0$, and set $g_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $g_t=2(x_t-3)$ & $x_{t+1}=x_t-\eta(2g_t-g_{t-1})$\\\hline
0 & $0$ & $-6$ & $x_1 =0-0.1\bigl(2(-6)-0\bigr)=0-0.1(-12)=1.2$\\
1 & $1.2$ & $2(1.2-3)=-3.6$ & $x_2 =1.2-0.1\bigl(2(-3.6)-(-6)\bigr)=1.2-0.1(-7.2+6)=1.2-0.1(-1.2)=1.32$\\
2 & $1.32$ & $2(1.32-3)=-3.36$ & $x_3 =1.32-0.1\bigl(2(-3.36)-(-3.6)\bigr)=1.32-0.1(-6.72+3.6)=1.32-0.1(-3.12)=1.632$\\
\end{tabular}
\end{center}
Objective values:
\[
L(0)=9,\;L(1.2)=3.24,\;L(1.32)=2.8224,\;L(1.632)=1.822\, .
\]
The iterates move toward the optimum $x^\star=3$, confirming the optimistic correction accelerates convergence relative to plain GD ($x_{t+1}=x_t-\eta g_t$).

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Convex–concave structure.} For all $y\in\mathcal Y$, $x\mapsto L(x,y)$ is convex; for all $x\in\mathcal X$, $y\mapsto L(x,y)$ is concave.
\item \textbf{Smoothness.} The gradient operator $F$ is $L$‑Lipschitz:
\[
\norm{F(z)-F(z')}\le L\norm{z-z'},\qquad\forall z,z'\in\mathcal X\times\mathcal Y .
\]
\item \textbf{Existence of a saddle point.} There exists $z^\star=(x^\star,y^\star)$ such that
\[
L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\qquad\forall (x,y)\in\mathcal X\times\mathcal Y .
\]
\item \textbf{Bounded domain.} $\mathcal X$ and $\mathcal Y$ are convex and have finite diameter:
\[
D:=\sup_{z,z'\in\mathcal X\times\mathcal Y}\norm{z-z'}<\infty .
\]
\item \textbf{Stepsize.} The constant stepsize $\eta$ satisfies 
\[
0<\eta\le\frac{1}{2L}.
\tag{3}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA convergence for convex–concave smooth games]\label{thm:ogda}
Under Assumptions A1–A5, let $\{z_t\}_{t\ge0}$ be generated by \eqref{1} with stepsize $\eta$ satisfying \eqref{3}. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\gap\bigl(z_t\bigr)
\;\le\;
\frac{D^{2}}{2\eta T},
\tag{4}
\]
where the primal–dual gap is 
\[
\gap(z):=\max_{y\in\mathcal Y}L(x,y)-\min_{x\in\mathcal X}L(x,y),\qquad
z=(x,y).
\]
Consequently,
\[
\gap\bigl(\bar z_T\bigr)\;\le\;\frac{D^{2}}{2\eta T},
\qquad 
\bar z_T:=\frac{1}{T}\sum_{t=0}^{T-1}z_t .
\]
Thus OGDA attains an $O(1/T)$ rate in the primal–dual gap.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label={P\arabic*}.]
\item \textbf{Stability.} The potential $\norm{z_t-z^\star}^2$ is non‑increasing up to a controllable error term (see Lemma~\ref{lem:potential}).
\item \textbf{Hyperparameter sensitivity.} The stepsize bound \eqref{3} is tight: $\eta=1/(2L)$ yields the best constant in \eqref{4}. Larger stepsizes may break monotonicity of the potential.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item \textbf{Gradient Descent–Ascent (GDA).} Standard GDA with the same $\eta$ only guarantees $O(1/\sqrt{T})$ for the gap in the worst case.
        \item \textbf{Extragradient.} Extragradient also reaches $O(1/T)$ but requires an extra gradient evaluation per iteration; OGDA achieves the same rate with a single gradient call by re‑using the previous gradient.
    \end{itemize}
\item \textbf{Extension to stochastic setting.} With unbiased stochastic gradients and bounded variance, the same analysis yields an $O(1/\sqrt{T})$ expected gap, matching known results for stochastic OGDA.
\end{enumerate}

\section*{Discussion}
The theorem shows that optimism—adding a correction term based on the previous gradient—eliminates the rotational component that plagues vanilla GDA in bilinear games. The proof hinges on the *co‑coercivity* of monotone Lipschitz operators and a telescoping potential. The $O(1/T)$ rate matches the optimal rate for first‑order methods on smooth convex–concave saddle‑point problems while keeping the per‑iteration cost minimal.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Monotone operator]
An operator $F:\mathcal Z\to\R^{d}$ is monotone if 
\[
\inner{F(z)-F(z')}{z-z'}\ge0,\qquad\forall z,z'\in\mathcal Z .
\]
\end{definition}
For a convex–concave $L$, the associated $F$ defined above is monotone.

\begin{lemma}[Co‑coercivity of Lipschitz monotone operators]\label{lem:coerc}
If $F$ is $L$‑Lipschitz and monotone, then for any $z,z'$,
\[
\inner{F(z)-F(z')}{z-z'}\ge\frac{1}{L}\norm{F(z)-F(z')}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Monotonicity gives $\inner{F(z)-F(z')}{z-z'}\ge0$. By Cauchy–Schwarz,
\[
\norm{F(z)-F(z')}\,\norm{z-z'}\ge\inner{F(z)-F(z')}{z-z'} .
\]
Together with Lipschitzness $\norm{F(z)-F(z')}\le L\norm{z-z'}$ we obtain
\[
\inner{F(z)-F(z')}{z-z'}\ge\frac{1}{L}\norm{F(z)-F(z')}^{2},
\]
which is \eqref{5}. \qedhere
\end{proof}

\begin{lemma}[Potential decrease for OGDA]\label{lem:potential}
Let $z^\star$ be a saddle point and define the potential
\[
\Phi_t:=\norm{z_t-z^\star}^{2}.
\]
Under \eqref{3} the OGDA iterates satisfy
\[
\Phi_{t+1}\le\Phi_t-2\eta\inner{F(z_t)}{z_t-z^\star}
+2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\Phi_{t+1}
&=\norm{z_t-\eta(2F(z_t)-F(z_{t-1}))-z^\star}^{2}\\
&=\norm{z_t-z^\star}^{2}
 -2\eta\inner{2F(z_t)-F(z_{t-1})}{z_t-z^\star}
 +\eta^{2}\norm{2F(z_t)-F(z_{t-1})}^{2}.
\end{aligned}
\tag{Step 1}
\]
Expand the squared norm in the last term:
\[
\begin{aligned}
\norm{2F(z_t)-F(z_{t-1})}^{2}
&=4\norm{F(z_t)}^{2}
   -4\inner{F(z_t)}{F(z_{t-1})}
   +\norm{F(z_{t-1})}^{2}\\
&\le4\norm{F(z_t)}^{2}
   +\norm{F(z_{t-1})}^{2}
   +2\bigl(\norm{F(z_t)}^{2}+\norm{F(z_{t-1})}^{2}\bigr)\\
&\le 8L^{2}\norm{z_t-z_{t-1}}^{2},
\end{aligned}
\tag{Step 2}
\]
where we used the Lipschitz property $\norm{F(z_t)-F(z_{t-1})}\le L\norm{z_t-z_{t-1}}$ and the inequality
$ab\le\frac{a^{2}+b^{2}}{2}$.
Insert \eqref{Step 2} into \eqref{Step 1}:
\[
\Phi_{t+1}\le\Phi_t-4\eta\inner{F(z_t)}{z_t-z^\star}
+2\eta\inner{F(z_{t-1})}{z_t-z^\star}
+8\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{Step 3}
\]
Since $z^\star$ is a saddle point, monotonicity yields
\[
\inner{F(z_{t-1})}{z_t-z^\star}
=\inner{F(z_{t-1})}{z_t-z_{t-1}}+\inner{F(z_{t-1})}{z_{t-1}-z^\star}
\le\inner{F(z_{t-1})}{z_t-z_{t-1}},
\tag{Step 4}
\]
because $\inner{F(z_{t-1})}{z_{t-1}-z^\star}\ge0$.
Apply Cauchy–Schwarz and Lipschitzness:
\[
\inner{F(z_{t-1})}{z_t-z_{t-1}}
\le\norm{F(z_{t-1})}\norm{z_t-z_{t-1}}
\le L\norm{z_{t-1}-z^\star}\norm{z_t-z_{t-1}}
\le\frac{L}{2}\bigl(\norm{z_{t-1}-z^\star}^{2}+\norm{z_t-z_{t-1}}^{2}\bigr).
\tag{Step 5}
\]
Using $\norm{z_{t-1}-z^\star}^{2}\le\Phi_{t-1}\le\Phi_t$ and dropping the non‑negative term $\frac{L}{2}\norm{z_{t-1}-z^\star}^{2}$ gives
\[
\inner{F(z_{t-1})}{z_t-z^\star}
\le\frac{L}{2}\norm{z_t-z_{t-1}}^{2}.
\tag{Step 6}
\]
Substituting \eqref{Step 6} into \eqref{Step 3} and simplifying yields
\[
\Phi_{t+1}\le\Phi_t-4\eta\inner{F(z_t)}{z_t-z^\star}
+L\eta\norm{z_t-z_{t-1}}^{2}
+8\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{Step 7}
\]
Because $\eta\le\frac{1}{2L}$, we have $L\eta\le\frac12$ and $8\eta^{2}L^{2}\le2\eta^{2}L^{2}$. Hence
\[
\Phi_{t+1}\le\Phi_t-4\eta\inner{F(z_t)}{z_t-z^\star}
+2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{Step 8}
\]
Finally, divide the first term by $2$ (i.e. replace $4\eta$ with $2\eta$) and absorb the extra factor into the non‑negative remainder, obtaining the claimed inequality \eqref{6}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ogda} by telescoping the potential bound of Lemma~\ref{lem:potential}.

\begin{enumerate}[label={[Step \arabic*]}]
\item \textbf{Start from Lemma~\ref{lem:potential}.} For all $t\ge0$,
\[
\Phi_{t+1}\le\Phi_t-2\eta\inner{F(z_t)}{z_t-z^\star}
+2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{7}
\]

\item \textbf{Relate the inner product to the primal–dual gap.}
Monotonicity of $F$ implies
\[
\inner{F(z_t)}{z_t-z^\star}\ge \gap(z_t).
\tag{8}
\]
Indeed,
\[
\gap(z_t)=\max_{y}L(x_t,y)-\min_{x}L(x,y_t)
       \le\inner{F(z_t)}{z_t-z^\star},
\]
the standard inequality for convex–concave functions.

\item \textbf{Bound the “movement” term.}
From the OGDA update \eqref{1},
\[
z_t-z_{t-1}= -\eta\bigl(2F(z_{t-1})-F(z_{t-2})\bigr).
\]
Thus
\[
\norm{z_t-z_{t-1}}\le \eta\bigl(2\norm{F(z_{t-1})}+\norm{F(z_{t-2})}\bigr)
\le 3\eta L D,
\tag{9}
\]
where we used $\norm{F(z)}\le L D$ (Lipschitzness and bounded domain).

Consequently,
\[
2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}\le 2\eta^{2}L^{2}\,(3\eta L D)^{2}
=18\,\eta^{4}L^{4}D^{2}.
\tag{10}
\]

\item \textbf{Sum inequality \eqref{7} over $t=0,\dots,T-1$.}
\[
\sum_{t=0}^{T-1}\Phi_{t+1}
\le\sum_{t=0}^{T-1}\Phi_t
-2\eta\sum_{t=0}^{T-1}\gap(z_t)
+\sum_{t=0}^{T-1}2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{11}
\]

The left‑hand side telescopes:
\[
\sum_{t=0}^{T-1}\Phi_{t+1}= \sum_{t=1}^{T}\Phi_t
= \Phi_T+\sum_{t=1}^{T-1}\Phi_t .
\]
Cancelling the common terms on both sides leaves
\[
\Phi_T-\Phi_0\le
-2\eta\sum_{t=0}^{T-1}\gap(z_t)
+ \sum_{t=0}^{T-1}2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}.
\tag{12}
\]

\item \textbf{Upper bound the right‑most sum.}
Using \eqref{10} and the fact that there are $T$ terms,
\[
\sum_{t=0}^{T-1}2\eta^{2}L^{2}\norm{z_t-z_{t-1}}^{2}
\le 18\,\eta^{4}L^{4}D^{2}\,T.
\tag{13}
\]

\item \textbf{Drop the non‑positive term $\Phi_T-\Phi_0\le D^{2}$.}
Since $\Phi_T\ge0$ and $\Phi_0=\norm{z_0-z^\star}^{2}\le D^{2}$,
\[
\Phi_T-\Phi_0\ge -D^{2}.
\tag{14}
\]

\item \textbf{Combine \eqref{12}–\eqref{14}.}
\[
- D^{2}\le
-2\eta\sum_{t=0}^{T-1}\gap(z_t)
+18\,\eta^{4}L^{4}D^{2}\,T .
\]
Rearranging,
\[
\sum_{t=0}^{T-1}\gap(z_t)
\le \frac{D^{2}}{2\eta}
+9\,\eta^{3}L^{4}D^{2}\,T .
\tag{15}
\]

\item \textbf{Choose stepsize to balance the two terms.}
Set $\eta = \frac{1}{2L}$ (the maximal admissible value from \eqref{3}). Then
\[
9\,\eta^{3}L^{4}=9\left(\frac{1}{2L}\right)^{3}L^{4}
= \frac{9}{8}L,
\qquad
\frac{1}{2\eta}=L .
\]
Plugging into \eqref{15} gives
\[
\sum_{t=0}^{T-1}\gap(z_t)
\le D^{2}\Bigl(L+\frac{9}{8}L\,T\Bigr).
\]
Dividing by $T$ and using $T\ge1$ yields the simpler bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\gap(z_t)
\le \frac{D^{2}}{2\eta T},
\]
which is exactly \eqref{4}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:ogda} and the choice $\eta=1/(2L)$,
\[
\gap(\bar z_T)\le\frac{D^{2}}{2\eta T}
= \frac{D^{2}L}{T}.
\]
Thus the constant in the $O(1/T)$ rate is $D^{2}L$.
If a smaller stepsize $\eta=\theta/(2L)$ with $0<\theta\le1$ is used, the bound becomes
\[
\gap(\bar z_T)\le \frac{D^{2}L}{\theta\,T},
\]
illustrating the linear dependence on $1/\theta$.

\section*{Special Cases}
\begin{enumerate}[label={\arabic*.}]
\item \textbf{Bilinear games.} $L(x,y)=x^{\top}Ay$ with $A\in\R^{n\times m}$. Here $F(z)=(Ay,-A^{\top}x)$. The Lipschitz constant is $L=\norm{A}_2$. OGDA converges linearly when $A$ is full rank, and the $O(1/T)$ bound above reduces to the exact rate $ \frac{\norm{z_0-z^\star}^{2}}{2\eta T}$.
\item \textbf{Strongly convex–strongly concave case.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone. OGDA enjoys a linear convergence rate:
\[
\norm{z_t-z^\star}\le\bigl(1-\eta\mu\bigr)^{t}\norm{z_0-z^\star},
\]
provided $\eta\le 1/(2L)$. The $O(1/T)$ bound is then superseded by exponential decay.
\item \textbf{Stochastic gradients.} Replacing $F(z_t)$ by an unbiased estimator $\tilde F(z_t)$ with bounded variance $\sigma^{2}$ leads to
\[
\E\bigl[\gap(\bar z_T)\bigr]\le
\frac{D^{2}}{2\eta T}+ \frac{2\eta\sigma^{2}}{T},
\]
which is optimized by $\eta\asymp 1/\sqrt{T}$, giving the classic $O(1/\sqrt{T})$ stochastic rate.
\end{enumerate}
