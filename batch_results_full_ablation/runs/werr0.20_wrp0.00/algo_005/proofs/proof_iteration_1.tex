%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) with Double Loop}  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x)$ where each $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is convex, $L$‑smooth and $\mu$‑strongly convex.  
Denote by $x^\star=\arg\min_x f(x)$.  

SVRG proceeds in \emph{epochs} $s=1,2,\dots,S$.  
At the beginning of epoch $s$ a \emph{reference point} $\tilde{x}^{\,s-1}$ is fixed and its \emph{full gradient}
\[
\tilde{g}^{\,s-1}\;:=\;\nabla f(\tilde{x}^{\,s-1}) \;=\; \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s-1})
\tag{1}
\]
is computed.  

Inside epoch $s$ we run $m$ inner stochastic updates.  
For $t=0,1,\dots,m-1$ let $x_t^{\,s}$ be the inner iterate (with $x_0^{\,s}= \tilde{x}^{\,s-1}$) and draw an index $i_t\!\in\!\{1,\dots,n\}$ uniformly at random.  
Define the \emph{variance‑reduced estimator}
\[
v_t^{\,s}\;:=\;\nabla f_{i_t}\!\bigl(x_t^{\,s}\bigr)\;-\;\nabla f_{i_t}\!\bigl(\tilde{x}^{\,s-1}\bigr)\;+\;\tilde{g}^{\,s-1}.
\tag{2}
\]
The inner update is
\[
x_{t+1}^{\,s}\;=\;x_t^{\,s}\;-\;\eta\, v_t^{\,s},
\tag{3}
\]
where $\eta>0$ is a constant stepsize.  

After $m$ inner steps we set the next reference point as the last inner iterate:
\[
\tilde{x}^{\,s}\;:=\;x_{m}^{\,s}.
\tag{4}
\]

The algorithm returns $\tilde{x}^{\,S}$ (or a random inner iterate).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (double‑loop)}\label{alg:svrg}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, inner length $m$, epochs $S$, initial point $\tilde{x}^{\,0}$
\FOR{epoch $s=1$ to $S$}
    \STATE Compute full gradient $\displaystyle \tilde{g}^{\,s-1}= \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s-1})$ \hfill\textbf{(outer loop)}
    \STATE $x_0^{\,s}\gets \tilde{x}^{\,s-1}$
    \FOR{inner step $t=0$ to $m-1$}
        \STATE Sample $i_t\sim \text{Unif}\{1,\dots,n\}$
        \STATE $v_t^{\,s}\gets \nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s-1})+\tilde{g}^{\,s-1}$ \hfill\textbf{(variance‑reduced estimator)}
        \STATE $x_{t+1}^{\,s}\gets x_t^{\,s}-\eta\, v_t^{\,s}$ \hfill\textbf{(inner SGD step)}
    \ENDFOR
    \STATE $\tilde{x}^{\,s}\gets x_{m}^{\,s}$ \hfill\textbf{(update reference point)}
\ENDFOR
\STATE \textbf{Output:} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple scalar problem
\[
f(w)=\bigl(w-3\bigr)^2,\qquad n=1,\; f_1\equiv f.
\]
Hence $\nabla f(w)=2(w-3)$, $L=2$, $\mu=2$.  
Set $w_0=\tilde{x}^{\,0}=0$, stepsize $\eta=0.1$, inner length $m=3$, and run a single epoch $S=1$.

\begin{enumerate}[label={\arabic*.}]
\item \textbf{Full gradient at reference point:}
\[
\tilde{g}^{\,0}= \nabla f(\tilde{x}^{\,0}) = 2(0-3) = -6 .
\]

\item \textbf{Inner iteration $t=0$:}
\begin{align*}
x_0^{\,1}&=0,\\
v_0^{\,1}&= \nabla f(x_0^{\,1})-\nabla f(\tilde{x}^{\,0})+\tilde{g}^{\,0}
         = (-6)-(-6)+(-6)= -6,\\
x_1^{\,1}&= x_0^{\,1}-\eta v_0^{\,1}=0-0.1(-6)=0.6.
\end{align*}
Objective after step: $f(0.6)=(0.6-3)^2=5.76$.

\item \textbf{Inner iteration $t=1$:}
\begin{align*}
v_1^{\,1}&= \nabla f(x_1^{\,1})-\nabla f(\tilde{x}^{\,0})+\tilde{g}^{\,0}
         = 2(0.6-3)-(-6)+(-6)\\
        &= -4.8-(-6)-6 = -4.8+6-6 = -4.8,\\
x_2^{\,1}&=0.6-0.1(-4.8)=1.08,\\
f(1.08)&=(1.08-3)^2=3.6864.
\end{align*}

\item \textbf{Inner iteration $t=2$:}
\begin{align*}
v_2^{\,1}&= 2(1.08-3)-(-6)+(-6)= -3.84+6-6 = -3.84,\\
x_3^{\,1}&=1.08-0.1(-3.84)=1.464,\\
f(1.464)&=(1.464-3)^2=2.363\,\text{(approx)}.
\end{align*}

\item \textbf{End of epoch:} $\tilde{x}^{\,1}=x_3^{\,1}=1.464$.
\end{enumerate}
The iterates move toward the optimum $w^\star=3$ and the objective value decreases.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,\qquad\forall x,y .
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The dataset size $n$ is finite and the stochastic index $i_t$ is drawn uniformly:
\[
\Pr(i_t=i)=\frac{1}{n},\quad i=1,\dots,n .
\]
\end{assumption}
All constants satisfy $0<\mu\le L$ and the stepsize $\eta$ will be chosen in $(0,\tfrac{1}{3L})$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:finite} hold. Choose a stepsize 
\[
\eta = \frac{1}{10L}
\tag{5}
\]
and an inner loop length 
\[
m \ge \frac{20L}{\mu}.
\tag{6}
\]
Let $\tilde{x}^{\,s}$ be the reference point after epoch $s$. Then for all $s\ge 1$
\[
\mathbb{E}\!\left[\,f(\tilde{x}^{\,s})-f(x^\star)\,\right]
\;\le\;
\left(1-\frac{\mu\eta}{2}\right)^{\!s}
\bigl[f(\tilde{x}^{\,0})-f(x^\star)\bigr].
\tag{7}
\]
Consequently $\tilde{x}^{\,s}\to x^\star$ at a geometric rate.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction factor $1-\frac{\mu\eta}{2}$ is independent of $n$; the algorithm remains stable as $n\to\infty$ provided $\eta\le\frac{1}{3L}$.
\item \textbf{Hyperparameter sensitivity.} Increasing $m$ improves the contraction constant but incurs higher cost per epoch. The bound \eqref{thm:linear} is optimal (up to constants) for the class of $L$‑smooth, $\mu$‑strongly convex functions.
\item \textbf{Comparison.} Classical SGD with diminishing stepsize yields $\mathcal{O}(1/T)$ sub‑linear rate, whereas SVRG attains the linear rate \eqref{7}. Moreover, the per‑iteration variance of $v_t^{\,s}$ is bounded by a term proportional to $\|x_t^{\,s}-\tilde{x}^{\,s-1}\|^2$, which vanishes as iterates converge.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that the double‑loop SVRG algorithm enjoys a deterministic‑like linear convergence despite using stochastic gradients. The proof hinges on two key lemmas: unbiasedness of the estimator $v_t^{\,s}$ and a variance bound that scales with the distance to the reference point. By carefully selecting $\eta$ and $m$, the error contributed by the stochastic variance can be dominated by the strong‑convexity contraction.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
For any inner iterate $x_t^{\,s}$,
\[
\mathbb{E}_{i_t}\!\bigl[\,v_t^{\,s}\mid x_t^{\,s},\tilde{x}^{\,s-1}\bigr]
\;=\;
\nabla f(x_t^{\,s}).
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly sampled index $i_t$,
\[
\begin{aligned}
\mathbb{E}_{i_t}[v_t^{\,s}]
&= \frac{1}{n}\sum_{i=1}^{n}\bigl(\nabla f_i(x_t^{\,s})-\nabla f_i(\tilde{x}^{\,s-1})\bigr)+\tilde{g}^{\,s-1}\\
&= \bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s-1})\bigr)+\nabla f(\tilde{x}^{\,s-1})
= \nabla f(x_t^{\,s}) .
\end{aligned}
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:varbound}
Conditional on $x_t^{\,s}$ and $\tilde{x}^{\,s-1}$,
\[
\mathbb{E}_{i_t}\!\bigl[\|v_t^{\,s}-\nabla f(x_t^{\,s})\|^2\bigr]
\;\le\;
4L\bigl(f(x_t^{\,s})-f(x^\star)+f(\tilde{x}^{\,s-1})-f(x^\star)\bigr).
\tag{8}
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the identity
\[
v_t^{\,s}-\nabla f(x_t^{\,s})
=
\nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s-1})
-\bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s-1})\bigr),
\]
we obtain
\[
\begin{aligned}
\mathbb{E}_{i_t}\!\bigl[\|v_t^{\,s}-\nabla f(x_t^{\,s})\|^2\bigr]
&= \frac{1}{n}\sum_{i=1}^{n}\bigl\|\nabla f_i(x_t^{\,s})-\nabla f_i(\tilde{x}^{\,s-1})
-\bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s-1})\bigr)\bigr\|^2\\
&\le \frac{1}{n}\sum_{i=1}^{n}\bigl\|\nabla f_i(x_t^{\,s})-\nabla f_i(\tilde{x}^{\,s-1})\bigr\|^2
\quad\text{[Jensen]}\\
&\le \frac{1}{n}\sum_{i=1}^{n} L^2\|x_t^{\,s}-\tilde{x}^{\,s-1}\|^2
= L^2\|x_t^{\,s}-\tilde{x}^{\,s-1}\|^2 .
\end{aligned}
\]
Now apply the standard inequality for $L$‑smooth functions
\[
\|x-y\|^2\le\frac{2}{\mu}\bigl(f(x)-f(x^\star)+f(y)-f(x^\star)\bigr),
\]
and use $\mu\le L$ to obtain \eqref{8}.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  Throughout we condition on the history up to the beginning of epoch $s$ and denote $\mathbb{E}_s[\cdot]$ the corresponding expectation.

\begin{enumerate}[label={[\textbf{Step \arabic*}]}]
\item \textbf{Descent Lemma for a single inner step.}  
From $L$‑smoothness,
\[
f(x_{t+1}^{\,s})\le f(x_t^{\,s})+\langle\nabla f(x_t^{\,s}),x_{t+1}^{\,s}-x_t^{\,s}\rangle
+\frac{L}{2}\|x_{t+1}^{\,s}-x_t^{\,s}\|^2 .
\tag{9}
\]
Using the update \eqref{3}, $x_{t+1}^{\,s}-x_t^{\,s}=-\eta v_t^{\,s}$,
\[
f(x_{t+1}^{\,s})\le f(x_t^{\,s})-\eta\langle\nabla f(x_t^{\,s}),v_t^{\,s}\rangle
+\frac{L\eta^{2}}{2}\|v_t^{\,s}\|^{2}.
\tag{10}
\]

\item \textbf{Take expectation w.r.t.\ $i_t$.}  
Apply Lemma \ref{lem:unbiased} and the law $\mathbb{E}\|v\|^{2}=\|\mathbb{E}v\|^{2}+\mathbb{E}\|v-\mathbb{E}v\|^{2}$:
\[
\begin{aligned}
\mathbb{E}_{i_t}[f(x_{t+1}^{\,s})]
&\le f(x_t^{\,s})-\eta\|\nabla f(x_t^{\,s})\|^{2}
+\frac{L\eta^{2}}{2}\Bigl(\|\nabla f(x_t^{\,s})\|^{2}
+\mathbb{E}_{i_t}\|v_t^{\,s}-\nabla f(x_t^{\,s})\|^{2}\Bigr)\\
&= f(x_t^{\,s})-\eta\Bigl(1-\frac{L\eta}{2}\Bigr)\|\nabla f(x_t^{\,s})\|^{2}
+\frac{L\eta^{2}}{2}\mathbb{E}_{i_t}\|v_t^{\,s}-\nabla f(x_t^{\,s})\|^{2}.
\end{aligned}
\tag{11}
\]

\item \textbf{Bound the gradient norm using strong convexity.}  
Strong convexity \eqref{ass:strong} implies
\[
\|\nabla f(x_t^{\,s})\|^{2}\ge 2\mu\bigl(f(x_t^{\,s})-f(x^\star)\bigr).
\tag{12}
\]

\item \textbf{Insert variance bound (Lemma \ref{lem:varbound}).}  
Using \eqref{8} in \eqref{11} and then \eqref{12} we obtain
\[
\begin{aligned}
\mathbb{E}_{i_t}[f(x_{t+1}^{\,s})]
&\le f(x_t^{\,s})-\eta\Bigl(1-\frac{L\eta}{2}\Bigr) 2\mu\bigl(f(x_t^{\,s})-f(x^\star)\bigr)\\
&\qquad+ \frac{L\eta^{2}}{2}\,4L\bigl(f(x_t^{\,s})-f(x^\star)+f(\tilde{x}^{\,s-1})-f(x^\star)\bigr)\\
&= f(x_t^{\,s})-\underbrace{2\mu\eta\Bigl(1-\frac{L\eta}{2}\Bigr)}_{\alpha}
\bigl(f(x_t^{\,s})-f(x^\star)\bigr)
+2L^{2}\eta^{2}\bigl(f(x_t^{\,s})-f(x^\star)\bigr)\\
&\qquad+2L^{2}\eta^{2}\bigl(f(\tilde{x}^{\,s-1})-f(x^\star)\bigr).
\end{aligned}
\tag{13}
\]

\item \textbf{Choose stepsize $\eta\le\frac{1}{3L}$.}  
With $\eta\le\frac{1}{3L}$ we have $L\eta\le\frac13$ and consequently
\[
\alpha =2\mu\eta\Bigl(1-\frac{L\eta}{2}\Bigr)\ge 2\mu\eta\Bigl(1-\frac{1}{6}\Bigr)=\frac{5}{3}\mu\eta .
\tag{14}
\]
Moreover $2L^{2}\eta^{2}\le \frac{2L^{2}}{9L^{2}}=\frac{2}{9}$. Hence
\[
2L^{2}\eta^{2}\le \frac{1}{2}\mu\eta
\quad\text{provided}\quad \eta\le\frac{\mu}{4L^{2}}.
\tag{15}
\]
We will later select $\eta=\frac{1}{10L}$, which satisfies both (14) and (15) because $\mu\le L$.

\item \textbf{Simplify recursion.}  
Using (14)–(15) in (13) yields
\[
\mathbb{E}_{i_t}[f(x_{t+1}^{\,s})-f(x^\star)]
\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)\bigl(f(x_t^{\,s})-f(x^\star)\bigr)
+\frac{\mu\eta}{2}\bigl(f(\tilde{x}^{\,s-1})-f(x^\star)\bigr).
\tag{16}
\]

\item \textbf{Take full expectation over the inner loop.}  
Define $\Delta_t^{\,s}:=\mathbb{E}\bigl[f(x_t^{\,s})-f(x^\star)\bigr]$.  
Iterating \eqref{16} for $t=0,\dots,m-1$ and noting $x_0^{\,s}=\tilde{x}^{\,s-1}$ gives
\[
\Delta_{t+1}^{\,s}\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)\Delta_t^{\,s}
+\frac{\mu\eta}{2}\Delta_0^{\,s}.
\tag{17}
\]

\item \textbf{Solve the linear recurrence.}  
Unfolding \eqref{17} yields
\[
\Delta_{m}^{\,s}
\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Delta_0^{\,s}
+\frac{\mu\eta}{2}\Delta_0^{\,s}\sum_{k=0}^{m-1}\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{k}.
\tag{18}
\]
The geometric sum evaluates to
\[
\sum_{k=0}^{m-1}\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{k}
= \frac{1-\bigl(1-\frac{\mu\eta}{2}\bigr)^{m}}{\frac{\mu\eta}{2}}
\le \frac{2}{\mu\eta}.
\tag{19}
\]
Plugging \eqref{19} into \eqref{18} gives
\[
\Delta_{m}^{\,s}\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Delta_0^{\,s}
+ \Delta_0^{\,s}
= \Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Delta_0^{\,s}
+ \Delta_0^{\,s}
- \Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Delta_0^{\,s}
= \Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Delta_0^{\,s}
+ \Bigl[1-\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\Bigr]\Delta_0^{\,s}
= \Delta_0^{\,s}.
\tag{20}
\]
Thus the inner loop does not increase the expected suboptimality.

\item \textbf{Choose inner length $m$.}  
Select $m$ such that
\[
\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}\le\frac{1}{2}.
\tag{21}
\]
Since $1-x\le e^{-x}$, a sufficient condition is
\[
e^{-\frac{\mu\eta}{2}m}\le\frac12\;\Longrightarrow\;
m\ge\frac{2\ln 2}{\mu\eta}\le\frac{20L}{\mu},
\]
where we used $\eta=\frac{1}{10L}$.

\item \textbf{Epoch‑wise contraction.}  
From \eqref{16} with $t=m-1$ and using (21) we obtain
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{\,s})-f(x^\star)\bigr]
\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{m}
\bigl[f(\tilde{x}^{\,s-1})-f(x^\star)\bigr]
\le\frac12\bigl[f(\tilde{x}^{\,s-1})-f(x^\star)\bigr].
\tag{22}
\]
Iterating (22) over epochs yields the linear rate \eqref{7}:
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{\,s})-f(x^\star)\bigr]
\le\Bigl(\tfrac12\Bigr)^{s}
\bigl[f(\tilde{x}^{\,0})-f(x^\star)\bigr]
\le\Bigl(1-\frac{\mu\eta}{2}\Bigr)^{s}
\bigl[f(\tilde{x}^{\,0})-f(x^\star)\bigr].
\]
\end{enumerate}
This completes the proof of Theorem \ref{thm:linear}. \hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From the derivation we have the explicit contraction factor
\[
\rho := 1-\frac{\mu\eta}{2}
= 1-\frac{\mu}{20L}.
\tag{23}
\]
Hence after $S$ epochs,
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{\,S})-f(x^\star)\bigr]
\le \rho^{\,S}\bigl[f(\tilde{x}^{\,0})-f(x^\star)\bigr].
\tag{24}
\]
The total number of stochastic gradient evaluations is $S\,(n+m)$.  Choosing $m=20L/\mu$ yields a complexity of
\[
\mathcal{O}\!\Bigl(\bigl(n+ \tfrac{L}{\mu}\bigr)\log\frac{1}{\varepsilon}\Bigr)
\]
to achieve $f(\tilde{x})-f(x^\star)\le\varepsilon$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label={\arabic*.}]
\item \textbf{Pure gradient descent.}  If $m=1$ and we replace the stochastic estimator by the full gradient, the recursion reduces to the classical GD rate $(1-\mu\eta)^{T}$.
\item \textbf{Large $n$ regime.}  When $n\gg L/\mu$, the term $n$ dominates the per‑epoch cost, and SVRG attains the optimal $\mathcal{O}\bigl((n+L/\mu)\log(1/\varepsilon)\bigr)$ complexity, outperforming both SGD ($\mathcal{O}(1/\varepsilon)$) and full GD ($\mathcal{O}(n\log(1/\varepsilon))$).
\item \textbf{Limit $\eta\to 0$.}  The contraction factor $\rho\to 1$, reproducing the sub‑linear $1/T$ behaviour of standard SGD.
\end{enumerate}
