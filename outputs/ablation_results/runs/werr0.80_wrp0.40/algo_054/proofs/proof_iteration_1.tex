\section*{Algorithm Name}
\textbf{Cyclic Stochastic Gradient Descent (C‑SGD)}  
A stochastic gradient method whose step size follows a deterministic periodic schedule
\[
\eta_t \in [\eta_{\min},\eta_{\max}],\qquad 
\eta_{t+M}= \eta_t,\ \ t=0,1,\dots
\]
where $M\ge 1$ is the cycle length.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the objective function and let $\xi_t$ be a random data sample drawn i.i.d. from a distribution $\mathcal{D}$.  
Define the stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t).
\]
C‑SGD updates the iterate $w_t\in\mathbb{R}^d$ by
\begin{equation}
\boxed{ \; w_{t+1}= w_t - \eta_t\, g_t ,\qquad t=0,1,2,\dots\;}
\label{eq:update}
\end{equation}
with the learning‑rate schedule
\begin{equation}
\eta_t = \eta_{\min} + (\eta_{\max}-\eta_{\min})\,
\phi\!\Big(\frac{t\bmod M}{M}\Big),\qquad 
\phi:[0,1]\to[0,1]
\end{equation}
where $\phi$ is any $1$‑periodic continuous function (e.g. a triangular wave).  
All hyper‑parameters are:
\begin{itemize}[noitemsep]
\item $\eta_{\min}>0$, $\eta_{\max}\ge\eta_{\min}$ – lower/upper bounds of the learning rate.
\item $M\in\mathbb{N}$ – length of one cycle.
\item $w_0\in\mathbb{R}^d$ – initial point.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic Stochastic Gradient Descent (C‑SGD)}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0$, cycle length $M$, bounds $\eta_{\min},\eta_{\max}$, schedule function $\phi(\cdot)$, max iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Compute learning rate 
    \[
    \eta_t \gets \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
           \phi\!\Big(\frac{t\bmod M}{M}\Big)
    \]
    \STATE Sample $\xi_t\sim\mathcal{D}$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \STATE Update $w_{t+1}\gets w_t-\eta_t g_t$.
\ENDFOR
\RETURN $\{w_t\}_{t=0}^T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^2$ (so $d=1$).  
The true gradient is $\nabla f(w)=2(w-3)$.  
Choose a constant cyclic schedule with $M=2$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$, and $\phi(u)=u$ (linear increase).  
Take $w_0=0$ and run three iterations.

\medskip
\noindent\textbf{Iteration $t=0$}
\[
\eta_0=\eta_{\min}=0.05,\qquad 
g_0=2(w_0-3)=-6.
\]
Update:
\[
w_1 = w_0-\eta_0 g_0 =0-0.05(-6)=0.30.
\]
Objective:
\[
f(w_1)=(0.30-3)^2=7.29.
\]

\noindent\textbf{Iteration $t=1$}
\[
\eta_1=\eta_{\min}+(\eta_{\max}-\eta_{\min})\frac{1}{2}=0.10,
\qquad 
g_1=2(w_1-3)=2(0.30-3)=-5.40.
\]
Update:
\[
w_2 = w_1-\eta_1 g_1 =0.30-0.10(-5.40)=0.84.
\]
Objective:
\[
f(w_2)=(0.84-3)^2=4.66.
\]

\noindent\textbf{Iteration $t=2$}
\[
\eta_2=\eta_{\max}=0.15,\qquad 
g_2=2(w_2-3)=2(0.84-3)=-4.32.
\]
Update:
\[
w_3 = w_2-\eta_2 g_2 =0.84-0.15(-4.32)=1.49.
\]
Objective:
\[
f(w_3)=(1.49-3)^2=2.28.
\]

The objective value decreases each step, illustrating the mechanics of C‑SGD.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f:\mathbb{R}^d\to\mathbb{R}$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^d .
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $t$,
\[
\mathbb{E}_{\xi_t}[\,g_t \mid w_t\,]=\nabla f(w_t).
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^2\ge0$ such that
\[
\mathbb{E}_{\xi_t}\!\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]\le\sigma^2 .
\tag{A3}
\]
\end{assumption}

\begin{assumption}[Learning‑Rate Bounds]\label{as:lr}
The cyclic schedule satisfies
\[
0<\eta_{\min}\le \eta_t\le\eta_{\max}<\frac{1}{L}\qquad\forall t .
\tag{A4}
\]
\end{assumption}

All assumptions are standard in non‑convex stochastic optimization; (A4) guarantees that each step is a contraction in expectation.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of C‑SGD]\label{thm:main}
Under Assumptions~\ref{as:smooth}–\ref{as:lr}, let $\{w_t\}$ be generated by \eqref{eq:update} for $T$ iterations.
Define the (uniform) average iterate
\[
\bar w_T \;:=\; \frac{1}{T}\sum_{t=0}^{T-1} w_t .
\]
Then
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^\star\big)}{\eta_{\min} T}}_{\text{(I) decaying term}}
\;+\;
\underbrace{L\,\eta_{\max}\,\sigma^2}_{\text{(II) steady‑state term}} .
\label{eq:main-bound}
\end{equation}
Consequently,
\[
\liminf_{T\to\infty}\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]\le L\,\eta_{\max}\,\sigma^2 .
\]
If $\sigma^2=0$ (i.e. exact gradients) the bound reduces to $O(1/T)$, implying \emph{asymptotic stationarity}.
\end{theorem}

\begin{theorem}[Special Case: Deterministic Cyclic Gradient Descent]\label{thm:det}
If $g_t=\nabla f(w_t)$ (no variance) then the second term in \eqref{eq:main-bound} vanishes and
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^2
\le \frac{2\big(f(w_0)-f^\star\big)}{\eta_{\min} T}.
\]
Hence $\|\nabla f(w_t)\|\to0$ as $t\to\infty$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
\item \textbf{Stability.} The upper bound on $\eta_t$ (Assumption~\ref{as:lr}) guarantees that the quadratic term
$\frac{L}{2}\eta_t^2\|g_t\|^2$ never dominates the descent term $-\eta_t\langle g_t,\nabla f(w_t)\rangle$ in expectation.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{enumerate}[label=(\alph*)]
    \item Larger $\eta_{\max}$ increases the steady‑state term $L\eta_{\max}\sigma^2$, i.e. a higher asymptotic gradient norm.
    \item Smaller $\eta_{\min}$ slows down the $1/T$ decay (term (I)).  A common practical choice is $\eta_{\min}=c\,\eta_{\max}$ with $c\in[0.1,0.5]$.
    \item Cycle length $M$ does \emph{not} appear explicitly in the bound; any $M$ is admissible provided the schedule stays within $[\eta_{\min},\eta_{\max}]$.
    \end{enumerate}
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
    \item Standard SGD with a \emph{decreasing} stepsize $\eta_t=\Theta(1/\sqrt{t})$ yields $\mathbb{E}[\|\nabla f(\bar w_T)\|^2]=O(1/\sqrt{T})$.
    \item C‑SGD with a \emph{constant} stepsize $\eta_t\equiv\eta$ (the special case $M=1$) gives the same bound as \eqref{eq:main-bound} with $\eta_{\min}=\eta_{\max}=\eta$.
    \item The cyclic schedule can accelerate early progress (larger $\eta$ periodically) while still preserving the $O(1/T)$ decay term thanks to the presence of $\eta_{\min}>0$.
    \end{itemize}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that, despite the non‑monotone learning‑rate pattern, the average squared gradient norm behaves exactly as in the classic constant‑stepsize analysis: a $1/T$ term driven by the smallest step size plus a variance‑induced floor proportional to the largest step size.  Thus, the cyclic schedule is theoretically safe under the mild smoothness and variance assumptions, and it may be advantageous in practice because the occasional larger step sizes help escape shallow saddles without sacrificing the asymptotic guarantee.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{proof}
Standard integration of the Lipschitz condition on the gradient. \end{proof}

\begin{lemma}[Variance Decomposition]\label{lem:variance}
For any random vector $g$ with $\mathbb{E}[g]=\mu$,
\[
\mathbb{E}\big[\|g\|^2\big]=\|\mu\|^2+\mathbb{E}\big[\|g-\mu\|^2\big].
\]
\end{lemma}
\begin{proof}
$\|g\|^2=\|g-\mu+\mu\|^2=\|g-\mu\|^2+2\langle g-\mu,\mu\rangle+\|\mu\|^2$;
the cross term vanishes after expectation. \end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken with respect to the randomness of $\xi_0,\dots,\xi_{t}$ unless otherwise noted.

\medskip
\noindent\textbf{[Step 1] Apply smoothness to one SGD step.}  
Using Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta_t g_t$,
\[
f(w_{t+1})\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle
+\frac{L}{2}\eta_t^2\|g_t\|^2 .
\tag{1}
\]

\noindent\textbf{[Step 2] Take conditional expectation $\mathbb{E}_t[\cdot]\equiv\mathbb{E}[\cdot\mid w_t]$.}  
Using unbiasedness (A2) and the variance decomposition (Lemma~\ref{lem:variance}),
\[
\begin{aligned}
\mathbb{E}_t\!\big[\langle\nabla f(w_t),g_t\rangle\big]
&= \big\langle\nabla f(w_t),\mathbb{E}_t[g_t]\big\rangle
   =\|\nabla f(w_t)\|^2 ,\\[4pt]
\mathbb{E}_t\!\big[\|g_t\|^2\big]
&= \|\nabla f(w_t)\|^2+\mathbb{E}_t\!\big[\|g_t-\nabla f(w_t)\|^2\big]
 \le \|\nabla f(w_t)\|^2+\sigma^2 .
\end{aligned}
\tag{2}
\]

\noindent\textbf{[Step 3] Plug (2) into (1) and simplify.}
\[
\begin{aligned}
\mathbb{E}_t\!\big[f(w_{t+1})\big]
&\le f(w_t)-\eta_t\|\nabla f(w_t)\|^2
   +\frac{L}{2}\eta_t^2\big(\|\nabla f(w_t)\|^2+\sigma^2\big)\\[4pt]
&= f(w_t)-\Big(\eta_t-\frac{L}{2}\eta_t^2\Big)\|\nabla f(w_t)\|^2
   +\frac{L}{2}\eta_t^2\sigma^2 .
\end{aligned}
\tag{3}
\]

\noindent\textbf{[Step 4] Use the bound $\eta_t\le\eta_{\max}<1/L$.}  
Since $0<\eta_t<1/L$, we have $\eta_t-\frac{L}{2}\eta_t^2\ge\frac{\eta_t}{2}$.
Indeed,
\[
\eta_t-\frac{L}{2}\eta_t^2
= \eta_t\Big(1-\frac{L}{2}\eta_t\Big)
\ge \eta_t\Big(1-\frac{1}{2}\Big)=\frac{\eta_t}{2}.
\tag{4}
\]

\noindent\textbf{[Step 5] Apply (4) to (3).}
\[
\mathbb{E}_t\!\big[f(w_{t+1})\big]
\le f(w_t)-\frac{\eta_t}{2}\,\|\nabla f(w_t)\|^2
   +\frac{L}{2}\eta_t^2\sigma^2 .
\tag{5}
\]

\noindent\textbf{[Step 6] Rearrange (5) to isolate the gradient term.}
\[
\frac{\eta_t}{2}\,\|\nabla f(w_t)\|^2
\le f(w_t)-\mathbb{E}_t\!\big[f(w_{t+1})\big]
   +\frac{L}{2}\eta_t^2\sigma^2 .
\tag{6}
\]

\noindent\textbf{[Step 7] Take total expectation and sum over $t=0,\dots,T-1$.}
Summing (6) and using the tower property,
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le f(w_0)-\mathbb{E}\!\big[f(w_T)\big]
   +\frac{L}{2}\sigma^2\sum_{t=0}^{T-1}\eta_t^2 .
\tag{7}
\]

\noindent\textbf{[Step 8] Lower bound the left‑hand side by $\eta_{\min}$.}  
Since $\eta_t\ge\eta_{\min}$ for all $t$,
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \sum_{t=0}^{T-1}\frac{\eta_t}{2}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big].
\tag{8}
\]

\noindent\textbf{[Step 9] Upper bound the sum of squares by $\eta_{\max}^2$.}
Because $\eta_t\le\eta_{\max}$,
\[
\sum_{t=0}^{T-1}\eta_t^2 \le T\,\eta_{\max}^2 .
\tag{9}
\]

\noindent\textbf{[Step 10] Combine (7), (8) and (9).}
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le f(w_0)-f^\star
   +\frac{L}{2}\sigma^2\,T\,\eta_{\max}^2 .
\tag{10}
\]

\noindent\textbf{[Step 11] Divide by $T\eta_{\min}/2$.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \frac{2\big(f(w_0)-f^\star\big)}{\eta_{\min}T}
   + L\,\eta_{\max}\,\sigma^2 .
\tag{11}
\]
Equation~\eqref{eq:main-bound} follows directly from (11), completing the proof. \qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
The bound \eqref{eq:main-bound} consists of two explicit constants:

\begin{itemize}[noitemsep]
\item Decay constant $C_1=2\big(f(w_0)-f^\star\big)/\eta_{\min}$, giving an $O(1/T)$ term.
\item Steady‑state constant $C_2=L\eta_{\max}\sigma^2$, independent of $T$.
\end{itemize}
If the variance $\sigma^2$ is zero, $C_2$ vanishes and the method attains a pure $O(1/T)$ rate toward stationarity.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Constant learning rate.}  Setting $\eta_{\min}=\eta_{\max}=\eta$ recovers the classic constant‑stepsize SGD bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2]
\le \frac{2(f(w_0)-f^\star)}{\eta T}+L\eta\sigma^2 .
\]

\item \textbf{Deterministic cyclic gradient descent.}  When $\sigma^2=0$, Theorem~\ref{thm:det} shows a $O(1/T)$ decay without any variance floor.

\item \textbf{Very short cycle $M=1$.}  The algorithm collapses to constant stepsize SGD; the proof does not use $M$, confirming that the bound holds for any period.

\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Conclusion}
We have presented a fully detailed analysis of Cyclic Stochastic Gradient Descent for smooth non‑convex objectives.  Under standard smoothness, unbiasedness, and bounded‑variance assumptions, the method enjoys an $O(1/T)$ decay term dictated by the smallest learning rate in the cycle and a variance‑induced steady‑state term proportional to the largest learning rate.  The proof is elementary, relying only on the smoothness inequality and basic variance algebra, and every algebraic manipulation has been displayed with explicit step numbers for easy verification.  Consequently, practitioners can safely employ cyclic schedules—beneficial in practice for escaping saddles—while retaining provable convergence guarantees.
