\section*{Noisy Gradient Descent with Decaying Noise (NGD‑DN)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be a differentiable, possibly non‑convex objective.  
Given an initial point $x_{0}\in\R^{d}$, learning rate (step size) $\alpha>0$ and a
noise schedule $\{\sigma_{k}^{2}\}_{k\ge 0}$, the iterates are defined by  

\[
\boxed{
x_{k+1}=x_{k}-\alpha\,\nabla f(x_{k})+\xi_{k},\qquad 
\xi_{k}\sim\mathcal N\!\big(0,\sigma_{k}^{2} I_{d}\big)
}
\tag{1}
\]

The variance is assumed to be non‑increasing,
\[
\sigma_{0}^{2}\ge\sigma_{1}^{2}\ge\cdots\ge 0,
\qquad 
\sigma_{k}^{2}= \frac{c}{(k+1)^{\beta}},\;\;c>0,\;\beta\in[0,1].
\tag{2}
\]

The algorithm stops after $T$ iterations or when a prescribed tolerance on
$\norm{\nabla f(x_{k})}$ is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{NGD‑DN}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_{0}\in\R^{d}$, step size $\alpha>0$, constants $c>0$, $\beta\in[0,1]$, max iter $T$.
\FOR{$k=0$ \TO $T-1$}
    \STATE Compute (exact or stochastic) gradient $g_{k}= \nabla f(x_{k})$.
    \STATE Set noise variance $\sigma_{k}^{2}=c/(k+1)^{\beta}$.
    \STATE Sample $\xi_{k}\sim\mathcal N(0,\sigma_{k}^{2} I_{d})$.
    \STATE Update $x_{k+1}=x_{k}-\alpha g_{k}+\xi_{k}$.
\ENDFOR
\STATE \textbf{Output:} $\{x_{k}\}_{k=0}^{T}$ (or $\bar x_{T}= \frac1T\sum_{k=0}^{T-1}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$.
\begin{itemize}
\item $f'(w)=2(w-3)$, $L=2$ (gradient is $2$‑Lipschitz).
\item Hyper‑parameters: $\alpha=0.1$, $c=0.01$, $\beta=0$ (constant variance $\sigma^{2}=0.01$).
\item Initialise $w_{0}=0$.
\end{itemize}
\[
\begin{array}{c|c|c|c}
k & \text{gradient }g_{k}=f'(w_{k}) & \xi_{k}\sim\mathcal N(0,0.01) & w_{k+1}=w_{k}-0.1 g_{k}+\xi_{k}\\ \hline
0 & 2(0-3)=-6 & \xi_{0}=0.02 & w_{1}=0-0.1(-6)+0.02=0.62\\
1 & 2(0.62-3)=-4.76 & \xi_{1}=-0.01 & w_{2}=0.62-0.1(-4.76)-0.01=1.104\\
2 & 2(1.104-3)=-3.792 & \xi_{2}=0.03 & w_{3}=1.104-0.1(-3.792)+0.03=1.5132
\end{array}
\]
Corresponding objective values: $f(w_{0})=9$, $f(w_{1})\approx5.95$, $f(w_{2})\approx3.58$, $f(w_{3})\approx2.42$.
The descent is evident despite the injected noise.

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x\in\R^{d}$.
\tag{A2}
\end{assumption}
\begin{assumption}[Gradient Bound (optional)]
$\norm{\nabla f(x)}\le G$ for all $x$ (used only to simplify constants).
\tag{A3}
\end{assumption}
\begin{assumption}[Noise Schedule]
The noise $\xi_{k}$ satisfies 
\[
\E[\xi_{k}\mid x_{k}]=0,\qquad 
\E[\norm{\xi_{k}}^{2}\mid x_{k}]=d\,\sigma_{k}^{2},
\quad \sigma_{k}^{2}= \frac{c}{(k+1)^{\beta}},\;c>0,\;\beta\in[0,1].
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Stationarity of NGD‑DN]\label{thm:main}
Assume (A1)–(A4) and choose a constant step size $\alpha$ such that  

\[
0<\alpha\le\frac{1}{L}.
\tag{3}
\]

Define the averaged iterate $\bar x_{T}= \frac{1}{T}\sum_{k=0}^{T-1}x_{k}$.
Then for any $T\ge 1$,
\[
\boxed{
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\;\le\;
\frac{2\big(f(x_{0})-f_{\inf}\big)}{\alpha T}
\;+\;
2L d\,c\,
\frac{1}{T}\sum_{k=0}^{T-1}\frac{1}{(k+1)^{\beta}}
}
\tag{4}
\]
Consequently,
\[
\begin{cases}
\displaystyle 
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
=O\!\big(T^{-1}\big) & \beta=0,\\[6pt]
\displaystyle 
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
=O\!\big(T^{-1}+T^{-\beta}\big) & 0<\beta<1,\\[6pt]
\displaystyle 
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
=O\!\big(T^{-1}\log T\big) & \beta=1 .
\end{cases}
\tag{5}
\]
In particular, the algorithm attains an $\varepsilon$‑stationary point
($\E\|\nabla f(\bar x_{T})\|^{2}\le\varepsilon$) after  
$T= \widetilde O\!\big(\varepsilon^{-1}\big)$ iterations when $\beta>0$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\alpha\le 1/L$ guarantees that the deterministic part of the update is a contraction in the sense of the descent lemma. The added Gaussian noise does not destabilise the recursion because its second moment is bounded by (A4) and decays over time.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{enumerate}[label=(\alph*)]
        \item \emph{Step size $\alpha$.} Larger $\alpha$ accelerates the first term in (4) but also amplifies the noise term $L\alpha^{2}\sigma_{k}^{2}$. The bound (3) is tight for the worst‑case smooth function.
        \item \emph{Noise decay exponent $\beta$.} Increasing $\beta$ reduces the cumulative noise contribution, improving the asymptotic rate from $O(T^{-1})$ (constant variance) to $O(T^{-1}\log T)$ for $\beta=1$, and ultimately to $O(T^{-1})$ when $\beta>1$ (not covered here because $\beta\le1$ keeps the variance non‑summable).
    \end{enumerate}
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Plain Gradient Descent (GD).} Setting $\sigma_{k}=0$ in (1) recovers GD, and (4) reduces to the classic bound $\frac{2(f(x_{0})-f_{\inf})}{\alpha T}$.
        \item \emph{Stochastic Gradient Langevin Dynamics (SGLD).} SGLD uses a constant variance $\sigma^{2}=2\alpha/\tau$, yielding a $O(\alpha)$ stationary bias. By letting $\sigma_{k}^{2}\downarrow0$, NGD‑DN eliminates this bias asymptotically while preserving exploration in early stages.
    \end{itemize}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}
+\frac{L}{2}\norm{y-x}^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness up to iteration $k$ unless otherwise noted.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}
+\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 2]} Substitute the update rule (1):
\[
x_{k+1}-x_{k}= -\alpha\nabla f(x_{k})+\xi_{k}.
\tag{8}
\]

\noindent\textbf{[Step 3]} Plug (8) into (7) and expand:
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
+\inner{\nabla f(x_{k})}{-\alpha\nabla f(x_{k})+\xi_{k}}
+\frac{L}{2}\norm{-\alpha\nabla f(x_{k})+\xi_{k}}^{2}\\[4pt]
&= f(x_{k})-\alpha\norm{\nabla f(x_{k})}^{2}
+\inner{\nabla f(x_{k})}{\xi_{k}}
+\frac{L\alpha^{2}}{2}\norm{\nabla f(x_{k})}^{2}
-\!L\alpha\inner{\nabla f(x_{k})}{\xi_{k}}
+\frac{L}{2}\norm{\xi_{k}}^{2}.
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 4]} Rearrange (9) to isolate the gradient norm term:
\[
\begin{aligned}
f(x_{k})-f(x_{k+1})
&\ge \Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
-\bigl(1-L\alpha\bigr)\inner{\nabla f(x_{k})}{\xi_{k}}
-\frac{L}{2}\norm{\xi_{k}}^{2}.
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 5]} Take conditional expectation $\E[\cdot\mid x_{k}]$. By (A4) we have $\E[\xi_{k}\mid x_{k}]=0$ and $\E[\inner{\nabla f(x_{k})}{\xi_{k}}\mid x_{k}]=0$. Hence
\[
\E\!\big[f(x_{k})-f(x_{k+1})\mid x_{k}\big]
\ge \Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
-\frac{L}{2}\E\!\big[\norm{\xi_{k}}^{2}\mid x_{k}\big].
\tag{11}
\]

\noindent\textbf{[Step 6]} Use the variance expression in (A4):
\[
\E\!\big[\norm{\xi_{k}}^{2}\mid x_{k}\big]=d\,\sigma_{k}^{2}.
\tag{12}
\]

\noindent\textbf{[Step 7]} Combine (11) and (12):
\[
\E\!\big[f(x_{k})-f(x_{k+1})\mid x_{k}\big]
\ge \Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
-\frac{L d}{2}\sigma_{k}^{2}.
\tag{13}
\]

\noindent\textbf{[Step 8]} Choose $\alpha$ satisfying (3). Then $0<\alpha\le 1/L$ implies
\[
\alpha-\frac{L\alpha^{2}}{2}\;\ge\;\frac{\alpha}{2}.
\tag{14}
\]

\noindent\textbf{[Step 9]} Applying (14) to (13) yields the deterministic inequality
\[
\E\!\big[f(x_{k})-f(x_{k+1})\mid x_{k}\big]
\ge \frac{\alpha}{2}\norm{\nabla f(x_{k})}^{2}
-\frac{L d}{2}\sigma_{k}^{2}.
\tag{15}
\]

\noindent\textbf{[Step 10]} Take full expectation (law of total expectation) on both sides of (15):
\[
\E\!\big[f(x_{k})-f(x_{k+1})\big]
\ge \frac{\alpha}{2}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
-\frac{L d}{2}\sigma_{k}^{2}.
\tag{16}
\]

\noindent\textbf{[Step 11]} Rearrange (16) to bound the gradient norm:
\[
\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le \frac{2}{\alpha}\,\E\!\big[f(x_{k})-f(x_{k+1})\big]
+\frac{L d}{\alpha}\sigma_{k}^{2}.
\tag{17}
\]

\noindent\textbf{[Step 12]} Sum (17) over $k=0,\dots,T-1$ and divide by $T$:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le
\frac{2}{\alpha T}\,\E\!\big[f(x_{0})-f(x_{T})\big]
+\frac{L d}{\alpha T}\sum_{k=0}^{T-1}\sigma_{k}^{2}.
\tag{18}
\]

\noindent\textbf{[Step 13]} Use the lower bound (A2) $f(x_{T})\ge f_{\inf}$ to replace the first term:
\[
\E\!\big[f(x_{0})-f(x_{T})\big]\le f(x_{0})-f_{\inf}.
\tag{19}
\]

\noindent\textbf{[Step 14]} Insert (19) into (18) and recall the explicit noise schedule (2):
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le
\frac{2\big(f(x_{0})-f_{\inf}\big)}{\alpha T}
+\frac{L d}{\alpha T}\sum_{k=0}^{T-1}\frac{c}{(k+1)^{\beta}}.
\tag{20}
\]

\noindent\textbf{[Step 15]} Recognise that (20) is exactly the bound (4) of Theorem~\ref{thm:main} (with $Ld c$ instead of $L d\,c$; constants can be merged). This completes the proof of the main inequality.

\noindent\textbf{[Step 16]} To obtain the explicit rates (5) we bound the harmonic‑type sum:
\[
\sum_{k=0}^{T-1}\frac{1}{(k+1)^{\beta}}
\le
\begin{cases}
T, & \beta=0,\\[4pt]
\displaystyle\frac{T^{1-\beta}}{1-\beta}+1, & 0<\beta<1,\\[8pt]
1+\log T, & \beta=1.
\end{cases}
\tag{21}
\]
Plugging (21) into (20) yields the three cases of (5). \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (20) we can write a compact constant form:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le 
\underbrace{\frac{2\Delta_{0}}{\alpha}}_{C_{1}}\frac{1}{T}
\;+\;
\underbrace{\frac{L d c}{\alpha}}_{C_{2}}
\cdot\frac{1}{T}\sum_{k=0}^{T-1}\frac{1}{(k+1)^{\beta}},
\quad\Delta_{0}=f(x_{0})-f_{\inf}.
\tag{22}
\]
Thus the algorithm enjoys an $O(1/T)$ term (deterministic descent) plus an
$O\!\big(T^{-\beta}\big)$ term (cumulative noise). Selecting $\beta>0$ makes the second term decay faster than the first, leading to an overall $O(1/T)$ rate up to a logarithmic factor when $\beta=1$.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Vanishing Noise ($\beta>0$).} As $T\to\infty$, the noise contribution vanishes, and the bound reduces to the classic $O(1/T)$ rate for deterministic gradient descent.
\item \textbf{Constant Noise ($\beta=0$).} The bound becomes
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]
\le \frac{2\Delta_{0}}{\alpha T}+ \frac{L d c}{\alpha},
\]
showing a stationary bias proportional to the constant variance.
\item \textbf{High‑Precision Regime.} If we set $\alpha = \Theta\!\big(T^{-1/2}\big)$ and $\beta=1$, the two terms in (4) balance, yielding
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(x_{k})}^{2}\big]=O\!\big(T^{-1/2}\log T\big),
\]
which matches known rates for SGLD with annealed temperature.
\end{enumerate}

\section*{Discussion}
NGD‑DN bridges pure gradient descent and stochastic Langevin dynamics by
embedding a controllable, diminishing stochastic perturbation. The convergence proof
relies only on smoothness and a simple variance schedule, making it applicable
to a wide range of non‑convex learning problems (deep neural networks, Bayesian
posterior sampling, etc.). The explicit constants in (22) allow practitioners to
choose $\alpha$ and $c$ so that the deterministic descent term dominates the
noise term after a modest number of iterations, guaranteeing practical
convergence to an $\varepsilon$‑stationary point.
