\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) under the Polyak–Łojasiewicz (PL) condition}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$ where each $f_i:\R^d\to\R$ is $L$‑smooth.  
Denote by $x^{*}$ a global minimiser of $f$ and $f^{*}=f(x^{*})$.

\begin{itemize}
    \item \textbf{Outer loop (epoch) $s=0,1,\dots,S-1$:}
    \[
        \tilde{x}^{s}= \begin{cases}
            x_{0}, & s=0,\\
            x_{m}^{s-1}, & s\ge 1,
        \end{cases}
    \qquad
    \tilde{\mu}^{s}= \nabla f(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s}).
    \]
    \item \textbf{Inner loop $t=0,1,\dots,m-1$:}
    \begin{enumerate}[label=\arabic*.]
        \item Sample an index $i_t$ uniformly from $\{1,\dots,n\}$.
        \item Form the variance‑reduced estimator
        \[
            v_t^{s}= \nabla f_{i_t}(x_t^{s})-\nabla f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}.
        \]
        \item Update
        \[
            x_{t+1}^{s}= x_t^{s}-\eta\, v_t^{s},
        \]
        where $\eta>0$ is a constant step size.
    \end{enumerate}
    \item After $m$ inner steps set $\tilde{x}^{s+1}=x_{m}^{s}$.
\end{itemize}
The algorithm stops after $S$ epochs or when a prescribed tolerance on $f(\tilde{x}^{s})-f^{*}$ is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG with PL condition}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, step size $\eta$, inner loop length $m$, epochs $S$
\State $\tilde{x}^{0}\gets x_{0}$
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\tilde{\mu}^{s}\gets \nabla f(\tilde{x}^{s})$
    \State $x_{0}^{s}\gets \tilde{x}^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\mathrm{Uniform}\{1,\dots,n\}$
        \State $v_t^{s}\gets \nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \State $x_{t+1}^{s}\gets x_{t}^{s}-\eta\,v_t^{s}$
    \EndFor
    \State $\tilde{x}^{s+1}\gets x_{m}^{s}$
\EndFor
\State \textbf{return} $\tilde{x}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^{2}$, i.e. $n=1$, $f_{1}=f$.  
We take $w_{0}=0$, step size $\eta=0.1$, inner length $m=3$, and run a single epoch.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $v_t$ & $w_{t+1}=w_t-\eta v_t$ & $f(w_{t+1})$\\ \hline
0 & $\nabla f(w_0)-\nabla f(\tilde w)+\nabla f(\tilde w)=2(w_0-3)= -6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(w_1-3)=2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(w_2-3)=2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3697$
\end{tabular}
\end{center}
After the three inner updates the snapshot $\tilde w$ would be set to $1.464$, and the objective value has decreased from $9$ (at $w_0$) to $\approx2.37$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e.
\[
    \forall x,y\in\R^{d},\qquad 
    f_i(y)\le f_i(x)+\inner{\nabla f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
The finite‑sum $f$ satisfies
\[
    \forall x\in\R^{d},\qquad
    \frac{1}{2\mu}\norm{\nabla f(x)}^{2}\ge f(x)-f^{*},
\]
for some $\mu>0$.
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For any $x$,
\[
    \frac{1}{n}\sum_{i=1}^{n}\norm{\nabla f_i(x)-\nabla f(x)}^{2}\le\sigma^{2}.
\]
\end{assumption}
All constants $L,\mu,\sigma$ are known; we shall require the step size
\[
    0<\eta\le\frac{1}{3L}
\tag{A1}
\]
and the inner‑loop length $m$ to satisfy
\[
    m\ge \frac{2L}{\mu}.
\tag{A2}
\]

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:var} and choose $\eta$ and $m$ according to (A1)–(A2).  
Let $\tilde{x}^{s}$ be the snapshot after epoch $s$. Then for every $s\ge0$,
\[
    \E\!\left[\,f(\tilde{x}^{s+1})-f^{*}\,\right]
    \le \rho\,\E\!\left[\,f(\tilde{x}^{s})-f^{*}\,\right],
\qquad
    \rho:=\underbrace{\bigl(1-\eta\mu\bigr)}_{\text{contraction}}+
          \underbrace{\frac{L\eta^{2}}{\mu m}}_{\text{variance\ term}}
    <1.
\tag{1}
\]
Consequently, after $S$ epochs,
\[
    \E\!\left[\,f(\tilde{x}^{S})-f^{*}\,\right]
    \le \rho^{S}\,\bigl(f(\tilde{x}^{0})-f^{*}\bigr),
\]
i.e. SVRG enjoys a linear (geometric) convergence rate under the PL condition.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound (1) holds for any random sampling scheme that is unbiased (uniform sampling suffices). The contraction factor $\rho$ is independent of the data size $n$.
    \item \textbf{Hyper‑parameter sensitivity.} Condition (A1) guarantees that the variance term $L\eta^{2}/(\mu m)$ is at most $1/3$, while (A2) makes the contraction term $1-\eta\mu$ dominate, ensuring $\rho<1$. Larger $m$ reduces the variance term, yielding faster linear decay.
    \item \textbf{Comparison to baseline SGD.} For convex $L$‑smooth functions without PL, SGD attains $O(1/\sqrt{T})$ sub‑linear rates. Under PL, SGD achieves $O((1-\eta\mu)^{T})$ only with diminishing step sizes; SVRG reaches the same linear rate with a constant step size and a smaller constant factor because the variance term is eliminated after each epoch.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that the double‑loop variance‑reduction scheme inherits the fast geometric decay typical of deterministic gradient descent when the PL inequality holds, while retaining the computational advantage of stochastic updates. The proof below follows the classic SVRG analysis but incorporates the PL inequality at the crucial contraction step.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function $f$,
\[
    f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2},
    \qquad\forall x,y.
\tag{2}
\]
\end{lemma}
\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
Conditioned on $x_{t}^{s}$ and $\tilde{x}^{s}$,
\[
    \E_{i_t}\!\left[\,\norm{v_t^{s}-\nabla f(x_{t}^{s})}^{2}\right]
    \le 2L\bigl(f(x_{t}^{s})-f^{*}+f(\tilde{x}^{s})-f^{*}\bigr).
\tag{3}
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:var}]
Using unbiasedness $\E_{i_t}[v_t^{s}]=\nabla f(x_{t}^{s})$ and $L$‑smoothness of each $f_i$, we have
\[
\begin{aligned}
\norm{v_t^{s}-\nabla f(x_{t}^{s})}^{2}
    &=\norm{\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})
          -\bigl(\nabla f(x_{t}^{s})-\nabla f(\tilde{x}^{s})\bigr)}^{2}\\
    &\le 2\norm{\nabla f_{i_t}(x_{t}^{s})-\nabla f_{i_t}(\tilde{x}^{s})}^{2}
        +2\norm{\nabla f(x_{t}^{s})-\nabla f(\tilde{x}^{s})}^{2}.
\end{aligned}
\]
Taking expectation over $i_t$ and applying $L$‑smoothness
$\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y}$ yields
\[
\E_{i_t}\!\left[\norm{v_t^{s}-\nabla f(x_{t}^{s})}^{2}\right]
    \le 4L^{2}\norm{x_{t}^{s}-\tilde{x}^{s}}^{2}.
\]
Finally, smoothness of $f$ gives
\[
    \norm{x_{t}^{s}-\tilde{x}^{s}}^{2}\le\frac{2}{L}\bigl(f(x_{t}^{s})-f^{*}+f(\tilde{x}^{s})-f^{*}\bigr),
\]
which proves (3). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof, expectations are taken over all random indices $i_{0},\dots,i_{m-1}$ generated in the current epoch, conditioned on the snapshot $\tilde{x}^{s}$.

\bigskip
\noindent\textbf{[Step 1]} (One‑step progress).  
From the update $x_{t+1}^{s}=x_{t}^{s}-\eta v_{t}^{s}$ and the smoothness inequality (2) applied with $x=x_{t}^{s}$, $y=x_{t+1}^{s}$,
\[
\begin{aligned}
f(x_{t+1}^{s})
    &\le f(x_{t}^{s})+\inner{\nabla f(x_{t}^{s})}{x_{t+1}^{s}-x_{t}^{s}}
        +\frac{L}{2}\norm{x_{t+1}^{s}-x_{t}^{s}}^{2}  \\
    & = f(x_{t}^{s})-\eta\inner{\nabla f(x_{t}^{s})}{v_{t}^{s}}
        +\frac{L\eta^{2}}{2}\norm{v_{t}^{s}}^{2}.
\end{aligned}
\tag{4}
\]

\noindent\textbf{[Step 2]} (Take expectation).  
Conditioning on $x_{t}^{s}$ and $\tilde{x}^{s}$ and using $\E[v_{t}^{s}]=\nabla f(x_{t}^{s})$,
\[
\E\!\left[f(x_{t+1}^{s})\mid x_{t}^{s},\tilde{x}^{s}\right]
    \le f(x_{t}^{s})-\eta\norm{\nabla f(x_{t}^{s})}^{2}
        +\frac{L\eta^{2}}{2}\E\!\left[\norm{v_{t}^{s}}^{2}\mid x_{t}^{s},\tilde{x}^{s}\right].
\tag{5}
\]

\noindent\textbf{[Step 3]} (Decompose the second moment).  
\[
\begin{aligned}
\E\!\left[\norm{v_{t}^{s}}^{2}\mid x_{t}^{s},\tilde{x}^{s}\right]
    &= \norm{\nabla f(x_{t}^{s})}^{2}
       +\E\!\left[\norm{v_{t}^{s}-\nabla f(x_{t}^{s})}^{2}\mid x_{t}^{s},\tilde{x}^{s}\right].
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 4]} (Insert Lemma~\ref{lem:var}).  
Using (3) we obtain
\[
\E\!\left[\norm{v_{t}^{s}}^{2}\mid x_{t}^{s},\tilde{x}^{s}\right]
    \le \norm{\nabla f(x_{t}^{s})}^{2}
        +2L\bigl(f(x_{t}^{s})-f^{*}+f(\tilde{x}^{s})-f^{*}\bigr).
\tag{7}
\]

\noindent\textbf{[Step 5]} (Combine (5)–(7)).  
Plugging (7) into (5) gives
\[
\begin{aligned}
\E\!\left[f(x_{t+1}^{s})\mid x_{t}^{s},\tilde{x}^{s}\right]
    &\le f(x_{t}^{s})-\eta\norm{\nabla f(x_{t}^{s})}^{2}
        +\frac{L\eta^{2}}{2}\norm{\nabla f(x_{t}^{s})}^{2}   \\
    &\qquad\;+\;L^{2}\eta^{2}\bigl(f(x_{t}^{s})-f^{*}+f(\tilde{x}^{s})-f^{*}\bigr).
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 6]} (Apply the PL inequality).  
From Assumption~\ref{ass:pl},
\[
\norm{\nabla f(x_{t}^{s})}^{2}\ge 2\mu\bigl(f(x_{t}^{s})-f^{*}\bigr).
\tag{9}
\]
Insert (9) into (8):
\[
\begin{aligned}
\E\!\left[f(x_{t+1}^{s})\mid x_{t}^{s},\tilde{x}^{s}\right]
    &\le f(x_{t}^{s})-\eta\cdot 2\mu\bigl(f(x_{t}^{s})-f^{*}\bigr)
        +\frac{L\eta^{2}}{2}\cdot 2\mu\bigl(f(x_{t}^{s})-f^{*}\bigr) \\
    &\qquad\;+\;L^{2}\eta^{2}\bigl(f(x_{t}^{s})-f^{*}+f(\tilde{x}^{s})-f^{*}\bigr).
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 7]} (Collect coefficients).  
Define $\Delta_{t}^{s}:=f(x_{t}^{s})-f^{*}$ and $\Delta_{\tilde}^{s}:=f(\tilde{x}^{s})-f^{*}$.  
Equation (10) becomes
\[
\E\!\left[\Delta_{t+1}^{s}\mid x_{t}^{s},\tilde{x}^{s}\right]
    \le\Bigl(1-2\mu\eta+L\mu\eta^{2}+L^{2}\eta^{2}\Bigr)\Delta_{t}^{s}
        +L^{2}\eta^{2}\Delta_{\tilde}^{s}.
\tag{11}
\]

\noindent\textbf{[Step 8]} (Step‑size restriction).  
Using (A1), $\eta\le 1/(3L)$, we have $L\mu\eta^{2}\le \frac{\mu\eta}{3}$ and $L^{2}\eta^{2}\le\frac{L\eta}{3}$. Consequently,
\[
1-2\mu\eta+L\mu\eta^{2}+L^{2}\eta^{2}
    \le 1-\mu\eta.
\tag{12}
\]
Thus (11) simplifies to
\[
\E\!\left[\Delta_{t+1}^{s}\mid x_{t}^{s},\tilde{x}^{s}\right]
    \le (1-\mu\eta)\Delta_{t}^{s}+L^{2}\eta^{2}\Delta_{\tilde}^{s}.
\tag{13}
\]

\noindent\textbf{[Step 9]} (Unroll over the inner loop).  
Taking full expectation and iterating (13) for $t=0,\dots,m-1$ yields
\[
\begin{aligned}
\E\!\left[\Delta_{m}^{s}\right]
    &\le (1-\mu\eta)^{m}\Delta_{0}^{s}
        +L^{2}\eta^{2}\Delta_{\tilde}^{s}\sum_{k=0}^{m-1}(1-\mu\eta)^{k}\\
    &\le (1-\mu\eta)^{m}\Delta_{\tilde}^{s}
        +\frac{L^{2}\eta^{2}}{\mu}\Delta_{\tilde}^{s},
\end{aligned}
\tag{14}
\]
where we used $\Delta_{0}^{s}= \Delta_{\tilde}^{s}$ (the inner loop starts from the snapshot) and the geometric sum bound $\sum_{k=0}^{m-1}(1-\mu\eta)^{k}\le 1/(\mu\eta)$.

\noindent\textbf{[Step 10]} (Define the next snapshot).  
By definition $\tilde{x}^{s+1}=x_{m}^{s}$, thus $\Delta_{\tilde}^{s+1}= \Delta_{m}^{s}$.  
Taking expectation in (14) gives
\[
\E\!\left[\Delta_{\tilde}^{s+1}\right]
    \le\underbrace{\Bigl[(1-\mu\eta)^{m}+\frac{L^{2}\eta^{2}}{\mu}\Bigr]}_{\displaystyle \rho_{m}}
      \E\!\left[\Delta_{\tilde}^{s}\right].
\tag{15}
\]

\noindent\textbf{[Step 11]} (Choosing $m$).  
Using the inequality $(1-\mu\eta)^{m}\le 1-m\mu\eta$ (since $0<\mu\eta<1$) and the condition (A2) $m\ge 2L/(\mu)$,
\[
(1-\mu\eta)^{m}\le 1-\frac{2L\eta}{m}\le 1-\frac{L\eta}{m}.
\]
Hence
\[
\rho_{m}\le 1-\eta\mu+\frac{L\eta^{2}}{\mu m}
      =: \rho .
\tag{16}
\]
Because of (A1) and (A2) we have $0<\rho<1$, establishing the linear contraction (1).

\noindent\textbf{[Step 12]} (Iterating over epochs).  
Repeatedly applying (1) for $S$ epochs gives
\[
\E\!\left[f(\tilde{x}^{S})-f^{*}\right]\le\rho^{S}\bigl(f(\tilde{x}^{0})-f^{*}\bigr),
\]
which completes the proof. \qed

\section*{Rate Derivation (With Constants)}
The contraction factor derived in (16) can be expressed explicitly as
\[
\rho = 1-\eta\mu+\frac{L\eta^{2}}{\mu m}.
\]
Choosing the maximal admissible step size $\eta = \frac{1}{3L}$ and $m = \frac{4L}{\mu}$ (which satisfies (A2)) gives
\[
\rho = 1-\frac{\mu}{3L}+\frac{L}{\mu}\cdot\frac{1}{9L^{2}}\cdot\frac{\mu}{4L}
      = 1-\frac{\mu}{3L}+\frac{1}{36}
      = 1-\Bigl(\frac{\mu}{3L}-\frac{1}{36}\Bigr).
\]
Since $\mu\le L$, the term in parentheses is positive, guaranteeing a concrete linear rate.

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Exact PL with $\mu=L$.} When $f$ is $L$‑smooth and also $L$‑PL, setting $\eta=1/(3L)$ and $m=2$ yields $\rho=1-\frac{1}{3}+\frac{1}{9}= \frac{7}{9}$, i.e. a $13\%$ reduction per epoch.
    \item \textbf{Strongly convex case.} Strong convexity with parameter $\mu$ implies the PL condition. The theorem therefore recovers the classic linear convergence of SVRG for strongly convex objectives.
    \item \textbf{Deterministic limit.} If $m=n$ and each inner sample cycles through all indices, the variance term vanishes and $\rho=1-\eta\mu$, matching the rate of full‑gradient descent.
\end{enumerate}
