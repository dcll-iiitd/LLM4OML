\section*{Algorithm Name}
Single-Sample Stochastic Gradient Method (SGM)

\section*{Mathematical Formulation}
We consider minimizing a (potentially non-convex) differentiable function $f:\mathbb{R}^d\to\mathbb{R}$ given access to a stochastic first-order oracle. At iteration $t$:
\begin{itemize}
    \item Sample a data point $z_t$ (or a random seed) from a fixed distribution $\mathcal{D}$.
    \item Compute a stochastic gradient $g_t \coloneqq \nabla f(w_t; z_t)$ that satisfies $\mathbb{E}[g_t \mid w_t] = \nabla f(w_t)$.
    \item Update the iterate by a stepsize $\eta_t>0$:
    \[
        w_{t+1} \coloneqq w_t - \eta_t \, g_t.
    \]
\end{itemize}
Hyperparameters: initialization $w_0\in\mathbb{R}^d$, stepsize schedule $\{\eta_t\}_{t\ge 0}$, number of iterations $T$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Single-Sample Stochastic Gradient Method (SGM)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, stepsizes $\{\eta_t\}_{t=0}^{T-1}$, total iterations $T$
\For{$t=0,1,2,\dots,T-1$}
    \State Sample $z_t \sim \mathcal{D}$
    \State Compute stochastic gradient $g_t \gets \nabla f(w_t; z_t)$
    \State Update $w_{t+1} \gets w_t - \eta_t g_t$
\EndFor
\State \textbf{Output:} either the last iterate $w_T$ or a randomly selected iterate from $\{w_t\}_{t=0}^{T-1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We apply the method to the deterministic quadratic $f(w)=(w-3)^2$ in one dimension with $w_0=0$, constant stepsize $\eta=0.1$, for $3$ iterations (no stochasticity; $g_t = \nabla f(w_t)=2(w_t-3)$).
\begin{itemize}
    \item Iteration $t=0$:
    \[
        w_0=0,\quad g_0=2(0-3)=-6,\quad w_1=w_0-\eta g_0=0-0.1(-6)=0.6,
    \]
    \[
        f(w_0)=(0-3)^2=9,\quad f(w_1)=(0.6-3)^2=(-2.4)^2=5.76.
    \]
    \item Iteration $t=1$:
    \[
        w_1=0.6,\quad g_1=2(0.6-3)=2(-2.4)=-4.8,\quad w_2=0.6-0.1(-4.8)=1.08,
    \]
    \[
        f(w_2)=(1.08-3)^2=(-1.92)^2=3.6864.
    \]
    \item Iteration $t=2$:
    \[
        w_2=1.08,\quad g_2=2(1.08-3)=2(-1.92)=-3.84,\quad w_3=1.08-0.1(-3.84)=1.464,
    \]
    \[
        f(w_3)=(1.464-3)^2=(-1.536)^2=2.359296.
    \]
\end{itemize}

\section*{Assumptions}
We analyze the method on non-convex smooth objectives under standard stochastic oracle conditions.
\begin{assumption}[Smoothness]
\label{assump:smooth}
The gradient of $f$ is $L$-Lipschitz (i.e., $f$ is $L$-smooth): for all $x,y\in\mathbb{R}^d$,
\[
\|\nabla f(x)-\nabla f(y)\| \le L \|x-y\|.
\]
Equivalently, for all $x,y\in\mathbb{R}^d$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{assumption}
\begin{assumption}[Unbiasedness]
\label{assump:unbiased}
The stochastic gradient is unbiased: for all $t$,
\[
\mathbb{E}[g_t \mid w_t] = \nabla f(w_t).
\]
\end{assumption}
\begin{assumption}[Bounded variance]
\label{assump:var}
There exists $\sigma^2 \ge 0$ such that for all $t$,
\[
\mathbb{E}\big[\|g_t - \nabla f(w_t)\|^2 \,\big|\, w_t\big] \le \sigma^2.
\]
\end{assumption}
\begin{assumption}[Lower boundedness]
\label{assump:lower}
$f$ is bounded below by $f_\star \in \mathbb{R}$: for all $x$, $f(x) \ge f_\star$.
\end{assumption}
\begin{assumption}[Stepsizes]
\label{assump:stepsizes}
Stepsizes satisfy $0<\eta_t \le \frac{1}{L}$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence to stationary points in non-convex smooth case]
\label{thm:nonconvex}
Under Assumptions \ref{assump:smooth}--\ref{assump:stepsizes}, define $\alpha_t \coloneqq \eta_t - \frac{L}{2}\eta_t^2$. Then for any $T\ge 1$,
\begin{equation}
\sum_{t=0}^{T-1} \alpha_t \,\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\;\le\; f(w_0) - f_\star \;+\; \frac{L}{2}\,\sigma^2 \sum_{t=0}^{T-1} \eta_t^2.
\label{eq:key-sum}
\end{equation}
In particular, if $\eta_t\equiv \eta \le \frac{1}{L}$ is constant, then $\alpha_t \ge \frac{\eta}{2}$ and
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\;\le\; \frac{2\,(f(w_0)-f_\star)}{\eta\,T} \;+\; L\,\sigma^2\,\eta.
\label{eq:avg-grad-const-eta}
\end{equation}
Choosing $\eta = \min\left\{\tfrac{1}{L}, \sqrt{\tfrac{2\,(f(w_0)-f_\star)}{L\,\sigma^2\,T}}\right\}$ yields the rate
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\;\le\; 2\sqrt{\frac{2L\,\sigma^2\,(f(w_0)-f_\star)}{T}}.
\label{eq:rate-O1sqrtT}
\end{equation}
Equivalently, if an output iterate $W_R$ is chosen uniformly at random from $\{w_t\}_{t=0}^{T-1}$ under constant stepsize $\eta\le 1/L$, then
\begin{equation}
\mathbb{E}\big[\|\nabla f(W_R)\|^2\big]
\;\le\; \frac{2\,(f(w_0)-f_\star)}{\eta\,T} \;+\; L\,\sigma^2\,\eta.
\label{eq:uniform-output}
\end{equation}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item Stability and noise floor: With constant stepsize, the bound \eqref{eq:avg-grad-const-eta} contains a bias term $2(f(w_0)-f_\star)/(\eta T)$ and a variance-induced floor $L\sigma^2\eta$. As $T\to\infty$, the average stationarity measure stabilizes near $O(\eta)$.
    \item Hyperparameter sensitivity: Smaller $\eta$ reduces the noise floor but slows transient decrease; optimizing $\eta$ as in \eqref{eq:rate-O1sqrtT} balances both.
    \item Comparison to deterministic gradient descent: When $\sigma^2=0$, the floor disappears and the average squared gradient decays as $O(1/T)$ with a sufficiently large constant stepsize $\eta\le 1/L$.
    \item Minibatching: If $g_t$ averages $B$ i.i.d.\ samples, $\sigma^2$ effectively scales as $\sigma^2/B$, improving the floor and the constants in \eqref{eq:avg-grad-const-eta}--\eqref{eq:rate-O1sqrtT}.
\end{itemize}

\section*{Discussion}
Although the original update is a subgradient step, under smoothness the subgradient equals the gradient, and the analysis specializes to classical stochastic gradient descent for non-convex smooth $f$. The single-sample nature yields the canonical $O(1/\sqrt{T})$ rate (in terms of the squared gradient norm), which is optimal up to constants for this oracle model without additional assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
We collect standard tools used in the proof.

\begin{definition}[Filtration]
Let $\mathcal{F}_t \coloneqq \sigma(w_0, z_0, \dots, z_{t-1})$ denote the canonical filtration up to time $t$, so that $w_t$ is $\mathcal{F}_t$-measurable, and $z_t$ (hence $g_t$) is independent of $\mathcal{F}_t$ except through $w_t$.
\end{definition}

\begin{lemma}[Descent lemma]
\label{lem:descent}
Under Assumption \ref{assump:smooth}, for all $x,y\in\mathbb{R}^d$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{lemma}
\begin{proof}
This is equivalent to $L$-Lipschitz continuity of $\nabla f$ (standard). A direct proof uses the fundamental theorem of calculus along the line segment from $x$ to $y$ and bounds the Hessian in operator norm by $L$ when it exists, or applies the integral form without requiring Hessian explicitly. We omit repeated details.
\end{proof}

\begin{lemma}[Second-moment decomposition]
\label{lem:second-moment}
For any random vector $X$ with $\mu=\mathbb{E}[X]$, one has $\mathbb{E}[\|X\|^2]=\|\mu\|^2+\mathbb{E}[\|X-\mu\|^2]$.
\end{lemma}
\begin{proof}
Expand $\|X\|^2 = \|X-\mu+\mu\|^2 = \|X-\mu\|^2 + 2\langle X-\mu,\mu\rangle + \|\mu\|^2$ and take expectations; the cross term vanishes since $\mathbb{E}[X-\mu]=0$.
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
We prove Theorem \ref{thm:nonconvex}.

Let $t\in\{0,1,\dots,T-1\}$ be arbitrary. Apply Lemma \ref{lem:descent} with $x=w_t$ and $y=w_{t+1}=w_t-\eta_t g_t$:
\begin{align}
f(w_{t+1})
&\le f(w_t) + \left\langle \nabla f(w_t),\, w_{t+1}-w_t \right\rangle + \frac{L}{2}\,\|w_{t+1}-w_t\|^2 \quad \text{[Step 1: smoothness inequality]} \nonumber\\
&= f(w_t) + \left\langle \nabla f(w_t),\, -\eta_t g_t \right\rangle + \frac{L}{2}\,\|\,-\eta_t g_t\|^2 \quad \text{[Step 2: substitute $w_{t+1}-w_t=-\eta_t g_t$]} \nonumber\\
&= f(w_t) - \eta_t \langle \nabla f(w_t), g_t \rangle + \frac{L}{2}\,\eta_t^2\,\|g_t\|^2. \quad \text{[Step 3: algebra]} \label{eq:one-step}
\end{align}
Take conditional expectation with respect to $\mathcal{F}_t$ on both sides of \eqref{eq:one-step}:
\begin{align}
\mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t]
&\le f(w_t) - \eta_t \,\mathbb{E}\big[\langle \nabla f(w_t), g_t \rangle \mid \mathcal{F}_t\big]
+ \frac{L}{2}\,\eta_t^2\, \mathbb{E}\big[\|g_t\|^2 \mid \mathcal{F}_t\big] \quad \text{[Step 4: linearity of conditional expectation]} \nonumber\\
&= f(w_t) - \eta_t \,\langle \nabla f(w_t), \mathbb{E}[g_t \mid \mathcal{F}_t]\rangle
+ \frac{L}{2}\,\eta_t^2\, \mathbb{E}\big[\|g_t\|^2 \mid \mathcal{F}_t\big] \quad \text{[Step 5: pull out $\nabla f(w_t)$, $\mathcal{F}_t$-measurable]} \nonumber\\
&= f(w_t) - \eta_t \,\|\nabla f(w_t)\|^2
+ \frac{L}{2}\,\eta_t^2\, \mathbb{E}\big[\|g_t\|^2 \mid \mathcal{F}_t\big] \quad \text{[Step 6: Assumption \ref{assump:unbiased}, $\mathbb{E}[g_t\mid \mathcal{F}_t]=\nabla f(w_t)$]} \label{eq:cond-exp-1}
\end{align}
Decompose the second moment using Lemma \ref{lem:second-moment} with $X=g_t$ and $\mu=\nabla f(w_t)$:
\begin{align}
\mathbb{E}\big[\|g_t\|^2 \mid \mathcal{F}_t\big]
&= \|\nabla f(w_t)\|^2 + \mathbb{E}\big[\|g_t - \nabla f(w_t)\|^2 \mid \mathcal{F}_t\big] \quad \text{[Step 7: conditional version of Lemma \ref{lem:second-moment}]} \nonumber\\
&\le \|\nabla f(w_t)\|^2 + \sigma^2. \quad \text{[Step 8: Assumption \ref{assump:var}]} \label{eq:var-bound}
\end{align}
Plug \eqref{eq:var-bound} into \eqref{eq:cond-exp-1}:
\begin{align}
\mathbb{E}[f(w_{t+1}) \mid \mathcal{F}_t]
&\le f(w_t) - \eta_t \,\|\nabla f(w_t)\|^2
+ \frac{L}{2}\,\eta_t^2\, \big(\|\nabla f(w_t)\|^2 + \sigma^2\big) \quad \text{[Step 9: substitution]} \nonumber\\
&= f(w_t) - \Big(\eta_t - \frac{L}{2}\eta_t^2\Big)\,\|\nabla f(w_t)\|^2 + \frac{L}{2}\,\eta_t^2\,\sigma^2. \quad \text{[Step 10: collect terms]} \label{eq:cond-recursion}
\end{align}
Define $\alpha_t \coloneqq \eta_t - \frac{L}{2}\eta_t^2$. Take total expectation and use tower property $\mathbb{E}[\mathbb{E}[X\mid \mathcal{F}_t]]=\mathbb{E}[X]$:
\begin{align}
\mathbb{E}[f(w_{t+1})]
&\le \mathbb{E}[f(w_t)] - \alpha_t \,\mathbb{E}\big[\|\nabla f(w_t)\|^2\big] + \frac{L}{2}\,\eta_t^2\,\sigma^2. \quad \text{[Step 11: tower property and linearity]} \label{eq:total-recursion}
\end{align}
Sum \eqref{eq:total-recursion} over $t=0$ to $T-1$:
\begin{align}
\mathbb{E}[f(w_T)]
&\le \mathbb{E}[f(w_0)] - \sum_{t=0}^{T-1} \alpha_t \,\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
+ \frac{L}{2}\,\sigma^2 \sum_{t=0}^{T-1} \eta_t^2 \quad \text{[Step 12: telescoping sum for $\mathbb{E}[f(w_t)]$]} \nonumber\\
\Longrightarrow\quad
\sum_{t=0}^{T-1} \alpha_t \,\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
&\le \mathbb{E}[f(w_0)] - \mathbb{E}[f(w_T)] + \frac{L}{2}\,\sigma^2 \sum_{t=0}^{T-1} \eta_t^2. \quad \text{[Step 13: rearrange terms]} \label{eq:pre-final}
\end{align}
Use Assumption \ref{assump:lower}, $f(w_T)\ge f_\star$ almost surely, hence $\mathbb{E}[f(w_T)]\ge f_\star$:
\begin{align}
\sum_{t=0}^{T-1} \alpha_t \,\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
&\le f(w_0) - f_\star + \frac{L}{2}\,\sigma^2 \sum_{t=0}^{T-1} \eta_t^2, \quad \text{[Step 14: apply $f(w_T)\ge f_\star$ inside expectation]} \label{eq:key-sum-proof}
\end{align}
which is exactly \eqref{eq:key-sum}. This proves the first claim.

Now suppose $\eta_t\equiv \eta \le \frac{1}{L}$. Then
\begin{align}
\alpha_t = \eta - \frac{L}{2}\eta^2 = \eta\Big(1-\frac{L}{2}\eta\Big) \ge \frac{\eta}{2}, \quad \text{[Step 15: since $\eta\le \tfrac{1}{L}\Rightarrow 1-\tfrac{L}{2}\eta \ge \tfrac{1}{2}$]} \label{eq:alpha-lb}
\end{align}
and $\sum_{t=0}^{T-1} \eta_t^2 = T\eta^2$. Divide both sides of \eqref{eq:key-sum-proof} by $T\,\frac{\eta}{2}$ and use \eqref{eq:alpha-lb}:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
&\le \frac{2\,(f(w_0)-f_\star)}{\eta\,T} + \frac{L}{2}\,\sigma^2 \cdot \frac{2\,T\eta^2}{T\eta} \quad \text{[Step 16: divide both sides, algebra]} \nonumber\\
&= \frac{2\,(f(w_0)-f_\star)}{\eta\,T} + L\,\sigma^2\,\eta, \quad \text{[Step 17: simplify]} \label{eq:avg-bound-proof}
\end{align}
which is \eqref{eq:avg-grad-const-eta}. Selecting the random output $W_R$ uniformly from $\{w_t\}$ yields \eqref{eq:uniform-output} immediately since the left side of \eqref{eq:avg-bound-proof} is exactly $\mathbb{E}[\|\nabla f(W_R)\|^2]$.

Finally, choose
\[
\eta^\star \coloneqq \min\left\{\frac{1}{L}, \sqrt{\frac{2\,(f(w_0)-f_\star)}{L\,\sigma^2\,T}}\right\}.
\]
If $\eta^\star = \sqrt{\frac{2(f(w_0)-f_\star)}{L\sigma^2 T}} \le \frac{1}{L}$, plug into \eqref{eq:avg-grad-const-eta}:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
&\le \frac{2\,(f(w_0)-f_\star)}{T\,\eta^\star} + L\,\sigma^2\,\eta^\star \quad \text{[Step 18: substitute $\eta^\star$]} \nonumber\\
&= \frac{2\,(f(w_0)-f_\star)}{T} \sqrt{\frac{L\,\sigma^2\,T}{2\,(f(w_0)-f_\star)}} \;+\; L\,\sigma^2 \sqrt{\frac{2\,(f(w_0)-f_\star)}{L\,\sigma^2\,T}} \quad \text{[Step 19: algebra]} \nonumber\\
&= \sqrt{\frac{2L\,\sigma^2\,(f(w_0)-f_\star)}{T}} \;+\; \sqrt{\frac{2L\,\sigma^2\,(f(w_0)-f_\star)}{T}} \quad \text{[Step 20: simplify radicals]} \nonumber\\
&= 2\sqrt{\frac{2L\,\sigma^2\,(f(w_0)-f_\star)}{T}}, \quad \text{[Step 21: sum identical terms]} \label{eq:rate-T-known}
\end{align}
which is \eqref{eq:rate-O1sqrtT}. If instead $\eta^\star=\frac{1}{L}$, then \eqref{eq:avg-grad-const-eta} gives
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
&\le \frac{2L\,(f(w_0)-f_\star)}{T} + \sigma^2, \quad \text{[Step 22: plug $\eta=1/L$]} \label{eq:noise-floor}
\end{align}
exhibiting a $\sigma^2$ noise floor for large $T$. This completes the proof.
\qedhere
\end{theorem}

\section*{Rate Derivation (With Constants)}
We summarize explicit rates under different stepsize policies.
\begin{itemize}
    \item Constant stepsize $\eta \le 1/L$:
    \[
    \frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\|\nabla f(w_t)\|^2
    \le \frac{2\,(f(w_0)-f_\star)}{\eta\,T} + L\,\sigma^2\,\eta.
    \]
    Optimizing over $\eta$ with horizon $T$ known gives
    \[
    \inf_{\eta\in(0,1/L]} \left\{\frac{2\,(f(w_0)-f_\star)}{\eta\,T} + L\,\sigma^2\,\eta \right\}
    \le 2\sqrt{\frac{2L\,\sigma^2\,(f(w_0)-f_\star)}{T}},
    \]
    achieved by $\eta^\star = \min\left\{\frac{1}{L}, \sqrt{\frac{2(f(w_0)-f_\star)}{L\sigma^2 T}}\right\}$.
    \item Diminishing stepsize $\eta_t = \frac{\eta_0}{\sqrt{t+1}}$ with $\eta_0 \le \frac{1}{L}$:
    Using \eqref{eq:key-sum} with $\alpha_t \ge \frac{\eta_t}{2}$ and $\sum_{t=0}^{T-1} \eta_t \ge \eta_0(\sqrt{T}-1)$, $\sum_{t=0}^{T-1} \eta_t^2 \le \eta_0^2(1+\log T)$, we obtain
    \[
    \frac{\sum_{t=0}^{T-1} \eta_t\, \mathbb{E}\|\nabla f(w_t)\|^2}{\sum_{t=0}^{T-1} \eta_t}
    \le \frac{2\,(f(w_0)-f_\star)}{\sum_{t=0}^{T-1} \eta_t} + L\,\sigma^2 \cdot \frac{\sum_{t=0}^{T-1} \eta_t^2}{\sum_{t=0}^{T-1} \eta_t}
    = O\!\left(\frac{1}{\sqrt{T}} + \frac{\log T}{\sqrt{T}}\right),
    \]
    i.e., $O(\log T/\sqrt{T})$ for the stepsize-weighted average; with slightly faster-decaying stepsizes the log factor can be removed in the average of iterates.
\end{itemize}

\section*{Special Cases}
\begin{itemize}
    \item Deterministic gradients ($\sigma^2=0$): From \eqref{eq:avg-grad-const-eta} with $\eta\le 1/L$,
    \[
    \frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\|\nabla f(w_t)\|^2
    \le \frac{2\,(f(w_0)-f_\star)}{\eta\,T},
    \]
    achieving $O(1/T)$ decay of the average squared gradient norm.
    \item Minibatch of size $B$: If $g_t$ averages $B$ i.i.d.\ samples, then Assumption \ref{assump:var} holds with $\sigma^2/B$, improving \eqref{eq:avg-grad-const-eta} to
    \[
    \frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\|\nabla f(w_t)\|^2
    \le \frac{2\,(f(w_0)-f_\star)}{\eta\,T} + \frac{L\,\sigma^2}{B}\,\eta.
    \]
    \item Projection-free unconstrained vs constrained: The analysis above is for unconstrained updates; with projections onto a convex set $\mathcal{W}$, the same proof applies replacing $w_{t+1}$ by $\Pi_{\mathcal{W}}(w_t-\eta_t g_t)$ and adding a standard non-expansiveness argument.
\end{itemize}
