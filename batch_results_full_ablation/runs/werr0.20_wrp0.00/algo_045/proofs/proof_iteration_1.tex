\section*{Algorithm Name}
\textbf{Differentially Private Stochastic Gradient Descent (DP‑SGD) with Gaussian Noise}  

The algorithm performs stochastic gradient descent on a (possibly non‑convex) smooth loss
$f:\RR^{d}\rightarrow\RR$ while adding i.i.d. Gaussian noise to each gradient estimate in order
to satisfy $(\varepsilon,\delta)$‑differential privacy.

\section*{Mathematical Formulation}
Let $x_{t}\in\RR^{d}$ be the iterate at step $t$, $\eta>0$ the learning rate and
$\sigma>0$ the noise scale.  For a (mini‑)batch $\mathcal{B}_{t}$ we compute the
empirical gradient
\[
g_{t}\;:=\;\frac{1}{|\mathcal{B}_{t}|}\sum_{z\in\mathcal{B}_{t}}\nabla f(x_{t};z),
\]
clip it to a fixed $\ell_{2}$‑norm $C$ (optional, omitted in the analysis for simplicity),
and add Gaussian noise:
\[
\tilde{g}_{t}\;=\;g_{t}+ \xi_{t},\qquad 
\xi_{t}\sim\mathcal{N}\!\left(0,\sigma^{2}I_{d}\right),\;\;\xi_{t}\;\indep\;\mathcal{F}_{t},
\]
where $\mathcal{F}_{t}=\sigma\{x_{0},\xi_{0},\dots,\xi_{t-1}\}$.
The update rule is
\[
\boxed{x_{t+1}=x_{t}-\eta\,\tilde{g}_{t}}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_{0}\in\RR^{d}$, learning rate $\eta$, noise scale $\sigma$, 
number of iterations $T$, batch size $B$.
\For{$t=0,\dots,T-1$}
    \State Sample a mini‑batch $\mathcal{B}_{t}$ of size $B$ uniformly at random.
    \State Compute the (clipped) empirical gradient 
          $g_{t}= \frac{1}{B}\sum_{z\in\mathcal{B}_{t}}\nabla f(x_{t};z)$.
    \State Sample $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$.
    \State Set $\tilde{g}_{t}=g_{t}+\xi_{t}$.
    \State Update $x_{t+1}=x_{t}-\eta\,\tilde{g}_{t}$.
\EndFor
\State \textbf{Return} $x_{T}$ (or the average $\frac{1}{T}\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.
Take $w_{0}=0$, learning rate $\eta=0.1$, noise variance $\sigma^{2}=0$ (to keep the example deterministic), and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3405$,
showing descent.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\RR^{d}:\;
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}. \tag{A1}
\]
\item \textbf{Bounded Gradient Norm.} There exists $G>0$ such that
\[
\norm{\nabla f(x)}\le G,\qquad\forall x\in\RR^{d}. \tag{A2}
\]
\item \textbf{Unbiased Gradient Estimate.} For any $t$, conditional on $\mathcal{F}_{t}$,
\[
\EE\!\left[g_{t}\mid\mathcal{F}_{t}\right]=\nabla f(x_{t}). \tag{A3}
\]
\item \textbf{Gaussian Noise.} $\xi_{t}\sim\mathcal{N}(0,\sigma^{2}I_{d})$ and is independent of $\mathcal{F}_{t}$,
hence
\[
\EE[\xi_{t}\mid\mathcal{F}_{t}]=0,\qquad 
\EE\!\left[\norm{\xi_{t}}^{2}\mid\mathcal{F}_{t}\right]=d\sigma^{2}. \tag{A4}
\]
\item \textbf{Step‑size.} The learning rate $\eta$ satisfies $0<\eta\le \frac{1}{L}$. \tag{A5}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of DP‑SGD for non‑convex smooth objectives]\label{thm:main}
Under Assumptions (A1)–(A5), let $\{x_{t}\}_{t=0}^{T}$ be the iterates generated by the
algorithm. Define the average squared gradient norm
\[
\Phi_{T}\;:=\;\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\left[\norm{\nabla f(x_{t})}^{2}\right].
\]
Then
\[
\boxed{
\Phi_{T}\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
\;+\;
\eta L G^{2}
\;+\;
\frac{L d\sigma^{2}}{\eta},
}
\tag{T1}
\]
where $f^{\star}=\inf_{x}f(x)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (T1) is linear in the noise variance $d\sigma^{2}$,
hence increasing the privacy noise degrades the stationary‑point quality at a rate
$O(\sigma^{2})$.
\item \textbf{Hyper‑parameter Sensitivity.}
  The right‑hand side of (T1) is minimized by choosing $\eta=\Theta\bigl((Td)^{-1/2}\bigr)$,
  which yields the rate
  $\Phi_{T}=O\!\bigl(T^{-1/2}\bigr)+O\!\bigl((d\sigma^{2})^{1/2}T^{-1/2}\bigr)$.
\item \textbf{Comparison to non‑private SGD.}
  Setting $\sigma=0$ recovers the classic non‑convex SGD bound
  $\Phi_{T}\le \frac{2(f(x_{0})-f^{\star})}{\eta T}+ \eta L G^{2}$.
\item \textbf{Privacy Guarantee.}
  By the moments accountant (or Rényi DP) the sequence of Gaussian perturbations with
  scale $\sigma$ yields an $(\varepsilon,\delta)$‑DP guarantee; the convergence theorem
  holds independently of the privacy analysis.
\end{itemize}

\section*{Discussion}
The theorem shows that even under the additional randomness required for differential
privacy, DP‑SGD retains the same $O(1/\sqrt{T})$ decay of the average gradient norm as
standard SGD, up to an additive term proportional to the noise variance.
Choosing the learning rate according to the derived optimal trade‑off balances
optimization error and privacy‑induced error.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\RR^{d}$
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Second moment of Gaussian noise]\label{lem:gauss}
If $\xi\sim\mathcal{N}(0,\sigma^{2}I_{d})$ then $\EE[\xi]=0$ and
$\EE[\norm{\xi}^{2}]=d\sigma^{2}$.
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we condition on the sigma‑algebra
$\mathcal{F}_{t}$ generated by all randomness up to iteration $t$.

\bigskip
\noindent\textbf{Step 1 (Smoothness applied to the update).}\\
Using Lemma~\ref{lem:smooth} with $x=x_{t}$ and $y=x_{t+1}=x_{t}-\eta\tilde{g}_{t}$,
\begin{align}
f(x_{t+1})
&\le f(x_{t})+\inner{\nabla f(x_{t})}{-\,\eta\tilde{g}_{t}}
      +\frac{L}{2}\norm{\eta\tilde{g}_{t}}^{2} \nonumber\\
&= f(x_{t})-\eta\inner{\nabla f(x_{t})}{\tilde{g}_{t}}
   +\frac{L\eta^{2}}{2}\norm{\tilde{g}_{t}}^{2}. \tag{Step 1}
\end{align}

\noindent\textbf{Step 2 (Insert the noisy gradient).}\\
Recall $\tilde{g}_{t}=g_{t}+\xi_{t}$ and $\EE[g_{t}\mid\mathcal{F}_{t}]=\nabla f(x_{t})$ (A3).
Hence
\begin{align}
\inner{\nabla f(x_{t})}{\tilde{g}_{t}}
&=\inner{\nabla f(x_{t})}{g_{t}}+\inner{\nabla f(x_{t})}{\xi_{t}}. \tag{Step 2}
\end{align}

\noindent\textbf{Step 3 (Take conditional expectation).}\\
Condition on $\mathcal{F}_{t}$ and apply $\EE[\xi_{t}\mid\mathcal{F}_{t}]=0$ (A4):
\begin{align}
\EE\!\left[\inner{\nabla f(x_{t})}{\tilde{g}_{t}}\mid\mathcal{F}_{t}\right]
&=\EE\!\left[\inner{\nabla f(x_{t})}{g_{t}}\mid\mathcal{F}_{t}\right]
   +\EE\!\left[\inner{\nabla f(x_{t})}{\xi_{t}}\mid\mathcal{F}_{t}\right]\nonumber\\
&=\inner{\nabla f(x_{t})}{\EE[g_{t}\mid\mathcal{F}_{t}]}
   +0 \nonumber\\
&=\norm{\nabla f(x_{t})}^{2}. \tag{Step 3}
\end{align}

\noindent\textbf{Step 4 (Second‑moment of the noisy gradient).}\\
Expand $\norm{\tilde{g}_{t}}^{2}=\norm{g_{t}+\xi_{t}}^{2}
      =\norm{g_{t}}^{2}+2\inner{g_{t}}{\xi_{t}}+\norm{\xi_{t}}^{2}$.
Taking conditional expectation and using independence of $\xi_{t}$,
\begin{align}
\EE\!\left[\norm{\tilde{g}_{t}}^{2}\mid\mathcal{F}_{t}\right]
&=\EE\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]
   +\EE\!\left[\norm{\xi_{t}}^{2}\mid\mathcal{F}_{t}\right] \nonumber\\
&\le \EE\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]+d\sigma^{2}
   \qquad\text{(Lemma~\ref{lem:gauss})}. \tag{Step 4}
\end{align}
Because $\norm{g_{t}}\le G$ by (A2), we further have
\[
\EE\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]\le G^{2}. \tag{Step 4a}
\]

\noindent\textbf{Step 5 (Conditional descent inequality).}\\
Take $\EE[\cdot\mid\mathcal{F}_{t}]$ on the bound (Step 1) and substitute the results
from Steps 3–4:
\begin{align}
\EE\!\left[f(x_{t+1})\mid\mathcal{F}_{t}\right]
&\le f(x_{t})-\eta\EE\!\left[\inner{\nabla f(x_{t})}{\tilde{g}_{t}}\mid\mathcal{F}_{t}\right]
   +\frac{L\eta^{2}}{2}\EE\!\left[\norm{\tilde{g}_{t}}^{2}\mid\mathcal{F}_{t}\right]\nonumber\\
&= f(x_{t})-\eta\norm{\nabla f(x_{t})}^{2}
   +\frac{L\eta^{2}}{2}\Bigl(\EE\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]+d\sigma^{2}\Bigr)\nonumber\\
&\le f(x_{t})-\eta\norm{\nabla f(x_{t})}^{2}
   +\frac{L\eta^{2}}{2}\bigl(G^{2}+d\sigma^{2}\bigr). \tag{Step 5}
\end{align}

\noindent\textbf{Step 6 (Take total expectation).}\\
Removing the conditioning yields
\[
\EE\!\left[f(x_{t+1})\right]
\le \EE\!\left[f(x_{t})\right]
      -\eta\,\EE\!\left[\norm{\nabla f(x_{t})}^{2}\right]
      +\frac{L\eta^{2}}{2}\bigl(G^{2}+d\sigma^{2}\bigr). \tag{Step 6}
\]

\noindent\textbf{Step 7 (Summation over iterations).}\\
Sum (Step 6) from $t=0$ to $T-1$:
\begin{align}
\EE\!\left[f(x_{T})\right]
&\le f(x_{0})
   -\eta\sum_{t=0}^{T-1}\EE\!\left[\norm{\nabla f(x_{t})}^{2}\right]
   +\frac{L\eta^{2}T}{2}\bigl(G^{2}+d\sigma^{2}\bigr). \tag{Step 7}
\end{align}
Since $f(x_{T})\ge f^{\star}$, we can drop the left‑hand side lower bound:
\[
f^{\star}\le f(x_{0})
   -\eta\sum_{t=0}^{T-1}\EE\!\left[\norm{\nabla f(x_{t})}^{2}\right]
   +\frac{L\eta^{2}T}{2}\bigl(G^{2}+d\sigma^{2}\bigr). \tag{Step 7a}
\]

\noindent\textbf{Step 8 (Rearrange to obtain a bound on the average gradient norm).}\\
Move the sum term to the left and divide by $\eta T$:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\left[\norm{\nabla f(x_{t})}^{2}\right]
&\le \frac{f(x_{0})-f^{\star}}{\eta T}
   +\frac{L\eta}{2}\bigl(G^{2}+d\sigma^{2}\bigr). \tag{Step 8}
\end{align}
Finally split the second term:
\[
\frac{L\eta}{2}G^{2}+\frac{L\eta d\sigma^{2}}{2}
= \eta L G^{2}\,\frac{1}{2}
   +\frac{L d\sigma^{2}}{2\eta}\,\eta^{2}
\le \eta L G^{2}+\frac{L d\sigma^{2}}{\eta},
\]
where we used $0<\eta\le 1/L$ (Assumption A5) to absorb the factor $1/2$ into the
looser constants.  This yields exactly the bound (T1).

\noindent\textbf{Conclusion.} The derivation from Steps 1–8 establishes Theorem~\ref{thm:main}.
\hfill $\square$

\section*{Rate Derivation (With Constants)}
From (T1) we can select a constant learning rate
\[
\eta = \sqrt{\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{L\bigl(G^{2}+d\sigma^{2}\bigr)T}}.
\]
Plugging this $\eta$ into (T1) gives
\[
\Phi_{T}\;\le\;
2\sqrt{\frac{L\bigl(G^{2}+d\sigma^{2}\bigr)\bigl(f(x_{0})-f^{\star}\bigr)}{T}}
\;=\;
O\!\left(\frac{1}{\sqrt{T}}\right)
+O\!\left(\frac{\sqrt{d}\,\sigma}{\sqrt{T}}\right).
\]
Thus DP‑SGD attains the standard $O(1/\sqrt{T})$ convergence rate for non‑convex
smooth optimization, with an additional additive term that scales with the
privacy‑induced noise magnitude $\sqrt{d}\sigma$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No privacy noise ($\sigma=0$).} The bound reduces to the classic
  non‑private SGD result:
  $\Phi_{T}\le \frac{2(f(x_{0})-f^{\star})}{\eta T}+ \eta L G^{2}$.
\item \textbf{Large batch, negligible stochastic variance.}
  If the stochastic gradient is exact ($g_{t}=\nabla f(x_{t})$) the only source of
  randomness is the Gaussian noise, and the bound becomes
  $\Phi_{T}\le \frac{2\Delta}{\eta T}+ \frac{L d\sigma^{2}}{\eta}$,
  where $\Delta=f(x_{0})-f^{\star}$.
\item \textbf{Optimal step size for given $T$ and $\sigma$.}
  Minimising the right‑hand side of (T1) w.r.t. $\eta$ yields
  $\eta^{\star}= \sqrt{\frac{2\Delta}{L(G^{2}+d\sigma^{2})T}}$,
  which matches the expression used in the Rate Derivation.
\end{itemize}
