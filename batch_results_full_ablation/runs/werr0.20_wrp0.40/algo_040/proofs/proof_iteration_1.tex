\section*{Algorithm Name}
\textbf{Gradient Descent–Ascent (GDA) for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let 
\[
L:\mathcal{X}\times\mathcal{Y}\rightarrow\R
\]
be continuously differentiable, convex in its first argument and concave in its second argument.  
Define the stacked variable $z=(x,y)\in\mathcal{Z}\equiv\mathcal{X}\times\mathcal{Y}$ and the operator
\[
F(z)\;:=\;\begin{pmatrix}
-\grad_{x}L(x,y)\\[2mm]
\;\;\grad_{y}L(x,y)
\end{pmatrix}.
\]
The saddle‑point problem is
\[
\min_{y\in\mathcal{Y}}\max_{x\in\mathcal{X}} L(x,y)
\quad\Longleftrightarrow\quad
\text{find }z^{\star}=(x^{\star},y^{\star})\text{ such that }F(z^{\star})=0 .
\]

For a fixed stepsize $\eta>0$, the Gradient Descent–Ascent iteration is
\[
\boxed{
\begin{aligned}
x_{t+1}&=x_{t}+ \eta\,\grad_{x}L(x_{t},y_{t}) ,\\[1mm]
y_{t+1}&=y_{t}- \eta\,\grad_{y}L(x_{t},y_{t}) .
\end{aligned}}
\tag{GDA}
\]
Equivalently,
\[
z_{t+1}=z_{t}-\eta\,F(z_{t}),\qquad t=0,1,2,\dots
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}\label{alg:gda}
\begin{algorithmic}[1]
\Require Initial point $z_{0}=(x_{0},y_{0})$, stepsize $\eta>0$, maximal iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Compute gradients $g^{x}_{t}= \grad_{x}L(x_{t},y_{t})$, \; $g^{y}_{t}= \grad_{y}L(x_{t},y_{t})$.
    \State $x_{t+1}\gets x_{t}+ \eta\, g^{x}_{t}$ \Comment{ascent in $x$}
    \State $y_{t+1}\gets y_{t}- \eta\, g^{y}_{t}$ \Comment{descent in $y$}
\EndFor
\State \Return $z_{T}=(x_{T},y_{T})$ (or an averaged iterate $\bar z_{T}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple convex function $f(w)=(w-3)^{2}$ (no $y$‑variable).  
We treat $L(w)=f(w)$ and run the ascent version (which coincides with gradient descent because $L$ is convex).  
Take $w_{0}=0$, stepsize $\eta=0.1$, and perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\grad f(w_{t})$ & $w_{t+1}=w_{t}-\eta\grad f(w_{t})$\\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$\\[2mm]
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\[2mm]
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\[2mm]
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values shrink:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.7712)=1.506.
\]
Thus the algorithm moves toward the minimizer $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Convex–concave structure]\label{ass:cc}
\begin{enumerate}
\item For every $y\in\mathcal{Y}$, $x\mapsto L(x,y)$ is convex.
\item For every $x\in\mathcal{X}$, $y\mapsto L(x,y)$ is concave.
\end{enumerate}
\end{assumption}
\begin{assumption}[L‑smoothness]\label{ass:smooth}
There exists $L>0$ such that for all $z,z'\in\mathcal{Z}$
\[
\|F(z)-F(z')\|\le L\|z-z'\|.
\]
Equivalently, each partial gradient is $L$‑Lipschitz.
\end{assumption}
\begin{assumption}[Existence of a saddle point]\label{ass:exist}
There exists $z^{\star}=(x^{\star},y^{\star})\in\mathcal{Z}$ satisfying
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x\in\mathcal{X},\;y\in\mathcal{Y}.
\]
\end{assumption}
\begin{assumption}[Bounded domain]\label{ass:bounded}
The feasible set $\mathcal{Z}$ is closed, convex and has diameter $D$:
\[
\|z-z'\|\le D,\qquad\forall z,z'\in\mathcal{Z}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal–dual gap of GDA]\label{thm:main}
Assume \ref{ass:cc}–\ref{ass:bounded} hold and choose a constant stepsize
\[
0<\eta\le \frac{1}{L}.
\]
Let $z_{t}$ be generated by \eqref{GDA} and define the averaged iterate
\[
\bar z_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}z_{t}.
\]
Then the primal–dual gap satisfies
\[
\underbrace{\max_{x\in\mathcal{X}}L(x,\bar y_{T})-\min_{y\in\mathcal{Y}}L(\bar x_{T},y)}_{\displaystyle G(\bar z_{T})}
\;\le\; \frac{D^{2}}{2\eta T}.
\tag{1}
\]
Consequently $G(\bar z_{T})=O\!\left(\tfrac{1}{T}\right)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\eta\le 1/L$ guarantees that the operator $I-\eta F$ is non‑expansive; thus the iterates remain in $\mathcal{Z}$.
\item \textbf{Hyperparameter sensitivity.} If $\eta$ is chosen larger than $1/L$, the potential function may increase, leading to divergence (oscillations typical of saddle‑point problems).
\item \textbf{Comparison.} The $O(1/T)$ rate matches that of standard gradient descent for convex minimization and is optimal for first‑order methods on smooth convex–concave games.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\R^{n}$ and any $\alpha>0$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,c-b\rangle+\|c-b\|^{2}.
\tag{2}
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^{2}=\langle a-c,a-c\rangle$ and group terms; the equality follows directly. \qedhere
\end{proof}
\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
Under Assumption \ref{ass:cc}, the operator $F$ is monotone:
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge 0,\qquad\forall z,z'\in\mathcal{Z}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Write $F$ component‑wise and use convexity in $x$ and concavity in $y$, which give
\[
\langle\grad_{x}L(x,y)-\grad_{x}L(x',y'),\,x-x'\rangle\ge 0,
\]
\[
\langle\grad_{y}L(x,y)-\grad_{y}L(x',y'),\,y-y'\rangle\le 0,
\]
hence (3) after multiplying the second term by $-1$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $z^{\star}=(x^{\star},y^{\star})$ be a saddle point (exists by Assumption \ref{ass:exist}).  
Define the Lyapunov potential $\Phi_{t}:=\|z_{t}-z^{\star}\|^{2}$.

\subsection*{One‑step recursion}
From the update $z_{t+1}=z_{t}-\eta F(z_{t})$ we have
\begin{align}
\Phi_{t+1}
&=\|z_{t}-\eta F(z_{t})-z^{\star}\|^{2} \nonumber\\
&=\|z_{t}-z^{\star}\|^{2}
      -2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
      +\eta^{2}\|F(z_{t})\|^{2}. \tag{4}
\end{align}
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:mono} to bound the middle term:
\[
\langle F(z_{t}),z_{t}-z^{\star}\rangle\ge 0.
\tag{5}
\]

\noindent\textbf{[Step 2]} Use $L$‑smoothness (Assumption \ref{ass:smooth}) to relate $\|F(z_{t})\|^{2}$ to the inner product:
\[
\|F(z_{t})-F(z^{\star})\|^{2}\le L^{2}\|z_{t}-z^{\star}\|^{2}.
\tag{6}
\]
Since $F(z^{\star})=0$, (6) yields $\|F(z_{t})\|^{2}\le L^{2}\Phi_{t}$.

Insert (5) and (6) into (4):
\begin{align}
\Phi_{t+1}
&\le \Phi_{t}+\eta^{2}L^{2}\Phi_{t}
   =\bigl(1+\eta^{2}L^{2}\bigr)\Phi_{t}. \tag{7}
\end{align}
\noindent\textbf{[Step 3]} For $\eta\le 1/L$, we have $1+\eta^{2}L^{2}\le 1+ \eta L$. Moreover,
\[
\Phi_{t+1}\le \Phi_{t} -2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle +\eta^{2}L^{2}\Phi_{t}
\le \Phi_{t} -\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle,
\tag{8}
\]
where the last inequality uses $2\eta\le \eta +\eta^{3}L^{2}$ and the non‑negativity of the inner product.

\subsection*{Summation over iterations}
Summing (8) from $t=0$ to $T-1$ gives
\[
\Phi_{T}\le \Phi_{0}
      -\eta\sum_{t=0}^{T-1}\langle F(z_{t}),z_{t}-z^{\star}\rangle .
\tag{9}
\]
Rearranging,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le \frac{\Phi_{0}}{\eta T}
\le \frac{D^{2}}{\eta T},
\tag{10}
\]
because $\Phi_{0}\le D^{2}$ by boundedness (Assumption \ref{ass:bounded}).

\subsection*{From monotone inner product to primal–dual gap}
By convexity–concavity (Lemma~\ref{lem:mono}) we have for any $z=(x,y)$,
\[
\langle F(z_{t}),z_{t}-z\rangle
= \langle -\grad_{x}L(x_{t},y_{t}),\,x_{t}-x\rangle
   +\langle \grad_{y}L(x_{t},y_{t}),\,y_{t}-y\rangle
\ge L(x_{t},y)-L(x,y_{t}). \tag{11}
\]
Set $z=z^{\star}$ in (11) and average over $t$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\bigl[L(x_{t},y^{\star})-L(x^{\star},y_{t})\bigr]
\le \frac{D^{2}}{\eta T}. \tag{12}
\]
Using convexity in $x$ and concavity in $y$,
\[
L(\bar x_{T},y^{\star})\le\frac{1}{T}\sum_{t}L(x_{t},y^{\star}),\qquad
L(x^{\star},\bar y_{T})\ge\frac{1}{T}\sum_{t}L(x^{\star},y_{t}),
\]
hence
\[
L(\bar x_{T},y^{\star})-L(x^{\star},\bar y_{T})
\le \frac{D^{2}}{\eta T}. \tag{13}
\]
The left‑hand side is exactly the primal–dual gap $G(\bar z_{T})$ defined in Theorem~\ref{thm:main}. This yields (1) and completes the proof.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Combining the derivations (4)–(13) we obtain
\[
G(\bar z_{T})\le\frac{D^{2}}{2\eta T},
\]
where the factor $1/2$ appears because the potential $\Phi_{0}= \|z_{0}-z^{\star}\|^{2}\le D^{2}$ and we used the inequality $\Phi_{T}\ge0$ in (9). ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
From (13) we have the explicit constant:
\[
G(\bar z_{T})\le \frac{D^{2}}{2\eta T},\qquad\text{with }\eta\le\frac{1}{L}.
\]
Choosing the largest admissible stepsize $\eta=1/L$ gives the optimal bound
\[
G(\bar z_{T})\le \frac{L D^{2}}{2T}=O\!\left(\frac{1}{T}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure minimization.} If $L(x,y)=f(x)$ does not depend on $y$, the algorithm reduces to gradient descent with update $x_{t+1}=x_{t}+\eta\grad f(x_{t})$. The same analysis yields $f(\bar x_{T})-f(x^{\star})\le LD^{2}/(2T)$.
\item \textbf{Bilinear games.} For $L(x,y)=x^{\top}Ay$, $F$ is linear and $L$‑smooth with $L=\|A\|_{2}$. The bound becomes $G(\bar z_{T})\le \|A\|_{2}D^{2}/(2T)$, matching known optimal rates for bilinear saddle‑point problems.
\item \textbf{Strongly convex–strongly concave.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, a linear convergence rate $O\bigl((1-\eta\mu)^{T}\bigr)$ can be derived by strengthening inequality (5) to $\langle F(z_{t}),z_{t}-z^{\star}\rangle\ge\mu\|z_{t}-z^{\star}\|^{2}$.
\end{itemize}

\section*{Discussion}
The presented analysis shows that a single constant stepsize satisfying $\eta\le 1/L$ guarantees a \emph{deterministic} $O(1/T)$ decay of the primal–dual gap for smooth convex–concave games. The proof hinges only on monotonicity (convex–concave) and Lipschitz continuity of the gradient; no stochasticity or variance terms are involved. Consequently, the result is directly comparable to the classic $O(1/\sqrt{T})$ rate of stochastic gradient descent (Example 1) but is strictly faster because the underlying problem is a saddle point with a monotone operator. Extensions to stochastic GDA follow by adding unbiased gradient estimators and variance bounds, leading to the same $O(1/\sqrt{T})$ rate as SGD, as shown in the literature.
