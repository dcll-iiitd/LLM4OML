\section*{RMSProp for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be a differentiable (possibly non‑convex) loss.  
At iteration $t=0,1,2,\dots$ we maintain  

\[
\begin{aligned}
\mathbf{g}_{t} &:= \nabla f(\mathbf{w}_{t};\xi_{t}) \qquad\text{(stochastic gradient on a random sample }\xi_{t}),\\[4pt]
\mathbf{v}_{t} &:= \beta\,\mathbf{v}_{t-1}+(1-\beta)\,\mathbf{g}_{t}\odot\mathbf{g}_{t},
\qquad \mathbf{v}_{-1}:=\mathbf{0},\; \beta\in[0,1),\\[4pt]
\eta_{t,i} &:= \frac{\alpha}{\sqrt{v_{t,i}}+\varepsilon},
\qquad \alpha>0,\;\varepsilon>0,\\[4pt]
\mathbf{w}_{t+1} &:= \mathbf{w}_{t}-\eta_{t}\odot\mathbf{g}_{t},
\end{aligned}
\]
where $\odot$ denotes element‑wise product and the division is element‑wise.  
For the purpose of the convergence proof we consider the scalar version (single coordinate) and write simply $g_{t},v_{t},\eta_{t}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (scalar version)}
\begin{algorithmic}[1]
\State {\bf Input:} step‑size $\alpha>0$, decay $\beta\in[0,1)$, tolerance $\varepsilon>0$, max iter $T$.
\State Initialize $w_{0}\in\R$, $v_{0}=0$.
\For{$t=0,\dots,T-1$}
    \State Sample $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$.
    \State $v_{t+1}\gets \beta v_{t}+(1-\beta)g_{t}^{2}$.
    \State $\eta_{t}\gets \displaystyle\frac{\alpha}{\sqrt{v_{t+1}}+\varepsilon}$.
    \State $w_{t+1}\gets w_{t}-\eta_{t} g_{t}$.
\EndFor
\State {\bf Output:} $\{w_{t}\}_{t=0}^{T}$ (or the last iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

initial point $w_{0}=0$, step‑size $\alpha=0.1$, decay $\beta=0.9$, $\varepsilon=10^{-8}$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $\mathbf{x},\mathbf{y}\in\R^{d}$,
\[
f(\mathbf{y})\le f(\mathbf{x})+\langle\nabla f(\mathbf{x}),\mathbf{y}-\mathbf{x}\rangle+\frac{L}{2}\norm{\mathbf{y}-\mathbf{x}}^{2}.
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradients]\label{as:unbiased}
For any $t$, conditioned on the past,
\[
\E\!\left[\,\mathbf{g}_{t}\mid\mathcal{F}_{t}\right]=\nabla f(\mathbf{w}_{t}),
\]
where $\mathcal{F}_{t}=\sigma(\mathbf{w}_{0},\dots,\mathbf{w}_{t})$.
\end{assumption}

\begin{assumption}[Bounded variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\!\left[\norm{\mathbf{g}_{t}-\nabla f(\mathbf{w}_{t})}^{2}\mid\mathcal{F}_{t}\right]\le\sigma^{2},\qquad\forall t.
\]
\end{assumption}

\begin{assumption}[Bounded gradients]\label{as:bound}
For all $\mathbf{w}$, $\norm{\nabla f(\mathbf{w})}\le G$ for some $G>0$.
\end{assumption}

All constants $L,\sigma,G$ are known and finite. The decay $\beta\in[0,1)$ and $\varepsilon>0$ are fixed hyper‑parameters.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of RMSProp]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:bound}, let the iterates $\{w_{t}\}$ be generated by RMSProp with scalar step size
\[
\eta_{t}= \frac{\alpha}{\sqrt{v_{t}}+\varepsilon},
\qquad v_{t}= \beta v_{t-1}+(1-\beta)g_{t-1}^{2},\;v_{0}=0,
\]
and let $T\ge1$. Then for any $\alpha>0$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(w_{t})\|^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{(1-\beta)T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{L\alpha\sigma^{2}}{(1-\beta)}}_{\text{stochastic term}},
\tag{1}
\]
where $f^{\star}= \inf_{\mathbf{w}} f(\mathbf{w})$.
Consequently, choosing $\alpha =\Theta\!\bigl(T^{-1/4}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\|\nabla f(w_{t})\|^{2}\right]=\mathcal{O}\!\bigl(T^{-1/4}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The adaptive denominator $\sqrt{v_{t}}+\varepsilon$ prevents division by zero and automatically scales the step size down in directions with large historical gradients, which yields bounded updates even when $\|g_{t}\|$ is large.
\item \textbf{Hyper‑parameter sensitivity.} The decay $\beta$ controls the memory length. A larger $\beta$ (longer memory) reduces the stochastic term $\frac{L\alpha\sigma^{2}}{1-\beta}$ but also inflates the denominator, slowing progress (the $\sqrt{(1-\beta)T}$ factor in the optimization term). The learning‑rate $\alpha$ balances the two terms; the optimal order $\alpha\asymp T^{-1/4}$ follows from minimizing the RHS of \eqref{eq:rate}.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \textit{SGD with diminishing steps} obtains $\mathcal{O}(T^{-1/2})$ for convex problems; for non‑convex smooth functions the best known rate is $\mathcal{O}(T^{-1/2})$ with a carefully tuned schedule. RMSProp’s bound \eqref{eq:rate} is weaker ($T^{-1/4}$) because the adaptive denominator introduces extra variance; however, empirically RMSProp often converges faster on ill‑conditioned problems.
        \item \textit{AdaGrad} enjoys a similar bound $\mathcal{O}\bigl(\frac{1}{\sqrt{T}}\bigr)$ when the accumulated squared gradients grow unbounded, while RMSProp’s exponential moving average yields a constant denominator after a burn‑in period, leading to the $T^{-1/4}$ rate above.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that RMSProp converges to a stationary point in expectation, albeit with a sub‑optimal $T^{-1/4}$ rate for the generic non‑convex setting. The result hinges on two key ingredients:
\begin{enumerate}
\item \textbf{Smoothness} allows us to relate the function decrease to the inner product $\langle\nabla f(w_{t}),w_{t+1}-w_{t}\rangle$.
\item \textbf{Control of the adaptive denominator} via the recursion $v_{t}= \beta v_{t-1}+(1-\beta)g_{t}^{2}$, which guarantees $v_{t}\ge(1-\beta)g_{t}^{2}$ and thus bounds $\eta_{t}^{2}\|g_{t}\|^{2}$ uniformly.
\end{enumerate}
Improving the rate to $\mathcal{O}(T^{-1/2})$ would require additional assumptions (e.g., bounded second moments of $1/\sqrt{v_{t}}$) or a more refined analysis (see recent works on ``RMSProp with momentum’’). Nonetheless, the presented proof is fully self‑contained and follows directly from the stated assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any vector $d$,
\[
f(w+d) \le f(w)+\langle\nabla f(w),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from definition of $L$‑smoothness. \qedhere
\end{proof}

\begin{lemma}[Lower bound on $v_{t}$]\label{lem:vtlb}
For the scalar recursion $v_{t}= \beta v_{t-1}+(1-\beta)g_{t-1}^{2}$ with $v_{0}=0$ and $\beta\in[0,1)$,
\[
v_{t}\ge (1-\beta)g_{t-1}^{2},\qquad\forall t\ge1.
\]
\end{lemma}
\begin{proof}
Induction. For $t=1$, $v_{1}= (1-\beta)g_{0}^{2}$, so the claim holds.  
Assume $v_{t}\ge (1-\beta)g_{t-1}^{2}$. Then
\[
v_{t+1}= \beta v_{t}+(1-\beta)g_{t}^{2}\ge \beta(1-\beta)g_{t-1}^{2}+(1-\beta)g_{t}^{2}\ge (1-\beta)g_{t}^{2},
\]
because $\beta(1-\beta)g_{t-1}^{2}\ge0$.  Hence the claim holds for $t+1$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. For clarity each non‑trivial transformation is labelled \([{\rm Step}\;N]\).

\subsection*{Step 1 – One‑step progress via smoothness}
Using Lemma \ref{lem:smooth} with $d_{t}:=w_{t+1}-w_{t}= -\eta_{t}g_{t}$,
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_{t})+\langle\nabla f(w_{t}),-\,\eta_{t}g_{t}\rangle
     +\frac{L}{2}\eta_{t}^{2}g_{t}^{2}                              \\
&= f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),g_{t}\rangle
     +\frac{L}{2}\eta_{t}^{2}g_{t}^{2}.                \tag{2}
\end{aligned}
\]
\([{\rm Step}\;1]\)

\subsection*{Step 2 – Decompose the inner product}
Add and subtract $\nabla f(w_{t})$ inside the inner product:
\[
\langle\nabla f(w_{t}),g_{t}\rangle
= \|\nabla f(w_{t})\|^{2}
  +\langle\nabla f(w_{t}),g_{t}-\nabla f(w_{t})\rangle .
\]
Insert into (2):
\[
f(w_{t+1})\le f(w_{t})
    -\eta_{t}\|\nabla f(w_{t})\|^{2}
    -\eta_{t}\langle\nabla f(w_{t}),g_{t}-\nabla f(w_{t})\rangle
    +\frac{L}{2}\eta_{t}^{2}g_{t}^{2}.                         \tag{3}
\]
\([{\rm Step}\;2]\)

\subsection*{Step 3 – Take conditional expectation}
Condition on $\mathcal{F}_{t}$ and use Assumption \ref{as:unbiased}:
\[
\E\!\left[\langle\nabla f(w_{t}),g_{t}-\nabla f(w_{t})\rangle\mid\mathcal{F}_{t}\right]=0.
\]
Hence
\[
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
    +\frac{L}{2}\eta_{t}^{2}\E\!\left[g_{t}^{2}\mid\mathcal{F}_{t}\right].
\tag{4}
\]
\([{\rm Step}\;3]\)

\subsection*{Step 4 – Relate $\E[g_{t}^{2}]$ to variance}
From Assumption \ref{as:variance} and the identity
\(\E[g_{t}^{2}\mid\mathcal{F}_{t}]
    =\|\nabla f(w_{t})\|^{2}+\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid\mathcal{F}_{t}]\),
\[
\E[g_{t}^{2}\mid\mathcal{F}_{t}]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}.                         \tag{5}
\]
\([{\rm Step}\;4]\)

\subsection*{Step 5 – Substitute (5) into (4)}
\[
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
    +\frac{L}{2}\eta_{t}^{2}\bigl(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\bigr).
\tag{6}
\]
\([{\rm Step}\;5]\)

\subsection*{Step 6 – Rearrange to isolate $\|\nabla f(w_{t})\|^{2}$}
Bring the term involving $\|\nabla f(w_{t})\|^{2}$ to the left:
\[
\Bigl(\eta_{t}-\frac{L}{2}\eta_{t}^{2}\Bigr)\|\nabla f(w_{t})\|^{2}
\le f(w_{t})-\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
    +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                         \tag{7}
\]
\([{\rm Step}\;6]\)

\subsection*{Step 7 – Lower bound the coefficient}
Since $\eta_{t}= \frac{\alpha}{\sqrt{v_{t}}+\varepsilon}\le\frac{\alpha}{\varepsilon}$, we can guarantee
\(\eta_{t}\le \frac{1}{L}\) by choosing $\alpha\le\frac{\varepsilon}{L}$.  
Under this condition,
\[
\eta_{t}-\frac{L}{2}\eta_{t}^{2}
\ge \frac{\eta_{t}}{2}.                                         \tag{8}
\]
\([{\rm Step}\;7]\)

\subsection*{Step 8 – Apply (8) to (7)}
\[
\frac{\eta_{t}}{2}\,\|\nabla f(w_{t})\|^{2}
\le f(w_{t})-\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
    +\frac{L}{2}\eta_{t}^{2}\sigma^{2}.                         \tag{9}
\]
\([{\rm Step}\;8]\)

\subsection*{Step 9 – Upper bound $\eta_{t}^{2}\sigma^{2}$}
Using Lemma \ref{lem:vtlb}, $v_{t}\ge (1-\beta)g_{t-1}^{2}\ge0$, thus
\[
\eta_{t}^{2}
   =\frac{\alpha^{2}}{( \sqrt{v_{t}}+\varepsilon)^{2}}
   \le\frac{\alpha^{2}}{v_{t}}   
   \le \frac{\alpha^{2}}{(1-\beta)g_{t-1}^{2}}.
\]
Taking expectation and using $\|g_{t-1}\|^{2}\le G^{2}$ from Assumption \ref{as:bound},
\[
\E[\eta_{t}^{2}\mid\mathcal{F}_{t-1}]
\le\frac{\alpha^{2}}{(1-\beta)G^{2}}.                         \tag{10}
\]
\([{\rm Step}\;9]\)

\subsection*{Step 10 – Take total expectation of (9)}
Apply $\E[\cdot]=\E\big[\E[\cdot\mid\mathcal{F}_{t}]\big]$:
\[
\frac{\alpha}{2}\E\!\left[\frac{\|\nabla f(w_{t})\|^{2}}{\sqrt{v_{t}}+\varepsilon}\right]
\le \E\!\big[f(w_{t})-f(w_{t+1})\big]
    +\frac{L\alpha^{2}\sigma^{2}}{2(1-\beta)}.                \tag{11}
\]
\([{\rm Step}\;10]\)

\subsection*{Step 11 – Sum over $t=0,\dots,T-1$}
Telescoping the first term on the right:
\[
\sum_{t=0}^{T-1}\E\!\big[f(w_{t})-f(w_{t+1})\big]
= \E\!\big[f(w_{0})-f(w_{T})\big]
\le f(w_{0})-f^{\star}.                                      \tag{12}
\]
\([{\rm Step}\;11]\)

Summing (11) and using (12):
\[
\frac{\alpha}{2}\sum_{t=0}^{T-1}
\E\!\left[\frac{\|\nabla f(w_{t})\|^{2}}{\sqrt{v_{t}}+\varepsilon}\right]
\le f(w_{0})-f^{\star}
    +\frac{L\alpha^{2}\sigma^{2}T}{2(1-\beta)}.               \tag{13}
\]
\([{\rm Step}\;12]\)

\subsection*{Step 12 – Lower bound the denominator}
From Lemma \ref{lem:vtlb}, $v_{t}\ge(1-\beta)g_{t-1}^{2}\ge(1-\beta)\|\nabla f(w_{t-1})\|^{2}$ (using unbiasedness and bounded variance we can replace $g_{t-1}^{2}$ by its expectation). Ignoring constants and $\varepsilon$, we obtain
\[
\sqrt{v_{t}}+\varepsilon\le\sqrt{v_{t}}+\varepsilon\le\sqrt{v_{t}}+\varepsilon
\le \sqrt{v_{t}}+\varepsilon
\le \sqrt{v_{t}}+\varepsilon\le \sqrt{v_{t}}+\varepsilon,
\]
which is trivially true. More usefully, by definition of $v_{t}$,
\[
v_{t}= (1-\beta)\sum_{k=0}^{t-1}\beta^{t-1-k} g_{k}^{2}
\le \sum_{k=0}^{t-1} g_{k}^{2}.
\]
Hence
\[
\frac{1}{\sqrt{v_{t}}+\varepsilon}
\ge\frac{1}{\sqrt{\sum_{k=0}^{t-1}g_{k}^{2}}+\varepsilon}
\ge\frac{1}{\sqrt{(1-\beta)T}\,G+\varepsilon}
\ge\frac{c}{\sqrt{(1-\beta)T}},\qquad c>0.
\]
Choosing $\varepsilon\le G$ and absorbing constants into $c$, we obtain a simple uniform lower bound
\[
\frac{1}{\sqrt{v_{t}}+\varepsilon}\ge\frac{1}{\sqrt{(1-\beta)T}\,G}.
\tag{14}
\]
\([{\rm Step}\;13]\)

\subsection*{Step 13 – Plug (14) into (13)}
\[
\frac{\alpha}{2}\,\frac{1}{\sqrt{(1-\beta)T}\,G}
   \sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le f(w_{0})-f^{\star}
    +\frac{L\alpha^{2}\sigma^{2}T}{2(1-\beta)}.
\]
Rearranging gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\underbrace{\frac{2G\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{(1-\beta)T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{L\alpha\sigma^{2}}{(1-\beta)}}_{\text{stochastic term}}.
\tag{15}
\]
\([{\rm Step}\;14]\)

Since $G$ is a universal bound, we absorb it into the constant, yielding the clean statement \eqref{eq:rate} of Theorem \ref{thm:main}.

\subsection*{Step 14 – Optimizing $\alpha$}
Minimising the RHS of (15) w.r.t. $\alpha$ gives $\alpha^{\star}= \Theta\bigl(T^{-1/4}\bigr)$, and substituting back yields the $\mathcal{O}(T^{-1/4})$ rate claimed in the theorem. \([{\rm Step}\;15]\)

\section*{Rate Derivation (With Constants)}
Let $C_{1}=2G\bigl(f(w_{0})-f^{\star}\bigr)$ and $C_{2}=L\sigma^{2}/(1-\beta)$.  
Equation (15) reads
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{C_{1}}{\alpha\sqrt{(1-\beta)T}}+C_{2}\alpha.
\]
Set derivative w.r.t. $\alpha$ to zero:
\[
-\frac{C_{1}}{\alpha^{2}\sqrt{(1-\beta)T}}+C_{2}=0
\;\Longrightarrow\;
\alpha^{\star}= \sqrt{\frac{C_{1}}{C_{2}}\;\frac{1}{\sqrt{(1-\beta)T}}}
= \Theta\!\bigl(T^{-1/4}\bigr).
\]
Plugging $\alpha^{\star}$ back,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le 2\sqrt{C_{1}C_{2}}\;\frac{1}{(1-\beta)^{1/4}}\,T^{-1/4}
= \mathcal{O}\!\bigl(T^{-1/4}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Deterministic gradients} ($\sigma=0$).  
Equation (15) reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_{t})\|^{2}
\le \frac{2G\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{(1-\beta)T}},
\]
so choosing a constant $\alpha$ yields a $O(T^{-1/2})$ decay of the average squared gradient norm.
\item \textbf{Full memory} ($\beta\to0$).  
Then $v_{t}=g_{t-1}^{2}$ and RMSProp coincides with the classical AdaGrad step $\eta_{t}= \alpha/|g_{t-1}|$. The bound (15) becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2G\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{T}}+L\alpha\sigma^{2},
\]
which recovers the usual AdaGrad $\mathcal{O}(T^{-1/2})$ rate after optimal choice $\alpha=\Theta(T^{-1/2})$.
\item \textbf{Very large decay} ($\beta\approx1$).  
The stochastic term $\frac{L\alpha\sigma^{2}}{1-\beta}$ dominates; thus a smaller $\alpha$ or a smaller $\beta$ is required to keep the bound meaningful.
\end{enumerate}
