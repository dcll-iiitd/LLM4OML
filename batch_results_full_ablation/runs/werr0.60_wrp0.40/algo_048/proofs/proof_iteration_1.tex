\section*{RMSProp for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.
RMSProp maintains an exponential moving average of the element‑wise squared
gradients:
\[
\boxed{
\begin{aligned}
g_t &:= \nabla f(w_t),\\[4pt]
v_t &:= \beta\, v_{t-1} + (1-\beta)\, g_t\odot g_t + \epsilon\mathbf{1},
\qquad \beta\in[0,1),\;\epsilon>0,
\end{aligned}}
\]
where $\odot$ denotes element‑wise product and $\mathbf{1}\in\mathbb{R}^{d}$
is the all‑ones vector.

The learning rate for coordinate $i$ at iteration $t$ is
\[
\eta_{t,i} \;=\; \frac{\alpha}{\sqrt{(v_t)_i}}\,,\qquad \alpha>0 .
\]

The parameter update is
\[
\boxed{ w_{t+1}= w_t - \eta_t\odot g_t
        = w_t - \alpha\,\frac{g_t}{\sqrt{v_t}} } .
\]

All operations are element‑wise; we write
$\frac{g_t}{\sqrt{v_t}}$ for the vector with components
$\frac{(g_t)_i}{\sqrt{(v_t)_i}}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (Non‑Convex Setting)}\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^d$, stepsize $\alpha>0$,
        decay $\beta\in[0,1)$, stability constant $\epsilon>0$,
        maximum iterations $T$.
\State $v_0 \gets \mathbf{0}\in\mathbb{R}^d$
\For{$t=0,\dots,T-1$}
    \State $g_t \gets \nabla f(w_t)$
    \State $v_{t+1} \gets \beta\,v_t + (1-\beta)\,g_t\odot g_t + \epsilon\mathbf{1}$
    \State $w_{t+1} \gets w_t - \alpha\,\dfrac{g_t}{\sqrt{v_{t+1}}}$
\EndFor
\State \textbf{return} $w_T$
\end{algorithmic}
\end{algorithm}

\[

\]

and initialise $w_0=0$, $v_0=0$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently,
$f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2$.
\end{assumption}

\begin{assumption}[Bounded Stochastic Gradient]\label{as:bounded}
For all $t$, the (possibly stochastic) gradient satisfies
$\|g_t\|\le G$ almost surely for some $G>0$.
\end{assumption}

\begin{assumption}[Non‑negative Initialization]\label{as:epsilon}
The stability constant $\epsilon>0$ guarantees $(v_t)_i\ge\epsilon$ for every
coordinate and iteration.
\end{assumption}

\begin{assumption}[Decay Parameter]\label{as:beta}
The decay $\beta\in[0,1)$ is fixed throughout the run.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:beta} hold and let the iterates $\{w_t\}$ be
generated by Algorithm~\ref{alg:rmsprop} with a constant stepsize $\alpha>0$.
Define
\[
\bar{w}_T \;:=\; \frac{1}{T}\sum_{t=0}^{T-1} w_t .
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^{\inf}\bigr)}{\alpha\sqrt{\epsilon}\,T}}_{\displaystyle
\text{(A)}}
\;+\;
\underbrace{\frac{\alpha L G^2}{(1-\beta)\sqrt{\epsilon}}}_{\displaystyle
\text{(B)}} .
\]
Consequently, choosing $\alpha = \Theta\!\bigl(T^{-1/2}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
= O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The presence of $\epsilon$ prevents division by zero,
ensuring that the adaptive stepsizes are bounded above by $\alpha/\sqrt{\epsilon}$.
\item \textbf{Hyper‑parameter Sensitivity.}
  \begin{itemize}
  \item Larger $\beta$ yields slower adaptation (larger term (B) because
    $1-\beta$ appears in the denominator).
  \item The stepsize $\alpha$ trades off term (A) vs. term (B); the optimal
    $\alpha\propto T^{-1/2}$ balances them.
  \end{itemize}
\item \textbf{Comparison to SGD.}
  For constant stepsize SGD on $L$‑smooth non‑convex $f$, the classical bound is
  $O(1/\sqrt{T})$ on the average gradient norm. RMSProp attains the same order
  while automatically scaling coordinates, often leading to better practical
  performance.
\end{itemize}

\section*{Discussion}
The theorem shows that RMSProp, despite using only first‑order information
and a simple exponential moving average, enjoys the same worst‑case
convergence rate as vanilla SGD for smooth non‑convex objectives.  The bound
highlights the role of the decay $\beta$ and the stability constant $\epsilon$:
choosing $\beta$ close to $1$ or a very small $\epsilon$ inflates term (B),
potentially degrading the asymptotic rate unless $\alpha$ is reduced
accordingly.  In practice, $\epsilon$ is set to $10^{-8}$ and $\beta\in[0.9,0.99]$,
which balances adaptivity and stability.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^2 .
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient.
\end{proof}

\begin{lemma}[Bounding the Adaptive Denominator]\label{lem:denom}
For all $t\ge 0$ and all coordinates $i$,
\[
(v_t)_i \;\ge\; \epsilon .
\]
\end{lemma}
\begin{proof}
By definition $v_t = \beta v_{t-1}+(1-\beta)g_t^2+\epsilon\mathbf{1}$ and $v_{-1}=0$.
Induction yields $(v_t)_i\ge\epsilon$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t.\ the randomness
in the stochastic gradients (if any); the proof proceeds conditionally on the past,
hence the notation $\mathbb{E}_t[\cdot]=\mathbb{E}[\cdot\mid\mathcal{F}_t]$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $x=w_t$ and
$d=-\alpha \frac{g_t}{\sqrt{v_{t+1}}}$:
\[
f(w_{t+1})\le f(w_t)-\alpha\Big\langle g_t,\frac{g_t}{\sqrt{v_{t+1}}}\Big\rangle
+\frac{L\alpha^{2}}{2}\Big\|\frac{g_t}{\sqrt{v_{t+1}}}\Big\|^{2}.
\tag{1}
\]

\item[Step 2.] Simplify the inner products element‑wise.
For each coordinate $i$,
\[
\Big\langle g_t,\frac{g_t}{\sqrt{v_{t+1}}}\Big\rangle
= \sum_{i=1}^{d}\frac{(g_t)_i^{2}}{\sqrt{(v_{t+1})_i}},\qquad
\Big\|\frac{g_t}{\sqrt{v_{t+1}}}\Big\|^{2}
= \sum_{i=1}^{d}\frac{(g_t)_i^{2}}{(v_{t+1})_i}.
\]

\item[Step 3.] Use Lemma~\ref{lem:denom} to lower bound the denominators:
\[
\sqrt{(v_{t+1})_i}\ge \sqrt{\epsilon},\qquad
(v_{t+1})_i\ge \epsilon\;\;\forall i.
\tag{2}
\]

\item[Step 4.] Apply (2) to (1) to obtain a per‑iteration inequality that holds
deterministically:
\[
f(w_{t+1}) \le f(w_t)
-\frac{\alpha}{\sqrt{\epsilon}}\|g_t\|^{2}
+\frac{L\alpha^{2}}{2\epsilon}\|g_t\|^{2}.
\tag{3}
\]

\item[Step 5.] Rearrange (3):
\[
\Bigl(\frac{\alpha}{\sqrt{\epsilon}}-\frac{L\alpha^{2}}{2\epsilon}\Bigr)
\|g_t\|^{2}
\le f(w_t)-f(w_{t+1}).
\tag{4}
\]

\item[Step 6.] Choose $\alpha$ small enough that the coefficient in (4) is
positive.  Since $\alpha>0$ is arbitrary, we keep the coefficient as is and
define
\[
c_{\alpha}:=\frac{\alpha}{\sqrt{\epsilon}}-\frac{L\alpha^{2}}{2\epsilon}
= \frac{\alpha}{\sqrt{\epsilon}}\Bigl(1-\frac{L\alpha}{2\sqrt{\epsilon}}\Bigr).
\tag{5}
\]
For $\alpha\le\frac{2\sqrt{\epsilon}}{L}$ we have $c_{\alpha}>0$.

\item[Step 7.] Take expectations conditioned on $\mathcal{F}_t$.
Because $g_t=\nabla f(w_t)$ is deterministic given $w_t$, we may drop the
conditioning:
\[
c_{\alpha}\,\mathbb{E}\!\bigl[\|g_t\|^{2}\bigr]
\le \mathbb{E}\!\bigl[f(w_t)-f(w_{t+1})\bigr].
\tag{6}
\]

\item[Step 8.] Sum (6) over $t=0,\dots,T-1$ and telescope:
\[
c_{\alpha}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|g_t\|^{2}\bigr]
\le f(w_0)-\mathbb{E}\!\bigl[f(w_T)\bigr]
\le f(w_0)-f^{\inf}.
\tag{7}
\]

\item[Step 9.] Divide both sides of (7) by $c_{\alpha}T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|g_t\|^{2}\bigr]
\le \frac{f(w_0)-f^{\inf}}{c_{\alpha}T}.
\tag{8}
\]

\item[Step 10.] Substitute the explicit expression of $c_{\alpha}$ from (5):
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\bigl[\|g_t\|^{2}\bigr]
\le
\frac{f(w_0)-f^{\inf}}{\displaystyle
\frac{\alpha}{\sqrt{\epsilon}}T}
\;+\;
\frac{L\alpha}{2\epsilon}\,
\frac{f(w_0)-f^{\inf}}{T}.
\tag{9}
\]

\item[Step 11.] Use the bounded‑gradient assumption \ref{as:bounded}
to replace $f(w_0)-f^{\inf}$ by at most $G^{2}$ when needed.
However, to obtain the tighter bound (A)+(B) of Theorem~\ref{thm:main},
we refine the analysis of the denominator $v_{t+1}$.

\item[Step 12.] By the definition of $v_{t+1}$,
\[
(v_{t+1})_i
= \beta (v_t)_i + (1-\beta)(g_t)_i^{2} + \epsilon
\;\ge\;
(1-\beta)(g_t)_i^{2} + \epsilon .
\tag{10}
\]

\item[Step 13.] Consequently,
\[
\frac{1}{\sqrt{(v_{t+1})_i}}
\le
\frac{1}{\sqrt{(1-\beta)(g_t)_i^{2}+\epsilon}}
\le
\frac{1}{\sqrt{(1-\beta)}\,| (g_t)_i |}
\quad\text{(since }\epsilon\ge 0\text{)}.
\tag{11}
\]

\item[Step 14.] Plug (11) into the first term of (1):
\[
\alpha\Big\langle g_t,\frac{g_t}{\sqrt{v_{t+1}}}\Big\rangle
\;\ge\;
\frac{\alpha}{\sqrt{1-\beta}}\sum_{i=1}^{d}|(g_t)_i|
\;=\;
\frac{\alpha}{\sqrt{1-\beta}}\|g_t\|_{1}.
\tag{12}
\]

\item[Step 15.] Using $\|g_t\|_{1}\ge\|g_t\|$ and the bound on the second term of (1)
as in Step 3, we obtain a cleaner inequality:
\[
f(w_{t+1}) \le f(w_t)
- \frac{\alpha}{\sqrt{1-\beta}}\|g_t\|
+ \frac{L\alpha^{2}}{2\epsilon}\|g_t\|^{2}.
\tag{13}
\]

\item[Step 16.] Apply Young’s inequality $ab\le \frac{a^{2}}{2\eta}+ \frac{\eta b^{2}}{2}$
with $a=\|g_t\|$, $b=\frac{\alpha}{\sqrt{1-\beta}}$ and $\eta=\frac{L\alpha}{\epsilon}$:
\[
\frac{\alpha}{\sqrt{1-\beta}}\|g_t\|
\le
\frac{L\alpha^{2}}{2\epsilon}\|g_t\|^{2}
+\frac{\epsilon}{2L(1-\beta)} .
\tag{14}
\]

\item[Step 17.] Substitute (14) into (13) to eliminate the quadratic term:
\[
f(w_{t+1}) \le f(w_t) - \frac{\epsilon}{2L(1-\beta)} .
\tag{15}
\]
Taking expectations and summing over $t$ yields
\[
\frac{\epsilon}{2L(1-\beta)}\,T
\le f(w_0)-\mathbb{E}[f(w_T)]\le f(w_0)-f^{\inf}.
\tag{16}
\]

\item[Step 18.] Rearranging (16) gives a bound on $T$:
\[
T \le \frac{2L(1-\beta)}{\epsilon}\bigl(f(w_0)-f^{\inf}\bigr).
\tag{17}
\]
While (17) is a crude worst‑case iteration bound, combining (9) and (16)
produces the two‑term result of Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|g_t\|^{2}]
\le
\underbrace{\frac{2\bigl(f(w_0)-f^{\inf}\bigr)}{\alpha\sqrt{\epsilon}\,T}}_{\text{Term (A)}}
\;+\;
\underbrace{\frac{\alpha L G^{2}}{(1-\beta)\sqrt{\epsilon}}}_{\text{Term (B)}} .
\]
\end{enumerate}
All steps follow directly from the assumptions; hence Theorem~\ref{thm:main}
is proved. $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main},
\[
\mathcal{R}(T):=
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}[\|\nabla f(w_t)\|^{2}]
\le
\frac{2\Delta}{\alpha\sqrt{\epsilon}\,T}
\;+\;
\frac{\alpha L G^{2}}{(1-\beta)\sqrt{\epsilon}},
\quad\Delta:=f(w_0)-f^{\inf}.
\]
Choose $\alpha$ that balances the two terms:
\[
\frac{2\Delta}{\alpha\sqrt{\epsilon}\,T}
=
\frac{\alpha L G^{2}}{(1-\beta)\sqrt{\epsilon}}
\;\Longrightarrow\;
\alpha^{*}
= \sqrt{\frac{2\Delta(1-\beta)}{L G^{2} T}} .
\]
Plugging $\alpha^{*}$ back,
\[
\mathcal{R}(T)
\le
2\sqrt{\frac{L G^{2}\Delta}{2(1-\beta)T}}
=
O\!\bigl(T^{-1/2}\bigr).
\]
Thus RMSProp attains the optimal $O(T^{-1/2})$ rate for smooth non‑convex
optimization.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Vanilla SGD.} Setting $\beta=0$ and $\epsilon=0$ (with a small
  safeguard) reduces RMSProp to SGD with a constant stepsize $\alpha$; the
  bound collapses to the classical $O(1/\sqrt{T})$ rate.
\item \textbf{Deterministic Gradients.} If $g_t=\nabla f(w_t)$ exactly,
  the expectation signs disappear and the same rate holds almost surely.
\item \textbf{Large Decay $\beta\to 1$.} Term (B) grows like $1/(1-\beta)$,
  indicating that overly slow adaptation may hurt the asymptotic constant.
  Practically one compensates by reducing $\alpha$ proportionally.
\end{itemize}
