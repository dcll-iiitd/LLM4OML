\section*{Algorithm Name}
\textbf{SAGA for Non‑Convex Smooth Finite‑Sum Optimization}  

We consider the finite‑sum problem  
\[
\min_{x\in\R^{d}} \; f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is possibly non‑convex but $L$‑smooth.

\section*{Mathematical Formulation}
Let $\{\phi_i^{k}\}_{i=1}^{n}\subset\R^{d}$ denote the stored “snapshot’’ points at iteration $k$
and define the table of past gradients
\[
g_i^{k}:=\nabla f_i(\phi_i^{k}),\qquad 
\bar g^{k}:=\frac{1}{n}\sum_{i=1}^{n} g_i^{k}.
\]
At iteration $k$ we pick an index $i_k\in\{1,\dots,n\}$ uniformly at random,
compute the stochastic gradient correction
\[
\widetilde{\nabla}_k \;:=\; \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar g^{k},
\]
and update
\[
x_{k+1}=x_{k}-\alpha\,\widetilde{\nabla}_k . \tag{1}
\]
After the update we replace the stored point for the sampled index:
\[
\phi_{i_k}^{k+1}=x_k,\qquad 
\phi_{j}^{k+1}=\phi_{j}^{k}\;\; \forall\,j\neq i_k . \tag{2}
\]
The stepsize $\alpha>0$ is a constant (or may follow a prescribed schedule).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex)}\label{alg:saga}
\begin{algorithmic}[1]
\Require Number of components $n$, stepsize $\alpha>0$, maximum iterations $K$
\State Initialize $x_0\in\R^{d}$ arbitrarily.
\For{$i=1,\dots,n$}
    \State $\phi_i^{0}\gets x_0$   \Comment{initial snapshot}
    \State $g_i^{0}\gets\nabla f_i(\phi_i^{0})$
\EndFor
\State $\bar g^{0}\gets \frac{1}{n}\sum_{i=1}^{n} g_i^{0}$
\For{$k=0$ \textbf{to} $K-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State $\widetilde{\nabla}_k\gets \nabla f_{i_k}(x_k)-g_{i_k}^{k}+\bar g^{k}$
    \State $x_{k+1}\gets x_k-\alpha\,\widetilde{\nabla}_k$ \Comment{Eq.~(1)}
    \State $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x_k)$
    \State $\bar g^{k+1}\gets \bar g^{k}+ \frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$
    \State $\phi_{i_k}^{k+1}\gets x_k$ \Comment{Eq.~(2)}
    \For{$j\neq i_k$}
        \State $g_{j}^{k+1}\gets g_{j}^{k},\;\phi_{j}^{k+1}\gets\phi_{j}^{k}$
    \EndFor
\EndFor
\State \Return $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑component problem ($n=1$) with  
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, stepsize $\alpha=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter} & $w_k$ & $\widetilde{\nabla}_k$ (here equal to $\nabla f(w_k)$) & $f(w_k)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3660$
\end{tabular}
\end{center}
Because $n=1$, the table never changes and the algorithm reduces to plain gradient descent.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e., for all $x,y\in\R^{d}$
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|.
\]
\end{assumption}
\begin{assumption}[Bounded Variance of the Gradient Table]\label{ass:bounded}
The stored gradients are uniformly bounded:
\[
\max_{i,k}\|\nabla f_i(\phi_i^{k})\|\le G.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The number of components $n$ is finite and known.
\end{assumption}
All expectations are taken with respect to the random index $i_k$ conditioned on the past.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:finite} and choose a constant stepsize 
\[
0<\alpha\le\frac{1}{3L}.
\]
Let $\{x_k\}$ be the iterates generated by SAGA. Then for any $K\ge1$
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K}
\;+\;
\frac{12L\alpha G^{2}}{n},
\tag{3}
\]
where $f^{\star}=\inf_{x}f(x)$. In particular, with the stepsize 
$\alpha=\Theta\!\bigl(\frac{1}{\sqrt{K}}\bigr)$ we obtain the non‑convex rate
\[
\min_{0\le k\le K-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]=O\!\Bigl(\frac{1}{\sqrt{K}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound (3) shows that the variance term 
$\frac{12L\alpha G^{2}}{n}$ vanishes as $n\to\infty$, reflecting the variance‑reduction nature of SAGA.
\item \textbf{Hyperparameter Sensitivity.} The stepsize must satisfy $\alpha\le 1/(3L)$; a larger $\alpha$ breaks the descent inequality in Lemma~\ref{lem:descent}.
\item \textbf{Comparison.} Compared with vanilla SGD, which yields $O(1/\sqrt{K})$ for the average gradient norm without the $1/n$ variance reduction, SAGA improves the constant by a factor $1/n$ and matches the optimal $O(1/K)$ rate for convex problems.
\end{enumerate}

\section*{Discussion}
The theorem is proved by constructing a Lyapunov function that couples the distance to the optimum with the error of the gradient table. The proof below follows the original SAGA analysis but is adapted to the non‑convex setting; all steps are fully expanded and numbered for easy reference.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Lyapunov function]
For iteration $k$ define
\[
\Phi_k\;:=\; f(x_k)-f^{\star}
\;+\;
\frac{c}{n}\sum_{i=1}^{n}\|\,\phi_i^{k}-x^{\star}\|^{2},
\tag{4}
\]
where $c>0$ will be chosen later and $x^{\star}$ is any point attaining $f^{\star}$ (if it exists; otherwise any minimizer of $f$).
\end{definition}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth $f_i$ and any $u,v\in\R^{d}$,
\[
f_i(u)\le f_i(v)+\langle\nabla f_i(v),u-v\rangle+\frac{L}{2}\|u-v\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{lemma}[Unbiasedness of the correction]\label{lem:unbiased}
Conditioned on the filtration $\mathcal{F}_k$ generated by $\{x_0,\dots,x_k\}$ and the table,
\[
\E\!\bigl[\widetilde{\nabla}_k\mid\mathcal{F}_k\bigr]
= \nabla f(x_k).
\tag{6}
\]
\end{lemma}
\begin{proof}
By definition,
\[
\widetilde{\nabla}_k
= \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar g^{k},
\]
and $\E[\nabla f_{i_k}(x_k)\mid\mathcal{F}_k]=\frac1n\sum_i\nabla f_i(x_k)=\nabla f(x_k)$.
Similarly $\E[\nabla f_{i_k}(\phi_{i_k}^{k})\mid\mathcal{F}_k]=\bar g^{k}$.
Hence the two extra terms cancel, yielding (6). \qedhere
\end{proof}
\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $u$ with $\E[u]=\mu$,
\[
\E\|u\|^{2}= \|\mu\|^{2}+ \E\|u-\mu\|^{2}.
\tag{7}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the random index $i_k$ and we use the filtration $\mathcal{F}_k$.

\subsection*{Step 1. One‑step progress of $f$}
From the $L$‑smoothness of $f$ (Lemma~\ref{lem:smooth}) and the update (1):
\begin{align}
f(x_{k+1})
&\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle
   +\frac{L}{2}\|x_{k+1}-x_k\|^{2} \nonumber\\
&= f(x_k)-\alpha\langle\nabla f(x_k),\widetilde{\nabla}_k\rangle
   +\frac{L\alpha^{2}}{2}\|\widetilde{\nabla}_k\|^{2}. \tag{8}
\end{align}
[Step 1] is the direct application of (5) with $u=x_{k+1},v=x_k$.

\subsection*{Step 2. Take expectation conditioned on $\mathcal{F}_k$}
Using Lemma~\ref{lem:unbiased} and the variance decomposition (7):
\begin{align}
\E\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +\frac{L\alpha^{2}}{2}\Bigl(\|\nabla f(x_k)\|^{2}
   +\E\!\bigl[\|\widetilde{\nabla}_k-\nabla f(x_k)\|^{2}\mid\mathcal{F}_k\bigr]\Bigr). \tag{9}
\end{align}
[Step 2] follows from (8) after inserting (7) with $u=\widetilde{\nabla}_k$, $\mu=\nabla f(x_k)$.

\subsection*{Step 3. Bound the correction variance}
Recall $\widetilde{\nabla}_k = \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar g^{k}$.
Conditioned on $\mathcal{F}_k$, the only randomness is $i_k$. Hence
\begin{align}
\E\!\bigl[\|\widetilde{\nabla}_k-\nabla f(x_k)\|^{2}\mid\mathcal{F}_k\bigr]
&= \frac{1}{n}\sum_{i=1}^{n}
   \bigl\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})-\bigl(\nabla f(x_k)-\bar g^{k}\bigr)\bigr\|^{2} \nonumber\\
&\le \frac{2}{n}\sum_{i=1}^{n}
   \Bigl(\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|^{2}
   +\|\nabla f(x_k)-\bar g^{k}\|^{2}\Bigr) \nonumber\\
&= \frac{2}{n}\sum_{i=1}^{n}\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|^{2}
   +2\|\nabla f(x_k)-\bar g^{k}\|^{2}. \tag{10}
\end{align}
[Step 3] uses $(a+b)^{2}\le2a^{2}+2b^{2}$.

Now apply $L$‑smoothness to each difference:
\[
\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|
\le L\|x_k-\phi_i^{k}\|,
\]
hence
\[
\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|^{2}
\le L^{2}\|x_k-\phi_i^{k}\|^{2}. \tag{11}
\]

Also note that $\bar g^{k}= \frac{1}{n}\sum_{i}\nabla f_i(\phi_i^{k})$, so
\[
\|\nabla f(x_k)-\bar g^{k}\|^{2}
\le \frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\|^{2}
\le \frac{L^{2}}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^{2}. \tag{12}
\]

Plugging (11)–(12) into (10) gives
\[
\E\!\bigl[\|\widetilde{\nabla}_k-\nabla f(x_k)\|^{2}\mid\mathcal{F}_k\bigr]
\le \frac{4L^{2}}{n}\sum_{i=1}^{n}\|x_k-\phi_i^{k}\|^{2}. \tag{13}
\]

\subsection*{Step 4. Lyapunov descent}
Define the Lyapunov function (4) with $c:=4\alpha L$.
Consider the change of the second term:
\[
\| \phi_{i_k}^{k+1}-x^{\star}\|^{2}
= \|x_k-x^{\star}\|^{2},
\qquad
\| \phi_{j}^{k+1}-x^{\star}\|^{2}
= \| \phi_{j}^{k}-x^{\star}\|^{2},\; j\neq i_k .
\]
Thus,
\[
\frac{c}{n}\sum_{i=1}^{n}\| \phi_i^{k+1}-x^{\star}\|^{2}
= \frac{c}{n}\Bigl(\sum_{i=1}^{n}\| \phi_i^{k}-x^{\star}\|^{2}
   -\| \phi_{i_k}^{k}-x^{\star}\|^{2}
   +\|x_k-x^{\star}\|^{2}\Bigr). \tag{14}
\]

Take expectation of (14) over $i_k$:
\[
\E\!\bigl[ \tfrac{c}{n}\sum_{i}\|\phi_i^{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\bigr]
= \frac{c}{n}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}
   -\frac{c}{n^{2}}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}
   +\frac{c}{n}\|x_k-x^{\star}\|^{2}. \tag{15}
\]

Now combine (9) with (13) and add the Lyapunov term (15). Using (9)–(13):
\begin{align}
\E\!\bigl[\Phi_{k+1}\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +\frac{L\alpha^{2}}{2}\|\nabla f(x_k)\|^{2}
   +\frac{2L^{3}\alpha^{2}}{n}\sum_{i}\|x_k-\phi_i^{k}\|^{2} \nonumber\\
&\quad +\frac{c}{n}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}
   -\frac{c}{n^{2}}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}
   +\frac{c}{n}\|x_k-x^{\star}\|^{2}. \tag{16}
\end{align}
[Step 4] consists of substituting (13) into (9) and adding (15).

Observe that
\[
\|x_k-\phi_i^{k}\|^{2}
\le 2\|x_k-x^{\star}\|^{2}+2\|\phi_i^{k}-x^{\star}\|^{2} . \tag{17}
\]
Insert (17) into the third term of (16):
\begin{align}
\frac{2L^{3}\alpha^{2}}{n}\sum_{i}\|x_k-\phi_i^{k}\|^{2}
&\le \frac{4L^{3}\alpha^{2}}{n}\sum_{i}\|x_k-x^{\star}\|^{2}
   +\frac{4L^{3}\alpha^{2}}{n}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2} \nonumber\\
&= 4L^{3}\alpha^{2}\|x_k-x^{\star}\|^{2}
   +4L^{3}\alpha^{2}\frac{1}{n}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}. \tag{18}
\end{align}

Now collect the coefficients of $\|x_k-x^{\star}\|^{2}$ and of $\frac{1}{n}\sum_i\|\phi_i^{k}-x^{\star}\|^{2}$ in (16). Using $c=4\alpha L$,
\begin{align}
\E\!\bigl[\Phi_{k+1}\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_k)\|^{2}
   +\Bigl(\frac{c}{n}+4L^{3}\alpha^{2}\Bigr)\|x_k-x^{\star}\|^{2} \nonumber\\
&\quad +\Bigl(\frac{c}{n}-\frac{c}{n^{2}}+4L^{3}\alpha^{2}\Bigr)\frac{1}{n}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}. \tag{19}
\end{align}
Because $c=4\alpha L$, we have $\frac{c}{n}= \frac{4\alpha L}{n}$.
Choose $\alpha\le \frac{1}{3L}$, then $1-\frac{L\alpha}{2}\ge \frac{1}{2}$ and also
\[
\frac{c}{n}+4L^{3}\alpha^{2}
= \frac{4\alpha L}{n}+4L^{3}\alpha^{2}
\le \frac{4\alpha L}{n}+ \frac{4\alpha L}{3n}= \frac{8\alpha L}{3n}.
\]
Similarly,
\[
\frac{c}{n}-\frac{c}{n^{2}}+4L^{3}\alpha^{2}
\le \frac{4\alpha L}{n}.
\]
Hence (19) simplifies to
\[
\E\!\bigl[\Phi_{k+1}\mid\mathcal{F}_k\bigr]
\le \Phi_k -\frac{\alpha}{2}\|\nabla f(x_k)\|^{2}
   +\frac{8\alpha L}{3n}\|x_k-x^{\star}\|^{2}
   +\frac{4\alpha L}{n^{2}}\sum_{i}\|\phi_i^{k}-x^{\star}\|^{2}. \tag{20}
\]

Drop the non‑negative terms $\|x_k-x^{\star}\|^{2}$ and $\|\phi_i^{k}-x^{\star}\|^{2}$ to obtain a clean recursion:
\[
\E\!\bigl[\Phi_{k+1}\mid\mathcal{F}_k\bigr]
\le \Phi_k -\frac{\alpha}{2}\|\nabla f(x_k)\|^{2}. \tag{21}
\]

\subsection*{Step 5. Telescoping sum}
Take total expectation and sum (21) for $k=0,\dots,K-1$:
\[
\Phi_0 - \E[\Phi_K] \;\ge\; \frac{\alpha}{2}\sum_{k=0}^{K-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr].
\]
Since $\Phi_K\ge 0$ and $\Phi_0 = f(x_0)-f^{\star}+ \frac{c}{n}\sum_i\|\phi_i^{0}-x^{\star}\|^{2}$,
and $\|\phi_i^{0}-x^{\star}\|\le G$ by Assumption~\ref{ass:bounded},
\[
\Phi_0 \le f(x_0)-f^{\star}+ \frac{c G^{2}}{n}
= f(x_0)-f^{\star}+ \frac{4\alpha L G^{2}}{n}. \tag{22}
\]

Thus,
\[
\sum_{k=0}^{K-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le \frac{2}{\alpha}\Bigl(f(x_0)-f^{\star}\Bigr)
   + \frac{8 L G^{2}}{n}. \tag{23}
\]

Dividing by $K$ yields exactly the bound (3) stated in Theorem~\ref{thm:main}.

\subsection*{Step 6. Choice of stepsize for the $O(1/\sqrt{K})$ rate}
Set $\alpha=\frac{1}{3L}\sqrt{\frac{n}{K}}$ (which respects $\alpha\le 1/(3L)$). Plugging into (3):
\[
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K}
= O\!\Bigl(\frac{1}{\sqrt{K}}\Bigr),\qquad
\frac{12L\alpha G^{2}}{n}= O\!\Bigl(\frac{1}{\sqrt{K}}\Bigr).
\]
Hence the dominant term is $O(1/\sqrt{K})$, proving the claimed rate.

\section*{Rate Derivation (With Constants)}
Collecting all constants explicitly, for any $K\ge1$ and stepsize $\alpha\le 1/(3L)$,
\[
\boxed{
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha K}
+\frac{12L\alpha G^{2}}{n}
}
\]
If $f$ satisfies the Polyak–Łojasiewicz condition, the same recursion (21) yields linear convergence, but this lies beyond the present non‑convex analysis.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Convex $f$.} When $f$ is convex, the same proof gives a $O(1/K)$ rate for the optimality gap $f(\bar x_K)-f^{\star}$.
\item \textbf{Infinite‑sum (online) limit.} As $n\to\infty$, the variance term $\frac{12L\alpha G^{2}}{n}$ vanishes, and SAGA behaves like full‑gradient descent with stepsize $\alpha$, recovering the deterministic $O(1/K)$ rate for smooth non‑convex functions.
\item \textbf{Mini‑batch variant.} Selecting a batch of $b$ indices per iteration simply replaces $\alpha$ by $\alpha b$ in (21) and improves the variance term by a factor $b/n$.
\end{enumerate}
