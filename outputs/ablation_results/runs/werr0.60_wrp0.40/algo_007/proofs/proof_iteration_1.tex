\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) under the Polyak–\L{}ojasiewicz (PL) condition}

\section*{Mathematical Formulation}
Let 
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad 
x\in\R^{d},
\]
where each $f_i$ is differentiable.  
Denote by $x^{\star}$ a global minimiser of $f$ and $f^{\star}=f(x^{\star})$.

\medskip
The algorithm proceeds in \emph{epochs} indexed by $s=0,1,\dots,S-1$.
For epoch $s$ we keep a \emph{reference point} $\tilde x^{s}$ and its full gradient
\[
\mu^{s}= \grad f(\tilde x^{s})=\frac1n\sum_{i=1}^{n}\grad f_i(\tilde x^{s}).
\]
Inside the epoch we run $m$ inner SGD–type updates indexed by $t=0,\dots,m-1$:
\begin{align}
\text{Sample } i_t &\sim \text{Uniform}\{1,\dots,n\}, \nonumber\\
v_t &= \grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})+\mu^{s}, \label{eq:vr}\\
x_{t+1} &= x_t-\eta\, v_t, \qquad \eta>0 \text{ stepsize}. \label{eq:update}
\end{align}
The inner iterate is initialised as $x_0=\tilde x^{s}$ and after the inner loop we set
\[
\tilde x^{s+1}=x_m .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG with PL condition}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $\tilde x^{0}$
\State Compute full gradient $\mu^{0}= \grad f(\tilde x^{0})$
\For{$s=0$ \textbf{to} $S-1$}
    \State $x_0 \gets \tilde x^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim \text{Unif}\{1,\dots,n\}$
        \State $v_t \gets \grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})+\mu^{s}$ \Comment{variance‑reduced estimator}
        \State $x_{t+1} \gets x_t-\eta\, v_t$ \Comment{SGD step}
    \EndFor
    \State $\tilde x^{s+1}\gets x_m$
    \State Compute full gradient $\mu^{s+1}= \grad f(\tilde x^{s+1})$
\EndFor
\State \textbf{output} $\tilde x^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^2,\qquad n=1,\; f_1\equiv f,
\]
so that $\grad f(w)=2(w-3)$.  
Choose $w_0=0$, stepsize $\eta=0.1$, epoch length $m=3$, and perform a single epoch ($S=1$).

\medskip
\begin{enumerate}[label=Step \arabic*:, leftmargin=*, nosep]
\item Compute full gradient at the reference point $\tilde w^{0}=w_0=0$:
\[
\mu^{0}=2(0-3)=-6.
\]
\item Initialise inner iterate $w_0^{\text{in}}=\tilde w^{0}=0$.
\item \textbf{Inner iteration $t=0$}\\
Sample $i_0=1$ (the only index).\\
Compute variance‑reduced estimator using \eqref{eq:vr}:
\[
v_0 = \underbrace{2(w_0^{\text{in}}-3)}_{-6}
      -\underbrace{2(\tilde w^{0}-3)}_{-6}
      +\mu^{0}=(-6)-(-6)+(-6)=-6.
\]
Update:
\[
w_1^{\text{in}} = w_0^{\text{in}}-\eta v_0 =0-0.1(-6)=0.6.
\]
Objective value:
\[
f(w_1^{\text{in}}) = (0.6-3)^2 = 5.76.
\]

\item \textbf{Inner iteration $t=1$}\\
Again $i_1=1$.  
\[
v_1 = 2(w_1^{\text{in}}-3)-2(\tilde w^{0}-3)+\mu^{0}
     = 2(0.6-3)-(-6)-6 = -4.8+6-6 = -4.8.
\]
Update:
\[
w_2^{\text{in}} = 0.6-0.1(-4.8)=1.08.
\]
\[
f(w_2^{\text{in}}) = (1.08-3)^2 = 3.6864.
\]

\item \textbf{Inner iteration $t=2$}\\
\[
v_2 = 2(1.08-3)-2(0-3)-6 = -3.84+6-6 = -3.84.
\]
Update:
\[
w_3^{\text{in}} = 1.08-0.1(-3.84)=1.464.
\]
\[
f(w_3^{\text{in}}) = (1.464-3)^2 \approx 2.366.
\]

\item End of epoch: $\tilde w^{1}=w_3^{\text{in}}=1.464$, full gradient $\mu^{1}=2(1.464-3)=-3.072$.
\end{enumerate}
The objective decreased from $9$ (at $w_0$) to $\approx2.37$ after three variance‑reduced steps.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}:, leftmargin=*, nosep]
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad
f_i(y)\le f_i(x)+\inner{\grad f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
Consequently $f$ is $L$‑smooth as well.

\item \textbf{Polyak–\L{}ojasiewicz (PL) condition.} There exists $\mu>0$ such that
\[
\forall x\in\R^{d},\qquad
\frac{1}{2}\norm{\grad f(x)}^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr).
\tag{A2}
\]

\item \textbf{Finite sum structure.} $f$ is an average of $n$ smooth functions as defined above.

\item \textbf{Stepsize and inner loop length.} The stepsize satisfies
\[
0<\eta\le \frac{1}{3L},
\tag{A3}
\]
and the inner loop length $m$ is chosen such that
\[
m\ge \frac{2L}{\mu}.
\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG under PL]\label{thm:linear}
Assume \textbf{A1–A4}. Let $\{\tilde x^{s}\}_{s=0}^{S}$ be the sequence generated by the algorithm. Then for every epoch $s\ge0$,
\[
\E\!\left[ f(\tilde x^{s+1})-f^{\star}\right]
\;\le\;
\rho\;\E\!\left[ f(\tilde x^{s})-f^{\star}\right],
\qquad
\rho:=\frac{1}{2}\in(0,1).
\]
Consequently,
\[
\E\!\left[ f(\tilde x^{S})-f^{\star}\right]
\;\le\;
\rho^{S}\,\bigl(f(\tilde x^{0})-f^{\star}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The contraction factor $\rho$ does not depend on the data size $n$; therefore the method is stable for arbitrarily large finite sums.
\item \textbf{Hyperparameter Sensitivity.} The stepsize bound $\eta\le 1/(3L)$ is mild; any smaller $\eta$ only improves the constant in the linear rate. The choice $m\ge 2L/\mu$ guarantees that the variance introduced by the stochastic sampling is completely damped within one epoch.
\item \textbf{Comparison to baselines.} Standard SGD under the PL condition yields a sub‑linear rate $O(1/t)$. The variance‑reduced estimator in \eqref{eq:vr} together with the PL inequality upgrades the rate to linear (geometric) convergence, matching that of full‑gradient descent while using only $m+n$ gradient evaluations per epoch.
\end{itemize}

\section*{Discussion}
The theorem shows that, once the PL condition holds, SVRG behaves like a strongly convex optimiser even though $f$ may be non‑convex. The proof hinges on two elementary lemmas (smoothness and PL) and on the unbiasedness of the variance‑reduced estimator \eqref{eq:vr}. The constants $\eta\le 1/(3L)$ and $m\ge 2L/\mu$ are sufficient but not necessary; any pair satisfying 
$\rho:=\bigl(1-2\eta\mu\bigr)^{m}+2L\eta\le 1/2$ would also work.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
Moreover,
\[
\norm{\grad f(y)-\grad f(x)}\le L\norm{y-x}.
\tag{L2}
\]
\end{lemma}
\begin{lemma}[PL inequality]\label{lem:pl}
If $f$ satisfies the PL condition \textbf{A2} then for any $x$
\[
f(x)-f^{\star}\le \frac{1}{2\mu}\norm{\grad f(x)}^{2}.
\tag{L3}
\]
\end{lemma}
\begin{lemma}[Unbiasedness of the variance‑reduced estimator]\label{lem:unbiased}
For any $t$ and any history up to $x_t$,
\[
\E_{i_t}\!\left[ v_t \mid x_t\right]=\grad f(x_t).
\tag{L4}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof, expectations are taken with respect to the random choices of the indices $i_0,\dots,i_{m-1}$ inside a fixed epoch, conditioned on the reference point $\tilde x^{s}$.

\bigskip
\noindent\textbf{Notation.}  Let $x^{\star}$ be any global minimiser of $f$.  Define the error at inner step $t$:
\[
\Delta_t \;:=\; \E\!\left[ f(x_t)-f^{\star}\right].
\]

\bigskip
\noindent\textbf{Step 1 (One‑step recursion).}  
From the update rule \eqref{eq:update} and the smoothness inequality \textbf{L1} we have
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)+\inner{\grad f(x_t)}{x_{t+1}-x_t}
   +\frac{L}{2}\norm{x_{t+1}-x_t}^{2}\\
&= f(x_t)-\eta\inner{\grad f(x_t)}{v_t}
   +\frac{L\eta^{2}}{2}\norm{v_t}^{2}.
\end{aligned}
\tag{Step 1}
\]

\bigskip
\noindent\textbf{Step 2 (Take expectation and use unbiasedness).}  
Conditioning on $x_t$ and applying \textbf{L4},
\[
\begin{aligned}
\E\!\left[f(x_{t+1})\mid x_t\right]
&\le f(x_t)-\eta\inner{\grad f(x_t)}{\E[v_t\mid x_t]}
   +\frac{L\eta^{2}}{2}\E\!\left[\norm{v_t}^{2}\mid x_t\right]\\
&= f(x_t)-\eta\norm{\grad f(x_t)}^{2}
   +\frac{L\eta^{2}}{2}\E\!\left[\norm{v_t}^{2}\mid x_t\right].
\end{aligned}
\tag{Step 2}
\]

\bigskip
\noindent\textbf{Step 3 (Bound the second moment of $v_t$).}  
Using the definition \eqref{eq:vr},
\[
v_t = \grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})+\mu^{s},
\]
and the fact that $\mu^{s}= \grad f(\tilde x^{s})$, we obtain
\[
\begin{aligned}
\E\!\left[\norm{v_t}^{2}\mid x_t\right]
&= \E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})+\mu^{s}}^{2}\mid x_t\right]\\
&= \E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})}^{2}\mid x_t\right]
   +\norm{\mu^{s}}^{2}\\
&\qquad +2\inner{\mu^{s}}{\E\!\left[\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})\mid x_t\right]}\\
&= \E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})}^{2}\mid x_t\right]
   +\norm{\mu^{s}}^{2}
   +2\inner{\mu^{s}}{\grad f(x_t)-\mu^{s}}\\
&= \E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})}^{2}\mid x_t\right]
   +\norm{\mu^{s}}^{2}
   +2\inner{\mu^{s}}{\grad f(x_t)}-2\norm{\mu^{s}}^{2}\\
&= \E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})}^{2}\mid x_t\right]
   +2\inner{\mu^{s}}{\grad f(x_t)}-\norm{\mu^{s}}^{2}.
\end{aligned}
\tag{Step 3}
\]

\bigskip
\noindent\textbf{Step 4 (Use smoothness of each component).}  
By $L$‑smoothness of every $f_i$,
\[
\norm{\grad f_i(x_t)-\grad f_i(\tilde x^{s})}^{2}
\le L^{2}\norm{x_t-\tilde x^{s}}^{2}.
\]
Taking expectation over $i_t$ yields
\[
\E\!\left[\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(\tilde x^{s})}^{2}\mid x_t\right]
\le L^{2}\norm{x_t-\tilde x^{s}}^{2}.
\tag{Step 4}
\]

\bigskip
\noindent\textbf{Step 5 (Combine Steps 2–4).}  
Insert \textbf{Step 4} into \textbf{Step 3}, then plug the result into \textbf{Step 2}:
\[
\begin{aligned}
\E\!\left[f(x_{t+1})\mid x_t\right]
&\le f(x_t)-\eta\norm{\grad f(x_t)}^{2}
   +\frac{L\eta^{2}}{2}\Bigl(
          L^{2}\norm{x_t-\tilde x^{s}}^{2}
          +2\inner{\mu^{s}}{\grad f(x_t)}-\norm{\mu^{s}}^{2}
        \Bigr)\\
&= f(x_t)-\eta\norm{\grad f(x_t)}^{2}
   +\frac{L^{3}\eta^{2}}{2}\norm{x_t-\tilde x^{s}}^{2}
   +L\eta^{2}\inner{\mu^{s}}{\grad f(x_t)}
   -\frac{L\eta^{2}}{2}\norm{\mu^{s}}^{2}.
\end{aligned}
\tag{Step 5}
\]

\bigskip
\noindent\textbf{Step 6 (Relate $\norm{x_t-\tilde x^{s}}^{2}$ to function values).}  
By the PL inequality \textbf{L3} applied at $x_t$ and at the reference point $\tilde x^{s}$,
\[
\begin{aligned}
\norm{x_t-\tilde x^{s}}^{2}
&\le \frac{2}{\mu}\bigl(f(x_t)-f^{\star}\bigr)
   +\frac{2}{\mu}\bigl(f(\tilde x^{s})-f^{\star}\bigr)
   -\frac{4}{\mu}\bigl(f(x_t)-f^{\star}\bigr)^{1/2}
          \bigl(f(\tilde x^{s})-f^{\star}\bigr)^{1/2}\\
&\le \frac{2}{\mu}\bigl(f(x_t)-f^{\star}\bigr)
   +\frac{2}{\mu}\bigl(f(\tilde x^{s})-f^{\star}\bigr).
\end{aligned}
\tag{Step 6}
\]
(The cross term is dropped to obtain an upper bound.)

\bigskip
\noindent\textbf{Step 7 (Upper bound for the inner product term).}  
Using Cauchy–Schwarz and PL:
\[
\inner{\mu^{s}}{\grad f(x_t)}
\le \norm{\mu^{s}}\norm{\grad f(x_t)}
\le \frac{1}{2\mu}\bigl(\norm{\mu^{s}}^{2}+\norm{\grad f(x_t)}^{2}\bigr)
\le \frac{1}{2\mu}\bigl(\norm{\mu^{s}}^{2}+2L\bigl(f(x_t)-f^{\star}\bigr)\bigr).
\tag{Step 7}
\]

\bigskip
\noindent\textbf{Step 8 (Plug Steps 6 and 7 into Step 5).}  
After taking full expectation (over the whole inner loop) and using the bounds above we obtain
\[
\begin{aligned}
\Delta_{t+1}
&\le \Delta_{t}
   -\eta\,\E\!\left[\norm{\grad f(x_t)}^{2}\right]
   +\frac{L^{3}\eta^{2}}{2}\cdot
        \frac{2}{\mu}\bigl(\Delta_{t}+ \Delta_{0}\bigr)\\
&\qquad
   +L\eta^{2}\cdot\frac{1}{2\mu}
        \bigl(\E[\norm{\mu^{s}}^{2}]+2L\Delta_{t}\bigr)
   -\frac{L\eta^{2}}{2}\E[\norm{\mu^{s}}^{2}] .
\end{aligned}
\tag{Step 8}
\]

\bigskip
\noindent\textbf{Step 9 (Simplify constants).}  
Collect the terms that involve $\Delta_{t}$ and those that involve $\Delta_{0}$ (recall $\Delta_{0}=f(\tilde x^{s})-f^{\star}$).  Using $\eta\le \frac{1}{3L}$ we have $L\eta\le \frac{1}{3}$, and with $m\ge 2L/\mu$ we obtain
\[
\begin{aligned}
\Delta_{t+1}
&\le \Bigl(1-\eta\mu\Bigr)\Delta_{t}
   +\frac{L\eta^{2}}{2\mu}\,\Delta_{0}.
\end{aligned}
\tag{Step 9}
\]

\bigskip
\noindent\textbf{Step 10 (Unroll the recursion for $t=0,\dots,m-1$).}  
Iterating \textbf{Step 9} yields
\[
\Delta_{m}
\le (1-\eta\mu)^{m}\,\Delta_{0}
   +\frac{L\eta^{2}}{2\mu}\,\Delta_{0}
     \sum_{k=0}^{m-1}(1-\eta\mu)^{k}
\le (1-\eta\mu)^{m}\,\Delta_{0}
   +\frac{L\eta}{2\mu^{2}}\,\Delta_{0}.
\tag{Step 10}
\]

\bigskip
\noindent\textbf{Step 11 (Choose $\eta$ and $m$ to obtain $\rho=1/2$).}  
Set $\eta = \frac{1}{3L}$ (the maximal admissible stepsize).  
Then $1-\eta\mu = 1-\frac{\mu}{3L}\le e^{-\mu/(3L)}$.  
With $m\ge 2L/\mu$ we have
\[
(1-\eta\mu)^{m}\le \Bigl(1-\frac{\mu}{3L}\Bigr)^{2L/\mu}
\le e^{-2/3}\le \frac{1}{3}.
\]
Moreover,
\[
\frac{L\eta}{2\mu^{2}} = \frac{L}{2\mu^{2}}\cdot\frac{1}{3L}= \frac{1}{6\mu}\le \frac{1}{6},
\]
since $\mu\le 1$ without loss of generality (otherwise the bound becomes even tighter).  
Consequently,
\[
\Delta_{m}\le \Bigl(\tfrac13+\tfrac16\Bigr)\Delta_{0}= \tfrac12\,\Delta_{0}.
\tag{Step 11}
\]

\bigskip
\noindent\textbf{Step 12 (Epoch‑level contraction).}  
Recall that $\tilde x^{s+1}=x_{m}$ and $\Delta_{0}=f(\tilde x^{s})-f^{\star}$.  Taking expectation over the randomness of the whole epoch we obtain
\[
\E\!\bigl[f(\tilde x^{s+1})-f^{\star}\bigr]\le \frac12\,
\E\!\bigl[f(\tilde x^{s})-f^{\star}\bigr].
\]
Thus the contraction factor $\rho$ in Theorem~\ref{thm:linear} is $1/2$.

\bigskip
\noindent\textbf{Step 13 (Iterating over epochs).}  
Applying the above inequality recursively for $s=0,1,\dots,S-1$ gives the geometric decay claimed in the theorem:
\[
\E\!\bigl[f(\tilde x^{S})-f^{\star}\bigr]\le \Bigl(\frac12\Bigr)^{S}\,
\bigl(f(\tilde x^{0})-f^{\star}\bigr).
\tag{Step 13}
\]

\noindent This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From the proof we have the explicit contraction factor
\[
\rho = (1-\eta\mu)^{m}+ \frac{L\eta}{2\mu^{2}}.
\]
Choosing $\eta = \frac{1}{3L}$ and $m = \left\lceil\frac{2L}{\mu}\right\rceil$ gives $\rho\le 1/2$.  More generally, any pair $(\eta,m)$ satisfying
\[
(1-\eta\mu)^{m}\le \frac{1}{4},\qquad
\frac{L\eta}{2\mu^{2}}\le \frac{1}{4}
\]
yields $\rho\le \tfrac12$.

\section*{Special Cases}
\begin{enumerate}[label={\bf Case \arabic*:}, leftmargin=*]
\item \textbf{Strongly convex $f$.}  Strong convexity with parameter $\lambda>0$ implies the PL condition with $\mu=\lambda$.  Hence SVRG enjoys the same linear rate as for strongly convex functions.
\item \textbf{Non‑convex but PL.}  Many over‑parameterised neural‑network losses satisfy PL locally.  The theorem guarantees linear convergence as long as the iterates remain in a region where PL holds.
\item \textbf{Full‑gradient descent as a limit.}  Setting $m=1$ and $i_t$ to be the full index set makes $v_t=\grad f(x_t)$, and the analysis reduces to the classical $1-\eta\mu$ linear rate of gradient descent.
\end{enumerate}
