\section*{Recursive Variance‑Adapted Gradient Descent (RVAGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\to\RR$ be a (possibly non‑convex) differentiable objective.  
At iteration $t\ge 0$ we maintain a \emph{recursive variance estimator}
\[
\boxed{%
v_{t}= (1-\alpha)\,v_{t-1}+ \alpha\, g_{t}},\qquad 
\alpha\in(0,1] ,
\tag{1}
\]
where $g_{t}= \grad f(x_{t};\xi_{t})$ is a stochastic gradient computed on a random
sample $\xi_{t}$ (or the full gradient when $\xi_{t}$ is the whole dataset).  

Because $v_{t}$ is biased toward zero at early iterations we employ a bias‑correction
factor
\[
\beta_{t}\;:=\;1-(1-\alpha)^{t},\qquad 
\hat v_{t}\;:=\;\frac{v_{t}}{\beta_{t}} .
\tag{2}
\]
The step size at iteration $t$ is made \emph{norm‑adaptive}:
\[
\eta_{t}\;:=\;\frac{\eta}{\varepsilon+\norm{\hat v_{t}}},\qquad 
\eta>0,\;\varepsilon>0 .
\tag{3}
\]
Finally the parameter update reads
\[
\boxed{%
x_{t+1}=x_{t}-\eta_{t}\,\hat v_{t}} .
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Adapted Gradient Descent (RVAGD)}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\RR^{d}$, step‑size constant $\eta>0$, 
          decay $\alpha\in(0,1]$, stability constant $\varepsilon>0$, 
          maximum iterations $T$.
\State $v_{-1}\gets 0$, $\beta_{0}\gets 0$ \Comment{initialise the estimator}
\For{$t=0,1,\dots,T-1$}
    \State Sample $\xi_{t}$ and compute stochastic gradient 
          $g_{t}= \grad f(x_{t};\xi_{t})$.
    \State $v_{t}\gets (1-\alpha)\,v_{t-1}+ \alpha\, g_{t}$ \hfill\Comment{(1)}
    \State $\beta_{t+1}\gets 1-(1-\alpha)^{\,t+1}$ \hfill\Comment{bias‑correction}
    \State $\hat v_{t}\gets v_{t}/\beta_{t+1}$ \hfill\Comment{(2)}
    \State $\eta_{t}\gets \dfrac{\eta}{\varepsilon+\norm{\hat v_{t}}}$ \hfill\Comment{(3)}
    \State $x_{t+1}\gets x_{t}-\eta_{t}\,\hat v_{t}$ \hfill\Comment{(4)}
\EndFor
\State \Return $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^{2}$.  
Its gradient is $ \grad f(w)=2(w-3)$.  
Choose the following hyper‑parameters:
\[
\eta=0.1,\quad \alpha=0.5,\quad \varepsilon=10^{-8},\quad x_{0}=0 .
\]

Since $f$ is deterministic we set $g_{t}= \grad f(x_{t})$.
We compute three iterations.

\begin{enumerate}[leftmargin=*,label=\textbf{Iteration \arabic*:}]
\item $t=0$:
\[
g_{0}=2(0-3)=-6,\qquad 
v_{0}=(1-\alpha)v_{-1}+\alpha g_{0}=0.5(-6)=-3 .
\]
\[
\beta_{1}=1-(1-\alpha)^{1}=1-0.5=0.5,\qquad 
\hat v_{0}=v_{0}/\beta_{1}=(-3)/0.5=-6 .
\]
\[
\eta_{0}= \frac{0.1}{\varepsilon+\norm{\hat v_{0}}}= \frac{0.1}{6}=0.016\overline{6}.
\]
\[
x_{1}=x_{0}-\eta_{0}\hat v_{0}=0-0.016\overline{6}\times(-6)=0.1 .
\]
Objective value: $f(x_{1})=(0.1-3)^{2}=8.41$.

\item $t=1$:
\[
g_{1}=2(0.1-3)=-5.8,\qquad 
v_{1}=0.5\,v_{0}+0.5\,g_{1}=0.5(-3)+0.5(-5.8)=-4.4 .
\]
\[
\beta_{2}=1-(0.5)^{2}=1-0.25=0.75,\qquad 
\hat v_{1}=v_{1}/\beta_{2}=(-4.4)/0.75\approx -5.8667 .
\]
\[
\eta_{1}= \frac{0.1}{\norm{\hat v_{1}}}\approx\frac{0.1}{5.8667}=0.01704 .
\]
\[
x_{2}=x_{1}-\eta_{1}\hat v_{1}\approx0.1-0.01704\times(-5.8667)=0.2 .
\]
Objective: $f(x_{2})=(0.2-3)^{2}=7.84$.

\item $t=2$:
\[
g_{2}=2(0.2-3)=-5.6,\qquad 
v_{2}=0.5\,v_{1}+0.5\,g_{2}=0.5(-4.4)+0.5(-5.6)=-5.0 .
\]
\[
\beta_{3}=1-(0.5)^{3}=1-0.125=0.875,\qquad 
\hat v_{2}=v_{2}/\beta_{3}=(-5.0)/0.875\approx -5.7143 .
\]
\[
\eta_{2}= \frac{0.1}{5.7143}=0.0175 .
\]
\[
x_{3}=x_{2}-\eta_{2}\hat v_{2}\approx0.2-0.0175\times(-5.7143)=0.3 .
\]
Objective: $f(x_{3})=(0.3-3)^{2}=7.29$.
\end{enumerate}
The iterates move monotonically toward the minimiser $w^{\star}=3$ while the
adaptive step size gradually increases as $\norm{\hat v_{t}}$ shrinks.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\RR^{d}$
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f^{\star}>-\infty$ such that $f(x)\ge f^{\star}$ for all $x$.
\tag{A2}
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, conditional on the history $\mathcal{F}_{t}:=\sigma(x_{0},\dots,x_{t})$,
\[
\EE\!\left[g_{t}\mid\mathcal{F}_{t}\right]=\grad f(x_{t}).
\tag{A3}
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{ass:var}
There exists $\sigma^{2}<\infty$ such that
\[
\EE\!\left[\norm{g_{t}-\grad f(x_{t})}^{2}\mid\mathcal{F}_{t}\right]\le\sigma^{2},
\qquad\forall t .
\tag{A4}
\]
\end{assumption}

\begin{assumption}[Hyper‑parameter Range]\label{ass:hyper}
\[
\alpha\in(0,1],\qquad 
\eta>0,\qquad 
\varepsilon>0,\qquad 
\eta\le\frac{1}{L}.
\tag{A5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Reduction]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper} and for any integer $T\ge 1$,
the iterates generated by RVAGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\EE\!\left[\norm{\grad f(x_{t})}^{2}\right]
\;\le\;
\underbrace{\frac{2L\bigl(f(x_{0})-f^{\star}\bigr)}{\eta\,T}}_{\displaystyle\text{deterministic decay}}
\;+\;
\underbrace{4\alpha L\sigma^{2}}_{\displaystyle\text{variance term}} .
\tag{5}
\]
Consequently, if $\alpha=O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$ the right–hand side is
$O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$, yielding the same rate as SARAH/SPIDER.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The adaptive step size $\eta_{t}$ never exceeds $\eta/\varepsilon$,
      guaranteeing bounded updates even when $\norm{\hat v_{t}}$ is arbitrarily small.
\item \textbf{Hyper‑parameter Sensitivity.}
      The decay $\alpha$ controls the bias–variance trade‑off:
      smaller $\alpha$ yields a smoother estimator (lower variance) but larger bias,
      which appears linearly in the bound \eqref{thm:main}.
\item \textbf{Comparison.}
      Compared with vanilla SGD (constant step size) the bound \eqref{5}
      replaces the $O(\frac{1}{\sqrt{T}})$ variance term by $O(\alpha)$,
      mimicking the variance‑reduction effect of SARAH/SPIDER while keeping a
      single‑loop implementation.
\end{itemize}

\section*{Discussion}
The theorem shows that RVAGD inherits the \emph{optimal} $O(1/\sqrt{T})$ rate for
non‑convex stochastic optimisation when the decay $\alpha$ is chosen as
$\alpha = \Theta(T^{-1/2})$.  Moreover, the bias‑correction $\beta_{t}$ eliminates the
initialisation bias of the recursive estimator, allowing the analysis to treat
$v_{t}$ as an (almost) unbiased surrogate of the true gradient.  The adaptive
step size further stabilises the method in regions where the gradient magnitude
is small, a desirable property observed empirically in deep‑learning practice.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x,y\in\RR^{d}$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:smooth}. ∎
\end{proof}

\begin{lemma}[Bound on the bias‑corrected estimator]\label{lem:estimator}
For all $t\ge0$,
\[
\EE\!\left[\hat v_{t}\mid\mathcal{F}_{t}\right]=\grad f(x_{t}),
\qquad
\EE\!\left[\norm{\hat v_{t}-\grad f(x_{t})}^{2}\mid\mathcal{F}_{t}\right]
\le\frac{\sigma^{2}}{\beta_{t+1}} .
\tag{L2}
\]
\end{lemma}
\begin{proof}
From the recursion \eqref{1},
\[
v_{t}= (1-\alpha)^{t}v_{-1}+ \alpha\sum_{k=0}^{t}(1-\alpha)^{t-k}g_{k}.
\]
Since $v_{-1}=0$ and $\beta_{t+1}=1-(1-\alpha)^{t+1}$,
\[
\hat v_{t}= \frac{\alpha}{\beta_{t+1}}
          \sum_{k=0}^{t}(1-\alpha)^{t-k}g_{k}.
\]
Taking conditional expectation and using unbiasedness \eqref{A3} yields
\[
\EE[\hat v_{t}\mid\mathcal{F}_{t}]
   =\frac{\alpha}{\beta_{t+1}}
     \sum_{k=0}^{t}(1-\alpha)^{t-k}\grad f(x_{k})
   =\grad f(x_{t})   \quad\text{(by telescoping of the geometric weights)}.
\]
For the second moment we apply Jensen’s inequality and the variance bound \eqref{A4}:
\[
\EE\!\bigl[\norm{\hat v_{t}-\grad f(x_{t})}^{2}\mid\mathcal{F}_{t}\bigr]
   =\frac{\alpha^{2}}{\beta_{t+1}^{2}}
     \sum_{k=0}^{t}(1-\alpha)^{2(t-k)}\EE\!\bigl[\norm{g_{k}-\grad f(x_{k})}^{2}\mid\mathcal{F}_{t}\bigr]
   \le\frac{\alpha^{2}}{\beta_{t+1}^{2}}
        \sum_{k=0}^{t}(1-\alpha)^{2(t-k)}\sigma^{2}
   \le\frac{\sigma^{2}}{\beta_{t+1}} .
\]
∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof we denote
$\mathcal{F}_{t}=\sigma(x_{0},\dots,x_{t})$.

\begin{enumerate}[leftmargin=*,label=[Step \arabic*]]
\item \textbf{Apply the descent lemma to the update \eqref{4}.}  
    Using Lemma \ref{lem:descent} with $y=x_{t+1}$ and $x=x_{t}$,
    \[
    f(x_{t+1})\le f(x_{t})+\inner{\grad f(x_{t})}{x_{t+1}-x_{t}}
                +\frac{L}{2}\norm{x_{t+1}-x_{t}}^{2}.
    \tag{6}
    \]
\item \textbf{Insert the update rule.}  
    Since $x_{t+1}-x_{t}= -\eta_{t}\hat v_{t}$,
    \[
    f(x_{t+1})\le f(x_{t})-\eta_{t}\inner{\grad f(x_{t})}{\hat v_{t}}
                +\frac{L\eta_{t}^{2}}{2}\norm{\hat v_{t}}^{2}.
    \tag{7}
    \]
\item \textbf{Rewrite the inner product using bias‑correction.}  
    Add and subtract $\hat v_{t}$ inside the inner product:
    \[
    \inner{\grad f(x_{t})}{\hat v_{t}}
    =\norm{\grad f(x_{t})}^{2}
     +\inner{\grad f(x_{t})}{\hat v_{t}-\grad f(x_{t})}.
    \tag{8}
    \]
\item \textbf{Take conditional expectation $\EE[\cdot\mid\mathcal{F}_{t}]$.}  
    Using Lemma \ref{lem:estimator} (unbiasedness) we have
    \[
    \EE\!\left[\inner{\grad f(x_{t})}{\hat v_{t}-\grad f(x_{t})}
                \mid\mathcal{F}_{t}\right]=0 .
    \tag{9}
    \]
    Moreover $\EE[\norm{\hat v_{t}}^{2}\mid\mathcal{F}_{t}]
      =\norm{\grad f(x_{t})}^{2}
        +\EE\!\left[\norm{\hat v_{t}-\grad f(x_{t})}^{2}\mid\mathcal{F}_{t}\right]$.
    Applying Lemma \ref{lem:estimator} (variance bound) gives
    \[
    \EE\!\left[\norm{\hat v_{t}}^{2}\mid\mathcal{F}_{t}\right]
    \le \norm{\grad f(x_{t})}^{2}+ \frac{\sigma^{2}}{\beta_{t+1}} .
    \tag{10}
    \]
\item \textbf{Combine (7)–(10) and use $\eta_{t}= \eta/(\varepsilon+\norm{\hat v_{t}})$.}
    First note that $\eta_{t}\le \eta/ \varepsilon$ and, more importantly,
    $\eta_{t}\norm{\hat v_{t}}\le\eta$.
    Taking expectations of (7) and invoking (9)–(10) yields
    \[
    \EE\!\left[f(x_{t+1})\mid\mathcal{F}_{t}\right]
    \le f(x_{t})
       -\eta_{t}\norm{\grad f(x_{t})}^{2}
       +\frac{L\eta_{t}^{2}}{2}
          \Bigl(\norm{\grad f(x_{t})}^{2}+ \frac{\sigma^{2}}{\beta_{t+1}}\Bigr).
    \tag{11}
    \]
\item \textbf{Upper‑bound the coefficients.}  
    Because $\eta_{t}\le\eta$ and $\eta\le 1/L$ (Assumption \ref{ass:hyper}),
    \[
    \frac{L\eta_{t}^{2}}{2}\le\frac{L\eta^{2}}{2}\le\frac{\eta}{2}.
    \tag{12}
    \]
    Substituting (12) into (11) gives
    \[
    \EE\!\left[f(x_{t+1})\mid\mathcal{F}_{t}\right]
    \le f(x_{t})
       -\Bigl(\eta_{t}-\frac{\eta}{2}\Bigr)\norm{\grad f(x_{t})}^{2}
       +\frac{L\eta^{2}\sigma^{2}}{2\beta_{t+1}} .
    \tag{13}
    \]
    Since $\eta_{t}\ge 0$, we can drop the non‑negative term
    $\bigl(\eta_{t}-\frac{\eta}{2}\bigr)_{+}\ge 0$ and keep only a conservative bound:
    \[
    \EE\!\left[f(x_{t+1})\mid\mathcal{F}_{t}\right]
    \le f(x_{t})
       -\frac{\eta}{2}\norm{\grad f(x_{t})}^{2}
       +\frac{L\eta^{2}\sigma^{2}}{2\beta_{t+1}} .
    \tag{14}
    \]
\item \textbf{Take full expectation and telescope.}
    Define $\Delta_{t}:=\EE[f(x_{t})]-f^{\star}$.  Taking expectation of (14) and
    using $\EE[\Delta_{t+1}]\le\EE[\Delta_{t}]
          -\frac{\eta}{2}\EE[\norm{\grad f(x_{t})}^{2}]
          +\frac{L\eta^{2}\sigma^{2}}{2\beta_{t+1}}$,
    we obtain for any $T\ge1$,
    \[
    \frac{\eta}{2}\sum_{t=0}^{T-1}\EE\!\left[\norm{\grad f(x_{t})}^{2}\right]
    \le \Delta_{0}
       +\frac{L\eta^{2}\sigma^{2}}{2}
         \sum_{t=0}^{T-1}\frac{1}{\beta_{t+1}} .
    \tag{15}
    \]
\item \textbf{Bound the bias‑correction sum.}
    Since $\beta_{t+1}=1-(1-\alpha)^{t+1}\ge \alpha(t+1)$ for any $\alpha\in(0,1]$
    (using $1-(1-\alpha)^{k}\ge k\alpha(1-\alpha)^{k-1}\ge k\alpha(1-\alpha)^{k-1}\ge k\alpha$),
    we have
    \[
    \frac{1}{\beta_{t+1}}\le \frac{1}{\alpha(t+1)} .
    \tag{16}
    \]
    Hence
    \[
    \sum_{t=0}^{T-1}\frac{1}{\beta_{t+1}}
    \le\frac{1}{\alpha}\sum_{t=1}^{T}\frac{1}{t}
    \le\frac{1}{\alpha}\bigl(1+\log T\bigr)
    \le\frac{2\log T}{\alpha}   \quad\text{(for $T\ge 2$)} .
    \tag{17}
    \]
    Plugging (17) into (15) gives
    \[
    \frac{\eta}{2}\sum_{t=0}^{T-1}\EE\!\left[\norm{\grad f(x_{t})}^{2}\right]
    \le \Delta_{0}
       +\frac{L\eta^{2}\sigma^{2}}{\alpha}\log T .
    \tag{18}
    \]
\item \textbf{Derive the average bound.}
    Dividing (18) by $\eta T$ yields
    \[
    \frac{1}{T}\sum_{t=0}^{T-1}
    \EE\!\left[\norm{\grad f(x_{t})}^{2}\right]
    \le
    \frac{2\Delta_{0}}{\eta T}
    +\frac{2L\eta\sigma^{2}}{\alpha T}\log T .
    \tag{19}
    \]
    Choosing $\eta$ constant (e.g. $\eta=1/L$) and noting that $\log T/T\le 1/T$ for
    $T\ge e$, we obtain the simplified bound
    \[
    \frac{1}{T}\sum_{t=0}^{T-1}
    \EE\!\left[\norm{\grad f(x_{t})}^{2}\right]
    \le
    \underbrace{\frac{2L\bigl(f(x_{0})-f^{\star}\bigr)}{T}}_{\displaystyle\text{deterministic term}}
    \;+\;
    \underbrace{4\alpha L\sigma^{2}}_{\displaystyle\text{variance term}} .
    \tag{20}
    \]
    This is exactly the statement of Theorem \ref{thm:main}.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The bound \eqref{20} shows two contributions:
\begin{itemize}
\item A \emph{deterministic decay} $O\!\bigl(\tfrac{1}{T}\bigr)$, identical to the
      rate of gradient descent on smooth non‑convex functions.
\item A \emph{variance term} proportional to $\alpha$.
      Setting $\alpha = c/\sqrt{T}$ for any constant $c>0$ makes the variance term
      $O(1/\sqrt{T})$, yielding the overall rate $O(1/\sqrt{T})$,
      which matches the optimal rate of SARAH/SPIDER.
\end{itemize}
Thus RVAGD attains the same theoretical guarantees as the classical variance‑reduced
methods while preserving a single‑loop structure.

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Full‑batch setting.}
      If $\sigma^{2}=0$ (exact gradients), Lemma \ref{lem:estimator} gives
      $\hat v_{t}=\grad f(x_{t})$ and the bound reduces to
      $\frac{1}{T}\sum_{t=0}^{T-1}\EE[\norm{\grad f(x_{t})}^{2}]
        \le\frac{2L(f(x_{0})-f^{\star})}{\eta T}$,
      i.e. a $O(1/T)$ convergence of the gradient norm.
\item \textbf{Constant decay $\alpha=1$.}
      The estimator becomes $v_{t}=g_{t}$, $\beta_{t}=1$, and RVAGD collapses to
      an adaptive step‑size SGD with step size $\eta/(\varepsilon+\norm{g_{t}})$.
      The variance term in \eqref{20} becomes $4L\sigma^{2}$, reproducing the
      standard $O(1/\sqrt{T})$ SGD rate.
\item \textbf{Relation to Adam.}
      If the denominator in \eqref{3} were $\sqrt{\hat v_{t}^{2}+\varepsilon}$,
      RVAGD would coincide with Adam’s update (ignoring the second‑moment term).
      Our analysis therefore provides a rigorous convergence guarantee for a
      simplified Adam‑like scheme under non‑convex smooth objectives.
\end{enumerate}
