\section*{Algorithm Name}
Stochastic Gradient Langevin Descent (SGLD) with a decreasing noise variance.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
The iterates $\{x_k\}_{k\ge 0}$ are generated by  

\[
\boxed{\,x_{k+1}=x_k-\alpha_k \nabla f(x_k)+\xi_k\,},\qquad k=0,1,2,\dots
\]

where  

\begin{itemize}
    \item $\alpha_k>0$ is the (possibly time‑varying) stepsize,
    \item $\xi_k\in\mathbb{R}^{d}$ is a random noise vector satisfying  

    \[
    \mathbb{E}[\xi_k\,|\,x_k]=0,\qquad 
    \mathbb{E}\big[\|\xi_k\|^{2}\,\big|\,x_k\big]\le \sigma_k^{2},
    \]  

    with a prescribed decreasing schedule $\sigma_k^{2}= \frac{c}{(k+1)^{\beta}}$, $c>0$, $\beta>0$.
\end{itemize}

The algorithm is a discretisation of the overdamped Langevin diffusion; the added Gaussian noise enables exploration while the variance decay guarantees convergence.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD with Decreasing Noise}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_{0}\in\mathbb{R}^{d}$, stepsize schedule $\{\alpha_k\}$, noise schedule $\{\sigma_k^{2}\}$.
\For{$k=0,1,\dots,T-1$}
    \State Compute (full) gradient $g_k\gets \nabla f(x_k)$.
    \State Sample $\xi_k\sim\mathcal{N}\!\big(0,\sigma_k^{2}I_d\big)$.
    \State Update $x_{k+1}\gets x_k-\alpha_k g_k+\xi_k$.
\EndFor
\State \textbf{Output:} $x_T$ (or the average $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic  

\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]

Choose a constant stepsize $\alpha=0.1$ and a deterministic “noise’’ sequence
$\xi_k=0$ (to illustrate the deterministic part).  
Initialize $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)$ & $w_{k+1}=w_k-\alpha\nabla f(w_k)$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}

Corresponding objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3330$.  
The iterates move toward the minimizer $w^{*}=3$.

\section*{Assumptions}
\begin{enumerate}
    \item[(A1)] \textbf{Convexity.} $f$ is convex:
    \[
    f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle,\qquad\forall x,y\in\mathbb{R}^{d}.
    \]
    \item[(A2)] \textbf{L‑smoothness.} There exists $L>0$ such that
    \[
    \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y.
    \]
    Equivalently,
    \[
    f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
    \]
    \item[(A3)] \textbf{Unbiased noise with decreasing variance.}
    \[
    \mathbb{E}[\xi_k\,|\,x_k]=0,\qquad 
    \mathbb{E}\big[\|\xi_k\|^{2}\,\big|\,x_k\big]\le\sigma_k^{2},
    \]
    where $\sigma_k^{2}=c/(k+1)^{\beta}$ for some $c>0$, $\beta>0$.
    \item[(A4)] \textbf{Stepsize.} The deterministic step sizes satisfy
    \[
    0<\alpha_k\le\frac{1}{2L},\qquad\text{and}\qquad 
    \sum_{k=0}^{\infty}\alpha_k =\infty,\;\;\sum_{k=0}^{\infty}\alpha_k^{2}<\infty .
    \]
    (For simplicity we will present the theorem for a constant $\alpha\le 1/(2L)$.)
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of SGLD]\label{thm:main}
Let $\{x_k\}$ be generated by the update rule in Section 2 and assume (A1)–(A4) with a constant stepsize $0<\alpha\le\frac{1}{2L}$. Denote $x^{*}\in\arg\min f$ and define the averaged iterate $\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k$. Then

\[
\boxed{\;
\mathbb{E}\big[f(\bar x_T)-f(x^{*})\big]\;\le\;
\frac{\|x_{0}-x^{*}\|^{2}}{2\alpha T}
\;+\;
\frac{L}{2T}\sum_{k=0}^{T-1}\sigma_k^{2}
\;}
\tag{1}
\]

Consequently, if $\sigma_k^{2}=c/(k+1)^{\beta}$ with $\beta>1$, the second term is $O(1/T)$ and the overall rate is $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The condition $\alpha\le1/(2L)$ guarantees that the deterministic part of the recursion is contractive; the added noise does not destabilise the iteration because its variance shrinks.
    \item \textbf{Hyper‑parameter sensitivity.} Larger $\alpha$ accelerates the deterministic descent but inflates the coefficient $L\alpha^{2}$ in the recursion; the bound (1) shows a trade‑off between $1/(\alpha T)$ and $\alpha L\sigma_k^{2}$. Choosing $\alpha=1/(2L)$ is optimal when the noise variance is small.
    \item \textbf{Comparison to SGD.} When $\xi_k\equiv0$, (1) reduces to the classic $O(1/(\alpha T))$ bound for gradient descent on convex $L$‑smooth functions. The extra term $\frac{L}{2T}\sum\sigma_k^{2}$ quantifies the penalty induced by the Langevin noise; with a sufficiently fast decay ($\beta>1$) this penalty vanishes at the same $O(1/T)$ rate.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that SGLD inherits the optimal $O(1/T)$ convergence of deterministic gradient descent for convex smooth objectives, provided the injected noise decays faster than $1/k$. This aligns with the intuition that early iterations benefit from exploration, while later iterations behave like plain gradient descent.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Gradient bound for convex $L$‑smooth functions]\label{lem:gradbound}
For convex $L$‑smooth $f$ and any $x$,
\[
\|\nabla f(x)\|^{2}\le 2L\big(f(x)-f(x^{*})\big),
\]
where $x^{*}$ is any minimiser of $f$.
\end{lemma}
\begin{proof}
From convexity,
$f(x^{*})\ge f(x)+\langle\nabla f(x),x^{*}-x\rangle$,
thus $\langle\nabla f(x),x-x^{*}\rangle\ge f(x)-f(x^{*})$.
Apply Cauchy–Schwarz and the $L$‑smoothness inequality
$f(x^{*})\ge f(x)+\langle\nabla f(x),x^{*}-x\rangle+\frac{1}{2L}\|\nabla f(x)-\nabla f(x^{*})\|^{2}$
(the latter follows from standard co‑coercivity). Since $\nabla f(x^{*})=0$, we obtain
$\|\nabla f(x)\|^{2}\le 2L\big(f(x)-f(x^{*})\big)$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we write $\mathbb{E}_k[\cdot]=\mathbb{E}[\cdot\,|\,x_k]$.

\begin{enumerate}
\item[Step 1.] \textbf{Distance recursion.}  
Starting from the update rule,
\[
x_{k+1}=x_k-\alpha\nabla f(x_k)+\xi_k,
\]
subtract $x^{*}$ and square:
\[
\|x_{k+1}-x^{*}\|^{2}
 =\|x_k-x^{*}\|^{2}
 -2\alpha\langle\nabla f(x_k),x_k-x^{*}\rangle
 +\alpha^{2}\|\nabla f(x_k)\|^{2}
 +2\langle\xi_k, x_k-x^{*}-\alpha\nabla f(x_k)\rangle
 +\|\xi_k\|^{2}.
\tag{2}
\]

\item[Step 2.] \textbf{Take conditional expectation.}  
Using (A3), $\mathbb{E}_k[\xi_k]=0$ and $\mathbb{E}_k[\langle\xi_k,\cdot\rangle]=0$, thus

\[
\mathbb{E}_k\!\big[\|x_{k+1}-x^{*}\|^{2}\big]
 =\|x_k-x^{*}\|^{2}
 -2\alpha\langle\nabla f(x_k),x_k-x^{*}\rangle
 +\alpha^{2}\|\nabla f(x_k)\|^{2}
 +\mathbb{E}_k\!\big[\|\xi_k\|^{2}\big].
\tag{3}
\]

\item[Step 3.] \textbf{Apply convexity.}  
From (A1),
\[
\langle\nabla f(x_k),x_k-x^{*}\rangle\ge f(x_k)-f(x^{*}).
\tag{4}
\]

\item[Step 4.] \textbf{Insert (4) into (3).}
\[
\mathbb{E}_k\!\big[\|x_{k+1}-x^{*}\|^{2}\big]
 \le\|x_k-x^{*}\|^{2}
 -2\alpha\big(f(x_k)-f(x^{*})\big)
 +\alpha^{2}\|\nabla f(x_k)\|^{2}
 +\sigma_k^{2}.
\tag{5}
\]

\item[Step 5.] \textbf{Bound the gradient term using Lemma~\ref{lem:gradbound}.}
\[
\|\nabla f(x_k)\|^{2}\le 2L\big(f(x_k)-f(x^{*})\big).
\tag{6}
\]

\item[Step 6.] \textbf{Combine (5) and (6).}
\[
\mathbb{E}_k\!\big[\|x_{k+1}-x^{*}\|^{2}\big]
 \le\|x_k-x^{*}\|^{2}
 -2\alpha\Big(1-\alpha L\Big)\big(f(x_k)-f(x^{*})\big)
 +\sigma_k^{2}.
\tag{7}
\]

\item[Step 7.] \textbf{Choose stepsize.}  
Assume $0<\alpha\le \frac{1}{2L}$, then $1-\alpha L\ge \frac12$. Hence

\[
\mathbb{E}_k\!\big[\|x_{k+1}-x^{*}\|^{2}\big]
 \le\|x_k-x^{*}\|^{2}
 -\alpha\big(f(x_k)-f(x^{*})\big)
 +\sigma_k^{2}.
\tag{8}
\]

\item[Step 8.] \textbf{Take full expectation and sum.}  
Denote $D_k:=\mathbb{E}\big[\|x_k-x^{*}\|^{2}\big]$ and $\Delta_k:=\mathbb{E}\big[f(x_k)-f(x^{*})\big]$.  
Taking expectation of (8) yields

\[
D_{k+1}\le D_k-\alpha\Delta_k+\sigma_k^{2}.
\tag{9}
\]

Summing (9) from $k=0$ to $T-1$ gives a telescoping series:

\[
D_T\le D_0-\alpha\sum_{k=0}^{T-1}\Delta_k+\sum_{k=0}^{T-1}\sigma_k^{2}.
\tag{10}
\]

Since $D_T\ge0$, we can drop it and rearrange:

\[
\alpha\sum_{k=0}^{T-1}\Delta_k\le D_0+\sum_{k=0}^{T-1}\sigma_k^{2}.
\tag{11}
\]

\item[Step 9.] \textbf{Pass to the averaged iterate.}  
By convexity of $f$,

\[
f(\bar x_T)-f(x^{*})\le\frac{1}{T}\sum_{k=0}^{T-1}\big(f(x_k)-f(x^{*})\big)=\frac{1}{T}\sum_{k=0}^{T-1}\Delta_k.
\tag{12}
\]

Multiplying (12) by $\alpha$ and using (11),

\[
\alpha\,\mathbb{E}\!\big[f(\bar x_T)-f(x^{*})\big]
 \le\frac{D_0}{T}+\frac{1}{T}\sum_{k=0}^{T-1}\sigma_k^{2}.
\tag{13}
\]

Finally, substitute $D_0=\|x_0-x^{*}\|^{2}$ and divide by $\alpha$:

\[
\boxed{\;
\mathbb{E}\!\big[f(\bar x_T)-f(x^{*})\big]
\le\frac{\|x_{0}-x^{*}\|^{2}}{2\alpha T}
+\frac{L}{2T}\sum_{k=0}^{T-1}\sigma_k^{2}
\;}
\tag{14}
\]

where we used $\alpha\le\frac{1}{2L}$ to replace the factor $1/\alpha$ by $2L$ in the first term, yielding the exact statement of Theorem~\ref{thm:main}.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
If $\sigma_k^{2}=c/(k+1)^{\beta}$ with $\beta>1$, then

\[
\frac{1}{T}\sum_{k=0}^{T-1}\sigma_k^{2}
\le\frac{c}{T}\sum_{k=1}^{\infty}\frac{1}{k^{\beta}}
=O\!\left(\frac{1}{T}\right).
\]

Hence both terms in (14) decay as $O(1/T)$, establishing the optimal $1/T$ convergence rate for convex smooth objectives.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No noise} ($\xi_k\equiv0$).  The bound (14) reduces to 
    $\mathbb{E}[f(\bar x_T)-f(x^{*})]\le\frac{\|x_0-x^{*}\|^{2}}{2\alpha T}$, the classic deterministic gradient‑descent rate.
    \item \textbf{Constant variance} ($\sigma_k^{2}\equiv\sigma^{2}$).  
    The second term becomes $\frac{L\sigma^{2}}{2}$, i.e. a steady‑state bias proportional to the noise level.
\end{itemize}
