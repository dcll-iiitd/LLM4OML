\section*{Algorithm Name}
\textbf{Proximal Gradient Descent (PGD)} for minimizing a composite convex objective
\[
\min_{x\in\mathbb{R}^d}\;F(x)\equiv f(x)+g(x),
\]
where $f$ is convex and $L$‑smooth, and $g$ is proper, closed and convex (possibly nonsmooth).

\section*{Mathematical Formulation}
Given a stepsize $\eta>0$, the iterates are
\begin{equation}
\boxed{
x_{k+1}= \operatorname{prox}_{\eta g}\bigl(x_k-\eta \nabla f(x_k)\bigr)},
\qquad k=0,1,2,\dots
\end{equation}
where the proximal operator of $g$ is defined by
\[
\operatorname{prox}_{\eta g}(v)\;:=\;\arg\min_{u\in\mathbb{R}^d}
\Bigl\{ g(u)+\frac{1}{2\eta}\|u-v\|^2\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Descent}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, max iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_k \gets \nabla f(x_k)$.
    \State Take a gradient step $y_k \gets x_k - \eta\,g_k$.
    \State Apply proximal map $x_{k+1} \gets \operatorname{prox}_{\eta g}(y_k)$.
\EndFor
\State \textbf{return} $x_K$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ with $g\equiv0$ (hence the proximal operator reduces to the identity).  
Choose $w_0=0$, stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $f(w_k)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1 = w_0-\eta\nabla f(w_0)=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2 = 0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3 = 1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3689$
\end{tabular}
\end{center}
The objective value decreases from $9$ to $2.37$ after three iterations, illustrating the descent property of PGD.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity.} $f$ and $g$ are convex; thus $F$ is convex.
\item \textbf{$L$‑smoothness of $f$.}
\[
\forall x,y\in\mathbb{R}^d:\;
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|
\]
equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{A1}
\]
\item \textbf{Proximal non‑expansiveness.} For any $u,v\in\mathbb{R}^d$,
\[
\big\|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\big\|
\le\|u-v\|.
\tag{A2}
\]
\item \textbf{Stepsize.} Choose $\eta\in(0,1/L]$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence rate of PGD]\label{thm:pgd}
Under the assumptions above, let $\{x_k\}$ be generated by PGD with stepsize
$\eta\in(0,1/L]$ and denote $x^\star\in\arg\min F$. Then for all $K\ge1$,
\[
F(x_K)-F(x^\star)\;\le\;\frac{\|x_0-x^\star\|^2}{2\eta K}.
\tag{T1}
\]
Consequently $F(x_K)\to F(x^\star)$ at the rate $\mathcal{O}(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The non‑expansiveness (A2) guarantees that the proximal step never amplifies the error introduced by the gradient step.
\item \textbf{Hyperparameter sensitivity.} The bound (T1) is monotone decreasing in $\eta$ up to $1/L$; choosing $\eta=1/L$ yields the smallest constant.
\item \textbf{Comparison.} When $g\equiv0$, PGD reduces to gradient descent and (T1) coincides with the classical $O(1/K)$ rate for convex smooth functions. With a non‑zero $g$, PGD enjoys the same rate without any additional variance terms, unlike stochastic methods.
\end{itemize}

\section*{Discussion}
The theorem shows that a single proximal step per iteration suffices to obtain the optimal $\mathcal{O}(1/K)$ convergence for convex composite problems. The proof relies only on convexity, smoothness of $f$, and the fundamental non‑expansiveness property (A2) of the proximal operator; no stochasticity or strong convexity is required.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point inequality for the proximal operator]\label{lem:three}
For any $x\in\mathbb{R}^d$ and any $u\in\mathbb{R}^d$,
\[
g(u)\ge g\bigl(\operatorname{prox}_{\eta g}(x)\bigr)
+ \frac{1}{2\eta}\Bigl\|\operatorname{prox}_{\eta g}(x)-x\Bigr\|^2
+ \frac{1}{2\eta}\Bigl\|u-\operatorname{prox}_{\eta g}(x)\Bigr\|^2
- \frac{1}{2\eta}\|u-x\|^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
The optimality condition of the proximal mapping yields
\[
0\in \partial g\bigl(\operatorname{prox}_{\eta g}(x)\bigr)
+\frac{1}{\eta}\bigl(\operatorname{prox}_{\eta g}(x)-x\bigr).
\]
Using convexity of $g$ and the standard three‑point identity gives \eqref{L1}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:pgd}. Let $y_k:=x_k-\eta\nabla f(x_k)$ and $x_{k+1}:=\operatorname{prox}_{\eta g}(y_k)$. Denote $x^\star\in\arg\min F$.

\bigskip
\noindent\textbf{Step 1.} Smoothness upper bound on $f(x_{k+1})$.
\begin{align}
f(x_{k+1})
&\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle
   +\frac{L}{2}\|x_{k+1}-x_k\|^2
   \qquad\text{[by (A1)]} \nonumber\\
&= f(x_k)-\frac{1}{\eta}\langle y_k-x_k,x_{k+1}-x_k\rangle
   +\frac{L}{2}\|x_{k+1}-x_k\|^2. \tag{Step 1}
\end{align}

\noindent\textbf{Step 2.} Expand the inner product using the identity
$\langle a,b\rangle=\frac12(\|a\|^2+\|b\|^2-\|a-b\|^2)$ with
$a=y_k-x_k$ and $b=x_{k+1}-x_k$:
\begin{align}
-\frac{1}{\eta}\langle y_k-x_k,x_{k+1}-x_k\rangle
&= -\frac{1}{2\eta}\Bigl(\|y_k-x_k\|^2+\|x_{k+1}-x_k\|^2
      -\|y_k-x_{k+1}\|^2\Bigr). \tag{Step 2}
\end{align}

\noindent\textbf{Step 3.} Substitute \eqref{Step 2} into \eqref{Step 1}:
\begin{align}
f(x_{k+1})
&\le f(x_k)
   -\frac{1}{2\eta}\|y_k-x_k\|^2
   -\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_k\|^2
   +\frac{1}{2\eta}\|y_k-x_{k+1}\|^2 .
   \tag{Step 3}
\end{align}
Since $\eta\le 1/L$, the coefficient $\frac{1}{2\eta}-\frac{L}{2}\ge0$.

\noindent\textbf{Step 4.} Apply Lemma~\ref{lem:three} with $x=y_k$ and $u=x^\star$:
\begin{align}
g(x_{k+1})
&\le g(x^\star)
   +\frac{1}{2\eta}\bigl(\|x^\star-y_k\|^2-\|x^\star-x_{k+1}\|^2
                         -\|x_{k+1}-y_k\|^2\bigr). \tag{Step 4}
\end{align}

\noindent\textbf{Step 5.} Add the inequalities of Step~3 and Step~4:
\begin{align}
F(x_{k+1}) &:= f(x_{k+1})+g(x_{k+1}) \nonumber\\
&\le f(x_k)+g(x^\star)
   -\frac{1}{2\eta}\|y_k-x_k\|^2
   -\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_k\|^2
   +\frac{1}{2\eta}\|y_k-x_{k+1}\|^2 \nonumber\\
&\qquad +\frac{1}{2\eta}\bigl(\|x^\star-y_k\|^2-\|x^\star-x_{k+1}\|^2
                         -\|x_{k+1}-y_k\|^2\bigr) \nonumber\\
&= f(x_k)+g(x^\star)
   -\frac{1}{2\eta}\|y_k-x_k\|^2
   -\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_k\|^2
   +\frac{1}{2\eta}\|x^\star-y_k\|^2
   -\frac{1}{2\eta}\|x^\star-x_{k+1}\|^2 . \tag{Step 5}
\end{align}
All terms $\|y_k-x_{k+1}\|^2$ cancel.

\noindent\textbf{Step 6.} Observe that
\[
y_k-x_k = -\eta\nabla f(x_k) \quad\Longrightarrow\quad
\|y_k-x_k\|^2 = \eta^2\|\nabla f(x_k)\|^2\ge0,
\]
hence we may drop the non‑positive term $-\frac{1}{2\eta}\|y_k-x_k\|^2$ to obtain a simpler inequality:
\begin{align}
F(x_{k+1})-F(x^\star)
&\le F(x_k)-F(x^\star)
   -\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_k\|^2
   +\frac{1}{2\eta}\bigl(\|x^\star-y_k\|^2-\|x^\star-x_k\|^2\bigr) \nonumber\\
&\qquad +\frac{1}{2\eta}\bigl(\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2\bigr). \tag{Step 6}
\end{align}
Since $\|x^\star-y_k\|^2=\|x^\star-(x_k-\eta\nabla f(x_k))\|^2
= \|x^\star-x_k\|^2-2\eta\langle\nabla f(x_k),x^\star-x_k\rangle
+\eta^2\|\nabla f(x_k)\|^2$, the middle two brackets combine to
$-\frac{1}{\eta}\langle\nabla f(x_k),x^\star-x_k\rangle$.
Using convexity of $f$,
$\langle\nabla f(x_k),x_k-x^\star\rangle\ge f(x_k)-f(x^\star)$,
we obtain
\begin{align}
F(x_{k+1})-F(x^\star)
&\le F(x_k)-F(x^\star)
   -\frac{1}{\eta}\bigl(f(x_k)-f(x^\star)\bigr)
   -\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_k\|^2 . \tag{Step 7}
\end{align}
Discarding the non‑negative term involving $\|x_{k+1}-x_k\|^2$ yields the clean recursion
\begin{align}
F(x_{k+1})-F(x^\star)
\le F(x_k)-F(x^\star)-\frac{1}{\eta}\bigl(f(x_k)-f(x^\star)\bigr). \tag{Step 8}
\end{align}
Because $g(x^\star)\le g(x_{k+1})$, $F(x_k)-F(x^\star)\ge f(x_k)-f(x^\star)$, so
\[
F(x_{k+1})-F(x^\star)\le F(x_k)-F(x^\star)-\frac{1}{\eta}\bigl(F(x_k)-F(x^\star)\bigr)
= \Bigl(1-\frac{1}{\eta}\Bigr)\bigl(F(x_k)-F(x^\star)\bigr). 
\]
However, we need a bound that does not explode when $\eta<1$. The standard technique is to work with the distance to the optimum.

\noindent\textbf{Step 9.} From Lemma~\ref{lem:three} with $u=x^\star$ and using $y_k=x_k-\eta\nabla f(x_k)$ we have
\[
\|x^\star-x_{k+1}\|^2
\le \|x^\star-y_k\|^2-\|x_{k+1}-y_k\|^2
\le \|x^\star-y_k\|^2 .
\]
Expanding $\|x^\star-y_k\|^2$ gives
\begin{align}
\|x^\star-x_{k+1}\|^2
\le \|x^\star-x_k\|^2-2\eta\langle\nabla f(x_k),x_k-x^\star\rangle
+\eta^2\|\nabla f(x_k)\|^2 . \tag{Step 10}
\end{align}
Using convexity of $f$,
$\langle\nabla f(x_k),x_k-x^\star\rangle\ge f(x_k)-f(x^\star)$,
and dropping the non‑negative $\eta^2\|\nabla f(x_k)\|^2$ term, we obtain
\begin{align}
\|x^\star-x_{k+1}\|^2
\le \|x^\star-x_k\|^2-2\eta\bigl(f(x_k)-f(x^\star)\bigr). \tag{Step 11}
\end{align}
Since $g(x^\star)\le g(x_{k+1})$, $F(x_k)-F(x^\star)\ge f(x_k)-f(x^\star)$, thus
\begin{align}
\|x^\star-x_{k+1}\|^2
\le \|x^\star-x_k\|^2-2\eta\bigl(F(x_k)-F(x^\star)\bigr). \tag{Step 12}
\end{align}

\noindent\textbf{Step 13.} Rearranging \eqref{Step 12} gives the fundamental descent inequality
\begin{align}
F(x_k)-F(x^\star)
\le \frac{1}{2\eta}\bigl(\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2\bigr). \tag{Step 13}
\end{align}
Summing \eqref{Step 13} for $k=0,\dots,K-1$ telescopes:
\begin{align}
\sum_{k=0}^{K-1}\bigl(F(x_k)-F(x^\star)\bigr)
\le \frac{1}{2\eta}\bigl(\|x^\star-x_0\|^2-\|x^\star-x_{K}\|^2\bigr)
\le \frac{\|x_0-x^\star\|^2}{2\eta}. \tag{Step 14}
\end{align}
Since the left‑hand side is a sum of $K$ non‑negative terms, each term is bounded by the average:
\begin{align}
F(x_{K})-F(x^\star)
\le \frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_k)-F(x^\star)\bigr)
\le \frac{\|x_0-x^\star\|^2}{2\eta K}. \tag{Step 15}
\end{align}
This is exactly the claim \eqref{T1}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:pgd} we have the explicit constant
\[
C(\eta)=\frac{\|x_0-x^\star\|^2}{2\eta},
\qquad
F(x_K)-F(x^\star)\le \frac{C(\eta)}{K}.
\]
Choosing the maximal admissible stepsize $\eta=1/L$ minimizes $C(\eta)$:
\[
C_{\min}= \frac{L}{2}\,\|x_0-x^\star\|^2,
\quad\Longrightarrow\quad
F(x_K)-F(x^\star)\le \frac{L\|x_0-x^\star\|^2}{2K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent} ($g\equiv0$). The proximal operator becomes the identity, and the proof reduces to the classical $O(1/K)$ result for smooth convex $f$.
\item \textbf{Indicator function $g=\iota_{\mathcal{C}}$}. Then $\operatorname{prox}_{\eta g}$ is the Euclidean projection onto a convex set $\mathcal{C}$, and PGD coincides with the projected gradient method; the same $O(1/K)$ bound holds.
\item \textbf{Strongly convex $F$}. If $F$ is $\mu$‑strongly convex, a slight modification of Step~13 yields a linear rate $O\big((1-\eta\mu)^K\big)$.
\end{itemize}
