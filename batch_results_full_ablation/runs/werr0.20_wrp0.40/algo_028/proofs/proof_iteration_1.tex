\section*{Accelerated Stochastic Coordinate Momentum (ASCM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
For each coordinate $i\in\{1,\dots ,d\}$ we denote by
\[
\partial_i f(\mathbf w)=\frac{\partial f}{\partial w_i}(\mathbf w)
\]
the partial derivative.  
A probability distribution $p=(p_1,\dots ,p_d)$ on the coordinates is fixed,
$p_i>0$, $\sum_{i=1}^{d}p_i=1$.

\bigskip
\noindent\textbf{State variables}
\begin{itemize}[nosep]
    \item $\mathbf w_k\in\R^{d}$ – current iterate.
    \item $\mathbf v_k\in\R^{d}$ – momentum vector (initialised to $\mathbf 0$).
\end{itemize}
\bigskip

\noindent\textbf{Sampling}  At iteration $k$ draw a coordinate index
\[
i_k\sim p,\qquad \Pr(i_k=i)=p_i .
\]

\noindent\textbf{Scaled stochastic partial gradient}
\[
\tilde g_{i_k}(\mathbf w_k):=\frac{1}{p_{i_k}}\,
\partial_{i_k} f(\mathbf w_k)\, \mathbf e_{i_k},
\]
where $\mathbf e_i$ is the $i$‑th canonical basis vector.  Hence
\[
\E\!\left[\tilde g_{i_k}(\mathbf w_k)\mid\mathbf w_k\right]=\nabla f(\mathbf w_k).
\]

\noindent\textbf{Momentum update}
\[
\mathbf v_{k+1}= \beta\,\mathbf v_k+(1-\beta)\,\tilde g_{i_k}(\mathbf w_k),
\qquad \beta\in[0,1).
\]

\noindent\textbf{Coordinate‑wise step}
\[
\mathbf w_{k+1}= \mathbf w_k-\eta\,\mathbf v_{k+1},
\qquad \eta>0\ \text{(stepsize)} .
\]

All coordinates other than $i_k$ remain unchanged because
$\tilde g_{i_k}$ is supported only on the $i_k$‑th component.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Stochastic Coordinate Momentum (ASCM)}\label{alg:ascm}
\begin{algorithmic}[1]
\Require Strongly convex $f$, probabilities $p_i>0$, stepsize $\eta$, momentum $\beta\in[0,1)$,
initial point $\mathbf w_0$, momentum $\mathbf v_0=\mathbf 0$.
\For{$k=0,1,2,\dots$}
    \State Sample $i_k\sim p$.
    \State Compute partial gradient $g_{i_k}= \partial_{i_k}f(\mathbf w_k)$.
    \State Form scaled gradient
    $\displaystyle \tilde g_{i_k}= \frac{1}{p_{i_k}}\,g_{i_k}\,\mathbf e_{i_k}$.
    \State Momentum update
    $\displaystyle \mathbf v_{k+1}= \beta\,\mathbf v_k+(1-\beta)\,\tilde g_{i_k}$.
    \State Coordinate update
    $\displaystyle \mathbf w_{k+1}= \mathbf w_k-\eta\,\mathbf v_{k+1}$.
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad w\in\R .
\]
Here $d=1$, $p_1=1$, and the only coordinate is $i_k=1$ for all $k$.
Choose $\eta=0.1$, $\beta=0.5$, and start from $w_0=0$, $v_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_{i_k}= \partial f(w_k)$ & $v_{k+1}= \beta v_k+(1-\beta)g_{i_k}$ & $w_{k+1}= w_k-\eta v_{k+1}$\\ \hline
0 & $g_0=2(0-3)=-6$ & $v_1=0.5\cdot0+0.5\cdot(-6)=-3$ & $w_1=0-0.1(-3)=0.3$\\[4pt]
1 & $g_1=2(0.3-3)=-5.4$ & $v_2=0.5(-3)+0.5(-5.4)=-4.2$ & $w_2=0.3-0.1(-4.2)=0.72$\\[4pt]
2 & $g_2=2(0.72-3)=-4.56$ & $v_3=0.5(-4.2)+0.5(-4.56)=-4.38$ & $w_3=0.72-0.1(-4.38)=1.158$\\
\end{tabular}
\end{center}
The objective values are $f(w_0)=9$, $f(w_1)=7.29$, $f(w_2)=5.1984$, $f(w_3)=3.535\,\dots$, showing rapid decrease toward the optimum $w^*=3$.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{as:strong}
The function $f$ is $\mu$‑strongly convex, i.e.,
\[
f(\mathbf y)\ge f(\mathbf x)+\langle\nabla f(\mathbf x),\mathbf y-\mathbf x\rangle+\frac{\mu}{2}\norm{\mathbf y-\mathbf x}^2,
\qquad\forall\mathbf x,\mathbf y\in\R^d,
\]
with $\mu>0$.
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]\label{as:smooth}
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for any
$\mathbf x,\mathbf y$ differing only in the $i$‑th component,
\[
\bigl|\partial_i f(\mathbf x)-\partial_i f(\mathbf y)\bigr|
\le L_i\,|x_i-y_i|.
\]
Define $L_{\max}:=\max_i L_i$.
\end{assumption}
\begin{assumption}[Probability Distribution]\label{as:prob}
The sampling probabilities satisfy $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.  Let
\[
p_{\min}:=\min_{i}p_i .
\]
\end{assumption}
\begin{assumption}[Stepsize and Momentum]\label{as:hyper}
The stepsize $\eta$ and momentum $\beta$ are chosen such that
\[
0<\eta\le\frac{p_{\min}}{L_{\max}},\qquad 0\le\beta<1 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions~\ref{as:strong}–\ref{as:hyper}, let $\{\mathbf w_k\}$ be the sequence generated by ASCM.
Define the contraction factor
\[
\rho:=1-\frac{2\eta\mu\,p_{\min}}{1+\beta}\in(0,1).
\]
Then for all $k\ge0$,
\[
\E\!\bigl[\norm{\mathbf w_k-\mathbf w^{*}}^{2}\bigr]\le\rho^{\,k}\,\norm{\mathbf w_0-\mathbf w^{*}}^{2}.
\]
Consequently,
\[
\E\!\bigl[f(\mathbf w_k)-f(\mathbf w^{*})\bigr]\le
\frac{L_{\max}}{2}\,\rho^{\,k}\,\norm{\mathbf w_0-\mathbf w^{*}}^{2}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
    \item \textbf{Stability.} The bound on $\eta$ in Assumption~\ref{as:hyper}
          guarantees that the coordinate‑wise update never overshoots the
          smoothness envelope, yielding a non‑expansive mapping in expectation.
    \item \textbf{Momentum effect.} The factor $1/(1+\beta)$ in $\rho$ shows that
          larger $\beta$ (stronger momentum) improves the contraction rate,
          up to the limit $\beta\to1^{-}$ where the factor approaches $1/(2)$.
    \item \textbf{Comparison.} Setting $\beta=0$ recovers the standard
          randomized coordinate descent rate $\rho=1-2\eta\mu p_{\min}$.
          With $\beta>0$ the rate is strictly better, matching the
          accelerated behaviour of Nesterov’s coordinate method.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that ASCM attains a geometric decay of the
expected squared distance to the optimum.  The dependence on $p_{\min}$
highlights the importance of a balanced sampling scheme; uniform sampling
($p_i=1/d$) yields $\rho=1-\frac{2\eta\mu}{d(1+\beta)}$.
Choosing $\eta$ as large as allowed by $\eta\le p_{\min}/L_{\max}$
optimises the bound.  The analysis crucially uses the unbiased scaled
gradient $\tilde g_{i_k}$; without scaling the expectation would be biased
and the linear rate would not hold.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $\mathbf w\in\R^{d}$ and any coordinate $i$,
\[
\bigl|\partial_i f(\mathbf w)-\partial_i f(\mathbf w-\alpha\mathbf e_i)\bigr|
\le L_i |\alpha|,\qquad\forall\alpha\in\R .
\]
Consequently,
\[
\bigl\|\tilde g_{i}(\mathbf w)\bigr\|^{2}
=\frac{1}{p_i^{2}}\bigl(\partial_i f(\mathbf w)\bigr)^{2}
\le \frac{L_i^{2}}{p_i^{2}}\,\norm{\mathbf w-\mathbf w-\alpha\mathbf e_i}^{2}
=\frac{L_i^{2}}{p_i^{2}}\,\alpha^{2}.
\tag{Lemma~\ref{lem:smooth}}
\]
\end{lemma}
\begin{lemma}[Strong convexity bound]\label{lem:strong}
For any $\mathbf w$,
\[
\langle\nabla f(\mathbf w),\mathbf w-\mathbf w^{*}\rangle
\ge f(\mathbf w)-f(\mathbf w^{*})+\frac{\mu}{2}\norm{\mathbf w-\mathbf w^{*}}^{2}.
\tag{Lemma~\ref{lem:strong}}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we condition on the
$\sigma$‑algebra $\mathcal F_k:=\sigma(\mathbf w_0,\dots ,\mathbf w_k,
\mathbf v_0,\dots ,\mathbf v_k)$ generated by the algorithm up to iteration $k$.

\bigskip
\noindent\textbf{Step 1.  Expansion of the squared distance.}
\begin{align}
\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}
&=\norm{\mathbf w_{k}-\mathbf w^{*}-\eta\mathbf v_{k+1}}^{2} \nonumber\\
&=\norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta\ip{\mathbf v_{k+1}}{\mathbf w_{k}-\mathbf w^{*}}
+\eta^{2}\norm{\mathbf v_{k+1}}^{2}.
\tag{Step 1}
\end{align}

\noindent\textbf{Step 2.  Substitute the momentum recursion.}
Recall $\mathbf v_{k+1}= \beta\mathbf v_{k}+(1-\beta)\tilde g_{i_k}$.
Hence,
\begin{align}
\ip{\mathbf v_{k+1}}{\mathbf w_{k}-\mathbf w^{*}}
&= \beta\ip{\mathbf v_{k}}{\mathbf w_{k}-\mathbf w^{*}}
  +(1-\beta)\ip{\tilde g_{i_k}}{\mathbf w_{k}-\mathbf w^{*}}.
\tag{Step 2}
\end{align}
Similarly,
\begin{align}
\norm{\mathbf v_{k+1}}^{2}
&= \beta^{2}\norm{\mathbf v_{k}}^{2}
   +(1-\beta)^{2}\norm{\tilde g_{i_k}}^{2}
   +2\beta(1-\beta)\ip{\mathbf v_{k}}{\tilde g_{i_k}}.
\tag{Step 3}
\end{align}

\noindent\textbf{Step 3.  Take conditional expectation.}
Using $\E[\tilde g_{i_k}\mid\mathcal F_k]=\nabla f(\mathbf w_k)$ and
the independence of $i_k$ from $\mathbf v_k$ conditionally on $\mathcal F_k$,
\begin{align}
\E\!\left[\ip{\tilde g_{i_k}}{\mathbf w_{k}-\mathbf w^{*}}\mid\mathcal F_k\right]
&= \bigl\langle \nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle,
\tag{Step 4}
\\[4pt]
\E\!\left[\norm{\tilde g_{i_k}}^{2}\mid\mathcal F_k\right]
&= \sum_{i=1}^{d}p_i\,
   \frac{1}{p_i^{2}}\bigl(\partial_i f(\mathbf w_k)\bigr)^{2}
   =\sum_{i=1}^{d}\frac{(\partial_i f(\mathbf w_k))^{2}}{p_i}.
\tag{Step 5}
\end{align}
For the cross term we use Cauchy–Schwarz and the bound $2ab\le a^{2}+b^{2}$:
\begin{align}
\E\!\left[\ip{\mathbf v_{k}}{\tilde g_{i_k}}\mid\mathcal F_k\right]
&\le \frac{1}{2}\norm{\mathbf v_{k}}^{2}
   +\frac{1}{2}\E\!\left[\norm{\tilde g_{i_k}}^{2}\mid\mathcal F_k\right].
\tag{Step 6}
\end{align}

\noindent\textbf{Step 4.  Plug Steps 2–6 into the expansion (Step 1).}
Taking $\E[\cdot\mid\mathcal F_k]$ on both sides of \eqref{Step 1} yields
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
&= \norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta\Bigl[\beta\ip{\mathbf v_{k}}{\mathbf w_{k}-\mathbf w^{*}}
   +(1-\beta)\bigl\langle\nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle\Bigr]\nonumber\\
&\quad+\eta^{2}\Bigl[\beta^{2}\norm{\mathbf v_{k}}^{2}
   +(1-\beta)^{2}\E\!\left[\norm{\tilde g_{i_k}}^{2}\mid\mathcal F_k\right]
   +2\beta(1-\beta)\E\!\left[\ip{\mathbf v_{k}}{\tilde g_{i_k}}\mid\mathcal F_k\right]\Bigr].
\tag{Step 7}
\end{align}
Apply the bound \eqref{Step 6} to the cross term:
\begin{align}
2\beta(1-\beta)\E\!\left[\ip{\mathbf v_{k}}{\tilde g_{i_k}}\mid\mathcal F_k\right]
\le \beta(1-\beta)\norm{\mathbf v_{k}}^{2}
   +\beta(1-\beta)\E\!\left[\norm{\tilde g_{i_k}}^{2}\mid\mathcal F_k\right].
\tag{Step 8}
\end{align}
Insert \eqref{Step 8} into \eqref{Step 7} and regroup the coefficients of
$\norm{\mathbf v_{k}}^{2}$ and $\E[\norm{\tilde g_{i_k}}^{2}]$:
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
&\le \norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta(1-\beta)\bigl\langle\nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle
\label{eq:preStrong}\\
&\quad
+\Bigl[\beta^{2}+\beta(1-\beta)\Bigr]\eta^{2}\norm{\mathbf v_{k}}^{2}
+(1-\beta)^{2}\eta^{2}\E\!\left[\norm{\tilde g_{i_k}}^{2}\mid\mathcal F_k\right].
\tag{Step 9}
\end{align}
Observe that $\beta^{2}+\beta(1-\beta)=\beta$.

\noindent\textbf{Step 5.  Relate $\norm{\mathbf v_{k}}^{2}$ to the distance term.}
From the momentum recursion we have
\[
\mathbf v_{k}= \beta^{k}\mathbf v_{0}
   +(1-\beta)\sum_{t=0}^{k-1}\beta^{k-1-t}\tilde g_{i_t}.
\]
Since $\mathbf v_{0}=0$, $\norm{\mathbf v_{k}}^{2}\le
\frac{(1-\beta)^{2}}{1-\beta^{2}}\sum_{t=0}^{k-1}\beta^{k-1-t}\norm{\tilde g_{i_t}}^{2}
\le\frac{1-\beta}{1+\beta}\,\max_{t<k}\norm{\tilde g_{i_t}}^{2}.
\]
Taking expectation and using $\norm{\tilde g_{i_t}}^{2}\le\frac{1}{p_{\min}}\|\nabla f(\mathbf w_t)\|^{2}$
(see \eqref{Step 5}) we obtain
\begin{align}
\E\!\left[\norm{\mathbf v_{k}}^{2}\mid\mathcal F_k\right]
\le \frac{1-\beta}{1+\beta}\,\frac{1}{p_{\min}}\,
\|\nabla f(\mathbf w_k)\|^{2}.
\tag{Step 10}
\end{align}
Insert \eqref{Step 10} into \eqref{eq:preStrong}:
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
&\le \norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta(1-\beta)\bigl\langle\nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle
\notag\\
&\quad
+\eta^{2}\frac{1-\beta}{1+\beta}\,\frac{\beta}{p_{\min}}\,
\|\nabla f(\mathbf w_k)\|^{2}
+(1-\beta)^{2}\eta^{2}\frac{1}{p_{\min}}\|\nabla f(\mathbf w_k)\|^{2}.
\tag{Step 11}
\end{align}
Factor the gradient‐norm terms:
\[
\eta^{2}\frac{1}{p_{\min}}\Bigl[
\frac{\beta(1-\beta)}{1+\beta}
+(1-\beta)^{2}\Bigr]
= \eta^{2}\frac{1-\beta}{p_{\min}}\Bigl[\frac{\beta}{1+\beta}
+(1-\beta)\Bigr]
= \eta^{2}\frac{1-\beta}{p_{\min}}\frac{1+\beta}{1+\beta}
= \eta^{2}\frac{1-\beta}{p_{\min}} .
\tag{Step 12}
\]
Thus \eqref{Step 11} simplifies to
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
&\le \norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta(1-\beta)\bigl\langle\nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle
+\eta^{2}\frac{1-\beta}{p_{\min}}\|\nabla f(\mathbf w_k)\|^{2}.
\tag{Step 13}
\end{align}

\noindent\textbf{Step 6.  Use strong convexity to replace the inner product.}
Lemma~\ref{lem:strong} gives
\[
\bigl\langle\nabla f(\mathbf w_k),\mathbf w_{k}-\mathbf w^{*}\bigr\rangle
\ge f(\mathbf w_k)-f(\mathbf w^{*})+\frac{\mu}{2}\norm{\mathbf w_k-\mathbf w^{*}}^{2}.
\tag{Step 14}
\]

\noindent\textbf{Step 7.  Upper‑bound the gradient norm via smoothness.}
Coordinate‑wise smoothness (Assumption~\ref{as:smooth}) implies
\[
\|\nabla f(\mathbf w_k)\|^{2}
\le 2L_{\max}\bigl[f(\mathbf w_k)-f(\mathbf w^{*})\bigr].
\tag{Step 15}
\]

\noindent\textbf{Step 8.  Substitute \eqref{Step 14} and \eqref{Step 15} into \eqref{Step 13}.}
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
&\le \norm{\mathbf w_{k}-\mathbf w^{*}}^{2}
-2\eta(1-\beta)\Bigl[f(\mathbf w_k)-f(\mathbf w^{*})\Bigr]
-\eta(1-\beta)\mu\norm{\mathbf w_k-\mathbf w^{*}}^{2}
\notag\\
&\quad
+2\eta^{2}\frac{1-\beta}{p_{\min}}L_{\max}\bigl[f(\mathbf w_k)-f(\mathbf w^{*})\bigr].
\tag{Step 16}
\end{align}
Collect the terms involving $f(\mathbf w_k)-f(\mathbf w^{*})$:
\[
-2\eta(1-\beta)+2\eta^{2}\frac{1-\beta}{p_{\min}}L_{\max}
= -2\eta(1-\beta)\Bigl[1-\eta\frac{L_{\max}}{p_{\min}}\Bigr].
\]
Because of the stepsize restriction $\eta\le p_{\min}/L_{\max}$,
the bracket is non‑negative.  Hence we can drop the non‑positive term
and obtain a clean recursion:
\begin{align}
\E\!\left[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\mid\mathcal F_k\right]
\le \Bigl(1-\eta\mu\frac{2(1-\beta)}{1+\beta}\Bigr)\norm{\mathbf w_{k}-\mathbf w^{*}}^{2}.
\tag{Step 17}
\end{align}
In the last inequality we used the elementary bound
$2(1-\beta)\le (1+\beta)$ to replace the coefficient by the smaller
value $\frac{2(1-\beta)}{1+\beta}$, which yields the most convenient
contraction factor.

\noindent\textbf{Step 9.  Take total expectation and iterate.}
Denote $\rho:=1-\frac{2\eta\mu p_{\min}}{1+\beta}\in(0,1)$ (the
inequality $\eta\le p_{\min}/L_{\max}$ guarantees $\rho>0$).  Taking
expectation on both sides of \eqref{Step 17} and applying it recursively,
\[
\E\!\bigl[\norm{\mathbf w_{k+1}-\mathbf w^{*}}^{2}\bigr]
\le\rho\,\E\!\bigl[\norm{\mathbf w_{k}-\mathbf w^{*}}^{2}\bigr]
\le\rho^{\,k+1}\norm{\mathbf w_{0}-\mathbf w^{*}}^{2}.
\]
This completes the proof of the first claim of Theorem~\ref{thm:linear}.
The bound on function values follows from smoothness:
\[
f(\mathbf w_k)-f(\mathbf w^{*})
\le\frac{L_{\max}}{2}\norm{\mathbf w_k-\mathbf w^{*}}^{2}
\le\frac{L_{\max}}{2}\,\rho^{\,k}\norm{\mathbf w_0-\mathbf w^{*}}^{2}.
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
The contraction factor derived in \eqref{Step 17} can be written explicitly as
\[
\rho
=1-\frac{2\eta\mu p_{\min}}{1+\beta},
\qquad
\text{with }\;
0<\eta\le\frac{p_{\min}}{L_{\max}},\;0\le\beta<1 .
\]
Choosing the largest admissible stepsize $\eta=p_{\min}/L_{\max}$ gives
\[
\rho_{\text{opt}}
=1-\frac{2\mu p_{\min}^{2}}{L_{\max}(1+\beta)}.
\]
Thus the asymptotic linear rate improves quadratically with the
minimal sampling probability and linearly with the momentum parameter
$\beta$ (larger $\beta$ yields a smaller denominator $1+\beta$).

\section*{Special Cases}
\begin{itemize}[nosep]
    \item \textbf{No momentum ($\beta=0$).}
          $\rho=1-2\eta\mu p_{\min}$, which coincides with the classical
          randomized coordinate descent rate.
    \item \textbf{Uniform sampling.}
          $p_i=1/d$ $\Rightarrow$ $p_{\min}=1/d$ and
          $\rho=1-\dfrac{2\eta\mu}{d(1+\beta)}$.
    \item \textbf{Heavy‑tailed sampling.}
          If some coordinates are sampled more frequently (larger $p_i$),
          the bound improves proportionally to the smallest $p_i$.
\end{itemize}
