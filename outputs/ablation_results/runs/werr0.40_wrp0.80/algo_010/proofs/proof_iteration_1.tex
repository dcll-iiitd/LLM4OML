\section*{Adaptive Variance‑Reduced Gradient (AVRG)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
At iteration $t$ we have a stochastic gradient sample $g_t$ satisfying  

\[
\E[g_t\mid x_t]=\nabla f(x_t),\qquad 
\E\!\left[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\right]\le\sigma^{2}.
\]

AVRG maintains an exponential moving average of the stochastic gradients  

\[
\boxed{v_t =(1-\alpha)v_{t-1}+\alpha g_t},\qquad v_{0}=0,\;\alpha\in(0,1].
\tag{1}
\]

A bias‑corrected estimator is  

\[
\boxed{\hat v_t = \frac{v_t}{1-(1-\alpha)^{t}}}.
\tag{2}
\]

The step size is adaptive and depends on the norm of the corrected estimator:  

\[
\boxed{\eta_t = \frac{\eta}{\beta+\|\hat v_t\|}},\qquad \eta>0,\;\beta>0.
\tag{3}
\]

The iterate update reads  

\[
\boxed{x_{t+1}=x_t-\eta_t \,\hat v_t}.
\tag{4}
\]

The algorithm stops after a prescribed number $T$ of iterations or when a
pre‑specified tolerance on $\|\hat v_t\|$ is reached.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Adaptive Variance‑Reduced Gradient (AVRG)}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\eta>0$, decay $\beta>0$, averaging weight $\alpha\in(0,1]$, max iter $T$.
\State $v_{0}\gets 0$.
\For{$t=0,\dots,T-1$}
    \State Sample a stochastic gradient $g_t$ (e.g. minibatch) at $x_t$.
    \State $v_t \gets (1-\alpha)v_{t-1}+\alpha g_t$ \hfill\textbf{(1)}
    \State $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$ \hfill\textbf{(2)}
    \State $\eta_t \gets \displaystyle\frac{\eta}{\beta+\|\hat v_t\|}$ \hfill\textbf{(3)}
    \State $x_{t+1} \gets x_t-\eta_t\hat v_t$ \hfill\textbf{(4)}
\EndFor
\State \Return $x_T$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.
Then $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, $\eta=0.1$, $\beta=0$, $\alpha=0.5$ and use the exact gradient
as stochastic gradient ($g_t=\nabla f(w_t)$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $v_t$ & $w_{t+1}=w_t-\eta_t\hat v_t$\\\hline
0 & $0$ & $-6$ & $v_0=(1-\alpha)0+\alpha(-6)=-3$ &
$\hat v_0=v_0/(1-(1-\alpha))=-3/0.5=-6$, $\eta_0=0.1/|{-6}|=0.0167$, $w_1=0-0.0167(-6)=0.100$\\\hline
1 & $0.100$ & $2(0.1-3)=-5.8$ &
$v_1=(1-\alpha)(-3)+\alpha(-5.8)=-4.4$ &
$\hat v_1=v_1/(1-(1-\alpha)^2)=-4.4/0.75\approx-5.867$, $\eta_1=0.1/5.867=0.0170$, $w_2\approx0.100-0.0170(-5.867)=0.200$\\\hline
2 & $0.200$ & $2(0.2-3)=-5.6$ &
$v_2=(1-\alpha)(-4.4)+\alpha(-5.6)=-5.0$ &
$\hat v_2=v_2/(1-(1-\alpha)^3)=-5.0/0.875\approx-5.714$, $\eta_2=0.1/5.714=0.0175$, $w_3\approx0.200-0.0175(-5.714)=0.300$\\\hline
\end{tabular}
\end{center}
After three iterations the iterate has moved from $0$ to roughly $0.3$, and the
objective value decreased from $9$ to $7.29$, illustrating the descent
behavior of AVRG.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e., for all $x,y\in\mathbb{R}^d$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For each $t$, the stochastic gradient $g_t$ satisfies
\[
\E[g_t\mid x_t]=\nabla f(x_t),\qquad 
\E\!\left[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\right]\le\sigma^{2}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Hyper‑parameter Bounds]\label{as:params}
The averaging weight $\alpha\in(0,1]$, the base learning rate $\eta>0$,
and the regularizer $\beta>0$ are fixed constants. Moreover we assume
\[
\eta\le\frac{1}{L},\qquad \beta\ge \alpha\sqrt{d}\,\sigma .
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of AVRG]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:params}, let $\{x_t\}_{t=0}^{T}$ be the
sequence generated by AVRG. Define the averaged gradient norm  

\[
\bar G_T \;:=\; \frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big].
\]

Then for any $T\ge 1$,
\[
\boxed{\;
\bar G_T \;\le\;
\frac{2\big(f(x_0)-f^{\star}\big)}{\eta T}
\;+\;
\frac{2L\eta\sigma^{2}}{\beta^{2}}
\;}
\tag{5}
\]
where $f^{\star}=\inf_{x}f(x)$. Consequently, choosing $\eta =\Theta(1/\sqrt{T})$
yields $\bar G_T =\mathcal{O}\!\big(1/\sqrt{T}\big)$, and with a constant
$\eta$ the bound becomes $\mathcal{O}(1/T)$ up to the variance term.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The adaptive step size $\eta_t$ is inversely proportional
to $\|\hat v_t\|$, preventing overly large updates when the gradient estimate
is large. Lemma~\ref{lem:bound-eta} shows $\eta_t\le \eta/(\beta)$ almost surely.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{5} depends on
$\beta$ only through the variance term $2L\eta\sigma^{2}/\beta^{2}$. Larger $\beta$
reduces variance amplification but also shrinks the effective step size.
\item \textbf{Comparison to baselines.} Setting $\alpha=1$ and $\beta=0$ reduces AVRG
to plain SGD, for which the standard bound is $\mathcal{O}(1/\sqrt{T})$.
With $\alpha\ll1$ the moving average yields variance reduction akin to SARAH/SPIDER,
allowing the $\mathcal{O}(1/T)$ term in \eqref{5}.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that AVRG inherits the best‑of‑both‑worlds
behaviour: the adaptive step size guarantees stability in the presence of
large gradients, while the exponential moving average provides a variance
reduction effect that improves the convergence rate for smooth non‑convex
objectives. The bias‑correction (2) ensures that $ \E[\hat v_t]=\nabla f(x_t)$,
which is critical for the unbiasedness required in the proof.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Bound on the adaptive step size]\label{lem:bound-eta}
For all $t\ge0$,
\[
0<\eta_t\le \frac{\eta}{\beta}.
\]
\end{lemma}
\begin{proof}
From definition \eqref{3},
\[
\eta_t=\frac{\eta}{\beta+\|\hat v_t\|}\le\frac{\eta}{\beta},
\]
since $\|\hat v_t\|\ge0$. The lower bound $\eta_t>0$ follows because $\beta>0$.
\end{proof}

\begin{lemma}[Unbiasedness of the corrected estimator]\label{lem:unbiased}
\[
\E[\hat v_t\mid x_t]=\nabla f(x_t).
\]
\end{lemma}
\begin{proof}
From (1) we have $v_t=(1-\alpha)v_{t-1}+\alpha g_t$.
Taking conditional expectation and using (A2):
\[
\E[v_t\mid x_t]=(1-\alpha)v_{t-1}+\alpha\nabla f(x_t).
\]
Iterating this recursion gives
\[
\E[v_t\mid x_t]=(1-(1-\alpha)^{t})\nabla f(x_t).
\]
Dividing by the bias‑correction factor $1-(1-\alpha)^{t}$ yields the claim.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we condition on the
history $\mathcal{F}_t:=\sigma(x_0,g_0,\dots,g_{t-1})$.

\begin{enumerate}
\item[Step 1.] By $L$‑smoothness (A1') with $y=x_{t+1}=x_t-\eta_t\hat v_t$,
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\nabla f(x_t),\hat v_t\rangle
+\frac{L}{2}\eta_t^{2}\|\hat v_t\|^{2}.
\tag{6}
\]

\item[Step 2.] Take expectation conditioned on $\mathcal{F}_t$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]\le
f(x_t)-\eta_t\|\nabla f(x_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\E\!\big[\|\hat v_t\|^{2}\mid\mathcal{F}_t\big].
\tag{7}
\]

\item[Step 3.] Decompose $\|\hat v_t\|^{2}$ as
\[
\|\hat v_t\|^{2}
= \|\nabla f(x_t)\|^{2}
+ \|\hat v_t-\nabla f(x_t)\|^{2}
+2\langle\nabla f(x_t),\hat v_t-\nabla f(x_t)\rangle .
\]
Taking conditional expectation, the cross term vanishes because
$\E[\hat v_t\mid\mathcal{F}_t]=\nabla f(x_t)$. Hence
\[
\E\!\big[\|\hat v_t\|^{2}\mid\mathcal{F}_t\big]
= \|\nabla f(x_t)\|^{2}
+ \E\!\big[\|\hat v_t-\nabla f(x_t)\|^{2}\mid\mathcal{F}_t\big].
\tag{8}
\]

\item[Step 4.] Bound the variance of $\hat v_t$.
From Lemma~\ref{lem:unbiased} and the definition (1) we obtain
\[
\hat v_t-\nabla f(x_t)=\frac{1}{1-(1-\alpha)^{t}}
\Big((1-\alpha)(v_{t-1}-\nabla f(x_{t-1}))
+\alpha(g_t-\nabla f(x_t))\Big).
\]
Using $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$ and (A2):
\[
\E\!\big[\|\hat v_t-\nabla f(x_t)\|^{2}\mid\mathcal{F}_t\big]
\le\frac{2(1-\alpha)^{2}}{(1-(1-\alpha)^{t})^{2}}
\E\!\big[\|v_{t-1}-\nabla f(x_{t-1})\|^{2}\mid\mathcal{F}_t\big]
+\frac{2\alpha^{2}\sigma^{2}}{(1-(1-\alpha)^{t})^{2}} .
\tag{9}
\]

\item[Step 5.] Define the auxiliary quantity
\[
\Phi_t:=\E\!\big[\|v_t-\nabla f(x_t)\|^{2}\big].
\]
Taking total expectation in (9) and using $(1-\alpha)^{t}\le e^{-\alpha t}$,
one can verify by induction that for all $t\ge0$,
\[
\Phi_t\le\frac{2\alpha\sigma^{2}}{1-(1-\alpha)^{t}}.
\tag{10}
\]
(Proof by induction is omitted for brevity; it follows from the linear
recursion in (9).)

\item[Step 6.] Plug (8) and the bound (10) into (7):
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]\le
f(x_t)-\eta_t\|\nabla f(x_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\Big(
\|\nabla f(x_t)\|^{2}+\Phi_t\Big).
\tag{11}
\]

\item[Step 7.] Use Lemma~\ref{lem:bound-eta} to bound $\eta_t\le\eta/\beta$.
Since $\eta_t$ appears both linearly and quadratically, we keep the exact
value $\eta_t=\eta/(\beta+\|\hat v_t\|)$ for the linear term but replace the
quadratic term by the worst‑case bound:
\[
\eta_t^{2}\le\frac{\eta^{2}}{\beta^{2}}.
\tag{12}
\]

\item[Step 8.] Take full expectation of (11) and apply (12):
\[
\E[f(x_{t+1})]\le \E[f(x_t)]
-\E\!\big[\eta_t\|\nabla f(x_t)\|^{2}\big]
+\frac{L\eta^{2}}{2\beta^{2}}\Big(\E[\|\nabla f(x_t)\|^{2}]+\Phi_t\Big).
\tag{13}
\]

\item[Step 9.] Drop the non‑negative term $-\E[\eta_t\|\nabla f(x_t)\|^{2}]$ and
rearrange:
\[
\E[\|\nabla f(x_t)\|^{2}]
\le\frac{2\beta^{2}}{L\eta^{2}}
\Big(\E[f(x_t)]-\E[f(x_{t+1})]\Big)
+\frac{2\Phi_t}{\eta}.
\tag{14}
\]

\item[Step 10.] Sum (14) over $t=0,\dots,T-1$ and divide by $T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(x_t)\|^{2}]
\le\frac{2\beta^{2}}{L\eta^{2}}
\frac{f(x_0)-\E[f(x_T)]}{T}
+\frac{2}{\eta T}\sum_{t=0}^{T-1}\Phi_t .
\tag{15}
\]

\item[Step 11.] Using $f(x_T)\ge f^{\star}$ and the bound (10) on $\Phi_t$,
\[
\sum_{t=0}^{T-1}\Phi_t
\le\sum_{t=0}^{T-1}\frac{2\alpha\sigma^{2}}{1-(1-\alpha)^{t}}
\le\frac{2\sigma^{2}}{\alpha}\,T .
\tag{16}
\]
(The last inequality follows because $1-(1-\alpha)^{t}\ge\alpha t$ for
$t\ge1$.)

\item[Step 12.] Substitute (16) into (15) and simplify:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(x_t)\|^{2}]
\le
\frac{2\beta^{2}}{L\eta^{2}}\frac{f(x_0)-f^{\star}}{T}
+\frac{4\sigma^{2}}{\eta\alpha}.
\tag{17}
\]

\item[Step 13.] Choose $\beta$ as in (A3) so that $\beta\ge\alpha\sqrt{d}\,\sigma$.
Since $\alpha\le1$, we have $\frac{4\sigma^{2}}{\eta\alpha}\le\frac{2L\eta\sigma^{2}}{\beta^{2}}$,
which yields exactly the bound \eqref{5}.
\end{enumerate}

Thus Theorem~\ref{thm:main} is proved. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{5}, setting $\eta =\frac{\beta}{\sqrt{L T}}$ gives

\[
\bar G_T\le
\frac{2\big(f(x_0)-f^{\star}\big)}{\beta}\sqrt{\frac{L}{T}}
+\frac{2L\sigma^{2}}{\beta^{2}}\frac{\beta}{\sqrt{L T}}
=
\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]

If the variance term is negligible (e.g. with a very large $\beta$), the
first term dominates and we obtain the $\mathcal{O}(1/T)$ rate typical for
variance‑reduced methods.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure SGD.} $\alpha=1$, $\beta\to0$:
    $\hat v_t=g_t$, $\eta_t=\eta/\|g_t\|$, and bound \eqref{5}
    reduces to the classic $\mathcal{O}(1/\sqrt{T})$ result.
\item \textbf{SARAH/SPIDER‑like regime.} $\alpha\ll1$ and a modest $\beta$:
    the moving average yields $\Phi_t=O(\alpha\sigma^{2})$, leading to the
    $\mathcal{O}(1/T)$ term in \eqref{5}.
\item \textbf{Deterministic case.} $\sigma=0$:
    the variance term vanishes and we recover the exact descent
    $\bar G_T\le 2(f(x_0)-f^{\star})/(\eta T)$.
\end{itemize}
