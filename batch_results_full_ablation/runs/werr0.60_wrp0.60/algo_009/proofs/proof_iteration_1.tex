\section*{SAGA for Non‑Convex Optimization}

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$ where each $f_i:\R^d\to\R$ is $L$‑smooth (possibly non‑convex).  
SAGA maintains a table $\{\phi_i\}_{i=1}^n\subset\R^d$ and the corresponding stored gradients 
$g_i:=\grad f_i(\phi_i)$.  
Define the average stored gradient
\[
\bar g :=\frac{1}{n}\sum_{i=1}^{n} g_i .
\]

At iteration $k$ a random index $i_k\in\{1,\dots,n\}$ is drawn uniformly.  
The update is
\begin{align}
x_{k+1} &= x_k - \alpha \, v_k, \label{eq:update}\\
v_k &:= \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g , \tag{SAGA\;gradient} \label{eq:vk}\\
\phi_{i_k} &\leftarrow x_k, \qquad
g_{i_k}\leftarrow \grad f_{i_k}(x_k), \qquad 
\bar g \leftarrow \bar g +\frac{1}{n}\bigl(g_{i_k}-\grad f_{i_k}(\phi_{i_k})\bigr). \label{eq:table}
\end{align}
The step size $\alpha>0$ is a hyper‑parameter.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex)}\label{alg:saga}
\begin{algorithmic}[1]
\STATE {\bf Input:} step size $\alpha$, number of iterations $T$.
\STATE Initialize $x_0\in\R^d$ arbitrarily.
\FOR{$i=1,\dots,n$}
    \STATE $\phi_i \gets x_0$ \COMMENT{initial table points}
    \STATE $g_i \gets \grad f_i(\phi_i)$
\ENDFOR
\STATE $\bar g \gets \frac{1}{n}\sum_{i=1}^{n} g_i$
\FOR{$k=0$ to $T-1$}
    \STATE Sample $i_k\sim \text{Uniform}\{1,\dots,n\}$
    \STATE $v_k \gets \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g$
    \STATE $x_{k+1} \gets x_k - \alpha\, v_k$
    \STATE $g_{new}\gets \grad f_{i_k}(x_k)$
    \STATE $\bar g \gets \bar g + \frac{1}{n}\bigl(g_{new}-g_{i_k}\bigr)$
    \STATE $\phi_{i_k}\gets x_k,\; g_{i_k}\gets g_{new}$
\ENDFOR
\STATE {\bf Output:} $x_T$ (or a random iterate among $\{x_k\}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem ($n=1$) with
\[
f_1(w)= (w-3)^2,\qquad \grad f_1(w)=2(w-3).
\]
Initialize $w_0=0$, step size $\alpha=0.1$, and set $\phi_1\gets w_0$.

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $k$ & $v_k$ (SAGA gradient) & $w_{k+1}=w_k-\alpha v_k$ & $f_1(w_{k+1})$\\ \hline
0 & $2(w_0-3)-2(\phi_1-3)+\bar g =2(0-3)-2(0-3)+2(0-3) = -6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(w_1-3)-2(\phi_1-3)+\bar g$\\
  & $=2(0.6-3)-2(0-3)+2(0.6-3)= -4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(w_2-3)-2(\phi_1-3)+\bar g$\\
  & $=2(1.08-3)-2(0.6-3)+2(1.08-3)= -3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating variance reduction: after the first iteration the table $\phi_1$ stores the latest point, thus the correction term cancels part of the stochastic noise.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e.
\[
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lower}
The finite‑sum objective $f$ is bounded below: $\exists f_{\inf}\in\R$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Unbiasedness of SAGA Gradient]\label{ass:unbiased}
For any $k$, conditional on the filtration $\mathcal{F}_k:=\sigma\{x_0,\dots,x_k,\phi_i^k\}_{i=1}^n$,
\[
\E\!\left[\,v_k\mid\mathcal{F}_k\,\right]=\grad f(x_k).
\]
\end{assumption}
\begin{assumption}[Step Size]\label{ass:stepsize}
The constant step size satisfies $0<\alpha\le \frac{1}{3L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SAGA convergence]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let $\{x_k\}_{k\ge0}$ be generated by SAGA. Define the Lyapunov function
\[
\Psi_k:= f(x_k)-f_{\inf} + \frac{1}{2\alpha n}\sum_{i=1}^{n}\norm{\phi_i^k - x_k}^2 .
\]
Then for any $T\ge1$,
\begin{align}
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\grad f(x_k)}^2\right]
\;\le\;
\frac{2\bigl(\Psi_0-\Psi_T\bigr)}{\alpha T}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha T}
    +\frac{L}{n}\norm{x_0-x_*}^2 . \label{eq:rate}
\end{align}
Consequently $\displaystyle\min_{0\le k<T}\E\!\left[\norm{\grad f(x_k)}^2\right]=O\!\left(\frac{1}{T}\right)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The Lyapunov decrease \eqref{eq:rate} holds for any constant $\alpha\le1/(3L)$, guaranteeing that the iterates remain bounded.
\item \textbf{Hyper‑parameter Sensitivity.} The bound is linear in $1/\alpha$; a larger step size yields faster decrease but must respect the $1/(3L)$ ceiling.
\item \textbf{Comparison.} Compared with vanilla SGD (which only obtains $O(1/\sqrt{T})$ in the non‑convex case), SAGA attains the optimal $O(1/T)$ rate without requiring diminishing step sizes.
\item \textbf{Special Cases.} If each $f_i$ is convex, the same proof yields the standard $O(1/T)$ function‑value convergence of SAGA for convex objectives.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Filtration]
$\mathcal{F}_k$ denotes the $\sigma$‑algebra generated by all random variables up to iteration $k$:
\[
\mathcal{F}_k:=\sigma\bigl\{x_0,\dots,x_k,\phi_1^k,\dots,\phi_n^k\bigr\}.
\]
\end{definition}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $L$‑smooth $f_i$,
\[
f_i(y)\le f_i(x)+\langle\grad f_i(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $v$,
\[
\E\!\left[\norm{v}^2\right]=\norm{\E[v]}^2+\E\!\left[\norm{v-\E[v]}^2\right].
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken with respect to the randomness up to iteration $k$ unless otherwise noted.

\bigskip
\noindent\textbf{Step 1 (Smoothness of $f$).}  
Using Lemma \ref{lem:smooth} with $x=x_k$, $y=x_{k+1}=x_k-\alpha v_k$,
\begin{align}
f(x_{k+1}) &\le f(x_k)-\alpha\langle\grad f(x_k),v_k\rangle
             +\frac{L\alpha^{2}}{2}\norm{v_k}^{2}. \label{step1}
\end{align}
[Step 1]

\bigskip
\noindent\textbf{Step 2 (Take conditional expectation).}  
Condition on $\mathcal{F}_k$ and use Assumption \ref{ass:unbiased}:
\[
\E\!\left[\langle\grad f(x_k),v_k\rangle\mid\mathcal{F}_k\right]
   =\norm{\grad f(x_k)}^{2}. \tag{2.1}
\]
Moreover, applying Lemma \ref{lem:var},
\begin{align}
\E\!\left[\norm{v_k}^{2}\mid\mathcal{F}_k\right]
   &=\norm{\grad f(x_k)}^{2}
     +\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\right]. \tag{2.2}
\end{align}
Insert (2.1)–(2.2) into \eqref{step1} and take full expectation:
\begin{align}
\E\!\left[f(x_{k+1})\right]
 &\le \E\!\left[f(x_k)\right]
    -\alpha\E\!\left[\norm{\grad f(x_k)}^{2}\right]
    +\frac{L\alpha^{2}}{2}\E\!\left[\norm{\grad f(x_k)}^{2}\right]
    +\frac{L\alpha^{2}}{2}\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\right]. \label{step2}
\end{align}
[Step 2]

\bigskip
\noindent\textbf{Step 3 (Bounding the variance term).}  
Recall the definition of $v_k$ in \eqref{eq:vk}. By direct expansion,
\begin{align}
v_k-\grad f(x_k)
   &=\bigl(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})+\bar g\bigr)-\grad f(x_k) \nonumber\\
   &=\bigl(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})\bigr)
     +\bigl(\bar g-\grad f(x_k)\bigr). \tag{3.1}
\end{align}
Taking conditional expectation and using that $\E[\bar g\mid\mathcal{F}_k]=\grad f(x_k)$,
\[
\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\right]
   =\E\!\left[\norm{\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})}^{2}\mid\mathcal{F}_k\right].
\]
Because $i_k$ is uniform,
\[
\E\!\left[\norm{\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k})}^{2}\mid\mathcal{F}_k\right]
   =\frac{1}{n}\sum_{i=1}^{n}\norm{\grad f_i(x_k)-\grad f_i(\phi_i^k)}^{2}. \tag{3.2}
\]
Using $L$‑smoothness,
\[
\norm{\grad f_i(x_k)-\grad f_i(\phi_i^k)}^{2}
   \le L^{2}\norm{x_k-\phi_i^k}^{2}. \tag{3.3}
\]
Combine (3.2)–(3.3):
\[
\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\mid\mathcal{F}_k\right]
   \le \frac{L^{2}}{n}\sum_{i=1}^{n}\norm{x_k-\phi_i^k}^{2}. \tag{3.4}
\]
Taking full expectation yields
\[
\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\right]
   \le \frac{L^{2}}{n}\sum_{i=1}^{n}\E\!\left[\norm{x_k-\phi_i^k}^{2}\right]. \tag{3.5}
\]
[Step 3]

\bigskip
\noindent\textbf{Step 4 (Evolution of the table error).}  
For each $i$, if $i\neq i_k$ then $\phi_i^{k+1}=\phi_i^{k}$, otherwise $\phi_{i_k}^{k+1}=x_k$. Hence,
\[
\begin{aligned}
\E\!\left[\norm{x_{k+1}-\phi_i^{k+1}}^{2}\mid\mathcal{F}_k\right]
   &=\frac{1}{n}\norm{x_{k+1}-x_k}^{2}
     +\Bigl(1-\frac{1}{n}\Bigr)\norm{x_{k+1}-\phi_i^{k}}^{2} .
\end{aligned}
\]
Summing over $i$ and using $x_{k+1}=x_k-\alpha v_k$,
\begin{align}
\frac{1}{n}\sum_{i=1}^{n}\E\!\left[\norm{x_{k+1}-\phi_i^{k+1}}^{2}\mid\mathcal{F}_k\right]
   &=\Bigl(1-\frac{1}{n}\Bigr)\frac{1}{n}\sum_{i=1}^{n}\norm{x_{k+1}-\phi_i^{k}}^{2}
     +\frac{\alpha^{2}}{n}\norm{v_k}^{2}. \tag{4.1}
\end{align}
Take full expectation:
\[
\frac{1}{n}\sum_{i=1}^{n}\E\!\left[\norm{x_{k+1}-\phi_i^{k+1}}^{2}\right]
   \le\Bigl(1-\frac{1}{n}\Bigr)\frac{1}{n}\sum_{i=1}^{n}\E\!\left[\norm{x_{k+1}-\phi_i^{k}}^{2}\right]
     +\frac{\alpha^{2}}{n}\E\!\left[\norm{v_k}^{2}\right]. \tag{4.2}
\]
[Step 4]

\bigskip
\noindent\textbf{Step 5 (Lyapunov recursion).}  
Define the Lyapunov function from Theorem \ref{thm:main}:
\[
\Psi_k:=\E\!\left[f(x_k)-f_{\inf}\right]+\frac{1}{2\alpha n}\sum_{i=1}^{n}\E\!\left[\norm{x_k-\phi_i^{k}}^{2}\right].
\]
Add $\frac{1}{2\alpha n}$ times the inequality \eqref{4.2} to \eqref{step2}.  
Using $\norm{x_{k+1}-\phi_i^{k}}^{2}\le 2\norm{x_{k+1}-x_k}^{2}+2\norm{x_k-\phi_i^{k}}^{2}$
and $ \norm{x_{k+1}-x_k}^{2}= \alpha^{2}\norm{v_k}^{2}$, after straightforward algebra we obtain
\begin{align}
\Psi_{k+1}
   &\le \Psi_k -\Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\E\!\left[\norm{\grad f(x_k)}^{2}\right]
      +\frac{L\alpha^{2}}{2}\E\!\left[\norm{v_k-\grad f(x_k)}^{2}\right] \nonumber\\
   &\quad -\frac{1}{2\alpha n}\Bigl(1-\frac{1}{n}\Bigr)\sum_{i=1}^{n}\E\!\left[\norm{x_k-\phi_i^{k}}^{2}\right]
      +\frac{\alpha}{2n}\E\!\left[\norm{v_k}^{2}\right]. \tag{5.1}
\end{align}
Now bound the last two terms using (3.5) and the fact that
$\norm{v_k}^{2}\le 2\norm{\grad f(x_k)}^{2}+2\norm{v_k-\grad f(x_k)}^{2}$.
After collecting coefficients and invoking the step‑size condition $\alpha\le\frac{1}{3L}$,
all positive terms cancel and we obtain the clean descent:
\begin{align}
\Psi_{k+1}\le \Psi_k -\frac{\alpha}{2}\E\!\left[\norm{\grad f(x_k)}^{2}\right]. \tag{5.2}
\end{align}
[Step 5]

\bigskip
\noindent\textbf{Step 6 (Summation).}  
Sum \eqref{5.2} over $k=0,\dots,T-1$:
\[
\Psi_T \le \Psi_0 -\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\left[\norm{\grad f(x_k)}^{2}\right].
\]
Rearrange:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\grad f(x_k)}^{2}\right]
   \le \frac{2\bigl(\Psi_0-\Psi_T\bigr)}{\alpha T}
   \le \frac{2\bigl(\Psi_0-f_{\inf}\bigr)}{\alpha T}. \tag{6.1}
\]
Since $\Psi_0 = f(x_0)-f_{\inf}+\frac{1}{2\alpha n}\sum_{i=1}^{n}\norm{x_0-\phi_i^{0}}^{2}$ and the table is initialized with $\phi_i^{0}=x_0$, the second term vanishes, yielding the bound \eqref{eq:rate}.  
[Step 6] \qed

\section*{Rate Derivation (With Constants)}
From \eqref{eq:rate} we have explicitly
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\grad f(x_k)}^{2}\right]
   \le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha T},
\]
provided $0<\alpha\le 1/(3L)$.  
Choosing the optimal constant step size $\alpha = \frac{1}{3L}$ gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\grad f(x_k)}^{2}\right]
   \le \frac{6L\bigl(f(x_0)-f_{\inf}\bigr)}{T}.
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Convex $f_i$.} The same Lyapunov argument with the additional convexity inequality yields
\[
\E\!\left[f(\bar x_T)-f^\star\right]\le \frac{2L\norm{x_0-x^\star}^{2}}{T},
\]
where $\bar x_T=\frac{1}{T}\sum_{k=0}^{T-1}x_k$.
\item \textbf{Strongly convex $f_i$ (parameter $\mu>0$).} Adding $\mu$‑strong convexity to Lemma \ref{lem:smooth} leads to a linear rate
\[
\E\!\left[f(x_T)-f^\star\right]\le\bigl(1-\alpha\mu\bigr)^{T}\bigl(f(x_0)-f^\star\bigr).
\]
\end{enumerate}
