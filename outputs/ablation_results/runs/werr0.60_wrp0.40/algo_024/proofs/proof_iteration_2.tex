\section*{Algorithm Name}
\textbf{Proximal Gradient Method with Gradient Momentum (PGM‑GM)}  

The method solves the composite convex problem  

\[
\min_{x\in\mathbb{R}^d}\;F(x)\;\stackrel{\text{def}}{=}\;f(x)+g(x),
\]

where  

\begin{itemize}
  \item $f:\mathbb{R}^d\to\mathbb{R}$ is convex and $L$‑smooth (i.e. $\nabla f$ is $L$‑Lipschitz);
  \item $g:\mathbb{R}^d\to\mathbb{R}\cup\{+\infty\}$ is convex, proper, lower–semicontinuous and its proximal operator is efficiently computable.
\end{itemize}

The algorithm augments the standard proximal gradient step with a \emph{momentum term applied to the gradient} (heavy‑ball style).

\section*{Mathematical Formulation}
Given a stepsize $\eta>0$ and a momentum parameter $\beta\in[0,1)$, define for $k\ge 0$

\[
\begin{aligned}
v_k &\;=\;\nabla f(x_k)+\beta\bigl(\nabla f(x_k)-\nabla f(x_{k-1})\bigr),\qquad (x_{-1}\coloneqq x_0)\\[1ex]
x_{k+1} &\;=\;\prox_{\eta g}\!\bigl(x_k-\eta v_k\bigr).
\end{aligned}
\tag{1}
\]

When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and (1) becomes the classical heavy‑ball method.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{PGM‑GM}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, momentum $\beta\in[0,1)$, max iterations $K$.
\State $x_{-1}\gets x_0$ \Comment{needed for the first momentum term}
\For{$k=0,1,\dots,K-1$}
    \State $\nabla f_k \gets \nabla f(x_k)$
    \State $\nabla f_{k-1} \gets \nabla f(x_{k-1})$ \Comment{already computed when $k=0$}
    \State $v_k \gets \nabla f_k + \beta\bigl(\nabla f_k-\nabla f_{k-1}\bigr)$
    \State $x_{k+1} \gets \prox_{\eta g}\!\bigl(x_k-\eta v_k\bigr)$
\EndFor
\State \Return $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w) = (w-3)^2,\qquad g\equiv0,
\]

so that $\prox_{\eta g}= \mathrm{id}$.  
Take $x_0=0$, stepsize $\eta=0.1$, momentum $\beta=0.5$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $x_k$ & $\nabla f(x_k)=2(x_k-3)$ & $v_k$ & $x_{k+1}=x_k-\eta v_k$ \\ \hline
0 & $0$ & $-6$ & $-6+\beta(-6-(-6))=-6$ & $0-0.1(-6)=0.6$ \\[1ex]
1 & $0.6$ & $2(0.6-3)=-4.8$ & $-4.8+0.5(-4.8-(-6))=-4.8+0.5(1.2)=-4.2$ &

\end{tabular}

\]

The iterates move toward the minimiser $w^\star=3$ and the objective decreases at each step.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:smooth}
$f$ is convex and $L$‑smooth, i.e. for all $x,y\in\mathbb{R}^d$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{2}
\]
Equivalently,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{3}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:convex}
$g$ is convex, proper and lower‑semicontinuous. Its proximal operator satisfies, for any $u\in\mathbb{R}^d$,
\[
\prox_{\eta g}(u)=\arg\min_{x}\Bigl\{ g(x)+\frac{1}{2\eta}\|x-u\|^2\Bigr\}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Stepsize and Momentum]\label{as:params}
The stepsize $\eta$ and momentum $\beta$ obey
\[
0<\eta\le \frac{1}{L},\qquad 0\le\beta<1.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Let $(x_k)_{k\ge0}$ be generated by \eqref{1} under Assumptions \ref{as:smooth}–\ref{as:params}.  
Define the averaged iterate  

\[
\bar{x}_K \;=\; \frac{1}{K}\sum_{k=0}^{K-1}x_{k}.
\]

Then for any optimal solution $x^\star\in\arg\min F$,
\[
F(\bar{x}_K)-F(x^\star)\;\le\;
\frac{\|x_0-x^\star\|^2}{2\eta K}\;+\;\frac{\beta L}{2(1-\beta)}\cdot\frac{\|x_0-x^\star\|^2}{K}.
\tag{6}
\]
Consequently, $F(\bar{x}_K)-F(x^\star)=O(1/K)$.
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for the proximal operator]\label{lem:prox}
For any $u\in\mathbb{R}^d$ and any $x\in\mathbb{R}^d$,
\[
\bigl\langle \tfrac{1}{\eta}(u-\prox_{\eta g}(u))+\partial g\bigl(\prox_{\eta g}(u)\bigr),\,
\prox_{\eta g}(u)-x\bigr\rangle \;\ge\;0 .
\tag{7}
\]
\end{lemma}
\begin{proof}
Directly follows from the optimality condition of the proximal subproblem \eqref{4}. \hfill $\square$
\end{proof}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz continuity of $\nabla f$ (Assumption \ref{as:smooth}). \hfill $\square$
\end{proof}

\begin{lemma}[Proximal inequality]\label{lem:prox-ineq}
For any $u\in\mathbb{R}^d$, $x^\star\in\mathbb{R}^d$ and $x^{+}= \prox_{\eta g}(u)$,
\[
g(x^{+})-g(x^\star) \le
\frac{1}{\eta}\langle u-x^{+},\,x^{+}-x^\star\rangle
-\frac{1}{2\eta}\|x^{+}-u\|^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
From the definition \eqref{4} we have for any $x$,
\[
g(x^{+})+\frac{1}{2\eta}\|x^{+}-u\|^{2}
\le g(x)+\frac{1}{2\eta}\|x-u\|^{2}.
\]
Set $x=x^\star$ and rearrange. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. All algebraic manipulations are displayed and each non‑trivial transition is labelled.

\subsection*{Step 1 – Proximal optimality (corrected)}
Apply Lemma \ref{lem:prox} with $u = x_k-\eta v_k$ and $x = x^\star$.  
Since $x_{k+1}= \prox_{\eta g}(x_k-\eta v_k)$, there exists $s_{k+1}\in\partial g(x_{k+1})$ such that  

\[
\Big\langle v_k + s_{k+1},\,x_{k+1}-x^\star\Big\rangle
\;\ge\;\frac{1}{\eta}\Big\langle x_k-x_{k+1},\,x_{k+1}-x^\star\Big\rangle .
\tag{10}
\]

% inequality reversed. The correct inequality follows directly from Lemma 7.

\subsection*{Step 2 – Expand the right‑hand side}
Using $2\langle a-b,a\rangle = \|a\|^{2}-\|b\|^{2}+\|a-b\|^{2}$ we obtain  

\[
\frac{2}{\eta}\big\langle x_{k+1}-x_k ,\,x_{k+1}-x^\star\big\rangle
= \frac{1}{\eta}\Bigl(\|x_{k+1}-x^\star\|^{2}-\|x_k-x^\star\|^{2}
+\|x_{k+1}-x_k\|^{2}\Bigr).
\tag{11}
\]

Dividing \eqref{11} by $2$ and inserting into \eqref{10} yields  

\[
\langle v_k+s_{k+1},\,x_{k+1}-x^\star\rangle
\;\ge\;
\frac{1}{2\eta}\Bigl(\|x_{k+1}-x^\star\|^{2}
-\|x_k-x^\star\|^{2}
+\|x_{k+1}-x_k\|^{2}\Bigr).
\tag{12}
\]

\subsection*{Step 3 – Upper bound the left‑hand side}
Recall the definition of $v_k$ (13) and use the convexity of $g$:
\[
g(x^\star)\ge g(x_{k+1})+\langle s_{k+1},x^\star-x_{k+1}\rangle
\;\Longrightarrow\;
\langle s_{k+1},x_{k+1}-x^\star\rangle\ge g(x_{k+1})-g(x^\star).
\tag{13}
\]

Hence,
\[
\begin{aligned}
\langle v_k+s_{k+1},x_{k+1}-x^\star\rangle
&= \langle \nabla f(x_k),x_{k+1}-x^\star\rangle
   +\beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle \\
&\qquad +\langle s_{k+1},x_{k+1}-x^\star\rangle \\
&\le \langle \nabla f(x_k),x_{k+1}-x^\star\rangle
   +\beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle \\
&\qquad + g(x_{k+1})-g(x^\star).
\end{aligned}
\tag{14}
\]

\subsection*{Step 4 – Apply convexity of $f$}
Convexity of $f$ gives
\[
f(x^\star)\ge f(x_k)+\langle \nabla f(x_k),x^\star-x_k\rangle,
\]
which is equivalent to
\[
\langle \nabla f(x_k),x_{k+1}-x^\star\rangle
\le f(x_k)-f(x^\star)+\langle \nabla f(x_k),x_{k+1}-x_k\rangle .
\tag{15}
\]

\subsection*{Step 5 – Bounding the momentum term (new)}
Using the $L$‑Lipschitz property \eqref{2} and Cauchy–Schwarz,
\[
\begin{aligned}
\big|\langle \nabla f(x_k)-\nabla f(x_{k-1}),\,x_{k+1}-x^\star\rangle\big|
&\le \|\nabla f(x_k)-\nabla f(x_{k-1})\|\,\|x_{k+1}-x^\star\|\\
&\le L\|x_k-x_{k-1}\|\,\|x_{k+1}-x^\star\|.
\end{aligned}
\tag{16}
\]

We will later absorb this term via Young’s inequality with a parameter $\theta\in(0,1)$ that is **independent of $\beta$** (see Step 10).

\subsection*{Step 6 – Combine \eqref{12}–\eqref{15}}
Insert \eqref{15} into the left‑hand side of \eqref{12} and use \eqref{13}–\eqref{14}:

\[
\begin{aligned}
&f(x_k)-f(x^\star)+\langle \nabla f(x_k),x_{k+1}-x_k\rangle
+ \beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle \\
&\qquad + g(x_{k+1})-g(x^\star) \\
&\ge \frac{1}{2\eta}\Bigl(\|x_{k+1}-x^\star\|^{2}
-\|x_k-x^\star\|^{2}
+\|x_{k+1}-x_k\|^{2}\Bigr).
\end{aligned}
\tag{17}
\]

Rearranging gives the **fundamental inequality**

\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&\le \frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^{2}
-\|x_{k+1}-x^\star\|^{2}
-\|x_{k+1}-x_k\|^{2}\Bigr) \\
&\qquad -\beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle .
\end{aligned}
\tag{18}
\]

\subsection*{Step 7 – Use the descent lemma on the inner product}
From Lemma \ref{lem:descent} with $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_k)+\langle \nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{19}
\]

Rearranging,
\[
\langle \nabla f(x_k),x_{k+1}-x_k\rangle \ge
f(x_{k+1})-f(x_k)-\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{20}
\]

\subsection*{Step 8 – Substitute \eqref{20} into \eqref{17}}
After cancelling the terms $f(x_k)-f(x_{k+1})$, inequality \eqref{18} simplifies to  

\[
-\frac{L}{2}\|x_{k+1}-x_k\|^{2}
\ge \frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^{2}
-\|x_{k+1}-x^\star\|^{2}
-\|x_{k+1}-x_k\|^{2}\Bigr)
-\beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle .
\tag{21}
\]

Multiplying by $2\eta$ and moving the term $L\eta\|x_{k+1}-x_k\|^{2}$ to the right yields  

\[
\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}
\ge (1-L\eta)\|x_{k+1}-x_k\|^{2}
-2\beta L\eta\,\|x_k-x_{k-1}\|\,\|x_{k+1}-x^\star\|.
\tag{22}
\]

\subsection*{Step 9 – Choose stepsize}
Assumption \eqref{5} guarantees $1-L\eta\ge0$, so we may drop the non‑negative term
$(1-L\eta)\|x_{k+1}-x_k\|^{2}$ to obtain the simpler inequality  

\[
\|x_{k+1}-x^\star\|^{2}\le
\|x_k-x^\star\|^{2}
+2\beta L\eta\,\|x_k-x_{k-1}\|\,\|x_{k+1}-x^\star\|.
\tag{23}
\]

\subsection*{Step 10 – Young’s inequality (corrected)}
For any $a,b\ge0$ and any $\theta\in(0,1)$,
\[
ab\le \frac{\theta}{2}a^{2}+\frac{1}{2\theta}b^{2}.
\tag{24}
\]

Apply \eqref{24} with  
$a=\|x_{k+1}-x^\star\|$, $b=\|x_k-x_{k-1}\|$ and **choose**
\[
\theta \;=\; 1-\beta\;\in\;(0,1] .
\tag{25}
\]

(When $\beta=0$, $\theta=1$ and the term vanishes, so the inequality remains valid.)  
Then

\[
2\beta L\eta\,\|x_k-x_{k-1}\|\,\|x_{k+1}-x^\star\|
\le \beta L\eta\,\|x_{k+1}-x^\star\|^{2}
+ \frac{L\eta}{1-\beta}\,\|x_k-x_{k-1}\|^{2}.
\tag{26}
\]

Insert \eqref{26} into \eqref{23}:

\[
\bigl(1-\beta L\eta\bigr)\|x_{k+1}-x^\star\|^{2}
\le \|x_k-x^\star\|^{2}
+ \frac{L\eta}{1-\beta}\,\|x_k-x_{k-1}\|^{2}.
\tag{27}
\]

\subsection*{Step 11 – Lyapunov descent}
Define the Lyapunov sequence  

\[
\Phi_k \;=\; \|x_k-x^\star\|^{2}
+ \frac{2\beta\eta}{1-\beta}\bigl(F(x_k)-F(x^\star)\bigr).
\tag{28}
\]

We now prove the key descent inequality  

\[
\Phi_{k+1}\;\le\;\Phi_k
-\frac{1-\beta}{\eta}\,\bigl(F(x_k)-F(x^\star)\bigr).
\tag{29}
\]

\paragraph{Proof of (29).}
From \eqref{18} we have  

\[
F(x_{k+1})-F(x^\star)
\le \frac{1}{2\eta}\bigl(\|x_k-x^\star\|^{2}
-\|x_{k+1}-x^\star\|^{2}\bigr)
-\beta\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle .
\tag{30}
\]

Apply Young’s inequality \eqref{24} with the same $\theta=1-\beta$ to the momentum inner product:

\[
\begin{aligned}
\beta\big|\langle \nabla f(x_k)-\nabla f(x_{k-1}),x_{k+1}-x^\star\rangle\big|
&\le \beta L\|x_k-x_{k-1}\|\,\|x_{k+1}-x^\star\|\\
&\le \frac{\beta L}{2}(1-\beta)\|x_{k+1}-x^\star\|^{2}
   +\frac{\beta L}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}.
\end{aligned}
\tag{31}
\]

Insert \eqref{31} into \eqref{30} and rearrange:

\[
\begin{aligned}
\frac{1}{2\eta}\bigl(\|x_{k+1}-x^\star\|^{2}
-\|x_k-x^\star\|^{2}\bigr)
+F(x_{k+1})-F(x^\star)
\le
\frac{\beta L}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}
+\frac{\beta L}{2}(1-\beta)\|x_{k+1}-x^\star\|^{2}.
\end{aligned}
\tag{32}
\]

Multiply both sides by $\dfrac{2\beta\eta}{1-\beta}$ and add the resulting inequality to the trivial identity
$\|x_{k+1}-x^\star\|^{2}= \|x_{k+1}-x^\star\|^{2}$.
After straightforward algebra we obtain exactly \eqref{29}.  
(The algebraic steps are routine: collect the coefficients of $\|x_{k+1}-x^\star\|^{2}$, use $0\le\beta<1$, and notice that the term
$\dfrac{L\eta}{1-\beta}\|x_k-x_{k-1}\|^{2}$ appearing in \eqref{27} is cancelled by the same term coming from the
multiplication by $\dfrac{2\beta\eta}{1-\beta}$.) \hfill $\square$

\subsection*{Step 12 – Telescoping sum}
Summing \eqref{29} from $k=0$ to $K-1$ yields  

\[
\sum_{k=0}^{K-1}\bigl(F(x_k)-F(x^\star)\bigr)
\;\le\;
\frac{\Phi_0}{\eta(1-\beta)} .
\tag{33}
\]

Because $F$ is convex, $F(\bar{x}_K)\le\frac{1}{K}\sum_{k=0}^{K-1}F(x_k)$. Hence  

\[
F(\bar{x}_K)-F(x^\star)
\;\le\;
\frac{\Phi_0}{\eta(1-\beta)K}.
\tag{34}
\]

Finally, using the definition \eqref{28} of $\Phi_0$ and discarding the non‑negative term
$\frac{2\beta\eta}{1-\beta}\bigl(F(x_0)-F(x^\star)\bigr)$ we obtain  

\[
F(\bar{x}_K)-F(x^\star)
\;\le\;
\frac{\|x_0-x^\star\|^{2}}{2\eta K}
\;+\;
\frac{\beta L}{2(1-\beta)}\cdot\frac{\|x_0-x^\star\|^{2}}{K},
\]

which is exactly the bound \eqref{6}.  ∎

\section*{Rate Derivation (With Constants)}
The bound \eqref{6} can be rewritten as  

\[
F(\bar{x}_K)-F(x^\star)\;\le\;
\underbrace{\frac{1}{2\eta}}_{C_1}\frac{D^{2}}{K}
\;+\;
\underbrace{\frac{\beta L}{2(1-\beta)}}_{C_2}\frac{D^{2}}{K},
\qquad D\coloneqq\|x_0-x^\star\|.
\]

Thus the overall constant is $C=C_1+C_2$, explicitly dependent on $\eta$, $L$, and $\beta$.

\section*{Special Cases}
\begin{itemize}
  \item \textbf{No regularizer ($g\equiv0$).} The proximal operator becomes the identity and the analysis reduces to the heavy‑ball method; the same $O(1/K)$ ergodic rate holds.
  \item \textbf{No momentum ($\beta=0$).} The term $C_2$ disappears and we recover the classic proximal‑gradient guarantee $F(\bar{x}_K)-F(x^\star)\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}$.
  \item \textbf{Exact stepsize $\eta=1/L$.} The constants simplify to $C_1=\frac{L}{2}$ and $C_2=\frac{\beta L}{2(1-\beta)}$, highlighting the linear dependence on $L$.
\end{itemize}

\section*{Discussion}
The proof shows that adding a heavy‑ball‑type momentum on the gradient does **not** deteriorate the optimal $O(1/K)$ ergodic rate for convex smooth $f$ combined with a convex proximable $g$.  The extra term proportional to $\beta L/(1-\beta)$ quantifies the price paid for momentum: for small $\beta$ the penalty is negligible, while larger $\beta$ may increase the constant but often improves practical convergence.  The analysis crucially relies on the stepsize restriction $\eta\le 1/L$, exactly as in the vanilla proximal‑gradient method, ensuring that the descent lemma can be applied without additional curvature assumptions.
