\section*{AdaGrad with Running Sum of Gradients}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be a convex objective.  
Denote the (stochastic or full) gradient observed at iteration $k$ by 
\[
g_{k}\;:=\;\nabla f(w_{k})\in\RR^{d}.
\]
Define the element‑wise accumulated squared gradients
\[
G_{k}\;:=\;\sum_{i=1}^{k} g_{i}^{2}\;\in\RR^{d},
\]
where the square and the sum are taken component‑wise.  
For a scalar stepsize hyperparameter $\eta>0$ the AdaGrad update is
\[
\boxed{%
w_{k+1}=w_{k}-\eta \, D_{k}^{-1}\, g_{k}},\qquad 
D_{k}:=\operatorname{diag}\!\bigl(\sqrt{G_{k}}+\epsilon\bigr),
\tag{1}
\]
with a small constant $\epsilon>0$ (often $\epsilon=10^{-8}$) to avoid division by zero.  
When $d=1$ the update simplifies to
\[
w_{k+1}=w_{k}-\frac{\eta}{\sqrt{\sum_{i=1}^{k} g_{i}^{2}}+\epsilon}\;g_{k}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (vector version)}\label{alg:adagrad}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\RR^{d}$, stepsize $\eta>0$, $\epsilon>0$
\State $G_{0}\gets 0\in\RR^{d}$ \Comment{accumulator}
\For{$k=0,1,\dots,T-1$}
    \State $g_{k}\gets\nabla f(w_{k})$ \Comment{gradient}
    \State $G_{k+1}\gets G_{k}+g_{k}^{2}$ \Comment{component‑wise square and sum}
    \State $D_{k+1}\gets\operatorname{diag}\!\bigl(\sqrt{G_{k+1}}+\epsilon\bigr)$
    \State $w_{k+1}\gets w_{k}-\eta\, D_{k+1}^{-1} g_{k}$
\EndFor
\State \textbf{Output:} $w_{T}$ (or the average $\bar w_T:=\tfrac1T\sum_{k=0}^{T-1}w_{k}$)
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Convexity]\label{as:convex}
$f:\RR^{d}\rightarrow\RR$ is convex, i.e.
\[
f(y)\ge f(x)+\inner{\nabla f(x)}{y-x},\qquad\forall x,y\in\RR^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradients]\label{as:bound}
There exists $G_{\infty}>0$ such that $\norm{g_{k}}\le G_{\infty}$ for all $k$.
\end{assumption}
\begin{assumption}[Domain Diameter]\label{as:diam}
The feasible set $\mathcal{W}\subset\RR^{d}$ is convex and has bounded diameter 
\[
D:=\max_{u,v\in\mathcal{W}}\norm{u-v}<\infty .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[AdaGrad Regret Bound]\label{thm:regret}
Under Assumptions \ref{as:convex}--\ref{as:diam}, for any $w^{*}\in\mathcal{W}$ the iterates generated by \eqref{eq:adagrad} satisfy
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\;\le\;
\frac{D^{2}}{2\eta}\sum_{j=1}^{d}\sqrt{G_{T,j}}
\;+\;
\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon},
\tag{3}
\]
where $G_{k,j}$ denotes the $j$‑th coordinate of $G_{k}$. Consequently, using convexity,
\[
f(\bar w_{T})-f(w^{*})\;\le\;
\frac{D^{2}}{2\eta T}\sum_{j=1}^{d}\sqrt{G_{T,j}}
\;+\;
\frac{\eta}{2T}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon},
\tag{4}
\]
where $\bar w_{T}:=\frac1T\sum_{k=0}^{T-1}w_{k}$.
\end{theorem}
When all gradients are bounded by $G_{\infty}$, $\sqrt{G_{T,j}}\le G_{\infty}\sqrt{T}$, and the right‑hand side of (4) becomes $O\!\bigl(\frac{DG_{\infty}\sqrt{d}}{\sqrt{T}}\bigr)$. Thus AdaGrad attains the optimal $O(1/\sqrt{T})$ rate for convex problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any $u,v,w\in\RR^{d}$ and any positive definite diagonal matrix $D$,
\[
\|u-v\|_{D}^{2}
= \|u-w\|_{D}^{2}+\|w-v\|_{D}^{2}
+2\inner{D(u-w)}{w-v},
\]
where $\|x\|_{D}^{2}:=x^{\top}Dx$.
\end{lemma}
\begin{proof}
Expand $\|u-v\|_{D}^{2}= (u-w+w-v)^{\top}D(u-w+w-v)$ and collect terms. \qedhere
\end{proof}

\begin{lemma}[Bound on Adaptive Steps]\label{lem:step}
For any $a\ge0$, $b\ge0$ and $\epsilon>0$,
\[
\frac{a^{2}}{\sqrt{b}+ \epsilon}\;\le\;2\bigl(\sqrt{b+a^{2}}-\sqrt{b}\bigr)+\epsilon .
\tag{5}
\]
\end{lemma}
\begin{proof}
Define $\phi(b)=\sqrt{b+a^{2}}-\sqrt{b}$. By the mean‑value theorem,
\[
\phi(b)=\frac{a^{2}}{2(\sqrt{b+\theta a^{2}})}\le\frac{a^{2}}{2(\sqrt{b}+\epsilon)},
\]
for some $\theta\in(0,1)$. Multiplying by $2$ and adding $\epsilon$ yields \eqref{eq:step}. \qedhere
\end{proof}

\begin{lemma}[Diameter–Weighted Distance]\label{lem:diam}
For any $w\in\mathcal{W}$ and any diagonal $D\succ0$,
\[
\|w-w^{*}\|_{D}^{2}\le D^{2}\,\max_{j}\,D_{jj}.
\]
\end{lemma}
\begin{proof}
Since $\|w-w^{*}\|^{2}\le D^{2}$ for all $w,w^{*}\in\mathcal{W}$ and $D$ is diagonal,
\[
\|w-w^{*}\|_{D}^{2}= \sum_{j} D_{jj}(w_{j}-w^{*}_{j})^{2}
\le \Bigl(\max_{j}D_{jj}\Bigr)\sum_{j}(w_{j}-w^{*}_{j})^{2}
\le D^{2}\,\max_{j}D_{jj}. \qedhere
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:regret}. Let $w^{*}\in\mathcal{W}$ be arbitrary. Define the diagonal matrix
\[
D_{k}:=\operatorname{diag}\!\bigl(\sqrt{G_{k}}+\epsilon\bigr),\qquad
D_{k}^{-1}:=\operatorname{diag}\!\Bigl(\frac{1}{\sqrt{G_{k}}+\epsilon}\Bigr).
\]

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:three} with $u=w_{k+1}$, $v=w^{*}$, $w=w_{k}$ and $D=D_{k+1}$:
\[
\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
= \|w_{k+1}-w_{k}\|_{D_{k+1}}^{2}
+ \|w_{k}-w^{*}\|_{D_{k+1}}^{2}
+2\inner{D_{k+1}(w_{k+1}-w_{k})}{w_{k}-w^{*}}.
\]
Using the update \eqref{eq:adagrad},
$w_{k+1}-w_{k}= -\eta D_{k+1}^{-1}g_{k}$,
and noting $D_{k+1}D_{k+1}^{-1}=I$, we obtain
\[
\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
= \|w_{k}-w^{*}\|_{D_{k+1}}^{2}
+\eta^{2}\|g_{k}\|_{D_{k+1}^{-1}}^{2}
+2\eta\inner{g_{k}}{w_{k}-w^{*}}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Since each diagonal entry of $D_{k+1}$ is at least the corresponding entry of $D_{k}$, we have
\[
\|w_{k}-w^{*}\|_{D_{k}}^{2}\le\|w_{k}-w^{*}\|_{D_{k+1}}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 3]} Combine \eqref{eq:6} and \eqref{eq:7}, then move the cross term to the left:
\[
2\eta\inner{g_{k}}{w_{k}-w^{*}}
\le \|w_{k}-w^{*}\|_{D_{k}}^{2}
-\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
+\eta^{2}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 4]} Sum \eqref{eq:8} over $k=0,\dots,T-1$:
\[
2\eta\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \|w_{0}-w^{*}\|_{D_{0}}^{2}
-\|w_{T}-w^{*}\|_{D_{T}}^{2}
+\eta^{2}\sum_{k=0}^{T-1}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{9}
\]

\noindent\textbf{[Step 5]} Bound the first term using Lemma~\ref{lem:diam}. Since $D_{0}=\operatorname{diag}(\epsilon)$,
\[
\|w_{0}-w^{*}\|_{D_{0}}^{2}
\le D^{2}\,\max_{j}(\epsilon)=D^{2}\epsilon .
\tag{10}
\]
The last term in \eqref{eq:9} is non‑negative, so we may drop $-\|w_{T}-w^{*}\|_{D_{T}}^{2}$.

\noindent\textbf{[Step 6]} Write the adaptive norm explicitly:
\[
\|g_{k}\|_{D_{k+1}^{-1}}^{2}
= \sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}.
\tag{11}
\]

\noindent\textbf{[Step 7]} Insert \eqref{eq:10}–\eqref{eq:11} into \eqref{eq:9} and divide by $2\eta$:
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \frac{D^{2}\epsilon}{2\eta}
+\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}.
\tag{12}
\]

\noindent\textbf{[Step 8]} Apply Lemma~\ref{lem:step} with $a=|g_{k,j}|$ and $b=G_{k,j}$ (note that $G_{k+1,j}=G_{k,j}+g_{k,j}^{2}$):
\[
\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}
\le 2\bigl(\sqrt{G_{k+1,j}}-\sqrt{G_{k,j}}\bigr)+\epsilon .
\tag{13}
\]

\noindent\textbf{[Step 9]} Summing \eqref{eq:13} over $k=0,\dots,T-1$ telescopes:
\[
\sum_{k=0}^{T-1}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}
\le 2\sqrt{G_{T,j}}+T\epsilon .
\tag{14}
\]

\noindent\textbf{[Step 10]} Plug \eqref{eq:14} into \eqref{eq:12} and use the elementary inequality
$\max_{j}(\sqrt{G_{T,j}}+\epsilon)\le \sum_{j}\bigl(\sqrt{G_{T,j}}+\epsilon\bigr)$ to obtain
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le
\frac{D^{2}}{2\eta}\sum_{j=1}^{d}\sqrt{G_{T,j}}
\;+\;
\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}
\;+\;
\underbrace{\frac{D^{2}\epsilon}{2\eta}+\frac{\eta T d\epsilon}{2}}_{\text{terms vanishing as }\epsilon\to0}.
\tag{15}
\]

\noindent\textbf{[Step 11]} Let $\epsilon\downarrow0$. Both extra terms are non‑negative and converge to $0$, so the inequality simplifies to exactly \eqref{eq:regret} (i.e., (3)). \qedhere

\section*{Rate Derivation (With Constants)}
From convexity,
\[
f(w_{k})-f(w^{*})\le\inner{g_{k}}{w_{k}-w^{*}}.
\]
Averaging over $k=0,\dots,T-1$ and using \eqref{eq:regret} yields
\[
f(\bar w_{T})-f(w^{*})
\le\frac{D^{2}}{2\eta T}\sum_{j=1}^{d}\sqrt{G_{T,j}}
+\frac{\eta}{2T}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}.
\]
Because $G_{k+1,j}\ge g_{k,j}^{2}$ we have $\sqrt{G_{k+1,j}}\ge|g_{k,j}|$, and together with the bounded‑gradient assumption,
\[
\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}\le |g_{k,j}|\le G_{\infty}.
\]
Moreover $\sqrt{G_{T,j}}\le G_{\infty}\sqrt{T}$. Hence
\[
f(\bar w_{T})-f(w^{*})
\le\frac{D^{2}G_{\infty}\sqrt{d}}{2\eta\sqrt{T}}
+\frac{\eta G_{\infty}d}{2}.
\]
Optimising the right‑hand side with respect to $\eta$ gives
\[
\eta^{\star}= \frac{D\sqrt{d}}{G_{\infty}\sqrt{T}},
\]
and substituting $\eta^{\star}$ yields the final rate
\[
f(\bar w_{T})-f(w^{*})\;\le\; \frac{DG_{\infty}\sqrt{d}}{\sqrt{T}}.
\tag{17}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{One‑dimensional case ($d=1$).} Equation (17) reduces to $O\bigl(\frac{DG_{\infty}}{\sqrt{T}}\bigr)$.
\item \textbf{Sparse gradients.} If only $s\ll d$ coordinates receive non‑zero gradients, then $\sum_{j=1}^{d}\sqrt{G_{T,j}}=s\cdot O(\sqrt{T})$, improving the constant factor to $\sqrt{s}$.
\item \textbf{Strongly convex $f$.} Adding $\mu$‑strong convexity yields a $O\bigl(\frac{\log T}{T}\bigr)$ bound after a similar derivation (omitted for brevity).
\end{itemize}
