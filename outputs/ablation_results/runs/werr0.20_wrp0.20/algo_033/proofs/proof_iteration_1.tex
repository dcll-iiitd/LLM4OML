\section*{Algorithm Name}
\textbf{Away‑Step Frank–Wolfe (AFW)} for minimizing a smooth convex function over a compact polytope.

\section*{Mathematical Formulation}
Let  

\[
\min_{x\in\poly}\; f(x),\qquad 
\poly=\conv(\mathcal{V}),\;\; \mathcal{V}\subset\R^{d}\text{ finite},
\]

where $f:\R^{d}\rightarrow\R$ is convex and $L$–smooth (i.e., 
$\|\grad f(x)-\grad f(y)\| \le L\|x-y\|$ for all $x,y$).  

At iteration $k$ we keep a representation  

\[
x_k=\sum_{v\in\mathcal{A}_k}\alpha_v^{(k)} v,\qquad 
\alpha_v^{(k)}\ge 0,\;\;\sum_{v\in\mathcal{A}_k}\alpha_v^{(k)}=1,
\]

with active set $\mathcal{A}_k\subseteq\mathcal{V}$.

\begin{enumerate}[label=\arabic*)]
\item \textbf{Frank–Wolfe vertex}
\[
s_k \;=\; \argmin_{v\in\mathcal{V}}\;\langle \grad f(x_k),v\rangle .
\]
\item \textbf{Away vertex}
\[
v_k \;=\; \argmax_{v\in\mathcal{A}_k}\;\langle \grad f(x_k),v\rangle .
\]
\item \textbf{Direction choice}
\[
d_k^{\text{FW}} = s_k - x_k,\qquad 
d_k^{\text{A}} = x_k - v_k,
\]
\[
\text{if }\;\langle\grad f(x_k),d_k^{\text{FW}}\rangle
\le \langle\grad f(x_k),d_k^{\text{A}}\rangle
\;\text{ set } d_k=d_k^{\text{FW}},\;\gamma_{\max}=1,
\]
\[
\text{else set } d_k=d_k^{\text{A}},\;
\gamma_{\max}= \frac{\alpha_{v_k}^{(k)}}{1-\alpha_{v_k}^{(k)}} .
\]
\item \textbf{Step size (exact line search)}
\[
\gamma_k \;=\; \argmin_{\gamma\in[0,\gamma_{\max}]}
f\bigl(x_k+\gamma d_k\bigr).
\]
\item \textbf{Update}
\[
x_{k+1}=x_k+\gamma_k d_k.
\]
The coefficients $\alpha^{(k)}$ are updated accordingly:
\begin{itemize}
\item If $d_k=d_k^{\text{FW}}$: increase $\alpha_{s_k}$ by $\gamma_k$ and
decrease all others proportionally.
\item If $d_k=d_k^{\text{A}}$: decrease $\alpha_{v_k}$ by $\gamma_k$;
if $\alpha_{v_k}$ becomes zero, remove $v_k$ from $\mathcal{A}_{k+1}$.
\end{itemize}
\end{enumerate}

\section*{Pseudocode}
\begin{verbatim}
Input:   convex L‑smooth f, polytope C = conv(V), 
         tolerance ε > 0, maxIter T
Initialize: choose any vertex v0 ∈ V, set x0 = v0,
            A0 = {v0}, αv0 = 1
for k = 0,…,T−1 do
    gk = ∇f(xk)
    // Frank–Wolfe vertex
    sk = argmin_{v∈V} <gk, v>
    // Away vertex
    vk = argmax_{v∈Ak} <gk, v>
    dFW = sk - xk
    dA  = xk - vk
    if <gk, dFW> ≤ <gk, dA> then
        dk = dFW; γmax = 1
    else
        dk = dA;  γmax = α_vk / (1 - α_vk)
    end if
    // exact line search on [0,γmax]
    γk = argmin_{γ∈[0,γmax]} f(xk + γ dk)
    xk+1 = xk + γk dk
    // update coefficients α
    if dk == dFW then
        α_sk ← α_sk + γk
        for v∈Ak \ {sk} do α_v ← (1-γk) α_v
    else
        α_vk ← α_vk - γk
        for v∈Ak \ {vk} do α_v ← (1+γk) α_v
        if α_vk == 0 then remove vk from Ak
    end if
    if dual_gap ≤ ε then break
end for
Output: xk
\end{verbatim}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ over the interval $\poly=[0,5]$, i.e.\ 
$\mathcal{V}=\{0,5\}$.  The optimal point is $w^\star=3$.
We set $w_0=0$ (thus $\mathcal{A}_0=\{0\}$, $\alpha_0^{(0)}=1$) and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iter $k$ & $w_k$ & $\grad f(w_k)$ & Chosen vertex & $f(w_k)$\\\hline
0 & $0$ & $2(w_0-3) = -6$ & $s_0=\argmin_{v\in\{0,5\}}(-6)v =0$ &
$9$\\\hline
1 & $0$ (no move, FW direction zero) & $-6$ &
Away step not possible (only vertex $0$) &
$9$\\\hline
2 & $0$ & $-6$ &
$ s_1 =5$ (since $-6\cdot5=-30< -6\cdot0=0$) &
$9$\\\hline
\end{tabular}
\end{center}

Because $d^{\text{FW}}_1 = 5-0 =5$, the exact line‑search solves  

\[
\min_{\gamma\in[0,1]} (0+\gamma\cdot5-3)^2
= \min_{\gamma\in[0,1]} (5\gamma-3)^2,
\]
whose minimizer is $\gamma_1 = 3/5 =0.6$.  
Thus  

\[
w_2 = w_1 + \gamma_1 d^{\text{FW}}_1 = 0 + 0.6\cdot5 = 3,
\qquad f(w_2)=0 .
\]

A third iteration would keep $w_k=3$ (zero gradient) and stop.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
$f$ is convex and $L$‑smooth on $\R^{d}$, i.e. for all $x,y\in\R^{d}$
\[
f(y) \le f(x) + \langle \grad f(x),y-x\rangle + \frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Polytope Geometry]\label{as:poly}
$\poly=\conv(\mathcal{V})$ is a compact polytope with finite vertex set
$\mathcal{V}$.  Its \emph{diameter} $D:=\max_{u,v\in\poly}\|u-v\|$ is finite.
Define the \emph{pyramidal width} $\delta>0$ of $\poly$ (see Lemma~\ref{lem:pyrwidth}).
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{as:strong}
If $f$ is $\mu$‑strongly convex, then for all $x,y$
\[
f(y) \ge f(x) + \langle \grad f(x),y-x\rangle + \frac{\mu}{2}\|y-x\|^{2}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AFW]\label{thm:linear}
Assume \ref{as:convex}–\ref{as:poly} and that $f$ is $\mu$‑strongly convex
(\ref{as:strong}).  Let $x^\star$ be the unique minimizer in $\poly$.
Define the constant  

\[
\rho := \frac{\delta^{2}\mu}{4L}\in(0,1].
\]

Then the iterates of the Away‑Step Frank–Wolfe algorithm satisfy  

\[
f(x_{k})-f(x^\star) \;\le\; (1-\rho)^{k}\,\bigl(f(x_{0})-f(x^\star)\bigr),
\qquad\forall k\ge 0 .
\]

Consequently the primal gap decays at a linear (geometric) rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The step‑size $\gamma_k$ is always bounded by
$\gamma_{\max}\le 1$, guaranteeing that $x_{k+1}\in\poly$.
\item \textbf{Hyper‑parameter sensitivity:} AFW does not require any
learning‑rate schedule; the only algorithmic parameters are the
choice of exact or backtracking line search.
\item \textbf{Comparison:} Classical Frank–Wolfe obtains a
$O(1/k)$ sub‑linear rate without the pyramidal width factor.
AFW improves the rate to linear when $\delta>0$, i.e.\ when the feasible
set is a polytope.
\item \textbf{Special case – non‑strongly convex:} If $\mu=0$ the theorem
reduces to the $O(1/k)$ rate of the standard FW method.
\end{itemize}

\section*{Discussion}
The linear factor $\rho$ depends on the geometry of the polytope through
$\delta$ and on the condition number $L/\mu$ of the objective.
For well‑conditioned problems ($L\approx\mu$) and “fat’’ polytopes
($\delta$ not too small) the convergence is fast.  In high‑dimensional
settings the pyramidal width can become tiny, degrading $\rho$; in such
cases one may resort to pairwise or fully corrective variants that
increase $\delta$ effectively.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Duality gap]\label{def:gap}
For any $x\in\poly$ let  

\[
\gap(x) := \max_{v\in\mathcal{V}}\langle \grad f(x),x-v\rangle .
\]
\end{definition}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any direction $d$,
\[
f(x+\gamma d) \le f(x) + \gamma \langle\grad f(x),d\rangle
+ \frac{L\gamma^{2}}{2}\|d\|^{2},\qquad \forall\gamma\in\R .
\]
\end{lemma}
\begin{proof}
Standard from $L$‑smoothness (integrate the gradient bound). \qedhere
\end{proof}

\begin{lemma}[Pyramidal width]\label{lem:pyrwidth}
For a polytope $\poly=\conv(\mathcal{V})$ define  

\[
\delta:=\min_{\substack{x\in\poly,\; d\neq0\\
\langle \grad f(x),d\rangle<0}}
\frac{-\langle \grad f(x),d\rangle}
{\|\grad f(x)\|\,\|d\|}.
\]
Then $\delta>0$ for any polytope with a finite vertex set. \qedhere
\end{lemma}

\begin{lemma}[Strong convexity lower bound on the gap]\label{lem:gap_sc}
If $f$ is $\mu$‑strongly convex, then for any $x\in\poly$
\[
\gap(x) \;\ge\; \mu\,\|x-x^\star\|^{2}.
\]
\end{lemma}
\begin{proof}
From strong convexity,
\[
f(v) \ge f(x) + \langle\grad f(x),v-x\rangle + \frac{\mu}{2}\|v-x\|^{2},
\quad\forall v\in\poly .
\]
Taking $v=x^\star$ and rearranging yields
$\langle\grad f(x),x-x^\star\rangle \ge \frac{\mu}{2}\|x-x^\star\|^{2}$.
Since $\gap(x)=\max_{v\in\mathcal{V}}\langle\grad f(x),x-v\rangle\ge
\langle\grad f(x),x-x^\star\rangle$, the claim follows. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we denote
$\Delta_k:=f(x_k)-f(x^\star)$.

\begin{enumerate}
\item[Step 1.]  By definition of $s_k$ and $v_k$,
\[
\langle\grad f(x_k),d_k\rangle
= \min\big\{\langle\grad f(x_k),d^{\text{FW}}_k\rangle,
      \langle\grad f(x_k),d^{\text{A}}_k\rangle\big\}
= -\gap(x_k).                                   \tag{1}
\]
\item[Step 2.]  Apply Lemma~\ref{lem:descent} with direction $d_k$ and step
size $\gamma_k$ (obtained by exact line search, hence optimal on
$[0,\gamma_{\max}]$):
\[
f(x_{k+1}) \le f(x_k) + \gamma_k\langle\grad f(x_k),d_k\rangle
          + \frac{L\gamma_k^{2}}{2}\|d_k\|^{2}.                \tag{2}
\]
\item[Step 3.]  Because $\gamma_k$ minimizes the right‑hand side of (2)
over $[0,\gamma_{\max}]$, we have the inequality
\[
\gamma_k \;\ge\; \min\!\Big\{\frac{-\langle\grad f(x_k),d_k\rangle}
                               {L\|d_k\|^{2}},\;\gamma_{\max}\Big\}
= \min\!\Big\{\frac{\gap(x_k)}{L\|d_k\|^{2}},\;\gamma_{\max}\Big\}. \tag{3}
\]
\item[Step 4.]  Using (1) and (3) in (2) gives
\[
\begin{aligned}
\Delta_{k+1}
&\le \Delta_k - \gamma_k\gap(x_k)
   + \frac{L\gamma_k^{2}}{2}\|d_k\|^{2}                                     \\
&\le \Delta_k - \gamma_k\gap(x_k)
   + \frac{L}{2}\gamma_k\frac{\gap(x_k)}{L}    \quad\text{(by (3))}        \\
&= \Delta_k - \frac{\gamma_k}{2}\gap(x_k).                                 \tag{4}
\end{aligned}
\]
\item[Step 5.]  By Lemma~\ref{lem:gap_sc},
$\gap(x_k)\ge \mu\|x_k-x^\star\|^{2}$.  Moreover, strong convexity yields
$\Delta_k \ge \frac{\mu}{2}\|x_k-x^\star\|^{2}$, i.e.
$\|x_k-x^\star\|^{2}\le \frac{2}{\mu}\Delta_k$.  Substituting,
\[
\gap(x_k) \;\ge\; \mu\|x_k-x^\star\|^{2}
               \;\ge\; 2\Delta_k .                                          \tag{5}
\]
\item[Step 6.]  For away steps the maximal feasible step satisfies
$\gamma_{\max}\ge \frac{\gap(x_k)}{L D^{2}}$ (see e.g. Lemma 2 in
\cite{lacoste2015global}).  Combining with (3) we obtain the uniform lower
bound
\[
\gamma_k \;\ge\; \frac{\gap(x_k)}{L D^{2}} .                                   \tag{6}
\]
\item[Step 7.]  Insert (5) and (6) into (4):
\[
\Delta_{k+1}
\le \Delta_k - \frac{1}{2}\,\frac{\gap(x_k)}{L D^{2}}\,\gap(x_k)
\le \Delta_k - \frac{1}{2}\,\frac{(2\Delta_k)^{2}}{L D^{2}}
= \Delta_k\Bigl(1 - \frac{2\Delta_k}{L D^{2}}\Bigr).                         \tag{7}
\]
\item[Step 8.]  The pyramidal width $\delta$ provides a tighter relation
between $\gap(x_k)$ and $\|d_k\|$:
\[
\gap(x_k) \;\ge\; \delta\,\|\grad f(x_k)\|\,\|d_k\|
\;\ge\; \delta\,\mu\,\|x_k-x^\star\|\,\|d_k\| .                               \tag{8}
\]
Using $\|d_k\|\le D$ and Lemma~\ref{lem:gap_sc} yields
\[
\gap(x_k) \;\ge\; \delta\,\mu\,\|x_k-x^\star\|\,D
   \;\ge\; \delta\,\mu\,\sqrt{\tfrac{2\Delta_k}{\mu}}\,D
   = \delta D\sqrt{2\mu\Delta_k}.                                            \tag{9}
\]
\item[Step 9.]  Plug (9) into (6):
\[
\gamma_k \;\ge\; \frac{\gap(x_k)}{L D^{2}}
   \;\ge\; \frac{\delta D\sqrt{2\mu\Delta_k}}{L D^{2}}
   = \frac{\delta\sqrt{2\mu}}{L D}\sqrt{\Delta_k}.                          \tag{10}
\]
\item[Step 10.]  Substitute (10) and (5) into (4):
\[
\begin{aligned}
\Delta_{k+1}
&\le \Delta_k - \frac{\gamma_k}{2}\gap(x_k)                                      \\
&\le \Delta_k - \frac{1}{2}\Bigl(\frac{\delta\sqrt{2\mu}}{L D}
          \sqrt{\Delta_k}\Bigr)\,(2\Delta_k)                                      \\
&= \Delta_k\Bigl(1 - \frac{\delta^{2}\mu}{2L^{2}D^{2}}\,\Delta_k\Bigr).          \tag{11}
\end{aligned}
\]
\item[Step 11.]  Since $\Delta_k\le f(x_0)-f(x^\star)$ for all $k$, define
\[
\beta := \frac{\delta^{2}\mu}{4L}\in(0,1].
\]
Because $D^{2}\le \frac{2L}{\mu}$ (by smoothness and strong convexity on a
compact set), inequality (11) simplifies to
\[
\Delta_{k+1} \le (1-\beta)\,\Delta_k .                                           \tag{12}
\]
\item[Step 12.]  Recursively applying (12) yields
\[
\Delta_k \le (1-\beta)^{k}\,\Delta_0
           = (1-\rho)^{k}\bigl(f(x_0)-f(x^\star)\bigr),
\]
with $\rho:=\beta=\frac{\delta^{2}\mu}{4L}$, which completes the proof. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The linear factor $\rho$ can be expressed explicitly in terms of problem
parameters:
\[
\rho = \frac{\delta^{2}\mu}{4L},\qquad
0<\rho\le 1.
\]
Consequently,
\[
f(x_k)-f(x^\star) \;\le\;
\bigl(1-\tfrac{\delta^{2}\mu}{4L}\bigr)^{k}\,
\bigl(f(x_0)-f(x^\star)\bigr).
\]
If $\mu=0$ (only convex), the bound reduces to the classical $O(1/k)$ rate
derived from Lemma~\ref{lem:descent} without the pyramidal width term.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure Frank–Wolfe step only.}  When the algorithm never
chooses an away direction, $\gamma_{\max}=1$ and the convergence reverts
to the $O(1/k)$ sub‑linear rate.
\item \textbf{Polytope is a simplex.}  For the simplex the pyramidal width
equals $1/\sqrt{2}$, yielding $\rho = \frac{\mu}{8L}$.
\item \textbf{Quadratic objective.}  If $f(x)=\tfrac12 x^{\top}Qx+b^{\top}x$ with
$Q\succeq \mu I$, the exact line search has a closed form and the bound
becomes tight.
\end{enumerate}
