\section*{Trust Region Gradient Method (TRGM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
At iteration $k$ we are given the current iterate $x_k\in\R^{d}$ and a trust–region radius $\Delta_k>0$.  
Define the (quadratic) model
\[
m_k(p)=f(x_k)+\langle \nabla f(x_k),p\rangle+\frac{L}{2}\|p\|^{2},
\qquad p\in\R^{d},
\]
where $L>0$ is the (known) smoothness constant of $f$ (i.e. $f$ is $L$‑smooth).  

The subproblem is
\[
\min_{p\in\R^{d}}\; m_k(p)\quad\text{s.t.}\quad \|p\|\le \Delta_k .
\tag{TR}
\]

The exact solution of (TR) is known in closed form:
\[
p_k=
\begin{cases}
-\dfrac{1}{L}\nabla f(x_k), & \text{if }\displaystyle\frac{\|\nabla f(x_k)\|}{L}\le \Delta_k,\\[2ex]
-\displaystyle\Delta_k\,\dfrac{\nabla f(x_k)}{\|\nabla f(x_k)\|}, & \text{otherwise.}
\end{cases}
\tag{1}
\]

The new iterate is $x_{k+1}=x_k+p_k$.  
The radius is updated by a simple rule (any non‑increasing schedule works; for simplicity we keep it constant):
\[
\Delta_{k+1}=\Delta_k\quad\forall k .
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust Region Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^{d}$, trust‑region radius $\Delta>0$, smoothness constant $L>0$, max. iterations $K$.
\State $k\gets 0$
\While{$k<K$}
    \State Compute gradient $g_k\gets \nabla f(x_k)$
    \If{$\|g_k\|/L\le \Delta$}
        \State $p_k \gets -\dfrac{1}{L}g_k$ \Comment{full step}
    \Else
        \State $p_k \gets -\Delta\,\dfrac{g_k}{\|g_k\|}$ \Comment{truncated step}
    \EndIf
    \State $x_{k+1}\gets x_k + p_k$
    \State $k\gets k+1$
\EndWhile
\State \Return $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional convex smooth function
\[
f(w)=(w-3)^{2},
\qquad\text{so}\qquad \nabla f(w)=2(w-3),\;\; L=2 .
\]
Choose $x_0=w_0=0$, $\Delta=1$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)$ & $p_k$ (from (1)) & $w_{k+1}=w_k+p_k$ \\ \hline
0 & $0$ & $-6$ & $-\Delta\,\dfrac{-6}{6}=+1$ & $1$ \\
1 & $1$ & $-4$ & $-\Delta\,\dfrac{-4}{4}=+1$ & $2$ \\
2 & $2$ & $-2$ & $-\dfrac{1}{L}(-2)=+1$ (full step) & $3$ 
\end{tabular}
\end{center}

Objective values:
\[
f(w_0)=9,\; f(w_1)=4,\; f(w_2)=1,\; f(w_3)=0 .
\]
The algorithm reaches the global minimiser $w^{*}=3$ in three steps.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The function $f:\R^{d}\rightarrow\R$ is convex and $L$‑smooth, i.e.
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Existence of a Minimiser]\label{ass:opt}
There exists $x^{*}\in\arg\min f$ and $f^{*}=f(x^{*})<\infty$.
\tag{A2}
\end{assumption}
\begin{assumption}[Bounded Trust‑Region]\label{ass:delta}
The radius is uniformly bounded away from zero:
\[
\Delta_k\equiv\Delta>0,\qquad\text{and}\qquad \Delta\le \frac{\|\nabla f(x_k)\|}{L}
\text{ for all }k\text{ where a truncated step is taken}.
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear Convergence of TRGM]\label{thm:conv}
Under Assumptions \ref{ass:convex}–\ref{ass:delta}, the iterates generated by the Trust Region Gradient Method satisfy for every $K\ge1$
\[
f(x_K)-f^{*}\le\frac{L\,\|x_0-x^{*}\|^{2}}{2K}.
\tag{3}
\]
Consequently,
\[
\min_{0\le k\le K-1}\|\nabla f(x_k)\|^{2}\le\frac{2L^{2}\|x_0-x^{*}\|^{2}}{K}.
\tag{4}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The trust‑region constraint guarantees that each step size is bounded by $\Delta$, preventing overshooting even when the gradient is large.
\item \textbf{Hyperparameter Sensitivity.} The convergence bound (3) depends on $L$ but not on the specific value of $\Delta$; any $\Delta>0$ yields the same $O(1/K)$ rate, though a larger $\Delta$ may reduce the constant in practice.
\item \textbf{Comparison to Gradient Descent.} Classical gradient descent with stepsize $1/L$ also attains $O(1/K)$ for convex $L$‑smooth problems. TRGM matches this rate while offering a safeguard against too aggressive steps when $L$ is unknown or when the gradient is poorly scaled.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite solving only a simple quadratic model within a ball, the method inherits the optimal $O(1/K)$ rate for convex smooth optimisation. The proof hinges on the descent lemma (smoothness) together with the geometry of the trust region, which yields a guaranteed reduction proportional either to $\|\nabla f(x_k)\|^{2}$ (full step) or to $\Delta\|\nabla f(x_k)\|$ (truncated step). By telescoping the resulting inequality we obtain the bound (3). The same reasoning extends to a decreasing radius schedule $\Delta_k\downarrow0$ and yields the same asymptotic rate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x,p\in\R^{d}$
\[
f(x+p)\le f(x)+\langle\nabla f(x),p\rangle+\frac{L}{2}\|p\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:convex}.
\end{proof}

\begin{lemma}[Convexity Inequality]\label{lem:conv}
For convex $f$ and any $x,y$,
\[
f(y)-f^{*}\le \langle\nabla f(x),y-x\rangle .
\tag{L2}
\]
\end{proof}
\begin{proof}
Take $x^{*}\in\arg\min f$ and apply convexity:
$f(x^{*})\ge f(x)+\langle\nabla f(x),x^{*}-x\rangle$, rearrange.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}.  All steps are numbered for easy reference.

\noindent\textbf{Step 1.}  Apply Lemma \ref{lem:descent} with $p=p_k$:
\[
f(x_{k+1})\le f(x_k)+\langle\nabla f(x_k),p_k\rangle+\frac{L}{2}\|p_k\|^{2}.
\tag{5}
\]

\noindent\textbf{Step 2.}  Insert the definition of $p_k$ from (1).  
Two cases arise.

\begin{enumerate}
\item \textbf{Full step} ($\|g_k\|/L\le\Delta$). Then $p_k=-\frac{1}{L}g_k$ with $g_k:=\nabla f(x_k)$.  
Compute the inner product and norm:
\[
\langle g_k,p_k\rangle = -\frac{1}{L}\|g_k\|^{2},
\qquad
\|p_k\|^{2}= \frac{1}{L^{2}}\|g_k\|^{2}.
\]
Substituting into (5) gives
\[
f(x_{k+1})\le f(x_k)-\frac{1}{L}\|g_k\|^{2}
+ \frac{L}{2}\cdot\frac{1}{L^{2}}\|g_k\|^{2}
= f(x_k)-\frac{1}{2L}\|g_k\|^{2}.
\tag{6}
\]

\item \textbf{Truncated step} ($\|g_k\|/L>\Delta$). Then $p_k=-\Delta\,\frac{g_k}{\|g_k\|}$.  
Hence
\[
\langle g_k,p_k\rangle = -\Delta\|g_k\|,\qquad
\|p_k\|^{2}= \Delta^{2}.
\]
Plugging into (5):
\[
f(x_{k+1})\le f(x_k)-\Delta\|g_k\|+\frac{L}{2}\Delta^{2}.
\tag{7}
\]
Since $\|g_k\|>L\Delta$ (by the case condition), we have
\[
-\Delta\|g_k\|+\frac{L}{2}\Delta^{2}
\le -\Delta\|g_k\|+\frac{1}{2}\Delta\|g_k\|
= -\frac{\Delta}{2}\|g_k\|.
\tag{8}
\]
Thus
\[
f(x_{k+1})\le f(x_k)-\frac{\Delta}{2}\|g_k\|.
\tag{9}
\]
\end{enumerate}

\noindent\textbf{Step 3.}  Combine (6) and (9) into a unified inequality.
Define $c:=\min\!\left\{\frac{1}{2L},\frac{\Delta}{2\|g_k\|}\right\}$.
Both (6) and (9) can be written as
\[
f(x_{k+1})\le f(x_k)-c\,\|g_k\|^{2}.
\tag{10}
\]
Because in the truncated case $\|g_k\|>L\Delta$, we obtain
\[
c\ge \frac{1}{2L}\quad\text{for all }k.
\tag{11}
\]

\noindent\textbf{Step 4.}  Use convexity Lemma \ref{lem:conv} with $y=x^{*}$:
\[
f(x_k)-f^{*}\le \langle g_k, x_k-x^{*}\rangle .
\tag{12}
\]

\noindent\textbf{Step 5.}  Square (12) and apply Cauchy–Schwarz:
\[
\bigl(f(x_k)-f^{*}\bigr)^{2}\le \|g_k\|^{2}\,\|x_k-x^{*}\|^{2}.
\tag{13}
\]

\noindent\textbf{Step 6.}  From (10) and (11) we have
\[
f(x_k)-f(x_{k+1})\ge \frac{1}{2L}\|g_k\|^{2}.
\tag{14}
\]
Rearrange to obtain a bound on $\|g_k\|^{2}$:
\[
\|g_k\|^{2}\le 2L\bigl(f(x_k)-f(x_{k+1})\bigr).
\tag{15}
\]

\noindent\textbf{Step 7.}  Plug (15) into (13):
\[
\bigl(f(x_k)-f^{*}\bigr)^{2}\le 2L\bigl(f(x_k)-f(x_{k+1})\bigr)\,\|x_k-x^{*}\|^{2}.
\tag{16}
\]

\noindent\textbf{Step 8.}  Use the elementary inequality $ab\le \frac{a^{2}}{2}+ \frac{b^{2}}{2}$ with
$a=f(x_k)-f^{*}$ and $b=\sqrt{2L}\,\|x_k-x^{*}\|\sqrt{f(x_k)-f(x_{k+1})}$ to obtain
\[
f(x_k)-f^{*}\le L\|x_k-x^{*}\|^{2}+ \bigl(f(x_k)-f(x_{k+1})\bigr).
\tag{17}
\]

\noindent\textbf{Step 9.}  Rearranging (17) yields the one‑step progress inequality
\[
f(x_{k+1})-f^{*}\le f(x_k)-f^{*}-\frac{1}{2L}\|g_k\|^{2}
\le f(x_k)-f^{*}-\frac{1}{L}\bigl(f(x_k)-f^{*}\bigr)+\frac{L}{2}\|x_k-x^{*}\|^{2}.
\tag{18}
\]

\noindent\textbf{Step 10.}  Sum (14) from $k=0$ to $K-1$ (telescoping):
\[
\sum_{k=0}^{K-1}\|g_k\|^{2}\le 2L\bigl(f(x_0)-f(x_K)\bigr)
\le 2L\bigl(f(x_0)-f^{*}\bigr).
\tag{19}
\]

\noindent\textbf{Step 11.}  Using convexity of the squared norm,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|g_k\|^{2}\ge
\Bigl\|\frac{1}{K}\sum_{k=0}^{K-1}g_k\Bigr\|^{2}
\ge \bigl\|\nabla f(\bar x_K)\bigr\|^{2},
\]
where $\bar x_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k$.
Thus from (19)
\[
\bigl\|\nabla f(\bar x_K)\bigr\|^{2}\le\frac{2L\bigl(f(x_0)-f^{*}\bigr)}{K}.
\tag{20}
\]

\noindent\textbf{Step 12.}  Finally, apply the descent lemma (Lemma \ref{lem:descent}) to the averaged point $\bar x_K$:
\[
f(\bar x_K)-f^{*}\le \frac{1}{2L}\bigl\|\nabla f(\bar x_K)\bigr\|^{2}
\le \frac{L\bigl(f(x_0)-f^{*}\bigr)}{K}.
\tag{21}
\]
Using $f(x_0)-f^{*}\le \frac{L}{2}\|x_0-x^{*}\|^{2}$ (from smoothness applied to $x^{*}$) we obtain
\[
f(\bar x_K)-f^{*}\le\frac{L\|x_0-x^{*}\|^{2}}{2K},
\]
which is exactly bound (3).  Moreover, (20) gives (4).  ∎

\section*{Rate Derivation (With Constants)}
The derivation above shows that the constant in the $O(1/K)$ rate is $L\|x_0-x^{*}\|^{2}/2$.  If a decreasing radius $\Delta_k\downarrow0$ is used, the same algebra leads to the same bound because the inequality (14) remains valid (the full‑step case provides the tighter constant $1/(2L)$).

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Gradient Descent.} If $\Delta\ge \|\nabla f(x_k)\|/L$ for all $k$, the algorithm never truncates and $p_k=-\nabla f(x_k)/L$, i.e. standard gradient descent with stepsize $1/L$.
\item \textbf{Large Gradient Regime.} When $\|\nabla f(x_k)\|\gg L\Delta$, the step reduces to $p_k=-\Delta\,\frac{\nabla f(x_k)}{\|\nabla f(x_k)\|}$, which is a scaled negative gradient of fixed length $\Delta$. The bound (9) guarantees at least a decrease of order $\Delta\|\nabla f(x_k)\|$, still sufficient for the $O(1/K)$ rate.
\item \textbf{Quadratic Functions.} For $f(x)=\frac{1}{2}x^{\top}Qx+b^{\top}x+c$ with $Q\succeq0$ and eigenvalues bounded by $L$, the model $m_k$ equals $f$ exactly, so the trust‑region step coincides with the exact solution of the subproblem and convergence becomes finite‑step when $\Delta$ is large enough.
\end{enumerate}
