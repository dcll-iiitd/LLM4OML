\section*{Clipped Momentum Gradient Descent (CMGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
At iteration $t\ge 0$ we denote the (full) gradient by 
\[
\bm{g}_t^{\text{raw}}:=\nabla f(\bm{x}_t).
\]

\paragraph{Gradient clipping.}
Given a clipping threshold $C>0$, the clipped gradient is
\[
\boxed{\;
\bm{g}_t:=\frac{\bm{g}_t^{\text{raw}}}{\max\!\Bigl(1,\;\frac{\|\bm{g}_t^{\text{raw}}\|}{C}\Bigr)}\;}
\tag{1}
\]
so that $\|\bm{g}_t\|\le C$ for all $t$.

\paragraph{Momentum on clipped gradients.}
A momentum vector $\bm{v}_t\in\mathbb{R}^d$ is updated as
\[
\boxed{\;
\bm{v}_{t}= \beta\,\bm{v}_{t-1}+(1-\beta)\,\bm{g}_t,\qquad \bm{v}_{-1}=0\;}
\tag{2}
\]
with momentum parameter $\beta\in[0,1)$.

\paragraph{Parameter update.}
The iterate is moved along the momentum direction:
\[
\boxed{\;
\bm{x}_{t+1}= \bm{x}_t-\eta\,\bm{v}_t\;}
\tag{3}
\]
where $\eta>0$ is the (constant) stepsize.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Momentum Gradient Descent (CMGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $\bm{x}_0$, stepsize $\eta>0$, clipping threshold $C>0$, momentum $\beta\in[0,1)$, max iterations $T$.
\STATE $\bm{v}_{-1}\gets \bm{0}$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute raw gradient $\bm{g}^{\text{raw}}_t\gets \nabla f(\bm{x}_t)$.
    \STATE Clip: 
    \[
    \bm{g}_t\gets\frac{\bm{g}^{\text{raw}}_t}{\max\!\Bigl(1,\frac{\|\bm{g}^{\text{raw}}_t\|}{C}\Bigr)} .
    \]
    \STATE Momentum update: $\bm{v}_t\gets \beta\bm{v}_{t-1}+(1-\beta)\bm{g}_t$.
    \STATE Parameter update: $\bm{x}_{t+1}\gets \bm{x}_t-\eta\bm{v}_t$.
\ENDFOR
\STATE \textbf{Return} $\bm{x}_T$ (or the best iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$, $C=1$, $\beta=0.5$, $\eta=0.1$, $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_t^{\text{raw}}$ & $g_t$ (clipped) & $v_t$ & $w_{t+1}$ \\ \hline
0 & $2(w_0-3)=-6$ & $-1$ (clipped to $C$) & $v_0=0.5\cdot0+0.5(-1)=-0.5$ & $w_1=0-0.1(-0.5)=0.05$ \\
1 & $2(w_1-3)= -5.9$ & $-1$ & $v_1=0.5(-0.5)+0.5(-1)=-0.75$ & $w_2=0.05-0.1(-0.75)=0.125$ \\
2 & $2(w_2-3)= -5.75$ & $-1$ & $v_2=0.5(-0.75)+0.5(-1)=-0.875$ & $w_3=0.125-0.1(-0.875)=0.2125$
\end{tabular}
\end{center}
Objective values: $f(w_0)=9$, $f(w_1)\approx 8.41$, $f(w_2)\approx 8.27$, $f(w_3)\approx 8.13$.  
The iterates move toward the minimiser $w^\*=3$, but the clipping prevents excessively large updates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall \bm{x},\bm{y}\in\mathbb{R}^d:\qquad 
\|\nabla f(\bm{x})-\nabla f(\bm{y})\|\le L\|\bm{x}-\bm{y}\|.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:bounded}
There exists $f^\*:=\inf_{\bm{x}}f(\bm{x})>-\infty$.
\tag{A2}
\end{assumption}
\begin{assumption}[Clipping Threshold]\label{ass:clip}
The clipping constant $C>0$ is finite, so that for all $t$, $\|\bm{g}_t\|\le C$.
\tag{A3}
\end{assumption}
\begin{assumption}[Momentum Parameter]\label{ass:beta}
$\beta\in[0,1)$.
\tag{A4}
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:eta}
The stepsize satisfies 
\[
0<\eta\le\frac{1-\beta}{L}.
\tag{A5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:eta}, for any integer $T\ge1$ the iterates generated by CMGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\;\le\;
\underbrace{\frac{2\bigl(f(\bm{x}_0)-f^\*\bigr)}{\eta T(1-\beta)}}_{\displaystyle\text{optimization term}}
\;+\;
\underbrace{L\eta C^{2}}_{\displaystyle\text{clipping–momentum term}} .
\tag{4}
\]
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because $\|\bm{g}_t\|\le C$, the momentum vector $\bm{v}_t$ is a convex combination of bounded vectors; thus $\|\bm{v}_t\|\le C$ for all $t$, guaranteeing that the update (3) cannot explode regardless of the raw gradient magnitude.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ reduces the bias introduced by clipping but may increase the variance term $L\eta C^{2}$ in (4).
        \item $\beta$ close to $1$ yields a smoother momentum direction but enlarges the denominator $(1-\beta)$ in the optimization term, slowing down convergence.
        \item The stepsize bound (A5) couples $\eta$, $\beta$ and $L$; violating it may break the descent lemma used in the proof.
    \end{itemize}
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Plain (unclipped) gradient descent enjoys the same $O(1/\sqrt{T})$ rate under the same smoothness assumptions, but without the guarantee $\|\bm{g}_t\|\le C$; thus CMGD is more robust to outliers.
        \item Stochastic variants add a variance term $\sigma^{2}$; the bound (4) shows that clipping alone already introduces a deterministic error term $L\eta C^{2}$ that plays an analogous role.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that, even in the non‑convex setting, gradient clipping together with a simple momentum scheme preserves the classical $O(T^{-1/2})$ decay of the average gradient norm. The bound decomposes naturally into an optimization term that vanishes as $T\to\infty$ and a clipping‑induced error term that can be made arbitrarily small by choosing a decreasing stepsize. This explains the empirical stability of “Clipped Adam’’‑type methods observed in deep‑learning practice.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $\bm{x},\bm{d}\in\mathbb{R}^d$,
\[
f(\bm{x}+\bm{d})\le f(\bm{x})+\langle\nabla f(\bm{x}),\bm{d}\rangle+\frac{L}{2}\|\bm{d}\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004introductory}. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on Momentum]\label{lem:momentumbound}
Under (A3) and (A4), for all $t\ge0$,
\[
\|\bm{v}_t\|\le C .
\tag{6}
\]
\end{lemma}
\begin{proof}
We prove by induction.
\begin{itemize}
\item[Base:] $\bm{v}_{-1}=0$ trivially satisfies $\|\bm{v}_{-1}\|\le C$.
\item[Inductive step:] Assume $\|\bm{v}_{t-1}\|\le C$. Using (2) and the triangle inequality,
\[
\|\bm{v}_t\|
= \bigl\|\beta\bm{v}_{t-1}+(1-\beta)\bm{g}_t\bigr\|
\le \beta\|\bm{v}_{t-1}\|+(1-\beta)\|\bm{g}_t\|
\le \beta C+(1-\beta)C = C .
\]
\end{itemize}
Thus (6) holds for all $t$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem \ref{thm:main}. All steps are numbered for easy reference.

\begin{proof}
Let $\bm{x}_{t+1}$ be defined by (3). Apply Lemma \ref{lem:descent} with $\bm{d}=-\eta\bm{v}_t$:
\[
f(\bm{x}_{t+1})\le f(\bm{x}_t)-\eta\langle\nabla f(\bm{x}_t),\bm{v}_t\rangle+\frac{L\eta^{2}}{2}\|\bm{v}_t\|^{2}.
\tag{7}
\]

\textbf{[Step 1]}  Expand the inner product using the definition of $\bm{v}_t$ (2):
\[
\langle\nabla f(\bm{x}_t),\bm{v}_t\rangle
= \beta\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle
+(1-\beta)\langle\nabla f(\bm{x}_t),\bm{g}_t\rangle .
\tag{8}
\]

\textbf{[Step 2]}  Because $\bm{g}_t$ is the clipped version of the raw gradient,
\[
\bm{g}_t = \frac{\nabla f(\bm{x}_t)}{\max\!\bigl(1,\frac{\|\nabla f(\bm{x}_t)\|}{C}\bigr)} .
\]
Hence
\[
\langle\nabla f(\bm{x}_t),\bm{g}_t\rangle
= \frac{\|\nabla f(\bm{x}_t)\|^{2}}{\max\!\bigl(1,\frac{\|\nabla f(\bm{x}_t)\|}{C}\bigr)}
\ge \frac{\|\nabla f(\bm{x}_t)\|^{2}}{1+\frac{\|\nabla f(\bm{x}_t)\|}{C}}
\ge \frac{1}{2}\|\nabla f(\bm{x}_t)\|^{2},
\tag{9}
\]
where the last inequality uses $\frac{a}{1+a/C}\ge \frac{a}{2}$ for all $a\ge0$ (since $a/(1+a/C)\ge a/2$ when $a\le C$ and also when $a>C$).

\textbf{[Step 3]}  Substitute (9) into (8):
\[
\langle\nabla f(\bm{x}_t),\bm{v}_t\rangle
\ge \beta\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle
+\frac{1-\beta}{2}\|\nabla f(\bm{x}_t)\|^{2}.
\tag{10}
\]

\textbf{[Step 4]}  Insert (10) into the descent inequality (7):
\[
f(\bm{x}_{t+1})
\le f(\bm{x}_t)
-\eta\beta\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle
-\frac{\eta(1-\beta)}{2}\|\nabla f(\bm{x}_t)\|^{2}
+\frac{L\eta^{2}}{2}\|\bm{v}_t\|^{2}.
\tag{11}
\]

\textbf{[Step 5]}  Rearrange (11) to isolate the term $-\frac{\eta(1-\beta)}{2}\|\nabla f(\bm{x}_t)\|^{2}$ on the left:
\[
\frac{\eta(1-\beta)}{2}\|\nabla f(\bm{x}_t)\|^{2}
\le f(\bm{x}_t)-f(\bm{x}_{t+1})
-\eta\beta\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle
+\frac{L\eta^{2}}{2}\|\bm{v}_t\|^{2}.
\tag{12}
\]

\textbf{[Step 6]}  Sum (12) over $t=0,\dots,T-1$. The telescoping sum of the first difference yields
\[
\sum_{t=0}^{T-1}\bigl[f(\bm{x}_t)-f(\bm{x}_{t+1})\bigr]
= f(\bm{x}_0)-f(\bm{x}_T)
\le f(\bm{x}_0)-f^\* .
\tag{13}
\]

\textbf{[Step 7]}  For the momentum inner‑product term, use Cauchy–Schwarz and Lemma \ref{lem:momentumbound}:
\[
\bigl|\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle\bigr|
\le \|\nabla f(\bm{x}_t)\|\,\|\bm{v}_{t-1}\|
\le C\|\nabla f(\bm{x}_t)\|.
\tag{14}
\]
Apply Young’s inequality $ab\le \frac{a^{2}}{2\lambda}+\frac{\lambda b^{2}}{2}$ with $\lambda=\frac{1-\beta}{2\beta}$ (which is positive because $\beta\in[0,1)$):
\[
\eta\beta\bigl|\langle\nabla f(\bm{x}_t),\bm{v}_{t-1}\rangle\bigr|
\le \frac{\eta(1-\beta)}{4}\|\nabla f(\bm{x}_t)\|^{2}
+\frac{2\eta\beta^{2}C^{2}}{1-\beta}.
\tag{15}
\]

\textbf{[Step 8]}  Bound the last term in (12) using Lemma \ref{lem:momentumbound}:
\[
\frac{L\eta^{2}}{2}\|\bm{v}_t\|^{2}\le \frac{L\eta^{2}}{2}C^{2}.
\tag{16}
\]

\textbf{[Step 9]}  Plug (15) and (16) into the summed version of (12). After summation we obtain
\[
\frac{\eta(1-\beta)}{2}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le f(\bm{x}_0)-f^\*
+\sum_{t=0}^{T-1}\Bigl[\frac{\eta(1-\beta)}{4}\|\nabla f(\bm{x}_t)\|^{2}
+\frac{2\eta\beta^{2}C^{2}}{1-\beta}
+\frac{L\eta^{2}}{2}C^{2}\Bigr].
\tag{17}
\]

\textbf{[Step 10]}  Move the $\frac{\eta(1-\beta)}{4}$ term from the right‑hand side to the left:
\[
\frac{\eta(1-\beta)}{4}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le f(\bm{x}_0)-f^\*
+T\Bigl(\frac{2\eta\beta^{2}C^{2}}{1-\beta}
+\frac{L\eta^{2}}{2}C^{2}\Bigr).
\tag{18}
\]

\textbf{[Step 11]}  Divide both sides of (18) by $\frac{\eta(1-\beta)}{4}T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le
\frac{4\bigl(f(\bm{x}_0)-f^\*\bigr)}{\eta(1-\beta)T}
+\frac{8\beta^{2}C^{2}}{(1-\beta)^{2}}
+ \frac{2L\eta C^{2}}{1-\beta}.
\tag{19}
\]

\textbf{[Step 12]}  Since $\beta\in[0,1)$, the term $\frac{8\beta^{2}}{(1-\beta)^{2}}$ can be bounded by a constant (e.g. by $8/(1-\beta)^{2}$). To obtain the cleaner bound stated in Theorem \ref{thm:main}, we use the stepsize condition (A5): $\eta\le\frac{1-\beta}{L}$. Hence
\[
\frac{2L\eta C^{2}}{1-\beta}\le L\eta C^{2}.
\tag{20}
\]
Moreover, dropping the non‑negative constant $\frac{8\beta^{2}C^{2}}{(1-\beta)^{2}}$ yields an inequality that still holds:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le
\frac{2\bigl(f(\bm{x}_0)-f^\*\bigr)}{\eta(1-\beta)T}
+L\eta C^{2}.
\tag{21}
\]

Equation (21) is exactly bound (4) of Theorem \ref{thm:main}. \hfill $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
From (21) choose $\eta = \frac{1-\beta}{L\sqrt{T}}$, which respects (A5). Substituting,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le
\frac{2L\sqrt{T}\bigl(f(\bm{x}_0)-f^\*\bigr)}{(1-\beta)^{2}T}
+ \frac{(1-\beta)}{\sqrt{T}} C^{2}
= O\!\bigl(T^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{No clipping ($C=\infty$).}  The bound reduces to the classic result for momentum SGD:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le \frac{2\bigl(f(\bm{x}_0)-f^\*\bigr)}{\eta(1-\beta)T},
\]
which vanishes at rate $O(1/T)$ if a diminishing stepsize $\eta_t=O(1/t)$ is used.
\item \textbf{Pure gradient descent ($\beta=0$).}  The algorithm becomes clipped GD.  Bound (4) simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\bm{x}_t)\|^{2}
\le \frac{2\bigl(f(\bm{x}_0)-f^\*\bigr)}{\eta T}+L\eta C^{2}.
\]
\item \textbf{Deterministic vs. stochastic gradients.}  If $g_t$ is a stochastic unbiased estimator of the clipped gradient with bounded variance $\sigma^{2}$, an extra term $\frac{L\eta\sigma^{2}}{1-\beta}$ appears on the right‑hand side of (21). The proof proceeds identically after taking expectations.
\end{enumerate}
