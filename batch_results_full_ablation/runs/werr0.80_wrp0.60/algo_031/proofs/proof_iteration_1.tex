\section*{Frank–Wolfe (Conditional Gradient) Algorithm}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$‑smooth function and let 
\[
\mathcal C\subset\R^{d}
\]
be a non‑empty compact convex set (the feasible domain).  
Denote by $x^{*}\in\arg\min_{x\in\mathcal C}f(x)$ a global minimizer and by
$D:=\max_{x,y\in\mathcal C}\|x-y\|$ the diameter of $\mathcal C$.

For a given stepsize sequence $\{\gamma_{k}\}_{k\ge 0}\subset(0,1]$ the
Frank–Wolfe iteration is

\[
\boxed{
\begin{aligned}
s_{k}      &:= \arg\min_{s\in\mathcal C}\,\langle \nabla f(x_{k}),\,s\rangle,
\\[4pt]
x_{k+1}    &:= (1-\gamma_{k})\,x_{k}+\gamma_{k}\,s_{k}.
\end{aligned}}
\tag{FW}
\]

A common choice is the \emph{open‑loop} schedule $\gamma_{k}=2/(k+2)$,
which yields the optimal $O(1/k)$ rate for smooth convex objectives.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe (Conditional Gradient)}\label{alg:FW}
\begin{algorithmic}[1]
\Require Convex $L$‑smooth $f$, feasible set $\mathcal C$, stepsize rule $\{\gamma_k\}$,
initial point $x_{0}\in\mathcal C$, maximum iterations $K$.
\State $k\gets 0$
\While{$k<K$}
    \State Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \State \textbf{Linear minimization oracle (LMO):}
          \[
          s_{k}\gets \arg\min_{s\in\mathcal C}\langle g_{k},\,s\rangle .
          \]
    \State Choose stepsize $\gamma_{k}\in(0,1]$ (e.g. $\gamma_{k}=2/(k+2)$).
    \State Update $x_{k+1}\gets (1-\gamma_{k})x_{k}+\gamma_{k}s_{k}$.
    \State $k\gets k+1$.
\EndWhile
\State \Return $x_{K}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate problem
\[
f(w)=(w-3)^{2},\qquad 
\mathcal C=[0,5],\qquad 
w_{0}=0,\qquad 
\gamma_{k}=0.1\;\;(\forall k).
\]

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item \textbf{Iteration $0$:}
\[
\nabla f(w_{0})=2(w_{0}-3)=-6.
\]
LMO:
\[
s_{0}= \arg\min_{s\in[0,5]}\langle -6,\,s\rangle
      =\arg\min_{s\in[0,5]}(-6s)=5.
\]
Update:
\[
w_{1}= (1-0.1)w_{0}+0.1\,s_{0}=0.9\cdot0+0.1\cdot5=0.5.
\]
Objective:
\[
f(w_{1})=(0.5-3)^{2}=6.25.
\]

\item \textbf{Iteration $1$:}
\[
\nabla f(w_{1})=2(0.5-3)=-5.
\]
LMO:
\[
s_{1}= \arg\min_{s\in[0,5]}(-5s)=5.
\]
Update:
\[
w_{2}=0.9\cdot0.5+0.1\cdot5=0.45+0.5=0.95.
\]
Objective:
\[
f(w_{2})=(0.95-3)^{2}=4.2025.
\]

\item \textbf{Iteration $2$:}
\[
\nabla f(w_{2})=2(0.95-3)=-4.1.
\]
LMO:
\[
s_{2}= \arg\min_{s\in[0,5]}(-4.1s)=5.
\]
Update:
\[
w_{3}=0.9\cdot0.95+0.1\cdot5=0.855+0.5=1.355.
\]
Objective:
\[
f(w_{3})=(1.355-3)^{2}\approx 2.714.
\]
\end{enumerate}
The sequence $\{w_{k}\}$ moves toward the optimum $w^{*}=3$ while the
objective values decrease, illustrating the mechanics of the
Frank–Wolfe iteration.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex on $\R^{d}$, i.e.
\[
f(y)\ge f(x)+\langle\nabla f(x),\,y-x\rangle,\quad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth on $\mathcal C$, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathcal C.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{1}
\]
\end{assumption}
\begin{assumption}[Compact feasible set]\label{ass:compact}
$\mathcal C$ is compact and convex with diameter
\[
D:=\max_{x,y\in\mathcal C}\|x-y\|<\infty .
\]
\end{assumption}
\begin{assumption}[Exact linear minimization oracle]\label{ass:lmo}
At each iteration the subproblem
\[
s_{k}=\arg\min_{s\in\mathcal C}\langle\nabla f(x_{k}),s\rangle
\]
is solved exactly.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal gap $O(1/k)$]\label{thm:rate}
Let Assumptions \ref{ass:convex}–\ref{ass:lmo} hold and
choose the open‑loop stepsize $\gamma_{k}=2/(k+2)$.  Denote
$x^{*}\in\arg\min_{x\in\mathcal C}f(x)$.  Then for all $k\ge 0$
\[
f(x_{k})-f(x^{*})\;\le\;\frac{2\,L\,D^{2}}{k+2}.
\tag{2}
\]
Consequently $\displaystyle\lim_{k\to\infty}f(x_{k})=f(x^{*})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The update is a convex combination of two feasible
points, therefore $x_{k}\in\mathcal C$ for all $k$.
\item \textbf{Hyper‑parameter sensitivity.}
The rate (2) depends only on $L$ and $D$, not on the exact choice of the
open‑loop schedule as long as $\gamma_{k}=O(1/k)$.  Larger constant stepsizes
yield a larger bound but still guarantee convergence.
\item \textbf{Comparison with projected gradient descent.}
Projected GD attains the same $O(1/k)$ rate under identical assumptions but
requires a projection onto $\mathcal C$, which may be computationally more
expensive than the linear minimization oracle used here.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Curvature constant]\label{def:curv}
The curvature of $f$ over $\mathcal C$ is
\[
C_{f}:=
\sup_{\substack{x,s\in\mathcal C\\ \gamma\in[0,1]}}
\frac{2}{\gamma^{2}}\Bigl(
f\bigl(x+\gamma(s-x)\bigr)-f(x)-\gamma\langle\nabla f(x),s-x\rangle
\Bigr).
\]
\end{definition}
\begin{lemma}[Curvature bound via smoothness]\label{lem:curv}
If $f$ is $L$‑smooth on $\mathcal C$, then $C_{f}\le L D^{2}$.
\end{lemma}
\begin{proof}
For any $x,s\in\mathcal C$ and $\gamma\in[0,1]$, set $y=x+\gamma(s-x)$.
By smoothness inequality (1):
\[
f(y)\le f(x)+\gamma\langle\nabla f(x),s-x\rangle
       +\frac{L}{2}\|y-x\|^{2}
     = f(x)+\gamma\langle\nabla f(x),s-x\rangle
       +\frac{L}{2}\gamma^{2}\|s-x\|^{2}.
\]
Rearranging,
\[
\frac{2}{\gamma^{2}}\!\Bigl(f(y)-f(x)-\gamma\langle\nabla f(x),s-x\rangle\Bigr)
\le L\|s-x\|^{2}\le L D^{2}.
\]
Taking the supremum yields $C_{f}\le L D^{2}$.\qedhere
\end{proof}
\begin{lemma}[Descent Lemma for Frank–Wolfe]\label{lem:descent}
For any $k\ge0$,
\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}\,
\underbrace{\bigl\langle\nabla f(x_{k}),x_{k}-s_{k}\bigr\rangle}_{\displaystyle\text{FW gap}}
+\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Apply the definition of curvature with $x=x_{k}$, $s=s_{k}$ and the chosen
$\gamma=\gamma_{k}$:
\[
f(x_{k+1})\le f(x_{k})+\gamma_{k}\langle\nabla f(x_{k}),s_{k}-x_{k}\rangle
               +\frac{\gamma_{k}^{2}}{2}C_{f}.
\]
Since $\langle\nabla f(x_{k}),s_{k}-x_{k}\rangle
      =-\langle\nabla f(x_{k}),x_{k}-s_{k}\rangle$, (3) follows.\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  Throughout the proof we use the notation
\[
\Delta_{k}:=f(x_{k})-f(x^{*})\ge0,\qquad
g_{k}:=\langle\nabla f(x_{k}),x_{k}-s_{k}\rangle\ge0
\]
(the Frank–Wolfe duality gap).

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{Upper bound on $g_{k}$.}  
By convexity of $f$ (Assumption \ref{ass:convex}) evaluated at $x^{*}$:
\[
f(x^{*})\ge f(x_{k})+\langle\nabla f(x_{k}),x^{*}-x_{k}\rangle.
\]
Rearranging,
\[
\langle\nabla f(x_{k}),x_{k}-x^{*}\rangle\ge \Delta_{k}.
\tag{4}
\]
Because $s_{k}$ solves the linear minimization oracle,
\[
\langle\nabla f(x_{k}),s_{k}\rangle
   \le \langle\nabla f(x_{k}),x^{*}\rangle.
\]
Thus,
\[
\langle\nabla f(x_{k}),x_{k}-s_{k}\rangle
   \ge \langle\nabla f(x_{k}),x_{k}-x^{*}\rangle
   \ge \Delta_{k}.
\tag{5}
\]
Hence $g_{k}\ge\Delta_{k}$.

\item \textbf{One‑step progress.}  
Insert (5) into Lemma \ref{lem:descent} (3):
\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}\Delta_{k}
            +\frac{\gamma_{k}^{2}}{2}C_{f}.
\]
Equivalently,
\[
\Delta_{k+1}\le (1-\gamma_{k})\Delta_{k}
               +\frac{\gamma_{k}^{2}}{2}C_{f}.
\tag{6}
\]

\item \textbf{Choice of stepsize.}  
Set $\displaystyle\gamma_{k}:=\frac{2}{k+2}$.
Observe that $0<\gamma_{k}\le1$ for all $k\ge0$.
Compute
\[
1-\gamma_{k}=1-\frac{2}{k+2}= \frac{k}{k+2},
\qquad
\gamma_{k}^{2}= \frac{4}{(k+2)^{2}}.
\tag{7}
\]

\item \textbf{Recursive inequality.}  
Plug (7) into (6):
\[
\Delta_{k+1}\le
\frac{k}{k+2}\Delta_{k}
+\frac{2}{(k+2)^{2}}C_{f}.
\tag{8}
\]

\item \textbf{Induction hypothesis.}  
We claim that for every $k\ge0$,
\[
\Delta_{k}\le \frac{2C_{f}}{k+2}.
\tag{9}
\]
The base case $k=0$:
\[
\Delta_{0}=f(x_{0})-f(x^{*})\le \frac{2C_{f}}{2}=C_{f},
\]
which holds because $C_{f}\ge0$.

\item \textbf{Inductive step.}  
Assume (9) holds for some $k\ge0$.  Using (8):
\[
\Delta_{k+1}\le
\frac{k}{k+2}\cdot\frac{2C_{f}}{k+2}
+\frac{2C_{f}}{(k+2)^{2}}
=
\frac{2C_{f}}{(k+2)^{2}}\bigl(k+1\bigr)
=
\frac{2C_{f}}{(k+1)+2}.
\]
Thus (9) also holds for $k+1$.

\item \textbf{Conclusion.}  
By induction, (9) is valid for all $k\ge0$.  Using Lemma \ref{lem:curv}
to replace $C_{f}$ with $L D^{2}$ we obtain
\[
\Delta_{k}=f(x_{k})-f(x^{*})
\le \frac{2\,L\,D^{2}}{k+2},
\]
which is exactly the bound (2) of Theorem \ref{thm:rate}.  $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The proof above yields the explicit constant $2L D^{2}$.
If a line‑search stepsize $\gamma_{k}= \arg\min_{\gamma\in[0,1]}
f\bigl((1-\gamma)x_{k}+\gamma s_{k}\bigr)$ is used, the same recursion
(6) holds with $\gamma_{k}$ replaced by the optimal $\gamma_{k}^{\star}$,
and one can show $\gamma_{k}^{\star}\ge \frac{\Delta_{k}}{C_{f}}$,
which leads to the slightly tighter bound
\[
\Delta_{k}\le\frac{2C_{f}}{k+2}
\quad\text{with }C_{f}\le L D^{2}.
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Strongly convex $f$.}  
If $f$ is $\mu$‑strongly convex, the primal gap decays linearly:
\[
\Delta_{k}\le\Bigl(1-\frac{\mu}{L}\Bigr)^{k}\Delta_{0}.
\]
The proof follows by replacing (5) with $g_{k}\ge\mu\|x_{k}-x^{*}\|^{2}$
and using $C_{f}\le L D^{2}$.

\item \textbf{Polyhedral feasible set.}  
When $\mathcal C$ is a polytope, the Frank–Wolfe algorithm enjoys the
\emph{away‑step} variant with a linear convergence rate
$O((1-\rho)^{k})$, where $\rho>0$ depends on the pyramidal width of
$\mathcal C$.

\item \textbf{Stochastic gradients.}  
If only an unbiased estimator $\tilde g_{k}$ of $\nabla f(x_{k})$ is
available, the same recursion holds in expectation, yielding an
$O(1/\sqrt{k})$ rate for the expected primal gap under bounded variance.
\end{enumerate}

\section*{Discussion}
The Frank–Wolfe algorithm trades the projection step of projected
gradient methods for a linear minimization oracle.  Under the minimal
assumptions of convexity, $L$‑smoothness, and compactness, the algorithm
guarantees a sublinear $O(1/k)$ decrease of the optimality gap with an
explicit constant $2L D^{2}$.  The proof relies only on elementary convex
analysis (convexity, smoothness) and the curvature definition; no
probabilistic arguments are needed.  This makes the method especially
attractive for large‑scale problems where the LMO is cheap (e.g.,
trace‑norm balls, simplex, or flow polytopes) while projections are
expensive.
