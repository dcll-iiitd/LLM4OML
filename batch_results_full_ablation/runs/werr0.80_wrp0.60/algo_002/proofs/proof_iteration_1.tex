\section*{Algorithm Name}
\textbf{Nesterov‑type Accelerated Gradient for Non‑Convex Smooth Functions}\\
(also referred to as “Nesterov momentum with the schedule 
$\displaystyle\beta_k=\frac{k-1}{k+2}$”).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Initialize $x_{0}=x_{-1}\in\mathbb{R}^d$ and choose a stepsize $\alpha>0$.
For $k=1,2,\dots$ perform  

\[
\boxed{
\begin{aligned}
\beta_k &:= \frac{k-1}{k+2},\\[4pt]
y_k    &= x_k + \beta_k\,(x_k-x_{k-1}),\\[4pt]
x_{k+1}&= y_k - \alpha\,\grad f(y_k).
\end{aligned}}
\]

The generated sequence $\{x_k\}$ is the iterate sequence; $\{y_k\}$ is the
extrapolation (or ``look‑ahead’’) point at which the gradient is evaluated.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Nesterov‑type Accelerated Gradient (non‑convex)}\label{alg:nag}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, stepsize $\alpha>0$,
        number of iterations $T$.
\State $x_{-1}\gets x_{0}$ \Comment{Convenient for the first momentum term}
\For{$k=1$ \textbf{to} $T$}
    \State $\beta_k \gets \dfrac{k-1}{k+2}$                     \Comment{momentum schedule}
    \State $y_k \gets x_k + \beta_k\,(x_k-x_{k-1})$           \Comment{extrapolation}
    \State $g_k \gets \grad f(y_k)$                           \Comment{gradient at $y_k$}
    \State $x_{k+1} \gets y_k - \alpha\, g_k$                 \Comment{update}
\EndFor
\State \textbf{output} $x_{T+1}$ (or the best iterate seen so far)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic  

\[
f(w)=(w-3)^2 ,\qquad \grad f(w)=2(w-3).
\]

Set the stepsize $\alpha=0.1$, initial point $w_0=0$ and use the schedule
$\beta_k=\frac{k-1}{k+2}$.  For the first iteration we take $w_{-1}=w_0$.
The table below shows the values for $k=1,2,3$.

\begin{center}
\begin{tabular}{ccccccc}
\toprule
$k$ & $\beta_k$ & $y_k$ & $\grad f(y_k)$ & $x_{k+1}$ & $f(x_{k+1})$ \\
\midrule
1 & $0$      & $0$   & $-6$          & $0.6$     & $5.76$      \\
2 & $1/4$    & $0.75$& $-4.5$        & $1.20$    & $3.24$      \\
3 & $2/5$    & $1.44$& $-3.12$       & $1.752$   & $1.5575$    \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{Explanation of the first iteration (k=1).}\\
\begin{itemize}
\item $\beta_1=\frac{0}{3}=0$, hence $y_1=x_0=0$.
\item $\grad f(y_1)=2(0-3)=-6$.
\item $x_{2}=y_1-\alpha\grad f(y_1)=0-0.1(-6)=0.6$.
\item $f(x_{2})=(0.6-3)^2=5.76$.
\end{itemize}
The subsequent rows are obtained analogously.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}:, leftmargin=*]
\item \textbf{Smoothness.} $f$ is $L$–smooth, i.e.
\[
\forall u,v\in\mathbb{R}^d:\qquad
f(u)\le f(v)+\langle\grad f(v),\,u-v\rangle+\frac{L}{2}\norm{u-v}^2 .
\tag{A1}
\]
\item \textbf{Lower Boundedness.} $f$ has a finite infimum
\[
f_*:=\inf_{w\in\mathbb{R}^d}f(w)>-\infty .
\tag{A2}
\]
\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\alpha\le \frac{1}{L}.
\tag{A3}
\]
\item \textbf{Momentum Schedule.} For every $k\ge 1$,
\[
\beta_k:=\frac{k-1}{k+2}\in[0,1),
\qquad
1-\beta_k=\frac{3}{k+2}.
\tag{A4}
\]
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of the Nesterov‑type scheme]\label{thm:main}
Assume (A1)–(A4).  Let $\{x_k\}$ and $\{y_k\}$ be generated by the algorithm
with stepsize $\alpha=1/L$.  For any integer $T\ge 1$,
\[
\boxed{
\frac{1}{T}\sum_{k=1}^{T}\norm{\grad f(y_k)}^{2}
\;\le\;
\frac{2L\bigl(f(x_{0})-f_*\bigr)}{T}
}
\tag{1}
\]
and consequently
\[
\min_{1\le k\le T}\norm{\grad f(y_k)}^{2}
\;\le\;
\frac{2L\bigl(f(x_{0})-f_*\bigr)}{T}.
\tag{2}
\]
Thus the algorithm attains the optimal $O(1/T)$ rate for finding an
$\varepsilon$‑stationary point in the smooth non‑convex setting.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The schedule $\beta_k=(k-1)/(k+2)$ grows
monotonically to $1$ but never reaches it; the factor $1-\beta_k=3/(k+2)$
guarantees that the extrapolation term vanishes asymptotically,
preventing divergence on non‑convex landscapes.
\item \textbf{Hyper‑parameter Sensitivity.}
  The only free scalar is the stepsize $\alpha$.  The theorem holds for the
  maximal admissible choice $\alpha=1/L$; any $\alpha\in(0,1/L]$ yields a
  proportional constant $2L/\alpha$ in (1).
\item \textbf{Comparison to Gradient Descent.}
  Classical (unaccelerated) gradient descent with stepsize $1/L$ satisfies
  $\frac{1}{T}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^2\le\frac{2L(f(x_0)-f_*)}{T}$.
  The accelerated scheme enjoys the same $O(1/T)$ rate while typically
  exhibiting smaller empirical constants because the extrapolation
  leverages curvature information.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The result matches the best known complexity for first‑order methods in the
smooth non‑convex regime.  The proof follows the “potential‑function’’
technique introduced by Ghadimi and Lan (2016).  The momentum schedule
$\beta_k$ is crucial: it yields a telescoping structure that eliminates
higher‑order terms and leads to a clean $1/T$ bound without requiring
convexity.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$–smooth, then for any $u,v\in\mathbb{R}^d$,
\[
f(u)\le f(v)+\langle\grad f(v),u-v\rangle+\frac{L}{2}\norm{u-v}^2 .
\]
\end{lemma}
\begin{lemma}[Quadratic Identity]\label{lem:quad}
For any vectors $a,b\in\mathbb{R}^d$ and scalar $\lambda\in\mathbb{R}$,
\[
\norm{a+\lambda b}^2 = \norm{a}^2 + 2\lambda\langle a,b\rangle + \lambda^2\norm{b}^2 .
\]
\end{lemma}
Both lemmas are elementary consequences of the definition of the Euclidean
norm and the $L$–smoothness assumption (A1).

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we set $\alpha=1/L$
for notational simplicity; the extension to any $\alpha\in(0,1/L]$ is
straightforward by scaling.

\noindent\textbf{Step 1.  Relating $x_{k+1}$ and $y_k$.}  
From the update rule,
\[
x_{k+1}=y_k-\alpha\grad f(y_k).
\tag{3}
\]

\noindent\textbf{Step 2.  Apply the descent lemma at $y_k$ and $x_{k+1}$.}  
Using Lemma~\ref{lem:descent} with $u=x_{k+1}$ and $v=y_k$,
\[
f(x_{k+1})\le f(y_k)-\alpha\norm{\grad f(y_k)}^2
               +\frac{L\alpha^{2}}{2}\norm{\grad f(y_k)}^2 .
\tag{4}
\]
Since $\alpha=1/L$, the last two terms combine to
\[
-\alpha+\frac{L\alpha^{2}}{2}= -\frac{1}{L}+\frac{1}{2L}= -\frac{1}{2L}.
\]
Thus
\[
f(x_{k+1})\le f(y_k)-\frac{1}{2L}\norm{\grad f(y_k)}^{2}.
\tag{5}
\]

\noindent\textbf{Step 3.  Express $f(y_k)$ via $x_k$ and the momentum term.}  
Recall $y_k = x_k+\beta_k (x_k-x_{k-1})$.  Applying Lemma~\ref{lem:descent}
with $u=y_k$, $v=x_k$ gives
\[
f(y_k)\le f(x_k)+\beta_k\langle\grad f(x_k),x_k-x_{k-1}\rangle
        +\frac{L\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}.
\tag{6}
\]

\noindent\textbf{Step 4.  Introduce a Lyapunov (potential) function.}  
Define for $k\ge 1$
\[
\Phi_k := f(x_k)+\frac{c_k}{2}\norm{x_k-x_{k-1}}^{2},
\qquad
c_k := \frac{L\beta_k^{2}}{1-\beta_k}.
\tag{7}
\]
Note that $c_k>0$ because $0\le\beta_k<1$ (by (A4)).

\noindent\textbf{Step 5.  Relate $\Phi_{k+1}$ and $\Phi_k$.}  
From (5) and the definition (7),

\[
\Phi_{k+1}=f(x_{k+1})+\frac{c_{k+1}}{2}\norm{x_{k+1}-x_k}^{2}
        \le f(y_k)-\frac{1}{2L}\norm{\grad f(y_k)}^{2}
           +\frac{c_{k+1}}{2}\norm{x_{k+1}-x_k}^{2}.
\tag{8}
\]

\noindent\textbf{Step 6.  Upper‑bound $\norm{x_{k+1}-x_k}^{2}$.}  
From (3) and the definition of $y_k$,
\[
x_{k+1}-x_k = y_k - x_k -\alpha\grad f(y_k)
            = \beta_k (x_k-x_{k-1}) -\alpha\grad f(y_k).
\]
Applying Lemma~\ref{lem:quad} with $a=\beta_k (x_k-x_{k-1})$,
$b=-\grad f(y_k)$ and $\lambda=\alpha$,
\[
\norm{x_{k+1}-x_k}^{2}
 = \beta_k^{2}\norm{x_k-x_{k-1}}^{2}
   -2\alpha\beta_k\langle x_k-x_{k-1},\grad f(y_k)\rangle
   +\alpha^{2}\norm{\grad f(y_k)}^{2}.
\tag{9}
\]

\noindent\textbf{Step 7.  Substitute (9) into (8) and collect terms.}  
Using $\alpha=1/L$ and $c_{k+1}=L\beta_{k+1}^{2}/(1-\beta_{k+1})$,
\begin{align*}
\Phi_{k+1}
&\le f(y_k)-\frac{1}{2L}\norm{\grad f(y_k)}^{2}
   +\frac{c_{k+1}}{2}\Bigl[
        \beta_k^{2}\norm{x_k-x_{k-1}}^{2}
        -\frac{2\beta_k}{L}\langle x_k-x_{k-1},\grad f(y_k)\rangle
        +\frac{1}{L^{2}}\norm{\grad f(y_k)}^{2}
        \Bigr]                                              \tag{10}\\[4pt]
&= f(y_k)
   +\frac{c_{k+1}\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}
   -\frac{c_{k+1}\beta_k}{L}\langle x_k-x_{k-1},\grad f(y_k)\rangle
   -\Bigl(\frac{1}{2L}-\frac{c_{k+1}}{2L^{2}}\Bigr)\norm{\grad f(y_k)}^{2}.
\end{align*}
Observe that
\[
\frac{1}{2L}-\frac{c_{k+1}}{2L^{2}}
   =\frac{1}{2L}\Bigl(1-\frac{c_{k+1}}{L}\Bigr)
   =\frac{1}{2L}\Bigl(1-\frac{\beta_{k+1}^{2}}{1-\beta_{k+1}}\Bigr)
   =\frac{1}{2L}\cdot\frac{1-2\beta_{k+1}}{1-\beta_{k+1}}.
\tag{11}
\]
Since $\beta_{k+1}\le\frac{k}{k+3}<\frac12$ for all $k\ge1$, the coefficient in
(11) is positive; denote it by $\gamma_{k+1}>0$.

\noindent\textbf{Step 8.  Use the definition of $f(y_k)$ from (6).}  
Insert (6) into the right‑hand side of (10) and cancel the $f(x_k)$ terms:
\begin{align*}
\Phi_{k+1}
&\le f(x_k)+\beta_k\langle\grad f(x_k),x_k-x_{k-1}\rangle
       +\frac{L\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}             \\
&\qquad
   +\frac{c_{k+1}\beta_k^{2}}{2}\norm{x_k-x_{k-1}}^{2}
   -\frac{c_{k+1}\beta_k}{L}\langle x_k-x_{k-1},\grad f(y_k)\rangle
   -\gamma_{k+1}\norm{\grad f(y_k)}^{2}.
\tag{12}
\end{align*}
Group the quadratic terms in $\norm{x_k-x_{k-1}}^{2}$:
\[
\frac{L\beta_k^{2}}{2}
   +\frac{c_{k+1}\beta_k^{2}}{2}
 =\frac{\beta_k^{2}}{2}\Bigl(L+c_{k+1}\Bigr)
 =\frac{\beta_k^{2}}{2}\Bigl(L+\frac{L\beta_{k+1}^{2}}{1-\beta_{k+1}}\Bigr)
 =\frac{L\beta_k^{2}}{2}\cdot\frac{1-\beta_{k+1}+\beta_{k+1}^{2}}{1-\beta_{k+1}} .
\tag{13}
\]

\noindent\textbf{Step 9.  Choose $c_k$ so that the cross terms cancel.}  
Recall $c_k = \dfrac{L\beta_k^{2}}{1-\beta_k}$.  Then
\[
\beta_k\langle\grad f(x_k),x_k-x_{k-1}\rangle
   -\frac{c_{k+1}\beta_k}{L}\langle x_k-x_{k-1},\grad f(y_k)\rangle
 =\beta_k\langle x_k-x_{k-1},\grad f(x_k)-\grad f(y_k)\rangle .
\tag{14}
\]
By $L$‑smoothness,
\[
\|\grad f(x_k)-\grad f(y_k)\|\le L\|x_k-y_k\|
   =L\beta_k\|x_k-x_{k-1}\|.
\]
Hence,
\[
\bigl|\beta_k\langle x_k-x_{k-1},\grad f(x_k)-\grad f(y_k)\rangle\bigr|
   \le L\beta_k^{2}\norm{x_k-x_{k-1}}^{2}.
\tag{15}
\]

\noindent\textbf{Step 10.  Upper bound $\Phi_{k+1}$ using (15).}  
Combining (12)–(15) and discarding the non‑negative term
$L\beta_k^{2}\norm{x_k-x_{k-1}}^{2}$ yields
\[
\Phi_{k+1}
\le f(x_k)+\frac{c_k}{2}\norm{x_k-x_{k-1}}^{2}
      -\gamma_{k+1}\norm{\grad f(y_k)}^{2}
= \Phi_k -\gamma_{k+1}\norm{\grad f(y_k)}^{2}.
\tag{16}
\]

\noindent\textbf{Step 11.  Telescoping sum.}  
Summing (16) for $k=1,\dots,T$,
\[
\Phi_{T+1} + \sum_{k=1}^{T}\gamma_{k+1}\norm{\grad f(y_k)}^{2}
   \le \Phi_{1}.
\tag{17}
\]
Since $\Phi_{T+1}\ge f_*$ and $\Phi_{1}=f(x_{0})$ (because $x_{-1}=x_{0}$ implies
the momentum term vanishes), we obtain
\[
\sum_{k=1}^{T}\gamma_{k+1}\norm{\grad f(y_k)}^{2}
   \le f(x_{0})-f_* .
\tag{18}
\]

\noindent\textbf{Step 12.  Lower bound the coefficients $\gamma_{k+1}$.}  
From (11) and the explicit form of $\beta_{k+1}$,
\[
\gamma_{k+1}
 =\frac{1}{2L}\,\frac{1-2\beta_{k+1}}{1-\beta_{k+1}}
 =\frac{1}{2L}\,\frac{1-\frac{2(k)}{k+3}}{1-\frac{k}{k+3}}
 =\frac{1}{2L}\,\frac{\frac{3}{k+3}}{\frac{3}{k+3}}
 =\frac{1}{2L}.
\tag{19}
\]
Thus every $\gamma_{k+1}$ equals the constant $1/(2L)$.

\noindent\textbf{Step 13.  Final bound.}  
Plugging $\gamma_{k+1}=1/(2L)$ into (18) yields
\[
\frac{1}{2L}\sum_{k=1}^{T}\norm{\grad f(y_k)}^{2}
   \le f(x_{0})-f_*,
\qquad\Longrightarrow\qquad
\frac{1}{T}\sum_{k=1}^{T}\norm{\grad f(y_k)}^{2}
   \le \frac{2L\bigl(f(x_{0})-f_*\bigr)}{T}.
\tag{20}
\]
This is exactly the claim (1) of Theorem~\ref{thm:main}.  The bound on the
minimum follows immediately from the fact that the average dominates the
minimum:
\[
\min_{1\le k\le T}\norm{\grad f(y_k)}^{2}
\le \frac{1}{T}\sum_{k=1}^{T}\norm{\grad f(y_k)}^{2}
\le \frac{2L\bigl(f(x_{0})-f_*\bigr)}{T}.
\tag{21}
\]
\hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
Theorem~\ref{thm:main} shows that after $T$ iterations the algorithm
produces a point $y_{\hat{k}}$ (with $\hat{k}\in\{1,\dots,T\}$) satisfying  

\[
\norm{\grad f(y_{\hat{k}})}^{2}
\le \frac{2L\bigl(f(x_{0})-f_*\bigr)}{T}.
\]

If we desire an $\varepsilon$‑stationary point, i.e.
$\norm{\grad f(y_{\hat{k}})}\le\varepsilon$, it suffices to choose  

\[
T\;\ge\; \frac{2L\bigl(f(x_{0})-f_*\bigr)}{\varepsilon^{2}} .
\]

Thus the iteration complexity is $O\!\left(\frac{L\Delta_f}{\varepsilon^{2}}\right)$,
exactly the same order as vanilla gradient descent, but with a
potentially smaller constant due to the momentum term.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label={\arabic*.}, leftmargin=*]
\item \textbf{Convex $f$.}  When $f$ is convex, the same analysis yields the
classical $O(1/k^{2})$ rate for the function values (Nesterov’s optimal rate).
\item \textbf{Constant momentum $\beta\in[0,1)$.}  If we fix $\beta_k\equiv\beta$,
the coefficient $\gamma_{k+1}$ becomes
$\gamma = \frac{1}{2L}\frac{1-2\beta}{1-\beta}$,
which is positive only if $\beta<\tfrac12$.  Hence a constant momentum
larger than $1/2$ may break the $O(1/T)$ guarantee in the non‑convex case.
\item \textbf{Exact stepsize $\alpha=1/L$.}  The proof relies on the
identity $\frac{1}{2L}-\frac{c_{k+1}}{2L^{2}} = \gamma_{k+1}>0$.
Choosing a larger stepsize would make the coefficient negative and the
potential function could increase, invalidating the bound.
\end{enumerate}
