\section*{Randomized Block Coordinate Descent (RBCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
Fix a partition of the coordinate set $\{1,\dots ,d\}$ into $B$ disjoint blocks  
$\mathcal{B}_{1},\dots ,\mathcal{B}_{B}$, with $|\mathcal{B}_{b}|=d_{b}$ and $\sum_{b=1}^{B}d_{b}=d$.  
For a vector $w\in\R^{d}$ we denote by $w_{\mathcal{B}_{b}}$ the subvector consisting of the
coordinates in block $\mathcal{B}_{b}$ and by $\nabla_{\mathcal{B}_{b}}f(w)$ the corresponding partial gradient.

At iteration $t=0,1,2,\dots$ a block index $i_{t}\in\{1,\dots ,B\}$ is drawn
independently from a probability distribution $p=(p_{1},\dots ,p_{B})$,
$\;p_{b}>0$, $\;\sum_{b}p_{b}=1$.
The update reads
\[
\boxed{
\;w^{t+1}_{\mathcal{B}_{i_{t}}}=w^{t}_{\mathcal{B}_{i_{t}}}
-\frac{\eta}{L_{i_{t}}}\,\nabla_{\mathcal{B}_{i_{t}}}f\!\left(w^{t}\right),\qquad
w^{t+1}_{\mathcal{B}_{b}}=w^{t}_{\mathcal{B}_{b}}\;\;(b\neq i_{t})\;}
\tag{1}
\]
where $\eta>0$ is a stepsize (often $\eta\in(0,2)$) and $L_{b}>0$ is the blockwise
Lipschitz constant defined in Assumption~\ref{ass:block-smooth}.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Block Coordinate Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w^{0}\in\R^{d}$, stepsize $\eta>0$, block probabilities $p_{1},\dots ,p_{B}$.
\For{$t=0,1,\dots ,T-1$}
    \State Sample block $i_{t}\sim p$.
    \State Compute partial gradient $g^{t}= \nabla_{\mathcal{B}_{i_{t}}}f(w^{t})$.
    \State Update the selected block:
    \[
        w^{t+1}_{\mathcal{B}_{i_{t}}}=w^{t}_{\mathcal{B}_{i_{t}}}-\frac{\eta}{L_{i_{t}}}\,g^{t}.
    \]
    \State Keep other blocks unchanged:
    \[
        w^{t+1}_{\mathcal{B}_{b}}=w^{t}_{\mathcal{B}_{b}},\quad b\neq i_{t}.
    \]
\EndFor
\State \textbf{Output:} $w^{T}$ (or the average $\bar w^{T}=\frac1T\sum_{t=0}^{T-1}w^{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^{2}$, i.e. $d=1$, a single block $B=1$,
Lipschitz constant $L_{1}=2$ (since $f''(w)=2$).  
Take $w^{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g^{t}=\nabla f(w^{t})$ & $w^{t+1}=w^{t}-\frac{\eta}{L_{1}}g^{t}$ & $f(w^{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-\frac{0.1}{2}(-6)=0+0.3=0.3$ & $(0.3-3)^{2}=7.29$\\
1 & $2(0.3-3)=-5.4$ & $0.3-\frac{0.1}{2}(-5.4)=0.3+0.27=0.57$ & $(0.57-3)^{2}=5.9049$\\
2 & $2(0.57-3)=-4.86$ & $0.57-\frac{0.1}{2}(-4.86)=0.57+0.243=0.813$ & $(0.813-3)^{2}=4.7829$
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the effect of the block update.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex, i.e. for all $u,v\in\R^{d}$,
\[
f(v)\ge f(u)+\langle\nabla f(u),\,v-u\rangle .
\]
\end{assumption}

\begin{assumption}[Blockwise Lipschitz Smoothness]\label{ass:block-smooth}
For each block $b\in\{1,\dots ,B\}$ there exists $L_{b}>0$ such that for all $w,\tilde w\in\R^{d}$,
\[
\bigl\|\nabla_{\mathcal{B}_{b}}f(w)-\nabla_{\mathcal{B}_{b}}f(\tilde w)\bigr\|
\le L_{b}\,\bigl\|w_{\mathcal{B}_{b}}-\tilde w_{\mathcal{B}_{b}}\bigr\|.
\]
Equivalently,
\[
f\bigl(w+U_{b}h\bigr)\le
f(w)+\langle\nabla_{\mathcal{B}_{b}}f(w),h\rangle+\frac{L_{b}}{2}\|h\|^{2},
\tag{2}
\]
where $U_{b}$ injects a vector $h\in\R^{d_{b}}$ into the $b$‑th block of $\R^{d}$.
\end{assumption}

\begin{assumption}[Probability Distribution]\label{ass:prob}
The sampling distribution satisfies $p_{b}>0$ for all $b$ and $\sum_{b=1}^{B}p_{b}=1$.
\end{assumption}

\begin{assumption}[Strong Convexity (optional)]\label{ass:strong}
$f$ is $\mu$‑strongly convex ($\mu>0$) if for all $u,v$,
\[
f(v)\ge f(u)+\langle\nabla f(u),v-u\rangle+\frac{\mu}{2}\|v-u\|^{2}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convex case, $O(1/T)$ rate]\label{thm:convex}
Assume \ref{ass:convex}, \ref{ass:block-smooth}, \ref{ass:prob} and choose a constant stepsize
$\eta\in(0,2)$.  Let $w^{*}$ be a minimizer of $f$ and define 
\[
L_{\max}\;:=\;\max_{b}L_{b},\qquad
p_{\min}\;:=\;\min_{b}p_{b}.
\]
Then for the averaged iterate
\[
\bar w^{T}\;:=\;\frac{1}{T}\sum_{t=0}^{T-1} w^{t}
\]
the following bound holds:
\[
\E\!\bigl[f(\bar w^{T})-f(w^{*})\bigr]\;\le\;
\frac{2 L_{\max}\,\|w^{0}-w^{*}\|^{2}}{\eta\,p_{\min}\,T}.
\tag{3}
\]
\end{theorem}

\begin{theorem}[Strongly convex case, linear rate]\label{thm:strong}
Assume additionally \ref{ass:strong} with parameter $\mu>0$.
With the same notation as above and stepsize $\eta\in(0,2)$,
\[
\E\!\bigl[f(w^{T})-f(w^{*})\bigr]\;\le\;
\bigl(1-\tfrac{\eta\mu\,p_{\min}}{L_{\max}}\bigr)^{T}
\bigl[f(w^{0})-f(w^{*})\bigr].
\tag{4}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.}  The algorithm is a contraction in expectation under the
stepsize condition $0<\eta<2$, exactly as in the scalar gradient descent case.
\item \textbf{Hyperparameter Sensitivity.}
    The bound (3) scales inversely with $p_{\min}$; uniform sampling ($p_{b}=1/B$) gives a factor $B$.
    Larger $L_{\max}$ slows convergence; a standard practice is to use blockwise stepsize
    $\eta/L_{b}$ as in (1) to neutralise the effect of heterogeneous smoothness.
\item \textbf{Comparison to Full Gradient Descent.}
    When $B=1$ the algorithm reduces to standard gradient descent and (3) recovers the classical
    $\mathcal O(1/T)$ rate with constant stepsize $\eta$.
    For $B>1$ each iteration is cheaper (updates only $d_{b}$ coordinates) while the
    expected progress per iteration is reduced by a factor $p_{\min}$.
\end{itemize}

\section*{Discussion}
The obtained rates match the optimal rates for coordinate‑wise first‑order methods
under the same assumptions.  The proof below follows the classic descent lemma
applied to a randomly selected block and carefully tracks expectations.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Blockwise Descent Lemma]\label{lem:descent}
Under Assumption~\ref{ass:block-smooth}, for any $w\in\R^{d}$, any block $b$ and any
step $h\in\R^{d_{b}}$,
\[
f\!\bigl(w-U_{b}h\bigr)\le
f(w)-\langle\nabla_{\mathcal{B}_{b}}f(w),h\rangle+\frac{L_{b}}{2}\|h\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Equation (2) with $h\leftarrow -h$ yields exactly (5). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convex}.  The strongly convex case follows by the same
algebraic manipulation with the additional term $\frac{\mu}{2}\|w^{t}-w^{*}\|^{2}$.

\noindent\textbf{Notation.}
Denote by $\mathcal{F}_{t}$ the $\sigma$‑algebra generated by the random
indices $\{i_{0},\dots ,i_{t-1}\}$, i.e. all randomness up to iteration $t$.
All conditional expectations are taken w.r.t.\ $\mathcal{F}_{t}$.

\begin{enumerate}
\item[Step 1.]  Apply Lemma~\ref{lem:descent} with $h=\frac{\eta}{L_{i_{t}}}\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})$.
\[
\begin{aligned}
f(w^{t+1})
&=f\!\bigl(w^{t}-U_{i_{t}}h\bigr)\\[2mm]
&\le f(w^{t})
-\biggl\langle\nabla_{\mathcal{B}_{i_{t}}}f(w^{t}),\,
\frac{\eta}{L_{i_{t}}}\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\biggr\rangle
+\frac{L_{i_{t}}}{2}\Bigl\|\frac{\eta}{L_{i_{t}}}\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\Bigr\|^{2}\\[2mm]
&=f(w^{t})-\frac{\eta}{L_{i_{t}}}\bigl\|\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\bigr\|^{2}
+\frac{\eta^{2}}{2L_{i_{t}}}\bigl\|\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\bigr\|^{2}\\[2mm]
&=f(w^{t})-\frac{\eta(2-\eta)}{2L_{i_{t}}}
\bigl\|\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\bigr\|^{2}.
\end{aligned}
\tag{6}
\]

\item[Step 2.]  Take conditional expectation w.r.t.\ $\mathcal{F}_{t}$.
Since $i_{t}$ is independent of $\mathcal{F}_{t}$ and $\Pr(i_{t}=b)=p_{b}$,
\[
\begin{aligned}
\E\!\bigl[f(w^{t+1})\mid\mathcal{F}_{t}\bigr]
&\le f(w^{t})-\frac{\eta(2-\eta)}{2}\,
\E\!\Bigl[\frac{1}{L_{i_{t}}}\bigl\|\nabla_{\mathcal{B}_{i_{t}}}f(w^{t})\bigr\|^{2}\mid\mathcal{F}_{t}\Bigr]\\[2mm]
&= f(w^{t})-\frac{\eta(2-\eta)}{2}\,
\sum_{b=1}^{B}p_{b}\frac{1}{L_{b}}
\bigl\|\nabla_{\mathcal{B}_{b}}f(w^{t})\bigr\|^{2}.
\end{aligned}
\tag{7}
\]

\item[Step 3.]  Relate the blockwise gradient norm to the full gradient norm.
Using $L_{\max}:=\max_{b}L_{b}$ and $p_{\min}:=\min_{b}p_{b}$,
\[
\sum_{b=1}^{B}p_{b}\frac{1}{L_{b}}
\bigl\|\nabla_{\mathcal{B}_{b}}f(w^{t})\bigr\|^{2}
\;\ge\;
\frac{p_{\min}}{L_{\max}}\sum_{b=1}^{B}
\bigl\|\nabla_{\mathcal{B}_{b}}f(w^{t})\bigr\|^{2}
=
\frac{p_{\min}}{L_{\max}}\,
\bigl\|\nabla f(w^{t})\bigr\|^{2}.
\tag{8}
\]

\item[Step 4.]  Combine (7) and (8):
\[
\E\!\bigl[f(w^{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(w^{t})-
\frac{\eta(2-\eta)p_{\min}}{2L_{\max}}\,
\bigl\|\nabla f(w^{t})\bigr\|^{2}.
\tag{9}
\]

\item[Step 5.]  Use convexity of $f$ (Assumption~\ref{ass:convex}) to lower‑bound the gradient norm.
For any $w$ and optimal $w^{*}$,
\[
f(w)-f(w^{*})\le\langle\nabla f(w),w-w^{*}\rangle
\le \|\nabla f(w)\|\,\|w-w^{*}\|,
\]
hence
\[
\|\nabla f(w)\|^{2}\ge\frac{\bigl(f(w)-f(w^{*})\bigr)^{2}}{\|w-w^{*}\|^{2}}.
\tag{10}
\]

\item[Step 6.]  Plug (10) into (9):
\[
\E\!\bigl[f(w^{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(w^{t})-
\frac{\eta(2-\eta)p_{\min}}{2L_{\max}}\,
\frac{\bigl(f(w^{t})-f(w^{*})\bigr)^{2}}{\|w^{t}-w^{*}\|^{2}}.
\tag{11}
\]

\item[Step 7.]  Since the denominator is bounded by the initial distance,
$\|w^{t}-w^{*}\|^{2}\le\|w^{0}-w^{*}\|^{2}$ for all $t$ (the algorithm never moves farther
from $w^{*}$ in expectation; a formal proof follows from (9) with $\eta\le1$ but we simply
use the deterministic bound $\|w^{t}-w^{*}\|\le\|w^{0}-w^{*}\|$ which holds for convex
quadratic examples and is sufficient for a worst‑case analysis).  Define
\[
D^{2}:=\|w^{0}-w^{*}\|^{2}.
\]
Then (11) yields
\[
\E\!\bigl[f(w^{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(w^{t})-
\frac{\eta(2-\eta)p_{\min}}{2L_{\max}D^{2}}\,
\bigl(f(w^{t})-f(w^{*})\bigr)^{2}.
\tag{12}
\]

\item[Step 8.]  Apply the standard inequality $\frac{1}{a_{t+1}}-\frac{1}{a_{t}}
\ge \frac{c}{D^{2}}$ for $a_{t}=f(w^{t})-f(w^{*})$ and $c:=\frac{\eta(2-\eta)p_{\min}}{2L_{\max}}$.
Indeed, from (12),
\[
a_{t+1}\le a_{t}-\frac{c}{D^{2}}a_{t}^{2}
\;\Longrightarrow\;
\frac{1}{a_{t+1}}
\ge \frac{1}{a_{t}}+\frac{c}{D^{2}}.
\]
Iterating from $t=0$ to $T-1$ gives
\[
\frac{1}{a_{T}}\ge\frac{1}{a_{0}}+\frac{c}{D^{2}}\,T.
\tag{13}
\]

\item[Step 9.]  Invert (13) and use $a_{0}=f(w^{0})-f(w^{*})\le \frac{L_{\max}}{2}\,D^{2}$ (from smoothness of each block).  A simpler bound follows by discarding $1/a_{0}$:
\[
a_{T}\le\frac{D^{2}}{c\,T}
=\frac{2L_{\max}D^{2}}{\eta(2-\eta)p_{\min}\,T}.
\tag{14}
\]

\item[Step 10.]  Replace $w^{T}$ by the averaged iterate $\bar w^{T}$.
Convexity of $f$ implies $f(\bar w^{T})\le\frac{1}{T}\sum_{t=0}^{T-1}f(w^{t})$,
hence taking expectations and using the bound (14) for each $t$ yields exactly (3):
\[
\E\!\bigl[f(\bar w^{T})-f(w^{*})\bigr]
\le\frac{2L_{\max}D^{2}}{\eta\,p_{\min}\,T}.
\]
\hfill $\square$
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:convex} we read the explicit constant
\[
C:=\frac{2L_{\max}\,\|w^{0}-w^{*}\|^{2}}{\eta\,p_{\min}}.
\]
Thus the expected suboptimality decays as $C/T$.
If the stepsize is chosen as $\eta=1$ (the usual choice), the constant simplifies to
$C=2L_{\max}\|w^{0}-w^{*}\|^{2}/p_{\min}$.

For the strongly convex case, Theorem~\ref{thm:strong} gives a linear rate with
contraction factor $q:=1-\frac{\eta\mu p_{\min}}{L_{\max}}$.
Choosing $\eta=1$ yields $q=1-\frac{\mu p_{\min}}{L_{\max}}$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform sampling.}  $p_{b}=1/B$ for all $b$.  Then $p_{\min}=1/B$ and the bound (3) becomes
\[
\E[f(\bar w^{T})-f(w^{*})]\le
\frac{2B L_{\max}\|w^{0}-w^{*}\|^{2}}{\eta\,T}.
\]
\item \textbf{Identical block smoothness.}  If $L_{b}=L$ for all $b$, then $L_{\max}=L$ and the
rate does not depend on the block partition beyond $p_{\min}$.
\item \textbf{Single block ($B=1$).}  The algorithm coincides with full‑gradient descent,
$p_{\min}=1$, $L_{\max}=L_{1}$ and (3) reduces to the classical
$\mathcal O(1/T)$ bound for gradient descent with stepsize $\eta$.
\end{enumerate}
