\section*{Algorithm Name}
\textbf{Online Gradient Descent (OGD)} for Online Convex Optimization (OCO).

The learner faces a sequence of convex loss functions $f_1, f_2, \ldots, f_T$ revealed
one at a time. At round $t$ the learner picks $w_t$, then observes $f_t$ (or a
(sub)gradient $g_t \in \partial f_t(w_t)$) and suffers loss $f_t(w_t)$. The aim is to
minimize the \emph{regret} relative to the best fixed decision in hindsight,
\[
\mathrm{Reg}_T \;=\; \sum_{t=1}^{T} f_t(w_t) \;-\; \min_{w \in \mathcal{W}} \sum_{t=1}^{T} f_t(w),
\]
seeking $\mathrm{Reg}_T = o(T)$ (sublinear regret), rather than convergence to a fixed point.

%==========================================================
\section*{Mathematical Formulation}

Let $\mathcal{W} \subseteq \mathbb{R}^d$ be a closed convex feasible set and
$\Pi_{\mathcal{W}}(x) = \arg\min_{w\in\mathcal{W}} \|w - x\|$ the Euclidean projection.

\textbf{Update rule (projected OGD):}
\[
g_t \in \partial f_t(w_t), \qquad
w_{t+1} = \Pi_{\mathcal{W}}\!\left( w_t - \eta_t\, g_t \right).
\]

\textbf{Hyperparameters:}
\begin{itemize}
  \item Step size schedule $\eta_t > 0$: either fixed $\eta_t = \eta$
        or decreasing $\eta_t = \dfrac{D}{G\sqrt{t}}$.
  \item Initial point $w_1 \in \mathcal{W}$.
\end{itemize}

\textbf{Quantities:}
\begin{itemize}
  \item $D = \mathrm{diam}(\mathcal{W}) = \sup_{u,v\in\mathcal{W}} \|u-v\|$.
  \item $G$ = uniform bound on (sub)gradient norms, $\|g_t\| \le G$.
\end{itemize}

%==========================================================
\section*{Pseudocode}

\begin{verbatim}
Input:  feasible set W, horizon T, step sizes {eta_t}, initial w_1 in W
Initialization:
        w_1 <- any point in W
For t = 1, 2, ..., T:
        Play w_t
        Observe loss function f_t (or subgradient g_t)
        Compute g_t in subdifferential of f_t at w_t
        Gradient step:   y_{t+1} = w_t - eta_t * g_t
        Projection:      w_{t+1} = Proj_W( y_{t+1} )
End For
Termination: stop after T rounds (or when regret bound target is met)
Output:  sequence w_1, ..., w_T  (or average wbar_T = (1/T) sum_t w_t)
\end{verbatim}

\section*{Dry Run Example}

We treat $f_t(w) = f(w) = (w-3)^2$ (same loss each round, $\mathcal{W}=\mathbb{R}$),
with $w_0 = 0$, $\eta = 0.1$, $3$ iterations. The gradient is
$\nabla f(w) = 2(w-3)$. With $\mathcal{W}=\mathbb{R}$ the projection is the identity.

\textbf{Iteration 1:}
\begin{align*}
g_0 &= 2(w_0 - 3) = 2(0-3) = -6,\\
w_1 &= w_0 - \eta g_0 = 0 - 0.1\cdot(-6) = 0.6,\\
f(w_1) &= (0.6-3)^2 = (-2.4)^2 = 5.76.
\end{align*}

\textbf{Iteration 2:}
\begin{align*}
g_1 &= 2(w_1 - 3) = 2(0.6-3) = -4.8,\\
w_2 &= w_1 - \eta g_1 = 0.6 - 0.1\cdot(-4.8) = 1.08,\\
f(w_2) &= (1.08-3)^2 = (-1.92)^2 = 3.6864.
\end{align*}

\textbf{Iteration 3:}
\begin{align*}
g_2 &= 2(w_2 - 3) = 2(1.08-3) = -3.84,\\
w_3 &= w_2 - \eta g_2 = 1.08 - 0.1\cdot(-3.84) = 1.464,\\
f(w_3) &= (1.464-3)^2 = (-1.536)^2 = 2.359296.
\end{align*}

The iterates $0 \to 0.6 \to 1.08 \to 1.464$ move monotonically toward the minimizer
$w^\star = 3$, and the loss decreases $5.76 \to 3.6864 \to 2.359296$.

\section*{Assumptions}

\begin{assumption}[Convex feasible set]\label{a:set}
$\mathcal{W} \subseteq \mathbb{R}^d$ is nonempty, closed, and convex, with finite diameter
\[
D = \sup_{u,v\in\mathcal{W}} \|u-v\| < \infty.
\]
\end{assumption}

\begin{assumption}[Convexity of losses]\label{a:convex}
Each $f_t:\mathcal{W}\to\mathbb{R}$ is convex, so for all $w,u\in\mathcal{W}$ and any
$g_t\in\partial f_t(w)$,
\[
f_t(u) \ge f_t(w) + \langle g_t,\, u - w \rangle.
\]
\end{assumption}

\begin{assumption}[Lipschitz / bounded subgradients]\label{a:lip}
Each $f_t$ is $G$-Lipschitz on $\mathcal{W}$, equivalently every subgradient is bounded:
\[
\|g_t\| \le G \qquad \forall\, g_t\in\partial f_t(w),\ w\in\mathcal{W}.
\]
\end{assumption}

\begin{assumption}[Step sizes]\label{a:step}
The step sizes satisfy $\eta_t>0$ and are nonincreasing ($\eta_{t+1}\le\eta_t$).
Typical choices: $\eta_t = \eta$ (fixed) or $\eta_t = \frac{D}{G\sqrt{t}}$.
\end{assumption}

%==========================================================
\section*{Main Convergence Theorem}

\begin{theorem}[Regret of OGD]\label{thm:main}
Under Assumptions~\ref{a:set}--\ref{a:step}, the iterates of projected OGD satisfy,
for any comparator $w^\star\in\mathcal{W}$,
\[
\mathrm{Reg}_T(w^\star)
= \sum_{t=1}^{T}\bigl(f_t(w_t) - f_t(w^\star)\bigr)
\;\le\;
\frac{\|w_1 - w^\star\|^2}{2\eta_T}
+ \frac{G^2}{2}\sum_{t=1}^{T}\eta_t.
\]
In particular:
\begin{itemize}
  \item \textbf{Decreasing steps} $\eta_t = \frac{D}{G\sqrt{t}}$ give
        $\mathrm{Reg}_T \le \tfrac{3}{2}\,D G \sqrt{T} = O(\sqrt{T})$.
  \item \textbf{Fixed step} $\eta = \frac{D}{G\sqrt{T}}$ gives
        $\mathrm{Reg}_T \le D G \sqrt{T} = O(\sqrt{T})$.
\end{itemize}
Consequently the average regret $\frac{1}{T}\mathrm{Reg}_T = O(1/\sqrt{T}) \to 0$.
\end{theorem}

%==========================================================
\section*{Theoretical Properties}

\begin{itemize}
  \item \textbf{Sublinearity:} $\mathrm{Reg}_T=O(\sqrt{T})$ implies vanishing average
        regret; the learner is asymptotically as good as the best fixed point.
  \item \textbf{Optimality:} For convex $G$-Lipschitz losses on a set of diameter $D$,
        the $\Omega(DG\sqrt{T})$ lower bound matches OGD up to constants;
        OGD is minimax-optimal.
  \item \textbf{Hyperparameter sensitivity:} The bound is minimized (for fixed $\eta$)
        at $\eta^\star = \frac{D}{G\sqrt{T}}$; mis-tuning by a factor $c$ multiplies the
        regret term by roughly $\tfrac12(c + 1/c)$, hence robust.
  \item \textbf{Stability (anytime):} Decreasing steps $\eta_t\propto 1/\sqrt{t}$ avoid
        needing $T$ in advance while preserving $O(\sqrt{T})$.
  \item \textbf{Strongly convex case:} If each $f_t$ is $\mu$-strongly convex, choosing
        $\eta_t=\frac{1}{\mu t}$ improves the bound to $O\!\left(\frac{G^2}{\mu}\log T\right)$.
\end{itemize}

%==========================================================
\section*{Discussion}

OGD is the canonical OCO algorithm: a single projected (sub)gradient step per round.
Unlike SGD which targets convergence of $f(w_T)\to f(w^\star)$ under stochastic noise,
OGD targets adversarial sequences and bounds cumulative regret. The $O(\sqrt{T})$ regret
is tight in the general convex-Lipschitz setting; online-to-batch conversion recovers the
familiar $O(1/\sqrt{T})$ optimization rate of Example~1 (SGD).

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Nonexpansiveness of Euclidean projection]\label{lem:proj}
For closed convex $\mathcal{W}$ and any $x\in\mathbb{R}^d$, $u\in\mathcal{W}$,
\[
\|\Pi_{\mathcal{W}}(x) - u\| \le \|x - u\|.
\]
\end{lemma}

\begin{proof}
Let $p=\Pi_{\mathcal{W}}(x)$. The projection's first-order optimality condition gives
$\langle x - p,\, u - p\rangle \le 0$ for all $u\in\mathcal{W}$. Then
\[
\|x-u\|^2 = \|x-p\|^2 + 2\langle x-p,\,p-u\rangle + \|p-u\|^2
\ge \|p-u\|^2,
\]
since $\langle x-p,\,p-u\rangle = -\langle x-p, u-p\rangle \ge 0$. Taking square roots
yields the claim.
\end{proof}

%==========================================================
\section*{Main Proof (Step-by-Step Derivation)}

We bound the per-round contribution $f_t(w_t) - f_t(w^\star)$ via the squared distance to
the comparator $w^\star\in\mathcal{W}$.

\textbf{[Step 1] Expand the squared distance after the gradient step.}
Let $y_{t+1} = w_t - \eta_t g_t$ (pre-projection). Then
\begin{align*}
\|y_{t+1} - w^\star\|^2
&= \|w_t - \eta_t g_t - w^\star\|^2 \\
&= \|w_t - w^\star\|^2 - 2\eta_t \langle g_t,\, w_t - w^\star\rangle
   + \eta_t^2 \|g_t\|^2.
\end{align*}

\textbf{[Step 2] Apply projection nonexpansiveness.}
Since $w_{t+1} = \Pi_{\mathcal{W}}(y_{t+1})$ and $w^\star\in\mathcal{W}$,
Lemma~\ref{lem:proj} gives $\|w_{t+1}-w^\star\| \le \|y_{t+1}-w^\star\|$, hence
\[
\|w_{t+1} - w^\star\|^2
\le \|w_t - w^\star\|^2 - 2\eta_t \langle g_t,\, w_t - w^\star\rangle
   + \eta_t^2 \|g_t\|^2.
\]

\textbf{[Step 3] Isolate the inner product.}
Rearranging,
\[
\langle g_t,\, w_t - w^\star\rangle
\le \frac{\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2}{2\eta_t}
   + \frac{\eta_t}{2}\|g_t\|^2.
\]

\textbf{[Step 4] Use convexity (Assumption~\ref{a:convex}).}
With $g_t\in\partial f_t(w_t)$ and convexity,
$f_t(w_t) - f_t(w^\star) \le \langle g_t,\, w_t - w^\star\rangle$. Therefore
\[
f_t(w_t) - f_t(w^\star)
\le \frac{\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2}{2\eta_t}
   + \frac{\eta_t}{2}\|g_t\|^2.
\]

\textbf{[Step 5] Bound the gradient term (Assumption~\ref{a:lip}).}
Since $\|g_t\|\le G$,
\[
f_t(w_t) - f_t(w^\star)
\le \frac{\|w_t - w^\star\|^2 - \|w_{t+1} - w^\star\|^2}{2\eta_t}
   + \frac{\eta_t}{2}G^2.
\]

\textbf{[Step 6] Sum over $t = 1,\ldots,T$.}
Define $A_t = \|w_t - w^\star\|^2$. Summing,
\[
\mathrm{Reg}_T
\le \sum_{t=1}^{T} \frac{A_t - A_{t+1}}{2\eta_t}
   + \frac{G^2}{2}\sum_{t=1}^{T}\eta_t.
\]

\textbf{[Step 7] Abel summation / telescoping of the first sum.}
Write
\[
\sum_{t=1}^{T}\frac{A_t - A_{t+1}}{2\eta_t}
= \sum_{t=1}^{T}\frac{A_t}{2\eta_t} - \sum_{t=1}^{T}\frac{A_{t+1}}{2\eta_t}.
\]
Re-index the second sum and regroup:
\begin{align*}
\sum_{t=1}^{T}\frac{A_t - A_{t+1}}{2\eta_t}
&= \frac{A_1}{2\eta_1}
 + \sum_{t=2}^{T} A_t\left(\frac{1}{2\eta_t} - \frac{1}{2\eta_{t-1}}\right)
 - \frac{A_{T+1}}{2\eta_T}.
\end{align*}

\textbf{[Step 8] Bound each term using $A_t \le D^2$ and step monotonicity.}
By Assumption~\ref{a:set}, $A_t = \|w_t-w^\star\|^2 \le D^2$, and $A_{T+1}\ge 0$.
By Assumption~\ref{a:step}, $\eta_t\le\eta_{t-1}$ so
$\frac{1}{2\eta_t}-\frac{1}{2\eta_{t-1}}\ge 0$. Hence
\begin{align*}
\sum_{t=1}^{T}\frac{A_t - A_{t+1}}{2\eta_t}
&\le \frac{D^2}{2\eta_1}
 + D^2\sum_{t=2}^{T}\left(\frac{1}{2\eta_t} - \frac{1}{2\eta_{t-1}}\right) \\
&= \frac{D^2}{2\eta_1}
 + D^2\left(\frac{1}{2\eta_T} - \frac{1}{2\eta_1}\right)
 = \frac{D^2}{2\eta_T}.
\end{align*}

\textbf{[Step 9] Combine.}
Substituting into Step 6,
\[
\boxed{\;\mathrm{Reg}_T
\le \frac{D^2}{2\eta_T} + \frac{G^2}{2}\sum_{t=1}^{T}\eta_t.\;}
\]
(Replacing $D^2$ by $\|w_1-w^\star\|^2$ in the alternative form of Step~7--8 with a
constant step gives the statement of Theorem~\ref{thm:main}.)

%==========================================================
\section*{Rate Derivation (With Constants)}

\textbf{Decreasing steps $\eta_t = \dfrac{D}{G\sqrt{t}}$.}

\textbf{[Step 10] Bound the harmonic-type sum.}
\[
\sum_{t=1}^{T}\eta_t
= \frac{D}{G}\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
\le \frac{D}{G}\int_{0}^{T}\frac{dt}{\sqrt{t}}
= \frac{D}{G}\cdot 2\sqrt{T}
= \frac{2D\sqrt{T}}{G}.
\]

\textbf{[Step 11] Evaluate the first term.}
$\eta_T = \dfrac{D}{G\sqrt{T}}$, so
\[
\frac{D^2}{2\eta_T} = \frac{D^2 G\sqrt{T}}{2D} = \frac{D G\sqrt{T}}{2}.
\]

\textbf{[Step 12] Add the two contributions.}
\[
\mathrm{Reg}_T
\le \frac{DG\sqrt{T}}{2} + \frac{G^2}{2}\cdot\frac{2D\sqrt{T}}{G}
= \frac{DG\sqrt{T}}{2} + D G\sqrt{T}
= \frac{3}{2}\,D G\sqrt{T}.
\]

\textbf{Fixed step $\eta = \dfrac{D}{G\sqrt{T}}$.}

\textbf{[Step 13] Specialize Step 9 with $\eta_t=\eta$, $\eta_T=\eta$.}
\[
\mathrm{Reg}_T
\le \frac{D^2}{2\eta} + \frac{G^2}{2}\,T\eta.
\]

\textbf{[Step 14] Plug in $\eta$.}
\[
\frac{D^2}{2\eta} = \frac{D^2 G\sqrt{T}}{2D} = \frac{DG\sqrt{T}}{2},
\qquad
\frac{G^2}{2}T\eta = \frac{G^2}{2}\cdot T\cdot\frac{D}{G\sqrt{T}}
= \frac{DG\sqrt{T}}{2}.
\]
Summing,
\[
\mathrm{Reg}_T \le DG\sqrt{T} = O(\sqrt{T}).
\]

\textbf{[Step 15] Optimal fixed step (verification).}
Minimizing $h(\eta)=\frac{D^2}{2\eta}+\frac{G^2 T}{2}\eta$ over $\eta>0$:
$h'(\eta) = -\frac{D^2}{2\eta^2}+\frac{G^2 T}{2}=0 \Rightarrow \eta^\star=\frac{D}{G\sqrt{T}}$,
confirming the chosen step is optimal and giving $h(\eta^\star)=DG\sqrt{T}$.

\section*{Special Cases}

\textbf{Strongly convex losses.}
Suppose each $f_t$ is $\mu$-strongly convex:
\[
f_t(w^\star) \ge f_t(w_t) + \langle g_t, w^\star - w_t\rangle + \frac{\mu}{2}\|w^\star-w_t\|^2.
\]

\textbf{[Step 16] Strengthen Step 4.}
This gives the sharper bound
\[
f_t(w_t) - f_t(w^\star)
\le \langle g_t, w_t - w^\star\rangle - \frac{\mu}{2}\|w_t - w^\star\|^2.
\]
Combining with Step 3,
\[
f_t(w_t)-f_t(w^\star)
\le \left(\frac{1}{2\eta_t}-\frac{\mu}{2}\right)A_t
 - \frac{1}{2\eta_t}A_{t+1} + \frac{\eta_t}{2}G^2.
\]

\textbf{[Step 17] Choose $\eta_t = \dfrac{1}{\mu t}$ and telescope.}
Then $\frac{1}{2\eta_t}-\frac{\mu}{2} = \frac{\mu t}{2}-\frac{\mu}{2} = \frac{\mu(t-1)}{2}
= \frac{1}{2\eta_{t-1}}$ (with $\frac{1}{2\eta_0}:=0$), so the distance terms telescope:
\[
\sum_{t=1}^{T}\left[\frac{\mu(t-1)}{2}A_t - \frac{\mu t}{2}A_{t+1}\right] \le 0.
\]
Hence
\[
\mathrm{Reg}_T \le \frac{G^2}{2}\sum_{t=1}^{T}\eta_t
= \frac{G^2}{2\mu}\sum_{t=1}^{T}\frac{1}{t}
\le \frac{G^2}{2\mu}\bigl(1+\ln T\bigr) = O\!\left(\frac{G^2}{\mu}\log T\right).
\]

\textbf{Online-to-batch (i.i.d.\ case).}
If $f_t(w)=f(w;z_t)$ with $z_t$ i.i.d.\ and $\bar w_T=\frac1T\sum_t w_t$, then by Jensen
and the regret bound,
\[
\mathbb{E}\bigl[F(\bar w_T) - F(w^\star)\bigr]
\le \frac{1}{T}\,\mathbb{E}[\mathrm{Reg}_T]
= O\!\left(\frac{DG}{\sqrt{T}}\right),
\]
recovering the SGD rate of Example~1.
