\section*{Frank–Wolfe (Conditional Gradient) Method}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be a convex and $L$–smooth function and let 
\[
\mathcal C\subset\mathbb{R}^d
\]
be a non‑empty compact convex set with diameter 
\[
D:=\max_{x,y\in\mathcal C}\|x-y\|_2<\infty .
\]
The Frank–Wolfe algorithm generates a sequence $\{x_k\}_{k\ge 0}\subset\mathcal C$ as follows:
\begin{align}
s_k &:= \argmin_{s\in\mathcal C}\;\langle \nabla f(x_k),\,s\rangle , \tag{FW‑1}\\[4pt]
x_{k+1} &:= (1-\gamma_k)\,x_k + \gamma_k\,s_k , \qquad \gamma_k\in[0,1]. \tag{FW‑2}
\end{align}
The scalar $\gamma_k$ can be chosen by an \emph{exact line search}
\[
\gamma_k:=\argmin_{\gamma\in[0,1]} f\bigl((1-\gamma)x_k+\gamma s_k\bigr),
\]
or by a prescribed schedule (e.g.\ $\gamma_k=\frac{2}{k+2}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe Method}\label{alg:FW}
\begin{algorithmic}[1]
\Require Convex $L$‑smooth $f$, compact convex $\mathcal C$, initial $x_0\in\mathcal C$, step‑size rule $\{\gamma_k\}_{k\ge0}$.
\For{$k=0,1,2,\dots$}
    \State Compute gradient $g_k := \nabla f(x_k)$.
    \State \textbf{Linear minimisation oracle:}
    \[
        s_k \gets \argmin_{s\in\mathcal C}\langle g_k,\,s\rangle .
    \] \Comment{solves a linear problem over $\mathcal C$}
    \State Choose step size $\gamma_k\in[0,1]$ (exact line search or predefined).
    \State Update 
    \[
        x_{k+1} \gets (1-\gamma_k)x_k + \gamma_k s_k .
    \]
    \If{termination criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the 1‑dimensional problem  

\[
f(w) = (w-3)^2 ,\qquad \mathcal C = [0,5],
\]
with initial point $w_0=0$ and a \emph{fixed} step size $\gamma_k=\gamma=0.1$ for all $k$.
The gradient is $\nabla f(w)=2(w-3)$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)$ & $s_k$ (oracle) & $w_{k+1}$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $\displaystyle s_0=\argmin_{s\in[0,5]}(-6)s =0$ & $0.9\cdot0+0.1\cdot0=0$ \\[4pt]
1 & $0$ & $-6$ & $s_1=0$ & $0$ \\[4pt]
2 & $0$ & $-6$ & $s_2=0$ & $0$ 
\end{tabular}
\end{center}

Because the linear minimisation oracle selects the left endpoint $0$ (the direction of steepest descent for the linearised model), the iterate stays at $0$ for the three displayed steps. The objective values are
\[
f(w_0)=9,\qquad f(w_1)=9,\qquad f(w_2)=9 .
\]
If we instead started from $w_0=4$, the oracle would pick $s_0=5$ (since $\nabla f(4)=2>0$) and the first update would be $w_1=0.9\cdot4+0.1\cdot5=4.1$, illustrating a non‑trivial move.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex-smooth}
The objective $f$ is convex and $L$–smooth on an open set containing $\mathcal C$, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|_2\le L\|x-y\|_2,\qquad\forall x,y\in\mathcal C .
\]
\end{assumption}
\begin{assumption}[Compact Feasible Set]\label{as:compact}
$\mathcal C$ is compact, convex and has finite diameter $D$.
\end{assumption}
\begin{assumption}[Linear Minimisation Oracle]\label{as:oracle}
For any $g\in\mathbb R^d$ the oracle returns an exact solution of 
\[
\argmin_{s\in\mathcal C}\langle g,s\rangle .
\]
\end{assumption}
\begin{assumption}[Step‑size]\label{as:stepsize}
The step size satisfies $\gamma_k\in[0,1]$ and either
\begin{enumerate}[label=(\alph*)]
\item \emph{Exact line search} $\displaystyle\gamma_k=\argmin_{\gamma\in[0,1]}f\bigl((1-\gamma)x_k+\gamma s_k\bigr)$, or
\item \emph{Predefined schedule} $\displaystyle\gamma_k=\frac{2}{k+2}$.
\end{enumerate}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal convergence of Frank–Wolfe]\label{thm:fw-rate}
Under Assumptions \ref{as:convex-smooth}–\ref{as:stepsize}, the iterates $\{x_k\}$ satisfy for all $k\ge 0$
\[
f(x_k)-f(x^\star)\;\le\; \frac{2L D^2}{k+2},
\]
where $x^\star\in\argmin_{x\in\mathcal C}f(x)$.
\end{theorem}
The bound is $O(1/k)$ and matches the classical Frank–Wolfe rate.

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The algorithm never leaves $\mathcal C$ because $x_{k+1}$ is a convex combination of two points in $\mathcal C$.
\item \textbf{Step‑size sensitivity:} Using the schedule $\gamma_k=2/(k+2)$ yields the optimal $O(1/k)$ bound without line search. Exact line search can improve constants but does not change the asymptotic order.
\item \textbf{Comparison to projected gradient:} Frank–Wolfe avoids projections, which is advantageous when the linear oracle is cheap while projections onto $\mathcal C$ are costly. The price is a slower $O(1/k)$ rate versus the $O(1/k)$ of projected gradient with the same smoothness constant, but Frank–Wolfe enjoys sparsity in the iterates when $\mathcal C$ is a polytope.
\end{itemize}

\section*{Discussion}
The theorem shows that the primal gap decays inversely with the iteration counter. When $f$ is additionally strongly convex, a linear (geometric) rate can be obtained by using a \emph{away‑step} or \emph{pairwise} variant, but the vanilla algorithm retains the $O(1/k)$ guarantee.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Duality gap]\label{def:gap}
For any $x\in\mathcal C$ define
\[
g(x) := \max_{s\in\mathcal C}\langle \nabla f(x),\,x-s\rangle
      = \langle \nabla f(x),\,x-s(x)\rangle ,
\]
where $s(x):=\argmin_{s\in\mathcal C}\langle \nabla f(x),s\rangle$ is the oracle output.
\end{definition}
The duality gap $g(x)$ upper bounds the primal suboptimality $f(x)-f(x^\star)$.

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\mathcal C$,
\[
f(y) \le f(x) + \langle \nabla f(x),y-x\rangle + \frac{L}{2}\|y-x\|_2^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004intro}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:fw-rate}.  
Let $x^\star\in\argmin_{\mathcal C}f$ and denote $h_k:=f(x_k)-f(x^\star)$.  

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $x=x_k$ and $y=x_{k+1}=(1-\gamma_k)x_k+\gamma_k s_k$:
\[
f(x_{k+1}) \le f(x_k)+\gamma_k\langle\nabla f(x_k),s_k-x_k\rangle+\frac{L\gamma_k^{2}}{2}\|s_k-x_k\|_2^{2}. \tag{1}
\]

\noindent\textbf{[Step 2]} By definition of $s_k$,
\[
\langle\nabla f(x_k),s_k-x_k\rangle
    = -\,\underbrace{\langle\nabla f(x_k),x_k-s_k\rangle}_{=\,g(x_k)}\;=\; -g(x_k). \tag{2}
\]

\noindent\textbf{[Step 3]} Because $s_k,x_k\in\mathcal C$, the distance is bounded by the diameter:
\[
\|s_k-x_k\|_2 \le D . \tag{3}
\]

\noindent\textbf{[Step 4]} Insert (2) and (3) into (1):
\[
f(x_{k+1}) \le f(x_k)-\gamma_k g(x_k)+\frac{L D^{2}}{2}\gamma_k^{2}. \tag{4}
\]

\noindent\textbf{[Step 5]} The duality gap majorises the primal gap:
\[
g(x_k)\;\ge\; f(x_k)-f(x^\star)=h_k . \tag{5}
\]
(The inequality follows from convexity:
$f(x^\star)\ge f(x_k)+\langle\nabla f(x_k),x^\star-x_k\rangle$,
rearranged gives (5).)

\noindent\textbf{[Step 6]} Substitute (5) into (4):
\[
h_{k+1}\le h_k-\gamma_k h_k+\frac{L D^{2}}{2}\gamma_k^{2}
          = (1-\gamma_k)h_k+\frac{L D^{2}}{2}\gamma_k^{2}. \tag{6}
\]

\noindent\textbf{[Step 7]} Choose the standard schedule $\displaystyle\gamma_k=\frac{2}{k+2}$ (satisfies Assumption \ref{as:stepsize} (b)). Then
\[
1-\gamma_k = 1-\frac{2}{k+2}= \frac{k}{k+2},\qquad
\gamma_k^{2}= \frac{4}{(k+2)^{2}} . \tag{7}
\]

\noindent\textbf{[Step 8]} Plug (7) into (6):
\[
h_{k+1}\le \frac{k}{k+2}h_k+\frac{L D^{2}}{2}\cdot\frac{4}{(k+2)^{2}}
        = \frac{k}{k+2}h_k+\frac{2L D^{2}}{(k+2)^{2}} . \tag{8}
\]

\noindent\textbf{[Step 9]} We prove by induction that
\[
h_k \le \frac{2L D^{2}}{k+2}\quad\forall k\ge0 . \tag{9}
\]
\emph{Base case $k=0$:} $h_0=f(x_0)-f(x^\star)\le \frac{2L D^{2}}{2}=L D^{2}$, which holds because $f$ is $L$‑smooth on a set of diameter $D$ (a crude bound suffices for the induction).

\noindent\textbf{[Step 10]} Inductive step: assume (9) holds for $k$. Using (8) and the induction hypothesis,
\[
\begin{aligned}
h_{k+1}
&\le \frac{k}{k+2}\cdot\frac{2L D^{2}}{k+2}
   +\frac{2L D^{2}}{(k+2)^{2}}  \\
&= \frac{2L D^{2}}{(k+2)^{2}}\bigl(k+1\bigr) \\
&= \frac{2L D^{2}}{(k+1)+2},
\end{aligned}
\]
which is exactly (9) with $k$ replaced by $k+1$. Hence the claim holds for all $k$.

\noindent\textbf{[Step 11]} Inequality (9) is precisely the statement of Theorem \ref{thm:fw-rate}.
\hfill $\square$

\section*{Rate Derivation (With Constants)}
The proof above yields the explicit constant $2L D^{2}$ in the bound
\[
f(x_k)-f(x^\star)\le \frac{2L D^{2}}{k+2}.
\]
If an exact line search is used, the same recurrence (6) holds with
\[
\gamma_k = \argmin_{\gamma\in[0,1]} f\bigl((1-\gamma)x_k+\gamma s_k\bigr),
\]
and a standard argument (see \cite{jaggi2013revisiting}) shows that the same $O(1/k)$ rate holds, often with a smaller empirical constant because the line search chooses $\gamma_k\le \frac{2}{k+2}$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Polytope $\mathcal C$:} When $\mathcal C=\operatorname{conv}\{v_1,\dots,v_m\}$, each $s_k$ is a vertex. Hence $x_k$ is a convex combination of at most $k+1$ vertices, yielding sparse iterates.
\item \textbf{Strongly convex $f$:} If $f$ is $\mu$‑strongly convex, the duality gap satisfies $g(x_k)\ge \mu\,\|x_k-x^\star\|_2^{2}$, and a variant with \emph{away steps} obtains a linear rate $O\bigl((1-\rho)^k\bigr)$ for some $\rho>0$.
\item \textbf{Quadratic $f$:} For $f(x)=\frac12 x^\top Q x + b^\top x$, the smoothness constant is $L=\lambda_{\max}(Q)$. The Frank–Wolfe iterates solve a sequence of linear programs, and the $O(1/k)$ bound remains tight.
\end{enumerate}
