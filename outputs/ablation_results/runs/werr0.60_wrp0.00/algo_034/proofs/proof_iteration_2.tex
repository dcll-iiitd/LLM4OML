\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be a (possibly non‑convex) differentiable loss.  
At iteration $t$ we draw a random sample $\xi_t$ and obtain a stochastic gradient
\[
g_t \;:=\; \nabla f(x_t;\xi_t) .
\]
Define the \emph{clipping operator} with threshold $C>0$ as
\[
\operatorname{Clip}_C(v)\;:=\;
\frac{v}{\max\!\bigl(1,\; \|v\|/C\bigr)}=
\begin{cases}
v, & \|v\|\le C,\\[4pt]
C\,\dfrac{v}{\|v\|}, & \|v\|>C .
\end{cases}
\]
The \textbf{clipped stochastic gradient} is
\[
\tilde g_t \;:=\; \operatorname{Clip}_C\!\bigl(g_t\bigr).
\]
With a (possibly time‑varying) stepsize $\eta_t>0$, the update rule reads
\[
\boxed{\,x_{t+1}=x_t-\eta_t\,\tilde g_t\,}. \tag{1}
\]

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
$\displaystyle\forall t,\qquad \E_{\xi_t}[g_t\mid x_t]=\nabla f(x_t)$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^2\ge0$ such that
\[
\E_{\xi_t}\!\bigl[\|g_t-\nabla f(x_t)\|^2\mid x_t\bigr]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Clipping Threshold]\label{as:threshold}
The clipping constant $C>0$ satisfies
\[
C\;\ge\;\sqrt{\frac{2\sigma^2}{L}} .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:threshold} and set a diminishing stepsize
\[
\eta_t \;=\; \frac{\eta}{\sqrt{t+1}},\qquad \eta\le \frac{1}{L}.
\]
Let $x_0$ be arbitrary and define the average iterate
$\displaystyle\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t$.
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{2L\eta\sigma^2}{\sqrt{T}}}_{\text{variance term}}
\;+\;
\underbrace{\frac{2L\eta C^{2}\bigl(1+\log T\bigr)}{\sqrt{T}}}_{\text{clipping term}} .
\tag{2}
\]
Consequently $\displaystyle\min_{0\le t<T}\E\!\bigl[\|\nabla f(x_t)\|^2\bigr]=O(T^{-1/2})$.
\end{theorem}

\section*{Preliminaries}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x$ and any vector $u$,
\[
f(x-u)-f(x) \le -\inner{\nabla f(x)}{u}+\frac{L}{2}\norm{u}^2 .
\]
\end{lemma}
\begin{proof}
Apply Definition~\ref{as:smooth} with $y=x-u$ and rearrange. \hfill $\square$
\end{proof}

\begin{lemma}[Clipping Bias Bound]\label{lem:bias}
For any $v\in\mathbb{R}^d$ and threshold $C>0$,
\[
\|v-\operatorname{Clip}_C(v)\|\le \frac{1}{C}\,\|v\|^2 .
\]
\end{lemma}
\begin{proof}
If $\|v\|\le C$, the left–hand side is $0$ and the inequality holds.
If $\|v\|>C$, then $\operatorname{Clip}_C(v)=C\,v/\|v\|$ and
\[
\|v-\operatorname{Clip}_C(v)\|
= \Bigl\|v-C\frac{v}{\|v\|}\Bigr\|
= \|v\|\Bigl(1-\frac{C}{\|v\|}\Bigr)
= \|v\|-C
\le \frac{\|v\|^2}{C},
\]
because $\|v\|-C\le \|v\|^2/C$ for $\|v\|>C$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, expectations are taken over all
randomness up to iteration $t$ unless otherwise noted.

\subsection*{Step 1: One‑step descent}
From Lemma~\ref{lem:smooth} with $u=\eta_t\tilde g_t$,
\[
f(x_{t+1})-f(x_t)\le
-\eta_t\inner{\nabla f(x_t)}{\tilde g_t}
+\frac{L\eta_t^{2}}{2}\norm{\tilde g_t}^{2}. \tag{3}
\]

\subsection*{Step 2: Decompose the inner product}
Add and subtract $g_t$:
\[
\inner{\nabla f(x_t)}{\tilde g_t}
= \inner{\nabla f(x_t)}{g_t}
+\inner{\nabla f(x_t)}{\tilde g_t-g_t}. \tag{4}
\]

\subsection*{Step 3: Take conditional expectation}
Taking $\E[\cdot\mid x_t]$ on (3) and using (4):
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})-f(x_t)\mid x_t\bigr]
\le
&-\eta_t\underbrace{\E\!\bigl[\inner{\nabla f(x_t)}{g_t}\mid x_t\bigr]}_{= \|\nabla f(x_t)\|^{2}} \\
&-\eta_t\E\!\bigl[\inner{\nabla f(x_t)}{\tilde g_t-g_t}\mid x_t\bigr]
+\frac{L\eta_t^{2}}{2}\E\!\bigl[\norm{\tilde g_t}^{2}\mid x_t\bigr]. \tag{5}
\end{aligned}
\]

\subsection*{Step 4: Bounding the bias term}
Using Cauchy–Schwarz and Lemma~\ref{lem:bias},
\[
\bigl|\inner{\nabla f(x_t)}{\tilde g_t-g_t}\bigr|
\le \norm{\nabla f(x_t)}\norm{\tilde g_t-g_t}
\le \norm{\nabla f(x_t)}\frac{\|g_t\|^{2}}{C}. \tag{6}
\]
Conditioned on $x_t$,
\[
\E\!\bigl[\|g_t\|^{2}\mid x_t\bigr]
= \|\nabla f(x_t)\|^{2}
+\E\!\bigl[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\bigr]
\le \|\nabla f(x_t)\|^{2}+\sigma^{2}.
\]
Hence
\[
\E\!\bigl[|\inner{\nabla f(x_t)}{\tilde g_t-g_t}|\mid x_t\bigr]
\le
\frac{1}{C}\norm{\nabla f(x_t)}\bigl(\|\nabla f(x_t)\|^{2}+\sigma^{2}\bigr). \tag{7}
\]

\subsection*{Step 5: Bounding the second‑moment of the clipped gradient}
By construction $\norm{\tilde g_t}\le C$, therefore
\[
\E\!\bigl[\norm{\tilde g_t}^{2}\mid x_t\bigr]\le C^{2}. \tag{8}
\]

\subsection*{Step 6: Combine (5)–(8)}
Plugging (7) and (8) into (5) gives
\[
\E\!\bigl[f(x_{t+1})-f(x_t)\mid x_t\bigr]
\le
-\eta_t\|\nabla f(x_t)\|^{2}
+\frac{\eta_t}{C}\norm{\nabla f(x_t)}\bigl(\|\nabla f(x_t)\|^{2}+\sigma^{2}\bigr)
+\frac{L\eta_t^{2}C^{2}}{2}. \tag{9}
\]

\subsection*{Step 7: Rearrangement}
Move the first term to the left‑hand side:
\[
\eta_t\|\nabla f(x_t)\|^{2}
\le
\E\!\bigl[f(x_t)-f(x_{t+1})\mid x_t\bigr]
+\frac{\eta_t}{C}\norm{\nabla f(x_t)}\bigl(\|\nabla f(x_t)\|^{2}+\sigma^{2}\bigr)
+\frac{L\eta_t^{2}C^{2}}{2}. \tag{10}
\]

\subsection*{Step 8: Bounding the mixed term via Young’s inequality}
For any $a,b\ge0$,
\[
ab\;\le\;\frac{a^{2}}{2}+\frac{b^{2}}{2}.
\]
Apply this with $a=\norm{\nabla f(x_t)}$ and $b=\frac{\|\nabla f(x_t)\|^{2}}{C}$
to the cubic part, and with $a=\norm{\nabla f(x_t)}$, $b=\frac{\sigma^{2}}{C}$
to the linear part.  This yields
\[
\frac{\eta_t}{C}\norm{\nabla f(x_t)}\|\nabla f(x_t)\|^{2}
\le
\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
+\frac{\eta_t}{2C^{2}}\|\nabla f(x_t)\|^{4},
\]
\[
\frac{\eta_t}{C}\norm{\nabla f(x_t)}\sigma^{2}
\le
\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
+\frac{\eta_t\sigma^{4}}{2C^{2}}. \tag{11}
\]
Discarding the non‑negative fourth‑order term $\frac{\eta_t}{2C^{2}}\|\nabla f(x_t)\|^{4}$
(which only makes the right‑hand side larger) we obtain from (10)
\[
\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
\le
\E\!\bigl[f(x_t)-f(x_{t+1})\mid x_t\bigr]
+\frac{\eta_t\sigma^{4}}{2C^{2}}
+\frac{L\eta_t^{2}C^{2}}{2}. \tag{12}
\]

\subsection*{Step 9: Using the threshold assumption}
Assumption~\ref{as:threshold} gives $C^{2}\ge 2\sigma^{2}/L$, i.e.
$\displaystyle \frac{\sigma^{2}}{C^{2}}\le\frac{L}{2}$.
Consequently
\[
\frac{\eta_t\sigma^{4}}{2C^{2}}
= \frac{\eta_t\sigma^{2}}{2}\,\frac{\sigma^{2}}{C^{2}}
\le \frac{L\eta_t\sigma^{2}}{4}. \tag{13}
\]
Insert (13) into (12):
\[
\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
\le
\E\!\bigl[f(x_t)-f(x_{t+1})\mid x_t\bigr]
+\frac{L\eta_t\sigma^{2}}{4}
+\frac{L\eta_t^{2}C^{2}}{2}. \tag{14}
\]

\subsection*{Step 10: Full expectation and summation}
Take full expectation and sum (14) over $t=0,\dots,T-1$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
f(x_0)-\E\!\bigl[f(x_T)\bigr]
+\frac{L\sigma^{2}}{4}\sum_{t=0}^{T-1}\eta_t
+\frac{L C^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}. \tag{15}
\]

\subsection*{Step 11: Bounding the step‑size sums}
Recall $\eta_t=\eta/\sqrt{t+1}$.
\[
\sum_{t=0}^{T-1}\eta_t
= \eta\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
\le 2\eta\sqrt{T}, \qquad
\sum_{t=0}^{T-1}\eta_t^{2}
= \eta^{2}\sum_{t=1}^{T}\frac{1}{t}
\le \eta^{2}\bigl(1+\log T\bigr). \tag{16}
\]

\subsection*{Step 12: Plugging the sums}
Using $f(x_T)\ge f^\star$ and (16) in (15) we obtain
\[
\frac{1}{2}\,2\eta\sqrt{T}\,
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
f(x_0)-f^\star
+L\sigma^{2}\eta\sqrt{T}
+L C^{2}\eta^{2}\bigl(1+\log T\bigr). \tag{17}
\]

\subsection*{Step 13: Isolating the average gradient norm}
Dividing both sides of (17) by $\eta\sqrt{T}$ yields the desired bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta\sqrt{T}}
+\frac{2L\eta\sigma^{2}}{\sqrt{T}}
+\frac{2L\eta C^{2}\bigl(1+\log T\bigr)}{\sqrt{T}} .
\tag{18}
\]
This is precisely inequality~\eqref{eq:2}, completing the proof. \qed

\section*{Rate Discussion}
The three contributions in~\eqref{eq:2} are:
\begin{itemize}
\item \textbf{Optimization term} $2\bigl(f(x_0)-f^\star\bigr)/(\eta\sqrt{T})$.
\item \textbf{Stochastic variance term} $2L\eta\sigma^{2}/\sqrt{T}$.
\item \textbf{Clipping term} $2L\eta C^{2}(1+\log T)/\sqrt{T}$.
\end{itemize}
Choosing $\eta=\Theta(1/\sqrt{L})$ balances the variance and clipping contributions,
yielding the overall rate $O(T^{-1/2})$ (up to a harmless $\log T$ factor).

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{No clipping} ($C\to\infty$). The clipping term disappears and we recover the classic non‑convex SGD bound $O(T^{-1/2})$.
\item \textbf{Heavy‑tailed noise}. Even when higher moments of the noise are infinite, the bound~\eqref{eq:2} remains valid because clipping enforces $\|\tilde g_t\|\le C$.
\item \textbf{Constant stepsize $\eta_t=\eta\le 1/L$}. The same derivation gives a steady‑state error of order $L\eta C^{2}$ together with an $O(1/T)$ optimization term.
\end{enumerate}
