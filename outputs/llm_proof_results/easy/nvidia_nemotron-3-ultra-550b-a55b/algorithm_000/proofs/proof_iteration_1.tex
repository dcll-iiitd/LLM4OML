\section*{Algorithm Name}
\textbf{Stochastic Gradient Descent (SGD) with Fixed Step Size}

\section*{Mathematical Formulation}
Let $f: \mathbb{R}^d \to \mathbb{R}$ be the objective function. At iteration $t$, we sample a minibatch $\xi_t$ (or a single data point) and compute a stochastic gradient $g_t = \nabla f(w_t; \xi_t)$. The update rule is:
\begin{align}
    w_{t+1} = w_t - \eta \, g_t, \qquad t = 0, 1, \dots, T-1
\end{align}
where $\eta > 0$ is a fixed step size (learning rate). The initial point $w_0 \in \mathbb{R}^d$ is given.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGD with Fixed Step Size}
\begin{algorithmic}
\STATE \textbf{Input:} Initial parameter $w_0$, step size $\eta > 0$, number of iterations $T$, stochastic gradient oracle $\nabla f(w; \xi)$.
\FOR{$t = 0$ to $T-1$}
    \STATE Sample minibatch $\xi_t$ (or random index).
    \STATE Compute stochastic gradient $g_t = \nabla f(w_t; \xi_t)$.
    \STATE Update $w_{t+1} = w_t - \eta \, g_t$.
\ENDFOR
\STATE \textbf{Output:} Final iterate $w_T$ (or average $\frac{1}{T}\sum_{t=0}^{T-1} w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w) = (w-3)^2$, which is convex and smooth (but we treat it as a non-convex smooth function for illustration). Let $w_0 = 0$, $\eta = 0.1$, and we use full gradient (minibatch size = 1, but deterministic for simplicity). The gradient is $\nabla f(w) = 2(w-3)$.

\begin{itemize}
    \item \textbf{Iteration 0:} $w_0 = 0$.
        \begin{align*}
            g_0 &= 2(0-3) = -6,\\
            w_1 &= 0 - 0.1 \times (-6) = 0.6,\\
            f(w_1) &= (0.6-3)^2 = 5.76.
        \end{align*}
    \item \textbf{Iteration 1:} $w_1 = 0.6$.
        \begin{align*}
            g_1 &= 2(0.6-3) = -4.8,\\
            w_2 &= 0.6 - 0.1 \times (-4.8) = 1.08,\\
            f(w_2) &= (1.08-3)^2 = 3.6864.
        \end{align*}
    \item \textbf{Iteration 2:} $w_2 = 1.08$.
        \begin{align*}
            g_2 &= 2(1.08-3) = -3.84,\\
            w_3 &= 1.08 - 0.1 \times (-3.84) = 1.464,\\
            f(w_3) &= (1.464-3)^2 = 2.359.
        \end{align*}
\end{itemize}
The objective value decreases monotonically. For stochastic version, we would add noise to gradients.

\section*{Assumptions}
We make the following standard assumptions for non-convex smooth optimization:

\begin{enumerate}
    \item \textbf{(L-smoothness)} $f$ is differentiable and its gradient is $L$-Lipschitz continuous: for all $x, y \in \mathbb{R}^d$,
        \begin{equation}
            \|\nabla f(x) - \nabla f(y)\| \leq L \|x - y\|.
        \end{equation}
        Equivalently, for all $x, y$,
        \begin{equation}
            f(y) \leq f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2} \|y-x\|^2. \tag{A1}
        \end{equation}
    \item \textbf{(Lower bounded)} There exists $f^* > -\infty$ such that $f(x) \geq f^*$ for all $x \in \mathbb{R}^d$.
    \item \textbf{(Unbiased stochastic gradient)} The stochastic gradient $g_t$ satisfies
        \begin{equation}
            \mathbb{E}[g_t \mid w_t] = \nabla f(w_t). \tag{A2}
        \end{equation}
    \item \textbf{(Bounded variance)} There exists $\sigma^2 \geq 0$ such that
        \begin{equation}
            \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 \mid w_t] \leq \sigma^2. \tag{A3}
        \end{equation}
    \item \textbf{(Fixed step size)} The step size $\eta$ is constant and satisfies $0 < \eta \leq \frac{1}{L}$.
\end{enumerate}

\section*{Main Convergence Theorem}
Under Assumptions 1--5, after $T$ iterations of SGD with fixed step size $\eta \leq 1/L$, the following holds:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2(f(w_0) - f^*)}{\eta T} + \eta L \sigma^2. \tag{1}
\end{equation}
In particular, if we choose $\eta = \min\left(\frac{1}{L}, \sqrt{\frac{2(f(w_0)-f^*)}{L\sigma^2 T}}\right)$, then
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2L(f(w_0)-f^*)}{T} + \sqrt{\frac{2L\sigma^2(f(w_0)-f^*)}{T}} = O\left(\frac{1}{\sqrt{T}}\right).
\end{equation}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Convergence to stationary point:} The bound (1) guarantees that the average expected squared gradient norm converges to zero at rate $O(1/\sqrt{T})$ with appropriate step size, implying convergence to a first-order stationary point.
    \item \textbf{Step size sensitivity:} The bound has two terms: a bias term decreasing with $\eta$ and a variance term increasing with $\eta$. The optimal $\eta$ balances them.
    \item \textbf{Comparison to GD:} For deterministic gradient descent ($\sigma=0$), the rate becomes $O(1/T)$ with $\eta = 1/L$. The stochastic noise introduces an $O(1/\sqrt{T})$ rate.
    \item \textbf{Stability:} The requirement $\eta \leq 1/L$ ensures the descent lemma holds in expectation.
\end{itemize}

\section*{Discussion}
The analysis assumes fixed step size. In practice, decreasing step sizes ($\eta_t \propto 1/\sqrt{t}$) can achieve $O(\log T/\sqrt{T})$ without knowing $T$ in advance. The bounded variance assumption (A3) can be relaxed to expected smoothness or other conditions. The result is standard for non-convex SGD and matches the lower bound for first-order stochastic methods.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for L-smooth functions]
If $f$ is $L$-smooth, then for any $x, y \in \mathbb{R}^d$,
\begin{equation}
    f(y) \leq f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2} \|y-x\|^2.
\end{equation}
\end{lemma}

\begin{lemma}[Variance decomposition]
For any random vector $X$ with $\mathbb{E}[X] = \mu$,
\begin{equation}
    \mathbb{E}[\|X\|^2] = \|\mu\|^2 + \mathbb{E}[\|X - \mu\|^2].
\end{equation}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}
We now prove the main convergence theorem. Let $w_t$ be the iterate at step $t$, and $g_t$ the stochastic gradient.

\begin{proof}
We start by applying the descent lemma (A1) with $x = w_t$ and $y = w_{t+1} = w_t - \eta g_t$:
\begin{equation}
    f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t), w_{t+1} - w_t \rangle + \frac{L}{2} \|w_{t+1} - w_t\|^2. \tag{Step 1}
\end{equation}

Substitute $w_{t+1} - w_t = -\eta g_t$:
\begin{equation}
    f(w_{t+1}) \leq f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L \eta^2}{2} \|g_t\|^2. \tag{Step 2}
\end{equation}

Take conditional expectation $\mathbb{E}[\cdot \mid w_t]$ on both sides. Using (A2) and (A3):
\begin{align}
    \mathbb{E}[f(w_{t+1}) \mid w_t] &\leq f(w_t) - \eta \langle \nabla f(w_t), \mathbb{E}[g_t \mid w_t] \rangle + \frac{L \eta^2}{2} \mathbb{E}[\|g_t\|^2 \mid w_t] \tag{Step 3} \\
    &= f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \mathbb{E}[\|g_t\|^2 \mid w_t]. \tag{Step 4}
\end{align}

Now we expand $\mathbb{E}[\|g_t\|^2 \mid w_t]$ using variance decomposition (Lemma 2) with $\mu = \nabla f(w_t)$:
\begin{equation}
    \mathbb{E}[\|g_t\|^2 \mid w_t] = \|\nabla f(w_t)\|^2 + \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 \mid w_t]. \tag{Step 5}
\end{equation}

By (A3), the second term is bounded by $\sigma^2$:
\begin{equation}
    \mathbb{E}[\|g_t\|^2 \mid w_t] \leq \|\nabla f(w_t)\|^2 + \sigma^2. \tag{Step 6}
\end{equation}

Plug (Step 6) into (Step 4):
\begin{align}
    \mathbb{E}[f(w_{t+1}) \mid w_t] &\leq f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \left( \|\nabla f(w_t)\|^2 + \sigma^2 \right) \tag{Step 7} \\
    &= f(w_t) - \left( \eta - \frac{L \eta^2}{2} \right) \|\nabla f(w_t)\|^2 + \frac{L \eta^2 \sigma^2}{2}. \tag{Step 8}
\end{align}

Since $\eta \leq 1/L$, we have $\eta - \frac{L \eta^2}{2} \geq \eta - \frac{\eta}{2} = \frac{\eta}{2}$. Thus,
\begin{equation}
    \mathbb{E}[f(w_{t+1}) \mid w_t] \leq f(w_t) - \frac{\eta}{2} \|\nabla f(w_t)\|^2 + \frac{L \eta^2 \sigma^2}{2}. \tag{Step 9}
\end{equation}

Take total expectation on both sides:
\begin{equation}
    \mathbb{E}[f(w_{t+1})] \leq \mathbb{E}[f(w_t)] - \frac{\eta}{2} \mathbb{E}[\|\nabla f(w_t)\|^2] + \frac{L \eta^2 \sigma^2}{2}. \tag{Step 10}
\end{equation}

Rearrange to isolate the gradient norm term:
\begin{equation}
    \frac{\eta}{2} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2}. \tag{Step 11}
\end{equation}

Sum this inequality for $t = 0, 1, \dots, T-1$:
\begin{equation}
    \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \sum_{t=0}^{T-1} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \right) + \frac{T L \eta^2 \sigma^2}{2}. \tag{Step 12}
\end{equation}

The sum telescopes:
\begin{equation}
    \sum_{t=0}^{T-1} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \right) = \mathbb{E}[f(w_0)] - \mathbb{E}[f(w_T)] \leq f(w_0) - f^*, \tag{Step 13}
\end{equation}
where we used $f(w_T) \geq f^*$ and $w_0$ is deterministic (so $\mathbb{E}[f(w_0)] = f(w_0)$).

Thus,
\begin{equation}
    \frac{\eta}{2} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq f(w_0) - f^* + \frac{T L \eta^2 \sigma^2}{2}. \tag{Step 14}
\end{equation}

Divide both sides by $\frac{\eta}{2} T$:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2(f(w_0) - f^*)}{\eta T} + \eta L \sigma^2. \tag{Step 15}
\end{equation}
This completes the proof of the main theorem.
\end{proof}

\section*{Rate Derivation (With Constants)}
From (Step 15), we have the exact bound:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \underbrace{\frac{2\Delta_0}{\eta T}}_{\text{bias term}} + \underbrace{\eta L \sigma^2}_{\text{variance term}}, \quad \Delta_0 = f(w_0) - f^*.
\end{equation}
To minimize the right-hand side over $\eta \in (0, 1/L]$, we consider the unconstrained minimizer of $\frac{2\Delta_0}{\eta T} + \eta L \sigma^2$, which is $\eta^* = \sqrt{\frac{2\Delta_0}{L\sigma^2 T}}$. If $\eta^* \leq 1/L$, we set $\eta = \eta^*$ and obtain:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq 2 \sqrt{\frac{2 L \sigma^2 \Delta_0}{T}} = O\left(\frac{1}{\sqrt{T}}\right).
\end{equation}
If $\eta^* > 1/L$, we set $\eta = 1/L$ and get:
\begin{equation}
    \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2 L \Delta_0}{T} + \sigma^2 = O\left(\frac{1}{T}\right) + \sigma^2.
\end{equation}
In the worst case (large $T$), the rate is $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Deterministic GD ($\sigma = 0$):} The bound becomes $\frac{1}{T}\sum_{t=0}^{T-1} \|\nabla f(w_t)\|^2 \leq \frac{2\Delta_0}{\eta T}$. With $\eta = 1/L$, we get $O(1/T)$ rate.
    \item \textbf{Convex functions:} If $f$ is also convex, we can derive a bound on function value suboptimality: $\mathbb{E}[f(\bar{w}_T) - f^*] \leq O(1/\sqrt{T})$ with $\bar{w}_T = \frac{1}{T}\sum_{t=0}^{T-1} w_t$.
    \item \textbf{Strongly convex:} With strong convexity, SGD with decreasing step size achieves $O(1/T)$.
    \item \textbf{Mini-batch SGD:} If we use mini-batches of size $B$, the variance reduces to $\sigma^2/B$, improving the variance term to $\eta L \sigma^2 / B$.
\end{enumerate}
