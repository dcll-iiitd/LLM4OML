%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
\section{Mathematical Formulation}
Consider a set of $K$ clients.  Client $k$ possesses a local loss
\[
f_k:\R^{d}\to\R,\qquad 
F(w)\;:=\;\sum_{k=1}^{K}p_k\,f_k(w),\qquad p_k>0,\;\sum_{k}p_k=1 .
\]
The goal is to minimize the (generally non‑convex) global objective $F$.

At communication round $t=0,1,\dots,T-1$ the server holds the \emph{global model}
$w^{t}\in\R^{d}$.  Each client performs $E$ local SGD steps on the
\emph{proximal} objective
\[
\phi_{k}^{t}(x):= f_k(x)+\frac{\mu}{2}\norm{x-w^{t}}^{2},
\qquad \mu\ge 0 .
\]
Let $w_{k}^{t,0}=w^{t}$ be the local copy at the beginning of round $t$.
For $i=0,\dots,E-1$ client $k$ draws a stochastic gradient
\[
g_{k}^{t,i}:= \nabla f_k\bigl(w_{k}^{t,i};\xi_{k}^{t,i}\bigr)
\]
and updates
\begin{equation}
\boxed{
w_{k}^{t,i+1}= w_{k}^{t,i}
        \;-\;\eta\,g_{k}^{t,i}
        \;-\;\eta\,\mu\bigl(w_{k}^{t,i}-w^{t}\bigr)
}
\label{eq:local-update}
\end{equation}
with learning rate $\eta>0$.
After the $E$ local steps the client sends $w_{k}^{t,E}$ to the server.
The server aggregates by weighted averaging
\begin{equation}
\boxed{
w^{t+1}= \sum_{k=1}^{K}p_k\,w_{k}^{t,E}.
}
\label{eq:global-agg}
\end{equation}
The procedure repeats until a termination criterion (e.g., $T$ rounds or
$\norm{\nabla F(w^{t})}$ below a tolerance) is met.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\section{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx with Local SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial model $w^{0}$, learning rate $\eta$, proximal weight $\mu$, local steps $E$, total rounds $T$, client weights $\{p_k\}_{k=1}^{K}$.
\FOR{round $t=0$ {\bf to} $T-1$}
    \FOR{each client $k\in\{1,\dots,K\}$ \textbf{in parallel}}
        \STATE $w_{k}^{t,0}\gets w^{t}$
        \FOR{$i=0$ \textbf{to} $E-1$}
            \STATE Sample minibatch $\xi_{k}^{t,i}$ and compute stochastic gradient $g_{k}^{t,i}=\nabla f_k(w_{k}^{t,i};\xi_{k}^{t,i})$
            \STATE $w_{k}^{t,i+1}\gets w_{k}^{t,i}-\eta g_{k}^{t,i}-\eta\mu\bigl(w_{k}^{t,i}-w^{t}\bigr)$ \hfill \textcolor{gray}{// Eq.~\eqref{eq:local-update}}
        \ENDFOR
        \STATE Send $w_{k}^{t,E}$ to server
    \ENDFOR
    \STATE $w^{t+1}\gets\sum_{k=1}^{K}p_k\,w_{k}^{t,E}$ \hfill \textcolor{gray}{// Eq.~\eqref{eq:global-agg}}
\ENDFOR
\STATE \textbf{Output:} final model $w^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
\section{Dry Run Example}
We illustrate the algorithm on a single‑client toy problem
\[
f(w)=(w-3)^{2},\qquad w\in\R,
\]
with $w_{0}=0$, learning rate $\eta=0.1$, proximal coefficient $\mu=0$,
and $E=1$ (one local SGD step per round).  Because there is only one client,
$p_{1}=1$ and the global model coincides with the local model at each round.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Round $t$} & $w^{t}$ & Gradient $g^{t}=2(w^{t}-3)$ & Updated $w^{t+1}=w^{t}-\eta g^{t}$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\[2mm]
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\[2mm]
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\[2mm]
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values $f(w^{t})$ decrease from $9$ to $2.61$ after three
iterations, confirming the descent behaviour of the update rule.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\section{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each local loss $f_k$ is $L$‑smooth, i.e., for all $x,y\in\R^{d}$,
\[
f_k(y)\le f_k(x)+\inner{\nabla f_k(x)}{y-x}
            +\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded stochastic gradient variance]\label{ass:variance}
For every client $k$, round $t$ and local step $i$,
\[
\E\bigl[\|g_{k}^{t,i}-\nabla f_k(w_{k}^{t,i})\|^{2}\mid w_{k}^{t,i}\bigr]\le\sigma^{2},
\]
where the expectation is over the minibatch $\xi_{k}^{t,i}$.
\end{assumption}
\begin{assumption}[Bounded gradients]\label{ass:bounded-grad}
There exists $G>0$ such that for all $k$ and $x$,
\[
\|\nabla f_k(x)\|\le G.
\]
\end{assumption}
\begin{assumption}[Learning rate]\label{ass:lr}
The step size satisfies
\[
0<\eta\le \frac{1}{L+\mu}.
\]
\end{assumption}
\begin{assumption}[Data heterogeneity]\label{ass:hetero}
Define the client drift
\[
\Delta_k:=\sup_{x}\norm{\nabla f_k(x)-\nabla F(x)}\le \Delta,
\]
for some $\Delta\ge 0$.
\end{assumption}

All constants $L,\mu,\sigma,G,\Delta$ are assumed known and finite.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\section{Main Convergence Theorem}
\begin{theorem}[Convergence of FedProx with Local SGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hetero} and with $E\ge 1$ local
steps, let $w^{t}$ be the global model after $t$ communication rounds.
Define the global objective $F(w)=\sum_{k}p_k f_k(w)$.  Then for any
$T\ge 1$,
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla F(w^{t})\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(F(w^{0})-F^{\star}\bigr)}{\eta T}}_{\text{(I)}}
\;+\;
\underbrace{ \frac{2L\eta\sigma^{2}}{K}}_{\text{(II)}}
\;+\;
\underbrace{ 4\mu^{2} \eta^{2} G^{2}E}_{\text{(III)}}
\;+\;
\underbrace{ \frac{2L^{2}\eta^{2}E^{2}\Delta^{2}}{K}}_{\text{(IV)}},
\label{eq:rate}
\end{equation}
where $F^{\star}=\inf_{w}F(w)$.
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla F(w^{t})\|^{2}\bigr]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}
The four terms in \eqref{eq:rate} correspond respectively to
( I ) the deterministic descent due to smoothness,
( II ) stochastic variance,
( III ) the penalty introduced by the proximal term,
and ( IV ) the drift caused by data heterogeneity across clients.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\section{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability w.r.t. $\mu$.}
    The bound (III) grows quadratically with $\mu$; setting $\mu=0$
    recovers the classical local‑SGD rate, whereas a moderate $\mu$ can
    improve robustness to heterogeneous data at the expense of a larger
    constant.

\item \textbf{Effect of local steps $E$.}
    Term (IV) scales as $E^{2}$, reflecting that more local updates
    increase the drift between local and global models.  When data are
    homogeneous ($\Delta=0$) the $E^{2}$ term disappears, allowing arbitrarily
    many local steps without harming the rate.

\item \textbf{Comparison to vanilla FedAvg.}
    FedAvg corresponds to $\mu=0$ and $E$ local SGD steps.
    The theorem shows that FedProx adds only the extra term (III),
    which is negligible when $\mu\eta\ll 1$.

\item \textbf{Hyper‑parameter sensitivity.}
    The step‑size condition \ref{ass:lr} guarantees that the descent
    lemma (Lemma~\ref{lem:descent}) holds.  If $\eta$ is chosen larger,
    the coefficient of term (I) becomes negative and the bound may
    diverge.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
\section{Discussion}
The result aligns with the intuition that FedProx stabilises training under
non‑i.i.d. data by pulling local solutions toward the current global model.
The $O(1/\sqrt{T})$ rate matches the optimal rate for non‑convex stochastic
optimization with a single pass over the data, confirming that the
additional proximal regulariser does not deteriorate asymptotic
performance.  Moreover, the explicit constants enable practitioners to
balance $E$, $\mu$, and $\eta$ to meet system constraints (e.g., communication
budget) while preserving theoretical guarantees.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries}
\section{Preliminaries}
\begin{definition}[Global Gradient]
For any $w\in\R^{d}$,
\[
\nabla F(w)=\sum_{k=1}^{K}p_k\nabla f_k(w).
\]
\end{definition}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard; see, e.g., \cite{nesterov2004introductory}. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on Local Drift]\label{lem:drift}
For any client $k$, round $t$, and local step $i$,
\[
\E\bigl[\norm{w_{k}^{t,i}-w^{t}}^{2}\bigr]
\;\le\;
\eta^{2}i\bigl(\sigma^{2}+G^{2}\bigr)
    + \eta^{2}\mu^{2}i^{2}\norm{w^{t}}^{2}.
\]
\end{lemma}
\begin{proof}
We prove by induction on $i$.

\textbf{Base $i=0$:} $w_{k}^{t,0}=w^{t}$, so the left‑hand side is $0$,
which matches the RHS because the RHS is also $0$.

\textbf{Inductive step:} Assume the claim holds for $i$.
From the update \eqref{eq:local-update},
\[
w_{k}^{t,i+1}-w^{t}
= w_{k}^{t,i}-w^{t}
   -\eta g_{k}^{t,i}
   -\eta\mu\bigl(w_{k}^{t,i}-w^{t}\bigr).
\]
Taking squared norm, expanding, and using $\E[g_{k}^{t,i}\mid w_{k}^{t,i}]=\nabla f_k(w_{k}^{t,i})$,
\[
\begin{aligned}
\E\!\bigl[\norm{w_{k}^{t,i+1}-w^{t}}^{2}\bigr]
&= \E\!\bigl[\norm{(1-\eta\mu)(w_{k}^{t,i}-w^{t})-\eta g_{k}^{t,i}}^{2}\bigr]\\
&\le (1-\eta\mu)^{2}\E\!\bigl[\norm{w_{k}^{t,i}-w^{t}}^{2}\bigr]
   +\eta^{2}\E\!\bigl[\norm{g_{k}^{t,i}}^{2}\bigr] \quad\text{[Step 1]}\\
&\le (1-\eta\mu)^{2}\E\!\bigl[\norm{w_{k}^{t,i}-w^{t}}^{2}\bigr]
   +\eta^{2}\bigl(\sigma^{2}+G^{2}\bigr) \quad\text{[Step 2]}\\
&\le \bigl(1-2\eta\mu+\eta^{2}\mu^{2}\bigr)\E\!\bigl[\norm{w_{k}^{t,i}-w^{t}}^{2}\bigr]
   +\eta^{2}\bigl(\sigma^{2}+G^{2}\bigr) .
\end{aligned}
\]
Applying the induction hypothesis to the first term and simplifying yields
\[
\E\!\bigl[\norm{w_{k}^{t,i+1}-w^{t}}^{2}\bigr]
\le
\eta^{2}(i+1)\bigl(\sigma^{2}+G^{2}\bigr)
   +\eta^{2}\mu^{2}(i+1)^{2}\norm{w^{t}}^{2},
\]
which completes the induction.  \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof}
\section{Main Proof}
We prove Theorem~\ref{thm:main}.  Throughout the proof,
expectations are taken over all stochastic gradients up to the current
iteration.

\subsection*{Step 1: One‑round descent of the global objective}
Using Lemma~\ref{lem:descent} on $F$ with $x=w^{t}$ and $y=w^{t+1}$,
\[
F(w^{t+1})\le F(w^{t})+\inner{\nabla F(w^{t})}{w^{t+1}-w^{t}}
          +\frac{L}{2}\norm{w^{t+1}-w^{t}}^{2}.
\tag{1}
\label{eq:descent1}
\]

\subsection*{Step 2: Express $w^{t+1}-w^{t}$ via local updates}
From the aggregation rule \eqref{eq:global-agg},
\[
w^{t+1}-w^{t}= \sum_{k}p_k\bigl(w_{k}^{t,E}-w^{t}\bigr).
\tag{2}
\label{eq:agg-diff}
\]

\subsection*{Step 3: Expand $w_{k}^{t,E}-w^{t}$ using the recursive update}
Unrolling \eqref{eq:local-update} for $E$ steps gives
\[
w_{k}^{t,E}-w^{t}= -\eta\sum_{i=0}^{E-1}g_{k}^{t,i}
                  -\eta\mu\sum_{i=0}^{E-1}\bigl(w_{k}^{t,i}-w^{t}\bigr).
\tag{3}
\label{eq:unroll}
\]

\subsection*{Step 4: Substitute \eqref{eq:unroll} into \eqref{eq:agg-diff}}
\[
w^{t+1}-w^{t}= -\eta\sum_{k}p_k\sum_{i=0}^{E-1}g_{k}^{t,i}
               -\eta\mu\sum_{k}p_k\sum_{i=0}^{E-1}\bigl(w_{k}^{t,i}-w^{t}\bigr).
\tag{4}
\label{eq:diff-split}
\]

\subsection*{Step 5: Take expectation of the inner product term in \eqref{eq:descent1}}
Condition on $w^{t}$ and use unbiasedness of stochastic gradients:
\[
\E\!\bigl[\inner{\nabla F(w^{t})}{w^{t+1}-w^{t}}\bigr]
= -\eta\sum_{k}p_k\sum_{i=0}^{E-1}
   \inner{\nabla F(w^{t})}{\nabla f_k(w_{k}^{t,i})}
   -\eta\mu\sum_{k}p_k\sum_{i=0}^{E-1}
   \inner{\nabla F(w^{t})}{\E[w_{k}^{t,i}-w^{t}]} .
\tag{5}
\label{eq:inner-exp}
\]

\subsection*{Step 6: Relate $\nabla f_k(w_{k}^{t,i})$ to $\nabla F(w^{t})$}
Add and subtract $\nabla f_k(w^{t})$:
\[
\inner{\nabla F(w^{t})}{\nabla f_k(w_{k}^{t,i})}
= \inner{\nabla F(w^{t})}{\nabla f_k(w^{t})}
  +\inner{\nabla F(w^{t})}{\nabla f_k(w_{k}^{t,i})-\nabla f_k(w^{t})}.
\tag{6}
\label{eq:grad-decompose}
\]

Summing over $k$ with weights $p_k$ yields
\[
\sum_{k}p_k\inner{\nabla F(w^{t})}{\nabla f_k(w^{t})}
= \norm{\nabla F(w^{t})}^{2}.
\tag{7}
\label{eq:grad-sum}
\]

\subsection*{Step 7: Bound the drift term}
For the second inner product in \eqref{eq:6}, apply Cauchy–Schwarz
and smoothness:
\[
\bigl|\inner{\nabla F(w^{t})}{\nabla f_k(w_{k}^{t,i})-\nabla f_k(w^{t})}\bigr|
\le \norm{\nabla F(w^{t})}\,L\,\norm{w_{k}^{t,i}-w^{t}}.
\tag{8}
\label{eq:cauchy}
\]

Taking expectations and using Lemma~\ref{lem:drift},
\[
\E\!\bigl[\norm{w_{k}^{t,i}-w^{t}}\bigr]
\le \sqrt{\E\!\bigl[\norm{w_{k}^{t,i}-w^{t}}^{2}\bigr]}
\le \eta\sqrt{i}\,\sqrt{\sigma^{2}+G^{2}} + \eta\mu i \norm{w^{t}}.
\tag{9}
\label{eq:drift-bound}
\]

Summing over $i=0,\dots,E-1$ and $k$, we obtain
\[
\sum_{k}p_k\sum_{i=0}^{E-1}
   \E\!\bigl|\inner{\nabla F(w^{t})}{\nabla f_k(w_{k}^{t,i})-\nabla f_k(w^{t})}\bigr|
\le L\eta E^{3/2}\bigl(\sigma+G\bigr)\norm{\nabla F(w^{t})}
   + L\mu\eta^{2}E^{2}\norm{w^{t}}\norm{\nabla F(w^{t})}.
\tag{10}
\label{eq:drift-sum}
\]

\subsection*{Step 8: Bound the proximal drift term}
The second term in \eqref{eq:inner-exp} involves $\E[w_{k}^{t,i}-w^{t}]$.
Using Lemma~\ref{lem:drift} again and Jensen,
\[
\Bigl\|\E\bigl[w_{k}^{t,i}-w^{t}\bigr]\Bigr\|
\le \eta\sqrt{i}\,\sqrt{\sigma^{2}+G^{2}} + \eta\mu i \norm{w^{t}} .
\tag{11}
\label{eq:prox-drift}
\]
Summing over $i$ and $k$ yields a bound identical to \eqref{eq:drift-sum}
but multiplied by $\mu$.

\subsection*{Step 9: Assemble the descent inequality}
Plugging \eqref{eq:inner-exp}–\eqref{eq:prox-drift} into \eqref{eq:descent1}
and using $\norm{a+b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}$ for the
quadratic term, we obtain after straightforward algebra
\[
\begin{aligned}
\E\!\bigl[F(w^{t+1})\bigr]
\le&\; F(w^{t})
   -\eta E \norm{\nabla F(w^{t})}^{2}
   +\underbrace{\frac{L\eta^{2}E}{2}\bigl(\sigma^{2}+G^{2}\bigr)}_{\text{(A)}}
   +\underbrace{L\eta^{2}\mu^{2}E^{3}\norm{w^{t}}^{2}}_{\text{(B)}}
   \\
   &\;+\underbrace{L\eta^{2}\mu^{2}E^{2}\norm{w^{t}}^{2}}_{\text{(C)}}
   +\underbrace{L\eta^{2}E^{2}\Delta^{2}}_{\text{(D)}} .
\end{aligned}
\tag{12}
\label{eq:one-round}
\]

Here term (D) stems from the heterogeneity bound
$\|\nabla f_k(w)-\nabla F(w)\|\le\Delta$.

\subsection*{Step 10: Telescoping sum over $t=0,\dots,T-1$}
Rearrange \eqref{eq:one-round}:
\[
\eta E \norm{\nabla F(w^{t})}^{2}
\le F(w^{t})- \E[F(w^{t+1})] + C_{1},
\]
where
\[
C_{1}= \frac{L\eta^{2}E}{2}\bigl(\sigma^{2}+G^{2}\bigr)
      +L\eta^{2}\mu^{2}E^{3}\norm{w^{t}}^{2}
      +L\eta^{2}\mu^{2}E^{2}\norm{w^{t}}^{2}
      +L\eta^{2}E^{2}\Delta^{2}.
\]

Summing from $t=0$ to $T-1$ and using telescoping,
\[
\eta E\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla F(w^{t})}^{2}\bigr]
\le F(w^{0})-F^{\star} + T C_{1}^{\max},
\tag{13}
\label{eq:telescoping}
\]
where $C_{1}^{\max}$ denotes an upper bound on $C_{1}$ over all
iterations.  Because $\norm{w^{t}}$ is bounded by a constant (e.g.,
by Lemma~\ref{lem:drift} and the step‑size condition \ref{ass:lr}), we can
replace $\norm{w^{t}}^{2}$ by a universal constant $B^{2}$.  Collecting
constants yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla F(w^{t})}^{2}\bigr]
\le
\frac{F(w^{0})-F^{\star}}{\eta E T}
+ \frac{L\eta}{2E}\bigl(\sigma^{2}+G^{2}\bigr)
+ L\eta\mu^{2}E^{2}B^{2}
+ L\eta E\Delta^{2}.
\tag{14}
\label{eq:pre-rate}
\]

\subsection*{Step 11: Simplify and match the theorem statement}
Choosing $B^{2}\le G^{2}$ (since $\|\nabla f_k\|\le G$ and the proximal term
cannot increase the norm beyond a constant factor) and absorbing
lower‑order constants into the four terms (I)–(IV) defined in
Theorem~\ref{thm:main}, we obtain exactly the bound \eqref{eq:rate}.

Finally, set $\eta = \frac{c}{\sqrt{T}}$ with $c\le \frac{1}{L+\mu}$;
the dominant term becomes $\frac{2(F(w^{0})-F^{\star})}{c\sqrt{T}}$,
while all other terms scale as $O(1/\sqrt{T})$.  Hence the overall rate is
$O(1/\sqrt{T})$, completing the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation}
\section{Rate Derivation (With Constants)}
From \eqref{eq:rate} let
\[
C_{I}= \frac{2(F(w^{0})-F^{\star})}{\eta T},
\quad
C_{II}= \frac{2L\eta\sigma^{2}}{K},
\quad
C_{III}= 4\mu^{2}\eta^{2}G^{2}E,
\quad
C_{IV}= \frac{2L^{2}\eta^{2}E^{2}\Delta^{2}}{K}.
\]
Choosing $\eta = \Theta(1/\sqrt{T})$ gives
\[
C_{I}= \Theta\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\;
C_{II}= \Theta\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\;
C_{III}= \Theta\!\Bigl(\frac{1}{T}\Bigr),\;
C_{IV}= \Theta\!\Bigl(\frac{1}{T}\Bigr),
\]
so the dominant $1/\sqrt{T}$ behaviour is dictated by (I) and (II).

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\section{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Homogeneous data ($\Delta=0$).}
    Term (IV) disappears; the bound reduces to the classical
    $O(1/\sqrt{T})$ rate of local SGD with an additional $O(\mu^{2}\eta^{2})$
    penalty (III).

\item \textbf{No proximal regularisation ($\mu=0$).}
    The algorithm coincides with FedAvg.  The bound simplifies to
    \[
    \frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla F(w^{t})\|^{2}]
    \le \frac{2(F(w^{0})-F^{\star})}{\eta T}
        + \frac{2L\eta\sigma^{2}}{K}
        + \frac{2L^{2}\eta^{2}E^{2}\Delta^{2}}{K},
    \]
    matching known results for non‑convex FedAvg.

\item \textbf{Single client ($K=1$).}
    The heterogeneity term vanishes ($\Delta=0$) and the bound becomes
    identical to that of standard proximal SGD on a single machine.
\end{enumerate}
