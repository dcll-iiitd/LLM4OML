\section*{Algorithm Name}
\textbf{Randomized Coordinate Descent (RCD) for Non-Convex Coordinate-wise Smooth Functions}

\section*{Mathematical Formulation}

Let $f: \R^d \to \R$ be a differentiable function. For each coordinate $i \in \{1,\dots,d\}$, assume the partial derivative $\nabla_i f$ is $L_i$-Lipschitz continuous (coordinate-wise smoothness). The algorithm maintains an iterate $w_t \in \R^d$ and updates a single randomly chosen coordinate at each step.

\begin{itemize}
    \item \textbf{Probability distribution:} Choose coordinate $i$ with probability $p_i > 0$, where $\sum_{i=1}^d p_i = 1$.
    \item \textbf{Step sizes:} $\gamma_i = \frac{1}{L_i}$ for each coordinate $i$ (or any $\gamma_i \in (0, 2/L_i]$; we use $\gamma_i = 1/L_i$ for optimal constant).
    \item \textbf{Update rule:} Given current iterate $w_t$, sample $i_t \sim \{p_1,\dots,p_d\}$ and set
    \begin{equation}
        w_{t+1}^{i_t} = w_t^{i_t} - \gamma_{i_t} \nabla_{i_t} f(w_t), \qquad
        w_{t+1}^j = w_t^j \;\; \forall j \neq i_t.
    \end{equation}
    In vector form, $w_{t+1} = w_t - \gamma_{i_t} \nabla_{i_t} f(w_t) e_{i_t}$, where $e_i$ is the $i$-th standard basis vector.
\end{itemize}

\section*{Pseudocode}

\begin{algorithm}
\caption{Randomized Coordinate Descent (RCD)}
\begin{algorithmic}
\REQUIRE Initial point $w_0 \in \R^d$, probabilities $\{p_i\}_{i=1}^d$, coordinate-wise Lipschitz constants $\{L_i\}_{i=1}^d$, max iterations $T$.
\STATE Compute step sizes $\gamma_i = 1/L_i$ for $i=1,\dots,d$.
\FOR{$t = 0, 1, \dots, T-1$}
    \STATE Sample coordinate $i_t \in \{1,\dots,d\}$ with $\Pr(i_t = i) = p_i$.
    \STATE Compute partial derivative $g_t = \nabla_{i_t} f(w_t)$.
    \STATE Update $w_{t+1} = w_t - \gamma_{i_t} g_t e_{i_t}$.
\ENDFOR
\RETURN $w_T$ (or average iterate $\bar{w}_T = \frac{1}{T}\sum_{t=0}^{T-1} w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

Consider the 1-dimensional function $f(w) = (w-3)^2$ with $w_0 = 0$, step size $\eta = 0.1$ (here $L=2$, so $\gamma = 1/L = 0.5$; we use the given $\eta=0.1$ for illustration). Since $d=1$, $p_1=1$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iteration $t$} & $w_t$ & $\nabla f(w_t) = 2(w_t-3)$ & $w_{t+1} = w_t - \eta \nabla f(w_t)$ & $f(w_{t+1})$ \\
\hline
0 & 0.0000 & $2(0-3) = -6.0000$ & $0 - 0.1(-6) = 0.6000$ & $(0.6-3)^2 = 5.7600$ \\
1 & 0.6000 & $2(0.6-3) = -4.8000$ & $0.6 - 0.1(-4.8) = 1.0800$ & $(1.08-3)^2 = 3.6864$ \\
2 & 1.0800 & $2(1.08-3) = -3.8400$ & $1.08 - 0.1(-3.84) = 1.4640$ & $(1.464-3)^2 = 2.3593$ \\
\end{tabular}
\end{center}

After 3 iterations, the objective value decreases from $f(0)=9$ to $f(1.464)\approx 2.36$.

\section*{Assumptions}

\begin{assumption}[Coordinate-wise Smoothness]
\label{assump:smooth}
For each $i \in \{1,\dots,d\}$, the partial derivative $\nabla_i f$ is $L_i$-Lipschitz continuous: for all $w \in \R^d$ and $h \in \R$,
\begin{equation}
|\nabla_i f(w + h e_i) - \nabla_i f(w)| \le L_i |h|.
\end{equation}
Equivalently, for all $w \in \R^d$ and $h \in \R$,
\begin{equation}
f(w + h e_i) \le f(w) + h \nabla_i f(w) + \frac{L_i}{2} h^2. \tag{Descent Lemma}
\end{equation}
\end{assumption}

\begin{assumption}[Lower Boundedness]
\label{assump:bounded}
The function $f$ is bounded below: there exists $f^* \in \R$ such that $f(w) \ge f^*$ for all $w \in \R^d$.
\end{assumption}

\begin{assumption}[Sampling Distribution]
\label{assump:sampling}
At each iteration $t$, the coordinate $i_t$ is sampled independently from a fixed distribution $\{p_i\}_{i=1}^d$ with $p_i > 0$ and $\sum_{i=1}^d p_i = 1$.
\end{assumption}

\begin{assumption}[Exact Partial Derivatives]
\label{assump:exact}
The algorithm has access to the exact partial derivative $\nabla_i f(w)$ for any $w$ and $i$.
\end{assumption}

\section*{Main Convergence Theorem}

\begin{theorem}[Convergence Rate for Non-Convex RCD]
\label{thm:main}
Under Assumptions \ref{assump:smooth}--\ref{assump:exact}, let $\{w_t\}_{t=0}^T$ be the sequence generated by RCD with step sizes $\gamma_i = 1/L_i$. Then for any $T \ge 1$,
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}^2 \right]
\le \frac{2 \max_{i} \frac{L_i}{p_i} \left( f(w_0) - f^* \right)}{T}.
\end{equation}
Equivalently, in terms of the weighted gradient norm $\norm{\nabla f(w)}_L^2 := \sum_{i=1}^d \frac{(\nabla_i f(w))^2}{L_i}$,
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}_L^2 \right]
\le \frac{2 \left( f(w_0) - f^* \right)}{T}.
\end{equation}
\end{theorem}

\section*{Theoretical Properties}

\begin{property}[Dimension Dependence]
The rate depends on $\max_i L_i/p_i$. With uniform sampling $p_i = 1/d$, the bound becomes $2d \max_i L_i (f(w_0)-f^*)/T$, which scales linearly with dimension $d$. With importance sampling $p_i \propto L_i$, the constant becomes $2 (\sum_i L_i) (f(w_0)-f^*)/T$, removing the explicit $d$ factor.
\end{property}

\begin{property}[Step Size Sensitivity]
The step size $\gamma_i = 1/L_i$ is the largest constant step size guaranteeing the descent lemma inequality. Larger step sizes may violate the bound; smaller step sizes yield a worse constant ($1 - L_i\gamma_i/2$ instead of $1/2$).
\end{property}

\begin{property}[Comparison to Gradient Descent]
Full gradient descent on an $L$-smooth non-convex function achieves $\frac{1}{T}\sum \E[\norm{\nabla f}^2] \le \frac{2L(f(w_0)-f^*)}{T}$. RCD with uniform sampling replaces $L$ by $d \max_i L_i$, reflecting the slower progress per iteration (only one coordinate updated). However, each RCD iteration is $d$ times cheaper (computes one partial derivative vs. full gradient).
\end{property}

\section*{Discussion}

The theorem establishes an $O(1/T)$ convergence rate in terms of the expected squared gradient norm (or weighted norm) for non-convex coordinate-wise smooth functions. This matches the rate of full gradient descent up to constants depending on the sampling distribution. The proof technique using $1/p_i$ weighting is crucial for handling arbitrary probabilities and obtaining the tight constant $\max_i L_i/p_i$. The result holds for any fixed probabilities $p_i>0$; adaptive or non-uniform sampling can improve the constant but does not change the $O(1/T)$ rate.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Coordinate-wise Descent Lemma]
Under Assumption \ref{assump:smooth}, for any $w \in \R^d$, coordinate $i$, and step $h \in \R$,
\begin{equation}
f(w + h e_i) \le f(w) + h \nabla_i f(w) + \frac{L_i}{2} h^2.
\end{equation}
\end{definition}

\begin{lemma}[One-Step Expected Decrease]
\label{lem:one_step}
Let $w_t$ be the current iterate and $i_t \sim \{p_i\}$. Using step size $\gamma_i = 1/L_i$, the next iterate satisfies
\begin{equation}
\E\left[ f(w_{t+1}) \mid w_t \right] \le f(w_t) - \frac{1}{2} \sum_{i=1}^d \frac{p_i}{L_i} \left( \nabla_i f(w_t) \right)^2.
\end{equation}
\end{lemma}
\begin{proof}
Apply the descent lemma with $h = -\gamma_{i_t} \nabla_{i_t} f(w_t)$:
\begin{align}
f(w_{t+1}) &\le f(w_t) - \gamma_{i_t} \left( \nabla_{i_t} f(w_t) \right)^2 + \frac{L_{i_t}}{2} \gamma_{i_t}^2 \left( \nabla_{i_t} f(w_t) \right)^2 \step{1} \\
&= f(w_t) - \gamma_{i_t} \left(1 - \frac{L_{i_t}\gamma_{i_t}}{2}\right) \left( \nabla_{i_t} f(w_t) \right)^2 \step{2} \\
&= f(w_t) - \frac{1}{2L_{i_t}} \left( \nabla_{i_t} f(w_t) \right)^2 \quad \text{(since $\gamma_i = 1/L_i$)} \step{3}.
\end{align}
Take conditional expectation w.r.t. $i_t$:
\begin{equation}
\E\left[ f(w_{t+1}) \mid w_t \right] \le f(w_t) - \sum_{i=1}^d p_i \frac{1}{2L_i} \left( \nabla_i f(w_t) \right)^2. \step{4}
\end{equation}
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}

We now prove Theorem \ref{thm:main} by telescoping the expected decrease over $T$ iterations.

\begin{proof}[Proof of Theorem \ref{thm:main}]

\step{5} From Lemma \ref{lem:one_step}, take total expectation over the history up to time $t$:
\begin{equation}
\E\left[ f(w_{t+1}) \right] \le \E\left[ f(w_t) \right] - \frac{1}{2} \E\left[ \sum_{i=1}^d \frac{p_i}{L_i} \left( \nabla_i f(w_t) \right)^2 \right].
\end{equation}

\step{6} Rearranging and summing from $t=0$ to $T-1$:
\begin{equation}
\sum_{t=0}^{T-1} \frac{1}{2} \E\left[ \sum_{i=1}^d \frac{p_i}{L_i} \left( \nabla_i f(w_t) \right)^2 \right]
\le \sum_{t=0}^{T-1} \left( \E[f(w_t)] - \E[f(w_{t+1})] \right)
= \E[f(w_0)] - \E[f(w_T)].
\end{equation}

\step{7} By Assumption \ref{assump:bounded}, $\E[f(w_T)] \ge f^*$. Also $\E[f(w_0)] = f(w_0)$ since $w_0$ is deterministic. Thus:
\begin{equation}
\sum_{t=0}^{T-1} \E\left[ \sum_{i=1}^d \frac{p_i}{L_i} \left( \nabla_i f(w_t) \right)^2 \right]
\le 2 \left( f(w_0) - f^* \right).
\end{equation}

\step{8} \textbf{Weighting by $1/p_i$ to obtain the full gradient norm.}
We relate the weighted sum $\sum_i \frac{p_i}{L_i} (\nabla_i f)^2$ to the standard squared gradient norm $\norm{\nabla f}^2 = \sum_i (\nabla_i f)^2$.
For any coordinate $i$, since $p_i > 0$, we have $\frac{p_i}{L_i} \ge \frac{p_i}{\max_j L_j} \cdot \frac{\max_j L_j}{L_i}$? Actually, we use the inequality:
\begin{equation}
\sum_{i=1}^d \frac{p_i}{L_i} (\nabla_i f)^2
\ge \frac{1}{\max_i \frac{L_i}{p_i}} \sum_{i=1}^d (\nabla_i f)^2
= \frac{1}{\max_i \frac{L_i}{p_i}} \norm{\nabla f}^2. \step{9}
\end{equation}
\step{9} \textit{Justification:} For each $i$, $\frac{p_i}{L_i} = \frac{1}{L_i/p_i} \ge \frac{1}{\max_j L_j/p_j}$. Multiplying by $(\nabla_i f)^2 \ge 0$ and summing gives the inequality.

\step{10} Substitute this lower bound into the summed inequality (Step 7):
\begin{equation}
\sum_{t=0}^{T-1} \E\left[ \frac{1}{\max_i \frac{L_i}{p_i}} \norm{\nabla f(w_t)}^2 \right]
\le 2 \left( f(w_0) - f^* \right).
\end{equation}

\step{11} Since $\max_i \frac{L_i}{p_i}$ is a constant, we can factor it out:
\begin{equation}
\frac{1}{\max_i \frac{L_i}{p_i}} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}^2 \right]
\le 2 \left( f(w_0) - f^* \right).
\end{equation}

\step{12} Divide both sides by $T$ and multiply by $\max_i \frac{L_i}{p_i}$ to obtain the final rate:
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}^2 \right]
\le \frac{2 \max_i \frac{L_i}{p_i} \left( f(w_0) - f^* \right)}{T}.
\end{equation}

\step{13} \textbf{Weighted norm version.}
Alternatively, from Step 7 directly, divide by $T$:
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \sum_{i=1}^d \frac{(\nabla_i f(w_t))^2}{L_i} \right]
\le \frac{2 \left( f(w_0) - f^* \right)}{T}.
\end{equation}
The left-hand side is exactly $\frac{1}{T}\sum_{t=0}^{T-1} \E[\norm{\nabla f(w_t)}_L^2]$. This completes the proof.
\end{proof}

\section*{Rate Derivation (With Constants)}

The explicit constants in the convergence rate are:
\begin{itemize}
    \item \textbf{Standard gradient norm:} $C = 2 \max_i \frac{L_i}{p_i} (f(w_0)-f^*)$.
    \item \textbf{Weighted gradient norm:} $C = 2 (f(w_0)-f^*)$.
\end{itemize}
The number of iterations to achieve $\frac{1}{T}\sum_{t=0}^{T-1} \E[\norm{\nabla f(w_t)}^2] \le \epsilon$ is
\begin{equation}
T \ge \frac{2 \max_i \frac{L_i}{p_i} (f(w_0)-f^*)}{\epsilon}.
\end{equation}
With uniform sampling $p_i = 1/d$, this becomes $T \ge \frac{2 d \max_i L_i (f(w_0)-f^*)}{\epsilon}$.
With importance sampling $p_i = L_i / \sum_j L_j$, this becomes $T \ge \frac{2 (\sum_j L_j) (f(w_0)-f^*)}{\epsilon}$.

\section*{Special Cases}

\begin{corollary}[Uniform Sampling]
If $p_i = 1/d$ for all $i$, then
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}^2 \right]
\le \frac{2 d \max_i L_i (f(w_0)-f^*)}{T}.
\end{equation}
\end{corollary}

\begin{corollary}[Importance Sampling]
If $p_i = L_i / \sum_{j=1}^d L_j$, then
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \E\left[ \norm{\nabla f(w_t)}^2 \right]
\le \frac{2 \left( \sum_{j=1}^d L_j \right) (f(w_0)-f^*)}{T}.
\end{equation}
\end{corollary}

\begin{corollary}[Convex Case]
If additionally $f$ is convex, the same proof yields convergence in function value: $\E[f(\bar{w}_T)] - f^* \le O(1/T)$ where $\bar{w}_T = \frac{1}{T}\sum_{t=0}^{T-1} w_t$. The proof uses convexity to relate $f(w_t)-f^*$ to $\inner{\nabla f(w_t)}{w_t - w^*}$ and a similar telescoping sum on $\norm{w_t - w^*}^2$.
\end{corollary}

\begin{corollary}[Stochastic Partial Derivatives]
If instead of exact $\nabla_i f(w_t)$ we use an unbiased stochastic estimator $g_t$ with $\E[g_t \mid w_t] = \nabla_{i_t} f(w_t)$ and bounded variance $\E[(g_t - \nabla_{i_t} f(w_t))^2 \mid w_t] \le \sigma^2$, the rate becomes $O(1/\sqrt{T})$ for non-convex functions. The proof adds a variance term $\frac{\gamma_{i_t}}{2}\sigma^2$ in the one-step decrease and uses decreasing step sizes $\gamma_i \propto 1/\sqrt{t}$.
\end{corollary}
