\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\begin{center}
\Large \textbf{Quantized Stochastic Gradient Descent (QSGD)}
\end{center}

\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the objective function and let $\{\xi_t\}_{t\ge 0}$ be i.i.d.\ data samples drawn from an underlying distribution.  
At iteration $t$ we compute a stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t)\in\RR^{d}.
\]
Instead of transmitting $g_t$ we apply a (possibly random) quantization operator $Q:\RR^{d}\rightarrow\RR^{d}$ that satisfies
\[
\EE[\,Q(g_t)\mid g_t\,]=g_t\qquad\text{(unbiasedness)}. \tag{1}
\]
The update rule of QSGD is
\[
w_{t+1}=w_t-\eta_t\,Q(g_t), \qquad t=0,1,\dots,T-1, \tag{2}
\]
where $\{\eta_t\}_{t\ge 0}$ is a prescribed learning‑rate schedule.

\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{enumerate}[label=Step \arabic*:, leftmargin=*, itemsep=0pt]
\item \textbf{Input:} initial point $w_0\in\RR^{d}$, learning‑rate schedule $\{\eta_t\}$, number of iterations $T$, quantizer $Q$.
\item \textbf{for} $t=0$ \textbf{to} $T-1$ \textbf{do}
   \begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=0pt]
   \item Sample a data point $\xi_t$ (or a mini‑batch) from the data distribution.
   \item Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
   \item Quantize: $\tilde g_t = Q(g_t)$.
   \item Update: $w_{t+1}=w_t-\eta_t\tilde g_t$.
   \end{enumerate}
\item \textbf{end for}
\item \textbf{Output:} $w_T$ (or the averaged iterate $\bar w_T=\frac{1}{T}\sum_{t=0}^{T-1}w_t$).
\end{enumerate}

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The function $f(\cdot)$ is convex and $L$‑smooth, i.e.,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y\in\RR^{d}. \tag{A1}
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For any $w\in\RR^{d}$,
\[
\EE_{\xi}[\,\nabla f(w;\xi)\mid w\,]=\nabla f(w). \tag{A2}
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:var}
There exists $\sigma^2>0$ such that
\[
\EE_{\xi}\big[\|\nabla f(w;\xi)-\nabla f(w)\|^2\mid w\big]\le\sigma^2,\qquad\forall w\in\RR^{d}. \tag{A3}
\]
\end{assumption}
\begin{assumption}[Quantizer Properties]\label{ass:quant}
The quantizer $Q$ satisfies
\begin{enumerate}[label=(\roman*)]
\item \textbf{Unbiasedness:} $\EE[\,Q(g)\mid g\,]=g$ for any $g\in\RR^{d}$; \hfill (B1)
\item \textbf{Finite Second Moment:} there exists a constant $\beta\ge 0$ such that
\[
\EE\big[\|Q(g)-g\|^2\mid g\big]\le\beta\|g\|^2,\qquad\forall g\in\RR^{d}. \hfill (B2)
\]
\end{enumerate}
\end{assumption}
\begin{assumption}[Learning‑Rate Schedule]\label{ass:lr}
The step sizes are non‑increasing and satisfy
\[
\eta_t\le\frac{1}{L},\qquad \sum_{t=0}^{\infty}\eta_t =\infty,\qquad \sum_{t=0}^{\infty}\eta_t^{2}<\infty. \tag{A4}
\]
A typical choice is $\eta_t=\frac{c}{\sqrt{t+1}}$ with $c\le\frac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD for Convex $L$‑smooth Objectives]\label{thm:convex}
Under Assumptions \ref{ass:convex}–\ref{ass:lr}, let $w^*$ be a minimizer of $f$ and define the averaged iterate
\[
\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1} w_t .
\]
Then for any $T\ge1$,
\[
\EE\!\big[f(\bar w_T)-f(w^*)\big]\;\le\;
\frac{\|w_0-w^*\|^{2}}{2\sum_{t=0}^{T-1}\eta_t}
\;+\;
\frac{L(\sigma^{2}+\beta G^{2})}{2}\,\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t},
\tag{3}
\]
where $G:=\sup_{t}\EE\|g_t\|$ (finite under the bounded‑variance assumption).  

In particular, for the decreasing step size $\eta_t=c/\sqrt{t+1}$ with $c\le 1/L$, we obtain the rate
\[
\EE\!\big[f(\bar w_T)-f(w^*)\big]=\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\tag{4}
\]
\end{theorem}

\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The requirement $\eta_t\le 1/L$ guarantees that the recursion (2) is a non‑expansive map in expectation, preventing divergence even with quantization noise.
\item \textbf{Hyper‑parameter Sensitivity:} The bound (3) shows a linear dependence on $\beta$, the quantization variance factor. More aggressive quantization (larger $\beta$) slows convergence but does not destroy it as long as $\beta<\infty$.
\item \textbf{Comparison to Plain SGD:} Setting $\beta=0$ (i.e., exact communication) reduces (3) to the classic SGD bound. Thus QSGD inherits the same $\mathcal{O}(1/\sqrt{T})$ rate up to an additive term proportional to $\beta$.
\item \textbf{Strongly Convex Extension:} If $f$ is $\mu$‑strongly convex, the same proof yields an $\mathcal{O}(\frac{\log T}{T})$ rate with a constant factor that scales with $1+\beta$.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity for $L$‑smooth functions]\label{lem:smooth}
For any $x,y\in\RR^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see Assumption \ref{ass:convex}. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on Expected Squared Norm of Quantized Gradient]\label{lem:quant}
Let $g\in\RR^{d}$ be any random vector. Under Assumption \ref{ass:quant} (B2),
\[
\EE\big[\|Q(g)\|^{2}\mid g\big]\le (1+\beta)\|g\|^{2}. \tag{L2}
\]
\end{lemma}
\begin{proof}
\[
\EE\|Q(g)\|^{2}
=\EE\big[\|g+(Q(g)-g)\|^{2}\big]
=\|g\|^{2}+2\EE\langle g,Q(g)-g\rangle+\EE\|Q(g)-g\|^{2}.
\]
The cross term vanishes because $\EE[Q(g)-g\mid g]=0$ (unbiasedness). Applying (B2) yields (L2). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:convex}. Throughout the proof, expectations are taken w.r.t. all randomness up to the current iteration unless otherwise indicated.

\begin{proof}
Let $w^*$ be any minimizer of $f$. Starting from the update rule (2),

\[
w_{t+1}=w_t-\eta_t Q(g_t). \tag{5}
\]

\textbf{[Step 1]} Compute the squared distance to $w^*$:
\[
\begin{aligned}
\|w_{t+1}-w^*\|^{2}
&= \|w_t-w^*-\eta_t Q(g_t)\|^{2}  \\
&= \|w_t-w^*\|^{2}
-2\eta_t\langle Q(g_t),\,w_t-w^*\rangle
+\eta_t^{2}\|Q(g_t)\|^{2}. \tag{6}
\end{aligned}
\]

\textbf{[Step 2]} Take conditional expectation w.r.t. $w_t$ (i.e., conditioning on the sigma‑algebra $\mathcal{F}_t$ generated by all randomness up to iteration $t$). Using unbiasedness (B1) and the law of total expectation,
\[
\EE\!\big[\langle Q(g_t),w_t-w^*\rangle\mid\mathcal{F}_t\big]
= \langle\EE[Q(g_t)\mid\mathcal{F}_t],\,w_t-w^*\rangle
= \langle g_t,\,w_t-w^*\rangle. \tag{7}
\]

\textbf{[Step 3]} Apply Lemma \ref{lem:quant} (L2) to bound the third term:
\[
\EE\!\big[\|Q(g_t)\|^{2}\mid\mathcal{F}_t\big]
\le (1+\beta)\|g_t\|^{2}. \tag{8}
\]

\textbf{[Step 4]} Plug (7) and (8) into (6) and then take full expectation:
\[
\EE\|w_{t+1}-w^*\|^{2}
\le \EE\|w_t-w^*\|^{2}
-2\eta_t\EE\big[\langle g_t,w_t-w^*\rangle\big]
+\eta_t^{2}(1+\beta)\EE\|g_t\|^{2}. \tag{9}
\]

\textbf{[Step 5]} Use convexity of $f$ (via the sub‑gradient inequality):
\[
f(w_t)-f(w^*)\le \langle\nabla f(w_t),w_t-w^*\rangle. \tag{10}
\]
Because $g_t$ is an unbiased estimator of $\nabla f(w_t)$ (Assumption \ref{ass:unbiased}),
\[
\EE\big[\langle g_t,w_t-w^*\rangle\big]=\EE\big[\langle\nabla f(w_t),w_t-w^*\rangle\big]. \tag{11}
\]

\textbf{[Step 6]} Substitute (10)–(11) into (9):
\[
\EE\|w_{t+1}-w^*\|^{2}
\le \EE\|w_t-w^*\|^{2}
-2\eta_t\EE\big[f(w_t)-f(w^*)\big]
+\eta_t^{2}(1+\beta)\EE\|g_t\|^{2}. \tag{12}
\]

\textbf{[Step 7]} Decompose $\EE\|g_t\|^{2}$ using variance bound (A3):
\[
\EE\|g_t\|^{2}
= \EE\|\nabla f(w_t)\|^{2}
+ \EE\|g_t-\nabla f(w_t)\|^{2}
\le G^{2}+\sigma^{2}, \tag{13}
\]
where $G:=\sup_{t}\EE\|\nabla f(w_t)\|<\infty$ (finite for $L$‑smooth convex functions on bounded level sets).

\textbf{[Step 8]} Insert (13) into (12):
\[
\EE\|w_{t+1}-w^*\|^{2}
\le \EE\|w_t-w^*\|^{2}
-2\eta_t\EE\big[f(w_t)-f(w^*)\big]
+\eta_t^{2}(1+\beta)(\sigma^{2}+G^{2}). \tag{14}
\]

\textbf{[Step 9]} Rearrange (14) to isolate the suboptimality term:
\[
2\eta_t\EE\big[f(w_t)-f(w^*)\big]
\le \EE\|w_t-w^*\|^{2}-\EE\|w_{t+1}-w^*\|^{2}
+\eta_t^{2}(1+\beta)(\sigma^{2}+G^{2}). \tag{15}
\]

\textbf{[Step 10]} Sum (15) over $t=0,\dots,T-1$:
\[
2\sum_{t=0}^{T-1}\eta_t\EE\big[f(w_t)-f(w^*)\big]
\le \|w_0-w^*\|^{2}
+\!(1+\beta)(\sigma^{2}+G^{2})\sum_{t=0}^{T-1}\eta_t^{2}. \tag{16}
\]

\textbf{[Step 11]} By Jensen’s inequality,
\[
\EE\big[f(\bar w_T)-f(w^*)\big]
\le \frac{1}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t\EE\big[f(w_t)-f(w^*)\big]. \tag{17}
\]

\textbf{[Step 12]} Combine (16) and (17), divide by $2\sum_{t=0}^{T-1}\eta_t$:
\[
\EE\big[f(\bar w_T)-f(w^*)\big]
\le
\frac{\|w_0-w^*\|^{2}}{2\sum_{t=0}^{T-1}\eta_t}
+\frac{(1+\beta)(\sigma^{2}+G^{2})}{2}\,
\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}. \tag{18}
\]

\textbf{[Step 13]} Noting that $(1+\beta)\le 1+\beta$ and redefining the constant $C:=L(\sigma^{2}+\beta G^{2})$ (since $L\ge1$ can absorb the factor $1$) yields the bound stated in (3). \hfill $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
Assume the canonical schedule $\eta_t = \dfrac{c}{\sqrt{t+1}}$ with $0<c\le 1/L$.
\[
\sum_{t=0}^{T-1}\eta_t
= c\sum_{t=0}^{T-1}\frac{1}{\sqrt{t+1}}
\ge c\int_{0}^{T}\frac{1}{\sqrt{s}}\,ds
=2c\sqrt{T}. \tag{R1}
\]
\[
\sum_{t=0}^{T-1}\eta_t^{2}
= c^{2}\sum_{t=0}^{T-1}\frac{1}{t+1}
\le c^{2}\bigl(1+\ln T\bigr)
\le 2c^{2}\ln T \quad\text{for }T\ge e. \tag{R2}
\]
Plugging (R1)–(R2) into (18) gives
\[
\EE\big[f(\bar w_T)-f(w^*)\big]
\le \frac{\|w_0-w^*\|^{2}}{4c\sqrt{T}}
+\frac{(1+\beta)(\sigma^{2}+G^{2})c}{\sqrt{T}}\,
\frac{\ln T}{\sqrt{T}}
= \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right). \tag{R3}
\]
Thus the $\mathcal{O}(1/\sqrt{T})$ rate holds, with an explicit constant
\[
C_{\text{QSGD}}=
\frac{\|w_0-w^*\|^{2}}{4c}
+\;(1+\beta)(\sigma^{2}+G^{2})c\ln T .
\]

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}[label=\arabic*., leftmargin=*, itemsep=0pt]
\item \textbf{Exact Communication:} $\beta=0$ reduces the bound to the classical SGD result.
\item \textbf{Deterministic Gradients:} $\sigma^{2}=0$ (full‑batch) yields
\[
\EE\big[f(\bar w_T)-f(w^*)\big]\le
\frac{\|w_0-w^*\|^{2}}{2\sum_{t=0}^{T-1}\eta_t}
+\frac{L\beta G^{2}}{2}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t},
\]
showing that quantization alone introduces an additive $\mathcal{O}(\beta/\sqrt{T})$ term.
\item \textbf{Strong Convexity ($\mu$‑strongly convex):} Repeating the proof with the inequality
\[
\langle\nabla f(w_t),w_t-w^*\rangle\ge f(w_t)-f(w^*)+\frac{\mu}{2}\|w_t-w^*\|^{2},
\]
leads to
\[
\EE\big[f(w_T)-f(w^*)\big]\le
\mathcal{O}\!\Big(\frac{(1+\beta)(\sigma^{2}+G^{2})}{\mu T}\Big).
\]
\end{enumerate}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The derivation above demonstrates that, under the modest unbiased‑quantizer assumption (B1) and a bounded variance condition (B2), QSGD inherits the optimal $\mathcal{O}(1/\sqrt{T})$ convergence rate of stochastic gradient methods for convex smooth objectives. The additional factor $\beta$ quantifies the loss incurred by communication compression; practical quantizers (e.g., stochastic rounding, top‑$k$ sparsification) satisfy (B2) with modest $\beta$, thus preserving the efficiency of distributed learning while drastically reducing bandwidth.
