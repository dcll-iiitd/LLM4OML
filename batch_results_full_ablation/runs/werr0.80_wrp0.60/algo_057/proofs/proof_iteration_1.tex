\section*{Randomized Block Coordinate Descent (RBCD) for Non‑Convex Smooth Functions}

\section*{Mathematical Formulation}
Let $w\in\R^{d}$ be partitioned into $m$ (non‑empty) blocks
\[
w=\bigl(w^{(1)},w^{(2)},\dots ,w^{(m)}\bigr),\qquad 
w^{(i)}\in\R^{d_i},\;\; \sum_{i=1}^{m}d_i=d .
\]
For a differentiable (possibly non‑convex) objective $f:\R^{d}\to\R$ define the block gradient
\[
\nabla_i f(w)\;:=\;\frac{\partial f(w)}{\partial w^{(i)}}\in\R^{d_i}.
\]

At iteration $t\ge0$ a block index $i_t\in\{1,\dots,m\}$ is drawn independently
according to a fixed distribution $p=(p_1,\dots ,p_m)$ with $p_i>0$ and
$\sum_{i=1}^{m}p_i=1$.
Given a stepsize $\eta_i>0$ for each block, the update reads
\[
\boxed{
\begin{aligned}
w_{t+1}^{(i_t)} &= w_{t}^{(i_t)}-\eta_{i_t}\,\nabla_{i_t}f(w_t),\\
w_{t+1}^{(j)}   &= w_{t}^{(j)}\qquad\text{for }j\neq i_t .
\end{aligned}}
\tag{RBCD}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Block Coordinate Descent (RBCD)}
\begin{algorithmic}[1]
\STATE {\bf Input:} initial point $w_0$, stepsizes $\{\eta_i\}_{i=1}^{m}$, block probabilities $\{p_i\}_{i=1}^{m}$, maximal iterations $T$.
\FOR{$t=0,\dots ,T-1$}
    \STATE Sample $i_t\sim p$.
    \STATE Compute block gradient $g_t:=\nabla_{i_t}f(w_t)$.
    \STATE Update block $w_{t+1}^{(i_t)} = w_t^{(i_t)}-\eta_{i_t}g_t$.
    \STATE For $j\neq i_t$, set $w_{t+1}^{(j)}=w_t^{(j)}$.
\ENDFOR
\STATE {\bf Output:} $\{w_t\}_{t=0}^{T}$ (or the last iterate $w_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^2$.
Here there is a single block ($m=1$), $w\in\R$, $\nabla f(w)=2(w-3)$,
and we choose $\eta=0.1$, $w_0=0$, $T=3$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)$ & $f(w_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3766$
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the algorithm.

\section*{Assumptions}
\begin{assumption}[Block‑wise $L_i$‑smoothness]\label{ass:smooth}
For each block $i\in\{1,\dots,m\}$ there exists $L_i>0$ such that for all
$u,v\in\R^{d}$ differing only in block $i$,
\[
f(u)\le f(v)+\langle\nabla_i f(v),\,u^{(i)}-v^{(i)}\rangle
+\frac{L_i}{2}\,\norm{u^{(i)}-v^{(i)}}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower boundedness]\label{ass:lb}
The objective is bounded below: $f^{\inf}:=\inf_{w\in\R^{d}}f(w)>-\infty$.
\tag{A2}
\end{assumption}
\begin{assumption}[Stepsizes]\label{ass:steps}
For each block $i$ the stepsize satisfies
\[
0<\eta_i\le \frac{1}{L_i}.
\tag{A3}
\]
\end{assumption}
All subsequent results rely only on \ref{ass:smooth}–\ref{ass:steps}.

\section*{Main Convergence Theorem}
\begin{theorem}[Expected stationary‑point rate]\label{thm:main}
Let $\{w_t\}_{t\ge0}$ be generated by \eqref{RBCD} under
Assumptions \ref{ass:smooth}–\ref{ass:steps}.
Define for each block $i$
\[
\gamma_i:=\eta_i-\frac{L_i\eta_i^{2}}{2}>0 .
\]
Let $\gamma_{\min}:=\min_{i}\gamma_i$ and $p_{\min}:=\min_{i}p_i$.
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\;\le\;
\frac{f(w_0)-f^{\inf}}{\gamma_{\min}\,p_{\min}\,T}.
\tag{T1}
\]
Consequently $\displaystyle \min_{0\le t<T}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]=O\!\Bigl(\frac{1}{T}\Bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\eta_i\le 1/L_i$, each update is a descent step for the selected block (see Lemma~\ref{lem:descent}), guaranteeing that the sequence $\{f(w_t)\}$ is non‑increasing in expectation.
\item \textbf{Hyper‑parameter sensitivity.} The constant $\gamma_i$ is maximised at $\eta_i=1/L_i$, yielding $\gamma_i=1/(2L_i)$. Choosing smaller stepsizes only slows the rate (larger denominator in \eqref{T1}).
\item \textbf{Comparison to full‑gradient SGD.} For $m=1$ and $p_1=1$ the bound \eqref{T1} reduces to the classic $O(1/T)$ rate for deterministic gradient descent on smooth non‑convex functions. Random block selection incurs an extra factor $1/p_{\min}$, reflecting the expected number of iterations needed to touch the “hard” blocks.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity the randomized block coordinate method converges to a stationary point in expectation, at the same $O(1/T)$ order as full‑gradient descent. The proof hinges only on block‑wise smoothness and a stepsize not exceeding the reciprocal of the corresponding Lipschitz constant. Extensions (e.g., variance‑reduced stochastic gradients, adaptive stepsizes) follow the same template.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single block]\label{lem:descent}
Let $i\in\{1,\dots,m\}$ be any block and let $w^{+}=w-\eta_i e_i\nabla_i f(w)$ denote the point obtained after updating only block $i$ (here $e_i$ is the block‑selection operator). Under Assumption~\ref{ass:smooth} and \ref{ass:steps},
\[
f(w^{+})\le f(w)-\Bigl(\eta_i-\frac{L_i\eta_i^{2}}{2}\Bigr)\norm{\nabla_i f(w)}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Fix a block $i$ and consider $u=w^{+}$, $v=w$. Since $u$ and $v$ differ only in block $i$, applying \eqref{A1} yields
\[
f(w^{+})\le f(w)+\langle\nabla_i f(w),\,w^{+(i)}-w^{(i)}\rangle
+\frac{L_i}{2}\norm{w^{+(i)}-w^{(i)}}^{2}.
\tag{Step 1}
\]
Because $w^{+(i)}-w^{(i)}=-\eta_i\nabla_i f(w)$, we have
\[
\langle\nabla_i f(w),\,w^{+(i)}-w^{(i)}\rangle
=-\eta_i\norm{\nabla_i f(w)}^{2},
\qquad
\norm{w^{+(i)}-w^{(i)}}^{2}
=\eta_i^{2}\norm{\nabla_i f(w)}^{2}.
\tag{Step 2}
\]
Substituting Step~2 into Step~1 gives
\[
f(w^{+})\le f(w)-\eta_i\norm{\nabla_i f(w)}^{2}
+\frac{L_i}{2}\eta_i^{2}\norm{\nabla_i f(w)}^{2}
= f(w)-\Bigl(\eta_i-\frac{L_i\eta_i^{2}}{2}\Bigr)\norm{\nabla_i f(w)}^{2}.
\tag{Step 3}
\]
Since $\eta_i\le 1/L_i$, the factor in parentheses is non‑negative, establishing \eqref{L1}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t. the random block $i_t$ conditioned on the history up to time $t$ (denoted $\mathcal{F}_t$).

\noindent\textbf{Step 1 (One‑step expected descent).}
Condition on $\mathcal{F}_t$ and apply Lemma~\ref{lem:descent} with the randomly chosen block $i_t$:
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\Bigl(\eta_{i_t}-\frac{L_{i_t}\eta_{i_t}^{2}}{2}\Bigr)
   \norm{\nabla_{i_t}f(w_t)}^{2}.
\tag{Step 1}
\]

\noindent\textbf{Step 2 (Take full expectation).}
Taking expectation over $i_t$ (using $P(i_t=i)=p_i$) yields
\[
\E\!\bigl[f(w_{t+1})\bigr]
\le f(w_t)-\sum_{i=1}^{m}p_i
   \Bigl(\eta_i-\frac{L_i\eta_i^{2}}{2}\Bigr)
   \norm{\nabla_{i}f(w_t)}^{2}.
\tag{Step 2}
\]

\noindent\textbf{Step 3 (Introduce the constants $\gamma_i$).}
Define $\gamma_i:=\eta_i-\frac{L_i\eta_i^{2}}{2}>0$ (by Assumption~\ref{ass:steps}).
Then Step~2 becomes
\[
\E\!\bigl[f(w_{t+1})\bigr]\le
f(w_t)-\sum_{i=1}^{m}p_i\gamma_i\norm{\nabla_{i}f(w_t)}^{2}.
\tag{Step 3}
\]

\noindent\textbf{Step 4 (Relate block sum to full gradient norm).}
Since $\norm{\nabla f(w)}^{2}=\sum_{i=1}^{m}\norm{\nabla_i f(w)}^{2}$,
and $p_i\ge p_{\min}$, $\gamma_i\ge\gamma_{\min}$, we have
\[
\sum_{i=1}^{m}p_i\gamma_i\norm{\nabla_i f(w_t)}^{2}
\;\ge\;
p_{\min}\gamma_{\min}\sum_{i=1}^{m}\norm{\nabla_i f(w_t)}^{2}
=
p_{\min}\gamma_{\min}\norm{\nabla f(w_t)}^{2}.
\tag{Step 4}
\]

\noindent\textbf{Step 5 (Combine Steps 3 and 4).}
\[
\E\!\bigl[f(w_{t+1})\bigr]\le
f(w_t)-p_{\min}\gamma_{\min}\,\norm{\nabla f(w_t)}^{2}.
\tag{Step 5}
\]

\noindent\textbf{Step 6 (Take total expectation).}
Because $f(w_t)$ is deterministic given the past, the same inequality holds after taking the total expectation:
\[
\E\!\bigl[f(w_{t+1})\bigr]\le
\E\!\bigl[f(w_t)\bigr]-p_{\min}\gamma_{\min}\,
      \E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr].
\tag{Step 6}
\]

\noindent\textbf{Step 7 (Telescoping sum).}
Sum the inequality in Step~6 over $t=0,\dots,T-1$:
\[
\E\!\bigl[f(w_T)\bigr]\le
f(w_0)-p_{\min}\gamma_{\min}\sum_{t=0}^{T-1}
      \E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr].
\tag{Step 7}
\]

\noindent\textbf{Step 8 (Use lower boundedness).}
By Assumption~\ref{ass:lb}, $f(w_T)\ge f^{\inf}$, thus
\[
f^{\inf}\le f(w_0)-p_{\min}\gamma_{\min}
      \sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr].
\tag{Step 8}
\]

\noindent\textbf{Step 9 (Rearrange to obtain the average bound).}
Dividing by $p_{\min}\gamma_{\min}T$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{f(w_0)-f^{\inf}}{p_{\min}\gamma_{\min}\,T}.
\tag{Step 9}
\]

Step~9 is exactly the bound \eqref{T1} claimed in Theorem~\ref{thm:main}. $\square$

\section*{Rate Derivation (With Constants)}
Recall $\gamma_i=\eta_i-\tfrac{L_i\eta_i^{2}}{2}$. The maximal admissible value is attained at $\eta_i=1/L_i$,
\[
\gamma_i^{\star}
= \frac{1}{L_i}-\frac{L_i}{2}\Bigl(\frac{1}{L_i}\Bigr)^{2}
= \frac{1}{2L_i}.
\]
Hence a convenient uniform choice $\eta_i=1/L_i$ for all blocks yields
\[
\gamma_{\min}=\frac{1}{2L_{\max}},\qquad
L_{\max}:=\max_{i}L_i .
\]
Plugging into \eqref{T1}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2L_{\max}\bigl(f(w_0)-f^{\inf}\bigr)}{p_{\min}\,T}.
\tag{R1}
\]
Thus the expected squared gradient norm decays as $O(1/T)$ with an explicit constant.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} If $p_i=1/m$ for all $i$, then $p_{\min}=1/m$ and the bound becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2mL_{\max}\bigl(f(w_0)-f^{\inf}\bigr)}{T}.
\]
The factor $m$ reflects the fact that only one out of $m$ blocks is updated per iteration.

\item \textbf{Single block ($m=1$).} The algorithm reduces to classical gradient descent and $p_{\min}=1$, giving the standard $O(1/T)$ rate.

\item \textbf{Non‑uniform probabilities.} Choosing larger $p_i$ for blocks with larger Lipschitz constants $L_i$ can improve the constant $p_{\min}\gamma_{\min}$, a common practical heuristic.
\end{itemize}
