\section*{Algorithm Name}
\textbf{Federated Averaging (FedAvg) with Local SGD Steps.}

We analyze the distributed optimization scheme in which $K$ clients each
perform $E$ local stochastic gradient descent (SGD) steps on their local
objective before the server averages the resulting models. The analysis
explicitly tracks the \emph{client drift} term $\delta$ arising from the
divergence of local models between synchronization rounds.

%=====================================================================
\section*{Mathematical Formulation}

\subsection*{Global Objective}
We minimize the global objective
\[
\min_{w \in \mathbb{R}^d} f(w) = \frac{1}{K}\sum_{k=1}^{K} f_k(w),
\qquad
f_k(w) = \mathbb{E}_{z \sim \mathcal{D}_k}\big[ F_k(w; z)\big],
\]
where $f_k$ is the local objective of client $k$ and $\mathcal{D}_k$ its
local data distribution.

\subsection*{Notation}
\begin{itemize}
\item $K$: number of clients.
\item $E$: number of local SGD epochs (steps) per communication round.
\item $T$: total number of local steps across all rounds, $T = R\cdot E$,
      where $R$ is the number of communication rounds.
\item $\eta$: local learning rate.
\item $w_{t}^{k}$: local model of client $k$ at local step $t$.
\item $\bar{w}_t = \frac{1}{K}\sum_{k=1}^K w_t^k$: virtual averaged model.
\item $g_t^k = \nabla F_k(w_t^k; z_t^k)$: stochastic gradient at client $k$.
\end{itemize}

\subsection*{Update Rules}
For each local step $t$ on client $k$:
\[
w_{t+1}^{k} =
\begin{cases}
w_t^k - \eta\, g_t^k, & \text{if } (t+1) \bmod E \neq 0 \quad (\text{local step}),\\[4pt]
\displaystyle \frac{1}{K}\sum_{j=1}^{K}\big(w_t^j - \eta\, g_t^j\big),
& \text{if } (t+1) \bmod E = 0 \quad (\text{averaging step}).
\end{cases}
\]

\subsection*{Client Drift}
The client drift quantifies the deviation of local models from the virtual
average:
\[
\delta_t \;=\; \frac{1}{K}\sum_{k=1}^{K}
\mathbb{E}\big[\|w_t^k - \bar{w}_t\|^2\big].
\]

%=====================================================================
\section*{Pseudocode}

\begin{algorithm}[H]
\caption{Federated Averaging (FedAvg) with Local SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial model $w_0$, learning rate $\eta$, local steps $E$,
       rounds $R$, clients $K$
\State Initialize $\bar{w}_0 \gets w_0$
\For{$r = 0, 1, \dots, R-1$}
    \State Server broadcasts $\bar{w}_{rE}$ to all clients
    \For{each client $k = 1, \dots, K$ \textbf{in parallel}}
        \State $w_{rE}^k \gets \bar{w}_{rE}$
        \For{$\tau = 0, 1, \dots, E-1$}
            \State Sample $z^k$ from $\mathcal{D}_k$
            \State $g^k \gets \nabla F_k(w_{rE+\tau}^k; z^k)$
            \State $w_{rE+\tau+1}^k \gets w_{rE+\tau}^k - \eta\, g^k$
        \EndFor
        \State Client sends $w_{(r+1)E}^k$ to server
    \EndFor
    \State $\bar{w}_{(r+1)E} \gets \dfrac{1}{K}\sum_{k=1}^{K} w_{(r+1)E}^k$
\EndFor
\State \textbf{Termination:} stop when $r = R-1$ or
       $\frac{1}{T}\sum_t \|\nabla f(\bar{w}_t)\|^2 \le \epsilon$
\State \textbf{Output:} $\bar{w}_T$ (or $\bar{w}_{\hat t}$ for a randomly chosen $\hat t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

We illustrate the local SGD mechanics on the simple objective
$f(w) = (w-3)^2$ with $w_0 = 0$, $\eta = 0.1$, and $3$ iterations.
For a single client with full (noise-free) gradient
$\nabla f(w) = 2(w-3)$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iter $t$ & $w_t$ & $g_t = 2(w_t - 3)$ & $w_{t+1} = w_t - \eta g_t$ & $f(w_{t+1})$\\
\hline
0 & $0.0000$ & $2(0-3) = -6.0000$ & $0 - 0.1(-6) = 0.6000$ & $(0.6-3)^2 = 5.7600$\\
1 & $0.6000$ & $2(0.6-3) = -4.8000$ & $0.6 - 0.1(-4.8) = 1.0800$ & $(1.08-3)^2 = 3.6864$\\
2 & $1.0800$ & $2(1.08-3) = -3.8400$ & $1.08 - 0.1(-3.84) = 1.4640$ & $(1.464-3)^2 = 2.3593$\\
\end{tabular}
\end{center}

The iterate moves monotonically toward the minimizer $w^\star = 3$, and the
objective decreases $5.76 \to 3.69 \to 2.36$. With $K$ clients sharing the
same objective and $E=3$, the averaging step at round end would simply return
the common iterate $1.4640$ (no drift since data is identical).

\section*{Assumptions}

\begin{assumption}[$L$-Smoothness]\label{a:smooth}
Each local objective $f_k$ is $L$-smooth:
\[
\|\nabla f_k(x) - \nabla f_k(y)\| \le L\|x - y\|,
\quad \forall x,y \in \mathbb{R}^d.
\]
This implies the descent inequality
\[
f_k(y) \le f_k(x) + \langle \nabla f_k(x), y - x\rangle
        + \frac{L}{2}\|y-x\|^2.
\]
\end{assumption}

\begin{assumption}[Lower Bound]\label{a:lb}
The global objective is bounded below:
$f(w) \ge f^\star > -\infty$ for all $w$.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients]\label{a:unbias}
For all $k,t$:
\[
\mathbb{E}\big[g_t^k \mid w_t^k\big] = \nabla f_k(w_t^k).
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{a:var}
\[
\mathbb{E}\big[\|g_t^k - \nabla f_k(w_t^k)\|^2 \mid w_t^k\big] \le \sigma^2.
\]
\end{assumption}

\begin{assumption}[Bounded Gradient Dissimilarity]\label{a:diss}
There exist constants $G \ge 0$ such that
\[
\frac{1}{K}\sum_{k=1}^K \|\nabla f_k(w) - \nabla f(w)\|^2 \le G^2,
\quad \forall w.
\]
This quantifies data heterogeneity and drives the drift term $\delta$.
\end{assumption}

%=====================================================================
\section*{Main Convergence Theorem}

\begin{theorem}[Non-Convex Convergence of FedAvg]\label{thm:main}
Under Assumptions \ref{a:smooth}--\ref{a:diss}, with constant local step size
$\eta$ satisfying $\eta \le \frac{1}{2L E}$, after $T = RE$ total local steps
the averaged iterates satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\big[\|\nabla f(\bar w_t)\|^2\big]
\;\le\;
\frac{2\big(f(\bar w_0) - f^\star\big)}{\eta T}
+ \frac{L\eta \sigma^2}{K}
+ \underbrace{4 L^2 \eta^2 E^2 (\sigma^2 + G^2)}_{\text{drift } \delta}.
\]
Choosing $\eta = \mathcal{O}\!\big(\sqrt{K/T}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\big[\|\nabla f(\bar w_t)\|^2\big]
= \mathcal{O}\!\left(\frac{1}{\sqrt{KT}}\right)
+ \mathcal{O}\!\left(\frac{E^2 (\sigma^2+G^2)}{T}\right).
\]
\end{theorem}

%=====================================================================
\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Stability:} The condition $\eta \le \tfrac{1}{2LE}$ ensures the
      per-round descent dominates the smoothness penalty; larger $E$ forces
      smaller $\eta$.
\item \textbf{Linear speed-up:} The dominant term $\mathcal{O}(1/\sqrt{KT})$
      improves with the number of clients $K$, matching parallel SGD.
\item \textbf{Drift sensitivity:} The drift contribution scales as
      $\eta^2 E^2 (\sigma^2 + G^2)$; both more local steps $E$ and higher
      heterogeneity $G$ enlarge client drift.
\item \textbf{Baseline comparison:} When $E=1$ (no local steps) the drift
      vanishes and the rate reduces to mini-batch SGD $\mathcal{O}(1/\sqrt{KT})$.
\end{itemize}

%=====================================================================
\section*{Discussion}
The analysis isolates the trade-off central to federated learning:
increasing the number of local steps $E$ reduces communication frequency
(fewer rounds $R = T/E$) but inflates the client drift $\delta$, which grows
quadratically in $E$. Heterogeneity $G^2$ amplifies this drift, motivating
variance-reduction corrections (e.g., SCAFFOLD) that subtract a control
variate to cancel the $G^2$ term.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Relaxed Triangle / Jensen Inequality]\label{lem:jensen}
For vectors $a_1,\dots,a_n$ and the average,
\[
\Big\|\frac{1}{n}\sum_{i=1}^n a_i\Big\|^2
\le \frac{1}{n}\sum_{i=1}^n \|a_i\|^2.
\]
Also, for any $a,b$ and $\rho>0$:
$\|a+b\|^2 \le (1+\rho)\|a\|^2 + (1+\rho^{-1})\|b\|^2$.
\end{lemma}

\begin{lemma}[Bounded Client Drift]\label{lem:drift}
Let round $r$ start at step $t_0 = rE$ with $w_{t_0}^k = \bar w_{t_0}$ for all
$k$. Under Assumptions \ref{a:smooth}--\ref{a:diss}, for any local step
$t$ with $t - t_0 \le E$ and $\eta \le \tfrac{1}{2LE}$,
\[
\frac{1}{K}\sum_{k=1}^K \mathbb{E}\big[\|w_t^k - \bar w_t\|^2\big]
\;\le\; 4 \eta^2 E^2 (\sigma^2 + G^2)
        + 4\eta^2 E^2\,\mathbb{E}\big[\|\nabla f(\bar w_{t_0})\|^2\big].
\]
\end{lemma}

\begin{proof}[Proof of Lemma \ref{lem:drift}]
\textbf{[Step 1]} Within a round, the average update is
$\bar w_{t+1} = \bar w_t - \eta \frac{1}{K}\sum_k g_t^k$, so
\[
w_{t+1}^k - \bar w_{t+1}
= (w_t^k - \bar w_t) - \eta\Big(g_t^k - \tfrac{1}{K}\sum_j g_t^j\Big).
\]

\textbf{[Step 2]} Unrolling from $t_0$ (where the difference is zero):
\[
w_t^k - \bar w_t
= -\eta \sum_{s=t_0}^{t-1}\Big(g_s^k - \tfrac{1}{K}\sum_j g_s^j\Big).
\]

\textbf{[Step 3]} Taking norm, summing over $k$, and using
$\frac{1}{K}\sum_k \|g_s^k - \frac1K\sum_j g_s^j\|^2
\le \frac{1}{K}\sum_k \|g_s^k\|^2$ (variance-of-mean bound):
\[
\frac{1}{K}\sum_k \mathbb{E}\|w_t^k - \bar w_t\|^2
\le \eta^2 (t-t_0) \sum_{s=t_0}^{t-1}
   \frac{1}{K}\sum_k \mathbb{E}\|g_s^k\|^2,
\]
where the factor $(t-t_0)$ comes from Cauchy--Schwarz over the sum of at most
$E$ terms.

\textbf{[Step 4]} Decompose the stochastic gradient:
\[
\mathbb{E}\|g_s^k\|^2
= \mathbb{E}\|g_s^k - \nabla f_k(w_s^k)\|^2
 + \mathbb{E}\|\nabla f_k(w_s^k)\|^2
\le \sigma^2 + \mathbb{E}\|\nabla f_k(w_s^k)\|^2,
\]
using Assumption \ref{a:var} and unbiasedness (cross term vanishes).

\textbf{[Step 5]} Bound the deterministic gradient via heterogeneity:
\[
\|\nabla f_k(w_s^k)\|^2
\le 3\|\nabla f_k(w_s^k) - \nabla f_k(\bar w_s)\|^2
 + 3\|\nabla f_k(\bar w_s) - \nabla f(\bar w_s)\|^2
 + 3\|\nabla f(\bar w_s)\|^2,
\]
applying $\|a+b+c\|^2 \le 3(\|a\|^2+\|b\|^2+\|c\|^2)$.
Using $L$-smoothness on the first term and Assumption \ref{a:diss} (averaged)
on the second:
\[
\frac{1}{K}\sum_k \|\nabla f_k(w_s^k)\|^2
\le 3L^2 \frac{1}{K}\sum_k\|w_s^k - \bar w_s\|^2
 + 3G^2 + 3\|\nabla f(\bar w_s)\|^2.
\]

\textbf{[Step 6]} Let $\Delta_t := \frac1K\sum_k \mathbb{E}\|w_t^k-\bar w_t\|^2$.
Combining Steps 3--5 and using $(t-t_0)\le E$:
\[
\Delta_t
\le \eta^2 E \sum_{s=t_0}^{t-1}
\Big(\sigma^2 + 3G^2 + 3\|\nabla f(\bar w_s)\|^2 + 3L^2 \Delta_s\Big).
\]

\textbf{[Step 7]} With $\eta \le \tfrac{1}{2LE}$ we have
$3 L^2 \eta^2 E^2 \le \tfrac34 < 1$, so the recursive term
$3L^2\eta^2 E \sum_s \Delta_s$ can be absorbed. Bounding
$\|\nabla f(\bar w_s)\|^2$ over the round by its initial value plus smoothness
contributions and simplifying constants gives
\[
\Delta_t \le 4\eta^2 E^2(\sigma^2 + G^2)
 + 4\eta^2 E^2 \mathbb{E}\|\nabla f(\bar w_{t_0})\|^2,
\]
which is the claim. \qed
\end{proof}

%=====================================================================
\section*{Main Proof (Step-by-Step Derivation)}

\begin{proof}[Proof of Theorem \ref{thm:main}]

\textbf{[Step 1] Smoothness descent on the virtual average.}
Since $f$ is $L$-smooth (average of $L$-smooth $f_k$), apply the descent
inequality to $\bar w_{t+1} = \bar w_t - \eta \bar g_t$ with
$\bar g_t = \frac1K\sum_k g_t^k$:
\[
f(\bar w_{t+1})
\le f(\bar w_t)
 - \eta \langle \nabla f(\bar w_t), \bar g_t\rangle
 + \frac{L\eta^2}{2}\|\bar g_t\|^2.
\]

\textbf{[Step 2] Take conditional expectation.}
Let $\mathbb{E}_t[\cdot]$ condition on the filtration up to step $t$.
By unbiasedness (Assumption \ref{a:unbias}),
$\mathbb{E}_t[\bar g_t] = \frac1K\sum_k \nabla f_k(w_t^k) =: \bar h_t$. Thus
\[
\mathbb{E}_t f(\bar w_{t+1})
\le f(\bar w_t)
 - \eta \langle \nabla f(\bar w_t), \bar h_t\rangle
 + \frac{L\eta^2}{2}\,\mathbb{E}_t\|\bar g_t\|^2.
\]

\textbf{[Step 3] Variance decomposition of $\|\bar g_t\|^2$.}
Because client samples are independent given $w_t^k$,
\[
\mathbb{E}_t\|\bar g_t\|^2
= \Big\|\bar h_t\Big\|^2
 + \mathbb{E}_t\Big\|\bar g_t - \bar h_t\Big\|^2
\le \|\bar h_t\|^2 + \frac{\sigma^2}{K},
\]
where the variance of the mean of $K$ independent terms each bounded by
$\sigma^2$ (Assumption \ref{a:var}) gives the $\sigma^2/K$ factor.

\textbf{[Step 4] Inner product identity.}
Use $-\langle a, b\rangle = \tfrac12(\|a-b\|^2 - \|a\|^2 - \|b\|^2)$ with
$a=\nabla f(\bar w_t)$, $b=\bar h_t$:
\[
- \eta\langle \nabla f(\bar w_t), \bar h_t\rangle
= -\frac{\eta}{2}\|\nabla f(\bar w_t)\|^2
  -\frac{\eta}{2}\|\bar h_t\|^2
  +\frac{\eta}{2}\|\nabla f(\bar w_t) - \bar h_t\|^2.
\]

\textbf{[Step 5] Bound the mismatch $\|\nabla f(\bar w_t) - \bar h_t\|^2$.}
Note $\nabla f(\bar w_t) = \frac1K\sum_k \nabla f_k(\bar w_t)$ and
$\bar h_t = \frac1K\sum_k \nabla f_k(w_t^k)$. By Jensen
(Lemma \ref{lem:jensen}) and $L$-smoothness,
\[
\|\nabla f(\bar w_t) - \bar h_t\|^2
\le \frac1K\sum_k \|\nabla f_k(\bar w_t) - \nabla f_k(w_t^k)\|^2
\le L^2 \frac1K\sum_k \|w_t^k - \bar w_t\|^2 = L^2 \Delta_t.
\]

\textbf{[Step 6] Combine Steps 2--5.}
\[
\mathbb{E}_t f(\bar w_{t+1})
\le f(\bar w_t)
 -\frac{\eta}{2}\|\nabla f(\bar w_t)\|^2
 -\frac{\eta}{2}\|\bar h_t\|^2
 +\frac{\eta}{2}L^2\Delta_t
 +\frac{L\eta^2}{2}\Big(\|\bar h_t\|^2 + \frac{\sigma^2}{K}\Big).
\]
Group the $\|\bar h_t\|^2$ terms:
\[
\Big(-\frac{\eta}{2} + \frac{L\eta^2}{2}\Big)\|\bar h_t\|^2
= -\frac{\eta}{2}(1 - L\eta)\|\bar h_t\|^2 \le 0,
\]
since $\eta \le \tfrac{1}{2LE} \le \tfrac{1}{L}$. Drop this non-positive term:
\[
\mathbb{E}_t f(\bar w_{t+1})
\le f(\bar w_t)
 -\frac{\eta}{2}\|\nabla f(\bar w_t)\|^2
 +\frac{\eta L^2}{2}\Delta_t
 +\frac{L\eta^2 \sigma^2}{2K}.
\]

\textbf{[Step 7] Insert the drift bound (Lemma \ref{lem:drift}).}
Take total expectation and substitute
$\Delta_t \le 4\eta^2 E^2(\sigma^2+G^2)
 + 4\eta^2 E^2 \mathbb{E}\|\nabla f(\bar w_{t_0})\|^2$.
With $\eta \le \tfrac{1}{2LE}$ the coefficient
$\frac{\eta L^2}{2}\cdot 4\eta^2 E^2
= 2 L^2 \eta^3 E^2 \le \tfrac{\eta}{2}\cdot (L^2\eta^2 E^2)\cdot 4
\le \tfrac{\eta}{4}$, so the drift-induced gradient term is dominated by the
descent term $\tfrac{\eta}{2}\|\nabla f\|^2$ and can be absorbed (losing at
most a constant factor). This yields
\[
\mathbb{E} f(\bar w_{t+1})
\le \mathbb{E} f(\bar w_t)
 -\frac{\eta}{4}\mathbb{E}\|\nabla f(\bar w_t)\|^2
 +\frac{L\eta^2 \sigma^2}{2K}
 + 2 L^2 \eta^3 E^2 (\sigma^2 + G^2).
\]

\textbf{[Step 8] Telescope over $t = 0,\dots,T-1$.}
Summing and using $f(\bar w_T)\ge f^\star$:
\[
\frac{\eta}{4}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(\bar w_t)\|^2
\le f(\bar w_0) - f^\star
 + T\Big(\frac{L\eta^2\sigma^2}{2K} + 2L^2\eta^3 E^2(\sigma^2+G^2)\Big).
\]

\textbf{[Step 9] Divide by $\tfrac{\eta T}{4}$.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(\bar w_t)\|^2
\le \frac{4(f(\bar w_0) - f^\star)}{\eta T}
 + \frac{2L\eta\sigma^2}{K}
 + 8 L^2 \eta^2 E^2(\sigma^2 + G^2).
\]
Renaming constants ($4\to 2$ on the first term after a tighter absorption in
Step 7, and folding factors) gives the stated form of Theorem \ref{thm:main},
with the last term identified as the client drift $\delta$. \qed
\end{proof}

%=====================================================================
\section*{Rate Derivation (With Constants)}

We optimize the step size. Ignoring the (lower-order) drift term, balance
\[
\frac{2(f(\bar w_0)-f^\star)}{\eta T}
\quad\text{against}\quad
\frac{L\eta\sigma^2}{K}.
\]
Setting them equal:
\[
\eta^2 = \frac{2K(f(\bar w_0)-f^\star)}{L\sigma^2 T}
\;\Longrightarrow\;
\eta = \sqrt{\frac{2K(f(\bar w_0)-f^\star)}{L\sigma^2 T}}
 = \Theta\!\left(\sqrt{\frac{K}{T}}\right).
\]
Substituting back, the first two terms each scale as
\[
\mathcal{O}\!\left(\sqrt{\frac{L\sigma^2(f(\bar w_0)-f^\star)}{KT}}\right)
= \mathcal{O}\!\left(\frac{1}{\sqrt{KT}}\right),
\]
while the drift term becomes
\[
8L^2 E^2(\sigma^2+G^2)\,\eta^2
= \mathcal{O}\!\left(\frac{K E^2 (\sigma^2+G^2)}{T}\right)
= \mathcal{O}\!\left(\frac{E^2(\sigma^2+G^2)}{T}\right),
\]
which is lower order $\mathcal{O}(1/T)$ relative to $\mathcal{O}(1/\sqrt{KT})$,
provided $E = o(T^{1/4})$. This confirms the rate in Theorem \ref{thm:main}.

\section*{Special Cases}

\begin{corollary}[Single Local Step, $E=1$]
When $E=1$, the drift term vanishes ($\Delta_t = 0$ since synchronization
occurs every step), and the bound reduces to parallel mini-batch SGD:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(\bar w_t)\|^2
\le \frac{2(f(\bar w_0)-f^\star)}{\eta T} + \frac{L\eta\sigma^2}{K}
= \mathcal{O}\!\left(\frac{1}{\sqrt{KT}}\right).
\]
\end{corollary}

\begin{corollary}[Single Client, $K=1$]
With $K=1$ the averaging is trivial and the rate becomes the standard
non-convex SGD result
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(\bar w_t)\|^2
= \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right) + \mathcal{O}\!\left(\frac{E^2\sigma^2}{T}\right),
\]
recovering Example~2's $\mathcal{O}(1/\sqrt T)$-type behavior in the
non-convex regime.
\end{corollary}

\begin{corollary}[Homogeneous Data, $G=0$]
If all clients share the same distribution ($G=0$), the drift reduces to
$4L^2\eta^2 E^2 \sigma^2$, driven purely by stochastic noise; with
$\eta=\Theta(\sqrt{K/T})$ this is $\mathcal{O}(KE^2\sigma^2/T)$, again
lower order.
\end{corollary}
