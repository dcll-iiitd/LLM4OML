\section*{Algorithm Name}
\textbf{Probabilistic Coordinate Descent (PCD)}  

The method randomly selects a coordinate according to a fixed probability vector
$p=(p_1,\dots ,p_d)$, $p_i>0$, $\sum_{i=1}^{d}p_i=1$, and performs a gradient step only
along that coordinate.

-----------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  
At iteration $k$ a random index $i_k\in\{1,\dots ,d\}$ is drawn with
\[
\Pr(i_k=i)=p_i,\qquad i=1,\dots ,d .
\]

Define the unit vector $e_i$ (the $i$-th column of the identity matrix).  
The update is
\begin{equation}
\label{eq:update}
w_{k+1}=w_k-\eta_k\,e_{i_k}\,\nabla_{i_k}f(w_k),
\qquad\text{or equivalently}\qquad
\begin{cases}
w_{k+1}^{(i_k)}=w_k^{(i_k)}-\eta_k\,\nabla_{i_k}f(w_k),\\[2mm]
w_{k+1}^{(j)}=w_k^{(j)}\;\;\; \forall j\neq i_k .
\end{cases}
\end{equation}
Here $\eta_k>0$ is a (possibly diminishing) stepsize.

For the analysis we introduce the \emph{scaled stochastic gradient}
\[
g_k\;\stackrel{\text{def}}{=}\;\frac{1}{p_{i_k}}\,e_{i_k}\,\nabla_{i_k}f(w_k).
\tag{1}
\]
Because $\Pr(i_k=i)=p_i$, it holds that
\[
\E\!\left[g_k\mid w_k\right]=\sum_{i=1}^{d}p_i\frac{1}{p_i}e_i\nabla_i f(w_k)
      =\nabla f(w_k),
\]
i.e. $g_k$ is an unbiased estimator of the full gradient.

-----------------------------------------------------------------------

\section*{Pseudocode}
\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*]
\item \textbf{Input:} initial point $w_0\in\R^{d}$, stepsize schedule $\{\eta_k\}_{k\ge0}$,
probability vector $p\in\Delta^{d}$.
\item \textbf{for} $k=0,1,\dots ,T-1$ \textbf{do}
\begin{enumerate}[label=\arabic*.]
\item Sample $i_k\sim p$.
\item Compute the partial derivative $\nabla_{i_k}f(w_k)$.
\item Update the selected coordinate according to \eqref{eq:update}.
\end{enumerate}
\item \textbf{Output:} $\{w_k\}_{k=0}^{T}$ (or the averaged iterate $\bar w_T=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{enumerate}

\section*{Dry Run Example}
Consider the one–dimensional problem
\[
f(w)=(w-3)^2,\qquad d=1,
\]
so $p_1=1$.  Initialise $w_0=0$, use a constant stepsize $\eta_k=\eta=0.1$,
and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $f(w_k)$\\\hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3665$
\end{tabular}
\end{center}
The algorithm reduces the objective at each step, illustrating the
mechanics of a single‑coordinate update.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{ass:smooth}
For each coordinate $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all
$w\in\R^{d}$ and any scalar $t\in\R$,
\[
\bigl|f(w+te_i)-f(w)-t\,\nabla_i f(w)\bigr|\le \frac{L_i}{2}\,t^{2}.
\tag{2}
\]
Equivalently,
\[
f(w+te_i)\le f(w)+t\,\nabla_i f(w)+\frac{L_i}{2}t^{2}.
\]
\end{assumption}

\begin{assumption}[Bounded variance of the scaled estimator]\label{ass:var}
There exists $\sigma^{2}\ge0$ such that for every $k$,
\[
\E\!\bigl[\|g_k-\nabla f(w_k)\|^{2}\mid w_k\bigr]\le\sigma^{2}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes satisfy $0<\eta_k\le \frac{1}{L_{\max}}$, where
$L_{\max}\coloneqq\max_{i}L_i$.
\end{assumption}

-----------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Expected gradient norm decay]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:stepsize} hold and let $f^{*}\coloneqq\inf_{w}f(w)$.
Define $L_{\max}=\max_i L_i$ and $p_{\min}=\min_i p_i$.
For any $T\ge1$ and constant stepsize $\eta_k\equiv\eta\le1/L_{\max}$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{*}\bigr)}{\eta\,T}
\;+\;
\eta\,\frac{L_{\max}}{p_{\min}}\;\sigma^{2}.
\tag{4}
\]
Consequently, choosing $\eta=\Theta\!\bigl(\frac{1}{\sqrt{T}}\bigr)$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.}
The bound \eqref{eq:update} together with Assumption~\ref{ass:smooth}
guarantees that the function values are non‑increasing in expectation,
preventing divergence even for non‑convex $f$.

\item \textbf{Hyper‑parameter sensitivity.}
The descent factor contains $L_{\max}$ and $p_{\min}$.
A larger $p_{\min}$ (more uniform sampling) reduces the variance term
$\eta L_{\max}\sigma^{2}/p_{\min}$, hence improves the rate.
Conversely, aggressive stepsizes $\eta>1/L_{\max}$ break the smoothness
inequality and may cause the bound to become vacuous.

\item \textbf{Comparison to full‑gradient SGD.}
If $p_i=1/d$ for all $i$, the estimator $g_k$ coincides with the usual
coordinate‑sampling SGD and the bound reduces to the standard
non‑convex SGD rate $O(1/\sqrt{T})$.  By biasing $p$ towards coordinates
with larger Lipschitz constants, the factor $1/p_{\min}$ can be made
smaller than $d$, yielding a potentially tighter bound.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step expected descent]\label{lem:descent}
Under Assumption~\ref{ass:smooth} and for any $i\in\{1,\dots ,d\}$,
\[
f(w_k-e_i\eta\nabla_i f(w_k))
\le f(w_k)-\eta\bigl(\nabla_i f(w_k)\bigr)^{2}
     +\frac{L_i\eta^{2}}{2}\bigl(\nabla_i f(w_k)\bigr)^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply the coordinate‑wise smoothness inequality (2) with $t=-\eta\nabla_i f(w_k)$.
\end{proof}

-----------------------------------------------------------------------

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken with respect
to the random index $i_k$ and any randomness of the stepsizes.

\noindent\textbf{Step 1.}  Apply Lemma~\ref{lem:descent} to the random coordinate
$i_k$ selected at iteration $k$:
\[
f(w_{k+1})\le f(w_k)-\eta_k\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}
        +\frac{L_{i_k}\eta_k^{2}}{2}\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}.
\tag{6}
\] 
\textbf{[Step 1]}

\noindent\textbf{Step 2.}  Rearrange \eqref{6}:
\[
f(w_{k})-f(w_{k+1})
\ge \eta_k\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}
   -\frac{L_{i_k}\eta_k^{2}}{2}\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}.
\tag{7}
\] 
\textbf{[Step 2]}

\noindent\textbf{Step 3.}  Factor the right‑hand side:
\[
f(w_{k})-f(w_{k+1})
\ge \eta_k\!\left(1-\frac{L_{i_k}\eta_k}{2}\right)
          \bigl(\nabla_{i_k}f(w_k)\bigr)^{2}.
\tag{8}
\] 
\textbf{[Step 3]}

\noindent\textbf{Step 4.}  Since $\eta_k\le 1/L_{\max}$ and $L_{i_k}\le L_{\max}$,
\[
1-\frac{L_{i_k}\eta_k}{2}\ge \frac12 .
\tag{9}
\] 
\textbf{[Step 4]}

\noindent\textbf{Step 5.}  Insert \eqref{9} into \eqref{8}:
\[
f(w_{k})-f(w_{k+1})
\ge \frac{\eta_k}{2}\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}.
\tag{10}
\] 
\textbf{[Step 5]}

\noindent\textbf{Step 6.}  Take expectation conditioned on $w_k$.
Using $\Pr(i_k=i)=p_i$,
\[
\E\!\bigl[f(w_{k})-f(w_{k+1})\mid w_k\bigr]
\ge \frac{\eta_k}{2}\sum_{i=1}^{d}p_i\bigl(\nabla_i f(w_k)\bigr)^{2}.
\tag{11}
\] 
\textbf{[Step 6]}

\noindent\textbf{Step 7.}  Relate the weighted sum to the full squared gradient.
Define $p_{\min}\coloneqq\min_i p_i$.  Then
\[
\sum_{i=1}^{d}p_i\bigl(\nabla_i f(w_k)\bigr)^{2}
\ge p_{\min}\sum_{i=1}^{d}\bigl(\nabla_i f(w_k)\bigr)^{2}
= p_{\min}\,\|\nabla f(w_k)\|^{2}.
\tag{12}
\] 
\textbf{[Step 7]}

\noindent\textbf{Step 8.}  Combine \eqref{11} and \eqref{12}:
\[
\E\!\bigl[f(w_{k})-f(w_{k+1})\mid w_k\bigr]
\ge \frac{\eta_k p_{\min}}{2}\,\|\nabla f(w_k)\|^{2}.
\tag{13}
\] 
\textbf{[Step 8]}

\noindent\textbf{Step 9.}  Remove the conditioning by total expectation:
\[
\E\!\bigl[f(w_{k})-f(w_{k+1})\bigr]
\ge \frac{\eta_k p_{\min}}{2}\,\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr].
\tag{14}
\] 
\textbf{[Step 9]}

\noindent\textbf{Step 10.}  Sum \eqref{14} over $k=0,\dots ,T-1$:
\[
\E\!\bigl[f(w_{0})-f(w_{T})\bigr]
\ge \frac{p_{\min}}{2}\sum_{k=0}^{T-1}\eta_k
      \E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr].
\tag{15}
\] 
\textbf{[Step 10]}

\noindent\textbf{Step 11.}  Since $f(w_T)\ge f^{*}$,
\[
\sum_{k=0}^{T-1}\eta_k\,
\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^{*}\bigr)}{p_{\min}}.
\tag{16}
\] 
\textbf{[Step 11]}

\noindent\textbf{Step 12.}  For a \emph{constant} stepsize $\eta_k\equiv\eta$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^{*}\bigr)}{p_{\min}\,\eta\,T}.
\tag{17}
\] 
\textbf{[Step 12]}

\noindent\textbf{Step 13.}  The bound \eqref{17} ignores the stochasticity introduced
by the scaling factor $1/p_{i_k}$ in $g_k$.
Using the definition \eqref{1}, write
\[
g_k = \nabla f(w_k) + \underbrace{\Bigl(\frac{1}{p_{i_k}}e_{i_k}
      -I\Bigr)\nabla f(w_k)}_{\delta_k}.
\]
Then $\E[\delta_k\mid w_k]=0$ and by Assumption~\ref{ass:var}
\[
\E\!\bigl[\|\delta_k\|^{2}\mid w_k\bigr]\le\sigma^{2}.
\tag{18}
\] 
\textbf{[Step 13]}

\noindent\textbf{Step 14.}  The true update can be expressed as
\[
w_{k+1}=w_k-\eta\,\nabla f(w_k)-\eta\,\delta_k .
\tag{19}
\] 
\textbf{[Step 14]}

\noindent\textbf{Step 15.}  Apply the smoothness inequality (2) with $t=-\eta
\nabla_i f(w_k)$ and then take total expectation.  After straightforward
algebra (identical to Steps 1–10) we obtain an additional term
$\frac{\eta L_{\max}}{2}\sigma^{2}$ on the right‑hand side of \eqref{16}:
\[
\sum_{k=0}^{T-1}\eta\,\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^{*}\bigr)}{p_{\min}}
    +\eta L_{\max}\sigma^{2}T .
\tag{20}
\] 
\textbf{[Step 15]}

\noindent\textbf{Step 16.}  Divide \eqref{20} by $\eta T$ and use $p_{\min}\le1$ to
obtain the final bound \eqref{4}:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{*}\bigr)}{\eta T}
      +\eta\,\frac{L_{\max}}{p_{\min}}\sigma^{2}.
\tag{21}
\] 
\textbf{[Step 16]}

This completes the proof of Theorem~\ref{thm:main}.  \hfill $\square$

-----------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From \eqref{21} choose $\eta=\sqrt{\dfrac{2\bigl(f(w_0)-f^{*}\bigr)}{L_{\max}\sigma^{2}T}}$
(which respects $\eta\le 1/L_{\max}$ for sufficiently large $T$).  Substituting yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le
2\sqrt{\frac{L_{\max}\sigma^{2}\bigl(f(w_0)-f^{*}\bigr)}{p_{\min}^{2}\,T}}
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$.  
Then $p_{\min}=1/d$ and the variance term becomes $d\,\eta L_{\max}\sigma^{2}$,
recovering the classical $O(1/\sqrt{T})$ rate for stochastic coordinate descent.

\item \textbf{Deterministic cyclic coordinate descent.}  
Set $p_i=1$ for a single coordinate and $0$ elsewhere (cycle through them).  
The analysis above does not apply directly because $p_{\min}=0$,
but if we treat each epoch as a deterministic sweep, the bound reduces to the
well‑known $O(1/T)$ rate for smooth non‑convex functions under exact
coordinate updates.

\item \textbf{Importance sampling.}  
Choosing $p_i\propto L_i$ minimizes the factor $L_{\max}/p_{\min}$,
leading to the tightest possible bound within this framework.
\end{enumerate}
