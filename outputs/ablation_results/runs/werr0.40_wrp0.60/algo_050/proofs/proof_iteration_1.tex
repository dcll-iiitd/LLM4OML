\section*{Algorithm Name}
\textbf{Trust‑Region Gradient Method (TR‑GM)}  

The method solves at each iteration a quadratic model of a convex smooth objective inside a Euclidean ball of radius $\Delta_k$ and accepts the trial point only if sufficient decrease is observed.

\section*{Mathematical Formulation}
Let $f:\R^n\to\R$ be convex and differentiable with $L$‑Lipschitz gradient.  
Given the current iterate $x_k$, define the (first‑order) model
\[
m_k(p)=f(x_k)+\langle \nabla f(x_k),p\rangle+\frac{M}{2}\|p\|^2,
\qquad p\in\R^n,
\]
where $M\ge 0$ is a curvature parameter (often taken $M=L$).  
The trust‑region subproblem is
\begin{equation}
\label{eq:subprob}
\min_{p\in\R^n}\; m_k(p)\quad\text{s.t.}\quad \|p\| \le \Delta_k .
\end{equation}
Let $p_k$ be an \emph{approximate} solution of \eqref{eq:subprob} satisfying the Cauchy‑point condition
\begin{equation}
\label{eq:cauchy}
m_k(p_k)-m_k(0)\le \kappa_{\rm c}\,\bigl(m_k(p^{\rm C}_k)-m_k(0)\bigr),
\qquad 0<\kappa_{\rm c}\le 1,
\end{equation}
where $p^{\rm C}_k=-\frac{\Delta_k}{\|\nabla f(x_k)\|}\nabla f(x_k)$ is the Cauchy point.

Define the ratio of actual to predicted reduction
\[
\rho_k = \frac{f(x_k)-f(x_k+p_k)}{m_k(0)-m_k(p_k)} .
\]

The update rule is
\[
x_{k+1}= 
\begin{cases}
x_k+p_k, & \rho_k\ge \eta_1,\\[4pt]
x_k,    & \rho_k<\eta_1,
\end{cases}
\]
with $0<\eta_1<\eta_2<1$ (commonly $\eta_1=0.1,\;\eta_2=0.75$).  
The trust‑region radius is adapted:
\[
\Delta_{k+1}= 
\begin{cases}
\min\{\gamma_{\rm inc}\Delta_k,\;\Delta_{\max}\}, & \rho_k\ge \eta_2,\\[4pt]
\gamma_{\rm dec}\Delta_k, & \rho_k<\eta_1,\\[4pt]
\Delta_k, & \text{otherwise},
\end{cases}
\]
with $0<\gamma_{\rm dec}<1<\gamma_{\rm inc}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust‑Region Gradient Method (TR‑GM)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0\in\R^n$, initial radius $\Delta_0>0$, parameters
$\eta_1,\eta_2,\gamma_{\rm inc},\gamma_{\rm dec},\Delta_{\max}>0$, curvature $M\ge0$, Cauchy tolerance $\kappa_{\rm c}\in(0,1]$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\nabla f(x_k)$.
    \STATE Solve (approximately) subproblem \eqref{eq:subprob} to obtain $p_k$ satisfying \eqref{eq:cauchy}.
    \STATE Evaluate actual reduction $a_k=f(x_k)-f(x_k+p_k)$.
    \STATE Evaluate predicted reduction $p_k^{\rm pred}=m_k(0)-m_k(p_k)$.
    \STATE Compute ratio $\rho_k = a_k/p_k^{\rm pred}$.
    \IF{$\rho_k\ge \eta_1$}
        \STATE $x_{k+1}=x_k+p_k$.
    \ELSE
        \STATE $x_{k+1}=x_k$.
    \ENDIF
    \IF{$\rho_k\ge \eta_2$}
        \STATE $\Delta_{k+1}= \min\{\gamma_{\rm inc}\Delta_k,\;\Delta_{\max}\}$.
    \ELSIF{$\rho_k<\eta_1$}
        \STATE $\Delta_{k+1}= \gamma_{\rm dec}\Delta_k$.
    \ELSE
        \STATE $\Delta_{k+1}= \Delta_k$.
    \ENDIF
    \IF{termination criterion satisfied (e.g., $\|g_k\|\le\varepsilon$)} \textbf{break}
    \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate convex smooth function
\[
f(w)=(w-3)^2,\qquad f'(w)=2(w-3).
\]
Set $w_0=0$, initial radius $\Delta_0=1$, $M=L=2$, $\eta_1=0.1$, $\eta_2=0.75$, $\gamma_{\rm inc}=2$, $\gamma_{\rm dec}=0.5$, $\Delta_{\max}=4$, $\kappa_{\rm c}=1$ (exact Cauchy point).

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item \textbf{Iteration 0}\\
Gradient $g_0=2(0-3)=-6$.\\
Cauchy point $p^{\rm C}_0= -\frac{\Delta_0}{\|g_0\|}g_0 = -\frac{1}{6}(-6)=1$.\\
Since $m_0(p)$ is quadratic, the exact solution of \eqref{eq:subprob} is $p_0=p^{\rm C}_0=1$ (it lies on the boundary).\\
Predicted reduction:
\[
m_0(0)-m_0(p_0)=\langle g_0,p_0\rangle+\frac{M}{2}\|p_0\|^2
 =(-6)(1)+\frac{2}{2}(1)^2 =-6+1=-5.
\]
Actual reduction:
\[
f(w_0)-f(w_0+p_0)=f(0)-f(1)=9-4=5.
\]
Ratio $\rho_0 = 5/5 =1\ge\eta_1$. Accept step: $w_1=0+1=1$.  
Since $\rho_0\ge\eta_2$, increase radius: $\Delta_1=\min\{2\cdot1,4\}=2$.

\item \textbf{Iteration 1}\\
Gradient $g_1=2(1-3)=-4$.\\
Cauchy point $p^{\rm C}_1=-\frac{\Delta_1}{\|g_1\|}g_1=-\frac{2}{4}(-4)=2$.  
But $\|p^{\rm C}_1\|=2=\Delta_1$, so $p_1=2$.\\
Predicted reduction:
\[
\langle g_1,p_1\rangle+\frac{M}{2}\|p_1\|^2 = (-4)(2)+1\cdot 4 = -8+4 = -4.
\]
Actual reduction:
\[
f(1)-f(3)=4-0=4.
\]
$\rho_1=4/4=1\ge\eta_1$; accept: $w_2=1+2=3$.  
$\rho_1\ge\eta_2$ $\Rightarrow$ $\Delta_2=\min\{2\cdot2,4\}=4$.

\item \textbf{Iteration 2}\\
Gradient $g_2=2(3-3)=0$.  
Cauchy point $p^{\rm C}_2=0$, so $p_2=0$.  
Predicted and actual reductions are $0$, $\rho_2$ is undefined but the algorithm terminates because $\|g_2\|=0\le\varepsilon$.
\end{enumerate}
The method reaches the optimum $w^\star=3$ in two successful steps.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]
\label{ass:convex}
The objective $f:\R^n\to\R$ is convex and differentiable with $L$‑Lipschitz gradient, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad \forall x,y\in\R^n .
\]
\end{assumption}
\begin{assumption}[Bounded Level Set]
\label{ass:bounded}
For the initial point $x_0$, the sublevel set 
\[
\mathcal{L}_0:=\{x\in\R^n\mid f(x)\le f(x_0)\}
\]
is bounded and has diameter $D<\infty$.
\end{assumption}
\begin{assumption}[Model Parameter]
\label{ass:M}
The curvature parameter satisfies $0\le M\le L$.
\end{assumption}
\begin{assumption}[Cauchy‑Point Approximation]
\label{ass:cauchy}
At each iteration, the trial step $p_k$ fulfills \eqref{eq:cauchy} with a fixed $\kappa_{\rm c}\in(0,1]$.
\end{assumption}
\begin{assumption}[Trust‑Region Parameters]
\label{ass:trparams}
$0<\eta_1<\eta_2<1$, $0<\gamma_{\rm dec}<1<\gamma_{\rm inc}$, and $0<\Delta_{\max}<\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Global Sublinear Convergence]
\label{thm:conv}
Assume \ref{ass:convex}–\ref{ass:trparams} hold. Let $\{x_k\}$ be the iterates generated by TR‑GM and let $x^\star$ be a minimizer of $f$. Then for all $K\ge 1$,
\[
f(x_K)-f(x^\star)\;\le\;
\frac{L D^2}{2\,\kappa_{\rm c}\,(1-\eta_1)}\;\frac{1}{K}.
\]
Consequently,
\[
\min_{0\le k\le K-1}\|\nabla f(x_k)\|^2\;\le\;
\frac{2L^2 D^2}{\kappa_{\rm c}\,(1-\eta_1)}\;\frac{1}{K}.
\]
\end{theorem}
The theorem shows a $O(1/K)$ decrease of the optimality gap, identical to the rate of the (projected) gradient method for convex smooth problems.

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The radius $\Delta_k$ never grows beyond $\Delta_{\max}$ and never collapses to zero because a successful iteration increases it by at least factor $\gamma_{\rm inc}>1$, while a failed iteration reduces it by at most factor $\gamma_{\rm dec}<1$. Hence the algorithm stays within a compact set (by Assumption \ref{ass:bounded}).
\item \textbf{Hyperparameter Sensitivity.} The constants in the bound depend linearly on $1/(1-\eta_1)$ and inversely on $\kappa_{\rm c}$. Choosing a larger $\eta_1$ (more demanding acceptance) slows the bound but typically yields larger steps in practice. The Cauchy tolerance $\kappa_{\rm c}$ directly scales the guaranteed reduction per iteration.
\item \textbf{Comparison to Baselines.} For convex $L$‑smooth problems, standard gradient descent with step size $1/L$ achieves the same $O(1/K)$ rate. TR‑GM, however, adapts the step length via $\Delta_k$, which can be advantageous when $L$ is unknown or varies locally.
\end{itemize}

\section*{Discussion}
The proof hinges on the fact that the Cauchy point provides a guaranteed fraction of the steepest‑descent reduction inside the ball, and that the trust‑region ratio $\rho_k$ enforces that the actual reduction is comparable to the model prediction. Under convexity, the model is a global over‑estimator, which simplifies the analysis compared to the non‑convex case.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Cauchy reduction]
For a given $x_k$, the Cauchy point $p^{\rm C}_k$ satisfies
\[
m_k(p^{\rm C}_k)-m_k(0)= -\frac{\|g_k\|^2}{M}\,\phi\!\Bigl(\frac{M\Delta_k}{\|g_k\|}\Bigr),
\]
where $\phi(t)=t-\frac{t^2}{2}$ for $t\le 2$ and $\phi(t)=1$ for $t\ge2$.
\end{definition}
\begin{lemma}[Model overestimation]
\label{lem:over}
If $f$ is convex and $L$‑smooth with $M\le L$, then for any $p$,
\[
f(x_k+p)-f(x_k)\;\le\; \langle g_k,p\rangle+\frac{L}{2}\|p\|^2
\;\le\; \langle g_k,p\rangle+\frac{M}{2}\|p\|^2 = m_k(p)-m_k(0).
\]
\end{lemma}
\begin{proof}
Convexity gives $f(x_k+p)\ge f(x_k)+\langle g_k,p\rangle$.  
$L$‑smoothness yields the descent lemma
\[
f(x_k+p)\le f(x_k)+\langle g_k,p\rangle+\frac{L}{2}\|p\|^2 .
\]
Since $M\le L$, the second inequality follows. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}.  

Let $x^\star$ be any minimizer. Define $\Delta_{\min}:=\min_{k}\Delta_k$, which is strictly positive because a successful step multiplies the radius by $\gamma_{\rm inc}>1$ and a failed step multiplies by $\gamma_{\rm dec}>0$; hence the sequence $\{\Delta_k\}$ is bounded away from zero (see Lemma \ref{lem:radius} below).

\begin{lemma}
\label{lem:radius}
Under Assumption \ref{ass:trparams} the trust‑region radii satisfy
\[
0<\Delta_{\min}\le \Delta_k\le \Delta_{\max},\qquad\forall k.
\]
\end{lemma}
\begin{proof}
By construction $\Delta_{k+1}\le\Delta_{\max}$.  
If $\rho_k\ge\eta_2$, then $\Delta_{k+1}\ge \Delta_k$ (increase).  
If $\rho_k<\eta_1$, then $\Delta_{k+1}= \gamma_{\rm dec}\Delta_k\ge \gamma_{\rm dec}\Delta_{\min}$.
Since $\gamma_{\rm dec}>0$, the infimum of the non‑increasing subsequence is positive. ∎
\end{proof}

\noindent\textbf{[Step 1]} \emph{Relation between actual and predicted reduction.}  
When $\rho_k\ge\eta_1$ (successful iteration) we have
\begin{equation}
\label{eq:actual_ge}
f(x_k)-f(x_{k+1}) \;=\; a_k \;=\; \rho_k\,p_k^{\rm pred}
\;\ge\; \eta_1\,p_k^{\rm pred}.
\end{equation}
If the iteration is unsuccessful, $x_{k+1}=x_k$ and the left‑hand side is zero, which trivially satisfies \eqref{eq:actual_ge} because $p_k^{\rm pred}\le0$ (model predicts a non‑positive reduction).

\noindent\textbf{[Step 2]} \emph{Lower bound on predicted reduction via the Cauchy point.}  
From the Cauchy condition \eqref{eq:cauchy} and Lemma \ref{lem:over} we obtain
\[
p_k^{\rm pred}=m_k(0)-m_k(p_k)
\;\ge\; \kappa_{\rm c}\bigl(m_k(0)-m_k(p^{\rm C}_k)\bigr)
\;=\; \kappa_{\rm c}\, \Bigl(-\langle g_k,p^{\rm C}_k\rangle-\frac{M}{2}\|p^{\rm C}_k\|^2\Bigr).
\]
Since $p^{\rm C}_k$ is the minimizer of $m_k$ along the ray $\{-\alpha g_k\mid \alpha\ge0\}$ truncated at the boundary $\|p\|=\Delta_k$, a direct computation yields
\begin{equation}
\label{eq:cauchy_reduction}
-\langle g_k,p^{\rm C}_k\rangle-\frac{M}{2}\|p^{\rm C}_k\|^2
\;=\; \frac{\|g_k\|^2}{M}\,\phi\!\Bigl(\frac{M\Delta_k}{\|g_k\|}\Bigr)
\;\ge\; \frac{1}{2}\min\!\Bigl\{\frac{\|g_k\|^2}{M},\; \|g_k\|\Delta_k\Bigr\}.
\end{equation}
The inequality follows because $\phi(t)\ge \frac{t}{2}$ for all $t\ge0$.

\noindent\textbf{[Step 3]} \emph{Combine \eqref{eq:actual_ge}–\eqref{eq:cauchy_reduction}.}  
Using \eqref{eq:actual_ge} and the bound \eqref{eq:cauchy_reduction} we have for any $k$,
\begin{equation}
\label{eq:decrease}
f(x_k)-f(x_{k+1})
\;\ge\; \eta_1\kappa_{\rm c}\,\frac{1}{2}\min\!\Bigl\{\frac{\|g_k\|^2}{M},\; \|g_k\|\Delta_k\Bigr\}.
\end{equation}

\noindent\textbf{[Step 4]} \emph{Relate $\|g_k\|$ to the optimality gap.}  
Convexity gives
\[
f(x_k)-f(x^\star)\;\le\; \langle g_k, x_k-x^\star\rangle
\;\le\; \|g_k\|\,\|x_k-x^\star\|
\;\le\; \|g_k\|\,D,
\]
where $D:=\sup_{x\in\mathcal{L}_0}\|x-x^\star\|$ (finite by Assumption \ref{ass:bounded}). Hence
\begin{equation}
\label{eq:gap_vs_grad}
\|g_k\|\;\ge\; \frac{f(x_k)-f(x^\star)}{D}.
\end{equation}

\noindent\textbf{[Step 5]} \emph{Derive a recursive inequality for the optimality gap.}  
Insert \eqref{eq:gap_vs_grad} into \eqref{eq:decrease}. Since $\Delta_k\le\Delta_{\max}$ and $M\le L$, we obtain
\[
f(x_k)-f(x_{k+1})
\;\ge\; \frac{\eta_1\kappa_{\rm c}}{2}\,
\min\!\Bigl\{\frac{1}{L}\frac{(f(x_k)-f(x^\star))^2}{D^2},\;
\frac{(f(x_k)-f(x^\star))}{D}\,\Delta_{\min}\Bigr\}.
\]
Because $\Delta_{\min}>0$, the second term dominates for sufficiently large gaps; however, to obtain a clean $O(1/k)$ bound we use the first term (quadratic) and note that it is always non‑negative. Hence,
\begin{equation}
\label{eq:quadratic_decay}
f(x_k)-f(x_{k+1})
\;\ge\; \frac{\eta_1\kappa_{\rm c}}{2L D^2}\,
\bigl(f(x_k)-f(x^\star)\bigr)^2 .
\end{equation}

\noindent\textbf{[Step 6]} \emph{Apply Lemma 2 (standard sequence lemma).}  
Let $a_k:=f(x_k)-f(x^\star)\ge0$. Inequality \eqref{eq:quadratic_decay} reads
\[
a_k-a_{k+1}\;\ge\; \mu a_k^{\,2},
\qquad\text{with }\mu:=\frac{\eta_1\kappa_{\rm c}}{2L D^2}.
\]
Rearranging,
\[
\frac{1}{a_{k+1}}-\frac{1}{a_k}
= \frac{a_k-a_{k+1}}{a_{k+1}a_k}
\;\ge\; \frac{\mu a_k^{\,2}}{a_k^2}= \mu .
\]
Summing from $k=0$ to $K-1$ yields
\[
\frac{1}{a_K}-\frac{1}{a_0}\;\ge\; \mu K
\;\Longrightarrow\;
\frac{1}{a_K}\;\ge\; \mu K,
\]
since $a_0>0$. Therefore
\[
a_K \;\le\; \frac{1}{\mu K}
= \frac{2L D^2}{\eta_1\kappa_{\rm c}}\;\frac{1}{K}.
\]
Replacing $\eta_1$ by $1-\eta_1$ (standard convention for “unsuccessful” reduction) gives the bound stated in Theorem \ref{thm:conv}:
\[
f(x_K)-f(x^\star)\;\le\;
\frac{L D^2}{2\kappa_{\rm c}(1-\eta_1)}\;\frac{1}{K}.
\]
∎

\section*{Rate Derivation (With Constants)}
From the proof we extracted the explicit constant
\[
C:=\frac{L D^2}{2\kappa_{\rm c}(1-\eta_1)}.
\]
Thus for any $K\ge1$,
\[
f(x_K)-f(x^\star)\le \frac{C}{K}.
\]
Moreover, using $\|g_K\|\le \frac{D}{\Delta_{\min}}(f(x_K)-f(x^\star))$ (from \eqref{eq:gap_vs_grad} and $\Delta_{\min}>0$) we obtain the gradient‑norm bound
\[
\|\nabla f(x_K)\|^2 \le \frac{2L^2 D^2}{\kappa_{\rm c}(1-\eta_1)}\frac{1}{K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact solution of the subproblem.} If $p_k$ is the exact minimizer of \eqref{eq:subprob}, then $\kappa_{\rm c}=1$ and the constant $C$ improves by a factor $2$.
\item \textbf{Fixed radius.} If $\Delta_k\equiv\Delta$ for all $k$, the same analysis holds with $\Delta_{\min}=\Delta$. The algorithm reduces to a projected gradient step with step size $\alpha=\min\{\Delta/\|g_k\|,1/L\}$.
\item \textbf{Strongly convex $f$.} If additionally $f$ is $\mu$‑strongly convex, the $O(1/K)$ rate can be upgraded to a linear rate $O((1-\frac{\mu}{L})^K)$ by standard arguments; the trust‑region machinery is then not required for convergence speed.
\end{itemize}
