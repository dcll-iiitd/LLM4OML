\section*{Algorithm Name}
\textbf{Bregman (Mirror) Gradient Descent (MGD)}  

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable objective and let  
$h:\R^{d}\rightarrow\R$ be a $\sigma$‑strongly convex \emph{prox‑function} w.r.t. a norm $\|\cdot\|$,
i.e.
\[
h(y)\ge h(x)+\inner{\grad h(x)}{y-x}+\frac{\sigma}{2}\|y-x\|^{2},\qquad\forall x,y\in\R^{d}.
\]
The associated Bregman divergence is
\[
\D(x,y)\;:=\;h(x)-h(y)-\inner{\grad h(y)}{x-y}.
\tag{1}
\]

Given a stepsize sequence $\{\eta_{k}\}_{k\ge0}$, the MGD update is
\[
\boxed{
x_{k+1}= \arg\min_{x\in\R^{d}}
\Big\{ \inner{\grad f(x_{k})}{x}
        +\frac{1}{\eta_{k}}\D(x,x_{k})\Big\}.
}
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Bregman (Mirror) Gradient Descent}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsizes $\{\eta_{k}\}_{k\ge0}>0$, prox‑function $h$.
\For{$k=0,1,2,\dots$}
    \State Compute gradient $g_{k}\gets\grad f(x_{k})$.
    \State $x_{k+1}\gets\displaystyle\arg\min_{x}
            \Big\{ \inner{g_{k}}{x}+\frac{1}{\eta_{k}}\D(x,x_{k})\Big\}$.
    \If{termination criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{return} $x_{k+1}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional problem  

\[
f(w)=(w-3)^{2},\qquad w_{0}=0,\qquad \eta_{k}\equiv\eta=0.1,
\]
and choose the Euclidean prox‑function $h(w)=\tfrac12 w^{2}$.
Then $\D(w,w_{k})=\tfrac12 (w-w_{k})^{2}$.

\medskip
\textbf{Iteration $k=0$}
\begin{align*}
g_{0}&=\grad f(w_{0})=2(w_{0}-3)=-6,\\[4pt]
w_{1}&=\arg\min_{w}\Big\{-6w+\frac{1}{0.1}\tfrac12(w-0)^{2}\Big\}
      =\arg\min_{w}\big\{-6w+5w^{2}\big\}.\\
\text{Setting derivative to zero: }&-6+10w=0\;\Rightarrow\; w_{1}=0.6,\\
f(w_{1})&=(0.6-3)^{2}=5.76.
\end{align*}

\textbf{Iteration $k=1$}
\begin{align*}
g_{1}&=2(w_{1}-3)=2(0.6-3)=-4.8,\\[4pt]
w_{2}&=\arg\min_{w}\Big\{-4.8w+5(w-0.6)^{2}\Big\}
      =\arg\min_{w}\big\{-4.8w+5(w^{2}-1.2w+0.36)\big\},\\
\text{Derivative: }&-4.8+10(w-0.6)=0\;\Rightarrow\;10w-6-4.8=0\;\Rightarrow\;w_{2}=1.08,\\
f(w_{2})&=(1.08-3)^{2}=3.6864.
\end{align*}

\textbf{Iteration $k=2$}
\begin{align*}
g_{2}&=2(w_{2}-3)=2(1.08-3)=-3.84,\\[4pt]
w_{3}&=\arg\min_{w}\Big\{-3.84w+5(w-1.08)^{2}\Big\},\\
\text{Derivative: }&-3.84+10(w-1.08)=0\;\Rightarrow\;10w-10.8-3.84=0\;\Rightarrow\;w_{3}=1.464,\\
f(w_{3})&=(1.464-3)^{2}=2.366\;.
\end{align*}

The iterates move monotonically toward the minimiser $w^{\star}=3$ and the objective values decrease.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]\label{as:convex}
$f$ is convex on $\R^{d}$ and $L$‑smooth w.r.t. the norm $\|\cdot\|$, i.e.
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity of the Prox‑function]\label{as:strong}
$h$ is $\sigma$‑strongly convex w.r.t. $\|\cdot\|$ (as defined above). Consequently,
\[
\D(x,y)\ge\frac{\sigma}{2}\|x-y\|^{2},\qquad\forall x,y\in\R^{d}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsizes satisfy $\eta_{k}>0$ for all $k$ and
\[
\sum_{k=0}^{\infty}\eta_{k}=+\infty,\qquad
\sum_{k=0}^{\infty}\eta_{k}^{2}<+\infty .
\tag{A3}
\]
A typical choice is constant $\eta_{k}=\eta\le \frac{\sigma}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Gradient Descent]\label{thm:conv}
Let Assumptions \ref{as:convex}--\ref{as:stepsize} hold and let $x^{\star}\in\arg\min f$. 
Define $R^{2}:=\D(x^{\star},x_{0})$. Then for any $K\ge0$,
\[
\boxed{
f(\bar{x}_{K})-f(x^{\star})
\;\le\;
\frac{R^{2}}{\displaystyle\sum_{k=0}^{K}\eta_{k}}
\;+\;
\frac{L}{2\sigma}\,\frac{\displaystyle\sum_{k=0}^{K}\eta_{k}^{2}}
{\displaystyle\sum_{k=0}^{K}\eta_{k}},
}
\tag{3}
\]
where $\bar{x}_{K}:=\big(\sum_{k=0}^{K}\eta_{k}x_{k+1}\big)/\big(\sum_{k=0}^{K}\eta_{k}\big)$.  
In particular, for a constant stepsize $\eta\le\sigma/L$,
\[
f(\bar{x}_{K})-f(x^{\star})\le\frac{R^{2}}{\eta (K+1)}.
\tag{4}
\]
Thus $f(\bar{x}_{K})\to f(x^{\star})$ at the rate $O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The algorithm is well‑defined because the sub‑problem (2) is strictly convex (quadratic term $\frac{1}{\eta_{k}}\D(\cdot,x_{k})$ dominates).  
\item \textbf{Stepsize Sensitivity:} Condition $\eta\le\sigma/L$ guarantees that the second term in (3) does not dominate, yielding the clean $O(1/K)$ rate. Larger stepsizes may still converge but the bound degrades.  
\item \textbf{Comparison to Euclidean Gradient Descent:} With $h(x)=\tfrac12\|x\|^{2}$, $\D$ reduces to $\tfrac12\|x-y\|^{2}$ and (2) becomes the standard gradient step $x_{k+1}=x_{k}-\eta_{k}\grad f(x_{k})$. The bound (4) coincides with the classical $O(1/K)$ rate for smooth convex functions.  
\item \textbf{Adaptivity:} By choosing a geometry‑matching $h$ (e.g., entropy for simplex constraints) the algorithm exploits problem structure and can obtain tighter constants $R^{2}$.
\end{itemize}

\section*{Discussion}
The theorem shows that the mirror‑gradient scheme inherits the optimal $O(1/K)$ convergence for smooth convex optimization while allowing non‑Euclidean geometries via $h$. The proof rests only on the three‑point identity of Bregman divergences and the smoothness inequality (A1). The result extends verbatim to stochastic gradients by taking expectations, leading to an $O(1/\sqrt{K})$ rate under bounded variance, similarly to classical SGD.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $x,y,z\in\R^{d}$,
\[
\D(x,z)=\D(x,y)+\D(y,z)+\inner{\grad h(y)-\grad h(z)}{x-y}.
\tag{5}
\]
\end{lemma}
\begin{proof}
By expanding the definition (1) for each pair and simplifying, we obtain (5). \hfill\qedsymbol
\end{proof}

\begin{lemma}[Smoothness bound]\label{lem:smooth}
If $f$ satisfies (A1) then for any $x,y$,
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\|y-x\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{as:convex}. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}. Throughout the proof $x^{\star}$ denotes an arbitrary minimiser of $f$.

\medskip
\noindent\textbf{Step 1 (Optimality condition of the sub‑problem).}  
From the definition (2) and convexity of the objective in the sub‑problem we have for any $x$,
\[
\inner{\grad f(x_{k})}{x_{k+1}-x}
    +\frac{1}{\eta_{k}}\inner{\grad_{1}\D(x_{k+1},x_{k})}{x_{k+1}-x}\le0.
\tag{7}
\]
Using $\grad_{1}\D(u,v)=\grad h(u)-\grad h(v)$, (7) rewrites as
\[
\inner{\grad f(x_{k})}{x_{k+1}-x}
    +\frac{1}{\eta_{k}}\inner{\grad h(x_{k+1})-\grad h(x_{k})}{x_{k+1}-x}\le0.
\tag{8}
\]

\medskip
\noindent\textbf{Step 2 (Rearrange the inner product).}  
Move the second term to the right‑hand side:
\[
\inner{\grad f(x_{k})}{x_{k+1}-x}
    \le\frac{1}{\eta_{k}}\inner{\grad h(x_{k})-\grad h(x_{k+1})}{x_{k+1}-x}.
\tag{9}
\]

\medskip
\noindent\textbf{Step 3 (Apply three‑point identity).}  
Set $y=x_{k+1}$, $z=x_{k}$ in Lemma \ref{lem:three} and rearrange:
\[
\D(x,x_{k})-\D(x,x_{k+1})-\D(x_{k+1},x_{k})
   =\inner{\grad h(x_{k})-\grad h(x_{k+1})}{x-x_{k+1}}.
\tag{10}
\]
Replace $x-x_{k+1}$ by $-(x_{k+1}-x)$ in (10) and substitute into (9):
\[
\inner{\grad f(x_{k})}{x_{k+1}-x}
    \le\frac{1}{\eta_{k}}\big[\,\D(x,x_{k})-\D(x,x_{k+1})-\D(x_{k+1},x_{k})\,\big].
\tag{11}
\]

\medskip
\noindent\textbf{Step 4 (Convexity of $f$).}  
Convexity yields
\[
f(x_{k})-f(x)\le\inner{\grad f(x_{k})}{x_{k}-x}.
\tag{12}
\]
Add and subtract $\inner{\grad f(x_{k})}{x_{k+1}-x}$:
\[
f(x_{k})-f(x)
    =\inner{\grad f(x_{k})}{x_{k+1}-x}
      +\inner{\grad f(x_{k})}{x_{k}-x_{k+1}}.
\tag{13}
\]

\medskip
\noindent\textbf{Step 5 (Bound the second term via smoothness).}  
Using Lemma \ref{lem:smooth} with $y=x_{k+1}$ and $x=x_{k}$,
\[
f(x_{k+1})\le f(x_{k})+\inner{\grad f(x_{k})}{x_{k+1}-x_{k}}+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{14}
\]
Rearrange (14) to obtain
\[
\inner{\grad f(x_{k})}{x_{k}-x_{k+1}}
    \le f(x_{k})-f(x_{k+1})+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{15}
\]

\medskip
\noindent\textbf{Step 6 (Combine (13), (11), (15)).}  
Insert (11) and (15) into (13):
\[
\begin{aligned}
f(x_{k})-f(x)
&\le \frac{1}{\eta_{k}}\big[\,\D(x,x_{k})-\D(x,x_{k+1})-\D(x_{k+1},x_{k})\,\big] \\
&\qquad + f(x_{k})-f(x_{k+1})+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{16}
\]

\medskip
\noindent\textbf{Step 7 (Cancel $f(x_{k})$ terms).}  
Subtract $f(x_{k})$ from both sides of (16):
\[
f(x_{k+1})-f(x)
\le \frac{1}{\eta_{k}}\big[\,\D(x,x_{k})-\D(x,x_{k+1})-\D(x_{k+1},x_{k})\,\big]
     +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{17}
\]

\medskip
\noindent\textbf{Step 8 (Relate Euclidean norm to Bregman divergence).}  
Strong convexity (A2) gives $\D(x_{k+1},x_{k})\ge \frac{\sigma}{2}\|x_{k+1}-x_{k}\|^{2}$. Hence,
\[
\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
\le \frac{L}{\sigma}\,\D(x_{k+1},x_{k}).
\tag{18}
\]

\medskip
\noindent\textbf{Step 9 (Plug (18) into (17)).}
\[
f(x_{k+1})-f(x)
\le \frac{1}{\eta_{k}}\big[\,\D(x,x_{k})-\D(x,x_{k+1})\,\big]
    +\Big(\frac{L}{\sigma}-\frac{1}{\eta_{k}}\Big)\D(x_{k+1},x_{k}).
\tag{19}
\]

\medskip
\noindent\textbf{Step 10 (Choose stepsize).}  
Assume $\eta_{k}\le \sigma/L$, i.e. $\frac{L}{\sigma}-\frac{1}{\eta_{k}}\le0$. Then the second term on the right‑hand side of (19) is non‑positive and can be dropped:
\[
f(x_{k+1})-f(x)
\le \frac{1}{\eta_{k}}\big[\,\D(x,x_{k})-\D(x,x_{k+1})\,\big].
\tag{20}
\]

\medskip
\noindent\textbf{Step 11 (Telescoping sum).}  
Sum (20) over $k=0,\dots,K$ and use $x=x^{\star}$:
\[
\sum_{k=0}^{K}\eta_{k}\big(f(x_{k+1})-f(x^{\star})\big)
    \le \D(x^{\star},x_{0})-\D(x^{\star},x_{K+1})
    \le R^{2},
\tag{21}
\]
where $R^{2}:=\D(x^{\star},x_{0})$.

\medskip
\noindent\textbf{Step 12 (Weighted average).}  
Define the weighted average $\bar{x}_{K}:=\frac{\sum_{k=0}^{K}\eta_{k}x_{k+1}}{\sum_{k=0}^{K}\eta_{k}}$.  
Convexity of $f$ gives
\[
f(\bar{x}_{K})-f(x^{\star})
\le \frac{\sum_{k=0}^{K}\eta_{k}\big(f(x_{k+1})-f(x^{\star})\big)}
          {\sum_{k=0}^{K}\eta_{k}}.
\tag{22}
\]
Combine (21) and (22) to obtain
\[
f(\bar{x}_{K})-f(x^{\star})
\le \frac{R^{2}}{\displaystyle\sum_{k=0}^{K}\eta_{k}}.
\tag{23}
\]

\medskip
\noindent\textbf{Step 13 (General stepsizes).}  
If the stepsizes are not bounded by $\sigma/L$, we retain the extra term in (19). Summing (19) yields
\[
\sum_{k=0}^{K}\eta_{k}\big(f(x_{k+1})-f(x^{\star})\big)
 \le R^{2}
   +\frac{L}{\sigma}\sum_{k=0}^{K}\eta_{k}^{2}.
\tag{24}
\]
Dividing by $\sum_{k=0}^{K}\eta_{k}$ and using (22) gives the bound (3).

\medskip
\noindent\textbf{Step 14 (Constant stepsize).}  
For $\eta_{k}\equiv\eta\le\sigma/L$, $\sum_{k=0}^{K}\eta_{k}=\eta(K+1)$ and the second term in (24) vanishes, yielding (4).

Thus Theorem \ref{thm:conv} is proved. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
From (23) with $\eta_{k}=\eta$,
\[
f(\bar{x}_{K})-f(x^{\star})\le\frac{R^{2}}{\eta(K+1)}.
\]
Choosing $\eta=\sigma/L$ (the largest admissible constant stepsize) gives the explicit constant
\[
f(\bar{x}_{K})-f(x^{\star})\le\frac{L}{\sigma}\,\frac{R^{2}}{K+1}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Euclidean geometry.} $h(x)=\tfrac12\|x\|_{2}^{2}$, $\sigma=1$, $\D(x,y)=\tfrac12\|x-y\|_{2}^{2}$. The algorithm reduces to standard gradient descent and Theorem \ref{thm:conv} recovers the classical $O(1/K)$ rate.
\item \textbf{Probability simplex.} $h(x)=\sum_{i}x_{i}\log x_{i}$ (negative entropy) yields the Kullback–Leibler divergence as $\D$. The same bound holds with $R^{2}=D_{\mathrm{KL}}(x^{\star}\|x_{0})$.
\item \textbf{Stochastic gradients.} If $g_{k}$ is an unbiased estimator of $\grad f(x_{k})$ with bounded variance $\sigma^{2}$, a standard martingale argument adds a term $O(\sigma/\sqrt{\sum\eta_{k}})$, leading to the $O(1/\sqrt{K})$ stochastic rate.
\end{enumerate}
