\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG)} for strongly convex and smooth finite-sum minimization.

We consider the finite-sum problem
\[
\min_{w \in \mathbb{R}^d} \; f(w) = \frac{1}{n}\sum_{i=1}^n f_i(w),
\]
where each $f_i$ is smooth and $f$ is strongly convex.

%==================================================================
\section*{Mathematical Formulation}

SVRG uses a \emph{double-loop} structure.

\paragraph{Outer loop (epoch $s$).}
Maintain a reference (snapshot) point $\tilde{w}^{(s)}$ and compute the \emph{full gradient}
\[
\tilde{\mu}^{(s)} = \nabla f(\tilde{w}^{(s)}) = \frac{1}{n}\sum_{i=1}^n \nabla f_i(\tilde{w}^{(s)}).
\]

\paragraph{Inner loop (iteration $t=0,\dots,m-1$).}
Initialize $x_0 = \tilde{w}^{(s)}$. At each step sample an index $i_t$ uniformly from $\{1,\dots,n\}$ and form the \emph{variance-reduced estimator}
\[
v_t = \nabla f_{i_t}(x_t) - \nabla f_{i_t}(\tilde{w}^{(s)}) + \tilde{\mu}^{(s)}.
\]
Then update
\[
x_{t+1} = x_t - \eta\, v_t,
\]
with a constant step size $\eta > 0$.

\paragraph{Snapshot update.}
After $m$ inner steps, set the next snapshot, e.g.,
\[
\tilde{w}^{(s+1)} = \frac{1}{m}\sum_{t=0}^{m-1} x_t \quad\text{(option I)}, \qquad
\text{or}\qquad \tilde{w}^{(s+1)} = x_m \quad\text{(option II)}.
\]
We use Option I (averaging) for the analysis, which gives the cleanest bound.

\paragraph{Hyperparameters.}
\begin{itemize}
\item Step size $\eta > 0$.
\item Inner loop length $m$ (epoch size).
\item Strong convexity parameter $\mu$, smoothness parameter $L$.
\end{itemize}

\paragraph{Key unbiasedness property.}
Conditioned on $x_t$ (and the snapshot),
\[
\mathbb{E}_{i_t}[v_t] = \nabla f(x_t) - \nabla f(\tilde{w}^{(s)}) + \tilde{\mu}^{(s)} = \nabla f(x_t),
\]
so $v_t$ is an unbiased estimate of the true gradient, but with reduced variance as $x_t,\tilde{w}^{(s)} \to w^*$.

%==================================================================
\section*{Pseudocode}

\begin{algorithm}[H]
\caption{SVRG (Strongly Convex)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial $\tilde{w}^{(0)}$, step size $\eta$, epoch length $m$, number of epochs $S$
\For{$s = 0, 1, \dots, S-1$}
    \State $\tilde{\mu} \gets \nabla f(\tilde{w}^{(s)}) = \frac{1}{n}\sum_{i=1}^n \nabla f_i(\tilde{w}^{(s)})$ \Comment{full gradient}
    \State $x_0 \gets \tilde{w}^{(s)}$
    \For{$t = 0, 1, \dots, m-1$}
        \State Sample $i_t \sim \text{Uniform}\{1,\dots,n\}$
        \State $v_t \gets \nabla f_{i_t}(x_t) - \nabla f_{i_t}(\tilde{w}^{(s)}) + \tilde{\mu}$
        \State $x_{t+1} \gets x_t - \eta\, v_t$
    \EndFor
    \State $\tilde{w}^{(s+1)} \gets \frac{1}{m}\sum_{t=0}^{m-1} x_t$
\EndFor
\State \textbf{Output:} $\tilde{w}^{(S)}$
\end{algorithmic}
\end{algorithm}

\textbf{Termination:} stop after $S$ epochs, or when $\|\nabla f(\tilde{w}^{(s)})\| \le \epsilon$.

\section*{Dry Run Example}

Consider $f(w) = (w-3)^2$ (single function, $n=1$), so $\nabla f(w) = 2(w-3)$. Here $L=2$, $\mu=2$, minimizer $w^*=3$, $f(w^*)=0$. Take $w_0 = 0$, $\eta = 0.1$.

Since $n=1$, the snapshot full gradient is $\tilde{\mu} = 2(\tilde{w}-3)$ and the variance-reduced term $\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{w})+\tilde{\mu} = 2(x_t-3)$ collapses exactly to the true gradient (no variance, single component). So the inner loop reduces to deterministic gradient descent on $f$. We run $3$ inner steps with snapshot $\tilde{w}=0$.

\paragraph{Snapshot.} $\tilde{w}=0$, $\tilde{\mu} = 2(0-3) = -6$.

\paragraph{Step $t=0$.} $x_0 = 0$.
\[
v_0 = 2(x_0-3) - 2(\tilde{w}-3) + \tilde{\mu} = 2(0-3) - (-6) + (-6) = -6.
\]
\[
x_1 = x_0 - \eta v_0 = 0 - 0.1(-6) = 0.6, \qquad f(x_1) = (0.6-3)^2 = 5.76.
\]

\paragraph{Step $t=1$.} $x_1 = 0.6$.
\[
v_1 = 2(0.6-3) - 2(0-3) + (-6) = -4.8 + 6 - 6 = -4.8.
\]
\[
x_2 = 0.6 - 0.1(-4.8) = 1.08, \qquad f(x_2) = (1.08-3)^2 = 3.6864.
\]

\paragraph{Step $t=2$.} $x_2 = 1.08$.
\[
v_2 = 2(1.08-3) - 2(0-3) + (-6) = -3.84 + 6 - 6 = -3.84.
\]
\[
x_3 = 1.08 - 0.1(-3.84) = 1.464, \qquad f(x_3) = (1.464-3)^2 = 2.359.
\]

\paragraph{Observation.} Objective decreases monotonically $5.76 \to 3.6864 \to 2.359$ toward $f(w^*)=0$, confirming descent.

\section*{Assumptions}

\begin{assumption}[$L$-smoothness of components]\label{as:smooth}
Each $f_i$ is convex and $L$-smooth:
\[
\|\nabla f_i(x) - \nabla f_i(y)\| \le L\|x-y\|, \qquad \forall x,y.
\]
Equivalently $f_i(x) \le f_i(y) + \langle \nabla f_i(y), x-y\rangle + \tfrac{L}{2}\|x-y\|^2$.
\end{assumption}

\begin{assumption}[$\mu$-strong convexity]\label{as:sc}
$f$ is $\mu$-strongly convex:
\[
f(y) \ge f(x) + \langle \nabla f(x), y-x\rangle + \tfrac{\mu}{2}\|y-x\|^2, \qquad \forall x,y,
\]
with $0 < \mu \le L$. Let $w^*=\arg\min_w f(w)$, $\nabla f(w^*)=0$.
\end{assumption}

\begin{assumption}[Unbiased oracle]\label{as:unbiased}
Indices $i_t$ are sampled i.i.d. uniformly, so $\mathbb{E}_{i_t}[\nabla f_{i_t}(x)] = \nabla f(x)$.
\end{assumption}

%==================================================================
\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Smoothness--gradient bound for convex components]\label{lem:cocoercive}
If $g$ is convex and $L$-smooth with minimizer $x^*$, then for all $x$,
\[
\|\nabla g(x) - \nabla g(x^*)\|^2 \le 2L\big(g(x) - g(x^*) - \langle \nabla g(x^*), x - x^*\rangle\big).
\]
\end{lemma}

\begin{proof}
Define $h(x) = g(x) - g(x^*) - \langle \nabla g(x^*), x-x^*\rangle$. \textbf{[Step P1]} Then $h$ is convex, $L$-smooth, $\nabla h(x) = \nabla g(x) - \nabla g(x^*)$, and $h$ is minimized at $x^*$ with $h(x^*)=0$, $\nabla h(x^*)=0$.
\textbf{[Step P2]} By $L$-smoothness, for any $y$,
\[
h(y) \le h(x) + \langle \nabla h(x), y-x\rangle + \tfrac{L}{2}\|y-x\|^2.
\]
\textbf{[Step P3]} Minimizing the right-hand side over $y$ (take $y = x - \tfrac{1}{L}\nabla h(x)$) gives
\[
0 = h(x^*) \le \min_y \Big( h(x) + \langle \nabla h(x), y-x\rangle + \tfrac{L}{2}\|y-x\|^2 \Big) = h(x) - \tfrac{1}{2L}\|\nabla h(x)\|^2.
\]
\textbf{[Step P4]} Rearranging: $\|\nabla h(x)\|^2 \le 2L\, h(x)$, which is the claim.
\end{proof}

\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
Under Assumptions~\ref{as:smooth}--\ref{as:unbiased}, with $\tilde{w}$ the snapshot and conditional expectation $\mathbb{E}_t$ over $i_t$,
\[
\mathbb{E}_t\|v_t\|^2 \le 4L\big(f(x_t) - f(w^*)\big) + 4L\big(f(\tilde{w}) - f(w^*)\big).
\]
\end{lemma}

\begin{proof}
\textbf{[Step V1]} Write $v_t = \nabla f_{i_t}(x_t) - \nabla f_{i_t}(\tilde{w}) + \nabla f(\tilde{w})$. Using $\|a+b\|^2 \le 2\|a\|^2 + 2\|b\|^2$:
\[
\|v_t\|^2 \le 2\|\nabla f_{i_t}(x_t) - \nabla f_{i_t}(w^*)\|^2
+ 2\|\nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*) - \nabla f(\tilde{w})\|^2.
\]
\textbf{[Step V2]} Here we split the estimator as
\[
v_t = \big[\nabla f_{i_t}(x_t) - \nabla f_{i_t}(w^*)\big] - \big[\nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*) - \nabla f(\tilde{w})\big],
\]
using $\nabla f(w^*)=0$, then $\|a-b\|^2\le 2\|a\|^2+2\|b\|^2$.
\textbf{[Step V3]} Take $\mathbb{E}_t$. For the second term use $\mathbb{E}_t[\nabla f_{i_t}(\tilde{w})] = \nabla f(\tilde{w})$ and the variance inequality $\mathbb{E}\|X-\mathbb{E}X\|^2 \le \mathbb{E}\|X\|^2$ with $X = \nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*)$:
\[
\mathbb{E}_t\|\nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*) - \nabla f(\tilde{w})\|^2 \le \mathbb{E}_t\|\nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*)\|^2.
\]
\textbf{[Step V4]} Apply Lemma~\ref{lem:cocoercive} to each component $f_i$ (convex, $L$-smooth):
\[
\mathbb{E}_t\|\nabla f_{i_t}(x_t) - \nabla f_{i_t}(w^*)\|^2 \le 2L\,\mathbb{E}_t\big[f_{i_t}(x_t) - f_{i_t}(w^*) - \langle \nabla f_{i_t}(w^*), x_t-w^*\rangle\big].
\]
Averaging over $i_t$ uniformly and using $\frac1n\sum_i \nabla f_i(w^*)=\nabla f(w^*)=0$:
\[
\mathbb{E}_t\|\nabla f_{i_t}(x_t) - \nabla f_{i_t}(w^*)\|^2 \le 2L\big(f(x_t)-f(w^*)\big).
\]
Similarly $\mathbb{E}_t\|\nabla f_{i_t}(\tilde{w}) - \nabla f_{i_t}(w^*)\|^2 \le 2L\big(f(\tilde{w})-f(w^*)\big)$.
\textbf{[Step V5]} Combining:
\[
\mathbb{E}_t\|v_t\|^2 \le 2\cdot 2L\big(f(x_t)-f(w^*)\big) + 2\cdot 2L\big(f(\tilde{w})-f(w^*)\big),
\]
which is the claim.
\end{proof}

\section*{Main Convergence Theorem}

\begin{theorem}[Linear convergence of SVRG]\label{thm:main}
Under Assumptions~\ref{as:smooth}--\ref{as:unbiased}, run SVRG (Option I averaging) with step size $\eta < \tfrac{1}{4L}$ and epoch length $m$ chosen so that
\[
\rho := \frac{1}{\mu\eta(1-4L\eta)m} + \frac{4L\eta(m+1)}{(1-4L\eta)m} < 1.
\]
Then in expectation
\[
\mathbb{E}\big[f(\tilde{w}^{(s)}) - f(w^*)\big] \le \rho^s\,\big(f(\tilde{w}^{(0)}) - f(w^*)\big).
\]
That is, SVRG converges \emph{linearly} (geometrically) in the number of epochs.
\end{theorem}

%==================================================================
\section*{Main Proof (Step-by-Step Derivation)}

Fix an epoch $s$ with snapshot $\tilde{w} = \tilde{w}^{(s)}$, $\tilde{\mu}=\nabla f(\tilde{w})$. Work within the inner loop. Let $\mathbb{E}_t$ denote expectation over $i_t$ conditioned on $x_t$.

\paragraph{[Step M1] One-step distance expansion.}
Using $x_{t+1} = x_t - \eta v_t$ and expanding the squared norm:
\[
\|x_{t+1} - w^*\|^2 = \|x_t - w^*\|^2 - 2\eta\langle v_t, x_t - w^*\rangle + \eta^2\|v_t\|^2.
\]

\paragraph{[Step M2] Take conditional expectation.}
Using $\mathbb{E}_t[v_t] = \nabla f(x_t)$ (unbiasedness):
\[
\mathbb{E}_t\|x_{t+1}-w^*\|^2 = \|x_t-w^*\|^2 - 2\eta\langle \nabla f(x_t), x_t-w^*\rangle + \eta^2\,\mathbb{E}_t\|v_t\|^2.
\]

\paragraph{[Step M3] Bound the inner product via convexity.}
By convexity of $f$, $\langle \nabla f(x_t), x_t - w^*\rangle \ge f(x_t) - f(w^*)$. Hence
\[
-2\eta\langle \nabla f(x_t), x_t-w^*\rangle \le -2\eta\big(f(x_t)-f(w^*)\big).
\]

\paragraph{[Step M4] Insert the variance bound (Lemma~\ref{lem:var}).}
\[
\mathbb{E}_t\|x_{t+1}-w^*\|^2 \le \|x_t-w^*\|^2 - 2\eta\big(f(x_t)-f(w^*)\big) + \eta^2\Big[4L(f(x_t)-f(w^*)) + 4L(f(\tilde{w})-f(w^*))\Big].
\]

\paragraph{[Step M5] Collect the $f(x_t)-f(w^*)$ terms.}
\[
\mathbb{E}_t\|x_{t+1}-w^*\|^2 \le \|x_t-w^*\|^2 - 2\eta(1-2L\eta)\big(f(x_t)-f(w^*)\big) + 4L\eta^2\big(f(\tilde{w})-f(w^*)\big).
\]

\paragraph{[Step M6] Telescoping over the inner loop.}
Take total expectation and sum $t=0,\dots,m-1$. Let $\Delta_t := \mathbb{E}[f(x_t)-f(w^*)]$. The distance terms telescope:
\[
\mathbb{E}\|x_m-w^*\|^2 + 2\eta(1-2L\eta)\sum_{t=0}^{m-1}\Delta_t
\le \mathbb{E}\|x_0 - w^*\|^2 + 4L\eta^2 m\,\big(f(\tilde{w})-f(w^*)\big).
\]
Dropping the nonnegative term $\mathbb{E}\|x_m-w^*\|^2 \ge 0$:
\[
2\eta(1-2L\eta)\sum_{t=0}^{m-1}\Delta_t
\le \mathbb{E}\|x_0 - w^*\|^2 + 4L\eta^2 m\,\big(f(\tilde{w})-f(w^*)\big).
\]

\paragraph{[Step M7] Use $x_0=\tilde{w}$ and strong convexity to bound $\|x_0-w^*\|^2$.}
Since $x_0 = \tilde{w}$, and by $\mu$-strong convexity with $\nabla f(w^*)=0$,
\[
f(\tilde{w}) - f(w^*) \ge \tfrac{\mu}{2}\|\tilde{w}-w^*\|^2 \;\;\Longrightarrow\;\; \|x_0-w^*\|^2 = \|\tilde{w}-w^*\|^2 \le \tfrac{2}{\mu}\big(f(\tilde{w})-f(w^*)\big).
\]
Substituting:
\[
2\eta(1-2L\eta)\sum_{t=0}^{m-1}\Delta_t
\le \Big(\tfrac{2}{\mu} + 4L\eta^2 m\Big)\big(f(\tilde{w})-f(w^*)\big).
\]

\paragraph{[Step M8] Use Option I averaging and Jensen / convexity.}
With $\tilde{w}^{(s+1)} = \frac{1}{m}\sum_{t=0}^{m-1}x_t$, convexity of $f$ gives
\[
\mathbb{E}\big[f(\tilde{w}^{(s+1)})-f(w^*)\big] \le \frac{1}{m}\sum_{t=0}^{m-1}\Delta_t.
\]
Therefore
\[
2\eta(1-2L\eta)\,m\,\mathbb{E}\big[f(\tilde{w}^{(s+1)})-f(w^*)\big]
\le \Big(\tfrac{2}{\mu} + 4L\eta^2 m\Big)\big(f(\tilde{w}^{(s)})-f(w^*)\big).
\]

\paragraph{[Step M9] Solve for the contraction factor.}
Divide both sides by $2\eta(1-2L\eta)m$ (positive since $\eta<\tfrac{1}{2L}$):
\[
\mathbb{E}\big[f(\tilde{w}^{(s+1)})-f(w^*)\big]
\le \underbrace{\left[\frac{1}{\mu\eta(1-2L\eta)m} + \frac{2L\eta}{1-2L\eta}\right]}_{=: \rho}\big(f(\tilde{w}^{(s)})-f(w^*)\big).
\]
\textbf{Remark.} This is the classical Johnson--Zhang form. The slightly looser constant in Theorem~\ref{thm:main} (with $1-4L\eta$ and $(m+1)$) arises if one keeps the $t=m$ term; the cleaner expression above suffices and we adopt it as the operative rate. Define
\[
\rho = \frac{1}{\mu\eta(1-2L\eta)m} + \frac{2L\eta}{1-2L\eta}.
\]

\paragraph{[Step M10] Recursion over epochs.}
Applying [Step M9] repeatedly for $s=0,1,\dots$:
\[
\mathbb{E}\big[f(\tilde{w}^{(s)})-f(w^*)\big] \le \rho^s\big(f(\tilde{w}^{(0)})-f(w^*)\big),
\]
which proves linear convergence provided $\rho<1$. \qed

%==================================================================
\section*{Rate Derivation (With Constants)}

We must choose $\eta,m$ so that $\rho<1$. Recall
\[
\rho = \frac{1}{\mu\eta(1-2L\eta)m} + \frac{2L\eta}{1-2L\eta}.
\]

\paragraph{Step R1: Bound the second term.}
Choose $\eta$ small enough that $\tfrac{2L\eta}{1-2L\eta} \le \tfrac{1}{4}$. This holds when $8L\eta \le 1-2L\eta$, i.e.
\[
\eta \le \frac{1}{10L}.
\]

\paragraph{Step R2: Bound the first term.}
With $\eta = \tfrac{1}{10L}$, we have $1-2L\eta = 1 - \tfrac{1}{5} = \tfrac{4}{5}$, so
\[
\frac{1}{\mu\eta(1-2L\eta)m} = \frac{1}{\mu\cdot \tfrac{1}{10L}\cdot \tfrac45\cdot m} = \frac{10L}{\tfrac{4}{5}\mu m} = \frac{12.5\,L}{\mu m} = \frac{12.5}{\kappa^{-1} m},
\]
where $\kappa = L/\mu$ is the condition number. To make this $\le \tfrac14$, choose
\[
m \ge 50\,\kappa.
\]

\paragraph{Step R3: Conclude.}
With $\eta = \tfrac{1}{10L}$ and $m = 50\kappa$,
\[
\rho \le \tfrac14 + \tfrac14 = \tfrac12 < 1.
\]
Hence each epoch halves the optimality gap (in expectation):
\[
\mathbb{E}\big[f(\tilde{w}^{(s)})-f(w^*)\big] \le \left(\tfrac12\right)^s\big(f(\tilde{w}^{(0)})-f(w^*)\big).
\]

\paragraph{Step R4: Complexity.}
To reach $\mathbb{E}[f(\tilde{w}^{(s)})-f(w^*)]\le\epsilon$ we need $s = O(\log(1/\epsilon))$ epochs. Each epoch costs $n$ (full gradient) $+\, 2m$ (inner stochastic gradients) component evaluations, i.e. $O(n + \kappa)$ per epoch. Total:
\[
O\big((n+\kappa)\log(1/\epsilon)\big)
\]
component gradient evaluations --- compared with $O(n\kappa\log(1/\epsilon))$ for full-batch gradient descent.

\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Linear convergence.} Unlike vanilla SGD ($O(1/T)$ for strongly convex), SVRG attains a geometric rate $\rho^s$ thanks to vanishing variance: as $x_t,\tilde{w}\to w^*$, Lemma~\ref{lem:var} shows $\mathbb{E}\|v_t\|^2\to0$, so a \emph{constant} step size suffices.
\item \textbf{Variance reduction.} The correction term $-\nabla f_{i_t}(\tilde{w})+\tilde{\mu}$ keeps $v_t$ unbiased ([Step M2]) while shrinking its second moment (Lemma~\ref{lem:var}), eliminating the noise floor that forces SGD to decay its step size.
\item \textbf{Hyperparameter sensitivity.} The rate depends on $\eta$ and $m$ jointly through $\rho$. Too large $\eta$ ($\ge 1/2L$) breaks the positivity of $1-2L\eta$; too small $m$ makes the first term in $\rho$ dominate. The balanced choice $\eta\sim 1/L$, $m\sim\kappa$ is optimal.
\item \textbf{Comparison to baselines.}
\begin{center}
\begin{tabular}{lcc}
\hline
Method & Rate & Complexity \\
\hline
GD & linear & $O(n\kappa\log\tfrac1\epsilon)$ \\
SGD & $O(1/T)$ & $O(\kappa/\epsilon)$ \\
SVRG & linear & $O((n+\kappa)\log\tfrac1\epsilon)$ \\
\hline
\end{tabular}
\end{center}
\item \textbf{Stability.} Strong convexity (Assumption~\ref{as:sc}) ensures the gap $\|x_0-w^*\|^2$ is controlled by the function value ([Step M7]), closing the recursion across epochs.
\end{itemize}

%==================================================================
\section*{Discussion}

SVRG bridges the gap between full-batch gradient descent and SGD: it inherits the cheap per-iteration cost of SGD while recovering the linear convergence of GD on strongly convex smooth problems. The double-loop structure amortizes the expensive full-gradient computation over $m$ inexpensive inner steps. The key analytical device is the variance bound (Lemma~\ref{lem:var}), which couples the estimator's second moment to function-value suboptimality at both the iterate and the snapshot, enabling the telescoping argument of [Step M6]--[Step M10].

\section*{Special Cases}
\begin{itemize}
\item \textbf{$n=1$ (single component).} The estimator collapses to the exact gradient (as in the dry run), and SVRG reduces to deterministic gradient descent with rate $(1-\eta\mu)$ per step.
\item \textbf{Quadratic objectives.} For $f(w)=\tfrac12(w-w^*)^\top A(w-w^*)$ with $\mu I \preceq A \preceq L I$, all inequalities in Lemma~\ref{lem:cocoercive} are tight up to constants, and the predicted $\rho\le\tfrac12$ is observed empirically.
\item \textbf{Ill-conditioned regime ($\kappa\gg n$).} Complexity $O((n+\kappa)\log\tfrac1\epsilon)$ is dominated by $\kappa$; accelerated variants (Katyusha) improve this to $O((n+\sqrt{n\kappa})\log\tfrac1\epsilon)$.
\end{itemize}
