\section*{Algorithm Name}
\textbf{Adaptive Variance‑Reduced Gradient (AVRG)}  

The method maintains an exponential moving average of the stochastic gradient (a variance estimator) and uses a step size that is inversely proportional to the norm of the bias‑corrected estimator.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable and let $g(x,\xi)$ denote a stochastic gradient evaluated at $x$ with random seed $\xi$ such that $\E_{\xi}[g(x,\xi)]=\nabla f(x)$.  
The algorithm generates a sequence $\{x_t\}_{t\ge 0}$ as follows:

\bigskip
\noindent\textbf{Parameters}
\begin{itemize}[noitemsep]
    \item stepsize base $\eta>0$,
    \item exponential decay $\alpha\in(0,1]$ for the variance estimator,
    \item small constant $\beta>0$ for numerical stability,
    \item maximal number of iterations $T$.
\end{itemize}
\bigskip

\noindent\textbf{Recursions}
\begin{align}
    g_t &:= g(x_t,\xi_t), \qquad\text{(stochastic gradient)} \label{eq:gt}\\
    v_t &:= (1-\alpha)v_{t-1}+\alpha g_t, \qquad v_{-1}:=0, \label{eq:vt}\\
    \hat v_t &:= \frac{v_t}{1-(1-\alpha)^{t+1}}, \qquad\text{(bias correction)} \label{eq:hatvt}\\
    \eta_t &:= \frac{\eta}{\beta+\norm{\hat v_t}}, \qquad\text{(adaptive stepsize)} \label{eq:etat}\\
    x_{t+1} &:= x_t-\eta_t\,\hat v_t . \label{eq:xt}
\end{align}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Adaptive Variance‑Reduced Gradient (AVRG)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, decay $\alpha\in(0,1]$, stability $\beta>0$, max‑iter $T$.
\STATE $v_{-1}\leftarrow 0$.
\FOR{$t=0,\dots,T-1$}
    \STATE Sample $\xi_t$ and compute $g_t\gets g(x_t,\xi_t)$.
    \STATE $v_t\gets (1-\alpha)v_{t-1}+\alpha g_t$.
    \STATE $\hat v_t\gets v_t/(1-(1-\alpha)^{t+1})$.
    \STATE $\eta_t\gets \eta/(\beta+\norm{\hat v_t})$.
    \STATE $x_{t+1}\gets x_t-\eta_t\hat v_t$.
\ENDFOR
\STATE \textbf{Output:} $\{x_t\}_{t=0}^T$ (or $x_{\text{out}}=x_{\arg\min_{0\le t<T}\norm{\hat v_t}}$).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For any $x$, $\E_{\xi}[g(x,\xi)]=\nabla f(x)$.
\end{assumption}
\begin{assumption}[Bounded Second Moment]\label{ass:bounded}
There exists $\sigma^2>0$ such that for all $x$,
\[
\E_{\xi}\big[\norm{g(x,\xi)}^2\big]\le \norm{\nabla f(x)}^2+\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter Range]\label{ass:hyper}
\[
\eta\le\frac{1}{L},\qquad \alpha\in(0,1],\qquad \beta>0 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper}, let $\{x_t\}_{t=0}^{T-1}$ be generated by AVRG. Define the random index $I$ uniformly on $\{0,\dots,T-1\}$. Then
\[
\E\!\left[\norm{\nabla f(x_I)}^2\right]\;\le\;
\frac{2\big(f(x_0)-f^\star\big)}{\eta T}
\;+\;
\frac{2L\eta\sigma^2}{\beta^2}\;,
\tag{1}
\]
where $f^\star=\inf_x f(x)$. Consequently, choosing $\eta =\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields the rate
\[
\E\!\left[\norm{\nabla f(x_I)}^2\right]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability.} The denominator $\beta+\norm{\hat v_t}$ prevents division by zero; the adaptive stepsize \eqref{eq:etat} is bounded above by $\eta/\beta$.
\item \textbf{Hyper‑parameter Sensitivity.} Larger $\alpha$ yields a variance estimator that tracks the current gradient more closely (smaller bias) but with higher variance; smaller $\alpha$ provides stronger smoothing at the cost of bias. The theorem holds for any $\alpha\in(0,1]$.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item With $\alpha=1$ and $\beta=0$ the algorithm reduces to standard SGD with stepsize $\eta_t=\eta/\norm{g_t}$, for which only $\mathcal{O}(1/\sqrt{T})$ rates are known in the non‑convex case.
        \item When $\alpha\approx 1$ and $\beta$ is modest, AVRG behaves like SARAH/SPIDER: the moving average $v_t$ approximates a variance‑reduced gradient, and the bound \eqref{eq:1} recovers the $\mathcal{O}(1/T)$ rate after proper tuning of $\eta$.
    \end{itemize}
\item \textbf{Non‑convexity.} The result makes no convexity assumption; it provides a bound on the expected squared norm of the true gradient, the standard stationarity measure for non‑convex optimization.
\end{enumerate}

\section*{Discussion}
The adaptive stepsize couples the magnitude of the variance‑reduced estimator to the learning rate, automatically shrinking the step when the estimator is large (high variance) and enlarging it when the estimator stabilises. The bias‑correction \eqref{eq:hatvt} guarantees that $\E[\hat v_t]=\nabla f(x_t)$, which is crucial for the unbiasedness required in the proof. The theorem shows that AVRG attains the same order of convergence as the best known variance‑reduced methods while being fully online (single‑loop) and requiring only one stochastic gradient per iteration.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\norm{d}^2 .
\]
\end{lemma}
\begin{lemma}[Bias‑corrected estimator is unbiased]\label{lem:unbiased}
For the recursion \eqref{eq:vt}–\eqref{eq:hatvt},
\[
\E[\hat v_t\mid\mathcal{F}_t]=\nabla f(x_t),
\]
where $\mathcal{F}_t$ is the $\sigma$‑algebra generated by $\{x_0,\dots,x_t\}$.
\end{lemma}
\begin{proof}[Proof of Lemma \ref{lem:unbiased}]
By induction on $t$.
\begin{enumerate}
    \item[(Base)] $t=0$: $v_0=\alpha g_0$, $\E[g_0]=\nabla f(x_0)$, and $1-(1-\alpha)^{1}=\alpha$, so $\hat v_0=v_0/\alpha=g_0$, hence $\E[\hat v_0]=\nabla f(x_0)$.
    \item[(Induction)] Assume $\E[\hat v_{t-1}]=\nabla f(x_{t-1})$. Taking conditional expectation of \eqref{eq:vt} gives
    \[
    \E[v_t\mid\mathcal{F}_t]=(1-\alpha)v_{t-1}+\alpha\nabla f(x_t).
    \]
    Dividing by $1-(1-\alpha)^{t+1}$ and using the geometric series identity yields
    \[
    \E[\hat v_t\mid\mathcal{F}_t]=\nabla f(x_t).
    \]
\end{enumerate}
\end{proof}
\begin{lemma}[Second‑moment bound for $\hat v_t$]\label{lem:var}
Under Assumption \ref{ass:bounded},
\[
\E\!\left[\norm{\hat v_t}^2\mid\mathcal{F}_t\right]\le
\norm{\nabla f(x_t)}^2+\frac{\sigma^2}{\big(1-(1-\alpha)^{t+1}\big)^2}.
\]
\end{lemma}
\begin{proof}[Proof of Lemma \ref{lem:var}]
From \eqref{eq:vt},
\[
v_t=(1-\alpha)v_{t-1}+\alpha g_t.
\]
Take squared norm, expand and use $\|a+b\|^2\le 2\|a\|^2+2\|b\|^2$,
\[
\|v_t\|^2\le 2(1-\alpha)^2\|v_{t-1}\|^2+2\alpha^2\|g_t\|^2 .
\]
Iterating the recursion and noting that $v_{-1}=0$ gives
\[
\|v_t\|^2\le 2\alpha\sum_{k=0}^{t}(1-\alpha)^{t-k}\|g_k\|^2 .
\]
Dividing by the square of the bias‑correction factor $c_t:=1-(1-\alpha)^{t+1}$ yields the claim after taking conditional expectation and applying Assumption \ref{ass:bounded}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof, let $\mathcal{F}_t$ be the filtration generated up to iteration $t$.

\begin{enumerate}
\item[Step 1.] Apply Lemma \ref{lem:descent} with $d=\eta_t\hat v_t$:
\begin{align}
f(x_{t+1}) &\le f(x_t)-\eta_t\langle\nabla f(x_t),\hat v_t\rangle
            +\frac{L\eta_t^{2}}{2}\norm{\hat v_t}^{2}. \label{eq:descent}
\end{align}
\item[Step 2.] Take conditional expectation w.r.t. $\mathcal{F}_t$ and use Lemma \ref{lem:unbiased}:
\begin{align}
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
&\le f(x_t)-\eta_t\norm{\nabla f(x_t)}^{2}
   +\frac{L\eta_t^{2}}{2}\E\!\left[\norm{\hat v_t}^{2}\mid\mathcal{F}_t\right]. \label{eq:exp-descent}
\end{align}
\item[Step 3.] Insert the adaptive stepsize \eqref{eq:etat}:
\[
\eta_t=\frac{\eta}{\beta+\norm{\hat v_t}}\quad\Longrightarrow\quad
\eta_t^{2}\norm{\hat v_t}^{2}= \frac{\eta^{2}\norm{\hat v_t}^{2}}{(\beta+\norm{\hat v_t})^{2}}
\le \frac{\eta^{2}}{\beta^{2}}\norm{\hat v_t}^{2},
\]
where the inequality follows from $\beta+\norm{\hat v_t}\ge\beta$.
\item[Step 4.] Substitute the bound from Step 3 into \eqref{eq:exp-descent}:
\begin{align}
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
&\le f(x_t)-\eta_t\norm{\nabla f(x_t)}^{2}
   +\frac{L\eta^{2}}{2\beta^{2}}
     \E\!\left[\norm{\hat v_t}^{2}\mid\mathcal{F}_t\right]. \label{eq:mid}
\end{align}
\item[Step 5.] Use Lemma \ref{lem:var} to bound the second moment:
\begin{align}
\E\!\left[\norm{\hat v_t}^{2}\mid\mathcal{F}_t\right]
&\le \norm{\nabla f(x_t)}^{2}
   +\frac{\sigma^{2}}{c_t^{2}},\qquad
c_t:=1-(1-\alpha)^{t+1}. \label{eq:varbound}
\end{align}
\item[Step 6.] Plug \eqref{eq:varbound} into \eqref{eq:mid}:
\begin{align}
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
&\le f(x_t)-\eta_t\norm{\nabla f(x_t)}^{2}
   +\frac{L\eta^{2}}{2\beta^{2}}
      \Bigl(\norm{\nabla f(x_t)}^{2}
            +\frac{\sigma^{2}}{c_t^{2}}\Bigr) \nonumber\\
&= f(x_t)-\Bigl(\eta_t-\frac{L\eta^{2}}{2\beta^{2}}\Bigr)\norm{\nabla f(x_t)}^{2}
   +\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}c_t^{2}} . \label{eq:aftervar}
\end{align}
\item[Step 7.] By Assumption \ref{ass:hyper} we enforce $\eta\le 1/L$; together with $\eta_t\le \eta/\beta$, we obtain
\[
\eta_t-\frac{L\eta^{2}}{2\beta^{2}}\ge \frac{\eta}{2\beta},
\]
because $L\eta\le 1$ implies $L\eta^{2}\le \eta$.
\item[Step 8.] Using the bound from Step 7 in \eqref{eq:aftervar}:
\begin{align}
\E\!\left[f(x_{t+1})\mid\mathcal{F}_t\right]
&\le f(x_t)-\frac{\eta}{2\beta}\norm{\nabla f(x_t)}^{2}
   +\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}c_t^{2}} . \label{eq:final-rec}
\end{align}
\item[Step 9.] Take full expectation and sum over $t=0,\dots,T-1$:
\begin{align}
\E[f(x_T)] &\le f(x_0)-\frac{\eta}{2\beta}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
            +\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}}\sum_{t=0}^{T-1}\frac{1}{c_t^{2}} . \label{eq:sum}
\end{align}
\item[Step 10.] Since $c_t=1-(1-\alpha)^{t+1}\ge \alpha$ for all $t\ge0$, we have $1/c_t^{2}\le 1/\alpha^{2}$. Hence
\[
\sum_{t=0}^{T-1}\frac{1}{c_t^{2}}\le \frac{T}{\alpha^{2}} .
\]
Insert into \eqref{eq:sum} and rearrange:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
&\le \frac{2\beta\big(f(x_0)-\E[f(x_T)]\big)}{\eta T}
   +\frac{L\eta\sigma^{2}}{\beta\alpha^{2}} . \label{eq:avg}
\end{align}
Since $f(x_T)\ge f^\star$, we replace $\E[f(x_T)]$ by $f^\star$ to obtain the clean bound
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right]
&\le \frac{2\beta\big(f(x_0)-f^\star\big)}{\eta T}
   +\frac{L\eta\sigma^{2}}{\beta\alpha^{2}} . \tag{*}
\end{align}
\item[Step 11.] Choose a uniformly random index $I\in\{0,\dots,T-1\}$ independent of the algorithm. By definition,
\[
\E\!\left[\norm{\nabla f(x_I)}^{2}\right]=\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(x_t)}^{2}\right].
\]
Thus, (*) yields exactly the bound \eqref{eq:1} stated in Theorem \ref{thm:main}.
\item[Step 12.] Optimising the RHS of \eqref{eq:1} with respect to $\eta$ (treating $\eta$ as a free scalar) gives $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$. Substituting this choice back yields the $\mathcal{O}(1/\sqrt{T})$ rate.
\end{enumerate}
The proof is complete. $\square$

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main} and the choice $\eta = \frac{\beta}{\sqrt{L\sigma^{2}T}}$ (which respects $\eta\le 1/L$ for $T$ large enough) we obtain
\[
\E\!\left[\norm{\nabla f(x_I)}^{2}\right]
\le
\underbrace{\frac{2\big(f(x_0)-f^\star\big)}{\beta}}_{C_1}
\frac{\sqrt{L\sigma^{2}}}{\sqrt{T}}
\;+\;
\underbrace{\frac{L\sigma^{2}}{\beta^{2}}}_{C_2}\frac{\beta}{\sqrt{L\sigma^{2}T}}
=
\big(C_1+C_2\big)\frac{1}{\sqrt{T}} .
\]
Hence the explicit constant is $C_1+C_2= \frac{2\big(f(x_0)-f^\star\big)}{\beta}+ \frac{L\sigma^{2}}{\beta^{2}}$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Exact gradients ($\sigma=0$).} The bound reduces to
\[
\E\!\left[\norm{\nabla f(x_I)}^{2}\right]\le\frac{2\beta\big(f(x_0)-f^\star\big)}{\eta T},
\]
i.e. a deterministic $\mathcal{O}(1/T)$ rate.
\item \textbf{Full bias correction ($\alpha=1$).} Then $c_t=1$ and the variance term becomes $\frac{L\eta\sigma^{2}}{\beta^{2}}$, matching the classic SGD bound with adaptive stepsize.
\item \textbf{Very small $\alpha$ (strong smoothing).} $c_t\approx \alpha(t+1)$ for early iterations, leading to a milder variance term $\propto 1/(\alpha^{2}t^{2})$, which can improve practical performance when gradients are noisy.
\end{enumerate}
