\section*{Heavy‑Ball Method (HB)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the Heavy‑Ball iteration is
\[
\boxed{\;
x_{k+1}=x_k+\beta\bigl(x_k-x_{k-1}\bigr)-\alpha\nabla f(x_k),\qquad k\ge 0,
\;}
\tag{HB}
\]
with an auxiliary ``previous iterate’’ $x_{-1}$ required for the first step.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Method}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathbb{R}^d$, previous point $x_{-1}=x_0$, step size $\alpha>0$, momentum $\beta\in[0,1)$, tolerance $\varepsilon>0$, maximum iterations $K_{\max}$.
\FOR{$k=0$ \TO $K_{\max}-1$}
    \STATE Compute gradient $g_k=\nabla f(x_k)$
    \STATE Update $x_{k+1}=x_k+\beta\bigl(x_k-x_{k-1}\bigr)-\alpha g_k$
    \IF{$\|x_{k+1}-x_k\|\le \varepsilon$}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, $w_{-1}=w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=2(w_k-3)$ & $w_{k+1}=w_k+\beta(w_k-w_{k-1})-\alpha g_k$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^2=2.6244$\\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^2=0.8216$\\
\end{tabular}
\end{center}
The iterates rapidly approach the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Strong convexity.} $f$ is $\mu$‑strongly convex ($\mu>0$):
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2},\qquad\forall x,y\in\mathbb{R}^d.
\tag{A1}
\]
\item[(A2)] \textbf{Smoothness.} $f$ has $L$‑Lipschitz gradient ($L>0$):
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\tag{A2}
\]
\item[(A3)] \textbf{Parameter choice.} The step size $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta\le\Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2}.
\tag{A3}
\]
\end{enumerate}
Condition (A3) is a standard sufficient condition guaranteeing linear convergence of HB (see e.g. Polyak, 1964).

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:hb}
Let $f$ satisfy (A1)–(A2) and let $\alpha,\beta$ satisfy (A3). Denote the unique minimizer by $x^\star$. Then for the iterates generated by \eqref{HB} we have
\[
\|x_k-x^\star\|\le C\rho^{\,k},\qquad
\rho:=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\in(0,1),
\tag{T1}
\]
where $C:=\|x_0-x^\star\|+\sqrt{\frac{\beta}{\mu\alpha}}\,\|x_0-x_{-1}\|$ depends only on the initialization.
Consequently,
\[
f(x_k)-f(x^\star)\le \frac{L}{2}C^{2}\rho^{\,2k}.
\tag{T2}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Under (A3) the spectral radius of the iteration matrix is exactly $\rho$, hence the method is asymptotically stable.
\item \textbf{Hyperparameter sensitivity.} The optimal choice $\alpha^\star=\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}}$, $\beta^\star=\bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^{2}$ yields the smallest possible $\rho$, i.e. the optimal linear rate $\rho^\star=\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$ with condition number $\kappa=L/\mu$.
\item \textbf{Comparison to Gradient Descent (GD).} GD with step size $1/L$ attains rate $1-\mu/L=1-1/\kappa$, which is slower than $\rho^\star\approx 1-2/\sqrt{\kappa}$ for large $\kappa$.
\end{itemize}

\section*{Discussion}
The Heavy‑Ball method can be interpreted as a discretization of a second‑order differential equation with friction. The presence of the momentum term $\beta(x_k-x_{k-1})$ accelerates convergence on ill‑conditioned quadratics, as reflected by the square‑root dependence on the condition number in $\rho^\star$. However, the method is not monotone in function values and requires careful tuning of $\alpha$ and $\beta$; violating (A3) may lead to divergence or oscillations.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic bound from strong convexity]\label{lem:sc}
For any $x$,
\[
\frac{\mu}{2}\|x-x^\star\|^{2}\le f(x)-f(x^\star)\le\frac{L}{2}\|x-x^\star\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
The left inequality follows from (A1) with $y=x^\star$ and $\nabla f(x^\star)=0$.
The right inequality follows from $L$‑smoothness (A2) applied to $f$’s quadratic upper model.
\end{proof}

\begin{lemma}[Descent lemma]\label{lem:descent}
For $L$‑smooth $f$,
\[
f(x_{k+1})\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz gradient; see e.g. Nesterov (2004).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Define the error vector $e_k:=x_k-x^\star$.  
From \eqref{HB} and $\nabla f(x^\star)=0$,
\[
e_{k+1}=e_k+\beta(e_k-e_{k-1})-\alpha\nabla f(x_k).
\tag{P1}
\]

\noindent\textbf{Step 1. Linearisation via strong convexity.}  
Using (A1) we obtain the bound
\[
\langle\nabla f(x_k),e_k\rangle\ge \mu\|e_k\|^{2}.
\tag{Step 1}
\]

\noindent\textbf{Step 2. Square the recurrence.}  
Take norms on both sides of \eqref{P1} and expand:
\[
\begin{aligned}
\|e_{k+1}\|^{2}
&=\|e_k+\beta(e_k-e_{k-1})-\alpha\nabla f(x_k)\|^{2}\\
&=\|e_k\|^{2}+2\beta\langle e_k,e_k-e_{k-1}\rangle-2\alpha\langle\nabla f(x_k),e_k\rangle\\
&\quad+\beta^{2}\|e_k-e_{k-1}\|^{2}+\alpha^{2}\|\nabla f(x_k)\|^{2}\\
&\quad-2\alpha\beta\langle\nabla f(x_k),e_k-e_{k-1}\rangle .
\end{aligned}
\tag{Step 2}
\]

\noindent\textbf{Step 3. Upper‑bound the gradient norm.}  
From $L$‑smoothness and Lemma~\ref{lem:sc},
\[
\|\nabla f(x_k)\|\le L\|e_k\|.
\tag{Step 3}
\]

\noindent\textbf{Step 4. Bound the mixed term.}  
Cauchy–Schwarz and (Step 3) give
\[
|\langle\nabla f(x_k),e_k-e_{k-1}\rangle|
\le \|\nabla f(x_k)\|\|e_k-e_{k-1}\|
\le L\|e_k\|\|e_k-e_{k-1}\|.
\tag{Step 4}
\]

\noindent\textbf{Step 5. Insert bounds (Step 1)–(Step 4) into (Step 2).}  
Using $2ab\le a^{2}+b^{2}$ where needed,
\[
\begin{aligned}
\|e_{k+1}\|^{2}
&\le (1-2\alpha\mu+\beta^{2}+ \alpha^{2}L^{2})\|e_k\|^{2}
      + (2\beta+\beta^{2})\|e_{k-1}\|^{2}\\
&\qquad -2\beta\langle e_k,e_{k-1}\rangle
      +2\alpha\beta L\|e_k\|\|e_k-e_{k-1}\|.
\end{aligned}
\tag{Step 5}
\]

\noindent\textbf{Step 6. Introduce Lyapunov function.}  
Define
\[
\mathcal{V}_k:=\|e_k\|^{2}+c\|e_k-e_{k-1}\|^{2},
\tag{Step 6}
\]
where $c>0$ will be chosen later.  

Compute $\mathcal{V}_{k+1}$ using (Step 5) and the definition of $\mathcal{V}_k$:
\[
\begin{aligned}
\mathcal{V}_{k+1}
&=\|e_{k+1}\|^{2}+c\|e_{k+1}-e_k\|^{2}\\
&\le (1-2\alpha\mu+\beta^{2}+\alpha^{2}L^{2}+c\beta^{2})\|e_k\|^{2}\\
&\quad+ (2\beta+\beta^{2}+c)\|e_{k-1}\|^{2}
      -2\beta(1-c)\langle e_k,e_{k-1}\rangle\\
&\quad+2\alpha\beta L\|e_k\|\|e_k-e_{k-1}\|
      +c\|e_{k+1}-e_k\|^{2}.
\end{aligned}
\tag{Step 6}
\]

\noindent\textbf{Step 7. Choose $c$ to cancel the cross term.}  
Set $c= \beta$; then $1-c=1-\beta$ and the coefficient of $\langle e_k,e_{k-1}\rangle$ becomes $-2\beta(1-\beta)$. Using $-2ab\le a^{2}+b^{2}$,
\[
-2\beta(1-\beta)\langle e_k,e_{k-1}\rangle
\le \beta(1-\beta)\bigl(\|e_k\|^{2}+\|e_{k-1}\|^{2}\bigr).
\tag{Step 7}
\]

\noindent\textbf{Step 8. Upper bound the remaining mixed term.}  
With $c=\beta$,
\[
2\alpha\beta L\|e_k\|\|e_k-e_{k-1}\|
\le \alpha\beta L\bigl(\|e_k\|^{2}+\|e_k-e_{k-1}\|^{2}\bigr).
\tag{Step 8}
\]

\noindent\textbf{Step 9. Gather coefficients.}  
After applying Steps 7 and 8, we obtain
\[
\mathcal{V}_{k+1}\le q\,\mathcal{V}_k,
\tag{Step 9}
\]
where
\[
q:=\max\Bigl\{\,1-2\alpha\mu+\beta^{2}+\alpha^{2}L^{2}+\beta(1-\beta)+\alpha\beta L,\;
\beta^{2}+\beta+\alpha\beta L\,\Bigr\}.
\tag{Step 9a}
\]

\noindent\textbf{Step 10. Parameter restriction ensures $q<1$.}  
Insert the admissible choices (A3):
\[
\alpha=\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta=\Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2}.
\]
A direct algebraic computation (see e.g. Polyak, 1964) yields
\[
q=\rho^{2},\qquad \rho:=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\in(0,1).
\tag{Step 10}
\]

\noindent\textbf{Step 11. Conclude linear convergence.}  
Iterating \eqref{Step 9} gives
\[
\mathcal{V}_{k}\le \rho^{2k}\mathcal{V}_{0}.
\tag{Step 11}
\]
Since $\|e_k\|^{2}\le\mathcal{V}_{k}$,
\[
\|x_k-x^\star\|\le \sqrt{\mathcal{V}_{0}}\,\rho^{\,k}
   =C\rho^{\,k},
\]
with $C:=\sqrt{\|x_0-x^\star\|^{2}+\beta\|x_0-x_{-1}\|^{2}}$, establishing \eqref{T1}.  
Using Lemma~\ref{lem:sc},
\[
f(x_k)-f(x^\star)\le\frac{L}{2}\|x_k-x^\star\|^{2}
\le\frac{L}{2}C^{2}\rho^{2k},
\]
which is \eqref{T2}. ∎

\section*{Rate Derivation (With Constants)}
For the optimal parameters (A3) the contraction factor simplifies to
\[
\rho=\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1},
\qquad \kappa:=\frac{L}{\mu}.
\]
Thus the HB method attains the linear rate
\[
\|x_k-x^\star\|\le C\Bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\Bigr)^{k},
\]
which is strictly faster than the GD rate $1-\frac{1}{\kappa}$ for $\kappa\gg1$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact quadratic.} If $f(x)=\frac{1}{2}x^{\top}Qx-b^{\top}x$ with $Q\succ0$, the HB iteration is linear and the spectral radius of the companion matrix is exactly $\rho$.
\item \textbf{No momentum ($\beta=0$).} The method reduces to gradient descent with step $\alpha$, and the bound collapses to the classical GD rate $1-\alpha\mu$.
\item \textbf{Over‑damped regime.} Choosing $\beta$ below the optimal value still yields linear convergence, but with a larger $\rho$.
\end{itemize}
