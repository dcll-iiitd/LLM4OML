\section{Problem Setting and Assumptions}
We consider the unconstrained optimization problem:
\begin{equation}
    \min_{x \in \mathbb{R}^d} f(x) \quad \text{where} \quad f(x) = \mathbb{E}_{\xi}[F(x; \xi)].
\end{equation}
We assume access to a stochastic oracle that returns unbiased gradient estimates with bounded variance.

\begin{assumption}[Lipschitz Gradient] \label{assump:lipschitz}
    The function $f$ is $L$-smooth: for all $x, y \in \mathbb{R}^d$,
    \begin{equation}
        \|\nabla f(x) - \nabla f(y)\| \le L \|x - y\|.
    \end{equation}
    Equivalently, $f(y) \le f(x) + \langle \nabla f(x), y-x \rangle + \frac{L}{2}\|y-x\|^2$.
\end{assumption}

\begin{assumption}[Unbiased Gradient with Bounded Variance] \label{assump:variance}
    The stochastic gradient $g(x; \xi)$ satisfies $\mathbb{E}_{\xi}[g(x; \xi)] = \nabla f(x)$ and
    \begin{equation}
        \mathbb{E}_{\xi}[\|g(x; \xi) - \nabla f(x)\|^2] \le \sigma^2 \quad \forall x.
    \end{equation}
\end{assumption}

\begin{assumption}[Lower Bounded Objective] \label{assump:lower}
    The objective function is bounded below: $f^* = \inf_x f(x) > -\infty$.
\end{assumption}

\section{Algorithm}
We analyze the standard Stochastic Gradient Descent (SGD) iteration:
\begin{equation}
    x_{k+1} = x_k - \eta_k g(x_k; \xi_k), \quad k = 0,1,\dots,T-1
\end{equation}
where $\eta_k > 0$ is the step size (learning rate) and $\xi_k$ are i.i.d. samples.

\section{Main Result}
\begin{theorem}[Convergence Rate of SGD] \label{thm:main}
    Under Assumptions \ref{assump:lipschitz}, \ref{assump:variance}, \ref{assump:lower}, if we choose a constant step size $\eta_k = \eta = \frac{1}{L\sqrt{T}}$ for $T \ge 1$, then the output of SGD after $T$ iterations satisfies
    \begin{equation}
        \frac{1}{T}\sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] \le \frac{2L(f(x_0)-f^*)}{\sqrt{T}} + \frac{L\sigma^2}{\sqrt{T}}.
    \end{equation}
    In particular, $\min_{0\le k < T} \mathbb{E}[\|\nabla f(x_k)\|^2] \le \mathcal{O}(1/\sqrt{T})$.
\end{theorem}

\section{Proof of Theorem \ref{thm:main}}

\begin{proof}
    We proceed by analyzing the one-step progress using smoothness.

    \noindent \textbf{[Step 1]} (Smoothness inequality) By Assumption \ref{assump:lipschitz}, for any $x_k$ and $x_{k+1} = x_k - \eta g_k$ where $g_k = g(x_k; \xi_k)$,
    \begin{equation}
        f(x_{k+1}) \le f(x_k) + \langle \nabla f(x_k), x_{k+1}-x_k \rangle + \frac{L}{2}\|x_{k+1}-x_k\|^2.
    \end{equation}
    Substituting the update rule:
    \begin{equation}
        f(x_{k+1}) \le f(x_k) - \eta \langle \nabla f(x_k), g_k \rangle + \frac{L\eta^2}{2}\|g_k\|^2. \tag{1}
    \end{equation}

    \noindent \textbf{[Step 2]} (Take expectation conditioned on $x_k$) Since $\mathbb{E}[g_k | x_k] = \nabla f(x_k)$ and $\mathbb{E}[\|g_k\|^2 | x_k] = \|\nabla f(x_k)\|^2 + \mathbb{E}[\|g_k - \nabla f(x_k)\|^2 | x_k] \le \|\nabla f(x_k)\|^2 + \sigma^2$ (by Assumption \ref{assump:variance}), we have
    \begin{align}
        \mathbb{E}[f(x_{k+1}) | x_k] &\le f(x_k) - \eta \|\nabla f(x_k)\|^2 + \frac{L\eta^2}{2}\left(\|\nabla f(x_k)\|^2 + \sigma^2\right) \\
        &= f(x_k) - \eta\left(1 - \frac{L\eta}{2}\right)\|\nabla f(x_k)\|^2 + \frac{L\eta^2\sigma^2}{2}. \tag{2}
    \end{align}

    \noindent \textbf{[Step 3]} (Choose step size to ensure positive coefficient) We require $1 - \frac{L\eta}{2} \ge \frac{1}{2}$, i.e., $\eta \le \frac{1}{L}$. With our choice $\eta = \frac{1}{L\sqrt{T}}$, this holds for all $T \ge 1$. Then
    \begin{equation}
        \mathbb{E}[f(x_{k+1}) | x_k] \le f(x_k) - \frac{\eta}{2}\|\nabla f(x_k)\|^2 + \frac{L\eta^2\sigma^2}{2}. \tag{3}
    \end{equation}

    \noindent \textbf{[Step 4]} (Telescoping sum) Take total expectation and sum from $k=0$ to $T-1$:
    \begin{equation}
        \mathbb{E}[f(x_T)] \le f(x_0) - \frac{\eta}{2}\sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] + \frac{T L\eta^2\sigma^2}{2}. \tag{4}
    \end{equation}

    \noindent \textbf{[Step 5]} (Lower bound and rearrangement) Since $f(x_T) \ge f^*$ by Assumption \ref{assump:lower}, we have
    \begin{equation}
        \frac{\eta}{2}\sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] \le f(x_0) - f^* + \frac{T L\eta^2\sigma^2}{2}. \tag{5}
    \end{equation}

    \noindent \textbf{[Step 6]} (Substitute $\eta = \frac{1}{L\sqrt{T}}$) Plugging in the step size:
    \begin{align}
        \frac{1}{2L\sqrt{T}}\sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] &\le f(x_0)-f^* + \frac{T L}{2}\cdot\frac{1}{L^2 T}\sigma^2 \\
        &= f(x_0)-f^* + \frac{\sigma^2}{2L}. \tag{6}
    \end{align}

    \noindent \textbf{[Step 7]} (Final rate) Multiply both sides by $2L\sqrt{T}$:
    \begin{equation}
        \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] \le 2L\sqrt{T}(f(x_0)-f^*) + \sigma^2\sqrt{T}. \tag{7}
    \end{equation}
    Dividing by $T$ yields the claimed bound:
    \begin{equation}
        \frac{1}{T}\sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(x_k)\|^2] \le \frac{2L(f(x_0)-f^*)}{\sqrt{T}} + \frac{L\sigma^2}{\sqrt{T}}. \tag{8}
    \end{equation}
    The minimum over $k$ is trivially bounded by the average, completing the proof.
\end{proof}

\section{Corollary: Convergence in Expectation}
\begin{corollary}
    Under the same conditions, if we pick $\hat{x}$ uniformly at random from $\{x_0,\dots,x_{T-1}\}$, then
    \begin{equation}
        \mathbb{E}[\|\nabla f(\hat{x})\|^2] \le \frac{2L(f(x_0)-f^*)}{\sqrt{T}} + \frac{L\sigma^2}{\sqrt{T}}.
    \end{equation}
\end{corollary}
