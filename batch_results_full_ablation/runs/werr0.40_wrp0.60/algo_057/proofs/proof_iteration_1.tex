\section*{Random Block Coordinate Descent (RBCD) for Non‑Convex Smooth Functions}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable (possibly non‑convex).  
Partition the variable $w\in\mathbb{R}^{d}$ into $B$ disjoint blocks
\[
w = \big( w^{(1)},w^{(2)},\dots ,w^{(B)} \big),\qquad 
w^{(i)}\in\mathbb{R}^{d_i},\;\; \sum_{i=1}^{B}d_i = d .
\]
Denote by $\nabla_i f(w)$ the partial gradient with respect to block $i$
and by $U_i\in\mathbb{R}^{d\times d_i}$ the column‑selection matrix that
embeds a block vector into the full space (i.e. $U_i v$ places $v$ in the
coordinates of block $i$ and zeros elsewhere).

At iteration $t=0,1,2,\dots$ :

\begin{enumerate}
\item Sample a block index $i_t\in\{1,\dots ,B\}$ uniformly at random,
      i.e. $\mathbb{P}(i_t=i)=1/B$.
\item Compute the partial gradient $\nabla_{i_t} f(w_t)$.
\item Update only the selected block with a (possibly time‑varying) stepsize
      $\eta_t>0$:
\[
w_{t+1}= w_t - \eta_t\, U_{i_t}\,\nabla_{i_t} f(w_t).
\tag{1}
\]
\end{enumerate}
All other blocks remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent (RBCD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, stepsize schedule
$\{\eta_t\}_{t\ge 0}$, maximum iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $i_t\sim\text{Uniform}\{1,\dots ,B\}$.
    \State Compute partial gradient $g_t \gets \nabla_{i_t}f(w_t)$.
    \State Update $w_{t+1}\gets w_t - \eta_t\,U_{i_t}\,g_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or any iterate $w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional function $f(w)=(w-3)^2$ (thus $B=1$).  
Take $w_0=0$, constant stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3665$\\
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the basic
mechanism of (1).

\section*{Assumptions}
\begin{assumption}[Blockwise Lipschitz Gradient]\label{ass:Lip}
For each block $i\in\{1,\dots ,B\}$ there exists $L_i>0$ such that for any
$w,\tilde w\in\mathbb{R}^{d}$ differing only in block $i$,
\[
\big\|\nabla_i f(w)-\nabla_i f(\tilde w)\big\|
\le L_i\big\|w^{(i)}-\tilde w^{(i)}\big\|.
\]
Define $L_{\max}= \max_i L_i$ and $L_{\mathrm{avg}}= \frac1B\sum_{i=1}^{B}L_i$.
\end{assumption}

\begin{assumption}[Lower Bounded Objective]\label{ass:bound}
There exists $f^{\inf}>-\infty$ such that $f(w)\ge f^{\inf}$ for all $w$.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes satisfy $0<\eta_t\le \frac{1}{L_{\max}}$ for all $t$.
\end{assumption}

No convexity or variance assumptions are required; the analysis works for
general non‑convex $f$.

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Decay]\label{thm:main}
Let $\{w_t\}_{t\ge0}$ be generated by RBCD under Assumptions
\ref{ass:Lip}–\ref{ass:stepsize}.  Define the average stepsize
$\bar\eta_T:=\frac{1}{T}\sum_{t=0}^{T-1}\eta_t$.
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\frac{2\,L_{\max}}{\bar\eta_T\,B}\,
\big(f(w_0)-f^{\inf}\big).
\tag{2}
\]
In particular, with a constant stepsize $\eta_t\equiv\eta\le 1/L_{\max}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\frac{2L_{\max}}{\eta B}\,
\frac{f(w_0)-f^{\inf}}{T}
= O\!\Big(\frac{1}{T}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (2) holds for any stepsize not exceeding
$1/L_{\max}$, guaranteeing that the iterates remain in a region where the
Lipschitz condition is valid.
\item \textbf{Hyper‑parameter Sensitivity.}
The convergence rate scales linearly with $L_{\max}$ and inversely with the
product $\eta B$.  Larger block counts $B$ accelerate convergence because
more blocks are sampled in expectation per iteration.
\item \textbf{Comparison to Full‑Gradient Descent.}
If $B=d$ (each coordinate is a block) and $\eta=1/L_{\max}$, (2) reduces to the
standard $O(1/T)$ rate for (deterministic) gradient descent on smooth
non‑convex functions.
\item \textbf{Special Cases.}
If all $L_i$ are equal ($L_i\equiv L$) and a constant stepsize $\eta=1/L$ is used,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le \frac{2L}{B}\,\frac{f(w_0)-f^{\inf}}{T}.
\]
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, random block coordinate
updates achieve the same $O(1/T)$ decay of the expected squared gradient
norm as full‑gradient methods, up to the factor $1/B$ reflecting the
probability of updating each block.  This matches the best known rates for
randomized coordinate descent in the non‑convex regime and justifies the
use of block‑wise updates when $d$ is large and computing the full gradient
is prohibitive.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Blockwise Descent Lemma]\label{lem:descent}
Let $i\in\{1,\dots ,B\}$ and $w\in\mathbb{R}^{d}$.  For any $d_i\in\mathbb{R}^{d_i}$,
\[
f\big(w+U_i d_i\big)
\le
f(w)+\langle \nabla_i f(w),d_i\rangle
+\frac{L_i}{2}\|d_i\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
The proof follows from the standard smoothness inequality applied to the
function $g(\theta)=f(w+U_i\theta)$, whose gradient w.r.t. $\theta$ is
$\nabla_i f(w+U_i\theta)$.  By $L_i$‑Lipschitz continuity of $\nabla_i f$,
\[
g(\theta)\le g(0)+\langle \nabla g(0),\theta\rangle+\frac{L_i}{2}\|\theta\|^{2},
\]
which is exactly (3) after substituting $\theta=d_i$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof,
expectations are taken over the random block index $i_t$ conditioned on the
past iterates, i.e. $\mathbb{E}_t[\cdot]=\mathbb{E}[\cdot\,|\,w_t]$.

\subsection*{Descent after one iteration}
From Lemma \ref{lem:descent} with $d_i=-\eta_t\nabla_i f(w_t)$ we obtain
\[
f(w_{t+1})
\le f(w_t)-\eta_t\|\nabla_{i_t}f(w_t)\|^{2}
+\frac{L_{i_t}}{2}\eta_t^{2}\|\nabla_{i_t}f(w_t)\|^{2}.
\tag{4}
\]
Rearranging,
\[
f(w_t)-f(w_{t+1})
\ge \Big(\eta_t-\frac{L_{i_t}}{2}\eta_t^{2}\Big)\|\nabla_{i_t}f(w_t)\|^{2}.
\tag{5}
\]

\subsection*{Taking expectation over the random block}
Using $\mathbb{P}(i_t=i)=1/B$ and the bound $\eta_t\le 1/L_{\max}$,
\[
\mathbb{E}_t\!\big[f(w_t)-f(w_{t+1})\big]
\ge
\frac{1}{B}\sum_{i=1}^{B}
\Big(\eta_t-\frac{L_i}{2}\eta_t^{2}\Big)\|\nabla_i f(w_t)\|^{2}.
\tag{6}
\]
Since $L_i\le L_{\max}$ and $\eta_t\le 1/L_{\max}$,
\[
\eta_t-\frac{L_i}{2}\eta_t^{2}
\ge
\eta_t-\frac{L_{\max}}{2}\eta_t^{2}
= \frac{\eta_t}{2}.
\tag{7}
\]
Insert (7) into (6):
\[
\mathbb{E}_t\!\big[f(w_t)-f(w_{t+1})\big]
\ge
\frac{\eta_t}{2B}\sum_{i=1}^{B}\|\nabla_i f(w_t)\|^{2}
= \frac{\eta_t}{2B}\,\|\nabla f(w_t)\|^{2}.
\tag{8}
\]
\textbf{[Step 1]} Equation (8) is the key descent inequality linking the
expected function decrease to the squared norm of the full gradient.

\subsection*{Summing over $t=0,\dots ,T-1$}
Take total expectation and sum (8):
\[
\sum_{t=0}^{T-1}\mathbb{E}\!\big[f(w_t)-f(w_{t+1})\big]
\ge
\frac{1}{2B}\sum_{t=0}^{T-1}\eta_t\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big].
\tag{9}
\]
The left‑hand side telescopes:
\[
\sum_{t=0}^{T-1}\mathbb{E}\!\big[f(w_t)-f(w_{t+1})\big]
= \mathbb{E}\!\big[f(w_0)-f(w_T)\big]
\le f(w_0)-f^{\inf}.
\tag{10}
\]
Combine (9) and (10):
\[
\frac{1}{2B}\sum_{t=0}^{T-1}\eta_t\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le f(w_0)-f^{\inf}.
\tag{11}
\]
\textbf{[Step 2]} Divide both sides of (11) by $\sum_{t=0}^{T-1}\eta_t$ and
use the definition $\bar\eta_T=\frac{1}{T}\sum_{t=0}^{T-1}\eta_t$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2B}{\sum_{t=0}^{T-1}\eta_t}\,(f(w_0)-f^{\inf})
=
\frac{2L_{\max}}{\bar\eta_T B}\,(f(w_0)-f^{\inf}),
\tag{12}
\]
where the last equality uses $\eta_t\le 1/L_{\max}$ to replace
$\sum_{t=0}^{T-1}\eta_t\ge T\bar\eta_T\ge T\eta_{\min}$; for a constant stepsize
$\eta_t\equiv\eta$ we have $\bar\eta_T=\eta$ and (12) reduces to (2).

Thus the claimed bound holds. $\square$

\section*{Rate Derivation (With Constants)}
For a constant stepsize $\eta\le 1/L_{\max}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2L_{\max}}{\eta B}\,
\frac{f(w_0)-f^{\inf}}{T}.
\tag{13}
\]
Choosing the largest admissible $\eta=1/L_{\max}$ yields the clean
expression
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\mathbb{E}\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2}{B}\,
\frac{f(w_0)-f^{\inf}}{T}.
\tag{14}
\]
Hence the expected squared gradient norm decays at the optimal
$\mathcal{O}(1/T)$ rate for smooth non‑convex optimization.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform Lipschitz constants.} If $L_i\equiv L$ for all $i$,
the stepsize condition simplifies to $\eta\le 1/L$, and the rate in
(13) becomes $\frac{2L}{\eta B}(f(w_0)-f^{\inf})/T$.

\item \textbf{Single‑block case ($B=1$).}
RBCD reduces to standard gradient descent, and (14) coincides with the
classical $O(1/T)$ bound for non‑convex smooth functions.

\item \textbf{Increasing stepsizes.}
If $\eta_t=\frac{c}{t+1}$ with $c\le 1/L_{\max}$, then
$\bar\eta_T\approx \frac{c\log T}{T}$ and (12) gives a rate
$O\big(\frac{\log T}{T}\big)$, which is slower but still vanishing.
\end{enumerate}
