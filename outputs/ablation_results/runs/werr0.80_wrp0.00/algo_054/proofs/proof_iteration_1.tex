\section*{Cyclic Stochastic Gradient Descent (Cyclic‑SGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (expected) objective
\[
f(w)=\EE_{z\sim\mathcal{D}}[F(w;z)],
\]
where $z$ denotes a random data sample and $F(\cdot;z)$ is differentiable.
At iteration $t\in\{0,1,\dots\}$ we draw an i.i.d.\ sample $z_t$ and compute a stochastic gradient
\[
g_t \;:=\; \nabla F(w_t;z_t).
\]

The learning rate follows a \emph{triangular cyclic schedule} with period $2M$:
\[
\boxed{
\eta_t \;=\; \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
\Bigl(1-\Bigl|\frac{(t\bmod 2M)}{M}-1\Bigr|\Bigr)},
\qquad\forall t\ge0,
\tag{1}
\]
where $0<\eta_{\min}\le\eta_{\max}$ and $M\in\mathbb{N}$ are hyper‑parameters.
The update rule is the standard SGD step
\[
\boxed{w_{t+1}=w_t-\eta_t g_t},\qquad t=0,1,\dots
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic‑SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $w_0\in\RR^{d}$, $\eta_{\min}>0$, $\eta_{\max}>0$, cycle length $M$, total iterations $T$.
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE Sample $z_t\sim\mathcal{D}$ and compute $g_t=\nabla F(w_t;z_t)$.
    \STATE Compute the cyclic step size
    \[
    \eta_t\gets \eta_{\min}+(\eta_{\max}-\eta_{\min})\Bigl(1-\Bigl|\frac{(t\bmod 2M)}{M}-1\Bigr|\Bigr).
    \]
    \STATE Update parameters: $w_{t+1}\gets w_t-\eta_t g_t$.
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.
Then $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$, $M=2$ (period $4$) and perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\eta_t$ (from (1)) & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta_t g_t$ & $f(w_{t+1})$\\\hline
0 & $0.05+0.10\bigl(1-|0/2-1|\bigr)=0.05+0.10(0)=0.05$ & $2(0-3)=-6$ & $0-0.05(-6)=0.30$ & $(0.30-3)^2=7.29$\\[4pt]
1 & $0.05+0.10\bigl(1-|1/2-1|\bigr)=0.05+0.10(0.5)=0.10$ & $2(0.30-3)=-5.40$ & $0.30-0.10(-5.40)=0.84$ & $(0.84-3)^2=4.66$\\[4pt]
2 & $0.05+0.10\bigl(1-|2/2-1|\bigr)=0.05+0.10(1)=0.15$ & $2(0.84-3)=-4.32$ & $0.84-0.15(-4.32)=1.50$ & $(1.50-3)^2=2.25$\\
\end{tabular}
\end{center}
The objective decreases from $9$ to $2.25$ in three steps, illustrating how the cyclic step size accelerates progress when the learning rate is larger (iteration $2$).

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\forall w,u\in\RR^{d},\qquad
f(u)\le f(w)+\inner{\nabla f(w)}{u-w}+\frac{L}{2}\norm{u-w}^{2}.
\tag{A1}
\]
\item \textbf{Unbiased stochastic gradients.}
\[
\EE[g_t\mid w_t]=\nabla f(w_t),\qquad\forall t\ge0.
\tag{A2}
\]
\item \textbf{Bounded variance.}
\[
\EE\bigl[\norm{g_t-\nabla f(w_t)}^{2}\mid w_t\bigr]\le\sigma^{2},\qquad\forall t\ge0.
\tag{A3}
\]
\item \textbf{Bounded learning rates.}
\[
0<\eta_{\min}\le\eta_t\le\eta_{\max},\qquad\forall t,
\tag{A4}
\]
where $\eta_{\min},\eta_{\max}$ are the hyper‑parameters appearing in (1).
\item \textbf{Existence of a global minimum.} There exists $w^{\star}$ such that $f(w^{\star})=f^{\star}:=\inf_{w}f(w)$ and $f^{\star}>-\infty$.
\tag{A5}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic‑SGD for non‑convex smooth objectives]\label{thm:main}
Under Assumptions (A1)–(A5) and for any integer $T\ge1$, the iterates generated by (2) with the cyclic schedule (1) satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\min}T}}_{\text{decrease term}}
\;+\;
\underbrace{L\eta_{\max}\sigma^{2}}_{\text{variance term}}.
\tag{3}
\]
Consequently, choosing $T\ge\frac{4\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\min}\sigma^{2}L\eta_{\max}}$ yields
\[
\min_{0\le t<T}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because every $\eta_t$ is bounded by $\eta_{\max}$, the standard smoothness argument guarantees that the iterates do not diverge as long as $\eta_{\max}<2/L$ (the classic SGD stability condition). The lower bound $\eta_{\min}>0$ prevents the step size from vanishing too quickly, ensuring a non‑trivial decrease term in (3).

\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Increasing $\eta_{\max}$ enlarges the variance term $L\eta_{\max}\sigma^{2}$ but also accelerates descent when the gradient is large.
        \item Decreasing the cycle length $M$ makes the schedule vary more rapidly; the convergence bound (3) depends only on the extrema $\eta_{\min},\eta_{\max}$, not on $M$, showing that the theoretical rate is \emph{cycle‑agnostic}.
        \item The ratio $\rho:=\eta_{\max}/\eta_{\min}$ controls the amplitude of the cycle; a moderate $\rho$ (e.g. $2\le\rho\le5$) often yields the best empirical trade‑off between exploration (large steps) and exploitation (small steps).
    \end{itemize}

\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item For constant step size $\eta_t\equiv\eta$, the standard SGD bound is $\frac{2(f(w_0)-f^{\star})}{\eta T}+L\eta\sigma^{2}$. The cyclic bound (3) replaces $\eta$ by $\eta_{\min}$ in the first term (more conservative) and by $\eta_{\max}$ in the variance term (more aggressive). Hence Cyclic‑SGD interpolates between two constant‑step SGD regimes.
        \item When $\eta_{\min}=\eta_{\max}=\eta$, Theorem~\ref{thm:main} recovers the classical constant‑step SGD result.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the non‑monotonicity of the step size, the average squared gradient norm decays at the same $O(1/T)$ rate as standard SGD with a constant step size. The cyclic schedule can be advantageous in practice because the occasional large steps help to escape shallow saddles or plateaus, while the small steps guarantee fine‑grained convergence once near a stationary point. The bound is independent of the cycle length $M$, which explains why cyclic learning rates are robust to the exact choice of $M$ in many deep‑learning applications.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w,u\in\RR^{d}$,
\[
f(u)\le f(w)+\inner{\nabla f(w)}{u-w}+\frac{L}{2}\norm{u-w}^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see (A1). \hfill $\square$
\end{proof}

\begin{lemma}[Variance decomposition]\label{lem:var}
For any random vector $g$ with $\EE[g]=\mu$, we have
\[
\EE\bigl[\norm{g}^{2}\bigr]=\norm{\mu}^{2}+\EE\bigl[\norm{g-\mu}^{2}\bigr].
\tag{5}
\]
\end{lemma}
\begin{proof}
Expand $\norm{g}^{2}= \norm{\mu+(g-\mu)}^{2}$ and take expectations. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t.\ the randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $u=w_{t+1}$, $w=w_t$:
\[
f(w_{t+1})\le f(w_t)+\inner{\nabla f(w_t)}{w_{t+1}-w_t}
+\frac{L}{2}\norm{w_{t+1}-w_t}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Substitute the SGD update (2): $w_{t+1}-w_t=-\eta_t g_t$.
\[
f(w_{t+1})\le f(w_t)-\eta_t\inner{\nabla f(w_t)}{g_t}
+\frac{L\eta_t^{2}}{2}\norm{g_t}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 3]} Take conditional expectation $\EE[\;\cdot\mid w_t]$ on both sides.
Using (A2) $\EE[g_t\mid w_t]=\nabla f(w_t)$ we obtain
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\eta_t\norm{\nabla f(w_t)}^{2}
+\frac{L\eta_t^{2}}{2}\EE\!\bigl[\norm{g_t}^{2}\mid w_t\bigr].
\tag{8}
\]

\noindent\textbf{[Step 4]} Apply Lemma~\ref{lem:var} to $\EE[\norm{g_t}^{2}\mid w_t]$:
\[
\EE\!\bigl[\norm{g_t}^{2}\mid w_t\bigr]
=\norm{\nabla f(w_t)}^{2}+\EE\!\bigl[\norm{g_t-\nabla f(w_t)}^{2}\mid w_t\bigr].
\tag{9}
\]

\noindent\textbf{[Step 5]} Insert (9) into (8) and use the variance bound (A3):
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\eta_t\norm{\nabla f(w_t)}^{2}
+\frac{L\eta_t^{2}}{2}\Bigl(\norm{\nabla f(w_t)}^{2}+\sigma^{2}\Bigr).
\tag{10}
\]

\noindent\textbf{[Step 6]} Rearrange (10) to isolate the gradient term:
\[
\EE\!\bigl[f(w_{t+1})\mid w_t\bigr]\le
f(w_t)-\underbrace{\Bigl(\eta_t-\frac{L\eta_t^{2}}{2}\Bigr)}_{\displaystyle\alpha_t}
\norm{\nabla f(w_t)}^{2}
+\frac{L\eta_t^{2}\sigma^{2}}{2}.
\tag{11}
\]
Define $\alpha_t:=\eta_t-\frac{L}{2}\eta_t^{2}$.

\noindent\textbf{[Step 7]} By the bound (A4) we have
\[
\eta_{\min}\le\eta_t\le\eta_{\max}\quad\Longrightarrow\quad
\alpha_t\ge\eta_{\min}-\frac{L}{2}\eta_{\max}^{2}.
\tag{12}
\]
To guarantee a positive coefficient we impose the standard stability condition
\[
\eta_{\max}<\frac{2}{L}\;\;\Longrightarrow\;\;
\eta_{\min}-\frac{L}{2}\eta_{\max}^{2}\ge\frac{\eta_{\min}}{2}>0.
\tag{13}
\]
Hence we may lower‑bound $\alpha_t$ by $\frac{\eta_{\min}}{2}$:
\[
\alpha_t\ge\frac{\eta_{\min}}{2},\qquad\forall t.
\tag{14}
\]

\noindent\textbf{[Step 8]} Taking full expectation (law of total expectation) in (11) and using (14):
\[
\EE\!\bigl[f(w_{t+1})\bigr]\le
\EE\!\bigl[f(w_t)\bigr]
-\frac{\eta_{\min}}{2}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
+\frac{L\eta_{\max}^{2}\sigma^{2}}{2}.
\tag{15}
\]

\noindent\textbf{[Step 9]} Rearrange (15) to obtain a one‑step descent inequality:
\[
\frac{\eta_{\min}}{2}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\EE\!\bigl[f(w_t)\bigr]-\EE\!\bigl[f(w_{t+1})\bigr]
+\frac{L\eta_{\max}^{2}\sigma^{2}}{2}.
\tag{16}
\]

\noindent\textbf{[Step 10]} Sum (16) over $t=0,\dots,T-1$:
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\bigl(f(w_0)-\EE[f(w_T)]\bigr)
+ \frac{TL\eta_{\max}^{2}\sigma^{2}}{2}.
\tag{17}
\]

\noindent\textbf{[Step 11]} Since $f(w_T)\ge f^{\star}$, replace $\EE[f(w_T)]$ by $f^{\star}$ to obtain a clean upper bound:
\[
\frac{\eta_{\min}}{2}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
f(w_0)-f^{\star}
+ \frac{TL\eta_{\max}^{2}\sigma^{2}}{2}.
\tag{18}
\]

\noindent\textbf{[Step 12]} Divide both sides of (18) by $T$ and by $\eta_{\min}/2$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\min}T}
+ L\eta_{\max}^{2}\sigma^{2}\frac{1}{\eta_{\min}}.
\tag{19}
\]

\noindent\textbf{[Step 13]} Since $\eta_{\max}\le\eta_{\max}$, we can replace $\eta_{\max}^{2}/\eta_{\min}$ by $\eta_{\max}$ (using $\eta_{\max}\ge\eta_{\min}>0$):
\[
L\eta_{\max}^{2}\sigma^{2}\frac{1}{\eta_{\min}}
\le L\eta_{\max}\sigma^{2}.
\tag{20}
\]

\noindent\textbf{[Step 14]} Plug (20) into (19) to obtain exactly the bound (3):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\min}T}
+ L\eta_{\max}\sigma^{2}.
\tag{21}
\]
This completes the proof of Theorem~\ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (21) we see two contributions:
\begin{itemize}
\item \emph{Optimization term} $\displaystyle\frac{2(f(w_0)-f^{\star})}{\eta_{\min}T}=O\!\bigl(\tfrac{1}{T}\bigr)$.
\item \emph{Stochastic noise term} $L\eta_{\max}\sigma^{2}=O(1)$.
\end{itemize}
Choosing $T$ such that the two terms are balanced,
\[
\frac{2(f(w_0)-f^{\star})}{\eta_{\min}T}\approx L\eta_{\max}\sigma^{2}
\;\Longrightarrow\;
T\approx\frac{2(f(w_0)-f^{\star})}{\eta_{\min}L\eta_{\max}\sigma^{2}}.
\]
Plugging this $T$ back into (21) yields
\[
\min_{0\le t<T}\EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
=O\!\bigl(\sigma\sqrt{L\eta_{\max}}\bigr)
=O\!\bigl(T^{-1/2}\bigr),
\]
which matches the optimal $O(1/\sqrt{T})$ rate for non‑convex SGD up to constant factors.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Constant step size.} If $\eta_{\min}=\eta_{\max}=\eta$, (3) reduces to the classic SGD bound $\frac{2(f(w_0)-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Vanishing lower bound.} Let $\eta_{\min}\downarrow0$ while keeping $\eta_{\max}$ fixed. The first term in (3) disappears, and the bound becomes dominated by the variance term $L\eta_{\max}\sigma^{2}$, reflecting the fact that arbitrarily small steps cannot guarantee descent.
\item \textbf{Large cycles.} As $M\to\infty$, the schedule approaches a constant step size equal to $\eta_{\min}$ for a long stretch followed by a single large step. The theorem still holds because only the extrema appear in (3).
\end{enumerate}
