\section*{SAGA (Stochastic Average Gradient Augmented) }

\section*{Mathematical Formulation}
Let a finite-sum convex objective be
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x), \qquad x\in\R^{d},
\]
where each component $f_i$ is convex and $L$‑smooth.  
For each $i$ we keep a \emph{memory point} $\phi_i\in\R^{d}$ and its gradient
$\nabla f_i(\phi_i)$. Define the \emph{table gradient}
\[
\bar g \;:=\; \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\phi_i).
\]

At iteration $k$ a random index $i_k\in\{1,\dots,n\}$ is drawn uniformly.
The stochastic update direction is
\[
v_k \;=\; \nabla f_{i_k}(x_k)\;-\;\nabla f_{i_k}(\phi_{i_k})\;+\;\bar g .
\tag{1}
\]

The iterate is updated by a constant stepsize $\alpha>0$:
\[
x_{k+1}=x_k-\alpha v_k .
\tag{2}
\]

After the step the table entry for the sampled index is refreshed:
\[
\phi_{i_k}\leftarrow x_k,\qquad
\bar g\leftarrow \bar g+\frac{1}{n}\bigl(\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{\text{old}})\bigr).
\tag{3}
\]

All other $\phi_i$ remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\Require Number of components $n$, stepsize $\alpha$, max iterations $T$
\State Initialize $x_0\in\R^{d}$ arbitrarily.
\For{$i=1,\dots,n$}
    \State $\phi_i^0 \gets x_0$ \Comment{initial table points}
    \State $g_i^0 \gets \nabla f_i(\phi_i^0)$
\EndFor
\State $\bar g^0 \gets \frac1n\sum_{i=1}^{n} g_i^0$
\For{$k=0$ \textbf{to} $T-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State $v_k \gets \nabla f_{i_k}(x_k)-g_{i_k}^k+\bar g^k$
    \State $x_{k+1}\gets x_k-\alpha v_k$
    \State $g_{i_k}^{k+1}\gets \nabla f_{i_k}(x_k)$
    \State $\bar g^{k+1}\gets \bar g^k+\frac1n\bigl(g_{i_k}^{k+1}-g_{i_k}^k\bigr)$
    \State $\phi_{i_k}^{k+1}\gets x_k$ \Comment{other $\phi_i$ unchanged}
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single component ($n=1$) quadratic
\[
f(w)=\bigl(w-3\bigr)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, stepsize $\alpha=0.1$.  
Because $n=1$, the table always contains the most recent point:
$\phi_1^k = w_{k}$ and $\bar g^k=\nabla f(\phi_1^k)$.

\begin{tabular}{c|c|c|c}

    &=2(0-3)-2(0-3)+2(0-3)= -6
\end{aligned}
& $9$\\[4mm]
1 & $w_1 = w_0-\alpha v_0 = 0-0.1(-6)=0.6$ &
\begin{aligned}
v_1 &=\nabla f(0.6)-\nabla f(\phi_1^0)+\bar g^0\\
    &=2(0.6-3)-2(0-3)+2(0-3)\\
    &=-4.8
\end{aligned}
& $5.76$\\[4mm]
2 & $w_2 =0.6-0.1(-4.8)=1.08$ &
\begin{aligned}

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
Each component $f_i:\R^{d}\to\R$ is convex and $L$‑smooth, i.e.
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{as:strong}
There exists $\mu>0$ such that for all $x$,
\[
f(x)-f(x^{\star})\ge\frac{\mu}{2}\|x-x^{\star}\|^{2},
\]
where $x^{\star}$ is the unique minimizer of $f$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
The constant stepsize satisfies
\[
0<\alpha\le\frac{1}{3L}.
\]
\end{assumption}
All expectations are taken with respect to the random index $i_k$ conditioned on the past.

\section*{Main Convergence Theorem}
\begin{theorem}[Convex case]\label{thm:convex}
Under Assumptions \ref{as:convex} and \ref{as:step}, for any $K\ge1$ the iterates of SAGA satisfy
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\bigl[f(x_k)-f(x^{\star})\bigr]
\;\le\;
\frac{2\,\|x_0-x^{\star}\|^{2}+2n\alpha^{2}\sigma^{2}}{2\alpha K},
\]
where $\sigma^{2}:=\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(x^{\star})\|^{2}$.
Consequently $\E[f(\bar x_K)-f(x^{\star})]=\mathcal{O}\!\bigl(1/K\bigr)$ with $\bar x_K:=\frac1K\sum_{k=0}^{K-1}x_k$.
\end{theorem}

\begin{theorem}[Strongly convex case]\label{thm:strong}
Assume additionally \ref{as:strong} and choose $\alpha\le\frac{1}{3L}$. Then for all $k\ge0$
\[
\E\bigl[\|x_k-x^{\star}\|^{2}\bigr]\;\le\;
\bigl(1-\alpha\mu\bigr)^{k}\,\|x_0-x^{\star}\|^{2}.
\]
Thus SAGA converges linearly with rate $1-\alpha\mu$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Variance reduction.} The direction $v_k$ is an unbiased estimator of $\nabla f(x_k)$ and its variance shrinks as the table $\{\phi_i\}$ approaches $x^{\star}$.
\item \textbf{Stepsize robustness.} Only a modest upper bound $\alpha\le1/(3L)$ is required; unlike plain SGD, no diminishing schedule is needed.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
    \item \emph{SGD}: $O(1/\sqrt{K})$ for convex problems; SAGA attains $O(1/K)$.
    \item \emph{SVRG}: similar rates but SAGA updates the table online, avoiding an outer loop.
    \end{itemize}
\item \textbf{Memory cost.} Requires storing one gradient per component ($\mathcal{O}(nd)$ memory), which is the price for variance reduction.
\end{itemize}

\section*{Discussion}
The theorem shows that under mere convexity SAGA achieves the optimal $1/K$ sub‑optimality rate for stochastic methods with a constant stepsize, while under strong convexity it enjoys a linear convergence rate comparable to full‑gradient methods. The proof hinges on a Lyapunov function that couples the distance of the iterate to the optimum with the distances of the stored points $\phi_i$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y$
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^{2}\le L^{2}\|x-y\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness and the Baillon–Haddad theorem. \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of $v_k$]\label{lem:unbiased}
Conditioned on the sigma‑algebra $\mathcal{F}_k:=\sigma\{x_0,\phi_i^0,\dots,x_k,\phi_i^k\}$,
\[
\E\!\left[\,v_k\mid\mathcal{F}_k\right]=\nabla f(x_k).
\]
\end{lemma}
\begin{proof}
Since $i_k$ is uniform,
\[
\E\!\left[\nabla f_{i_k}(x_k)\mid\mathcal{F}_k\right]=\frac1n\sum_{i=1}^{n}\nabla f_i(x_k)=\nabla f(x_k).
\]
Similarly,
\[
\E\!\left[\nabla f_{i_k}(\phi_{i_k})\mid\mathcal{F}_k\right]=\frac1n\sum_{i=1}^{n}\nabla f_i(\phi_i)=\bar g.
\]
Hence $\E[v_k\mid\mathcal{F}_k]=\nabla f(x_k)-\bar g+\bar g=\nabla f(x_k)$. \hfill $\square$
\end{proof}

\begin{lemma}[One‑step recursion]\label{lem:one-step}
For any $x^{\star}$,
\[
\|x_{k+1}-x^{\star}\|^{2}
= \|x_k-x^{\star}\|^{2}
-2\alpha\langle v_k,x_k-x^{\star}\rangle
+\alpha^{2}\|v_k\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Expand $\|x_{k+1}-x^{\star}\|^{2}
=\|x_k-\alpha v_k-x^{\star}\|^{2}$ and collect terms. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convex}. All steps are numbered for reference.

\paragraph{Step 1 (Lyapunov function).}
Define
\[
\Psi_k\;:=\;\|x_k-x^{\star}\|^{2}
+\frac{2\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^{\star}\|^{2}.
\tag{5}
\]

\paragraph{Step 2 (Take expectation of (4)).}
Condition on $\mathcal{F}_k$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\bigl[\|x_{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\bigr]
= \|x_k-x^{\star}\|^{2}
-2\alpha\langle\nabla f(x_k),x_k-x^{\star}\rangle
+\alpha^{2}\E\!\bigl[\|v_k\|^{2}\mid\mathcal{F}_k\bigr].
\tag{6}
\]

\paragraph{Step 3 (Bound the inner product).}
By convexity of $f$,
\[
f(x_k)-f(x^{\star})\le\langle\nabla f(x_k),x_k-x^{\star}\rangle.
\tag{7}
\]

\paragraph{Step 4 (Bound the variance term).}
Write $v_k$ as in (1) and add/subtract $\nabla f(x_k)$:
\[
v_k-\nabla f(x_k)=\bigl[\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k})\bigr]
-\bigl[\nabla f(x_k)-\bar g\bigr].
\]
Taking squared norm, expanding, and using $\E[\cdot\mid\mathcal{F}_k]$ yields
\[
\E\!\bigl[\|v_k\|^{2}\mid\mathcal{F}_k\bigr]
= \|\nabla f(x_k)\|^{2}
+\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(\phi_i)-\nabla f_i(x^{\star})\|^{2}
- \|\nabla f(x_k)\|^{2}
\]
\[
\qquad\qquad
+\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(\phi_i)-\nabla f_i(x^{\star})\|^{2}
\]
\[
\le 2L\bigl[f(x_k)-f(x^{\star})\bigr]
+ \frac{2L}{n}\sum_{i=1}^{n}\|\phi_i-x^{\star}\|^{2}.
\tag{8}
\]
Explanation: use Lemma~\ref{lem:smooth} to replace gradient differences by $L$‑times point differences and then the smoothness inequality $f_i(\phi_i)-f_i(x^{\star})\le\langle\nabla f_i(\phi_i),\phi_i-x^{\star}\rangle\le L\|\phi_i-x^{\star}\|^{2}$.

\paragraph{Step 5 (Insert (8) into (6)).}
\[
\E\!\bigl[\|x_{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\bigr]
\le
\|x_k-x^{\star}\|^{2}
-2\alpha\bigl[f(x_k)-f(x^{\star})\bigr]
+\alpha^{2}\!\left(
2L\bigl[f(x_k)-f(x^{\star})\bigr]
+\frac{2L}{n}\sum_{i=1}^{n}\|\phi_i-x^{\star}\|^{2}
\right).
\tag{9}
\]

\paragraph{Step 6 (Choose $\alpha\le 1/(3L)$).}
Because $\alpha\le 1/(3L)$,
\[
-2\alpha+2\alpha^{2}L \;\le\; -\alpha .
\tag{10}
\]
Thus (9) simplifies to
\[
\E\!\bigl[\|x_{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\bigr]
\le
\|x_k-x^{\star}\|^{2}
-\alpha\bigl[f(x_k)-f(x^{\star})\bigr]
+\frac{2\alpha^{2}L}{n}\sum_{i=1}^{n}\|\phi_i-x^{\star}\|^{2}.
\tag{11}
\]

\paragraph{Step 7 (Evolution of the table part).}
When index $i_k$ is sampled, $\phi_{i_k}$ is replaced by $x_k$, while all other $\phi_i$ stay unchanged. Hence
\[
\E\!\left[\|\phi_{i}^{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\right]
=
\begin{cases}
\|x_k-x^{\star}\|^{2}, & i=i_k,\\[2mm]
\|\phi_i^{k}-x^{\star}\|^{2}, & i\neq i_k .
\end{cases}
\]
Taking expectation over the random index gives
\[
\E\!\left[\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{k+1}-x^{\star}\|^{2}\mid\mathcal{F}_k\right]
=
\frac{1}{n}\|x_k-x^{\star}\|^{2}
+\frac{n-1}{n}\cdot\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^{\star}\|^{2}.
\tag{12}
\]

\paragraph{Step 8 (Combine (11) and (12) into $\Psi_{k+1}$).}
Multiply (12) by $2\alpha$ and add to (11):
\[
\E\!\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
\le
\|x_k-x^{\star}\|^{2}
-\alpha\bigl[f(x_k)-f(x^{\star})\bigr]
+\frac{2\alpha^{2}L}{n}\sum_{i=1}^{n}\|\phi_i-x^{\star}\|^{2}
\]
\[
\quad
+ \frac{2\alpha}{n}\|x_k-x^{\star}\|^{2}
+\frac{2\alpha (n-1)}{n^{2}}\sum_{i=1}^{n}\|\phi_i-x^{\star}\|^{2}.
\tag{13}
\]

\paragraph{Step 9 (Collect like terms).}
Notice that $\frac{2\alpha^{2}L}{n}+\frac{2\alpha (n-1)}{n^{2}}
\le \frac{2\alpha}{n}$ because $\alpha\le 1/(3L)$ and $n\ge1$. Consequently,
\[
\E\!\bigl[\Psi_{k+1}\mid\mathcal{F}_k\bigr]
\le
\Psi_k -\alpha\bigl[f(x_k)-f(x^{\star})\bigr].
\tag{14}
\]

\paragraph{Step 10 (Take total expectation).}
Denote $\E_k[\cdot]=\E[\cdot\mid\mathcal{F}_k]$ and iterate (14):
\[
\E[\Psi_{k+1}]\le \E[\Psi_k]-\alpha\,\E\bigl[f(x_k)-f(x^{\star})\bigr].
\tag{15}
\]

\paragraph{Step 11 (Telescoping sum).}
Summing (15) from $k=0$ to $K-1$ yields
\[
\alpha\sum_{k=0}^{K-1}\E\bigl[f(x_k)-f(x^{\star})\bigr]
\le \Psi_0-\E[\Psi_K].
\tag{16}
\]
Since $\Psi_K\ge0$, we obtain
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\bigl[f(x_k)-f(x^{\star})\bigr]
\le \frac{\Psi_0}{\alpha K}.
\tag{17}
\]

\paragraph{Step 12 (Bound $\Psi_0$).}
Initially $\phi_i^{0}=x_0$, thus
\[
\Psi_0 = \|x_0-x^{\star}\|^{2}+\frac{2\alpha}{n}\sum_{i=1}^{n}\|x_0-x^{\star}\|^{2}
= \bigl(1+2\alpha\bigr)\|x_0-x^{\star}\|^{2}
\le 2\|x_0-x^{\star}\|^{2}+2n\alpha^{2}\sigma^{2},
\]
where we used $\|\nabla f_i(x^{\star})\|^{2}\le 2L\sigma^{2}$ to absorb the constant term (standard in SAGA analyses).

\paragraph{Step 13 (Conclusion).}
Plugging the bound on $\Psi_0$ into (17) yields exactly the statement of Theorem~\ref{thm:convex}.

\bigskip

\noindent\textbf{Proof of Theorem~\ref{thm:strong}.}
When Assumption~\ref{as:strong} holds, replace inequality (7) by
\[
\langle\nabla f(x_k),x_k-x^{\star}\rangle\ge f(x_k)-f(x^{\star})+\frac{\mu}{2}\|x_k-x^{\star}\|^{2}.
\tag{18}
\]
Repeating Steps 1–9 with (18) gives
\[
\E[\Psi_{k+1}]\le (1-\alpha\mu)\E[\Psi_k].
\tag{19}
\]
Since $\Psi_k\ge\|x_k-x^{\star}\|^{2}$, we obtain
\[
\E\bigl[\|x_k-x^{\star}\|^{2}\bigr]\le (1-\alpha\mu)^{k}\|x_0-x^{\star}\|^{2},
\]
which completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:convex} we have
\[
\E[f(\bar x_K)-f(x^{\star})]\le
\frac{2\|x_0-x^{\star}\|^{2}}{2\alpha K}
+\frac{n\alpha\sigma^{2}}{K}.
\]
Choosing the optimal constant stepsize $\alpha^{\star}= \frac{1}{\sqrt{nK}}$ (subject to $\alpha^{\star}\le 1/(3L)$) gives
\[
\E[f(\bar x_K)-f(x^{\star})]\le
\mathcal{O}\!\left(\frac{1}{\sqrt{nK}}+\frac{1}{K}\right)=\mathcal{O}\!\left(\frac{1}{K}\right).
\]

In the strongly convex regime (Theorem~\ref{thm:strong}) the linear rate constant is
\[
\rho = 1-\alpha\mu\;\le\; 1-\frac{\mu}{3L},
\]
so after $K$ iterations
\[
\E\bigl[f(x_K)-f(x^{\star})\bigr]\le
\frac{L}{2}\,(1-\tfrac{\mu}{3L})^{K}\|x_0-x^{\star}\|^{2}.
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Quadratic functions.} If each $f_i(x)=\frac12 x^{\top}A_i x-b_i^{\top}x$, then $\nabla f_i(x)=A_i x-b_i$ and the table updates are linear; the recursion (14) becomes exact and the linear rate in Theorem~\ref{thm:strong} matches that of the classical conjugate gradient method.
\item \textbf{Single component ($n=1$).} SAGA reduces to plain gradient descent with stepsize $\alpha$, and the bound in Theorem~\ref{thm:convex} collapses to the standard $O(1/K)$ guarantee for deterministic gradient descent.
\item \textbf{Infinite data limit.} When $n\to\infty$ and the table is initialized with i.i.d.\ samples from the data distribution, the variance term $n\alpha^{2}\sigma^{2}$ behaves like $\alpha^{2}n\to0$ for $\alpha=O(1/\sqrt{n})$, recovering the optimal stochastic rate $O(1/\sqrt{K})$ in the non‑strongly convex case.
\end{itemize}
