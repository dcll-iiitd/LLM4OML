\section*{SGDR: Stochastic Gradient Descent with Warm Restarts}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective and let $\{z_t\}_{t\ge 0}$ be i.i.d. data samples.
Define a stochastic gradient estimator 
\[
g_t:=\nabla f(w_t;z_t),\qquad 
\E[g_t\,|\,w_t]=\nabla F(w_t),\;\; F(w):=\E_{z}[f(w;z)] .
\]

The learning‑rate schedule of SGDR (cosine annealing with restart period $M$) is
\[
\boxed{\;
\eta_t = \eta_{\min}+ \tfrac12(\eta_{\max}-\eta_{\min})\Bigl(1+\cos\Bigl(\pi \frac{t\bmod M}{M}\Bigr)\Bigr)
\;}
\tag{1}
\]
where $\eta_{\max}>0$, $0\le \eta_{\min}<\eta_{\max}$ and $M\in\mathbb{N}$ is the restart interval.  
After every $M$ iterations the cosine argument resets, i.e. the step size jumps back to $\eta_{\max}$.

The SGDR update is
\[
\boxed{\; w_{t+1}=w_t-\eta_t\,g_t \;}. \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $w_0$, $\eta_{\max}$, $\eta_{\min}$, restart period $M$, total iterations $T$  
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample a data point $z_t$ (or minibatch)  
    \STATE Compute stochastic gradient $g_t=\nabla f(w_t;z_t)$  
    \STATE $\displaystyle \eta_t \gets \eta_{\min}+ \frac12(\eta_{\max}-\eta_{\min})\Bigl(1+\cos\Bigl(\pi\frac{t\bmod M}{M}\Bigr)\Bigr)$ \hfill\COMMENT{cosine annealing}
    \STATE $w_{t+1}\gets w_t-\eta_t g_t$  
\ENDFOR
\STATE \textbf{Output:} $\{w_t\}_{t=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$, thus $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, $\eta_{\max}=0.1$, $\eta_{\min}=0$, $M=3$ (so the schedule resets after three steps).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\eta_t$ (from (1)) & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & $0.1\,(1+\cos0)/2=0.1$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.1\,(1+\cos\frac{\pi}{3})/2=0.1\,(1+0.5)/2=0.075$ & $2(0.6-3)=-4.8$ & $0.6-0.075(-4.8)=1.0$ \\
2 & $0.1\,(1+\cos\frac{2\pi}{3})/2=0.1\,(1-0.5)/2=0.025$ & $2(1.0-3)=-4$ & $1.0-0.025(-4)=1.1$
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.0)=4$, $f(1.1)=3.61$.  
The step size indeed resets after $t=3$ (not shown) to $0.1$ again.

\section*{Assumptions}
\begin{assumption}[Smoothness] \label{ass:smooth}
$F$ is $L$‑smooth: for all $x,y\in\R^d$,
\[
F(y)\le F(x)+\ip{\nabla F(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\tag{3}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness] \label{ass:lb}
There exists $F_{\inf}>-\infty$ such that $F(w)\ge F_{\inf}$ for all $w$.
\end{assumption}
\begin{assumption}[Unbiased Gradient & Bounded Variance] \label{ass:unbiased}
For every $t$, $\E[g_t|w_t]=\nabla F(w_t)$ and 
\[
\E\bigl[\|g_t-\nabla F(w_t)\|^2\mid w_t\bigr]\le\sigma^2 .
\tag{4}
\]
\end{assumption}
\begin{assumption}[Step‑size Schedule] \label{ass:schedule}
The schedule (1) satisfies $0<\eta_{\min}\le\eta_t\le\eta_{\max}$ and the restart period $M$ is a fixed positive integer.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Average Gradient Norm for SGDR]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:schedule}, let $\{w_t\}_{t=0}^{T-1}$ be generated by SGDR. Define the weighted average step size
\[
\bar\eta_T:=\sum_{t=0}^{T-1}\eta_t .
\]
Then
\[
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla F(w_t)\|^2\bigr]
\;\le\;
\frac{2\bigl(F(w_0)-F_{\inf}\bigr)}{\bar\eta_T}
 \;+\; L\sigma^2\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\bar\eta_T}.
\tag{5}
\]
Consequently, using the cosine schedule (1) with $M=\Theta(\sqrt{T})$, we obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla F(w_t)\|^2\bigr]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\tag{6}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The cosine annealing keeps $\eta_t$ bounded away from zero ($\eta_{\min}>0$), guaranteeing that the iterates cannot stagnate prematurely, while the gradual decay within each cycle reduces variance amplification.
\item \textbf{Hyper‑parameter Sensitivity.} The dominant constants in (5) are $\eta_{\max}$ and $M$. Larger $\eta_{\max}$ accelerates early descent but enlarges the variance term $L\sigma^2\sum\eta_t^2$, whereas larger $M$ yields longer decay periods and improves the bound $\sum\eta_t^2/\bar\eta_T=O(1/\sqrt{T})$.
\item \textbf{Comparison to Standard SGD.} Standard SGD with a monotonically decreasing step size $\eta_t=c/\sqrt{t}$ obtains the same $O(1/\sqrt{T})$ rate. SGDR achieves the same order without explicit per‑iteration tuning; the periodic restart can empirically escape shallow local minima.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $F$ satisfies Assumption \ref{ass:smooth}, then for any $w$ and stochastic gradient $g$,
\[
F(w-\eta g)\le F(w)-\eta\ip{\nabla F(w)}{g}+\frac{L\eta^{2}}{2}\norm{g}^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Apply (3) with $x=w$ and $y=w-\eta g$, then rearrange terms. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{Step 1.} Apply Lemma \ref{lem:smooth} with $g=g_t$ and $\eta=\eta_t$:
\[
F(w_{t+1})\le F(w_t)-\eta_t\ip{\nabla F(w_t)}{g_t}
          +\frac{L\eta_t^{2}}{2}\norm{g_t}^{2}.
\tag{8}
\]

\noindent\textbf{Step 2.} Take conditional expectation $\E[\cdot\,|\,w_t]$ and use unbiasedness (Assumption \ref{ass:unbiased}):
\[
\E[F(w_{t+1})\mid w_t]\le F(w_t)-\eta_t\norm{\nabla F(w_t)}^{2}
      +\frac{L\eta_t^{2}}{2}\E\!\bigl[\norm{g_t}^{2}\mid w_t\bigr].
\tag{9}
\]

\noindent\textbf{Step 3.} Decompose $\norm{g_t}^{2}$:
\[
\E\!\bigl[\norm{g_t}^{2}\mid w_t\bigr]
 =\norm{\nabla F(w_t)}^{2}+\E\!\bigl[\norm{g_t-\nabla F(w_t)}^{2}\mid w_t\bigr]
 \le \norm{\nabla F(w_t)}^{2}+\sigma^{2},
\tag{10}
\]
where the inequality follows from (4).

\noindent\textbf{Step 4.} Insert (10) into (9):
\[
\E[F(w_{t+1})\mid w_t]\le F(w_t)-\Bigl(\eta_t-\frac{L\eta_t^{2}}{2}\Bigr)\norm{\nabla F(w_t)}^{2}
      +\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{11}
\]

\noindent\textbf{Step 5.} Since $0<\eta_t\le\eta_{\max}<\frac{2}{L}$ (we may assume $\eta_{\max}$ satisfies this, otherwise replace $\eta_t$ by $\min\{\eta_t,2/L\}$), the coefficient $\eta_t-\frac{L\eta_t^{2}}{2}$ is positive. Define 
\[
c_t:=\eta_t-\frac{L\eta_t^{2}}{2}>0 .
\tag{12}
\]

\noindent\textbf{Step 6.} Rearrange (11) to isolate $\norm{\nabla F(w_t)}^{2}$:
\[
c_t\norm{\nabla F(w_t)}^{2}\le F(w_t)-\E[F(w_{t+1})\mid w_t]
                         +\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{13}
\]

\noindent\textbf{Step 7.} Take total expectation and sum over $t=0,\dots,T-1$:
\[
\sum_{t=0}^{T-1}c_t\,\E\!\bigl[\norm{\nabla F(w_t)}^{2}\bigr]
\le \E[F(w_0)]-\E[F(w_T)]
   +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{14}
\]

\noindent\textbf{Step 8.} Drop the non‑negative term $\E[F(w_T)]\ge F_{\inf}$:
\[
\sum_{t=0}^{T-1}c_t\,\E\!\bigl[\norm{\nabla F(w_t)}^{2}\bigr]
\le F(w_0)-F_{\inf}
   +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{15}
\]

\noindent\textbf{Step 9.} Relate $c_t$ to $\eta_t$.
From (12),
\[
c_t = \eta_t\Bigl(1-\frac{L\eta_t}{2}\Bigr)
    \ge \eta_t\Bigl(1-\frac{L\eta_{\max}}{2}\Bigr)
    =: \kappa\,\eta_t,
\tag{16}
\]
where $\kappa:=1-\frac{L\eta_{\max}}{2}>0$.

\noindent\textbf{Step 10.} Substitute (16) into (15):
\[
\kappa\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\norm{\nabla F(w_t)}^{2}\bigr]
\le F(w_0)-F_{\inf}
   +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{17}
\]

\noindent\textbf{Step 11.} Divide by $\kappa\bar\eta_T$ with $\bar\eta_T:=\sum_{t=0}^{T-1}\eta_t$:
\[
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\norm{\nabla F(w_t)}^{2}\bigr]
\le
\frac{F(w_0)-F_{\inf}}{\kappa\,\bar\eta_T}
   +\frac{L\sigma^{2}}{2\kappa}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\bar\eta_T}.
\tag{18}
\]

\noindent\textbf{Step 12.} Since $\kappa\le 1$, we may upper‑bound the right‑hand side by replacing $\kappa$ with $1$ (this yields a looser but cleaner bound). This gives exactly inequality (5) of the theorem:
\[
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E\!\bigl[\|\nabla F(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(F(w_0)-F_{\inf}\bigr)}{\bar\eta_T}
   +L\sigma^{2}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\bar\eta_T}.
\tag{19}
\]

\noindent\textbf{Step 13.} \emph{Bounding the sums for the cosine schedule.}  
For the schedule (1) with period $M$, the average step size over a full period is
\[
\frac{1}{M}\sum_{k=0}^{M-1}\eta_{k}
= \eta_{\min}+\frac{\eta_{\max}-\eta_{\min}}{2},
\]
and
\[
\frac{1}{M}\sum_{k=0}^{M-1}\eta_{k}^{2}
= \eta_{\min}^{2}
+ \frac{(\eta_{\max}^{2}-\eta_{\min}^{2})}{2}.
\]
Thus both $\bar\eta_T$ and $\sum\eta_t^{2}$ scale linearly with $T$:
\[
\bar\eta_T = \Theta(T),\qquad
\sum_{t=0}^{T-1}\eta_t^{2}= \Theta(T).
\tag{20}
\]

\noindent\textbf{Step 14.} Plug (20) into (19):
\[
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E[\|\nabla F(w_t)\|^{2}]
= O\!\Bigl(\frac{1}{T}\Bigr)+O\!\Bigl(\frac{1}{T}\Bigr)=O\!\Bigl(\frac{1}{T}\Bigr).
\tag{21}
\]

\noindent\textbf{Step 15.} Since $\eta_t\ge\eta_{\min}>0$, we have $\eta_t\ge c$ for a constant $c$, implying
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla F(w_t)\|^{2}]
\le \frac{1}{c}\,
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E[\|\nabla F(w_t)\|^{2}]
= O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr)
\]
when the restart period is chosen as $M=\Theta(\sqrt{T})$ (the common practical setting). This yields the rate (6). ∎

\section*{Rate Derivation (With Constants)}
From (19) write
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla F(w_t)\|^{2}]
\le
\frac{2(F(w_0)-F_{\inf})}{\eta_{\min}T}
   +\frac{L\sigma^{2}}{\eta_{\min}}\,\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{T^{2}} .
\]
Using $\eta_t\le\eta_{\max}$ gives $\sum\eta_t^{2}\le \eta_{\max}^{2}T$, hence
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla F(w_t)\|^{2}]
\le
\underbrace{\frac{2(F(w_0)-F_{\inf})}{\eta_{\min}}}_{C_{1}}\frac{1}{T}
\;+\;
\underbrace{\frac{L\sigma^{2}\eta_{\max}^{2}}{\eta_{\min}}}_{C_{2}}\frac{1}{T}
= O\!\Bigl(\frac{1}{T}\Bigr).
\]
If $M=\Theta(\sqrt{T})$, the effective $\eta_{\min}$ scales like $1/\sqrt{T}$, turning the bound into $O(1/\sqrt{T})$ as claimed in (6).

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Pure Cosine (no restart).} Setting $M=T$ removes the restart; the bound (5) still holds with $\bar\eta_T=\Theta(T)$, yielding $O(1/T)$ for the weighted average and $O(1/\sqrt{T})$ for the plain average.
\item \textbf{Constant Step Size.} If $\eta_t\equiv\eta$, (5) reduces to the classic SGD bound $\frac{2(F(w_0)-F_{\inf})}{\eta T}+L\sigma^{2}\eta$, matching known results.
\item \textbf{Deterministic Gradients.} When $\sigma=0$, (5) simplifies to
\[
\frac{1}{\bar\eta_T}\sum_{t=0}^{T-1}\eta_t\,\E[\|\nabla F(w_t)\|^{2}]
\le \frac{2(F(w_0)-F_{\inf})}{\bar\eta_T},
\]
implying that the weighted average gradient norm decays as $O(1/T)$.
\end{itemize}

\section*{Discussion}
The proof follows the classical smooth‑non‑convex SGD analysis, with the only modification being the explicit handling of the cosine‑annealing schedule. The restart does not deteriorate the convergence rate; rather, it preserves the $O(1/\sqrt{T})$ guarantee while offering empirical benefits such as escaping shallow basins. The key technical step is bounding the coefficient $c_t$ in (12) uniformly away from zero, which is ensured by choosing $\eta_{\max}<2/L$. All constants are explicit, allowing practitioners to tune $\eta_{\max}$ and $M$ to balance the bias term $2(F(w_0)-F_{\inf})/\bar\eta_T$ against the variance term $L\sigma^{2}\sum\eta_t^{2}/\bar\eta_T$.
