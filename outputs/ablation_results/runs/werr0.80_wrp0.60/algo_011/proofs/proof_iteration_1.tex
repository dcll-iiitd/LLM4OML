\section*{Recursive Variance‑Adapted Gradient (RVAG) }

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
The algorithm maintains a recursive estimator of the gradient
\[
\boxed{%
v_{t}= (1-\alpha)\,v_{t-1}+ \alpha\,\nabla f(x_{t}) ,\qquad t\ge 1,
}
\tag{1}
\]
with $v_{0}=0$ and a stepsize that adapts to the magnitude of the estimator:
\[
\boxed{%
\eta_{t}= \frac{\eta}{1+\beta\| \widehat v_{t}\|},\qquad 
\widehat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t}} .
}
\tag{2}
\]
The bias‑corrected vector $\widehat v_{t}$ satisfies $\E[\widehat v_{t}\mid x_{t}]=\nabla f(x_{t})$ (see Lemma~1).  
The iterate update is
\[
\boxed{%
x_{t+1}=x_{t}-\eta_{t}\,\widehat v_{t}.
}
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Adapted Gradient (RVAG)}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize base $\eta>0$, 
decay parameters $\alpha\in(0,1]$, $\beta\ge0$, maximum iterations $T$.
\State $v_{0}\gets 0$.
\For{$t=0,\dots,T-1$}
    \State Compute stochastic gradient $g_{t}\gets\nabla f(x_{t})$  
          (or an unbiased estimator thereof).
    \State $v_{t+1}\gets (1-\alpha)v_{t}+ \alpha g_{t}$ \hfill // (1)
    \State $\widehat v_{t+1}\gets v_{t+1}/\bigl(1-(1-\alpha)^{t+1}\bigr)$ \hfill // bias correction
    \State $\eta_{t+1}\gets \displaystyle\frac{\eta}{1+\beta\|\widehat v_{t+1}\|}$ \hfill // (2)
    \State $x_{t+1}\gets x_{t}-\eta_{t+1}\widehat v_{t+1}$ \hfill // (3)
\EndFor
\State \Return $x_{T}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Set $w_{0}=0$, stepsize base $\eta=0.1$, $\alpha=0.5$, $\beta=0$ (so $\eta_{t}=0.1$).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $g_{t}= \nabla f(w_{t})$ & $v_{t}$ (by (1)) & $f(w_{t})$\\ \hline
0 & $0$ & $-6$ & $v_{0}=0$ & $9$\\
1 & $w_{1}=0-0.1\cdot(-6)=0.6$ & $g_{1}=2(0.6-3)=-4.8$ &
$v_{1}=0.5\cdot0+0.5(-6)=-3$ & $5.76$\\
2 & $w_{2}=0.6-0.1\cdot(-3) =0.9$ &
$g_{2}=2(0.9-3)=-4.2$ &
$v_{2}=0.5(-3)+0.5(-4.8)=-3.9$ & $4.41$\\
3 & $w_{3}=0.9-0.1\cdot(-3.9)=1.29$ &
$g_{3}=2(1.29-3)=-3.42$ &
$v_{3}=0.5(-3.9)+0.5(-4.2)=-4.05$ & $3.0361$\\
\end{tabular}
\end{center}
The objective value decreases from $9$ to $3.04$ in three iterations, illustrating the effect of the recursive variance estimator.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\qquad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+ \frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{as:pl}
There exists $\mu>0$ such that
\[
\forall x\in\R^{d}:\qquad 
\frac{1}{2}\|\nabla f(x)\|^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\]
where $f^{\star}= \min_{x}f(x)$.
\end{assumption}
\begin{assumption}[Stochastic Gradient Oracle]\label{as:stoch}
For every $x$, we can draw a random vector $g(x)$ satisfying
\[
\E[g(x)\mid x]=\nabla f(x),\qquad 
\E\!\bigl[\|g(x)-\nabla f(x)\|^{2}\mid x\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter range]\label{as:hyper}
\[
\alpha\in(0,1],\qquad \beta\ge0,\qquad 
0<\eta\le \frac{1}{L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Assume \ref{as:smooth}–\ref{as:hyper} hold. Let the iterates $\{x_{t}\}$ be generated by RVAG. Define the bias‑corrected estimator $\widehat v_{t}$ as in (2). Then for any $T\ge1$,
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]
\;\le\;
\bigl(1-\eta\mu\bigr)^{T}\bigl(f(x_{0})-f^{\star}\bigr)
\;+\;
\frac{\eta\sigma^{2}}{2\mu}.
\tag{4}
\]
Consequently, if $\sigma=0$ (exact gradients) the method enjoys a linear rate
$\E[f(x_{T})-f^{\star}]\le (1-\eta\mu)^{T}(f(x_{0})-f^{\star})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The adaptive stepsize $\eta_{t}$ is bounded above by $\eta$; the denominator $1+\beta\|\widehat v_{t}\|$ prevents explosion even when the variance estimator is large.
\item \textbf{Hyper‑parameter sensitivity.} 
\begin{itemize}
\item Larger $\alpha$ places more weight on the newest stochastic gradient, reducing bias but increasing variance.
\item $\beta$ controls the strength of the norm‑based decay; $\beta=0$ recovers a constant stepsize.
\item The linear rate requires $\eta\le 1/L$; a smaller $\eta$ yields a slower but more robust convergence.
\end{itemize}
\item \textbf{Comparison to baselines.}
\begin{itemize}
\item With $\alpha=1$ and $\beta=0$, RVAG reduces to standard SGD with constant stepsize and inherits its $O(1/T)$ sub‑linear rate under PL (see e.g.~\cite{karimi2016linear}).
\item The recursive estimator reduces the variance term from $\sigma^{2}$ to $\frac{\alpha}{2-\alpha}\sigma^{2}$ (Lemma~2), yielding a strictly smaller steady‑state error than SGD.
\item Compared with Adam, RVAG enjoys a provable linear rate under PL without requiring convexity of the second‑moment estimate.
\end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that the combination of a PL condition, $L$‑smoothness and an exponentially weighted gradient estimator yields a linear convergence guarantee up to a variance floor proportional to $\sigma^{2}$. The bias‑correction term exactly removes the deterministic bias introduced by the recursion, making the estimator unbiased (Lemma~1). The adaptive stepsize further stabilises the algorithm when the estimator norm grows, a property that is crucial in non‑convex extensions (see Section~5 of the extended manuscript).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the bias‑corrected estimator]\label{lem:unbiased}
Under Assumption~\ref{as:stoch}, for every $t\ge1$,
\[
\E\bigl[\widehat v_{t}\mid x_{t}\bigr]=\nabla f(x_{t}).
\]
\end{lemma}
\begin{proof}
From the recursion (1),
\[
v_{t}= (1-\alpha)^{t}v_{0}+ \alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}\,g_{t-1-k}.
\]
Since $v_{0}=0$ and $\E[g_{s}\mid x_{s}]=\nabla f(x_{s})$,
\[
\E[v_{t}\mid x_{t}]
 = \alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}\,\nabla f(x_{t-1-k}).
\]
Multiplying by the correction factor $c_{t}:=1/(1-(1-\alpha)^{t})$ gives
\[
\E[\widehat v_{t}\mid x_{t}]
 = c_{t}\,\alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}\,\nabla f(x_{t-1-k})
 = \nabla f(x_{t}),
\]
because the coefficients form a geometric series that sums to $c_{t}^{-1}$.  ∎
\end{proof}

\begin{lemma}[Variance bound]\label{lem:var}
Let $\widehat v_{t}$ be defined as in (2). Under Assumption~\ref{as:stoch},
\[
\E\!\bigl[\|\widehat v_{t}-\nabla f(x_{t})\|^{2}\mid x_{t}\bigr]
\le
\frac{\alpha}{2-\alpha}\,\sigma^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
From the expression of $v_{t}$,
\[
v_{t}= \alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}g_{t-1-k}.
\]
Since the $g$’s are conditionally independent and have bounded variance,
\[
\E\!\bigl[\|v_{t}-\E[v_{t}\mid x_{t}]\|^{2}\mid x_{t}\bigr]
 = \alpha^{2}\sum_{k=0}^{t-1}(1-\alpha)^{2k}\,\E\!\bigl[\|g_{t-1-k}-\nabla f(x_{t-1-k})\|^{2}\mid x_{t}\bigr]
 \le \alpha^{2}\sigma^{2}\sum_{k=0}^{\infty}(1-\alpha)^{2k}.
\]
The infinite geometric series sums to $\frac{1}{1-(1-\alpha)^{2}}=\frac{1}{\alpha(2-\alpha)}$, yielding
\[
\E\!\bigl[\|v_{t}-\E[v_{t}\mid x_{t}]\|^{2}\mid x_{t}\bigr]
 \le \frac{\alpha}{2-\alpha}\sigma^{2}.
\]
Because $\widehat v_{t}=v_{t}/c_{t}$ with $c_{t}\le1$, the same bound holds for $\widehat v_{t}$, establishing (5). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof, expectations are taken over all randomness up to iteration $t$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1] Smoothness inequality.}  
By $L$‑smoothness (Assumption~\ref{as:smooth}) and the update (3),
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t}) + \langle\nabla f(x_{t}),x_{t+1}-x_{t}\rangle
      +\frac{L}{2}\|x_{t+1}-x_{t}\|^{2} \\
&= f(x_{t}) -\eta_{t}\langle\nabla f(x_{t}),\widehat v_{t}\rangle
      +\frac{L}{2}\eta_{t}^{2}\|\widehat v_{t}\|^{2}. \tag{6}
\end{aligned}
\]

\noindent\textbf{[Step 2] Decompose the inner product.}  
Add and subtract $\widehat v_{t}$ inside the inner product:
\[
\langle\nabla f(x_{t}),\widehat v_{t}\rangle
= \|\widehat v_{t}\|^{2}
  +\langle\nabla f(x_{t})-\widehat v_{t},\widehat v_{t}\rangle. \tag{7}
\]

\noindent\textbf{[Step 3] Plug (7) into (6).}
\[
f(x_{t+1})
\le f(x_{t}) -\eta_{t}\|\widehat v_{t}\|^{2}
      -\eta_{t}\langle\nabla f(x_{t})-\widehat v_{t},\widehat v_{t}\rangle
      +\frac{L}{2}\eta_{t}^{2}\|\widehat v_{t}\|^{2}. \tag{8}
\]

\noindent\textbf{[Step 4] Take conditional expectation.}  
Condition on the sigma‑algebra $\mathcal{F}_{t}$ generated by $\{x_{0},\dots ,x_{t}\}$.  
Using Lemma~\ref{lem:unbiased}, $\E[\widehat v_{t}\mid\mathcal{F}_{t}]=\nabla f(x_{t})$, therefore the cross term vanishes:
\[
\E\!\bigl[\langle\nabla f(x_{t})-\widehat v_{t},\widehat v_{t}\rangle\mid\mathcal{F}_{t}\bigr]=0. \tag{9}
\]

\noindent\textbf{[Step 5] Upper bound the remaining terms.}
Since $\eta_{t}\le\eta$ (by definition (2) and $\beta\ge0$),
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(x_{t}) -\eta\,\E\!\bigl[\|\widehat v_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]
    +\frac{L}{2}\eta^{2}\E\!\bigl[\|\widehat v_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]. \tag{10}
\]

\noindent\textbf{[Step 6] Relate $\|\widehat v_{t}\|^{2}$ to $\|\nabla f(x_{t})\|^{2}$.}  
Write $\widehat v_{t}= \nabla f(x_{t}) + (\widehat v_{t}-\nabla f(x_{t}))$ and expand:
\[
\|\widehat v_{t}\|^{2}
= \|\nabla f(x_{t})\|^{2}
  +2\langle\nabla f(x_{t}),\widehat v_{t}-\nabla f(x_{t})\rangle
  +\|\widehat v_{t}-\nabla f(x_{t})\|^{2}.
\]
Taking conditional expectation, the cross term disappears (again by unbiasedness), yielding
\[
\E\!\bigl[\|\widehat v_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]
= \|\nabla f(x_{t})\|^{2}
  +\E\!\bigl[\|\widehat v_{t}-\nabla f(x_{t})\|^{2}\mid\mathcal{F}_{t}\bigr]. \tag{11}
\]

\noindent\textbf{[Step 7] Insert the variance bound (Lemma~\ref{lem:var}).}
Using (5),
\[
\E\!\bigl[\|\widehat v_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]
\le \|\nabla f(x_{t})\|^{2} + \frac{\alpha}{2-\alpha}\sigma^{2}. \tag{12}
\]

\noindent\textbf{[Step 8] Substitute (12) into (10).}
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(x_{t}) -\eta\|\nabla f(x_{t})\|^{2}
    +\frac{L}{2}\eta^{2}\|\nabla f(x_{t})\|^{2}
    +\Bigl(\frac{L}{2}\eta^{2}-\eta\Bigr)\frac{\alpha}{2-\alpha}\sigma^{2}. \tag{13}
\]

\noindent\textbf{[Step 9] Choose $\eta\le 1/L$.}  
If $\eta\le 1/L$, then $ \frac{L}{2}\eta^{2}\le \frac{\eta}{2}$, hence
\[
-\eta +\frac{L}{2}\eta^{2}\le -\frac{\eta}{2}. \tag{14}
\]
Applying (14) to (13) gives
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(x_{t}) -\frac{\eta}{2}\|\nabla f(x_{t})\|^{2}
    +\frac{\eta}{2}\,\frac{\alpha}{2-\alpha}\sigma^{2}. \tag{15}
\]

\noindent\textbf{[Step 10] Use the PL inequality (Assumption~\ref{as:pl}).}  
From PL,
\[
\|\nabla f(x_{t})\|^{2}\ge 2\mu\bigl(f(x_{t})-f^{\star}\bigr). \tag{16}
\]

\noindent\textbf{[Step 11] Plug (16) into (15).}
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal{F}_{t}\bigr]
\le f(x_{t}) -\eta\mu\bigl(f(x_{t})-f^{\star}\bigr)
    +\frac{\eta}{2}\,\frac{\alpha}{2-\alpha}\sigma^{2}. \tag{17}
\]

\noindent\textbf{[Step 12] Rearrange (17).}
\[
\E\!\bigl[f(x_{t+1})-f^{\star}\mid\mathcal{F}_{t}\bigr]
\le \bigl(1-\eta\mu\bigr)\bigl(f(x_{t})-f^{\star}\bigr)
    +\frac{\eta}{2}\,\frac{\alpha}{2-\alpha}\sigma^{2}. \tag{18}
\]

\noindent\textbf{[Step 13] Unroll the recursion.}  
Taking total expectation and iterating (18) yields
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]
\le (1-\eta\mu)^{T}\bigl(f(x_{0})-f^{\star}\bigr)
   +\frac{\eta}{2}\,\frac{\alpha}{2-\alpha}\sigma^{2}
     \sum_{k=0}^{T-1}(1-\eta\mu)^{k}. \tag{19}
\]

\noindent\textbf{[Step 14] Sum the geometric series.}
\[
\sum_{k=0}^{T-1}(1-\eta\mu)^{k}
= \frac{1-(1-\eta\mu)^{T}}{\eta\mu}
\le \frac{1}{\eta\mu}. \tag{20}
\]

\noindent\textbf{[Step 15] Insert (20) into (19).}
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]
\le (1-\eta\mu)^{T}\bigl(f(x_{0})-f^{\star}\bigr)
   +\frac{\sigma^{2}}{2\mu}\,\frac{\alpha}{2-\alpha}. \tag{21}
\]

\noindent\textbf{[Step 16] Simplify the constant.}  
Since $\frac{\alpha}{2-\alpha}\le1$ for $\alpha\in(0,1]$, we obtain the cleaner bound
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]
\le (1-\eta\mu)^{T}\bigl(f(x_{0})-f^{\star}\bigr)
   +\frac{\eta\sigma^{2}}{2\mu},
\]
which is exactly the statement (4) of Theorem~\ref{thm:linear}. ∎

\section*{Rate Derivation (With Constants)}
From (4) we can read off two regimes:

\begin{itemize}
\item \textbf{Exact gradients ($\sigma=0$).} The error decays geometrically with factor $1-\eta\mu$, i.e. linear convergence.
\item \textbf{Noisy gradients.} After $T\ge\frac{1}{\eta\mu}\log\frac{f(x_{0})-f^{\star}}{\varepsilon}$ iterations the first term drops below $\varepsilon$, and the residual error is bounded by $\frac{\eta\sigma^{2}}{2\mu}$. Choosing $\eta=\Theta\!\bigl(\frac{1}{L}\bigr)$ minimises the steady‑state term, giving
\[
\text{steady‑state error }=O\!\bigl(\frac{\sigma^{2}}{L\mu}\bigr).
\]
\end{itemize}

\section*{Special Cases}
\begin{enumerate}
\item \textbf{SGD as a limit.} Setting $\alpha=1$ makes $v_{t}=g_{t}$ and $c_{t}=1$, so $\widehat v_{t}=g_{t}$. The algorithm reduces to standard SGD with adaptive stepsize $\eta_{t}= \eta/(1+\beta\|g_{t}\|)$. Theorem~\ref{thm:linear} collapses to the known linear rate for SGD under PL (see~\cite{karimi2016linear}).
\item \textbf{Heavy‑ball limit.} If $\beta=0$ and $\alpha$ is small, $v_{t}$ evolves slowly, providing a momentum‑like effect. The bound (21) still holds, but the variance reduction factor $\frac{\alpha}{2-\alpha}$ becomes more pronounced.
\item \textbf{Deterministic case.} When $\sigma=0$, Lemma~\ref{lem:var} is trivial and the algorithm attains the optimal rate $O((1-\eta\mu)^{T})$ for smooth PL functions, matching the rate of full‑gradient gradient descent with stepsize $\eta$.
\end{enumerate}

\bigskip
\noindent\textbf{Conclusion.}  
We have presented the RVAG method, derived its unbiased recursive variance estimator, proved a linear convergence guarantee under the PL condition, and highlighted how the hyper‑parameters influence both the convergence speed and the variance floor. The proof is fully elementary, relying only on smoothness, the PL inequality, unbiasedness, and a simple geometric‑series argument.
