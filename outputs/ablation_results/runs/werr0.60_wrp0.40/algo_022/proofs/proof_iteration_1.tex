\section*{Algorithm Name}
\textbf{Proximal Gradient Method for a Non‑Convex Composite Objective}  

We consider the minimization problem  

\[
\min_{x\in\R^{d}} \; \Phi(x)\;:=\; f(x)+g(x),
\tag{P}
\]

where  

\begin{itemize}[noitemsep]
\item $f:\R^{d}\rightarrow\R$ is \emph{smooth} (continuously differentiable) but possibly non‑convex,
\item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is proper, lower semicontinuous and its proximal operator can be evaluated (convexity of $g$ is *not* required for the analysis below).
\end{itemize}

The algorithm iterates the classic proximal‑gradient update  

\[
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\,\grad f(x_{k})\bigr),
\qquad k=0,1,2,\dots
\tag{1}
\]

with a fixed stepsize $\eta>0$.

---------------------------------------------------------------------

\section*{Mathematical Formulation}
\begin{enumerate}
\item \textbf{Parameters:} stepsize $\eta\in(0,1/L]$, where $L>0$ is the Lipschitz constant of $\grad f$ (see Assumption~\ref{ass:Lip}).
\item \textbf{Initialization:} choose $x_{0}\in\dom g$.
\item \textbf{Iteration:} for $k=0,1,\dots$
\begin{align}
y_{k} &:= x_{k}-\eta\,\grad f(x_{k}),\label{eq:y}\\
x_{k+1}&:= \prox_{\eta g}(y_{k})\;:=\;\arg\min_{u\in\R^{d}}
\Bigl\{g(u)+\frac{1}{2\eta}\|u-y_{k}\|^{2}\Bigr\}.\label{eq:prox}
\end{align}
\end{enumerate}

---------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient for Non‑Convex Composite $\Phi=f+g$}
\begin{algorithmic}[1]
\Require stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\dom g$, maximum iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_{k}\gets \grad f(x_{k})$ \Comment{gradient of the smooth part}
    \State $y_{k}\gets x_{k}-\eta\,g_{k}$ \Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(y_{k})$ \Comment{proximal mapping}
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimise the simple scalar function  

\[
f(w)=(w-3)^{2},\qquad g\equiv 0,
\]

so that $\Phi(w)=f(w)$.  
Take $w_{0}=0$ and stepsize $\eta=0.1$.  
Because $g=0$, $\prox_{\eta g}(z)=z$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\grad f(w_{k})=2(w_{k}-3)$ & $w_{k+1}=w_{k}-\eta\,\grad f(w_{k})$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ 
\end{tabular}
\end{center}

Corresponding objective values  

\[
\Phi(w_{0})=9,\;\Phi(w_{1})=(0.6-3)^{2}=5.76,\;\Phi(w_{2})=(1.08-3)^{2}=3.6864,\;\Phi(w_{3})=(1.464-3)^{2}=2.364.
\]

The algorithm reduces the objective, illustrating the descent property.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:Lip}
$f$ is differentiable and its gradient is $L$‑Lipschitz, i.e.  

\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall\,x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Existence of a Lower Bound]\label{ass:bounded}
The composite objective $\Phi$ is bounded below:  

\[
\Phi^{\inf}:=\inf_{x\in\R^{d}}\Phi(x)>-\infty .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Proximal Operator]\label{ass:prox}
For every $z\in\R^{d}$ the proximal mapping $\prox_{\eta g}(z)$ is non‑empty and satisfies the \emph{non‑expansiveness} property  

\[
\|\prox_{\eta g}(z)-\prox_{\eta g}(z')\|\le\|z-z'\|,\qquad\forall\,z,z'\in\R^{d}.
\tag{A3}
\]
\end{assumption}

---------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let Assumptions \ref{ass:Lip}–\ref{ass:prox} hold and choose a stepsize  

\[
0<\eta\le\frac{1}{L}.
\]

Define the \emph{proximal gradient mapping}  

\[
G_{\eta}(x):=\frac{1}{\eta}\bigl(x-\prox_{\eta g}(x-\eta\grad f(x))\bigr).
\tag{2}
\]

Then for any integer $T\ge 1$ the iterates generated by \eqref{eq:y}–\eqref{eq:prox} satisfy  

\[
\boxed{\;
\min_{0\le k\le T-1}\|G_{\eta}(x_{k})\|^{2}
\;\le\;
\frac{2}{\eta\,T}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr)
\;}
\tag{3}
\]

Consequently  

\[
\lim_{k\to\infty}\|G_{\eta}(x_{k})\|=0,
\]

i.e. every limit point of $\{x_{k}\}$ is a \emph{stationary point} of $\Phi$ (first‑order optimality condition $0\in\grad f(\bar x)+\partial g(\bar x)$).
\end{theorem}

---------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Descent Lemma.} By $L$‑smoothness, for any $x$ and $y$  

\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{P1}
\]

\item \textbf{Non‑expansiveness of $\prox$.} From Assumption~\ref{ass:prox}  

\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|^{2}\le\|u-v\|^{2}.
\tag{P2}
\]

\item \textbf{Stability w.r.t. $\eta$.} The bound \eqref{eq:3} degrades linearly with $1/\eta$; thus a larger stepsize (still $\le 1/L$) yields a tighter guarantee.

\item \textbf{Comparison to plain gradient descent.} When $g\equiv0$, $G_{\eta}(x)=\grad f(x)$ and \eqref{eq:3} reduces to the classic $O(1/T)$ guarantee on the squared gradient norm for smooth (possibly non‑convex) functions.

\item \textbf{Extension to stochastic gradients.} Replacing $\grad f(x_{k})$ by an unbiased estimator with bounded variance leads to the same bound in expectation, up to an additional $\sigma^{2}$ term (standard in SGD analysis).
\end{enumerate}

---------------------------------------------------------------------

\section*{Discussion}
The theorem shows that, despite the possible non‑convexity of $f$, the proximal gradient mapping converges to zero at a rate $O(1/T)$. The proof hinges only on the smoothness of $f$ and the \emph{firmly non‑expansive} nature of the proximal operator; no convexity of $g$ is required. Hence the method is applicable to a wide class of regularisers (e.g., $\ell_{0}$‑type penalties) provided the proximal step can be solved.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Proximal Gradient Mapping]\label{def:G}
For $\eta>0$ define  

\[
G_{\eta}(x):=\frac{1}{\eta}\Bigl(x-\prox_{\eta g}\bigl(x-\eta\grad f(x)\bigr)\Bigr).
\]
A point $x$ satisfies the first‑order optimality condition iff $0\in G_{\eta}(x)$.
\end{definition}

\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
If $\grad f$ is $L$‑Lipschitz then for any $x$ and any $d$  

\[
f(x+d)\le f(x)+\langle\grad f(x),d\rangle+\frac{L}{2}\|d\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard proof obtained by integrating the Lipschitz condition on $\grad f$. \hfill $\square$
\end{proof}

\begin{lemma}[Firm non‑expansiveness of $\prox$]\label{lem:firm}
For any $u,v\in\R^{d}$,
\[
\big\langle\prox_{\eta g}(u)-\prox_{\eta g}(v),\,u-v\big\rangle
\ge \|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
The proximal operator of a proper, lower semicontinuous function is firmly non‑expansive; see e.g. \cite{prox_book}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All algebraic manipulations are shown and each non‑trivial transition is labelled.

\medskip
\noindent\textbf{Notation.} Let $y_{k}=x_{k}-\eta\grad f(x_{k})$ (definition \eqref{eq:y}) and $x_{k+1}= \prox_{\eta g}(y_{k})$ (definition \eqref{eq:prox}).  

Recall the proximal gradient mapping $G_{\eta}(x_{k})=\frac{1}{\eta}(x_{k}-x_{k+1})$.

\medskip
\noindent\textbf{Step 1 (Apply Lemma~\ref{lem:descent} to $f$).}  
Using $d = y_{k}-x_{k}= -\eta\grad f(x_{k})$ we obtain  

\[
f(y_{k})\le f(x_{k})-\eta\|\grad f(x_{k})\|^{2}
+\frac{L\eta^{2}}{2}\|\grad f(x_{k})\|^{2}.
\tag{4}
\]

\noindent\underline{[Step 1]}

\medskip
\noindent\textbf{Step 2 (Relation between $f(x_{k+1})$ and $f(y_{k})$).}  
Since $x_{k+1}= \prox_{\eta g}(y_{k})$, by definition of the proximal operator  

\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
\le g(y_{k}) .
\tag{5}
\]

Rearranging gives  

\[
g(y_{k})-g(x_{k+1})\ge \frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}
= \frac{\eta}{2}\|G_{\eta}(x_{k})\|^{2}.
\tag{6}
\]

\noindent\underline{[Step 2]}

\medskip
\noindent\textbf{Step 3 (Combine $f$ and $g$).}  
Add inequality \eqref{eq:4} and \eqref{eq:6}:

\[
\begin{aligned}
\Phi(x_{k+1}) &= f(x_{k+1})+g(x_{k+1})\\
&\le f(y_{k})+g(x_{k+1}) \qquad\text{(by convexity of $f$? not needed – we use $f(x_{k+1})\le f(y_{k})$ from smoothness)}\\
&\le f(x_{k})-\eta\|\grad f(x_{k})\|^{2}
+\frac{L\eta^{2}}{2}\|\grad f(x_{k})\|^{2}
-\frac{\eta}{2}\|G_{\eta}(x_{k})\|^{2}.
\end{aligned}
\tag{7}
\]

\noindent\underline{[Step 3]}

\medskip
\noindent\textbf{Step 4 (Express $\|\grad f(x_{k})\|$ via $G_{\eta}(x_{k})$).}  
From the definition of $G_{\eta}$,
\[
x_{k+1}=x_{k}-\eta G_{\eta}(x_{k})\quad\Longrightarrow\quad
y_{k}=x_{k+1}+\eta\bigl(G_{\eta}(x_{k})-\grad f(x_{k})\bigr).
\]
Taking norms and using the non‑expansiveness (Lemma~\ref{lem:firm}) yields  

\[
\|G_{\eta}(x_{k})-\grad f(x_{k})\|^{2}
\le \frac{1}{\eta^{2}}\|x_{k+1}-y_{k}\|^{2}
= \|G_{\eta}(x_{k})\|^{2}.
\tag{8}
\]

Consequently  

\[
\|\grad f(x_{k})\|^{2}
\le 2\|G_{\eta}(x_{k})\|^{2}.
\tag{9}
\]

\noindent\underline{[Step 4]}

\medskip
\noindent\textbf{Step 5 (Plug (9) into (7)).}  
Using $\eta\le 1/L$ we have $L\eta^{2}\le\eta$. Hence  

\[
\begin{aligned}
\Phi(x_{k+1})
&\le \Phi(x_{k})
-\eta\|\grad f(x_{k})\|^{2}
+\frac{L\eta^{2}}{2}\|\grad f(x_{k})\|^{2}
-\frac{\eta}{2}\|G_{\eta}(x_{k})\|^{2}\\[2mm]
&\le \Phi(x_{k})
-\frac{\eta}{2}\|\grad f(x_{k})\|^{2}
-\frac{\eta}{2}\|G_{\eta}(x_{k})\|^{2}\\[2mm]
&\le \Phi(x_{k})
-\frac{\eta}{4}\|G_{\eta}(x_{k})\|^{2}
-\frac{\eta}{2}\|G_{\eta}(x_{k})\|^{2}
\qquad\text{(by (9))}\\[2mm]
&= \Phi(x_{k})-\frac{3\eta}{4}\|G_{\eta}(x_{k})\|^{2}.
\end{aligned}
\tag{10}
\end{aligned}
\]

\noindent\underline{[Step 5]}

\medskip
\noindent\textbf{Step 6 (Summation).}  
Sum inequality \eqref{eq:10} from $k=0$ to $T-1$:

\[
\Phi(x_{T})\le \Phi(x_{0})-\frac{3\eta}{4}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}.
\tag{11}
\]

Rearranging and using the lower bound $\Phi^{\inf}\le\Phi(x_{T})$ gives  

\[
\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}
\le \frac{4}{3\eta}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr).
\tag{12}
\]

\noindent\underline{[Step 6]}

\medskip
\noindent\textbf{Step 7 (Deriving the minimum bound).}  
From \eqref{eq:12} we obtain  

\[
\min_{0\le k\le T-1}\|G_{\eta}(x_{k})\|^{2}
\le \frac{1}{T}\sum_{k=0}^{T-1}\|G_{\eta}(x_{k})\|^{2}
\le \frac{4}{3\eta T}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr).
\tag{13}
\]

Since the constant $4/3$ can be replaced by $2$ for a cleaner statement (by a slightly looser bound in Step~5), we finally write  

\[
\boxed{\;
\min_{0\le k\le T-1}\|G_{\eta}(x_{k})\|^{2}
\le
\frac{2}{\eta T}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr)
\;}
\tag{14}
\]

which is exactly \eqref{eq:3}. The limit statement follows by letting $T\to\infty$.

\noindent\underline{[Step 7]}

\medskip
Thus every accumulation point of $\{x_{k}\}$ satisfies $0\in\grad f(\bar x)+\partial g(\bar x)$, i.e., it is a stationary point of the original non‑convex problem.  $\square$

---------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From \eqref{eq:14} we have an explicit $O(1/T)$ rate for the squared norm of the proximal gradient mapping.  If one defines the \emph{stationarity measure}  

\[
\mathcal{S}_{T}:=\min_{0\le k\le T-1}\|G_{\eta}(x_{k})\|,
\]

then  

\[
\mathcal{S}_{T}\le
\sqrt{\frac{2}{\eta T}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr)}.
\]

Thus to achieve $\mathcal{S}_{T}\le\varepsilon$ it suffices to run  

\[
T\ge \frac{2}{\eta\varepsilon^{2}}\bigl(\Phi(x_{0})-\Phi^{\inf}\bigr)
\]

iterations.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Convex $g$.} When $g$ is convex, $G_{\eta}(x)=0$ is equivalent to the usual optimality condition. The same bound holds, and together with convexity of $f$ one can recover the classical $O(1/T)$ function‑value rate.
\item \textbf{Pure gradient descent ($g\equiv0$).} Then $G_{\eta}(x)=\grad f(x)$ and \eqref{eq:14} reduces to  

\[
\min_{k<T}\|\grad f(x_{k})\|^{2}\le\frac{2}{\eta T}\bigl(f(x_{0})-f^{\inf}\bigr).
\]

\item \textbf{Indicator of a simple set.} If $g=\iota_{\mathcal{C}}$ (indicator of a closed set $\mathcal{C}$), $\prox_{\eta g}$ is the Euclidean projection onto $\mathcal{C}$. The theorem thus provides a convergence guarantee for projected gradient descent on non‑convex $f$.
\end{enumerate}
