\section*{Algorithm Name}
\textbf{Stochastic Gradient Descent with Warm Restarts (SGDR)} – a variant of SGD
where the learning rate follows a cosine annealing schedule and is reset to a
maximum value every $M$ iterations (a ``restart'').

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective function.
At iteration $t\in\{0,1,\dots,T-1\}$ we draw a random data sample $\xi_t$,
obtain a stochastic gradient $g_t:=\nabla f(w_t;\xi_t)$ and update the
parameter vector $w_t\in\R^{d}$ according to
\[
\boxed{w_{t+1}=w_t-\eta_t\,g_t},
\tag{1}
\]
where the learning‑rate $\eta_t$ is defined by the \emph{cosine restart
schedule}
\[
\boxed{
\eta_t \;=\;
\frac{\eta_{\max}}{2}\Bigl(1+\cos\Bigl(\pi\frac{t\bmod M}{M}\Bigr)\Bigr)},
\qquad
\eta_{\max}>0,\;M\in\mathbb{N}.
\tag{2}
\]
Thus $\eta_t$ starts from $\eta_{\max}$, decays smoothly to
$\eta_{\max}/2$ at the end of each cycle of length $M$, and then is
reset to $\eta_{\max}$ again.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\R^{d}$, maximal step size $\eta_{\max}$,
restart period $M$, total number of iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a mini‑batch $\xi_t$ and compute stochastic gradient
          $g_t=\nabla f(w_t;\xi_t)$.
    \State Compute learning rate
          \[
          \eta_t\gets\frac{\eta_{\max}}{2}\Bigl(1+\cos\Bigl(\pi\frac{t\bmod M}{M}\Bigr)\Bigr).
          \]
    \State Update parameters:
          \[
          w_{t+1}\gets w_t-\eta_t\,g_t.
          \]
\EndFor
\State \Return $w_T$ (or optionally the averaged iterate
          $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, $\eta_{\max}=0.1$, restart period $M=5$ and run three
iterations.  The cosine schedule (2) yields

\[
\begin{array}{c|c|c}
t & \displaystyle\eta_t
     =\tfrac{\eta_{\max}}{2}\bigl(1+\cos(\pi t/M)\bigr) & 
\displaystyle\text{numerical value}\\ \hline
0 & \tfrac{0.1}{2}(1+\cos 0)=0.1 & 0.1000\\
1 & \tfrac{0.1}{2}\bigl(1+\cos(\pi/5)\bigr)
   \approx0.09045 & 0.09045\\
2 & \tfrac{0.1}{2}\bigl(1+\cos(2\pi/5)\bigr)
   \approx0.06545 & 0.06545
\end{array}
\]

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall w,v\in\R^{d}:\qquad
f(v)\le f(w)+\ip{\nabla f(w)}{v-w}+\frac{L}{2}\norm{v-w}^{2},
\]
for some constant $L>0$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, conditioned on $w_t$,
\[
\E\bigl[g_t\mid w_t\bigr]=\nabla f(w_t).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^{2}\ge0$ such that
\[
\E\bigl[\norm{g_t-\nabla f(w_t)}^{2}\mid w_t\bigr]\le\sigma^{2},
\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]\label{ass:gradbound}
There is $G>0$ with $\norm{\nabla f(w)}\le G$ for all $w$.
\end{assumption}
All constants $L,\sigma,G$ are assumed known and independent of $T$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR]\label{thm:sgdr}
Let $\{w_t\}_{t=0}^{T}$ be generated by the SGDR iteration (1)–(2) under
Assumptions~\ref{ass:smooth}–\ref{ass:gradbound}. Define the average
iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$.
Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T/2}}_{\displaystyle O\!\left(\frac{1}{T}\right)}
\;+\;
\underbrace{L\eta_{\max}\sigma^{2}}_{\displaystyle O(\eta_{\max})},
\tag{3}
\]
where $f^{\star}=\inf_{w}f(w)$. Consequently, if the maximal step size
is chosen as $\eta_{\max}=c/\sqrt{T}$ for some $c>0$, we obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\tag{4}
\]
\end{theorem}
The theorem shows that SGDR attains the same $O(1/\sqrt{T})$ rate as
classical SGD while enjoying periodic learning‑rate resets.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability.}
The cosine schedule stays within $[0,\eta_{\max}]$ and its average over a
full cycle equals $\eta_{\max}/2$. Hence the recursion (1) never uses a step
larger than $\eta_{\max}$, guaranteeing stability under the same
conditions as ordinary SGD with constant step size $\eta_{\max}$.

\item \textbf{Hyper‑parameter Sensitivity.}
The period $M$ only influences the variance of the step size but not the
asymptotic bound (3) because $\frac1M\sum_{t=0}^{M-1}\eta_t=\eta_{\max}/2$.
Thus SGDR is robust to the exact choice of $M$, whereas the maximal
learning rate $\eta_{\max}$ directly controls the bias–variance trade‑off
as reflected in (3).

\item \textbf{Comparison to Baselines.}
For plain SGD with a constant step size $\eta$, the standard bound reads
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}+L\eta\sigma^{2}.
\]
Replacing $\eta$ by the average learning rate $\eta_{\max}/2$ gives exactly
the bound (3).  Hence SGDR inherits the optimal $O(1/\sqrt{T})$ rate while
potentially escaping shallow local minima thanks to the periodic resets.

\end{enumerate}

\section*{Discussion}
The proof of Theorem~\ref{thm:sgdr} follows the classic SGD analysis,
the only modification being the replacement of a constant step size by the
time‑varying cosine schedule. Because the schedule is deterministic and
bounded, the same telescoping argument works after averaging over a cycle.
The result therefore demonstrates that warm restarts do not degrade the
theoretical guarantees for non‑convex smooth optimization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth (Assumption~\ref{ass:smooth}), then for any $w$ and
any stochastic gradient $g$,
\[
f(w-\eta g)\le f(w)-\eta\ip{\nabla f(w)}{g}
+\frac{L\eta^{2}}{2}\norm{g}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply the definition of $L$‑smoothness with $v=w-\eta g$ and rearrange.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:sgdr}.  All expectations are taken with respect to the
randomness of the stochastic gradients $\{g_t\}$.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $w=w_t$, $g=g_t$ and
$\eta=\eta_t$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\ip{\nabla f(w_t)}{g_t}
+\frac{L\eta_t^{2}}{2}\norm{g_t}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Take conditional expectation w.r.t.\ $\xi_t$
( i.e.\ given $w_t$) and use unbiasedness (Assumption~\ref{ass:unbiased}):
\[
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\eta_t\norm{\nabla f(w_t)}^{2}
+\frac{L\eta_t^{2}}{2}\E\!\bigl[\norm{g_t}^{2}\mid w_t\bigr].
\tag{7}
\]

\noindent\textbf{[Step 3]} Expand $\norm{g_t}^{2}$ using variance decomposition:
\[
\E\!\bigl[\norm{g_t}^{2}\mid w_t\bigr]
=\norm{\nabla f(w_t)}^{2}
+\E\!\bigl[\norm{g_t-\nabla f(w_t)}^{2}\mid w_t\bigr]
\le\norm{\nabla f(w_t)}^{2}+\sigma^{2},
\tag{8}
\]
where the inequality follows from Assumption~\ref{ass:variance}.

\noindent\textbf{[Step 4]} Substitute (8) into (7):
\[
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t)-\eta_t\norm{\nabla f(w_t)}^{2}
+\frac{L\eta_t^{2}}{2}\bigl(\norm{\nabla f(w_t)}^{2}+\sigma^{2}\bigr).
\tag{9}
\]

\noindent\textbf{[Step 5]} Rearrange (9) to isolate the gradient term:
\[
\bigl(\eta_t-\tfrac{L\eta_t^{2}}{2}\bigr)\norm{\nabla f(w_t)}^{2}
\le f(w_t)-\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
+\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{10}
\]

\noindent\textbf{[Step 6]} Because $0\le\eta_t\le\eta_{\max}$ and
$\eta_{\max}\le\frac{1}{L}$ (a standard step‑size restriction; we may
choose $\eta_{\max}$ to satisfy it), we have
\[
\eta_t-\frac{L\eta_t^{2}}{2}\;\ge\;\frac{\eta_t}{2}.
\tag{11}
\]
Thus (10) yields
\[
\frac{\eta_t}{2}\norm{\nabla f(w_t)}^{2}
\le f(w_t)-\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
+\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 7]} Take total expectation on both sides of (12):
\[
\frac{\eta_t}{2}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le \E\!\bigl[f(w_t)\bigr]-\E\!\bigl[f(w_{t+1})\bigr]
+\frac{L\eta_t^{2}}{2}\sigma^{2}.
\tag{13}
\]

\noindent\textbf{[Step 8]} Sum (13) over $t=0,\dots,T-1$.
The telescoping sum on the right‑hand side collapses:
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-\E\!\bigl[f(w_T)\bigr]
+\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{14}
\]

\noindent\textbf{[Step 9]} Since $f(w_T)\ge f^{\star}$,
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-f^{\star}
+\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{15}
\]

\noindent\textbf{[Step 10]} Evaluate the sums of the learning rates.
From the definition (2) the schedule is periodic with period $M$ and
\[
\frac{1}{M}\sum_{k=0}^{M-1}\eta_k
= \frac{\eta_{\max}}{2},
\qquad
\frac{1}{M}\sum_{k=0}^{M-1}\eta_k^{2}
\le \eta_{\max}^{2}.
\]
Hence for any $T$,
\[
\sum_{t=0}^{T-1}\eta_t \;\ge\; \frac{\eta_{\max}}{2}\,T,
\qquad
\sum_{t=0}^{T-1}\eta_t^{2}\;\le\; \eta_{\max}^{2}\,T.
\tag{16}
\]

\noindent\textbf{[Step 11]} Plug (16) into (15):
\[
\frac{\eta_{\max}}{4}\,T\;
\cdot\;
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-f^{\star}
+\frac{L\sigma^{2}}{2}\,\eta_{\max}^{2}T.
\tag{17}
\]

\noindent\textbf{[Step 12]} Divide both sides of (17) by $\frac{\eta_{\max}}{4}\,T$,
which yields the desired bound (3):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T/2}
+L\eta_{\max}\sigma^{2}.
\tag{18}
\]

\noindent\textbf{[Step 13]} Choose $\eta_{\max}=c/\sqrt{T}$ with a constant
$c>0$ satisfying $c/\sqrt{T}\le 1/L$ for all $T$ of interest.  Substituting
into (18) gives (4):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (18) we can write the explicit constant‑level bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le
\underbrace{\frac{4\bigl(f(w_0)-f^{\star}\bigr)}{\eta_{\max}T}}_{\displaystyle
\text{bias term}}
\;+\;
\underbrace{L\eta_{\max}\sigma^{2}}_{\displaystyle\text{variance term}}.
\]
If we set $\eta_{\max}=c/\sqrt{T}$, the bias term becomes
$\frac{4\bigl(f(w_0)-f^{\star}\bigr)}{c}\,T^{-1/2}$ and the variance term
is $L c\sigma^{2}T^{-1/2}$, confirming the $O(T^{-1/2})$ rate.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Deterministic Gradient (σ=0).}
When exact gradients are available, the variance term disappears and (18)
reduces to a pure $O(1/T)$ decay of the average squared gradient norm.
\item \textbf{Very Long Restart Period ($M\to\infty$).}
The cosine schedule degenerates to a simple linear decay from $\eta_{\max}$
to $0$, and the bound (3) remains unchanged because only the average value
of $\eta_t$ matters.
\item \textbf{Constant Step Size ($\eta_t\equiv\eta_{\max}$).}
This is a special case of SGDR where the cosine factor is always $1$.
The bound (3) becomes the classic SGD bound with constant step size.
\end{enumerate}

\section*{Conclusion}
We have presented a complete mathematical description of SGDR, illustrated its
behaviour on a toy problem, and established a rigorous convergence guarantee
for non‑convex smooth objectives.  The proof shows that the warm‑restart
cosine schedule does not affect the asymptotic $O(1/\sqrt{T})$ rate of SGD,
while in practice it often yields faster empirical progress by periodically
re‑energising the optimization dynamics.
