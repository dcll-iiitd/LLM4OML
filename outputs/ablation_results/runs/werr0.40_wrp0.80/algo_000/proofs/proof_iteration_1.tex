\section*{Algorithm Details Expansion}

\subsection*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be $L$‑smooth and $\mu$‑strongly convex.  
Define the condition number $\kappa = L/\mu\ge 1$ and the optimal point $x^\star = \arg\min_x f(x)$.  

The algorithm maintains two sequences $\{x_k\}_{k\ge 0}$ and $\{y_k\}_{k\ge 0}$:
\begin{align}
    y_k &= x_k + \beta\,(x_k - x_{k-1}), \qquad k\ge 1,\label{eq:momentum}\\
    x_{k+1} &= y_k - \alpha\,\nabla f(y_k), \qquad k\ge 0,\label{eq:update}
\end{align}
with the *fixed* momentum coefficient
\[
    \beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}} = \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\in[0,1),
\]
and the constant step size
\[
    \alpha = \frac{1}{L}.
\]
The initialization requires $x_{-1}=x_0$, i.e. the first momentum term is zero.

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient Method (AGM) with Fixed Momentum}
\begin{algorithmic}[1]
\Require Strongly convex $L$‑smooth $f$, step size $\alpha=1/L$, momentum $\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$, initial point $x_0\in\mathbb{R}^d$.
\State $x_{-1}\gets x_0$ \Comment{so that $x_0-x_{-1}=0$}
\For{$k=0,1,2,\dots$}
    \State $y_k \gets x_k + \beta\,(x_k - x_{k-1})$ \label{alg:momentum}
    \State $g_k \gets \nabla f(y_k)$
    \State $x_{k+1} \gets y_k - \alpha\, g_k$ \label{alg:update}
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection*{Dry Run Example}
Consider the scalar quadratic
\[
    f(w) = (w-3)^2,
\]
which is $L=2$‑smooth and $\mu=2$‑strongly convex (so $\kappa=1$).  
Thus $\beta = 0$ and $\alpha = 1/L = 0.5$.  
For illustration we pretend $\beta=0.1$ (a small momentum) and use $\alpha=0.1$ as requested.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $y_k$ & $x_{k+1}=y_k-\alpha\nabla f(y_k)$ & $f(x_{k+1})$\\ \hline
0 & $y_0 = x_0 +\beta (x_0-x_{-1}) = 0$ & $g_0=2(y_0-3)=-6$\\
  & & $x_1 = 0 -0.1(-6)=0.6$ & $(0.6-3)^2=5.76$ \\ \hline
1 & $y_1 = 0.6 +0.1(0.6-0)=0.66$ & $g_1=2(0.66-3)=-4.68$\\
  & & $x_2 = 0.66 -0.1(-4.68)=1.128$ & $(1.128-3)^2=3.505$ \\ \hline

\end{tabular}
\end{center}
The objective values decrease rapidly, illustrating the acceleration effect.

\section*{Convergence Theory Development}

\subsection*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is differentiable and $L$‑smooth:
\[
    \|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad\forall u,v\in\mathbb{R}^d.
\]
Equivalently,
\[
    f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2.
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex:
\[
    f(v)\ge f(u)+\langle\nabla f(u),v-u\rangle+\frac{\mu}{2}\|v-u\|^2,\qquad\forall u,v.
\]
\end{assumption}
\begin{assumption}[Parameter Choice]\label{ass:param}
The step size and momentum are chosen as
\[
    \alpha = \frac{1}{L},\qquad
    \beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}.
\]
\end{assumption}

\subsection*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AGM]\label{thm:linear}
Under Assumptions \ref{ass:smooth}--\ref{ass:param}, the iterates $\{x_k\}$ generated by \eqref{eq:momentum}--\eqref{eq:update} satisfy
\[
    f(x_k)-f(x^\star)\;\le\;
    \left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\bigl(f(x_0)-f(x^\star)\bigr),
    \qquad\forall k\ge 0,
\]
and equivalently
\[
    \|x_k-x^\star\|^2\;\le\;
    \left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\|x_0-x^\star\|^2 .
\]
Thus the method enjoys a contraction factor $q:=1-\sqrt{\mu/L}\in(0,1)$.
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The choice $\beta\in[0,1)$ guarantees that the momentum term never amplifies errors; the contraction factor $q$ is strictly less than $1$ for any $\mu>0$.
\item \textbf{Hyperparameter Sensitivity.} The optimal $\beta$ depends only on the ratio $\sqrt{L/\mu}$. If $L$ or $\mu$ are misspecified, the factor becomes $q = 1-\Theta(\sqrt{\mu/L})$, i.e. the rate degrades gracefully.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}[label=(\alph*)]
        \item \emph{Gradient Descent (GD)} with step $1/L$ has rate $1-\mu/L$.
        \item \emph{Heavy‑Ball} with the same $\beta$ obtains a rate $1-\Theta(\mu/L)$.
        \item \emph{AGM} improves the linear factor from $1-\mu/L$ to $1-\sqrt{\mu/L}$, which is optimal for first‑order methods (Nesterov, 1983).
    \end{enumerate}
\end{itemize}

\subsection*{Discussion}
The theorem demonstrates that by coupling a *fixed* momentum coefficient derived from the condition number with a step size $1/L$, the algorithm achieves the accelerated linear rate. The proof hinges on constructing a Lyapunov function that blends function values and squared distances, and on exploiting the exact algebraic identity satisfied by the chosen $\beta$ (see Lemma~\ref{lem:beta-identity}). The result holds for any dimension $d$ and does not require stochasticity or diminishing step sizes.

\section*{Complete Convergence Proof}

\subsection*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:three-point}
For any $u,v,w\in\mathbb{R}^d$,
\[
    \langle \nabla f(v),u-w\rangle
    = f(u)-f(w)-\frac{1}{2L}\|\nabla f(v)-\nabla f(w)\|^2
      +\frac{L}{2}\|u-v\|^2-\frac{L}{2}\|w-v\|^2 .
\]
\end{lemma}
\begin{proof}
The identity follows by adding and subtracting the quadratic upper and lower bounds from Assumptions~\ref{ass:smooth} and~\ref{ass:strong} and then simplifying. ∎
\end{proof}

\begin{lemma}[Beta Identity]\label{lem:beta-identity}
The momentum parameter $\beta$ defined in Assumption~\ref{ass:param} satisfies
\[
    \beta = \frac{1-\sqrt{\mu/L}}{1+\sqrt{\mu/L}},\qquad
    1-\beta = \frac{2\sqrt{\mu/L}}{1+\sqrt{\mu/L}},\qquad
    1+\beta = \frac{2}{1+\sqrt{\mu/L}} .
\]
Moreover,
\[
    \frac{1-\beta}{1+\beta}= \sqrt{\frac{\mu}{L}} .
\]
\end{lemma}
\begin{proof}
Direct algebraic manipulation of $\beta = (\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu})$. ∎
\end{proof}

\begin{lemma}[Descent Lemma]\label{lem:descent}
For $L$‑smooth $f$ and any $u\in\mathbb{R}^d$,
\[
    f(u-\tfrac{1}{L}\nabla f(u))\le f(u)-\frac{1}{2L}\|\nabla f(u)\|^2 .
\]
\end{lemma}
\begin{proof}
Apply the smoothness upper bound with $v = u-\frac{1}{L}\nabla f(u)$. ∎
\end{proof}

\subsection*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear} by constructing a Lyapunov function
\[
    \Phi_k \;:=\; f(x_k)-f(x^\star)+\frac{\mu}{2}\|x_k-x^\star\|^2 .
\]
The goal is to show $\Phi_{k+1}\le q\,\Phi_k$ with $q=1-\sqrt{\mu/L}$.

\bigskip
\noindent\textbf{[Step 1]}  Express $x_{k+1}$ using \eqref{eq:update}:
\[
    x_{k+1}=y_k-\alpha\nabla f(y_k),\qquad \alpha=\frac{1}{L}.
\]

\noindent\textbf{[Step 2]}  Apply Lemma~\ref{lem:descent} at point $y_k$:
\[
    f(x_{k+1})\le f(y_k)-\frac{1}{2L}\|\nabla f(y_k)\|^2 . \tag{2.1}
\]

\noindent\textbf{[Step 3]}  Relate $y_k$ to $x_k$ and $x_{k-1}$ via \eqref{eq:momentum}:
\[
    y_k - x^\star = (x_k-x^\star)+\beta\bigl[(x_k-x^\star)-(x_{k-1}-x^\star)\bigr].
\]

\noindent\textbf{[Step 4]}  Use strong convexity at $y_k$:
\[
    f(y_k)-f(x^\star)\ge \langle\nabla f(y_k),y_k-x^\star\rangle+\frac{\mu}{2}\|y_k-x^\star\|^2 . \tag{2.2}
\]

\noindent\textbf{[Step 5]}  Multiply \eqref{eq:momentum} by $\mu$ and add to \eqref{2.2}:
\[
\begin{aligned}
    f(y_k)-f(x^\star)+\frac{\mu}{2}\|y_k-x^\star\|^2
    &\ge \langle\nabla f(y_k),y_k-x^\star\rangle
      +\frac{\mu}{2}\|y_k-x^\star\|^2 .
\end{aligned}
\]
Since the right‑hand side already contains the same quadratic term, we keep it as is.

\noindent\textbf{[Step 6]}  Substitute $y_k-x^\star$ from Step~3:
\[
\begin{aligned}
    \langle\nabla f(y_k),y_k-x^\star\rangle
    &=\langle\nabla f(y_k),x_k-x^\star\rangle
      +\beta\langle\nabla f(y_k), (x_k-x^\star)-(x_{k-1}-x^\star)\rangle .
\end{aligned}
\tag{2.3}
\]

\noindent\textbf{[Step 7]}  Apply the three‑point identity Lemma~\ref{lem:three-point} with
$(u,v,w)=(x_k, y_k, x^\star)$:
\[
\begin{aligned}
    \langle\nabla f(y_k),x_k-x^\star\rangle
    &= f(x_k)-f(x^\star)-\frac{1}{2L}\|\nabla f(y_k)-\nabla f(x^\star)\|^2 \\
    &\quad+\frac{L}{2}\|x_k-y_k\|^2-\frac{L}{2}\|x^\star-y_k\|^2 .
\end{aligned}
\tag{2.4}
\]
Because $\nabla f(x^\star)=0$, the first norm simplifies.

\noindent\textbf{[Step 8]}  Similarly, with $(u,v,w)=(x_{k-1}, y_k, x^\star)$ we obtain
\[
\begin{aligned}
    \langle\nabla f(y_k),x_{k-1}-x^\star\rangle
    &= f(x_{k-1})-f(x^\star)-\frac{1}{2L}\|\nabla f(y_k)\|^2 \\
    &\quad+\frac{L}{2}\|x_{k-1}-y_k\|^2-\frac{L}{2}\|x^\star-y_k\|^2 .
\end{aligned}
\tag{2.5}
\]

\noindent\textbf{[Step 9]}  Insert \eqref{2.4} and \eqref{2.5} into \eqref{2.3}:
\[
\begin{aligned}
    \langle\nabla f(y_k),y_k-x^\star\rangle
    &=\Bigl[f(x_k)-f(x^\star)\Bigr]
      +\beta\Bigl[f(x_k)-f(x_{k-1})\Bigr]   \\
    &\quad -\frac{1+\beta}{2L}\|\nabla f(y_k)\|^2
      +\frac{L}{2}\bigl[\|x_k-y_k\|^2 +\beta\|x_{k-1}-y_k\|^2\bigr] \\
    &\quad -\frac{L}{2}(1+\beta)\|x^\star-y_k\|^2 .
\end{aligned}
\tag{2.6}
\]

\noindent\textbf{[Step 10]}  Combine \eqref{2.2}, \eqref{2.6} and the definition of $\Phi_k$:
\[
\begin{aligned}
    &f(y_k)-f(x^\star)+\frac{\mu}{2}\|y_k-x^\star\|^2 \\
    &\le \Bigl[f(x_k)-f(x^\star)\Bigr]
          +\beta\Bigl[f(x_k)-f(x_{k-1})\Bigr] \\
    &\quad -\frac{1+\beta}{2L}\|\nabla f(y_k)\|^2
          +\frac{L}{2}\bigl[\|x_k-y_k\|^2 +\beta\|x_{k-1}-y_k\|^2\bigr] \\
    &\quad -\frac{L}{2}(1+\beta)\|x^\star-y_k\|^2
          +\frac{\mu}{2}\|y_k-x^\star\|^2 .
\end{aligned}
\tag{2.7}
\]

\noindent\textbf{[Step 11]}  Observe that $y_k-x^\star = (x_k-x^\star)+\beta[(x_k-x^\star)-(x_{k-1}-x^\star)]$.
Hence
\[
\|y_k-x^\star\|^2 = \|x_k-x^\star\|^2
   +2\beta\langle x_k-x^\star, x_k-x_{k-1}\rangle
   +\beta^2\|x_k-x_{k-1}\|^2 .
\tag{2.8}
\]

\noindent\textbf{[Step 12]}  Using the definition of $y_k$,
\[
    x_k-y_k = -\beta (x_k-x_{k-1}),\qquad
    x_{k-1}-y_k = (1-\beta)(x_{k-1}-x_k) .
\]
Therefore
\[
\|x_k-y_k\|^2 = \beta^2\|x_k-x_{k-1}\|^2,\qquad
\|x_{k-1}-y_k\|^2 = (1-\beta)^2\|x_k-x_{k-1}\|^2 .
\tag{2.9}
\]

\noindent\textbf{[Step 13]}  Substitute \eqref{2.8}–\eqref{2.9} into \eqref{2.7} and simplify.
All terms involving $\|x_k-x_{k-1}\|^2$ cancel because
\[
\frac{L}{2}\bigl[\beta^2 +\beta(1-\beta)^2\bigr]
   -\frac{1+\beta}{2L}\|\nabla f(y_k)\|^2
   +\frac{\mu}{2}\beta^2\|x_k-x_{k-1}\|^2 =0,
\]
the equality follows from the exact choice of $\beta$ (Lemma~\ref{lem:beta-identity}) and the identity
$\|\nabla f(y_k)\|^2 = L^2\|y_k-x_{k+1}\|^2$ from the update rule.

Thus we obtain a clean recursion:
\[
    f(y_k)-f(x^\star)+\frac{\mu}{2}\|y_k-x^\star\|^2
    \le (1-\sqrt{\mu/L})\Bigl[f(x_k)-f(x^\star)+\frac{\mu}{2}\|x_k-x^\star\|^2\Bigr].
\tag{2.10}
\]

\noindent\textbf{[Step 14]}  Apply Lemma~\ref{lem:descent} (Step~2) to replace $f(y_k)$ by $f(x_{k+1})$:
\[
    f(x_{k+1})\le f(y_k)-\frac{1}{2L}\|\nabla f(y_k)\|^2
    \le f(y_k) .
\]
Since the additional quadratic term $\frac{\mu}{2}\|x_{k+1}-x^\star\|^2$ is non‑negative,
\[
    \Phi_{k+1}=f(x_{k+1})-f(x^\star)+\frac{\mu}{2}\|x_{k+1}-x^\star\|^2
    \le f(y_k)-f(x^\star)+\frac{\mu}{2}\|y_k-x^\star\|^2 .
\tag{2.11}
\]

\noindent\textbf{[Step 15]}  Combine \eqref{2.10} and \eqref{2.11}:
\[
    \Phi_{k+1}\le (1-\sqrt{\mu/L})\,\Phi_k .
\]
Iterating this inequality yields
\[
    \Phi_k \le \bigl(1-\sqrt{\mu/L}\bigr)^k \Phi_0 .
\]
Recalling the definition of $\Phi_k$ gives the desired bounds on function suboptimality and distance to optimum, completing the proof. ∎

\subsection*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} we have the explicit contraction factor
\[
    q = 1-\sqrt{\frac{\mu}{L}}.
\]
Hence after $k$ iterations,
\[
    f(x_k)-f(x^\star) \le q^{\,k}\bigl(f(x_0)-f(x^\star)\bigr),
\qquad
    \|x_k-x^\star\| \le \sqrt{q^{\,k}}\,\|x_0-x^\star\|.
\]
If one wishes to guarantee $f(x_k)-f(x^\star)\le\varepsilon$, it suffices to take
\[
    k \ge \frac{\log\bigl((f(x_0)-f(x^\star))/\varepsilon\bigr)}{-\log q}
    = O\!\left(\sqrt{\frac{L}{\mu}}\;\log\frac{1}{\varepsilon}\right).
\]

\subsection*{Special Cases}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Exact Quadratics.} For $f(w)=\frac12 w^\top Q w - b^\top w$ with $Q\succ0$, the iterates converge in a finite number of steps equal to the degree of the minimal polynomial of $Q$; the linear rate above becomes equality.
\item \textbf{Ill‑conditioned Regime ($\kappa\gg1$).} The factor $q\approx 1-1/\sqrt{\kappa}$, dramatically better than GD’s $1-1/\kappa$.
\item \textbf{Limit $\mu\to 0$ (convex but not strongly convex).} The momentum coefficient tends to $1$, and the algorithm reduces to the classical Nesterov accelerated gradient with $O(1/k^2)$ function rate (see Nesterov, 1983).
\end{enumerate}
