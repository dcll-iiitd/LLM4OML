\section*{Stochastic Gradient Descent (SGD) for Non-Convex Smooth Functions}

\section*{Mathematical Formulation}
The algorithm iteratively updates the parameters by computing a stochastic gradient using a single randomly sampled data point and taking a step in the negative stochastic gradient direction. Let $w_t \in \mathbb{R}^d$ be the parameter at iteration $t$. The update rule is:
\begin{itemize}
    \item \textbf{Update rule:} $w_{t+1} = w_t - \eta_t g_t$
    \item \textbf{Stochastic gradient:} $g_t = \nabla f(w_t; z_t)$ is a stochastic gradient computed on a random sample $z_t$ at time $t$
    \item \textbf{Learning rate schedule:} Fixed ($\eta_t = \eta$) or decreasing.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Gradient Descent (SGD)}
\begin{algorithmic}[1]
\REQUIRE Initial parameter $w_0 \in \mathbb{R}^d$, learning rate sequence $\{\eta_t\}_{t=0}^{T-1}$, number of iterations $T$
\FOR{$t = 0, 1, \dots, T-1$}
    \STATE Sample data point $z_t$ uniformly at random from the dataset
    \STATE Compute stochastic gradient $g_t = \nabla f(w_t; z_t)$
    \STATE Update parameter $w_{t+1} = w_t - \eta_t g_t$
\ENDFOR
\RETURN $w_{\text{out}}$ chosen uniformly at random from $\{w_0, w_1, \dots, w_{T-1}\}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the non-convex function $f(w) = -(w-3)^2$ (note: $(w-3)^2$ is convex, but the task specifies non-convex and smooth functions. We apply SGD to $f(w)=-(w-3)^2$ which is non-convex and smooth). 
Let $w_0 = 0$, $\eta = 0.1$, and 3 iterations. The true gradient is $\nabla f(w) = -2(w-3)$. We assume the sampled data point yields the true gradient (i.e., $g_t = \nabla f(w_t)$ for this deterministic dry run).

\textbf{Iteration 1:}
\begin{itemize}
    \item Gradient: $g_0 = -2(0 - 3) = 6$
    \item Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1(6) = -0.6$
    \item Objective: $f(w_1) = -(-0.6 - 3)^2 = -12.96$
\end{itemize}

\textbf{Iteration 2:}
\begin{itemize}
    \item Gradient: $g_1 = -2(-0.6 - 3) = 7.2$
    \item Update: $w_2 = w_1 - \eta g_1 = -0.6 - 0.1(7.2) = -1.32$
    \item Objective: $f(w_2) = -(-1.32 - 3)^2 = -18.6624$
\end{itemize}

\textbf{Iteration 3:}
\begin{itemize}
    \item Gradient: $g_2 = -2(-1.32 - 3) = 8.64$
    \item Update: $w_3 = w_2 - \eta g_2 = -1.32 - 0.1(8.64) = -2.184$
    \item Objective: $f(w_3) = -(-2.184 - 3)^2 = -25.9336$
\end{itemize}

\section*{Assumptions}
\begin{definition}
We make the following formal assumptions:
\end{definition}
\begin{enumerate}
    \item \textbf{Non-convexity and Smoothness:} The objective function $f(w)$ is differentiable and $L$-smooth, meaning for all $w, w' \in \mathbb{R}^d$:
    $$\|\nabla f(w) - \nabla f(w')\| \leq L \|w - w'\|$$
    Equivalently, this implies:
    $$f(w') \leq f(w) + \langle \nabla f(w), w' - w \rangle + \frac{L}{2} \|w' - w\|^2$$
    \item \textbf{Unbiased Stochastic Gradients:} The stochastic gradient $g_t$ is an unbiased estimator of the true gradient:
    $$\mathbb{E}[g_t | w_t] = \nabla f(w_t)$$
    \item \textbf{Bounded Variance:} The variance of the stochastic gradient is bounded:
    $$\mathbb{E}[\|g_t - \nabla f(w_t)\|^2 | w_t] \leq \sigma^2$$
    \item \textbf{Bounded Initial Suboptimality:} The initial parameter $w_0$ satisfies:
    $$f(w_0) - f(w^*) \leq \Delta$$
    where $w^* = \arg\min_w f(w)$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
Under the assumptions stated above, and using a fixed learning rate $\eta_t = \eta = \frac{1}{L}$ for $t = 0, 1, \dots, T-1$, Stochastic Gradient Descent satisfies:
$$\frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2L(f(w_0) - f(w^*))}{T} + \frac{L\sigma^2}{T}$$
More generally, for any fixed learning rate $\eta \leq \frac{1}{L}$:
$$\frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2\Delta}{\eta T} + L\sigma^2 \eta$$
By choosing $\eta = \frac{1}{\sqrt{T}}$ (assuming $T \geq L^2$), the convergence rate to a stationary point is:
$$\frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq O\left(\frac{1}{\sqrt{T}}\right)$$
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The $L$-smoothness assumption guarantees that the function value cannot increase arbitrarily in a single step, provided the learning rate is sufficiently small ($\eta \leq 1/L$).
    \item \textbf{Hyperparameter Sensitivity:} The convergence bound reveals a fundamental trade-off. The term $\frac{2\Delta}{\eta T}$ decreases with larger $\eta$, while the term $L\sigma^2 \eta$ increases with larger $\eta$. This reflects the bias-variance trade-off inherent in SGD.
    \item \textbf{Comparison to Baselines:} For non-convex functions, standard Gradient Descent (GD) achieves a rate of $O(1/T)$ without the variance term $\sigma^2$. SGD suffers from an $O(1/\sqrt{T})$ rate due to gradient noise, but enjoys a per-iteration computational cost of $O(1)$ compared to $O(N)$ for GD, making it advantageous for large datasets.
\end{itemize}

\section*{Discussion}
The convergence criterion for non-convex functions is finding an $\epsilon$-approximate stationary point, defined as $\mathbb{E}[\|\nabla f(w)\|^2] \leq \epsilon$. The theorem shows that SGD requires $O(1/\epsilon^2)$ iterations to reach this, which matches the information-theoretic lower bound for first-order methods in the non-convex stochastic setting.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]
If $f(w)$ is $L$-smooth, then for any $w_{t+1} = w_t - \eta g_t$:
$$f(w_{t+1}) \leq f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L \eta^2}{2} \|g_t\|^2$$
\end{lemma}
\begin{proof}
Directly applying the definition of $L$-smoothness with $w' = w_{t+1}$ and $w = w_t$:
$$f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t), w_{t+1} - w_t \rangle + \frac{L}{2} \|w_{t+1} - w_t\|^2$$
Substituting $w_{t+1} - w_t = -\eta g_t$:
$$f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t), -\eta g_t \rangle + \frac{L}{2} \|-\eta g_t\|^2$$
$$f(w_{t+1}) \leq f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L \eta^2}{2} \|g_t\|^2$$
\end{proof}

\begin{lemma}[Variance Decomposition]
For any random vector $X$ with $\mathbb{E}[X] = \mu$, we have $\mathbb{E}[\|X\|^2] = \|\mu\|^2 + \mathbb{E}[\|X - \mu\|^2]$.
\end{lemma}
\begin{proof}
$\mathbb{E}[\|X\|^2] = \mathbb{E}[\|X - \mu + \mu\|^2] = \mathbb{E}[\|X - \mu\|^2 + 2\langle X - \mu, \mu \rangle + \|\mu\|^2]$. Since $\mathbb{E}[X - \mu] = 0$, the cross-term vanishes, yielding the result.
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
We begin with the descent lemma and take the conditional expectation given $w_t$.
\begin{align*}
\mathbb{E}[f(w_{t+1}) | w_t] &\leq \mathbb{E}\left[ f(w_t) - \eta \langle \nabla f(w_t), g_t \rangle + \frac{L \eta^2}{2} \|g_t\|^2 \bigg| w_t \right] \quad \text{[Step 1: Apply Lemma 1]} \\
&= f(w_t) - \eta \langle \nabla f(w_t), \mathbb{E}[g_t | w_t] \rangle + \frac{L \eta^2}{2} \mathbb{E}[\|g_t\|^2 | w_t] \quad \text{[Step 2: Linearity of expectation]} \\
&= f(w_t) - \eta \langle \nabla f(w_t), \nabla f(w_t) \rangle + \frac{L \eta^2}{2} \mathbb{E}[\|g_t\|^2 | w_t] \quad \text{[Step 3: Unbiased assumption]} \\
&= f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \mathbb{E}[\|g_t\|^2 | w_t] \quad \text{[Step 4: Norm property]} \\
&= f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \left( \|\nabla f(w_t)\|^2 + \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 | w_t] \right) \quad \text{[Step 5: Apply Lemma 2]} \\
&\leq f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L \eta^2}{2} \left( \|\nabla f(w_t)\|^2 + \sigma^2 \right) \quad \text{[Step 6: Bounded variance assumption]} \\
&= f(w_t) - \left( \eta - \frac{L \eta^2}{2} \right) \|\nabla f(w_t)\|^2 + \frac{L \eta^2 \sigma^2}{2} \quad \text{[Step 7: Algebraic rearrangement]}
\end{align*}
Rearranging terms to isolate $\|\nabla f(w_t)\|^2$:
\begin{align*}
\left( \eta - \frac{L \eta^2}{2} \right) \|\nabla f(w_t)\|^2 &\leq f(w_t) - \mathbb{E}[f(w_{t+1}) | w_t] + \frac{L \eta^2 \sigma^2}{2} \quad \text{[Step 8: Move expected function decrease]} \\
\left( \eta - \frac{L \eta^2}{2} \right) \mathbb{E}[\|\nabla f(w_t)\|^2] &\leq \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2} \quad \text{[Step 9: Take total expectation via Tower Property]}
\end{align*}
Assuming $\eta \leq \frac{1}{L}$, we have $\eta - \frac{L\eta^2}{2} \geq \frac{\eta}{2}$. Specifically, $\eta - \frac{L\eta^2}{2} = \eta(1 - \frac{L\eta}{2}) \geq \eta(1 - 1/2) = \frac{\eta}{2}$.
\begin{align*}
\frac{\eta}{2} \mathbb{E}[\|\nabla f(w_t)\|^2] &\leq \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2} \quad \text{[Step 10: Bound the coefficient]} \\
\mathbb{E}[\|\nabla f(w_t)\|^2] &\leq \frac{2}{\eta} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L \eta^2 \sigma^2}{2} \right) \quad \text{[Step 11: Divide by }\eta/2] \\
\mathbb{E}[\|\nabla f(w_t)\|^2] &\leq \frac{2}{\eta} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \right) + L \eta \sigma^2 \quad \text{[Step 12: Distribute } 2/\eta]
\end{align*}
Summing this inequality from $t=0$ to $T-1$:
\begin{align*}
\sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] &\leq \frac{2}{\eta} \sum_{t=0}^{T-1} \left( \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] \right) + T L \eta \sigma^2 \quad \text{[Step 13: Summation]} \\
&= \frac{2}{\eta} \left( \mathbb{E}[f(w_0)] - \mathbb{E}[f(w_T)] \right) + T L \eta \sigma^2 \quad \text{[Step 14: Telescoping sum]} \\
&\leq \frac{2}{\eta} \left( f(w_0) - f(w^*) \right) + T L \eta \sigma^2 \quad \text{[Step 15: Lower bound } f(w_T) \text{ by } f(w^*)] \\
&= \frac{2\Delta}{\eta} + T L \eta \sigma^2 \quad \text{[Step 16: Apply initial suboptimality bound]}
\end{align*}

\section*{Rate Derivation (With Constants)}
Dividing the summed inequality by $T$:
$$ \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2\Delta}{\eta T} + L \eta \sigma^2 \quad \text{[Step 17: Average over } T \text{ iterations]} $$
To balance the two terms, we set $\frac{2\Delta}{\eta T} = L \eta \sigma^2$, which yields $\eta^2 = \frac{2\Delta}{L \sigma^2 T}$. Thus, we choose the optimal constant learning rate $\eta = \sqrt{\frac{2\Delta}{L \sigma^2 T}}$. 
Plugging this back into the bound:
$$ \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq 2 \sqrt{\frac{2\Delta L \sigma^2}{T}} = O\left(\frac{1}{\sqrt{T}}\right) \quad \text{[Step 18: Final rate derivation]} $$
Alternatively, as stated in the theorem, choosing $\eta = \frac{1}{\sqrt{T}}$ (assuming $T \geq L^2$ to satisfy $\eta \leq 1/L$) gives:
$$ \frac{1}{T} \sum_{t=0}^{T-1} \mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{2\Delta}{\sqrt{T}} + \frac{L\sigma^2}{\sqrt{T}} = O\left(\frac{1}{\sqrt{T}}\right) \quad \text{[Step 19: Simplified rate derivation]} $$

\section*{Special Cases}
\textbf{Case 1: Zero Variance ($\sigma^2 = 0$).} This corresponds to standard Gradient Descent (GD). The bound becomes $\frac{1}{T} \sum_{t=0}^{T-1} \|\nabla f(w_t)\|^2 \leq \frac{2\Delta}{\eta T}$. By choosing $\eta = \frac{1}{L}$, the rate becomes $O(1/T)$, recovering the standard convergence rate for GD on smooth non-convex functions.

\textbf{Case 2: Convex Functions.} If $f(w)$ is additionally convex, the $O(1/\sqrt{T})$ rate for the average gradient norm still holds, but stronger results can be derived for the function value suboptimality, $f(\bar{w}_T) - f(w^*)$, where $\bar{w}_T$ is the average of the iterates.
