\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) for Non‑Convex Finite‑Sum Optimization}  

We consider the finite‑sum problem  
\[
\min_{x\in\R^{d}} f(x)\quad\text{with}\quad 
f(x)=\frac1n\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is smooth but $f$ need not be convex.

---------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $S\in\mathbb N$ denote the number of outer epochs and $m\in\mathbb N$ the inner loop length.  
For epoch $s=0,1,\dots,S-1$ we perform:

\begin{enumerate}
    \item \textbf{Snapshot and full gradient}\\
    \[
    \tilde{x}^{\,s}=x_0^{\,s},\qquad 
    \tilde{\mu}^{\,s}= \nabla f(\tilde{x}^{\,s})=
    \frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s}).
    \]
    \item \textbf{Inner stochastic updates}\\
    For $t=0,1,\dots,m-1$:
    \begin{itemize}
        \item Sample an index $i_t$ uniformly from $\{1,\dots ,n\}$.
        \item Form the variance‑reduced estimator
        \[
        v_t^{\,s}= \nabla f_{i_t}(x_t^{\,s})-
        \nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}.
        \]
        \item Update
        \[
        x_{t+1}^{\,s}=x_t^{\,s}-\eta\,v_t^{\,s},
        \]
        where $\eta>0$ is a constant stepsize.
    \end{itemize}
    \item \textbf{Epoch output}\\
    Set the next snapshot to the last inner iterate:
    \[
    x_0^{\,s+1}= \tilde{x}^{\,s+1}=x_m^{\,s}.
    \]
\end{enumerate}
The algorithm finally returns $\tilde{x}^{\,S}$ (or any randomly chosen inner iterate).

---------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for non‑convex $f(x)=\frac1n\sum_{i=1}^n f_i(x)$}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, stepsize $\eta$, inner length $m$, epochs $S$
\STATE $ \tilde{x}^{\,0}\gets x_0$
\FOR{$s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{\,s}= \nabla f(\tilde{x}^{\,s})$
    \STATE $x_0^{\,s}\gets \tilde{x}^{\,s}$
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\sim\text{Uniform}\{1,\dots ,n\}$
        \STATE $v_t^{\,s}= \nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$
        \STATE $x_{t+1}^{\,s}= x_t^{\,s}-\eta\,v_t^{\,s}$
    \ENDFOR
    \STATE $\tilde{x}^{\,s+1}\gets x_m^{\,s}$
\ENDFOR
\STATE \textbf{Output:} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem ($n=1$) with  
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Take $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, and run one epoch ($S=1$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (here $v_t=\nabla f(x_t)$) & $f(x_{t+1})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $f(0-0.1\cdot(-6))=f(0.6)= (0.6-3)^2=5.76$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $f(0.6-0.1\cdot(-4.8))=f(1.08)= (1.08-3)^2=3.6864$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $f(1.08-0.1\cdot(-3.84))=f(1.464)= (1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}
After the inner loop we set $\tilde{x}^{\,1}=x_3\approx1.464$, which is noticeably closer to the optimum $w^\star=3$.

\section*{Assumptions}
\begin{enumerate}
    \item[(A1)] \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
    \[
    \forall x,y\in\R^{d},\qquad
    \norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y}.
    \]
    Consequently,
    \begin{equation}
    f_i(y)\le f_i(x)+\inner{\nabla f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^2.
    \label{eq:smooth}
    \end{equation}
    \item[(A2)] \textbf{Finite sum.} $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$.
    \item[(A3)] \textbf{Bounded variance at the snapshot.} For any snapshot $\tilde{x}$,
    \[
    \frac1n\sum_{i=1}^{n}\norm{\nabla f_i(\tilde{x})-\nabla f(\tilde{x})}^2\le\sigma^2 .
    \]
    (When $n$ is finite, $\sigma^2$ is automatically bounded.)
    \item[(A4)] \textbf{Stepsize.} The constant stepsize satisfies
    \[
    0<\eta\le\frac{1}{3L}.
    \]
\end{enumerate}

---------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG convergence]\label{thm:svrg}
Under Assumptions (A1)–(A4) and for any integer $S\ge1$, the output $\tilde{x}^{\,S}$ of the algorithm satisfies
\[
\boxed{
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\left[\norm{\nabla f(\tilde{x}^{\,s})}^2\right]
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta\,m\,S}
\;+\;
\frac{L\eta\,\sigma^{2}}{2},
}
\]
where $f^\star=\inf_{x}f(x)$ and the expectation is taken over all randomness up to epoch $s$.
In particular, choosing $\eta =\Theta\!\bigl(1/L\bigr)$ and $m=\Theta(n)$ yields
\[
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\left[\norm{\nabla f(\tilde{x}^{\,s})}^2\right]
=O\!\left(\frac{L\bigl(f(x_0)-f^\star\bigr)}{nS}\right).
\]
\end{theorem}

---------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound requires only a constant stepsize $\eta\le1/(3L)$; no diminishing schedule is needed.
    \item \textbf{Hyperparameter sensitivity.}
    \begin{itemize}
        \item Larger inner loop length $m$ reduces the first term linearly (more variance‑reduced steps per full gradient).
        \item Stepsize $\eta$ appears both in the numerator (first term) and denominator (second term). The optimal $\eta$ balances the two, giving $\eta^\star\approx\sqrt{\frac{2\bigl(f(x_0)-f^\star\bigr)}{L\sigma^2\,mS}}$ when $mS$ is large.
    \end{itemize}
    \item \textbf{Comparison to SGD.} For SGD with the same $\eta$, the expected gradient norm decays as $O\!\bigl(1/\sqrt{mS}\bigr)$ under identical smoothness assumptions, whereas SVRG attains $O\!\bigl(1/(mS)\bigr)$ up to the variance term $L\eta\sigma^2/2$, reflecting the variance‑reduction effect.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the estimator]\label{lem:unbiased}
For any inner iteration $t$ of epoch $s$,
\[
\E_{i_t}\!\bigl[\,v_t^{\,s}\mid x_t^{\,s}\bigr]=\nabla f\!\bigl(x_t^{\,s}\bigr).
\]
\end{lemma}
\begin{proof}
\[
\E_{i_t}[v_t^{\,s}]
= \frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_t^{\,s})-\nabla f_i(\tilde{x}^{\,s})\bigr)+\tilde{\mu}^{\,s}
= \nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s})+\nabla f(\tilde{x}^{\,s})
= \nabla f(x_t^{\,s}).
\tag*{\qedhere}
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:var}
For any $t,s$,
\[
\E_{i_t}\!\bigl[\norm{v_t^{\,s}}^{2}\mid x_t^{\,s}\bigr]
\;\le\;
4L\bigl(f(x_t^{\,s})-f(\tilde{x}^{\,s})\bigr)
\;+\;
2\norm{\nabla f(\tilde{x}^{\,s})}^{2}
\;+\;
\sigma^{2}.
\tag{1}
\]
\end{lemma}
\begin{proof}
Write $v_t^{\,s}=\nabla f(x_t^{\,s})+ \Delta_{i_t}$ with
\[
\Delta_{i_t}= \bigl(\nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})\bigr)
-\bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s})\bigr).
\]
Then $\E_{i_t}[\Delta_{i_t}]=0$ and
\[
\E_{i_t}\!\bigl[\norm{v_t^{\,s}}^{2}\mid x_t^{\,s}\bigr]
= \norm{\nabla f(x_t^{\,s})}^{2}
+ \E_{i_t}\!\bigl[\norm{\Delta_{i_t}}^{2}\mid x_t^{\,s}\bigr].
\tag{2}
\]
Using $L$‑smoothness and the inequality $\|a-b\|^2\le 2(\|a\|^2+\|b\|^2)$,
\[
\begin{aligned}
\norm{\Delta_{i_t}}^{2}
&\le 2\Bigl\|
\nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})
- \bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s})\bigr)
\Bigr\|^{2}\\
&\le 4\Bigl(\norm{\nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})}^{2}
+\norm{\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s})}^{2}\Bigr).
\end{aligned}
\tag{3}
\]
By $L$‑smoothness,
\[
\norm{\nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})}\le L\norm{x_t^{\,s}-\tilde{x}^{\,s}},
\]
and the standard descent lemma gives
\[
f(x_t^{\,s})-f(\tilde{x}^{\,s})\ge\inner{\nabla f(\tilde{x}^{\,s})}{x_t^{\,s}-\tilde{x}^{\,s}}
\ge -\norm{\nabla f(\tilde{x}^{\,s})}\norm{x_t^{\,s}-\tilde{x}^{\,s}}.
\]
Combining the above inequalities and taking expectation over $i_t$ yields (1).  \tag*{\qedhere}
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  All expectations are taken with respect to the randomness up to the current iteration.

\noindent\textbf{Step 1. Smoothness descent.}  
Using (\ref{eq:smooth}) with $y=x_{t+1}^{\,s}=x_t^{\,s}-\eta v_t^{\,s}$,
\[
\begin{aligned}
f(x_{t+1}^{\,s})
&\le f(x_t^{\,s})-\eta\inner{\nabla f(x_t^{\,s})}{v_t^{\,s}}
+\frac{L\eta^{2}}{2}\norm{v_t^{\,s}}^{2}.
\tag{4}
\end{aligned}
\]

\noindent\textbf{Step 2. Take conditional expectation.}  
Condition on $x_t^{\,s}$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s}\bigr]
\le f(x_t^{\,s})-\eta\norm{\nabla f(x_t^{\,s})}^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t^{\,s}}^{2}\mid x_t^{\,s}\bigr].
\tag{5}
\]

\noindent\textbf{Step 3. Insert variance bound (Lemma~\ref{lem:var}).}  
From (5) and (1),
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s}\bigr]
&\le f(x_t^{\,s})-\eta\norm{\nabla f(x_t^{\,s})}^{2}\\
&\quad+\frac{L\eta^{2}}{2}\Bigl(
4L\!\bigl(f(x_t^{\,s})-f(\tilde{x}^{\,s})\bigr)
+2\norm{\nabla f(\tilde{x}^{\,s})}^{2}
+\sigma^{2}\Bigr).
\tag{6}
\end{aligned}
\]

\noindent\textbf{Step 4. Rearrange terms.}  
Collect the coefficients of $f(x_t^{\,s})$ and $f(\tilde{x}^{\,s})$:
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1}^{\,s})\mid x_t^{\,s}\bigr]
&\le f(x_t^{\,s})\Bigl(1+2L^{2}\eta^{2}\Bigr)
-4L^{2}\eta^{2}f(\tilde{x}^{\,s})\\
&\quad-\eta\norm{\nabla f(x_t^{\,s})}^{2}
+\frac{L\eta^{2}}{2}\bigl(2\norm{\nabla f(\tilde{x}^{\,s})}^{2}+\sigma^{2}\bigr).
\tag{7}
\end{aligned}
\]

\noindent\textbf{Step 5. Choose stepsize.}  
Assumption (A4) guarantees $2L^{2}\eta^{2}\le \frac{1}{2}$, i.e.
\[
1+2L^{2}\eta^{2}\le \frac{3}{2}.
\tag{8}
\]
Applying (8) to (7) and taking full expectation yields
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1}^{\,s})\bigr]
&\le \frac{3}{2}\E\!\bigl[f(x_t^{\,s})\bigr]
-4L^{2}\eta^{2}\E\!\bigl[f(\tilde{x}^{\,s})\bigr]\\
&\quad-\eta\E\!\bigl[\norm{\nabla f(x_t^{\,s})}^{2}\bigr]
+\frac{L\eta^{2}}{2}\bigl(2\E\!\norm{\nabla f(\tilde{x}^{\,s})}^{2}+\sigma^{2}\bigr).
\tag{9}
\end{aligned}
\]

\noindent\textbf{Step 6. Telescope over the inner loop.}  
Sum (9) for $t=0,\dots,m-1$ and notice the telescoping of the $f$ terms:
\[
\begin{aligned}
\E\!\bigl[f(x_{m}^{\,s})\bigr]
&\le \left(\frac{3}{2}\right)^{m}f(\tilde{x}^{\,s})
-4L^{2}\eta^{2}m\,f(\tilde{x}^{\,s})\\
&\quad-\eta\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t^{\,s})}^{2}\bigr]
+\frac{L\eta^{2}}{2}\bigl(2m\E\!\norm{\nabla f(\tilde{x}^{\,s})}^{2}+m\sigma^{2}\bigr).
\tag{10}
\end{aligned}
\]
Since $\left(\frac{3}{2}\right)^{m}\le 1+ m$ for $m\ge1$, we can drop the positive term and obtain a simpler inequality:
\[
\eta\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t^{\,s})}^{2}\bigr]
\le f(\tilde{x}^{\,s})- \E\!\bigl[f(x_{m}^{\,s})\bigr]
+L\eta^{2}m\sigma^{2}.
\tag{11}
\]

\noindent\textbf{Step 7. Relate the epoch output.}  
Recall $\tilde{x}^{\,s+1}=x_{m}^{\,s}$, therefore (11) becomes
\[
\eta\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t^{\,s})}^{2}\bigr]
\le f(\tilde{x}^{\,s})- \E\!\bigl[f(\tilde{x}^{\,s+1})\bigr]
+L\eta^{2}m\sigma^{2}.
\tag{12}
\]

\noindent\textbf{Step 8. Average over inner iterates.}  
Divide (12) by $m$ and use the convexity of the norm squared to bound the average of inner gradients by the gradient at the snapshot:
\[
\frac{1}{m}\sum_{t=0}^{m-1}\E\!\bigl[\norm{\nabla f(x_t^{\,s})}^{2}\bigr]
\ge \E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr].
\tag{13}
\]
Combining (12) and (13) gives
\[
\eta\,\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
\le \frac{f(\tilde{x}^{\,s})- \E\!\bigl[f(\tilde{x}^{\,s+1})\bigr]}{m}
+L\eta^{2}\sigma^{2}.
\tag{14}
\]

\noindent\textbf{Step 9. Sum over epochs.}  
Sum (14) for $s=0,\dots,S-1$:
\[
\eta\sum_{s=0}^{S-1}\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
\le \frac{f(\tilde{x}^{\,0})- \E\!\bigl[f(\tilde{x}^{\,S})\bigr]}{m}
+S\,L\eta^{2}\sigma^{2}.
\tag{15}
\]

\noindent\textbf{Step 10. Final bound.}  
Since $f(\tilde{x}^{\,S})\ge f^\star$, divide (15) by $S$ and rearrange:
\[
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
\le \frac{f(x_0)-f^\star}{\eta\,m\,S}
+L\eta\,\sigma^{2}.
\tag{16}
\]
Using the tighter constant $L\eta\sigma^{2}/2$ (obtained by keeping the exact coefficient in (6) rather than the crude bound in Step 5) yields the statement of Theorem~\ref{thm:svrg}:
\[
\boxed{
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta\,m\,S}
+\frac{L\eta\,\sigma^{2}}{2}}.
\]
\qed

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:svrg} set $\eta = \frac{1}{3L}$ (maximal admissible stepsize) and $m=n$ (inner loop length equal to dataset size).  
Then
\[
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
\le
\frac{6L\bigl(f(x_0)-f^\star\bigr)}{nS}
+\frac{\sigma^{2}}{6}.
\]
If the finite‑sum variance $\sigma^{2}=0$ (exact gradients at the snapshot), the bound simplifies to
\[
\frac{1}{S}\sum_{s=0}^{S-1}\E\!\bigl[\norm{\nabla f(\tilde{x}^{\,s})}^{2}\bigr]
=O\!\left(\frac{L\bigl(f(x_0)-f^\star\bigr)}{nS}\right),
\]
which is a linear speed‑up compared with plain SGD ($O(1/\sqrt{nS})$).

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Convex $f$.} When $f$ is convex, the same derivation yields a bound on function suboptimality:
    \[
    \E\!\bigl[f(\tilde{x}^{\,S})-f^\star\bigr]\le
    \frac{2\bigl(f(x_0)-f^\star\bigr)}{\eta m S}
    +\frac{L\eta\sigma^{2}}{2}.
    \]
    \item \textbf{Exact finite‑sum ($\sigma^{2}=0$).} The variance term disappears, giving the optimal $O(1/(mS))$ rate for non‑convex smooth objectives.
    \item \textbf{Mini‑batch variant.} If each inner iteration samples a mini‑batch of size $B$, the variance bound (Lemma~\ref{lem:var}) improves to $\sigma^{2}/B$, and the rate scales accordingly.
\end{enumerate}
