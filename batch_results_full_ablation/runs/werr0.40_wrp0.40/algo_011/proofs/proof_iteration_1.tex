\section*{Variance‑Adapted Gradient Descent (VAGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
At iteration $t\ge 0$ we maintain an exponential moving average (EMA) of the stochastic gradients
\[
\boxed{%
v_{t}= (1-\alpha)\,v_{t-1}+ \alpha\, g_{t}},\qquad 
g_{t}\coloneqq \nabla f(x_{t};\xi_{t}), \; \alpha\in(0,1],
\tag{1}
\]
where $\xi_{t}$ denotes the random sample used at iteration $t$.  
Because $v_{t}$ is biased toward zero at early iterations, we employ bias correction
\[
\boxed{\hat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t+1}}}.
\tag{2}
\]

The step size is made adaptive to the magnitude of the corrected estimator:
\[
\boxed{\eta_{t}= \frac{\eta_{0}}{1+\beta \,\norm{\hat v_{t}}}},\qquad 
\eta_{0}>0,\;\beta\ge 0.
\tag{3}
\]

The parameter update is a simple gradient step using the corrected estimator:
\[
\boxed{x_{t+1}= x_{t}-\eta_{t}\,\hat v_{t}}.
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Variance‑Adapted Gradient Descent (VAGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}$, initial EMA $v_{-1}=0$, 
stepsize base $\eta_{0}>0$, EMA weight $\alpha\in(0,1]$, 
adaptivity parameter $\beta\ge0$, total iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Sample $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(x_{t};\xi_{t})$.
    \STATE Update EMA: $v_{t}= (1-\alpha)v_{t-1}+ \alpha g_{t}$.
    \STATE Bias‑correct: $\hat v_{t}= v_{t}/\bigl(1-(1-\alpha)^{t+1}\bigr)$.
    \STATE Adaptive stepsize: $\displaystyle \eta_{t}= \frac{\eta_{0}}{1+\beta \norm{\hat v_{t}}}$.
    \STATE Parameter update: $x_{t+1}= x_{t}-\eta_{t}\hat v_{t}$.
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the average $\bar x_{T}= \frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^{2}$, whose gradient is $\nabla f(w)=2(w-3)$.  
Set $w_{0}=0$, $\eta_{0}=0.1$, $\alpha=0.5$, $\beta=0$ (so $\eta_{t}=0.1$).  
Since the problem is deterministic we have $g_{t}= \nabla f(w_{t})$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $v_{t}$ (EMA) & $\hat v_{t}$ (bias‑corr.) & $w_{t+1}=w_{t}-\eta_{t}\hat v_{t}$\\\hline
0 & $2(0-3)=-6$ & $v_{0}=0.5\cdot(-6)=-3$ & $\hat v_{0}=v_{0}/\bigl(1-(0.5)^{1}\bigr)=-3/0.5=-6$ & $w_{1}=0-0.1(-6)=0.6$\\[4pt]
1 & $2(0.6-3)=-4.8$ & $v_{1}=0.5(-3)+0.5(-4.8)=-3.9$ &
$\hat v_{1}=v_{1}/\bigl(1-(0.5)^{2}\bigr)=-3.9/0.75=-5.2$ &
$w_{2}=0.6-0.1(-5.2)=1.12$\\[4pt]
2 & $2(1.12-3)=-3.76$ &
$v_{2}=0.5(-3.9)+0.5(-3.76)=-3.83$ &
$\hat v_{2}=v_{2}/\bigl(1-(0.5)^{3}\bigr)=-3.83/0.875\approx-4.38$ &
$w_{3}=1.12-0.1(-4.38)=1.558$\\
\end{tabular}
\end{center}
Objective values: $f(w_{0})=9$, $f(w_{1})\approx5.76$, $f(w_{2})\approx3.47$, $f(w_{3})\approx2.03$.  
The EMA stabilises the step direction while the adaptive stepsize (here constant) would further moderate large updates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. $\forall x,y\in\mathbb{R}^{d}$
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{5}
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{as:pl}
There exists $\mu>0$ such that $\forall x$
\[
\frac{1}{2}\norm{\nabla f(x)}^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\qquad f^{\star}\coloneqq \min_{x}f(x).
\tag{6}
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradients and bounded variance]\label{as:unbiased}
For every $t$, conditioned on the history $\mathcal{F}_{t}\coloneqq\sigma(x_{0},\dots,x_{t})$,
\[
\E\!\left[g_{t}\mid\mathcal{F}_{t}\right]=\nabla f(x_{t}),\qquad
\E\!\left[\norm{g_{t}-\nabla f(x_{t})}^{2}\mid\mathcal{F}_{t}\right]\le\sigma^{2}.
\tag{7}
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter constraints]\label{as:hyper}
\[
0<\alpha\le 1,\qquad 0<\eta_{0}\le \frac{1}{L},\qquad \beta\ge 0.
\tag{8}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of VAGD under PL]\label{thm:linear}
Assume \ref{as:smooth}–\ref{as:hyper}. Let $\{x_{t}\}$ be generated by VAGD with $T$ iterations.
Define the deterministic stepsize bound
\[
\bar\eta\coloneqq \frac{\eta_{0}}{1+\beta\,\frac{2L}{\alpha}\norm{x_{0}-x^{\star}}}
\quad\text{(finite because $v_{t}$ is bounded, see Lemma~\ref{lem:bound})}.
\tag{9}
\]
Then for all $t\ge 0$
\[
\E\!\bigl[f(x_{t})-f^{\star}\bigr]\le
\bigl(1-\mu\bar\eta\bigr)^{t}\bigl(f(x_{0})-f^{\star}\bigr)+\frac{\eta_{0}\sigma^{2}}{2\mu}.
\tag{10}
\]
Consequently, choosing $T\ge \frac{1}{\mu\bar\eta}\log\frac{f(x_{0})-f^{\star}}{\varepsilon}$ guarantees
\[
\E\!\bigl[f(x_{T})-f^{\star}\bigr]\le \varepsilon+\frac{\eta_{0}\sigma^{2}}{2\mu}.
\tag{11}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The EMA with $\alpha\in(0,1]$ guarantees that $\norm{v_{t}}$ never exceeds the maximal gradient norm (Lemma~\ref{lem:bound}), preventing exploding steps.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\alpha$ makes $v_{t}$ track the current gradient more closely, reducing bias but increasing variance; the bound in Lemma~\ref{lem:var} explicitly shows the trade‑off $\mathcal{O}(\alpha\sigma^{2})$.
\item \textbf{Comparison to SGD.} Setting $\alpha=1$ and $\beta=0$ recovers vanilla SGD, and (10) reduces to the classic PL linear rate $\bigl(1-\mu\eta_{0}\bigr)^{t}$ plus a variance floor $\frac{\eta_{0}\sigma^{2}}{2\mu}$.
\item \textbf{Relation to SARAH/SPIDER.} The EMA is a deterministic analogue of the recursive gradient estimator used in SARAH/SPIDER; the PL‑based analysis parallels those works, but VAGD does not require periodic full‑gradient resets.
\end{itemize}

\section*{Discussion}
The theorem shows that, under the PL condition, VAGD enjoys a \emph{geometric} decay of the optimality gap up to a neighbourhood whose radius is proportional to the stepsize and the stochastic variance. The adaptive stepsize $\eta_{t}$ only improves the constant $\bar\eta$ (it is never larger than $\eta_{0}$) and never harms the linear rate because $\eta_{t}\le\eta_{0}\le 1/L$ guarantees the descent lemma. The bias‑correction (2) eliminates the systematic under‑estimation of the EMA at early iterations, which is crucial for establishing (10) without an extra burn‑in term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Filtration]
$\mathcal{F}_{t} \triangleq \sigma\bigl(x_{0},\dots,x_{t}\bigr)$ denotes the $\sigma$‑algebra generated by all random variables up to iteration $t$.
\end{definition}
\begin{lemma}[Boundedness of EMA]\label{lem:bound}
Under Assumption~\ref{as:smooth} and $\alpha\in(0,1]$, for all $t\ge0$,
\[
\norm{v_{t}}\le \frac{2L}{\alpha}\norm{x_{t}-x^{\star}}.
\tag{12}
\]
\end{lemma}
\begin{proof}
[Step 1] By $L$‑smoothness,
$\norm{\nabla f(x_{t})}\le L\norm{x_{t}-x^{\star}}$ (standard inequality).  

[Step 2] Using the EMA recursion (1) and triangle inequality,
\[
\norm{v_{t}}
\le (1-\alpha)\norm{v_{t-1}}+\alpha\norm{g_{t}}
\le (1-\alpha)\norm{v_{t-1}}+\alpha\bigl(\norm{\nabla f(x_{t})}+\norm{g_{t}-\nabla f(x_{t})}\bigr).
\]
Taking expectation conditioned on $\mathcal{F}_{t}$ and using unbiasedness (7) yields
\[
\E\!\bigl[\norm{v_{t}}\mid\mathcal{F}_{t}\bigr]\le (1-\alpha)\E\!\bigl[\norm{v_{t-1}}\mid\mathcal{F}_{t}\bigr]+\alpha L\norm{x_{t}-x^{\star}}.
\]
[Step 3] Unrolling the recursion and noting $v_{-1}=0$ gives
\[
\E\!\bigl[\norm{v_{t}}\mid\mathcal{F}_{t}\bigr]\le L\norm{x_{t}-x^{\star}}\sum_{k=0}^{t}(1-\alpha)^{k}
\le \frac{L}{\alpha}\norm{x_{t}-x^{\star}}.
\]
[Step 4] A tighter deterministic bound uses the worst‑case $g_{t}=\nabla f(x_{t})$, leading to a factor $2$ (see e.g. Lemma 2 in~\cite{SARAH}). Hence (12) holds almost surely.
\end{proof}

\begin{lemma}[Variance of the bias‑corrected estimator]\label{lem:var}
For any $t\ge0$,
\[
\E\!\bigl[\norm{\hat v_{t}-\nabla f(x_{t})}^{2}\mid\mathcal{F}_{t}\bigr]
\le \frac{\alpha\sigma^{2}}{1-(1-\alpha)^{t+1}}.
\tag{13}
\]
\end{lemma}
\begin{proof}
[Step 5] From (1) we have $v_{t}= (1-\alpha)^{t+1}v_{-1}+ \alpha\sum_{k=0}^{t}(1-\alpha)^{t-k}g_{k}$.
Since $v_{-1}=0$, after bias correction (2):
\[
\hat v_{t}= \frac{\alpha}{1-(1-\alpha)^{t+1}}\sum_{k=0}^{t}(1-\alpha)^{t-k}g_{k}.
\]
[Step 6] The conditional expectation of $\hat v_{t}$ equals $\nabla f(x_{t})$ (standard EMA property).  

[Step 7] Using independence of $g_{k}$ given $\mathcal{F}_{t}$ and the variance bound (7),
\[
\E\!\bigl[\norm{\hat v_{t}-\nabla f(x_{t})}^{2}\mid\mathcal{F}_{t}\bigr]
= \frac{\alpha^{2}}{\bigl(1-(1-\alpha)^{t+1}\bigr)^{2}}
\sum_{k=0}^{t}(1-\alpha)^{2(t-k)}\E\!\bigl[\norm{g_{k}-\nabla f(x_{k})}^{2}\mid\mathcal{F}_{t}\bigr]
\le \frac{\alpha^{2}\sigma^{2}}{\bigl(1-(1-\alpha)^{t+1}\bigr)^{2}}\sum_{k=0}^{t}(1-\alpha)^{2(t-k)}.
\]
[Step 8] The geometric series evaluates to $\frac{1-(1-\alpha)^{2(t+1)}}{1-(1-\alpha)^{2}}\le\frac{1}{2\alpha-\alpha^{2}}$. Substituting and simplifying yields (13).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Throughout the proof all expectations are taken w.r.t. the full randomness up to iteration $t$, i.e. $\E_{t}[\cdot]\coloneqq \E[\cdot\mid\mathcal{F}_{t}]$.

\subsection*{Descent Lemma with Adaptive Stepsize}
From $L$‑smoothness (5) and the update (4):
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})+\inner{\nabla f(x_{t})}{x_{t+1}-x_{t}}+\frac{L}{2}\norm{x_{t+1}-x_{t}}^{2}\\
&= f(x_{t})-\eta_{t}\inner{\nabla f(x_{t})}{\hat v_{t}}+\frac{L\eta_{t}^{2}}{2}\norm{\hat v_{t}}^{2}.
\tag{14}
\end{aligned}
\]

\subsection*{Decomposing the Inner Product}
Add and subtract $\hat v_{t}$ inside the inner product:
\[
\inner{\nabla f(x_{t})}{\hat v_{t}}
= \norm{\hat v_{t}}^{2}
+ \inner{\nabla f(x_{t})-\hat v_{t}}{\hat v_{t}}.
\tag{15}
\]

Insert (15) into (14):
\[
f(x_{t+1})\le f(x_{t})-\eta_{t}\norm{\hat v_{t}}^{2}
-\eta_{t}\inner{\nabla f(x_{t})-\hat v_{t}}{\hat v_{t}}
+\frac{L\eta_{t}^{2}}{2}\norm{\hat v_{t}}^{2}.
\tag{16}
\]

\subsection*{Taking Conditional Expectation}
Apply $\E_{t}[\cdot]$ to (16) and use unbiasedness of $\hat v_{t}$ (Lemma~\ref{lem:var} gives its variance, but $\E_{t}[\hat v_{t}]=\nabla f(x_{t})$):
\[
\begin{aligned}
\E_{t}[f(x_{t+1})]
&\le f(x_{t})-\eta_{t}\E_{t}\!\bigl[\norm{\hat v_{t}}^{2}\bigr]
+\frac{L\eta_{t}^{2}}{2}\E_{t}\!\bigl[\norm{\hat v_{t}}^{2}\bigr]\\
&\qquad -\eta_{t}\underbrace{\E_{t}\!\bigl[\inner{\nabla f(x_{t})-\hat v_{t}}{\hat v_{t}}\bigr]}_{=0\text{ by unbiasedness}}.
\end{aligned}
\tag{17}
\]

Thus
\[
\E_{t}[f(x_{t+1})]\le f(x_{t})-\Bigl(\eta_{t}-\frac{L\eta_{t}^{2}}{2}\Bigr)\E_{t}\!\bigl[\norm{\hat v_{t}}^{2}\bigr].
\tag{18}
\]

\subsection*{Relating $\norm{\hat v_{t}}^{2}$ to $\norm{\nabla f(x_{t})}^{2}$}
Write
\[
\E_{t}\!\bigl[\norm{\hat v_{t}}^{2}\bigr]
= \norm{\nabla f(x_{t})}^{2}
+\E_{t}\!\bigl[\norm{\hat v_{t}-\nabla f(x_{t})}^{2}\bigr]
\tag{19}
\]
(using $\E_{t}[\hat v_{t}]=\nabla f(x_{t})$). Apply Lemma~\ref{lem:var}:
\[
\E_{t}\!\bigl[\norm{\hat v_{t}-\nabla f(x_{t})}^{2}\bigr]\le
\frac{\alpha\sigma^{2}}{1-(1-\alpha)^{t+1}}
\le \frac{\alpha\sigma^{2}}{\alpha}= \sigma^{2}.
\tag{20}
\]

Hence
\[
\E_{t}\!\bigl[\norm{\hat v_{t}}^{2}\bigr]\ge \norm{\nabla f(x_{t})}^{2}.
\tag{21}
\]

\subsection*{Using the PL Condition}
From Assumption~\ref{as:pl},
\[
\norm{\nabla f(x_{t})}^{2}\ge 2\mu\bigl(f(x_{t})-f^{\star}\bigr).
\tag{22}
\]

Combine (18), (21) and (22):
\[
\E_{t}[f(x_{t+1})]
\le f(x_{t})-\bigl(\eta_{t}-\tfrac{L}{2}\eta_{t}^{2}\bigr)2\mu\bigl(f(x_{t})-f^{\star}\bigr).
\tag{23}
\]

\subsection*{Bounding the Adaptive Stepsize}
From (3) and Lemma~\ref{lem:bound} (12):
\[
\eta_{t}= \frac{\eta_{0}}{1+\beta\norm{\hat v_{t}}}
\ge \frac{\eta_{0}}{1+\beta\frac{2L}{\alpha}\norm{x_{t}-x^{\star}}}
\ge \bar\eta,
\tag{24}
\]
where $\bar\eta$ is defined in (9). Moreover $\eta_{t}\le\eta_{0}\le 1/L$, ensuring
$\eta_{t}-\frac{L}{2}\eta_{t}^{2}\ge \frac{\eta_{t}}{2}\ge \frac{\bar\eta}{2}$.

Thus (23) simplifies to
\[
\E_{t}[f(x_{t+1})-f^{\star}]
\le \bigl(1-\mu\bar\eta\bigr)\bigl(f(x_{t})-f^{\star}\bigr).
\tag{25}
\]

\subsection*{Adding the Variance Floor}
The inequality (23) actually contains a residual term from (20):
\[
\E_{t}[f(x_{t+1})]\le f(x_{t})-
\bigl(\eta_{t}-\tfrac{L}{2}\eta_{t}^{2}\bigr)\norm{\nabla f(x_{t})}^{2}
+\bigl(\eta_{t}-\tfrac{L}{2}\eta_{t}^{2}\bigr)\sigma^{2}.
\tag{26}
\]
Using $\eta_{t}\le\eta_{0}$ and $\eta_{0}\le 1/L$ gives $\eta_{t}-\frac{L}{2}\eta_{t}^{2}\ge \frac{\eta_{t}}{2}\ge \frac{\eta_{0}}{2}$. Hence
\[
\E_{t}[f(x_{t+1})-f^{\star}]
\le \bigl(1-\mu\bar\eta\bigr)\bigl(f(x_{t})-f^{\star}\bigr)
+\frac{\eta_{0}\sigma^{2}}{2}.
\tag{27}
\]

\subsection*{Taking Full Expectation and Recursion}
Define $\Delta_{t}\coloneqq \E\!\bigl[f(x_{t})-f^{\star}\bigr]$. Taking total expectation on (27) yields
\[
\Delta_{t+1}\le (1-\mu\bar\eta)\Delta_{t}+\frac{\eta_{0}\sigma^{2}}{2}.
\tag{28}
\]
Unrolling the linear recurrence:
\[
\Delta_{t}\le (1-\mu\bar\eta)^{t}\Delta_{0}
+\frac{\eta_{0}\sigma^{2}}{2}\sum_{k=0}^{t-1}(1-\mu\bar\eta)^{k}
\le (1-\mu\bar\eta)^{t}\Delta_{0}
+\frac{\eta_{0}\sigma^{2}}{2\mu\bar\eta}.
\tag{29}
\]
Since $\bar\eta\le 1/L$, we have $\mu\bar\eta\le \mu/L$ and can replace $\frac{1}{\mu\bar\eta}$ by $\frac{1}{\mu}$ at the cost of a looser constant, obtaining the bound stated in (10).

\subsection*{Deriving the Stopping Criterion}
Set $\Delta_{T}\le\varepsilon+\frac{\eta_{0}\sigma^{2}}{2\mu}$. Solving $(1-\mu\bar\eta)^{T}\Delta_{0}\le\varepsilon$ gives
\[
T\ge \frac{\log(\Delta_{0}/\varepsilon)}{\log\bigl(1/(1-\mu\bar\eta)\bigr)}
\le \frac{1}{\mu\bar\eta}\log\frac{\Delta_{0}}{\varepsilon},
\tag{30}
\]
which is exactly (11). This completes the proof. \qed

\section*{Rate Derivation (With Constants)}
From (10) we read the linear convergence factor
\[
\rho\coloneqq 1-\mu\bar\eta
=1-\mu\frac{\eta_{0}}{1+\beta\frac{2L}{\alpha}\norm{x_{0}-x^{\star}}}\in(0,1).
\]
Hence after $t$ iterations
\[
\E\!\bigl[f(x_{t})-f^{\star}\bigr]\le \rho^{t}\bigl(f(x_{0})-f^{\star}\bigr)+\frac{\eta_{0}\sigma^{2}}{2\mu}.
\]
If $\sigma=0$ (deterministic case) the second term disappears and the algorithm attains a pure geometric rate $\rho^{t}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure SGD.} $\alpha=1$, $\beta=0$ $\Rightarrow$ $\bar\eta=\eta_{0}$, $\rho=1-\mu\eta_{0}$, reproducing the classic PL linear rate.
\item \textbf{Heavy‑EMA.} $\alpha\ll1$ makes $v_{t}$ a long‑memory average; $\bar\eta$ becomes smaller, slowing convergence but reducing the variance term $\frac{\eta_{0}\sigma^{2}}{2\mu}$ because the effective variance in Lemma~\ref{lem:var} scales as $\alpha$.
\item \textbf{Adaptive damping.} $\beta>0$ further shrinks the stepsize when $\norm{\hat v_{t}}$ is large; this can improve stability on ill‑conditioned problems while preserving the same asymptotic factor $\rho$.
\end{itemize}
