\section{Introduction}
This document provides a rigorous proof of convergence for Stochastic Gradient Descent (SGD) applied to $L$-smooth, convex objective functions. The proof explicitly extracts the convergence rate and carefully applies all underlying assumptions.

\section{Assumptions}
We operate under the following standard assumptions:
\begin{itemize}
    \item[\textbf{A1}] (Convexity) The objective function $f: \mathbb{R}^d \to \mathbb{R}$ is convex.
    \item[\textbf{A2}] (Smoothness) The function $f$ is $L$-smooth, i.e., for all $x, y \in \mathbb{R}^d$,
    $$ \|\nabla f(x) - \nabla f(y)\| \le L \|x - y\|. $$
    \item[\textbf{A3}] (Bounded Variance) The stochastic gradient $\nabla F(x; \xi)$ is an unbiased estimator of $\nabla f(x)$ with bounded variance: $\mathbb{E}[\nabla F(x; \xi)] = \nabla f(x)$ and $\mathbb{E}[\|\nabla F(x; \xi) - \nabla f(x)\|^2] \le \sigma^2$.
\end{itemize}

\section{Main Result}

\begin{theorem}
Let Assumptions \textbf{A1}, \textbf{A2}, and \textbf{A3} hold. Consider the SGD iteration $x_{t+1} = x_t - \eta_t \nabla F(x_t; \xi_t)$ with step size $\eta_t = \frac{1}{L}$. Then, for the averaged iterate $\bar{x}_T = \frac{1}{T} \sum_{t=1}^T x_t$, we have:
$$ \mathbb{E}[f(\bar{x}_T)] - f(x^*) \le \frac{L \|x_1 - x^*\|^2}{2T} + \frac{\sigma^2}{2L} $$
where $x^* \in \arg\min_{x} f(x)$.
\end{theorem}

\section{Proof}

\begin{proof}
Let $x^*$ be an optimal solution. We begin by bounding the distance between the current iterate and the optimum.

% ADDED: Missing step establishing the descent inequality using smoothness
\textbf{[Step 1]} By the $L$-smoothness assumption (\textbf{A2}), we have the standard quadratic upper bound:
$$ f(y) \le f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2} \|y - x\|^2, \quad \forall x, y \in \mathbb{R}^d. $$

% ADDED: Missing step applying the descent inequality to the SGD update
\textbf{[Step 2]} Substituting $y = x_{t+1}$ and $x = x_t$, and using the update rule $x_{t+1} = x_t - \eta_t \nabla F(x_t; \xi_t)$, we obtain:
$$ f(x_{t+1}) \le f(x_t) - \eta_t \langle \nabla f(x_t), \nabla F(x_t; \xi_t) \rangle + \frac{L \eta_t^2}{2} \|\nabla F(x_t; \xi_t)\|^2. $$

% ADDED: Missing step taking the conditional expectation
\textbf{[Step 3]} Taking the conditional expectation $\mathbb{E}_t[\cdot] = \mathbb{E}[\cdot | x_t]$ and applying the unbiasedness property from (\textbf{A3}), $\mathbb{E}_t[\nabla F(x_t; \xi_t)] = \nabla f(x_t)$:
$$ \mathbb{E}_t[f(x_{t+1})] \le f(x_t) - \eta_t \|\nabla f(x_t)\|^2 + \frac{L \eta_t^2}{2} \mathbb{E}_t[\|\nabla F(x_t; \xi_t)\|^2]. $$

% ADDED: Missing step decomposing the stochastic gradient norm using variance
\textbf{[Step 4]} We decompose the expected squared norm of the stochastic gradient using the variance. By (\textbf{A3}):
$$ \mathbb{E}_t[\|\nabla F(x_t; \xi_t)\|^2] = \|\mathbb{E}_t[\nabla F(x_t; \xi_t)]\|^2 + \mathbb{E}_t[\|\nabla F(x_t; \xi_t) - \nabla f(x_t)\|^2] \le \|\nabla f(x_t)\|^2 + \sigma^2. $$

% ADDED: Missing step substituting the variance bound
\textbf{[Step 5]} Substituting the result of [Step 4] into [Step 3] yields:
$$ \mathbb{E}_t[f(x_{t+1})] \le f(x_t) - \eta_t \|\nabla f(x_t)\|^2 + \frac{L \eta_t^2}{2} (\|\nabla f(x_t)\|^2 + \sigma^2). $$
Rearranging terms gives:
$$ \mathbb{E}_t[f(x_{t+1})] \le f(x_t) - \left( \eta_t - \frac{L \eta_t^2}{2} \right) \|\nabla f(x_t)\|^2 + \frac{L \eta_t^2}{2} \sigma^2. $$

% ADDED: Missing step applying convexity to relate gradient to function suboptimality
\textbf{[Step 6]} By the convexity assumption (\textbf{A1}), we have $f(x^*) \ge f(x_t) + \langle \nabla f(x_t), x^* - x_t \rangle$, which implies:
$$ \langle \nabla f(x_t), x_t - x^* \rangle \ge f(x_t) - f(x^*). $$
Furthermore, by the Cauchy-Schwarz and Young's inequalities, $\|\nabla f(x_t)\|^2 \ge \frac{2}{\|x_t - x^*\|^2} \langle \nabla f(x_t), x_t - x^* \rangle^2$. A more direct standard bound for convex functions is:
$$ \|\nabla f(x_t)\|^2 \ge \frac{2(f(x_t) - f(x^*))}{\|x_t - x^*\|^2} \quad \text{(if we use strong convexity, but we only have convexity here).} $$
% CORRECTED: Since we only assume convexity (A1), we must use the standard distance-based upper bound instead of gradient norm bounds.
Instead, we bound the distance to the optimum. Subtracting $f(x^*)$ from both sides of [Step 5] and using the convexity lower bound $\langle \nabla f(x_t), x_t - x^* \rangle \ge f(x_t) - f(x^*)$:
$$ \mathbb{E}_t[f(x_{t+1})] \le f(x_t) - \left( \eta_t - \frac{L \eta_t^2}{2} \right) \|\nabla f(x_t)\|^2 + \frac{L \eta_t^2}{2} \sigma^2. $$
To introduce $x^*$, we use the identity $\|x_{t+1} - x^*\|^2 = \|x_t - x^*\|^2 - 2\eta_t \langle \nabla F(x_t; \xi_t), x_t - x^* \rangle + \eta_t^2 \|\nabla F(x_t; \xi_t)\|^2$.

% ADDED: Missing step taking the conditional expectation of the distance identity
\textbf{[Step 7]} Taking the conditional expectation of the distance identity:
$$ \mathbb{E}_t[\|x_{t+1} - x^*\|^2] = \|x_t - x^*\|^2 - 2\eta_t \langle \nabla f(x_t), x_t - x^* \rangle + \eta_t^2 \mathbb{E}_t[\|\nabla F(x_t; \xi_t)\|^2]. $$
Applying [Step 4] and the convexity assumption (\textbf{A1}) $\langle \nabla f(x_t), x_t - x^* \rangle \ge f(x_t) - f(x^*)$:
$$ \mathbb{E}_t[\|x_{t+1} - x^*\|^2] \le \|x_t - x^*\|^2 - 2\eta_t (f(x_t) - f(x^*)) + \eta_t^2 (\|\nabla f(x_t)\|^2 + \sigma^2). $$

% ADDED: Missing step combining the descent and distance inequalities
\textbf{[Step 8]} We eliminate $\|\nabla f(x_t)\|^2$ by combining [Step 5] and [Step 7]. Multiply [Step 5] by $\eta_t$:
$$ \eta_t \mathbb{E}_t[f(x_{t+1})] \le \eta_t f(x_t) - (\eta_t^2 - \frac{L \eta_t^3}{2}) \|\nabla f(x_t)\|^2 + \frac{L \eta_t^3}{2} \sigma^2. $$
Multiply [Step 7] by $\frac{1}{2}$:
$$ \frac{1}{2} \mathbb{E}_t[\|x_{t+1} - x^*\|^2] \le \frac{1}{2} \|x_t - x^*\|^2 - \eta_t (f(x_t) - f(x^*)) + \frac{\eta_t^2}{2} \|\nabla f(x_t)\|^2 + \frac{\eta_t^2}{2} \sigma^2. $$
Setting $\eta_t = \frac{1}{L}$, we have $\eta_t^2 = \frac{L \eta_t^3}{2} = \frac{1}{L^2}$. Adding the two equations cancels the gradient norm terms exactly:
$$ \eta_t \mathbb{E}_t[f(x_{t+1})] + \frac{1}{2} \mathbb{E}_t[\|x_{t+1} - x^*\|^2] \le \eta_t f(x_t) - \eta_t (f(x_t) - f(x^*)) + \frac{1}{2} \|x_t - x^*\|^2 + \frac{\eta_t^2}{2} \sigma^2 + \frac{L \eta_t^3}{2} \sigma^2. $$
Simplifying the function values:
$$ \eta_t \mathbb{E}_t[f(x_{t+1})] + \frac{1}{2} \mathbb{E}_t[\|x_{t+1} - x^*\|^2] \le \eta_t f(x^*) + \frac{1}{2} \|x_t - x^*\|^2 + \eta_t^2 \sigma^2. $$
Subtracting $\frac{1}{2} \mathbb{E}_t[\|x_{t+1} - x^*\|^2]$ and taking total expectation yields:
$$ \eta_t \mathbb{E}[f(x_{t+1}) - f(x^*)] \le \frac{1}{2} \mathbb{E}[\|x_t - x^*\|^2] - \frac{1}{2} \mathbb{E}[\|x_{t+1} - x^*\|^2] + \eta_t^2 \sigma^2. $$

% ADDED: Missing step telescoping sum to extract the convergence rate
\textbf{[Step 9]} Summing the above inequality from $t=1$ to $T$:
$$ \sum_{t=1}^T \eta_t \mathbb{E}[f(x_{t+1}) - f(x^*)] \le \frac{1}{2} \|x_1 - x^*\|^2 - \frac{1}{2} \mathbb{E}[\|x_{T+1} - x^*\|^2] + T \eta_t^2 \sigma^2 \le \frac{1}{2} \|x_1 - x^*\|^2 + T \eta_t^2 \sigma^2. $$
Since $\eta_t = \frac{1}{L}$ is constant, we divide by $T \eta_t$:
$$ \frac{1}{T} \sum_{t=1}^T \mathbb{E}[f(x_{t+1}) - f(x^*)] \le \frac{L \|x_1 - x^*\|^2}{2T} + \frac{\eta_t \sigma^2}{L} = \frac{L \|x_1 - x^*\|^2}{2T} + \frac{\sigma^2}{2L}. $$

% ADDED: Missing step applying Jensen's inequality for the averaged iterate
\textbf{[Step 10]} By Jensen's inequality and the convexity of $f$ (\textbf{A1}), we have $\mathbb{E}[f(\bar{x}_{T+1})] \le \frac{1}{T} \sum_{t=1}^T \mathbb{E}[f(x_{t+1})]$. Applying this to the previous step gives the final convergence rate:
$$ \mathbb{E}[f(\bar{x}_{T+1})] - f(x^*) \le \frac{L \|x_1 - x^*\|^2}{2T} + \frac{\sigma^2}{2L}. $$
Shifting the index $T+1$ to $T$ for the averaged iterate yields the stated theorem:
$$ \mathbb{E}[f(\bar{x}_T)] - f(x^*) \le \frac{L \|x_1 - x^*\|^2}{2T} + \frac{\sigma^2}{2L}. $$
This completes the proof.
\end{proof}
