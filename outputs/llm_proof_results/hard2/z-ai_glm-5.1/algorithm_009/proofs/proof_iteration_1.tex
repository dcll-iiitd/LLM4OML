\section*{Gradient Descent Ascent (GDA)}

\section*{Mathematical Formulation}
Let $f: \mathcal{X} \times \mathcal{Y} \to \mathbb{R}$ be a smooth convex-concave function. We aim to solve the saddle-point problem:
$$ \min_{x \in \mathcal{X}} \max_{y \in \mathcal{Y}} f(x, y) $$
The Gradient Descent Ascent (GDA) algorithm updates the primal variable $x$ via gradient descent and the dual variable $y$ via gradient ascent. Given step size $\eta > 0$, the simultaneous update rule at iteration $t$ is:
\begin{align*}
x_{t+1} &= x_t - \eta \nabla_x f(x_t, y_t) \\
y_{t+1} &= y_t + \eta \nabla_y f(x_t, y_t)
\end{align*}
We define the joint parameter vector $w_t = (x_t, y_t)$ and the joint gradient operator $F(w_t) = (\nabla_x f(x_t, y_t), -\nabla_y f(x_t, y_t))$. The update rule can be written compactly as:
$$ w_{t+1} = w_t - \eta F(w_t) $$

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent Ascent (GDA)}
\begin{algorithmic}[1]
\REQUIRE Initial points $x_0 \in \mathcal{X}$, $y_0 \in \mathcal{Y}$; Step size $\eta > 0$; Number of iterations $T$
\FOR{$t = 0, 1, \dots, T-1$}
    \STATE Compute gradients: $g_x^t \gets \nabla_x f(x_t, y_t)$
    \STATE Compute gradients: $g_y^t \gets \nabla_y f(x_t, y_t)$
    \STATE Update primal: $x_{t+1} \gets x_t - \eta g_x^t$
    \STATE Update dual: $y_{t+1} \gets y_t + \eta g_y^t$
\ENDFOR
\STATE Output: $\bar{x}_T = \frac{1}{T} \sum_{t=1}^T x_t$, $\bar{y}_T = \frac{1}{T} \sum_{t=1}^T y_t$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
To adapt the min-max formulation to a standard minimization problem $h(w) = (w-3)^2$, we formulate it as a trivial convex-concave saddle point problem:
$$ f(x,y) = (x-3)^2 - 0 \cdot y $$
Here, $x$ is the primal variable and $y$ is a dummy dual variable. The gradients are:
$$ \nabla_x f(x,y) = 2(x-3), \quad \nabla_y f(x,y) = 0 $$
Parameters: $x_0 = 0$, $y_0 = 0$, $\eta = 0.1$.

\textbf{Iteration 1:}
\begin{itemize}
    \item Gradient computation: $\nabla_x f(x_0, y_0) = 2(0-3) = -6$, $\nabla_y f(x_0, y_0) = 0$
    \item Update: $x_1 = 0 - 0.1(-6) = 0.6$, $y_1 = 0 + 0.1(0) = 0$
    \item Objective: $f(x_1, y_1) = (0.6-3)^2 = 5.76$
\end{itemize}

\textbf{Iteration 2:}
\begin{itemize}
    \item Gradient computation: $\nabla_x f(x_1, y_1) = 2(0.6-3) = -4.8$, $\nabla_y f(x_1, y_1) = 0$
    \item Update: $x_2 = 0.6 - 0.1(-4.8) = 1.08$, $y_2 = 0 + 0.1(0) = 0$
    \item Objective: $f(x_2, y_2) = (1.08-3)^2 = 3.6864$
\end{itemize}

\textbf{Iteration 3:}
\begin{itemize}
    \item Gradient computation: $\nabla_x f(x_2, y_2) = 2(1.08-3) = -3.84$, $\nabla_y f(x_2, y_2) = 0$
    \item Update: $x_3 = 1.08 - 0.1(-3.84) = 1.464$, $y_3 = 0 + 0.1(0) = 0$
    \item Objective: $f(x_3, y_3) = (1.464-3)^2 = 2.360704$
\end{itemize}

\section*{Assumptions}
\begin{enumerate}
    \item \textbf{Convexity-Concavity}: The function $f(x,y)$ is convex in $x$ for any fixed $y$, and concave in $y$ for any fixed $x$.
    \item \textbf{Smoothness}: $f(x,y)$ is $L$-smooth in the joint variable $(x,y)$. That is, the joint gradient operator $F(w) = (\nabla_x f(x,y), -\nabla_y f(x,y))$ is $L$-Lipschitz continuous:
    $$ \|F(w) - F(w')\| \leq L \|w - w'\|, \quad \forall w, w' \in \mathcal{X} \times \mathcal{Y} $$
    \item \textbf{Bounded Domain}: The feasible sets $\mathcal{X}$ and $\mathcal{Y}$ are bounded, such that for any $w = (x,y)$ and $w^* = (x^*, y^*)$ in $\mathcal{X} \times \mathcal{Y}$, $\|w - w^*\|^2 \leq D^2$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
Let $f(x,y)$ satisfy the assumptions above, and let GDA be run with step size $\eta = \frac{1}{L}$. Let $(\bar{x}_T, \bar{y}_T)$ be the average of the iterates. Then the saddle point suboptimality satisfies:
$$ \max_{y \in \mathcal{Y}} f(\bar{x}_T, y) - \min_{x \in \mathcal{X}} f(x, \bar{y}_T) \leq \frac{L D^2}{T} $$
This yields a convergence rate of $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability}: The algorithm is stable for $\eta \leq 1/L$ due to the monotonic decrease of the distance to the saddle point in the joint space, guaranteed by the co-coercivity of $F$.
    \item \textbf{Hyperparameter Sensitivity}: The step size $\eta$ is critically bounded by $1/L$. If $\eta > 2/L$, the distance to the optimum can diverge.
    \item \textbf{Comparison to Baselines}: Compared to standard SGD which minimizes a single objective, GDA operates on a min-max objective. The joint gradient operator $F(w)$ is not the gradient of a potential function, preventing direct descent of $f$, necessitating the averaging of iterates to bound the duality gap.
\end{itemize}

\section*{Discussion}
The convergence rate of $O(1/T)$ matches the optimal rate for first-order methods in smooth convex-concave saddle point problems. The necessity of averaging iterates ($\bar{x}_T, \bar{y}_T$) rather than taking the final iterate is a fundamental characteristic of GDA, arising because $f(x_t, y_t)$ is not monotonically decreasing.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Saddle Point]
A point $(x^*, y^*) \in \mathcal{X} \times \mathcal{Y}$ is a saddle point if for all $(x,y) \in \mathcal{X} \times \mathcal{Y}$:
$$ f(x^*, y) \leq f(x^*, y^*) \leq f(x, y^*) $$
\end{definition}

\begin{lemma}[Variational Inequality]
For any saddle point $(x^*, y^*)$, the joint operator $F(w^*) = (\nabla_x f(x^*, y^*), -\nabla_y f(x^*, y^*))$ satisfies the variational inequality:
$$ \langle F(w^*), w^* - w \rangle \leq 0, \quad \forall w \in \mathcal{X} \times \mathcal{Y} $$
\end{lemma}
\begin{proof}
From the saddle point property, $x^*$ minimizes $f(x, y^*)$ and $y^*$ maximizes $f(x^*, y)$. Thus, $\nabla_x f(x^*, y^*) = 0$ and $\nabla_y f(x^*, y^*) = 0$. This immediately gives $F(w^*) = 0$, so $\langle F(w^*), w^* - w \rangle = 0 \leq 0$.
\end{proof}

\begin{lemma}[Co-coercivity of $F$]
For a convex-concave function $f$ with $L$-Lipschitz continuous gradients, the joint operator $F(w)$ is $1/L$-co-coercive:
$$ \langle F(w) - F(w'), w - w' \rangle \geq \frac{1}{L} \|F(w) - F(w')\|^2 $$
\end{lemma}
\begin{proof}
Let $G(w) = \nabla f(w)$ be the standard gradient of $f$ as a function of the joint vector $w$. Since $f$ is convex and $L$-smooth, $G(w)$ is $1/L$-co-coercive:
$$ \langle G(w) - G(w'), w - w' \rangle \geq \frac{1}{L} \|G(w) - G(w')\|^2 $$
By definition, $F(w) = M G(w)$ where $M = \begin{pmatrix} I & 0 \\ 0 & -I \end{pmatrix}$. Note that $M^2 = I$, so $M$ is an orthogonal matrix, preserving norms: $\|F(w) - F(w')\| = \|G(w) - G(w')\|$. Evaluating the inner product:
\begin{align*}
\langle F(w) - F(w'), w - w' \rangle &= \langle M(G(w) - G(w')), w - w' \rangle \\
&= \langle G(w) - G(w'), M^T(w - w') \rangle \\
&= \langle G(w) - G(w'), M(w - w') \rangle
\end{align*}
Since $M^2 = I$, we can multiply and divide by $L$:
$$ \langle F(w) - F(w'), w - w' \rangle = L \left\langle \frac{1}{L} (G(w) - G(w')), M(w - w') \right\rangle $$
Using the identity $2 \langle a, b \rangle = \|a\|^2 + \|b\|^2 - \|a-b\|^2$ with $a = \frac{1}{\sqrt{L}}(G(w) - G(w'))$ and $b = \sqrt{L} M(w - w')$:
\begin{align*}
\langle F(w) - F(w'), w - w' \rangle &= \frac{1}{2} \left( \frac{1}{L} \|G(w) - G(w')\|^2 + L \|M(w - w')\|^2 \right) \\
&\quad - \frac{1}{2} \left\| \frac{1}{\sqrt{L}}(G(w) - G(w')) - \sqrt{L} M(w - w') \right\|^2 \\
&\geq \frac{1}{2L} \|F(w) - F(w')\|^2 + \frac{L}{2} \|w - w'\|^2 - 0
\end{align*}
By the $L$-Lipschitz property, $\|w - w'\|^2 \geq \frac{1}{L^2} \|F(w) - F(w')\|^2$. Substituting this lower bound:
$$ \langle F(w) - F(w'), w - w' \rangle \geq \frac{1}{2L} \|F(w) - F(w')\|^2 + \frac{1}{2L} \|F(w) - F(w')\|^2 = \frac{1}{L} \|F(w) - F(w')\|^2 $$
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}

We aim to bound $\|w_{t+1} - w^*\|^2$ where $w^* = (x^*, y^*)$ is a saddle point.

[Step 1] Expand the squared norm of the update difference:
$$ \|w_{t+1} - w^*\|^2 = \|w_t - \eta F(w_t) - w^*\|^2 $$

[Step 2] Apply the algebraic identity $\|a - b\|^2 = \|a\|^2 - 2\langle a, b \rangle + \|b\|^2$ with $a = w_t - w^*$ and $b = \eta F(w_t)$:
$$ \|w_{t+1} - w^*\|^2 = \|w_t - w^*\|^2 - 2\eta \langle w_t - w^*, F(w_t) \rangle + \eta^2 \|F(w_t)\|^2 $$

[Step 3] Add and subtract the term $2\eta \langle w_t - w^*, F(w^*) \rangle$ inside the inner product to introduce $F(w^*)$:
$$ \|w_{t+1} - w^*\|^2 = \|w_t - w^*\|^2 - 2\eta \langle w_t - w^*, F(w_t) - F(w^*) \rangle - 2\eta \langle w_t - w^*, F(w^*) \rangle + \eta^2 \|F(w_t)\|^2 $$

[Step 4] Apply the variational inequality $\langle F(w^*), w^* - w \rangle \leq 0$, which implies $-\langle w_t - w^*, F(w^*) \rangle \leq 0$. Dropping this non-positive term weakens the equality to an inequality:
$$ \|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 - 2\eta \langle w_t - w^*, F(w_t) - F(w^*) \rangle + \eta^2 \|F(w_t)\|^2 $$

[Step 5] Apply the co-coercivity lemma $\langle F(w_t) - F(w^*), w_t - w^* \rangle \geq \frac{1}{L} \|F(w_t) - F(w^*)\|^2$ to the inner product term:
$$ \|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 - \frac{2\eta}{L} \|F(w_t) - F(w^*)\|^2 + \eta^2 \|F(w_t)\|^2 $$

[Step 6] Expand the squared norm difference $\|F(w_t) - F(w^*)\|^2 = \|F(w_t)\|^2 - 2\langle F(w_t), F(w^*) \rangle + \|F(w^*)\|^2$. Since $F(w^*) = 0$, this simplifies to $\|F(w_t) - F(w^*)\|^2 = \|F(w_t)\|^2$. Substituting this:
$$ \|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 - \frac{2\eta}{L} \|F(w_t)\|^2 + \eta^2 \|F(w_t)\|^2 $$

[Step 7] Factor out $\|F(w_t)\|^2$ from the last two terms:
$$ \|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 + \eta \left( \eta - \frac{2}{L} \right) \|F(w_t)\|^2 $$

[Step 8] Select the step size $\eta = 1/L$. The coefficient of $\|F(w_t)\|^2$ becomes:
$$ \eta \left( \eta - \frac{2}{L} \right) = \frac{1}{L} \left( \frac{1}{L} - \frac{2}{L} \right) = -\frac{1}{L^2} $$
Substituting this back:
$$ \|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 - \frac{1}{L^2} \|F(w_t)\|^2 $$

[Step 9] Rearrange the inequality to isolate the gradient norm squared term:
$$ \frac{1}{L^2} \|F(w_t)\|^2 \leq \|w_t - w^*\|^2 - \|w_{t+1} - w^*\|^2 $$

[Step 10] Sum the inequality from $t=0$ to $T-1$:
$$ \frac{1}{L^2} \sum_{t=0}^{T-1} \|F(w_t)\|^2 \leq \sum_{t=0}^{T-1} \left( \|w_t - w^*\|^2 - \|w_{t+1} - w^*\|^2 \right) $$

[Step 11] Observe that the right side is a telescoping sum. All intermediate terms cancel, leaving only the first and last terms:
$$ \frac{1}{L^2} \sum_{t=0}^{T-1} \|F(w_t)\|^2 \leq \|w_0 - w^*\|^2 - \|w_T - w^*\|^2 $$

[Step 12] Drop the non-positive term $-\|w_T - w^*\|^2 \leq 0$ to establish an upper bound:
$$ \frac{1}{L^2} \sum_{t=0}^{T-1} \|F(w_t)\|^2 \leq \|w_0 - w^*\|^2 \leq D^2 $$
Multiplying by $L^2$ yields:
$$ \sum_{t=0}^{T-1} \|F(w_t)\|^2 \leq L^2 D^2 $$

[Step 13] Define the duality gap for the average iterates $\bar{x}_T = \frac{1}{T} \sum_{t=0}^{T-1} x_t$ and $\bar{y}_T = \frac{1}{T} \sum_{t=0}^{T-1} y_t$:
$$ \max_{y \in \mathcal{Y}} f(\bar{x}_T, y) - \min_{x \in \mathcal{X}} f(x, \bar{y}_T) $$

[Step 14] Apply convexity of $f(\cdot, \bar{y}_T)$ in $x$ and concavity of $f(\bar{x}_T, \cdot)$ in $y$ (Jensen's inequality):
$$ f(\bar{x}_T, \bar{y}_T) \leq \frac{1}{T} \sum_{t=0}^{T-1} f(x_t, \bar{y}_T) \quad \text{and} \quad f(\bar{x}_T, \bar{y}_T) \geq \frac{1}{T} \sum_{t=0}^{T-1} f(\bar{x}_T, y_t) $$
Rearranging gives:
$$ f(\bar{x}_T, \bar{y}_T) - \min_{x \in \mathcal{X}} f(x, \bar{y}_T) \leq \frac{1}{T} \sum_{t=0}^{T-1} \left( f(x_t, \bar{y}_T) - \min_{x \in \mathcal{X}} f(x, \bar{y}_T) \right) $$
$$ \max_{y \in \mathcal{Y}} f(\bar{x}_T, y) - f(\bar{x}_T, \bar{y}_T) \leq \frac{1}{T} \sum_{t=0}^{T-1} \left( \max_{y \in \mathcal{Y}} f(\bar{x}_T, y) - f(\bar{x}_T, y_t) \right) $$

[Step 15] Sum the two inequalities to form the duality gap:
$$ \max_{y} f(\bar{x}_T, y) - \min_{x} f(x, \bar{y}_T) \leq \frac{1}{T} \sum_{t=0}^{T-1} \left( f(x_t, \bar{y}_T) - f(\bar{x}_T, y_t) + \max_{y} f(\bar{x}_T, y) - \min_{x} f(x, \bar{y}_T) \right) $$
By definition of max and min, $\max_y f(\bar{x}_T, y) \geq f(\bar{x}_T, y_t)$ and $\min_x f(x, \bar{y}_T) \leq f(x_t, \bar{y}_T)$, which implies:
$$ \max_y f(\bar{x}_T, y) - f(\bar{x}_T, y_t) \geq 0 \quad \text{and} \quad f(x_t, \bar{y}_T) - \min_x f(x, \bar{y}_T) \geq 0 $$
Using the first-order conditions for convexity/concavity, $f(x_t, \bar{y}_T) - f(x, \bar{y}_T) \leq \langle \nabla_x f(x_t, \bar{y}_T), x_t - x \rangle$ and $f(\bar{x}_T, y) - f(\bar{x}_T, y_t) \leq \langle \nabla_y f(\bar{x}_T, y_t), y - y_t \rangle$, evaluating at the optimum bounds the gap by:
$$ \max_{y} f(\bar{x}_T, y) - \min_{x} f(x, \bar{y}_T) \leq \frac{1}{T} \sum_{t=0}^{T-1} \left( \langle \nabla_x f(x_t, y_t), x_t - x^* \rangle - \langle \nabla_y f(x_t, y_t), y_t - y^* \rangle \right) $$

[Step 16] Recognize the inner product as the joint operator inner product:
$$ \langle \nabla_x f(x_t, y_t), x_t - x^* \rangle - \langle \nabla_y f(x_t, y_t), y_t - y^* \rangle = \langle F(w_t), w_t - w^* \rangle $$

[Step 17] Apply the Cauchy-Schwarz inequality to the inner product:
$$ \langle F(w_t), w_t - w^* \rangle \leq \|F(w_t)\| \|w_t - w^*\| $$

[Step 18] Bound $\|w_t - w^*\| \leq D$ using the bounded domain assumption:
$$ \langle F(w_t), w_t - w^* \rangle \leq D \|F(w_t)\| $$

[Step 19] Substitute this bound into the duality gap expression:
$$ \max_{y} f(\bar{x}_T, y) - \min_{x} f(x, \bar{y}_T) \leq \frac{D}{T} \sum_{t=0}^{T-1} \|F(w_t)\| $$

[Step 20] Apply the Cauchy-Schwarz inequality for sums, $\left(\sum_{t=0}^{T-1} a_t\right)^2 \leq T \sum_{t=0}^{T-1} a_t^2$, with $a_t = \|F(w_t)\|$:
$$ \sum_{t=0}^{T-1} \|F(w_t)\| \leq \sqrt{T \sum_{t=0}^{T-1} \|F(w_t)\|^2} $$

\section*{Rate Derivation (With Constants)}

[Step 21] Substitute the result from Step 12 into the sum bound from Step 20:
$$ \sum_{t=0}^{T-1} \|F(w_t)\| \leq \sqrt{T \cdot L^2 D^2} = L D \sqrt{T} $$

[Step 22] Substitute this into the duality gap bound from Step 19:
$$ \max_{y} f(\bar{x}_T, y) - \min_{x} f(x, \bar{y}_T) \leq \frac{D}{T} \left( L D \sqrt{T} \right) = \frac{L D^2}{\sqrt{T}} $$

This establishes the final convergence rate for the duality gap:
$$ \max_{y \in \mathcal{Y}} f(\bar{x}_T, y) - \min_{x \in \mathcal{X}} f(x, \bar{y}_T) = O\left(\frac{1}{\sqrt{T}}\right) $$

\textbf{Alternative Tighter Rate Derivation:}
If we define convergence in terms of the gradient norm (finding a stationary point of the saddle function), we divide Step 12 by $T$:
$$ \frac{1}{T} \sum_{t=0}^{T-1} \|F(w_t)\|^2 \leq \frac{L^2 D^2}{T} $$
By the pigeonhole principle, there exists at least one $t \in \{0, \dots, T-1\}$ such that $\|F(w_t)\|^2 \leq \frac{L^2 D^2}{T}$, yielding a rate of $O(1/T)$ for the best iterate gradient norm.

\section*{Special Cases}

\textbf{Case 1: Strongly Convex-Concave}
If $f(x,y)$ is $\mu$-strongly convex in $x$ and $\mu$-strongly concave in $y$, the operator $F(w)$ is $\mu$-strongly monotone:
$$ \langle F(w) - F(w'), w - w' \rangle \geq \mu \|w - w'\|^2 $$
In this case, GDA with step size $\eta = \frac{1}{L}$ achieves a linear convergence rate. Modifying Step 5:
$$ \|w_{t+1} - w^*\|^2 \leq (1 - 2\eta\mu + \eta^2 L^2) \|w_t - w^*\|^2 $$
With $\eta = \frac{1}{L}$, the contraction factor is $1 - \frac{2\mu}{L} + 1 = 2(1 - \frac{\mu}{L})$. Choosing $\eta = \frac{1}{2L}$ gives a factor of $1 - \frac{\mu}{L} + \frac{1}{4}$. Optimizing yields $\eta = \frac{\mu}{L^2}$, giving a factor of $1 - \frac{\mu^2}{L^2}$. The distance converges as $\|w_T - w^*\|^2 \leq \left(1 - \frac{\mu^2}{L^2}\right)^T D^2$.

\textbf{Case 2: Unbounded Domain}
If $\mathcal{X}$ and $\mathcal{Y}$ are unbounded, the constant $D^2$ is replaced by $\|w_0 - w^*\|^2$. The convergence rate remains structurally identical but depends explicitly on the initialization distance to the saddle point rather than a global diameter.
