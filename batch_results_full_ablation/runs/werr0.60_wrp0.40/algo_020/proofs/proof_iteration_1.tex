\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimization problem  

\[
\min_{x\in\R^{d}} \; F(x)\;\stackrel{\text{def}}{=}\; f(x)+g(x),
\]

where  

\begin{itemize}
  \item $f:\R^{d}\to\R$ is \emph{convex} and $L$-\emph{smooth} (i.e., $\nabla f$ is $L$-Lipschitz);
  \item $g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).
\end{itemize}

Given a stepsize $\eta>0$, the proximal gradient iteration is  

\[
\boxed{%
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta \nabla f(x_{k})\bigr)},\qquad k=0,1,\dots
}
\tag{1}
\]

The proximal operator of $g$ with parameter $\eta$ is  

\[
\prox_{\eta g}(v)\; \stackrel{\text{def}}{=}\; \arg\min_{u\in\R^{d}}
\Bigl\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Bigr\}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$.
\For{$k=0,1,2,\dots$}
    \State Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \State $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta g_{k}\bigr)$.
\EndFor
\end{algorithmic}
\end{algorithm}
The algorithm is typically terminated when $\|x_{k+1}-x_{k}\|\leq\varepsilon$ or after a fixed number of iterations.

\section*{Dry Run Example}
Let  

\[
f(w)=(w-3)^{2},\qquad g\equiv 0,\qquad \eta=0.1,\qquad w_{0}=0.
\]

Since $g\equiv0$, $\prox_{\eta g}(v)=v$. The iteration reduces to ordinary gradient descent:
\[
w_{k+1}=w_{k}-\eta \nabla f(w_{k}),\qquad 
\nabla f(w)=2(w-3).
\]

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})$ & $F(w_{k})=(w_{k}-3)^{2}$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3673$
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the descent property of (1).

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]\label{as:convexsmooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient, i.e.,  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\quad\forall x,y\in\R^{d}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{as:gconvex}
$g$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is well defined for every $\eta>0$.
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsize satisfies  

\[
0<\eta\le \frac{1}{L}.
\tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) convergence of PGM]\label{thm:main}
Let $\{x_{k}\}$ be generated by \eqref{1} under Assumptions \ref{as:convexsmooth}–\ref{as:stepsize} and let $x^{\star}\in\arg\min F$. Then for all $k\ge 1$  

\[
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta k}.
\tag{5}
\]

Consequently, $F(x_{k})\to F(x^{\star})$ and the rate $O(1/k)$ is optimal for first‑order methods on the class of convex $L$‑smooth + convex functions.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability}: The iteration is a non‑expansive map when $\eta\le1/L$ (see Lemma~\ref{lem:nonexpansive}). Hence the sequence $\{x_{k}\}$ remains bounded.
  \item \textbf{Hyperparameter sensitivity}: The bound \eqref{5} degrades linearly with $1/\eta$, so a stepsize close to $1/L$ yields the smallest constant.
  \item \textbf{Comparison to baselines}: For $g\equiv0$, \eqref{5} reduces to the classical gradient‑descent bound. When $g$ is the indicator of a convex set $C$, \eqref{1} becomes projected gradient descent with the same $O(1/k)$ guarantee.
\end{itemize}

\section*{Discussion}
The proof below relies only on two elementary facts:
\begin{enumerate}
  \item The \emph{descent lemma} for $L$‑smooth functions (Lemma~\ref{lem:descent}).
  \item The \emph{non‑expansiveness} of the proximal operator (Lemma~\ref{lem:nonexpansive}).
\end{enumerate}
No stochasticity or additional regularity is required.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ satisfies \eqref{3}, then for any $x,y\in\R^{d}$  

\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
The standard integral form of the Lipschitz condition yields \eqref{6}. ∎
\end{proof}

\begin{lemma}[Non‑expansiveness of the proximal operator]\label{lem:nonexpansive}
For any $\eta>0$ and any $u,v\in\R^{d}$,  

\[
\bigl\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\bigr\|\le \|u-v\|.
\tag{7}
\]
\end{lemma}
\begin{proof}
Let $p=\prox_{\eta g}(u)$ and $q=\prox_{\eta g}(v)$. By the optimality conditions,
\[
\frac{1}{\eta}(u-p)\in\partial g(p),\qquad 
\frac{1}{\eta}(v-q)\in\partial g(q).
\]
Monotonicity of $\partial g$ gives
\[
\big\langle \tfrac{1}{\eta}(u-p)-\tfrac{1}{\eta}(v-q),p-q\big\rangle\ge0,
\]
which after rearranging yields \eqref{7}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{\star}\in\arg\min F$.  Define the proximal gradient step  

\[
y_{k}\;\stackrel{\text{def}}{=}\;x_{k}-\eta\nabla f(x_{k}),\qquad
x_{k+1}= \prox_{\eta g}(y_{k}). \tag{8}
\]

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:nonexpansive} with $u=y_{k}$ and $v=x^{\star}$ (note that $x^{\star}= \prox_{\eta g}(x^{\star}-\eta\nabla f(x^{\star}) )$ because $0\in\partial F(x^{\star})$). We obtain  

\[
\|x_{k+1}-x^{\star}\|^{2}\le\|y_{k}-x^{\star}\|^{2}. \tag{9}
\]

\noindent\textbf{[Step 2]} Expand the right‑hand side of \eqref{9} using the definition of $y_{k}$:

\[
\|y_{k}-x^{\star}\|^{2}
=\|x_{k}-x^{\star}\|^{2}
-2\eta\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle
+\eta^{2}\|\nabla f(x_{k})\|^{2}. \tag{10}
\]

\noindent\textbf{[Step 3]} Use the convexity of $f$ to lower‑bound the inner product:

\[
\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle
\ge f(x_{k})-f(x^{\star}). \tag{11}
\]

\noindent\textbf{[Step 4]} Apply the descent lemma \eqref{6} with $x=x_{k}$ and $y=x^{\star}$, then rearrange to obtain an upper bound on $\|\nabla f(x_{k})\|^{2}$:

\[
f(x^{\star})\le f(x_{k})+\langle\nabla f(x_{k}),x^{\star}-x_{k}\rangle+\frac{L}{2}\|x^{\star}-x_{k}\|^{2}.
\]
Rewriting,
\[
\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\le f(x_{k})-f(x^{\star})+\frac{L}{2}\|x_{k}-x^{\star}\|^{2}. \tag{12}
\]

\noindent\textbf{[Step 5]} Insert \eqref{12} into \eqref{10}:

\[
\|y_{k}-x^{\star}\|^{2}
\le\|x_{k}-x^{\star}\|^{2}
-2\eta\bigl[f(x_{k})-f(x^{\star})\bigr]
+\eta L\|x_{k}-x^{\star}\|^{2}
+\eta^{2}\|\nabla f(x_{k})\|^{2}. \tag{13}
\]

\noindent\textbf{[Step 6]} Since $g$ is convex, the subgradient inequality yields  

\[
g(x_{k+1})-g(x^{\star})\le \langle s_{k+1},x_{k+1}-x^{\star}\rangle,
\qquad s_{k+1}\in\partial g(x_{k+1}). \tag{14}
\]

From the optimality condition of the proximal step (see Lemma~\ref{lem:nonexpansive} proof),

\[
\frac{1}{\eta}\bigl(y_{k}-x_{k+1}\bigr)\in\partial g(x_{k+1})\quad\Rightarrow\quad
s_{k+1}= \frac{1}{\eta}(y_{k}-x_{k+1}). \tag{15}
\]

\noindent\textbf{[Step 7]} Multiply \eqref{15} by $x_{k+1}-x^{\star}$ and use \eqref{14}:

\[
g(x_{k+1})-g(x^{\star})
\le \frac{1}{\eta}\langle y_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle.
\tag{16}
\]

\noindent\textbf{[Step 8]} Expand the inner product:

\[
\langle y_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
= \frac{1}{2}\Bigl(\|y_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|y_{k}-x_{k+1}\|^{2}\Bigr). \tag{17}
\]

\noindent\textbf{[Step 9]} Plug \eqref{17} into \eqref{16} and multiply by $2\eta$:

\[
2\eta\bigl[g(x_{k+1})-g(x^{\star})\bigr]
\le \|y_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|y_{k}-x_{k+1}\|^{2}. \tag{18}
\]

\noindent\textbf{[Step 10]} Rearranging \eqref{18} and discarding the non‑negative term $\|y_{k}-x_{k+1}\|^{2}$ gives

\[
\|x_{k+1}-x^{\star}\|^{2}
\le \|y_{k}-x^{\star}\|^{2}
-2\eta\bigl[g(x_{k+1})-g(x^{\star})\bigr]. \tag{19}
\]

\noindent\textbf{[Step 11]} Combine \eqref{19} with the inequality \eqref{9} (which already states $\|x_{k+1}-x^{\star}\|^{2}\le\|y_{k}-x^{\star}\|^{2}$) to keep the tighter bound \eqref{19}. Now substitute the expansion \eqref{13} for $\|y_{k}-x^{\star}\|^{2}$:

\[
\begin{aligned}
\|x_{k+1}-x^{\star}\|^{2}
&\le \|x_{k}-x^{\star}\|^{2}
-2\eta\bigl[f(x_{k})-f(x^{\star})\bigr]
+\eta L\|x_{k}-x^{\star}\|^{2}
+\eta^{2}\|\nabla f(x_{k})\|^{2} \\
&\qquad -2\eta\bigl[g(x_{k+1})-g(x^{\star})\bigr].
\end{aligned}
\tag{20}
\]

\noindent\textbf{[Step 12]} Because $\eta\le 1/L$, we have $\eta L\le1$, hence $1-\eta L\ge0$. Drop the non‑negative term $\eta^{2}\|\nabla f(x_{k})\|^{2}$ to obtain a simpler inequality:

\[
\|x_{k+1}-x^{\star}\|^{2}
\le \|x_{k}-x^{\star}\|^{2}
-2\eta\Bigl[\,\underbrace{f(x_{k})-f(x^{\star})}_{\text{(a)}}
+\underbrace{g(x_{k+1})-g(x^{\star})}_{\text{(b)}}\Bigr]
+ (1-\eta L)\|x_{k}-x^{\star}\|^{2}. \tag{21}
\]

\noindent\textbf{[Step 13]} Rearranging terms gives the one‑step descent inequality for the composite objective $F$:

\[
2\eta\bigl[F(x_{k+1})-F(x^{\star})\bigr]
\le \|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}
+(1-\eta L)\|x_{k}-x^{\star}\|^{2}. \tag{22}
\]

Since $(1-\eta L)\|x_{k}-x^{\star}\|^{2}\ge0$, we can drop it to obtain a clean bound:

\[
2\eta\bigl[F(x_{k+1})-F(x^{\star})\bigr]
\le \|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}. \tag{23}
\]

\noindent\textbf{[Step 14]} Summing \eqref{23} over $k=0,\dots, K-1$ telescopes the right‑hand side:

\[
2\eta\sum_{k=0}^{K-1}\bigl[F(x_{k+1})-F(x^{\star})\bigr]
\le \|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}
\le \|x_{0}-x^{\star}\|^{2}. \tag{24}
\]

\noindent\textbf{[Step 15]} Because $F(x_{k})$ is non‑increasing (from \eqref{23}), each term in the sum is bounded above by the average. Hence

\[
K\bigl[F(x_{K})-F(x^{\star})\bigr]
\le \sum_{k=0}^{K-1}\bigl[F(x_{k+1})-F(x^{\star})\bigr]
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}. \tag{25}
\]

Dividing by $K$ yields the desired rate:

\[
F(x_{K})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K},
\qquad K\ge1. \tag{26}
\]

Equation \eqref{26} is exactly \eqref{5}, completing the proof. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{5} depends only on the initial distance to the optimum and the stepsize. Choosing the largest admissible stepsize $\eta=1/L$ minimizes the bound, giving  

\[
F(x_{k})-F(x^{\star})\le \frac{L\|x_{0}-x^{\star}\|^{2}}{2k}.
\tag{27}
\]

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Pure gradient descent} ($g\equiv0$): $\prox_{\eta g}$ is the identity, and the proof reduces to the classic $O(1/k)$ result for smooth convex $f$.
  \item \textbf{Projected gradient descent}: $g=\iota_{C}$ is the indicator of a closed convex set $C$. Then $\prox_{\eta g}=\Pi_{C}$ (Euclidean projection). The same bound \eqref{5} holds with $x^{\star}\in C$.
  \item \textbf{Lasso (ℓ\(_1\) regularization)}: $g(x)=\lambda\|x\|_{1}$ gives the soft‑thresholding operator as proximal map. The theorem guarantees $O(1/k)$ decay of the objective $f(x)+\lambda\|x\|_{1}$.
\end{itemize}
