\section*{Top‑K SGD}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a differentiable (possibly non‑convex) loss.  
At iteration $t$ a stochastic gradient $g_{t}$ is sampled:
\[
g_{t}= \nabla f(w_{t};\xi_{t}),\qquad \xi_{t}\sim\mathcal{D},
\]
where $\mathcal{D}$ denotes the data distribution.  

Define the \emph{Top‑K operator} $T_{K}:\R^{d}\rightarrow\R^{d}$ as
\[
\big[T_{K}(v)\big]_{i}= 
\begin{cases}
v_{i}, & i\in\mathcal{I}_{K}(v),\\[2mm]
0, & \text{otherwise},
\end{cases}
\]
where $\mathcal{I}_{K}(v)$ indexes the $K$ coordinates of $v$ having largest absolute value.  

The update rule of **Top‑K SGD** is
\[
\boxed{w_{t+1}=w_{t}-\eta_{t}\,T_{K}(g_{t})},
\tag{1}
\]
with step‑size (learning‑rate) schedule $\{\eta_{t}\}_{t\ge 0}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\R^{d}$, learning‑rate schedule $\{\eta_{t}\}$, sparsity level $K$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample a mini‑batch (or data point) $\xi_{t}\sim\mathcal{D}$.
    \STATE Compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$.
    \STATE $\displaystyle \tilde g_{t}=T_{K}(g_{t})$   \hfill\# keep only top‑$K$ entries.
    \STATE $w_{t+1}= w_{t}-\eta_{t}\,\tilde g_{t}$.
\ENDFOR
\STATE \textbf{Output:} $w_{T}$ (or averaged iterate $\bar w_{T}=\frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$ (so $d=1$).  
Gradient: $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, constant step‑size $\eta=0.1$, and $K=1$ (the operator does nothing in $d=1$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3689$
\end{tabular}
\end{center}
The iterates move toward the minimiser $w^{\star}=3$, and the objective value decreases.

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ has $L$‑Lipschitz continuous gradient:
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient and Bounded Variance]
\[
\E_{\xi_{t}}\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),\qquad
\E_{\xi_{t}}\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Compression Property of Top‑K]
Define the compression operator $\mathcal{C}(v)=T_{K}(v)$.  
There exists a constant $\tau\in(0,1]$ such that for all $v\in\R^{d}$,
\[
\E\!\left[\norm{\mathcal{C}(v)-v}^{2}\right]\le (1-\tau)\norm{v}^{2},
\qquad\text{with }\tau=\frac{K}{d}.
\tag{A3}
\]
\end{assumption}

\begin{assumption}[Bounded Second Moment of Gradients]
\[
\forall t,\qquad \E\!\left[\norm{g_{t}}^{2}\right]\le G^{2}.
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Assume (A1)–(A4) and choose a constant step size
\[
\eta_{t}\equiv\eta\le\frac{\tau}{L}.
\]
Then for any $T\ge1$ the iterates of Top‑K SGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\;\le\;
\frac{2\big(f(w_{0})-f^{\star}\big)}{\tau\,\eta\,T}
\;+\;
\frac{L\eta G^{2}}{\tau}.
\tag{2}
\]
Consequently, setting $\eta=\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because the compression operator satisfies a contractive bound (A3), the error introduced by sparsification is proportional to $(1-\tau)$ and does not explode. The condition $\eta\le\tau/L$ guarantees that the descent term dominates the compression error term.

\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item Larger $K$ (hence larger $\tau$) permits a larger admissible step size and improves the constant in front of the $1/T$ term.
        \item The bound degrades linearly with $L$ and $G^{2}$, matching the classical SGD rate when $\tau=1$ (i.e., no compression).
    \end{itemize}
\item \textbf{Comparison to Baselines.} When $K=d$ (no sparsification) we have $\tau=1$ and Theorem~\ref{thm:main} reduces to the standard non‑convex SGD bound $\mathcal{O}(1/\sqrt{T})$. For $K\ll d$, the rate is slowed only by the factor $1/\tau$, confirming that Top‑K sparsification incurs a mild penalty.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]
Under (A1),
\[
f(x-\eta\,u)\le f(x)-\eta\inner{\nabla f(x)}{u}
+\frac{L\eta^{2}}{2}\norm{u}^{2},
\qquad\forall x,u\in\R^{d},\;\eta>0.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient (see e.g. Nesterov). \qedhere
\end{proof}

\begin{lemma}[Compression Error Bound]
For the Top‑K operator $\mathcal{C}=T_{K}$ the property (A3) holds with $\tau=K/d$.
\tag{4}
\end{lemma}
\begin{proof}
Let $v\in\R^{d}$ and denote by $\mathcal{I}_{K}(v)$ the index set of the $K$ largest magnitudes.
Then
\[
\norm{\mathcal{C}(v)-v}^{2}
= \sum_{i\notin\mathcal{I}_{K}(v)} v_{i}^{2}
\le \frac{d-K}{d}\sum_{i=1}^{d}v_{i}^{2}
=(1-\tau)\norm{v}^{2}.
\]
Taking expectation (deterministic operator) yields (4). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We bound the expected decrease of the objective after one iteration.

\begin{align}
\text{[Step 1]} \quad
&f(w_{t+1}) \overset{(1)}{=} f\!\big(w_{t}-\eta\,\mathcal{C}(g_{t})\big) \nonumber\\
&\le f(w_{t})-\eta\inner{\nabla f(w_{t})}{\mathcal{C}(g_{t})}
+\frac{L\eta^{2}}{2}\norm{\mathcal{C}(g_{t})}^{2}
\qquad\text{(by Lemma 1)}. \tag{5}
\end{align}

Take conditional expectation w.r.t.\ $\xi_{t}$ (the randomness producing $g_{t}$) while fixing $w_{t}$.

\begin{align}
\text{[Step 2]} \quad
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})-\eta\,\E\!\big[\inner{\nabla f(w_{t})}{\mathcal{C}(g_{t})}\mid w_{t}\big]
+\frac{L\eta^{2}}{2}\E\!\big[\norm{\mathcal{C}(g_{t})}^{2}\mid w_{t}\big]. \tag{6}
\end{align}

Decompose the inner product using unbiasedness (A2) and add–subtract $g_{t}$:

\begin{align}
\text{[Step 3]} \quad
\E\!\big[\inner{\nabla f(w_{t})}{\mathcal{C}(g_{t})}\mid w_{t}\big]
&=\inner{\nabla f(w_{t})}{\E[\mathcal{C}(g_{t})\mid w_{t}] } \nonumber\\
&=\inner{\nabla f(w_{t})}{\nabla f(w_{t})}
   +\inner{\nabla f(w_{t})}{\E[\mathcal{C}(g_{t})-g_{t}\mid w_{t}] } \nonumber\\
&= \norm{\nabla f(w_{t})}^{2}
   +\inner{\nabla f(w_{t})}{\E[\mathcal{C}(g_{t})-g_{t}\mid w_{t}] }.
\tag{7}
\end{align}
Because $\mathcal{C}$ is deterministic given $g_{t}$, the expectation is over $g_{t}$.  
Apply Cauchy–Schwarz and Lemma~2:

\begin{align}
\text{[Step 4]} \quad
\big|\inner{\nabla f(w_{t})}{\E[\mathcal{C}(g_{t})-g_{t}\mid w_{t}] }\big|
&\le \norm{\nabla f(w_{t})}\,
   \E\!\big[\norm{\mathcal{C}(g_{t})-g_{t}}\mid w_{t}\big] \nonumber\\
&\le \norm{\nabla f(w_{t})}\,
   \sqrt{\E\!\big[\norm{\mathcal{C}(g_{t})-g_{t}}^{2}\mid w_{t}\big]} \nonumber\\
&\le \norm{\nabla f(w_{t})}\,
   \sqrt{(1-\tau)\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]} .
\tag{8}
\end{align}
Using (A4) we bound $\E[\norm{g_{t}}^{2}\mid w_{t}]\le G^{2}$, thus

\[
\big|\inner{\nabla f(w_{t})}{\E[\mathcal{C}(g_{t})-g_{t}\mid w_{t}] }\big|
\le \sqrt{1-\tau}\,G\,\norm{\nabla f(w_{t})}.
\tag{9}
\]

Insert (7)–(9) into (6):

\begin{align}
\text{[Step 5]} \quad
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})-\eta\norm{\nabla f(w_{t})}^{2}
   +\eta\sqrt{1-\tau}\,G\,\norm{\nabla f(w_{t})}
   +\frac{L\eta^{2}}{2}\E\!\big[\norm{\mathcal{C}(g_{t})}^{2}\mid w_{t}\big].
\tag{10}
\end{align}

Now bound $\E[\norm{\mathcal{C}(g_{t})}^{2}\mid w_{t}]$.
By the identity $\norm{\mathcal{C}(v)}^{2}= \norm{v}^{2}-\norm{\mathcal{C}(v)-v}^{2}$ and Lemma~2,

\begin{align}
\text{[Step 6]} \quad
\E\!\big[\norm{\mathcal{C}(g_{t})}^{2}\mid w_{t}\big]
&= \E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]
   -\E\!\big[\norm{\mathcal{C}(g_{t})-g_{t}}^{2}\mid w_{t}\big] \nonumber\\
&\le \E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]
   -(1-\tau)\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big] \nonumber\\
&= \tau\,\E\!\big[\norm{g_{t}}^{2}\mid w_{t}\big]
\le \tau G^{2}.
\tag{11}
\end{align}

Insert (11) into (10) and use the elementary inequality
$ab\le\frac{\tau}{2}a^{2}+\frac{1}{2\tau}b^{2}$ with $a=\norm{\nabla f(w_{t})}$,
$b=\sqrt{1-\tau}\,G$:

\begin{align}
\text{[Step 7]} \quad
\eta\sqrt{1-\tau}\,G\,\norm{\nabla f(w_{t})}
&\le \frac{\eta\tau}{2}\norm{\nabla f(w_{t})}^{2}
   +\frac{\eta(1-\tau)}{2\tau}G^{2}. 
\tag{12}
\end{align}

Combine (10), (11) and (12):

\begin{align}
\text{[Step 8]} \quad
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})
   -\eta\Big(1-\frac{\tau}{2}\Big)\norm{\nabla f(w_{t})}^{2}
   +\frac{\eta(1-\tau)}{2\tau}G^{2}
   +\frac{L\eta^{2}\tau}{2}G^{2}. 
\tag{13}
\end{align}

Because $\eta\le \tau/L$ (step‑size condition), we have $L\eta^{2}\tau\le \eta\tau$ and thus

\[
\frac{L\eta^{2}\tau}{2}G^{2}\le\frac{\eta\tau}{2}G^{2}.
\tag{14}
\]

Insert (14) into (13) and simplify the coefficients of $\norm{\nabla f(w_{t})}^{2}$:

\begin{align}
\text{[Step 9]} \quad
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})
   -\frac{\eta\tau}{2}\norm{\nabla f(w_{t})}^{2}
   +\frac{\eta}{2\tau}G^{2}.
\tag{15}
\end{align}

Take total expectation and sum over $t=0,\dots,T-1$:

\begin{align}
\text{[Step 10]} \quad
\E[f(w_{T})]
&\le f(w_{0})
   -\frac{\eta\tau}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_{t})}^{2}\big]
   +\frac{\eta T}{2\tau}G^{2}.
\tag{16}
\end{align}

Rearrange (16) and use $f(w_{T})\ge f^{\star}$:

\begin{align}
\text{[Step 11]} \quad
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_{t})}^{2}\big]
&\le
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta\tau T}
+\frac{G^{2}}{\tau}.
\tag{17}
\end{align}

Finally, replace $\eta$ by any admissible constant $\eta\le\tau/L$.
If we set $\eta=\frac{\tau}{L\sqrt{T}}$, then $\eta\le\tau/L$ holds and (17) becomes

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_{t})}^{2}\big]
\le
\underbrace{\frac{2L\big(f(w_{0})-f^{\star}\big)}{\tau^{2}}}_{\mathcal{O}(1)}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{G^{2}}{\tau}}_{\text{bias term}}.
\]

When $T$ is large the $\mathcal{O}(1/\sqrt{T})$ term dominates, establishing the claimed rate. \qed

\section*{Rate Derivation (With Constants)}
From (15) we have the per‑iteration descent
\[
\Delta_{t}\triangleq \E[f(w_{t})]-\E[f(w_{t+1})]
\ge \frac{\eta\tau}{2}\E\!\big[\norm{\nabla f(w_{t})}^{2}\big]-\frac{\eta}{2\tau}G^{2}.
\]
Summing $\Delta_{t}$ yields (16). Solving for the average gradient norm gives (2) with explicit constants
\[
C_{1}= \frac{2\big(f(w_{0})-f^{\star}\big)}{\tau},\qquad
C_{2}= \frac{G^{2}}{\tau},\qquad
\text{so}\;\;
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_{t})}^{2}\big]\le
\frac{C_{1}}{\eta T}+C_{2}\eta .
\]
Optimising w.r.t. $\eta$ (subject to $\eta\le \tau/L$) gives $\eta^{\star}= \min\!\big\{\sqrt{C_{1}/(C_{2}T)},\,\tau/L\big\}$,
which yields the $\mathcal{O}(1/\sqrt{T})$ rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No compression ($K=d$).} Then $\tau=1$, (A3) becomes an equality, and Theorem~\ref{thm:main} reduces to the standard non‑convex SGD bound.
\item \textbf{Extreme sparsity ($K=1$).} Here $\tau=1/d$, so the admissible step size shrinks by $1/d$ and the constant factor in front of $1/\sqrt{T}$ grows linearly with $d$. This matches known lower bounds for unbiased $1/d$‑compression.
\item \textbf{Deterministic Top‑K vs. Random $K$‑sparsification.} The proof only uses the contraction property (A3); any compressor satisfying the same $\tau$ yields the identical rate.
\end{itemize}
