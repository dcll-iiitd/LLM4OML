\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let 
\[
F(x)=f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}
  \item $f:\R^{d}\rightarrow\R$ is convex, differentiable and its gradient is $L$‑Lipschitz:
        \[
        \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
        \]
  \item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is proper, closed and convex (possibly nonsmooth).  
\end{itemize}
For a stepsize $\eta>0$, the \textbf{proximal gradient update} is
\begin{equation}
\boxed{\;
x_{k+1}= \prox_{\eta g}\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),\qquad k=0,1,2,\dots
\;}
\label{eq:update}
\end{equation}
where the proximal operator of $g$ is defined by
\[
\prox_{\eta g}(u)=\arg\min_{z\in\R^{d}}\Bigl\{g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\R^{d}$, max iterations $K$.
\FOR{$k=0$ \textbf{to} $K-1$}
    \STATE Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \STATE Take a gradient step $y_{k}\gets x_{k}-\eta g_{k}$.
    \STATE Apply proximal map $x_{k+1}\gets \prox_{\eta g}(y_{k})$.
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or the average $\bar x_{K}= \tfrac1K\sum_{k=1}^{K}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^{2}$, $g\equiv0$ (so $\prox_{\eta g}(u)=u$).  
Take $w_{0}=0$ and stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$\\\hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values $F(w_{k})=f(w_{k})$ are
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.7712)=1.5259,
\]
showing monotone decrease toward the optimum $f^{\star}=0$ at $w^{\star}=3$.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*.}, leftmargin=*]
\item \label{A1} $f$ is convex and differentiable with $L$‑Lipschitz gradient.
\item \label{A2} $g$ is proper, closed and convex.
\item \label{A3} The stepsize satisfies $0<\eta\le\frac{1}{L}$.
\item \label{A4} An optimal solution $x^{\star}\in\arg\min_{x}F(x)$ exists and $F(x^{\star})$ is finite.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence rate of PGM]\label{thm:rate}
Under Assumptions \ref{A1}–\ref{A4}, the iterates generated by \eqref{eq:update} satisfy for every $k\ge1$
\begin{equation}
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta k}.
\label{eq:rate}
\end{equation}
Consequently $F(x_{k})\to F(x^{\star})$ and the worst‑case decay is $\mathcal{O}(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The update \eqref{eq:update} is a non‑expansive mapping when $\eta\le 1/L$, guaranteeing that the sequence $\{x_{k}\}$ remains bounded.
\item \textbf{Hyperparameter sensitivity:} The constant $1/(2\eta)$ in \eqref{eq:rate} shows that a larger stepsize (up to $1/L$) yields a tighter bound; too large a stepsize destroys the descent property.
\item \textbf{Comparison:} Compared with plain gradient descent ($g\equiv0$) the rate is identical, while the proximal term enables handling nonsmooth regularizers (e.g., $\ell_{1}$) without sacrificing the $\mathcal{O}(1/k)$ guarantee.
\end{itemize}

\section*{Discussion}
The theorem is proved by exploiting two key facts:
\begin{enumerate}
\item The \emph{descent lemma} for $f$ (smoothness).
\item The \emph{non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{NE}
\]
These two ingredients allow us to relate $F(x_{k+1})$ to $F(x_{k})$ and to $F(x^{\star})$, leading to a telescoping series.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Proximal optimality condition]\label{lem:prox_opt}
For any $u\in\R^{d}$ and $\eta>0$, $z=\prox_{\eta g}(u)$ iff
\[
0\in \partial g(z)+\frac{1}{\eta}(z-u),
\]
i.e.
\[
\frac{1}{\eta}(u-z)\in \partial g(z).
\tag{PO}
\]
\end{lemma}
\begin{lemma}[Non‑expansiveness]\label{lem:nonexp}
For any $u,v\in\R^{d}$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
\tag{NE}
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x_{k+1}$ be defined by \eqref{eq:update} and denote $y_{k}=x_{k}-\eta\nabla f(x_{k})$.
We prove \eqref{eq:rate} by establishing a one‑step inequality and then telescoping.

\bigskip
\noindent\textbf{Step 1 (Apply optimality condition of the prox).}\\
From Lemma~\ref{lem:prox_opt} with $u=y_{k}$ we have
\[
\frac{1}{\eta}(y_{k}-x_{k+1})\in\partial g(x_{k+1}).
\tag{S1}
\]

\noindent\textbf{Step 2 (Convexity of $g$).}\\
For any $x^{\star}$,
\[
g(x_{k+1})-g(x^{\star})\le
\Big\langle\frac{1}{\eta}(y_{k}-x_{k+1}),\,x_{k+1}-x^{\star}\Big\rangle .
\tag{S2}
\]

\noindent\textbf{Step 3 (Expand $y_{k}=x_{k}-\eta\nabla f(x_{k})$).}\\
Insert $y_{k}$ into (S2):
\[
g(x_{k+1})-g(x^{\star})\le
\Big\langle\frac{1}{\eta}\bigl(x_{k}-\eta\nabla f(x_{k})-x_{k+1}\bigr),\,x_{k+1}-x^{\star}\Big\rangle .
\tag{S3}
\]

\noindent\textbf{Step 4 (Separate terms).}\\
Rewrite (S3) as
\[
g(x_{k+1})-g(x^{\star})\le
\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
-\langle\nabla f(x_{k}),x_{k+1}-x^{\star}\rangle .
\tag{S4}
\]

\noindent\textbf{Step 5 (Add $f$ and use descent lemma).}\\
Add $f(x_{k+1})-f(x^{\star})$ to both sides of (S4). By Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{S5}
\]
Hence
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le f(x_{k})-f(x^{\star})
+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle\\
&\quad+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{\eta}\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
-\langle\nabla f(x_{k}),x_{k+1}-x^{\star}\rangle .
\end{aligned}
\tag{S6}
\]

\noindent\textbf{Step 6 (Cancel the gradient inner products).}\\
Combine the terms containing $\nabla f(x_{k})$:
\[
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle-\langle\nabla f(x_{k}),x_{k+1}-x^{\star}\rangle
=-\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle .
\tag{S7}
\]

\noindent\textbf{Step 7 (Re‑write the bilinear term).}\\
Recall the identity
\[
2\langle a-b,c-b\rangle = \|a-c\|^{2}-\|a-b\|^{2}-\|c-b\|^{2}.
\tag{S8}
\]
Apply \eqref{S8} with $a=x_{k}$, $b=x_{k+1}$, $c=x^{\star}$:
\[
2\langle x_{k}-x_{k+1},x_{k+1}-x^{\star}\rangle
= \|x_{k}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}.
\tag{S9}
\]

\noindent\textbf{Step 8 (Insert (S7)–(S9) into (S6)).}\\
Dividing (S9) by $2\eta$ and substituting yields
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&\le f(x_{k})-f(x^{\star})-\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\\
&\qquad+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\end{aligned}
\tag{S10}
\]

\noindent\textbf{Step 9 (Use convexity of $f$).}\\
Convexity of $f$ gives
\[
f(x_{k})-f(x^{\star})\le\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle .
\tag{S11}
\]
Plugging \eqref{S11} into \eqref{S10} cancels the first two terms:
\[
F(x_{k+1})-F(x^{\star})
\le \frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\tag{S12}
\]

\noindent\textbf{Step 10 (Gather the quadratic terms).}\\
Combine the coefficients of $\|x_{k+1}-x_{k}\|^{2}$:
\[
\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}-\frac{1}{2\eta}\|x_{k}-x_{k+1}\|^{2}
=\frac{1}{2}\Bigl(L-\frac{1}{\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\tag{S13}
\]

\noindent\textbf{Step 11 (Choose stepsize $\eta\le 1/L$).}\\
Because $\eta\le 1/L$, we have $L-\frac{1}{\eta}\le0$, therefore the term \eqref{S13} is non‑positive and can be dropped:
\[
F(x_{k+1})-F(x^{\star})\le
\frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\tag{S14}
\]

\noindent\textbf{Step 12 (One‑step inequality).}\\
Inequality \eqref{S14} is the fundamental descent relation:
\[
\boxed{\;
2\eta\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}.
\;}
\tag{S15}
\]

\noindent\textbf{Step 13 (Telescoping sum).}\\
Sum \eqref{S15} for $k=0,\dots, K-1$:
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}
\le\|x_{0}-x^{\star}\|^{2}.
\tag{S16}
\]

\noindent\textbf{Step 14 (Average iterate bound).}\\
Since each term $F(x_{k+1})-F(x^{\star})\ge0$, we have
\[
K\bigl(F(x_{K})-F(x^{\star})\bigr)
\le\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\tag{S17}
\]

\noindent\textbf{Step 15 (Final rate).}\\
Dividing by $K$ yields exactly the bound \eqref{eq:rate}:
\[
F(x_{K})-F(x^{\star})\le\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{S18}
\]
This completes the proof.

\section*{Rate Derivation (With Constants)}
The constant in \eqref{eq:rate} is $C=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}$.  
If the optimal solution is unknown, one can use the bound
\[
F(\bar x_{K})-F(x^{\star})\le\frac{C}{K},
\quad\text{with }\bar x_{K}=\frac1K\sum_{k=1}^{K}x_{k},
\]
which follows from convexity of $F$ and the same telescoping argument.

\section*{Special Cases}
\begin{enumerate}[label=\textbf{Case \arabic*.}, leftmargin=*]
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal map reduces to the identity, and the proof collapses to the classical $O(1/k)$ rate for smooth convex optimization.
\item \textbf{Indicator of a convex set $C$ (projected gradient).} If $g=\iota_{C}$, then $\prox_{\eta g}$ is the Euclidean projection onto $C$. The same analysis applies, yielding the same $O(1/k)$ convergence for projected gradient descent.
\item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, a tighter inequality
\[
F(x_{k})-F(x^{\star})\le\bigl(1-\eta\mu\bigr)^{k}\bigl(F(x_{0})-F(x^{\star})\bigr)
\]
can be obtained by inserting the strong convexity inequality into Step 9.
\end{enumerate}

\section*{Conclusion}
We have presented a fully detailed derivation of the $\mathcal{O}(1/k)$ convergence rate for the proximal gradient method under standard convexity and smoothness assumptions. Every algebraic manipulation and inequality has been annotated with explicit step numbers, enabling precise verification of the logical flow.
