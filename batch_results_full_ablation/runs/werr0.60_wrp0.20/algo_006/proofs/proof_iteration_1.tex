\section*{SVRG (Stochastic Variance‑Reduced Gradient) for Non‑Convex Smooth Finite‑Sum Problems}

\section*{Mathematical Formulation}
Consider the finite‑sum objective
\[
\min_{x\in\mathbb{R}^d} f(x)\;,\qquad 
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is differentiable and $L$‑smooth (i.e., $\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|$ for all $u,v$).

SVRG proceeds in epochs $s=0,1,\dots$. At the beginning of epoch $s$ a \emph{snapshot} point $\tilde{x}^{\,s}$ is fixed and the full gradient
\[
\tilde{\mu}^{\,s}:=\grad f(\tilde{x}^{\,s})=\frac{1}{n}\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s})
\]
is computed.  
Inside the epoch $m$ inner updates are performed. Let $x_{t}^{\,s}$ denote the iterate after $t$ inner steps of epoch $s$ (with $x_{0}^{\,s}=\tilde{x}^{\,s}$). For a randomly drawn index $i_t\in\{1,\dots,n\}$ the \emph{variance‑reduced} stochastic gradient is
\[
v_{t}^{\,s}= \grad f_{i_t}\!\bigl(x_{t}^{\,s}\bigr)-\grad f_{i_t}\!\bigl(\tilde{x}^{\,s}\bigr)+\tilde{\mu}^{\,s}.
\tag{1}
\]
The inner update is a standard SGD step with a constant stepsize $\eta>0$:
\[
x_{t+1}^{\,s}=x_{t}^{\,s}-\eta\, v_{t}^{\,s},\qquad t=0,\dots,m-1.
\tag{2}
\]
After the inner loop the next snapshot is set to the last inner iterate,
\[
\tilde{x}^{\,s+1}=x_{m}^{\,s}.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for Non‑Convex $L$‑Smooth Finite‑Sum}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta$, initial point $x_0$
\State $\tilde{x}^{0}\gets x_0$
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\tilde{\mu}^{\,s}\gets \frac{1}{n}\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s})$
    \State $x_{0}^{\,s}\gets \tilde{x}^{\,s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t$ uniformly from $\{1,\dots,n\}$
        \State $v_{t}^{\,s}\gets \grad f_{i_t}(x_{t}^{\,s})-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$ \Comment{variance‑reduced estimator}
        \State $x_{t+1}^{\,s}\gets x_{t}^{\,s}-\eta\,v_{t}^{\,s}$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_{m}^{\,s}$
\EndFor
\State \Return $\hat{x}$ chosen uniformly at random from $\{x_{t}^{\,s}\mid s=0,\dots,S-1,\;t=0,\dots,m-1\}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the algorithm on the simple one‑dimensional quadratic
\[
f(w)=\frac{1}{2}(w-3)^2,
\qquad\text{so that } \grad f(w)=w-3.
\]
Here $n=1$, $f_1\equiv f$, thus the full gradient equals the stochastic gradient.

\begin{enumerate}[label=\textbf{Step \arabic*:},leftmargin=*]
\item \textbf{Initialization.} $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$ (single epoch).
\item \textbf{Full gradient at snapshot.} $\tilde{\mu}= \grad f(w_0)=0-3=-3$.
\item \textbf{Inner iteration $t=0$.}
    \begin{align*}
    v_0 &= \grad f(w_0)-\grad f(w_0)+\tilde{\mu}= -3,\\
    w_1 &= w_0-\eta v_0 = 0-0.1(-3)=0.3,\\
    f(w_1)&=\tfrac12(0.3-3)^2= \tfrac12( -2.7)^2=3.645.
    \end{align*}
\item \textbf{Inner iteration $t=1$.}
    \begin{align*}
    v_1 &= \grad f(w_1)-\grad f(w_0)+\tilde{\mu}
          =(0.3-3)-(-3)+(-3) = -2.7-(-3)-3 = -2.7+3-3 = -2.7,\\
    w_2 &= w_1-\eta v_1 = 0.3-0.1(-2.7)=0.57,\\
    f(w_2)&=\tfrac12(0.57-3)^2= \tfrac12( -2.43)^2\approx2.952.
    \end{align*}
\item \textbf{Inner iteration $t=2$.}
    \begin{align*}
    v_2 &= \grad f(w_2)-\grad f(w_0)+\tilde{\mu}
          =(0.57-3)-(-3)+(-3)= -2.43+3-3 = -2.43,\\
    w_3 &= w_2-\eta v_2 = 0.57-0.1(-2.43)=0.813,\\
    f(w_3)&=\tfrac12(0.813-3)^2\approx2.394.
    \end{align*}
\item The new snapshot would be $\tilde{x}^{1}=w_3=0.813$.
\end{enumerate}
The objective value decreases from $4.5$ (at $w_0$) to $2.394$ after three inner steps, demonstrating the variance‑reduced progress.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|,\qquad\forall u,v\in\mathbb{R}^d .
\]
Consequently $f$ is also $L$‑smooth.
\end{assumption}
\begin{assumption}[Bounded Below]\label{ass:lower}
The objective $f$ is bounded below, i.e., there exists $f^{\star}>-\infty$ such that $f(x)\ge f^{\star}$ for all $x$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le\frac{1}{3L}.
\]
\end{assumption}
\begin{assumption}[Inner Loop Length]\label{ass:inner}
The inner loop length $m$ is a positive integer. In the theorem we will set $m=2n$ (any $m\ge 1$ works, but $m=O(n)$ yields the best complexity).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let Assumptions \ref{ass:smooth}–\ref{ass:inner} hold. Run SVRG for $S$ epochs with inner length $m$ and stepsize $\eta\le 1/(3L)$. Define the random output $\hat{x}$ as a uniformly chosen iterate among all $Sm$ inner points. Then
\[
\EE\!\left[\,\|\grad f(\hat{x})\|^{2}\right]
\;\le\;
\frac{4L\bigl(f(x_{0})-f^{\star}\bigr)}{S\,m } .
\tag{4}
\]
Consequently, to obtain $\EE[\|\grad f(\hat{x})\|^{2}]\le\varepsilon$ it suffices to choose
\[
S = \mathcal{O}\!\left(\frac{L\bigl(f(x_{0})-f^{\star}\bigr)}{m\varepsilon}\right)
\quad\Longleftrightarrow\quad
\text{total stochastic gradient evaluations } = \mathcal{O}\!\left(\left(n+\frac{L}{\varepsilon}\right)\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (4) depends only on $L$, $m$, and the initial optimality gap; it is independent of the variance of stochastic gradients because the estimator (1) is unbiased and its variance vanishes at the snapshot.
\item \textbf{Hyperparameter Sensitivity.} The stepsize condition $\eta\le 1/(3L)$ is mild; in practice larger $\eta$ often works, but the proof guarantees convergence only within this range. The inner length $m$ trades off the cost of computing the full gradient ($\mathcal{O}(n)$) against the reduction of variance; choosing $m\approx 2n$ yields the optimal $\mathcal{O}(n+L/\varepsilon)$ complexity.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Plain SGD.} For non‑convex $L$‑smooth problems SGD with diminishing stepsizes attains $\EE[\|\grad f(\hat{x})\|^{2}]=\mathcal{O}(1/\sqrt{T})$, i.e. $\mathcal{O}(1/\varepsilon^{2})$ stochastic gradients.
        \item \emph{SVRG (Theorem \ref{thm:main})}. Achieves a linear dependence on $1/\varepsilon$, matching the best known rates for finite‑sum non‑convex optimization.
    \end{enumerate}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $u,v\in\mathbb{R}^d$,
\[
f(v)\le f(u)+\langle \grad f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Unbiasedness and Variance of $v_t^{\,s}$]\label{lem:unbiased}
For a fixed snapshot $\tilde{x}$,
\[
\EE_{i_t}\!\bigl[v_t\bigr]=\grad f(x_t),\qquad
\EE_{i_t}\!\bigl[\|v_t-\grad f(x_t)\|^{2}\bigr]
\le L^{2}\|x_t-\tilde{x}\|^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
\[
\EE_{i_t}[v_t]=\frac{1}{n}\sum_{i=1}^{n}\bigl(\grad f_i(x_t)-\grad f_i(\tilde{x})\bigr)+\grad f(\tilde{x})
      =\grad f(x_t).
\]
For the variance, using smoothness of each $f_i$,
\[
\|\grad f_i(x_t)-\grad f_i(\tilde{x})\|
    \le L\|x_t-\tilde{x}\|,
\]
hence
\[
\|v_t-\grad f(x_t)\|
    =\bigl\|\grad f_i(x_t)-\grad f_i(\tilde{x})-\bigl(\grad f(x_t)-\grad f(\tilde{x})\bigr)\bigr\|
    \le 2L\|x_t-\tilde{x}\|,
\]
and squaring and taking expectation yields (6). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout we fix an epoch $s$ and suppress the superscript $s$ for readability.

\noindent\textbf{[Step 1]} (Descent Lemma). By Lemma \ref{lem:smooth} with $u=x_t$ and $v=x_{t+1}=x_t-\eta v_t$,
\[
f(x_{t+1})\le f(x_t)-\eta\langle\grad f(x_t),v_t\rangle+\frac{L\eta^{2}}{2}\|v_t\|^{2}.
\tag{7}
\]

\noindent\textbf{[Step 2]} (Take expectation w.r.t. $i_t$). Using unbiasedness of $v_t$,
\[
\EE_{i_t}\!\bigl[\langle\grad f(x_t),v_t\rangle\bigr]
    =\|\grad f(x_t)\|^{2}.
\tag{8}
\]

\noindent\textbf{[Step 3]} (Bound the second moment). From $\|a\|^{2}= \|\EE a\|^{2}+\EE\|a-\EE a\|^{2}$,
\[
\EE_{i_t}\!\bigl[\|v_t\|^{2}\bigr]
    =\|\grad f(x_t)\|^{2}+ \EE_{i_t}\!\bigl[\|v_t-\grad f(x_t)\|^{2}\bigr]
    \le\|\grad f(x_t)\|^{2}+L^{2}\|x_t-\tilde{x}\|^{2},
\tag{9}
\]
where we used Lemma \ref{lem:unbiased}.

\noindent\textbf{[Step 4]} (Insert (8)–(9) into (7) and take full expectation). Let $\mathcal{F}_t$ be the sigma‑algebra generated by all randomness up to step $t$. Then
\[
\EE\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
\le f(x_t)-\eta\|\grad f(x_t)\|^{2}
      +\frac{L\eta^{2}}{2}\Bigl(\|\grad f(x_t)\|^{2}+L^{2}\|x_t-\tilde{x}\|^{2}\Bigr).
\tag{10}
\]

\noindent\textbf{[Step 5]} (Choose $\eta\le 1/(3L)$). Since $\eta\le 1/(3L)$,
\[
\eta-\frac{L\eta^{2}}{2}\ge \frac{\eta}{2}.
\tag{11}
\]
Applying (11) to (10) yields
\[
\EE\!\bigl[f(x_{t+1})\mid\mathcal{F}_t\bigr]
\le f(x_t)-\frac{\eta}{2}\|\grad f(x_t)\|^{2}
      +\frac{L^{3}\eta^{2}}{2}\|x_t-\tilde{x}\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 6]} (Recursion for the distance to the snapshot). By the update rule,
\[
x_{t+1}-\tilde{x}=x_t-\tilde{x}-\eta v_t,
\]
hence
\[
\|x_{t+1}-\tilde{x}\|^{2}
= \|x_t-\tilde{x}\|^{2}-2\eta\langle x_t-\tilde{x},v_t\rangle+\eta^{2}\|v_t\|^{2}.
\tag{13}
\]
Taking expectation conditioned on $\mathcal{F}_t$ and using $\EE[v_t]=\grad f(x_t)$ gives
\[
\EE\!\bigl[\|x_{t+1}-\tilde{x}\|^{2}\mid\mathcal{F}_t\bigr]
\le \|x_t-\tilde{x}\|^{2}
     -2\eta\langle x_t-\tilde{x},\grad f(x_t)\rangle
     +\eta^{2}\Bigl(\|\grad f(x_t)\|^{2}+L^{2}\|x_t-\tilde{x}\|^{2}\Bigr).
\tag{14}
\]

\noindent\textbf{[Step 7]} (Use smoothness to bound the cross term). From $L$‑smoothness,
\[
f(x_t)-f(\tilde{x})\le\langle\grad f(x_t),x_t-\tilde{x}\rangle
                     +\frac{L}{2}\|x_t-\tilde{x}\|^{2},
\]
so that
\[
\langle x_t-\tilde{x},\grad f(x_t)\rangle\ge f(x_t)-f(\tilde{x})-\frac{L}{2}\|x_t-\tilde{x}\|^{2}.
\tag{15}
\]

\noindent\textbf{[Step 8]} (Plug (15) into (14) and simplify). After substitution and rearranging,
\[
\EE\!\bigl[\|x_{t+1}-\tilde{x}\|^{2}\mid\mathcal{F}_t\bigr]
\le (1+L\eta^{2})\|x_t-\tilde{x}\|^{2}
    -2\eta\bigl(f(x_t)-f(\tilde{x})\bigr)
    +\eta^{2}\|\grad f(x_t)\|^{2}.
\tag{16}
\]

\noindent\textbf{[Step 9]} (Define a Lyapunov function). Let
\[
\Phi_t:=\EE\!\bigl[f(x_t)\bigr]
       +\alpha\,\EE\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr],
\]
where $\alpha>0$ will be chosen later. Taking total expectation in (12) and (16) and adding $\alpha$ times (16) to (12) gives
\[
\Phi_{t+1}
\le \Phi_t
   -\Bigl(\frac{\eta}{2}-\alpha\eta^{2}\Bigr)\EE\!\bigl[\|\grad f(x_t)\|^{2}\bigr]
   -2\alpha\eta\,\EE\!\bigl[f(x_t)-f(\tilde{x})\bigr]
   +\Bigl(\frac{L^{3}\eta^{2}}{2}+\alpha L\eta^{2}\Bigr)\EE\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr].
\tag{17}
\]

\noindent\textbf{[Step 10]} (Choose $\alpha$). Set $\alpha=\frac{L}{2}$. With the stepsize bound $\eta\le 1/(3L)$ we have
\[
\frac{\eta}{2}-\alpha\eta^{2}
= \frac{\eta}{2}-\frac{L}{2}\eta^{2}
\ge \frac{\eta}{3},
\qquad
\frac{L^{3}\eta^{2}}{2}+\alpha L\eta^{2}
= \frac{L^{3}\eta^{2}}{2}+\frac{L^{2}\eta^{2}}{2}
\le \frac{2L^{3}\eta^{2}}{2}=L^{3}\eta^{2}\le \frac{L}{3}\eta,
\]
where the last inequality again uses $\eta\le 1/(3L)$. Hence (17) simplifies to
\[
\Phi_{t+1}
\le \Phi_t
   -\frac{\eta}{3}\EE\!\bigl[\|\grad f(x_t)\|^{2}\bigr]
   -L\eta\,\EE\!\bigl[f(x_t)-f(\tilde{x})\bigr]
   +\frac{L}{3}\eta\,\EE\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr].
\tag{18}
\]

\noindent\textbf{[Step 11]} (Telescoping over the inner loop). Summing (18) for $t=0,\dots,m-1$ and noting that $\Phi_0 = f(\tilde{x})$ (since $\|x_0-\tilde{x}\|=0$) yields
\[
\sum_{t=0}^{m-1}\EE\!\bigl[\|\grad f(x_t)\|^{2}\bigr]
\le \frac{3}{\eta}\Bigl(f(\tilde{x})- \EE[f(x_{m})]\Bigr)
     + 3L\sum_{t=0}^{m-1}\EE\!\bigl[f(x_t)-f(\tilde{x})\bigr]
     + L\sum_{t=0}^{m-1}\EE\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr].
\tag{19}
\]

\noindent\textbf{[Step 12]} (Bounding the last two sums). Using $f(x_t)\ge f^{\star}$ and $\|x_t-\tilde{x}\|^{2}\ge0$, the non‑negative terms can be dropped, giving the crude inequality
\[
\sum_{t=0}^{m-1}\EE\!\bigl[\|\grad f(x_t)\|^{2}\bigr]
\le \frac{3}{\eta}\bigl(f(\tilde{x})-f^{\star}\bigr).
\tag{20}
\]

\noindent\textbf{[Step 13]} (Average over all inner iterates of all epochs). After $S$ epochs, the total number of inner iterates is $Sm$. Choose $\hat{x}$ uniformly at random among them. Taking expectation over the random choice,
\[
\EE\!\bigl[\|\grad f(\hat{x})\|^{2}\bigr]
= \frac{1}{Sm}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
   \EE\!\bigl[\|\grad f(x_{t}^{\,s})\|^{2}\bigr]
\le \frac{1}{Sm}\sum_{s=0}^{S-1}\frac{3}{\eta}
        \bigl(f(\tilde{x}^{\,s})-f^{\star}\bigr).
\tag{21}
\]

\noindent\textbf{[Step 14]} (Monotonicity of the snapshot values). From (12) with $t=m-1$ and $\eta\le 1/(3L)$ we have $f(\tilde{x}^{\,s+1})\le f(\tilde{x}^{\,s})$, i.e. the sequence $\{f(\tilde{x}^{\,s})\}$ is non‑increasing. Hence each term $f(\tilde{x}^{\,s})-f^{\star}\le f(x_{0})-f^{\star}$.

\noindent\textbf{[Step 15]} (Finalize the bound). Substituting the uniform bound into (21) gives
\[
\EE\!\bigl[\|\grad f(\hat{x})\|^{2}\bigr]
\le \frac{3}{\eta Sm}\bigl(f(x_{0})-f^{\star}\bigr).
\]
Choosing $\eta = \frac{1}{3L}$ (the largest admissible stepsize) yields exactly the statement of Theorem \ref{thm:main}:
\[
\EE\!\bigl[\|\grad f(\hat{x})\|^{2}\bigr]
\le \frac{4L\bigl(f(x_{0})-f^{\star}\bigr)}{Sm}.
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main}, to achieve $\EE[\|\grad f(\hat{x})\|^{2}]\le\varepsilon$ it suffices that
\[
Sm \;\ge\; \frac{4L\bigl(f(x_{0})-f^{\star}\bigr)}{\varepsilon}.
\]
Each epoch costs $n$ gradient evaluations for the full gradient plus $m$ stochastic updates, i.e.
\[
\text{grad\_evals}= S\,(n+m).
\]
Setting $m=2n$ (a common choice) gives
\[
\text{grad\_evals}
= S\,(3n)
= \mathcal{O}\!\left(n + \frac{L}{\varepsilon}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $L$‑smooth case.} The same proof without the non‑negativity drop in Step 12 yields a faster $\mathcal{O}(1/T)$ decrease of the optimality gap $f(\hat{x})-f^{\star}$.
\item \textbf{Infinite‑sum (online) setting.} If $n\to\infty$, the full gradient is replaced by a large mini‑batch estimate; the analysis follows similarly with an additional variance term.
\item \textbf{Choosing $m\neq 2n$.} The bound (4) holds for any $m$, but the total complexity is minimized when $m=\Theta(n)$, balancing the $n$ full‑gradient cost with the inner‑loop progress.
\end{itemize}
