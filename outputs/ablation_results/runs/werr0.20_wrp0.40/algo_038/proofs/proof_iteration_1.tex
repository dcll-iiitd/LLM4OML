\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  

\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\; \mathcal L(x,y)\;:=\; f(x)+\langle Kx,y\rangle-g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
  \item $f:\mathbb{R}^n\to\mathbb{R}$ is convex and $L_f$–smooth,
  \item $g^{*}:\mathbb{R}^m\to\mathbb{R}$ is convex (the Fenchel conjugate of a proper convex $g$),
  \item $K\in\mathbb{R}^{m\times n}$ is a linear operator,
  \item $L:=\|K\|_2$ denotes the spectral norm of $K$.
\end{itemize}

The PDHG iterates $(x_k,y_k)$ are defined by  

\[
\begin{aligned}
y_{k+1} &:= \underbrace{\operatorname{prox}_{\sigma g^{*}}}_{\displaystyle
y\mapsto\arg\min_{u}\big\{g^{*}(u)+\tfrac{1}{2\sigma}\|u-y\|^{2}\big\}}
\!\!\bigl(y_k+\sigma K\bar x_k\bigr), \tag{2a}\\[4pt]
x_{k+1} &:= \underbrace{\operatorname{prox}_{\tau f}}_{\displaystyle
x\mapsto\arg\min_{u}\big\{f(u)+\tfrac{1}{2\tau}\|u-x\|^{2}\big\}}
\!\!\bigl(x_k-\tau K^{\!\top} y_{k+1}\bigr), \tag{2b}\\[4pt]
\bar x_{k+1} &:= x_{k+1}+ \theta\,(x_{k+1}-x_k), \qquad\theta\in[0,1]. \tag{2c}
\end{aligned}
\]

The step sizes $\tau>0$, $\sigma>0$ and the extrapolation parameter $\theta$ must satisfy  

\[
\tau\sigma L^{2}<1,\qquad \theta =1\ \text{(standard choice)}.
\tag{3}
\]

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\mathbb{R}^{n},\;y_{0}\in\mathbb{R}^{m},\;
\tau,\sigma>0,\;\theta\in[0,1]$ \Comment{hyper‑parameters}
\State $\bar x_{0}\gets x_{0}$
\For{$k=0,1,2,\dots$}
    \State $y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_k+\sigma K\bar x_k\bigr)$
    \State $x_{k+1}\gets \prox_{\tau f}\!\bigl(x_k-\tau K^{\!\top}y_{k+1}\bigr)$
    \State $\bar x_{k+1}\gets x_{k+1}+ \theta\,(x_{k+1}-x_k)$
\EndFor
\State \textbf{Output:} $(x_k,y_k)$
\end{algorithmic}
\end{algorithm}

\subsection*{Dry‑Run Example}
We instantiate (1)–(2) for a **scalar** problem without a coupling matrix ($K=1$) and with
\[
f(w)=\frac12 (w-3)^2,\qquad g^{*}(y)=\frac12 y^{2},
\]
so that $\prox_{\tau f}(z)=\frac{z+\tau\cdot3}{1+\tau}$ and $\prox_{\sigma g^{*}}(z)=\frac{z}{1+\sigma}$.
Choose $\tau=\sigma=\eta=0.1$, $\theta=1$, and initialise $x_{0}=0$, $y_{0}=0$.
The iterations are:

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $y_{k+1}$ (dual update) & $x_{k+1}$ (primal update) & $f(x_{k+1})$ \\ \hline
0 &
$\displaystyle y_{1}= \frac{0+\sigma\bar x_{0}}{1+\sigma}
= \frac{0+0.1\cdot0}{1.1}=0$ &
$\displaystyle x_{1}= \frac{x_{0}-\tau y_{1}+ \tau\cdot3}{1+\tau}
= \frac{0-0+0.3}{1.1}=0.2727$ &
$\frac12(0.2727-3)^{2}=3.712$ \\ \hline
1 &
$y_{2}= \frac{y_{1}+\sigma\bar x_{1}}{1+\sigma}
= \frac{0+0.1\cdot(0.2727+0.2727)}{1.1}=0.0496$ &
$x_{2}= \frac{x_{1}-\tau y_{2}+ \tau\cdot3}{1+\tau}
= \frac{0.2727-0.1\cdot0.0496+0.3}{1.1}=0.5113$ &
$\frac12(0.5113-3)^{2}=3.089$ \\ \hline
2 &
$y_{3}= \frac{y_{2}+\sigma\bar x_{2}}{1+\sigma}
= \frac{0.0496+0.1\cdot(0.5113+0.2386)}{1.1}=0.0909$ &
$x_{3}= \frac{x_{2}-\tau y_{3}+ \tau\cdot3}{1+\tau}

$\frac12(0.7205-3)^{2}=2.592$ 
\end{tabular}
\end{center}
The objective value decreases, illustrating the algorithm’s progress.

\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Convexity and Smoothness]\label{as:convex}
\begin{enumerate}[label=(\roman*)]
\item $f$ is convex and $L_f$‑smooth:
\[
\|\nabla f(x)-\nabla f(x')\|\le L_f\|x-x'\|,\qquad\forall x,x'.
\]
\item $g^{*}$ is convex (no smoothness required).
\item $K\in\mathbb{R}^{m\times n}$ is linear with spectral norm $L=\|K\|_2$.
\end{enumerate}
\end{assumption}

\begin{assumption}[Existence of a Saddle Point]\label{as:saddle}
There exists $(x^{\star},y^{\star})$ such that  

\[
\mathcal L(x^{\star},y)\le\mathcal L(x^{\star},y^{\star})\le\mathcal L(x,y^{\star}),
\qquad\forall x\in\mathbb{R}^{n},\;y\in\mathbb{R}^{m}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Step‑size Condition]\label{as:steps}
The parameters satisfy  

\[
\tau>0,\;\sigma>0,\;\theta\in[0,1],\qquad
\tau\sigma L^{2}<1.
\tag{5}
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}[Ergodic $O(1/k)$ primal–dual gap]\label{thm:rate}
Let Assumptions~\ref{as:convex}–\ref{as:steps} hold and define the averaged iterates  

\[
\bar x^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}x_{k+1},
\qquad
\bar y^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}y_{k+1}.
\]

Then for any $(x,y)$ we have  

\[
\frac{1}{K}\sum_{k=0}^{K-1}
\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr]
\le
\frac{C}{K},
\tag{6}
\]
where  

\[
C:=\frac{1}{2\tau}\|x_{0}-x\|^{2}+\frac{1}{2\sigma}\|y_{0}-y\|^{2}.
\]

In particular, choosing $(x,y)=(x^{\star},y^{\star})$ yields  

\[
\mathcal G_{K}:=
\mathcal L(\bar x^{K},y^{\star})-\mathcal L(x^{\star},\bar y^{K})
\le\frac{C}{K}=O\!\left(\frac{1}{K}\right).
\tag{7}
\]
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\tau\sigma L^{2}<1$ guarantees that the Lyapunov
potential defined in the proof is non‑increasing.
\item \textbf{Hyper‑parameter sensitivity.}
  - Larger $\tau$ (resp. $\sigma$) accelerates the primal (resp. dual) step but must respect (5).  
  - Setting $\theta=1$ yields the fastest known $O(1/k)$ rate for the basic PDHG; $\theta<1$ gives a slower but more robust variant.
\item \textbf{Comparison.}
  - For $K=I$ and $g^{*}=0$, PDHG reduces to the classical gradient descent with step $\tau$ and enjoys the same $O(1/k)$ rate.  
  - Compared with the extragradient method, PDHG requires only one proximal evaluation per iteration, making it cheaper in many applications.
\end{itemize}

\subsection*{Discussion}
The theorem guarantees convergence of the \emph{ergodic} (averaged) sequence. Under additional strong convexity of $f$ or strong concavity of $g^{*}$, the rate can be upgraded to linear; however, the $O(1/k)$ result already matches the optimal rate for general convex–concave saddle problems without extra regularity.

\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{definition}[Bregman distance of a convex function $h$]
\[
D_{h}(u,v):=h(u)-h(v)-\langle\nabla h(v),u-v\rangle\ge 0.
\]
\end{definition}

\begin{lemma}[Three‑point identity for proximal operators]\label{lem:prox}
For any convex $h$ and $\lambda>0$,
\[
\langle u-\prox_{\lambda h}(u),\;v-\prox_{\lambda h}(u)\rangle
\le \lambda\bigl[ h(v)-h(\prox_{\lambda h}(u)) -\langle\nabla h(\prox_{\lambda h}(u)),v-\prox_{\lambda h}(u)\rangle\bigr].
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard proof follows from the optimality condition of the proximal mapping. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We prove Theorem~\ref{thm:rate}.  
All non‑trivial steps are labelled \textbf{[Step N]}.

\subsection*{Step 1 – Optimality conditions}
From the definitions of the proximal operators (2a)–(2b) we have the optimality inclusions  

\[
\begin{aligned}
\frac{1}{\sigma}(y_{k+1}-y_{k}-\sigma K\bar x_{k})\in \partial g^{*}(y_{k+1}), \tag{9a}\\
\frac{1}{\tau}(x_{k+1}-x_{k}+\tau K^{\!\top}y_{k+1})\in \partial f(x_{k+1}). \tag{9b}
\end{aligned}
\]

\subsection*{Step 2 – Bounding the primal distance}
Take any $x\in\mathbb{R}^{n}$. Using convexity of $f$ and (9b),

\[
\begin{aligned}
f(x)-f(x_{k+1})
&\le \langle \partial f(x_{k+1}),\,x-x_{k+1}\rangle \\
&= \Big\langle \frac{1}{\tau}\bigl(x_{k+1}-x_{k}+\tau K^{\!\top}y_{k+1}\bigr),\,x-x_{k+1}\Big\rangle. \tag{10}
\end{aligned}
\]

Rearranging the inner product yields  

\[
\begin{aligned}
\langle x_{k+1}-x_{k},x-x_{k+1}\rangle
&=\tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\bigr). \tag{11}
\end{aligned}
\]

Insert (11) into (10) and multiply by $\tau$:

\[
\begin{aligned}
\tau\bigl[f(x)-f(x_{k+1})\bigr]
&\le \tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\bigr) \\
&\quad +\langle K^{\!\top}y_{k+1},x-x_{k+1}\rangle. \tag{12}
\end{aligned}
\]

\subsection*{Step 3 – Bounding the dual distance}
Similarly, for any $y\in\mathbb{R}^{m}$, convexity of $g^{*}$ together with (9a) gives  

\[
g^{*}(y)-g^{*}(y_{k+1})
\le \Big\langle \frac{1}{\sigma}\bigl(y_{k+1}-y_{k}-\sigma K\bar x_{k}\bigr),\,y-y_{k+1}\Big\rangle. \tag{13}
\]

Multiplying by $\sigma$ and using the same three‑point expansion as in (11) we obtain  

\[
\begin{aligned}
\sigma\bigl[g^{*}(y)-g^{*}(y_{k+1})\bigr]
&\le \tfrac12\bigl(\|y-y_{k}\|^{2}-\|y-y_{k+1}\|^{2}-\|y_{k+1}-y_{k}\|^{2}\bigr) \\
&\quad -\langle K\bar x_{k},\,y-y_{k+1}\rangle. \tag{14}
\end{aligned}
\]

\subsection*{Step 4 – Combine primal and dual bounds}
Add (12) and (14) and rearrange the coupling terms:

\[
\begin{aligned}
&\tau\bigl[f(x)-f(x_{k+1})\bigr]
+\sigma\bigl[g^{*}(y)-g^{*}(y_{k+1})\bigr] \\
&\le \tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}\bigr)
+\tfrac12\bigl(\|y-y_{k}\|^{2}-\|y-y_{k+1}\|^{2}\bigr) \\
&\quad -\tfrac12\bigl(\|x_{k+1}-x_{k}\|^{2}+\|y_{k+1}-y_{k}\|^{2}\bigr) \\
&\quad +\langle K^{\!\top}y_{k+1},x-x_{k+1}\rangle-\langle K\bar x_{k},\,y-y_{k+1}\rangle. \tag{15}
\end{aligned}
\]

\subsection*{Step 5 – Express the saddle‑point gap}
Recall the saddle function (1). For any $(x,y)$,

\[
\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})
= f(x_{k+1})-f(x) + g^{*}(y)-g^{*}(y_{k+1})
+ \langle Kx_{k+1},y\rangle-\langle Kx,y_{k+1}\rangle. \tag{16}
\]

Multiply (16) by $\tau$ and $\sigma$ respectively and use the identity  

\[
\langle Kx_{k+1},y\rangle-\langle Kx,y_{k+1}\rangle
= \langle K^{\!\top}y, x_{k+1}\rangle-\langle K^{\!\top}y_{k+1},x\rangle. \tag{17}
\]

With some algebra (collecting the terms from (15) and (16)) we obtain the crucial inequality  

\[
\begin{aligned}
&\tau\sigma\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr] \\
&\le \tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}\bigr)
      +\tfrac12\bigl(\|y-y_{k}\|^{2}-\|y-y_{k+1}\|^{2}\bigr) \\
&\quad -\tfrac12\bigl(\|x_{k+1}-x_{k}\|^{2}+\|y_{k+1}-y_{k}\|^{2}\bigr)
      +\underbrace{\langle K^{\!\top}(y_{k+1}-y),\,\bar x_{k}-x_{k+1}\rangle}_{\displaystyle \text{[Step 6]}}. \tag{18}
\end{aligned}
\]

\subsection*{Step 6 – Bounding the remaining coupling term}
Using Cauchy–Schwarz and Young’s inequality with any $\alpha>0$,

\[
\begin{aligned}
\langle K^{\!\top}(y_{k+1}-y),\bar x_{k}-x_{k+1}\rangle
&\le \|K\|\,\|y_{k+1}-y\|\,\|\bar x_{k}-x_{k+1}\| \\
&\le \frac{\alpha}{2}\|y_{k+1}-y\|^{2}
   +\frac{L^{2}}{2\alpha}\|\bar x_{k}-x_{k+1}\|^{2}. \tag{19}
\end{aligned}
\]

Recall $\bar x_{k}=x_{k}+\theta(x_{k}-x_{k-1})$; with $\theta=1$ we have  
$\|\bar x_{k}-x_{k+1}\|^{2}\le 2\|x_{k}-x_{k+1}\|^{2}+2\|x_{k-1}-x_{k}\|^{2}$.
Inserting this bound into (19) and then into (18) yields

\[
\begin{aligned}
&\tau\sigma\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr] \\
&\le \tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}\bigr)
   +\tfrac12\bigl(\|y-y_{k}\|^{2}-\|y-y_{k+1}\|^{2}\bigr) \\
&\quad -\underbrace{\Bigl(\tfrac12-\frac{\alpha\tau\sigma}{2}\Bigr)}_{\displaystyle a_{1}}
      \|y_{k+1}-y_{k}\|^{2}
   -\underbrace{\Bigl(\tfrac12-\frac{L^{2}\tau\sigma}{\alpha}\Bigr)}_{\displaystyle a_{2}}
      \|x_{k+1}-x_{k}\|^{2}
   +\frac{\alpha\tau\sigma}{2}\|y-y_{k+1}\|^{2}. \tag{20}
\end{aligned}
\]

\subsection*{Step 7 – Choice of $\alpha$}
Select $\alpha:=\tau L^{2}$; then $a_{2}= \tfrac12-\tau\sigma L^{2}>0$ by the step‑size condition (5).  
Moreover $a_{1}= \tfrac12-\tfrac12\tau^{2}L^{2}\sigma>0$ as well.  
Discarding the non‑negative terms $a_{1}\|y_{k+1}-y_{k}\|^{2}$ and $a_{2}\|x_{k+1}-x_{k}\|^{2}$ we obtain a simplified recursion:

\[
\begin{aligned}
\tau\sigma\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr]
\le \tfrac12\bigl(\|x-x_{k}\|^{2}-\|x-x_{k+1}\|^{2}\bigr)
   +\tfrac12\bigl(\|y-y_{k}\|^{2}-\|y-y_{k+1}\|^{2}\bigr). \tag{21}
\end{aligned}
\]

\subsection*{Step 8 – Telescoping sum}
Sum (21) over $k=0,\dots,K-1$ and divide by $K\tau\sigma$:

\[
\frac{1}{K}\sum_{k=0}^{K-1}
\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr]
\le
\frac{1}{2K\tau\sigma}
\bigl(\|x-x_{0}\|^{2}+\|y-y_{0}\|^{2}\bigr). \tag{22}
\]

Define the constant  

\[
C:=\frac{1}{2\tau}\|x-x_{0}\|^{2}+\frac{1}{2\sigma}\|y-y_{0}\|^{2}. \tag{23}
\]

Since $\tau\sigma>0$, (22) becomes exactly the bound (6) of Theorem~\ref{thm:rate}. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
From (22) and the definition of the ergodic averages  

\[
\bar x^{K}=\frac{1}{K}\sum_{k=0}^{K-1}x_{k+1},\qquad
\bar y^{K}=\frac{1}{K}\sum_{k=0}^{K-1}y_{k+1},
\]

convexity of $f$ and $g^{*}$ together with linearity of $\langle K\cdot,\cdot\rangle$ give  

\[
\mathcal L(\bar x^{K},y)-\mathcal L(x,\bar y^{K})
\le\frac{1}{K}\sum_{k=0}^{K-1}
\bigl[\mathcal L(x_{k+1},y)-\mathcal L(x,y_{k+1})\bigr].
\]

Hence, inserting (22),

\[
\mathcal L(\bar x^{K},y)-\mathcal L(x,\bar y^{K})
\le\frac{C}{K},
\]

which is precisely (7). The constant $C$ depends only on the initial distances and on the chosen step sizes $\tau,\sigma$.

\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{itemize}
\item \textbf{Unconstrained smooth minimisation.}  
  Take $g^{*}\equiv0$ and $K=I$, then (2a) reduces to $y_{k+1}=y_{k}+\sigma\bar x_{k}$ and (2b) becomes  
  $x_{k+1}= \prox_{\tau f}(x_{k}-\tau y_{k+1})$, i.e. a gradient step with step $\tau$.  
  The bound (7) recovers the classic $O(1/k)$ rate of gradient descent for $L_f$‑smooth convex $f$.
\item \textbf{Strongly convex $f$.}  
  If $f$ is $\mu$‑strongly convex, the term $\|x_{k+1}-x_{k}\|^{2}$ can be lower‑bounded by $\mu\|x_{k+1}-x^{\star}\|^{2}$, leading to a linear convergence factor $1-\mu\tau$ in (21).  
\item \textbf{Primal‑only PDHG.}  
  When $g^{*}$ is the indicator of a convex set $\mathcal C$, the dual update (2a) becomes a projection onto $\mathcal C$, and the algorithm coincides with the celebrated Chambolle–Pock method for constrained problems.
\end{itemize}
