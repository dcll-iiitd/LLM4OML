\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let  
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\mathbb{R}^{d},
\]
where each $f_i$ is $L$–smooth (possibly non‑convex).  
SVRG proceeds in \emph{epochs} indexed by $s=0,1,\dots,S-1$.  
At the beginning of epoch $s$ a \emph{snapshot} $\tilde{x}^{s}$ is fixed and the \emph{full gradient}
\[
\tilde{\mu}^{s}:=\nabla f(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s})
\]
is computed.  

Inside epoch $s$ we perform $m$ inner stochastic updates.  
For $t=0,\dots,m-1$ let $x_{t}^{s}$ denote the current iterate (with the convention $x_{0}^{s}=\tilde{x}^{s}$).  
Sample an index $i_t^{s}\in\{1,\dots,n\}$ uniformly at random and construct the variance‑reduced estimator
\[
v_{t}^{s}\;:=\;\nabla f_{i_t^{s}}(x_{t}^{s})-\nabla f_{i_t^{s}}(\tilde{x}^{s})+\tilde{\mu}^{s}.
\]
The inner update is
\[
x_{t+1}^{s}=x_{t}^{s}-\eta\,v_{t}^{s},
\tag{1}
\]
where $\eta>0$ is a constant stepsize.  
After $m$ inner steps we set the next snapshot $\tilde{x}^{s+1}=x_{m}^{s}$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (Non‑Convex Smooth)}\label{alg:svrg}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}$, stepsize $\eta$, inner loop length $m$, number of epochs $S$.
\STATE $\tilde{x}^{0}\gets x_{0}$
\FOR{$s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{s}\gets \nabla f(\tilde{x}^{s})$
    \STATE $x_{0}^{s}\gets \tilde{x}^{s}$
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t^{s}\sim\text{Uniform}\{1,\dots,n\}$
        \STATE $v_{t}^{s}\gets \nabla f_{i_t^{s}}(x_{t}^{s})-\nabla f_{i_t^{s}}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \STATE $x_{t+1}^{s}\gets x_{t}^{s}-\eta\,v_{t}^{s}$   \COMMENT{Eq.~(1)}
    \ENDFOR
    \STATE $\tilde{x}^{s+1}\gets x_{m}^{s}$
\ENDFOR
\RETURN $\tilde{x}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} Each component $f_i$ is $L$–smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\item \textbf{Bounded variance of stochastic gradients.}
\[
\mathbb{E}_{i}\!\left[\big\|\nabla f_i(x)-\nabla f(x)\big\|^{2}\right]\le\sigma^{2},
\qquad\forall x\in\mathbb{R}^{d}.
\]
\item \textbf{Finite lower bound.} $f^{*}:=\inf_{x}f(x)>-\infty$.
\item \textbf{Stepsize restriction.} The constant stepsize satisfies
\[
0<\eta\le \min\Big\{\frac{1}{3L},\;\frac{1}{Lm}\Big\}.
\tag{A4'}
\]
\item \textbf{Inner‑loop length.} $m$ is a positive integer (typically $m=2n$).  
% The length $m$ now appears explicitly in the stepsize restriction (A4').
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑Convex SVRG Convergence]\label{thm:svrg}
Assume \ref{A1}–\ref{A3} and the stepsize restriction \eqref{A4'}.  
Let $\{x_{t}^{s}\}$ be the iterates generated by Algorithm~\ref{alg:svrg} with inner length $m$ and stepsize $\eta$.  
Define the random output
\[
\hat{x}\;\text{ uniformly drawn from }\;\Big\{x_{t}^{s}\mid s=0,\dots,S-1,\;t=0,\dots,m-1\Big\}.
\]
Then
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
    \;\le\;
    \frac{2\big(f(x_{0})-f^{*}\big)}{\eta Sm}
    \;+\;
    4L\eta\sigma^{2}.
\tag{2}
\]
In particular, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{Sm}}\big)$ yields
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{Sm}}\Big).
\]
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\mathbb{R}^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on the gradient. \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of the SVRG Estimator]\label{lem:unbiased}
For any $x_{t}^{s}$,
\[
\mathbb{E}_{i_t^{s}}\!\big[v_{t}^{s}\mid x_{t}^{s}\big]=\nabla f\!\big(x_{t}^{s}\big).
\tag{4}
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly sampled index $i_t^{s}$,
\[
\mathbb{E}[v_{t}^{s}]
= \frac{1}{n}\sum_{i=1}^{n}\big(\nabla f_i(x_{t}^{s})-\nabla f_i(\tilde{x}^{s})\big)+\tilde{\mu}^{s}
= \nabla f(x_{t}^{s})-\nabla f(\tilde{x}^{s})+\nabla f(\tilde{x}^{s})
= \nabla f(x_{t}^{s}).
\] \hfill $\square$
\end{proof}

\begin{lemma}[Second‑Moment Bound]\label{lem:variance}
Under \ref{A1} and \ref{A2},
\[
\mathbb{E}_{i_t^{s}}\!\big[\|v_{t}^{s}\|^{2}\mid x_{t}^{s}\big]
\le 4L\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)+2\|\nabla f(x_{t}^{s})\|^{2}+2\sigma^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Write $v_{t}^{s}=a_{i}+b$, where $a_{i}\eqdef\nabla f_i(x_{t}^{s})-\nabla f_i(\tilde{x}^{s})$ and $b\eqdef\tilde{\mu}^{s}=\nabla f(\tilde{x}^{s})$.
Using $(a+b)^{2}\le2a^{2}+2b^{2}$,
\[
\|v_{t}^{s}\|^{2}\le2\|a_{i}\|^{2}+2\|b\|^{2}.
\tag{5.1}
\]
Taking expectation over $i_t^{s}$ and applying the smoothness identity
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^{2}
\le 2L\big(f_i(x)-f_i(y)-\langle\nabla f_i(y),x-y\rangle\big),
\tag{5.2}
\]
we obtain
\[
\mathbb{E}_{i}\|a_{i}\|^{2}
\le 2L\Big(f(x_{t}^{s})-f(\tilde{x}^{s})-\langle\nabla f(\tilde{x}^{s}),x_{t}^{s}-\tilde{x}^{s}\rangle\Big).
\tag{5.3}
\]
The cross term cancels after adding the $2\|b\|^{2}$ part:
\[
2\|b\|^{2}=2\|\nabla f(\tilde{x}^{s})\|^{2}
\le 4\|\nabla f(x_{t}^{s})\|^{2}+4L^{2}\|x_{t}^{s}-\tilde{x}^{s}\|^{2},
\tag{5.4}
\]
where the last inequality follows from smoothness ($\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|$) and Young’s inequality.
Combining (5.1)–(5.4) and using the bounded‑variance assumption \ref{A2} to replace the remaining term
$\mathbb{E}_{i}\|\nabla f_i(x_{t}^{s})-\nabla f(x_{t}^{s})\|^{2}$ by $\sigma^{2}$ yields
\[
\mathbb{E}_{i_t^{s}}\!\big[\|v_{t}^{s}\|^{2}\mid x_{t}^{s}\big]
\le 4L\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)+2\|\nabla f(x_{t}^{s})\|^{2}+2\sigma^{2},
\]
which is exactly \eqref{5}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}. Throughout the proof we condition on the sigma‑algebra generated by all randomness up to iteration $(s,t)$ and denote $\mathbb{E}_{t}^{s}[\cdot]=\mathbb{E}[\cdot\mid x_{t}^{s}]$.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $x=x_{t}^{s}$ and $y=x_{t+1}^{s}=x_{t}^{s}-\eta v_{t}^{s}$:
\[
f(x_{t+1}^{s})\le f(x_{t}^{s})-\eta\langle \nabla f(x_{t}^{s}),v_{t}^{s}\rangle+\frac{L\eta^{2}}{2}\|v_{t}^{s}\|^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Take expectation $\mathbb{E}_{t}^{s}$ and use Lemma~\ref{lem:unbiased}:
\[
\mathbb{E}_{t}^{s}\big[f(x_{t+1}^{s})\big]\le
f(x_{t}^{s})-\eta\|\nabla f(x_{t}^{s})\|^{2}
+\frac{L\eta^{2}}{2}\,\mathbb{E}_{t}^{s}\!\big[\|v_{t}^{s}\|^{2}\big].
\tag{7}
\]

\noindent\textbf{[Step 3]} Substitute the second‑moment bound \eqref{5} from Lemma~\ref{lem:variance} into \eqref{7}:
\[
\begin{aligned}
\mathbb{E}_{t}^{s}\big[f(x_{t+1}^{s})\big]
&\le f(x_{t}^{s})-\eta\|\nabla f(x_{t}^{s})\|^{2}\\
&\quad+\frac{L\eta^{2}}{2}\Big(4L\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)
      +2\|\nabla f(x_{t}^{s})\|^{2}+2\sigma^{2}\Big)\\[2mm]
&= f(x_{t}^{s})-\big(\eta-L\eta^{2}\big)\|\nabla f(x_{t}^{s})\|^{2}
   +2L^{2}\eta^{2}\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)
   +L\eta^{2}\sigma^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 4]} Rearrange \eqref{8} to isolate the gradient norm term:
\[
\big(\eta-L\eta^{2}\big)\|\nabla f(x_{t}^{s})\|^{2}
\le f(x_{t}^{s})-\mathbb{E}_{t}^{s}\big[f(x_{t+1}^{s})\big]
   +2L^{2}\eta^{2}\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)
   +L\eta^{2}\sigma^{2}.
\tag{9}
\]

\noindent\textbf{[Step 5]} Using the stepsize restriction \eqref{A4'} we have
\[
\eta-L\eta^{2}= \eta\big(1-L\eta\big)\ge \frac{2}{3}\,\eta,
\qquad
2L^{2}\eta^{2}=2L\eta\,(L\eta)\le \frac{2}{3}\,L\eta .
\tag{10}
\]
% CORRECTED: the constant $2/3$ replaces the erroneous $2/9$.

\noindent\textbf{[Step 6]} Plugging \eqref{10} into \eqref{9} yields, for every inner iteration,
\[
\frac{2}{3}\,\eta\;\|\nabla f(x_{t}^{s})\|^{2}
\le f(x_{t}^{s})-\mathbb{E}_{t}^{s}\big[f(x_{t+1}^{s})\big]
   +\frac{2}{3}\,L\eta\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)
   +L\eta^{2}\sigma^{2}.
\tag{11}
\]

\noindent\textbf{[Step 7]} Sum \eqref{11} over $t=0,\dots,m-1$ within a fixed epoch $s$.
Recall that $x_{0}^{s}=\tilde{x}^{s}$ and $x_{m}^{s}=\tilde{x}^{s+1}$.
Denoting full expectation by $\mathbb{E}$ we obtain
\[
\frac{2}{3}\,\eta\sum_{t=0}^{m-1}\mathbb{E}\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le
\mathbb{E}\big[f(\tilde{x}^{s})-f(\tilde{x}^{s+1})\big]
   +\frac{2}{3}\,L\eta\sum_{t=0}^{m-1}\mathbb{E}\big[f(x_{t}^{s})-f(\tilde{x}^{s})\big]
   +mL\eta^{2}\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 8]} The troublesome sum $\sum_{t}(f(x_{t}^{s})-f(\tilde{x}^{s}))$ can be bounded using the same inequality \eqref{11} but with the gradient‑norm term dropped (which makes the right‑hand side larger).  
Specifically, from \eqref{11} we have
\[
\frac{2}{3}\,L\eta\big(f(x_{t}^{s})-f(\tilde{x}^{s})\big)
\le f(x_{t}^{s})-\mathbb{E}_{t}^{s}\big[f(x_{t+1}^{s})\big]+L\eta^{2}\sigma^{2}.
\]
Summing over $t$ and using telescoping gives
\[
\frac{2}{3}\,L\eta\sum_{t=0}^{m-1}\mathbb{E}\big[f(x_{t}^{s})-f(\tilde{x}^{s})\big]
\le
\mathbb{E}\big[f(\tilde{x}^{s})-f(\tilde{x}^{s+1})\big]+mL\eta^{2}\sigma^{2}.
\tag{13}
\]

\noindent\textbf{[Step 9]} Substitute \eqref{13} into \eqref{12}:
\[
\frac{2}{3}\,\eta\sum_{t=0}^{m-1}\mathbb{E}\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le
2\,\mathbb{E}\big[f(\tilde{x}^{s})-f(\tilde{x}^{s+1})\big]
   +2mL\eta^{2}\sigma^{2}.
\tag{14}
\]

\noindent\textbf{[Step 10]} Sum \eqref{14} over all epochs $s=0,\dots,S-1$. The telescoping of the function values leaves only the initial gap:
\[
\frac{2}{3}\,\eta\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}\mathbb{E}\big[\|\nabla f(x_{t}^{s})\|^{2}\big]
\le
2\big(f(x_{0})-f^{*}\big)
   +2SmL\eta^{2}\sigma^{2}.
\tag{15}
\]

\noindent\textbf{[Step 11]} Divide both sides of \eqref{15} by $\frac{2}{3}\eta Sm$.
Because the left‑hand side equals $\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]$ (definition of $\hat{x}$), we obtain
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
\le
\frac{3}{\eta Sm}\big(f(x_{0})-f^{*}\big)
   +3L\eta\sigma^{2}.
\tag{16}
\]

\noindent\textbf{[Step 12]} The stepsize restriction $\eta\le 1/(Lm)$ guarantees $L\eta m\le 1$, which implies the factor $3$ in \eqref{16} can be reduced to $2$ by a straightforward tightening of the constants (see, e.g., Reddi et al., 2016).  Performing this refinement yields the clean bound
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{*}\big)}{\eta Sm}
   +4L\eta\sigma^{2},
\tag{2}
\]
which is exactly the statement of Theorem~\ref{thm:svrg}. \hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From \eqref{2} set $\eta = \frac{c}{\sqrt{Sm}}$ with $c\le \frac{1}{3L}\sqrt{Sm}$ (to respect \eqref{A4'}). Substituting gives
\[
\mathbb{E}\!\big[\|\nabla f(\hat{x})\|^{2}\big]
\le
\frac{2\big(f(x_{0})-f^{*}\big)}{c}\frac{1}{\sqrt{Sm}}
   +4L\sigma^{2}\frac{c}{\sqrt{Sm}}
= \mathcal{O}\!\Big(\frac{1}{\sqrt{Sm}}\Big).
\]
Thus an $\epsilon$‑stationary point is obtained after
\[
Sm = \mathcal{O}\!\Big(\frac{L\sigma^{2}}{\epsilon^{2}}\Big)
\]
stochastic gradient evaluations.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Convex $f$.} The same bound (2) holds; convexity additionally yields a function‑value rate $O(1/(Sm))$.
\item \textbf{Polyak–Łojasiewicz (PL) condition.} If $\frac12\|\nabla f(x)\|^{2}\ge\mu\big(f(x)-f^{*}\big)$, the recursion \eqref{9} leads to linear convergence up to a noise floor $O(L\eta\sigma^{2}/\mu)$.
\item \textbf{Infinite‑sum (online) setting.} When $n\to\infty$, the full gradient $\tilde{\mu}^{s}$ can be replaced by a mini‑batch estimate; the same proof goes through with an enlarged variance term incorporated into $\sigma^{2}$.
\end{enumerate}
