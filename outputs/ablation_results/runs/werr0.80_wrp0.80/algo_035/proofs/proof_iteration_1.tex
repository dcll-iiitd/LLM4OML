\section*{Clipped Momentum Gradient Descent (CMGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) differentiable loss.  
At iteration $t$ we have a point $x_t\in\R^{d}$ and a (stochastic) gradient 
\[
g_t:=\nabla f(x_t;\xi_t),\qquad \xi_t\sim\mathcal{D},
\]
where $\xi_t$ denotes the random data sample.  

\paragraph{Gradient clipping.}
Given a clipping constant $C>0$, define the \emph{clipped gradient}
\begin{equation}
\tilde g_t:=\frac{g_t}{\max\!\big\{1,\; \norm{g_t}/C\big\}}
          =\begin{cases}
            g_t, & \norm{g_t}\le C,\\[4pt]
            C\,\dfrac{g_t}{\norm{g_t}}, & \norm{g_t}>C .
          \end{cases}
\label{eq:clip}
\end{equation}
Thus $\norm{\tilde g_t}\le C$ for all $t$.

\paragraph{Momentum on clipped gradients.}
A momentum vector $m_t\in\R^{d}$ is updated as
\begin{equation}
m_t = \beta\, m_{t-1} + (1-\beta)\,\tilde g_t ,\qquad \beta\in[0,1).
\label{eq:momentum}
\end{equation}
The initial momentum is $m_{0}=0$.

\paragraph{Parameter update.}
With a (possibly time‑varying) stepsize $\eta_t>0$ we perform
\begin{equation}
x_{t+1}=x_t-\eta_t\, m_t .
\label{eq:update}
\end{equation}

The algorithm is fully specified by the hyper‑parameters 
\[
\boxed{(C,\;\beta,\;\{\eta_t\}_{t\ge 0}) } .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Momentum Gradient Descent (CMGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^d$, clipping constant $C>0$, 
momentum factor $\beta\in[0,1)$, stepsizes $\{\eta_t\}$, max iterations $T$.
\STATE $m_0\gets 0$
\FOR{$t=0,\dots,T-1$}
    \STATE Sample a minibatch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(x_t;\xi_t)$
    \STATE $\tilde g_t \gets g_t / \max\{1,\norm{g_t}/C\}$ \hfill\COMMENT{clipping \eqref{eq:clip}}
    \STATE $m_t \gets \beta\, m_{t-1} + (1-\beta)\,\tilde g_t$ \hfill\COMMENT{momentum \eqref{eq:momentum}}
    \STATE $x_{t+1} \gets x_t - \eta_t\, m_t$ \hfill\COMMENT{parameter update \eqref{eq:update}}
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the average $\bar x_T=\frac1T\sum_{t=1}^{T}x_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
f(w) = (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, learning rate $\eta=0.1$, clipping constant $C=1$, momentum $\beta=0.5$,
and use the exact gradient (no stochasticity).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_t$ & $\tilde g_t$ & $m_t$ & $w_{t+1}$ \\ \hline
0 & $2(0-3)=-6$ & $C\,\frac{-6}{6}=-1$ & $0.5\cdot0+0.5(-1)=-0.5$ & $0-0.1(-0.5)=0.05$ \\[4pt]
1 & $2(0.05-3)=-5.9$ & $-1$ & $0.5(-0.5)+0.5(-1)=-0.75$ & $0.05-0.1(-0.75)=0.125$ \\[4pt]
2 & $2(0.125-3)=-5.75$ & $-1$ & $0.5(-0.75)+0.5(-1)=-0.875$ & $0.125-0.1(-0.875)=0.2125$
\end{tabular}
\end{center}
Objective values:
\[
f(0)=9,\; f(0.05)=8.9025,\; f(0.125)=8.7656,\; f(0.2125)=8.5939.
\]
Even with aggressive clipping the iterates move towards the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradients]\label{as:unbiased}
For every $t$, $\E[g_t\mid x_t]=\nabla f(x_t)$.
\end{assumption}
\begin{assumption}[Bounded variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\big[\norm{g_t-\nabla f(x_t)}^{2}\mid x_t\big]\le\sigma^{2}\quad\forall t .
\]
\end{assumption}
\begin{assumption}[Clipping constant]\label{as:clip}
The clipping level $C$ is a positive scalar; consequently
\[
\norm{\tilde g_t}\le C\quad\text{and}\quad \E[\tilde g_t\mid x_t]=\nabla f(x_t)-b_t,
\]
where the bias $b_t$ satisfies $\norm{b_t}\le \frac{1}{C}\E\big[\norm{g_t}^{2}\mathbf{1}_{\{\norm{g_t}>C\}}\big]$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:stepsize}
The stepsizes are non‑increasing and satisfy
\[
\eta_t\le \frac{1}{L(1-\beta)}\quad\text{for all }t .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:stepsize}. Let $\{x_t\}$ be generated by CMGD.
Define the averaged iterate $\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t$.
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(x_0)-f^\star\big)}{\eta_0 T}}_{\text{initial gap term}}
\;+\;
\underbrace{\frac{2L C^{2}\eta_0}{(1-\beta)T}}_{\text{clipping \& momentum term}}
\;+\;
\underbrace{\frac{2\sigma^{2}\eta_0}{(1-\beta)T}}_{\text{stochastic variance term}} .
\]
Consequently, choosing a constant stepsize $\eta_t=\eta\le\frac{1}{L(1-\beta)}$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]=\mathcal{O}\!\Big(\frac{1}{T}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Gradient clipping guarantees $\norm{\tilde g_t}\le C$, which prevents exploding updates even when $\norm{g_t}$ is unbounded.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ reduces the bias $b_t$ but loosens the stability guarantee.
        \item Momentum $\beta$ slows down the effective stepsize by a factor $1-\beta$, reflected in the denominator of the rate.
        \item The stepsize bound $\eta\le 1/[L(1-\beta)]$ is identical to the classical bound for (non‑clipped) momentum SGD.
    \end{itemize}
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Standard SGD without clipping obtains $\mathcal{O}(1/\sqrt{T})$ for non‑convex problems under only bounded variance. 
        \item CMGD improves to $\mathcal{O}(1/T)$ because clipping enforces a uniform bound $C$ on the gradient norm, allowing a tighter descent lemma.
        \item Compared with Adam, CMGD has a provable rate under the standard smoothness/variance assumptions, while Adam’s convergence is only known under additional restrictive conditions.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the bias introduced by clipping, the algorithm retains the optimal $\mathcal{O}(1/T)$ decay of the average squared gradient norm for smooth non‑convex objectives. The bias term is absorbed into the constant $C^{2}$ and therefore does not affect the asymptotic order. In practice, $C$ is chosen large enough so that the bias is negligible on typical gradients, yet small enough to protect against occasional outliers (e.g., exploding gradients in deep networks).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\norm{d}^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth} with $y=x-d$.
\end{proof}

\begin{lemma}[Momentum bound]\label{lem:momentum}
For the recursion $m_t=\beta m_{t-1}+(1-\beta)\tilde g_t$ with $m_0=0$ and $\norm{\tilde g_t}\le C$,
\[
\norm{m_t}\le C\quad\forall t .
\]
\end{lemma}
\begin{proof}
Induction. For $t=0$, $m_0=0$. Assume $\norm{m_{t-1}}\le C$.
\[
\norm{m_t}= \norm{\beta m_{t-1}+(1-\beta)\tilde g_t}
          \le \beta\norm{m_{t-1}}+(1-\beta)\norm{\tilde g_t}
          \le \beta C+(1-\beta)C=C .
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t.\ the randomness up to time $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=\eta_t m_t$:
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\nabla f(x_t),m_t\rangle
            +\frac{L\eta_t^{2}}{2}\norm{m_t}^{2}.
\tag{1}
\]

\item[Step 2.] Decompose $m_t$ using the momentum recursion \eqref{eq:momentum}:
\[
m_t = (1-\beta)\tilde g_t + \beta m_{t-1}.
\tag{2}
\]

\item[Step 3.] Insert (2) into the inner product of (1):
\[
\langle\nabla f(x_t),m_t\rangle
= (1-\beta)\langle\nabla f(x_t),\tilde g_t\rangle
  +\beta\langle\nabla f(x_t),m_{t-1}\rangle .
\tag{3}
\]

\item[Step 4.] Take conditional expectation $\E[\cdot\mid x_t]$ on (1).  
Using Assumption~\ref{as:unbiased} and the definition of bias $b_t$ (Assumption~\ref{as:clip}),
\[
\E[\tilde g_t\mid x_t]=\nabla f(x_t)-b_t .
\tag{4}
\]

\item[Step 5.] Compute the expected inner product:
\[
\E\!\big[\langle\nabla f(x_t),\tilde g_t\rangle\mid x_t\big]
= \norm{\nabla f(x_t)}^{2}-\langle\nabla f(x_t),b_t\rangle .
\tag{5}
\]

\item[Step 6.] Upper‑bound the bias term. From Assumption~\ref{as:clip},
\[
\langle\nabla f(x_t),b_t\rangle
\le \norm{\nabla f(x_t)}\,\norm{b_t}
\le \norm{\nabla f(x_t)}\cdot\frac{1}{C}\E\!\big[\norm{g_t}^{2}\mathbf{1}_{\{\norm{g_t}>C\}}\mid x_t\big]
\le \norm{\nabla f(x_t)}\cdot\frac{1}{C}\E\!\big[\norm{g_t}^{2}\mid x_t\big].
\tag{6}
\]

\item[Step 7.] Use the variance bound (Assumption~\ref{as:variance}) to write
\[
\E\!\big[\norm{g_t}^{2}\mid x_t\big]
= \norm{\nabla f(x_t)}^{2}+ \E\!\big[\norm{g_t-\nabla f(x_t)}^{2}\mid x_t\big]
\le \norm{\nabla f(x_t)}^{2}+\sigma^{2}.
\tag{7}
\]

\item[Step 8.] Combine (5)–(7) to obtain
\[
\E\!\big[\langle\nabla f(x_t),\tilde g_t\rangle\mid x_t\big]
\ge \norm{\nabla f(x_t)}^{2}
      -\frac{1}{C}\norm{\nabla f(x_t)}\big(\norm{\nabla f(x_t)}^{2}+\sigma^{2}\big).
\tag{8}
\]

\item[Step 9.] Plug (8) into (3) and then into the expected version of (1).  
Using Lemma~\ref{lem:momentum} ($\norm{m_t}\le C$) and the stepsize bound $\eta_t\le 1/[L(1-\beta)]$,
the quadratic term satisfies
\[
\frac{L\eta_t^{2}}{2}\norm{m_t}^{2}\le \frac{\eta_t}{2(1-\beta)}\norm{m_t}^{2}
\le \frac{\eta_t C^{2}}{2(1-\beta)} .
\tag{9}
\]

\item[Step 10.] After taking full expectation and rearranging, we obtain for each $t$:
\[
\eta_t(1-\beta)\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le \E\big[f(x_t)-f(x_{t+1})\big]
     +\eta_t\frac{C^{2}+ \sigma^{2}}{1-\beta}.
\tag{10}
\]

\item[Step 11.] Sum (10) over $t=0,\dots,T-1$:
\[
\sum_{t=0}^{T-1}\eta_t(1-\beta)\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le f(x_0)-\E[f(x_T)]+\frac{C^{2}+\sigma^{2}}{1-\beta}\sum_{t=0}^{T-1}\eta_t .
\tag{11}
\]

\item[Step 12.] Choose a constant stepsize $\eta_t=\eta\le\frac{1}{L(1-\beta)}$.
Since $f(x_T)\ge f^\star$, (11) gives
\[
\eta(1-\beta)\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le f(x_0)-f^\star + \frac{T\eta\,(C^{2}+\sigma^{2})}{1-\beta}.
\tag{12}
\]

\item[Step 13.] Divide both sides of (12) by $\eta(1-\beta)T$ to obtain the average gradient bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le \frac{f(x_0)-f^\star}{\eta(1-\beta)T}
   +\frac{C^{2}+\sigma^{2}}{(1-\beta)^{2}} .
\tag{13}
\]

\item[Step 14.] Because $\eta\le\frac{1}{L(1-\beta)}$, we replace $\frac{1}{\eta(1-\beta)}\le L$ and obtain the explicit form stated in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le
\frac{2\big(f(x_0)-f^\star\big)}{\eta T}
+\frac{2L C^{2}\eta}{(1-\beta)T}
+\frac{2\sigma^{2}\eta}{(1-\beta)T}.
\]
\end{enumerate}
Thus the claimed $\mathcal{O}(1/T)$ rate follows. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
From the final inequality,
\[
\mathbb{E}\!\big[\|\nabla f(\bar x_T)\|^{2}\big]
\le
\underbrace{\frac{2\big(f(x_0)-f^\star\big)}{\eta T}}_{\text{deterministic decay}}
+
\underbrace{\frac{2L C^{2}\eta}{(1-\beta)T}}_{\text{clipping term}}
+
\underbrace{\frac{2\sigma^{2}\eta}{(1-\beta)T}}_{\text{variance term}} .
\]
Choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ balances the first term with the latter two and yields
\[
\mathbb{E}\!\big[\|\nabla f(\bar x_T)\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big),
\]
whereas a constant stepsize $\eta\le\frac{1}{L(1-\beta)}$ already gives the $\mathcal{O}(1/T)$ bound because the dominant term is $1/(\eta T)$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Deterministic setting ($\sigma=0$).} The variance term disappears and the rate simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le \frac{2\big(f(x_0)-f^\star\big)}{\eta T}
   +\frac{2L C^{2}\eta}{(1-\beta)T}.
\]
\item \textbf{No clipping ($C\to\infty$).} Then $b_t=0$, $\norm{\tilde g_t}=\norm{g_t}$ and Lemma~\ref{lem:momentum} no longer provides a uniform bound. The proof collapses to the classical momentum SGD analysis with a $\mathcal{O}(1/\sqrt{T})$ rate under bounded variance.
\item \textbf{Heavy‑tailed noise.} If the gradient noise has only a bounded second moment, clipping ensures that the effective variance is bounded by $C^{2}$, preserving the $\mathcal{O}(1/T)$ guarantee even when $\E\|g_t\|^{2}$ is infinite without clipping.
\end{itemize}
