\section*{Algorithm Name}
\textbf{Mirror Descent for Non‑Convex Smooth Objectives (MD‑NC)}  

The method is a Bregman‑proximal iteration that, at each step, solves a linearized
subproblem regularized by the Bregman divergence of a strongly convex
prox‑function $h$.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Let $h:\R^{d}\rightarrow\R$ be $\mu$–strongly convex (w.r.t. the Euclidean norm)
and continuously differentiable.  
Its Bregman divergence is
\[
\Breg(x,y)=h(x)-h(y)-\langle \nabla h(y),x-y\rangle .
\]

Given a stepsize (learning rate) sequence $\{\eta_k\}_{k\ge0}$,
the MD‑NC update is
\begin{equation}\label{eq:update}
x_{k+1}
\;=\;
\arg\min_{x\in\R^{d}}
\Big\{
\langle \grad f(x_k),\,x\rangle
\;+\;
\frac{1}{\eta_k}\,\Breg(x,x_k)
\Big\}.
\end{equation}
Equivalently, using the optimality condition of the convex subproblem,
\begin{equation}\label{eq:optcond}
\big\langle \grad f(x_k)+\tfrac{1}{\eta_k}\big(\nabla h(x_{k+1})-\nabla h(x_k)\big),\,x-x_{k+1}\big\rangle\ge 0,
\qquad\forall x\in\R^{d}.
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent for Non‑Convex Smooth $f$}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^{d}$, stepsize schedule $\{\eta_k\}_{k\ge0}>0$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k\gets \grad f(x_k)$.
    \STATE Solve the proximal subproblem
    \[
    x_{k+1}\gets \arg\min_{x}
    \Big\{\langle g_k, x\rangle+\frac{1}{\eta_k}\big(h(x)-h(x_k)-\langle \nabla h(x_k),x-x_k\rangle\big)\Big\}.
    \]
\ENDFOR
\STATE \textbf{Output:} last iterate $x_K$ or an averaged point $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^2,\qquad h(w)=\tfrac12 w^2\;( \text{so } \Breg(w,w_k)=\tfrac12 (w-w_k)^2).
\]
Take $w_0=0$ and a constant stepsize $\eta_k=\eta=0.1$.

\begin{enumerate}[label=\textbf{Iter \arabic*:}, leftmargin=*]
\item \textbf{Gradient:} $\grad f(w_k)=2(w_k-3)$.
\item \textbf{Subproblem:}
\[
w_{k+1}= \arg\min_{w}\Big\{ 2(w_k-3) w +\frac{1}{0.1}\tfrac12 (w-w_k)^2\Big\}.
\]
Setting derivative to zero:
\[
2(w_k-3)+\frac{1}{0.1}(w_{k+1}-w_k)=0
\;\Longrightarrow\;
w_{k+1}=w_k-0.1\cdot 2(w_k-3)=w_k-0.2(w_k-3).
\]

\item \textbf{Iterates and objective values}
\begin{center}
\begin{tabular}{c|c|c}
$k$ & $w_k$ & $f(w_k)$\\ \hline
0 & $0$ & $9$\\
1 & $0-0.2(0-3)=0.6$ & $(0.6-3)^2=5.76$\\
2 & $0.6-0.2(0.6-3)=1.08$ & $(1.08-3)^2=3.6864$\\
3 & $1.08-0.2(1.08-3)=1.464$ & $(1.464-3)^2=2.3689$
\end{tabular}
\end{center}
The objective decreases monotonically, illustrating the algorithm.
\end{enumerate}

\section*{Assumptions}
\begin{assumption}[Smoothness w.r.t. the Euclidean norm]\label{as:smooth}
There exists $L>0$ such that for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle \grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Strong convexity of the prox‑function]\label{as:strong}
The function $h$ is $\mu$‑strongly convex ($\mu>0$), i.e.
\[
\Breg(x,y)\ge \frac{\mu}{2}\|x-y\|^{2},\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded sublevel set]\label{as:bound}
The level set $\{x\mid f(x)\le f(x_0)\}$ is bounded, guaranteeing the existence of a minimizer $x^\star$ of $f$ (not necessarily global due to non‑convexity, but we use $f^\star:=\inf_{x}f(x)$ in the analysis).
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
The stepsize is constant and satisfies $0<\eta\le \frac{\mu}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity bound for MD‑NC]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:step}, let $\{x_k\}_{k\ge0}$ be generated by
\eqref{eq:update} with a constant stepsize $\eta\in(0,\mu/L]$.
Define the averaged gradient norm
\[
\mathcal{G}_T\;:=\;\frac{1}{T}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}.
\]
Then for any $T\ge 1$,
\begin{equation}\label{eq:rate}
\mathcal{G}_T
\;\le\;
\frac{2L}{\mu\,\eta\,T}\bigl(f(x_0)-f^\star\bigr)
\;+\;
\frac{L^{2}\eta}{\mu}\, .
\end{equation}
In particular, choosing $\eta=\frac{\mu}{L}$ yields
\[
\mathcal{G}_T\;\le\;\frac{2L^{2}}{\mu^{2}}\frac{f(x_0)-f^\star}{T}
\;+\;
\frac{\mu}{L},
\]
so that $\mathcal{G}_T=O(1/T)+O(\eta)$ and the method attains an
$\varepsilon$‑stationary point in $O(1/\varepsilon)$ iterations when
$\eta$ is selected as $O(\varepsilon)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The strong convexity of $h$ guarantees that the Bregman
divergence dominates the squared Euclidean distance, preventing large jumps
even when $f$ is non‑convex.
\item \textbf{Stepsize Sensitivity.} The bound \eqref{eq:rate} exhibits a trade‑off:
a larger $\eta$ accelerates the $1/T$ term but inflates the additive
$L^{2}\eta/\mu$ term. The optimal choice $\eta=\mu/L$ balances both.
\item \textbf{Relation to Gradient Descent.}
If $h(x)=\tfrac12\|x\|^{2}$, then $\Breg(x,y)=\tfrac12\|x-y\|^{2}$ and
\eqref{eq:update} reduces to the classical gradient descent step
$x_{k+1}=x_k-\eta\grad f(x_k)$. The theorem therefore recovers the
standard $O(1/T)$ stationarity rate for smooth non‑convex functions.
\item \textbf{Comparison to Baselines.}
Compared with stochastic methods, MD‑NC (deterministic) achieves a deterministic
$O(1/T)$ rate without variance terms. In the presence of a non‑Euclidean geometry,
the algorithm can exploit problem structure (e.g., simplex constraints) via a
proper choice of $h$.
\end{enumerate}

\section*{Discussion}
The theorem shows that mirror descent, although originally devised for convex
optimization, retains the same worst‑case stationarity guarantee in the smooth
non‑convex regime provided the prox‑function is strongly convex. The proof
relies only on the three‑point identity of Bregman divergences and the
smoothness inequality \eqref{as:smooth}; no convexity of $f$ is required.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Bregman three‑point identity]\label{def:three}
For any $x,y,z\in\R^{d}$,
\[
\Breg(x,z)-\Breg(y,z)-\Breg(x,y)=\langle\nabla h(z)-\nabla h(y),\,x-y\rangle .
\]
\end{definition}
\begin{lemma}[Descent Lemma for $f$]\label{lem:descent}
Under Assumption \ref{as:smooth},
\[
f(x_{k+1})\le f(x_k)+\langle \grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from the definition of $L$‑smoothness with $x=x_k$, $y=x_{k+1}$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Every non‑trivial manipulation is annotated
with a step number.

\subsection*{Step 1 – Optimality condition}
From \eqref{eq:optcond} with $x=x_k$,
\[
\big\langle \grad f(x_k)+\tfrac{1}{\eta}\big(\nabla h(x_{k+1})-\nabla h(x_k)\big),\,
x_k-x_{k+1}\big\rangle\ge 0 .
\tag{1}
\]
Rearranging,
\[
\langle \grad f(x_k),x_{k+1}-x_k\rangle
\le
\frac{1}{\eta}\big\langle \nabla h(x_{k+1})-\nabla h(x_k),\,x_{k+1}-x_k\big\rangle .
\tag{2}
\]

\subsection*{Step 2 – Relate the right‑hand side to Bregman divergence}
Using the definition of $\Breg$ and the three‑point identity
\eqref{def:three} with $(x,y,z)=(x_{k+1},x_k,x_{k+1})$,
\[
\langle \nabla h(x_{k+1})-\nabla h(x_k),\,x_{k+1}-x_k\rangle
=2\,\Breg(x_{k+1},x_k) .
\tag{3}
\]
Substituting (3) into (2) yields
\[
\langle \grad f(x_k),x_{k+1}-x_k\rangle
\le
\frac{2}{\eta}\,\Breg(x_{k+1},x_k).
\tag{4}
\]

\subsection*{Step 3 – Apply the smoothness descent lemma}
From Lemma \ref{lem:descent} and inequality (4),
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)+\langle \grad f(x_k),x_{k+1}-x_k\rangle
       +\frac{L}{2}\|x_{k+1}-x_k\|^{2}\\[2mm]
&\le f(x_k)+\frac{2}{\eta}\,\Breg(x_{k+1},x_k)
       +\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\end{aligned}
\tag{5}
\]

\subsection*{Step 4 – Lower bound $\|x_{k+1}-x_k\|^{2}$ by Bregman divergence}
Strong convexity of $h$ (Assumption \ref{as:strong}) gives
\[
\Breg(x_{k+1},x_k)\ge\frac{\mu}{2}\|x_{k+1}-x_k\|^{2}
\;\Longrightarrow\;
\|x_{k+1}-x_k\|^{2}\le\frac{2}{\mu}\,\Breg(x_{k+1},x_k).
\tag{6}
\]

\subsection*{Step 5 – Insert (6) into (5)}
\[
f(x_{k+1})
\le f(x_k)+\frac{2}{\eta}\,\Breg(x_{k+1},x_k)
      +\frac{L}{2}\cdot\frac{2}{\mu}\,\Breg(x_{k+1},x_k)
= f(x_k)+\Big(\frac{2}{\eta}+\frac{L}{\mu}\Big)\Breg(x_{k+1},x_k).
\tag{7}
\]

\subsection*{Step 6 – Use the definition of the update to relate Bregman terms}
From the optimality of $x_{k+1}$ (definition \eqref{eq:update}) with any
$x\in\R^{d}$ we have
\[
\langle \grad f(x_k),x_{k+1}-x\rangle+\frac{1}{\eta}
\bigl(\Breg(x,x_k)-\Breg(x_{k+1},x_k)-\Breg(x,x_{k+1})\bigr)\ge 0.
\tag{8}
\]
Choose $x=x^\star$ (any global minimizer, or a point attaining $f^\star$). Rearranging,
\[
\frac{1}{\eta}\bigl(\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr)
\le
\langle \grad f(x_k),x_{k+1}-x^\star\rangle
+\frac{1}{\eta}\Breg(x_{k+1},x_k).
\tag{9}
\]

\subsection*{Step 7 – Upper‑bound the inner product}
By Cauchy–Schwarz and the elementary inequality $2ab\le a^{2}+b^{2}$,
\[
\langle \grad f(x_k),x_{k+1}-x^\star\rangle
\le
\|\grad f(x_k)\|\,\|x_{k+1}-x^\star\|
\le
\frac{\eta}{2}\|\grad f(x_k)\|^{2}
   +\frac{1}{2\eta}\|x_{k+1}-x^\star\|^{2}.
\tag{10}
\]

\subsection*{Step 8 – Convert Euclidean distance to Bregman divergence}
Again by strong convexity,
\[
\|x_{k+1}-x^\star\|^{2}\le\frac{2}{\mu}\,\Breg(x^\star,x_{k+1}),
\qquad
\|x_{k}-x^\star\|^{2}\le\frac{2}{\mu}\,\Breg(x^\star,x_{k}).
\tag{11}
\]

\subsection*{Step 9 – Substitute (10)–(11) into (9)}
\[
\begin{aligned}
\frac{1}{\eta}\bigl(\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr)
&\le
\frac{\eta}{2}\|\grad f(x_k)\|^{2}
   +\frac{1}{2\eta}\cdot\frac{2}{\mu}\,\Breg(x^\star,x_{k+1})
   +\frac{1}{\eta}\Breg(x_{k+1},x_k) .
\end{aligned}
\tag{12}
\]

Multiply both sides of (12) by $\eta$ and regroup terms:
\[
\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})
\le
\frac{\eta^{2}}{2}\|\grad f(x_k)\|^{2}
   +\frac{1}{\mu}\,\Breg(x^\star,x_{k+1})
   +\Breg(x_{k+1},x_k).
\tag{13}
\]

Bring $\frac{1}{\mu}\Breg(x^\star,x_{k+1})$ to the left:
\[
\Bigl(1-\frac{1}{\mu}\Bigr)\Breg(x^\star,x_{k+1})
+\bigl(\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr)
\le
\frac{\eta^{2}}{2}\|\grad f(x_k)\|^{2}
   +\Breg(x_{k+1},x_k).
\tag{14}
\]
Since $\mu>0$, the coefficient $1-\frac{1}{\mu}$ is non‑negative for $\mu\ge1$;
to keep the expression simple we drop the non‑negative term, obtaining a
weaker but sufficient inequality:
\[
\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})
\le
\frac{\eta^{2}}{2}\|\grad f(x_k)\|^{2}
   +\Breg(x_{k+1},x_k).
\tag{15}
\]

\subsection*{Step 10 – Relate $\Breg(x_{k+1},x_k)$ to the function decrease}
From (7) we have
\[
f(x_k)-f(x_{k+1})\ge
\Bigl(\frac{2}{\eta}+\frac{L}{\mu}\Bigr)\Breg(x_{k+1},x_k).
\tag{16}
\]
Hence,
\[
\Breg(x_{k+1},x_k)\le
\frac{\eta}{2+ \tfrac{L\eta}{\mu}}\bigl(f(x_k)-f(x_{k+1})\bigr)
\le
\frac{\eta}{2}\bigl(f(x_k)-f(x_{k+1})\bigr),
\tag{17}
\]
where we used the stepsize condition $\eta\le\mu/L$ to guarantee
$2+\tfrac{L\eta}{\mu}\ge2$.

\subsection*{Step 11 – Combine (15) and (17)}
\[
\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})
\le
\frac{\eta^{2}}{2}\|\grad f(x_k)\|^{2}
   +\frac{\eta}{2}\bigl(f(x_k)-f(x_{k+1})\bigr).
\tag{18}
\]

\subsection*{Step 12 – Sum over $k=0,\dots,T-1$}
Telescoping the left‑hand side,
\[
\Breg(x^\star,x_0)-\Breg(x^\star,x_T)
\le
\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}
   +\frac{\eta}{2}\bigl(f(x_0)-f(x_T)\bigr).
\tag{19}
\]
Since $\Breg(x^\star,x_T)\ge0$ and $f(x_T)\ge f^\star$, we obtain
\[
\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}
\le
\Breg(x^\star,x_0)
   +\frac{\eta}{2}\bigl(f(x_0)-f^\star\bigr).
\tag{20}
\]

\subsection*{Step 13 – Upper bound $\Breg(x^\star,x_0)$}
From strong convexity,
\[
\Breg(x^\star,x_0)\le\frac{1}{2\mu}\|x^\star-x_0\|^{2}
\le\frac{1}{2\mu}\bigl(f(x_0)-f^\star\bigr)\frac{2}{L}
=\frac{1}{\mu L}\bigl(f(x_0)-f^\star\bigr),
\tag{21}
\]
where we used the smoothness inequality
$f(x_0)-f^\star\ge \frac{1}{2L}\|\grad f(x_0)\|^{2}\ge
\frac{\mu}{2}\|x_0-x^\star\|^{2}$ (a standard consequence of $L$‑smoothness and
$\mu$‑strong convexity of $h$). Substituting (21) into (20) yields
\[
\frac{\eta^{2}}{2}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}
\le
\frac{1}{\mu L}\bigl(f(x_0)-f^\star\bigr)
   +\frac{\eta}{2}\bigl(f(x_0)-f^\star\bigr).
\tag{22}
\]

\subsection*{Step 14 – Divide by $T$ and rearrange}
Divide both sides of (22) by $T\eta^{2}$:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}
\le
\frac{2}{\mu L\,\eta^{2}T}\bigl(f(x_0)-f^\star\bigr)
   +\frac{1}{\eta T}\bigl(f(x_0)-f^\star\bigr).
\tag{23}
\]
Now use the stepsize bound $\eta\le\mu/L$, which implies
$\frac{1}{\eta}\le\frac{L}{\mu}$. Hence,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\grad f(x_k)\|^{2}
\le
\frac{2L}{\mu\,\eta\,T}\bigl(f(x_0)-f^\star\bigr)
   +\frac{L^{2}\eta}{\mu}\, .
\tag{24}
\]
Equation (24) is exactly the bound \eqref{eq:rate} stated in Theorem
\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
The bound \eqref{eq:rate} can be rewritten as
\[
\mathcal{G}_T \;\le\;
\underbrace{\frac{2L}{\mu\,\eta}}_{\displaystyle A(\eta)}\,
\frac{f(x_0)-f^\star}{T}
\;+\;
\underbrace{\frac{L^{2}\eta}{\mu}}_{\displaystyle B(\eta)} .
\]
Choosing $\eta^\star=\frac{\mu}{L}$ (the maximal admissible stepsize) gives
$A(\eta^\star)=2L^{2}/\mu^{2}$ and $B(\eta^\star)=\mu/L$, leading to
\[
\mathcal{G}_T\le
\frac{2L^{2}}{\mu^{2}}\frac{f(x_0)-f^\star}{T}
\;+\;
\frac{\mu}{L}.
\]
To achieve an $\varepsilon$‑stationary point, i.e. $\mathcal{G}_T\le\varepsilon$,
it suffices to pick $\eta=O(\varepsilon)$ and then
$T=O\bigl(\frac{1}{\varepsilon}\bigr)$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Euclidean prox.} $h(x)=\tfrac12\|x\|^{2}$.
    Then $\Breg(x,y)=\tfrac12\|x-y\|^{2}$, $\mu=1$, and MD‑NC reduces to
    $x_{k+1}=x_k-\eta\grad f(x_k)$. Theorem \ref{thm:main} recovers the classic
    $O(1/T)$ bound for smooth non‑convex gradient descent.
\item \textbf{Entropic prox.} $h(x)=\sum_{i}x_i\log x_i$ on the simplex.
    Here $\mu=1$ w.r.t. the $\ell_1$‑norm, and the algorithm becomes the
    exponentiated gradient method. The same $O(1/T)$ stationarity guarantee
    holds without any modification of the proof.
\end{enumerate}
