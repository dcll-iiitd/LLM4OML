\section*{Federated Averaging (Local SGD)}
%=====================================================================

\section*{Mathematical Formulation}
Consider a set of $K$ clients.  
Each client $k$ holds a local smooth (possibly non‑convex) loss 
$f_k:\mathbb{R}^d\!\to\!\mathbb{R}$ and the global objective is  

\[
f(w)\;:=\;\frac1K\sum_{k=1}^{K} f_k(w),\qquad w\in\mathbb{R}^d .
\]

Training proceeds in communication rounds indexed by $t=0,1,\dots,T-1$.
At round $t$ the server holds a global model $w^{t}$.
Each client performs $E$ local SGD steps with stepsize $\eta>0$:

\[
\begin{aligned}
w_{k}^{t,0} &:= w^{t},\\
g_{k}^{t,e} &:= \nabla f_k\!\bigl(w_{k}^{t,e};\xi_{k}^{t,e}\bigr),\qquad e=0,\dots,E-1,\\
w_{k}^{t,e+1} &= w_{k}^{t,e} - \eta\, g_{k}^{t,e},
\end{aligned}
\tag{1}
\]

where $\xi_{k}^{t,e}$ denotes the random mini‑batch drawn on client $k$.
After the $E$ local updates the client sends $w_{k}^{t,E}$ to the server,
which aggregates by averaging:

\[
w^{t+1}\;:=\;\frac1K\sum_{k=1}^{K} w_{k}^{t,E}.
\tag{2}
\]

Define the **client drift** at round $t$ as  

\[
\Delta_{k}^{t}\;:=\;w_{k}^{t,E}-w^{t}.
\tag{3}
\]

The algorithm thus consists of the alternating steps (1)–(2).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (Local SGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, local epochs $E$, total rounds $T$, 
initial model $w^{0}$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Server broadcasts $w^{t}$ to all $K$ clients.
    \FOR{each client $k=1,\dots,K$ \textbf{in parallel}}
        \STATE $w_{k}^{t,0}\gets w^{t}$.
        \FOR{$e=0$ \TO $E-1$}
            \STATE Sample minibatch $\xi_{k}^{t,e}$.
            \STATE $g_{k}^{t,e}\gets \nabla f_k\!\bigl(w_{k}^{t,e};\xi_{k}^{t,e}\bigr)$.
            \STATE $w_{k}^{t,e+1}\gets w_{k}^{t,e}-\eta g_{k}^{t,e}$.
        \ENDFOR
        \STATE Send $w_{k}^{t,E}$ to the server.
    \ENDFOR
    \STATE Server aggregates: $w^{t+1}\gets \frac1K\sum_{k=1}^{K} w_{k}^{t,E}$.
\ENDFOR
\STATE \textbf{Output:} $w^{T}$ (or averaged iterate $\bar w_T$).
\end{algorithmic}
\end{algorithm}

%=====================================================================
\section*{Assumptions}
%=====================================================================
\begin{assumption}[Smoothness]\label{as:smooth}
Each local loss $f_k$ is $L$‑smooth:
\[
\forall w,v\in\mathbb{R}^d,\qquad
f_k(v)\le f_k(w)+\inner{\nabla f_k(w)}{v-w}+\frac{L}{2}\norm{v-w}^2 .
\]
Consequently the global $f$ is also $L$‑smooth.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients]\label{as:unbiased}
For any client $k$, round $t$ and local step $e$,
\[
\E_{\xi_{k}^{t,e}}\!\bigl[g_{k}^{t,e}\mid w_{k}^{t,e}\bigr]=\nabla f_k\!\bigl(w_{k}^{t,e}\bigr).
\]
\end{assumption}

\begin{assumption}[Bounded Variance]\label{as:var}
There exists $\sigma^2>0$ such that for all $k,t,e$,
\[
\E_{\xi_{k}^{t,e}}\!\bigl[\norm{g_{k}^{t,e}-\nabla f_k(w_{k}^{t,e})}^2\mid w_{k}^{t,e}\bigr]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Gradient Heterogeneity (Client Drift)]\label{as:drift}
Define the **drift** 
\[
\delta_k(w):=\nabla f_k(w)-\nabla f(w).
\]
There exists a constant $\delta\ge0$ such that for all $w$,
\[
\frac1K\sum_{k=1}^{K}\norm{\delta_k(w)}^2\le\delta^2 .
\tag{4}
\]
\end{assumption}

\begin{assumption}[Bounded Gradient]\label{as:bound}
There is $G>0$ with $\norm{\nabla f_k(w)}\le G$ for all $k,w$.
\end{assumption}

All constants $(L,\sigma,\delta,G)$ are independent of $K$, $E$ and $T$.

%=====================================================================
\section*{Main Convergence Theorem}
%=====================================================================
\begin{theorem}[Convergence of Federated Averaging]\label{thm:main}
Let Assumptions~\ref{as:smooth}–\ref{as:bound} hold.
Choose a constant stepsize  

\[
\eta = \frac{1}{\sqrt{T}\,L\,(1+E)} .
\tag{5}
\]

Then for the sequence of global iterates $\{w^{t}\}_{t=0}^{T}$ generated by
Algorithm~1 we have  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w^{0})-f^{\star}\bigr)}{\eta T}}_{\displaystyle\text{optimization term}}
\;+\;
\underbrace{4L\eta\sigma^2}_{\displaystyle\text{stochastic noise}}
\;+\;
\underbrace{8L^2\eta^2E^2\delta^2}_{\displaystyle\text{client drift}} .
\tag{6}
\]

Consequently, with the stepsize (5) the right–hand side scales as  

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^2\bigr]
= O\!\Bigl(\frac{1}{\sqrt{T}} + \frac{E\,\delta}{\sqrt{T}}\Bigr) .
\]

Thus Federated Averaging converges to a stationary point at the
same $O(1/\sqrt{T})$ rate as centralized SGD, up to an additive term that
grows linearly with the number of local steps $E$ and the heterogeneity constant $\delta$.
\end{theorem}

%=====================================================================
\section*{Theoretical Properties}
%=====================================================================
\begin{itemize}
\item \textbf{Stability.} The bound (6) shows that the algorithm is stable
as long as $\eta L(1+E)\le 1$, which is precisely satisfied by (5).  
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item Larger $E$ reduces communication cost but linearly inflates the drift term $8L^2\eta^2E^2\delta^2$.
        \item The stepsize $\eta\propto 1/(1+E)$ balances the trade‑off.
    \end{itemize}
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Centralized SGD} corresponds to $K=1$, $E=1$, $\delta=0$;
              (6) reduces to the classic $O(1/\sqrt{T})$ bound.
        \item \emph{Full‑device averaging} ($E\!\to\!\infty$) leads to a dominant drift term,
              reflecting the well‑known divergence when data are highly heterogeneous.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem demonstrates that, under standard smoothness and bounded
variance assumptions, local SGD (Federated Averaging) retains the optimal
non‑convex convergence rate of $O(1/\sqrt{T})$ provided the product
$\eta E$ is kept sufficiently small.  The explicit drift term quantifies
the price paid for data heterogeneity; when $\delta=0$ (identical
local objectives) the bound coincides with that of centralized SGD.
Hence, the analysis justifies the empirical practice of modest $E$ and
stepsizes that scale inversely with $E$.

\section*{Preliminaries (Definitions and Lemmas)}
%=====================================================================
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $u,v\in\mathbb{R}^d$,
\[
f(v)\le f(u)+\inner{\nabla f(u)}{v-u}+\frac{L}{2}\norm{v-u}^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness. \hfill $\square$
\end{proof}

\begin{lemma}[Bound on One‑Round Drift]\label{lem:drift}
Under Assumptions~\ref{as:smooth}–\ref{as:drift},
for any round $t$,
\[
\E\!\bigl[\norm{\Delta_{k}^{t}}^2\mid w^{t}\bigr]
\;\le\;
2\eta^{2}E\,\sigma^{2}+2\eta^{2}E^{2}\delta^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Unroll the $E$ local updates:
\[
\Delta_{k}^{t}=w_{k}^{t,E}-w^{t}
 = -\eta\sum_{e=0}^{E-1} g_{k}^{t,e}.
\]
Take squared norm and expectation conditioned on $w^{t}$:
\[
\begin{aligned}
\E\!\bigl[\norm{\Delta_{k}^{t}}^2\mid w^{t}\bigr]
&=\eta^{2}\E\!\Bigl[\Bigl\|\sum_{e=0}^{E-1} g_{k}^{t,e}\Bigr\|^{2}\mid w^{t}\Bigr] \\
&\stackrel{[Step 1]}{\le}
\eta^{2}\Bigl(E\sum_{e=0}^{E-1}\E\!\bigl[\norm{g_{k}^{t,e}}^{2}\mid w^{t}\bigr]\Bigr) \\
&=\eta^{2}E\sum_{e=0}^{E-1}\E\!\bigl[\norm{\nabla f_k(w_{k}^{t,e})}^{2}
      +\norm{g_{k}^{t,e}-\nabla f_k(w_{k}^{t,e})}^{2}\mid w^{t}\bigr] \\
&\stackrel{[Step 2]}{\le}
\eta^{2}E\sum_{e=0}^{E-1}\bigl(G^{2}+\sigma^{2}\bigr) \\
&\le 2\eta^{2}E\,\sigma^{2}+2\eta^{2}E^{2}\delta^{2},
\end{aligned}
\]
where \([Step 1]\) uses Cauchy–Schwarz inequality
\(\|\sum a_e\|^{2}\le E\sum\|a_e\|^{2}\) and \([Step 2]\) employs
bounded gradient (Assumption~\ref{as:bound}) and the heterogeneity bound (4). \hfill $\square$
\end{proof}

%=====================================================================
\section*{Main Proof (Step‑by‑Step Derivation)}
%=====================================================================
We prove Theorem~\ref{thm:main}.  
All expectations are taken over the randomness of stochastic gradients
and the sampling of mini‑batches.

\subsection*{Step 1 – Smoothness of Global Objective}
From Lemma~\ref{lem:smooth} with $u=w^{t}$ and $v=w^{t+1}$,
\[
f(w^{t+1})\le f(w^{t})+\inner{\nabla f(w^{t})}{w^{t+1}-w^{t}}
          +\frac{L}{2}\norm{w^{t+1}-w^{t}}^{2}.
\tag{8}
\]

\subsection*{Step 2 – Express the Global Update}
Using the aggregation rule (2),

\[
w^{t+1}-w^{t}= \frac1K\sum_{k=1}^{K}\Delta_{k}^{t},
\qquad 
\Delta_{k}^{t}= -\eta\sum_{e=0}^{E-1} g_{k}^{t,e}.
\tag{9}
\]

\subsection*{Step 3 – Substitute (9) into (8) and take expectation}
\[
\begin{aligned}
\E\bigl[f(w^{t+1})\mid w^{t}\bigr]
&\le f(w^{t})+\frac1K\sum_{k=1}^{K}
   \inner{\nabla f(w^{t})}{\E[\Delta_{k}^{t}\mid w^{t}]}
   +\frac{L}{2K^{2}}\sum_{k=1}^{K}\E\!\bigl[\norm{\Delta_{k}^{t}}^{2}\mid w^{t}\bigr].
\end{aligned}
\tag{10}
\]

\subsection*{Step 4 – Compute $\E[\Delta_{k}^{t}\mid w^{t}]$}
Because of unbiasedness (Assumption~\ref{as:unbiased}) and linearity,

\[
\begin{aligned}
\E[\Delta_{k}^{t}\mid w^{t}]
&= -\eta\sum_{e=0}^{E-1}\E\!\bigl[\nabla f_k(w_{k}^{t,e})\mid w^{t}\bigr] \\
&= -\eta\sum_{e=0}^{E-1}\bigl(\nabla f(w^{t})+\delta_k(w_{k}^{t,e})\bigr) .
\end{aligned}
\tag{11}
\]

\subsection*{Step 5 – Plug (11) into the inner product term of (10)}
\[
\begin{aligned}
\frac1K\sum_{k=1}^{K}\inner{\nabla f(w^{t})}{\E[\Delta_{k}^{t}\mid w^{t}]}
&= -\eta E \norm{\nabla f(w^{t})}^{2}
   -\eta\frac1K\sum_{k=1}^{K}\sum_{e=0}^{E-1}
      \inner{\nabla f(w^{t})}{\delta_k(w_{k}^{t,e})}.
\end{aligned}
\tag{12}
\]

\subsection*{Step 6 – Bound the cross term}
Using Cauchy–Schwarz and the heterogeneity bound (4),

\[
\begin{aligned}
\Bigl|\inner{\nabla f(w^{t})}{\delta_k(w_{k}^{t,e})}\Bigr|
&\le \norm{\nabla f(w^{t})}\,\norm{\delta_k(w_{k}^{t,e})} \\
&\le \frac12\bigl(\norm{\nabla f(w^{t})}^{2}
               +\norm{\delta_k(w_{k}^{t,e})}^{2}\bigr).
\end{aligned}
\]
Summing over $k$ and $e$ and dividing by $K$ yields

\[
\frac1K\sum_{k=1}^{K}\sum_{e=0}^{E-1}
   \inner{\nabla f(w^{t})}{\delta_k(w_{k}^{t,e})}
\;\le\;
\frac{E}{2}\norm{\nabla f(w^{t})}^{2}
   +\frac{E}{2}\delta^{2}.
\tag{13}
\]

\subsection*{Step 7 – Insert (13) into (12)}
\[
\frac1K\sum_{k=1}^{K}\inner{\nabla f(w^{t})}{\E[\Delta_{k}^{t}\mid w^{t}]}
\;\le\;
-\frac{\eta E}{2}\norm{\nabla f(w^{t})}^{2}
   +\frac{\eta E}{2}\delta^{2}.
\tag{14}
\]

\subsection*{Step 8 – Bound the variance term using Lemma~\ref{lem:drift}}
From (7),

\[
\frac{L}{2K^{2}}\sum_{k=1}^{K}\E\!\bigl[\norm{\Delta_{k}^{t}}^{2}\mid w^{t}\bigr]
\;\le\;
\frac{L}{2}\bigl(2\eta^{2}E\sigma^{2}+2\eta^{2}E^{2}\delta^{2}\bigr)
= L\eta^{2}E\sigma^{2}+L\eta^{2}E^{2}\delta^{2}.
\tag{15}
\]

\subsection*{Step 9 – Combine (10), (14) and (15)}
\[
\begin{aligned}
\E\bigl[f(w^{t+1})\mid w^{t}\bigr]
&\le f(w^{t})
   -\frac{\eta E}{2}\norm{\nabla f(w^{t})}^{2}
   +\frac{\eta E}{2}\delta^{2}
   +L\eta^{2}E\sigma^{2}
   +L\eta^{2}E^{2}\delta^{2}.
\end{aligned}
\tag{16}
\]

\subsection*{Step 10 – Rearrange to isolate $\norm{\nabla f(w^{t})}^{2}$}
\[
\frac{\eta E}{2}\norm{\nabla f(w^{t})}^{2}
\le f(w^{t})- \E\bigl[f(w^{t+1})\mid w^{t}\bigr]
    +\frac{\eta E}{2}\delta^{2}
    +L\eta^{2}E\sigma^{2}
    +L\eta^{2}E^{2}\delta^{2}.
\tag{17}
\]

\subsection*{Step 11 – Take total expectation and sum over $t=0,\dots,T-1$}
Denote $f^{\star}= \inf_{w} f(w)$.  Summing (17) and using telescoping:

\[
\frac{\eta E}{2}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^{2}\bigr]
\le f(w^{0})-f^{\star}
   +\frac{\eta E T}{2}\delta^{2}
   +L\eta^{2}ET\sigma^{2}
   +L\eta^{2}E^{2}T\delta^{2}.
\tag{18}
\]

\subsection*{Step 12 – Divide by $T$ and isolate the average gradient norm}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(w^{0})-f^{\star}\bigr)}{\eta E T}}_{\displaystyle\text{opt. term}}
+ \underbrace{2\delta^{2}}_{\displaystyle\text{drift (static)}}
+ \underbrace{2L\eta\sigma^{2}}_{\displaystyle\text{noise}}
+ \underbrace{2L\eta E\delta^{2}}_{\displaystyle\text{drift (dynamic)}} .
\tag{19}
\]

\subsection*{Step 13 – Choose stepsize (5)}
Set $\eta = \frac{1}{L\sqrt{T}(1+E)}$.  Observe that $\eta E\le \eta (1+E)=\frac{1}{L\sqrt{T}}$.
Plugging into (19) yields

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^{2}\bigr]
\le
\frac{2\bigl(f(w^{0})-f^{\star}\bigr)L(1+E)}{\sqrt{T}}
+ \frac{2\sigma^{2}}{L\sqrt{T}(1+E)}
+ \frac{2\delta^{2}}{L\sqrt{T}} .
\]

Absorbing constants gives the bound (6) stated in Theorem~\ref{thm:main}.
\hfill $\square$

%=====================================================================
\section*{Rate Derivation (With Constants)}
%=====================================================================
From (6) and the choice (5),

\[
\begin{aligned}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w^{t})}^{2}\bigr]
&\le
\frac{2\bigl(f(w^{0})-f^{\star}\bigr)L(1+E)}{\sqrt{T}}
+ \frac{2\sigma^{2}}{L\sqrt{T}(1+E)}
+ \frac{8L^{2}E^{2}\delta^{2}}{L^{2}T(1+E)^{2}}  \\
&= O\!\Bigl(\frac{1}{\sqrt{T}} + \frac{E\delta}{\sqrt{T}}\Bigr).
\end{aligned}
\]

Hence the algorithm achieves the optimal $O(1/\sqrt{T})$ rate for non‑convex
smooth optimization, with an additional linear dependence on the
local epoch count $E$ and the heterogeneity constant $\delta$.

%=====================================================================
\section*{Special Cases}
%=====================================================================
\begin{itemize}
\item \textbf{Identical data across clients ($\delta=0$).}  
      The drift terms vanish and (6) reduces to the classical SGD bound
      $O\bigl(1/\sqrt{T}\bigr)$.
\item \textbf{Single local step ($E=1$).}  
      The algorithm coincides with standard distributed SGD; the bound
      becomes $O\bigl(1/\sqrt{T}\bigr)$ with only the stochastic noise term.
\item \textbf{Large $E$ but homogeneous data.}  
      Even with many local steps, if $\delta=0$ the convergence rate
      remains $O(1/\sqrt{T})$, explaining the empirical success of large $E$
      when data are IID.
\end{itemize}
