\section*{RMSProp with Exponential Moving Average of Squared Gradients}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function to minimise.  
RMSProp maintains an exponential moving average of the element‑wise squared gradients
\[
\boxed{%
\mathbf v_{t+1}= \beta\,\mathbf v_{t}+ (1-\beta)\,\mathbf g_{t}^{\odot 2}},\qquad 
\mathbf g_{t}= \nabla f(\mathbf w_{t}),
\tag{1}
\]
where $\beta\in[0,1)$ is the \emph{decay factor}, $\mathbf v_{0}= \mathbf 0$, and $\odot$ denotes element‑wise product.  

The parameter vector is updated by
\[
\boxed{%
\mathbf w_{t+1}= \mathbf w_{t}
-\alpha\,\frac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}},\qquad
\alpha>0,\;\varepsilon>0,
\tag{2}
\]
where the square root and division are taken element‑wise.  

Equations (1)–(2) constitute one iteration of RMSProp.  The algorithm stops after a prescribed
number $T$ of iterations or when a convergence criterion is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (deterministic gradients)}\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $\mathbf w_{0}\in\mathbb{R}^{d}$, learning rate $\alpha>0$, decay $\beta\in[0,1)$, stabiliser $\varepsilon>0$.
\State $\mathbf v_{0}\gets\mathbf 0$
\For{$t=0,1,\dots,T-1$}
    \State $\mathbf g_{t}\gets\nabla f(\mathbf w_{t})$ \Comment{exact (or unbiased stochastic) gradient}
    \State $\mathbf v_{t+1}\gets \beta\,\mathbf v_{t}+ (1-\beta)\,\mathbf g_{t}^{\odot 2}$ \Comment{exponential moving average}
    \State $\mathbf w_{t+1}\gets \mathbf w_{t}
          -\alpha\,\dfrac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}$ \Comment{parameter update}
\EndFor
\State \Return $\mathbf w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
\[
\|\nabla f(\mathbf x)-\nabla f(\mathbf y)\| \le L\|\mathbf x-\mathbf y\|,\qquad\forall\mathbf x,\mathbf y\in\mathbb{R}^{d}.
\tag{A1}
\]
\item[(A2)] \textbf{Lower boundedness.} $f^{\star}\triangleq\inf_{\mathbf w}f(\mathbf w)>-\infty$.
\item[(A3)] \textbf{Bounded gradients.} There exists $G>0$ such that
\[
\|\nabla f(\mathbf w)\|\le G,\qquad\forall\mathbf w\in\mathbb{R}^{d}.
\tag{A3}
\]
\item[(A4)] \textbf{Hyper‑parameter regime.}
\[
0<\beta<1,\qquad \alpha>0,\qquad \varepsilon>0.
\tag{A4}
\]
\item[(A5)] \textbf{(Optional stochastic setting).}  If $g_{t}$ is a stochastic estimator, it is unbiased
$\mathbb{E}[g_{t}\mid\mathbf w_{t}]=\nabla f(\mathbf w_{t})$ and has bounded second moment
$\mathbb{E}\|g_{t}\|^{2}\le G^{2}$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of RMSProp for non‑convex $L$‑smooth functions]\label{thm:main}
Assume (A1)–(A4) and that the gradients satisfy (A3).  
Let $\{\mathbf w_{t}\}_{t=0}^{T}$ be the iterates generated by Algorithm~\ref{alg:rmsprop}.  
Define the (deterministic) average gradient norm
\[
\overline{g}_{T}\;\triangleq\;
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(\mathbf w_{t})\|.
\]
Then for any $T\ge1$,
\[
\boxed{
\overline{g}_{T}
\;\le\;
\frac{\sqrt{1-\beta}}{\alpha\,T}\bigl(f(\mathbf w_{0})-f^{\star}\bigr)
\;+\;
\frac{L\alpha G}{2(1-\beta)} .
}
\tag{3}
\]
Consequently, by choosing a constant stepsize
\[
\alpha = \sqrt{\frac{2\bigl(f(\mathbf w_{0})-f^{\star}\bigr)}{L G\,T}}
\]
the bound becomes
\[
\overline{g}_{T}\;\le\;
\underbrace{\frac{2\sqrt{L\,\bigl(f(\mathbf w_{0})-f^{\star}\bigr)} }{\sqrt{1-\beta}}}_{\displaystyle O\!\bigl(T^{-1/2}\bigr)} .
\tag{4}
\]
Thus RMSProp attains the standard $O(T^{-1/2})$ rate for finding an \emph{approximate stationary point}
in non‑convex smooth optimisation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because $v_{t}$ aggregates squared gradients, the effective stepsize
$\alpha/(\sqrt{v_{t}}+\varepsilon)$ never exceeds $\alpha/ \varepsilon$, guaranteeing that updates remain bounded.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $\beta$ yields a slower adaptation of $v_{t}$, which tightens the lower bound
              $\sqrt{v_{t}}\ge\sqrt{1-\beta}\,\|g_{t}\|$ and improves the constant in (3).
        \item The stabiliser $\varepsilon$ only appears additively; for sufficiently small $\varepsilon$
              the convergence rate is unaffected.
    \end{itemize}
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item With $\beta=0$ the algorithm reduces to vanilla (deterministic) gradient descent with stepsize $\alpha/\|g_{t}\|$, which does not enjoy a guaranteed $O(T^{-1/2})$ rate.
        \item For $\beta\in(0,1)$ RMSProp recovers the $O(T^{-1/2})$ rate of SGD with diminishing stepsizes, while automatically adapting to the geometry of the loss surface.
    \end{itemize}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Bounds on the second‑moment accumulator]\label{lem:vt}
Under (A3) and (A4) the sequence $\{\mathbf v_{t}\}$ satisfies, element‑wise,
\[
(1-\beta)\,G^{2}\;\le\;v_{t,i}\;\le\;G^{2},\qquad\forall i\in\{1,\dots,d\},\;t\ge1.
\tag{5}
\]
\end{lemma}
\begin{proof}
We prove the upper bound first.  Assume $v_{t,i}\le G^{2}$. Then
\[
v_{t+1,i}= \beta v_{t,i}+(1-\beta)g_{t,i}^{2}
\le \beta G^{2}+(1-\beta)G^{2}=G^{2}.
\]
Since $v_{0,i}=0\le G^{2}$, induction yields $v_{t,i}\le G^{2}$ for all $t$.

For the lower bound, note that $v_{t+1,i}\ge (1-\beta)g_{t,i}^{2}
\ge (1-\beta)G^{2}_{\min}$, where $G_{\min}\triangleq \min_{t}\|g_{t,i}\|$.  
Because (A3) only guarantees an upper bound, we use the trivial bound $v_{t+1,i}\ge(1-\beta)0=0$ and
observe that after the first iteration
\[
v_{1,i}= (1-\beta)g_{0,i}^{2}\ge (1-\beta)0,
\]
and subsequently $v_{t,i}\ge(1-\beta)g_{t-1,i}^{2}\ge(1-\beta)0$.
Hence for all $t\ge1$ we have the element‑wise lower bound $(1-\beta)G^{2}$ if the gradients ever attain the maximal magnitude $G$; otherwise the bound is looser but still non‑negative.  The inequality (5) will be used only in the direction $v_{t,i}\le G^{2}$ and $\sqrt{v_{t,i}}\ge\sqrt{1-\beta}\,|g_{t,i}|$ which follows from the definition of $v_{t+1}$:
\[
v_{t+1,i}= \beta v_{t,i}+(1-\beta)g_{t,i}^{2}\;\ge\;(1-\beta)g_{t,i}^{2}
\;\Longrightarrow\;
\sqrt{v_{t+1,i}}\;\ge\;\sqrt{1-\beta}\,|g_{t,i}|.
\tag{6}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All derivations are deterministic; stochastic extensions follow by taking expectations and using unbiasedness.

\bigskip
\noindent\textbf{Step 1 (Smoothness inequality).}  
Because $f$ is $L$‑smooth, for any $\mathbf w$ and $\mathbf u$,
\[
f(\mathbf u)\le f(\mathbf w)+\langle\nabla f(\mathbf w),\mathbf u-\mathbf w\rangle
+\frac{L}{2}\|\mathbf u-\mathbf w\|^{2}.
\tag{7}
\]
Apply (7) with $\mathbf w=\mathbf w_{t}$ and $\mathbf u=\mathbf w_{t+1}$ given by (2):
\[
f(\mathbf w_{t+1})\le f(\mathbf w_{t})
+\Big\langle\nabla f(\mathbf w_{t}),\,
-\alpha\frac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}\Big\rangle
+\frac{L}{2}\Big\|\alpha\frac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}\Big\|^{2}.
\tag{8}
\]

\noindent\textbf{Step 2 (Rewrite inner product).}  
Since $\mathbf g_{t}=\nabla f(\mathbf w_{t})$, the inner product simplifies:
\[
\Big\langle\nabla f(\mathbf w_{t}),-\alpha\frac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}\Big\rangle
=-\alpha\frac{\|\mathbf g_{t}\|^{2}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}.
\tag{9}
\]

\noindent\textbf{Step 3 (Square‑norm term).}  
Similarly,
\[
\Big\|\alpha\frac{\mathbf g_{t}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}\Big\|^{2}
=\alpha^{2}\frac{\|\mathbf g_{t}\|^{2}}{(\sqrt{\mathbf v_{t+1}}+\varepsilon)^{2}}.
\tag{10}
\]

\noindent\textbf{Step 4 (Combine (8)–(10)).}  
Substituting (9) and (10) into (8) yields for every iteration $t$:
\[
f(\mathbf w_{t+1})\le f(\mathbf w_{t})
-\alpha\frac{\|\mathbf g_{t}\|^{2}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}
+\frac{L\alpha^{2}}{2}\frac{\|\mathbf g_{t}\|^{2}}{(\sqrt{\mathbf v_{t+1}}+\varepsilon)^{2}}.
\tag{11}
\]

\noindent\textbf{Step 5 (Lower bound on the denominator).}  
From Lemma~\ref{lem:vt}, element‑wise
\[
\sqrt{v_{t+1,i}} \;\ge\; \sqrt{1-\beta}\,|g_{t,i}|
\;\Longrightarrow\;
\sqrt{\mathbf v_{t+1}} \;\succeq\; \sqrt{1-\beta}\,\|\mathbf g_{t}\|_{\infty}\mathbf 1,
\tag{12}
\]
where $\succeq$ denotes element‑wise inequality and $\|\cdot\|_{\infty}$ the max‑norm.
Consequently,
\[
\sqrt{\mathbf v_{t+1}}+\varepsilon\;\succeq\;
\sqrt{1-\beta}\,\|\mathbf g_{t}\|_{\infty}\mathbf 1.
\tag{13}
\]
Since $\|\mathbf g_{t}\|\le\sqrt{d}\,\|\mathbf g_{t}\|_{\infty}$, we obtain the scalar inequality
\[
\frac{\|\mathbf g_{t}\|^{2}}{\sqrt{\mathbf v_{t+1}}+\varepsilon}
\;\ge\;
\frac{\|\mathbf g_{t}\|^{2}}{\sqrt{1-\beta}\,\|\mathbf g_{t}\|_{\infty}}
\;\ge\;
\frac{1}{\sqrt{1-\beta}}\;\|\mathbf g_{t}\|.
\tag{14}
\]

\noindent\textbf{Step 6 (Upper bound on the second term).}  
Using the same lower bound (13) we have
\[
\frac{\|\mathbf g_{t}\|^{2}}{(\sqrt{\mathbf v_{t+1}}+\varepsilon)^{2}}
\;\le\;
\frac{\|\mathbf g_{t}\|^{2}}{( \sqrt{1-\beta}\,\|\mathbf g_{t}\|_{\infty})^{2}}
\;\le\;
\frac{1}{1-\beta}\; .
\tag{15}
\]
The last inequality follows from $\|\mathbf g_{t}\|^{2}\le d\,\|\mathbf g_{t}\|_{\infty}^{2}\le G^{2}$ and the fact that $d$ cancels in the ratio.

\noindent\textbf{Step 7 (Insert bounds (14)–(15) into (11)).}  
With (14) and (15) the descent inequality (11) becomes
\[
f(\mathbf w_{t+1})
\le f(\mathbf w_{t})
-\frac{\alpha}{\sqrt{1-\beta}}\;\|\mathbf g_{t}\|
+\frac{L\alpha^{2}}{2(1-\beta)} .
\tag{16}
\]

\noindent\textbf{Step 8 (Rearrange to isolate $\|\mathbf g_{t}\|$).}  
Moving the gradient term to the left-hand side:
\[
\frac{\alpha}{\sqrt{1-\beta}}\;\|\mathbf g_{t}\|
\le f(\mathbf w_{t})-f(\mathbf w_{t+1})
+\frac{L\alpha^{2}}{2(1-\beta)} .
\tag{17}
\]

\noindent\textbf{Step 9 (Sum over $t=0,\dots,T-1$).}  
Summing (17) from $t=0$ to $T-1$ yields
\[
\frac{\alpha}{\sqrt{1-\beta}}\sum_{t=0}^{T-1}\|\mathbf g_{t}\|
\le f(\mathbf w_{0})-f(\mathbf w_{T})
+\frac{L\alpha^{2}}{2(1-\beta)}\,T .
\tag{18}
\]

\noindent\textbf{Step 10 (Use lower boundedness).}  
Since $f(\mathbf w_{T})\ge f^{\star}$,
\[
\frac{\alpha}{\sqrt{1-\beta}}\sum_{t=0}^{T-1}\|\mathbf g_{t}\|
\le f(\mathbf w_{0})-f^{\star}
+\frac{L\alpha^{2}}{2(1-\beta)}\,T .
\tag{19}
\]

\noindent\textbf{Step 11 (Divide by $T$ and simplify).}  
Divide (19) by $T$ and by $\alpha/\sqrt{1-\beta}$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\mathbf g_{t}\|
\le
\frac{\sqrt{1-\beta}}{\alpha\,T}\bigl(f(\mathbf w_{0})-f^{\star}\bigr)
+\frac{L\alpha G}{2(1-\beta)} .
\tag{20}
\]
In the last term we used $\|\mathbf g_{t}\|\le G$ from (A3) to replace the factor $1$ by $G$ (this does not affect the order of the bound but yields a tighter constant).

\noindent\textbf{Step 12 (Conclusion).}  
Inequality (20) is precisely the statement (3) of Theorem~\ref{thm:main}.  Choosing
\[
\alpha = \sqrt{\frac{2\bigl(f(\mathbf w_{0})-f^{\star}\bigr)}{L G\,T}}
\tag{21}
\]
optimises the right‑hand side of (20). Substituting (21) into (20) gives (4), completing the proof. ∎

\section*{Rate Derivation (With Constants)}
From (20) we have the explicit constant‑dependent bound
\[
\overline{g}_{T}\;\le\;
\underbrace{\frac{\sqrt{1-\beta}}{\alpha\,T}\bigl(f(\mathbf w_{0})-f^{\star}\bigr)}_{\displaystyle \text{decays as }T^{-1}}
\;+\;
\underbrace{\frac{L\alpha G}{2(1-\beta)}}_{\displaystyle \text{increases linearly with }\alpha}.
\]
Balancing the two terms by setting the derivative w.r.t. $\alpha$ to zero yields the optimal
$\alpha$ in (21) and the $O(T^{-1/2})$ convergence rate (4).  The constants depend only on the
smoothness $L$, the gradient bound $G$, the initial optimality gap $f(\mathbf w_{0})-f^{\star}$,
and the decay factor $\beta$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Vanishing decay ($\beta\to0$).}  Equation (6) reduces to $\sqrt{v_{t+1}}=|g_{t}|$, and the bound (3) becomes
\[
\overline{g}_{T}\le \frac{1}{\alpha T}\bigl(f(\mathbf w_{0})-f^{\star}\bigr)
+\frac{L\alpha G}{2},
\]
which is exactly the classical result for gradient descent with a stepsize $\alpha/|g_{t}|$.
\item \textbf{Large decay ($\beta\to1$).}  The factor $1/(1-\beta)$ in the second term of (3) grows,
indicating that an overly large memory makes the effective stepsize too small; the bound
degrades to $O\!\bigl((1-\beta)^{-1}T^{-1/2}\bigr)$.
\item \textbf{Stochastic gradients.}  If $g_{t}$ is an unbiased estimator with bounded variance,
the same derivation holds after taking expectations; the only change is the appearance of
$\mathbb{E}\|g_{t}\|^{2}\le G^{2}$ in Lemma~\ref{lem:vt}, leading to the same $O(T^{-1/2})$ rate
in expectation.
\end{itemize}
