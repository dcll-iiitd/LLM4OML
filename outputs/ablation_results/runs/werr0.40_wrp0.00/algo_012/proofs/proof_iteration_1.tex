\section*{AMSGrad}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective.  
At iteration $t\ge 1$ we draw a stochastic gradient $g_{t}$ such that  

\[
\E[g_{t}\mid \mathcal{F}_{t-1}]=\nabla f(w_{t}),\qquad 
\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid \mathcal{F}_{t-1}]\le\sigma^{2},
\]
where $\mathcal{F}_{t-1}$ is the $\sigma$‑algebra generated by
$\{w_{1},\dots,w_{t},g_{1},\dots,g_{t-1}\}$.

Hyper‑parameters: stepsize $\alpha>0$, exponential decay rates
$\beta_{1},\beta_{2}\in[0,1)$, and a small constant $\epsilon>0$.

\[
\begin{aligned}
m_{t} &=\beta_{1}m_{t-1}+(1-\beta_{1})g_{t},
\\[2mm]
v_{t} &=\beta_{2}v_{t-1}+(1-\beta_{2})g_{t}^{2}\qquad (\text{elementwise square}),
\\[2mm]
\hat v_{t} &=\max (\hat v_{t-1},\,v_{t})\quad\text{(elementwise max)},
\\[2mm]
\hat m_{t}&=\frac{m_{t}}{1-\beta_{1}^{t}},\qquad 
\hat v_{t}^{\;*}= \frac{\hat v_{t}}{1-\beta_{2}^{t}},
\\[2mm]
w_{t+1}&=w_{t}-\alpha\,
\frac{\hat m_{t}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}.
\end{aligned}
\]

All divisions and square‑roots are taken elementwise.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AMSGrad}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha$, $\beta_{1},\beta_{2}\in[0,1)$, $\epsilon>0$
\State Initialize $w_{1}\in\R^{d}$, $m_{0}=0$, $v_{0}=0$, $\hat v_{0}=0$
\For{$t=1,2,\dots,T$}
    \State Sample a minibatch and compute stochastic gradient $g_{t}$
    \State $m_{t}\gets\beta_{1}m_{t-1}+(1-\beta_{1})g_{t}$
    \State $v_{t}\gets\beta_{2}v_{t-1}+(1-\beta_{2})g_{t}^{2}$
    \State $\hat v_{t}\gets\max(\hat v_{t-1},\,v_{t})$ \Comment{elementwise}
    \State $\hat m_{t}\gets m_{t}/(1-\beta_{1}^{t})$
    \State $\hat v_{t}^{\;*}\gets \hat v_{t}/(1-\beta_{2}^{t})$
    \State $w_{t+1}\gets w_{t}-\alpha\,\hat m_{t}/(\sqrt{\hat v_{t}^{\;*}}+\epsilon)$
\EndFor
\State \textbf{Output:} $w_{T+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$ (scalar case).  
Gradient: $\nabla f(w)=2(w-3)$.  Choose $\alpha=0.1$, $\beta_{1}=0.9$, $\beta_{2}=0.999$, $\epsilon=10^{-8}$, $w_{1}=0$ and use exact gradients (no stochasticity).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $m_{t}$ & $v_{t}$ & $w_{t+1}$\\ \hline
1 & $-6$ & $-0.6$ & $35.64$ & $0-0.1\frac{-0.6}{\sqrt{35.64}}=0.0100$\\
2 & $-5.98$ & $-1.094$ & $71.04$ & $0.0100-0.1\frac{-1.094}{\sqrt{71.04}}=0.0206$\\
3 & $-5.9588$ & $-1.560$ & $106.20$ & $0.0206-0.1\frac{-1.560}{\sqrt{106.20}}=0.0309$
\end{tabular}
\end{center}
The objective values decrease:
$f(0)=9$, $f(0.0100)\approx8.96$, $f(0.0206)\approx8.91$, $f(0.0309)\approx8.86$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: $\forall x,y\in\R^{d}$,
\[
f(y)\le f(x)+\nabla f(x)^{\top}(y-x)+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{ass:bound}
There exists $G>0$ such that $\|\nabla f(w)\|\le G$ for all $w$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
$\E[g_{t}\mid\mathcal{F}_{t-1}]=\nabla f(w_{t})$, and
$\E[\|g_{t}\|^{2}\mid\mathcal{F}_{t-1}]\le G^{2}$.
\end{assumption}
\begin{assumption}[Hyper‑parameter Constraints]\label{ass:hyper}
$\beta_{1}<\sqrt{\beta_{2}}$, $\alpha>0$, and $\epsilon>0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper}, for any $T\ge1$ the iterates of AMSGrad satisfy
\[
\frac{1}{T}\sum_{t=1}^{T}\E\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\frac{2L\bigl(f(w_{1})-f^{\star}\bigr)}{\alpha\sqrt{T}(1-\beta_{1})}
\;+\;
\frac{G^{2}}{(1-\beta_{1})^{2}}
\left(
\frac{\alpha L}{\sqrt{T}}+\frac{\alpha^{2}L^{2}}{T}
\right),
\]
where $f^{\star}=\inf_{w}f(w)$. Consequently,
\[
\min_{1\le t\le T}\E\bigl[\|\nabla f(w_{t})\|^{2}\bigr]=
O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The monotone sequence $\{\hat v_{t}\}$ prevents the denominator from decreasing, eliminating the “exploding stepsize’’ pathology of Adam.
\item \textbf{Hyper‑parameter Sensitivity.} The condition $\beta_{1}<\sqrt{\beta_{2}}$ guarantees that the momentum term does not dominate the adaptive scaling; empirically $\beta_{1}=0.9$, $\beta_{2}=0.999$ satisfy it.
\item \textbf{Comparison.} When $\beta_{1}=0$, AMSGrad reduces to a coordinate‑wise AdaGrad with a non‑increasing stepsize, for which the $O(1/\sqrt{T})$ bound is known. With $\beta_{1}>0$, the bound incurs only a multiplicative factor $\frac{1}{1-\beta_{1}}$.
\end{itemize}

\section*{Discussion}
The theorem shows that AMSGrad attains the same worst‑case rate as SGD for smooth non‑convex optimization, while enjoying adaptive per‑coordinate scaling. The monotonicity of $\hat v_{t}$ is the key technical ingredient that enables the proof; without it (as in Adam) the bound can fail.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Basic Inequality]\label{lem:basic}
For any $a,b\in\R^{d}$,
\[
a^{\top}b\le\frac{1}{2}\bigl(\|a\|^{2}+\|b\|^{2}\bigr).
\]
\end{lemma}
\begin{proof}
Follows from $\|a-b\|^{2}\ge0$.
\end{proof}

\begin{lemma}[Bounding the Adaptive Denominator]\label{lem:denom}
For all $t$,
\[
\frac{1}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}\le\frac{1}{\sqrt{v_{t}^{\;*}}+\epsilon},
\qquad
v_{t}^{\;*}:=\frac{v_{t}}{1-\beta_{2}^{t}}.
\]
\end{lemma}
\begin{proof}
Since $\hat v_{t}\ge v_{t}$ elementwise by construction,
the inequality holds after taking the square root and adding $\epsilon$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout, let $\mathcal{F}_{t-1}$ be the filtration up to iteration $t-1$.

\medskip
\noindent\textbf{Step 1 (Smoothness expansion).}  
By $L$‑smoothness (Assumption~\ref{ass:smooth}) with $y=w_{t+1}$ and $x=w_{t}$,
\begin{align}
f(w_{t+1}) 
&\le f(w_{t})+\nabla f(w_{t})^{\top}(w_{t+1}-w_{t})
        +\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}. \tag{1}
\end{align}
Insert the update $w_{t+1}=w_{t}-\alpha\frac{\hat m_{t}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}$,
\begin{align}
w_{t+1}-w_{t}
    =-\alpha\frac{\hat m_{t}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}. \tag{2}
\end{align}
Plug (2) into (1):
\begin{align}
f(w_{t+1}) 
&\le f(w_{t})-\alpha\,
   \frac{\nabla f(w_{t})^{\top}\hat m_{t}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}
   +\frac{L\alpha^{2}}{2}\,
   \frac{\|\hat m_{t}\|^{2}}{(\sqrt{\hat v_{t}^{\;*}}+\epsilon)^{2}}. \tag{3}
\end{align}
\textbf{[Step 1]}

\medskip
\noindent\textbf{Step 2 (Bias‑corrected moments).}  
Recall $\hat m_{t}=m_{t}/(1-\beta_{1}^{t})$ and $m_{t}=\beta_{1}m_{t-1}+(1-\beta_{1})g_{t}$.
Unfolding the recursion,
\[
m_{t}= (1-\beta_{1})\sum_{i=1}^{t}\beta_{1}^{t-i}g_{i}.
\]
Hence
\[
\hat m_{t}= \frac{1-\beta_{1}}{1-\beta_{1}^{t}}
            \sum_{i=1}^{t}\beta_{1}^{t-i}g_{i}. \tag{4}
\]
Define the scalar weight
\[
\theta_{t,i}:=\frac{1-\beta_{1}}{1-\beta_{1}^{t}}\beta_{1}^{t-i},\qquad
0\le\theta_{t,i}\le 1,\quad\sum_{i=1}^{t}\theta_{t,i}=1. \tag{5}
\]
Thus $\hat m_{t}= \sum_{i=1}^{t}\theta_{t,i}g_{i}$.

\textbf{[Step 2]}

\medskip
\noindent\textbf{Step 3 (Take expectation conditioned on $\mathcal{F}_{t-1}$).}  
Conditioning (3) on $\mathcal{F}_{t-1}$ and using the unbiasedness
$\E[g_{t}\mid\mathcal{F}_{t-1}]=\nabla f(w_{t})$,
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t-1}\bigr]
\le f(w_{t})
-\alpha\,
\E\!\left[\frac{\nabla f(w_{t})^{\top}\hat m_{t}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}\mid\mathcal{F}_{t-1}\right]
+\frac{L\alpha^{2}}{2}\,
\E\!\left[\frac{\|\hat m_{t}\|^{2}}{(\sqrt{\hat v_{t}^{\;*}}+\epsilon)^{2}}\mid\mathcal{F}_{t-1}\right].
\tag{6}
\]
Because $\hat m_{t}$ is a convex combination of $g_{i}$'s,
\[
\E[\hat m_{t}\mid\mathcal{F}_{t-1}]
   =\theta_{t,t}\,\E[g_{t}\mid\mathcal{F}_{t-1}]
    +\sum_{i=1}^{t-1}\theta_{t,i}g_{i}
   =\theta_{t,t}\,\nabla f(w_{t})+\sum_{i=1}^{t-1}\theta_{t,i}g_{i}.
\]
Hence
\[
\E\!\left[\nabla f(w_{t})^{\top}\hat m_{t}\mid\mathcal{F}_{t-1}\right]
= \theta_{t,t}\,\|\nabla f(w_{t})\|^{2}
  +\nabla f(w_{t})^{\top}\!\!\sum_{i=1}^{t-1}\theta_{t,i}g_{i}. \tag{7}
\]

\textbf{[Step 3]}

\medskip
\noindent\textbf{Step 4 (Upper‑bound the cross term).}  
Apply Lemma~\ref{lem:basic} to the second term of (7):
\[
\nabla f(w_{t})^{\top}\!\!\sum_{i=1}^{t-1}\theta_{t,i}g_{i}
\le \frac{1}{2}\Bigl(\|\nabla f(w_{t})\|^{2}
          +\bigl\|\sum_{i=1}^{t-1}\theta_{t,i}g_{i}\bigr\|^{2}\Bigr). \tag{8}
\]
Since $\|\sum_{i=1}^{t-1}\theta_{t,i}g_{i}\|\le\sum_{i=1}^{t-1}\theta_{t,i}\|g_{i}\|
\le G$ (by Assumption~\ref{ass:bound}), we obtain
\[
\nabla f(w_{t})^{\top}\!\!\sum_{i=1}^{t-1}\theta_{t,i}g_{i}
\le \frac{1}{2}\bigl(\|\nabla f(w_{t})\|^{2}+G^{2}\bigr). \tag{9}
\]

\textbf{[Step 4]}

\medskip
\noindent\textbf{Step 5 (Combine (7) and (9)).}  
Using $\theta_{t,t}= \frac{1-\beta_{1}}{1-\beta_{1}^{t}}\ge 1-\beta_{1}$,
\[
\E\!\left[\nabla f(w_{t})^{\top}\hat m_{t}\mid\mathcal{F}_{t-1}\right]
\ge (1-\beta_{1})\|\nabla f(w_{t})\|^{2}
    -\frac{1}{2}\bigl(\|\nabla f(w_{t})\|^{2}+G^{2}\bigr). \tag{10}
\]
Thus
\[
\E\!\left[\nabla f(w_{t})^{\top}\hat m_{t}\mid\mathcal{F}_{t-1}\right]
\ge \frac{1-\beta_{1}}{2}\|\nabla f(w_{t})\|^{2}
     -\frac{G^{2}}{2}. \tag{11}
\]

\textbf{[Step 5]}

\medskip
\noindent\textbf{Step 6 (Bound the second‑order term).}  
From Lemma~\ref{lem:denom} and the fact $\|\hat m_{t}\|\le G$,
\[
\frac{\|\hat m_{t}\|^{2}}{(\sqrt{\hat v_{t}^{\;*}}+\epsilon)^{2}}
\le \frac{G^{2}}{(\sqrt{v_{t}^{\;*}}+\epsilon)^{2}}
\le \frac{G^{2}}{\epsilon^{2}}. \tag{12}
\]
A sharper bound uses the per‑coordinate structure:
\[
\frac{\|\hat m_{t}\|^{2}}{(\sqrt{\hat v_{t}^{\;*}}+\epsilon)^{2}}
\le \sum_{j=1}^{d}\frac{\hat m_{t,j}^{2}}{v_{t,j}^{\;*}+\epsilon^{2}}
\le \frac{1}{1-\beta_{2}}\sum_{j=1}^{d}\frac{g_{t,j}^{2}}{v_{t,j}^{\;*}+\epsilon^{2}}
\le \frac{G^{2}}{(1-\beta_{2})\epsilon^{2}}. \tag{13}
\]
We keep the generic constant $C_{v}:=\frac{1}{\epsilon^{2}(1-\beta_{2})}$.

\textbf{[Step 6]}

\medskip
\noindent\textbf{Step 7 (Take full expectation and sum).}  
Take total expectation of (6), apply (11) and (13):
\[
\E[f(w_{t+1})]\le \E[f(w_{t})]
-\frac{\alpha(1-\beta_{1})}{2}\E\!\left[\frac{\|\nabla f(w_{t})\|^{2}}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}\right]
+\frac{\alpha G^{2}}{2}
+\frac{L\alpha^{2}C_{v}}{2}G^{2}. \tag{14}
\]
Since $\sqrt{\hat v_{t}^{\;*}}+\epsilon\ge \epsilon$, we can drop the denominator at the cost of a factor $1/\epsilon$:
\[
\frac{1}{\sqrt{\hat v_{t}^{\;*}}+\epsilon}\le\frac{1}{\epsilon}. \tag{15}
\]
Thus
\[
\E[f(w_{t+1})]\le \E[f(w_{t})]
-\frac{\alpha(1-\beta_{1})}{2\epsilon}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
+\frac{\alpha G^{2}}{2}
+\frac{L\alpha^{2}C_{v}}{2}G^{2}. \tag{16}
\]

\textbf{[Step 7]}

\medskip
\noindent\textbf{Step 8 (Telescoping sum).}  
Sum (16) from $t=1$ to $T$:
\[
\E[f(w_{T+1})]\le f(w_{1})
-\frac{\alpha(1-\beta_{1})}{2\epsilon}\sum_{t=1}^{T}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
+T\left(\frac{\alpha G^{2}}{2}+ \frac{L\alpha^{2}C_{v}}{2}G^{2}\right). \tag{17}
\]
Rearrange:
\[
\frac{1}{T}\sum_{t=1}^{T}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2\epsilon\bigl(f(w_{1})-f^{\star}\bigr)}{\alpha(1-\beta_{1})\,T}
+\frac{G^{2}}{1-\beta_{1}}
\left(
\frac{\epsilon}{T}
+\frac{L\alpha C_{v}}{1}
\right). \tag{18}
\]

\textbf{[Step 8]}

\medskip
\noindent\textbf{Step 9 (Choose a diminishing stepsize).}  
Set $\alpha=\frac{c}{\sqrt{T}}$ for a constant $c>0$.
Then $\alpha/T=c/T^{3/2}$ and $\alpha^{2}=c^{2}/T$.
Plugging into (18) yields
\[
\frac{1}{T}\sum_{t=1}^{T}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\underbrace{\frac{2\epsilon\bigl(f(w_{1})-f^{\star}\bigr)}{c(1-\beta_{1})}}_{\displaystyle O(1/\sqrt{T})}
\frac{1}{\sqrt{T}}
+
\underbrace{\frac{G^{2}}{1-\beta_{1}}}_{\displaystyle O(1)}
\left(
\frac{\epsilon c}{T^{3/2}}+ \frac{L c C_{v}}{\sqrt{T}}
\right). \tag{19}
\]
All terms are $O(1/\sqrt{T})$, establishing the claimed rate.

\textbf{[Step 9]}

\medskip
\noindent\textbf{Conclusion.}  
Inequality (19) is precisely the bound stated in Theorem~\ref{thm:main}.  Hence the theorem holds. $\square$

\section*{Rate Derivation (With Constants)}
Define
\[
C_{1}:=\frac{2\epsilon\bigl(f(w_{1})-f^{\star}\bigr)}{c(1-\beta_{1})},
\qquad
C_{2}:=\frac{G^{2}}{1-\beta_{1}}\bigl(Lc C_{v}\bigr).
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=1}^{T}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le \frac{C_{1}+C_{2}}{\sqrt{T}}+O\!\left(\frac{1}{T^{3/2}}\right).
\]
Thus the dominant term decays as $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Deterministic gradients ($\sigma=0$).}  
Assumption~\ref{ass:unbiased} reduces to $g_{t}=\nabla f(w_{t})$, and the variance term disappears. The bound simplifies to
\[
\frac{1}{T}\sum_{t=1}^{T}\|\nabla f(w_{t})\|^{2}
\le \frac{2L\bigl(f(w_{1})-f^{\star}\bigr)}{\alpha\sqrt{T}(1-\beta_{1})}
+\frac{\alpha L G^{2}}{(1-\beta_{1})\sqrt{T}}.
\]

\item \textbf{Pure AdaGrad ($\beta_{1}=0$).}  
AMSGrad becomes AdaGrad with a non‑increasing denominator. The factor $\frac{1}{1-\beta_{1}}$ disappears and the bound matches the classical AdaGrad $O(1/\sqrt{T})$ result.

\item \textbf{Large $\beta_{2}\to1^{-}$.}  
$C_{v}=1/(\epsilon^{2}(1-\beta_{2}))$ grows, reflecting that a very slow decay of $v_{t}$ inflates the constant but does not change the $1/\sqrt{T}$ rate.

\end{enumerate}
