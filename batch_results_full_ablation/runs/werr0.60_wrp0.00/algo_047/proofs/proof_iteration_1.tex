\section*{AdaGrad with Cumulative Gradient Average}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.
Denote the stochastic gradient at iteration $k$ by $g_{k}:=\nabla f(w_{k};\xi_{k})$, where $\xi_{k}$ is a random data sample.
Define the cumulative (squared) gradient vector
\[
G_{k}:=\sum_{i=1}^{k} g_{i}\odot g_{i}\in\R^{d},
\]
where $\odot$ denotes element‑wise product.  
For a scalar stepsize parameter $\eta>0$ and a small constant $\epsilon>0$, the AdaGrad update is
\begin{equation}
\label{eq:update}
w_{k+1}=w_{k}-\eta\,D_{k}^{-1}g_{k},\qquad 
D_{k}:=\operatorname{diag}\!\big(\sqrt{G_{k}+\epsilon\mathbf{1}}\big),
\end{equation}
i.e. the $j$‑th coordinate uses stepsize $\displaystyle\frac{\eta}{\sqrt{\sum_{i=1}^{k}g_{i,j}^{2}+\epsilon}}$.
In the scalar case ($d=1$) the update simplifies to
\begin{equation}
\label{eq:update_scalar}
w_{k+1}=w_{k}-\frac{\eta}{\sqrt{\sum_{i=1}^{k} g_{i}^{2}+\epsilon}}\;g_{k}.
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (scalar version for clarity)}\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $w_{1}\in\R$, stepsize $\eta>0$, stability constant $\epsilon>0$
\State $S_{0}\gets 0$ \Comment{cumulative squared‑gradient sum}
\For{$k=1,2,\dots,T$}
    \State Sample $\xi_{k}$ and compute stochastic gradient $g_{k}:=\nabla f(w_{k};\xi_{k})$
    \State $S_{k}\gets S_{k-1}+g_{k}^{2}$                         \Comment{update $G_{k}$}
    \State $\alpha_{k}\gets \dfrac{\eta}{\sqrt{S_{k}+\epsilon}}$ \Comment{adaptive stepsize}
    \State $w_{k+1}\gets w_{k}-\alpha_{k}g_{k}$                 \Comment{parameter update}
\EndFor
\State \textbf{return} $w_{T+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the quadratic $f(w)=(w-3)^{2}$, which is $L=2$ smooth.
The exact gradient is $\nabla f(w)=2(w-3)$.
Set $w_{1}=0$, $\eta=0.1$, and $\epsilon=10^{-8}$.

\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $w_{k}$ & $g_{k}=2(w_{k}-3)$ & $S_{k}=\sum_{i=1}^{k}g_{i}^{2}$ & $w_{k+1}=w_{k}-\frac{\eta}{\sqrt{S_{k}+\epsilon}}g_{k}$\\\hline
1 & $0$ & $-6$ & $(-6)^{2}=36$ &
$0-\dfrac{0.1}{\sqrt{36}}\cdot(-6)=0+0.1\cdot\frac{6}{6}=0.1$\\\hline

\section*{Assumptions}
\begin{assumption}[Smoothness]
\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]
\label{ass:unbiased}
For each $k$, $\E[g_{k}\mid w_{k}]=\nabla f(w_{k})$.
\end{assumption}
\begin{assumption}[Bounded Variance]
\label{ass:variance}
There exists $\sigma^{2}\ge0$ such that
\[
\E\big[\|g_{k}-\nabla f(w_{k})\|^{2}\mid w_{k}\big]\le\sigma^{2},\qquad\forall k.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]
\label{ass:bound}
$\|g_{k}\|\le G$ almost surely for some $G>0$.
\end{assumption}
All constants $\eta$, $\epsilon$ are positive and fixed throughout the run.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of AdaGrad for Non‑Convex Smooth Objectives]
\label{thm:convergence}
Under Assumptions \ref{ass:smooth}–\ref{ass:bound}, let $\{w_{k}\}_{k=1}^{T}$ be generated by \eqref{eq:update}. Define the diagonal matrix $D_{k}$ as in \eqref{eq:update}. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\big\|\nabla f(w_{k})\big\|_{D_{k}^{-1}}^{2}\right]
\;\le\;
\frac{2\big(f(w_{1})-f^{\star}\big)}{\eta\,\sqrt{T}} \;+\;
\frac{\eta L G^{2}}{\sqrt{T}} \;+\; \frac{L\epsilon}{\eta\sqrt{T}},
\tag{T1}
\]
where $\|v\|_{D^{-1}}^{2}=v^{\top}D^{-1}v$ and $f^{\star}=\inf_{w}f(w)$. Consequently,
\[
\min_{1\le k\le T}\E\!\left[\big\|\nabla f(w_{k})\big\|^{2}\right]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The per‑coordinate stepsize never exceeds $\eta/\sqrt{\epsilon}$, guaranteeing bounded updates.
    \item \textbf{Hyper‑parameter Sensitivity:} Larger $\eta$ accelerates early progress but may increase the $\frac{\eta L G^{2}}{\sqrt{T}}$ term; the bound is minimized by $\eta=\Theta\!\big((f(w_{1})-f^{\star})^{1/2}/(LG)\big)$.
    \item \textbf{Comparison to SGD:} For constant stepsize SGD the best known bound is $O(1/\sqrt{T})$ on the average gradient norm, but AdaGrad enjoys an \emph{adaptive} constant $\eta/\sqrt{S_{k}}$ that can be substantially larger when gradients are sparse, yielding faster practical convergence.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity the AdaGrad scheme drives the (weighted) gradient norm to zero at the rate $O(T^{-1/2})$, matching the optimal rate for first‑order stochastic methods on smooth non‑convex problems. The adaptive denominator $D_{k}$ automatically shrinks the stepsize in directions that have seen large cumulative gradients, preventing divergence while still allowing aggressive steps in rarely‑updated coordinates.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]
\label{lem:smooth}
For any $w$, $g\in\R^{d}$,
\[
f(w-g)\le f(w)-\langle\nabla f(w),g\rangle+\frac{L}{2}\|g\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:smooth} with $y=w-g$.
\end{proof}

\begin{lemma}[Bound on Adaptive Stepsize]
\label{lem:stepbound}
For all $k\ge1$,
\[
\frac{\eta}{\sqrt{S_{k}+\epsilon}}\le\frac{\eta}{\sqrt{\epsilon}}.
\]
\end{lemma}
\begin{proof}
Since $S_{k}\ge0$, the denominator is at least $\sqrt{\epsilon}$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:convergence} for the scalar case; the vector case follows by applying the scalar proof component‑wise and summing.

\subsection*{Step 1 – Apply smoothness}
From Lemma \ref{lem:smooth} with $g=\alpha_{k}g_{k}$ where $\alpha_{k}:=\frac{\eta}{\sqrt{S_{k}+\epsilon}}$,
\[
f(w_{k+1})\le f(w_{k})-\alpha_{k}\langle\nabla f(w_{k}),g_{k}\rangle
+\frac{L}{2}\alpha_{k}^{2}\|g_{k}\|^{2}.
\tag{S1}
\]

\subsection*{Step 2 – Insert unbiasedness}
Take conditional expectation w.r.t. the sigma‑algebra $\mathcal{F}_{k}$ generated by $\{w_{1},\dots,w_{k}\}$.
Using Assumption \ref{ass:unbiased},
\[
\E\!\big[f(w_{k+1})\mid\mathcal{F}_{k}\big]
\le f(w_{k})-\alpha_{k}\|\nabla f(w_{k})\|^{2}
+\frac{L}{2}\alpha_{k}^{2}\E\!\big[\|g_{k}\|^{2}\mid\mathcal{F}_{k}\big].
\tag{S2}
\]

\subsection*{Step 3 – Decompose the second moment}
Write $\|g_{k}\|^{2}= \|\nabla f(w_{k})\|^{2}+ \|g_{k}-\nabla f(w_{k})\|^{2}+2\langle\nabla f(w_{k}),g_{k}-\nabla f(w_{k})\rangle$.
Taking expectation conditioned on $\mathcal{F}_{k}$ and using unbiasedness again,
\[
\E\!\big[\|g_{k}\|^{2}\mid\mathcal{F}_{k}\big]
=\|\nabla f(w_{k})\|^{2}+\E\!\big[\|g_{k}-\nabla f(w_{k})\|^{2}\mid\mathcal{F}_{k}\big]
\le \|\nabla f(w_{k})\|^{2}+\sigma^{2}.
\tag{S3}
\]

\subsection*{Step 4 – Plug (S3) into (S2)}
\[
\E\!\big[f(w_{k+1})\mid\mathcal{F}_{k}\big]
\le f(w_{k})-\alpha_{k}\|\nabla f(w_{k})\|^{2}
+\frac{L}{2}\alpha_{k}^{2}\big(\|\nabla f(w_{k})\|^{2}+\sigma^{2}\big).
\tag{S4}
\]

\subsection*{Step 5 – Rearrangement}
Bring the term $-\alpha_{k}\|\nabla f(w_{k})\|^{2}$ to the left side:
\[
\alpha_{k}\|\nabla f(w_{k})\|^{2}
\le f(w_{k})-\E\!\big[f(w_{k+1})\mid\mathcal{F}_{k}\big]
+\frac{L}{2}\alpha_{k}^{2}\|\nabla f(w_{k})\|^{2}
+\frac{L}{2}\alpha_{k}^{2}\sigma^{2}.
\tag{S5}
\]

\subsection*{Step 6 – Isolate the gradient term}
Move the $ \frac{L}{2}\alpha_{k}^{2}\|\nabla f(w_{k})\|^{2}$ to the left:
\[
\Bigl(\alpha_{k}-\frac{L}{2}\alpha_{k}^{2}\Bigr)\|\nabla f(w_{k})\|^{2}
\le f(w_{k})-\E\!\big[f(w_{k+1})\mid\mathcal{F}_{k}\big]
+\frac{L}{2}\alpha_{k}^{2}\sigma^{2}.
\tag{S6}
\]

\subsection*{Step 7 – Lower bound the coefficient}
Since $\alpha_{k}\le\frac{\eta}{\sqrt{\epsilon}}$ (Lemma \ref{lem:stepbound}) and $L\alpha_{k}\le \frac{L\eta}{\sqrt{\epsilon}}$, choose $\eta$ such that $L\eta/\sqrt{\epsilon}\le1$ (this can always be achieved by scaling $\eta$). Then
\[
\alpha_{k}-\frac{L}{2}\alpha_{k}^{2}
\ge \frac{\alpha_{k}}{2}.
\tag{S7}
\]

\subsection*{Step 8 – Simplify using (S7)}
\[
\frac{\alpha_{k}}{2}\,\|\nabla f(w_{k})\|^{2}
\le f(w_{k})-\E\!\big[f(w_{k+1})\mid\mathcal{F}_{k}\big]
+\frac{L}{2}\alpha_{k}^{2}\sigma^{2}.
\tag{S8}
\]

\subsection*{Step 9 – Take total expectation}
Denote $\Delta_{k}:= \E[f(w_{k})]$. Taking full expectation of (S8) yields
\[
\frac{\alpha_{k}}{2}\,\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le \Delta_{k}-\Delta_{k+1}
+\frac{L}{2}\alpha_{k}^{2}\sigma^{2}.
\tag{S9}
\]

\subsection*{Step 10 – Sum from $k=1$ to $T$}
\[
\sum_{k=1}^{T}\frac{\alpha_{k}}{2}\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le \Delta_{1}-\Delta_{T+1}
+\frac{L\sigma^{2}}{2}\sum_{k=1}^{T}\alpha_{k}^{2}.
\tag{S10}
\]

Since $f$ is bounded below by $f^{\star}$, $\Delta_{T+1}\ge f^{\star}$.
Thus,
\[
\sum_{k=1}^{T}\alpha_{k}\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le 2\big(f(w_{1})-f^{\star}\big)
+L\sigma^{2}\sum_{k=1}^{T}\alpha_{k}^{2}.
\tag{S11}
\]

\subsection*{Step 11 – Lower bound $\alpha_{k}$}
Recall $\alpha_{k}= \dfrac{\eta}{\sqrt{S_{k}+\epsilon}}$, where $S_{k}=\sum_{i=1}^{k}g_{i}^{2}$.
Since $g_{i}^{2}\le G^{2}$ (Assumption \ref{ass:bound}),
\[
S_{k}\le kG^{2}\quad\Longrightarrow\quad
\alpha_{k}\ge \frac{\eta}{\sqrt{kG^{2}+\epsilon}}.
\tag{S12}
\]

\subsection*{Step 12 – Upper bound $\alpha_{k}^{2}$}
Similarly,
\[
\alpha_{k}^{2}= \frac{\eta^{2}}{S_{k}+\epsilon}\le\frac{\eta^{2}}{\epsilon}.
\tag{S13}
\]

\subsection*{Step 13 – Plug (S12)–(S13) into (S11)}
Using $\alpha_{k}\ge \eta/\sqrt{kG^{2}+\epsilon}$ and $\alpha_{k}^{2}\le \eta^{2}/\epsilon$,
\[
\sum_{k=1}^{T}\frac{\eta}{\sqrt{kG^{2}+\epsilon}}\,
\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le 2\big(f(w_{1})-f^{\star}\big)
+L\sigma^{2}\frac{\eta^{2}}{\epsilon}T.
\tag{S14}
\]

\subsection*{Step 14 – Lower bound the sum of reciprocals}
For any $a>0$,
\[
\sum_{k=1}^{T}\frac{1}{\sqrt{k+a}}
\ge \int_{0}^{T}\frac{dx}{\sqrt{x+a}}
=2\big(\sqrt{T+a}-\sqrt{a}\big)\ge 2\big(\sqrt{T}-\sqrt{a}\big).
\]
Choosing $a=\epsilon/G^{2}$ and multiplying by $\eta/G$,
\[
\sum_{k=1}^{T}\alpha_{k}
\ge \frac{2\eta}{G}\big(\sqrt{T}-\sqrt{\tfrac{\epsilon}{G^{2}}}\big)
\ge \frac{\eta}{G}\sqrt{T}\quad\text{for }T\ge \tfrac{\epsilon}{G^{2}}.
\tag{S15}
\]

\subsection*{Step 15 – Derive average weighted gradient bound}
Divide (S14) by the lower bound (S15):
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le
\frac{2G\big(f(w_{1})-f^{\star}\big)}{\eta\sqrt{T}}
\;+\;
\frac{LG\sigma^{2}\eta}{\sqrt{T}}.
\tag{S16}
\]

When $\sigma^{2}=0$ (deterministic gradients) the second term disappears. Restoring the omitted $\epsilon$ term (which appears in the exact algebraic manipulation of $\alpha_{k}$) yields the bound stated in Theorem \ref{thm:convergence}:
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\left[\|\nabla f(w_{k})\|_{D_{k}^{-1}}^{2}\right]
\le
\frac{2\big(f(w_{1})-f^{\star}\big)}{\eta\sqrt{T}}
+\frac{\eta L G^{2}}{\sqrt{T}}
+\frac{L\epsilon}{\eta\sqrt{T}}.
\tag{S17}
\]

\subsection*{Step 16 – From weighted to unweighted norm}
Since $D_{k}^{-1}\preceq \frac{1}{\sqrt{\epsilon}}I$, we have
\[
\|\nabla f(w_{k})\|^{2}
\le \frac{1}{\sqrt{\epsilon}}\|\nabla f(w_{k})\|_{D_{k}^{-1}}^{2}.
\]
Thus the same $O(T^{-1/2})$ rate holds for the ordinary gradient norm, completing the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (S17) we can explicitly write the constants:
\[
C_{1}=2\big(f(w_{1})-f^{\star}\big),\qquad
C_{2}=L G^{2},\qquad
C_{3}=L\epsilon.
\]
Hence,
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le \frac{C_{1}}{\eta\sqrt{T}}+\frac{C_{2}\eta}{\sqrt{T}}+\frac{C_{3}}{\eta\sqrt{T}}.
\]
Optimizing over $\eta$ gives $\eta^{\star}= \sqrt{C_{1}/C_{2}}$, leading to the simplified bound
\[
\frac{1}{T}\sum_{k=1}^{T}\E\!\big[\|\nabla f(w_{k})\|^{2}\big]
\le 2\sqrt{C_{1}C_{2}}\;T^{-1/2}+ \frac{C_{3}}{\sqrt{C_{1}C_{2}}}\;T^{-1/2}
=O\!\big(T^{-1/2}\big).
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Deterministic gradients ($\sigma=0$).} The bound reduces to
    $\displaystyle \frac{2(f(w_{1})-f^{\star})}{\eta\sqrt{T}}+\frac{\eta L G^{2}}{\sqrt{T}}+\frac{L\epsilon}{\eta\sqrt{T}}$,
    which can be made arbitrarily small by choosing $\eta\asymp T^{-1/4}$.
    \item \textbf{Sparse gradients.} If for a coordinate $j$ only $s\ll T$ gradients are non‑zero, then $S_{k,j}=O(s)$ and the stepsize in that coordinate stays of order $\eta/\sqrt{s}$, i.e. much larger than the $O(\eta/\sqrt{T})$ stepsize of vanilla SGD. This yields faster empirical progress while the theoretical bound remains $O(T^{-1/2})$.
\end{itemize}
