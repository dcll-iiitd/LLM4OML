\section*{Optimistic Gradient Descent Ascent (OGDA)}

\section*{Mathematical Formulation}
Let $L:\R^{n}\times\R^{m}\rightarrow\R$ be a continuously differentiable function that is convex in its first argument $x\in\R^{n}$ and concave in its second argument $y\in\R^{m}$.  
Define the monotone operator
\[
F(z)\;:=\;\begin{pmatrix}
\grad_{x}L(x,y)\\[2mm]
-\grad_{y}L(x,y)
\end{pmatrix},
\qquad 
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\R^{n+m}.
\]

The OGDA iteration with stepsize $\eta>0$ is
\begin{align}
z_{t+1}
&=z_{t}-2\eta F(z_{t})+\eta F(z_{t-1}),\qquad t\ge 0,
\label{eq:ogda}
\end{align}
with an auxiliary initialization $F(z_{-1})\equiv 0$.  In component form,
\begin{aligned}
x_{t+1}&=x_{t}-2\eta\,\grad_{x}L(x_{t},y_{t})+\eta\,\grad_{x}L(x_{t-1},y_{t-1}),\\
y_{t+1}&=y_{t}+2\eta\,\grad_{y}L(x_{t},y_{t})-\eta\,\grad_{y}L(x_{t-1},y_{t-1}).
\end{aligned}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent Ascent (OGDA)}
\begin{algorithmic}[1]
\Require Initial point $z_{0}=(x_{0},y_{0})$, stepsize $\eta>0$, maximum iteration $T$.
\State $F(z_{-1})\gets 0$ \Comment{auxiliary gradient}
\For{$t=0$ \textbf{to} $T-1$}
    \State Compute $F(z_{t})=\bigl(\grad_{x}L(x_{t},y_{t}),\,-\grad_{y}L(x_{t},y_{t})\bigr)$
    \State $z_{t+1}\gets z_{t}-2\eta\,F(z_{t})+\eta\,F(z_{t-1})$   \Comment{Eq.~\eqref{eq:ogda}}
\EndFor
\State \Return $\{z_{t}\}_{t=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^{2}$ (i.e. a degenerate min–max problem with only a primal variable).  
Here $F(w) = \grad f(w)=2(w-3)$.  
We set $w_{0}=0$, stepsize $\eta=0.1$, and initialise $F(w_{-1})=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=F(w_{t})$ & $w_{t+1}=w_{t}-2\eta g_{t}+\eta g_{t-1}$ & $f(w_{t+1})$\\ \hline
0 & $g_{0}=2(0-3)=-6$ & $w_{1}=0-2(0.1)(-6)+0=1.2$ & $(1.2-3)^{2}=3.24$\\[2mm]
1 & $g_{1}=2(1.2-3)=-3.6$ & $w_{2}=1.2-2(0.1)(-3.6)+0.1(-6)=1.68$ & $(1.68-3)^{2}=1.7424$\\[2mm]
2 & $g_{2}=2(1.68-3)=-2.64$ & $w_{3}=1.68-2(0.1)(-2.64)+0.1(-3.6)=1.992$ & $(1.992-3)^{2}=1.016064$\\
\end{tabular}
\end{center}
The iterates move steadily toward the minimiser $w^{*}=3$, illustrating the optimistic correction term $+\eta g_{t-1}$.

\section*{Assumptions}
\begin{assumption}[Convex–concave and smoothness]\label{ass:smooth}
The function $L(x,y)$ satisfies:
\begin{enumerate}
\item For every fixed $y$, $x\mapsto L(x,y)$ is convex; for every fixed $x$, $y\mapsto L(x,y)$ is concave.
\item $L$ is $L$–smooth, i.e.
\[
\norm{F(z)-F(z')}\le L\,\norm{z-z'}\qquad\forall\,z,z'\in\R^{n+m}.
\]
\end{enumerate}
\end{assumption}

\begin{assumption}[Strong convexity–strong concavity (optional)]\label{ass:strong}
There exists $\mu>0$ such that for all $z,z'$,
\[
\inner{F(z)-F(z')}{z-z'}\;\ge\;\mu\,\norm{z-z'}^{2}.
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:step}
The stepsize satisfies 
\[
0<\eta\le \frac{1}{4L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA convergence for convex–concave $L$]\label{thm:conv}
Let Assumptions \ref{ass:smooth} and \ref{ass:step} hold and let $z^{*}=(x^{*},y^{*})$ be any saddle point of $L$, i.e. $F(z^{*})=0$.  
Define the Lyapunov potential
\[
\Phi_{t}:=\norm{z_{t}-z^{*}}^{2}+\norm{z_{t}-z_{t-1}}^{2}.
\]
Then for every $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\;\le\;
\frac{4}{\eta\,T}\,\Phi_{0}.
\tag{1}
\]
Consequently the averaged iterate $\bar z_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}z_{t}$ satisfies
\[
\max_{y}\,L(\bar x_{T},y)-\min_{x}\,L(x,\bar y_{T})\;\le\;
\frac{2L}{T}\,\Phi_{0}.
\]
\end{theorem}

\begin{theorem}[Linear rate under strong monotonicity]\label{thm:linear}
If, in addition, Assumption \ref{ass:strong} holds, then for the same stepsize range,
\[
\norm{z_{t}-z^{*}}^{2}\;\le\;\bigl(1-2\eta\mu\bigr)^{t}\,\norm{z_{0}-z^{*}}^{2}.
\tag{2}
\]
Thus OGDA converges linearly to the unique saddle point.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The potential $\Phi_{t}$ is non‑increasing as long as $\eta\le1/(4L)$; this guarantees bounded iterates.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{eq:ogda} shows the rate constant scales as $1/(\eta)$, while the admissible stepsize range shrinks linearly with the Lipschitz constant $L$.
\item \textbf{Comparison.} Standard Gradient Descent Ascent (GDA) enjoys an $O(1/\sqrt{T})$ rate for the same class of problems, whereas OGDA improves to $O(1/T)$ (Theorem~\ref{thm:conv}) and even linear convergence under strong convexity–concavity (Theorem~\ref{thm:linear}).
\end{itemize}

\section*{Discussion}
The optimistic correction term $+\eta F(z_{t-1})$ can be interpreted as a first‑order prediction of the next gradient, turning the method into a single‑loop analogue of the extragradient scheme.  The analysis above follows the classic potential‑function technique for monotone operators and highlights that the extra term exactly cancels the destabilising component present in vanilla GDA.  Consequently, OGDA attains the optimal $O(1/T)$ rate for smooth convex–concave saddle‑point problems without requiring an inner loop.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Lipschitz continuity of $F$]\label{lem:lipschitz}
Under Assumption~\ref{ass:smooth}, for any $z,z'$,
\[
\norm{F(z)-F(z')}\le L\,\norm{z-z'}.
\]
\end{lemma}
\begin{proof}
Directly from the $L$‑smoothness of $L$ applied to both coordinates. \qedhere
\end{proof}

\begin{lemma}[Monotonicity]\label{lem:mono}
For a convex–concave $L$, the operator $F$ is monotone:
\[
\inner{F(z)-F(z')}{z-z'}\ge0,\qquad\forall z,z'.
\]
\end{lemma}
\begin{proof}
Monotonicity follows from convexity in $x$ and concavity in $y$ (see e.g. \cite{rockafellar1970}). \qedhere
\end{proof}

\begin{lemma}[Co‑coercivity of Lipschitz monotone operators]\label{lem:coercive}
If $F$ is $L$‑Lipschitz and monotone, then
\[
\inner{F(z)-F(z')}{z-z'}\;\ge\;\frac{1}{L}\norm{F(z)-F(z')}^{2}.
\]
\end{lemma}
\begin{proof}
Combine Lemma~\ref{lem:lipschitz} with monotonicity and apply the Cauchy–Schwarz inequality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}.  All non‑trivial steps are numbered.

\begin{proof}
Let $z^{*}$ be a saddle point, i.e. $F(z^{*})=0$.  Define the error vectors
\[
e_{t}:=z_{t}-z^{*},\qquad d_{t}:=z_{t}-z_{t-1}\;(t\ge0,\;z_{-1}=z_{0}).
\]

\noindent\textbf{[Step 1]}  Expand $\|e_{t+1}\|^{2}$ using the OGDA update \eqref{eq:ogda}:
\begin{align}
\|e_{t+1}\|^{2}
&=\norm{e_{t}-2\eta F(z_{t})+\eta F(z_{t-1})}^{2}\nonumber\\
&=\norm{e_{t}}^{2}
-4\eta\inner{F(z_{t})}{e_{t}}
+2\eta\inner{F(z_{t-1})}{e_{t}}
+4\eta^{2}\norm{F(z_{t})}^{2}
-4\eta^{2}\inner{F(z_{t})}{F(z_{t-1})}
+\eta^{2}\norm{F(z_{t-1})}^{2}.
\label{eq:exp1}
\end{align}

\noindent\textbf{[Step 2]}  Add $\|d_{t+1}\|^{2}$ to both sides, where $d_{t+1}=z_{t+1}-z_{t}= -2\eta F(z_{t})+\eta F(z_{t-1})$:
\begin{align}
\|d_{t+1}\|^{2}
&= \norm{-2\eta F(z_{t})+\eta F(z_{t-1})}^{2}\nonumber\\
&=4\eta^{2}\norm{F(z_{t})}^{2}
-4\eta^{2}\inner{F(z_{t})}{F(z_{t-1})}
+\eta^{2}\norm{F(z_{t-1})}^{2}.
\label{eq:exp2}
\end{align}
Summing \eqref{eq:exp1} and \eqref{eq:exp2} yields the Lyapunov increment
\begin{align}
\Phi_{t+1}
&:=\|e_{t+1}\|^{2}+\|d_{t+1}\|^{2}\nonumber\\
&=\|e_{t}\|^{2}+\|d_{t}\|^{2}
-4\eta\inner{F(z_{t})}{e_{t}}
+2\eta\inner{F(z_{t-1})}{e_{t}}
+8\eta^{2}\norm{F(z_{t})}^{2}
-8\eta^{2}\inner{F(z_{t})}{F(z_{t-1})}
+2\eta^{2}\norm{F(z_{t-1})}^{2}.
\label{eq:phi_inc}
\end{align}

\noindent\textbf{[Step 3]}  Use monotonicity (Lemma~\ref{lem:mono}) with $z^{*}$ to replace the inner product term:
\[
\inner{F(z_{t})}{e_{t}}=\inner{F(z_{t})-F(z^{*})}{z_{t}-z^{*}}\ge0.
\]
Hence $-4\eta\inner{F(z_{t})}{e_{t}}\le0$ and can be dropped for an upper bound.

\noindent\textbf{[Step 4]}  Bound the mixed term $2\eta\inner{F(z_{t-1})}{e_{t}}$ by Cauchy–Schwarz and Young’s inequality:
\[
2\eta\inner{F(z_{t-1})}{e_{t}}
\le 2\eta\norm{F(z_{t-1})}\norm{e_{t}}
\le \eta\bigl(\norm{e_{t}}^{2}+\norm{F(z_{t-1})}^{2}\bigr).
\tag{3}
\]

\noindent\textbf{[Step 5]}  Apply Lemma~\ref{lem:coercive} to the term $-8\eta^{2}\inner{F(z_{t})}{F(z_{t-1})}$:
\[
-8\eta^{2}\inner{F(z_{t})}{F(z_{t-1})}
\le -8\eta^{2}\Bigl(\frac{1}{2}\norm{F(z_{t})}^{2}+\frac{1}{2}\norm{F(z_{t-1})}^{2}
-\frac{1}{2L}\norm{F(z_{t})-F(z_{t-1})}^{2}\Bigr)
\]
\[
= -4\eta^{2}\norm{F(z_{t})}^{2}
-4\eta^{2}\norm{F(z_{t-1})}^{2}
+\frac{4\eta^{2}}{L}\norm{F(z_{t})-F(z_{t-1})}^{2}.
\tag{4}
\]

\noindent\textbf{[Step 6]}  Substitute (3) and (4) into \eqref{eq:phi_inc} and collect like terms:
\begin{align}
\Phi_{t+1}
&\le \bigl(1+\eta\bigr)\norm{e_{t}}^{2}+\norm{d_{t}}^{2}
+\Bigl(8\eta^{2}-4\eta^{2}\Bigr)\norm{F(z_{t})}^{2}
+\Bigl(2\eta^{2}-4\eta^{2}\Bigr)\norm{F(z_{t-1})}^{2}
+\frac{4\eta^{2}}{L}\norm{F(z_{t})-F(z_{t-1})}^{2}\nonumber\\
&= \norm{e_{t}}^{2}+\norm{d_{t}}^{2}
+\eta\norm{e_{t}}^{2}
+4\eta^{2}\norm{F(z_{t})}^{2}
-2\eta^{2}\norm{F(z_{t-1})}^{2}
+\frac{4\eta^{2}}{L}\norm{F(z_{t})-F(z_{t-1})}^{2}.
\label{eq:phi_mid}
\end{align}

\noindent\textbf{[Step 7]}  Use the Lipschitz bound $\norm{F(z_{t})-F(z_{t-1})}\le L\norm{d_{t}}$ (Lemma~\ref{lem:lipschitz}) to handle the last term:
\[
\frac{4\eta^{2}}{L}\norm{F(z_{t})-F(z_{t-1})}^{2}
\le 4\eta^{2}L\,\norm{d_{t}}^{2}.
\]
Insert this into \eqref{eq:phi_mid}:
\begin{align}
\Phi_{t+1}
&\le \bigl(1+\eta\bigr)\norm{e_{t}}^{2}
+\bigl(1+4\eta^{2}L\bigr)\norm{d_{t}}^{2}
+4\eta^{2}\norm{F(z_{t})}^{2}
-2\eta^{2}\norm{F(z_{t-1})}^{2}.
\tag{5}
\end{align}

\noindent\textbf{[Step 8]}  Since $\Phi_{t}= \norm{e_{t}}^{2}+\norm{d_{t}}^{2}$, inequality (5) can be rewritten as
\[
\Phi_{t+1}\le \Phi_{t}
-\underbrace{\bigl(1-\eta\bigr)}_{\ge 0}\norm{e_{t}}^{2}
-\underbrace{\bigl(1-4\eta^{2}L\bigr)}_{\ge0}\norm{d_{t}}^{2}
+4\eta^{2}\norm{F(z_{t})}^{2}
-2\eta^{2}\norm{F(z_{t-1})}^{2}.
\]
Using the stepsize condition $\eta\le 1/(4L)$ we have $1-4\eta^{2}L\ge0$.  Dropping the non‑positive terms $-\norm{e_{t}}^{2}$ and $-\norm{d_{t}}^{2}$ yields a simpler descent inequality
\[
\Phi_{t+1}\le \Phi_{t}+4\eta^{2}\norm{F(z_{t})}^{2}-2\eta^{2}\norm{F(z_{t-1})}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 9]}  Rearranging (6) gives
\[
4\eta^{2}\norm{F(z_{t})}^{2}\le \Phi_{t}-\Phi_{t+1}+2\eta^{2}\norm{F(z_{t-1})}^{2}.
\]
Summing from $t=0$ to $T-1$ telescopes the potential terms:
\[
4\eta^{2}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\le \Phi_{0}-\Phi_{T}+2\eta^{2}\sum_{t=0}^{T-2}\norm{F(z_{t})}^{2}.
\]
Move the second sum to the left:
\[
2\eta^{2}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\le \Phi_{0},
\qquad\text{hence}\qquad
\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\le \frac{\Phi_{0}}{2\eta^{2}}.
\]
Dividing by $T$ and recalling $\Phi_{0}=\norm{z_{0}-z^{*}}^{2}+\norm{z_{0}-z_{-1}}^{2}\le 2\norm{z_{0}-z^{*}}^{2}$ gives the claimed bound \eqref{1}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\le \frac{4}{\eta T}\,\Phi_{0}.
\qedhere
\]
\end{proof}

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:conv}, set $\Phi_{0}= \norm{z_{0}-z^{*}}^{2}+\norm{z_{0}-z_{-1}}^{2}\le 2R^{2}$ where $R$ bounds the initial distance.  Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}\le \frac{8R^{2}}{\eta T}.
\]
Using $L$‑smoothness,
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\;\le\; \frac{L}{T}\sum_{t=0}^{T-1}\norm{z_{t}-z_{t+1}}^{2}
\le \frac{2L}{T}\Phi_{0}
\le \frac{4LR^{2}}{T}.
\]
Thus OGDA attains the optimal $O(1/T)$ decay.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex–strongly concave $L$.}  Under Assumption~\ref{ass:strong} the monotone operator $F$ is $\mu$‑strongly monotone.  Re‑applying the potential analysis with the stronger inequality
\[
\inner{F(z_{t})}{e_{t}}\ge \mu\norm{e_{t}}^{2}
\]
yields
\[
\Phi_{t+1}\le (1-2\eta\mu)\Phi_{t},
\]
which directly implies the linear rate \eqref{2} (Theorem~\ref{thm:linear}).
\item \textbf{Zero‑sum bilinear games.}  For $L(x,y)=x^{\top}Ay$ with $A$ symmetric, $F$ is linear.  The OGDA iteration reduces to a linear recurrence whose spectral radius equals $|1-2\eta\lambda|$ for eigenvalues $\lambda$ of $A$.  Choosing $\eta<1/(2\max|\lambda|)$ guarantees contraction, matching the general theory.
\end{itemize}
