\section*{Online Gradient Descent}

\section*{Mathematical Formulation}
Let $\mathcal{W} \subseteq \mathbb{R}^d$ be a convex decision set. At each time step $t = 1, 2, \dots, T$, the algorithm chooses a point $w_t \in \mathcal{W}$. An adversarial convex function $f_t : \mathcal{W} \to \mathbb{R}$ is then revealed, and the algorithm incurs a loss $f_t(w_t)$. The update rule for Online Gradient Descent (OGD) with projection is:
\begin{align*}
w_{t+1} = \Pi_{\mathcal{W}} \left( w_t - \eta_t \nabla f_t(w_t) \right)
\end{align*}
where $\eta_t$ is the learning rate at time $t$, $\nabla f_t(w_t)$ is the gradient of the loss function at the current point, and $\Pi_{\mathcal{W}}(x) = \arg\min_{w \in \mathcal{W}} \|w - x\|_2$ is the Euclidean projection onto the set $\mathcal{W}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Online Gradient Descent (OGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} Convex set $\mathcal{W}$, learning rate schedule $\{\eta_t\}_{t=1}^{T-1}$, horizon $T$
\STATE Initialize $w_1 \in \mathcal{W}$ arbitrarily
\FOR{$t = 1, 2, \dots, T$}
\STATE Predict $w_t$
\STATE Observe loss function $f_t$ and incur loss $f_t(w_t)$
\STATE Compute gradient $g_t = \nabla f_t(w_t)$
\STATE Update $w_{t+1} = \Pi_{\mathcal{W}} \left( w_t - \eta_t g_t \right)$
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let $f(w) = (w - 3)^2$. The gradient is $\nabla f(w) = 2(w - 3)$. 
We set $\mathcal{W} = \mathbb{R}$ (no projection needed), $w_0 = 0$, and a fixed learning rate $\eta = 0.1$. We perform 3 iterations. Note that in OGD, the iterates are $w_1, w_2, w_3$ and updates happen from $w_t$ to $w_{t+1}$. For this dry run, we align with the provided initial point $w_0=0$.

\textbf{Iteration 1:}
\begin{itemize}
\item Current parameter: $w_0 = 0$
\item Gradient computation: $g_0 = 2(w_0 - 3) = 2(0 - 3) = -6$
\item Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6$
\item Objective value: $f(w_1) = (0.6 - 3)^2 = (-2.4)^2 = 5.76$
\end{itemize}

\textbf{Iteration 2:}
\begin{itemize}
\item Current parameter: $w_1 = 0.6$
\item Gradient computation: $g_1 = 2(w_1 - 3) = 2(0.6 - 3) = -4.8$
\item Update: $w_2 = w_1 - \eta g_1 = 0.6 - 0.1(-4.8) = 0.6 + 0.48 = 1.08$
\item Objective value: $f(w_2) = (1.08 - 3)^2 = (-1.92)^2 = 3.6864$
\end{itemize}

\textbf{Iteration 3:}
\begin{itemize}
\item Current parameter: $w_2 = 1.08$
\item Gradient computation: $g_2 = 2(w_2 - 3) = 2(1.08 - 3) = -3.84$
\item Update: $w_3 = w_2 - \eta g_2 = 1.08 - 0.1(-3.84) = 1.08 + 0.384 = 1.464$
\item Objective value: $f(w_3) = (1.464 - 3)^2 = (-1.536)^2 = 2.3593$
\end{itemize}

\section*{Assumptions}
\begin{definition}[Convexity]
For all $t = 1, \dots, T$, the online functions $f_t : \mathcal{W} \to \mathbb{R}$ are convex. That is, for all $w, u \in \mathcal{W}$ and $\lambda \in [0, 1]$,
$f_t(\lambda w + (1-\lambda)u) \leq \lambda f_t(w) + (1-\lambda) f_t(u)$.
\end{definition}

\begin{definition}[Lipschitz Continuity]
For all $t = 1, \dots, T$, the online functions $f_t$ are $G$-Lipschitz continuous with respect to the $L_2$ norm. That is, there exists a constant $G > 0$ such that for all $w \in \mathcal{W}$,
$|f_t(w) - f_t(u)| \leq G \|w - u\|_2$.
This implies that the subgradients (or gradients) are bounded: $\|\nabla f_t(w)\|_2 \leq G$ for all $w \in \mathcal{W}$.
\end{definition}

\begin{definition}[Bounded Decision Set]
The decision set $\mathcal{W} \subseteq \mathbb{R}^d$ is convex and bounded, such that its diameter is bounded by $D > 0$:
$\max_{w, u \in \mathcal{W}} \|w - u\|_2 \leq D$.
\end{definition}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear Regret of OGD]
Let $\mathcal{W} \subseteq \mathbb{R}^d$ be a convex set with diameter $D$. Let $\{f_t\}_{t=1}^T$ be a sequence of convex and $G$-Lipschitz continuous online functions. If Online Gradient Descent is run with a fixed learning rate $\eta = \frac{D}{G\sqrt{T}}$, then the cumulative regret satisfies:
\begin{align*}
R_T = \sum_{t=1}^T f_t(w_t) - \min_{w \in \mathcal{W}} \sum_{t=1}^T f_t(w) \leq D G \sqrt{T}
\end{align*}
Consequently, the average regret vanishes: $\frac{R_T}{T} \leq \frac{DG}{\sqrt{T}} \to 0$ as $T \to \infty$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The projection step ensures that the iterates $w_t$ remain within the bounded set $\mathcal{W}$, preventing the parameters from diverging to infinity regardless of the adversarial loss sequence.
\item \textbf{Hyperparameter Sensitivity:} The optimal rate requires knowing the horizon $T$ in advance to set $\eta = \frac{D}{G\sqrt{T}}$. If $T$ is unknown, a doubling trick or a decreasing learning rate schedule $\eta_t = \frac{D}{G\sqrt{t}}$ can be employed to achieve the same $O(\sqrt{T})$ regret without foreknowledge of $T$.
\item \textbf{Comparison to Baselines:} Online Gradient Descent achieves the minimax optimal regret $O(\sqrt{T})$ for the class of convex and Lipschitz online functions. Simpler strategies like Follow The Leader (FTL) can suffer linear regret $O(T)$ on adversarial sequences, making OGD fundamentally more robust.
\end{itemize}

\section*{Discussion}
The $O(\sqrt{T})$ regret bound is the standard benchmark for online convex optimization with convex, Lipschitz functions. It establishes that the algorithm's average performance approaches that of the best fixed point in hindsight. The projection operation is essential to guarantee bounded iterates and control the cumulative error terms arising from the gradient steps.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Projection Lemma]
For any closed convex set $\mathcal{W} \subseteq \mathbb{R}^d$ and any point $x \in \mathbb{R}^d$, the Euclidean projection $\Pi_{\mathcal{W}}(x)$ satisfies:
\begin{align*}
\|\Pi_{\mathcal{W}}(x) - w\|_2^2 \leq \|x - w\|_2^2 - \|\Pi_{\mathcal{W}}(x) - x\|_2^2 \quad \forall w \in \mathcal{W}
\end{align*}
\end{lemma}
\begin{proof}
By the optimality condition of the projection, for $y = \Pi_{\mathcal{W}}(x)$, we have $\langle x - y, w - y \rangle \leq 0$ for all $w \in \mathcal{W}$. Expanding the squared norm:
\begin{align*}
\|x - w\|_2^2 &= \|(x - y) + (y - w)\|_2^2 \\
&= \|x - y\|_2^2 + \|y - w\|_2^2 + 2 \langle x - y, y - w \rangle
\end{align*}
Since $\langle x - y, y - w \rangle = -\langle x - y, w - y \rangle \geq 0$, we obtain $\|x - w\|_2^2 \geq \|x - y\|_2^2 + \|y - w\|_2^2$, which rearranges to the stated inequality.
\end{proof}

\begin{lemma}[Convexity and Subgradient Inequality]
For any convex function $f_t$ and any $w, u \in \mathcal{W}$:
\begin{align*}
f_t(w) - f_t(u) \leq \langle \nabla f_t(w), w - u \rangle
\end{align*}
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}
Let $w^* = \arg\min_{w \in \mathcal{W}} \sum_{t=1}^T f_t(w)$ be the best fixed decision in hindsight. We will bound the instantaneous regret $f_t(w_t) - f_t(w^*)$.

\textbf{[Step 1]} Apply the subgradient inequality (Lemma 2) to bound the instantaneous regret:
\begin{align*}
f_t(w_t) - f_t(w^*) \leq \langle \nabla f_t(w_t), w_t - w^* \rangle
\end{align*}

\textbf{[Step 2]} Define the pre-projection update $y_{t+1} = w_t - \eta_t \nabla f_t(w_t)$. Note that $w_{t+1} = \Pi_{\mathcal{W}}(y_{t+1})$. Apply the Projection Lemma (Lemma 1) with $x = y_{t+1}$ and $w = w^*$:
\begin{align*}
\|w_{t+1} - w^*\|_2^2 \leq \|y_{t+1} - w^*\|_2^2 - \|w_{t+1} - y_{t+1}\|_2^2
\end{align*}

\textbf{[Step 3]} Substitute $y_{t+1} = w_t - \eta_t \nabla f_t(w_t)$ into the first term on the right-hand side and expand the squared norm:
\begin{align*}
\|y_{t+1} - w^*\|_2^2 &= \|(w_t - \eta_t \nabla f_t(w_t)) - w^*\|_2^2 \\
&= \|(w_t - w^*) - \eta_t \nabla f_t(w_t)\|_2^2 \\
&= \|w_t - w^*\|_2^2 - 2\eta_t \langle \nabla f_t(w_t), w_t - w^* \rangle + \eta_t^2 \|\nabla f_t(w_t)\|_2^2
\end{align*}

\textbf{[Step 4]} Substitute the expanded expression from Step 3 back into the inequality from Step 2:
\begin{align*}
\|w_{t+1} - w^*\|_2^2 \leq \|w_t - w^*\|_2^2 - 2\eta_t \langle \nabla f_t(w_t), w_t - w^* \rangle + \eta_t^2 \|\nabla f_t(w_t)\|_2^2 - \|w_{t+1} - y_{t+1}\|_2^2
\end{align*}

\textbf{[Step 5]} Rearrange the inequality to isolate the inner product term $\langle \nabla f_t(w_t), w_t - w^* \rangle$:
\begin{align*}
2\eta_t \langle \nabla f_t(w_t), w_t - w^* \rangle \leq \|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2 + \eta_t^2 \|\nabla f_t(w_t)\|_2^2 - \|w_{t+1} - y_{t+1}\|_2^2
\end{align*}

\textbf{[Step 6]} Divide both sides by $2\eta_t$ (since $\eta_t > 0$, the inequality direction is preserved):
\begin{align*}
\langle \nabla f_t(w_t), w_t - w^* \rangle \leq \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} + \frac{\eta_t}{2} \|\nabla f_t(w_t)\|_2^2 - \frac{\|w_{t+1} - y_{t+1}\|_2^2}{2\eta_t}
\end{align*}

\textbf{[Step 7]} Apply the Lipschitz assumption (Assumption 2), which guarantees $\|\nabla f_t(w_t)\|_2 \leq G$. We relax the inequality by dropping the non-negative projection distance term $-\frac{\|w_{t+1} - y_{t+1}\|_2^2}{2\eta_t} \leq 0$:
\begin{align*}
\langle \nabla f_t(w_t), w_t - w^* \rangle \leq \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} + \frac{\eta_t G^2}{2}
\end{align*}

\textbf{[Step 8]} Combine Step 1 with Step 7 to bound the instantaneous regret:
\begin{align*}
f_t(w_t) - f_t(w^*) \leq \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} + \frac{\eta_t G^2}{2}
\end{align*}

\textbf{[Step 9]} Sum the instantaneous regret bound from $t=1$ to $T$:
\begin{align*}
\sum_{t=1}^T \left( f_t(w_t) - f_t(w^*) \right) \leq \sum_{t=1}^T \left( \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} \right) + \sum_{t=1}^T \frac{\eta_t G^2}{2}
\end{align*}

\textbf{[Step 10]} For a fixed learning rate $\eta_t = \eta$, the second sum evaluates to $\frac{\eta G^2 T}{2}$. For the first sum, we unpack the telescoping series carefully:
\begin{align*}
\sum_{t=1}^T \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta} &= \frac{1}{2\eta} \left( \|w_1 - w^*\|_2^2 - \|w_2 - w^*\|_2^2 + \|w_2 - w^*\|_2^2 - \|w_3 - w^*\|_2^2 + \dots + \|w_T - w^*\|_2^2 - \|w_{T+1} - w^*\|_2^2 \right) \\
&= \frac{1}{2\eta} \left( \|w_1 - w^*\|_2^2 - \|w_{T+1} - w^*\|_2^2 \right)
\end{align*}

\textbf{[Step 11]} Apply the bounded domain assumption (Assumption 3). Since $w_1, w^* \in \mathcal{W}$, we have $\|w_1 - w^*\|_2^2 \leq D^2$. Since $w_{T+1}, w^* \in \mathcal{W}$, we have $\|w_{T+1} - w^*\|_2^2 \geq 0$. Thus:
\begin{align*}
\frac{1}{2\eta} \left( \|w_1 - w^*\|_2^2 - \|w_{T+1} - w^*\|_2^2 \right) \leq \frac{D^2}{2\eta}
\end{align*}

\textbf{[Step 12]} Substitute the result from Step 11 back into the cumulative sum from Step 9:
\begin{align*}
R_T = \sum_{t=1}^T f_t(w_t) - \sum_{t=1}^T f_t(w^*) \leq \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}
\end{align*}

\section*{Rate Derivation (With Constants)}
To find the optimal learning rate $\eta$ that minimizes the upper bound on the regret, we treat the bound $\frac{D^2}{2\eta} + \frac{\eta G^2 T}{2}$ as a function of $\eta > 0$.

\textbf{[Step 13]} Take the derivative of the regret bound with respect to $\eta$ and set it to zero:
\begin{align*}
\frac{d}{d\eta} \left( \frac{D^2}{2\eta} + \frac{\eta G^2 T}{2} \right) = -\frac{D^2}{2\eta^2} + \frac{G^2 T}{2} = 0
\end{align*}

\textbf{[Step 14]} Solve for the optimal learning rate $\eta^*$:
\begin{align*}
\frac{D^2}{2\eta^2} &= \frac{G^2 T}{2} \\
D^2 &= \eta^2 G^2 T \\
\eta^2 &= \frac{D^2}{G^2 T} \\
\eta^* &= \frac{D}{G\sqrt{T}}
\end{align*}

\textbf{[Step 15]} Substitute the optimal learning rate $\eta^*$ back into the regret bound derived in Step 12:
\begin{align*}
R_T &\leq \frac{D^2}{2 \left( \frac{D}{G\sqrt{T}} \right)} + \frac{\left( \frac{D}{G\sqrt{T}} \right) G^2 T}{2} \\
&= \frac{D^2 G \sqrt{T}}{2 D} + \frac{D G^2 T}{2 G \sqrt{T}} \\
&= \frac{D G \sqrt{T}}{2} + \frac{D G \sqrt{T}}{2} \\
&= D G \sqrt{T}
\end{align*}
This yields the final sublinear regret bound $R_T \leq D G \sqrt{T}$, confirming that the average regret $\frac{R_T}{T} \leq \frac{DG}{\sqrt{T}}$ vanishes as $T \to \infty$.

\section*{Special Cases}
\textbf{Case 1: Unknown Horizon $T$.} If the total number of iterations $T$ is not known in advance, we cannot set the fixed optimal $\eta$. Instead, we use a decreasing learning rate schedule $\eta_t = \frac{D}{G\sqrt{t}}$. 

\textbf{[Step 16]} Returning to the bound in Step 9 with varying $\eta_t$, the first sum no longer telescopes cleanly. We write it as:
\begin{align*}
\sum_{t=1}^T \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} = \frac{\|w_1 - w^*\|_2^2}{2\eta_1} + \sum_{t=2}^T \|w_t - w^*\|_2^2 \left( \frac{1}{2\eta_t} - \frac{1}{2\eta_{t-1}} \right) - \frac{\|w_{T+1} - w^*\|_2^2}{2\eta_T}
\end{align*}

\textbf{[Step 17]} Since $\eta_t = \frac{D}{G\sqrt{t}}$ is a decreasing sequence, $\frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \geq 0$. Because $\|w_t - w^*\|_2^2 \leq D^2$, we can upper bound the sum by replacing $\|w_t - w^*\|_2^2$ with $D^2$:
\begin{align*}
\sum_{t=1}^T \frac{\|w_t - w^*\|_2^2 - \|w_{t+1} - w^*\|_2^2}{2\eta_t} &\leq D^2 \left( \frac{1}{2\eta_1} + \sum_{t=2}^T \left( \frac{1}{2\eta_t} - \frac{1}{2\eta_{t-1}} \right) \right) \\
&= D^2 \left( \frac{1}{2\eta_1} + \frac{1}{2\eta_2} - \frac{1}{2\eta_1} + \dots + \frac{1}{2\eta_T} - \frac{1}{2\eta_{T-1}} \right) \\
&= \frac{D^2}{2\eta_T} = \frac{D^2}{2 \left(\frac{D}{G\sqrt{T}}\right)} = \frac{DG\sqrt{T}}{2}
\end{align*}

\textbf{[Step 18]} For the second sum with varying $\eta_t = \frac{D}{G\sqrt{t}}$, we evaluate:
\begin{align*}
\sum_{t=1}^T \frac{\eta_t G^2}{2} = \sum_{t=1}^T \frac{D G^2}{2 G \sqrt{t}} = \frac{DG}{2} \sum_{t=1}^T \frac{1}{\sqrt{t}}
\end{align*}

\textbf{[Step 19]} We bound the harmonic sum using the integral test:
\begin{align*}
\sum_{t=1}^T \frac{1}{\sqrt{t}} \leq 1 + \int_1^T \frac{1}{\sqrt{x}} dx = 1 + \left[ 2\sqrt{x} \right]_1^T = 1 + 2\sqrt{T} - 2 = 2\sqrt{T} - 1 \leq 2\sqrt{T}
\end{align*}

\textbf{[Step 20]} Combine the results from Step 17 and Step 19:
\begin{align*}
R_T \leq \frac{DG\sqrt{T}}{2} + \frac{DG}{2} (2\sqrt{T}) = \frac{DG\sqrt{T}}{2} + DG\sqrt{T} = \frac{3}{2} DG\sqrt{T}
\end{align*}
Thus, an adaptive learning rate $\eta_t = \frac{D}{G\sqrt{t}}$ still guarantees sublinear regret $O(\sqrt{T})$, with only a constant factor of $1.5$ degradation compared to the optimal offline setting.
