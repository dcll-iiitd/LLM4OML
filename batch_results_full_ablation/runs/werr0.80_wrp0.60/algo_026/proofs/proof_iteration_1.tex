\section*{Randomized Coordinate Descent with Non‑Uniform Sampling (RCD‑NU)}

---------------------------------------------------------------------

\section*{Algorithm Details Expansion}

\subsection*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be a differentiable convex function.  
For each coordinate $i\in\{1,\dots ,d\}$ let $L_i>0$ be a coordinate‑wise smoothness constant, i.e.

\[
\big| \partial_i f(x+ \alpha e_i)-\partial_i f(x) \big|\le L_i |\alpha|\qquad \forall x\in\R^{d},\;\forall \alpha\in\R .
\tag{A1}
\]

Fix a probability vector $p=(p_1,\dots ,p_d)$ with $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.  
At iteration $t=0,1,\dots$ we draw an index $i_t$ according to $p$ and perform an update only on that coordinate:

\[
\boxed{
w_{t+1}= w_{t}-\frac{1}{L_{i_t}}\; \partial_{i_t} f(w_t)\, e_{i_t}},
\tag{1}
\]

where $e_i$ denotes the $i$‑th standard basis vector.  
All other coordinates remain unchanged: $[w_{t+1}]_j=[w_t]_j$ for $j\neq i_t$.

\subsection*{Pseudocode}
\begin{enumerate}[leftmargin=*]
\item \textbf{Input:} initial point $w_0\in\R^{d}$, smoothness constants $\{L_i\}_{i=1}^{d}$, probability vector $p$,
      number of iterations $T$.
\item \textbf{for} $t=0$ \textbf{to} $T-1$ \textbf{do}
   \begin{enumerate}
   \item Sample $i_t\sim p$.
   \item Compute the partial derivative $g_{i_t}^{(t)}:=\partial_{i_t} f(w_t)$.
   \item Update the selected coordinate
   \[
   [w_{t+1}]_{i_t}= [w_t]_{i_t}-\frac{1}{L_{i_t}}\,g_{i_t}^{(t)} .
   \]
   \item For $j\neq i_t$ set $[w_{t+1}]_{j}= [w_t]_{j}$.
   \end{enumerate}
\item \textbf{Output:} $\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1}w_t$ (or the last iterate $w_T$).
\end{enumerate}

\subsection*{Dry Run Example}
Consider the one–dimensional problem $f(w)=(w-3)^2$, thus $d=1$, $L_1=2$ (since $f''(w)=2$), and we set $p_1=1$.  
Take $w_0=0$ and step size $\eta_1=1/L_1=0.5$ (equivalently $\eta=0.1$ would be a smaller step; we keep the exact $1/L_i$ for the theory).  

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{1}^{(t)}=\partial_1 f(w_t)$ & $w_{t+1}=w_t-\frac{1}{L_1}g_{1}^{(t)}$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.5(-6)=3$ & $(3-3)^2=0$\\
1 & $2(3-3)=0$   & $3-0.5(0)=3$                & $0$\\
2 & $0$          & $3$                         & $0$
\end{tabular}
\end{center}
After the first iteration the algorithm reaches the optimum $w^{*}=3$, and the objective value stays at $0$ thereafter.  This illustrates the coordinate update rule (1) in the simplest possible setting.

\section*{Convergence Theory Development}

\subsection*{Assumptions}
\begin{assumption}[Convexity]\label{as:convex}
$f$ is convex, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\ge f(x)+\ip{\grad f(x)}{y-x}.
\]
\end{assumption}

\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{as:smooth}
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $x\in\R^{d}$ and $\alpha\in\R$,
\[
\big|\partial_i f(x+\alpha e_i)-\partial_i f(x)\big|\le L_i|\alpha|.
\tag{2}
\]
Equivalently,
\[
f\big(x+\alpha e_i\big)\le f(x)+\alpha\,\partial_i f(x)+\frac{L_i}{2}\alpha^{2}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Sampling probabilities]\label{as:prob}
$p_i>0$ for all $i$ and $\sum_{i=1}^{d}p_i=1$.
\end{assumption}

\subsection*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of RCD‑NU]\label{thm:main}
Let $w^{*}\in\arg\min_{w\in\R^{d}}f(w)$.  Run the algorithm (1) for $T$ iterations with step sizes $\eta_i=1/L_i$.  Define the averaged iterate
\[
\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1}w_t .
\]
Then
\[
\E\!\big[\,f(\bar w_T)-f(w^{*})\,\big]\;\le\;
\frac{1}{2T}\Big(\sum_{i=1}^{d}\frac{L_i}{p_i}\Big)\,
\norm{w_{0}-w^{*}}^{2}.
\tag{4}
\]
Consequently the method attains an $O(1/T)$ convergence rate in expectation.
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because each coordinate step uses the exact Lipschitz constant $L_i$, the update (1) never increases the objective (Lemma~\ref{lem:descent}).
\item \textbf{Hyper‑parameter sensitivity.}  The only free parameters are the probabilities $p_i$.  Choosing $p_i\propto L_i^{-1/2}$ balances the term $\sum L_i/p_i$ and can improve the bound.
\item \textbf{Comparison to uniform sampling.}  If $p_i=1/d$, bound (4) reduces to $\frac{d}{2T}\big(\max_i L_i\big)\norm{w_0-w^{*}}^{2}$, which matches the classic uniform randomized coordinate descent bound.  Non‑uniform sampling can strictly improve the constant when the $L_i$ vary widely.
\end{itemize}

\section*{Complete Convergence Proof}

\subsection*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate descent lemma]\label{lem:descent}
Under Assumption~\ref{as:smooth}, for any $x\in\R^{d}$ and any coordinate $i$,
\[
f\!\big(x-\tfrac{1}{L_i}\partial_i f(x)\,e_i\big)\le 
f(x)-\frac{1}{2L_i}\big(\partial_i f(x)\big)^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Apply inequality (3) with $\alpha=-\partial_i f(x)/L_i$:
\[
\begin{aligned}
f\!\big(x-\tfrac{1}{L_i}\partial_i f(x)\,e_i\big)
&\le f(x)-\frac{1}{L_i}\big(\partial_i f(x)\big)^{2}
   +\frac{L_i}{2}\Big(\frac{\partial_i f(x)}{L_i}\Big)^{2}\\
&= f(x)-\frac{1}{2L_i}\big(\partial_i f(x)\big)^{2}.
\end{aligned}
\]
\end{proof}

\begin{lemma}[Weighted expected descent]\label{lem:expdescent}
Let $i_t$ be drawn from $p$.  Then for the iterate $w_t$,
\[
\E_{i_t}\!\big[\,f(w_{t+1})\mid w_t\,\big]
\le f(w_t)-\frac{1}{2}\sum_{i=1}^{d}\frac{p_i}{L_i}
\big(\partial_i f(w_t)\big)^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Condition on $w_t$ and use Lemma~\ref{lem:descent}:
\[
\begin{aligned}
\E_{i_t}\!\big[\,f(w_{t+1})\mid w_t\,\big]
&=\sum_{i=1}^{d}p_i\,
   f\!\big(w_t-\tfrac{1}{L_i}\partial_i f(w_t)\,e_i\big)\\
&\overset{[Step 1]}{\le}
\sum_{i=1}^{d}p_i\Big(
   f(w_t)-\frac{1}{2L_i}\big(\partial_i f(w_t)\big)^{2}\Big)\\
&= f(w_t)-\frac{1}{2}\sum_{i=1}^{d}\frac{p_i}{L_i}
   \big(\partial_i f(w_t)\big)^{2}.
\end{aligned}
\]
The inequality in \([Step 1]\) is exactly Lemma~\ref{lem:descent}.
\end{proof}

\subsection*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main}.  Throughout the proof we denote by
$\E_t[\cdot]=\E[\cdot\mid w_0,\dots ,w_t]$ the conditional expectation given the history up to iteration $t$.

\begin{proof}
\textbf{Step 1 (Potential decrease).}  
Consider the squared Euclidean distance to the optimum:
\[
V_t:=\norm{w_t-w^{*}}^{2}.
\]
Using the update (1) and the fact that only coordinate $i_t$ changes,
\[
\begin{aligned}
V_{t+1}
&=\norm{w_t-w^{*}-\tfrac{1}{L_{i_t}}\partial_{i_t} f(w_t)\,e_{i_t}}^{2}\\
&=V_t- \frac{2}{L_{i_t}}\partial_{i_t} f(w_t)\big([w_t]_{i_t}-[w^{*}]_{i_t}\big)
   +\frac{1}{L_{i_t}^{2}}\big(\partial_{i_t} f(w_t)\big)^{2}.
\end{aligned}
\tag{7}
\]

\textbf{Step 2 (Take expectation over $i_t$).}  
Conditioning on $w_t$ and using $\E_{i_t}[\,\cdot\,]=\sum_i p_i(\cdot)$,
\[
\begin{aligned}
\E_{i_t}[V_{t+1}\mid w_t]
&=V_t-\sum_{i=1}^{d}\frac{2p_i}{L_i}\partial_i f(w_t)
   \big([w_t]_i-[w^{*}]_i\big)
   +\sum_{i=1}^{d}\frac{p_i}{L_i^{2}}\big(\partial_i f(w_t)\big)^{2}.
\end{aligned}
\tag{8}
\]

\textbf{Step 3 (Convexity substitution).}  
By convexity (Assumption~\ref{as:convex}),
\[
f(w_t)-f(w^{*})\le \ip{\grad f(w_t)}{w_t-w^{*}}
   =\sum_{i=1}^{d}\partial_i f(w_t)\big([w_t]_i-[w^{*}]_i\big).
\tag{9}
\]
Insert (9) into (8):
\[
\begin{aligned}
\E_{i_t}[V_{t+1}\mid w_t]
&\le V_t- \frac{2}{\max_i L_i}\big(f(w_t)-f(w^{*})\big)
   +\sum_{i=1}^{d}\frac{p_i}{L_i^{2}}\big(\partial_i f(w_t)\big)^{2}.
\end{aligned}
\tag{10}
\]

\textbf{Step 4 (Bound the residual term).}  
Apply the elementary inequality $ab\le \frac{a^{2}}{2\lambda}+ \frac{\lambda b^{2}}{2}$ with
$\lambda=L_i/p_i$ and $a=\partial_i f(w_t)$, $b=[w_t]_i-[w^{*}]_i$:
\[
\frac{2p_i}{L_i}\partial_i f(w_t)\big([w_t]_i-[w^{*}]_i\big)
\ge
\frac{p_i}{L_i^{2}}\big(\partial_i f(w_t)\big)^{2}
-\frac{p_i}{L_i}\big([w_t]_i-[w^{*}]_i\big)^{2}.
\tag{11}
\]
Rearranging (11) gives
\[
\frac{p_i}{L_i^{2}}\big(\partial_i f(w_t)\big)^{2}
\le
\frac{2p_i}{L_i}\partial_i f(w_t)\big([w_t]_i-[w^{*}]_i\big)
   +\frac{p_i}{L_i}\big([w_t]_i-[w^{*}]_i\big)^{2}.
\tag{12}
\]

\textbf{Step 5 (Insert (12) into (8)).}  
Using (12) for each coordinate in the last term of (8) we obtain
\[
\begin{aligned}
\E_{i_t}[V_{t+1}\mid w_t]
&\le V_t- \sum_{i=1}^{d}\frac{p_i}{L_i}
   \big([w_t]_i-[w^{*}]_i\big)^{2}.
\end{aligned}
\tag{13}
\]
Thus the expected potential never increases and the decrease is at least the weighted squared distance term.

\textbf{Step 6 (Summation and telescoping).}  
Take total expectation and sum (13) over $t=0,\dots ,T-1$:
\[
\begin{aligned}
\E[V_T]
&\le V_0-\sum_{t=0}^{T-1}\sum_{i=1}^{d}\frac{p_i}{L_i}
   \E\!\big[\big([w_t]_i-[w^{*}]_i\big)^{2}\big].
\end{aligned}
\tag{14}
\]

\textbf{Step 7 (Relate distances to objective).}  
From Lemma~\ref{lem:expdescent} (6) and convexity we have for every $t$,
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid w_t\big]
&\le f(w_t)-\frac{1}{2}\sum_{i=1}^{d}\frac{p_i}{L_i}
   \big(\partial_i f(w_t)\big)^{2}.
\end{aligned}
\tag{15}
\]
Using the Cauchy–Schwarz inequality together with (9) yields
\[
\big(f(w_t)-f(w^{*})\big)^{2}
\le
\Big(\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\partial_i f(w_t)\big)^{2}\Big)
   \Big(\sum_{i=1}^{d}\frac{L_i}{p_i}
          \big([w_t]_i-[w^{*}]_i\big)^{2}\Big).
\tag{16}
\]

\textbf{Step 8 (Upper bound the gradient term).}  
Rearrange (16) to obtain
\[
\sum_{i=1}^{d}\frac{p_i}{L_i}\big(\partial_i f(w_t)\big)^{2}
\ge
\frac{\big(f(w_t)-f(w^{*})\big)^{2}}
     {\sum_{i=1}^{d}\frac{L_i}{p_i}\norm{w_t-w^{*}}^{2}}.
\tag{17}
\]

\textbf{Step 9 (Plug (17) into (15)).}  
Taking total expectation and using $\E[V_t]=\E\norm{w_t-w^{*}}^{2}$,
\[
\E\!\big[f(w_{t+1})\big]
\le \E\!\big[f(w_t)\big]
   -\frac{1}{2}
     \frac{\E\!\big[\big(f(w_t)-f(w^{*})\big)^{2}\big]}
          {\sum_{i=1}^{d}\frac{L_i}{p_i}\,\E[V_t]}.
\tag{18}
\]

\textbf{Step 10 (Define a scalar potential).}  
Let $A:=\sum_{i=1}^{d}\frac{L_i}{p_i}$ (a constant).  Since $V_t\ge 0$, we have $\E[V_t]\le \norm{w_0-w^{*}}^{2}$ for all $t$ by (13).  Hence from (18),
\[
\E\!\big[f(w_{t+1})\big]
\le \E\!\big[f(w_t)\big]
   -\frac{1}{2A\,\norm{w_0-w^{*}}^{2}}
       \E\!\big[\big(f(w_t)-f(w^{*})\big)^{2}\big].
\tag{19}
\]

\textbf{Step 11 (Recursive inequality).}  
Denote $\Delta_t:=\E\!\big[f(w_t)-f(w^{*})\big]$.  By Jensen,
$\E[(f(w_t)-f(w^{*}))^{2}]\ge \Delta_t^{2}$.  Thus (19) yields
\[
\Delta_{t+1}\le \Delta_t-\frac{\Delta_t^{2}}{2A\,\norm{w_0-w^{*}}^{2}}
\quad\Longrightarrow\quad
\frac{1}{\Delta_{t+1}}\ge \frac{1}{\Delta_t}+\frac{1}{2A\,\norm{w_0-w^{*}}^{2}}.
\tag{20}
\]

\textbf{Step 12 (Summation).}  
Iterating (20) from $t=0$ to $T-1$ gives
\[
\frac{1}{\Delta_T}\ge \frac{1}{\Delta_0}
   +\frac{T}{2A\,\norm{w_0-w^{*}}^{2}}
   \;\ge\; \frac{T}{2A\,\norm{w_0-w^{*}}^{2}},
\]
because $\Delta_0\le f(w_0)-f(w^{*})\le \frac{L_{\max}}{2}\norm{w_0-w^{*}}^{2}$ and the term $1/\Delta_0$ is non‑negative.  Inverting,
\[
\Delta_T\le \frac{2A\,\norm{w_0-w^{*}}^{2}}{T}.
\tag{21}
\]

\textbf{Step 13 (From the last iterate to the average iterate).}  
Convexity of $f$ implies
\[
f(\bar w_T)-f(w^{*})\le \frac{1}{T}\sum_{t=0}^{T-1}\big(f(w_t)-f(w^{*})\big)
   =\frac{1}{T}\sum_{t=0}^{T-1}\Delta_t
   \le \Delta_T,
\]
where the last inequality follows from the monotonicity of $\Delta_t$ (the sequence is non‑increasing by (15)).  Combining with (21) yields exactly the bound (4):
\[
\E\!\big[f(\bar w_T)-f(w^{*})\big]\le
\frac{A}{2T}\norm{w_0-w^{*}}^{2}
=\frac{1}{2T}\Big(\sum_{i=1}^{d}\frac{L_i}{p_i}\Big)
   \norm{w_0-w^{*}}^{2}.
\qquad\blacksquare
\]
\end{proof}

\subsection*{Rate Derivation (With Constants)}
The constant appearing in (4) is $C:=\frac{1}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}$.  If all $L_i$ are equal to $L$, then $C=\frac{L}{2}\sum_i\frac{1}{p_i}= \frac{L d}{2}$ for uniform sampling ($p_i=1/d$).  For a highly non‑uniform smoothness profile, choosing $p_i\propto L_i$ reduces $C$ to $\frac{1}{2}\sum_i \sqrt{L_i}$, which can be substantially smaller.

\subsection*{Special Cases}
\begin{enumerate}
\item \textbf{Uniform sampling.}  $p_i=1/d$ gives the classic $O(dL/T)$ bound.
\item \textbf{Importance sampling.}  Setting $p_i\propto L_i$ yields
\[
\sum_{i=1}^{d}\frac{L_i}{p_i}= \Big(\sum_{i=1}^{d}\sqrt{L_i}\Big)^{2},
\]
which is the tightest bound obtainable within this framework.
\item \textbf{Strongly convex $f$.}  If, in addition, $f$ is $\mu$‑strongly convex,
the same proof yields a linear rate
\[
\E\!\big[f(w_t)-f(w^{*})\big]\le
\big(1-\tfrac{\mu}{\sum_i L_i/p_i}\big)^{t}\big(f(w_0)-f(w^{*})\big).
\]
\end{enumerate}

\section*{Discussion}
The proof above follows the classic template for randomized coordinate descent but explicitly keeps the sampling probabilities $p_i$ inside every expectation.  The crucial step \([Step 1]\) in Lemma~\ref{lem:expdescent} shows how the factor $1/p_i$ naturally appears when taking expectations with respect to a non‑uniform distribution.  Consequently the convergence constant involves the weighted sum $\sum_i L_i/p_i$, which can be dramatically smaller than the uniform counterpart when the coordinates have heterogeneous smoothness.

The analysis requires only convexity and coordinate‑wise smoothness; no global Lipschitz gradient bound is needed.  Moreover, the algorithm is completely parameter‑free apart from the choice of $p$, making it attractive for large‑scale problems where per‑coordinate step sizes can be pre‑computed cheaply.
