\section*{Algorithm Name}
\textbf{Randomized Coordinate Descent with Non‑Uniform Sampling (RCD‑NU).}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be convex and coordinate‑wise $L_i$‑smooth, i.e.
\[
\bigl|\partial_i f(x+\alpha e_i)-\partial_i f(x)\bigr|\le L_i|\alpha|,
\qquad\forall x\in\mathbb{R}^d,\;\forall\alpha\in\mathbb{R},\; i=1,\dots,d .
\tag{1}
\]
Denote by $p_i>0$ the probability of selecting coordinate $i$, with $\sum_{i=1}^{d}p_i=1$.
Given a stepsize $\eta>0$, the iteration at time $t$ is
\[
\boxed{
\begin{aligned}
i_t &\sim \text{Multinomial}(p_1,\dots,p_d),\\
w_{t+1}&=w_t-\frac{\eta}{p_{i_t}}\,\partial_{i_t}f(w_t)\,e_{i_t},
\end{aligned}}
\tag{2}
\]
where $e_i$ is the $i$‑th canonical unit vector.  
The factor $1/p_{i_t}$ compensates for the non‑uniform sampling and will appear in the expectation calculations.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD‑NU}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, stepsize $\eta>0$, probabilities $\{p_i\}_{i=1}^d$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample coordinate $i_t\in\{1,\dots,d\}$ with $\Pr(i_t=i)=p_i$.
    \STATE Compute partial derivative $g_{i_t}\gets\partial_{i_t}f(w_t)$.
    \STATE Update the selected coordinate
    \[
    w_{t+1}=w_t-\frac{\eta}{p_{i_t}}\,g_{i_t}\,e_{i_t}.
    \]
\ENDFOR
\STATE \textbf{Return} $w_T$ (or an averaged iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate convex quadratic
\[
f(w)=(w-3)^2,\qquad f'(w)=2(w-3).
\]
Here $d=1$, $p_1=1$, $L_1=2$ (since $|f''(w)|=2$).  
Choose $w_0=0$ and stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_{i_t}=f'(w_t)$ & $f(w_t)$\\\hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-\frac{0.1}{1}(-6)=0.6$ & $f'(0.6)=2(0.6-3)=-4.8$ & $f(0.6)=5.76$\\
2 & $w_2=0.6-\frac{0.1}{1}(-4.8)=1.08$ & $f'(1.08)=2(1.08-3)=-3.84$ & $f(1.08)=3.6864$\\
3 & $w_3=1.08-\frac{0.1}{1}(-3.84)=1.464$ & $f'(1.464)=-3.072$ & $f(1.464)=2.352$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm’s behaviour on a simple problem.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex, i.e. for all $x,y\in\mathbb{R}^d$,
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x}.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]\label{ass:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that (1) holds.
\end{assumption}
\begin{assumption}[Sampling Probabilities]\label{ass:prob}
$p_i>0$ for all $i$ and $\sum_{i=1}^d p_i =1$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:step}
The stepsize satisfies
\[
0<\eta\le \frac{1}{2\,\max_{i}\frac{L_i}{p_i}}.
\tag{4}
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Expected $O(1/T)$ convergence]\label{thm:main}
Let $\{w_t\}$ be generated by (2) under Assumptions \ref{ass:convex}–\ref{ass:step} and let $w^\star\in\arg\min f$. Define
\[
C\;:=\;\max_{i=1,\dots,d}\frac{L_i}{p_i}.
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f(w^\star)\bigr]
\;\le\;
\frac{2C}{\eta\,T}\,\norm{w_0-w^\star}^{2}.
\tag{5}
\]
Consequently, choosing a random iterate $\tilde w$ uniformly from $\{w_0,\dots,w_{T-1}\}$ yields
\[
\E\!\bigl[f(\tilde w)-f(w^\star)\bigr]\;\le\;\frac{2C}{\eta\,T}\,\norm{w_0-w^\star}^{2}.
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{ass:step} guarantees that the term $1-\eta C$ appearing in the descent inequality is non‑negative; thus the expected distance to the optimum never grows.
\item \textbf{Hyper‑parameter sensitivity.} The bound depends on the product $\eta C$. A larger stepsize speeds up convergence up to the limit $\eta\le 1/(2C)$; beyond this the guarantee breaks down.
\item \textbf{Comparison to uniform sampling.} If $p_i=1/d$, then $C=\max_i L_i d$ and the rate recovers the classic $O(d\,\max_i L_i/T)$ bound for uniform randomized coordinate descent.
\item \textbf{Special case – exact line search.} Setting $\eta = p_{i_t}$ and using the exact line‐search step $\alpha_t = \frac{1}{L_{i_t}}\partial_{i_t}f(w_t)$ eliminates the $1/p_i$ factor, yielding the standard $O(1/T)$ rate without the constant $C$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Convexity inequality]\label{lem:conv}
For any $w\in\mathbb{R}^d$ and optimal point $w^\star$,
\[
f(w)-f(w^\star)\le \ip{\nabla f(w)}{w-w^\star}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Directly from Assumption \ref{ass:convex} with $x=w$, $y=w^\star$.
\end{proof}

\begin{lemma}[Coordinate‑wise smoothness bound]\label{lem:smooth}
For each $i$ and any $w\in\mathbb{R}^d$,
\[
\bigl(\partial_i f(w)\bigr)^2\le 2L_i\bigl(f(w)-f(w-e_i\frac{1}{L_i}\partial_i f(w))\bigr).
\tag{7}
\]
\end{lemma}
\begin{proof}
From $L_i$‑smoothness we have
\[
f\!\bigl(w-\tfrac{1}{L_i}\partial_i f(w)e_i\bigr)
\le f(w)-\frac{1}{2L_i}\bigl(\partial_i f(w)\bigr)^2,
\]
which after rearranging gives (7).
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof expectation $\E_t[\cdot]$ denotes conditional expectation w.r.t. the random choice $i_t$ given $w_t$.

\noindent\textbf{Step 1.}  Expand the squared distance after one update.
\[
\begin{aligned}
\norm{w_{t+1}-w^\star}^{2}
&=\norm{w_t-\frac{\eta}{p_{i_t}}\partial_{i_t}f(w_t)e_{i_t}-w^\star}^{2}\\
&=\norm{w_t-w^\star}^{2}
-2\frac{\eta}{p_{i_t}}\partial_{i_t}f(w_t)\bigl(w_t-w^\star\bigr)_{i_t}
+\frac{\eta^{2}}{p_{i_t}^{2}}\bigl(\partial_{i_t}f(w_t)\bigr)^{2}.
\end{aligned}
\tag{8}
\]
\textbf{[Step 1]}

\noindent\textbf{Step 2.}  Take conditional expectation using $\Pr(i_t=i)=p_i$.
\[
\begin{aligned}
\E_t\!\bigl[\norm{w_{t+1}-w^\star}^{2}\bigr]
&=\norm{w_t-w^\star}^{2}
-2\eta\sum_{i=1}^{d}\partial_i f(w_t)\bigl(w_t-w^\star\bigr)_i
+\eta^{2}\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(w_t)\bigr)^{2}.
\end{aligned}
\tag{9}
\]
\textbf{[Step 2]}

\noindent\textbf{Step 3.}  Apply Lemma \ref{lem:conv} to lower‑bound the linear term.
\[
\sum_{i=1}^{d}\partial_i f(w_t)\bigl(w_t-w^\star\bigr)_i
\;\ge\; f(w_t)-f(w^\star).
\tag{10}
\]
\textbf{[Step 3]}

\noindent\textbf{Step 4.}  Use Lemma \ref{lem:smooth} to bound the quadratic term.
From (7) we have for each $i$,
\[
\bigl(\partial_i f(w_t)\bigr)^{2}
\le 2L_i\Bigl(f(w_t)-f\!\bigl(w_t-\tfrac{1}{L_i}\partial_i f(w_t)e_i\bigr)\Bigr)
\le 2L_i\bigl(f(w_t)-f(w^\star)\bigr),
\]
where the last inequality follows from convexity of $f$.
Hence,
\[
\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\partial_i f(w_t)\bigr)^{2}
\le 2\Bigl(\max_{i}\frac{L_i}{p_i}\Bigr)\bigl(f(w_t)-f(w^\star)\bigr)
=2C\bigl(f(w_t)-f(w^\star)\bigr).
\tag{11}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5.}  Substitute (10) and (11) into (9).
\[
\E_t\!\bigl[\norm{w_{t+1}-w^\star}^{2}\bigr]
\le \norm{w_t-w^\star}^{2}
-2\eta\bigl(f(w_t)-f(w^\star)\bigr)
+\eta^{2}\,2C\bigl(f(w_t)-f(w^\star)\bigr).
\tag{12}
\]
\textbf{[Step 5]}

\noindent\textbf{Step 6.}  Choose $\eta$ satisfying (4) so that $2\eta C\le 1$.
Then $-2\eta+2\eta^{2}C\le -\eta$, and (12) simplifies to
\[
\E_t\!\bigl[\norm{w_{t+1}-w^\star}^{2}\bigr]
\le \norm{w_t-w^\star}^{2}
-\eta\bigl(f(w_t)-f(w^\star)\bigr).
\tag{13}
\]
\textbf{[Step 6]}

\noindent\textbf{Step 7.}  Take full expectation and sum over $t=0,\dots,T-1$.
\[
\begin{aligned}
\E\!\bigl[\norm{w_{T}-w^\star}^{2}\bigr]
&\le \norm{w_0-w^\star}^{2}
-\eta\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f(w^\star)\bigr].
\end{aligned}
\tag{14}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8.}  Rearrange (14) to obtain the averaged bound.
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f(w^\star)\bigr]
\le \frac{\norm{w_0-w^\star}^{2}}{\eta\,T}.
\tag{15}
\]
\textbf{[Step 8]}

\noindent\textbf{Step 9.}  Replace $C$ using (4). Since $\eta\le \frac{1}{2C}$, we have $\frac{1}{\eta}\le 2C$, which yields the final bound (5):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f(w^\star)\bigr]
\le \frac{2C}{\eta\,T}\,\norm{w_0-w^\star}^{2}.
\]
\textbf{[Step 9]}

Thus Theorem \ref{thm:main} is proved. \hfill $\blacksquare$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From the proof we see that the constant governing the speed is $C=\max_i L_i/p_i$.  
If all coordinates are sampled uniformly ($p_i=1/d$) then $C=d\max_i L_i$ and (5) becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f^\star\bigr]
\le \frac{2d\,\max_i L_i}{\eta\,T}\norm{w_0-w^\star}^{2},
\]
which matches the classic bound for randomized coordinate descent.

If we choose probabilities proportional to the smoothness constants,
\[
p_i=\frac{L_i}{\sum_{j}L_j},
\]
then $C=\sum_{j}L_j$, i.e. the dependence on the worst‑case coordinate disappears and the rate improves.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact line search.} Setting $\eta=p_{i_t}$ and using the step
$\alpha_t=\frac{1}{L_{i_t}}\partial_{i_t}f(w_t)$ gives
\[
w_{t+1}=w_t-\frac{1}{L_{i_t}}\partial_{i_t}f(w_t)e_{i_t},
\]
which removes the $1/p_i$ factor. The same proof with $\eta=1$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[f(w_t)-f^\star\bigr]
\le \frac{2\max_i L_i}{T}\norm{w_0-w^\star}^{2}.
\]

\item \textbf{Strongly convex case.} If $f$ is $\mu$‑strongly convex,
\[
f(w)-f^\star\ge \frac{\mu}{2}\norm{w-w^\star}^{2},
\]
combining with (13) gives a linear convergence rate:
\[
\E\!\bigl[f(w_t)-f^\star\bigr]\le\bigl(1-\eta\mu\bigr)^{t}\bigl(f(w_0)-f^\star\bigr).
\]
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The analysis shows that the compensation factor $1/p_i$ in the update (2) exactly cancels the bias introduced by non‑uniform sampling, leading to a clean $O(1/T)$ expected suboptimality bound. The constant $C$ captures the interaction between coordinate smoothness $L_i$ and sampling probabilities $p_i$; choosing $p_i$ proportional to $L_i$ minimizes $C$ and therefore yields the fastest guarantee. The proof follows the same elementary steps as the classic SGD analysis—expansion of the squared distance, convexity inequality, and a smoothness‑based bound—yet each algebraic manipulation is displayed explicitly and labeled for easy verification.
