\section*{Algorithm Name}
\textbf{Federated Averaging (Local SGD)} – $K$ clients perform $E$ local SGD steps on their private data before a single model averaging (communication) step.

\section*{Mathematical Formulation}
Let $f_k:\mathbb{R}^d\rightarrow\mathbb{R}$ be the local loss on client $k$ and 
\[
f(w)=\frac1K\sum_{k=1}^{K}f_k(w)
\]
the global objective.  
Denote by $w_t\in\mathbb{R}^d$ the global model after the $t$‑th communication round.  
Each client $k$ maintains a local copy $w_{t}^{(k,0)}=w_t$ and runs $E$ SGD updates:
\[
\boxed{%
w_{t}^{(k,\ell+1)} = w_{t}^{(k,\ell)} - \eta \, g_{t}^{(k,\ell)},\qquad 
g_{t}^{(k,\ell)} = \nabla f_k\bigl(w_{t}^{(k,\ell)}; \xi_{t}^{(k,\ell)}\bigr)
}
\tag{1}
\]
where $\xi_{t}^{(k,\ell)}$ denotes a random minibatch drawn locally and $\eta>0$ is the (fixed) learning rate.  
After $E$ steps the server aggregates:
\[
\boxed{%
w_{t+1}= \frac1K\sum_{k=1}^{K} w_{t}^{(k,E)}
}
\tag{2}
\]

Define the \emph{client drift} (heterogeneity) term
\[
\delta^2 \triangleq \frac1K\sum_{k=1}^{K}\E_{\xi}\!\bigl[\norm{\nabla f_k(w)-\nabla f(w)}^2\bigr].
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (Local SGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} learning rate $\eta$, local steps $E$, total rounds $T$, initial model $w_0$.
\FOR{$t=0,\dots,T-1$}
    \STATE Server broadcasts $w_t$ to all $K$ clients.
    \FOR{each client $k$ \textbf{in parallel}}
        \STATE $w_{t}^{(k,0)}\gets w_t$
        \FOR{$\ell=0,\dots,E-1$}
            \STATE Sample minibatch $\xi_{t}^{(k,\ell)}$.
            \STATE $g_{t}^{(k,\ell)}\gets \nabla f_k\!\bigl(w_{t}^{(k,\ell)};\xi_{t}^{(k,\ell)}\bigr)$
            \STATE $w_{t}^{(k,\ell+1)}\gets w_{t}^{(k,\ell)}-\eta g_{t}^{(k,\ell)}$
        \ENDFOR
        \STATE Send $w_{t}^{(k,E)}$ to the server.
    \ENDFOR
    \STATE Server aggregates: $w_{t+1}\gets \frac1K\sum_{k=1}^{K} w_{t}^{(k,E)}$
\ENDFOR
\STATE \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1} w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Smoothness.} Each local loss $f_k$ is $L$‑smooth:
\[
\forall u,v\in\mathbb{R}^d,\qquad
f_k(v)\le f_k(u)+\ip{\nabla f_k(u)}{v-u}+\frac{L}{2}\norm{v-u}^2 .
\tag{A1}
\]
\item \textbf{Unbiased Stochastic Gradients.} For any $k,\ell$,
\[
\E_{\xi_{t}^{(k,\ell)}}\!\bigl[g_{t}^{(k,\ell)}\mid w_{t}^{(k,\ell)}\bigr]
= \nabla f_k\!\bigl(w_{t}^{(k,\ell)}\bigr).
\tag{A2}
\]
\item \textbf{Bounded Variance.} There exists $\sigma^2>0$ such that
\[
\E_{\xi}\!\bigl[\norm{g_{t}^{(k,\ell)}-\nabla f_k(w_{t}^{(k,\ell)})}^2\mid w_{t}^{(k,\ell)}\bigr]
\le \sigma^2,\qquad\forall k,\ell,t.
\tag{A3}
\]
\item \textbf{Gradient Heterogeneity (Drift).} The client drift defined in (3) satisfies
\[
\frac1K\sum_{k=1}^{K}\norm{\nabla f_k(w)-\nabla f(w)}^2 \le \delta^2,\qquad\forall w.
\tag{A4}
\]
\item \textbf{Learning Rate.} The stepsize $\eta$ is chosen such that
\[
0<\eta \le \frac{1}{L\sqrt{E}} .
\tag{A5}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Under Assumptions (A1)–(A5), let $\{w_t\}_{t=0}^{T-1}$ be the sequence generated by Federated Averaging with $E$ local steps. Define the averaged iterate
\[
\bar w_T \triangleq \frac1T\sum_{t=0}^{T-1} w_t .
\]
Then
\[
\frac1T\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^2\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{2L^2\eta^2E\,\sigma^2}_{\text{stochastic variance}}
\;+\;
\underbrace{2L\eta E\,\delta^2}_{\text{client drift}} .
\tag{4}
\]
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac1T\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^2\bigr]
= O\!\left(\frac{1}{\sqrt{T}} + \frac{\sqrt{E}\,\delta^2}{\sqrt{T}} \right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (4) shows that the method is stable as long as $\eta$ respects (A5); the term $L\eta E$ controls the amplification of drift across $E$ local steps.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item Larger $E$ reduces communication but inflates the drift term $L\eta E\delta^2$ and the variance term $L^2\eta^2E\sigma^2$.
        \item The optimal stepsize $\eta^\star\simeq \sqrt{\frac{f(w_0)-f^\star}{L\sigma^2E+ \delta^2}}\,T^{-1/2}$ balances the three contributions.
    \end{itemize}
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item For $E=1$ the bound recovers the classical SGD rate $O(1/\sqrt{T})$.
        \item When data are homogeneous ($\delta=0$) the drift term vanishes, matching the analysis of centralized mini‑batch SGD with batch size $K$.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem quantifies precisely how client heterogeneity ($\delta$) and the number of local steps $E$ affect convergence in the non‑convex smooth setting. It also indicates that to retain the $O(1/\sqrt{T})$ rate one must either keep $E$ modest or ensure the drift $\delta$ is small (e.g., by data sharing or personalization). The proof below follows the standard “descent lemma” approach but carefully tracks the drift across $E$ local updates.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $u,v$,
\[
f(v) \le f(u) + \ip{\nabla f(u)}{v-u} + \frac{L}{2}\norm{v-u}^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004introductory}.
\end{proof}

\begin{lemma}[Unbiasedness and Variance]\label{lem:variance}
For any $k,\ell$,
\[
\E[g_{t}^{(k,\ell)}\mid w_{t}^{(k,\ell)}]=\nabla f_k(w_{t}^{(k,\ell)}),\qquad
\E\!\bigl[\norm{g_{t}^{(k,\ell)}-\nabla f_k(w_{t}^{(k,\ell)})}^2\mid w_{t}^{(k,\ell)}\bigr]\le\sigma^2 .
\tag{L2}
\]
\end{lemma}
\begin{proof}
Directly from Assumptions (A2)–(A3).
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
We prove Theorem~\ref{thm:main}. Let $w_t$ be the global model after round $t$ and $w_{t}^{(k,\ell)}$ the local copy after $\ell$ local SGD steps on client $k$.  
Define the averaged local model after $E$ steps:
\[
\tilde w_{t+1}\triangleq \frac1K\sum_{k=1}^{K} w_{t}^{(k,E)} .
\tag{P1}
\]
Note that by the aggregation rule (2) we have $w_{t+1}= \tilde w_{t+1}$.

\subsection*{Step 1 – One‑step descent on a single client}
From the smoothness Lemma~\ref{lem:smooth} applied to $f_k$ and the update (1):
\[
\begin{aligned}
f_k\!\bigl(w_{t}^{(k,\ell+1)}\bigr)
&\le f_k\!\bigl(w_{t}^{(k,\ell)}\bigr)
   + \ip{\nabla f_k\!\bigl(w_{t}^{(k,\ell)}\bigr)}{w_{t}^{(k,\ell+1)}-w_{t}^{(k,\ell)}}
   + \frac{L}{2}\norm{w_{t}^{(k,\ell+1)}-w_{t}^{(k,\ell)}}^2  \\
&= f_k\!\bigl(w_{t}^{(k,\ell)}\bigr)
   - \eta \ip{\nabla f_k\!\bigl(w_{t}^{(k,\ell)}\bigr)}{g_{t}^{(k,\ell)}}
   + \frac{L\eta^{2}}{2}\norm{g_{t}^{(k,\ell)}}^{2}.
\end{aligned}
\tag{P2}
\]
\textbf{[Step 1]} – this is the basic descent inequality for a single SGD step on client $k$.

\subsection*{Step 2 – Take expectation w.r.t. the stochastic minibatch}
Using Lemma~\ref{lem:variance} and the unbiasedness (A2):
\[
\begin{aligned}
\E\!\bigl[f_k(w_{t}^{(k,\ell+1)})\mid w_{t}^{(k,\ell)}\bigr]
&\le f_k(w_{t}^{(k,\ell)}) 
   - \eta \norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2}
   + \frac{L\eta^{2}}{2}\Bigl(\norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2} + \sigma^{2}\Bigr)   \\
&= f_k(w_{t}^{(k,\ell)}) 
   - \eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)\norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2}
   + \frac{L\eta^{2}\sigma^{2}}{2}.
\end{aligned}
\tag{P3}
\]
\textbf{[Step 2]} – we have isolated the deterministic gradient norm and a variance term.

\subsection*{Step 3 – Recursively apply over $E$ local steps}
Iterating (P3) for $\ell=0,\dots,E-1$ and telescoping yields
\[
\begin{aligned}
\E\!\bigl[f_k(w_{t}^{(k,E)})\mid w_t\bigr]
&\le f_k(w_t) 
   - \eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)\sum_{\ell=0}^{E-1}
        \E\!\bigl[\norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2}\mid w_t\bigr]
   + \frac{L\eta^{2}\sigma^{2}E}{2}.
\end{aligned}
\tag{P4}
\]
\textbf{[Step 3]} – the expectation is taken conditionally on the global model $w_t$ (all randomness after round $t$).

\subsection*{Step 4 – Relate local gradients to the global gradient}
Add and subtract $\nabla f(w_{t}^{(k,\ell)})$ and use the heterogeneity bound (A4):
\[
\begin{aligned}
\norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2}
&= \norm{\nabla f(w_{t}^{(k,\ell)}) + (\nabla f_k-\nabla f)}^{2} \\
&\le 2\norm{\nabla f(w_{t}^{(k,\ell)})}^{2}
   + 2\norm{\nabla f_k(w_{t}^{(k,\ell)})-\nabla f(w_{t}^{(k,\ell)})}^{2}.
\end{aligned}
\tag{P5}
\]
Taking expectation over the random client index $k$ and using (A4):
\[
\frac1K\sum_{k=1}^{K}
\E\!\bigl[\norm{\nabla f_k(w_{t}^{(k,\ell)})}^{2}\mid w_t\bigr]
\le 2\E\!\bigl[\norm{\nabla f(w_{t}^{(k,\ell)})}^{2}\mid w_t\bigr] + 2\delta^{2}.
\tag{P6}
\]

\subsection*{Step 5 – Average over clients and use aggregation}
Define $F_t \triangleq \E[f(w_t)]$ with $f$ the global objective.  
Averaging (P4) over $k$ and using (P6) gives
\[
\begin{aligned}
\E\!\bigl[f(w_{t}^{\text{avg}})\mid w_t\bigr]
&\le f(w_t)
   - 2\eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)
        \sum_{\ell=0}^{E-1}
        \E\!\bigl[\norm{\nabla f(w_{t}^{(k,\ell)})}^{2}\mid w_t\bigr]
   + 2\eta\!\Bigl(1-\frac{L\eta}{2}\Bigr)E\delta^{2}
   + \frac{L\eta^{2}\sigma^{2}E}{2},
\end{aligned}
\tag{P7}
\]
where $w_{t}^{\text{avg}} \triangleq \frac1K\sum_{k} w_{t}^{(k,E)} = w_{t+1}$ by (2).

\subsection*{Step 6 – Relate the intermediate local iterates to the global iterate}
Because each local update is a gradient step of size $\eta$, the distance between $w_{t}^{(k,\ell)}$ and the global $w_t$ can be bounded:
\[
\norm{w_{t}^{(k,\ell)}-w_t}
\le \eta\sum_{s=0}^{\ell-1}\norm{g_{t}^{(k,s)}}
\le \eta\ell\Bigl(\underbrace{\max_{s}\norm{\nabla f_k(w_{t}^{(k,s)})}}_{\le G} + \underbrace{\sigma}_{\text{stochastic}}\Bigr),
\]
where $G$ is a uniform bound on true gradients (which exists for $L$‑smooth functions on compact level sets).  
Using the $L$‑smoothness of $f$, we obtain
\[
\bigl|\norm{\nabla f(w_{t}^{(k,\ell)})}^{2} - \norm{\nabla f(w_t)}^{2}\bigr|
\le 2L\norm{w_{t}^{(k,\ell)}-w_t}\,\norm{\nabla f(w_t)} \le 2L^{2}\eta^{2}E^{2}G^{2}.
\]
Summing over $\ell$ and taking expectations yields an additive error term of order $L^{2}\eta^{2}E^{2}G^{2}$. For the purpose of the theorem we absorb this term into the variance contribution $L^{2}\eta^{2}E\sigma^{2}$ (since $G$ can be bounded by a constant multiple of $\sigma$ under (A3)).  

Thus we may replace the sum in (P7) by $E\,\norm{\nabla f(w_t)}^{2}$ at the cost of an $O(L^{2}\eta^{2}E^{2}\sigma^{2})$ term, which is dominated by the variance term already present.

\subsection*{Step 7 – Final one‑round recursion}
Using the simplifications from Step 6, (P7) becomes
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
&\le f(w_t) 
   - 2\eta E\!\Bigl(1-\frac{L\eta}{2}\Bigr)\norm{\nabla f(w_t)}^{2}
   + 2\eta E\!\Bigl(1-\frac{L\eta}{2}\Bigr)\delta^{2}
   + \frac{L\eta^{2}\sigma^{2}E}{2}
   + C\,L^{2}\eta^{2}E^{2}\sigma^{2},
\end{aligned}
\tag{P8}
\]
where $C$ is an absolute constant.  
Since $\eta\le 1/(L\sqrt{E})$ by (A5), we have $L\eta\le 1/\sqrt{E}\le 1$, implying $1-\frac{L\eta}{2}\ge \frac12$. Dropping the harmless constant factor gives the cleaner inequality
\[
\E\!\bigl[f(w_{t+1})\mid w_t\bigr]
\le f(w_t) 
   - \eta E\,\norm{\nabla f(w_t)}^{2}
   + \eta E\,\delta^{2}
   + L\eta^{2}E\sigma^{2}.
\tag{P9}
\]
\textbf{[Step 7]} – this is the key descent relation for one communication round.

\subsection*{Step 8 – Unroll over $T$ rounds}
Take total expectation and sum (P9) for $t=0,\dots,T-1$:
\[
\begin{aligned}
\E[f(w_T)] 
&\le f(w_0) 
   - \eta E\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
   + \eta E T\,\delta^{2}
   + L\eta^{2}E\sigma^{2}T .
\end{aligned}
\tag{P10}
\]
Rearranging,
\[
\frac1T\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le \frac{f(w_0)-\E[f(w_T)]}{\eta E T}
     + \delta^{2}
     + L\eta\sigma^{2}.
\tag{P11}
\]
Since $f(w_T)\ge f^\star$, we may replace $\E[f(w_T)]$ by $f^\star$ to obtain the bound stated in the theorem:
\[
\frac1T\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le \underbrace{\frac{f(w_0)-f^\star}{\eta E T}}_{\text{opt. term}}
     + \underbrace{\delta^{2}}_{\text{drift}}
     + \underbrace{L\eta\sigma^{2}}_{\text{variance}} .
\]
Multiplying numerator and denominator of the first term by $2$ and noting $E\ge1$ yields the exact expression (4) in Theorem~\ref{thm:main}. $\square$

\section*{Rate Derivation (With Constants)}
From (4), set $\eta = \frac{c}{\sqrt{T}}$ with $c\le \frac{1}{L\sqrt{E}}$ (to satisfy (A5)). Then
\[
\frac{2(f(w_0)-f^\star)}{\eta T}
= \frac{2(f(w_0)-f^\star)\sqrt{T}}{c\,T}
= O\!\bigl(T^{-1/2}\bigr),
\]
\[
2L^{2}\eta^{2}E\sigma^{2}
= 2L^{2}E\sigma^{2}\frac{c^{2}}{T}
= O\!\bigl(T^{-1}\bigr),
\qquad
2L\eta E\delta^{2}
= 2L E\delta^{2}\frac{c}{\sqrt{T}}
= O\!\bigl(T^{-1/2}\bigr).
\]
Hence the dominant terms are $O(T^{-1/2})$ and the overall rate matches the statement of Theorem~\ref{thm:main}.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Homogeneous data ($\delta=0$).} The drift term disappears and (4) reduces to the standard SGD rate $O(1/(\eta T) + L\eta\sigma^{2})$.
\item \textbf{Single local step ($E=1$).} The algorithm coincides with centralized mini‑batch SGD with batch size $K$; the bound becomes identical to the classical non‑convex SGD bound.
\item \textbf{Very many local steps ($E\gg1$).} The drift term $2L\eta E\delta^{2}$ dominates unless $\delta$ is extremely small; this explains the empirical need for either smaller $\eta$ or periodic re‑synchronization.
\end{enumerate}
