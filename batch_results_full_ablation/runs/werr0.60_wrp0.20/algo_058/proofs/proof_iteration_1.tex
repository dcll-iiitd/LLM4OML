\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$-smooth objective.  
At iteration $t$ we sample a stochastic gradient $g_t:=\nabla f(w_t;\xi_t)$, where $\xi_t$ is a random data sample.  
A (possibly lossy) quantization operator $Q:\R^{d}\rightarrow\R^{d}$ satisfies  

\[
\E[Q(g_t)\mid w_t]=g_t ,\qquad 
\E\!\left[\norm{Q(g_t)-g_t}^{2}\mid w_t\right]\leq \beta \,\norm{g_t}^{2},
\tag{1}
\]

with $0\le\beta<1$ quantifying the distortion.  

The update rule of QSGD is  

\[
\boxed{\,w_{t+1}=w_{t}-\eta_t\, Q(g_t)\,},\qquad t=0,1,\dots,T-1,
\tag{2}
\]

where $\{\eta_t\}_{t\ge0}$ is a stepsize schedule (constant or diminishing).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, stepsizes $\{\eta_t\}$, number of iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State Quantize: $\tilde g_t = Q(g_t)$.
    \State Update: $w_{t+1}=w_t-\eta_t \tilde g_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the deterministic quadratic $f(w)=(w-3)^2$ on $\R$ (so $L=2$).  
Take the identity quantizer $Q(g)=g$, stepsize $\eta=0.1$, and $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\ \hline
0 & $2(w_0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3660$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm’s behaviour.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{asm:convex}
$f$ is convex and differentiable and $L$-smooth:
\[
f(y)\le f(x)+\iprod{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2},
\quad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Variance]\label{asm:variance}
There exists $\sigma^{2}\ge0$ such that for all $t$,
\[
\E\!\left[\norm{g_t-\nabla f(w_t)}^{2}\mid w_t\right]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Quantizer with Bounded Distortion]\label{asm:quant}
The quantizer $Q$ satisfies (1) with a constant $\beta\in[0,1)$.
\end{assumption}
\begin{assumption}[Stepsize]\label{asm:stepsize}
We use a diminishing stepsize $\eta_t=\frac{c}{\sqrt{t+1}}$ with $c>0$ chosen so that $c\le\frac{1}{L(1+\beta)}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of QSGD]\label{thm:main}
Under Assumptions \ref{asm:convex}–\ref{asm:stepsize}, let $w^{*}\in\arg\min f$ and define the averaged iterate $\bar w_T=\frac{1}{T}\sum_{t=0}^{T-1} w_t$. Then
\[
\E\!\left[f(\bar w_T)-f(w^{*})\right]\;\le\;
\frac{\norm{w_0-w^{*}}^{2}}{2c\sqrt{T}}
\;+\;
\frac{cL(1+\beta)}{2\sqrt{T}}\;\sigma^{2}.
\tag{3}
\]
Consequently $\E[f(\bar w_T)-f(w^{*})]=\mathcal O\!\left(\frac{1}{\sqrt{T}}\right)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The recursion (Step 4 below) shows a contraction in expectation provided $c\le\frac{1}{L(1+\beta)}$.
    \item \textbf{Sensitivity to $\beta$:} Larger quantization distortion $\beta$ appears linearly in the bound (3); the algorithm degrades gracefully.
    \item \textbf{Comparison to vanilla SGD:} Setting $\beta=0$ recovers the classic $\mathcal O(1/\sqrt{T})$ rate for SGD on convex smooth functions.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that unbiased quantization does not alter the asymptotic convergence order; it only inflates the constant by a factor $1+\beta$. This validates the use of low‑precision communication in distributed learning where bandwidth is scarce. The proof below follows the standard SGD analysis, with additional handling of the quantization error term (Step 6).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\iprod{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Non‑expansive Expectation]\label{lem:exp}
For any random vector $Z$ with $\E[Z]=\mu$, $\E[\norm{Z}^{2}]=\norm{\mu}^{2}+\E[\norm{Z-\mu}^{2}]$.
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  All expectations are taken w.r.t. the history up to time $t$, denoted $\mathcal F_t$.

\begin{proof}
\textbf{[Step 1]}  Apply Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta_t Q(g_t)$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\iprod{\nabla f(w_t)}{Q(g_t)}+\frac{L\eta_t^{2}}{2}\norm{Q(g_t)}^{2}.
\tag{4}
\]

\textbf{[Step 2]}  Take conditional expectation $\E[\cdot|\mathcal F_t]$ and use unbiasedness of $Q$ (Assumption~\ref{asm:quant}):
\[
\E\!\left[\iprod{\nabla f(w_t)}{Q(g_t)}\mid\mathcal F_t\right]
= \iprod{\nabla f(w_t)}{\E[Q(g_t)\mid\mathcal F_t]}
= \iprod{\nabla f(w_t)}{g_t}.
\tag{5}
\]

\textbf{[Step 3]}  Expand $\norm{Q(g_t)}^{2}$ using Lemma~\ref{lem:exp} and (1):
\[
\begin{aligned}
\E\!\left[\norm{Q(g_t)}^{2}\mid\mathcal F_t\right]
&= \norm{g_t}^{2}+\E\!\left[\norm{Q(g_t)-g_t}^{2}\mid\mathcal F_t\right] \\
&\le (1+\beta)\norm{g_t}^{2}.
\end{aligned}
\tag{6}
\]

\textbf{[Step 4]}  Substitute (5)–(6) into (4) and then take full expectation:
\[
\begin{aligned}
\E[f(w_{t+1})]
&\le \E[f(w_t)]-\eta_t\E\!\left[\iprod{\nabla f(w_t)}{g_t}\right]
+\frac{L\eta_t^{2}}{2}(1+\beta)\E\!\left[\norm{g_t}^{2}\right].
\end{aligned}
\tag{7}
\]

\textbf{[Step 5]}  Decompose $g_t=\nabla f(w_t)+\zeta_t$ where $\zeta_t:=g_t-\nabla f(w_t)$.
Using $\E[\zeta_t\mid\mathcal F_t]=0$ and Assumption~\ref{asm:variance} we have
\[
\E\!\left[\norm{g_t}^{2}\right]=\E\!\left[\norm{\nabla f(w_t)}^{2}\right]+\E\!\left[\norm{\zeta_t}^{2}\right]
\le \E\!\left[\norm{\nabla f(w_t)}^{2}\right]+\sigma^{2}.
\tag{8}
\]

\textbf{[Step 6]}  Moreover,
\[
\E\!\left[\iprod{\nabla f(w_t)}{g_t}\right]
=\E\!\left[\norm{\nabla f(w_t)}^{2}\right],
\tag{9}
\]
because $\E[\zeta_t]=0$.

\textbf{[Step 7]}  Plug (8)–(9) into (7):
\[
\E[f(w_{t+1})]\le \E[f(w_t)]-\eta_t\E\!\left[\norm{\nabla f(w_t)}^{2}\right]
+\frac{L\eta_t^{2}}{2}(1+\beta)\Bigl(\E\!\left[\norm{\nabla f(w_t)}^{2}\right]+\sigma^{2}\Bigr).
\tag{10}
\]

\textbf{[Step 8]}  Rearrange (10) to isolate the gradient term:
\[
\Bigl(\eta_t-\frac{L\eta_t^{2}}{2}(1+\beta)\Bigr)\E\!\left[\norm{\nabla f(w_t)}^{2}\right]
\le \E[f(w_t)]-\E[f(w_{t+1})]+\frac{L\eta_t^{2}}{2}(1+\beta)\sigma^{2}.
\tag{11}
\]

\textbf{[Step 9]}  By the stepsize choice $\eta_t\le\frac{1}{L(1+\beta)}$, the coefficient in front of the gradient satisfies
\[
\eta_t-\frac{L\eta_t^{2}}{2}(1+\beta)\ge \frac{\eta_t}{2}.
\tag{12}
\]

\textbf{[Step 10]}  Using (12) in (11) yields
\[
\frac{\eta_t}{2}\E\!\left[\norm{\nabla f(w_t)}^{2}\right]
\le \E[f(w_t)]-\E[f(w_{t+1})]+\frac{L\eta_t^{2}}{2}(1+\beta)\sigma^{2}.
\tag{13}
\]

\textbf{[Step 11]}  Sum (13) over $t=0,\dots,T-1$:
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_t \E\!\left[\norm{\nabla f(w_t)}^{2}\right]
\le f(w_0)-\E[f(w_T)]+\frac{L(1+\beta)\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{14}
\]

\textbf{[Step 12]}  Since $f$ is convex, for the averaged iterate $\bar w_T$ we have
\[
f(\bar w_T)-f(w^{*})\le \frac{1}{T}\sum_{t=0}^{T-1}\bigl(f(w_t)-f(w^{*})\bigr).
\tag{15}
\]

\textbf{[Step 13]}  Using $f(w^{*})\le f(w_T)$ and discarding the non‑negative term $f(w_T)-f(w^{*})$ from (14) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[f(w_t)-f(w^{*})\right]
\le \frac{ \norm{w_0-w^{*}}^{2}}{2\sum_{t=0}^{T-1}\eta_t}
    +\frac{L(1+\beta)\sigma^{2}}{2T}\,\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{16}
\]

\textbf{[Step 14]}  With $\eta_t=c/\sqrt{t+1}$ we have the standard bounds
\[
\sum_{t=0}^{T-1}\eta_t \ge 2c\sqrt{T},\qquad
\sum_{t=0}^{T-1}\eta_t^{2}\le c^{2}\log T + c^{2}.
\tag{17}
\]

\textbf{[Step 15]}  Plugging (17) into (16) and using $\log T\le \sqrt{T}$ for $T\ge 1$ yields
\[
\E\!\left[f(\bar w_T)-f(w^{*})\right]
\le \frac{\norm{w_0-w^{*}}^{2}}{2c\sqrt{T}}
   +\frac{cL(1+\beta)}{2\sqrt{T}}\;\sigma^{2}.
\tag{18}
\]

Equation (18) is exactly the claimed bound (3).  ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
From (18) we identify the two constants
\[
C_{1}= \frac{\norm{w_0-w^{*}}^{2}}{2c},\qquad
C_{2}= \frac{cL(1+\beta)}{2}\sigma^{2},
\]
so that for all $T\ge1$
\[
\E\!\left[f(\bar w_T)-f(w^{*})\right]\le \frac{C_{1}+C_{2}}{\sqrt{T}}.
\]
Choosing $c=\frac{1}{L(1+\beta)}$ balances the two terms and gives the optimal constant
\[
C_{1}+C_{2}= \frac{L(1+\beta)}{2}\Bigl(\norm{w_0-w^{*}}^{2}+ \sigma^{2}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No quantization ($\beta=0$):} The bound reduces to the classical SGD rate.
    \item \textbf{Deterministic gradients ($\sigma^{2}=0$):} The second term disappears, yielding $O(1/\sqrt{T})$ with a smaller constant.
    \item \textbf{Constant stepsize $\eta_t=\eta\le\frac{1}{L(1+\beta)}$:} One obtains a steady‑state error floor
    $\E[f(\bar w_T)-f(w^{*})]\le \frac{\eta L(1+\beta)}{2}\sigma^{2}$,
    matching the standard SGD analysis.
\end{itemize}
