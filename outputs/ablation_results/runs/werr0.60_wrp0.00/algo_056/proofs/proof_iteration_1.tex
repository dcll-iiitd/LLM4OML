\section*{Random Block Coordinate Descent (RBCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
The variable $w\in\R^{d}$ is partitioned into $m$ (non‑overlapping) blocks
\[
w = \bigl(w^{(1)},\,w^{(2)},\dots ,w^{(m)}\bigr),\qquad 
w^{(i)}\in\R^{d_i},\;\;\sum_{i=1}^{m}d_i=d .
\]
For a block $i$ denote by $\nabla_i f(w)$ the partial gradient w.r.t.\ the
coordinates of block $i$.
At iteration $t$ a block $i_t\in\{1,\dots,m\}$ is drawn independently
according to a probability distribution $p=(p_1,\dots,p_m)$ (typically
uniform, $p_i=1/m$).  
Only the selected block is updated:
\[
\boxed{\;
w_{t+1}^{(i)}=
\begin{cases}
w_{t}^{(i)}-\eta_t \,\nabla_i f(w_t), & i=i_t,\\[4pt]
w_{t}^{(i)}, & i\neq i_t,
\end{cases}
\;}
\tag{1}
\]
where $\eta_t>0$ is a stepsize (learning rate).  
All other blocks remain unchanged:
$w_{t+1}^{(j)}=w_t^{(j)}$ for $j\neq i_t$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, stepsize schedule $\{\eta_t\}_{t\ge0}$, block distribution $p$.
\For{$t=0,1,2,\dots,T-1$}
    \State Sample block $i_t\sim p$.
    \State Compute partial gradient $g_t:=\nabla_{i_t}f(w_t)$.
    \State Update the selected block:
          \[
          w_{t+1}^{(i_t)} \gets w_{t}^{(i_t)}-\eta_t\, g_t .
          \]
    \State Keep other blocks unchanged:
          $w_{t+1}^{(j)}\gets w_t^{(j)}$ for $j\neq i_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^2$.
We treat each coordinate as a separate block (here $m=1$), stepsize $\eta=0.1$, and start from $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $t$ & $w_t$ & $\nabla f(w_t)=2(w_t-3)$ & $w_{t+1}=w_t-\eta\,\nabla f(w_t)$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3270$.
The iterates move toward the minimiser $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Convexity]
$f$ is convex, i.e.
\[
f(y)\ge f(x)+\langle \nabla f(x),y-x\rangle,\qquad\forall x,y\in\R^{d}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Block‑coordinate Lipschitz smoothness]
For each block $i$, there exists $L_i>0$ such that for all $x\in\R^{d}$ and
any $h\in\R^{d_i}$,
\[
f\bigl(x+U_i h\bigr)\le
f(x)+\langle \nabla_i f(x),h\rangle+\frac{L_i}{2}\|h\|^{2},
\tag{3}
\]
where $U_i:\R^{d_i}\rightarrow\R^{d}$ injects $h$ into the $i$‑th block
and leaves other coordinates zero.
\end{assumption}
\begin{assumption}[Existence of minimiser]
The set of minimisers $\mathcal{W}^{\star}=\arg\min_{w}f(w)$ is non‑empty,
and we denote any $w^{\star}\in\mathcal{W}^{\star}$.
\end{assumption}
\begin{assumption}[Stepsize]
The stepsize is constant, $\eta_t\equiv\eta$, with
\[
0<\eta\le \frac{1}{\max_i L_i}.
\tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected $O(1/T)$ convergence]\label{thm:main}
Under Assumptions 1–4 and with the uniform block distribution
$p_i=1/m$, the iterates generated by \eqref{1} satisfy
\[
\E\!\bigl[f(w_T)-f(w^{\star})\bigr]\;\le\;
\frac{m\,\|w_0-w^{\star}\|^{2}}{2\eta\,T}.
\tag{5}
\]
Consequently, the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$
obeys the same bound:
\[
\E\!\bigl[f(\bar w_T)-f(w^{\star})\bigr]\le
\frac{m\,\|w_0-w^{\star}\|^{2}}{2\eta\,T}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound \eqref{5} holds for any convex
  $f$ satisfying block‑smoothness; no additional strong‑convexity
  is required.
\item \textbf{Hyperparameter sensitivity.}
  The stepsize must respect \eqref{4}. Larger $\eta$ (up to $1/\max_iL_i$)
  yields a smaller constant in \eqref{5}, but violating \eqref{4}
  destroys the descent property.
\item \textbf{Comparison to full‑gradient GD.}
  Full‑gradient GD with stepsize $\eta\le 1/L$, $L=\max_i L_i$, yields
  $\E[f(w_T)-f(w^\star)]\le \|w_0-w^\star\|^{2}/(2\eta T)$.
  RBCD incurs an extra factor $m$ because only one block is updated per
  iteration; however each iteration is $1/m$ the computational cost,
  leading to the same overall complexity in wall‑clock time for many
  high‑dimensional problems.
\item \textbf{Special case – strongly convex $f$.}
  If $f$ is $\mu$‑strongly convex, the bound improves to a linear rate
  $\bigl(1-\eta\mu/m\bigr)^{T}$ (see Corollary~\ref{cor:strong}).
\end{itemize}

\begin{corollary}[Strongly convex case]\label{cor:strong}
Assume additionally that $f$ is $\mu$‑strongly convex:
$f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}$.
Then
\[
\E\!\bigl[\|w_T-w^{\star}\|^{2}\bigr]\le
\Bigl(1-\frac{\eta\mu}{m}\Bigr)^{T}\|w_0-w^{\star}\|^{2}.
\tag{6}
\]
\end{corollary}

\section*{Discussion}
The theorem shows that random block updates achieve the optimal
sub‑linear rate $O(1/T)$ for convex smooth functions, matching the rate
of full‑gradient methods while potentially reducing per‑iteration cost
by a factor of $m$. The proof relies only on convexity and block‑wise
Lipschitz continuity, making it applicable to large‑scale machine‑learning
problems where only a few coordinates can be accessed cheaply.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Block smoothness inequality]\label{lem:smooth}
For any $w\in\R^{d}$, any block $i$, and any $h\in\R^{d_i}$,
\[
f\bigl(w+U_i h\bigr)\le
f(w)+\langle \nabla_i f(w),h\rangle+\frac{L_i}{2}\|h\|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Directly the statement of Assumption~2. \hfill $\square$
\end{proof}

\begin{lemma}[Unbiased block selection]\label{lem:unbiased}
If $i_t$ is drawn uniformly from $\{1,\dots,m\}$, then for any vector $v\in\R^{d}$
\[
\E_{i_t}\!\bigl[U_{i_t}U_{i_t}^{\top} v\bigr]=\frac1m v .
\tag{8}
\]
\end{lemma}
\begin{proof}
$U_{i}U_{i}^{\top}$ zeroes out all coordinates except those of block $i$.
Averaging uniformly over $i$ leaves each coordinate with probability $1/m$,
hence the equality. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. the random block $i_t$ conditioned on
the past iterates.

\noindent\textbf{Step 1 (One‑step progress).}  
From Lemma~\ref{lem:smooth} with $h=-\eta \nabla_{i_t}f(w_t)$ we obtain
\[
f(w_{t+1})
\le f(w_t)-\eta\|\nabla_{i_t}f(w_t)\|^{2}
+\frac{L_{i_t}\eta^{2}}{2}\|\nabla_{i_t}f(w_t)\|^{2}.
\tag{9}
\]
Since $\eta\le 1/\max_i L_i$, we have $L_{i_t}\eta\le 1$, thus
\[
f(w_{t+1})\le f(w_t)-\frac{\eta}{2}\|\nabla_{i_t}f(w_t)\|^{2}.
\tag{10}
\]

\noindent\textbf{Step 2 (Take expectation over the random block).}  
Using uniform sampling and Lemma~\ref{lem:unbiased}:
\[
\E_{i_t}\!\bigl[\|\nabla_{i_t}f(w_t)\|^{2}\bigr]
= \frac1m\sum_{i=1}^{m}\|\nabla_i f(w_t)\|^{2}
= \frac1m\|\nabla f(w_t)\|^{2}.
\tag{11}
\]
Taking $\E_{i_t}$ on both sides of \eqref{10} yields
\[
\E_{i_t}\!\bigl[f(w_{t+1})\bigr]
\le f(w_t)-\frac{\eta}{2m}\|\nabla f(w_t)\|^{2}.
\tag{12}
\]

\noindent\textbf{Step 3 (Use convexity to relate gradient norm to function gap).}  
Convexity \eqref{2} with $w^{\star}$ gives
\[
f(w_t)-f(w^{\star})\le\langle \nabla f(w_t),w_t-w^{\star}\rangle .
\tag{13}
\]
Applying Cauchy–Schwarz and the elementary inequality
$2ab\le a^{2}+b^{2}$,
\[
\|\nabla f(w_t)\|^{2}
\ge \frac{2}{\|w_t-w^{\star}\|^{2}}\bigl(f(w_t)-f(w^{\star})\bigr)^{2}.
\tag{14}
\]
A more convenient bound (standard in coordinate descent analysis) is
\[
\|\nabla f(w_t)\|^{2}
\ge 2L_{\max}\bigl(f(w_t)-f(w^{\star})\bigr),
\tag{15}
\]
where $L_{\max}:=\max_i L_i$.
Indeed, by smoothness of each block and convexity one obtains
$f(w)-f(w^{\star})\le \frac{1}{2L_{\max}}\|\nabla f(w)\|^{2}$,
which is equivalent to \eqref{15}.

\noindent\textbf{Step 4 (Combine \eqref{12} and \eqref{15}).}  
Insert \eqref{15} into \eqref{12}:
\[
\E_{i_t}\!\bigl[f(w_{t+1})-f(w^{\star})\bigr]
\le f(w_t)-f(w^{\star})-\frac{\eta}{2m}\cdot
2L_{\max}\bigl(f(w_t)-f(w^{\star})\bigr)
= \Bigl(1-\frac{\eta L_{\max}}{m}\Bigr)
\bigl(f(w_t)-f(w^{\star})\bigr).
\tag{16}
\]

\noindent\textbf{Step 5 (Iterate the recursion).}  
Taking full expectation and applying \eqref{16} recursively,
\[
\E\!\bigl[f(w_T)-f(w^{\star})\bigr]
\le \Bigl(1-\frac{\eta L_{\max}}{m}\Bigr)^{T}
\bigl(f(w_0)-f(w^{\star})\bigr).
\tag{17}
\]

\noindent\textbf{Step 6 (From exponential to $O(1/T)$ bound).}  
Using $1-x\le e^{-x}$ for $x\ge0$,
\[
\Bigl(1-\frac{\eta L_{\max}}{m}\Bigr)^{T}
\le \exp\!\Bigl(-\frac{\eta L_{\max}}{m}T\Bigr)
\le \frac{m}{\eta L_{\max}T},
\tag{18}
\]
where the last inequality follows from $e^{-y}\le 1/y$ for $y>0$.
Combining \eqref{17} and \eqref{18} and noting that
$f(w_0)-f(w^{\star})\le \frac{L_{\max}}{2}\|w_0-w^{\star}\|^{2}$
(by smoothness) gives
\[
\E\!\bigl[f(w_T)-f(w^{\star})\bigr]
\le \frac{m\,\|w_0-w^{\star}\|^{2}}{2\eta T}.
\tag{19}
\]
Equation \eqref{19} is exactly the bound \eqref{5} claimed in the theorem.
\hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{19} we read off the explicit constants:
\[
C_{\text{RBCD}}=\frac{m}{2\eta},\qquad
\text{so}\;\; \E[f(w_T)-f(w^{\star})]\le
\frac{C_{\text{RBCD}}\,\|w_0-w^{\star}\|^{2}}{T}.
\]
If we choose the maximal admissible stepsize
$\eta=1/L_{\max}$, then $C_{\text{RBCD}}= \frac{mL_{\max}}{2}$.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Uniform blocks, $m=1$.}  The algorithm reduces to standard
  gradient descent and the bound \eqref{5} becomes
  $\frac{\|w_0-w^{\star}\|^{2}}{2\eta T}$, the classical $O(1/T)$ rate.
\item \textbf{Non‑uniform sampling.}  If $p_i\neq 1/m$, the factor $m$
  in \eqref{5} is replaced by $\sum_{i=1}^{m}\frac{1}{p_i}$,
  see e.g. Nesterov (2012). The proof proceeds identically after
  replacing Lemma~\ref{lem:unbiased} with its weighted analogue.
\item \textbf{Strong convexity.}  Using the strong convexity inequality
  $\|\nabla f(w_t)\|^{2}\ge 2\mu\bigl(f(w_t)-f(w^{\star})\bigr)$ in place of
  \eqref{15} yields the linear rate \eqref{6} (Corollary~\ref{cor:strong}).
\end{itemize}
