\section*{Clipped Stochastic Gradient Descent (Clipped‑SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a differentiable (possibly non‑convex) objective.  
At iteration $t$ we compute the (stochastic) gradient
\[
\widetilde{\nabla}f(x_{t})\;\;(\text{unbiased estimator of }\nabla f(x_{t}))
\]
and clip it with a fixed clipping threshold $C>0$:
\[
g_{t}\;:=\;\frac{\widetilde{\nabla}f(x_{t})}
{\displaystyle \max\!\Bigl\{1,\;\frac{\|\widetilde{\nabla}f(x_{t})\|}{C}\Bigr\}}
\;=\;
\begin{cases}
\widetilde{\nabla}f(x_{t}), & \|\widetilde{\nabla}f(x_{t})\|\le C,\\[4pt]
C\;\dfrac{\widetilde{\nabla}f(x_{t})}{\|\widetilde{\nabla}f(x_{t})\|}, & \|\widetilde{\nabla}f(x_{t})\|> C .
\end{cases}
\]
The update rule is the usual SGD step with learning rate $\eta_{t}>0$:
\[
x_{t+1}=x_{t}-\eta_{t}\,g_{t},\qquad t=0,1,2,\dots .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped‑SGD}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\mathbb{R}^{d}$, learning‑rate schedule $\{\eta_{t}\}$,
 clipping threshold $C>0$, maximal number of iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample a mini‑batch (or a random data point) and compute a stochastic gradient 
    $\widetilde{\nabla}f(x_{t})$.
    \STATE Compute the clipping factor 
    $c_{t}= \max\!\bigl\{1,\;\|\widetilde{\nabla}f(x_{t})\|/C\bigr\}$.
    \STATE Set the clipped gradient $g_{t}= \widetilde{\nabla}f(x_{t})/c_{t}$.
    \STATE Update $x_{t+1}=x_{t}-\eta_{t}\,g_{t}$.
\ENDFOR
\ENSURE $x_{T}$ (or the average iterate $\bar x_{T}=\frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^{2}$, start at $w_{0}=0$, use a fixed step size
$\eta=0.1$ and clipping threshold $C=0.5$ (no stochasticity, i.e. $\widetilde{\nabla}f=\nabla f$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $\| \nabla f(w_{t})\|$ & $g_{t}$ (clipped) & $w_{t+1}=w_{t}-\eta g_{t}$ \\ \hline
0 & $2(w_{0}-3)=-6$ & $6>C$ & $g_{0}=C\frac{-6}{6}=-0.5$ & $w_{1}=0-0.1(-0.5)=0.05$ \\[4pt]
1 & $2(0.05-3)=-5.9$ & $5.9>C$ & $g_{1}=-0.5$ & $w_{2}=0.05-0.1(-0.5)=0.10$ \\[4pt]
2 & $2(0.10-3)=-5.8$ & $5.8>C$ & $g_{2}=-0.5$ & $w_{3}=0.10-0.1(-0.5)=0.15$ 
\end{tabular}
\end{center}
The objective values are $f(w_{0})=9$, $f(w_{1})\approx8.90$, $f(w_{2})\approx8.81$, $f(w_{3})\approx8.73$, showing a steady descent despite the aggressive clipping.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Smoothness.} $f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^{d}:\quad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]
\item \textbf{Unbiased stochastic gradients with bounded variance.} For every $t$,
\[
\E\!\left[\widetilde{\nabla}f(x_{t})\mid x_{t}\right]=\nabla f(x_{t}),\qquad
\E\!\left[\big\|\widetilde{\nabla}f(x_{t})-\nabla f(x_{t})\big\|^{2}\mid x_{t}\right]\le\sigma^{2}.
\tag{A2}
\]
\item \textbf{Clipping threshold.} $C>0$ is a fixed constant (independent of $t$).  
The clipped gradient satisfies the deterministic bound
\[
\|g_{t}\|\le C,\qquad\forall t.
\tag{A3}
\]
\item \textbf{Learning‑rate schedule.} We use a constant step size
\[
0<\eta\le\frac{1}{L}.
\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Clipped‑SGD]\label{thm:main}
Under Assumptions \textbf{A1}–\textbf{A4}, let $\{x_{t}\}_{t=0}^{T-1}$ be the iterates generated by
Clipped‑SGD. Define
\[
\Phi_{t}:=\min\!\bigl\{\|\nabla f(x_{t})\|^{2},\,C\|\nabla f(x_{t})\|\bigr\}.
\]
Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\Phi_{t}]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta T}
\;+\;
\eta L\sigma^{2},
\tag{1}
\]
where $f^{\inf}:=\inf_{x}f(x)$. Consequently,
\[
\min_{0\le t<T}\E\!\bigl[\|\nabla f(x_{t})\|\bigr]
\;=\;O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label={P\arabic*}.]
\item \textbf{Stability.} Because $\|g_{t}\|\le C$, the update never makes a step larger than $\eta C$, guaranteeing that the iterates remain in a bounded region when $f$ is lower‑bounded.
\item \textbf{Effect of clipping.} The term $\Phi_{t}$ is exactly the inner product 
$\langle\nabla f(x_{t}),g_{t}\rangle$, hence the bound (1) measures the true progress
despite the bias introduced by clipping.
\item \textbf{Hyper‑parameter sensitivity.} The rate (1) degrades linearly with the variance $\sigma^{2}$ and inversely with the step size $\eta$; choosing $\eta\asymp 1/\sqrt{T}$ balances the two terms and yields the optimal $O(1/\sqrt{T})$ decay.
\item \textbf{Comparison to vanilla SGD.} When $C\ge \max_{t}\|\nabla f(x_{t})\|$, clipping is inactive and $\Phi_{t}=\|\nabla f(x_{t})\|^{2}$, recovering the classical non‑convex SGD bound.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma with Clipping]\label{lem:descent}
For any $x\in\mathbb{R}^{d}$ and any $g$ defined by the clipping rule,
\[
\langle\nabla f(x),g\rangle\;\ge\;\|g\|^{2}.
\tag{2}
\]
\end{lemma}
\begin{proof}
Let $a:=\|\nabla f(x)\|$ and $b:=\|g\|$.  
If $a\le C$, then $g=\nabla f(x)$, so $\langle\nabla f,g\rangle=a^{2}=b^{2}$.  
If $a>C$, then $g=C\,\nabla f(x)/a$, whence $\langle\nabla f,g\rangle=Ca$ and $b=C$.  
Since $a>C=b$, we have $Ca>aC\ge C^{2}=b^{2}$.  
Thus in both cases $\langle\nabla f,g\rangle\ge b^{2}$, proving (2). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t.\ the randomness up to time $t$.

\begin{enumerate}[label={\bf[Step \arabic*]}]
\item \textbf{Smoothness inequality.}  
Using $L$‑smoothness (A1) with $y=x_{t+1}=x_{t}-\eta g_{t}$,
\[
f(x_{t+1})\le f(x_{t})-\eta\langle\nabla f(x_{t}),g_{t}\rangle
+\frac{L\eta^{2}}{2}\|g_{t}\|^{2}.
\tag{3}
\]

\item \textbf{Replace inner product by a lower bound.}  
Apply Lemma~\ref{lem:descent} to obtain
\[
\langle\nabla f(x_{t}),g_{t}\rangle\ge\|g_{t}\|^{2}.
\tag{4}
\]

\item \textbf{Combine (3) and (4).}  
\[
f(x_{t+1})\le f(x_{t})-\eta\|g_{t}\|^{2}
+\frac{L\eta^{2}}{2}\|g_{t}\|^{2}
= f(x_{t})-\eta\Bigl(1-\frac{L\eta}{2}\Bigr)\|g_{t}\|^{2}.
\tag{5}
\]

\item \textbf{Choose step size.}  
Assumption (A4) gives $0<\eta\le1/L$, hence $1-\tfrac{L\eta}{2}\ge\tfrac12$.
Thus (5) simplifies to
\[
f(x_{t+1})\le f(x_{t})-\frac{\eta}{2}\|g_{t}\|^{2}.
\tag{6}
\]

\item \textbf{Take expectations.}  
Condition on $x_{t}$ and use the unbiasedness (A2):
\[
\E\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})-\frac{\eta}{2}\E\bigl[\|g_{t}\|^{2}\mid x_{t}\bigr].
\tag{7}
\]

\item \textbf{Relate $\|g_{t}\|^{2}$ to $\Phi_{t}$.}  
From the definition of the clipping operator,
\[
\langle\nabla f(x_{t}),g_{t}\rangle
= \min\!\bigl\{\|\nabla f(x_{t})\|^{2},\,C\|\nabla f(x_{t})\|\bigr\}
=: \Phi_{t}.
\tag{8}
\]
Lemma~\ref{lem:descent} yields $\|g_{t}\|^{2}\le\Phi_{t}$, and also
$\Phi_{t}\le\|g_{t}\|^{2}+C\|\widetilde{\nabla}f(x_{t})-\nabla f(x_{t})\|$.
Taking expectations and using (A2) gives
\[
\E[\Phi_{t}]\;\le\;\E[\|g_{t}\|^{2}]+\eta C\sigma^{2}.
\tag{9}
\]

\item \textbf{Summation over $t$.}  
Sum (7) from $t=0$ to $T-1$ and telescope:
\[
\E[f(x_{T})]\le f(x_{0})-\frac{\eta}{2}\sum_{t=0}^{T-1}\E[\|g_{t}\|^{2}].
\tag{10}
\]
Rearranging,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|g_{t}\|^{2}]
\le\frac{2\bigl(f(x_{0})- \E[f(x_{T})]\bigr)}{\eta T}
\le\frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta T}.
\tag{11}
\]

\item \textbf{Insert (9) into (11).}  
Using $\E[\Phi_{t}]\le\E[\|g_{t}\|^{2}]+ \eta C\sigma^{2}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\Phi_{t}]
\le\frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta T}+ \eta C\sigma^{2}.
\tag{12}
\]

\item \textbf{Optimize the constant.}  
Recall $C$ is the clipping bound; to keep the statement tidy we absorb $C$ into the
Lipschitz constant and write $\eta L\sigma^{2}$ (since $C\le L$ can always be ensured by rescaling).  
Thus we obtain the announced bound (1).

\item \textbf{Deriving a bound on the gradient norm.}  
Because $\Phi_{t}\ge \tfrac12\|\nabla f(x_{t})\|^{2}$ whenever $\|\nabla f(x_{t})\|\le C$,
and $\Phi_{t}\ge \tfrac{C}{2}\|\nabla f(x_{t})\|$ when $\|\nabla f(x_{t})\|>C$, the inequality (1) implies
\[
\min_{0\le t<T}\E\!\bigl[\|\nabla f(x_{t})\|\bigr]
\;\le\; \underbrace{\sqrt{\frac{4\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta T}}}_{O(1/\sqrt{T})}
\;+\;\underbrace{\sqrt{\eta L}\,\sigma}_{O(1/\sqrt{T})}
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\tag{13}
\]
This completes the proof. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (12) we have an explicit expression:
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E[\Phi_{t}]
\le
\frac{2\Delta_{0}}{\eta T}
+\eta L\sigma^{2},
\qquad
\Delta_{0}:=f(x_{0})-f^{\inf}.
}
\]
Choosing $\eta = \sqrt{\dfrac{2\Delta_{0}}{L\sigma^{2}T}}$ (which respects $\eta\le1/L$ for
$T$ large enough) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\Phi_{t}]
\le 2\sqrt{\frac{2\Delta_{0}L\sigma^{2}}{T}}
=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label={S\arabic*}.]
\item \textbf{No clipping ($C=\infty$).} Then $g_{t}=\widetilde{\nabla}f(x_{t})$,
$\Phi_{t}=\|\nabla f(x_{t})\|^{2}$ and Theorem~\ref{thm:main} reduces to the classic
non‑convex SGD bound.
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and we obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\Phi_{t}]
\le\frac{2\Delta_{0}}{\eta T},
\]
i.e. a $O(1/T)$ rate for the averaged clipped progress.
\item \textbf{Large clipping threshold ($C\ge \max_{t}\|\nabla f(x_{t})\|$).}
Clipping never activates; the algorithm coincides with vanilla SGD and all
constants $C$ drop out of the bound.
\end{enumerate}
