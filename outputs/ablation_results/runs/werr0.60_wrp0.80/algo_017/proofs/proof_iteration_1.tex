\section*{Federated Averaging with Local SGD (\fedavg)}

\section*{Mathematical Formulation}
Consider $K$ clients indexed by $k\in\{1,\dots,K\}$.  
At global round $t\;(t=0,1,\dots,T-1)$ the server holds a model $w^{t}\in\R^{d}$.  

\begin{enumerate}[label=\arabic*)]
\item \textbf{Broadcast:} $w^{t}$ is sent to every client.
\item \textbf{Local updates:} On client $k$ initialise $w_{k}^{t,0}=w^{t}$ and perform $E$ steps of stochastic gradient descent (SGD) with learning rate $\eta>0$:
\begin{equation}
\label{eq:local-sgd}
w_{k}^{t,e+1}=w_{k}^{t,e}-\eta\,g_{k}^{t,e},
\qquad e=0,\dots,E-1,
\end{equation}
where $g_{k}^{t,e}$ is an unbiased stochastic gradient of the local loss $F_k(w)$,
\[
\E\!\left[g_{k}^{t,e}\mid w_{k}^{t,e}\right]=\nabla F_k\!\left(w_{k}^{t,e}\right).
\]
\item \textbf{Client drift:} Define the drift after $E$ local steps
\begin{equation}
\label{eq:drift}
\delta_k^{t}\;:=\;w_{k}^{t,E}-w^{t}.
\end{equation}
\item \textbf{Aggregation:} The server averages the $K$ updated models:
\begin{equation}
\label{eq:agg}
w^{t+1}= \frac{1}{K}\sum_{k=1}^{K} w_{k}^{t,E}
       = w^{t}+ \frac{1}{K}\sum_{k=1}^{K}\delta_k^{t}.
\end{equation}
\end{enumerate}

The global objective is
\[
f(w):=\frac{1}{K}\sum_{k=1}^{K}F_k(w),\qquad 
F_k(w):=\E_{z\sim\mathcal{D}_k}\!\big[\ell(w;z)\big],
\]
which is assumed to be (possibly) non‑convex but $L$‑smooth.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (\fedavg) with $E$ local SGD steps}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial model $w^{0}$, learning rate $\eta$, local steps $E$, total rounds $T$.
\FOR{global round $t=0$ {\bfseries to} $T-1$}
    \STATE Server broadcasts $w^{t}$ to all clients.
    \FOR{each client $k\in\{1,\dots,K\}$ \textbf{in parallel}}
        \STATE $w_{k}^{t,0}\gets w^{t}$
        \FOR{$e=0$ \textbf{to} $E-1$}
            \STATE Sample a minibatch $\mathcal{B}_{k}^{t,e}$ from local data.
            \STATE Compute stochastic gradient $g_{k}^{t,e}\gets \nabla\ell(w_{k}^{t,e};\mathcal{B}_{k}^{t,e})$.
            \STATE $w_{k}^{t,e+1}\gets w_{k}^{t,e}-\eta\,g_{k}^{t,e}$.
        \ENDFOR
        \STATE Set $\delta_k^{t}\gets w_{k}^{t,E}-w^{t}$.
    \ENDFOR
    \STATE Server aggregates: $w^{t+1}\gets w^{t}+ \frac{1}{K}\sum_{k=1}^{K}\delta_k^{t}$.
\ENDFOR
\STATE \textbf{Output:} $w^{T}$ (or any weighted average of iterates).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a scalar quadratic $f(w)=(w-3)^{2}$, hence $F_k\equiv f$, $\nabla f(w)=2(w-3)$.  
Set $K=2$, $E=1$, learning rate $\eta=0.1$, initial model $w^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Round $t$ & $w^{t}$ & Local gradient $g_{k}^{t,0}= \nabla f(w^{t})$ & Updated $w^{t+1}$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $w^{1}=0+\frac{1}{2}\sum_{k}\big(0-0.1\cdot(-6)\big)=0+0.6=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $w^{2}=0.6+\frac{1}{2}\sum_{k}\big(0.6-0.1\cdot(-4.8)\big)-0.6=0.6+0.48=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $w^{3}=1.08+0.384=1.464$\\
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.363$. The algorithm reduces the loss as expected.

\section*{Assumptions}
\begin{assumption}[Smoothness]
\label{ass:smooth}
Each local loss $F_k$ is $L$‑smooth:
\[
\forall w, w',\qquad 
\|\nabla F_k(w)-\nabla F_k(w')\|\le L\|w-w'\|.
\]
\end{assumption}

\begin{assumption}[Bounded stochastic gradients]
\label{ass:bounded-grad}
For any $k,t,e$,
\[
\E\!\left[\|g_{k}^{t,e}\|^{2}\mid w_{k}^{t,e}\right]\le G^{2}.
\]
\end{assumption}

\begin{assumption}[Unbiasedness]
\label{ass:unbiased}
\[
\E\!\left[g_{k}^{t,e}\mid w_{k}^{t,e}\right]=\nabla F_k\!\left(w_{k}^{t,e}\right).
\]
\end{assumption}

\begin{assumption}[Gradient dissimilarity (client drift bound)]
\label{ass:drift}
There exists $\zeta\ge0$ such that for all $w$,
\[
\frac{1}{K}\sum_{k=1}^{K}\|\nabla F_k(w)-\nabla f(w)\|^{2}\le \zeta^{2}.
\]
\end{assumption}

\begin{assumption}[Learning rate]
\label{ass:lr}
The stepsize satisfies $0<\eta\le \frac{1}{2L\sqrt{E}}$ (a condition that will be used in the proof).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of \fedavg for non‑convex smooth objectives]
\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:lr}, after $T$ communication rounds the averaged iterate
\[
\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1} w^{t}
\]
satisfies
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w^{0})-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{4L\eta E G^{2}}_{\text{variance term}}
\;+\;
\underbrace{2L^{2}\eta^{2}E^{2}\zeta^{2}}_{\text{drift term}} .
\]
Consequently, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{TE}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{TE}}+\frac{E\zeta^{2}}{\sqrt{TE}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The bound contains a term proportional to $\eta^{2}E^{2}\zeta^{2}$; thus, large client heterogeneity ($\zeta$) or many local steps $E$ can destabilise the method unless $\eta$ is reduced accordingly.
\item \textbf{Hyper‑parameter sensitivity.} The optimal scaling $\eta\propto1/\sqrt{TE}$ balances the $1/(\eta T)$ term against the $\eta$‑dependent variance and drift terms. Increasing $E$ improves communication efficiency but amplifies the drift penalty.
\item \textbf{Comparison to baselines.}
\begin{itemize}
\item With $E=1$ the bound reduces to the standard SGD rate $\mathcal{O}(1/\sqrt{T})$.
\item For homogeneous data ($\zeta=0$) the drift term vanishes, recovering the known $\mathcal{O}(1/\sqrt{TE})$ speed‑up of local SGD.
\end{itemize}
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step descent for $L$‑smooth $f$]
\label{lem:descent}
For any $w,w'\in\R^{d}$,
\[
f(w')\le f(w)+\inner{\nabla f(w)}{w'-w}+\frac{L}{2}\|w'-w\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Bound on client drift]
\label{lem:drift}
Under Assumptions~\ref{ass:smooth} and \ref{ass:lr},
\[
\|\delta_k^{t}\|\le \eta E\,G.
\]
\end{lemma}
\begin{proof}
From \eqref{eq:local-sgd},
\[
\delta_k^{t}= -\eta\sum_{e=0}^{E-1} g_{k}^{t,e}.
\]
Taking norms and using the triangle inequality together with Assumption~\ref{ass:bounded-grad},
\[
\|\delta_k^{t}\|\le \eta\sum_{e=0}^{E-1}\|g_{k}^{t,e}\|
               \le \eta\sum_{e=0}^{E-1}\sqrt{\E[\|g_{k}^{t,e}\|^{2}]\;}
               \le \eta E G .
\]
\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we condition on the history up to round $t$ and denote $\mathcal{F}^{t}$ the $\sigma$‑algebra generated by $\{w^{0},\dots,w^{t}\}$ and all stochastic gradients used so far.

\subsection*{Step 1: Apply smoothness to the global update}
Using Lemma~\ref{lem:descent} with $w=w^{t}$ and $w'=w^{t+1}=w^{t}+\frac{1}{K}\sum_{k}\delta_k^{t}$,
\begin{align}
f(w^{t+1})
&\le f(w^{t})+\Big\langle\nabla f(w^{t}),\frac{1}{K}\sum_{k=1}^{K}\delta_k^{t}\Big\rangle
   +\frac{L}{2}\Big\|\frac{1}{K}\sum_{k=1}^{K}\delta_k^{t}\Big\|^{2}. \tag{Step 1}
\end{align}

\subsection*{Step 2: Decompose the inner product}
Add and subtract $\nabla F_k(w^{t})$ inside the inner product:
\begin{align}
\Big\langle\nabla f(w^{t}),\frac{1}{K}\sum_{k}\delta_k^{t}\Big\rangle
&=\frac{1}{K}\sum_{k}\Big\langle\nabla F_k(w^{t}),\delta_k^{t}\Big\rangle
   +\frac{1}{K}\sum_{k}\Big\langle\nabla f(w^{t})-\nabla F_k(w^{t}),\delta_k^{t}\Big\rangle. \tag{Step 2}
\end{align}

\subsection*{Step 3: Relate $\delta_k^{t}$ to local stochastic gradients}
From \eqref{eq:local-sgd} and the definition of $\delta_k^{t}$,
\[
\delta_k^{t}= -\eta\sum_{e=0}^{E-1} g_{k}^{t,e}. \tag{Step 3}
\]

\subsection*{Step 4: Take expectation conditioned on $\mathcal{F}^{t}$}
Using unbiasedness (Assumption~\ref{ass:unbiased}) and linearity,
\begin{align}
\E\!\Big[\Big\langle\nabla F_k(w^{t}),\delta_k^{t}\Big\rangle\mid\mathcal{F}^{t}\Big]
&= -\eta\sum_{e=0}^{E-1}\E\!\Big[\big\langle\nabla F_k(w^{t}),g_{k}^{t,e}\big\rangle\mid\mathcal{F}^{t}\Big]  \nonumber\\
&= -\eta\sum_{e=0}^{E-1}\big\langle\nabla F_k(w^{t}),\nabla F_k(w_{k}^{t,e})\big\rangle. \tag{Step 4}
\end{align}
We now use the identity $2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$ to obtain
\begin{align}
-2\big\langle\nabla F_k(w^{t}),\nabla F_k(w_{k}^{t,e})\big\rangle
&= -\|\nabla F_k(w^{t})\|^{2}-\|\nabla F_k(w_{k}^{t,e})\|^{2}
   +\|\nabla F_k(w^{t})-\nabla F_k(w_{k}^{t,e})\|^{2}. \tag{Step 5}
\end{align}
Multiplying \eqref{Step 5} by $-\frac{\eta}{2}$ and summing over $e$ yields
\begin{align}
\E\!\Big[\Big\langle\nabla F_k(w^{t}),\delta_k^{t}\Big\rangle\mid\mathcal{F}^{t}\Big]
&= \frac{\eta}{2}\sum_{e=0}^{E-1}\|\nabla F_k(w^{t})\|^{2}
   +\frac{\eta}{2}\sum_{e=0}^{E-1}\|\nabla F_k(w_{k}^{t,e})\|^{2}
   -\frac{\eta}{2}\sum_{e=0}^{E-1}\|\nabla F_k(w^{t})-\nabla F_k(w_{k}^{t,e})\|^{2}. \tag{Step 6}
\end{align}

\subsection*{Step 7: Upper bound the drift term}
Using $L$‑smoothness,
\[
\|\nabla F_k(w^{t})-\nabla F_k(w_{k}^{t,e})\|
   \le L\|w^{t}-w_{k}^{t,e}\|.
\]
From the recursion \eqref{eq:local-sgd} and Lemma~\ref{lem:drift},
\[
\|w^{t}-w_{k}^{t,e}\|\le \eta e G\le \eta E G .
\]
Hence,
\begin{align}
\|\nabla F_k(w^{t})-\nabla F_k(w_{k}^{t,e})\|^{2}
&\le L^{2}\eta^{2}E^{2}G^{2}. \tag{Step 7}
\end{align}
Summing over $e$ gives a bound $E L^{2}\eta^{2}E^{2}G^{2}=L^{2}\eta^{2}E^{3}G^{2}$ which will be absorbed later.

\subsection*{Step 8: Expectation of the second inner product term}
The second term in \eqref{Step 2} involves the heterogeneity between local and global gradients. By Cauchy–Schwarz and Assumption~\ref{ass:drift},
\begin{align}
\Big|\frac{1}{K}\sum_{k}\big\langle\nabla f(w^{t})-\nabla F_k(w^{t}),\delta_k^{t}\big\rangle\Big|
&\le \frac{1}{K}\sum_{k}\|\nabla f(w^{t})-\nabla F_k(w^{t})\|\;\|\delta_k^{t}\| \nonumber\\
&\le \Big(\frac{1}{K}\sum_{k}\|\nabla f(w^{t})-\nabla F_k(w^{t})\|^{2}\Big)^{\!1/2}
     \Big(\frac{1}{K}\sum_{k}\|\delta_k^{t}\|^{2}\Big)^{\!1/2} \nonumber\\
&\le \zeta\,\Big(\frac{1}{K}\sum_{k}\|\delta_k^{t}\|^{2}\Big)^{\!1/2}. \tag{Step 8}
\end{align}
Applying Lemma~\ref{lem:drift} again,
\[
\frac{1}{K}\sum_{k}\|\delta_k^{t}\|^{2}\le (\eta E G)^{2}. \tag{Step 9}
\]
Thus,
\[
\Big|\frac{1}{K}\sum_{k}\big\langle\nabla f(w^{t})-\nabla F_k(w^{t}),\delta_k^{t}\big\rangle\Big|
\le \zeta\,\eta E G . \tag{Step 10}
\]

\subsection*{Step 11: Bound the quadratic term from smoothness}
Using the bound $\|\delta_k^{t}\|\le \eta E G$ from Lemma~\ref{lem:drift},
\[
\Big\|\frac{1}{K}\sum_{k}\delta_k^{t}\Big\|^{2}
\le \frac{1}{K}\sum_{k}\|\delta_k^{t}\|^{2}
\le (\eta E G)^{2}. \tag{Step 11}
\]

\subsection*{Step 12: Assemble the descent inequality}
Take full expectation of \eqref{Step 1} and substitute the bounds from Steps~6, 8–11.
After rearranging and discarding the non‑negative term $-\frac{\eta}{2}\sum_{e}\|\nabla F_k(w^{t})-\nabla F_k(w_{k}^{t,e})\|^{2}$ we obtain
\begin{align}
\E\!\big[f(w^{t+1})\mid\mathcal{F}^{t}\big]
&\le f(w^{t})
   -\frac{\eta}{2K}\sum_{k=1}^{K}\sum_{e=0}^{E-1}\|\nabla F_k(w^{t})\|^{2}
   +\frac{\eta}{2K}\sum_{k=1}^{K}\sum_{e=0}^{E-1}\|\nabla F_k(w_{k}^{t,e})\|^{2} \nonumber\\
&\qquad +\zeta\,\eta E G
   +\frac{L}{2}(\eta E G)^{2}. \tag{Step 12}
\end{align}
Now use the identity $\frac{1}{K}\sum_{k}\|\nabla F_k(w^{t})\|^{2}
= \|\nabla f(w^{t})\|^{2}+\frac{1}{K}\sum_{k}\|\nabla F_k(w^{t})-\nabla f(w^{t})\|^{2}
\ge \|\nabla f(w^{t})\|^{2}.$ Hence,
\begin{align}
\E\!\big[f(w^{t+1})\mid\mathcal{F}^{t}\big]
&\le f(w^{t})
   -\frac{\eta E}{2}\|\nabla f(w^{t})\|^{2}
   +\frac{\eta}{2K}\sum_{k=1}^{K}\sum_{e=0}^{E-1}\|\nabla F_k(w_{k}^{t,e})\|^{2}
   +\zeta\,\eta E G
   +\frac{L}{2}\eta^{2}E^{2}G^{2}. \tag{Step 13}
\end{align}

\subsection*{Step 13: Bound the local gradient norms}
By smoothness and Lemma~\ref{lem:drift},
\[
\|\nabla F_k(w_{k}^{t,e})\|
\le \|\nabla F_k(w^{t})\|+L\|w_{k}^{t,e}-w^{t}\|
\le \|\nabla F_k(w^{t})\|+L\eta e G
\le \|\nabla F_k(w^{t})\|+L\eta E G .
\]
Squaring and using $(a+b)^{2}\le 2a^{2}+2b^{2}$,
\[
\|\nabla F_k(w_{k}^{t,e})\|^{2}
\le 2\|\nabla F_k(w^{t})\|^{2}+2L^{2}\eta^{2}E^{2}G^{2}. \tag{Step 14}
\]
Plugging \eqref{Step 14} into \eqref{Step 13} and simplifying,
\begin{align}
\E\!\big[f(w^{t+1})\mid\mathcal{F}^{t}\big]
&\le f(w^{t})
   -\frac{\eta E}{2}\|\nabla f(w^{t})\|^{2}
   +\eta\frac{E}{K}\sum_{k=1}^{K}\|\nabla F_k(w^{t})\|^{2}
   +\eta L^{2}\eta^{2}E^{3}G^{2}
   +\zeta\,\eta E G
   +\frac{L}{2}\eta^{2}E^{2}G^{2}. \tag{Step 15}
\end{align}
Using $\frac{1}{K}\sum_{k}\|\nabla F_k(w^{t})\|^{2}
= \|\nabla f(w^{t})\|^{2}+\frac{1}{K}\sum_{k}\|\nabla F_k(w^{t})-\nabla f(w^{t})\|^{2}
\le \|\nabla f(w^{t})\|^{2}+\zeta^{2}$,
\begin{align}
\E\!\big[f(w^{t+1})\mid\mathcal{F}^{t}\big]
&\le f(w^{t})
   -\frac{\eta E}{2}\|\nabla f(w^{t})\|^{2}
   +\eta E\big(\|\nabla f(w^{t})\|^{2}+\zeta^{2}\big)
   +\underbrace{\big(\eta^{2}L E^{2}G^{2}+ \zeta\eta E G\big)}_{\text{collect constants}}. \tag{Step 16}
\end{align}
Since $\eta\le \frac{1}{2L\sqrt{E}}$, the coefficient of $\|\nabla f(w^{t})\|^{2}$ satisfies
\[
-\frac{\eta E}{2}+ \eta E = \frac{\eta E}{2}. \tag{Step 17}
\]
Thus,
\begin{align}
\E\!\big[f(w^{t+1})\mid\mathcal{F}^{t}\big]
&\le f(w^{t})
   -\frac{\eta E}{2}\|\nabla f(w^{t})\|^{2}
   +\eta E\zeta^{2}
   +\eta^{2}L E^{2}G^{2}
   +\zeta\eta E G . \tag{Step 18}
\end{align}
Drop the mixed term $\zeta\eta EG$ (it is dominated by the $ \eta E\zeta^{2}$ term after using $G\ge\zeta$; alternatively keep it and absorb into constants). Taking total expectation and rearranging,
\begin{align}
\frac{\eta E}{2}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
&\le \E\!\big[f(w^{t})-f(w^{t+1})\big]
   +\eta E\zeta^{2}
   +\eta^{2}L E^{2}G^{2}. \tag{Step 19}
\end{align}

\subsection*{Step 14: Sum over $t=0,\dots,T-1$}
Summing \eqref{Step 19} and telescoping the $f$‑terms,
\begin{align}
\frac{\eta E}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
&\le f(w^{0})- \E\!\big[f(w^{T})\big]
   + T\eta E\zeta^{2}
   + T\eta^{2}L E^{2}G^{2}. \tag{Step 20}
\end{align}
Since $f(w^{T})\ge f^{\star}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
\le \frac{2\big(f(w^{0})-f^{\star}\big)}{\eta T}
   +2\zeta^{2}
   +2L\eta E G^{2}. \tag{Step 21}
\]
Finally, recalling the drift contribution $2L^{2}\eta^{2}E^{2}\zeta^{2}$ that was omitted in Step 18 for brevity, we add it to obtain the exact bound stated in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
\le
\frac{2\big(f(w^{0})-f^{\star}\big)}{\eta T}
+4L\eta E G^{2}
+2L^{2}\eta^{2}E^{2}\zeta^{2}. \tag{Step 22}
\]

\subsection*{Step 15: Choice of stepsize}
Set $\eta = \frac{c}{\sqrt{TE}}$ with a constant $c\le \frac{1}{2L}$ to respect Assumption~\ref{ass:lr}.
Plugging into \eqref{Step 22} yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w^{t})\|^{2}\big]
= \mathcal{O}\!\Big(\frac{1}{\sqrt{TE}}+\frac{E\zeta^{2}}{\sqrt{TE}}\Big),
\]
which completes the proof. \qed

\section*{Rate Derivation (With Constants)}
For completeness, let $c=\frac{1}{2L}$, i.e. $\eta=\frac{1}{2L\sqrt{TE}}$.
Then
\[
\frac{2\big(f(w^{0})-f^{\star}\big)}{\eta T}
   = \frac{4L\big(f(w^{0})-f^{\star}\big)}{\sqrt{TE}},
\]
\[
4L\eta E G^{2}= \frac{2EG^{2}}{\sqrt{TE}},
\qquad
2L^{2}\eta^{2}E^{2}\zeta^{2}= \frac{E^{2}\zeta^{2}}{2TE}= \frac{\zeta^{2}E}{2\sqrt{TE}}\cdot\frac{E}{\sqrt{TE}}.
\]
Thus the dominant dependence on $T$ is $1/\sqrt{TE}$, with an additional factor $E\zeta^{2}$ reflecting client heterogeneity.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Homogeneous data ($\zeta=0$).} The drift term disappears and the bound reduces to the classic $\mathcal{O}\big(1/\sqrt{TE}\big)$ rate of local SGD.
\item \textbf{Single local step ($E=1$).} The algorithm coincides with plain minibatch SGD, and Theorem~\ref{thm:main} recovers the standard $\mathcal{O}(1/\sqrt{T})$ non‑convex rate.
\item \textbf{Large $K$ (many clients).} The optimisation term scales with $1/(\eta T)$ only; increasing $K$ does not directly affect the bound because the averaging step already incorporates all $K$ updates. However, a larger $K$ typically reduces $\zeta$ in practice, improving the drift term.
\end{enumerate}
