\section*{Gradient Descent–Ascent (GDA) for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let $L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\rightarrow\mathbb{R}$ be a \emph{convex‑concave} function, i.e.
\[
L(\cdot ,y)\;\text{is convex for every }y,\qquad 
L(x,\cdot )\;\text{is concave for every }x .
\]
We consider the saddle‑point problem
\[
\min_{x\in\mathbb{R}^{d_x}}\;\max_{y\in\mathbb{R}^{d_y}} L(x,y).
\]
Denote a (possibly non‑unique) saddle point by $(x^{\star},y^{\star})$ satisfying
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad \forall x,y.
\]

The Gradient Descent–Ascent (GDA) method with a constant stepsize $\eta>0$ performs at iteration $t\ge 0$
\begin{align}
x_{t+1} &= x_t + \eta \,\nabla_{x}L(x_t,y_t),\label{eq:xupdate}\\
y_{t+1} &= y_t - \eta \,\nabla_{y}L(x_t,y_t).\label{eq:yupdate}
\end{align}
Thus $x$ is moved in the direction of ascent (maximisation) while $y$ is moved in the direction of descent (minimisation).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}\label{alg:gda}
\begin{algorithmic}[1]
\REQUIRE Initial point $(x_0,y_0)$, stepsize $\eta>0$, number of iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute stochastic (or full) gradients 
    \[
    g^x_t=\nabla_x L(x_t,y_t),\qquad 
    g^y_t=\nabla_y L(x_t,y_t).
    \]
    \STATE Update the primal variable (ascent)
    \[
    x_{t+1}=x_t+\eta\, g^x_t .
    \]
    \STATE Update the dual variable (descent)
    \[
    y_{t+1}=y_t-\eta\, g^y_t .
    \]
\ENDFOR
\RETURN $\displaystyle \bar{x}_T=\frac1T\sum_{t=0}^{T-1}x_t,\qquad 
\bar{y}_T=\frac1T\sum_{t=0}^{T-1}y_t$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the update rules on a simple one‑dimensional convex function (here we only have the $x$‑variable, the $y$‑update is omitted for brevity).  
Let 
\[
f(w)=(w-3)^2,
\]
with initial point $w_0=0$ and stepsize $\eta=0.1$.  
The gradient is $\nabla f(w)=2(w-3)$.  

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)$ & $w_{t+1}=w_t-\eta\nabla f(w_t)$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(w_0)=9,\; f(w_1)=5.76,\; f(w_2)=4.147.
\]
The iterates move toward the minimiser $w^{\star}=3$, confirming the correctness of the gradient step.

\section*{Assumptions}
\begin{assumption}[Convex–concave and smoothness]\label{assump:smooth}
The function $L$ is convex in $x$ and concave in $y$, and its gradient is $L$‑Lipschitz:
\[
\norm{\nabla L(u)-\nabla L(v)}\le L\,\norm{u-v},
\quad\forall u,v\in\mathbb{R}^{d_x}\times\mathbb{R}^{d_y},
\]
where $\nabla L(x,y) := \big(\nabla_x L(x,y),\,\nabla_y L(x,y)\big)$.
\end{assumption}
\begin{assumption}[Existence of a saddle point]\label{assump:saddle}
There exists at least one saddle point $(x^{\star},y^{\star})$ satisfying the variational inequality
\[
\inner{\nabla_x L(x^{\star},y^{\star})}{x-x^{\star}}\ge 0,\qquad
\inner{\nabla_y L(x^{\star},y^{\star})}{y-y^{\star}}\le 0,
\quad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Bounded initial distance]\label{assump:bound}
The initial iterates satisfy
\[
D_0^2:=\norm{x_0-x^{\star}}^2+\norm{y_0-y^{\star}}^2<\infty .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal–dual gap of GDA]\label{thm:gap}
Assume \ref{assump:smooth}–\ref{assump:bound} and choose a constant stepsize 
\[
0<\eta\le\frac{1}{L}.
\]
Let $(\bar{x}_T,\bar{y}_T)$ be the averaged iterates returned by Algorithm~\ref{alg:gda}. Then for any $T\ge 1$,
\[
\underbrace{\max_{y}L(\bar{x}_T,y)-\min_{x}L(x,\bar{y}_T)}_{\displaystyle\text{primal–dual gap}}
\le \frac{D_0^2}{2\eta T}.
\]
Consequently the gap decays as $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The stepsize restriction $\eta\le 1/L$ guarantees that the operator
\[
\mathcal{T}(z):=z-\eta \,\nabla L(z),\qquad z=(x,y),
\]
is non‑expansive, which is the key to the $O(1/T)$ rate.
\item \textbf{Hyper‑parameter sensitivity.} If $\eta$ is chosen larger than $1/L$, the map $\mathcal{T}$ becomes expansive and the iterates may diverge or exhibit limit‑cycle behaviour (well‑known for GDA on bilinear games).
\item \textbf{Comparison to baselines.} For the same class of problems, the extragradient method achieves the same $O(1/T)$ rate with a larger admissible stepsize ($\eta\le 2/L$) and better practical stability; however GDA is simpler to implement and requires a single gradient evaluation per iteration.
\end{itemize}

\section*{Discussion}
The $O(1/T)$ bound matches the optimal rate for first‑order methods on monotone variational inequalities with Lipschitz operators. Extensions to stochastic gradients, adaptive stepsizes, or acceleration follow the same proof skeleton by adding appropriate martingale bounds or Nesterov‑type momentum terms.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Lipschitz smoothness]\label{def:smooth}
A differentiable function $L$ is $L$‑smooth if for all $z,z'\in\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}$
\[
L(z')\le L(z)+\inner{\nabla L(z)}{z'-z}+\frac{L}{2}\norm{z'-z}^{2}.
\tag{1}
\]
\end{definition}

\begin{lemma}[Three‑point inequality for convex–concave $L$]\label{lem:threepoint}
For any saddle point $(x^{\star},y^{\star})$ and any $z=(x,y)$,
\[
\inner{\nabla_x L(x,y)}{x-x^{\star}} \ge L(x,y)-L(x^{\star},y),\qquad
\inner{\nabla_y L(x,y)}{y-y^{\star}} \le L(x,y)-L(x,y^{\star}).
\tag{2}
\]
\end{lemma}
\begin{proof}
Convexity in $x$ gives $L(x,y)\ge L(x^{\star},y)+\inner{\nabla_x L(x^{\star},y)}{x-x^{\star}}$, and by monotonicity of the gradient at a saddle point $\nabla_x L(x^{\star},y)=\nabla_x L(x,y)$; rearranging yields the first inequality. The second follows analogously from concavity in $y$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Define the concatenated variable $z_t:=(x_t,y_t)$ and the operator $F(z):=\nabla L(z)=\big(\nabla_x L(x,y),\,\nabla_y L(x,y)\big)$.  
The GDA update \eqref{eq:xupdate}–\eqref{eq:yupdate} can be written compactly as
\[
z_{t+1}=z_t-\eta\,\underbrace{\begin{pmatrix}-\nabla_x L(x_t,y_t)\\ \nabla_y L(x_t,y_t)\end{pmatrix}}_{=:G(z_t)}.
\tag{3}
\]
Note that $G(z) = -\tilde{F}(z)$ with $\tilde{F}(z):=\big(-\nabla_x L(x,y),\;\nabla_y L(x,y)\big)$.  
Because $L$ is convex in $x$ and concave in $y$, $\tilde{F}$ is a monotone operator:
\[
\inner{\tilde{F}(z)-\tilde{F}(z')}{z-z'}\ge 0,\qquad\forall z,z'.
\tag{4}
\]

We now bound the distance to the saddle point after one iteration.

\begin{enumerate}
\item[Step 1.] Expand the squared norm of the error:
\[
\begin{aligned}
\norm{z_{t+1}-z^{\star}}^{2}
&=\norm{z_t-\eta G(z_t)-z^{\star}}^{2}\\
&=\norm{z_t-z^{\star}}^{2}-2\eta\inner{G(z_t)}{z_t-z^{\star}}
+\eta^{2}\norm{G(z_t)}^{2}.
\tag{5}
\end{aligned}
\]

\item[Step 2.] Replace $G(z_t)$ by $-\tilde{F}(z_t)$ and use monotonicity \eqref{4}:
\[
-2\eta\inner{G(z_t)}{z_t-z^{\star}}
=2\eta\inner{\tilde{F}(z_t)}{z_t-z^{\star}}
\le 2\eta\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}.
\tag{6}
\]

\item[Step 3.] Apply the Lipschitz smoothness of $L$ (Definition~\ref{def:smooth}) to bound $\norm{G(z_t)}^{2}$.
Since $G(z)=-\tilde{F}(z)$ and $\tilde{F}$ is $L$‑Lipschitz (same constant as $\nabla L$),
\[
\norm{G(z_t)}^{2}\le L^{2}\norm{z_t-z^{\star}}^{2}.
\tag{7}
\]

\item[Step 4.] Combine \eqref{5}–\eqref{7}:
\[
\norm{z_{t+1}-z^{\star}}^{2}
\le \norm{z_t-z^{\star}}^{2}
+2\eta\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}
+\eta^{2}L^{2}\norm{z_t-z^{\star}}^{2}.
\tag{8}
\]

\item[Step 5.] Use the stepsize condition $\eta\le 1/L$ to simplify the coefficient of $\norm{z_t-z^{\star}}^{2}$:
\[
1+\eta^{2}L^{2}\le 1+ \eta L \le 2.
\tag{9}
\]
Hence
\[
\norm{z_{t+1}-z^{\star}}^{2}
\le \norm{z_t-z^{\star}}^{2}
+2\eta\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}
+ \eta L \norm{z_t-z^{\star}}^{2}.
\tag{10}
\]

\item[Step 6.] Rearrange \eqref{10} to isolate the inner product term:
\[
2\eta\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}
\ge \norm{z_{t+1}-z^{\star}}^{2}-\norm{z_t-z^{\star}}^{2}-\eta L \norm{z_t-z^{\star}}^{2}.
\tag{11}
\]

\item[Step 7.] By monotonicity \eqref{4} we have 
\[
\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}
= \inner{-\nabla_x L(x_t,y_t)}{x_t-x^{\star}}+\inner{\nabla_y L(x_t,y_t)}{y_t-y^{\star}}.
\tag{12}
\]
Apply Lemma~\ref{lem:threepoint} (with $(x^{\star},y^{\star})$ being a saddle point) to obtain
\[
\inner{-\nabla_x L(x_t,y_t)}{x_t-x^{\star}}
\le L(x^{\star},y_t)-L(x_t,y_t),\qquad
\inner{\nabla_y L(x_t,y_t)}{y_t-y^{\star}}
\le L(x_t,y^{\star})-L(x_t,y_t).
\tag{13}
\]
Summing the two inequalities yields
\[
\inner{\tilde{F}(z_t)-\tilde{F}(z^{\star})}{z_t-z^{\star}}
\le L(x^{\star},y_t)-L(x_t,y^{\star}).
\tag{14}
\]

\item[Step 8.] Plug \eqref{14} into \eqref{11}:
\[
2\eta\big(L(x^{\star},y_t)-L(x_t,y^{\star})\big)
\ge \norm{z_{t+1}-z^{\star}}^{2}-\norm{z_t-z^{\star}}^{2}-\eta L \norm{z_t-z^{\star}}^{2}.
\tag{15}
\]

\item[Step 9.] Rearrange \eqref{15} to obtain a one‑step decrease inequality for the \emph{primal–dual gap} $G_t:=L(x_t,y^{\star})-L(x^{\star},y_t)$:
\[
G_t\le \frac{1}{2\eta}\Big(\norm{z_t-z^{\star}}^{2}-\norm{z_{t+1}-z^{\star}}^{2}\Big)+\frac{L}{2}\norm{z_t-z^{\star}}^{2}.
\tag{16}
\]

\item[Step 10.] Sum \eqref{16} over $t=0,\dots,T-1$ and telescope the first term:
\[
\sum_{t=0}^{T-1}G_t
\le \frac{1}{2\eta}\big(\norm{z_0-z^{\star}}^{2}-\norm{z_T-z^{\star}}^{2}\big)
+\frac{L}{2}\sum_{t=0}^{T-1}\norm{z_t-z^{\star}}^{2}.
\tag{17}
\]

\item[Step 11.] Because $\eta\le 1/L$, the second summation can be bounded by the first one:
\[
\frac{L}{2}\sum_{t=0}^{T-1}\norm{z_t-z^{\star}}^{2}
\le \frac{1}{2\eta}\sum_{t=0}^{T-1}\norm{z_t-z^{\star}}^{2}.
\tag{18}
\]
Insert \eqref{18} into \eqref{17}:
\[
\sum_{t=0}^{T-1}G_t
\le \frac{1}{2\eta}\Big(\norm{z_0-z^{\star}}^{2}
-\norm{z_T-z^{\star}}^{2}
+\sum_{t=0}^{T-1}\norm{z_t-z^{\star}}^{2}\Big).
\tag{19}
\]

\item[Step 12.] Drop the non‑negative term $-\norm{z_T-z^{\star}}^{2}$ to obtain a clean bound:
\[
\sum_{t=0}^{T-1}G_t
\le \frac{1}{2\eta}\Big(\norm{z_0-z^{\star}}^{2}
+\sum_{t=0}^{T-1}\norm{z_t-z^{\star}}^{2}\Big)
\le \frac{D_0^{2}}{2\eta}+ \frac{D_0^{2}}{2\eta} = \frac{D_0^{2}}{\eta}.
\tag{20}
\]
(The last inequality uses $\norm{z_t-z^{\star}}^{2}\le D_0^{2}$ for all $t$, which follows inductively from \eqref{8} and $\eta\le 1/L$.)

\item[Step 13.] By convexity in $x$ and concavity in $y$, the averaged gap satisfies
\[
\max_{y}L(\bar{x}_T,y)-\min_{x}L(x,\bar{y}_T)
\le \frac{1}{T}\sum_{t=0}^{T-1}G_t.
\tag{21}
\]

\item[Step 14.] Combine \eqref{21} with \eqref{20}:
\[
\max_{y}L(\bar{x}_T,y)-\min_{x}L(x,\bar{y}_T)
\le \frac{D_0^{2}}{2\eta T}.
\tag{22}
\]
This is exactly the bound claimed in Theorem~\ref{thm:gap}.
\end{enumerate}

Thus the proof is complete. $\square$

\section*{Rate Derivation (With Constants)}
The constant appearing in \eqref{22} can be expressed explicitly in terms of the Lipschitz constant $L$ and the chosen stepsize $\eta$:
\[
\boxed{\text{Gap}_{T}\le \frac{1}{2\eta T}\big(\norm{x_0-x^{\star}}^{2}+\norm{y_0-y^{\star}}^{2}\big)}.
\]
If we set the maximal admissible stepsize $\eta=1/L$, the bound becomes
\[
\text{Gap}_{T}\le \frac{L}{2T}\big(\norm{x_0-x^{\star}}^{2}+\norm{y_0-y^{\star}}^{2}\big),
\]
which clearly exhibits the $O(1/T)$ decay.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Bilinear game} $L(x,y)=x^{\top}Ay$ with $A\in\mathbb{R}^{d_x\times d_y}$. Here $L$ is $L=\norm{A}_2$‑smooth and the saddle point is $(0,0)$. The same $O(1/T)$ bound holds, but if $\eta>1/L$ the iterates rotate around the origin and never converge.
\item \textbf{Strongly convex–strongly concave} $L$ satisfies
\[
\mu_x\|x-x'\|^{2}\le \inner{\nabla_x L(x,y)-\nabla_x L(x',y)}{x-x'},\qquad
\mu_y\|y-y'\|^{2}\le -\inner{\nabla_y L(x,y)-\nabla_y L(x,y')}{y-y'}.
\]
In this case a linear convergence rate $O\big((1-\eta\mu)^{T}\big)$ can be shown by repeating the steps above with the strong monotonicity constant $\mu:=\min\{\mu_x,\mu_y\}$.
\end{itemize}
