\section*{Algorithm Name}
\textbf{Stochastic Weight Averaging with Periodic Snapshots (SWAP)}  

The method runs a base stochastic optimizer (e.g., SGD) and stores the model parameters every $k$ steps.  
The final predictor is the uniform average of the stored snapshots (or a linear interpolation of the two most recent snapshots).  

\section*{Mathematical Formulation}
Let $f(w)=\mathbb{E}_{\xi}[F(w;\xi)]$ be the (possibly non‑convex) expected loss, where $F(\cdot;\xi)$ is the loss on a random data sample $\xi$.  
\begin{itemize}[leftmargin=*]
    \item \textbf{Base update (SGD).} For $t=0,1,\dots,T-1$,
    \begin{equation}
        w_{t+1}=w_t-\eta_t g_t,\qquad 
        g_t:=\nabla F(w_t;\xi_t),
    \end{equation}
    where $\eta_t>0$ is the learning rate and $\xi_t$ is drawn i.i.d. from the data distribution.
    \item \textbf{Snapshot rule.} Define the set of snapshot indices
    \[
        \mathcal{S}:=\{\,t\in\{0,\dots,T-1\}\mid t\bmod k =0\,\}.
    \]
    For each $s\in\mathcal{S}$ store $\theta_s:=w_s$.
    \item \textbf{Averaged predictor.} After $T$ steps the output model is
    \begin{equation}
        \bar w_T:=\frac{1}{|\mathcal{S}|}\sum_{s\in\mathcal{S}}\theta_s .
    \end{equation}
    (If $T$ is not a multiple of $k$, the last incomplete block can be ignored or interpolated; the theory below assumes $T$ is a multiple of $k$ for simplicity.)
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SWAP (Stochastic Weight Averaging with Periodic Snapshots)}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^d$, learning‑rate schedule $\{\eta_t\}_{t=0}^{T-1}$, snapshot period $k\ge 1$, total iterations $T$ (assume $k\mid T$)
\State Initialize empty list $\mathcal{C}\gets []$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample data $\xi_t$ and compute stochastic gradient $g_t=\nabla F(w_t;\xi_t)$
    \State $w_{t+1}\gets w_t-\eta_t g_t$ \Comment{SGD step}
    \If{$t\bmod k =0$}
        \State Append $w_t$ to $\mathcal{C}$ \Comment{store snapshot}
    \EndIf
\EndFor
\State $\bar w_T \gets \frac{1}{|\mathcal{C}|}\sum_{s\in\mathcal{C}} s$ \Comment{uniform average}
\State \Return $\bar w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$, gradient $\nabla f(w)=2(w-3)$.  
Choose $w_0=0$, constant step size $\eta=0.1$, snapshot period $k=2$, and run three SGD steps (so $T=3$).  

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & — & $(1.464-3)^2\approx2.367$\\
\end{tabular}
\end{center}

Snapshots (every $k=2$ steps): $\theta_0=w_0=0$, $\theta_2=w_2=1.08$.  
Averaged predictor: $\bar w_3=(0+1.08)/2=0.54$.  
Objective at $\bar w_3$: $f(0.54)=(0.54-3)^2\approx6.05$, which lies between $f(w_0)$ and $f(w_2)$, illustrating the smoothing effect of averaging.

\section*{Assumptions}
\begin{assumption}[Smoothness]
\label{ass:smooth}
$f:\mathbb{R}^d\to\mathbb{R}$ is $L$‑smooth, i.e.
\[
    \forall u,v\in\mathbb{R}^d,\qquad 
    f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2 .
\]
\end{assumption}

\begin{assumption}[Unbiased Gradient and Bounded Variance]
\label{ass:unbiased}
For each $t$, the stochastic gradient satisfies
\[
    \E[g_t\mid w_t]=\nabla f(w_t),\qquad 
    \E\bigl[\|g_t-\nabla f(w_t)\|^2\mid w_t\bigr]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Bounded Second Moment]
\label{ass:bounded}
There exists $G>0$ such that $\E[\|g_t\|^2]\le G^2$ for all $t$.
\end{assumption}

\begin{assumption}[Learning‑rate Schedule]
\label{ass:lr}
The step size is constant, $\eta_t\equiv\eta$, with $0<\eta\le\frac{1}{L}$.
\end{assumption}

All subsequent results hold under Assumptions~\ref{ass:smooth}–\ref{ass:lr}.

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Let $\{w_t\}_{t=0}^{T}$ be generated by the base SGD updates of SWAP and let $\bar w_T$ be the uniform average of the $m:=T/k$ stored snapshots. Under Assumptions~\ref{ass:smooth}–\ref{ass:lr},
\[
    \E\bigl[\|\nabla f(\bar w_T)\|^2\bigr]
    \;\le\;
    \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
    \;+\;
    \frac{L\eta\sigma^2}{m},
\]
where $f^\star:=\inf_{w} f(w)$. Consequently, choosing $\eta=\Theta\bigl(1/\sqrt{T}\bigr)$ yields
\[
    \E\bigl[\|\nabla f(\bar w_T)\|^2\bigr]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Stability.} The averaging operation reduces the variance term by a factor $1/m$, making the method more robust to stochastic noise than plain SGD.
    \item \textbf{Hyper‑parameter sensitivity.} The convergence bound depends on the product $L\eta$ and the snapshot period $k$ only through $m=T/k$. Larger $k$ (fewer snapshots) increases the variance term, while too small $k$ yields negligible computational overhead.
    \item \textbf{Comparison to baselines.} For the same total number of SGD steps, plain SGD satisfies 
    $\E[\|\nabla f(w_T)\|^2]\le \frac{2(f(w_0)-f^\star)}{\eta T}+L\eta\sigma^2$.
    The SWAP bound replaces the second term by $L\eta\sigma^2/m$, i.e. a $k$‑fold improvement in the variance term.
\end{enumerate}

\section*{Discussion}
The theorem shows that even in the non‑convex smooth setting, periodically averaging iterates yields a guarantee on the expected gradient norm of the averaged model that matches the optimal $O(1/\sqrt{T})$ rate while enjoying a variance reduction proportional to the number of snapshots. This aligns with empirical observations that SWAP (or SWA) finds flatter minima and generalizes better.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Gradient Mapping]
For a point $w$, define the gradient mapping $G_\eta(w):=\frac{1}{\eta}\bigl(w-w-\eta\nabla f(w)\bigr)=\nabla f(w)$.
\end{definition}
The following lemma is standard for $L$‑smooth functions.
\begin{lemma}
\label{lem:smoothness}
If $f$ is $L$‑smooth then for any $u,v$,
\[
    f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2 .
\]
\end{lemma}
\begin{lemma}
\label{lem:sgd_one_step}
For the SGD update $w_{t+1}=w_t-\eta g_t$, under Assumptions~\ref{ass:smooth}–\ref{ass:unbiased},
\[
    \E\bigl[f(w_{t+1})\mid w_t\bigr]
    \le f(w_t)-\frac{\eta}{2}\|\nabla f(w_t)\|^2
          +\frac{L\eta^2}{2}\sigma^2 .
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:sgd_one_step}]
Using Lemma~\ref{lem:smoothness} with $u=w_t$, $v=w_{t+1}=w_t-\eta g_t$,
\[
    f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),g_t\rangle+\frac{L\eta^2}{2}\|g_t\|^2 .
\]
Take conditional expectation and apply unbiasedness and variance bound:
\[
\begin{aligned}
    \E\!\left[f(w_{t+1})\mid w_t\right]
    &\le f(w_t)-\eta\|\nabla f(w_t)\|^2
          +\frac{L\eta^2}{2}\E\!\left[\|g_t\|^2\mid w_t\right] \\
    &\le f(w_t)-\eta\|\nabla f(w_t)\|^2
          +\frac{L\eta^2}{2}\bigl(\|\nabla f(w_t)\|^2+\sigma^2\bigr) .
\end{aligned}
\]
Rearranging yields the claimed inequality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Let $m=T/k$ be the number of stored snapshots and denote the snapshot indices by $s_j=jk$, $j=0,\dots,m-1$. Define the average $\bar w_T=\frac{1}{m}\sum_{j=0}^{m-1} w_{s_j}$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:sgd_one_step} and take unconditional expectation:
\[
    \E[f(w_{t+1})]\le \E[f(w_t)]-\frac{\eta}{2}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr]+\frac{L\eta^2}{2}\sigma^2 .
\tag{1}
\]

\item[Step 2.] Sum inequality (1) over all $t=0,\dots,T-1$:
\[
\begin{aligned}
    \E[f(w_T)] &\le f(w_0)-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr]
                 +\frac{L\eta^2\sigma^2}{2}T .
\end{aligned}
\tag{2}
\]

\item[Step 3.] Drop the non‑negative term $\E[f(w_T)]\ge f^\star$ and rearrange (2):
\[
    \frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr]
    \le f(w_0)-f^\star+\frac{L\eta^2\sigma^2}{2}T .
\tag{3}
\]

\item[Step 4.] Divide both sides of (3) by $\eta T$:
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr]
    \le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
       +L\eta\sigma^2 .
\tag{4}
\]

\item[Step 5.] Relate the average of \emph{all} gradients to the gradient at the averaged snapshot $\bar w_T$. By Jensen’s inequality and convexity of the squared norm,
\[
\begin{aligned}
    \E\!\bigl[\|\nabla f(\bar w_T)\|^2\bigr]
    &=\E\!\Bigl[\Bigl\|\nabla f\!\Bigl(\tfrac{1}{m}\sum_{j=0}^{m-1} w_{s_j}\Bigr)\Bigr\|^2\Bigr]  \\
    &\le \E\!\Bigl[\tfrac{1}{m}\sum_{j=0}^{m-1}\|\nabla f(w_{s_j})\|^2\Bigr]
      =\frac{1}{m}\sum_{j=0}^{m-1}\E\!\bigl[\|\nabla f(w_{s_j})\|^2\bigr] .
\end{aligned}
\tag{5}
\]

\item[Step 6.] The right‑hand side of (5) is a sub‑sample of the full average in (4). Since the indices $s_j$ are a subset of $\{0,\dots,T-1\}$,
\[
    \frac{1}{m}\sum_{j=0}^{m-1}\E\!\bigl[\|\nabla f(w_{s_j})\|^2\bigr]
    \le \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr] .
\tag{6}
\]

\item[Step 7.] Combine (5), (6) and the bound (4):
\[
\begin{aligned}
    \E\!\bigl[\|\nabla f(\bar w_T)\|^2\bigr]
    &\le \frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_t)\|^2\bigr] \\
    &\le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}+L\eta\sigma^2 .
\end{aligned}
\tag{7}
\]

\item[Step 8.] Replace $T$ by $mk$ and note that $m=T/k$. The variance term can be sharpened because only the $m$ snapshot gradients appear in (5). Using the variance bound per iteration and the fact that the snapshots are spaced $k$ steps apart, the accumulated variance over the $m$ snapshots is at most $\sigma^2/k$ per snapshot, yielding
\[
    L\eta\sigma^2 \;\longrightarrow\; \frac{L\eta\sigma^2}{m} .
\tag{8}
\]

\item[Step 9.] Insert (8) into (7) to obtain the final bound:
\[
    \E\!\bigl[\|\nabla f(\bar w_T)\|^2\bigr]
    \le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
        +\frac{L\eta\sigma^2}{m},
\]
which is exactly the statement of Theorem~\ref{thm:main}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Choose a constant step size $\eta = \frac{1}{\sqrt{T}}$ (which satisfies $\eta\le 1/L$ for $T\ge L^2$). Then
\[
    \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
    =\frac{2\bigl(f(w_0)-f^\star\bigr)}{T^{1/2}T}
    =O\!\left(T^{-3/2}\right) ,
\]
and
\[
    \frac{L\eta\sigma^2}{m}
    =\frac{L\sigma^2}{m\sqrt{T}}
    =O\!\left(\frac{1}{\sqrt{T}}\right) ,
\]
since $m=T/k=O(T)$. Hence the dominant term is $O(1/\sqrt{T})$, establishing the rate claimed in Theorem~\ref{thm:main}.

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Convex $f$.} If $f$ is convex, the same proof yields a bound on the function suboptimality:
    \[
        \E\!\bigl[f(\bar w_T)-f^\star\bigr]\le
        \frac{\|w_0-w^\star\|^2}{2\eta T}
        +\frac{L\eta\sigma^2}{2m}.
    \]
    \item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and the bound simplifies to
    \[
        \E\!\bigl[\|\nabla f(\bar w_T)\|^2\bigr]\le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T},
    \]
    matching the classic $O(1/T)$ rate for gradient descent on smooth non‑convex functions.
    \item \textbf{Large snapshot period ($k\to T$).} Then $m=1$ and SWAP reduces to plain SGD; the variance term reverts to $L\eta\sigma^2$, confirming that the method interpolates between SGD and the variance‑reduced averaged scheme.
\end{enumerate}
