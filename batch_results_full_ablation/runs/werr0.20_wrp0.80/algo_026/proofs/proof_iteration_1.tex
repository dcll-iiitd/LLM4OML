\section*{Randomized Coordinate Descent with Non‑Uniform Sampling (RCD‑NU)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be convex and continuously differentiable.  
For each coordinate $i\in\{1,\dots,d\}$ let $L_i>0$ be a coordinate‑wise smoothness constant, i.e.

\[
\big| \partial_i f(\mathbf{w}+ \alpha \evec_i)-\partial_i f(\mathbf{w})\big|\le L_i |\alpha|
\qquad\forall \mathbf{w}\in\R^{d},\;\forall \alpha\in\R .
\tag{1}
\]

We are given a probability vector $\pvec=(p_1,\dots,p_d)$ with $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.  
At iteration $k$ we draw a random index $i_k$ according to $\pvec$ and form the unbiased stochastic gradient estimator

\[
\mathbf{g}_k \;:=\; \frac{1}{p_{i_k}}\; \partial_{i_k} f(\mathbf{w}_k)\; \evec_{i_k}\in\R^{d},
\tag{2}
\]

where $\evec_i$ denotes the $i$‑th canonical basis vector.  
The update rule is the standard SGD step

\[
\mathbf{w}_{k+1}= \mathbf{w}_k - \eta\,\mathbf{g}_k ,
\qquad\eta>0 \text{ (stepsize)} .
\tag{3}
\]

Because of the factor $1/p_{i_k}$ the estimator (2) satisfies  

\[
\E\big[\mathbf{g}_k\mid \mathbf{w}_k\big]=\grad f(\mathbf{w}_k) .
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD‑NU}
\begin{algorithmic}[1]
\Require Convex $f$, smoothness constants $\{L_i\}_{i=1}^d$, probability vector $\pvec$, stepsize $\eta$, initial point $\mathbf{w}_0\in\R^d$, max iter $T$.
\For{$k=0,1,\dots,T-1$}
    \State Sample $i_k\in\{1,\dots,d\}$ with $\Pr(i_k=i)=p_i$.
    \State Compute partial gradient $g_{k}^{(i_k)}:=\partial_{i_k} f(\mathbf{w}_k)$.
    \State Form stochastic gradient $\mathbf{g}_k:=\frac{1}{p_{i_k}}g_{k}^{(i_k)}\evec_{i_k}$.
    \State Update $\mathbf{w}_{k+1}:=\mathbf{w}_k-\eta\,\mathbf{g}_k$.
\EndFor
\State \Return $\mathbf{w}_T$ (or the average $\bar{\mathbf{w}}_T:=\frac1T\sum_{k=0}^{T-1}\mathbf{w}_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem $f(w)=(w-3)^2$.
Here $d=1$, $p_1=1$, $L_1=2$, and the partial gradient is $\partial_1 f(w)=2(w-3)$.
Take $\eta=0.1$ and $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_k^{(1)}=2(w_k-3)$ & $w_{k+1}=w_k-\eta\frac{1}{p_1}g_k^{(1)}$ & $f(w_{k+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{as:conv}
$f$ is convex, i.e. for all $\mathbf{u},\mathbf{v}\in\R^{d}$
\[
f(\mathbf{v})\ge f(\mathbf{u})+\ip{\grad f(\mathbf{u})}{\mathbf{v}-\mathbf{u}} .
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Coordinate‑wise Smoothness]\label{as:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that (1) holds.
Define $L_{\max}:=\max_i L_i$.
\tag{A2}
\end{assumption}

\begin{assumption}[Probability Vector]\label{as:prob}
$p_i>0$ for all $i$ and $\sum_{i=1}^{d}p_i=1$.
\tag{A3}
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The stepsize satisfies $0<\eta\le \dfrac{1}{L_{\max}}$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of RCD‑NU]\label{thm:main}
Let $\mathbf{w}^{*}\in\arg\min_{\mathbf{w}}f(\mathbf{w})$ and assume \ref{as:conv}–\ref{as:step}.
Define the weighted norm $\norm{\mathbf{x}}_{P^{-1}}^{2}:=\sum_{i=1}^{d}\dfrac{x_i^{2}}{p_i}$.
Then for any $T\ge1$ the iterates generated by \eqref{3} satisfy  

\[
\E\!\big[\,f(\bar{\mathbf{w}}_{T})-f(\mathbf{w}^{*})\,\big]
\;\le\;
\frac{\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}}{2\eta T},
\qquad 
\bar{\mathbf{w}}_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}\mathbf{w}_{k}.
\tag{5}
\]
Consequently, the expected optimality gap decays as $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The condition $\eta\le 1/L_{\max}$ guarantees that each coordinate update is a contraction in the corresponding smoothness metric; thus the iterates remain bounded.
\item \textbf{Hyperparameter Sensitivity.} The bound \eqref{5} scales linearly with $\eta^{-1}$; a larger stepsize yields a tighter bound up to the stability limit $1/L_{\max}$.
\item \textbf{Comparison to Uniform Sampling.} If $p_i=1/d$ for all $i$, then $\norm{\cdot}_{P^{-1}}^{2}=d\,\norm{\cdot}_{2}^{2}$ and \eqref{5} reduces to the classical $O(d/T)$ rate for uniform coordinate descent. Non‑uniform $p_i$ can exploit cheap coordinates (small $L_i$) by assigning them larger probability, thereby reducing the constant $\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}$.
\end{enumerate}

\section*{Discussion}
The theorem shows that, despite updating a single coordinate per iteration, the algorithm attains the optimal $O(1/T)$ convergence for convex smooth objectives when the stochastic gradient estimator is properly re‑scaled by $1/p_i$. The proof follows the standard SGD analysis but requires careful handling of the weighted norm induced by the sampling distribution.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the estimator]\label{lem:unbiased}
For any $\mathbf{w}\in\R^{d}$,
\[
\E\big[\mathbf{g}\mid \mathbf{w}\big]=\grad f(\mathbf{w}),
\qquad\text{with }\mathbf{g}\text{ defined in }(2).
\]
\end{lemma}
\begin{proof}
\[
\E[\mathbf{g}\mid\mathbf{w}]
= \sum_{i=1}^{d} p_i \frac{1}{p_i}\partial_i f(\mathbf{w})\evec_i
= \sum_{i=1}^{d}\partial_i f(\mathbf{w})\evec_i
= \grad f(\mathbf{w}) .
\tag{6}
\]
\end{proof}

\begin{lemma}[Bound on the second moment]\label{lem:second}
For any $\mathbf{w}$,
\[
\E\!\big[\norm{\mathbf{g}}_{P}^{2}\mid\mathbf{w}\big]
\;\le\;
\sum_{i=1}^{d}\frac{1}{p_i}\big(\partial_i f(\mathbf{w})\big)^{2},
\qquad 
\norm{\mathbf{x}}_{P}^{2}:=\sum_{i=1}^{d}p_i x_i^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Using definition (2),
\[
\norm{\mathbf{g}}_{P}^{2}
= \sum_{i=1}^{d}p_i\Big(\frac{1}{p_i}\partial_i f(\mathbf{w})\mathbf{1}_{\{i=i_k\}}\Big)^{2}
= \sum_{i=1}^{d}\frac{1}{p_i}\big(\partial_i f(\mathbf{w})\big)^{2}\mathbf{1}_{\{i=i_k\}} .
\]
Taking expectation over $i_k$ yields (7).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. the random index $i_k$ conditioned on the history up to iteration $k$; we denote this by $\E_k[\cdot]$.

\noindent\textbf{Step 1 (Distance recursion).}  
From the update \eqref{3} and the definition of the weighted norm,
\begin{align}
\norm{\mathbf{w}_{k+1}-\mathbf{w}^{*}}_{P^{-1}}^{2}
&= \norm{\mathbf{w}_{k}-\eta\mathbf{g}_k-\mathbf{w}^{*}}_{P^{-1}}^{2}\nonumber\\
&= \norm{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}^{2}
     -2\eta\!\ip{\mathbf{g}_k}{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}
     +\eta^{2}\norm{\mathbf{g}_k}_{P^{-1}}^{2}.
\tag{8}
\end{align}
[Step 1]

\noindent\textbf{Step 2 (Taking conditional expectation).}  
Apply $\E_k[\cdot]$ to \eqref{8} and use Lemma~\ref{lem:unbiased}:
\begin{align}
\E_k\!\big[\norm{\mathbf{w}_{k+1}-\mathbf{w}^{*}}_{P^{-1}}^{2}\big]
&= \norm{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}^{2}
   -2\eta\ip{\grad f(\mathbf{w}_{k})}{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}
   +\eta^{2}\E_k\!\big[\norm{\mathbf{g}_k}_{P^{-1}}^{2}\big].
\tag{9}
\end{align}
[Step 2]

\noindent\textbf{Step 3 (Bounding the variance term).}  
From Lemma~\ref{lem:second} and the definition of the $P^{-1}$‑norm,
\[
\E_k\!\big[\norm{\mathbf{g}_k}_{P^{-1}}^{2}\big]
= \sum_{i=1}^{d}\frac{1}{p_i}\big(\partial_i f(\mathbf{w}_{k})\big)^{2}
= \norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}.
\tag{10}
\]
[Step 3]

\noindent\textbf{Step 4 (Using coordinate‑wise smoothness).}  
Coordinate‑wise $L_i$‑smoothness implies for any $\mathbf{u},\mathbf{v}$,
\[
f(\mathbf{v})\le f(\mathbf{u})+\ip{\grad f(\mathbf{u})}{\mathbf{v}-\mathbf{u}}
      +\frac12\sum_{i=1}^{d}L_i (v_i-u_i)^{2}.
\tag{11}
\]
Set $\mathbf{v}=\mathbf{w}_{k}-\eta\mathbf{g}_k$ and use $\mathbf{g}_k$ from (2). Because only coordinate $i_k$ is changed,
\[
(v_i-u_i)^{2}= \eta^{2}\frac{1}{p_{i_k}^{2}}\big(\partial_{i_k}f(\mathbf{w}_{k})\big)^{2}\mathbf{1}_{\{i=i_k\}} .
\]
Taking expectation and employing $p_i\le 1$ yields
\[
\E_k\!\big[f(\mathbf{w}_{k+1})\big]
\le f(\mathbf{w}_{k})-\eta\norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}
    +\frac{\eta^{2}L_{\max}}{2}\norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}.
\tag{12}
\]
[Step 4]

\noindent\textbf{Step 5 (Combining distance and function decrease).}  
From convexity (A1) we have
\[
f(\mathbf{w}_{k})-f(\mathbf{w}^{*})\le
\ip{\grad f(\mathbf{w}_{k})}{\mathbf{w}_{k}-\mathbf{w}^{*}} .
\tag{13}
\]
Multiplying \eqref{13} by $2\eta$ and inserting into \eqref{9} gives
\begin{align}
2\eta\big(f(\mathbf{w}_{k})-f(\mathbf{w}^{*})\big)
&\le
\norm{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}^{2}
-\E_k\!\big[\norm{\mathbf{w}_{k+1}-\mathbf{w}^{*}}_{P^{-1}}^{2}\big]
+\eta^{2}\norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}.
\tag{14}
\end{align}
[Step 5]

\noindent\textbf{Step 6 (Using the stepsize bound).}  
Assumption \ref{as:step} implies $\eta L_{\max}\le1$, hence from \eqref{12}
\[
\E_k\!\big[f(\mathbf{w}_{k+1})\big]
\le f(\mathbf{w}_{k})-\frac{\eta}{2}\norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}.
\tag{15}
\]
Rearranging,
\[
\eta\norm{\grad f(\mathbf{w}_{k})}_{P^{-1}}^{2}
\le 2\big(f(\mathbf{w}_{k})-\E_k[f(\mathbf{w}_{k+1})]\big).
\tag{16}
\]
[Step 6]

\noindent\textbf{Step 7 (Eliminate the gradient norm).}  
Insert \eqref{16} into \eqref{14}:
\begin{align}
2\eta\big(f(\mathbf{w}_{k})-f(\mathbf{w}^{*})\big)
&\le
\norm{\mathbf{w}_{k}-\mathbf{w}^{*}}_{P^{-1}}^{2}
-\E_k\!\big[\norm{\mathbf{w}_{k+1}-\mathbf{w}^{*}}_{P^{-1}}^{2}\big]
+2\big(f(\mathbf{w}_{k})-\E_k[f(\mathbf{w}_{k+1})]\big).
\tag{17}
\end{align}
[Step 7]

\noindent\textbf{Step 8 (Take full expectation and telescope).}  
Taking total expectation $\E[\cdot]$ in \eqref{17} and summing from $k=0$ to $T-1$ yields
\begin{align}
2\eta \sum_{k=0}^{T-1}\E\!\big[f(\mathbf{w}_{k})-f(\mathbf{w}^{*})\big]
&\le
\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}
- \E\!\big[\norm{\mathbf{w}_{T}-\mathbf{w}^{*}}_{P^{-1}}^{2}\big]
+2\big(f(\mathbf{w}_{0})-\E[f(\mathbf{w}_{T})]\big).
\tag{18}
\end{align}
Since the last term is non‑negative we can drop it, obtaining
\[
2\eta \sum_{k=0}^{T-1}\E\!\big[f(\mathbf{w}_{k})-f(\mathbf{w}^{*})\big]
\le
\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}.
\tag{19}
\]
[Step 8]

\noindent\textbf{Step 9 (From the average to a single point).}  
By convexity of $f$, the average iterate $\bar{\mathbf{w}}_{T}$ satisfies
\[
f(\bar{\mathbf{w}}_{T})\le \frac{1}{T}\sum_{k=0}^{T-1} f(\mathbf{w}_{k}) .
\]
Taking expectation and using \eqref{19} gives
\[
\E\!\big[f(\bar{\mathbf{w}}_{T})-f(\mathbf{w}^{*})\big]
\le
\frac{\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}}{2\eta T},
\]
which is exactly the bound \eqref{5}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{5} depends on the weighted distance to optimum:
\[
C:=\frac12\,\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{P^{-1}}^{2}
= \frac12\sum_{i=1}^{d}\frac{\big(w_{0}^{(i)}-w^{* (i)}\big)^{2}}{p_i}.
\]
If the optimal point is sparse and the sampling probabilities allocate larger $p_i$ to coordinates that are likely zero, the constant $C$ can be dramatically smaller than the unweighted Euclidean counterpart $ \frac{d}{2}\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{2}^{2}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$.
Then $\norm{\cdot}_{P^{-1}}^{2}=d\,\norm{\cdot}_{2}^{2}$ and \eqref{5} becomes
\[
\E\!\big[f(\bar{\mathbf{w}}_{T})-f(\mathbf{w}^{*})\big]
\le \frac{d\,\norm{\mathbf{w}_{0}-\mathbf{w}^{*}}_{2}^{2}}{2\eta T},
\]
the classical rate for uniform randomized coordinate descent.

\item \textbf{Importance sampling.} Choose $p_i\propto L_i$.
Then $L_{\max}$ in the stepsize condition can be replaced by $\bar L:=\sum_i p_i L_i$, leading to a potentially larger admissible $\eta$ and a tighter constant $C$.

\item \textbf{Deterministic cyclic coordinate descent.}
If $p_i=1$ for a single coordinate and $0$ for the rest (not allowed by \ref{as:prob}), the analysis collapses to a single‑coordinate gradient step; the bound reduces to the standard 1‑dimensional SGD bound.
\end{itemize}
