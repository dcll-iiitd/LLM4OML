\section*{Algorithm Name}
\textbf{Heavy–Ball Gradient Method (HB)}  

The method augments ordinary gradient descent with a momentum term that uses the previous displacement (not a look‑ahead gradient).

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Given an initial pair $(x_{0},x_{-1})\in\R^{d}\times\R^{d}$ (usually $x_{-1}=x_{0}$), step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the iterates are defined for $k=0,1,2,\dots$ by  

\[
\boxed{
\begin{aligned}
x_{k+1}&=x_{k}+\beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha \nabla f(x_{k})   \tag{HB}\\
s_{k}&\stackrel{\text{def}}{=}x_{k}-x_{k-1}\qquad\text{(displacement)} .
\end{aligned}}
\]

Equivalently,
\[
s_{k+1}=x_{k+1}-x_{k}= \beta s_{k}-\alpha \nabla f(x_{k}). \tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy–Ball Gradient Method}
\begin{algorithmic}[1]
\REQUIRE $f$, step size $\alpha>0$, momentum $\beta\in[0,1)$, 
initial point $x_{0}\in\R^{d}$, optional $x_{-1}=x_{0}$,
maximum iterations $K$.
\STATE $x_{-1}\gets x_{0}$
\FOR{$k=0$ \TO $K-1$}
    \STATE $g_{k}\gets \nabla f(x_{k})$
    \STATE $x_{k+1}\gets x_{k}+ \beta (x_{k}-x_{k-1}) - \alpha g_{k}$   \COMMENT{HB update}
\ENDFOR
\RETURN $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  
\[
f(w)=(w-3)^{2},
\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, $w_{0}=0$, $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_{k}= \nabla f(w_{k})$ & $w_{k+1}=w_{k}+ \beta(w_{k}-w_{k-1})-\alpha g_{k}$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^{2}=2.6244$\\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^{2}=0.821\,.$
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $w_{0}$) to $0.821$ after three iterations.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Lower boundedness.} $f$ has a finite infimum $f_{\inf}>-\infty$.
\item \textbf{$L$–smoothness.} $f$ is differentiable and its gradient is $L$–Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\| \le L\|x-y\|,\qquad\forall x,y\in\R^{d}. \tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{A2}
\]
\item \textbf{Step‑size and momentum constraints.}
\[
0\le\beta<1,\qquad 0<\alpha\le\frac{1-\beta}{L}. \tag{A3}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity of HB for non‑convex $L$‑smooth $f$]\label{thm:hb}
Under assumptions (A1)–(A3), define the Lyapunov sequence  

\[
\Phi_{k}\stackrel{\text{def}}{=}f(x_{k})+\frac{c}{2}\|x_{k}-x_{k-1}\|^{2},
\qquad c\;:=\;\frac{1-\beta}{\alpha}-L>0.
\]

Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\;\le\;
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha(1-\beta)T}. \tag{2}
\]
Consequently $\displaystyle\liminf_{k\to\infty}\|\nabla f(x_{k})\|=0$ and every limit point of $\{x_{k}\}$ is a stationary point of $f$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\alpha\le (1-\beta)/L$ guarantees that the Lyapunov function $\Phi_{k}$ is non‑increasing; thus the iterates remain in a bounded sublevel set of $f$.
\item \textbf{Hyperparameter sensitivity.} Larger $\beta$ accelerates progress when the local curvature is close to $L$, but the admissible $\alpha$ shrinks linearly with $1-\beta$ (see (A3)). The bound (2) shows a trade‑off: the factor $1/(1-\beta)$ appears in the denominator, so aggressive momentum (β close to 1) slows the worst‑case $1/T$ rate unless $\alpha$ is reduced accordingly.
\item \textbf{Comparison to Gradient Descent (GD).} Setting $\beta=0$ recovers standard GD. The bound (2) reduces to $\frac{2(f(x_{0})-f_{\inf})}{\alpha T}$, the classical $O(1/T)$ result for smooth non‑convex GD. Hence HB never worsens the worst‑case rate while potentially offering empirical speed‑up.
\end{itemize}

\section*{Discussion}
The theorem establishes that Heavy‑Ball converges to first‑order stationary points at the same $O(1/T)$ rate as plain gradient descent despite the inclusion of momentum. The proof relies on a simple Lyapunov function that couples function values and the kinetic energy $\|x_{k}-x_{k-1}\|^{2}$. The requirement $\alpha\le (1-\beta)/L$ is tight for the analysis; in practice larger steps are often used with good empirical performance, but the theoretical guarantee would then need additional curvature assumptions (e.g., Polyak–Łojasiewicz condition).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$–smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. 
\]
\end{lemma}

\begin{lemma}[Quadratic expansion]\label{lem:expansion}
For any vectors $a,b\in\R^{d}$ and scalar $\eta>0$,
\[
\|a+b\|^{2}= \|a\|^{2}+2\langle a,b\rangle+\|b\|^{2}. 
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:hb}. All non‑trivial algebraic manipulations are numbered for easy reference.

\medskip
\noindent\textbf{Notation.} $g_{k}\stackrel{\text{def}}{=}\nabla f(x_{k})$, $s_{k}=x_{k}-x_{k-1}$, and recall \eqref{1}:
\[
s_{k+1}= \beta s_{k}-\alpha g_{k}. \tag{3}
\]

\medskip
\noindent\textbf{Step 1 (Smoothness applied to the next iterate).}  
Using Lemma~\ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}=x_{k}+s_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle g_{k},s_{k+1}\rangle+\frac{L}{2}\|s_{k+1}\|^{2}. \tag{4}
\]

\noindent\textbf{Step 2 (Insert the expression of $s_{k+1}$).}  
From \eqref{3},
\[
\langle g_{k},s_{k+1}\rangle
 =\langle g_{k},\beta s_{k}-\alpha g_{k}\rangle
 =\beta\langle g_{k},s_{k}\rangle-\alpha\|g_{k}\|^{2}. \tag{5}
\]

\noindent\textbf{Step 3 (Expand $\|s_{k+1}\|^{2}$).}  
Apply Lemma~\ref{lem:expansion} to $s_{k+1}= \beta s_{k}-\alpha g_{k}$:
\[
\|s_{k+1}\|^{2}= \beta^{2}\|s_{k}\|^{2}-2\alpha\beta\langle s_{k},g_{k}\rangle+\alpha^{2}\|g_{k}\|^{2}. \tag{6}
\]

\noindent\textbf{Step 4 (Plug (5) and (6) into (4)).}  
\[
\begin{aligned}
f(x_{k+1}) 
&\le f(x_{k})+\beta\langle g_{k},s_{k}\rangle-\alpha\|g_{k}\|^{2}
   +\frac{L}{2}\Bigl(\beta^{2}\|s_{k}\|^{2}
        -2\alpha\beta\langle s_{k},g_{k}\rangle
        +\alpha^{2}\|g_{k}\|^{2}\Bigr)\\[2mm]
&= f(x_{k}) 
   +\Bigl(\beta-\alpha L\beta\Bigr)\langle g_{k},s_{k}\rangle
   +\Bigl(\frac{L\alpha^{2}}{2}-\alpha\Bigr)\|g_{k}\|^{2}
   +\frac{L\beta^{2}}{2}\|s_{k}\|^{2}. \tag{7}
\end{aligned}
\]

\noindent\textbf{Step 5 (Add a weighted kinetic term).}  
Define $c>0$ (to be chosen later) and consider the Lyapunov quantity  

\[
\Phi_{k}=f(x_{k})+\frac{c}{2}\|s_{k}\|^{2}. \tag{8}
\]

Compute $\Phi_{k+1}-\Phi_{k}$ using (7) and the definition of $s_{k+1}$:

\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&= f(x_{k+1})-f(x_{k})+\frac{c}{2}\bigl(\|s_{k+1}\|^{2}-\|s_{k}\|^{2}\bigr)\\
&\le \Bigl(\beta-\alpha L\beta\Bigr)\langle g_{k},s_{k}\rangle
   +\Bigl(\frac{L\alpha^{2}}{2}-\alpha\Bigr)\|g_{k}\|^{2}
   +\frac{L\beta^{2}}{2}\|s_{k}\|^{2}\\
&\quad +\frac{c}{2}\Bigl(\beta^{2}\|s_{k}\|^{2}
   -2\alpha\beta\langle s_{k},g_{k}\rangle
   +\alpha^{2}\|g_{k}\|^{2}
   -\|s_{k}\|^{2}\Bigr) \qquad\text{[using (6)]}. \tag{9}
\end{aligned}
\]

\noindent\textbf{Step 6 (Collect coefficients of $\langle g_{k},s_{k}\rangle$, $\|g_{k}\|^{2}$ and $\|s_{k}\|^{2}$).}  
After simple algebra:

\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&\le
\underbrace{\Bigl(\beta-\alpha L\beta -c\alpha\beta\Bigr)}_{\text{coeff. of }\langle g_{k},s_{k}\rangle}\langle g_{k},s_{k}\rangle \\
&\quad
+\underbrace{\Bigl(\frac{L\alpha^{2}}{2}-\alpha+\frac{c\alpha^{2}}{2}\Bigr)}_{\text{coeff. of }\|g_{k}\|^{2}}\|g_{k}\|^{2}\\
&\quad
+\underbrace{\Bigl(\frac{L\beta^{2}}{2}+\frac{c}{2}(\beta^{2}-1)\Bigr)}_{\text{coeff. of }\|s_{k}\|^{2}}\|s_{k}\|^{2}. \tag{10}
\end{aligned}
\]

\noindent\textbf{Step 7 (Choose $c$ to cancel the mixed term).}  
Set the coefficient of $\langle g_{k},s_{k}\rangle$ to zero:

\[
\beta-\alpha L\beta-c\alpha\beta=0
\;\Longrightarrow\;
c=\frac{1-\alpha L}{\alpha}. \tag{11}
\]

Because of (A3), $\alpha L\le 1-\beta<1$, hence $c>0$.

\noindent\textbf{Step 8 (Simplify the remaining coefficients).}  
Insert \eqref{11} into the coefficients of $\|g_{k}\|^{2}$ and $\|s_{k}\|^{2}$.

\[
\begin{aligned}
\text{Coeff}_{\|g_{k}\|^{2}}
&= \frac{L\alpha^{2}}{2}-\alpha+\frac{\alpha}{2}(1-\alpha L)\\
&= -\frac{\alpha}{2}\bigl(1-\alpha L\bigr). \tag{12}
\end{aligned}
\]

\[
\begin{aligned}
\text{Coeff}_{\|s_{k}\|^{2}}
&= \frac{L\beta^{2}}{2}+\frac{1-\alpha L}{2\alpha}\bigl(\beta^{2}-1\bigr)\\
&= -\frac{1-\beta}{2\alpha}\bigl(1-\alpha L\bigr). \tag{13}
\end{aligned}
\]

\noindent\textbf{Step 9 (Descent inequality).}  
Plugging (12)–(13) into (10) yields

\[
\Phi_{k+1}-\Phi_{k}
\le -\frac{\alpha}{2}\bigl(1-\alpha L\bigr)\|g_{k}\|^{2}
   -\frac{1-\beta}{2\alpha}\bigl(1-\alpha L\bigr)\|s_{k}\|^{2}. \tag{14}
\]

Since $0<\alpha\le (1-\beta)/L$ we have $1-\alpha L\ge\beta>0$, therefore both terms on the right‑hand side are non‑positive. Consequently $\{\Phi_{k}\}$ is non‑increasing and bounded below by $f_{\inf}$.

\noindent\textbf{Step 10 (Summation).}  
Sum \eqref{14} over $k=0,\dots,T-1$:

\[
\Phi_{T}-\Phi_{0}
\le -\frac{\alpha}{2}\bigl(1-\alpha L\bigr)\sum_{k=0}^{T-1}\|g_{k}\|^{2}
   -\frac{1-\beta}{2\alpha}\bigl(1-\alpha L\bigr)\sum_{k=0}^{T-1}\|s_{k}\|^{2}. \tag{15}
\]

Drop the non‑positive second sum and use $\Phi_{T}\ge f_{\inf}$:

\[
f_{\inf}-\Phi_{0}
\le -\frac{\alpha}{2}\bigl(1-\alpha L\bigr)\sum_{k=0}^{T-1}\|g_{k}\|^{2}. \tag{16}
\]

Recall $\Phi_{0}=f(x_{0})+\frac{c}{2}\|s_{0}\|^{2}\ge f(x_{0})$. Rearranging (16) gives

\[
\sum_{k=0}^{T-1}\|g_{k}\|^{2}
\le \frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha\bigl(1-\alpha L\bigr)}. \tag{17}
\]

\noindent\textbf{Step 11 (Use the admissible step size).}  
From (A3), $\alpha L\le 1-\beta$, thus $1-\alpha L\ge 1-(1-\beta)=\beta$. Since $\beta<1$, we can bound $1-\alpha L\ge (1-\beta)$. Hence

\[
\sum_{k=0}^{T-1}\|g_{k}\|^{2}
\le \frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha(1-\beta)}. \tag{18}
\]

Dividing by $T$ yields the desired rate \eqref{2}.

\noindent\textbf{Step 12 (Vanishing gradient subsequence).}  
Inequality \eqref{2} implies $\displaystyle\liminf_{k\to\infty}\|g_{k}\|^{2}=0$. Any accumulation point $\bar{x}$ of $\{x_{k}\}$ satisfies $\nabla f(\bar{x})=0$ because $f$ is continuous and $\{x_{k}\}$ stays in a compact sublevel set (by the descent of $\Phi_{k}$). ∎

\section*{Rate Derivation (With Constants)}
The bound \eqref{2} can be written explicitly as  

\[
\boxed{
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\le
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha(1-\beta)T}
}
\]

where:
\begin{itemize}
\item $f(x_{0})-f_{\inf}$ is the initial optimality gap,
\item $\alpha$ is the step size,
\item $\beta$ is the momentum coefficient,
\item $T$ is the number of performed iterations.
\end{itemize}
Thus Heavy‑Ball attains the classical $O(1/T)$ convergence of the squared gradient norm for smooth non‑convex optimization.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{No momentum ($\beta=0$).}  The algorithm reduces to vanilla gradient descent and $c=(1-\alpha L)/\alpha$. The bound \eqref{2} becomes $\frac{2(f(x_{0})-f_{\inf})}{\alpha T}$, matching the standard GD result.
\item \textbf{Exact Polyak step ($\alpha = \frac{1-\beta}{L}$).}  The admissible upper bound in (A3) is attained, yielding $c=0$ and the Lyapunov function collapses to $f(x_{k})$; the descent inequality simplifies to $f(x_{k+1})\le f(x_{k})-\frac{1-\beta}{2L}\|g_{k}\|^{2}$.
\item \textbf{Strongly convex $f$.}  Adding the Polyak–Łojasiewicz inequality $\frac{1}{2}\|\nabla f(x)\|^{2}\ge\mu\bigl(f(x)-f_{\inf}\bigr)$ converts \eqref{2} into a linear convergence rate $f(x_{k})-f_{\inf}=O\bigl((1-\alpha\mu)^{k}\bigr)$, showing that the same analysis can be refined under curvature.
\end{enumerate}
