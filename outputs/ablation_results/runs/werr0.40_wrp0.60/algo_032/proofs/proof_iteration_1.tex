\section*{Algorithm Name}
\textbf{Frank–Wolfe (Conditional Gradient) Method for Non‑Convex Smooth Optimization}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let  

\[
\min_{x\in\mathcal C} \; f(x),
\]

where  

\begin{itemize}
  \item $\mathcal C\subset\mathbb R^{d}$ is a non‑empty compact convex set,
  \item $f:\mathbb R^{d}\rightarrow\mathbb R$ is differentiable and $L$‑smooth on an open set containing $\mathcal C$.
\end{itemize}

For iteration $k=0,1,2,\dots$ define the \emph{linear minimization oracle} (LMO)

\[
s_{k}\;:=\;\arg\min_{s\in\mathcal C}\;\big\langle \nabla f(x_{k}) ,\, s\big\rangle .
\tag{1}
\]

The \emph{Frank–Wolfe gap} at $x_{k}$ is  

\[
\gap(x_{k})\;:=\;\max_{s\in\mathcal C}\;\big\langle \nabla f(x_{k}),\,x_{k}-s\big\rangle
               \;=\;\big\langle \nabla f(x_{k}),\,x_{k}-s_{k}\big\rangle .
\tag{2}
\]

Given a stepsize $\gamma_{k}\in(0,1]$, the update is  

\[
x_{k+1}\;=\;(1-\gamma_{k})\,x_{k}+ \gamma_{k}\,s_{k}
          \;=\;x_{k}+ \gamma_{k}\,(s_{k}-x_{k}).                           
\tag{3}
\]

Typical choices for $\gamma_{k}$ are  

\begin{itemize}
  \item \textbf{Fixed schedule:} $\displaystyle\gamma_{k}= \frac{2}{k+2}$,
  \item \textbf{Line‑search:} $\displaystyle\gamma_{k}= \arg\min_{\gamma\in[0,1]}
          f\big(x_{k}+ \gamma (s_{k}-x_{k})\big)$,
  \item \textbf{Adaptive schedule:} $\displaystyle\gamma_{k}= \min\Big\{1,\,
          \frac{\gap(x_{k})}{L\,\diam^{2}}\Big\}$,
\end{itemize}
where $\diam:=\max_{x,y\in\mathcal C}\|x-y\|$ denotes the diameter of $\mathcal C$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe for Non‑Convex Smooth $f$}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_{0}\in\mathcal C$, stepsize rule $\{\gamma_{k}\}$, 
maximum iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_{k}\gets \nabla f(x_{k})$.
    \State \textbf{LMO:} $s_{k}\gets \arg\min_{s\in\mathcal C}\langle g_{k},s\rangle$.
    \State Compute Frank–Wolfe gap $\gap_{k}\gets \langle g_{k},x_{k}-s_{k}\rangle$.
    \State Choose stepsize $\gamma_{k}$ (e.g. $\gamma_{k}=2/(k+2)$ or adaptive).
    \State Update $x_{k+1}\gets (1-\gamma_{k})x_{k}+ \gamma_{k}s_{k}$.
\EndFor
\State \textbf{Output:} $\displaystyle \hat x = \arg\min_{0\le k\le K-1}\gap(x_{k})$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem  

\[
f(w)=(w-3)^{2},\qquad \mathcal C=[0,5].
\]

The gradient is $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, and use the fixed stepsize $\gamma_{k}=0.1$ for three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})$ & $s_{k}=\arg\min_{s\in[0,5]}\langle\nabla f(w_{k}),s\rangle$ & $w_{k+1}= (1-\gamma)w_{k}+\gamma s_{k}$ \\ \hline
0 & $0$ & $-6$ & $\displaystyle s_{0}=5$ (since $\langle-6,s\rangle$ minimized at largest $s$) & $w_{1}=0.9\cdot0+0.1\cdot5=0.5$ \\
1 & $0.5$ & $-5$ & $s_{1}=5$ & $w_{2}=0.9\cdot0.5+0.1\cdot5=0.95$ \\
2 & $0.95$ & $-4.1$ & $s_{2}=5$ & $w_{3}=0.9\cdot0.95+0.1\cdot5=1.355$ 
\end{tabular}
\end{center}

Objective values: $f(w_{0})=9$, $f(w_{1})=6.25$, $f(w_{2})=4.2025$, $f(w_{3})=2.7025$.  
The Frank–Wolfe gap at each step equals $|\nabla f(w_{k})|\cdot|w_{k}-s_{k}|$ and decreases accordingly.

\section*{Assumptions}
\begin{assumption}[Compact Convex Feasible Set]\label{as:C}
The set $\mathcal C\subset\mathbb R^{d}$ is convex, compact and has finite diameter  

\[
\diam:=\max_{x,y\in\mathcal C}\|x-y\|<\infty .
\]
\end{assumption}

\begin{assumption}[Smoothness]\label{as:L}
The objective $f$ is differentiable on an open neighbourhood of $\mathcal C$ and its gradient is $L$‑Lipschitz, i.e.  

\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathcal C .
\tag{4}
\]
\end{assumption}

\begin{assumption}[Bounded Suboptimality]\label{as:Delta}
Denote $f^{\star}:=\min_{x\in\mathcal C}f(x)$.  The initial optimality gap is finite  

\[
\Delta_{0}:=f(x_{0})-f^{\star}<\infty .
\tag{5}
\]
\end{assumption}

No convexity of $f$ is required; the analysis holds for arbitrary non‑convex $L$‑smooth functions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Frank–Wolfe Gap]\label{thm:main}
Assume \ref{as:C}–\ref{as:Delta}.  Let the stepsizes be  

\[
\gamma_{k}= \min\Big\{1,\;\frac{\gap(x_{k})}{L\,\diam^{2}}\Big\},\qquad k\ge0.
\tag{6}
\]

Then for any integer $K\ge1$ the iterates satisfy  

\[
\min_{0\le k\le K-1}\gap(x_{k})\;\le\;
\sqrt{\frac{2L\diam^{2}\,\Delta_{0}}{K}} .
\tag{7}
\]

Consequently $\gap(x_{k})\to0$ as $k\to\infty$, and any accumulation point of $\{x_{k}\}$ is a stationary point of $f$ on $\mathcal C$.
\end{theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} The update is a convex combination of two points in $\mathcal C$, hence $x_{k}\in\mathcal C$ for all $k$ (by convexity of $\mathcal C$).
  \item \textbf{Hyperparameter Sensitivity.} The adaptive stepsize (6) automatically balances progress (large gap) and curvature (via $L$). Fixed schedules such as $\gamma_{k}=2/(k+2)$ also yield the $O(1/\sqrt{K})$ rate, but with larger constants.
  \item \textbf{Comparison to Baselines.}
    \begin{itemize}
      \item Gradient descent with projection enjoys $O(1/K)$ for non‑convex smooth problems, but requires costly projection onto $\mathcal C$.
      \item Frank–Wolfe avoids projections; the price is a slower $O(1/\sqrt{K})$ rate for the gap, which is optimal among projection‑free methods under the same assumptions.
    \end{itemize}
\end{itemize}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem shows that even without convexity the conditional‑gradient scheme makes the Frank–Wolfe gap vanish at the sublinear rate $O(1/\sqrt{K})$.  The gap is a natural optimality measure for constrained problems because it upper‑bounds the primal suboptimality when $f$ is convex, and it coincides with the norm of the projected gradient when $\mathcal C$ is a Euclidean ball.  Thus the method is suitable for large‑scale non‑convex learning tasks where projections are expensive but a linear oracle is cheap (e.g., structured sparsity, low‑rank matrix sets).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Upper Bound]\label{lem:smooth}
If $f$ satisfies \ref{as:L}, then for any $x,y\in\mathcal C$

\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; integrate the gradient along the line segment $[x,y]$.
\end{proof}

\begin{lemma}[Diameter Bound]\label{lem:diam}
For any $x\in\mathcal C$ and any $s\in\mathcal C$ we have  

\[
\|s-x\|\le\diam .
\tag{9}
\]
\end{lemma}
\begin{proof}
Directly from the definition of $\diam$.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we write $\gap_{k}:=\gap(x_{k})$.

\medskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}=x_{k}+\gamma_{k}(s_{k}-x_{k})$:

\[
f(x_{k+1})\le f(x_{k})+\gamma_{k}\langle\nabla f(x_{k}),s_{k}-x_{k}\rangle
            +\frac{L}{2}\gamma_{k}^{2}\|s_{k}-x_{k}\|^{2}.
\tag{10}
\]

\noindent\textbf{[Step 2]} By definition of $s_{k}$ (the LMO) we have  

\[
\langle\nabla f(x_{k}),s_{k}\rangle
   =\min_{s\in\mathcal C}\langle\nabla f(x_{k}),s\rangle,
\]
hence  

\[
\langle\nabla f(x_{k}),x_{k}-s_{k}\rangle
   =\max_{s\in\mathcal C}\langle\nabla f(x_{k}),x_{k}-s\rangle
   =\gap_{k}.
\tag{11}
\]

Thus $\langle\nabla f(x_{k}),s_{k}-x_{k}\rangle=-\gap_{k}$.
Insert (11) into (10):

\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}\gap_{k}
            +\frac{L}{2}\gamma_{k}^{2}\|s_{k}-x_{k}\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 3]} Use Lemma~\ref{lem:diam} to bound the norm term:

\[
\|s_{k}-x_{k}\|^{2}\le\diam^{2}.
\tag{13}
\]

Insert (13) into (12):

\[
f(x_{k+1})\le f(x_{k})-\gamma_{k}\gap_{k}
            +\frac{L}{2}\gamma_{k}^{2}\diam^{2}.
\tag{14}
\]

\noindent\textbf{[Step 4]} Rearrange (14) to isolate the progress term:

\[
\gamma_{k}\gap_{k}\le f(x_{k})-f(x_{k+1})
                   +\frac{L}{2}\gamma_{k}^{2}\diam^{2}.
\tag{15}
\]

\noindent\textbf{[Step 5]} Sum (15) over $k=0,\dots,K-1$:

\[
\sum_{k=0}^{K-1}\gamma_{k}\gap_{k}
\le f(x_{0})-f(x_{K})
   +\frac{L\diam^{2}}{2}\sum_{k=0}^{K-1}\gamma_{k}^{2}.
\tag{16}
\]

Since $f(x_{K})\ge f^{\star}$, we may replace $-f(x_{K})$ by $-f^{\star}$ and use $\Delta_{0}=f(x_{0})-f^{\star}$:

\[
\sum_{k=0}^{K-1}\gamma_{k}\gap_{k}
\le \Delta_{0}
   +\frac{L\diam^{2}}{2}\sum_{k=0}^{K-1}\gamma_{k}^{2}.
\tag{17}
\]

\noindent\textbf{[Step 6]} By the adaptive rule (6), $\gamma_{k}= \min\{1,\gap_{k}/(L\diam^{2})\}$.  
Hence for every $k$ we have two possibilities:

\begin{enumerate}
  \item If $\gap_{k}\le L\diam^{2}$, then $\gamma_{k}= \gap_{k}/(L\diam^{2})$ and
        $\gamma_{k}^{2}= \gap_{k}^{2}/(L^{2}\diam^{4})$.
  \item If $\gap_{k}> L\diam^{2}$, then $\gamma_{k}=1$ and $\gamma_{k}^{2}=1$.
\end{enumerate}

In both cases the inequality  

\[
\gamma_{k}\gap_{k}\;\ge\;\frac{\gap_{k}^{2}}{2L\diam^{2}}
\tag{18}
\]

holds (it is easy to verify by checking the two regimes).  Moreover  

\[
\gamma_{k}^{2}\;\le\;\frac{2\gap_{k}}{L\diam^{2}}.
\tag{19}
\]

\noindent\textbf{[Step 7]} Substitute (18) and (19) into (17):

\[
\sum_{k=0}^{K-1}\frac{\gap_{k}^{2}}{2L\diam^{2}}
\le \Delta_{0}
   +\frac{L\diam^{2}}{2}\sum_{k=0}^{K-1}\frac{2\gap_{k}}{L\diam^{2}}
   =\Delta_{0}+ \sum_{k=0}^{K-1}\gap_{k}.
\tag{20}
\]

\noindent\textbf{[Step 8]} Let $g_{\min}:=\min_{0\le k\le K-1}\gap_{k}$.  Since each $\gap_{k}\ge g_{\min}$, we have  

\[
\sum_{k=0}^{K-1}\gap_{k}\ge K\,g_{\min},\qquad
\sum_{k=0}^{K-1}\gap_{k}^{2}\ge K\,g_{\min}^{2}.
\tag{21}
\]

Insert (21) into (20):

\[
\frac{K g_{\min}^{2}}{2L\diam^{2}}
\le \Delta_{0}+ K\,g_{\min}.
\tag{22}
\]

\noindent\textbf{[Step 9]} Rearrange (22) as a quadratic inequality in $g_{\min}$:

\[
\frac{K}{2L\diam^{2}}\,g_{\min}^{2} - K\,g_{\min} -\Delta_{0}\le 0.
\tag{23}
\]

Dividing by $K$ and solving the quadratic inequality yields  

\[
g_{\min}\le
\frac{L\diam^{2}}{2}
\Bigl(1+\sqrt{1+\frac{2\Delta_{0}}{L\diam^{2}K}}\Bigr)
\;-\;L\diam^{2}.
\tag{24}
\]

A simpler, looser bound follows from $\sqrt{1+u}\le 1+u/2$ for $u\ge0$:

\[
g_{\min}\le \sqrt{\frac{2L\diam^{2}\,\Delta_{0}}{K}} .
\tag{25}
\]

Equation (25) is exactly the claim (7) of Theorem~\ref{thm:main}.  ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
The bound (7) can be written as  

\[
\boxed{\displaystyle
\min_{0\le k\le K-1}\gap(x_{k})\;\le\;
\underbrace{\sqrt{2L\diam^{2}}}_{C}\;\frac{\sqrt{\Delta_{0}}}{\sqrt{K}} }.
\]

Thus the Frank–Wolfe gap decays as $O(1/\sqrt{K})$ with constant $C=\sqrt{2L\diam^{2}}$ that depends only on smoothness and geometry of $\mathcal C$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}
  \item \textbf{Convex $f$.} When $f$ is convex, the Frank–Wolfe gap upper‑bounds the primal suboptimality:
        $f(x_{k})-f^{\star}\le \gap(x_{k})$, hence the same $O(1/\sqrt{K})$ rate holds for the objective value.
  \item \textbf{Polytope $\mathcal C$.} If $\mathcal C$ is a polytope, the LMO reduces to a linear program that can be solved in $O(d)$ time for many structures (e.g., simplex, flow polytope).
  \item \textbf{Quadratic $f$.} For $f(x)=\tfrac12 x^{\top}Qx+b^{\top}x$ with $L=\lambda_{\max}(Q)$, the bound simplifies to
        $\min_{k}\gap(x_{k})\le \sqrt{2\lambda_{\max}(Q)D^{2}\Delta_{0}/K}$.
\end{itemize}
