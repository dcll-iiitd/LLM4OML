\section{Problem Setting and Assumptions}

We consider the unconstrained optimization problem:
\begin{equation}
    \min_{x \in \R^d} f(x)
\end{equation}
where $f: \R^d \to \R$ satisfies the following assumptions.

\begin{assumption}[L-smoothness] \label{assump:smooth}
    The function $f$ is differentiable and its gradient is $L$-Lipschitz continuous for some $L > 0$:
    \begin{equation}
        \norm{\nabla f(x) - \nabla f(y)} \le L \norm{x - y}, \quad \forall x, y \in \R^d.
    \end{equation}
\end{assumption}

\begin{assumption}[Convexity] \label{assump:convex}
    The function $f$ is convex:
    \begin{equation}
        f(y) \ge f(x) + \inner{\nabla f(x), y - x}, \quad \forall x, y \in \R^d.
    \end{equation}
\end{assumption}

\begin{assumption}[Existence of minimizer] \label{assump:minimizer}
    The set of minimizers $X^* = \arg\min_{x \in \R^d} f(x)$ is nonempty. Let $x^* \in X^*$ and $f^* = f(x^*)$.
\end{assumption}

\section{Algorithm}

We analyze the gradient descent algorithm with fixed step size $\eta > 0$:
\begin{equation}
    x_{k+1} = x_k - \eta \nabla f(x_k), \quad k = 0, 1, 2, \dots
\end{equation}

\section{Main Theorem}

\begin{theorem}[Convergence Rate] \label{thm:convergence}
    Under Assumptions \ref{assump:smooth}, \ref{assump:convex}, and \ref{assump:minimizer}, if the step size satisfies $0 < \eta \le \frac{1}{L}$, then for all $k \ge 1$:
    \begin{equation}
        f(x_k) - f^* \le \frac{\norm{x_0 - x^*}^2}{2\eta k}.
    \end{equation}
    In particular, the convergence rate is $O(1/k)$.
\end{theorem}

\section{Proof of Theorem \ref{thm:convergence}}

The proof proceeds through a series of lemmas and inequalities.

\step{1: Descent Lemma}
By $L$-smoothness (Assumption \ref{assump:smooth}), we have the standard descent lemma:
\begin{equation}
    f(y) \le f(x) + \inner{\nabla f(x), y - x} + \frac{L}{2} \norm{y - x}^2, \quad \forall x, y \in \R^d.
\end{equation}
% CORRECTED: This is a well-known consequence of L-smoothness; proof omitted for brevity but can be derived by integrating the gradient Lipschitz condition.

\step{2: One-step progress}
Apply the descent lemma with $x = x_k$ and $y = x_{k+1} = x_k - \eta \nabla f(x_k)$:
\begin{align}
    f(x_{k+1}) &\le f(x_k) + \inner{\nabla f(x_k), -\eta \nabla f(x_k)} + \frac{L}{2} \norm{-\eta \nabla f(x_k)}^2 \\
    &= f(x_k) - \eta \norm{\nabla f(x_k)}^2 + \frac{L\eta^2}{2} \norm{\nabla f(x_k)}^2 \\
    &= f(x_k) - \eta\left(1 - \frac{L\eta}{2}\right) \norm{\nabla f(x_k)}^2.
\end{align}
Since $\eta \le 1/L$, we have $1 - \frac{L\eta}{2} \ge \frac{1}{2} > 0$, so
\begin{equation} \label{eq:descent}
    f(x_{k+1}) \le f(x_k) - \frac{\eta}{2} \norm{\nabla f(x_k)}^2.
\end{equation}

\step{3: Relating gradient norm to suboptimality}
By convexity (Assumption \ref{assump:convex}) with $y = x^*$:
\begin{equation}
    f^* \ge f(x_k) + \inner{\nabla f(x_k), x^* - x_k}.
\end{equation}
Rearranging and using Cauchy-Schwarz:
\begin{equation}
    f(x_k) - f^* \le \inner{\nabla f(x_k), x_k - x^*} \le \norm{\nabla f(x_k)} \norm{x_k - x^*}.
\end{equation}
This gives a lower bound on the gradient norm:
\begin{equation} \label{eq:grad_lower}
    \norm{\nabla f(x_k)} \ge \frac{f(x_k) - f^*}{\norm{x_k - x^*}}.
\end{equation}

\step{4: Distance to optimum does not increase}
We show $\norm{x_{k+1} - x^*}^2 \le \norm{x_k - x^*}^2$. Compute:
\begin{align}
    \norm{x_{k+1} - x^*}^2 &= \norm{x_k - \eta \nabla f(x_k) - x^*}^2 \\
    &= \norm{x_k - x^*}^2 - 2\eta \inner{\nabla f(x_k), x_k - x^*} + \eta^2 \norm{\nabla f(x_k)}^2.
\end{align}
By convexity, $\inner{\nabla f(x_k), x_k - x^*} \ge f(x_k) - f^* \ge 0$. Also, from the descent lemma (Step 1) with $y = x^*$:
\begin{equation}
    f^* \le f(x_k) + \inner{\nabla f(x_k), x^* - x_k} + \frac{L}{2} \norm{x^* - x_k}^2.
\end{equation}
Rearranging: $\inner{\nabla f(x_k), x_k - x^*} \le f(x_k) - f^* + \frac{L}{2} \norm{x_k - x^*}^2$.
However, a simpler argument: since $\eta \le 1/L$, we can use the co-coercivity of the gradient (which follows from smoothness and convexity):
\begin{equation}
    \inner{\nabla f(x_k) - \nabla f(x^*), x_k - x^*} \ge \frac{1}{L} \norm{\nabla f(x_k) - \nabla f(x^*)}^2.
\end{equation}
Since $\nabla f(x^*) = 0$, this gives $\inner{\nabla f(x_k), x_k - x^*} \ge \frac{1}{L} \norm{\nabla f(x_k)}^2$.
Then:
\begin{align}
    \norm{x_{k+1} - x^*}^2 &\le \norm{x_k - x^*}^2 - 2\eta \frac{1}{L} \norm{\nabla f(x_k)}^2 + \eta^2 \norm{\nabla f(x_k)}^2 \\
    &= \norm{x_k - x^*}^2 - \eta\left(\frac{2}{L} - \eta\right) \norm{\nabla f(x_k)}^2.
\end{align}
Because $\eta \le 1/L$, we have $\frac{2}{L} - \eta \ge \frac{1}{L} > 0$, so the distance is non-increasing:
\begin{equation} \label{eq:distance_nonincrease}
    \norm{x_{k+1} - x^*}^2 \le \norm{x_k - x^*}^2 \le \dots \le \norm{x_0 - x^*}^2.
\end{equation}

\step{5: Key inequality combining descent and distance bound}
From \eqref{eq:descent} and \eqref{eq:grad_lower}, and using $\norm{x_k - x^*} \le \norm{x_0 - x^*}$ from \eqref{eq:distance_nonincrease}:
\begin{align}
    f(x_{k+1}) &\le f(x_k) - \frac{\eta}{2} \norm{\nabla f(x_k)}^2 \\
    &\le f(x_k) - \frac{\eta}{2} \frac{(f(x_k) - f^*)^2}{\norm{x_k - x^*}^2} \\
    &\le f(x_k) - \frac{\eta}{2\norm{x_0 - x^*}^2} (f(x_k) - f^*)^2.
\end{align}
Let $\Delta_k = f(x_k) - f^* \ge 0$ and $C = \frac{\eta}{2\norm{x_0 - x^*}^2}$. Then:
\begin{equation} \label{eq:recurrence}
    \Delta_{k+1} \le \Delta_k - C \Delta_k^2.
\end{equation}

\step{6: Solving the recurrence}
We prove by induction that $\Delta_k \le \frac{1}{C k}$ for all $k \ge 1$.
Base case $k=1$: From \eqref{eq:descent}, $\Delta_1 \le \Delta_0 - \frac{\eta}{2}\norm{\nabla f(x_0)}^2 \le \Delta_0$. Also, by convexity and smoothness, one can show $\Delta_0 \le \frac{L}{2}\norm{x_0 - x^*}^2$. With $\eta \le 1/L$, we have $C = \frac{\eta}{2\norm{x_0 - x^*}^2} \le \frac{1}{2L\norm{x_0 - x^*}^2}$. Then $\frac{1}{C} \ge 2L\norm{x_0 - x^*}^2 \ge 4\Delta_0 \ge 4\Delta_1$, so $\Delta_1 \le \frac{1}{4C} \le \frac{1}{C}$. The base case holds.
Inductive step: Assume $\Delta_k \le \frac{1}{C k}$. Then from \eqref{eq:recurrence}:
\begin{align}
    \Delta_{k+1} &\le \frac{1}{C k} - C \left(\frac{1}{C k}\right)^2 = \frac{1}{C k} - \frac{1}{C k^2} = \frac{k-1}{C k^2}.
\end{align}
We need to show $\frac{k-1}{C k^2} \le \frac{1}{C(k+1)}$, i.e., $(k-1)(k+1) \le k^2$, i.e., $k^2 - 1 \le k^2$, which is true. Hence $\Delta_{k+1} \le \frac{1}{C(k+1)}$.

\step{7: Final rate}
Substituting $C = \frac{\eta}{2\norm{x_0 - x^*}^2}$ gives:
\begin{equation}
    f(x_k) - f^* \le \frac{2\norm{x_0 - x^*}^2}{\eta k} \cdot \frac{1}{2}? \text{ Wait, check: } \frac{1}{C k} = \frac{2\norm{x_0 - x^*}^2}{\eta k}.
\end{equation}
But the theorem states $\frac{\norm{x_0 - x^*}^2}{2\eta k}$. There is a discrepancy by a factor of 4. Let's re-examine the recurrence constant.

Actually, from \eqref{eq:descent}: $f(x_{k+1}) \le f(x_k) - \frac{\eta}{2} \norm{\nabla f(x_k)}^2$.
From convexity: $f(x_k) - f^* \le \inner{\nabla f(x_k), x_k - x^*} \le \norm{\nabla f(x_k)} \norm{x_k - x^*}$.
So $\norm{\nabla f(x_k)} \ge \frac{\Delta_k}{\norm{x_k - x^*}}$.
Then $\Delta_{k+1} \le \Delta_k - \frac{\eta}{2} \frac{\Delta_k^2}{\norm{x_k - x^*}^2} \le \Delta_k - \frac{\eta}{2\norm{x_0 - x^*}^2} \Delta_k^2$.
So $C = \frac{\eta}{2\norm{x_0 - x^*}^2}$.
Then $\frac{1}{C k} = \frac{2\norm{x_0 - x^*}^2}{\eta k}$.
But the standard rate is $\frac{\norm{x_0 - x^*}^2}{2\eta k}$ (e.g., Nesterov's book). Where is the factor 4 difference?
Let's check the standard proof: usually one uses $f(x_{k+1}) \le f(x_k) - \frac{1}{2L} \norm{\nabla f(x_k)}^2$ for $\eta = 1/L$. Then with $\eta=1/L$, $C = \frac{1}{2L\norm{x_0 - x^*}^2}$, giving $\frac{2L\norm{x_0 - x^*}^2}{k}$. The standard bound is $\frac{L\norm{x_0 - x^*}^2}{2k}$? Actually, Nesterov (2004) Theorem 2.1.15 gives $f(x_k) - f^* \le \frac{2L\norm{x_0 - x^*}^2}{k+4}$ for $\eta=1/L$. For general $\eta \le 1/L$, the bound is $\frac{\norm{x_0 - x^*}^2}{2\eta k}$? Alternative approach: Use the inequality $f(x_{k+1}) \le f(x_k) - \frac{\eta}{2} \norm{\nabla f(x_k)}^2$ and also the bound $\norm{\nabla f(x_k)}^2 \ge 2L (f(x_k) - f^*)$? No, that's for strongly convex.

Wait, the standard proof for convex smooth uses a different Lyapunov function. Let's use the standard proof to get the correct constant.

\step{8: Standard proof with correct constant}
Consider the following inequality (from smoothness and convexity):
\begin{equation}
    f(y) \le f(x) + \inner{\nabla f(x), y - x} + \frac{L}{2} \norm{y - x}^2.
\end{equation}
Minimizing the RHS over $y$ gives $y = x - \frac{1}{L} \nabla f(x)$ and
\begin{equation}
    f\left(x - \frac{1}{L} \nabla f(x)\right) \le f(x) - \frac{1}{2L} \norm{\nabla f(x)}^2.
\end{equation}
For general $\eta \le 1/L$, we have $f(x - \eta \nabla f(x)) \le f(x) - \frac{\eta}{2} \norm{\nabla f(x)}^2$ (as in Step 2).

Now, by convexity, for any $x^*$:
\begin{equation}
    f(x^*) \ge f(x_k) + \inner{\nabla f(x_k), x^* - x_k}.
\end{equation}
Also, from the update:
\begin{align}
    \norm{x_{k+1} - x^*}^2 &= \norm{x_k - x^*}^2 - 2\eta \inner{\nabla f(x_k), x_k - x^*} + \eta^2 \norm{\nabla f(x_k)}^2 \\
    &\le \norm{x_k - x^*}^2 - 2\eta (f(x_k) - f^*) + \eta^2 \norm{\nabla f(x_k)}^2.
\end{align}
Using the descent inequality $\frac{\eta}{2} \norm{\nabla f(x_k)}^2 \le f(x_k) - f(x_{k+1})$, we get $\eta^2 \norm{\nabla f(x_k)}^2 \le 2\eta (f(x_k) - f(x_{k+1}))$.
Thus:
\begin{align}
    \norm{x_{k+1} - x^*}^2 &\le \norm{x_k - x^*}^2 - 2\eta (f(x_k) - f^*) + 2\eta (f(x_k) - f(x_{k+1})) \\
    &= \norm{x_k - x^*}^2 - 2\eta (f(x_{k+1}) - f^*).
\end{align}
Rearranging:
\begin{equation}
    f(x_{k+1}) - f^* \le \frac{1}{2\eta} \left( \norm{x_k - x^*}^2 - \norm{x_{k+1} - x^*}^2 \right).
\end{equation}
Summing from $k=0$ to $K-1$:
\begin{equation}
    \sum_{k=0}^{K-1} (f(x_{k+1}) - f^*) \le \frac{1}{2\eta} \norm{x_0 - x^*}^2.
\end{equation}
Since $f(x_k)$ is non-increasing (by descent lemma), $f(x_K) - f^* \le \frac{1}{K} \sum_{k=1}^{K} (f(x_k) - f^*) \le \frac{\norm{x_0 - x^*}^2}{2\eta K}$.
This yields the exact constant in the theorem.

\step{9: Conclusion}
Therefore, for all $k \ge 1$,
\begin{equation}
    f(x_k) - f^* \le \frac{\norm{x_0 - x^*}^2}{2\eta k}.
\end{equation}
This completes the proof of Theorem \ref{thm:convergence}. $\square$

\section{Discussion}

The proof uses the standard technique of summing the one-step progress inequality. The key steps are:
\begin{itemize}
    \item Descent lemma from smoothness (Step 1, 2)
    \item Convexity to relate gradient to suboptimality (Step 3, 8)
    \item Telescoping sum of distance squares (Step 8)
\end{itemize}
The step size condition $\eta \le 1/L$ ensures the descent coefficient is positive and the distance to optimum does not increase.
