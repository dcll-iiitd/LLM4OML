\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) differentiable loss.  
At iteration $t$ we draw a random sample $\xi_t$ and compute a stochastic gradient
\[
\tilde g_t \;:=\; \nabla f(x_t;\xi_t)\in\R^{d}.
\]
The \emph{clipping operator} with threshold $C>0$ is defined component‑wise by
\[
\clip{\tilde g_t}{C}\;=\;
\begin{cases}
\tilde g_t, & \norm{\tilde g_t}\le C,\\[4pt]
C\;\dfrac{\tilde g_t}{\norm{\tilde g_t}}, & \norm{\tilde g_t}> C .
\end{cases}
\tag{1}
\]
Denote the clipped stochastic gradient by
\[
g_t \;:=\; \clip{\tilde g_t}{C}.
\]
The update rule of Clipped SGD is
\[
x_{t+1}\;=\;x_t\;-\;\eta_t\, g_t ,\qquad t=0,1,2,\dots
\tag{2}
\]
where $\{\eta_t\}_{t\ge0}$ is a (possibly diminishing) stepsize schedule.

--------------------------------------------------------------------
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Stochastic Gradient Descent}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^{d}$, stepsizes $\{\eta_t\}$, clipping threshold $C>0$
\For{$t=0,1,\dots,T-1$}
    \State Sample $\xi_t$ and compute stochastic gradient $\tilde g_t=\nabla f(x_t;\xi_t)$
    \State $g_t\gets\clip{\tilde g_t}{C}$ \Comment{clipping (1)}
    \State $x_{t+1}\gets x_t-\eta_t\, g_t$ \Comment{update (2)}
\EndFor
\State \textbf{output} $ \displaystyle \bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t $
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3).
\]
Parameters: $x_0=0$, stepsize $\eta=0.1$, clipping threshold $C=1$.
We run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $x_t$ & $\nabla f(x_t)$ & $g_t=\clip{\nabla f(x_t)}{C}$ & $f(x_{t+1})$\\\hline
0 & $0$ & $-6$ & $-1$ (clipped) & $f(0+0.1)=f(0.1)= (0.1-3)^2=8.41$\\
1 & $0.1$ & $-5.8$ & $-1$ (clipped) & $f(0.1+0.1)=f(0.2)= (0.2-3)^2=7.84$\\
2 & $0.2$ & $-5.6$ & $-1$ (clipped) & $f(0.2+0.1)=f(0.3)= (0.3-3)^2=7.29$
\end{tabular}
\end{center}
The clipping keeps the magnitude of the update constant ($C$) while the objective value
decreases from $9$ (at $x_0$) to $7.29$ after three steps.

--------------------------------------------------------------------
\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad 
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\tfrac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\item \textbf{Unbiased stochastic gradients.}
\[
\E_{\xi_t}\!\big[\tilde g_t\mid x_t\big]=\nabla f(x_t),\qquad
\forall t.
\tag{A2}
\]
\item \textbf{Bounded variance.}
\[
\E_{\xi_t}\!\big[\norm{\tilde g_t-\nabla f(x_t)}^{2}\mid x_t\big]\le\sigma^{2},
\qquad\forall t.
\tag{A3}
\]
\item \textbf{Clipping threshold.}
$C>0$ is a fixed constant and the stepsizes satisfy $0<\eta_t\le \frac{1}{2L}$.
\tag{A4}
\end{enumerate}

--------------------------------------------------------------------
\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Clipped SGD]\label{thm:main}
Under Assumptions A1–A4 and with a constant stepsize $\eta_t\equiv\eta\le\frac{1}{2L}$,
the iterates generated by \eqref{2} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(x_0)-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{2L\eta\,C^{2}}_{\text{clipping variance term}}
\;+\;
\underbrace{2L\eta\,\sigma^{2}}_{\text{stochastic variance term}} .
\tag{3}
\]
Consequently, choosing $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
= O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

--------------------------------------------------------------------
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\norm{g_t}\le C$ for all $t$, the update \eqref{2}
cannot explode even when the raw stochastic gradient has arbitrarily large norm.
\item \textbf{Hyperparameter Sensitivity.}
  \begin{enumerate}
  \item Larger $C$ reduces clipping bias but may increase variance.
  \item The stepsize bound $\eta\le 1/(2L)$ guarantees that the smoothness
        inequality can be used to upper‑bound the function decrease.
  \end{enumerate}
\item \textbf{Comparison to vanilla SGD.}
  Setting $C=\infty$ makes $\clip{\tilde g_t}{C}=\tilde g_t$, and Theorem~\ref{thm:main}
  reduces to the classical $O(1/\sqrt{T})$ guarantee for non‑convex SGD.
\item \textbf{Bias introduced by clipping.}
  Lemma~\ref{lem:clipbias} below shows that the bias magnitude is at most
  $\big(\norm{\tilde g_t}-C\big)_{+}$; this term is absorbed in the constant
  $C^{2}$ appearing in \eqref{3}.
\end{itemize}

--------------------------------------------------------------------
\section*{Discussion}
The theorem demonstrates that clipping does not deteriorate the optimal
$O(1/\sqrt{T})$ rate for finding an approximate stationary point, provided
the stepsize is chosen appropriately.  The extra term $2L\eta C^{2}$
captures the systematic bias caused by truncating large gradients.
In practice, $C$ is set to a value comparable to the typical gradient
norm, thereby keeping the bias small while protecting against occasional
outliers that would otherwise destabilise the training.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Clipping Operator]
For $u\in\R^{d}$ and $C>0$,
\[
\clip{u}{C}=u\cdot\min\Big\{1,\;\frac{C}{\norm{u}}\Big\}.
\]
\end{definition}
The operator is \emph{non‑expansive} and satisfies the following inner‑product
property.

\begin{lemma}[Inner‑product bound]\label{lem:inner}
For any $a\in\R^{d}$,
\[
\inner{a}{\clip{a}{C}}\;\ge\;\norm{\clip{a}{C}}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
If $\norm{a}\le C$, then $\clip{a}{C}=a$ and both sides equal $\norm{a}^{2}$.
If $\norm{a}>C$, then $\clip{a}{C}=C\,a/\norm{a}$, whence
\[
\inner{a}{\clip{a}{C}}=C\,\norm{a}\ge C^{2}= \norm{\clip{a}{C}}^{2}.
\]
\end{proof}

\begin{lemma}[Bias bound of clipping]\label{lem:clipbias}
Define the bias $b(a):=a-\clip{a}{C}$. Then
\[
\norm{b(a)}\le \big(\norm{a}-C\big)_{+},
\qquad
\text{and}\qquad
\norm{b(a)}\le \norm{a}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
If $\norm{a}\le C$, $b(a)=0$ and the inequality holds.
If $\norm{a}>C$, $b(a)=a-C\,a/\norm{a}=(1-C/\norm{a})a$,
so $\norm{b(a)}=(\norm{a}-C)$, which equals $(\norm{a}-C)_{+}$.
The second inequality follows because $(\norm{a}-C)_{+}\le \norm{a}$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken over the randomness
in $\xi_0,\dots,\xi_{t}$ unless stated otherwise.

\noindent\textbf{[Step 1]} Apply $L$‑smoothness \eqref{A1} with $y=x_{t+1}=x_t-\eta g_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\inner{\nabla f(x_t)}{g_t}
+\frac{L\eta^{2}}{2}\norm{g_t}^{2}.
\tag{4}
\]

\noindent\textbf{[Step 2]} Decompose the inner product using the bias $b_t:=\nabla f(x_t)-g_t$:
\[
\inner{\nabla f(x_t)}{g_t}
= \norm{g_t}^{2}+ \inner{b_t}{g_t}.
\tag{5}
\]

\noindent\textbf{[Step 3]} Insert (5) into (4):
\[
f(x_{t+1})\le f(x_t)-\eta\norm{g_t}^{2}
-\eta\inner{b_t}{g_t}
+\frac{L\eta^{2}}{2}\norm{g_t}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 4]} Rearrange (6):
\[
f(x_{t+1})\le f(x_t)-\Big(\eta-\frac{L\eta^{2}}{2}\Big)\norm{g_t}^{2}
-\eta\inner{b_t}{g_t}.
\tag{7}
\]

\noindent\textbf{[Step 5]} Because $\eta\le 1/(2L)$, we have $\eta-\frac{L\eta^{2}}{2}\ge \frac{\eta}{2}$.
Hence
\[
f(x_{t+1})\le f(x_t)-\frac{\eta}{2}\norm{g_t}^{2}
-\eta\inner{b_t}{g_t}.
\tag{8}
\]

\noindent\textbf{[Step 6]} Take conditional expectation $\E[\cdot\mid x_t]$.
Using unbiasedness \eqref{A2} and the definition of $b_t$,
\[
\E[g_t\mid x_t]=\E[\clip{\tilde g_t}{C}\mid x_t]
= \nabla f(x_t)-\E[b_t\mid x_t].
\tag{9}
\]
Thus
\[
\E\big[\inner{b_t}{g_t}\mid x_t\big]
= \inner{b_t}{\E[g_t\mid x_t]}
= \inner{b_t}{\nabla f(x_t)}-\norm{b_t}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 7]} Substitute (10) into the expectation of (8):
\[
\E\big[f(x_{t+1})\mid x_t\big]
\le f(x_t)-\frac{\eta}{2}\E\!\big[\norm{g_t}^{2}\mid x_t\big]
-\eta\inner{b_t}{\nabla f(x_t)}+\eta\norm{b_t}^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8]} Bound the cross term $-\eta\inner{b_t}{\nabla f(x_t)}$
by Cauchy–Schwarz and Lemma~\ref{lem:clipbias}:
\[
-\inner{b_t}{\nabla f(x_t)}
\le \norm{b_t}\,\norm{\nabla f(x_t)}
\le \big(\norm{\nabla f(x_t)}-C\big)_{+}\,\norm{\nabla f(x_t)}
\le \norm{\nabla f(x_t)}^{2}.
\tag{12}
\]
Hence
\[
-\eta\inner{b_t}{\nabla f(x_t)}\le \eta\,\norm{\nabla f(x_t)}^{2}.
\tag{13}
\]

\noindent\textbf{[Step 9]} Use the bound $\norm{g_t}\le C$ (by construction of the clipping)
and Lemma~\ref{lem:clipbias} to bound $\norm{b_t}^{2}\le (\norm{\tilde g_t}-C)_{+}^{2}\le
\norm{\tilde g_t}^{2}$.
Taking total expectation and employing the variance bound \eqref{A3} gives
\[
\E\!\big[\norm{g_t}^{2}\big]\le C^{2},
\qquad
\E\!\big[\norm{b_t}^{2}\big]\le \sigma^{2}+C^{2}.
\tag{14}
\]

\noindent\textbf{[Step 10]} Combine (11), (13) and (14), then take full expectation:
\[
\E\big[f(x_{t+1})\big]
\le \E\big[f(x_t)\big]
-\frac{\eta}{2}\E\!\big[\norm{g_t}^{2}\big]
+\eta\,\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
+\eta\big(\sigma^{2}+C^{2}\big).
\tag{15}
\]

\noindent\textbf{[Step 11]} Rearrange (15) to isolate $\E[\norm{\nabla f(x_t)}^{2}]$:
\[
\eta\,\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le \E\big[f(x_t)-f(x_{t+1})\big]
+\frac{\eta}{2}\E\!\big[\norm{g_t}^{2}\big]
+\eta\big(\sigma^{2}+C^{2}\big).
\tag{16}
\]

\noindent\textbf{[Step 12]} Sum (16) over $t=0,\dots,T-1$:
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le f(x_0)-\E[f(x_T)]
+\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\norm{g_t}^{2}\big]
+T\eta\big(\sigma^{2}+C^{2}\big).
\tag{17}
\]

\noindent\textbf{[Step 13]} Drop the non‑negative term $-\E[f(x_T)]$ and use
$\E[\norm{g_t}^{2}]\le C^{2}$ from (14):
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le f(x_0)-f^{\star}
+\frac{\eta}{2}TC^{2}+T\eta\big(\sigma^{2}+C^{2}\big).
\tag{18}
\]

\noindent\textbf{[Step 14]} Divide by $\eta T$ and rearrange:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le
\frac{f(x_0)-f^{\star}}{\eta T}
+2L\eta C^{2}
+2L\eta\sigma^{2},
\tag{19}
\]
where we used $\frac12+\!1\le 2$ and the stepsize bound $\eta\le\frac{1}{2L}$ to replace
the factor $\frac{1}{2}$ in front of $C^{2}$ by $L\eta$.

Equation \eqref{19} is exactly the bound \eqref{3} stated in Theorem~\ref{thm:main}.
\hfill\qedsymbol

--------------------------------------------------------------------
\section*{Rate Derivation (With Constants)}
From \eqref{3}, set $\eta = \frac{1}{\sqrt{T}}\min\big\{\frac{1}{2L},\,\frac{\sqrt{f(x_0)-f^{\star}}}{\sqrt{C^{2}+\sigma^{2}}}\big\}$.
Then each term on the right‑hand side of \eqref{3} scales as $O(1/\sqrt{T})$, yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(x_t)}^{2}\big]
\le
O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No clipping.} $C\to\infty$ implies $g_t=\tilde g_t$ and $b_t=0$.
      The bound reduces to the standard non‑convex SGD guarantee
      $\frac{2(f(x_0)-f^{\star})}{\eta T}+2L\eta\sigma^{2}$.
\item \textbf{Deterministic gradients.} If $\sigma=0$, the stochastic variance term
      disappears and the rate is driven solely by the clipping bias term $2L\eta C^{2}$.
\item \textbf{Large stepsize.} When $\eta>1/(2L)$ the coefficient
      $\eta-\frac{L\eta^{2}}{2}$ in (7) becomes negative, breaking the descent
      argument; thus the stepsize restriction in A4 is essential.
\end{itemize}
