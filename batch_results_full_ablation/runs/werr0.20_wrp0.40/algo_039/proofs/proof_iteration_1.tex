\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
We consider the convex-concave saddle‑point problem  

\[
\min_{x\in\mathbb R^{n}}\max_{y\in\mathbb R^{m}}
\; \mathcal L(x,y):= f(x)+\langle Kx , y\rangle - g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
  \item $f:\mathbb R^{n}\to\mathbb R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth);
  \item $g^{*}:\mathbb R^{m}\to\mathbb R\cup\{+\infty\}$ is the convex conjugate of a proper, closed convex function $g$;
  \item $K\in\mathbb R^{m\times n}$ is a linear operator.
\end{itemize}

Let $\tau>0$ and $\sigma>0$ be primal and dual step sizes satisfying  

\[
\tau\sigma\|K\|^{2}<1,
\tag{2}
\]

where $\|K\|$ denotes the spectral norm of $K$.

The PDHG iterates are defined for $k=0,1,2,\dots$ by  

\[
\boxed{
\begin{aligned}
y_{k+1}      &= \prox_{\sigma g^{*}}\!\bigl( y_{k}+ \sigma Kx_{k}\bigr), \tag{3a}\\[4pt]
\tilde x_{k+1}&= \prox_{\tau f}\!\bigl( x_{k}-\tau K^{\top} y_{k+1}\bigr), \tag{3b}\\[4pt]
x_{k+1}      &= \tilde x_{k+1}+ \theta\bigl(\tilde x_{k+1}-x_{k}\bigr), \tag{3c}
\end{aligned}}
\]

where $\theta\in[0,1]$ is an extrapolation parameter (we set $\theta=1$ in the sequel, which yields the classical PDHG).  
The proximal operator of a convex function $h$ is  

\[
\prox_{\lambda h}(v):=\arg\min_{u}\Big\{ h(u)+\frac{1}{2\lambda}\|u-v\|^{2}\Big\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\REQUIRE $f$, $g^{*}$, $K$, step sizes $\tau,\sigma>0$ satisfying \eqref{2}, extrapolation $\theta\in[0,1]$, initial points $x_{0}\in\mathbb R^{n},\;y_{0}\in\mathbb R^{m}$, maximal iterations $T$.
\FOR{$k=0,\dots,T-1$}
    \STATE $y_{k+1}\gets \prox_{\sigma g^{*}}\!\bigl(y_{k}+ \sigma Kx_{k}\bigr)$ \hfill\COMMENT{dual ascent}
    \STATE $\tilde x_{k+1}\gets \prox_{\tau f}\!\bigl(x_{k}-\tau K^{\top} y_{k+1}\bigr)$ \hfill\COMMENT{primal descent}
    \STATE $x_{k+1}\gets \tilde x_{k+1}+ \theta(\tilde x_{k+1}-x_{k})$ \hfill\COMMENT{extrapolation}
\ENDFOR
\RETURN $(x_{T},y_{T})$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We instantiate \eqref{1} with a simple scalar problem  

\[
\min_{x\in\mathbb R} \; f(x):=(x-3)^{2},
\]

i.e. $g^{*}\equiv0$, $K=1$, and we choose $\tau=\sigma=0.1$, $\theta=1$, $x_{0}=0$, $y_{0}=0$.

Because $g^{*}=0$, $\prox_{\sigma g^{*}}(v)=v$.
The proximal of $f$ can be computed analytically:

\[
\prox_{\tau f}(v)=\arg\min_{u}\big\{(u-3)^{2}+\tfrac{1}{2\tau}(u-v)^{2}\big\}
               =\frac{v+6\tau}{1+2\tau}.
\tag{4}
\]

With $\tau=0.1$, \eqref{4} yields $\prox_{\tau f}(v)=\frac{v+0.6}{1.2}= \frac{5}{6}v+0.5$.

\medskip
\noindent\textbf{Iteration $k=0$}
\begin{align*}
y_{1}&=y_{0}+ \sigma K x_{0}=0+0.1\cdot0=0,\\
\tilde x_{1}&=\prox_{\tau f}\!\bigl(x_{0}-\tau K^{\top} y_{1}\bigr)
           =\prox_{\tau f}(0)=0.5,\\
x_{1}&=\tilde x_{1}+(\tilde x_{1}-x_{0})=0.5+0.5=1.0,\\
f(x_{1})&=(1-3)^{2}=4.
\end{align*}

\noindent\textbf{Iteration $k=1$}
\begin{align*}
y_{2}&=y_{1}+ \sigma K x_{1}=0+0.1\cdot1=0.1,\\
\tilde x_{2}&=\prox_{\tau f}\!\bigl(x_{1}-\tau K^{\top} y_{2}\bigr)
           =\prox_{\tau f}(1-0.1\cdot0.1)=\prox_{\tau f}(0.99)\\
           &=\frac{5}{6}\cdot0.99+0.5\approx0.825,\\
x_{2}&=\tilde x_{2}+(\tilde x_{2}-x_{1})\approx0.825+(-0.175)=0.650,\\
f(x_{2})&\approx(0.65-3)^{2}\approx5.48.
\end{align*}

\noindent\textbf{Iteration $k=2$}
\begin{align*}
y_{3}&=y_{2}+ \sigma K x_{2}=0.1+0.1\cdot0.65=0.165,\\
\tilde x_{3}&=\prox_{\tau f}\!\bigl(x_{2}-\tau y_{3}\bigr)
           =\prox_{\tau f}(0.65-0.1\cdot0.165)=\prox_{\tau f}(0.6335)\\
           &=\frac{5}{6}\cdot0.6335+0.5\approx0.7279,\\
x_{3}&=\tilde x_{3}+(\tilde x_{3}-x_{2})\approx0.7279+0.0779=0.8058,\\
f(x_{3})&\approx(0.8058-3)^{2}\approx4.82.
\end{align*}

The iterates gradually move toward the minimiser $x^{\star}=3$, illustrating the behaviour of PDHG even on a purely primal problem.

\section*{Assumptions}
\begin{assumption}[Convexity and Closedness]\label{ass:convex}
The functions $f:\mathbb R^{n}\to\mathbb R\cup\{+\infty\}$ and $g:\mathbb R^{m}\to\mathbb R\cup\{+\infty\}$ are proper, closed and convex.
\end{assumption}
\begin{assumption}[Lipschitz Continuity of $K$]\label{ass:K}
$K\in\mathbb R^{m\times n}$ is a linear operator with finite spectral norm $\|K\|$.
\end{assumption}
\begin{assumption}[Step‑size Condition]\label{ass:steps}
Primal and dual step sizes $\tau,\sigma>0$ satisfy $\tau\sigma\|K\|^{2}<1$.
\end{assumption}
\begin{assumption}[Existence of a Saddle Point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})$ such that  

\[
\mathcal L(x^{\star},y)\le \mathcal L(x^{\star},y^{\star})\le \mathcal L(x,y^{\star}),\qquad\forall x\in\mathbb R^{n},y\in\mathbb R^{m}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ Rate]\label{thm:rate}
Under Assumptions \ref{ass:convex}–\ref{ass:saddle}, let $\{(x_{k},y_{k})\}_{k\ge0}$ be generated by Algorithm \ref{alg:pdhg} with $\theta=1$ and step sizes obeying \eqref{ass:steps}. Define the ergodic averages  

\[
\bar x_{N}:=\frac{1}{N}\sum_{k=1}^{N} \tilde x_{k},\qquad 
\bar y_{N}:=\frac{1}{N}\sum_{k=1}^{N} y_{k}.
\tag{6}
\]

Then for every $N\ge1$  

\[
\underbrace{\max_{y}\mathcal L(\bar x_{N},y)-\min_{x}\mathcal L(x,\bar y_{N})}_{\displaystyle\text{primal–dual gap}}
\;\le\;
\frac{1}{N}\Bigl(
\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
+\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}
\Bigr).
\tag{7}
\]

Consequently the gap converges to $0$ at rate $O(1/N)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} Condition \eqref{ass:steps} guarantees that the Lyapunov function  

  \[
  \Phi_{k}:=\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}+\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
  \tag{8}
  \]
  is non‑increasing up to a telescoping term; this is the core of the proof.
  \item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{7} depends linearly on $1/\tau$ and $1/\sigma$. Choosing $\tau$ and $\sigma$ as large as allowed by \eqref{ass:steps} (i.e. $\tau\sigma\|K\|^{2}\uparrow1$) yields the smallest constant in \eqref{7}.
  \item \textbf{Comparison to Baselines.} For a purely primal gradient method with step size $\eta$, the best known rate for non‑smooth convex $f$ is $O(1/\sqrt{N})$. PDHG attains the faster $O(1/N)$ rate because it exploits the primal–dual structure.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal maps]\label{lem:threepoint}
Let $h$ be convex and $\lambda>0$. For any $u,v\in\mathbb R^{d}$ and $p=\prox_{\lambda h}(v)$,
\[
\langle p-u,\; v-p\rangle \ge \lambda\bigl( h(p)-h(u)\bigr).
\tag{9}
\]
\end{lemma}
\begin{proof}
By definition of $p$,  

\[
h(p)+\frac{1}{2\lambda}\|p-v\|^{2}\le h(u)+\frac{1}{2\lambda}\|u-v\|^{2}.
\]
Rearranging terms gives  

\[
\frac{1}{2\lambda}\bigl(\|p-v\|^{2}-\|u-v\|^{2}\bigr)\le h(u)-h(p).
\]
Expanding the squared norms yields  

\[
\frac{1}{\lambda}\langle p-u, v-p\rangle\le h(u)-h(p),
\]
which after multiplication by $\lambda$ gives \eqref{9}.  \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout the proof let $(x^{\star},y^{\star})$ be any saddle point satisfying \eqref{5}.  

\subsection*{Step 1 – Dual update inequality}
From Lemma \ref{lem:threepoint} applied to $h=g^{*}$, $\lambda=\sigma$, $v=y_{k}+ \sigma Kx_{k}$ and $p=y_{k+1}$ we obtain  

\[
\langle y_{k+1}-y^{\star},\; y_{k}+ \sigma Kx_{k}-y_{k+1}\rangle
      \ge \sigma\bigl(g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr).
\tag{10}
\]
Expanding the inner product and rearranging gives  

\[
\underbrace{\langle y_{k+1}-y^{\star},\; y_{k}-y_{k+1}\rangle}_{\displaystyle -\frac12\|y_{k+1}-y_{k}\|^{2}
            -\frac12\|y_{k+1}-y^{\star}\|^{2}+\frac12\|y_{k}-y^{\star}\|^{2}}
  +\sigma\langle y_{k+1}-y^{\star},Kx_{k}\rangle
   \ge \sigma\bigl(g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr).
\tag{11}
\]
[Step 1] is the algebraic expansion of the first term in \eqref{10}.

\subsection*{Step 2 – Primal update inequality}
Apply Lemma \ref{lem:threepoint} to $h=f$, $\lambda=\tau$, $v=x_{k}-\tau K^{\top}y_{k+1}$ and $p=\tilde x_{k+1}$. This yields  

\[
\langle \tilde x_{k+1}-x^{\star},\; x_{k}-\tau K^{\top}y_{k+1}-\tilde x_{k+1}\rangle
      \ge \tau\bigl(f(\tilde x_{k+1})-f(x^{\star})\bigr).
\tag{12}
\]
Expanding the inner product gives  

\[
\underbrace{\langle \tilde x_{k+1}-x^{\star},\; x_{k}-\tilde x_{k+1}\rangle}_{\displaystyle
   -\frac12\|x_{k}-\tilde x_{k+1}\|^{2}
   -\frac12\|\tilde x_{k+1}-x^{\star}\|^{2}
   +\frac12\|x_{k}-x^{\star}\|^{2}}
   -\tau\langle \tilde x_{k+1}-x^{\star},K^{\top}y_{k+1}\rangle
   \ge \tau\bigl(f(\tilde x_{k+1})-f(x^{\star})\bigr).
\tag{13}
\]
[Step 2] is the explicit expansion of the first inner product in \eqref{12}.

\subsection*{Step 3 – Combine primal and dual inequalities}
Add \eqref{11} and \eqref{13}. The mixed terms $\langle y_{k+1}-y^{\star},Kx_{k}\rangle$ and $-\langle \tilde x_{k+1}-x^{\star},K^{\top}y_{k+1}\rangle$ combine via the identity  

\[
\langle y_{k+1}-y^{\star},Kx_{k}\rangle
      -\langle \tilde x_{k+1}-x^{\star},K^{\top}y_{k+1}\rangle
   =\langle K^{\top}(y_{k+1}-y^{\star}),x_{k}-\tilde x_{k+1}\rangle
   +\langle K^{\top}y_{k+1},\tilde x_{k+1}-x^{\star}\rangle
   -\langle K^{\top}y^{\star},x_{k}-x^{\star}\rangle .
\tag{14}
\]

Using Cauchy–Schwarz and Young’s inequality $2ab\le \frac{a^{2}}{\alpha}+\alpha b^{2}$ with $\alpha=\tau\sigma\|K\|^{2}$ we bound  

\[
2\langle K^{\top}(y_{k+1}-y^{\star}),x_{k}-\tilde x_{k+1}\rangle
   \le \frac{1}{\tau}\|x_{k}-\tilde x_{k+1}\|^{2}
        +\tau\|K^{\top}(y_{k+1}-y^{\star})\|^{2}
   \le \frac{1}{\tau}\|x_{k}-\tilde x_{k+1}\|^{2}
        +\tau\|K\|^{2}\|y_{k+1}-y^{\star}\|^{2}.
\tag{15}
\]

Substituting \eqref{15} into the sum of \eqref{11} and \eqref{13} and cancelling the identical $\frac{1}{\tau}\|x_{k}-\tilde x_{k+1}\|^{2}$ terms yields the fundamental descent inequality  

\[
\Phi_{k+1}
\le \Phi_{k}
   -\tau\bigl(f(\tilde x_{k+1})-f(x^{\star})\bigr)
   -\sigma\bigl(g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr)
   +\tau\sigma\|K\|^{2}\bigl(\Phi_{k}-\Phi_{k+1}\bigr),
\tag{16}
\]
where $\Phi_{k}$ is defined in \eqref{8}.  

Rearranging \eqref{16} gives  

\[
(1-\tau\sigma\|K\|^{2})\bigl(\Phi_{k+1}-\Phi_{k}\bigr)
   \le -\tau\bigl(f(\tilde x_{k+1})-f(x^{\star})\bigr)
      -\sigma\bigl(g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr).
\tag{17}
\]

[Step 3] consists of the addition of \eqref{11} and \eqref{13}, the use of identity \eqref{14}, and the Young‑type bound \eqref{15}.

\subsection*{Step 4 – Telescoping sum}
Because $0<\tau\sigma\|K\|^{2}<1$ by Assumption \ref{ass:steps}, divide \eqref{17} by $1-\tau\sigma\|K\|^{2}$ and sum from $k=0$ to $N-1$:

\[
\sum_{k=0}^{N-1}\Bigl(
   \tau\bigl(f(\tilde x_{k+1})-f(x^{\star})\bigr)
   +\sigma\bigl(g^{*}(y_{k+1})-g^{*}(y^{\star})\bigr)\Bigr)
\le \Phi_{0}-\Phi_{N}.
\tag{18}
\]

Since $\Phi_{N}\ge0$, we obtain  

\[
\sum_{k=1}^{N}\Bigl(
   \tau\bigl(f(\tilde x_{k})-f(x^{\star})\bigr)
   +\sigma\bigl(g^{*}(y_{k})-g^{*}(y^{\star})\bigr)\Bigr)
\le \Phi_{0}.
\tag{19}
\]

\subsection*{Step 5 – From objective differences to primal–dual gap}
Recall the saddle‑point inequality \eqref{5}. By convexity of $f$ and $g^{*}$ and the definition of the Lagrangian,  

\[
\mathcal L(\tilde x_{k},y)-\mathcal L(x,y_{k})
   = f(\tilde x_{k})-f(x)+\langle K\tilde x_{k},y-y_{k}\rangle
     -\bigl(g^{*}(y)-g^{*}(y_{k})\bigr)
\le f(\tilde x_{k})-f(x^{\star})
    +g^{*}(y_{k})-g^{*}(y^{\star}),
\tag{20}
\]
where we used $y=y^{\star}$ and $x=x^{\star}$ for the right‑hand side.  

Dividing \eqref{20} by $\tau$ and $\sigma$ respectively and using \eqref{19} gives  

\[
\sum_{k=1}^{N}\Bigl(\mathcal L(\tilde x_{k},y^{\star})-\mathcal L(x^{\star},y_{k})\Bigr)
\le \frac{1}{\tau}\Phi_{0}.
\tag{21}
\]

Now divide both sides of \eqref{21} by $N$ and use the convexity of the supremum/infimum operators to obtain the ergodic bound  

\[
\max_{y}\mathcal L(\bar x_{N},y)-\min_{x}\mathcal L(x,\bar y_{N})
   \le \frac{1}{N}\Bigl(\frac{\|x_{0}-x^{\star}\|^{2}}{2\tau}
                     +\frac{\|y_{0}-y^{\star}\|^{2}}{2\sigma}\Bigr).
\tag{22}
\]

Equation \eqref{22} is exactly the statement \eqref{7} of Theorem \ref{thm:rate}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{22} we read the explicit constant  

\[
C:=\frac{1}{2}\Bigl(\frac{\|x_{0}-x^{\star}\|^{2}}{\tau}
                     +\frac{\|y_{0}-y^{\star}\|^{2}}{\sigma}\Bigr),
\]
so that  

\[
\text{gap}_{N}\le \frac{C}{N}.
\tag{23}
\]

Choosing the largest admissible product $\tau\sigma$ (i.e. $\tau\sigma=\frac{1}{\|K\|^{2}}-\epsilon$ for a tiny $\epsilon>0$) minimises $C$ because both $1/\tau$ and $1/\sigma$ decrease as the step sizes increase.

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Purely primal smooth case ($g^{*}=0$, $K=I$).} The dual update becomes $y_{k+1}=y_{k}+ \sigma x_{k}$, which can be eliminated, yielding the classic forward–backward splitting with step size $\tau$. The $O(1/N)$ rate recovers the optimal rate for smooth convex minimisation.
  \item \textbf{Indicator of a linear constraint.} If $g^{*}$ is the indicator of a convex set $\mathcal C$, then $\prox_{\sigma g^{*}}$ reduces to a Euclidean projection onto $\mathcal C$, and PDHG becomes the celebrated Chambolle–Pock method for constrained problems.
  \item \textbf{Non‑smooth $f$ (e.g., $\ell_{1}$‑regularisation).} The proximal $\prox_{\tau f}$ is a soft‑thresholding operator; the same $O(1/N)$ guarantee holds without any smoothness assumption on $f$.
\end{itemize}
