\section*{Algorithm Name}
RMSProp (without bias correction; per-coordinate exponential moving average of squared gradients; per-coordinate normalization by the square root of the running average).

\section*{Mathematical Formulation}
We consider minimizing a differentiable (possibly non-convex) function $f:\RR^d\to\RR$. Let $\{w_t\}_{t\ge 0}\subset\RR^d$ denote the iterates. RMSProp maintains a running average of squared gradients
\[
v_t \;=\; \beta\, v_{t-1} + (1-\beta)\, g_t \odot g_t,\quad t\ge 1,
\]
where $g_t = \nabla f(w_t)$, $\beta\in[0,1)$ is a fixed decay parameter, $v_0\in\RR_+^d$ is an initial positive vector, and $\odot$ denotes elementwise product. The update is
\[
w_{t+1} \;=\; w_t - \alpha\, P_t\, g_t,\qquad P_t \;=\; \diag\!\big( v_t^{-1/2}\big),
\]
where $v_t^{-1/2}$ is taken elementwise and $\alpha>0$ is a base stepsize. In coordinates,
\[
(v_t)_j = \beta (v_{t-1})_j + (1-\beta) (g_t)_j^2,\qquad (w_{t+1})_j = (w_t)_j - \alpha \frac{(g_t)_j}{\sqrt{(v_t)_j}},\quad j=1,\dots,d.
\]
To avoid division-by-zero, assume $(v_0)_j>0$ for all $j$; the analysis below uses $(v_0)_j\ge 0$ and the fact that for $g_t\neq 0$, $(v_t)_j>0$ automatically when $\beta<1$.

\section*{Pseudocode}
\begin{flushleft}
Input: $w_0\in\RR^d$, $v_0\in\RR_+^d$ (e.g., $v_0=\delta \cdot \one$ with $\delta>0$), base stepsize $\alpha>0$, decay $\beta\in[0,1)$, maximum iterations $T$, tolerance tol.

For $t=1,2,\dots,T$:
\begin{enumerate}
\item Compute gradient: $g_t \leftarrow \nabla f(w_t)$
\item Update squared-average: $v_t \leftarrow \beta v_{t-1} + (1-\beta)\, g_t \odot g_t$
\item Compute preconditioner: $P_t \leftarrow \diag(v_t^{-1/2})$
\item Update parameters: $w_{t+1} \leftarrow w_t - \alpha\, P_t\, g_t$
\item If $\norm{g_t}_2 \le \text{tol}$ then stop
\end{enumerate}
Output: $w_{\text{out}} = w_{t^\star}$ where $t^\star$ is the last iteration.
\end{flushleft}

\section*{Dry Run Example}
We minimize $f(w)=(w-3)^2$ in $d=1$ starting at $w_0=0$ with $\alpha=0.1$, $\beta=0.9$, $v_0=0$. The gradient is $g(w)=2(w-3)$.

Iteration 1:
\[
g_1 = 2(0-3) = -6,\quad v_1 = 0.9\cdot 0 + 0.1\cdot 36 = 3.6,\quad
w_1 = 0 - 0.1\cdot\frac{-6}{\sqrt{3.6}} \approx 0.316227766.
\]
Objective: $f(w_1) = (0.316227766-3)^2 = 9 - 2\cdot 3 \cdot 0.316227766 + 0.1 \approx 7.202633404$.

Iteration 2:
\[
g_2 = 2(0.316227766-3) \approx -5.367544468,\;\; g_2^2 \approx 28.810533616,
\]
\[
v_2 = 0.9\cdot 3.6 + 0.1\cdot 28.810533616 \approx 6.121053362,\quad
w_2 = 0.316227766 - 0.1\cdot\frac{-5.367544468}{\sqrt{6.121053362}} \approx 0.533158166.
\]
Objective: $f(w_2) \approx (0.533158166-3)^2 \approx 6.085309$.

Iteration 3:
\[
g_3 = 2(0.533158166-3) \approx -4.933683668,\;\; g_3^2 \approx 24.341234,
\]
\[
v_3 = 0.9\cdot 6.121053362 + 0.1\cdot 24.341234 \approx 7.9430715,\quad
w_3 = 0.533158166 - 0.1\cdot\frac{-4.933683668}{\sqrt{7.9430715}} \approx 0.708217.
\]
Objective: $f(w_3) \approx (0.708217-3)^2 \approx 5.25228$.

\section*{Assumptions}
We study the deterministic setting $g_t = \nabla f(w_t)$ under:
\begin{enumerate}
\item $f$ is lower bounded: $\inf_{w\in\RR^d} f(w) =: f_\star > -\infty$.
\item $f$ is $L$-smooth: $\norm{\nabla f(x)-\nabla f(y)}_2 \le L\norm{x-y}_2$ for all $x,y\in\RR^d$ (equivalently, $\nabla f$ is $L$-Lipschitz).
\item Bounded gradients in $\ell_\infty$: there exists $G_\infty>0$ such that $\norm{\nabla f(w)}_\infty \le G_\infty$ for all iterates $w$ produced by the algorithm.
\item RMSProp parameters: $\beta\in[0,1)$ is fixed, $v_0\in\RR_+^d$ with $(v_0)_j\ge 0$ for all $j$ (for well-definedness we may take $(v_0)_j>0$).
\item Base stepsize $\alpha>0$ (possibly chosen as a function of the horizon $T$).
\end{enumerate}

\section*{Main Convergence Theorem}
Let $V:=\max\{G_\infty^2,\ \max_{j}(v_0)_j\}$. Under the assumptions above, the RMSProp iterates satisfy, for any $T\ge 1$,
\[
\min_{1\le t\le T}\norm{\nabla f(w_t)}_2^2
\;\le\;
\frac{G_\infty}{\alpha T}\,\big(f(w_1)-f_\star\big)
\;+\;
\frac{G_\infty\,\alpha\,L\,d}{2(1-\beta)}.
\]
In particular, choosing $\alpha=\sqrt{\frac{2(1-\beta)\,\big(f(w_1)-f_\star\big)}{L\,d\,G_\infty\,T}}$ yields
\[
\min_{1\le t\le T}\norm{\nabla f(w_t)}_2^2
\;\le\;
G_\infty\,\sqrt{\frac{2\,L\,d\,\big(f(w_1)-f_\star\big)}{(1-\beta)\,T}}
\;=\; \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Theoretical Properties}
\begin{itemize}
\item Stability: The per-iteration scaled step satisfies $\norm{P_t g_t}_2^2 = \sum_{j=1}^d \frac{(g_t)_j^2}{(v_t)_j} \le \frac{d}{1-\beta}$, uniformly in $t$.
\item Descent coefficient: The effective curvature along $g_t$ is at least $\lambda_{\min}(P_t)\ge 1/\sqrt{V}\ge 1/G_\infty$, ensuring a uniform descent component along the gradient direction.
\item Hyperparameter sensitivity: Larger $\beta$ increases memory but also the factor $1/(1-\beta)$ in the variance term, which degrades constants in the rate bound.
\item Comparison to SGD: Compared to SGD with stepsize $\eta$, RMSProp adaptively rescales coordinates by $1/\sqrt{v_t}$, yielding bounded effective step energy $\norm{P_t g_t}^2$, which can be advantageous under gradient anisotropy.
\end{itemize}

\section*{Discussion}
The bound shows the standard nonconvex rate $\mathcal{O}(1/\sqrt{T})$ on the minimum squared gradient norm with explicit constants highlighting the roles of $(1-\beta)$, dimension $d$, smoothness $L$, and the $\ell_\infty$ gradient bound $G_\infty$. The proof is deterministic; expectations are therefore trivial if introduced (i.e., $\E[\cdot]=(\cdot)$).

\section*{Preliminaries (Definitions and Lemmas)}
We collect standard smoothness and RMSProp-specific bounds needed in the proof.

\paragraph{Smoothness inequality.}
If $f$ is $L$-smooth, then for any $x,s\in\RR^d$,
\begin{equation}
f(x+s)\;\le\; f(x) + \ip{\nabla f(x)}{s} + \frac{L}{2}\norm{s}_2^2.
\tag{P1}[Step 1]
\end{equation}

\paragraph{Bounds on $v_t$.}
Let $g_t=\nabla f(w_t)$ and $v_t=\beta v_{t-1} + (1-\beta) g_t\odot g_t$ with $\beta\in[0,1)$.
\begin{itemize}
\item Upper bound: For each coordinate $j$,
\begin{equation}
0\;\le\; (v_t)_j \;\le\; \max\Big\{(v_0)_j,\ \sup_{s\le t}(g_s)_j^2\Big\}
\;\le\; \max\{(v_0)_j, G_\infty^2\}\;\le\; V.
\tag{P2}[Step 2]
\end{equation}
Proof: By induction in $t$. If $(v_{t-1})_j\le V$ and $(g_t)_j^2\le G_\infty^2\le V$, then $(v_t)_j=\beta (v_{t-1})_j + (1-\beta)(g_t)_j^2\le \beta V + (1-\beta) V = V$. Nonnegativity is obvious since both terms are nonnegative. [Step 2]
\item Lower bound relative to $g_t$: For each $j$,
\begin{equation}
(v_t)_j \;=\; \beta (v_{t-1})_j + (1-\beta) (g_t)_j^2 \;\ge\; (1-\beta) (g_t)_j^2,
\tag{P3}[Step 3]
\end{equation}
hence for all $j$,
\begin{equation}
\frac{(g_t)_j^2}{(v_t)_j} \;\le\; \frac{1}{1-\beta}.
\tag{P4}[Step 4]
\end{equation}
\end{itemize}

\paragraph{Matrix bounds for the preconditioner.}
Let $P_t=\diag(v_t^{-1/2})$. Then $P_t$ is positive definite as long as $(v_t)_j>0$ for all $j$. Moreover, using (P2) we have
\begin{equation}
\lambda_{\min}(P_t) \;=\; \min_j \frac{1}{\sqrt{(v_t)_j}} \;\ge\; \frac{1}{\sqrt{V}} \;\ge\; \frac{1}{G_\infty}.
\tag{P5}[Step 5]
\end{equation}
Consequently, for any $x\in\RR^d$,
\begin{equation}
x^\top P_t x \;\ge\; \lambda_{\min}(P_t)\,\norm{x}_2^2 \;\ge\; \frac{1}{\sqrt{V}}\,\norm{x}_2^2 \;\ge\; \frac{1}{G_\infty}\,\norm{x}_2^2.
\tag{P6}[Step 6]
\end{equation}
Also, by (P4),
\begin{equation}
\norm{P_t g_t}_2^2 \;=\; \sum_{j=1}^d \frac{(g_t)_j^2}{(v_t)_j} \;\le\; \sum_{j=1}^d \frac{1}{1-\beta} \;=\; \frac{d}{1-\beta}.
\tag{P7}[Step 7]
\end{equation}

\section*{Main Proof (Step-by-Step Derivation)}
We prove the main descent inequality per iteration and then sum across iterations.

Let $s_t := -\alpha P_t g_t$. By $L$-smoothness (P1) with $x=w_t$ and $s=s_t$,
\begin{align}
f(w_{t+1})
&= f(w_t + s_t)
\;\le\;
f(w_t) + \ip{\nabla f(w_t)}{s_t} + \frac{L}{2}\norm{s_t}_2^2.
\tag{M1}[Step 8]
\end{align}
Since $s_t=-\alpha P_t g_t$ and $g_t=\nabla f(w_t)$ (deterministic setting),
\begin{align}
\ip{\nabla f(w_t)}{s_t}
&= -\alpha \ip{g_t}{P_t g_t}
= -\alpha\, g_t^\top P_t g_t,
\tag{M2}[Step 9]
\\
\norm{s_t}_2^2
&= \alpha^2 \norm{P_t g_t}_2^2.
\tag{M3}[Step 10]
\end{align}
Substituting (M2)–(M3) into (M1) yields
\begin{equation}
f(w_{t+1})
\;\le\;
f(w_t)
\;-\;\alpha\, g_t^\top P_t g_t
\;+\;\frac{L\alpha^2}{2}\,\norm{P_t g_t}_2^2.
\tag{M4}[Step 11]
\end{equation}
By (P6) with $x=g_t$ (i.e., $g_t^\top P_t g_t \ge \frac{1}{\sqrt{V}}\norm{g_t}_2^2 \ge \frac{1}{G_\infty}\norm{g_t}_2^2$),
\begin{equation}
g_t^\top P_t g_t \;\ge\; \frac{1}{G_\infty}\,\norm{g_t}_2^2,
\tag{M5}[Step 12]
\end{equation}
and by (P7),
\begin{equation}
\norm{P_t g_t}_2^2 \;\le\; \frac{d}{1-\beta}.
\tag{M6}[Step 13]
\end{equation}
Applying (M5)–(M6) to (M4) gives the per-iteration descent bound
\begin{equation}
f(w_{t+1})
\;\le\;
f(w_t)
\;-\;\frac{\alpha}{G_\infty}\,\norm{g_t}_2^2
\;+\;\frac{L\alpha^2}{2}\cdot \frac{d}{1-\beta}.
\tag{M7}[Step 14]
\end{equation}
Summing (M7) over $t=1,\dots,T$ yields
\begin{align}
f(w_{T+1}) - f(w_1)
&\le
-\frac{\alpha}{G_\infty}\sum_{t=1}^T \norm{g_t}_2^2
+ \frac{L\alpha^2}{2}\cdot \frac{d}{1-\beta}\cdot T.
\tag{M8}[Step 15]
\end{align}
Since $f(w_{T+1})\ge f_\star$,
\begin{equation}
f_\star - f(w_1)
\;\le\;
-\frac{\alpha}{G_\infty}\sum_{t=1}^T \norm{g_t}_2^2
+ \frac{L\alpha^2}{2}\cdot \frac{d}{1-\beta}\cdot T,
\tag{M9}[Step 16]
\end{equation}
which rearranges to
\begin{equation}
\sum_{t=1}^T \norm{\nabla f(w_t)}_2^2
\;\le\;
\frac{G_\infty}{\alpha}\,\big(f(w_1)-f_\star\big)
+ \frac{G_\infty L \alpha d}{2(1-\beta)}\,T.
\tag{M10}[Step 17]
\end{equation}
Dividing both sides by $T$ and using $\min_{1\le t\le T} a_t \le \frac{1}{T}\sum_{t=1}^T a_t$ for nonnegative $a_t$,
\begin{equation}
\min_{1\le t\le T}\norm{\nabla f(w_t)}_2^2
\;\le\;
\frac{1}{T}\sum_{t=1}^T \norm{\nabla f(w_t)}_2^2
\;\le\;
\frac{G_\infty}{\alpha T}\,\big(f(w_1)-f_\star\big)
+ \frac{G_\infty L \alpha d}{2(1-\beta)}.
\tag{M11}[Step 18]
\end{equation}
This proves the first claimed bound in the Main Convergence Theorem. [Step 18]

\section*{Rate Derivation (With Constants)}
Consider the upper bound in (M11) as a function of $\alpha>0$:
\[
\phi_T(\alpha) \;=\; \frac{A}{\alpha T} + B\,\alpha,\qquad
A := G_\infty\big(f(w_1)-f_\star\big),\quad B := \frac{G_\infty L d}{2(1-\beta)}.
\]
Minimizing $\phi_T(\alpha)$ over $\alpha>0$ gives
\begin{equation}
\alpha^\star \;=\; \sqrt{\frac{A}{B T}} \;=\; \sqrt{\frac{2(1-\beta)\,\big(f(w_1)-f_\star\big)}{L\,d\,G_\infty\,T}},
\tag{R1}[Step 19]
\end{equation}
and the minimized value
\begin{align}
\phi_T(\alpha^\star)
&= \frac{A}{\alpha^\star T} + B \alpha^\star
= 2\sqrt{\frac{AB}{T}}
= 2\sqrt{\frac{G_\infty\big(f(w_1)-f_\star\big)}{T}\cdot \frac{G_\infty L d}{2(1-\beta)}}
\notag\\
&= G_\infty\,\sqrt{\frac{2\,L\,d\,\big(f(w_1)-f_\star\big)}{(1-\beta)\,T}}.
\tag{R2}[Step 20]
\end{align}
Substituting $\alpha=\alpha^\star$ into (M11) yields
\[
\min_{1\le t\le T}\norm{\nabla f(w_t)}_2^2 \;\le\; \phi_T(\alpha^\star)
= G_\infty\,\sqrt{\frac{2\,L\,d\,\big(f(w_1)-f_\star\big)}{(1-\beta)\,T}}.
\]

\section*{Special Cases}
\begin{itemize}
\item $\beta=0$ (no memory): Then $v_t = g_t\odot g_t$ and $P_t=\diag(|g_t|^{-1})$, so the update reduces to coordinatewise sign descent $w_{t+1}=w_t-\alpha\,\mathrm{sign}(g_t)$. The bound specializes with $(1-\beta)=1$.
\item $d=1$ (scalar case): The constants simplify to $B = \frac{G_\infty L}{2(1-\beta)}$ and the same $\mathcal{O}(1/\sqrt{T})$ rate holds.
\item Large $\beta\uparrow 1$: The variance term scales like $\frac{1}{1-\beta}$, so while memory increases, the worst-case bound deteriorates in its constant; careful tuning of $\alpha$ becomes more important.
\end{itemize}
