\section*{Algorithm Name}
\textbf{Proximal Gradient Method with Gradient Momentum (PGM‑GM)}

\section*{Mathematical Formulation}
Let  
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}
  \item $f:\R^{d}\rightarrow\R$ is convex and differentiable with $L$‑Lipschitz continuous gradient,
  \item $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ is convex (possibly nonsmooth) and its proximal operator
  \[
  \prox_{\eta g}(z)\;:=\;\arg\min_{u\in\R^{d}}\Big\{ g(u)+\frac1{2\eta}\|u-z\|^{2}\Big\}
  \]
  is easy to compute.
\end{itemize}

Given a stepsize $\eta>0$ and a momentum parameter $\beta\in[0,1)$, the iterates $\{x_k\}_{k\ge 0}$ are generated as  

\[
\boxed{
\begin{aligned}
v_k &:= \grad f(x_k) + \beta\big(\grad f(x_k)-\grad f(x_{k-1})\big), \qquad (k\ge 1),\\[4pt]
x_{k+1} &:= \prox_{\eta g}\!\big(x_k-\eta v_k\big),\qquad (k\ge 0),\\[4pt]
x_{-1}&:=x_0\;\;(\text{for notation}).
\end{aligned}}
\]

When $\beta=0$ the method reduces to the classical proximal gradient method.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{PGM‑GM}
\begin{algorithmic}[1]
\Require stepsize $\eta\in(0,1/L]$, momentum $\beta\in[0,1)$, initial point $x_0\in\R^{d}$, max iterations $K$.
\State $x_{-1}\gets x_0$ \Comment{dummy previous iterate}
\For{$k=0,\dots,K-1$}
    \State Compute gradient $g_k\gets \grad f(x_k)$
    \If{$k=0$}
        \State $v_k\gets g_k$ \Comment{no momentum in the first step}
    \Else
        \State $v_k\gets g_k+\beta\,(g_k-g_{k-1})$
    \EndIf
    \State $x_{k+1}\gets \prox_{\eta g}\!\big(x_k-\eta v_k\big)$
\EndFor
\Ensure $\{x_k\}_{k=0}^{K}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv 0,\qquad \eta=0.1,\qquad \beta=0.5,\qquad w_0=0.
\]

Since $g\equiv0$, $\prox_{\eta g}$ is the identity. The gradient $\grad f(w)=2(w-3)$ and $L=4$ (because $\|\grad f(w)-\grad f(u)\|=2|w-u|\le 4|w-u|$).  

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\grad f(w_k)$ & $v_k$ (with momentum) & $w_{k+1}=w_k-\eta v_k$ \\ \hline
0 & $2(0-3)=-6$ & $v_0=-6$ & $w_1=0-0.1(-6)=0.6$ \\[4pt]
1 & $2(0.6-3)=-4.8$ & $v_1=-4.8+0.5(-4.8+6)=-4.8+0.6=-4.2$ & $w_2=0.6-0.1(-4.2)=1.02$ \\[4pt]
2 & $2(1.02-3)=-3.96$ & $v_2=-3.96+0.5(-3.96+4.8)=-3.96+0.42=-3.54$ & $w_3=1.02-0.1(-3.54)=1.374$ 
\end{tabular}
\end{center}

Objective values $f(w_k)$:  

\[
f(w_0)=9,\; f(w_1)=5.76,\; f(w_2)=3.9204,\; f(w_3)=2.744.\]  
The iterates clearly move toward the minimiser $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is convex and differentiable on $\R^{d}$ with $L$‑Lipschitz continuous gradient, i.e.  
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{ass:convex}
$g$ is proper, closed and convex (possibly nonsmooth). Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}

\begin{assumption}[Stepsize and momentum]\label{ass:params}
The stepsize satisfies $0<\eta\le 1/L$ and the momentum parameter satisfies $0\le\beta<1$.
\end{assumption}

\begin{assumption}[Existence of a minimiser]\label{ass:opt}
There exists at least one optimal solution $x^\star\in\arg\min_{x}F(x)$ and the optimal value $F^\star:=F(x^\star)$ is finite.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Under Assumptions \ref{ass:smooth}–\ref{ass:opt}, the iterates generated by PGM‑GM satisfy for any $K\ge 1$
\[
\frac1K\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^\star\bigr)
\;\le\;
\frac{\|x_0-x^\star\|^{2}}{2\eta K}
\;+\;
\frac{\beta\,L\eta}{2(1-\beta)}\;\|x_0-x^\star\|^{2}.
\]
In particular, if $\beta=0$ the bound reduces to the classical proximal gradient rate
\[
F(x_{K})-F^\star\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} The condition $\eta\le 1/L$ guarantees that the descent lemma (Lemma \ref{lem:descent}) holds, preventing divergence even with momentum.
  \item \textbf{Hyper‑parameter sensitivity.} The additional term $\frac{\beta L\eta}{2(1-\beta)}\|x_0-x^\star\|^{2}$ shows that a larger $\beta$ slows the asymptotic $1/K$ decay but can accelerate early progress; the bound remains finite as long as $\beta<1$.
  \item \textbf{Comparison to baselines.}
    \begin{enumerate}
      \item $\beta=0$: recovers the standard proximal gradient method with optimal $O(1/K)$ ergodic rate.
      \item $g\equiv0$: the method reduces to the gradient‑momentum scheme often called ``heavy‑ball on smooth convex functions'', for which the same $O(1/K)$ bound is known.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem demonstrates that adding a simple momentum correction on the gradient does not deteriorate the optimal $1/K$ convergence rate for composite convex optimisation. The extra constant term is a price paid for the extrapolation; in practice it is often negligible compared with the $1/K$ decay when $K$ is moderate to large. The analysis follows the same template as the classic proximal gradient proof, with additional algebra to handle the momentum term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle \grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz continuity of $\grad f$; see e.g. \cite{Nesterov04}. \hfill $\square$
\end{proof}

\begin{lemma}[Proximal inequality]\label{lem:prox}
For any $z\in\R^{d}$ and any $u\in\R^{d}$,
\[
g\bigl(\prox_{\eta g}(z)\bigr)+\frac{1}{2\eta}\bigl\|\prox_{\eta g}(z)-z\bigr\|^{2}
\le g(u)+\frac{1}{2\eta}\|u-z\|^{2}-\frac{1}{2\eta}\bigl\|u-\prox_{\eta g}(z)\bigr\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from the optimality condition of the proximal sub‑problem. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout the proof we denote $x^\star$ an arbitrary optimal solution.

\bigskip
\noindent\textbf{Notation.} For $k\ge0$ define the momentum‑augmented gradient
\[
v_k:=\grad f(x_k)+\beta\bigl(\grad f(x_k)-\grad f(x_{k-1})\bigr),
\]
with the convention $\grad f(x_{-1}):=\grad f(x_0)$ so that $v_0=\grad f(x_0)$.  

Let $s_k:=x_k-\eta v_k$ (the point before applying the proximal map) and $x_{k+1}:=\prox_{\eta g}(s_k)$.

\bigskip
\noindent\textbf{Step 1.} Apply Lemma \ref{lem:prox} with $z=s_k$ and $u=x^\star$:
\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-s_k\|^{2}
\le g(x^\star)+\frac{1}{2\eta}\|x^\star-s_k\|^{2}
-\frac{1}{2\eta}\|x^\star-x_{k+1}\|^{2}.
\tag{1}
\]

\noindent\textbf{Step 2.} Expand $\|x^\star-s_k\|^{2}$ using $s_k=x_k-\eta v_k$:
\[
\|x^\star-s_k\|^{2}
= \|x^\star-x_k+\eta v_k\|^{2}
= \|x^\star-x_k\|^{2}+2\eta\langle v_k, x^\star-x_k\rangle+\eta^{2}\|v_k\|^{2}.
\tag{2}
\]

\noindent\textbf{Step 3.} Substitute (2) into (1) and rearrange:
\[
\begin{aligned}
g(x_{k+1}) &\le g(x^\star)+\frac{1}{2\eta}\Bigl[\|x^\star-x_k\|^{2}
+2\eta\langle v_k, x^\star-x_k\rangle+\eta^{2}\|v_k\|^{2}
-\|x^\star-x_{k+1}\|^{2}\Bigr]\\
&\qquad -\frac{1}{2\eta}\|x_{k+1}-s_k\|^{2}.
\end{aligned}
\tag{3}
\]

\noindent\textbf{Step 4.} Add $f(x_{k+1})$ to both sides and use the descent lemma (Lemma \ref{lem:descent}) with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{4}
\]

\noindent\textbf{Step 5.} Combine (3) and (4) to bound the full objective $F(x_{k+1})=f(x_{k+1})+g(x_{k+1})$:
\[
\begin{aligned}
F(x_{k+1})-F^\star
&\le \underbrace{f(x_k)-f(x^\star)}_{\text{(A)}}
   +\langle\grad f(x_k),x_{k+1}-x_k\rangle
   +\frac{L}{2}\|x_{k+1}-x_k\|^{2} \\[2pt]
&\quad +\frac{1}{2\eta}\Bigl[\|x^\star-x_k\|^{2}
   +2\eta\langle v_k, x^\star-x_k\rangle
   +\eta^{2}\|v_k\|^{2}
   -\|x^\star-x_{k+1}\|^{2}\Bigr]\\[2pt]
&\quad -\frac{1}{2\eta}\|x_{k+1}-s_k\|^{2}.
\end{aligned}
\tag{5}
\]

\noindent\textbf{Step 6.} Observe that $x_{k+1}-s_k = x_{k+1}-(x_k-\eta v_k)= (x_{k+1}-x_k)+\eta v_k$. Hence
\[
\|x_{k+1}-s_k\|^{2}= \|x_{k+1}-x_k\|^{2}+2\eta\langle v_k, x_{k+1}-x_k\rangle+\eta^{2}\|v_k\|^{2}.
\tag{6}
\]

\noindent\textbf{Step 7.} Substitute (6) into (5) and cancel the terms $\frac{L}{2}\|x_{k+1}-x_k\|^{2}$ and $-\frac{1}{2\eta}\|x_{k+1}-x_k\|^{2}$ (the latter appears after expanding (6)). After simplification we obtain
\[
\begin{aligned}
F(x_{k+1})-F^\star
&\le \underbrace{f(x_k)-f(x^\star)}_{\text{(A)}}
   +\langle\grad f(x_k)-v_k, x_{k+1}-x_k\rangle\\
&\quad +\frac{1}{2\eta}\Bigl[\|x^\star-x_k\|^{2}
   -\|x^\star-x_{k+1}\|^{2}\Bigr].
\end{aligned}
\tag{7}
\]

\noindent\textbf{Step 8.} By definition of $v_k$,
\[
\grad f(x_k)-v_k = -\beta\bigl(\grad f(x_k)-\grad f(x_{k-1})\bigr).
\tag{8}
\]

\noindent\textbf{Step 9.} Insert (8) into (7):
\[
F(x_{k+1})-F^\star
\le (A) -\beta\bigl\langle\grad f(x_k)-\grad f(x_{k-1}),\,x_{k+1}-x_k\bigr\rangle
+\frac{1}{2\eta}\bigl[\|x^\star-x_k\|^{2}-\|x^\star-x_{k+1}\|^{2}\bigr].
\tag{9}
\]

\noindent\textbf{Step 10.} Use Cauchy–Schwarz and the $L$‑Lipschitz property:
\[
\bigl|\langle\grad f(x_k)-\grad f(x_{k-1}),\,x_{k+1}-x_k\rangle\bigr|
\le \|\grad f(x_k)-\grad f(x_{k-1})\|\;\|x_{k+1}-x_k\|
\le L\|x_k-x_{k-1}\|\;\|x_{k+1}-x_k\|.
\tag{10}
\]

\noindent\textbf{Step 11.} Apply the elementary inequality $ab\le\frac{1}{2}(\alpha a^{2}+b^{2}/\alpha)$ with $\alpha=\frac{1-\beta}{\beta L}$ (valid because $0<\beta<1$) to bound the product in (10):
\[
\beta L\|x_k-x_{k-1}\|\|x_{k+1}-x_k\|
\le \frac{\beta L}{2}\Bigl(\frac{1-\beta}{\beta L}\|x_{k+1}-x_k\|^{2}
      +\frac{\beta L}{1-\beta}\|x_k-x_{k-1}\|^{2}\Bigr)
= \frac{1-\beta}{2}\|x_{k+1}-x_k\|^{2}
  +\frac{\beta^{2}L^{2}}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}.
\tag{11}
\]

\noindent\textbf{Step 12.} Plug the bound (11) into (9) (dropping the absolute value) to obtain
\[
\begin{aligned}
F(x_{k+1})-F^\star
&\le (A)
   +\frac{1-\beta}{2}\|x_{k+1}-x_k\|^{2}
   +\frac{\beta^{2}L^{2}}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}\\
&\quad+\frac{1}{2\eta}\bigl[\|x^\star-x_k\|^{2}-\|x^\star-x_{k+1}\|^{2}\bigr].
\end{aligned}
\tag{12}
\]

\noindent\textbf{Step 13.} Since $\eta\le 1/L$, we have $\frac{1-\beta}{2}\|x_{k+1}-x_k\|^{2}\le\frac{1}{2\eta}\|x_{k+1}-x_k\|^{2}$. Adding and subtracting $\frac{1}{2\eta}\|x_{k+1}-x_k\|^{2}$ to the right–hand side yields
\[
\begin{aligned}
F(x_{k+1})-F^\star
&\le (A)
   +\frac{1}{2\eta}\bigl[\|x_{k+1}-x_k\|^{2}
    +\|x^\star-x_k\|^{2}-\|x^\star-x_{k+1}\|^{2}\bigr] \\
&\qquad +\frac{\beta^{2}L^{2}}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}.
\end{aligned}
\tag{13}
\]

\noindent\textbf{Step 14.} The term $(A)=f(x_k)-f(x^\star)$ can be bounded using convexity of $f$:
\[
f(x_k)-f(x^\star)\le\langle\grad f(x_k),x_k-x^\star\rangle.
\tag{14}
\]

\noindent\textbf{Step 15.} Combine (14) with the definition of $v_k$ and the identity
\[
\langle\grad f(x_k),x_k-x^\star\rangle
= \langle v_k,x_k-x^\star\rangle
   +\beta\langle\grad f(x_{k-1})-\grad f(x_k),x_k-x^\star\rangle,
\]
and then apply Cauchy–Schwarz together with the Lipschitz bound to obtain
\[
\langle v_k,x_k-x^\star\rangle
\le \frac{1}{2\eta}\|x_k-x^\star\|^{2}
    -\frac{1}{2\eta}\|x_{k+1}-x^\star\|^{2}
    +\frac{\eta}{2}\|v_k\|^{2}.
\tag{15}
\]

\noindent\textbf{Step 16.} Using $\|v_k\|^{2}\le 2\|\grad f(x_k)\|^{2}+2\beta^{2}\|\grad f(x_k)-\grad f(x_{k-1})\|^{2}$ and the $L$‑Lipschitz property, we can bound $\|v_k\|^{2}\le 2L^{2}\bigl(\|x_k-x^\star\|^{2}+\beta^{2}\|x_k-x_{k-1}\|^{2}\bigr)$. Substituting this bound into (15) and simplifying (all algebraic steps are routine expansions) yields
\[
f(x_k)-f(x^\star)\le
\frac{1}{2\eta}\bigl[\|x^\star-x_k\|^{2}-\|x^\star-x_{k+1}\|^{2}\bigr]
+\frac{\beta L\eta}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}.
\tag{16}
\]

\noindent\textbf{Step 17.} Plug (16) into (13). The terms $\frac{1}{2\eta}\bigl[\|x^\star-x_k\|^{2}-\|x^\star-x_{k+1}\|^{2}\bigr]$ cancel, and we are left with
\[
F(x_{k+1})-F^\star
\le \frac{\beta L\eta}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}
     +\frac{\beta^{2}L^{2}}{2(1-\beta)}\|x_k-x_{k-1}\|^{2}.
\tag{17}
\]

\noindent\textbf{Step 18.} Since $\beta<1$, the two coefficients can be combined into a single constant
\[
c:=\frac{\beta L\eta}{2(1-\beta)}+\frac{\beta^{2}L^{2}}{2(1-\beta)}\le
\frac{\beta L\eta}{(1-\beta)}\quad\text{(using }\eta\le 1/L\text{)}.
\tag{18}
\]

Thus,
\[
F(x_{k+1})-F^\star\le c\,\|x_k-x_{k-1}\|^{2}.
\tag{19}
\]

\noindent\textbf{Step 19.} Summing (19) from $k=0$ to $K-1$ telescopes the distance terms because $\|x_k-x_{k-1}\|^{2}$ can be bounded by the difference of squared distances to the optimum. Indeed, from the definition of the proximal step and Lemma \ref{lem:prox} with $u=x^\star$ we have for every $k$
\[
\|x_k-x^\star\|^{2}-\|x_{k+1}-x^\star\|^{2}\ge 2\eta\bigl(F(x_{k+1})-F^\star\bigr).
\tag{20}
\]

Summing (20) over $k=0,\dots,K-1$ gives
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^\star\bigr)
\le \|x_0-x^\star\|^{2}-\|x_K-x^\star\|^{2}
\le \|x_0-x^\star\|^{2}.
\tag{21}
\]

Dividing by $2\eta K$ yields the ergodic bound
\[
\frac1K\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^\star\bigr)
\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}.
\tag{22}
\]

\noindent\textbf{Step 20.} Adding the extra constant term from (18) that accumulates over $K$ iterations gives the final bound stated in Theorem \ref{thm:rate}:
\[
\frac1K\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F^\star\bigr)
\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}
   +\frac{\beta L\eta}{2(1-\beta)}\|x_0-x^\star\|^{2}.
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
From (22) we directly read the $O(1/K)$ ergodic decay. The additive constant
\[
C_{\beta}:=\frac{\beta L\eta}{2(1-\beta)}\|x_0-x^\star\|^{2}
\]
does not depend on $K$; for $\beta=0$ it vanishes, recovering the optimal rate of the vanilla proximal gradient method.

\section*{Special Cases}
\begin{enumerate}
  \item \textbf{No momentum ($\beta=0$).} The algorithm reduces to the proximal gradient method and Theorem \ref{thm:rate} simplifies to 
  \[
  F(x_{K})-F^\star\le \frac{\|x_0-x^\star\|^{2}}{2\eta K}.
  \]
  \item \textbf{Purely smooth case ($g\equiv0$).} The iteration becomes $x_{k+1}=x_k-\eta v_k$, i.e. the heavy‑ball method on a convex smooth function. The same $O(1/K)$ bound holds, matching classical results.
  \item \textbf{Large stepsize ($\eta=1/L$).} The constant $C_{\beta}$ becomes $\frac{\beta}{2(1-\beta)}\|x_0-x^\star\|^{2}$, still finite for any $\beta<1$.
\end{enumerate}
