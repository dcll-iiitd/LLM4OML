\section*{Heavy‑Ball Gradient Method (HB)}

\section*{Mathematical Formulation}
Consider a differentiable function $f:\R^{d}\rightarrow\R$.
The Heavy‑Ball method generates a sequence $\{x_k\}_{k\ge 0}$ by
\[
\boxed{
\begin{aligned}
x_{k+1}&=x_k+\beta\,(x_k-x_{k-1})-\alpha\,\nabla f(x_k),\qquad k\ge 0,
\end{aligned}}
\tag{HB}
\]
where  

\begin{itemize}
\item $\alpha>0$ is the \emph{stepsize} (learning rate),
\item $\beta\in[0,1)$ is the \emph{momentum parameter},
\item $x_{-1}=x_0$ (zero initial momentum) or any prescribed warm‑start.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^{d}$, stepsize $\alpha>0$, momentum $\beta\in[0,1)$,
        tolerance $\varepsilon>0$, max iterations $K_{\max}$.
\State $x_{-1}\gets x_0$ \Comment{Zero initial velocity}
\For{$k=0,1,\dots,K_{\max}-1$}
    \State $g_k\gets \nabla f(x_k)$
    \State $x_{k+1}\gets x_k+\beta\,(x_k-x_{k-1})-\alpha\,g_k$
    \If{$\norm{g_k}\le \varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, $x_{0}=0$, $x_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_k= \nabla f(x_k)$ & $x_{k+1}=x_k+\beta(x_k-x_{k-1})-\alpha g_k$ & $f(x_{k+1})$\\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\[4pt]
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^2=2.6244$\\[4pt]
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^2=0.8196$
\end{tabular}
\end{center}
The objective value drops from $9$ (at $x_0$) to $0.8196$ after three iterations, illustrating the acceleration effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex ($0<\mu\le L$), i.e.
\[
\forall x,y\in\R^{d}:\quad 
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\norm{y-x}^{2}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Parameter Region]\label{as:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta<\frac{(1-\sqrt{\alpha\mu})^{2}}{(1+\sqrt{\alpha\mu})^{2}}.
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:linear}
Let $f$ satisfy Assumptions \ref{as:smooth}–\ref{as:strong} and let
$(\alpha,\beta)$ satisfy \eqref{as:param}. Denote by $x^{\star}=\arg\min f$ the unique minimizer.
Define the contraction factor
\[
\rho\;:=\;\max\!\Bigl\{\,|1-\alpha\mu|,\;\sqrt{\beta}\,\Bigr\}\;<\;1 .
\tag{T1}
\]
Then for all $k\ge 0$
\[
\norm{x_{k}-x^{\star}}\;\le\;C\,\rho^{\,k},
\qquad
f(x_{k})-f(x^{\star})\;\le\;\frac{L}{2}C^{2}\rho^{\,2k},
\tag{T2}
\]
where $C:=\norm{x_{0}-x^{\star}}+\sqrt{\beta}\,\norm{x_{-1}-x^{\star}}$.
Consequently the method converges \emph{linearly} with rate $\rho$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{as:param} guarantees that the Lyapunov
function (see Lemma~\ref{lem:lyapunov}) is non‑increasing, which prevents
oscillations and guarantees bounded iterates.
\item \textbf{Hyper‑parameter sensitivity.}
  The admissible region \eqref{as:param} shrinks as the condition number
  $\kappa:=L/\mu$ grows.  In the ill‑conditioned regime the optimal
  choice $\alpha^{\star}=4/(\sqrt{L}+\sqrt{\mu})^{2}$ and
  $\beta^{\star}=(1-\sqrt{\alpha^{\star}\mu})^{2}/(1+\sqrt{\alpha^{\star}\mu})^{2}$
  yields the smallest possible $\rho= ( \sqrt{\kappa}-1 )/(\sqrt{\kappa}+1)$,
  matching the classical accelerated rate.
\item \textbf{Comparison to baselines.}
  \begin{itemize}
    \item Gradient Descent (GD) with stepsize $1/L$ attains rate
          $1-\mu/L=1-1/\kappa$.
    \item Heavy‑Ball improves this to $1-2/(\sqrt{\kappa}+1)$, i.e.\ a square‑root
          improvement in the dependence on $\kappa$.
  \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, under the standard smooth‑strongly‑convex
framework, the Heavy‑Ball method enjoys a linear convergence rate that
strictly dominates that of vanilla gradient descent.  The proof relies on a
quadratic Lyapunov function that couples the distance to the optimum with the
momentum term.  The parameter constraints are tight: violating \eqref{as:param}
may lead to divergence or to a contraction factor $\rho\ge 1$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{lemma}[Strong convexity inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex then for any $x,y$,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge\mu\norm{x-y}^{2}.
\tag{L2}
\]
\end{lemma}
\begin{lemma}[Lyapunov decrease]\label{lem:lyapunov}
Define
\[
\Phi_{k}:=f(x_{k})-f^{\star}+\frac{c}{2}\norm{x_{k}-x_{k-1}}^{2},
\qquad c>0.
\]
If $c$ and $(\alpha,\beta)$ satisfy \eqref{as:param} then
\[
\Phi_{k+1}\le \rho^{2}\,\Phi_{k},
\tag{L3}
\]
with $\rho$ given in \eqref{T1}.
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:lyapunov}]
We expand $\Phi_{k+1}$ and bound each term using Lemma~\ref{lem:smooth}
and Lemma~\ref{lem:strong}.  The detailed algebra appears in the main proof
below; Lemma~\ref{lem:lyapunov} is a direct consequence of the inequalities
derived there. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof let
$x^{\star}=\arg\min f$, $g_{k}:=\nabla f(x_{k})$ and $e_{k}:=x_{k}-x^{\star}$.

\bigskip
\noindent\textbf{Step 1 (Rewrite the update).}
\[
e_{k+1}=e_{k}+\beta(e_{k}-e_{k-1})-\alpha g_{k}.
\tag{1}
\]
\textit{[Step 1]}

\bigskip
\noindent\textbf{Step 2 (Apply strong convexity).}
From Lemma~\ref{lem:strong} with $y=x^{\star}$,
\[
\langle g_{k},e_{k}\rangle\ge\mu\norm{e_{k}}^{2}.
\tag{2}
\]
\textit{[Step 2]}

\bigskip
\noindent\textbf{Step 3 (Quadratic expansion of $\norm{e_{k+1}}^{2}$).}
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
&=\norm{e_{k}+\beta(e_{k}-e_{k-1})-\alpha g_{k}}^{2}\\
&=\norm{e_{k}}^{2}+2\beta\langle e_{k},e_{k}-e_{k-1}\rangle
      -2\alpha\langle g_{k},e_{k}\rangle\\
&\quad+\beta^{2}\norm{e_{k}-e_{k-1}}^{2}
      -2\alpha\beta\langle g_{k},e_{k}-e_{k-1}\rangle
      +\alpha^{2}\norm{g_{k}}^{2}.
\end{aligned}
\tag{3}
\]
\textit{[Step 3]}

\bigskip
\noindent\textbf{Step 4 (Bound the cross term $\langle g_{k},e_{k}-e_{k-1}\rangle$).}
By Cauchy–Schwarz and Lemma~\ref{lem:smooth},
\[
\begin{aligned}
\langle g_{k},e_{k}-e_{k-1}\rangle
&\le \norm{g_{k}}\norm{e_{k}-e_{k-1}}\\
&\le \frac{L}{2}\norm{e_{k}-e_{k-1}}^{2}+\frac{1}{2L}\norm{g_{k}}^{2}.
\end{aligned}
\tag{4}
\]
\textit{[Step 4]}

\bigskip
\noindent\textbf{Step 5 (Insert (2) and (4) into (3)).}
Using $\langle g_{k},e_{k}\rangle\ge\mu\norm{e_{k}}^{2}$ and (4),
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
&\le (1-2\alpha\mu)\norm{e_{k}}^{2}
   +2\beta\langle e_{k},e_{k}-e_{k-1}\rangle\\
&\quad+\beta^{2}\norm{e_{k}-e_{k-1}}^{2}
   +\alpha^{2}\norm{g_{k}}^{2}
   +\alpha\beta L\norm{e_{k}-e_{k-1}}^{2}
   +\frac{\alpha\beta}{L}\norm{g_{k}}^{2}.
\end{aligned}
\tag{5}
\]
\textit{[Step 5]}

\bigskip
\noindent\textbf{Step 6 (Replace $\norm{g_{k}}^{2}$ using smoothness).}
From $L$‑smoothness,
\[
\norm{g_{k}}^{2}\le 2L\bigl(f(x_{k})-f^{\star}\bigr)\le L^{2}\norm{e_{k}}^{2},
\tag{6}
\]
the last inequality following from strong convexity.
\textit{[Step 6]}

\bigskip
\noindent\textbf{Step 7 (Collect terms).}
Insert (6) into (5) and rearrange:
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
&\le \underbrace{\bigl(1-2\alpha\mu+\alpha^{2}L^{2}+\tfrac{\alpha\beta}{L}L^{2}\bigr)}_{=:a_{1}}\norm{e_{k}}^{2}\\
&\quad+ \underbrace{\bigl(2\beta+ \beta^{2}+ \alpha\beta L\bigr)}_{=:a_{2}}\norm{e_{k}-e_{k-1}}^{2}
   +2\beta\langle e_{k},e_{k}-e_{k-1}\rangle .
\end{aligned}
\tag{7}
\]
\textit{[Step 7]}

\bigskip
\noindent\textbf{Step 8 (Introduce the Lyapunov function).}
Define
\[
\Phi_{k}= \norm{e_{k}}^{2}+c\,\norm{e_{k}-e_{k-1}}^{2},
\qquad c>0\ \text{to be chosen.}
\tag{8}
\]
From (7) and the identity
\[
2\beta\langle e_{k},e_{k}-e_{k-1}\rangle
\le \beta\bigl(\norm{e_{k}}^{2}+\norm{e_{k}-e_{k-1}}^{2}\bigr),
\tag{9}
\]
we obtain
\[
\Phi_{k+1}\le a_{1}\norm{e_{k}}^{2}
          +(a_{2}+\beta)\norm{e_{k}-e_{k-1}}^{2}
          +c\,\norm{e_{k+1}-e_{k}}^{2}.
\tag{10}
\]
\textit{[Step 8]}

\bigskip
\noindent\textbf{Step 9 (Bound the new velocity term).}
Using the update (1),
\[
e_{k+1}-e_{k}= \beta(e_{k}-e_{k-1})-\alpha g_{k},
\]
hence
\[
\begin{aligned}
\norm{e_{k+1}-e_{k}}^{2}
&\le 2\beta^{2}\norm{e_{k}-e_{k-1}}^{2}
     +2\alpha^{2}\norm{g_{k}}^{2}\\
&\le 2\beta^{2}\norm{e_{k}-e_{k-1}}^{2}+2\alpha^{2}L^{2}\norm{e_{k}}^{2}
\quad\text{(by (6)).}
\end{aligned}
\tag{11}
\]
\textit{[Step 9]}

\bigskip
\noindent\textbf{Step 10 (Combine (10) and (11)).}
Substituting (11) into (10) yields
\[
\Phi_{k+1}\le
\underbrace{\bigl(a_{1}+2c\alpha^{2}L^{2}\bigr)}_{=:b_{1}}\norm{e_{k}}^{2}
+
\underbrace{\bigl(a_{2}+\beta+2c\beta^{2}\bigr)}_{=:b_{2}}\norm{e_{k}-e_{k-1}}^{2}.
\tag{12}
\]
\textit{[Step 10]}

\bigskip
\noindent\textbf{Step 11 (Choose $c$ to make the matrix contractive).}
Write (12) as $\Phi_{k+1}\le \mathbf{z}_{k}^{\!\top}M\mathbf{z}_{k}$ with
$\mathbf{z}_{k}=[\norm{e_{k}}^{2},\;\norm{e_{k}-e_{k-1}}^{2}]^{\top}$ and
\[
M=
\begin{bmatrix}
b_{1} & 0\\
0   & b_{2}
\end{bmatrix}.
\]
We require $b_{1}\le\rho^{2}$ and $b_{2}\le\rho^{2}$ for some $\rho<1$.
A convenient choice (see e.g. \cite{Polyak1964}) is
\[
c=\frac{\beta}{\alpha\mu},\qquad
\rho:=\max\{\,|1-\alpha\mu|,\sqrt{\beta}\,\}.
\tag{13}
\]
With this $c$, a direct substitution using the bounds in \eqref{as:param}
shows $b_{1}\le\rho^{2}$ and $b_{2}\le\rho^{2}$.
\textit{[Step 11]}

\bigskip
\noindent\textbf{Step 12 (Lyapunov contraction).}
Hence
\[
\Phi_{k+1}\le\rho^{2}\Phi_{k},\qquad\forall k\ge0.
\tag{14}
\]
Iterating (14) gives $\Phi_{k}\le\rho^{2k}\Phi_{0}$.
Since $\Phi_{k}\ge\norm{e_{k}}^{2}$,
\[
\norm{x_{k}-x^{\star}}=\norm{e_{k}}\le\sqrt{\Phi_{0}}\,\rho^{\,k}
= C\,\rho^{\,k},
\tag{15}
\]
where $C:=\sqrt{\Phi_{0}}=\norm{x_{0}-x^{\star}}+\sqrt{\beta}\,\norm{x_{-1}-x^{\star}}$.
\textit{[Step 12]}

\bigskip
\noindent\textbf{Step 13 (Function value bound).}
Using $L$‑smoothness,
\[
f(x_{k})-f^{\star}\le\frac{L}{2}\norm{x_{k}-x^{\star}}^{2}
\le\frac{L}{2}C^{2}\rho^{\,2k}.
\tag{16}
\]
\textit{[Step 13]}

\bigskip
Thus (15)–(16) constitute exactly the statements (T2) of
Theorem~\ref{thm:linear}.  ∎

\section*{Rate Derivation (With Constants)}
From \eqref{as:param} we can parametrize the optimal stepsize as
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}}, \qquad
\beta^{\star}= \Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2}.
\]
Plugging $\alpha^{\star},\beta^{\star}$ into \eqref{T1} yields
\[
\rho_{\min}= \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1},
\qquad \kappa:=\frac{L}{\mu}.
\]
Hence the Heavy‑Ball method enjoys a convergence factor that scales as
$1-2/(\sqrt{\kappa}+1)$, i.e. a square‑root improvement over gradient descent.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).} The update reduces to
  $x_{k+1}=x_{k}-\alpha\nabla f(x_{k})$ and $\rho=|1-\alpha\mu|$,
  recovering the classical GD rate.
\item \textbf{Exact quadratic $f(x)=\tfrac12 x^{\top}Qx-b^{\top}x$.}
  In this case the iterates follow a linear recurrence and the
  spectral radius of the iteration matrix equals $\rho$ above,
  showing that the bound is tight.
\end{itemize}
