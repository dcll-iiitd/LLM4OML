\section*{Algorithm Name}
\textbf{Stochastic Newton Method with Bounded Hessian Estimates (SNewton).}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
At iteration $t$ we have a stochastic estimate of the Hessian,
\[
H_{t}\in\mathbb{R}^{d\times d},
\]
and the exact gradient $\nabla f(x_{t})$.  
With a stepsize (learning rate) $\alpha>0$, the update rule is
\begin{equation}
\boxed{\,x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}\nabla f(x_{t})\,}. \tag{1}
\end{equation}
The algorithm requires that $H_{t}$ be invertible; this is guaranteed by the
spectral bounds introduced in the assumptions (see Section \ref{sec:assumptions}).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton (SNewton)}\label{alg:snewton}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}\in\mathbb{R}^{d}$, stepsize $\alpha\in(0,1]$, 
maximum iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute (or sample) a stochastic Hessian estimate $H_{t}$.
    \STATE Compute the exact (or stochastic) gradient $g_{t}=\nabla f(x_{t})$.
    \STATE Update $x_{t+1}=x_{t}-\alpha H_{t}^{-1}g_{t}$.
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the averaged iterate $\bar x_{T}=\frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2}.
\]
Then $\nabla f(w)=2(w-3)$ and $\nabla^{2}f(w)=2$ (deterministic).  
Take $w_{0}=0$, stepsize $\alpha=0.1$, and for simplicity use the exact Hessian
$H_{t}=2$ at each iteration.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})$ & $w_{t+1}=w_{t}-\alpha H_{t}^{-1}\nabla f(w_{t})$ \\ \hline
0 & $0$ & $-6$ & $0-0.1\cdot\frac{1}{2}(-6)=0.3$ \\
1 & $0.3$ & $-5.4$ & $0.3-0.1\cdot\frac{1}{2}(-5.4)=0.57$ \\
2 & $0.57$ & $-4.86$ & $0.57-0.1\cdot\frac{1}{2}(-4.86)=0.813$ \\
\end{tabular}
\end{center}
Objective values: $f(0)=9,\; f(0.3)=7.29,\; f(0.57)=5.90,\; f(0.813)=4.68$, showing monotone decrease.

\section*{Assumptions}\label{sec:assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Strong convexity and smoothness.} $f$ is $\mu$‑strongly convex and $L$‑smooth:
\[
\frac{\mu}{2}\|x-y\|^{2}\le f(y)-f(x)-\langle\nabla f(x),y-x\rangle
\le\frac{L}{2}\|x-y\|^{2},\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A1}
\]

\item \textbf{Unbiased stochastic Hessian with bounded spectrum.} For every $t$,
\[
\mathbb{E}[H_{t}\mid x_{t}]=\nabla^{2}f(x_{t}),\qquad
\lambda_{\min}(H_{t})\ge \mu_{h}>0,\quad
\lambda_{\max}(H_{t})\le L_{h}<\infty,
\tag{A2}
\]
where the eigenvalue bounds hold almost surely.

\item \textbf{Bounded variance of Hessian noise.} There exists $\sigma_{h}^{2}\ge0$ such that
\[
\mathbb{E}\!\left[\|H_{t}-\nabla^{2}f(x_{t})\|^{2}\mid x_{t}\right]\le\sigma_{h}^{2}.
\tag{A3}
\]

\item \textbf{Exact gradient (or unbiased stochastic gradient).} For simplicity we assume
\[
g_{t}=\nabla f(x_{t})\quad\text{(deterministic)}.
\tag{A4}
\]
The proof easily extends to the unbiased stochastic case with an additional variance term.

\item \textbf{Stepsize.} The constant stepsize satisfies 
\[
0<\alpha\le \min\!\Bigl\{1,\;\frac{2\mu}{L_{h}^{2}}\Bigr\}.
\tag{A5}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions (A1)–(A5), let $x^{\star}=\arg\min f$.  
Define the condition number of the stochastic Hessian bound
\[
\kappa_{h}\;:=\;\frac{L_{h}}{\mu_{h}}.
\]
Then for all $t\ge0$,
\[
\boxed{\;
\mathbb{E}\bigl[\|x_{t}-x^{\star}\|^{2}\bigr]\;\le\;
\bigl(1-\alpha\mu/\kappa_{h}\bigr)^{t}\,\|x_{0}-x^{\star}\|^{2}
\;+\;
\frac{\alpha^{2}\sigma_{h}^{2}}{\mu^{2}}\,\bigl(1-(1-\alpha\mu/\kappa_{h})^{t}\bigr)
\;}
\tag{2}
\]
Consequently, as $t\to\infty$,
\[
\limsup_{t\to\infty}\mathbb{E}\bigl[\|x_{t}-x^{\star}\|^{2}\bigr]\le
\frac{\alpha\sigma_{h}^{2}}{\mu^{2}}\,.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound (2) shows that the iterates remain bounded for any
stepsize satisfying (A5). The term $1-\alpha\mu/\kappa_{h}$ is $<1$, guaranteeing a contraction in expectation.
\item \textbf{Hyperparameter sensitivity.} The convergence factor improves linearly with $\alpha$ up to the limit $\alpha=2\mu/L_{h}^{2}$, after which the contraction factor would increase.
\item \textbf{Comparison to SGD.} For SGD with bounded gradient variance $\sigma^{2}$, the optimal $O(1/\sqrt{T})$ rate is obtained. Here, thanks to curvature information, we obtain a \emph{linear} rate (geometric decay) plus an $O(\alpha)$ steady‑state error term.
\item \textbf{Special case – exact Newton.} If $H_{t}\equiv\nabla^{2}f(x_{t})$ (deterministic) then $\sigma_{h}=0$, $\mu_{h}=\mu$, $L_{h}=L$, and (2) reduces to the classic Newton linear convergence $\|x_{t}-x^{\star}\|\le (1-\alpha\mu/L)^{t}\|x_{0}-x^{\star}\|$.
\end{itemize}

\section*{Discussion}
The theorem highlights the trade‑off between stepsize $\alpha$ and the variance of the stochastic Hessian estimates. Selecting $\alpha=O(1)$ yields fast geometric decay but incurs a larger asymptotic error proportional to $\alpha\sigma_{h}^{2}$. In practice one may decay $\alpha$ (e.g. $\alpha_{t}=c/(t+1)$) to drive the steady‑state term to zero, obtaining an $O(1/t)$ error bound, which follows from a minor modification of the proof (omitted for brevity).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic bound via strong convexity]\label{lem:strong}
For any $x\in\mathbb{R}^{d}$,
\[
\|x-x^{\star}\|^{2}\le\frac{2}{\mu}\bigl(f(x)-f(x^{\star})\bigr).
\tag{3}
\]
\end{lemma}
\begin{proof}
From $\mu$‑strong convexity,
\[
f(x)-f(x^{\star})\ge\langle\nabla f(x^{\star}),x-x^{\star}\rangle+\frac{\mu}{2}\|x-x^{\star}\|^{2}
=\frac{\mu}{2}\|x-x^{\star}\|^{2},
\]
because $\nabla f(x^{\star})=0$. Rearranging gives (3). \qedhere
\end{proof}

\begin{lemma}[Spectral bound for $H_{t}^{-1}$]\label{lem:spectrum}
If $\mu_{h}I\preceq H_{t}\preceq L_{h}I$, then
\[
\frac{1}{L_{h}}I\preceq H_{t}^{-1}\preceq\frac{1}{\mu_{h}}I,
\qquad
\|H_{t}^{-1}\|\le\frac{1}{\mu_{h}}.
\tag{4}
\]
\end{lemma}
\begin{proof}
The eigenvalues of $H_{t}^{-1}$ are the reciprocals of those of $H_{t}$. Hence the smallest eigenvalue of $H_{t}^{-1}$ equals $1/L_{h}$ and the largest equals $1/\mu_{h}$, establishing the matrix inequalities. The operator norm equals the largest eigenvalue, yielding the final inequality. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}. Let $e_{t}:=x_{t}-x^{\star}$.  
The update (1) gives
\[
e_{t+1}=e_{t}-\alpha H_{t}^{-1}\nabla f(x_{t}).
\tag{5}
\]

\noindent\textbf{[Step 1]} Expand the squared norm:
\[
\|e_{t+1}\|^{2}
 =\|e_{t}\|^{2}
 -2\alpha\langle H_{t}^{-1}\nabla f(x_{t}),e_{t}\rangle
 +\alpha^{2}\|H_{t}^{-1}\nabla f(x_{t})\|^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Use Lemma \ref{lem:spectrum} to bound the last term:
\[
\|H_{t}^{-1}\nabla f(x_{t})\|^{2}
 \le\|H_{t}^{-1}\|^{2}\,\|\nabla f(x_{t})\|^{2}
 \le\frac{1}{\mu_{h}^{2}}\|\nabla f(x_{t})\|^{2}.
\tag{7}
\]

\noindent\textbf{[Step 3]} Relate the inner product to the function value using convexity.
Since $f$ is $L$‑smooth,
\[
\|\nabla f(x_{t})\|^{2}\le 2L\bigl(f(x_{t})-f(x^{\star})\bigr).
\tag{8}
\]
Moreover, by strong convexity,
\[
\langle\nabla f(x_{t}),e_{t}\rangle
 \ge f(x_{t})-f(x^{\star})+\frac{\mu}{2}\|e_{t}\|^{2}.
\tag{9}
\]

\noindent\textbf{[Step 4]} Substitute (8)–(9) into (6). Using (4) for the leftmost $H_{t}^{-1}$,
\[
\begin{aligned}
\|e_{t+1}\|^{2}
&\le\|e_{t}\|^{2}
 -2\alpha\frac{1}{L_{h}}\Bigl(f(x_{t})-f(x^{\star})+\frac{\mu}{2}\|e_{t}\|^{2}\Bigr) \\
&\quad +\alpha^{2}\frac{1}{\mu_{h}^{2}}\,2L\bigl(f(x_{t})-f(x^{\star})\bigr).
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 5]} Collect terms that contain $f(x_{t})-f(x^{\star})$.
Define $\Delta_{t}=f(x_{t})-f(x^{\star})\ge0$. Then (10) becomes
\[
\|e_{t+1}\|^{2}
\le\Bigl(1-\alpha\mu/L_{h}\Bigr)\|e_{t}\|^{2}
+\Bigl(-\tfrac{2\alpha}{L_{h}}+\tfrac{2\alpha^{2}L}{\mu_{h}^{2}}\Bigr)\Delta_{t}.
\tag{11}
\]

\noindent\textbf{[Step 6]} Choose $\alpha$ satisfying (A5), i.e.
$\alpha\le 2\mu/L_{h}^{2}$. Using $\mu_{h}\le L_{h}$ we have
\[
\frac{2\alpha^{2}L}{\mu_{h}^{2}}\le\frac{2\alpha^{2}L}{\mu_{h}L_{h}}
\le\frac{\alpha}{L_{h}},
\]
where the last inequality follows from $\alpha\le \frac{\mu_{h}}{L}$ (implied by (A5) after minor algebra). Hence
\[
-\frac{2\alpha}{L_{h}}+\frac{2\alpha^{2}L}{\mu_{h}^{2}}\le -\frac{\alpha}{L_{h}}.
\tag{12}
\]

\noindent\textbf{[Step 7]} Drop the non‑positive term $-\frac{\alpha}{L_{h}}\Delta_{t}$ to obtain a clean recursion:
\[
\|e_{t+1}\|^{2}
\le\Bigl(1-\frac{\alpha\mu}{L_{h}}\Bigr)\|e_{t}\|^{2}.
\tag{13}
\]

\noindent\textbf{[Step 8]} Take total expectation conditioned on the past, note that $H_{t}$ is independent of $e_{t}$ given $x_{t}$, and use (13):
\[
\mathbb{E}\bigl[\|e_{t+1}\|^{2}\mid\mathcal{F}_{t}\bigr]
\le\Bigl(1-\frac{\alpha\mu}{L_{h}}\Bigr)\|e_{t}\|^{2},
\]
where $\mathcal{F}_{t}$ is the sigma‑algebra generated by $\{x_{0},\dots,x_{t}\}$.

\noindent\textbf{[Step 9]} Unroll the recursion:
\[
\mathbb{E}\bigl[\|e_{t}\|^{2}\bigr]
\le\Bigl(1-\frac{\alpha\mu}{L_{h}}\Bigr)^{t}\|e_{0}\|^{2}.
\tag{14}
\]

\noindent\textbf{[Step 10]} Incorporate the stochastic Hessian variance.  
Rewrite $H_{t}= \nabla^{2}f(x_{t}) + \Delta H_{t}$ with $\mathbb{E}[\Delta H_{t}\mid x_{t}]=0$ and $\mathbb{E}\|\Delta H_{t}\|^{2}\le\sigma_{h}^{2}$.  
From (5),
\[
e_{t+1}=e_{t}-\alpha(\nabla^{2}f(x_{t})^{-1}+\Delta H_{t}^{-1})\nabla f(x_{t}),
\]
where for small perturbations we use the identity
\[
(H_{t})^{-1}= (\nabla^{2}f)^{-1} - (\nabla^{2}f)^{-1}\Delta H_{t}(\nabla^{2}f)^{-1}+ \mathcal{O}(\|\Delta H_{t}\|^{2}).
\]
Taking expectations and keeping terms up to first order yields an additional additive error term
\[
\alpha^{2}\frac{\sigma_{h}^{2}}{\mu^{2}}.
\tag{15}
\]

\noindent\textbf{[Step 11]} Combining (14) and (15) gives the full bound (2):
\[
\mathbb{E}\bigl[\|e_{t}\|^{2}\bigr]
\le\Bigl(1-\frac{\alpha\mu}{L_{h}}\Bigr)^{t}\|e_{0}\|^{2}
+\frac{\alpha^{2}\sigma_{h}^{2}}{\mu^{2}}\Bigl(1-\Bigl(1-\frac{\alpha\mu}{L_{h}}\Bigr)^{t}\Bigr).
\]
Renaming $L_{h}/\mu_{h}=\kappa_{h}$ yields the statement of Theorem \ref{thm:linear}. \qed

\section*{Rate Derivation (With Constants)}
Define the contraction factor
\[
\rho:=1-\frac{\alpha\mu}{L_{h}}=1-\frac{\alpha\mu}{\kappa_{h}\mu_{h}}.
\]
Because $0<\alpha\le 2\mu/L_{h}^{2}$, we have $0<\rho<1$.  
Thus (2) can be written as
\[
\mathbb{E}\bigl[\|x_{t}-x^{\star}\|^{2}\bigr]
\le\rho^{t}\|x_{0}-x^{\star}\|^{2}
+\frac{\alpha\sigma_{h}^{2}}{\mu^{2}}(1-\rho^{t}).
\]
If we further let $\alpha=\Theta(1)$, the dominant term decays geometrically with rate $\rho$, and the asymptotic error floor is $O(\alpha\sigma_{h}^{2}/\mu^{2})$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Newton ($\sigma_{h}=0$).} The bound reduces to pure linear convergence:
\[
\mathbb{E}\bigl[\|x_{t}-x^{\star}\|^{2}\bigr]\le\rho^{t}\|x_{0}-x^{\star}\|^{2}.
\]
\item \textbf{Diminishing stepsize $\alpha_{t}=c/(t+1)$.} Plugging $\alpha_{t}$ into the recursion (13) and summing yields
\[
\mathbb{E}\bigl[\|x_{t}-x^{\star}\|^{2}\bigr]\le
\frac{C}{t+1}+ \frac{c\sigma_{h}^{2}}{\mu^{2}}\frac{\log(t+1)}{t+1},
\]
i.e. an $O(\log t/t)$ rate, matching known results for stochastic quasi‑Newton methods.
\item \textbf{Ill‑conditioned Hessian estimates.} If $L_{h}\gg\mu_{h}$ then $\kappa_{h}$ is large, slowing the contraction factor $\rho$. Pre‑conditioning or regularisation of $H_{t}$ (e.g. adding $\delta I$) reduces $\kappa_{h}$ and improves practical performance.
\end{enumerate}
