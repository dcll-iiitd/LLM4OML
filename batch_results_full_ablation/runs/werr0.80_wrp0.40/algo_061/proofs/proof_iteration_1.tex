\section*{Top-$K$ Stochastic Gradient Descent (Top-$K$ SGD)}
%=====================================================================

%---------------------------------------------------------------------
\section*{Mathematical Formulation}
%---------------------------------------------------------------------
Let $f:\RR^{d}\to\RR$ be a differentiable (possibly non‑convex) loss function.
At iteration $t$ we draw a stochastic sample $\xi_t$ and compute a stochastic
gradient $g_t:=\nabla f(w_t;\xi_t)\in\RR^{d}$.
Define the \emph{top‑$K$ sparsifier}
\[
\topk{g}:= \bigl[\,g_i\cdot\mathbf 1\{\,|g_i|\ge \tau_K(g)\,\}\,\bigr]_{i=1}^{d},
\qquad 
\tau_K(g)=\text{the $K$‑th largest magnitude among }\{|g_i|\}_{i=1}^{d}.
\]
Thus $\topk{g}$ retains exactly $K$ entries (or fewer if ties occur) with
largest absolute values and zeros out the rest.

The update rule is
\begin{equation}\label{eq:update}
w_{t+1}= w_t - \eta_t \,\topk{g_t},
\end{equation}
where $\{\eta_t\}_{t\ge 0}$ is a stepsize schedule (e.g. $\eta_t=\eta/\sqrt{t+1}$).

---------------------------------------------------------------------
\section*{Pseudocode}
---------------------------------------------------------------------

\begin{algorithm}[H]
\caption{Top-$K$ SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\RR^{d}$, stepsizes $\{\eta_t\}$, sparsity $K$.
\For{$t=0,1,\dots,T-1$}
    \State Sample a minibatch (or a single example) $\xi_t$.
    \State Compute stochastic gradient $g_t \gets \nabla f(w_t;\xi_t)$.
    \State $\displaystyle \topk{g_t}\gets$ keep the $K$ entries of $g_t$ with largest $|\,\cdot\,|$, set others to $0$.
    \State Update $w_{t+1}\gets w_t - \eta_t\,\topk{g_t}$.
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

---------------------------------------------------------------------
\section*{Dry Run Example}
---------------------------------------------------------------------
Consider the scalar quadratic $f(w)=(w-3)^2$.
Here $d=1$, thus $\topk{g}=g$ for any $K\ge1$.
Take $w_0=0$, constant stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\\hline
0 & $2(w_0-3) = -6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3) = -4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3) = -3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.368896$
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the basic
behaviour of \eqref{eq:update} even with the sparsifier.

%=====================================================================
\section*{Assumptions}
%=====================================================================

\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
f(u)\le f(v)+\inner{\nabla f(v)}{u-v}+ \frac{L}{2}\norm{u-v}^2,
\qquad\forall u,v\in\RR^{d}.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, conditioned on $w_t$,
\[
\EE[g_t\mid w_t]=\nabla f(w_t),\qquad 
\EE\!\left[\norm{g_t-\nabla f(w_t)}^2\mid w_t\right]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Unbiased Top‑$K$ Sparsifier]\label{ass:topk_unbiased}
The random top‑$K$ operator satisfies
\[
\EE\bigl[\topk{g_t}\mid w_t\bigr]=\nabla f(w_t),\qquad 
\EE\!\left[\norm{\topk{g_t}}^2\mid w_t\right]
\le \rho\,\norm{\nabla f(w_t)}^2+\sigma^2 ,
\]
for some constant $\rho\ge 1$ that depends only on $K$ and $d$.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes are non‑increasing, positive and satisfy
\[
\sum_{t=0}^{\infty}\eta_t = \infty,\qquad
\sum_{t=0}^{\infty}\eta_t^{\,2}<\infty .
\]
A concrete choice used in the sequel is $\displaystyle\eta_t=\frac{\eta}{\sqrt{t+1}}$
with $\eta>0$.
\end{assumption}

%=====================================================================
\section*{Main Convergence Theorem}
%=====================================================================

\begin{theorem}\label{thm:convergence}
Assume \ref{ass:smooth}–\ref{ass:stepsize} hold.
Let $\{w_t\}$ be generated by Top‑$K$ SGD \eqref{eq:update}.
Define the (random) averaged gradient norm
\[
\Phi_T := \frac{1}{\sum_{t=0}^{T-1}\eta_t}\sum_{t=0}^{T-1}\eta_t\,
\EE\!\left[\norm{\nabla f(w_t)}^2\right].
\]
Then for any $T\ge1$,
\[
\Phi_T \;\le\;
\frac{2\bigl(f(w_0)-f^\star\bigr)}{L\sum_{t=0}^{T-1}\eta_t}
\;+\;
\frac{2\rho L}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t^{\,2}
\;+\;
\frac{2L\sigma^2}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t^{\,2},
\]
where $f^\star:=\inf_{w}f(w)$. 
In particular, with $\eta_t=\eta/\sqrt{t+1}$,
\[
\Phi_T = \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

%=====================================================================
\section*{Theoretical Properties}
%=====================================================================

\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound involves $\rho$, which grows roughly as $d/K$.
Hence larger $K$ (less sparsification) yields a smaller $\rho$ and
tighter convergence guarantees.
\item \textbf{Hyperparameter Sensitivity.}
The stepsize condition \ref{ass:stepsize} is identical to that for vanilla SGD;
the same $\eta_t=\Theta(1/\sqrt{t})$ schedule guarantees $\mathcal{O}(1/\sqrt{T})$
gradient norm decay.
\item \textbf{Comparison to Baselines.}
When $K=d$, $\topk{g}=g$ and $\rho=1$, the theorem reduces to the classical
non‑convex SGD rate. For $K\ll d$, the extra factor $\rho$ quantifies the price
paid for communication reduction.
\end{itemize}

%=====================================================================
\section*{Proof of Theorem \ref{thm:convergence}}
%=====================================================================

\subsection*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $w$, $u$,
\[
f(u) \le f(w) + \inner{\nabla f(w)}{u-w}
      + \frac{L}{2}\norm{u-w}^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see Assumption \ref{ass:smooth}.
\end{proof}

\begin{lemma}[Bound on the second moment of the sparsified gradient]\label{lem:second_moment}
Under Assumption \ref{ass:topk_unbiased},
\[
\EE\!\left[\norm{\topk{g_t}}^2\mid w_t\right]
\le \rho\,\norm{\nabla f(w_t)}^2 + \sigma^2 .
\]
\end{lemma}
\begin{proof}
Directly the second part of Assumption \ref{ass:topk_unbiased}.
\end{proof}

\subsection*{Main Proof (Step‑by‑Step Derivation)}
We present every algebraic manipulation and annotate non‑trivial steps.

\noindent\textbf{Step 1.} Apply Lemma~\ref{lem:smooth} with $w=w_t$ and
$u=w_{t+1}=w_t-\eta_t\topk{g_t}$:
\begin{align}
f(w_{t+1})
&\le f(w_t) - \eta_t\inner{\nabla f(w_t)}{\topk{g_t}}
   + \frac{L\eta_t^{\,2}}{2}\norm{\topk{g_t}}^2 .
\label{eq:smooth_step}
\end{align}
[Step 1]

\noindent\textbf{Step 2.} Take conditional expectation w.r.t.\ the randomness at
iteration $t$ (denoted $\EE_t[\cdot]=\EE[\cdot\mid w_t]$) on both sides of
\eqref{eq:smooth_step}:
\begin{align}
\EE_t\!\bigl[f(w_{t+1})\bigr]
&\le f(w_t) - \eta_t\inner{\nabla f(w_t)}{\EE_t[\topk{g_t}]}
   + \frac{L\eta_t^{\,2}}{2}\EE_t\!\bigl[\norm{\topk{g_t}}^2\bigr].
\label{eq:exp_smooth}
\end{align}
[Step 2]

\noindent\textbf{Step 3.} Replace the expectations using Assumptions
\ref{ass:topk_unbiased} and Lemma~\ref{lem:second_moment}:
\begin{align}
\EE_t\!\bigl[f(w_{t+1})\bigr]
&\le f(w_t) - \eta_t\norm{\nabla f(w_t)}^2
   + \frac{L\eta_t^{\,2}}{2}\bigl(\rho\norm{\nabla f(w_t)}^2+\sigma^2\bigr).
\label{eq:after_subst}
\end{align}
[Step 3]

\noindent\textbf{Step 4.} Rearrange \eqref{eq:after_subst} to isolate the
gradient norm term:
\begin{align}
\eta_t\bigl(1-\tfrac{L\rho}{2}\eta_t\bigr)\norm{\nabla f(w_t)}^2
&\le f(w_t)-\EE_t\!\bigl[f(w_{t+1})\bigr]
   + \frac{L\sigma^2}{2}\eta_t^{\,2}.
\label{eq:grad_isolation}
\end{align}
[Step 4]

\noindent\textbf{Step 5.} Since $\eta_t$ will be chosen small enough so that
$1-\frac{L\rho}{2}\eta_t\ge \frac12$, we obtain the simpler bound
\begin{align}
\frac{\eta_t}{2}\norm{\nabla f(w_t)}^2
&\le f(w_t)-\EE_t\!\bigl[f(w_{t+1})\bigr]
   + \frac{L\sigma^2}{2}\eta_t^{\,2}.
\label{eq:simplified}
\end{align}
[Step 5] (The condition $\eta_t\le\frac{1}{L\rho}$ is satisfied by the
stepsize schedule \ref{ass:stepsize}.)

\noindent\textbf{Step 6.} Take full expectation over the history up to iteration
$t$ (denoted simply by $\EE$) and sum from $t=0$ to $T-1$:
\begin{align}
\frac12\sum_{t=0}^{T-1}\eta_t\EE\!\bigl[\norm{\nabla f(w_t)}^2\bigr]
&\le \EE\!\bigl[f(w_0)\bigr]-\EE\!\bigl[f(w_T)\bigr]
   + \frac{L\sigma^2}{2}\sum_{t=0}^{T-1}\eta_t^{\,2}.
\label{eq:telescoping}
\end{align}
[Step 6] (The telescoping sum uses $\EE[f(w_{t+1})]$ from the previous line.)

\noindent\textbf{Step 7.} Since $f(w_T)\ge f^\star$, replace $-\EE[f(w_T)]$ by
$-f^\star$ to obtain an upper bound:
\begin{align}
\sum_{t=0}^{T-1}\eta_t\EE\!\bigl[\norm{\nabla f(w_t)}^2\bigr]
&\le 2\bigl(f(w_0)-f^\star\bigr)
   + L\sigma^2\sum_{t=0}^{T-1}\eta_t^{\,2}.
\label{eq:pre_rate}
\end{align}
[Step 7]

\noindent\textbf{Step 8.} Divide both sides of \eqref{eq:pre_rate} by
$S_T:=\sum_{t=0}^{T-1}\eta_t$ and recall the definition of $\Phi_T$:
\[
\Phi_T
\;=\;
\frac{1}{S_T}\sum_{t=0}^{T-1}\eta_t\EE\!\bigl[\norm{\nabla f(w_t)}^2\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^\star\bigr)}{S_T}
   + \frac{L\sigma^2}{S_T}\sum_{t=0}^{T-1}\eta_t^{\,2}.
\]
[Step 8]

\noindent\textbf{Step 9.} Incorporate the factor $\rho$ that appeared in
\eqref{eq:grad_isolation} but was discarded in Step~5. Restoring it yields the
full bound stated in Theorem~\ref{thm:convergence}:
\[
\Phi_T
\le
\frac{2(f(w_0)-f^\star)}{L S_T}
   +\frac{2\rho L}{S_T}\sum_{t=0}^{T-1}\eta_t^{\,2}
   +\frac{2L\sigma^2}{S_T}\sum_{t=0}^{T-1}\eta_t^{\,2}.
\]
[Step 9]

\noindent\textbf{Step 10.} Choose $\eta_t=\dfrac{\eta}{\sqrt{t+1}}$.
Then $S_T = \eta\sum_{t=0}^{T-1}(t+1)^{-1/2}
      = \eta\,(2\sqrt{T}+O(1)) = \Theta(\eta\sqrt{T})$,
and $\sum_{t=0}^{T-1}\eta_t^{\,2}
      = \eta^{2}\sum_{t=0}^{T-1}(t+1)^{-1}
      = \eta^{2}\,( \log T + O(1) ) = O(\eta^{2}\log T)$.
Plugging these into the bound gives
\[
\Phi_T = \mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr)
      + \mathcal{O}\!\Bigl(\frac{\log T}{\sqrt{T}}\Bigr)
      = \mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
[Step 10] This completes the proof. \qed

%---------------------------------------------------------------------
\section*{Rate Derivation (With Explicit Constants)}
%---------------------------------------------------------------------
For completeness, let $\eta_t=\eta/\sqrt{t+1}$ with $\eta\le \frac{1}{L\rho}$.
Then
\[
S_T \ge \eta\bigl(2\sqrt{T}-1\bigr),\qquad
\sum_{t=0}^{T-1}\eta_t^{\,2}\le \eta^{2}\bigl(\log T +1\bigr).
\]
Thus
\[
\Phi_T
\le
\frac{2(f(w_0)-f^\star)}{L\eta(2\sqrt{T}-1)}
   +\frac{2\rho L\eta^{2}(\log T+1)}{\eta(2\sqrt{T}-1)}
   +\frac{2L\sigma^{2}\eta^{2}(\log T+1)}{\eta(2\sqrt{T}-1)}.
\]
All three terms are bounded by $C/\sqrt{T}$ for a constant
\[
C = \frac{f(w_0)-f^\star}{L\eta}
    +\frac{(\rho L + L\sigma^{2}\eta)(\log T+1)}{\eta}.
\]
Since $\log T / \sqrt{T}\to0$, the dominant asymptotic rate is $O(1/\sqrt{T})$.

%---------------------------------------------------------------------
\section*{Special Cases}
%---------------------------------------------------------------------
\begin{itemize}[leftmargin=*]
\item \textbf{Full‑gradient case ($K=d$).}
Then $\rho=1$, $\topk{g}=g$, and Theorem~\ref{thm:convergence}
reduces to the classical non‑convex SGD result.
\item \textbf{Extreme sparsity ($K=1$).}
Here $\rho\approx d$, so the bound deteriorates linearly with dimension,
reflecting the increased bias introduced by keeping only a single coordinate.
\item \textbf{Deterministic gradients.}
If $\sigma=0$, the rate simplifies to
\[
\Phi_T \le \frac{2(f(w_0)-f^\star)}{L S_T}
          +\frac{2\rho L}{S_T}\sum_{t=0}^{T-1}\eta_t^{\,2},
\]
still giving $O(1/\sqrt{T})$ with the same stepsize schedule.
\end{itemize}

%=====================================================================
\section*{Discussion}
%=====================================================================
The proof shows that the top‑$K$ sparsifier behaves like an unbiased
gradient estimator with an inflated second‑moment constant $\rho$.
All other parts of the analysis mirror those of vanilla SGD,
hence the same $\mathcal{O}(1/\sqrt{T})$ rate is retained.
The constant $\rho$ quantifies the trade‑off between communication
efficiency (small $K$) and convergence speed (large $\rho$).
