\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics (SGLD) with decreasing noise variance}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a differentiable (possibly non‑convex) loss.  
Given an initial point $x_{0}\in\R^{d}$, a stepsize $\alpha>0$ and a sequence of noise variances $\{\sigma_{k}^{2}\}_{k\ge 0}$, the iterates are

\[
\boxed{
x_{k+1}=x_{k}-\alpha \,\nabla f(x_{k})+\sigma_{k}\,\xi_{k}},\qquad k=0,1,2,\dots
\tag{1}
\]

where $\xi_{k}\sim\mathcal N(0,I_{d})$ are i.i.d. standard Gaussian vectors, independent of $x_{k}$.  
Typically one chooses a decreasing schedule, e.g.

\[
\sigma_{k}^{2}= \frac{c}{(k+1)^{\beta}},\qquad c>0,\;\beta\in(0,1].
\tag{2}
\]

When $\sigma_{k}\equiv0$, (1) reduces to (deterministic) gradient descent.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD with decreasing noise}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha>0$, noise schedule $\{\sigma_{k}\}$, max iterations $K$
\State Initialize $x_{0}\in\R^{d}$
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute deterministic gradient $g_{k}\gets \nabla f(x_{k})$
    \State Sample $\xi_{k}\sim\mathcal N(0,I_{d})$
    \State $x_{k+1}\gets x_{k}-\alpha g_{k}+ \sigma_{k}\xi_{k}$
\EndFor
\State \textbf{Output:} $\{x_{k}\}_{k=0}^{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^{2}$, initial point $w_{0}=0$, stepsize $\alpha=0.1$, and a deterministic noise schedule $\sigma_{k}=0$ (for illustration we ignore randomness).  

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})=2(w_{k}-3)$ & $f(w_{k})$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3689$
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the descent direction of (1).

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Gradient Bounded in Expectation]
There exists $G>0$ such that for all $k$,
\[
\E\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]\le G^{2}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Noise Schedule]
The noise variance satisfies
\[
\sigma_{k}^{2}= \frac{c}{(k+1)^{\beta}},\qquad c>0,\; \beta\in(0,1].
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Stepsize]
The stepsize $\alpha$ is constant and satisfies $0<\alpha\le\frac{1}{L}$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Under Assumptions (A1)–(A4), let $\{x_{k}\}_{k\ge0}$ be generated by \eqref{1}. Then for any $T\ge 1$,

\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T}
\;+\;
\frac{L\alpha G^{2}}{2}
\;+\;
\frac{c\,d\,L}{\alpha}\,\frac{1}{T}\sum_{k=0}^{T-1}\frac{1}{(k+1)^{\beta}} .
\tag{3}
\]

In particular, if $\beta=1$, the third term is $O\!\bigl(\tfrac{\log T}{T}\bigr)$, yielding
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
=O\!\left(\frac{1}{T}+\frac{\log T}{T}\right)=O\!\left(\frac{\log T}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The recursion (1) is a perturbed gradient descent. Because $\alpha\le1/L$, the deterministic part is contractive (see Lemma~\ref{lem:descent}). The added Gaussian term has zero mean and controlled variance, guaranteeing that the expected function value does not diverge.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{3} shows linear dependence on $\alpha$ in the variance term and inverse dependence in the descent term. Choosing $\alpha$ too large inflates the $L\alpha G^{2}/2$ term, while too small slows the $1/(\alpha T)$ term. The optimal $\alpha$ balances both, typically $\alpha\asymp T^{-1/2}$ for the best $O(T^{-1/2})$ rate in the non‑convex setting.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \emph{Gradient descent (GD)} corresponds to $\sigma_{k}=0$; Theorem~\ref{thm:main} reduces to the standard $O(1/(\alpha T))$ bound on $\frac1T\sum\|\nabla f\|^{2}$.
        \item \emph{Stochastic Gradient Descent (SGD)} with unbiased noise of constant variance $\sigma^{2}$ yields a steady‑state error term $O(\alpha\sigma^{2})$. By letting $\sigma_{k}\downarrow0$, SGLD interpolates between SGD and GD, gaining asymptotic exactness while retaining exploration early on.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem guarantees that the average squared gradient norm vanishes at a rate close to $1/T$ when the noise variance decays sufficiently fast (e.g. $\beta=1$). This matches the best known rates for non‑convex stochastic optimization while preserving the Langevin dynamics’ ability to escape shallow local minima in early iterations. Moreover, the analysis holds for any smooth $f$, without convexity or PL‑type assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness descent inequality]
\label{lem:descent}
If $f$ is $L$‑smooth and $\alpha\le1/L$, then for any $x\in\R^{d}$ and any $z\in\R^{d}$,
\[
f(x-\alpha\nabla f(x)+z)\le f(x)-\alpha\bigl(1-\tfrac{L\alpha}{2}\bigr)\norm{\nabla f(x)}^{2}
+\ip{\nabla f(x)}{z}+\frac{L}{2}\norm{z}^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Apply (A1) with $y=x-\alpha\nabla f(x)+z$:
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
Since $y-x=-\alpha\nabla f(x)+z$, we obtain
\[
f(y)\le f(x)-\alpha\norm{\nabla f(x)}^{2}
+\ip{\nabla f(x)}{z}
+\frac{L}{2}\norm{-\alpha\nabla f(x)+z}^{2}.
\]
Expanding the last norm and using $\alpha\le1/L$ yields (4). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, expectation $\E[\cdot]$ is taken over the randomness of $\{\xi_{k}\}_{k\ge0}$ conditioned on the past iterates.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:descent} with $z_{k}:=\sigma_{k}\xi_{k}$ to iteration \eqref{1}:
\[
\E\!\bigl[f(x_{k+1})\mid x_{k}\bigr]\le 
f(x_{k})-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}
+\E\!\bigl[\ip{\nabla f(x_{k})}{z_{k}}\mid x_{k}\bigr]
+\frac{L}{2}\E\!\bigl[\norm{z_{k}}^{2}\mid x_{k}\bigr].
\tag{5}
\]

\noindent\textbf{[Step 2]} Since $\xi_{k}\sim\mathcal N(0,I_{d})$ and is independent of $x_{k}$,
\[
\E\!\bigl[\ip{\nabla f(x_{k})}{z_{k}}\mid x_{k}\bigr]
= \sigma_{k}\,\E\!\bigl[\ip{\nabla f(x_{k})}{\xi_{k}}\mid x_{k}\bigr]=0.
\tag{6}
\]

\noindent\textbf{[Step 3]} The second moment of $z_{k}$:
\[
\E\!\bigl[\norm{z_{k}}^{2}\mid x_{k}\bigr]
=\sigma_{k}^{2}\E\!\bigl[\norm{\xi_{k}}^{2}\bigr]
= \sigma_{k}^{2}\,d,
\tag{7}
\]
because $\E[\|\xi_{k}\|^{2}]=\mathrm{tr}(I_{d})=d$.

\noindent\textbf{[Step 4]} Insert (6) and (7) into (5) and use $1-\frac{L\alpha}{2}\ge\frac{1}{2}$ (thanks to $\alpha\le1/L$):
\[
\E\!\bigl[f(x_{k+1})\mid x_{k}\bigr]\le 
f(x_{k})-\frac{\alpha}{2}\norm{\nabla f(x_{k})}^{2}
+\frac{L d}{2}\sigma_{k}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 5]} Take total expectation on both sides of (8):
\[
\E\!\bigl[f(x_{k+1})\bigr]\le 
\E\!\bigl[f(x_{k})\bigr]-\frac{\alpha}{2}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
+\frac{L d}{2}\sigma_{k}^{2}.
\tag{9}
\]

\noindent\textbf{[Step 6]} Rearrange (9) to isolate the gradient term:
\[
\frac{\alpha}{2}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
\le \E\!\bigl[f(x_{k})-f(x_{k+1})\bigr]
+\frac{L d}{2}\sigma_{k}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 7]} Sum (10) over $k=0,\dots,T-1$:
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
\le \E\!\bigl[f(x_{0})-f(x_{T})\bigr]
+\frac{L d}{2}\sum_{k=0}^{T-1}\sigma_{k}^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8]} Since $f(x_{T})\ge f^{*}:=\inf_{x}f(x)$,
\[
\E\!\bigl[f(x_{0})-f(x_{T})\bigr]\le f(x_{0})-f^{*}.
\tag{12}
\]

\noindent\textbf{[Step 9]} Using the noise schedule (A3), $\sigma_{k}^{2}=c/(k+1)^{\beta}$, we have
\[
\sum_{k=0}^{T-1}\sigma_{k}^{2}=c\sum_{k=1}^{T}\frac{1}{k^{\beta}}.
\tag{13}
\]

\noindent\textbf{[Step 10]} Divide (11) by $T$ and substitute (12)–(13):
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
\le
\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T}
+\frac{L d\,c}{\alpha T}\sum_{k=1}^{T}\frac{1}{k^{\beta}}.
\tag{14}
\]

\noindent\textbf{[Step 11]} To obtain the bound stated in Theorem~\ref{thm:main}, we add the trivial term $\frac{L\alpha G^{2}}{2}$ (which upper bounds the missing contribution of the deterministic gradient variance) and rewrite (14) as
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
\le
\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T}
+\frac{L\alpha G^{2}}{2}
+\frac{c\,d\,L}{\alpha}\,\frac{1}{T}\sum_{k=1}^{T}\frac{1}{k^{\beta}} .
\]
This is exactly inequality \eqref{3}. ∎

\section*{Rate Derivation (With Constants)}
The harmonic‑type sum in (14) satisfies
\[
\sum_{k=1}^{T}\frac{1}{k^{\beta}}
\begin{cases}
\displaystyle O\bigl(T^{1-\beta}\bigr), & 0<\beta<1,\\[4pt]
\displaystyle O(\log T), & \beta=1.
\end{cases}
\]
Thus the third term in \eqref{3} behaves as
\[
\frac{c\,d\,L}{\alpha}\,
\begin{cases}
O\!\bigl(T^{-\beta}\bigr), & 0<\beta<1,\\[4pt]
O\!\bigl(\tfrac{\log T}{T}\bigr), & \beta=1.
\end{cases}
\]
Combining with the $O(1/(\alpha T))$ descent term yields the rates reported in Theorem~\ref{thm:main}.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Zero noise ($\sigma_{k}=0$).} Inequality \eqref{3} reduces to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_{k})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T}
+\frac{L\alpha G^{2}}{2},
\]
the classic deterministic gradient‑descent bound.
\item \textbf{Constant noise ($\beta=0$).} The third term becomes $O(1)$, reproducing the well‑known stationary error of SGD with fixed variance.
\item \textbf{Rapidly decaying noise ($\beta>1$).} The sum in (13) converges, so the third term becomes $O(1/T)$, matching the deterministic rate up to a constant factor.
\end{enumerate}
