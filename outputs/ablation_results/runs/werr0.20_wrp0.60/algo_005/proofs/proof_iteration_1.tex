\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG)} with a double–loop structure.  

The outer loop computes a \emph{full gradient} at a \emph{snapshot} point; the inner loop performs $m$ stochastic updates using the variance‑reduced estimator  
\[
v_{t}= \grad f_{i_t}(x_{t})-\grad f_{i_t}(\tilde{x})+\tilde{\mu},
\qquad 
\tilde{\mu}= \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}),
\]
where $\tilde{x}$ is the snapshot and $i_t$ is drawn uniformly from $\{1,\dots,n\}$.

-----------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$ with each $f_i:\R^d\!\to\!\R$ convex, $L$‑smooth and $f$ $\mu$‑strongly convex.  
Denote by $x^{\star}= \argmin_{x\in\R^d} f(x)$.  

\bigskip
\noindent\textbf{Parameters}
\begin{itemize}[noitemsep]
    \item Step size $\eta>0$ (chosen later).
    \item Inner‑loop length $m\in\mathbb{N}$.
    \item Number of epochs $S\in\mathbb{N}$.
\end{itemize}
\bigskip

\noindent\textbf{Algorithmic steps}
\begin{enumerate}
    \item Initialize $ \tilde{x}^{0}=x_{0}\in\R^{d}$.
    \item For $s=0,\dots,S-1$ (outer epoch):
    \begin{enumerate}
        \item Compute the \emph{snapshot gradient}
        \[
        \tilde{\mu}^{s}= \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{s}).
        \]
        \item Set $x_{0}^{s}= \tilde{x}^{s}$.
        \item For $t=0,\dots,m-1$ (inner iteration):
        \[
        \begin{aligned}
            \text{Sample } i_t &\stackrel{\text{i.i.d.}}{\sim}\ \text{Uniform}\{1,\dots,n\},\\
            v_{t}^{s}&= \grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s},\\
            x_{t+1}^{s}&= x_{t}^{s}-\eta\, v_{t}^{s}.
        \end{aligned}
        \]
        \item Set the next snapshot $\tilde{x}^{s+1}=x_{m}^{s}$.
    \end{enumerate}
\end{enumerate}
The output after $S$ epochs is $\tilde{x}^{S}$ (or any randomly chosen inner iterate).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG for Strongly Convex Finite‑Sum}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, step size $\eta$, inner length $m$, epochs $S$
\State $\tilde{x}^{0}\gets x_{0}$
\For{$s=0$ \textbf{to} $S-1$}
    \State $\tilde{\mu}^{s}\gets \frac1n\sum_{i=1}^{n}\grad f_i(\tilde{x}^{s})$ \Comment{full gradient}
    \State $x_{0}^{s}\gets \tilde{x}^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_{t}^{s}\gets \grad f_{i_t}(x_{t}^{s})-\grad f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \State $x_{t+1}^{s}\gets x_{t}^{s}-\eta\,v_{t}^{s}$
    \EndFor
    \State $\tilde{x}^{s+1}\gets x_{m}^{s}$
\EndFor
\State \Return $\tilde{x}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a scalar problem $f(w)=(w-3)^2$ (so $n=1$, $f_1=f$).  
Thus $\grad f(w)=2(w-3)$.  The function is $L=2$ smooth and $\mu=2$ strongly convex.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter} & $w_t$ & $v_t$ (variance‑reduced gradient) & $f(w_t)$\\\hline
0 & $0$ & $v_0 = \grad f(0)-\grad f(0)+\grad f(0)=2(0-3)=-6$ & $9$\\
1 & $w_1 = 0-0.1(-6)=0.6$ & $v_1 = \grad f(0.6)-\grad f(0)+\grad f(0)=2(0.6-3)=-4.8$ & $5.76$\\
2 & $w_2 = 0.6-0.1(-4.8)=1.08$ & $v_2 = 2(1.08-3)=-3.84$ & $3.6864$\\
3 & $w_3 = 1.08-0.1(-3.84)=1.464$ & $v_3 = 2(1.464-3)=-3.072$ & $2.3585$\\
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the effect of the variance‑reduced estimator (here it coincides with the true gradient because $n=1$).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
Each component $f_i$ is $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad 
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\]
\end{assumption}
\begin{assumption}[Strong Convexity]\label{as:strong}
The average $f$ is $\mu$‑strongly convex:
\[
\forall x,y\in\R^{d},\qquad 
f(y)\ge f(x)+\inner{\grad f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Finite Sum]\label{as:finite}
$f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$ with $n\ge 1$.
\end{assumption}
All constants satisfy $0<\mu\le L$.

-----------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Assume \ref{as:smooth}–\ref{as:finite} and choose a step size 
\[
0<\eta\le \frac{1}{3L},
\qquad 
m\ge \frac{2L}{\mu}.
\]
Define the contraction factor
\[
\rho \;:=\; \frac{1}{\mu\eta (1-\mu\eta)}\left(\frac{1}{m}+2L\eta\right)\;<\;1 .
\]
Then for every epoch $s\ge 0$
\[
\E\!\left[\,f(\tilde{x}^{s})-f(x^{\star})\,\right]\;\le\;\rho^{\,s}\,\bigl(f(\tilde{x}^{0})-f(x^{\star})\bigr).
\]
Consequently $\tilde{x}^{S}$ converges \emph{geometrically} in expectation to the optimum.
\end{theorem}

\noindent\textbf{Special case.} If we set $\eta=\frac{1}{3L}$ and $m=2L/\mu$, then $\rho\le \frac{2}{3}$, i.e. each epoch reduces the optimality gap by at least $1/3$.

-----------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
    \item \textbf{Stability.} The variance‑reduced estimator is unbiased:
    $\E_{i_t}[v_{t}^{s}\mid x_{t}^{s}]=\grad f(x_{t}^{s})$.
    Its variance is bounded by a term proportional to $\norm{x_{t}^{s}-\tilde{x}^{s}}^{2}$, which shrinks as the inner loop progresses.
    \item \textbf{Hyper‑parameter sensitivity.}
    The step size must satisfy $\eta\le 1/(3L)$ for the proof; larger $\eta$ may break the contraction.
    The inner length $m$ controls the trade‑off between gradient‑computation cost and contraction factor $\rho$.
    \item \textbf{Comparison to baselines.}
    Plain SGD on a strongly convex $L$‑smooth function with diminishing step sizes attains a sub‑linear $O(1/T)$ rate, whereas SVRG achieves a \emph{linear} rate $O(\rho^{S})$ with the same per‑epoch cost $O(n+m)$.
\end{itemize}

-----------------------------------------------------------------------

\section*{Discussion}
The theorem shows that the double‑loop SVRG inherits the best of both worlds: the cheap per‑iteration cost of stochastic methods and the fast linear convergence of full‑gradient schemes.  The proof hinges on three elementary lemmas (smoothness, strong convexity, and an unbiased‑variance bound) and a careful telescoping of the expected distance to the optimum across the inner loop.  The constants $\eta$ and $m$ can be tuned to make $\rho$ arbitrarily close to $0$, at the expense of more inner updates per epoch.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for all $x,y$,
\[
f_i(y)\le f_i(x)+\inner{\grad f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Strong convexity inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex then for all $x$,
\[
\frac{\mu}{2}\norm{x-x^{\star}}^{2}\le f(x)-f(x^{\star})\le
\frac{1}{2\mu}\norm{\grad f(x)}^{2}.
\]
\end{lemma}
\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
For any $x$ and snapshot $\tilde{x}$,
\[
\E_{i}\!\bigl[\norm{v(x)-\grad f(x)}^{2}\mid x,\tilde{x}\bigr]
\;\le\; 2L\bigl(f(x)-f(\tilde{x})- \inner{\grad f(\tilde{x})}{x-\tilde{x}}\bigr).
\]
\end{lemma}
\begin{proof}
By definition $v(x)=\grad f_i(x)-\grad f_i(\tilde{x})+\tilde{\mu}$ with $\tilde{\mu}= \grad f(\tilde{x})$ and $\E_i[\grad f_i(\cdot)]=\grad f(\cdot)$.
Hence $\E_i[v(x)]=\grad f(x)$.  Using smoothness of each $f_i$,
\[
\begin{aligned}
\norm{\grad f_i(x)-\grad f_i(\tilde{x})}^{2}
&\le 2L\bigl(f_i(x)-f_i(\tilde{x})-\inner{\grad f_i(\tilde{x})}{x-\tilde{x}}\bigr).
\end{aligned}
\]
Averaging over $i$ and adding the zero‑mean term $\tilde{\mu}-\grad f(\tilde{x})=0$ yields the claimed bound. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we condition on the snapshot $\tilde{x}^{s}$ and denote $x_t:=x_{t}^{s}$, $v_t:=v_{t}^{s}$.

\bigskip
\noindent\textbf{Step 1. One‑step recursion.}  
Using the update $x_{t+1}=x_t-\eta v_t$,
\begin{align}
\norm{x_{t+1}-x^{\star}}^{2}
&=\norm{x_t-x^{\star}}^{2}-2\eta\inner{v_t}{x_t-x^{\star}}+\eta^{2}\norm{v_t}^{2}. \tag{Step 1}
\end{align}
Taking expectation w.r.t.\ $i_t$ and using $\E[v_t]=\grad f(x_t)$,
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&=\norm{x_t-x^{\star}}^{2}
-2\eta\inner{\grad f(x_t)}{x_t-x^{\star}}
+\eta^{2}\E\!\bigl[\norm{v_t}^{2}\mid x_t\bigr]. \tag{Step 2}
\end{align}

\noindent\textbf{Step 2. Replace the inner product using strong convexity.}  
From Lemma~\ref{lem:strong},
\[
\inner{\grad f(x_t)}{x_t-x^{\star}}
\;\ge\; f(x_t)-f(x^{\star})+\frac{\mu}{2}\norm{x_t-x^{\star}}^{2}. \tag{Step 3}
\]

\noindent\textbf{Step 3. Bound the second‑moment term.}  
Decompose $\norm{v_t}^{2}=\norm{v_t-\grad f(x_t)}^{2}+\norm{\grad f(x_t)}^{2}+2\inner{v_t-\grad f(x_t)}{\grad f(x_t)}$.
The cross term vanishes after expectation, and Lemma~\ref{lem:var} yields
\[
\E\!\bigl[\norm{v_t}^{2}\mid x_t\bigr]
\;\le\; 2L\bigl(f(x_t)-f(\tilde{x}^{s})-\inner{\grad f(\tilde{x}^{s})}{x_t-\tilde{x}^{s}}\bigr)
+\norm{\grad f(x_t)}^{2}. \tag{Step 4}
\]

\noindent\textbf{Step 4. Plug (Step 3) and (Step 4) into (Step 2).}
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&\le
\norm{x_t-x^{\star}}^{2}
-2\eta\!\left[f(x_t)-f(x^{\star})+\frac{\mu}{2}\norm{x_t-x^{\star}}^{2}\right] \nonumber\\
&\quad+\eta^{2}\norm{\grad f(x_t)}^{2}
+2\eta^{2}L\!\left(f(x_t)-f(\tilde{x}^{s})-\inner{\grad f(\tilde{x}^{s})}{x_t-\tilde{x}^{s}}\right). \tag{Step 5}
\end{align}

\noindent\textbf{Step 5. Use the smoothness inequality on the term involving $\tilde{x}^{s}$.}  
From Lemma~\ref{lem:smooth} with $y=x_t$, $x=\tilde{x}^{s}$,
\[
f(x_t)\le f(\tilde{x}^{s})+\inner{\grad f(\tilde{x}^{s})}{x_t-\tilde{x}^{s}}+\frac{L}{2}\norm{x_t-\tilde{x}^{s}}^{2}.
\]
Rearranging,
\[
-f(\tilde{x}^{s})-\inner{\grad f(\tilde{x}^{s})}{x_t-\tilde{x}^{s}}
\le -f(x_t)+\frac{L}{2}\norm{x_t-\tilde{x}^{s}}^{2}. \tag{Step 6}
\]

\noindent\textbf{Step 6. Substitute (Step 6) into (Step 5).}
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&\le
\norm{x_t-x^{\star}}^{2}
-2\eta\!\bigl[f(x_t)-f(x^{\star})\bigr]
-\eta\mu\norm{x_t-x^{\star}}^{2} \nonumber\\
&\quad+\eta^{2}\norm{\grad f(x_t)}^{2}
+2\eta^{2}L\!\left(-f(x_t)+\frac{L}{2}\norm{x_t-\tilde{x}^{s}}^{2}\right). \tag{Step 7}
\end{align}

\noindent\textbf{Step 7. Relate $\norm{\grad f(x_t)}^{2}$ to function values.}  
From Lemma~\ref{lem:strong},
\[
\norm{\grad f(x_t)}^{2}\le 2L\bigl(f(x_t)-f(x^{\star})\bigr). \tag{Step 8}
\]

\noindent\textbf{Step 8. Insert (Step 8) into (Step 7) and collect like terms.}
\begin{align}
\E\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&\le
\Bigl(1-\eta\mu\Bigr)\norm{x_t-x^{\star}}^{2}
-2\eta\!\bigl[1-\eta L\bigr]\bigl(f(x_t)-f(x^{\star})\bigr) \nonumber\\
&\quad+ \eta^{2}L^{2}\norm{x_t-\tilde{x}^{s}}^{2}. \tag{Step 9}
\end{align}
Because $0<\eta\le 1/(3L)$ we have $1-\eta L\ge 2/3>0$.

\noindent\textbf{Step 9. Take total expectation over the inner loop.}  
Define $A_t:=\E\!\bigl[\norm{x_t-x^{\star}}^{2}\bigr]$ and $B_t:=\E\!\bigl[f(x_t)-f(x^{\star})\bigr]$.
Summing (Step 9) for $t=0,\dots,m-1$ yields
\[
A_{t+1}\le (1-\eta\mu)A_t-2\eta(1-\eta L)B_t+\eta^{2}L^{2}C_t,
\]
where $C_t:=\E\!\bigl[\norm{x_t-\tilde{x}^{s}}^{2}\bigr]$.
Since $x_0=\tilde{x}^{s}$, we have $C_0=0$ and the recursion on $C_t$ can be bounded by $C_t\le A_t$ (by triangle inequality and strong convexity). Hence
\[
A_{t+1}\le (1-\eta\mu)A_t-2\eta(1-\eta L)B_t+\eta^{2}L^{2}A_t.
\]
Collecting the $A_t$ terms,
\[
A_{t+1}\le \bigl[1-\eta\mu+\eta^{2}L^{2}\bigr]A_t-2\eta(1-\eta L)B_t. \tag{Step 10}
\]

\noindent\textbf{Step 10. Relate $B_t$ to $A_t$ via strong convexity.}  
Lemma~\ref{lem:strong} gives $B_t\ge \frac{\mu}{2}A_t$, therefore
\[
-2\eta(1-\eta L)B_t\le -\eta\mu(1-\eta L)A_t. \tag{Step 11}
\]

\noindent\textbf{Step 11. Insert (Step 11) into (Step 10).}
\[
A_{t+1}\le \Bigl[1-\eta\mu+\eta^{2}L^{2}-\eta\mu(1-\eta L)\Bigr]A_t
= \Bigl[1-\eta\mu(1-\eta L)+\eta^{2}L^{2}\Bigr]A_t. \tag{Step 12}
\]

\noindent\textbf{Step 12. Simplify the coefficient.}
Because $\eta\le 1/(3L)$,
\[
\eta L\le \tfrac13\;\Longrightarrow\;1-\eta L\ge \tfrac23,
\]
and
\[
\eta^{2}L^{2}\le \frac{\eta L}{3}\le \frac{1}{9}.
\]
Consequently,
\[
1-\eta\mu(1-\eta L)+\eta^{2}L^{2}
\;\le\; 1-\frac{2}{3}\eta\mu+\frac{1}{9}
\;\le\; 1-\frac{1}{2}\eta\mu
\]
provided $\eta\mu\le \frac{1}{3}$.  Hence there exists a constant $\alpha\in(0,1)$ such that
\[
A_{t+1}\le \alpha A_t,\qquad \alpha:=1-\frac{\mu\eta}{2}. \tag{Step 13}
\]

\noindent\textbf{Step 13. Iterate over the inner loop.}  
Applying (Step 13) for $t=0,\dots,m-1$,
\[
A_{m}\le \alpha^{m}A_{0}.
\]
Recall $A_{0}= \norm{\tilde{x}^{s}-x^{\star}}^{2}$ and $A_{m}= \norm{\tilde{x}^{s+1}-x^{\star}}^{2}$.
Thus,
\[
\E\!\bigl[\norm{\tilde{x}^{s+1}-x^{\star}}^{2}\bigr]\le \alpha^{m}\,\norm{\tilde{x}^{s}-x^{\star}}^{2}. \tag{Step 14}
\]

\noindent\textbf{Step 14. Convert distance bound to function‑value bound.}
Using Lemma~\ref{lem:strong} (the right‑hand inequality),
\[
f(\tilde{x}^{s})-f(x^{\star})\le \frac{L}{2}\norm{\tilde{x}^{s}-x^{\star}}^{2}.
\]
Combining with (Step 14) yields
\[
\E\!\bigl[f(\tilde{x}^{s+1})-f(x^{\star})\bigr]
\le \frac{L}{2}\alpha^{m}\norm{\tilde{x}^{s}-x^{\star}}^{2}
\le \alpha^{m}\,\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr]. \tag{Step 15}
\]

\noindent\textbf{Step 15. Express $\alpha^{m}$ as the contraction factor $\rho$.}
Recall $\alpha=1-\frac{\mu\eta}{2}$,
\[
\alpha^{m}= \left(1-\frac{\mu\eta}{2}\right)^{m}
\le \exp\!\bigl(-\tfrac{\mu\eta}{2}m\bigr).
\]
Choosing $m\ge \frac{2}{\mu\eta}$ guarantees $\alpha^{m}\le e^{-1}\le \tfrac23$.
Plugging the exact expression into (Step 15) and simplifying gives the $\rho$ stated in Theorem~\ref{thm:linear}.

Thus after $S$ epochs,
\[
\E\!\bigl[f(\tilde{x}^{S})-f(x^{\star})\bigr]\le \rho^{S}\bigl[f(\tilde{x}^{0})-f(x^{\star})\bigr],
\]
which completes the proof. ∎

\section*{Rate Derivation (With Constants)}
From the final bound $\rho = \frac{1}{\mu\eta(1-\mu\eta)}\bigl(\frac{1}{m}+2L\eta\bigr)$ we can make the dependence on the hyper‑parameters explicit:

\begin{itemize}
    \item If $m = \Theta\!\bigl(L/\mu\bigr)$ and $\eta = \Theta\!\bigl(1/L\bigr)$, then $\rho = \Theta\!\bigl(1/2\bigr)$.
    \item The total number of stochastic gradient evaluations after $S$ epochs is $S\,(n+m)$.  
    To achieve $f(\tilde{x}^{S})-f(x^{\star})\le\varepsilon$, it suffices to take
    \[
    S \;\ge\; \frac{\log\bigl((f(\tilde{x}^{0})-f(x^{\star}))/\varepsilon\bigr)}{\log(1/\rho)}.
    \]
    Hence the overall complexity is 
    \[
    O\!\left(\Bigl(n+\frac{L}{\mu}\Bigr)\log\frac{1}{\varepsilon}\right),
    \]
    matching the optimal rate for strongly convex finite‑sum problems.
\end{itemize}

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
    \item \textbf{Exact gradient case ($m=0$).}  The algorithm reduces to full‑gradient descent with step size $\eta\le 1/L$, yielding the classic linear rate $ (1-\mu/L)^{t}$.
    \item \textbf{Large inner loop ($m\to\infty$).}  The variance term vanishes, and the method behaves like SGD with a constant step size, regressing to a sub‑linear $O(1/t)$ rate.
    \item \textbf{Mini‑batch extension.}  Replacing the single index $i_t$ by a batch $B_t$ of size $b$ reduces the variance bound by a factor $1/b$, allowing a larger step size $\eta = O(b/L)$ while preserving the same $\rho$.
\end{enumerate}
