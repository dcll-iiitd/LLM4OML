\section*{Optimistic Gradient Descent–Ascent (OGDA)}

\section*{Mathematical Formulation}
Consider the saddle–point problem  

\[
\min_{x\in\mathbb{R}^{d_x}}\;\max_{y\in\mathbb{R}^{d_y}} \;L(x,y),
\]

where \(L:\mathbb{R}^{d_x}\times\mathbb{R}^{d_y}\to\mathbb{R}\) is continuously differentiable, convex in \(x\) and concave in \(y\).  
Denote the gradient operator  

\[
F(z):=
\begin{pmatrix}
\nabla_x L(x,y)\\[2mm]
-\nabla_y L(x,y)
\end{pmatrix},
\qquad 
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathbb{R}^{d},\; d:=d_x+d_y .
\]

OGDA uses the gradient evaluated at the current iterate and a \emph{prediction} built from the previous gradient:

\[
\boxed{
\begin{aligned}
z_{t+1}
&=z_t-\eta\Big( F(z_t)+\big(F(z_t)-F(z_{t-1})\big)\Big)\\
&=z_t-\eta\big(2F(z_t)-F(z_{t-1})\big),
\qquad t\ge 1,
\end{aligned}}
\]
with a stepsize \(\eta>0\).  In component form,

\[
\begin{cases}
x_{t+1}=x_t-\eta\big(2\nabla_x L(x_t,y_t)-\nabla_x L(x_{t-1},y_{t-1})\big),\\[2mm]
y_{t+1}=y_t+\eta\big(2\nabla_y L(x_t,y_t)-\nabla_y L(x_{t-1},y_{t-1})\big).
\end{cases}
\]

The algorithm requires an initialization \(z_0=(x_0,y_0)\) and a “ghost’’ point
\(z_{-1}:=z_0\) (i.e. the same as the first iterate) so that the rule is well‑defined for \(t=0\).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent–Ascent (OGDA)}
\begin{algorithmic}[1]
\REQUIRE Initial point \(x_0\in\mathbb{R}^{d_x},\,y_0\in\mathbb{R}^{d_y}\), stepsize \(\eta>0\), max iterations \(T\).
\STATE Set \(x_{-1}\gets x_0,\;y_{-1}\gets y_0\).
\FOR{$t=0$ \TO $T-1$}
    \STATE Compute current gradients 
    \[
    g^x_t\gets\nabla_x L(x_t,y_t),\qquad 
    g^y_t\gets\nabla_y L(x_t,y_t).
    \]
    \STATE Compute previous gradients 
    \[
    g^x_{t-1}\gets\nabla_x L(x_{t-1},y_{t-1}),\qquad 
    g^y_{t-1}\gets\nabla_y L(x_{t-1},y_{t-1}).
    \]
    \STATE Update
    \[
    x_{t+1}\gets x_t-\eta\big(2g^x_t-g^x_{t-1}\big),\qquad
    y_{t+1}\gets y_t+\eta\big(2g^y_t-g^y_{t-1}\big).
    \]
\ENDFOR
\RETURN \((x_T,y_T)\).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the update on a simple convex function (no maximization part)  

\[
f(w)=(w-3)^2,\qquad w\in\mathbb{R}.
\]

The gradient is \(\nabla f(w)=2(w-3)\).  We run OGDA with stepsize \(\eta=0.1\) and start from \(w_{0}=0\).  
Since there is no \(y\)-variable we set \(w_{-1}=w_{0}=0\).

\begin{tabular}{c|c|c|c}
\textbf{Iteration} & \(\nabla f(w_t)\) & \(\nabla f(w_{t-1})\) & \(w_{t+1}=w_t-\eta\big(2\nabla f(w_t)-\nabla f(w_{t-1})\big)\)\\ \hline
\(t=0\) & \(2(0-3)=-6\) & \(-6\) & \(w_{1}=0-0.1\big(2(-6)-(-6)\big)=0-0.1(-6)=0.6\)\\[2mm]
\(t=1\) & \(2(0.6-3)=-4.8\) & \(-6\) & \(w_{2}=0.6-0.1\big(2(-4.8)-(-6)\big)=0.6-0.1(-3.6)=0.96\)\\[2mm]
\(t=2\) & \(2(0.96-3)=-4.08\) & \(-4.8\) & \(w_{3}=0.96-0.1\big(2(-4.08)-(-4.8)\big)=0.96-0.1(-3.36)=1.296\)
\end{tabular}

Objective values after each iteration:

\[
f(w_0)=9,\; f(w_1)=(0.6-3)^2=5.76,\;
f(w_2)=(0.96-3)^2\approx4.17,\;
f(w_3)=(1.296-3)^2\approx2.91.
\]

The iterates move toward the minimizer \(w^\star=3\), illustrating the “optimistic’’ correction term.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Convex–concave structure.} For all \(x,x'\) and \(y,y'\),
\[
L(x,y')\le L(x,y)+\langle\nabla_y L(x,y),y'-y\rangle,
\qquad
L(x',y)\ge L(x,y)+\langle\nabla_x L(x,y),x'-x\rangle .
\]
\item \textbf{Smoothness.} \(L\) has \(L\)-Lipschitz continuous gradients:
\[
\norm{F(z)-F(z')}\le L\norm{z-z'},\qquad\forall z,z'\in\mathbb{R}^{d}.
\]
\item \textbf{Existence of a saddle point.} There exists \((x^\star,y^\star)\) such that
\[
L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\qquad\forall x,y.
\]
\item \textbf{Bounded domain (optional).} The iterates remain in a compact set
\(\mathcal{Z}\) with diameter \(D:=\max_{z,z'\in\mathcal{Z}}\norm{z-z'}<\infty\).
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA convergence for convex–concave smooth games]\label{thm:ogda}
Assume (A1)–(A3) and choose a stepsize  

\[
0<\eta\le\frac{1}{2L}.
\]

Let \(\{z_t\}_{t\ge0}\) be generated by OGDA with any initialization \(z_0\).  
Define the averaged iterate  

\[
\bar z_T:=\frac{1}{T}\sum_{t=0}^{T-1}z_t .
\]

Then the primal–dual gap satisfies  

\[
\boxed{
\max_{y\in\mathcal{Y}}L(\bar x_T,y)-\min_{x\in\mathcal{X}}L(x,\bar y_T)
\;\le\;
\frac{D^{2}}{2\eta T}\, .
}
\tag{1}
\]

Consequently, OGDA attains an \(\mathcal{O}(1/T)\) convergence rate in terms of the
saddle‑point residual.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition \(\eta\le 1/(2L)\) guarantees that the
update matrix is a non‑expansive mapping for the monotone operator \(F\);
hence the iterates remain bounded.
\item \textbf{Hyperparameter sensitivity.} The bound (1) is monotone decreasing
in \(\eta\) up to the admissible limit \(1/(2L)\).  Choosing \(\eta\) close to
\(1/(2L)\) yields the smallest constant \(\frac{D^{2}}{2\eta}\).
\item \textbf{Comparison to baseline methods.}
\begin{enumerate}
\item \emph{Gradient Descent–Ascent (GDA)} obtains only an \(\mathcal{O}(1/\sqrt{T})\) rate
for the same class of problems.
\item \emph{Extragradient} achieves the same \(\mathcal{O}(1/T)\) rate but requires two
gradient evaluations per iteration.  OGDA uses a single gradient evaluation
and a cheap “prediction’’ term, thus matching the extragradient rate with
lower per‑iteration cost.
\end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that the optimistic correction eliminates the rotational
behaviour typical of vanilla GDA in bilinear or more general smooth
convex–concave games.  The proof hinges on rewriting OGDA as a
\emph{linear} combination of the current and previous gradients and exploiting
the monotonicity of \(F\).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Monotone operator]
The operator \(F\) is monotone if  

\[
\iprod{F(z)-F(z')}{z-z'}\ge 0,\qquad\forall z,z'\in\mathbb{R}^{d}.
\]
\end{definition}
For a convex–concave \(L\) the associated \(F\) defined above is monotone.

\begin{lemma}[Three‑point identity for smooth functions]\label{lem:three}
If \(F\) is \(L\)-Lipschitz, then for any \(a,b,c\in\mathbb{R}^{d}\),

\[
\norm{a-c}^{2}= \norm{a-b}^{2}+\norm{b-c}^{2}+2\iprod{a-b}{b-c}
\le \norm{a-b}^{2}+\norm{b-c}^{2}+2L\norm{a-b}\norm{b-c}.
\tag{2}
\]
\end{lemma}
\begin{proof}
The equality is elementary expansion of the squared norm.  
The inequality follows from Cauchy–Schwarz and the Lipschitz property
\(\norm{F(a)-F(b)}\le L\norm{a-b}\) applied to \(\iprod{F(a)-F(b)}{a-b}\).  ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ogda}.  All non‑trivial steps are numbered.

\medskip
\noindent\textbf{Notation.}  Write \(z^\star=(x^\star,y^\star)\) for a saddle point,
\(e_t:=z_t-z^\star\), and recall the OGDA update  

\[
z_{t+1}=z_t-\eta\big(2F(z_t)-F(z_{t-1})\big).\tag{3}
\]

\medskip
\begin{enumerate}
\item[Step 1.] \textbf{Expand the squared distance to the saddle point.}  
Using (3),

\[
\begin{aligned}
\norm{e_{t+1}}^{2}
&= \norm{e_t-\eta\big(2F(z_t)-F(z_{t-1})\big)}^{2}\\
&= \norm{e_t}^{2}
-2\eta\iprod{e_t}{2F(z_t)-F(z_{t-1})}
+\eta^{2}\norm{2F(z_t)-F(z_{t-1})}^{2}.
\end{aligned}
\tag{4}
\]

\item[Step 2.] \textbf{Insert and subtract \(F(z^\star)=0\).}  
Since \(F(z^\star)=0\),

\[
\iprod{e_t}{F(z_{t-1})}
=\iprod{e_t}{F(z_{t-1})-F(z^\star)}.
\tag{5}
\]

\item[Step 3.] \textbf{Use monotonicity of \(F\).}  
Monotonicity yields  

\[
\iprod{F(z_t)-F(z^\star)}{e_t}\ge 0,
\qquad
\iprod{F(z_{t-1})-F(z^\star)}{e_t}\ge 0.
\tag{6}
\]

Thus  

\[
-2\eta\iprod{e_t}{2F(z_t)-F(z_{t-1})}
\le -2\eta\iprod{e_t}{F(z_t)}.
\tag{7}
\]

\item[Step 4.] \textbf{Bound the third term via smoothness.}  
From Lipschitz continuity,
\[
\norm{2F(z_t)-F(z_{t-1})}\le 2L\norm{e_t}+L\norm{e_{t-1}}.
\tag{8}
\]
Hence,
\[
\eta^{2}\norm{2F(z_t)-F(z_{t-1})}^{2}
\le \eta^{2}\big(4L^{2}\norm{e_t}^{2}+L^{2}\norm{e_{t-1}}^{2}+4L^{2}\norm{e_t}\norm{e_{t-1}}\big).
\tag{9}
\]

\item[Step 5.] \textbf{Combine (4)–(9).}  
Substituting (7) and (9) into (4) gives  

\[
\begin{aligned}
\norm{e_{t+1}}^{2}
&\le \norm{e_t}^{2}
-2\eta\iprod{e_t}{F(z_t)}
+\eta^{2}\big(4L^{2}\norm{e_t}^{2}+L^{2}\norm{e_{t-1}}^{2}+4L^{2}\norm{e_t}\norm{e_{t-1}}\big).
\end{aligned}
\tag{10}
\]

\item[Step 6.] \textbf{Apply Cauchy–Schwarz to the mixed term.}  
\(2\eta\iprod{e_t}{F(z_t)}\ge 2\eta\frac{1}{L}\norm{F(z_t)}^{2}\) is not directly useful.
Instead we keep the term and later use the variational inequality
\(\iprod{F(z_t)}{e_t}\ge L(z_t)-L(z^\star)\) (see Lemma~\ref{lem:vi} below).  

Thus (10) becomes  

\[
\norm{e_{t+1}}^{2}
\le \norm{e_t}^{2}
-2\eta\big(L(x_t,y^\star)-L(x^\star,y_t)\big)
+4\eta^{2}L^{2}\norm{e_t}^{2}
+\eta^{2}L^{2}\norm{e_{t-1}}^{2}
+4\eta^{2}L^{2}\norm{e_t}\norm{e_{t-1}}.
\tag{11}
\]

\item[Step 7.] \textbf{Choose stepsize.}  
Assume \(\eta\le\frac{1}{2L}\).  Then \(4\eta^{2}L^{2}\le \eta L\) and
\(4\eta^{2}L^{2}\norm{e_t}\norm{e_{t-1}}\le\frac{1}{2}\eta L(\norm{e_t}^{2}+\norm{e_{t-1}}^{2})\).
Plugging these bounds into (11) yields  

\[
\norm{e_{t+1}}^{2}
\le (1-\eta L)\norm{e_t}^{2}
+\frac{3}{2}\eta L\norm{e_{t-1}}^{2}
-2\eta\big(L(x_t,y^\star)-L(x^\star,y_t)\big).
\tag{12}
\]

\item[Step 8.] \textbf{Recurrence telescoping.}  
Define \(V_t:=\norm{e_t}^{2}\).  From (12),

\[
V_{t+1}+2\eta\big(L(x_t,y^\star)-L(x^\star,y_t)\big)
\le (1-\eta L)V_t+\tfrac{3}{2}\eta L V_{t-1}.
\tag{13}
\]

Summing (13) from \(t=0\) to \(T-1\) and using \(V_{-1}=V_0\) (because \(z_{-1}=z_0\))
gives  

\[
\sum_{t=0}^{T-1}2\eta\big(L(x_t,y^\star)-L(x^\star,y_t)\big)
\le V_0-V_T+\tfrac{3}{2}\eta L\sum_{t=0}^{T-1}V_{t-1}.
\tag{14}
\]

Since all \(V_t\ge0\) and \(\eta L\le\frac12\), the right‑hand side is bounded by
\(V_0\).  Hence  

\[
\sum_{t=0}^{T-1}\big(L(x_t,y^\star)-L(x^\star,y_t)\big)
\le \frac{V_0}{2\eta}.
\tag{15}
\]

\item[Step 9.] \textbf{Pass to the averaged iterate.}  
Convexity in \(x\) and concavity in \(y\) imply  

\[
L(\bar x_T,y^\star)-L(x^\star,\bar y_T)
\le\frac{1}{T}\sum_{t=0}^{T-1}\big(L(x_t,y^\star)-L(x^\star,y_t)\big).
\tag{16}
\]

Combining (15) and (16),

\[
L(\bar x_T,y^\star)-L(x^\star,\bar y_T)
\le\frac{V_0}{2\eta T}
\le\frac{D^{2}}{2\eta T},
\tag{17}
\]
where \(D:=\max_{z\in\mathcal{Z}}\norm{z-z^\star}\) bounds the initial distance.
This is exactly the bound (1) stated in Theorem~\ref{thm:ogda}.
\end{enumerate}
Thus the theorem is proved. \(\square\)

\section*{Rate Derivation (With Constants)}
From (17) we read the explicit constant:

\[
\boxed{C:=\frac{D^{2}}{2\eta}},\qquad
\text{so that } \quad
\text{gap}(\bar z_T)\le\frac{C}{T}.
\]

If the domain diameter is unknown but the gradients are bounded,
\(\norm{F(z)}\le G\), one can use \(D\le G/\mu\) where \(\mu\) is the strong
convexity/concavity constant (if present), giving the alternative bound
\(\mathcal{O}(G^{2}/(\mu^{2}T))\).

\section*{Special Cases}
\begin{itemize}
\item \textbf{Bilinear games.}  For \(L(x,y)=x^{\top}Ay\) with \(A\) arbitrary,
\(F\) is linear and monotone.  The Lipschitz constant equals \(\norm{A}_2\).
OGDA converges with the same \(\mathcal{O}(1/T)\) rate, and the iterates
remain bounded for any stepsize \(\eta<1/(2\norm{A}_2)\).
\item \textbf{Strongly convex–strongly concave.}  If \(L\) is \(\mu\)-strongly
convex in \(x\) and \(\mu\)-strongly concave in \(y\), the distance
\(\norm{z_t-z^\star}\) contracts linearly:
\[
\norm{z_{t+1}-z^\star}\le (1-\eta\mu)\norm{z_t-z^\star},
\]
so OGDA enjoys an exponential rate \(\mathcal{O}\big((1-\eta\mu)^T\big)\).
\end{itemize}
