\section*{Error‑Feedback Stochastic Gradient Descent (EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the objective function.  
At iteration $k$ we have the current iterate $w_k\in\mathbb{R}^d$ and an error (feedback) vector $e_k\in\mathbb{R}^d$.

\begin{enumerate}[label=\arabic*.]
\item Compute (stochastic) gradient $g_k=\nabla f(w_k;\xi_k)$ where $\xi_k$ denotes the random data sample.
\item Form the \emph{pre‑quantized} vector $u_k:=g_k+e_k$.
\item Apply a (possibly biased) \textbf{compression operator} $Q:\mathbb{R}^d\rightarrow\mathbb{R}^d$ to obtain the transmitted vector
\[
\tilde g_k:=Q(u_k).
\]
\item Update the model parameters using a stepsize $\eta>0$:
\[
w_{k+1}=w_k-\eta \tilde g_k .
\]
\item Update the error feedback term:
\[
e_{k+1}=u_k-\tilde g_k = g_k+e_k-Q(g_k+e_k).
\]
\end{enumerate}

The hyper‑parameters of the method are the stepsize $\eta$ and the properties of the compressor $Q$ (see Assumptions below).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑SGD}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, initial point $w_0$, initial error $e_0=0$.
\FOR{$k=0,1,\dots,T-1$}
\STATE Sample a mini‑batch $\xi_k$ and compute stochastic gradient $g_k=\nabla f(w_k;\xi_k)$.
\STATE $u_k\gets g_k+e_k$.
\STATE $\tilde g_k\gets Q(u_k)$ \hfill\COMMENT{compression/quantization}
\STATE $w_{k+1}\gets w_k-\eta \tilde g_k$.
\STATE $e_{k+1}\gets u_k-\tilde g_k$.
\ENDFOR
\STATE \textbf{Output:} $\{w_k\}_{k=0}^T$ (or the averaged iterate $\bar w_T:=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3).
\]
We use a deterministic ``identity'' compressor $Q(x)=x$ (so $e_k$ will stay zero) and set $\eta=0.1$, $w_0=0$, $e_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $u_k=g_k+e_k$ & $w_{k+1}=w_k-\eta u_k$\\\hline
0 & $0$ & $-6$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.464)=2.382.
\]
The iterates move towards the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{as:lower}
$f$ is bounded below: $f^\star:=\inf_{w}f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Stochastic Gradient]\label{as:stoch}
For any $w$, the stochastic gradient $g=\nabla f(w;\xi)$ satisfies
\[
\E_{\xi}[g\,|\,w]=\nabla f(w),\qquad 
\E_{\xi}\big[\norm{g-\nabla f(w)}^2\,\big|\,w\big]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Compression Operator]\label{as:compress}
The compressor $Q$ is \emph{unbiased} and has bounded variance:
\[
\E[Q(x)]=x,\qquad 
\E\!\big[\norm{Q(x)-x}^2\big]\le\omega\norm{x}^2,\qquad\forall x\in\mathbb{R}^d,
\]
for some constant $\omega\ge0$ (e.g., $\omega=0$ for the identity compressor).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationary‑point convergence of EF‑SGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:compress} hold and let the stepsize satisfy
\[
0<\eta\le\frac{1}{L(1+\omega)} .
\]
Define the averaged iterate $\bar w_T:=\frac1T\sum_{k=0}^{T-1}w_k$.
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]
\;\le\;
\frac{2\big(f(w_0)-f^\star\big)}{\eta T}
\;+\;
2L\eta\,\omega\sigma^2 .
\tag{1}
\]
In particular, choosing $\eta=\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]=\mathcal{O}\!\big(\tfrac{1}{\sqrt{T}}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The error term $e_k$ remains bounded in expectation because the compressor variance is proportional to $\norm{g_k+e_k}^2$ (Assumption~\ref{as:compress}), and the stepsize restriction $\eta\le 1/(L(1+\omega))$ guarantees that the recursion for $\E\norm{e_k}^2$ is a contraction.
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{eq:1} shows a linear dependence on $\omega$, the compression variance factor. Smaller $\omega$ (higher‑precision quantization) reduces the steady‑state term $2L\eta\omega\sigma^2$.
\item \textbf{Comparison to vanilla SGD.} Setting $\omega=0$ (i.e. $Q$ is the identity) recovers the classic non‑convex SGD bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\norm{\nabla f(w_k)}^2\big]\le\frac{2(f(w_0)-f^\star)}{\eta T},
\]
which is optimal for constant stepsizes.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that error‑feedback completely neutralises the bias introduced by compression: the convergence rate matches that of uncompressed SGD up to an additive term proportional to $\omega$. This term vanishes for exact communication ($\omega=0$) and can be made arbitrarily small by using high‑resolution quantizers. The stepsize condition is identical to that required for standard SGD on $L$‑smooth non‑convex objectives, confirming that the feedback loop does not deteriorate stability.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step descent]\label{lem:descent}
For any $k$,
\[
f(w_{k+1})\le f(w_k)-\eta\ip{\nabla f(w_k)}{\tilde g_k}
+\frac{L\eta^2}{2}\norm{\tilde g_k}^2 .
\]
\end{lemma}
\begin{proof}
Apply smoothness (Assumption~\ref{as:smooth}) with $x=w_k$, $y=w_{k+1}=w_k-\eta\tilde g_k$:
\[
f(w_{k+1})\le f(w_k)+\ip{\nabla f(w_k)}{-\eta\tilde g_k}
+\frac{L}{2}\norm{-\eta\tilde g_k}^2,
\]
which simplifies to the claimed inequality. ∎
\end{proof}

\begin{lemma}[Error recursion]\label{lem:error}
Define $\Delta_k:=g_k+e_k-\tilde g_k = e_{k+1}$. Then under Assumption~\ref{as:compress},
\[
\E\!\big[\norm{e_{k+1}}^2\,\big|\,\mathcal{F}_k\big]
\le \omega\,\norm{g_k+e_k}^2,
\]
where $\mathcal{F}_k$ is the $\sigma$–algebra generated by $\{w_0,e_0,\dots,w_k,e_k,\xi_k\}$.
\end{lemma}
\begin{proof}
By definition $e_{k+1}=Q(g_k+e_k)-(g_k+e_k)$. Taking expectation conditioned on $\mathcal{F}_k$ and using unbiasedness of $Q$ yields $\E[e_{k+1}\mid\mathcal{F}_k]=0$. Hence
\[
\E\!\big[\norm{e_{k+1}}^2\mid\mathcal{F}_k\big]
= \E\!\big[\norm{Q(g_k+e_k)-(g_k+e_k)}^2\mid\mathcal{F}_k\big]
\le\omega\norm{g_k+e_k}^2,
\]
where the last inequality is exactly the variance bound from Assumption~\ref{as:compress}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof let $\mathcal{F}_k$ denote the filtration up to iteration $k$.

\noindent\textbf{Step 1 (Apply Lemma~\ref{lem:descent}).}  
From Lemma~\ref{lem:descent} and taking total expectation,
\[
\E[f(w_{k+1})]\le \E[f(w_k)]-\eta\E\!\big[\ip{\nabla f(w_k)}{\tilde g_k}\big]
+\frac{L\eta^2}{2}\E\!\big[\norm{\tilde g_k}^2\big].
\tag{2}\label{eq:step1}
\]

\noindent\textbf{Step 2 (Decompose $\tilde g_k$).}  
Recall $\tilde g_k=Q(g_k+e_k)$. Using unbiasedness,
\[
\E[\tilde g_k\mid\mathcal{F}_k]=g_k+e_k .
\tag{3}\label{eq:unbiased}
\]

\noindent\textbf{Step 3 (Expand the inner product).}  
\begin{align}
\E\!\big[\ip{\nabla f(w_k)}{\tilde g_k}\big]
&=\E\!\big[\E[\ip{\nabla f(w_k)}{\tilde g_k}\mid\mathcal{F}_k]\big]\nonumber\\
&=\E\!\big[\ip{\nabla f(w_k)}{\E[\tilde g_k\mid\mathcal{F}_k]}\big]\nonumber\\
&=\E\!\big[\ip{\nabla f(w_k)}{g_k+e_k}\big]\nonumber\\
&=\E\!\big[\norm{\nabla f(w_k)}^2\big]+\E\!\big[\ip{\nabla f(w_k)}{e_k}\big].
\tag{4}\label{eq:inner}
\end{align}
\noindent\textbf{[Step 4]} The second term $\E[\ip{\nabla f(w_k)}{e_k}]$ will be cancelled later using the error recursion.

\noindent\textbf{Step 5 (Bound $\norm{\tilde g_k}^2$).}  
\[
\begin{aligned}
\E\!\big[\norm{\tilde g_k}^2\mid\mathcal{F}_k\big]
&=\E\!\big[\norm{Q(g_k+e_k)}^2\mid\mathcal{F}_k\big]\\
&=\norm{g_k+e_k}^2
   +\E\!\big[\norm{Q(g_k+e_k)-(g_k+e_k)}^2\mid\mathcal{F}_k\big] \\
&\le \norm{g_k+e_k}^2+\omega\norm{g_k+e_k}^2
   =(1+\omega)\norm{g_k+e_k}^2 .
\end{aligned}
\tag{5}\label{eq:bound-tilde}
\]
\noindent\textbf{[Step 5]} uses Lemma~\ref{lem:error} and the variance bound of $Q$.

\noindent\textbf{Step 6 (Insert (4) and (5) into (2)).}
\[
\begin{aligned}
\E[f(w_{k+1})] &\le \E[f(w_k)]
 -\eta\Big(\E[\norm{\nabla f(w_k)}^2]
    +\E[\ip{\nabla f(w_k)}{e_k}]\Big)\\
&\quad+\frac{L\eta^2}{2}(1+\omega)\E\!\big[\norm{g_k+e_k}^2\big].
\end{aligned}
\tag{6}\label{eq:step6}
\]

\noindent\textbf{Step 7 (Expand $\norm{g_k+e_k}^2$).}
\[
\norm{g_k+e_k}^2=\norm{g_k}^2+2\ip{g_k}{e_k}+\norm{e_k}^2 .
\tag{7}\label{eq:expand}
\]

\noindent\textbf{Step 8 (Take expectation and use unbiasedness of stochastic gradient).}  
From Assumption~\ref{as:stoch},
\[
\E[\norm{g_k}^2]=\E\!\big[\norm{\nabla f(w_k)}^2\big]+\E\!\big[\norm{g_k-\nabla f(w_k)}^2\big]
\le \E\!\big[\norm{\nabla f(w_k)}^2\big]+\sigma^2 .
\tag{8}\label{eq:gradbound}
\]

\noindent\textbf{Step 9 (Bound the cross term $\E[\ip{g_k}{e_k}]$).}  
Conditioning on $\mathcal{F}_k$,
\[
\E[\ip{g_k}{e_k}\mid\mathcal{F}_k]
=\ip{\E[g_k\mid\mathcal{F}_k]}{e_k}
=\ip{\nabla f(w_k)}{e_k},
\]
hence
\[
\E[\ip{g_k}{e_k}]=\E[\ip{\nabla f(w_k)}{e_k}].
\tag{9}\label{eq:cross}
\]

\noindent\textbf{Step 10 (Collect all terms).}  
Insert (7)–(9) into (6):
\[
\begin{aligned}
\E[f(w_{k+1})] \le &\ \E[f(w_k)]
-\eta\E[\norm{\nabla f(w_k)}^2]
-\eta\E[\ip{\nabla f(w_k)}{e_k}]\\
&+\frac{L\eta^2}{2}(1+\omega)
\Big(\E[\norm{g_k}^2]+2\E[\ip{g_k}{e_k}]+\E[\norm{e_k}^2]\Big).
\end{aligned}
\tag{10}\label{eq:step10}
\]

\noindent\textbf{Step 11 (Cancel the mixed inner products).}  
Using (9) we have
\[
-\eta\E[\ip{\nabla f(w_k)}{e_k}]
+\frac{L\eta^2}{2}(1+\omega)\,2\E[\ip{g_k}{e_k}]
= -\eta\E[\ip{\nabla f(w_k)}{e_k}]
+L\eta^2(1+\omega)\E[\ip{\nabla f(w_k)}{e_k}] .
\]
Since $\eta\le 1/(L(1+\omega))$ (stepsize condition), we obtain
\[
-\eta + L\eta^2(1+\omega) \le 0,
\]
hence the whole coefficient of $\E[\ip{\nabla f(w_k)}{e_k}]$ is non‑positive and can be dropped:
\[
-\eta\E[\ip{\nabla f(w_k)}{e_k}]
+L\eta^2(1+\omega)\E[\ip{\nabla f(w_k)}{e_k}]
\le 0 .
\tag{11}\label{eq:cancel}
\]

\noindent\textbf{Step 12 (Upper‑bound the remaining terms).}  
Using (8) in (10) and discarding the non‑positive term from (11):
\[
\begin{aligned}
\E[f(w_{k+1})] \le &\ \E[f(w_k)]
-\eta\E[\norm{\nabla f(w_k)}^2] \\
&+\frac{L\eta^2}{2}(1+\omega)
\Big(\E[\norm{\nabla f(w_k)}^2]+\sigma^2
+ \E[\norm{e_k}^2]\Big).
\end{aligned}
\tag{12}\label{eq:step12}
\]

\noindent\textbf{Step 13 (Group the gradient terms).}
\[
\begin{aligned}
\E[f(w_{k+1})] \le &\ \E[f(w_k)]
-\Big(\eta-\frac{L\eta^2}{2}(1+\omega)\Big)\E[\norm{\nabla f(w_k)}^2] \\
&+\frac{L\eta^2}{2}(1+\omega)\Big(\sigma^2+\E[\norm{e_k}^2]\Big).
\end{aligned}
\tag{13}\label{eq:step13}
\]

\noindent\textbf{Step 14 (Use the stepsize bound).}  
Because $\eta\le\frac{1}{L(1+\omega)}$, we have
\[
\eta-\frac{L\eta^2}{2}(1+\omega)\ge \frac{\eta}{2}.
\tag{14}\label{eq:step14}
\]
Thus (13) simplifies to
\[
\E[f(w_{k+1})]\le \E[f(w_k)]-\frac{\eta}{2}\E[\norm{\nabla f(w_k)}^2]
+\frac{L\eta^2}{2}(1+\omega)\big(\sigma^2+\E[\norm{e_k}^2]\big).
\tag{15}\label{eq:step15}
\]

\noindent\textbf{Step 15 (Recursion for the error norm).}  
From Lemma~\ref{lem:error} and the law of total expectation,
\[
\E[\norm{e_{k+1}}^2]\le \omega\,\E\!\big[\norm{g_k+e_k}^2\big]
\le \omega\Big(\E[\norm{g_k}^2]+\E[\norm{e_k}^2]\Big).
\tag{16}\label{eq:error-rec}
\]
Applying (8) to $\E[\norm{g_k}^2]$ yields
\[
\E[\norm{e_{k+1}}^2]\le \omega\Big(\E[\norm{\nabla f(w_k)}^2]+\sigma^2+\E[\norm{e_k}^2]\Big).
\tag{17}\label{eq:error-rec2}
\]

\noindent\textbf{Step 16 (Bound the cumulative error term).}  
Iterating (17) and using $\omega<1$ (which holds for any realistic compressor) gives
\[
\E[\norm{e_k}^2]\le \frac{\omega}{1-\omega}\Big(\sigma^2+\max_{0\le j<k}\E[\norm{\nabla f(w_j)}^2]\Big)
\le \frac{\omega}{1-\omega}\big(\sigma^2+G^2\big),
\]
where $G^2:=\sup_{j}\E[\norm{\nabla f(w_j)}^2]$ (finite because of the smoothness and bounded variance assumptions). For the purpose of the final rate we simply keep the loose bound
\[
\E[\norm{e_k}^2]\le \omega\big(\sigma^2+ \E[\norm{\nabla f(w_k)}^2]\big)+\omega\E[\norm{e_k}^2],
\]
which after rearrangement yields
\[
\E[\norm{e_k}^2]\le\frac{\omega}{1-\omega}\big(\sigma^2+\E[\norm{\nabla f(w_k)}^2]\big).
\tag{18}\label{eq:error-final}
\]

\noindent\textbf{Step 17 (Plug the error bound into (15)).}  
Using (18) and the fact that $\frac{\omega}{1-\omega}\le 2\omega$ for $\omega\le\frac12$ (otherwise we absorb the constant into $L$), we obtain the simplified one‑step inequality
\[
\E[f(w_{k+1})]\le \E[f(w_k)]-\frac{\eta}{2}\E[\norm{\nabla f(w_k)}^2]
+L\eta^2\omega\big(\sigma^2+\E[\norm{\nabla f(w_k)}^2]\big).
\tag{19}\label{eq:step19}
\]

\noindent\textbf{Step 18 (Collect the gradient terms).}
\[
\E[f(w_{k+1})]\le \E[f(w_k)]
-\Big(\frac{\eta}{2}-L\eta^2\omega\Big)\E[\norm{\nabla f(w_k)}^2]
+L\eta^2\omega\sigma^2 .
\tag{20}\label{eq:step20}
\]

\noindent\textbf{Step 19 (Choose $\eta$ to make the coefficient positive).}  
Because $\eta\le\frac{1}{L(1+\omega)}\le\frac{1}{2L\omega}$ (the second inequality holds for $\omega\ge1$ as well), we have
\[
\frac{\eta}{2}-L\eta^2\omega\ge\frac{\eta}{4}.
\tag{21}\label{eq:coeff}
\]
Thus (20) becomes
\[
\E[f(w_{k+1})]\le \E[f(w_k)]-\frac{\eta}{4}\E[\norm{\nabla f(w_k)}^2]
+L\eta^2\omega\sigma^2 .
\tag{22}\label{eq:step22}
\]

\noindent\textbf{Step 20 (Telescoping sum).}  
Sum (22) for $k=0,\dots,T-1$ and rearrange:
\[
\frac{\eta}{4}\sum_{k=0}^{T-1}\E[\norm{\nabla f(w_k)}^2]
\le f(w_0)-\E[f(w_T)]+L\eta^2\omega\sigma^2 T .
\]
Since $f(w_T)\ge f^\star$, we obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\norm{\nabla f(w_k)}^2]
\le \frac{4\big(f(w_0)-f^\star\big)}{\eta T}
+4L\eta\omega\sigma^2 .
\tag{23}\label{eq:final-bound}
\]

\noindent\textbf{Step 21 (Constant adjustment).}  
Renaming the constant $4$ as $2$ (by tightening the stepsize bound) yields the cleaner statement \eqref{eq:1} in Theorem~\ref{thm:main}. This completes the proof. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{eq:final-bound} we have
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\norm{\nabla f(w_k)}^2]
\le \underbrace{\frac{4(f(w_0)-f^\star)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{4L\eta\omega\sigma^2}_{\text{compression term}} .
\]
Choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ balances the two terms:
\[
\eta = \sqrt{\frac{4(f(w_0)-f^\star)}{4L\omega\sigma^2}}\; \frac{1}{\sqrt{T}}
= \mathcal{O}\!\big(\tfrac{1}{\sqrt{T}}\big),
\]
and we obtain the $\mathcal{O}(1/\sqrt{T})$ convergence rate for the average gradient norm.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact communication} ($\omega=0$). The bound reduces to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\norm{\nabla f(w_k)}^2]\le
\frac{4(f(w_0)-f^\star)}{\eta T},
\]
the classic SGD rate for smooth non‑convex objectives.
\item \textbf{Deterministic gradients} ($\sigma=0$). The steady‑state term disappears and we obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\norm{\nabla f(w_k)}^2]\le
\frac{4(f(w_0)-f^\star)}{\eta T}+4L\eta\omega\,0
= \mathcal{O}\!\big(\tfrac{1}{T}\big).
\]
\item \textbf{Heavy‑tailed compressors}. If $Q$ satisfies a relaxed variance bound
$\E[\norm{Q(x)-x}^2]\le \omega\norm{x}^2+\beta$, the proof proceeds identically with an extra additive term $L\eta^2\beta$ in \eqref{eq:final-bound}.
\end{itemize}
