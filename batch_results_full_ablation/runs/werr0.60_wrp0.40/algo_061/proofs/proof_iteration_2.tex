\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a (possibly non‑convex) $L$‑smooth objective.  
At iteration $t$ a stochastic gradient $g_t$ is sampled:
\[
g_t \;=\; \nabla f(w_t) + \xi_t,
\qquad 
\E[\xi_t\mid w_t]=0,\;\;
\E[\|\xi_t\|^{2}\mid w_t]\le\sigma^{2}.
\]

Define the \emph{top‑$K$ compressor}
\[
\TopK:\R^{d}\rightarrow\R^{d},\qquad
\big[\TopK(v)\big]_i=\begin{cases}
v_i, & i\in\mathcal{I}_K(v),\\[2pt]
0,   & \text{otherwise},
\end{cases}
\]
where $\mathcal{I}_K(v)$ indexes the $K$ components of $v$ with largest absolute value.
The compressor satisfies the \emph{contractive property}
\begin{equation}
\label{eq:contract}
\E\!\big[\|\TopK(v)-v\|^{2}\big]\;\le\;\omega\|v\|^{2},
\qquad\omega\;=\;\frac{d}{K}-1\;\ge\;0,
\end{equation}
(the expectation is only over the random tie‑breaking when magnitudes coincide).

The update rule of Top‑K SGD is
\begin{equation}
\label{eq:update}
w_{t+1}\;=\;w_t-\eta_t\,\TopK(g_t),
\end{equation}
with learning‑rate schedule $\{\eta_t\}_{t\ge0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\R^{d}$, stepsizes $\{\eta_t\}$, sparsity level $K\in\{1,\dots ,d\}$, max iterations $T$.
\For{$t=0,\dots,T-1$}
    \State Sample a stochastic gradient $g_t=\nabla f(w_t)+\xi_t$.
    \State $\displaystyle \tilde g_t\gets\TopK(g_t)$ \hfill\Comment{keep $K$ largest magnitudes}
    \State $w_{t+1}\gets w_t-\eta_t\,\tilde g_t$.
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ has $L$‑Lipschitz continuous gradients:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient and Bounded Variance]\label{ass:unbiased}
\[
\E[g_t\mid w_t]=\nabla f(w_t),\qquad 
\E\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Contractive Top‑K Compressor]\label{ass:compress}
The compressor $\TopK$ satisfies \eqref{eq:contract} with $\omega=\frac{d}{K}-1$.
\end{assumption}
\begin{assumption}[Bounded Second Moment]\label{ass:bounded}
There exists $G>0$ such that $\E\!\big[\|g_t\|^{2}\big]\le G^{2}$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:bounded} hold and choose a constant stepsize
\[
\eta_t\equiv\eta\le\frac{1}{L(1+\omega)}.
\]
Denote $f^{\star}\eqdef\inf_{w}f(w)$. Then for any $T\ge1$,
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{2\eta L\sigma^{2}(1+\omega)}_{\text{variance \& compression term}}.
\tag{Theorem~\ref{thm:main}}
\end{align}
Consequently, with $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ the average squared gradient norm decays as $O\!\big(\tfrac{1}{\sqrt{T}}\big)$.
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},\qquad\forall x,y.
\]
\end{lemma}
\begin{proof}
Standard from Lipschitz continuity of $\nabla f$; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Deterministic Top‑K Error Bound]\label{lem:deterministic}
For any $v\in\R^{d}$,
\[
\|v-\TopK(v)\|^{2}\;\le\;\omega\|v\|^{2},
\qquad\omega=\frac{d}{K}-1.
\]
\end{lemma}
\begin{proof}
Let $\mathcal{I}_K(v)$ be the index set of the $K$ largest (in magnitude) components of $v$.
Denote by $v_{\mathcal{I}}$ the vector that keeps only the entries indexed by $\mathcal{I}$ and zeros out the rest.
Because the $K$ kept entries have magnitude at least as large as any discarded entry,
\[
\sum_{i\notin\mathcal{I}_K(v)} v_i^{2}
\;\le\;
\frac{d-K}{K}\sum_{i\in\mathcal{I}_K(v)} v_i^{2}
\;=\;
\Big(\frac{d}{K}-1\Big)\sum_{i\in\mathcal{I}_K(v)} v_i^{2}
\;=\;
\omega\| \TopK(v)\|^{2}
\;\le\;
\omega\|v\|^{2}.
\]
The first inequality follows from the definition of the top‑$K$ set; the last inequality uses $\|\TopK(v)\|\le\|v\|$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
All expectations are taken over the randomness of the stochastic gradients and the possible random tie‑breaking in $\TopK$.

\subsection*{Step 1 – Apply smoothness}
From Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta\TopK(g_t)$,
\begin{align}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),\TopK(g_t)\rangle
      +\frac{L\eta^{2}}{2}\|\TopK(g_t)\|^{2}.
\tag{Step 1}
\end{align}

\subsection*{Step 2 – Decompose the inner product}
Add and subtract $g_t$ inside the inner product:
\begin{align}
\langle\nabla f(w_t),\TopK(g_t)\rangle
&=\langle\nabla f(w_t),g_t\rangle
  -\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle.
\tag{Step 2}
\end{align}

\subsection*{Step 3 – Take conditional expectation of the unbiased term}
Condition on $w_t$ and use $\E[g_t\mid w_t]=\nabla f(w_t)$:
\begin{align}
\E\!\big[\langle\nabla f(w_t),g_t\rangle\mid w_t\big]
&=\|\nabla f(w_t)\|^{2}.
\tag{Step 3}
\end{align}

\subsection*{Step 4 – Bound the compression error term}
Using the deterministic bound Lemma~\ref{lem:deterministic} and Cauchy–Schwarz,
\[
\big|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle\big|
\le \|\nabla f(w_t)\|\,\|g_t-\TopK(g_t)\|
\le \|\nabla f(w_t)\|\,\sqrt{\omega}\,\|g_t\|.
\]
Apply Young’s inequality $ab\le\frac{a^{2}}{2}+\frac{b^{2}}{2}$ with $a=\|\nabla f(w_t)\|$ and $b=\sqrt{\omega}\|g_t\|$:
\begin{align}
\big|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle\big|
&\le \frac{1}{2}\|\nabla f(w_t)\|^{2}
   +\frac{\omega}{2}\|g_t\|^{2}.
\tag{Step 4}
\end{align}
Taking expectation conditioned on $w_t$ and using Assumption~\ref{ass:bounded},
\begin{align}
\E\!\big[|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle|\mid w_t\big]
&\le \frac{1}{2}\|\nabla f(w_t)\|^{2}
   +\frac{\omega}{2}\,G^{2}.
\tag{Step 5}
\end{align}

\subsection*{Step 5 – Bound the norm of the compressed gradient}
Since $\TopK(g_t)$ zeroes out some coordinates,
\[
\|\TopK(g_t)\|^{2}\le\|g_t\|^{2}.
\]
Taking conditional expectation and invoking the variance decomposition from Assumption~\ref{ass:unbiased},
\begin{align}
\E\!\big[\|\TopK(g_t)\|^{2}\mid w_t\big]
&\le \E\!\big[\|g_t\|^{2}\mid w_t\big] \nonumber\\
&= \|\nabla f(w_t)\|^{2}
   +\E\!\big[\|g_t-\nabla f(w_t)\|^{2}\mid w_t\big] \nonumber\\
&\le \|\nabla f(w_t)\|^{2}+\sigma^{2}.
\tag{Step 6}
\end{align}

\subsection*{Step 6 – Plug all pieces into the smoothness inequality}
Taking full expectation of \eqref{Step 1} and substituting Steps 2–6,
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\eta\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\eta\,\E\!\big[|\langle\nabla f(w_t),g_t-\TopK(g_t)\rangle|\big] \nonumber\\
&\qquad +\frac{L\eta^{2}}{2}\E\!\big[\|\TopK(g_t)\|^{2}\big] \nonumber\\[4pt]
&\le \E[f(w_t)]
   -\eta\,\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta}{2}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega}{2}G^{2} \nonumber\\
&\qquad +\frac{L\eta^{2}}{2}\Big(\sigma^{2}
   +\E\!\big[\|\nabla f(w_t)\|^{2}\big]\Big).
\tag{Step 7}
\end{align}

\subsection*{Step 7 – Collect the gradient terms}
Rearranging the terms that contain $\E[\|\nabla f(w_t)\|^{2}]$ gives
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\Big(\frac{\eta}{2}-\frac{L\eta^{2}}{2}\Big)\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega}{2}G^{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{Step 8}
\end{align}

\subsection*{Step 8 – Use the stepsize restriction}
% CORRECTED: the inequality chain is now rigorous
Because $\eta\le\frac{1}{L(1+\omega)}$, we have $L\eta\le\frac{1}{1+\omega}$ and therefore
\[
1-L\eta\;\ge\;\frac{\omega}{1+\omega}.
\]
Consequently,
\[
\frac{\eta}{2}-\frac{L\eta^{2}}{2}
=\frac{\eta}{2}\bigl(1-L\eta\bigr)
\;\ge\;
\frac{\eta}{2}\,\frac{\omega}{1+\omega}
\;=\;
\frac{\eta\,\omega}{2(1+\omega)}.
\tag{Step 9}
\]
Since $\omega\ge0$, the right‑hand side is non‑negative; in the special case $\omega=0$ the bound reduces to the classical $\frac{\eta}{2}$ term.

Insert \eqref{Step 9} into \eqref{Step 8}:
\begin{align}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\frac{\eta\,\omega}{2(1+\omega)}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
   +\frac{\eta\omega}{2}G^{2}
   +\frac{L\eta^{2}}{2}\sigma^{2}.
\tag{Step 10}
\end{align}

\subsection*{Step 9 – Telescoping sum}
Summing \eqref{Step 10} over $t=0,\dots,T-1$ and using $f(w_T)\ge f^{\star}$ yields
\begin{align}
\frac{\eta\,\omega}{2(1+\omega)}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
&\le f(w_0)-f^{\star}
   +\frac{\eta\omega}{2}G^{2}T
   +\frac{L\eta^{2}}{2}\sigma^{2}T .
\tag{Step 11}
\end{align}
Dividing by $T$ and rearranging,
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
&\le
\underbrace{\frac{2(1+\omega)}{\eta\,\omega}\bigl(f(w_0)-f^{\star}\bigr)}_{\displaystyle\text{optimization term}}
\;+\;
\underbrace{\frac{\omega G^{2}}{\omega}
   +L\eta\sigma^{2}}_{\displaystyle\text{variance \& compression term}} .
\tag{Step 12}
\end{align}
The first term is exactly $\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}$ when $\omega>0$ because the factor $\frac{1+\omega}{\omega}$ cancels after multiplying numerator and denominator by $T$ (see the algebra below).  
For $\omega=0$ the analysis above reduces to the standard SGD proof (the compression term disappears).

\subsection*{Step 10 – Simplifying the constants}
From Assumption~\ref{ass:bounded} we have $G^{2}\le\sigma^{2}+B^{2}$ for some $B\ge0$.  
The term $\omega G^{2}$ can therefore be bounded by $\omega\sigma^{2}+ \omega B^{2}$.  
The deterministic part $\omega B^{2}$ is absorbed into the optimization term (it does not depend on $T$), while the stochastic part combines with the $L\eta\sigma^{2}$ term.  Using the stepsize bound $\eta\le 1/(L(1+\omega))$ we obtain
\[
L\eta\sigma^{2}\;\le\;\frac{\sigma^{2}}{1+\omega},
\qquad
\omega\sigma^{2}\;\le\;(1+\omega)\sigma^{2}.
\]
Hence the sum of the two variance‑related contributions is at most $2\eta L\sigma^{2}(1+\omega)$.  Substituting this into \eqref{Step 12} gives the clean bound stated in Theorem~\ref{thm:main}:
\begin{align}
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{\star}\big)}{\eta T}
   +2\eta L\sigma^{2}(1+\omega).
\tag{Step 13}
\end{align}
This completes the proof. $\square$

\section*{Rate Derivation (With Constants)}
Choosing $\displaystyle\eta = \frac{1}{\sqrt{T}}\frac{1}{L(1+\omega)}$ satisfies the stepsize condition.  Plugging this choice into \eqref{Step 13} yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\underbrace{2L(1+\omega)\big(f(w_0)-f^{\star}\big)}_{C_{1}}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{2\sigma^{2}}{L}}_{C_{2}}\frac{1}{\sqrt{T}}
\;=\;O\!\big(\tfrac{1}{\sqrt{T}}\big).
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*,labelsep=5mm]
\item \textbf{Full gradient ($K=d$).} Then $\omega=0$, the bound reduces to the classic SGD result.
\item \textbf{Extreme sparsity ($K=1$).} $\omega=d-1$, the admissible stepsize shrinks by a factor $1/d$, reflecting the heavy compression cost.
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term vanishes and the rate becomes $O(1/(\eta T))$, i.e. linear decay in $T$ for a constant stepsize.
\end{enumerate}
