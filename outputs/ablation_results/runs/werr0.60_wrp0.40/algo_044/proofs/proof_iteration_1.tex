\section*{Algorithm Name}
\textbf{DP‑SGD with Gaussian Gradient Noise} (DP‑SGD\textsubscript{G})

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the empirical risk 
\[
f(w)=\frac{1}{n}\sum_{i=1}^{n}\ell(w;z_i),
\]
where each $\ell(\cdot;z_i)$ is convex and $L$‑smooth.  
At iteration $t$ we draw a minibatch $\mathcal{B}_t\subset\{1,\dots,n\}$ of size $b$ uniformly at random and compute the (clipped) stochastic gradient
\[
\tilde{g}_t=\frac{1}{b}\sum_{i\in\mathcal{B}_t}\operatorname{clip}_{C}\!\big(\nabla\ell(w_t;z_i)\big),\qquad 
\operatorname{clip}_{C}(u)=\frac{u}{\max\{1,\|u\|_2/C\}} .
\]
Gaussian noise is added:
\[
g_t=\tilde{g}_t+ \xi_t,\qquad \xi_t\sim\mathcal{N}\!\big(0,\sigma^{2}I_{d}\big).
\]
The update rule is
\[
\boxed{w_{t+1}=w_t-\eta_t g_t},
\]
with learning‑rate schedule $\{\eta_t\}_{t\ge 0}$ (constant or decreasing).  
The noise scale $\sigma$ is chosen to satisfy a given $(\varepsilon,\delta)$‑DP budget via the moments accountant (or Rényi DP) for the sampling‑and‑adding‑noise mechanism.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD\textsubscript{G}}
\begin{algorithmic}[1]
\State \textbf{Input:} data $\{z_i\}_{i=1}^{n}$, loss $\ell$, clipping norm $C$, noise scale $\sigma$, batch size $b$, steps $T$, learning‑rate schedule $\{\eta_t\}$.
\State Initialize $w_0\in\R^{d}$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a minibatch $\mathcal{B}_t$ of size $b$ uniformly without replacement.
    \State Compute per‑example gradients $g_i^{(t)}=\nabla\ell(w_t;z_i)$ for $i\in\mathcal{B}_t$.
    \State Clip: $\hat g_i^{(t)}=\operatorname{clip}_{C}(g_i^{(t)})$.
    \State Aggregate: $\tilde g_t=\frac{1}{b}\sum_{i\in\mathcal{B}_t}\hat g_i^{(t)}$.
    \State Sample noise $\xi_t\sim\mathcal{N}(0,\sigma^{2}I_d)$.
    \State Form noisy gradient $g_t=\tilde g_t+\xi_t$.
    \State Update $w_{t+1}=w_t-\eta_t g_t$.
\EndFor
\State \textbf{Output:} $w_T$ (or average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the 1‑dimensional quadratic $f(w)=(w-3)^2$.  
Here $\nabla f(w)=2(w-3)$.  Set $w_0=0$, constant step size $\eta=0.1$, clipping bound $C=+\infty$ (no clipping), noise variance $\sigma^2=0.01$, and use a batch of size $1$ (deterministic gradient).  
We illustrate three iterations (noise draws are shown explicitly).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_t)$ & $\xi_t\sim\mathcal{N}(0,0.01)$ & $w_{t+1}=w_t-\eta(\nabla f(w_t)+\xi_t)$ \\ \hline
0 & $2(0-3)=-6$ & $\xi_0=0.05$ & $w_1=0-0.1(-6+0.05)=0.595$ \\ 
1 & $2(0.595-3)=-4.81$ & $\xi_1=-0.02$ & $w_2=0.595-0.1(-4.81-0.02)=1.078$ \\

\end{tabular}
\end{center}
Corresponding objective values $f(w_t)$:
\[
f(w_0)=9,\quad f(w_1)=5.78,\quad f(w_2)=3.70,\quad f(w_3)=2.91.
\]
The sequence shows descent despite the injected Gaussian noise.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
For all $w,w'\in\R^{d}$,
\[
f(w')\ge f(w)+\langle\nabla f(w),w'-w\rangle,
\qquad 
\|\nabla f(w)-\nabla f(w')\|_2\le L\|w-w'\|_2 .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient after Clipping]\label{ass:bound}
The clipping operation guarantees $\|\tilde g_t\|_2\le C$ almost surely for every $t$.
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
Conditioned on $w_t$, the minibatch gradient satisfies
\[
\E[\tilde g_t\mid w_t]=\nabla f(w_t).
\]
\end{assumption}
\begin{assumption}[Noise Independence]\label{ass:noise}
The Gaussian noise $\{\xi_t\}_{t\ge0}$ is independent of the data and of each other, with
\[
\E[\xi_t]=0,\qquad \E[\|\xi_t\|_2^{2}]=d\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Learning‑Rate]\label{ass:lr}
The step sizes are non‑increasing and satisfy $\eta_t\le \frac{1}{L}$ for all $t$.
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:conv}
Assume \ref{ass:convex}–\ref{ass:lr} hold.  Let $w^{*}\in\arg\min_{w}f(w)$.  Define the averaged iterate
\[
\bar w_T:=\frac{1}{T}\sum_{t=0}^{T-1}w_t .
\]
If a constant step size $\eta_t\equiv\eta\le\frac{1}{2L}$ is used, then
\[
\boxed{
\E\!\big[f(\bar w_T)-f(w^{*})\big]\le
\frac{\|w_0-w^{*}\|_2^{2}}{2\eta T}
+\frac{\eta C^{2}}{2}
+\frac{\eta d\sigma^{2}}{2}
}.
\tag{1}
\]
Choosing $\eta=\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ balances the three terms and yields
\[
\E\!\big[f(\bar w_T)-f(w^{*})\big]=O\!\left(\frac{1}{\sqrt{T}}\right)+O\!\left(\frac{d\sigma^{2}}{\sqrt{T}}\right).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound (1) is linear in the noise variance $\sigma^{2}$; larger privacy guarantees (larger $\sigma$) degrade utility at a rate $O(\sigma^{2}/\sqrt{T})$.
\item \textbf{Hyper‑parameter Sensitivity.} The optimal $\eta$ scales as $1/\sqrt{T}$; using a too‑large constant step size violates the condition $\eta\le 1/(2L)$ and can cause divergence.
\item \textbf{Comparison to Non‑Private SGD.} Setting $\sigma=0$ recovers the classic SGD bound $\frac{\|w_0-w^{*}\|^{2}}{2\eta T}+\frac{\eta C^{2}}{2}$.
\item \textbf{Effect of Batch Size.} Larger minibatch $b$ reduces the variance of $\tilde g_t$ (not shown explicitly because we assume clipping makes $\|\tilde g_t\|\le C$).  The privacy‑noise term $d\sigma^{2}$ is independent of $b$ under the moments‑accountant accounting.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The theorem demonstrates that DP‑SGD with Gaussian noise retains the $O(1/\sqrt{T})$ convergence rate of standard SGD for convex, smooth objectives, up to an additive term proportional to the privacy noise variance.  Consequently, for a fixed privacy budget, increasing the number of iterations $T$ improves utility only up to the point where the $1/\sqrt{T}$ term and the noise term balance; beyond that, additional iterations yield diminishing returns.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity for $L$‑smooth functions]\label{lem:3pt}
If $f$ is $L$‑smooth, then for any $u,v\in\R^{d}$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|_2^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Non‑expansiveness of Projection onto Convex Set]\label{lem:proj}
If $P_{\mathcal{K}}$ denotes Euclidean projection onto a closed convex set $\mathcal{K}$, then for any $x,y$,
\[
\|P_{\mathcal{K}}(x)-P_{\mathcal{K}}(y)\|_2\le\|x-y\|_2 .
\]
\end{lemma}
\begin{proof}
Follows from the firm non‑expansiveness of projections. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}.  Throughout the proof we condition on the history $\mathcal{F}_t:=\sigma(w_0,\xi_0,\dots,\xi_{t-1})$.

\noindent\textbf{[Step 1]}  Expand the squared distance after one update:
\begin{align}
\|w_{t+1}-w^{*}\|_2^{2}
&=\|w_t-\eta g_t-w^{*}\|_2^{2} \nonumber\\
&=\|w_t-w^{*}\|_2^{2}
-2\eta\langle g_t,w_t-w^{*}\rangle
+\eta^{2}\|g_t\|_2^{2}. \tag{2}
\end{align}

\noindent\textbf{[Step 2]}  Decompose $g_t=\tilde g_t+\xi_t$ and take expectation conditioned on $\mathcal{F}_t$:
\begin{align}
\E[\langle g_t,w_t-w^{*}\rangle\mid\mathcal{F}_t]
&=\E[\langle\tilde g_t,w_t-w^{*}\rangle\mid\mathcal{F}_t]
+\E[\langle\xi_t,w_t-w^{*}\rangle\mid\mathcal{F}_t] \nonumber\\
&=\langle\E[\tilde g_t\mid\mathcal{F}_t],w_t-w^{*}\rangle
+0 \quad\text{(by Assumption~\ref{ass:noise})} \nonumber\\
&=\langle\nabla f(w_t),w_t-w^{*}\rangle
\quad\text{(by Assumption~\ref{ass:unbiased})}. \tag{3}
\end{align}

\noindent\textbf{[Step 3]}  Bound the squared norm term:
\begin{align}
\E[\|g_t\|_2^{2}\mid\mathcal{F}_t]
&=\E[\|\tilde g_t+\xi_t\|_2^{2}\mid\mathcal{F}_t] \nonumber\\
&=\E[\|\tilde g_t\|_2^{2}\mid\mathcal{F}_t]
+2\E[\langle\tilde g_t,\xi_t\rangle\mid\mathcal{F}_t]
+\E[\|\xi_t\|_2^{2}] \nonumber\\
&=\E[\|\tilde g_t\|_2^{2}\mid\mathcal{F}_t]
+0
+d\sigma^{2} \quad\text{(noise independent, zero mean)}. \tag{4}
\end{align}
Because of clipping, $\|\tilde g_t\|_2\le C$, hence
\[
\E[\|\tilde g_t\|_2^{2}\mid\mathcal{F}_t]\le C^{2}. \tag{5}
\]

\noindent\textbf{[Step 4]}  Insert (3)–(5) into (2) and take full expectation:
\begin{align}
\E\big[\|w_{t+1}-w^{*}\|_2^{2}\big]
&\le \E\big[\|w_t-w^{*}\|_2^{2}\big]
-2\eta\,\E\!\big[\langle\nabla f(w_t),w_t-w^{*}\rangle\big]
+\eta^{2}\big(C^{2}+d\sigma^{2}\big). \tag{6}
\end{align}

\noindent\textbf{[Step 5]}  Use convexity of $f$:
\[
f(w_t)-f(w^{*})\le\langle\nabla f(w_t),w_t-w^{*}\rangle . \tag{7}
\]
Plug (7) into (6):
\begin{align}
2\eta\,\E\!\big[f(w_t)-f(w^{*})\big]
&\le \E\big[\|w_t-w^{*}\|_2^{2}\big]
-\E\big[\|w_{t+1}-w^{*}\|_2^{2}\big]
+\eta^{2}\big(C^{2}+d\sigma^{2}\big). \tag{8}
\end{align}

\noindent\textbf{[Step 6]}  Sum (8) over $t=0,\dots,T-1$:
\begin{align}
2\eta\sum_{t=0}^{T-1}\E\!\big[f(w_t)-f(w^{*})\big]
&\le \|w_0-w^{*}\|_2^{2}
-\E\!\big[\|w_T-w^{*}\|_2^{2}\big]
+T\eta^{2}\big(C^{2}+d\sigma^{2}\big) \nonumber\\
&\le \|w_0-w^{*}\|_2^{2}
+T\eta^{2}\big(C^{2}+d\sigma^{2}\big). \tag{9}
\end{align}
(The last inequality drops the non‑negative term $-\E\|w_T-w^{*}\|^2$.)

\noindent\textbf{[Step 7]}  Divide by $2\eta T$ and use convexity of $f$ again to replace the average of function values by the function at the average iterate (Jensen):
\[
\E\!\big[f(\bar w_T)-f(w^{*})\big]
\le\frac{\|w_0-w^{*}\|_2^{2}}{2\eta T}
+\frac{\eta}{2}\big(C^{2}+d\sigma^{2}\big). \tag{10}
\]
Equation (10) is exactly the bound claimed in (1).  ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From (10) set $\eta=\frac{\|w_0-w^{*}\|_2}{\sqrt{T\,(C^{2}+d\sigma^{2})}}$, which respects $\eta\le 1/(2L)$ for sufficiently large $T$ (by Assumption~\ref{ass:lr}).  Substituting gives
\[
\E\!\big[f(\bar w_T)-f(w^{*})\big]
\le \frac{\|w_0-w^{*}\|_2\sqrt{C^{2}+d\sigma^{2}}}{\sqrt{T}}
=O\!\left(\frac{1}{\sqrt{T}}\right)
+O\!\left(\frac{d\sigma^{2}}{\sqrt{T}}\right).
\]
Thus the convergence rate is $O(1/\sqrt{T})$ with an additive penalty proportional to the privacy‑noise variance.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{No privacy noise ($\sigma=0$).} Bound (10) reduces to the classic SGD guarantee $\frac{\|w_0-w^{*}\|^{2}}{2\eta T}+\frac{\eta C^{2}}{2}$.
\item \textbf{Strongly convex $f$ (parameter $\mu>0$).} A similar derivation using the strong convexity inequality
$\langle\nabla f(w_t),w_t-w^{*}\rangle\ge \mu\|w_t-w^{*}\|^{2}$
yields a linear convergence term $O\big((1-\eta\mu)^{T}\big)$ plus an $O(\eta d\sigma^{2})$ steady‑state error.
\item \textbf{Increasing batch size $b$.} If the clipping bound $C$ is kept fixed, larger $b$ reduces the variance of $\tilde g_t$ but does not affect the bound because we already use the worst‑case $C$; privacy‑noise variance $d\sigma^{2}$ dominates when $\sigma$ is set by the DP accountant.
\end{itemize}
