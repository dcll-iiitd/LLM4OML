\section*{Algorithm Name}
\textbf{Stochastic Subsampled Newton (SSN)}  

The method uses a stochastic estimate of the Hessian matrix obtained by sampling a mini‑batch of data at each iteration.  

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the empirical risk 
\[
f(x)=\frac1n\sum_{i=1}^n \ell_i(x),
\]
where each component loss $\ell_i$ is twice continuously differentiable.  
At iteration $t$ we draw a random subset $S_t\subset\{1,\dots,n\}$ of size $|S_t|=b$ and define  

\[
\begin{aligned}
g_t &:= \nabla f_{S_t}(x_t)=\frac1b\sum_{i\in S_t}\nabla \ell_i(x_t),\\[4pt]
H_t &:= \nabla^2 f_{S_t}(x_t)+\lambda \Id
      =\frac1b\sum_{i\in S_t}\nabla^2 \ell_i(x_t)+\lambda \Id,
\end{aligned}
\]
where $\lambda>0$ is a damping parameter guaranteeing $H_t\succ0$.  
The update rule is  

\[
\boxed{\,x_{t+1}=x_t-\alpha H_t^{-1}g_t\,},
\]
with step‑size (learning rate) $\alpha\in(0,1]$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Subsampled Newton (SSN)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\mathbb{R}^d$, step size $\alpha\in(0,1]$, batch size $b$, damping $\lambda>0$, maximum iterations $T$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample a mini‑batch $S_t\subset\{1,\dots,n\}$ uniformly with $|S_t|=b$.
    \STATE Compute stochastic gradient $g_t=\nabla f_{S_t}(x_t)$.
    \STATE Compute stochastic Hessian $H_t=\nabla^2 f_{S_t}(x_t)+\lambda\Id$.
    \STATE Solve $p_t=H_t^{-1}g_t$ (e.g. by CG or direct factorisation).
    \STATE Update $x_{t+1}=x_t-\alpha p_t$.
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the best iterate according to $f$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$ with exact Hessian $2$.  
We set $\alpha=0.1$, $\lambda=0$ (so $H_t=2$ exactly), and run three iterations.

\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\alpha H_t^{-1}g_t$ \\ \hline
0 & $0$ & $-6$ & $0-0.1\cdot(1/2)(-6)=0.3$ \\
1 & $0.3$ & $-5.4$ & $0.3-0.1\cdot(1/2)(-5.4)=0.57$ \\
2 & $0.57$ & $-4.86$ & $0.57-0.1\cdot(1/2)(-4.86)=0.813$ \\
\end{tabular}

Objective values: $f(0)=9$, $f(0.3)=7.29$, $f(0.57)=5.90$, $f(0.813)=4.78$, showing descent.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ has $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Hessian Lipschitz]\label{as:hessian}
The Hessian is $\rho$‑Lipschitz:
\[
\|\nabla^2 f(x)-\nabla^2 f(y)\|\le \rho\|x-y\|,\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{as:grad-unbiased}
\[
\E[g_t\mid \mathcal{F}_t]=\nabla f(x_t),
\]
where $\mathcal{F}_t=\sigma\{x_0,S_0,\dots,S_{t-1}\}$.
\end{assumption}
\begin{assumption}[Bounded variance of gradient]\label{as:grad-var}
\[
\E\big[\|g_t-\nabla f(x_t)\|^2\mid\mathcal{F}_t\big]\le\sigma_g^2.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic Hessian]\label{as:hess-unbiased}
\[
\E[H_t\mid\mathcal{F}_t]=\nabla^2 f(x_t)+\lambda\Id .
\]
\end{assumption}
\begin{assumption}[Deterministic bound on the inverse Hessian]\label{as:hess-inv}
Because of the damping $\lambda>0$, for every $t$
\[
H_t\succeq \lambda\Id\quad\Longrightarrow\quad 
\|H_t^{-1}\|\le \frac1\lambda\; .
\]
Define the constant
\[
c_H:=\frac1\lambda .
\tag{A6'}
\]
\end{assumption}
All constants $L,\rho,\sigma_g,\lambda$ are assumed known and finite.

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Stationarity]\label{thm:main}
Assume \ref{as:smooth}, \ref{as:grad-unbiased}, \ref{as:grad-var} and \ref{as:hess-inv} hold.
Let the step size satisfy 
\[
0<\alpha\le \frac{1}{L\,c_H^{3}} .
\tag{*}
\]
Let $x_T$ be the iterate after $T$ steps of SSN. Then  

\[
\frac1T\sum_{t=0}^{T-1}\E\big[\|\nabla f(x_t)\|^2\big]
\;\le\;
\frac{2c_H\big(f(x_0)-f^\star\big)}{\alpha T}
\;+\;
L\alpha c_H^{2}\sigma_g^{2}.
\tag{1}
\]

Consequently, choosing $\alpha=\Theta\!\big(T^{-1/2}\big)$ yields  

\[
\min_{0\le t<T}\E\big[\|\nabla f(x_t)\|^2\big]=\mathcal{O}\!\big(T^{-1/2}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The damping $\lambda>0$ together with Lemma \ref{lem:inv-bound} guarantees that the Newton direction never explodes, even when the subsampled Hessian is rank‑deficient.
\item \textbf{Hyper‑parameter Sensitivity.} The bound (1) shows a linear dependence on $\alpha$ in the variance term and an inverse dependence on $\alpha$ in the bias term, leading to the optimal $\alpha\propto T^{-1/2}$.
\item \textbf{Comparison to SGD.} For the same variance $\sigma_g^2$, SSN improves the bias term by a factor $c_H\le 1$ (since $c_H=1/\lambda$ can be much smaller than $1$ when strong curvature is present), yielding faster decay of the gradient norm.
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item If $b=n$ (full batch) then $H_t=\nabla^2 f(x_t)+\lambda\Id$ and $c_H=1/(\lambda_{\min}+\lambda)$. Under additional strong‑convexity the rate becomes $\mathcal{O}(T^{-1})$.
        \item If $\lambda\to0$ and $H_t$ is exact, the method reduces to the classic Newton method with quadratic convergence locally.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem establishes that SSN attains the same $\mathcal{O}(T^{-1/2})$ rate as stochastic gradient methods for non‑convex smooth objectives, while exploiting curvature information to reduce the constant factor. The analysis hinges on the smoothness of $f$, unbiasedness of the stochastic gradient, and a deterministic bound on the norm of the inverse stochastic Hessian. Extensions to adaptive batch sizes or variance‑reduced Hessian estimators follow the same template.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies Assumption~\ref{as:smooth}, then for any $x,y$,
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^2 .
\tag{2}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient (integrate the gradient bound). \end{proof}

\begin{lemma}[Bound on the inverse Hessian]\label{lem:inv-bound}
Under Assumption~\ref{as:hess-inv} we have for every $t$
\[
\|H_t^{-1}\|\le c_H\qquad\text{with }c_H:=1/\lambda .
\tag{3}
\]
\end{lemma}
\begin{proof}
Since $H_t=\frac1b\sum_{i\in S_t}\nabla^2\ell_i(x_t)+\lambda\Id\succeq\lambda\Id$, all eigenvalues of $H_t$ are at least $\lambda$, and therefore the eigenvalues of $H_t^{-1}$ are at most $1/\lambda$. The spectral norm equals the largest eigenvalue, giving the claim. \end{proof}

\begin{lemma}[Cross‑term bound]\label{lem:cross}
For any vectors $a,b\in\mathbb{R}^d$ and any $\gamma>0$,
\[
\big|\langle a,b\rangle\big|\le \frac{\gamma}{2}\|a\|^2+\frac{1}{2\gamma}\|b\|^2 .
\tag{4}
\]
\end{lemma}
\begin{proof}
Cauchy–Schwarz and the elementary inequality $2uv\le \gamma u^2+v^2/\gamma$. \end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout let $\mathcal{F}_t$ be the $\sigma$‑algebra generated by $\{x_0,S_0,\dots,S_{t-1}\}$.

\subsection*{Descent decomposition}
\begin{enumerate}[label={[\arabic*]}]
\item From Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_t-\alpha H_t^{-1}g_t$,
\[
f(x_{t+1})\le f(x_t)-\alpha\ip{\nabla f(x_t)}{H_t^{-1}g_t}
       +\frac{L\alpha^{2}}{2}\,\|H_t^{-1}g_t\|^{2}.
\tag{5}
\]
\item Take conditional expectation w.r.t.\ $\mathcal{F}_t$:
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]
\le f(x_t)-\alpha\,\E\!\big[\ip{\nabla f(x_t)}{H_t^{-1}g_t}\mid\mathcal{F}_t\big]
   +\frac{L\alpha^{2}}{2}\,\E\!\big[\|H_t^{-1}g_t\|^{2}\mid\mathcal{F}_t\big].
\tag{6}
\]

\item \textbf{Handling the inner product.}  
Write $g_t=\nabla f(x_t)+\xi_t$ where $\E[\xi_t\mid\mathcal{F}_t]=0$ and
$\E[\|\xi_t\|^{2}\mid\mathcal{F}_t]\le\sigma_g^{2}$ (Assumption~\ref{as:grad-var}).
Then
\[
\ip{\nabla f(x_t)}{H_t^{-1}g_t}
 =\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
   +\ip{\nabla f(x_t)}{H_t^{-1}\xi_t}.
\tag{7}
\]
Taking conditional expectation and using $\E[\xi_t\mid\mathcal{F}_t]=0$ gives
\[
\E\!\big[\ip{\nabla f(x_t)}{H_t^{-1}g_t}\mid\mathcal{F}_t\big]
 =\E\!\big[\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}\mid\mathcal{F}_t\big].
\tag{8}
\]
No independence between $H_t^{-1}$ and $\xi_t$ is required; the second term vanishes because the
conditional mean of $\xi_t$ is zero.

\item \textbf{Lower‑bounding the deterministic term.}  
Since $H_t\succeq\lambda\Id$, we have $H_t^{-1}\preceq c_H\Id$ with $c_H=1/\lambda$.
Consequently,
\[
\ip{\nabla f(x_t)}{H_t^{-1}\nabla f(x_t)}
 \ge \frac1{c_H}\,\|\nabla f(x_t)\|^{2}.
\tag{9}
\]
Taking expectation preserves the inequality, so
\[
\E\!\big[\ip{\nabla f(x_t)}{H_t^{-1}g_t}\mid\mathcal{F}_t\big]
 \ge \frac1{c_H}\,\|\nabla f(x_t)\|^{2}.
\tag{10}
\]

\item \textbf{Bounding the quadratic term.}  
Using $g_t=\nabla f(x_t)+\xi_t$ and Lemma~\ref{lem:cross} with $\gamma=1$,
\[
\|H_t^{-1}g_t\|^{2}
 =\|H_t^{-1}\nabla f(x_t)\|^{2}
   +2\ip{H_t^{-1}\nabla f(x_t)}{H_t^{-1}\xi_t}
   +\|H_t^{-1}\xi_t\|^{2}
\]
\[
\le \|H_t^{-1}\nabla f(x_t)\|^{2}
   +\|H_t^{-1}\nabla f(x_t)\|^{2}
   +\|H_t^{-1}\xi_t\|^{2}
 =2\|H_t^{-1}\nabla f(x_t)\|^{2}+\|H_t^{-1}\xi_t\|^{2}.
\tag{11}
\]
Applying Lemma~\ref{lem:inv-bound} ($\|H_t^{-1}\|\le c_H$) yields
\[
\|H_t^{-1}\nabla f(x_t)\|^{2}\le c_H^{2}\|\nabla f(x_t)\|^{2},
\qquad
\|H_t^{-1}\xi_t\|^{2}\le c_H^{2}\|\xi_t\|^{2}.
\]
Hence, after taking conditional expectation and using the variance bound,
\[
\E\!\big[\|H_t^{-1}g_t\|^{2}\mid\mathcal{F}_t\big]
\le 2c_H^{2}\|\nabla f(x_t)\|^{2}+c_H^{2}\sigma_g^{2}.
\tag{12}
\]

\item Substitute (10) and (12) into (6):
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]
\le f(x_t)-\frac{\alpha}{c_H}\|\nabla f(x_t)\|^{2}
   +\frac{L\alpha^{2}}{2}\big(2c_H^{2}\|\nabla f(x_t)\|^{2}+c_H^{2}\sigma_g^{2}\big).
\tag{13}
\]

\item Collect the $\|\nabla f(x_t)\|^{2}$ terms:
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]
\le f(x_t)-\Big(\frac{\alpha}{c_H}-L\alpha^{2}c_H^{2}\Big)\|\nabla f(x_t)\|^{2}
   +\frac{L\alpha^{2}c_H^{2}}{2}\sigma_g^{2}.
\tag{14}
\]

\item \textbf{Step‑size condition.}  
We require the coefficient of $\|\nabla f(x_t)\|^{2}$ to be at least $\frac{\alpha}{2c_H}$.
Thus we need
\[
\frac{\alpha}{c_H}-L\alpha^{2}c_H^{2}\ge\frac{\alpha}{2c_H}
\;\Longleftrightarrow\;
\alpha\le\frac{1}{L\,c_H^{3}} .
\tag{15}
\]
This is exactly the condition stated in \eqref{*}. Under \eqref{*},
\[
\E\!\big[f(x_{t+1})\mid\mathcal{F}_t\big]
\le f(x_t)-\frac{\alpha}{2c_H}\|\nabla f(x_t)\|^{2}
   +\frac{L\alpha^{2}c_H^{2}}{2}\sigma_g^{2}.
\tag{16}
\]

\item Take total expectation and sum over $t=0,\dots,T-1$:
\[
\E[f(x_T)]\le f(x_0)-\frac{\alpha}{2c_H}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
   +\frac{L\alpha^{2}c_H^{2}}{2}\sigma_g^{2}T .
\tag{17}
\]
Rearranging gives
\[
\frac1T\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le \frac{2c_H\big(f(x_0)-\E[f(x_T)]\big)}{\alpha T}
   +L\alpha c_H^{2}\sigma_g^{2}.
\tag{18}
\]
Since $f(x_T)\ge f^\star$, we obtain bound \eqref{1}.
\end{enumerate}

\subsection*{Rate derivation}
From \eqref{1},
\[
\frac1T\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le \frac{2c_H\Delta_0}{\alpha T}+L\alpha c_H^{2}\sigma_g^{2},
\qquad\Delta_0:=f(x_0)-f^\star .
\]
Choosing 
\[
\alpha = \sqrt{\frac{2\Delta_0}{L\,c_H^{3}\sigma_g^{2}}}\;T^{-1/2},
\]
which respects $\alpha\le 1/(L c_H^{3})$ for all sufficiently large $T$, yields
\[
\frac1T\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le 2\sqrt{2L\Delta_0\sigma_g^{2}}\;c_H^{3/2}\;T^{-1/2}.
\]
Hence $\min_{t<T}\E[\|\nabla f(x_t)\|^{2}]=\mathcal{O}(T^{-1/2})$.

\section*{Special Cases}
\begin{enumerate}[label={\arabic*.}]
\item \textbf{Full‑batch Newton.} If $b=n$ and $\lambda=0$, then $H_t=\nabla^2 f(x_t)$. Under strong convexity $\mu I\preceq\nabla^2 f(x)\preceq L I$, we have $c_H=1/\mu$ and the deterministic Newton step enjoys a linear rate locally; the stochastic bound reduces to the classic deterministic descent inequality.
\item \textbf{Large damping.} If $\lambda\gg L$, then $c_H\approx 1/\lambda$ and SSN behaves like SGD with effective step size $\alpha/\lambda$, recovering the standard $\mathcal{O}(T^{-1/2})$ rate.
\item \textbf{Increasing batch size.} If $b$ grows with $t$ such that $\sigma_g^{2}=O(1/b)$, the variance term in \eqref{1} vanishes, and a constant step size satisfying \eqref{*} yields an $\mathcal{O}(1/T)$ decay of the gradient norm.
\end{enumerate}
