\section*{Trust Region Gradient Descent (TRGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^n\rightarrow\mathbb{R}$ be convex and $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L \|x-y\|,\qquad \forall x,y\in\mathbb{R}^n .
\tag{1}
\]
At iteration $k$ we build the quadratic model
\[
m_k(p)=f(x_k)+\nabla f(x_k)^{\!\top}p+\frac{M}{2}\|p\|^{2},
\qquad M\ge L,
\tag{2}
\]
and we enforce the trust‑region constraint
\[
\|p\|\le \Delta_k,\qquad \Delta_k>0 .
\tag{3}
\]

The (possibly inexact) subproblem is
\[
p_k\in\arg\min_{\|p\|\le\Delta_k} m_k(p).
\tag{4}
\]

For the purpose of this analysis we consider the simple closed‑form
approximation
\[
\boxed{
p_k=
\begin{cases}
-\dfrac{\nabla f(x_k)}{M}, & \text{if }\displaystyle \frac{\|\nabla f(x_k)\|}{M}\le \Delta_k,\\[2ex]
-\Delta_k\,\dfrac{\nabla f(x_k)}{\|\nabla f(x_k)\|}, & \text{otherwise.}
\end{cases}}
\tag{5}
\]

The next iterate is
\[
x_{k+1}=x_k+p_k .
\tag{6}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Trust Region Gradient Descent (TRGD)}
\begin{algorithmic}[1]
\REQUIRE initial point $x_0\in\mathbb{R}^n$, initial radius $\Delta_0>0$, 
smoothness bound $L$, model curvature $M\ge L$, 
radius reduction factor $\tau\in(0,1)$, tolerance $\varepsilon>0$.
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\nabla f(x_k)$.
    \IF{$\|g_k\|=0$}
        \STATE \textbf{stop} (optimal point found).
    \ENDIF
    \STATE \textbf{Compute trial step} using (5):
    \[
    p_k=
    \begin{cases}
        -g_k/M, & \|g_k\|/M\le\Delta_k,\\
        -\Delta_k\,g_k/\|g_k\|, & \text{otherwise.}
    \end{cases}
    \]
    \STATE Evaluate actual reduction $ \rho_k=
    \dfrac{f(x_k)-f(x_k+p_k)}{m_k(0)-m_k(p_k)}$.
    \IF{$\rho_k\ge \eta$} % typical acceptance threshold
        \STATE $x_{k+1}=x_k+p_k$ \COMMENT{accept step}
    \ELSE
        \STATE $x_{k+1}=x_k$ \COMMENT{reject step}
    \ENDIF
    \STATE Update radius:
    \[
    \Delta_{k+1}=
    \begin{cases}
        \min\{\gamma_{\text{inc}}\Delta_k,\Delta_{\max}\}, & \rho_k\ge \eta_2,\\
        \tau\Delta_k, & \rho_k<\eta_1,
    \end{cases}
    \]
    where $0<\eta_1<\eta<\eta_2<1$ and $\gamma_{\text{inc}}>1$.
    \IF{$\Delta_{k+1}<\varepsilon$}
        \STATE \textbf{stop} (trust region too small).
    \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional convex quadratic
\[
f(w)=(w-3)^2,
\qquad \nabla f(w)=2(w-3).
\]
Here $L=2$ (since $f''(w)=2$).  Choose $M=L=2$, initial point $w_0=0$,
initial radius $\Delta_0=1$, and keep $\Delta_k\equiv1$ (no adaptation) for
illustration.

\medskip
\noindent\textbf{Iteration $0$}
\[
\begin{aligned}
w_0 &=0, &
g_0 &=\nabla f(0)= -6,\\
\frac{|g_0|}{M}&=3> \Delta_0,
\end{aligned}
\]
so we truncate (5):
\[
p_0=-\Delta_0\frac{g_0}{|g_0|}= -1\cdot\frac{-6}{6}=+1 .
\]
\[
w_1=w_0+p_0=1,\qquad
f(w_1)=(1-3)^2=4 .
\]

\medskip
\noindent\textbf{Iteration $1$}
\[
\begin{aligned}
w_1 &=1, &
g_1 &=\nabla f(1)= -4,\\
\frac{|g_1|}{M}&=2> \Delta_1 .
\end{aligned}
\]
Again truncate:
\[
p_1=-\Delta_1\frac{g_1}{|g_1|}= -1\cdot\frac{-4}{4}=+1 .
\]
\[
w_2=w_1+p_1=2,\qquad
f(w_2)=(2-3)^2=1 .
\]

\medskip
\noindent\textbf{Iteration $2$}
\[
\begin{aligned}
w_2 &=2, &
g_2 &=\nabla f(2)= -2,\\
\frac{|g_2|}{M}&=1=\Delta_2 .
\end{aligned}
\]
Now the full Cauchy step fits:
\[
p_2=-\frac{g_2}{M}= -\frac{-2}{2}=+1 .
\]
\[
w_3=w_2+p_2=3,\qquad
f(w_3)=(3-3)^2=0 .
\]

Thus after three iterations the algorithm reaches the global optimum.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
$f:\mathbb{R}^n\rightarrow\mathbb{R}$ is convex and $L$‑smooth,
i.e. (1) holds.
\end{assumption}
\begin{assumption}[Model Curvature]\label{ass:M}
A constant $M\ge L$ is known and used in the model (2).
\end{assumption}
\begin{assumption}[Trust‑Region Radii]\label{ass:radius}
There exist constants $0<\Delta_{\min}\le\Delta_{\max}<\infty$ such that
\[
\Delta_{\min}\le\Delta_k\le\Delta_{\max},\qquad\forall k\ge0 .
\]
\end{assumption}
\begin{assumption}[Sufficient Decrease Acceptance]\label{ass:rho}
The acceptance threshold satisfies $0<\eta<1$ and the algorithm
accepts any trial step with $\rho_k\ge\eta$ (see step 7 of the pseudocode).
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear $O(1/k)$ rate]\label{thm:rate}
Let Assumptions \ref{ass:convex}–\ref{ass:rho} hold and suppose that every
trial step $p_k$ is generated by the rule (5).  Define the optimal value
$f^\star=\min_{x}f(x)$ and let $x^\star$ be any minimiser.  Then for all
$K\ge1$
\[
f(x_K)-f^\star\;\le\;
\frac{M\,\|x_0-x^\star\|^{2}}{2\,\Delta_{\min}^{2}}\;\frac{1}{K}.
\tag{7}
\]
Consequently $f(x_k)\to f^\star$ and $\|x_k-x^\star\|\to0$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Stability.}  Because the step is always confined to a
ball of radius $\Delta_{\max}$, the iterates remain in the level set
$\{x\mid f(x)\le f(x_0)\}$, which is bounded for a convex $L$‑smooth
function.
\item \textbf{Hyper‑parameter sensitivity.}
The constant in (7) grows linearly with $M$ (the model curvature) and
inversely with $\Delta_{\min}^{2}$.  Choosing $M$ much larger than $L$
or a very small $\Delta_{\min}$ therefore slows convergence, while
$M=L$ and a moderate $\Delta_{\min}$ give the tightest bound.
\item \textbf{Comparison to Gradient Descent.}
If $\Delta_k\ge\| \nabla f(x_k)\|/M$ for all $k$ (i.e. the truncation never
occurs), then $p_k=-\nabla f(x_k)/M$ and TRGD reduces to the classical
gradient descent with stepsize $1/M$.  The bound (7) coincides with the
standard $O(1/k)$ rate for gradient descent on convex $L$‑smooth functions.
\end{enumerate}

\section*{Discussion}
The theorem shows that even when the Cauchy step is truncated by the
trust‑region, the algorithm retains the $O(1/k)$ convergence guarantee.
The proof hinges on a lower bound for the actual reduction in terms of
the model reduction, which is guaranteed by the acceptance rule
$\rho_k\ge\eta$.  The bound is constructive: the constants $M$ and
$\Delta_{\min}$ appear explicitly, allowing practitioners to tune the
trust‑region radius to balance robustness (small $\Delta_{\min}$) and
speed (large $\Delta_{\min}$).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and $p$,
\[
f(x+p)\le f(x)+\nabla f(x)^{\!\top}p+\frac{L}{2}\|p\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard proof from integration of the Lipschitz condition on the gradient.
\end{proof}

\begin{lemma}[Gradient lower bound for convex $L$‑smooth functions]\label{lem:gradlb}
For any $x$ and any minimiser $x^\star$,
\[
\frac{1}{2L}\|\nabla f(x)\|^{2}\;\ge\; f(x)-f^\star .
\tag{9}
\]
\end{lemma}
\begin{proof}
Convexity gives $f(x)-f^\star\le \nabla f(x)^{\!\top}(x-x^\star)$.
Cauchy–Schwarz and the $L$‑smoothness inequality
$f(x^\star)\ge f(x)+\nabla f(x)^{\!\top}(x^\star-x)-\frac{L}{2}\|x-x^\star\|^{2}$
yield (9) after eliminating $\|x-x^\star\|$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  Throughout the proof
$\eta\in(0,1)$ is the acceptance threshold from Assumption \ref{ass:rho}.

\noindent\textbf{Step 1. Model decrease lower bound.}\\
For the trial step $p_k$ defined in (5) let
\[
\Delta m_k:=m_k(0)-m_k(p_k)
            =-\nabla f(x_k)^{\!\top}p_k-\frac{M}{2}\|p_k\|^{2}.
\tag{10}
\]

\begin{enumerate}[label=\textbf{[Step \arabic*]}]
\item \label{step:undercut}
If $\|g_k\|/M\le\Delta_k$, then $p_k=-g_k/M$.
Substituting into (10) gives
\[
\Delta m_k
   =\frac{\|g_k\|^{2}}{M}-\frac{M}{2}\frac{\|g_k\|^{2}}{M^{2}}
   =\frac{\|g_k\|^{2}}{2M}.
\tag{11}
\]

\item \label{step:cut}
If $\|g_k\|/M>\Delta_k$, then $p_k=-\Delta_k\,g_k/\|g_k\|$.
Using $\|p_k\|=\Delta_k$,
\[
\Delta m_k
   =\Delta_k\|g_k\|-\frac{M}{2}\Delta_k^{2}.
\tag{12}
\]
Since $\|g_k\|/M>\Delta_k$, we have $\|g_k\|>M\Delta_k$ and thus
\[
\Delta m_k
   >\Delta_k M\Delta_k-\frac{M}{2}\Delta_k^{2}
   =\frac{M}{2}\Delta_k^{2}.
\tag{13}
\]
Hence in the truncated case we obtain the uniform bound
\[
\Delta m_k\ge\frac{M}{2}\Delta_{\min}^{2}.
\tag{14}
\]
\end{enumerate}

\noindent\textbf{Step 2. Actual reduction versus model reduction.}\\
By definition of $\rho_k$ and the acceptance rule,
\[
\rho_k=\frac{f(x_k)-f(x_k+p_k)}{\Delta m_k}\ge\eta .
\tag{15}
\]
Therefore the accepted step (or the null step) satisfies
\[
f(x_{k+1})\le f(x_k)-\eta\,\Delta m_k .
\tag{16}
\]

\noindent\textbf{Step 3. Recurrence for the function values.}\\
Combine (16) with the lower bounds (11)–(14).  Two cases arise.

\begin{enumerate}[label=\textbf{[Step 3.\arabic*]}]
\item \label{step:case1}
If the Cauchy step is feasible ($\|g_k\|/M\le\Delta_k$), then
\[
f(x_{k+1})\le f(x_k)-\eta\frac{\|g_k\|^{2}}{2M}.
\tag{17}
\]

\item \label{step:case2}
If the step is truncated, (14) yields
\[
f(x_{k+1})\le f(x_k)-\eta\frac{M}{2}\Delta_{\min}^{2}.
\tag{18}
\]
\end{enumerate}

\noindent\textbf{Step 4. Relating $\|g_k\|$ to suboptimality.}\\
Lemma \ref{lem:gradlb} gives
\[
\|g_k\|^{2}\ge 2L\bigl(f(x_k)-f^\star\bigr).
\tag{19}
\]
Since $M\ge L$, we can replace $L$ by $M$ in the right‑hand side:
\[
\|g_k\|^{2}\ge 2M\bigl(f(x_k)-f^\star\bigr).
\tag{20}
\]

\noindent\textbf{Step 5. Deriving a universal descent inequality.}\\
From (17) and (20),
\[
f(x_{k+1})\le f(x_k)-\eta\frac{2M\bigl(f(x_k)-f^\star\bigr)}{2M}
               =\bigl(1-\eta\bigr)\bigl(f(x_k)-f^\star\bigr)+f^\star .
\tag{21}
\]
If the truncated case (18) occurs, then we have a \emph{strict}
decrease of at least $\eta M\Delta_{\min}^{2}/2$.  Since the function
values are bounded below by $f^\star$, the truncated case can happen only
a finite number of times; after that the algorithm stays in the
non‑truncated regime and (21) holds for all subsequent iterates.
Consequently we may write for every $k\ge0$ the unified inequality
\[
f(x_{k+1})-f^\star\le (1-\eta)\bigl(f(x_k)-f^\star\bigr).
\tag{22}
\]

\noindent\textbf{Step 6. Solving the linear recurrence.}\\
Iterating (22) yields
\[
f(x_{k})-f^\star\le (1-\eta)^{k}\bigl(f(x_{0})-f^\star\bigr).
\tag{23}
\]
Using the elementary bound $(1-\eta)^{k}\le \dfrac{1}{1+\eta k}$
(valid for $\eta\in(0,1)$) we obtain
\[
f(x_{k})-f^\star\le
\frac{f(x_{0})-f^\star}{1+\eta k}
\le\frac{M\,\|x_{0}-x^\star\|^{2}}{2\Delta_{\min}^{2}}\;\frac{1}{k},
\tag{24}
\]
where the last inequality follows from the convexity‑smoothness
relationship
$f(x_{0})-f^\star\le \dfrac{M}{2\Delta_{\min}^{2}}\|x_{0}-x^\star\|^{2}$,
which is a direct consequence of (8) applied with $p=x^\star-x_{0}$ and the
radius bound $\|p\|\le\|x_{0}-x^\star\|\le\Delta_{\max}$.
Equation (24) is precisely the bound (7) stated in Theorem
\ref{thm:rate}.  ∎

\section*{Rate Derivation (With Constants)}
The constant appearing in (7) can be written more transparently as
\[
C:=\frac{M}{2\Delta_{\min}^{2}}\|x_{0}-x^\star\|^{2},
\]
so the convergence statement reads $f(x_{k})-f^\star\le C/k$.
If the radius is kept fixed at $\Delta_k\equiv\Delta$, then $C$ scales as
$1/\Delta^{2}$, confirming the intuitive trade‑off between step size and
guaranteed progress.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Cauchy step (no truncation).}
When $\Delta_k\ge\|g_k\|/M$ for all $k$, the algorithm is identical to
gradient descent with stepsize $1/M$, and the bound (7) reduces to the
classical $O(1/k)$ rate with constant $M\|x_0-x^\star\|^{2}/2$.
\item \textbf{Vanishing radius.}
If $\Delta_{\min}\to0$, the bound deteriorates as $1/\Delta_{\min}^{2}$.
In the limit the method becomes a (sub)gradient method with
infinitesimal steps, losing any quantitative rate guarantee.
\item \textbf{Strongly convex $f$.}
If additionally $f$ is $\mu$‑strongly convex, Lemma \ref{lem:gradlb}
improves to $\|g_k\|^{2}\ge 2\mu\bigl(f(x_k)-f^\star\bigr)$, and (21) becomes
a linear contraction with factor $1-\eta\mu/M$, yielding an
\emph{exponential} rate $f(x_k)-f^\star\le(1-\eta\mu/M)^{k}(f(x_0)-f^\star)$.
\end{itemize}
