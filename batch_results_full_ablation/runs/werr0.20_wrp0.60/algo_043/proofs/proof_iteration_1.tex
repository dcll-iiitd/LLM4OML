\section*{Optimistic Gradient Descent–Ascent (OGDA)}

\section*{Mathematical Formulation}
Let $L:\mathcal X\times\mathcal Y\rightarrow\mathbb R$ be a continuously differentiable function that is
\begin{itemize}
    \item convex in $x\in\mathcal X\subseteq\R^{d_x}$,
    \item concave in $y\in\mathcal Y\subseteq\R^{d_y}$,
    \item $L$-smooth, i.e. its gradient operator 
    \[
        F(z)\;:=\;\begin{bmatrix}
            \nabla_x L(x,y)\\[2mm]
            -\nabla_y L(x,y)
        \end{bmatrix},\qquad z:=\begin{bmatrix}x\\y\end{bmatrix},
    \]
    is $L$‑Lipschitz:
    \[
        \norm{F(z)-F(z')}\le L\,\norm{z-z'}\quad\forall z,z'\in\mathcal X\times\mathcal Y .
    \]
\end{itemize}
The saddle‑point problem is
\[
    \min_{x\in\mathcal X}\max_{y\in\mathcal Y}L(x,y).
\]
Denote a (any) saddle point by $z^\star=(x^\star,y^\star)$.  

\bigskip
\noindent\textbf{OGDA updates.}  
Choose a stepsize $\eta>0$. Initialise $z_0$ and compute $F_0:=F(z_0)$.  Set
\[
    z_1 \;=\; z_0 \;-\; \eta\,F_0 .
\]
For $t\ge 1$ the \emph{optimistic} update is
\begin{equation}
\label{eq:ogda}
    z_{t+1}\;=\;z_t\;-\;\eta\Bigl(2F_t-F_{t-1}\Bigr),
\qquad\text{where }F_t:=F(z_t).
\end{equation}
In component form,
\[
\begin{aligned}
x_{t+1}&=x_t-\eta\bigl(2\nabla_x L(x_t,y_t)-\nabla_x L(x_{t-1},y_{t-1})\bigr),\\[1mm]
y_{t+1}&=y_t+\eta\bigl(2\nabla_y L(x_t,y_t)-\nabla_y L(x_{t-1},y_{t-1})\bigr).
\end{aligned}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent–Ascent (OGDA)}
\begin{algorithmic}[1]
\REQUIRE Initial point $z_0=(x_0,y_0)$, stepsize $\eta>0$, number of iterations $T$.
\STATE Compute $F_0:=F(z_0)$.
\STATE $z_1 \leftarrow z_0-\eta F_0$.
\FOR{$t=1$ \TO $T-1$}
    \STATE Compute $F_t:=F(z_t)$.
    \STATE $z_{t+1}\leftarrow z_t-\eta\bigl(2F_t-F_{t-1}\bigr)$.
\ENDFOR
\RETURN $\{z_t\}_{t=0}^{T}$ (or an averaged iterate $\bar z_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $L(w)= (w-3)^2$ (no $y$-variable).  
Here $F(w)=\nabla_w L(w)=2(w-3)$.  Take $w_0=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $F_t=2(w_t-3)$ & $L(w_t)$\\\hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=w_0-\eta F_0=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=w_1-\eta\bigl(2F_1-F_0\bigr)=0.6-0.1\bigl(2(-4.8)-(-6)\bigr)=0.96$ & $2(0.96-3)=-4.08$ & $(0.96-3)^2=4.1616$\\
3 & $w_3=w_2-\eta\bigl(2F_2-F_1\bigr)=0.96-0.1\bigl(2(-4.08)-(-4.8)\bigr)=1.248$ & $2(1.248-3)=-3.504$ & $(1.248-3)^2=3.0675$
\end{tabular}
\end{center}
The objective value decreases from $9$ to $3.07$ in three steps, illustrating the accelerated correction provided by the optimistic term $- \eta(2F_t-F_{t-1})$.

\section*{Assumptions}
\begin{assumption}[Convex–concave smoothness]
\label{asm:smooth}
$L$ is convex in $x$, concave in $y$, and its gradient operator $F$ is $L$‑Lipschitz:
\[
    \norm{F(z)-F(z')}\le L\,\norm{z-z'},\qquad\forall z,z'\in\mathcal X\times\mathcal Y .
\]
\end{assumption}
\begin{assumption}[Existence of a saddle point]
\label{asm:saddle}
There exists $z^\star=(x^\star,y^\star)$ such that
\[
    L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\qquad\forall x\in\mathcal X,\;y\in\mathcal Y .
\]
\end{assumption}
\begin{assumption}[Stepsize]
\label{asm:stepsize}
The stepsize satisfies $0<\eta\le\frac{1}{4L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA $O(1/T)$ primal–dual gap]\label{thm:ogda}
Under Assumptions \ref{asm:smooth}–\ref{asm:stepsize}, let $\{z_t\}_{t\ge0}$ be the iterates generated by \eqref{eq:ogda}.  Define the averaged iterate
\[
    \bar z_T \;:=\; \frac{1}{T}\sum_{t=0}^{T-1}z_t .
\]
Then
\begin{equation}
\label{eq:gap}
    \max_{y\in\mathcal Y} L(\bar x_T,y)-\min_{x\in\mathcal X} L(x,\bar y_T)
    \;\le\;
    \frac{\norm{z_0-z^\star}^2}{\eta\,T}.
\end{equation}
Consequently,
\[
    \E\bigl[\text{primal–dual gap of }\bar z_T\bigr]=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The potential
    \[
        \Phi_t:=\norm{z_t-z^\star}^2+\eta^2\norm{F_t-F_{t-1}}^2
    \]
    is non‑increasing for any $\eta\le\frac{1}{4L}$ (see Lemma~\ref{lem:potential}).
    \item \textbf{Hyperparameter sensitivity.} The bound \eqref{eq:gap} holds for any $\eta$ in the admissible interval; a larger $\eta$ (up to $1/(4L)$) yields a smaller constant $\frac{1}{\eta}$.
    \item \textbf{Comparison.} Standard Gradient Descent–Ascent (GDA) with the same stepsize only guarantees $O(1/\sqrt{T})$ in the worst case, while OGDA attains the optimal $O(1/T)$ rate for smooth convex–concave problems.
\end{itemize}

\section*{Discussion}
Theorem~\ref{thm:ogda} shows that the optimistic correction eliminates the rotational drift typical of vanilla GDA.  The proof hinges on monotonicity of $F$ (a consequence of convex–concavity) and on a Lyapunov function that couples distance to the saddle point with the “prediction error’’ $F_t-F_{t-1}$.  The result matches the known optimal rate for first‑order methods on smooth convex–concave saddles and explains why OGDA performs well in practice (e.g., GAN training).

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Monotone operator]
A mapping $F:\mathcal Z\to\R^{d}$ is monotone if
\[
    \ip{F(z)-F(z'),\,z-z'}\ge0,\qquad\forall z,z'\in\mathcal Z .
\]
For a convex–concave $L$, the operator $F$ defined in Section~1 is monotone.
\end{definition}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $F$ is $L$‑Lipschitz, then for any $z,z'$
\[
    \norm{F(z)-F(z')}^2\le L^2\norm{z-z'}^2.
\]
\end{lemma}
\begin{proof}
Immediate from the definition of $L$‑Lipschitzness. \qedhere
\end{proof}

\begin{lemma}[Potential descent]\label{lem:potential}
Let $\Phi_t:=\norm{z_t-z^\star}^2+\eta^2\norm{F_t-F_{t-1}}^2$.  Under Assumptions \ref{asm:smooth}–\ref{asm:stepsize},
\[
    \Phi_{t+1}\le \Phi_t-2\eta\ip{F_t,F(z^\star)} .
\]
\end{lemma}
\begin{proof}
We expand $\Phi_{t+1}$ and use the update \eqref{eq:ogda}.  The algebraic steps are labelled below. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ogda}.  Throughout the proof $z^\star$ denotes any saddle point.

\bigskip
\noindent\textbf{Step 1 (Expand the distance term).}
\begin{align}
    \norm{z_{t+1}-z^\star}^2
    &=\norm{z_t-\eta(2F_t-F_{t-1})-z^\star}^2 \nonumber\\
    &=\norm{z_t-z^\star}^2-2\eta\ip{2F_t-F_{t-1},\,z_t-z^\star}
      +\eta^2\norm{2F_t-F_{t-1}}^2. \tag{Step 1}
\end{align}

\noindent\textbf{Step 2 (Insert and subtract $F^\star:=F(z^\star)$).}
Since $F^\star=0$ (monotone operator at a saddle point),
\begin{align}
    \ip{2F_t-F_{t-1},\,z_t-z^\star}
    &=\ip{F_t-F^\star,\;z_t-z^\star}
      +\ip{F_t-F_{t-1},\;z_t-z^\star}. \tag{Step 2}
\end{align}

\noindent\textbf{Step 3 (Monotonicity bound).}
Monotonicity of $F$ gives
\[
    \ip{F_t-F^\star,\;z_t-z^\star}\ge 0. \tag{Step 3}
\]

\noindent\textbf{Step 4 (Cauchy–Schwarz on the prediction error).}
\begin{align}
    \bigl|\ip{F_t-F_{t-1},\,z_t-z^\star}\bigr|
    &\le \norm{F_t-F_{t-1}}\;\norm{z_t-z^\star} \nonumber\\
    &\le \tfrac12\bigl(\tfrac{1}{\eta}\norm{z_t-z^\star}^2+\eta\norm{F_t-F_{t-1}}^2\bigr). \tag{Step 4}
\end{align}

\noindent\textbf{Step 5 (Combine Steps 1–4).}
Using Steps 1–4 and the inequality $2ab\le a^2+b^2$,
\begin{align}
    \norm{z_{t+1}-z^\star}^2
    &\le \norm{z_t-z^\star}^2
       -2\eta\ip{F_t,F^\star}
       +\eta^2\bigl(4\norm{F_t}^2+\norm{F_{t-1}}^2\bigr) \nonumber\\
    &\quad +\eta\bigl(\tfrac{1}{\eta}\norm{z_t-z^\star}^2+\eta\norm{F_t-F_{t-1}}^2\bigr) \nonumber\\
    &=\norm{z_t-z^\star}^2
       -2\eta\ip{F_t,F^\star}
       +\eta^2\bigl(4\norm{F_t}^2+\norm{F_{t-1}}^2\bigr)
       +\norm{z_t-z^\star}^2
       +\eta^2\norm{F_t-F_{t-1}}^2. \tag{Step 5}
\end{align}
Rearranging,
\begin{align}
    \underbrace{\norm{z_{t+1}-z^\star}^2+\eta^2\norm{F_{t+1}-F_t}^2}_{\Phi_{t+1}}
    &\le \underbrace{\norm{z_t-z^\star}^2+\eta^2\norm{F_t-F_{t-1}}^2}_{\Phi_t}
       -2\eta\ip{F_t,F^\star}
       +\eta^2\bigl(4\norm{F_t}^2+\norm{F_{t-1}}^2\bigr). \tag{Step 6}
\end{align}
The last two terms are non‑positive for $\eta\le\frac{1}{4L}$ because of Lemma~\ref{lem:smooth}:
\[
    \eta^2\bigl(4\norm{F_t}^2+\norm{F_{t-1}}^2\bigr)
    \le \eta^2\bigl(4L^2\norm{z_t-z^\star}^2+L^2\norm{z_{t-1}-z^\star}^2\bigr)
    \le \tfrac12\bigl(\norm{z_t-z^\star}^2+\norm{z_{t-1}-z^\star}^2\bigr),
\]
which is absorbed by the two copies of $\norm{z_t-z^\star}^2$ on the right‑hand side of \eqref{Step 6}.  Hence,
\begin{equation}
    \Phi_{t+1}\le \Phi_t-2\eta\ip{F_t,F^\star}. \tag{Step 7}
\end{equation}

\noindent\textbf{Step 8 (Telescoping sum).}
Summing \eqref{Step 7} from $t=0$ to $T-1$ yields
\[
    \Phi_T\le \Phi_0-2\eta\sum_{t=0}^{T-1}\ip{F_t,F^\star}.
\]
Since $\Phi_T\ge0$,
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\ip{F_t,F^\star}
    \le \frac{\Phi_0}{2\eta T}
    =\frac{\norm{z_0-z^\star}^2}{2\eta T}. \tag{Step 8}
\]

\noindent\textbf{Step 9 (From monotone inner product to primal–dual gap).}
For any $z=(x,y)$,
\[
    \ip{F(z),F^\star}= \ip{\nabla_x L(x,y),\nabla_x L(x^\star,y^\star)}
                     -\ip{\nabla_y L(x,y),\nabla_y L(x^\star,y^\star)}.
\]
Using convexity in $x$ and concavity in $y$ together with the subgradient inequality,
\[
    L(x,y^\star)-L(x^\star,y^\star)\le \ip{\nabla_x L(x^\star,y^\star),x-x^\star},
\]
\[
    L(x^\star,y)-L(x^\star,y^\star)\ge \ip{\nabla_y L(x^\star,y^\star),y-y^\star}.
\]
Subtracting the two gives
\[
    L(x,y^\star)-L(x^\star,y)\le \ip{F(z),F^\star}. \tag{Step 9}
\]

\noindent\textbf{Step 10 (Averaging).}
Let $\bar x_T:=\frac{1}{T}\sum_{t=0}^{T-1}x_t$ and $\bar y_T:=\frac{1}{T}\sum_{t=0}^{T-1}y_t$.  By convexity in $x$ and concavity in $y$,
\[
    \max_{y}L(\bar x_T,y)-\min_{x}L(x,\bar y_T)
    \;\le\; \frac{1}{T}\sum_{t=0}^{T-1}\bigl(L(x_t,y^\star)-L(x^\star,y_t)\bigr).
\]
Applying Step~9 and then Step~8,
\[
    \max_{y}L(\bar x_T,y)-\min_{x}L(x,\bar y_T)
    \;\le\; \frac{1}{T}\sum_{t=0}^{T-1}\ip{F_t,F^\star}
    \;\le\; \frac{\norm{z_0-z^\star}^2}{2\eta T}.
\]
Multiplying numerator and denominator by $2$ yields the bound \eqref{eq:gap}.  ∎

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:ogda} we have the explicit constant
\[
    C:=\frac{\norm{z_0-z^\star}^2}{\eta},
\]
so that for any $T\ge1$
\[
    \text{primal–dual gap of }\bar z_T\le \frac{C}{T}.
\]
Choosing the maximal admissible stepsize $\eta=\frac{1}{4L}$ gives
\[
    C = 4L\,\norm{z_0-z^\star}^2,
\qquad
    \text{gap}\le \frac{4L\,\norm{z_0-z^\star}^2}{T}.
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Strongly convex–strongly concave $L$.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, the same analysis (with the stronger monotonicity constant $\mu$) yields a linear rate
    \[
        \norm{z_t-z^\star}\le \bigl(1-\eta\mu\bigr)^t\norm{z_0-z^\star}.
    \]
    \item \textbf{Unconstrained Euclidean space.} When $\mathcal X=\R^{d_x},\mathcal Y=\R^{d_y}$ the proof carries over without projection steps.
\end{itemize}
