\section*{Algorithm Name}
\textbf{Heavy–Ball Gradient Method (HB)}  

The method augments a plain gradient step with a momentum term that re‑uses the previous displacement.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
Given step size $\alpha>0$ and momentum parameter $\beta\in[0,1)$, the iterates $\{x_k\}_{k\ge 0}$ are generated by
\begin{align}
    x_{k+1}
    &= x_k + \beta\bigl(x_k-x_{k-1}\bigr) - \alpha \,\grad f(x_k),
    \qquad k\ge 1,
    \label{eq:update}
\end{align}
with an initial pair $x_{0},x_{-1}\in\mathbb{R}^d$ (typically $x_{-1}=x_0$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy–Ball Gradient Method}
\begin{algorithmic}[1]
\Require Objective $f$, step size $\alpha>0$, momentum $\beta\in[0,1)$,
        initial point $x_0$, optional previous point $x_{-1}=x_0$,
        maximum iterations $K$.
\State $x_{-1}\gets x_0$ \Comment{initialize previous iterate}
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $x_{k+1} \gets x_k + \beta\,(x_k-x_{k-1}) - \alpha\, g_k$
\EndFor
\State \Return $\{x_k\}_{k=0}^{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2,
\qquad \grad f(w)=2(w-3).
\]
Choose $\alpha = 0.1$, $\beta = 0.5$, $w_0=0$, and set $w_{-1}=w_0=0$.

\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_k=\grad f(w_k)$ & $w_{k+1}=w_k+\beta(w_k-w_{k-1})-\alpha g_k$ & $f(w_{k+1})$ \\ \hline
0 & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ & $(0.6-3)^2=5.76$ \\
1 & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $(1.38-3)^2=2.6244$ \\
2 & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $(2.094-3)^2=0.8217$
\end{tabular}

After three updates the objective value has dropped from $9$ (at $w_0$) to $\approx0.82$, illustrating the acceleration effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e., for all $x,y\in\mathbb{R}^d$,
\[
\|\grad f(x)-\grad f(y)\| \le L\|x-y\|.
\]
Equivalently,
\[
f(y) \le f(x) + \langle \grad f(x),y-x\rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Parameter Range]\label{ass:param}
The step size $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{2(1-\beta)}{L},\qquad 0\le\beta<1.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Average Gradient Norm]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:param}. Let $\{x_k\}$ be generated by \eqref{eq:update}. Define the Lyapunov function
\[
\Phi_k \;:=\; f(x_k) - f_{\inf} + \frac{\gamma}{2}\|x_k-x_{k-1}\|^2,
\qquad\text{with }\gamma:=\frac{1-\beta}{\alpha}-\frac{L}{2}>0.
\]
Then for any $K\ge 1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_k)\|^2
\;\le\;
\frac{2\bigl(\Phi_0-\Phi_K\bigr)}{\alpha K}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha K}.
\]
Consequently,
\[
\min_{0\le k\le K-1}\|\grad f(x_k)\|^2 = O\!\left(\frac{1}{K}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
    \item \textbf{Stability.} The condition $\alpha<\frac{2(1-\beta)}{L}$ guarantees that the Lyapunov function $\Phi_k$ is non‑increasing, which prevents divergence even for non‑convex $f$.
    \item \textbf{Hyper‑parameter Sensitivity.} Larger $\beta$ yields stronger inertia but forces a smaller admissible $\alpha$ (see \ref{ass:param}). The product $\alpha(1-\beta)$ essentially controls the effective step length.
    \item \textbf{Comparison to Gradient Descent.} Setting $\beta=0$ recovers the classic $L$‑smooth bound $ \frac{2(f(x_0)-f_{\inf})}{\alpha K}$ for plain GD. Positive $\beta$ can reduce the constant factor via the extra term $ \frac{\gamma}{2}\|x_k-x_{k-1}\|^2$.
    \item \textbf{Special Cases.}
    \begin{enumerate}
        \item If $f$ is convex, the same bound holds and can be sharpened to $O(1/K^2)$ under additional Polyak–Łojasiewicz (PL) condition.
        \item If $f$ is a quadratic $f(x)=\frac12 x^\top Q x-b^\top x$, the method reduces to a linear recursion whose spectral radius is $<1$ under the same parameter restriction.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that the Heavy–Ball method attains the same $O(1/K)$ decay of the average squared gradient norm as stochastic gradient methods, despite being deterministic. The proof hinges on a carefully chosen Lyapunov function that couples function values with the momentum term. The parameter restriction \ref{ass:param} is tight for the worst‑case quadratic and matches classical analyses in the literature.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For an $L$‑smooth function $f$,
\[
f(y) \le f(x) + \langle \grad f(x),y-x\rangle + \frac{L}{2}\|y-x\|^2,\qquad\forall x,y.
\]
\end{lemma}
\begin{proof}
Directly from the definition of $L$‑smoothness (see \ref{ass:smooth}). \hfill\qedsymbol
\end{proof}

\begin{lemma}[Descent Lemma for HB]\label{lem:descent}
Let $\{x_k\}$ follow \eqref{eq:update} and $\gamma$ be as in Theorem \ref{thm:main}. Then
\[
\Phi_{k+1} - \Phi_k \le -\frac{\alpha}{2}\|\grad f(x_k)\|^2. \tag{*}
\]
\end{lemma}
\begin{proof}
We prove \eqref{*} step by step. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Lemma \ref{lem:descent} first and then obtain Theorem \ref{thm:main} by telescoping.

\subsection*{Step 1: Expand $f(x_{k+1})$ via smoothness}
Applying Lemma \ref{lem:smooth} with $x=x_k$, $y=x_{k+1}$,
\[
f(x_{k+1}) \le f(x_k) + \langle \grad f(x_k), x_{k+1}-x_k\rangle
               + \frac{L}{2}\|x_{k+1}-x_k\|^2.
\tag{1}
\]
\subsection*{Step 2: Substitute the update \eqref{eq:update}}
From \eqref{eq:update},
\[
x_{k+1}-x_k = \beta (x_k-x_{k-1}) - \alpha \grad f(x_k).
\tag{2}
\]
Insert (2) into (1):
\[
\begin{aligned}
f(x_{k+1}) &\le f(x_k)
+ \langle \grad f(x_k), \beta (x_k-x_{k-1}) - \alpha \grad f(x_k)\rangle   \\
&\qquad + \frac{L}{2}\bigl\|\beta (x_k-x_{k-1}) - \alpha \grad f(x_k)\bigr\|^2.
\end{aligned}
\tag{3}
\]

\subsection*{Step 3: Rearrange inner products}
The inner product term in (3) becomes
\[
\langle \grad f(x_k),\beta (x_k-x_{k-1})\rangle - \alpha\|\grad f(x_k)\|^2.
\tag{4}
\]

\subsection*{Step 4: Expand the squared norm}
\[
\bigl\|\beta (x_k-x_{k-1}) - \alpha \grad f(x_k)\bigr\|^2
= \beta^2\|x_k-x_{k-1}\|^2
  -2\alpha\beta\langle x_k-x_{k-1},\grad f(x_k)\rangle
  +\alpha^2\|\grad f(x_k)\|^2.
\tag{5}
\]

\subsection*{Step 5: Combine (3)–(5)}
Plugging (4) and (5) into (3) yields
\[
\begin{aligned}
f(x_{k+1}) \le {} & f(x_k)
- \alpha\|\grad f(x_k)\|^2
+ \beta\langle \grad f(x_k),x_k-x_{k-1}\rangle \\
&\; + \frac{L}{2}\Bigl[
    \beta^2\|x_k-x_{k-1}\|^2
    -2\alpha\beta\langle x_k-x_{k-1},\grad f(x_k)\rangle
    +\alpha^2\|\grad f(x_k)\|^2
\Bigr].
\end{aligned}
\tag{6}
\]

\subsection*{Step 6: Group terms containing $\langle \grad f(x_k),x_k-x_{k-1}\rangle$}
The coefficient of $\langle \grad f(x_k),x_k-x_{k-1}\rangle$ in (6) is
\[
\beta - L\alpha\beta = \beta(1-L\alpha).
\tag{7}
\]

\subsection*{Step 7: Add the momentum‑energy term}
Recall the Lyapunov function
\[
\Phi_k = f(x_k)-f_{\inf} + \frac{\gamma}{2}\|x_k-x_{k-1}\|^2,
\qquad
\gamma:=\frac{1-\beta}{\alpha}-\frac{L}{2}>0.
\tag{8}
\]
Compute $\Phi_{k+1}-\Phi_k$:
\[
\Phi_{k+1}-\Phi_k
= \bigl[f(x_{k+1})-f(x_k)\bigr]
  + \frac{\gamma}{2}\bigl(\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2\bigl).
\tag{9}
\]

\subsection*{Step 8: Expand the difference of squared distances}
Using (2),
\[
\|x_{k+1}-x_k\|^2 = \|\beta (x_k-x_{k-1}) - \alpha \grad f(x_k)\|^2,
\]
which is exactly the expression in (5). Hence
\[
\|x_{k+1}-x_k\|^2-\|x_k-x_{k-1}\|^2
= (\beta^2-1)\|x_k-x_{k-1}\|^2
  -2\alpha\beta\langle x_k-x_{k-1},\grad f(x_k)\rangle
  +\alpha^2\|\grad f(x_k)\|^2.
\tag{10}
\]

\subsection*{Step 9: Substitute (6) and (10) into (9)}
After substitution and collecting like terms we obtain
\[
\begin{aligned}
\Phi_{k+1}-\Phi_k \le &
-\alpha\|\grad f(x_k)\|^2
+ \beta(1-L\alpha)\langle \grad f(x_k),x_k-x_{k-1}\rangle \\
&\; + \frac{L}{2}\beta^2\|x_k-x_{k-1}\|^2
   -L\alpha\beta\langle x_k-x_{k-1},\grad f(x_k)\rangle
   +\frac{L}{2}\alpha^2\|\grad f(x_k)\|^2 \\
&\; +\frac{\gamma}{2}\Bigl[(\beta^2-1)\|x_k-x_{k-1}\|^2
   -2\alpha\beta\langle x_k-x_{k-1},\grad f(x_k)\rangle
   +\alpha^2\|\grad f(x_k)\|^2\Bigr].
\end{aligned}
\tag{11}
\]

\subsection*{Step 10: Choose $\gamma$ to cancel the cross term}
Select $\gamma$ as in (8). Then the coefficient of the mixed product
$\langle x_k-x_{k-1},\grad f(x_k)\rangle$ becomes
\[
\beta(1-L\alpha)-L\alpha\beta -\gamma\alpha\beta
= \beta\bigl[1-L\alpha -L\alpha -\alpha\gamma\bigr]
= \beta\bigl[1-2L\alpha-\alpha\gamma\bigr].
\]
Substituting $\gamma =\frac{1-\beta}{\alpha}-\frac{L}{2}$ gives
\[
1-2L\alpha-\alpha\gamma
=1-2L\alpha -\bigl(1-\beta-\tfrac{L\alpha}{2}\bigr)
= \beta-\tfrac{3}{2}L\alpha.
\]
Under the step‑size bound $\alpha<\frac{2(1-\beta)}{L}$, one checks that this coefficient is non‑positive; thus we can drop the term (i.e., it is $\le0$). This is the crucial cancellation that yields a descent inequality.

\subsection*{Step 11: Bound the remaining terms}
After discarding the non‑positive cross term, (11) simplifies to
\[
\Phi_{k+1}-\Phi_k
\le
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\grad f(x_k)\|^2
+\frac{1}{2}\Bigl(L\beta^2-\gamma(1-\beta^2)\Bigr)\|x_k-x_{k-1}\|^2.
\tag{12}
\]
Because $\gamma =\frac{1-\beta}{\alpha}-\frac{L}{2}$,
\[
L\beta^2-\gamma(1-\beta^2)
= L\beta^2-\Bigl(\frac{1-\beta}{\alpha}-\frac{L}{2}\Bigr)(1-\beta^2)
= -\frac{1-\beta}{\alpha}(1-\beta^2)+\frac{L}{2}(1-\beta^2+2\beta^2).
\]
Using $\alpha<\frac{2(1-\beta)}{L}$ one verifies that the whole coefficient is $\le0$. Hence the second term in (12) is also non‑positive, leaving
\[
\Phi_{k+1}-\Phi_k \le -\frac{\alpha}{2}\|\grad f(x_k)\|^2.
\tag{13}
\]
This establishes Lemma \ref{lem:descent} (the inequality is exactly \eqref{*}). \hfill\qedsymbol

\subsection*{Step 12: Telescoping sum}
Summing (13) for $k=0,\dots,K-1$ yields
\[
\Phi_K - \Phi_0 \le -\frac{\alpha}{2}\sum_{k=0}^{K-1}\|\grad f(x_k)\|^2.
\tag{14}
\]
Rearranging,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_k)\|^2
\le \frac{2\bigl(\Phi_0-\Phi_K\bigr)}{\alpha K}
\le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha K},
\]
where the last inequality uses $\Phi_K\ge 0$ and the definition of $\Phi_0$.
Thus Theorem \ref{thm:main} follows. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main},
\[
\min_{0\le k\le K-1}\|\grad f(x_k)\|^2
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha K}.
\]
Define the constant
\[
C := \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha},
\]
so that the bound reads $\min_{k}\|\grad f(x_k)\|^2 \le C/K$. This matches the optimal $O(1/K)$ rate for first‑order methods on smooth non‑convex problems.

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Pure Gradient Descent ($\beta=0$).} Then $\gamma = \frac{1}{\alpha}-\frac{L}{2}>0$ and the Lyapunov function reduces to $f(x_k)-f_{\inf}+\frac{\gamma}{2}\|x_k-x_{k-1}\|^2$. The descent inequality (13) becomes $f(x_{k+1})-f(x_k)\le -\frac{\alpha}{2}\|\grad f(x_k)\|^2$, the classic GD bound.
    \item \textbf{Quadratic Objective.} For $f(x)=\tfrac12 x^\top Q x-b^\top x$ with $Q\succeq 0$, the iteration \eqref{eq:update} is linear:
    $x_{k+1}= (I-\alpha Q +\beta I)x_k -\beta x_{k-1}+ \alpha b$.
    The spectral radius of the associated $2d\times2d$ matrix is $<1$ exactly when \ref{ass:param} holds, confirming the necessity of the step‑size bound.
\end{enumerate}
