\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\begin{center}
\textbf{Trust‑Region Method with Approximate Subproblem Solution (TR‑A)}
\end{center}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\R^{n}\rightarrow\R$ be twice continuously differentiable and bounded below.  
At iteration $k$ we have a current iterate $x_k\in\R^{n}$ and a trust‑region radius $\Delta_k>0$.  

\medskip
\noindent\textbf{Quadratic model.}
\[
m_k(p)\;:=\;f(x_k)+g_k^{\top}p+\tfrac12 p^{\top}B_k p,
\qquad 
g_k:=\nabla f(x_k),\;
B_k\approx \nabla^{2}f(x_k).
\]

\noindent\textbf{Subproblem (TR).}
\[
\min_{p\in\R^{n}}\; m_k(p)\quad\text{s.t.}\quad \norm{p}\le \Delta_k .
\tag{TR$_k$}
\]

\noindent\textbf{Approximate solution.}
We do \emph{not} require the exact minimiser $p_k^{\star}$ of (TR$_k$).  
Instead we accept any $p_k$ satisfying the \emph{Cauchy‑type sufficient decrease} condition
\[
m_k(p_k)\;\le\; m_k(0)-\eta\,\norm{g_k}\min\!\Big\{\Delta_k,\,
\frac{\norm{g_k}}{\norm{B_k}}\Big\},
\qquad \eta\in(0,1).
\tag{A1}
\]

\noindent\textbf{Actual vs. predicted reduction.}
\[
\rho_k \;:=\;
\frac{f(x_k)-f(x_k+p_k)}{m_k(0)-m_k(p_k)} .
\tag{A2}
\]

\noindent\textbf{Radius update.} Choose constants
\[
0<\eta_1<\eta_2<1,\qquad
\gamma_{\text{inc}}>1,\qquad
0<\gamma_{\text{dec}}<1 .
\]
Then
\[
\Delta_{k+1}= 
\begin{cases}
\min\{\gamma_{\text{inc}}\Delta_k,\;\Delta_{\max}\}, & \rho_k\ge \eta_2,\\[4pt]
\Delta_k, & \eta_1\le\rho_k<\eta_2,\\[4pt]
\gamma_{\text{dec}}\Delta_k, & \rho_k<\eta_1 .
\end{cases}
\tag{A3}
\]

\noindent\textbf{Iterate update.}
\[
x_{k+1}= 
\begin{cases}
x_k+p_k , & \rho_k\ge \eta_1,\\[4pt]
x_k , & \rho_k<\eta_1 .
\end{cases}
\tag{A4}
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{enumerate}[label=Step \arabic*:,leftmargin=*,itemsep=0.3em]
\item \textbf{Input:} initial point $x_0$, initial radius $\Delta_0$, parameters 
$\eta,\eta_1,\eta_2,\gamma_{\text{inc}},\gamma_{\text{dec}},\Delta_{\max}$.
\item \textbf{For} $k=0,1,2,\dots$ \textbf{do}
\begin{enumerate}[label= \arabic*.,leftmargin=*,itemsep=0.2em]
\item Compute gradient $g_k=\nabla f(x_k)$ and (optional) Hessian approximation $B_k$.
\item Find $p_k$ satisfying the sufficient decrease condition (A1).  
(Any practical method – e.g. truncated CG, dog‑leg – can be used.)
\item Compute actual reduction $ared_k=f(x_k)-f(x_k+p_k)$ and predicted reduction $pred_k=m_k(0)-m_k(p_k)$.
\item Compute ratio $\rho_k$ via (A2).
\item \textbf{If} $\rho_k\ge \eta_1$ \textbf{then} set $x_{k+1}=x_k+p_k$ \textbf{else} $x_{k+1}=x_k$.
\item Update the trust‑region radius according to (A3).
\item \textbf{If} stopping criterion (e.g. $\norm{g_k}\le \varepsilon$ or $k\ge K_{\max}$) \textbf{then break.}
\end{enumerate}
\item \textbf{Output:} last iterate $x_{k}$.
\end{enumerate}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2 .
\]
Hence $g(w)=\nabla f(w)=2(w-3)$ and $B(w)=\nabla^2 f(w)=2$ (constant).  
Choose
\[
x_0=0,\qquad \Delta_0=1,\qquad 
\eta=0.1,\;\eta_1=0.25,\;\eta_2=0.75,\;
\gamma_{\text{inc}}=2,\;\gamma_{\text{dec}}=0.5,\;
\Delta_{\max}=10 .
\]

Because $f$ is quadratic, the exact minimiser of the model is $p_k^{\star}=-(g_k/B_k)$.  
We will use the Cauchy point $p_k=-\Delta_k\,\operatorname{sign}(g_k)$, which satisfies (A1) for any $\eta\in(0,1)$.

\medskip
\noindent\textbf{Iteration 0}
\begin{itemize}
\item $g_0=2(0-3)=-6$, $B_0=2$.
\item Cauchy step: $p_0=-\Delta_0\operatorname{sign}(g_0)= -1\cdot(-1)=1$.
\item $f(x_0)=9$, $f(x_0+p_0)=f(1)=(1-3)^2=4$.
\item $ared_0=9-4=5$, $pred_0=m_0(0)-m_0(p_0)=0-[-6\cdot1+0.5\cdot2\cdot1^2]=5$.
\item $\rho_0=5/5=1\ge\eta_1$ $\Rightarrow$ accept step: $x_1=1$.
\item Since $\rho_0\ge\eta_2$, enlarge radius: $\Delta_1=\min\{2\cdot1,10\}=2$.
\end{itemize}

\noindent\textbf{Iteration 1}
\begin{itemize}
\item $g_1=2(1-3)=-4$, $B_1=2$.
\item Cauchy step: $p_1=-\Delta_1\operatorname{sign}(g_1)=-2\cdot(-1)=2$.
\item $f(x_1)=4$, $f(x_1+p_1)=f(3)=0$.
\item $ared_1=4$, $pred_1= -(-4)2+0.5\cdot2\cdot2^2=8-4=4$.
\item $\rho_1=1\ge\eta_1$ $\Rightarrow$ $x_2=3$ (optimal point reached).
\item $\rho_1\ge\eta_2$ $\Rightarrow$ $\Delta_2=\min\{2\cdot2,10\}=4$.
\end{itemize}

\noindent\textbf{Iteration 2}
\begin{itemize}
\item $g_2=2(3-3)=0$, $B_2=2$.
\item Cauchy step: $p_2=0$ (since $g_2=0$).
\item $f(x_2)=0$, no reduction, $\rho_2$ undefined but we treat $\rho_2=1$.
\item Algorithm terminates because $\norm{g_2}=0\le\varepsilon$.
\end{itemize}

The sequence of iterates is $0\rightarrow 1\rightarrow 3$, and the objective values are $9\rightarrow4\rightarrow0$, confirming that TR‑A converges in two successful steps for this simple problem.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f\in C^{2}(\R^{n})$ and there exists $L>0$ such that for all $x,y\in\R^{n}$
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently, the Hessian is $L$‑Lipschitz:
\[
\|\nabla^{2}f(x)-\nabla^{2}f(y)\|\le L\|x-y\|.
\]
\end{assumption}

\begin{assumption}[Bounded Below]\label{as:bound}
There exists $f_{\inf}>-\infty$ with $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}

\begin{assumption}[Model Accuracy]\label{as:model}
The matrix $B_k$ satisfies
\[
\|B_k-\nabla^{2}f(x_k)\|\le \kappa_B,\qquad \kappa_B\ge0.
\]
\end{assumption}

\begin{assumption}[Parameter Choice]\label{as:param}
The algorithmic constants obey
\[
0<\eta_1<\eta_2<1,\qquad
0<\eta<1,\qquad
\gamma_{\text{inc}}>1,\qquad
0<\gamma_{\text{dec}}<1,\qquad
\Delta_{\max}>0.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Global Convergence and Rate]\label{thm:global}
Assume \ref{as:smooth}–\ref{as:param} hold and the subproblem solution $p_k$ satisfies (A1).  
Define the set of \emph{successful} iterations
\[
\mathcal{S}:=\{k\ge0\mid \rho_k\ge\eta_1\}.
\]
Then:
\begin{enumerate}[label=(\roman*),leftmargin=*,itemsep=0.3em]
\item \textbf{Stationarity.} Every limit point $\bar x$ of $\{x_k\}$ satisfies $\nabla f(\bar x)=0$.
\item \textbf{Complexity.} For any tolerance $\varepsilon>0$, let
\[
T_{\varepsilon}:=\min\{k\ge0\mid \norm{\nabla f(x_k)}\le\varepsilon\}.
\]
If the Cauchy‑type condition (A1) is used, then
\[
T_{\varepsilon}\;\le\; \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta\,\varepsilon^{2}}\,\Bigl(L+\kappa_B\Bigr)
\;=\;O\!\left(\frac{1}{\varepsilon^{2}}\right).
\tag{T1}
\]
Consequently,
\[
\min_{0\le k\le T_{\varepsilon}} \norm{\nabla f(x_k)}\;\le\;O\!\left(\frac{1}{\sqrt{T_{\varepsilon}}}\right).
\]
\end{enumerate}
\end{theorem}
The theorem mirrors the classic $O(1/\varepsilon^{2})$ bound for trust‑region methods with approximate subproblem solutions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability of the radius.} The update rule (A3) guarantees $\Delta_k$ stays in $[\Delta_{\min},\Delta_{\max}]$ where $\Delta_{\min}>0$ can be chosen arbitrarily small; this prevents stagnation.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\eta_1$ makes the acceptance test stricter, possibly leading to more radius reductions but tighter steps. The complexity bound (T1) depends only on $\eta$ (the constant in (A1)), not on $\eta_1,\eta_2$, showing robustness.
\item \textbf{Comparison with baseline methods.}
\begin{enumerate}[label=\alph*.,leftmargin=*,itemsep=0.2em]
\item \emph{Gradient descent} with fixed step size obtains $O(1/\varepsilon^{2})$ only under strong convexity; TR‑A works without convexity.
\item \emph{Stochastic methods} such as SGD achieve $O(1/\sqrt{T})$ in expectation but require diminishing steps and variance control. TR‑A provides deterministic guarantees.
\item \emph{Exact trust‑region} (solving (TR$_k$) exactly) yields the same $O(1/\varepsilon^{2})$ rate but at higher per‑iteration cost. The sufficient‑decrease condition (A1) allows cheap subproblem solvers while preserving the rate.
\end{enumerate}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The analysis shows that even a very coarse approximation of the trust‑region subproblem—merely a Cauchy step—suffices to guarantee convergence to first‑order stationary points at the optimal $O(1/\varepsilon^{2})$ iteration complexity.  The proof hinges on two elementary facts:
\begin{enumerate}[label=\roman*),leftmargin=*,itemsep=0.2em]
\item The model decrease is lower‑bounded by a constant times $\min\{\Delta_k,\|g_k\|/\|B_k\|\}\|g_k\|$ (Cauchy bound).
\item Lipschitz continuity of the gradient translates model decrease into actual function decrease up to a factor $1-\rho_k$.
\end{enumerate}
Thus, the algorithm adapts the radius automatically and never gets stuck unless the gradient is already small.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Cauchy decrease]\label{lem:cauchy}
For any $k$, the Cauchy point $p_k^{C}:=-\tau_k g_k$ with
\[
\tau_k:=\min\!\Bigl\{\frac{\norm{g_k}}{\norm{B_k} \norm{g_k}},\;\frac{\Delta_k}{\norm{g_k}}\Bigr\}
\]
satisfies
\[
m_k(0)-m_k(p_k^{C}) \;\ge\; \frac{\eta}{2}\,\norm{g_k}\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{\norm{B_k}}\Bigr\},
\tag{L1}
\]
where $\eta\in(0,1)$ is the constant appearing in (A1).
\end{lemma}
\begin{proof}
\[
m_k(p)=g_k^{\top}p+\tfrac12 p^{\top}B_k p .
\]
Insert $p=-\tau g$:
\[
m_k(p)=-\tau\norm{g_k}^{2}+\tfrac12\tau^{2} g_k^{\top}B_k g_k
\le -\tau\norm{g_k}^{2}+\tfrac12\tau^{2}\norm{B_k}\norm{g_k}^{2}.
\]
Choosing $\tau$ as the minimiser of the right‑hand side yields
\[
\tau^{\star}= \frac{\norm{g_k}}{\norm{B_k}} .
\]
If $\tau^{\star}\le\Delta_k/\norm{g_k}$ we set $\tau=\tau^{\star}$, otherwise $\tau=\Delta_k/\norm{g_k}$.
In both cases,
\[
m_k(0)-m_k(p_k^{C})\ge \frac12\norm{g_k}\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{\norm{B_k}}\Bigr\}.
\]
Multiplying by $\eta\in(0,1)$ gives (L1). \qedhere
\end{proof}

\begin{lemma}[Model vs. actual reduction]\label{lem:ratio}
Assume $f$ has $L$‑Lipschitz gradient. Then for any step $p$ with $\norm{p}\le\Delta_k$,
\[
\bigl|\,f(x_k+p)-m_k(p)\,\bigr|
\le \frac{L}{2}\norm{p}^{2} + \frac{\kappa_B}{2}\norm{p}^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
By the Taylor expansion with remainder,
\[
f(x_k+p)=f(x_k)+g_k^{\top}p+\frac12 p^{\top}\nabla^{2}f(\xi)p,
\]
for some $\xi$ on the segment $[x_k,x_k+p]$. Hence
\[
f(x_k+p)-m_k(p)=\frac12 p^{\top}\bigl(\nabla^{2}f(\xi)-B_k\bigr)p .
\]
Using $\|\nabla^{2}f(\xi)-B_k\|\le \|\nabla^{2}f(\xi)-\nabla^{2}f(x_k)\|+\|B_k-\nabla^{2}f(x_k)\|
\le L\|p\|+\kappa_B$ (Lipschitz Hessian + Assumption \ref{as:model}) we obtain
\[
\bigl|f(x_k+p)-m_k(p)\bigr|
\le \frac12\bigl(L\|p\|+\kappa_B\bigr)\|p\|^{2}
\le \frac{L+\kappa_B}{2}\|p\|^{2}.
\]
\qedhere
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:global}.  The proof follows the classical trust‑region analysis but we annotate every non‑trivial manipulation.

\subsection*{Step 1 – Lower bound on predicted reduction}
From Lemma \ref{lem:cauchy} and the sufficient‑decrease condition (A1) we have for every iteration $k$
\[
m_k(0)-m_k(p_k)\;\ge\;\eta\,\norm{g_k}\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{\norm{B_k}}\Bigr\}.
\tag{S1}
\]

\subsection*{Step 2 – Relating actual to predicted reduction}
By Lemma \ref{lem:ratio} with $p=p_k$,
\[
\bigl|f(x_k)-f(x_k+p_k)-\bigl(m_k(0)-m_k(p_k)\bigr)\bigr|
\le \frac{L+\kappa_B}{2}\norm{p_k}^{2}.
\tag{S2}
\]

\subsection*{Step 3 – Bounding the ratio $\rho_k$}
Recall $\rho_k$ definition (A2). Using (S2) we obtain
\[
\rho_k
=1-\frac{f(x_k)-f(x_k+p_k)-\bigl(m_k(0)-m_k(p_k)\bigr)}{m_k(0)-m_k(p_k)}
\ge 1-\frac{\frac{L+\kappa_B}{2}\norm{p_k}^{2}}{m_k(0)-m_k(p_k)}.
\tag{S3}
\]

\subsection*{Step 4 – Relating $\norm{p_k}$ to the model decrease}
From (S1) and the definition of $p_k$ we have
\[
\norm{p_k}\le\Delta_k,\qquad
m_k(0)-m_k(p_k)\ge \eta\,\norm{g_k}\,\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{\norm{B_k}}\Bigr\}.
\tag{S4}
\]
Moreover $\norm{B_k}\le \|\nabla^{2}f(x_k)\|+\kappa_B\le L+\kappa_B$ (by smoothness and Assumption \ref{as:model}), so
\[
\frac{\norm{g_k}}{\norm{B_k}}\ge \frac{\norm{g_k}}{L+\kappa_B}.
\tag{S5}
\]

\subsection*{Step 5 – Lower bound on the denominator in (S3)}
Using (S4) and (S5),
\[
m_k(0)-m_k(p_k)\;\ge\;\eta\,\norm{g_k}\,\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{L+\kappa_B}\Bigr\}.
\tag{S6}
\]

\subsection*{Step 6 – Upper bound on $\rho_k$ when the iteration is \emph{unsuccessful}}
If $\rho_k<\eta_1$, then from (S3) and (S6)
\[
\eta_1
> 1-\frac{\frac{L+\kappa_B}{2}\norm{p_k}^{2}}{\eta\,\norm{g_k}\,\min\!\{\Delta_k,\frac{\norm{g_k}}{L+\kappa_B}\}} .
\]
Rearranging gives
\[
\frac{L+\kappa_B}{2}\norm{p_k}^{2}
> (1-\eta_1)\,\eta\,\norm{g_k}\,\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{L+\kappa_B}\Bigr\}.
\tag{S7}
\]

\subsection*{Step 7 – Consequence for the radius}
By the update rule (A3), an unsuccessful iteration ($\rho_k<\eta_1$) forces
\[
\Delta_{k+1}= \gamma_{\text{dec}}\Delta_k,\qquad 0<\gamma_{\text{dec}}<1.
\tag{S8}
\]

\subsection*{Step 8 – Telescoping sum of function values}
Consider only \emph{successful} iterations $k\in\mathcal{S}$. For such $k$ we have $\rho_k\ge\eta_1$ and the step is accepted, so
\[
f(x_{k+1}) = f(x_k) - \underbrace{\bigl(f(x_k)-f(x_k+p_k)\bigr)}_{\text{actual reduction}} .
\]
Using the definition of $\rho_k$,
\[
f(x_k)-f(x_{k+1})
= \rho_k\bigl(m_k(0)-m_k(p_k)\bigr)
\ge \eta_1\bigl(m_k(0)-m_k(p_k)\bigr).
\tag{S9}
\]

Insert the lower bound (S6) into (S9):
\[
f(x_k)-f(x_{k+1})
\ge \eta_1\eta\,\norm{g_k}\,\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{L+\kappa_B}\Bigr\}.
\tag{S10}
\]

\subsection*{Step 9 – Bounding the decrease by a quadratic term}
Observe that $\min\{a,b\}\ge \frac{ab}{a+b}$ for any $a,b>0$. Apply with 
$a=\Delta_k$, $b=\frac{\norm{g_k}}{L+\kappa_B}$:
\[
\min\!\Bigl\{\Delta_k,\frac{\norm{g_k}}{L+\kappa_B}\Bigr\}
\ge
\frac{\Delta_k\,\norm{g_k}}{\Delta_k(L+\kappa_B)+\norm{g_k}} .
\]
Thus (S10) yields
\[
f(x_k)-f(x_{k+1})
\ge
\frac{\eta_1\eta\,\Delta_k\,\norm{g_k}^{2}}{\Delta_k(L+\kappa_B)+\norm{g_k}} .
\tag{S11}
\]

Since $\Delta_k\le\Delta_{\max}$, we obtain the uniform bound
\[
f(x_k)-f(x_{k+1})
\ge
\frac{\eta_1\eta}{\Delta_{\max}(L+\kappa_B)+1}\,\norm{g_k}^{2}
\;:=\;c\,\norm{g_k}^{2},
\tag{S12}
\]
where $c>0$ is a constant depending only on the algorithmic parameters and $L,\kappa_B$.

\subsection*{Step 10 – Summation over successful iterations}
Let $K$ be any integer and denote by $S_K:=\{k< K\mid k\in\mathcal{S}\}$ the set of successful indices up to $K-1$. Summing (S12) over $k\in S_K$ gives
\[
\sum_{k\in S_K}\norm{g_k}^{2}
\le \frac{1}{c}\sum_{k\in S_K}\bigl(f(x_k)-f(x_{k+1})\bigr)
\le \frac{f(x_0)-f_{\inf}}{c}.
\tag{S13}
\]

\subsection*{Step 11 – Relating $|S_K|$ to $K$}
Every iteration is either successful or, if unsuccessful, forces a reduction of the radius by the factor $\gamma_{\text{dec}}<1$ (see (S8)). Since the radius is bounded below by a positive constant (e.g., $\Delta_{\min}>0$), the number of consecutive unsuccessful iterations is uniformly bounded by a constant
\[
N_{\max}:= \Bigl\lceil\frac{\log(\Delta_{\min}/\Delta_{0})}{\log\gamma_{\text{dec}}}\Bigr\rceil .
\]
Consequently,
\[
|S_K|\;\ge\;\frac{K}{N_{\max}+1}.
\tag{S14}
\]

\subsection*{Step 12 – Deriving the complexity bound}
Combine (S13) and (S14):
\[
\frac{K}{N_{\max}+1}\;\cdot\;\min_{0\le k<K}\norm{g_k}^{2}
\;\le\;
\sum_{k\in S_K}\norm{g_k}^{2}
\;\le\;
\frac{f(x_0)-f_{\inf}}{c}.
\]
Hence
\[
\min_{0\le k<K}\norm{g_k}^{2}
\;\le\;
\frac{(N_{\max}+1)(f(x_0)-f_{\inf})}{c\,K}.
\tag{S15}
\]
Setting $K=T_{\varepsilon}$ and solving for $K$ such that the left‑hand side is $\le\varepsilon^{2}$ yields
\[
T_{\varepsilon}
\;\le\;
\frac{(N_{\max}+1)(f(x_0)-f_{\inf})}{c\,\varepsilon^{2}}
\;=\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta\,\varepsilon^{2}}\,(L+\kappa_B),
\]
where we used the explicit expression of $c$ from (S12). This is exactly the bound (T1) claimed in Theorem \ref{thm:global}. ∎

\subsection*{Step 13 – Stationarity of limit points}
From (S15) we have $\norm{g_k}\to0$ along the subsequence of successful iterates. Since the distance between any two consecutive iterates is bounded by $\Delta_k\le\Delta_{\max}$, the whole sequence $\{x_k\}$ is bounded. Any accumulation point $\bar x$ is the limit of a subsequence where $\norm{\nabla f}$ tends to zero, thus $\nabla f(\bar x)=0$. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
Collecting the constants introduced:
\[
c \;=\; \frac{\eta_1\eta}{\Delta_{\max}(L+\kappa_B)+1},
\qquad
N_{\max}+1 \;=\; 1+\Bigl\lceil\frac{\log(\Delta_{\min}/\Delta_{0})}{\log\gamma_{\text{dec}}}\Bigr\rceil .
\]
Plugging into (S15) gives the explicit rate
\[
\min_{0\le k<T}\norm{\nabla f(x_k)}\;\le\;
\sqrt{\frac{(N_{\max}+1)(f(x_0)-f_{\inf})}{c\,T}}
\;=\;
\sqrt{\frac{(N_{\max}+1)\bigl(\Delta_{\max}(L+\kappa_B)+1\bigr)}{\eta_1\eta\,T}\,\bigl(f(x_0)-f_{\inf}\bigr)} .
\]
Thus the algorithm enjoys an $O(1/\sqrt{T})$ decay of the gradient norm, which is optimal for first‑order methods on non‑convex smooth problems.

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}[label=\alph*.,leftmargin=*,itemsep=0.3em]
\item \textbf{Exact subproblem solution.} If $p_k$ is the exact minimiser of (TR$_k$), then the sufficient‑decrease condition holds with $\eta=1$. Consequently $c$ improves, yielding a tighter bound
\[
T_{\varepsilon}\le \frac{(f(x_0)-f_{\inf})}{\eta_1 (L+\kappa_B)}\frac{1}{\varepsilon^{2}} .
\]
\item \textbf{Quadratic objective.} When $f$ is a quadratic with constant Hessian, the model is exact ($B_k=\nabla^{2}f$) and Lemma \ref{lem:ratio} gives zero error. The ratio $\rho_k$ becomes $1$, every iteration is successful, and the algorithm terminates in at most one step (the Cauchy point coincides with the Newton step).
\item \textbf{Convex case.} If $f$ is convex, the same $O(1/\varepsilon^{2})$ bound holds, but one can further show that the function values converge at rate $O(1/T)$ by exploiting convexity of $f$ in the telescoping sum.
\end{enumerate}
