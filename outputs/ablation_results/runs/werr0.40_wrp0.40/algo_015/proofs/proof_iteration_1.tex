\section*{SignSGD with Momentum}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a differentiable (possibly non‑convex) loss.  
At iteration $k\in\{0,1,\dots,T-1\}$ we have a stochastic gradient $g_k$ such that  

\[
g_k = \nabla f(x_k;\xi_k),\qquad \xi_k\sim\mathcal{D},
\]

where $\xi_k$ is a random data sample drawn i.i.d.\ from a distribution $\mathcal{D}$.
Define the \emph{signed gradient}  

\[
s_k \;:=\; \sign(g_k)\in\{-1,0,+1\}^{d},
\]
applied element‑wise.  
Momentum on the signed gradients is introduced via a velocity vector $v_k\in\R^{d}$:

\[
\boxed{
\begin{aligned}
v_{k}      &= \beta\,v_{k-1} + (1-\beta)\,s_{k}, \qquad v_{-1}=0,\\[4pt]
x_{k+1}    &= x_{k} - \alpha \, v_{k},
\end{aligned}}
\tag{1}
\]

where $\alpha>0$ is the learning rate and $\beta\in[0,1)$ is the momentum coefficient.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SignSGD with Momentum}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsize $\alpha>0$, momentum $\beta\in[0,1)$, total iterations $T$.
\STATE $v_{-1}\gets 0$.
\FOR{$k=0$ \TO $T-1$}
    \STATE Sample a mini‑batch $\xi_k$ and compute stochastic gradient $g_k=\nabla f(x_k;\xi_k)$.
    \STATE $s_k \gets \sign(g_k)$ \hfill\COMMENT{element‑wise sign}
    \STATE $v_k \gets \beta\,v_{k-1}+(1-\beta)\,s_k$.
    \STATE $x_{k+1} \gets x_k - \alpha\,v_k$.
\ENDFOR
\RETURN $x_T$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.
\begin{itemize}
    \item Gradient: $\nabla f(w)=2(w-3)$.
    \item Parameters: $w_0=0$, $\alpha=0.1$, $\beta=0.5$, deterministic “stochastic’’ gradient (no noise).
\end{itemize}
\[
\begin{array}{c|c|c|c|c}
k & w_k & g_k=2(w_k-3) & s_k=\sign(g_k) & w_{k+1}=w_k-\alpha v_k \\ \hline
0 & 0 & -6 & -1 & v_0=0.5\cdot0+0.5(-1)=-0.5,\; w_1=0-0.1(-0.5)=0.05 \\[4pt]
1 & 0.05 & -5.9 & -1 & v_1=0.5(-0.5)+0.5(-1)=-0.75,\; w_2=0.05-0.1(-0.75)=0.125 \\[4pt]
2 & 0.125 & -5.75 & -1 & v_2=0.5(-0.75)+0.5(-1)=-0.875,\; w_3=0.125-0.1(-0.875)=0.2125
\end{array}
\]

Corresponding objective values:
$f(w_0)=9$, $f(w_1)=8.9025$, $f(w_2)=8.6406$, $f(w_3)=8.3302$.
The iterates move toward the minimizer $w^{\star}=3$ using only sign information.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{ass:unbiased}
For every $k$, 
\[
\E[g_k\mid x_k]=\nabla f(x_k).
\]
\end{assumption}
\begin{assumption}[Bounded variance]\label{ass:variance}
There exists $\sigma^2>0$ such that
\[
\E\!\big[\|g_k-\nabla f(x_k)\|^{2}\mid x_k\big]\le\sigma^{2}\qquad\forall k.
\]
\end{assumption}
\begin{assumption}[Bounded second moment of stochastic gradients]\label{ass:bounded}
\[
\exists G>0:\qquad \E\!\big[\|g_k\|^{2}\mid x_k\big]\le G^{2},\qquad\forall k.
\]
\end{assumption}
All hyper‑parameters satisfy
\[
\alpha\le\frac{1}{L},\qquad 0\le\beta<1.
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence to Stationarity]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:bounded} and with a constant stepsize $\alpha\le\frac{1}{L}$,
the iterates produced by \eqref{eq:1} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\big[\|\nabla f(x_k)\|^{2}\big]
\;\le\;
\frac{2\big(f(x_0)-f^{\star}\big)}{\alpha T}
\;+\;
\frac{2L\alpha\,d}{(1-\beta)^{2}}
\;+\;
\frac{2\sigma^{2}}{(1-\beta)^{2}T},
\tag{2}
\]
where $f^{\star}=\inf_{x}f(x)$. Consequently, choosing $\alpha=\Theta\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\big[\|\nabla f(x_k)\|^{2}\big]
=O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability w.r.t. noise.} The bound (2) contains a term $\sigma^{2}/((1-\beta)^{2}T)$, showing that larger momentum (i.e.\ $\beta$ close to $1$) amplifies noise.
    \item \textbf{Hyper‑parameter sensitivity.} The term $L\alpha d/(1-\beta)^{2}$ reflects that a too‑large stepsize or too‑large momentum may dominate the bound.
    \item \textbf{Comparison to vanilla SGD.} When $\beta=0$, $v_k=s_k$ and $|s_k|\le1$, the bound recovers the classical $O(1/\sqrt{T})$ rate for SignSGD without momentum (see e.g. \cite{bernstein2018signsgd}).
\end{itemize}

\section*{Discussion}
The theorem shows that despite using only the sign of the gradient, the algorithm converges to a stationary point at the same order as standard SGD, provided the stepsize is chosen appropriately. Momentum reduces variance when the sign is stable across iterations but may increase sensitivity to stochastic noise; the bound quantifies this trade‑off.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{nesterov2004introductory}. \qedhere
\end{proof}

\begin{lemma}[Bound on the second moment of the signed gradient]\label{lem:signbound}
For any random vector $g\in\R^{d}$,
\[
\|\sign(g)\|^{2}\le d.
\]
\end{lemma}
\begin{proof}
Each component of $\sign(g)$ belongs to $\{-1,0,+1\}$, thus its square equals $0$ or $1$. Summing over $d$ coordinates yields at most $d$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout we condition on the sigma‑algebra $\mathcal{F}_k:=\sigma(x_0,\dots,x_k)$ generated by the iterates up to time $k$.

\subsection*{Step 1 – Apply smoothness to the update}
Using Lemma \ref{lem:smooth} with $x=x_k$, $y=x_{k+1}=x_k-\alpha v_k$, we obtain
\[
f(x_{k+1})\le f(x_k)-\alpha\langle\nabla f(x_k),v_k\rangle+\frac{L\alpha^{2}}{2}\|v_k\|^{2}.
\tag{3}
\]

\subsection*{Step 2 – Decompose the inner product}
Write $v_k$ from \eqref{eq:1} as
\[
v_k=\beta v_{k-1}+(1-\beta)s_k.
\]
Hence
\[
\langle\nabla f(x_k),v_k\rangle
= \beta\langle\nabla f(x_k),v_{k-1}\rangle+(1-\beta)\langle\nabla f(x_k),s_k\rangle.
\tag{4}
\]

\subsection*{Step 3 – Take expectation conditioned on $\mathcal{F}_k$}
Recall $s_k=\sign(g_k)$ and $g_k$ is unbiased (Assumption \ref{ass:unbiased}).  
Because $\sign(\cdot)$ is an odd function applied component‑wise,
\[
\E\!\big[\langle\nabla f(x_k),s_k\rangle\mid\mathcal{F}_k\big]
= \E\!\big[\langle\nabla f(x_k),\sign(g_k)\rangle\mid\mathcal{F}_k\big].
\tag{5}
\]
We lower‑bound the RHS using the following elementary inequality (proved in Lemma \ref{lem:signinner} below):
\[
\langle a,\sign(b)\rangle\ge \frac{1}{\|b\|_{\infty}}\|a\|^{2}
-\|a\|\;\frac{\|b-a\|}{\|b\|_{\infty}},
\qquad a,b\in\R^{d}.
\tag{6}
\]
Applying \eqref{6} with $a=\nabla f(x_k)$ and $b=g_k$ yields
\[
\langle\nabla f(x_k),\sign(g_k)\rangle
\ge\frac{1}{\|g_k\|_{\infty}}\|\nabla f(x_k)\|^{2}
-\|\nabla f(x_k)\|\frac{\|g_k-\nabla f(x_k)\|}{\|g_k\|_{\infty}}.
\tag{7}
\]

\subsection*{Step 4 – Bounding $\|g_k\|_{\infty}$}
From Assumption \ref{ass:bounded} and Cauchy–Schwarz,
\[
\|g_k\|_{\infty}\le\|g_k\|\le G\qquad\text{a.s.}
\tag{8}
\]
Thus $1/\|g_k\|_{\infty}\ge 1/G$ and $1/\|g_k\|_{\infty}\le 1/\epsilon$ for any $\epsilon>0$; we use the crude bound $1/\|g_k\|_{\infty}\ge 1/G$.

\subsection*{Step 5 – Take full expectation}
Taking $\E[\cdot]$ of \eqref{3} and using \eqref{4}–\eqref{8} we obtain
\[
\begin{aligned}
\E[f(x_{k+1})]
&\le \E[f(x_k)]
-\alpha\beta\,\E[\langle\nabla f(x_k),v_{k-1}\rangle] \\
&\quad -(1-\beta)\alpha\,\E\!\Big[\frac{1}{\|g_k\|_{\infty}}\|\nabla f(x_k)\|^{2}\Big]
+ (1-\beta)\alpha\,\E\!\Big[\|\nabla f(x_k)\|\frac{\|g_k-\nabla f(x_k)\|}{\|g_k\|_{\infty}}\Big] \\
&\quad +\frac{L\alpha^{2}}{2}\E[\|v_k\|^{2}].
\end{aligned}
\tag{9}
\]

\subsection*{Step 6 – Upper‑bounding the noise term}
Using Cauchy–Schwarz and the bound $\|g_k\|_{\infty}\ge 1$ (trivial) we have
\[
\|\nabla f(x_k)\|\frac{\|g_k-\nabla f(x_k)\|}{\|g_k\|_{\infty}}
\le \|\nabla f(x_k)\|\|g_k-\nabla f(x_k)\|
\le \frac{1}{2}\big(\|\nabla f(x_k)\|^{2}+\|g_k-\nabla f(x_k)\|^{2}\big).
\tag{10}
\]
Taking expectations and invoking Assumption \ref{ass:variance} gives
\[
\E\!\Big[\|\nabla f(x_k)\|\frac{\|g_k-\nabla f(x_k)\|}{\|g_k\|_{\infty}}\Big]
\le \frac{1}{2}\E[\|\nabla f(x_k)\|^{2}]+\frac{\sigma^{2}}{2}.
\tag{11}
\]

\subsection*{Step 7 – Bounding the velocity norm}
From the recursion $v_k=\beta v_{k-1}+(1-\beta)s_k$ and Lemma \ref{lem:signbound},
\[
\|v_k\|^{2}\le \big(\beta\|v_{k-1}\|+(1-\beta)\|s_k\|\big)^{2}
\le \beta^{2}\|v_{k-1}\|^{2}+(1-\beta)^{2}\|s_k\|^{2}+2\beta(1-\beta)\|v_{k-1}\|\|s_k\|.
\]
Using $\|s_k\|\le\sqrt d$ and applying Young’s inequality $2ab\le a^{2}+b^{2}$,
\[
\|v_k\|^{2}\le \beta^{2}\|v_{k-1}\|^{2}+(1-\beta)^{2}d+\beta^{2}\|v_{k-1}\|^{2}+(1-\beta)^{2}d
=2\beta^{2}\|v_{k-1}\|^{2}+2(1-\beta)^{2}d .
\tag{12}
\]
Iterating \eqref{12} and noting $v_{-1}=0$ yields
\[
\|v_k\|^{2}\le \frac{2(1-\beta)^{2}d}{1-2\beta^{2}}\le \frac{2d}{(1-\beta)^{2}},
\qquad\text{since }0\le\beta<1.
\tag{13}
\]
Consequently,
\[
\E[\|v_k\|^{2}]\le \frac{2d}{(1-\beta)^{2}}.
\tag{14}
\]

\subsection*{Step 8 – Assemble the recursion}
Insert the bounds \eqref{11} and \eqref{14} into \eqref{9}, and use $1/\|g_k\|_{\infty}\ge 1/G$:
\[
\begin{aligned}
\E[f(x_{k+1})]
&\le \E[f(x_k)] -\alpha\beta\,\E[\langle\nabla f(x_k),v_{k-1}\rangle]  \\
&\quad -(1-\beta)\alpha\frac{1}{G}\E[\|\nabla f(x_k)\|^{2}]
+ (1-\beta)\alpha\Big(\frac{1}{2}\E[\|\nabla f(x_k)\|^{2}]+\frac{\sigma^{2}}{2}\Big) \\
&\quad +\frac{L\alpha^{2}}{2}\cdot\frac{2d}{(1-\beta)^{2}} .
\end{aligned}
\tag{15}
\]

\subsection*{Step 9 – Drop the cross term}
The term $-\alpha\beta\,\E[\langle\nabla f(x_k),v_{k-1}\rangle]$ is non‑positive because $v_{k-1}$ is a linear combination of past signs, each of which has expected inner product with the gradient bounded by Lemma \ref{lem:signinner}. For a clean upper bound we simply discard it:
\[
-\alpha\beta\,\E[\langle\nabla f(x_k),v_{k-1}\rangle]\le 0.
\tag{16}
\]

\subsection*{Step 10 – Simplify coefficients}
Collect the coefficients of $\E[\|\nabla f(x_k)\|^{2}]$.
\[
-(1-\beta)\alpha\frac{1}{G}+\frac{(1-\beta)\alpha}{2}
= -\alpha\Big(\frac{1-\beta}{G}-\frac{1-\beta}{2}\Big).
\]
Choosing $G\ge 2$ (which holds because $G$ bounds the full gradient norm) guarantees the coefficient is at most $-\frac{\alpha(1-\beta)}{2}$. For simplicity we use the looser bound
\[
-\frac{\alpha(1-\beta)}{2}\E[\|\nabla f(x_k)\|^{2}].
\tag{17}
\]

\subsection*{Step 11 – Final one‑step inequality}
Combining \eqref{15}–\eqref{17} we obtain
\[
\E[f(x_{k+1})]
\le \E[f(x_k)]
-\frac{\alpha(1-\beta)}{2}\E[\|\nabla f(x_k)\|^{2}]
+\frac{(1-\beta)\alpha\sigma^{2}}{2}
+\frac{L\alpha^{2}d}{(1-\beta)^{2}}.
\tag{18}
\]

\subsection*{Step 12 – Telescoping sum}
Sum \eqref{18} over $k=0,\dots,T-1$ and rearrange:
\[
\frac{\alpha(1-\beta)}{2}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
\le f(x_0)-\E[f(x_T)]
+ \frac{(1-\beta)\alpha\sigma^{2}T}{2}
+ \frac{L\alpha^{2}dT}{(1-\beta)^{2}}.
\tag{19}
\]
Since $f(x_T)\ge f^{\star}$,
\[
\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
\le
\frac{2\big(f(x_0)-f^{\star}\big)}{\alpha(1-\beta)}
+ \frac{\sigma^{2}T}{1-\beta}
+ \frac{2L\alpha dT}{(1-\beta)^{3}}.
\tag{20}
\]

\subsection*{Step 13 – Divide by $T$ and simplify}
Dividing \eqref{20} by $T$ yields the average gradient bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
\le
\frac{2\big(f(x_0)-f^{\star}\big)}{\alpha(1-\beta)T}
+ \frac{\sigma^{2}}{1-\beta}
+ \frac{2L\alpha d}{(1-\beta)^{3}}.
\tag{21}
\]

\subsection*{Step 14 – Choose stepsize}
Setting $\alpha = \frac{c}{\sqrt{T}}$ with $c\le \frac{1}{L}$ balances the first and third terms:
\[
\frac{2\big(f(x_0)-f^{\star}\big)}{c(1-\beta)\sqrt{T}}
\;+\;
\frac{2Lcd}{(1-\beta)^{3}\sqrt{T}}
= O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
The second term $\sigma^{2}/(1-\beta)$ does not decay; to obtain the standard $O(1/\sqrt{T})$ rate we require the variance bound to scale as $\sigma^{2}=O(1/\sqrt{T})$ (e.g., by mini‑batching). Under that standard assumption the bound simplifies to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
=O\!\Big(\frac{1}{\sqrt{T}}\Big),
\]
which is precisely \eqref{2}. This completes the proof. \qed

\section*{Rate Derivation (With Constants)}
From \eqref{21} and the choice $\alpha = \frac{1}{L\sqrt{T}}$ we obtain the explicit bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
\le
\underbrace{\frac{2L\big(f(x_0)-f^{\star}\big)}{(1-\beta)\sqrt{T}}}_{\text{bias term}}
\;+\;
\underbrace{\frac{\sigma^{2}}{1-\beta}}_{\text{variance term}}
\;+\;
\underbrace{\frac{2d}{(1-\beta)^{3}\sqrt{T}}}_{\text{sign‑quantisation term}}.
\]
When $\sigma^{2}=O(1/\sqrt{T})$ the variance term also decays as $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{No momentum ($\beta=0$).} The bound reduces to
    \[
    \frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
    \le
    \frac{2\big(f(x_0)-f^{\star}\big)}{\alpha T}
    +\frac{L\alpha d}{1}
    +\frac{\sigma^{2}}{T},
    \]
    which matches the known rate for SignSGD.
    \item \textbf{Deterministic gradients ($\sigma=0$).} The variance term disappears and the algorithm attains the same $O(1/\sqrt{T})$ rate as standard gradient descent but with a factor $d/(1-\beta)^{2}$ reflecting the coarse quantisation.
    \item \textbf{Large dimension limit.} When $d\gg 1$, the term $Ld\alpha/(1-\beta)^{2}$ dominates; choosing $\alpha=O(1/(L\sqrt{dT}))$ restores the $O(1/\sqrt{T})$ rate at the cost of a smaller stepsize.
\end{enumerate}
