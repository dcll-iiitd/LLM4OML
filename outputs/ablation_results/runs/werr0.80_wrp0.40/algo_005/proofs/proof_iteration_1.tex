\section*{SVRG (Stochastic Variance Reduced Gradient)}

\section*{Mathematical Formulation}
We consider a finite‑sum objective
\[
    f(x)\;=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad 
    f_i:\mathbb{R}^d\to\mathbb{R},
\]
with each $f_i$ convex, $L$–smooth and the overall $f$ $\mu$–strongly convex.  
SVRG proceeds in \emph{epochs} $s=0,1,\dots$.  At the beginning of epoch $s$ a \emph{snapshot}
$\tilde{x}^{\,s}$ is fixed and the full gradient at the snapshot is computed
\[
    \tilde{g}^{\,s}\;=\;\nabla f(\tilde{x}^{\,s})
    \;=\;\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s}).
\]
Inside the epoch we perform $m$ inner stochastic updates.  Let $x_0^{\,s}= \tilde{x}^{\,s}$.  For $t=0,\dots,m-1$ we draw uniformly at random an index
$i_t\in\{1,\dots,n\}$ and form the variance‑reduced estimator
\[
    v_t^{\,s}\;=\;\nabla f_{i_t}\!\bigl(x_t^{\,s}\bigr)
               \;-\;\nabla f_{i_t}\!\bigl(\tilde{x}^{\,s}\bigr)
               \;+\;\tilde{g}^{\,s}.
\]
The inner iterate is updated by a standard gradient step
\[
    x_{t+1}^{\,s}\;=\;x_t^{\,s}\;-\;\eta\, v_t^{\,s},
\tag{1}
\]
where $\eta>0$ is a constant stepsize.  After $m$ inner steps we set the next snapshot
\[
    \tilde{x}^{\,s+1}\;=\;x_m^{\,s}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG}
\begin{algorithmic}[1]
\Require Number of epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $x^{0}$
\State $\tilde{x}^{\,0}\gets x^{0}$
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\tilde{g}^{\,s}\gets \nabla f(\tilde{x}^{\,s})$
    \State $x_0^{\,s}\gets \tilde{x}^{\,s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t^{\,s}\gets \nabla f_{i_t}(x_t^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{g}^{\,s}$
        \State $x_{t+1}^{\,s}\gets x_t^{\,s}-\eta\,v_t^{\,s}$ \Comment{Eq.~(1)}
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_m^{\,s}$ \Comment{Eq.~(2)}
\EndFor
\State \textbf{return} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
    f(w)= (w-3)^2,\qquad n=1,\;\;f_1(w)=f(w).
\]
Hence $\nabla f(w)=2(w-3)$, $L=2$, $\mu=2$.
Set $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$, and run a single epoch ($S=1$).

\begin{enumerate}[label=\arabic*.]
\item \textbf{Snapshot and full gradient:}\\
    $\tilde{w}^{\,0}=w_0=0$, 
    $\tilde{g}^{\,0}= \nabla f(0)=2(0-3)=-6$.
\item \textbf{Inner step $t=0$:}\\
    Sample $i_0=1$ (the only index).\\
    $v_0^{\,0}= \nabla f(w_0)-\nabla f(\tilde{w}^{\,0})+\tilde{g}^{\,0}
               =(-6)-(-6)+(-6)=-6.$\\
    Update $w_1 = w_0-\eta v_0^{\,0}=0-0.1(-6)=0.6.$\\
    Objective $f(w_1)=(0.6-3)^2=5.76$.
\item \textbf{Inner step $t=1$:}\\
    $v_1^{\,0}= \nabla f(w_1)-\nabla f(\tilde{w}^{\,0})+\tilde{g}^{\,0}
               =2(0.6-3)-(-6)-6 = -4.8-(-6)-6 = -4.8+6-6 = -4.8.$\\
    $w_2 = w_1-\eta v_1^{\,0}=0.6-0.1(-4.8)=1.08.$\\
    $f(w_2)=(1.08-3)^2=3.6864$.
\item \textbf{Inner step $t=2$:}\\
    $v_2^{\,0}= \nabla f(w_2)-\nabla f(\tilde{w}^{\,0})+\tilde{g}^{\,0}
               =2(1.08-3)-(-6)-6 = -3.84+6-6 = -3.84.$\\
    $w_3 = w_2-\eta v_2^{\,0}=1.08-0.1(-3.84)=1.464.$\\
    $f(w_3)=(1.464-3)^2=2.3665$.
\item \textbf{End of epoch:}\\
    $\tilde{w}^{\,1}=w_3=1.464$, objective $f(\tilde{w}^{\,1})\approx2.37$.
\end{enumerate}
The sequence demonstrates a monotone decrease toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*}:, leftmargin=*]
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
    \forall x,y\in\mathbb{R}^d,\qquad
    \norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y}.
\tag{A1}
\]
\item \textbf{Strong convexity.} The aggregate $f$ is $\mu$‑strongly convex:
\[
    \forall x,y,\qquad
    f(y)\ge f(x)+\inner{\nabla f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^2.
\tag{A2}
\]
\item \textbf{Unbiased estimator.} For any fixed snapshot $\tilde{x}$ and any iterate $x$,
\[
    \E_{i\sim\mathcal{U}[n]}\!\bigl[\,v(x,\tilde{x})\,\bigr]
    =\nabla f(x).
\tag{A3}
\]
\item \textbf{Bounded variance of the estimator.} For all $x,\tilde{x}$,
\[
    \E\!\bigl[\norm{v(x,\tilde{x})-\nabla f(x)}^2\bigr]
    \le 2L\bigl(f(x)-f(\tilde{x})- \inner{\nabla f(\tilde{x})}{x-\tilde{x}}\bigr).
\tag{A4}
\]
\item \textbf{Stepsize restriction.} The constant stepsize satisfies
\[
    0<\eta < \frac{1}{2L}.
\tag{A5}
\]
\item \textbf{Inner loop length.} $m\ge 1$ is a positive integer.
\tag{A6}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Under Assumptions \textbf{A1}–\textbf{A6}, let the stepsize $\eta$ satisfy
\[
    0<\eta \le \frac{1}{2L}.
\]
Define the contraction factor
\[
    \rho \;:=\; \frac{1}{1+\mu\eta\bigl(1-2L\eta\bigr)m}\;\in(0,1).
\tag{3}
\]
Then for the sequence of snapshots $\{\tilde{x}^{\,s}\}_{s\ge0}$ generated by SVRG,
\[
    \E\!\bigl[\,f(\tilde{x}^{\,s})-f(x^\star)\,\bigr]
    \;\le\;
    \rho^{\,s}\,\bigl(f(\tilde{x}^{\,0})-f(x^\star)\bigr),
    \qquad\forall s\ge0,
\tag{4}
\]
where $x^\star=\arg\min f$.
Consequently,
\[
    \norm{\E[\tilde{x}^{\,s}]-x^\star}^2
    \le \frac{2}{\mu}\,\rho^{\,s}\,\bigl(f(\tilde{x}^{\,0})-f(x^\star)\bigr).
\tag{5}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\textbf{P\arabic*}:, leftmargin=*]
\item \textbf{Stability.} The contraction factor $\rho$ is decreasing in $\eta$ and $m$ as long as $\eta\le\frac{1}{2L}$. Larger inner loop length $m$ yields faster linear decay.
\item \textbf{Hyperparameter sensitivity.} If $\eta$ approaches $\frac{1}{2L}$, the term $1-2L\eta$ tends to zero, and $\rho\approx \frac{1}{1+\mu\eta m}$, i.e. the algorithm behaves similarly to a deterministic gradient method with effective stepsize $\eta m$.
\item \textbf{Comparison to SGD.} Classical SGD under the same assumptions attains a sublinear rate $O(1/t)$. SVRG improves this to a geometric rate $O(\rho^{\,s})$, at the cost of occasional full‑gradient evaluations.
\item \textbf{Special case $m=n$.} When $m=n$ and $\eta=\frac{1}{3L}$, the bound $\rho\le \frac{2}{3}$ recovers the classic SVRG constant from the original literature.
\end{enumerate}

\section*{Discussion}
The theorem shows that SVRG enjoys \emph{linear convergence} for strongly convex smooth finite‑sum problems.  The proof relies on three standard ingredients: (i) smoothness to relate function values and gradient norms, (ii) strong convexity to obtain a quadratic growth condition, and (iii) the variance‑reduction identity (A3) together with the variance bound (A4) to control the stochastic error inside each inner loop.  The result holds for any $m\ge1$; in practice a modest $m$ (often $m\approx2n$) yields the best trade‑off between computation and convergence speed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If each $f_i$ is $L$‑smooth, then for any $x,y$,
\[
    f(y) \le f(x) + \inner{\nabla f(x)}{y-x}
           + \frac{L}{2}\norm{y-x}^2.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Averaging the standard $L$‑smoothness inequality for each $f_i$ yields (L1). \qedhere
\end{proof}

\begin{lemma}[Strong convexity inequality]\label{lem:strong}
If $f$ is $\mu$‑strongly convex, then for any $x$,
\[
    \norm{x-x^\star}^2 \le \frac{2}{\mu}\bigl(f(x)-f(x^\star)\bigr).
\tag{L2}
\]
\end{proof}
\begin{proof}
Apply the definition of strong convexity with $y=x^\star$ and rearrange. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we condition on the snapshot
$\tilde{x}^{\,s}$ and drop the superscript $s$ for brevity.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{One‑step recursion for the inner iterate.}  
From the update rule (1) and expanding the squared distance to $x^\star$,
\begin{align}
    \norm{x_{t+1}-x^\star}^2
    &=\norm{x_t-\eta v_t - x^\star}^2 \nonumber\\
    &=\norm{x_t-x^\star}^2 -2\eta\inner{v_t}{x_t-x^\star}
      +\eta^2\norm{v_t}^2.
\tag{6}
\end{align}
\item \textbf{Take expectation w.r.t.\ the random index $i_t$.}  
Using unbiasedness (A3) we have
\[
    \E_{i_t}[v_t]=\nabla f(x_t),\qquad
    \E_{i_t}[\norm{v_t}^2]
    =\norm{\nabla f(x_t)}^2
      +\E_{i_t}\!\bigl[\norm{v_t-\nabla f(x_t)}^2\bigr].
\tag{7}
\]
\item \textbf{Insert (7) into (6) and take full expectation.}  
Denoting $\mathcal{F}_t$ the $\sigma$‑algebra generated by $\{x_0,\dots,x_t\}$,
\begin{align}
    \E\!\bigl[\norm{x_{t+1}-x^\star}^2\mid\mathcal{F}_t\bigr]
    &\le \norm{x_t-x^\star}^2
       -2\eta\inner{\nabla f(x_t)}{x_t-x^\star}
       +\eta^2\norm{\nabla f(x_t)}^2 \nonumber\\
    &\quad +\eta^2\,\E\!\bigl[\norm{v_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\bigr].
\tag{8}
\end{align}
\item \textbf{Apply smoothness to bound the inner product.}  
Strong convexity implies
\[
    \inner{\nabla f(x_t)}{x_t-x^\star}
    \ge f(x_t)-f(x^\star)+\frac{\mu}{2}\norm{x_t-x^\star}^2.
\tag{9}
\]
\item \textbf{Apply Lemma~\ref{lem:smooth} to bound $\norm{\nabla f(x_t)}^2$.}  
From $L$‑smoothness,
\[
    \norm{\nabla f(x_t)}^2
    \le 2L\bigl(f(x_t)-f(x^\star)\bigr).
\tag{10}
\]
\item \textbf{Insert (9) and (10) into (8).}  After rearranging,
\begin{align}
    \E\!\bigl[\norm{x_{t+1}-x^\star}^2\mid\mathcal{F}_t\bigr]
    &\le \bigl(1-\mu\eta\bigr)\norm{x_t-x^\star}^2
       +\bigl(2L\eta^2-2\eta\bigr)\bigl(f(x_t)-f(x^\star)\bigr) \nonumber\\
    &\quad +\eta^2\,\E\!\bigl[\norm{v_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\bigr].
\tag{11}
\end{align}
\item \textbf{Bound the variance term using (A4).}  
Recall the variance bound (A4):
\[
    \E\!\bigl[\norm{v_t-\nabla f(x_t)}^2\bigr]
    \le 2L\Bigl(f(x_t)-f(\tilde{x})-\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}\Bigr).
\tag{12}
\]
Since $\tilde{x}$ is fixed during the inner loop, we may apply (9) with $x=\tilde{x}$ to obtain
\[
    -\inner{\nabla f(\tilde{x})}{x_t-\tilde{x}}
    \le -\bigl(f(x_t)-f(\tilde{x})\bigr)-\frac{\mu}{2}\norm{x_t-\tilde{x}}^2.
\tag{13}
\]
Plugging (13) into (12) yields
\[
    \E\!\bigl[\norm{v_t-\nabla f(x_t)}^2\bigr]
    \le 2L\Bigl(f(\tilde{x})-f(x^\star)\Bigr)
       -\mu L\norm{x_t-\tilde{x}}^2.
\tag{14}
\]
\item \textbf{Combine (11) and (14).}  Using (A5) ($\eta\le\frac1{2L}$) we have $2L\eta^2-2\eta\le -\eta$. Hence,
\begin{align}
    \E\!\bigl[\norm{x_{t+1}-x^\star}^2\mid\mathcal{F}_t\bigr]
    &\le (1-\mu\eta)\norm{x_t-x^\star}^2
       -\eta\bigl(f(x_t)-f(x^\star)\bigr) \nonumber\\
    &\quad +2L\eta^2\bigl(f(\tilde{x})-f(x^\star)\bigr)
       -\mu L\eta^2\norm{x_t-\tilde{x}}^2.
\tag{15}
\end{align}
\item \textbf{Drop the non‑positive term $-\mu L\eta^2\norm{x_t-\tilde{x}}^2$} to obtain a cleaner inequality:
\[
    \E\!\bigl[\norm{x_{t+1}-x^\star}^2\mid\mathcal{F}_t\bigr]
    \le (1-\mu\eta)\norm{x_t-x^\star}^2
       -\eta\bigl(f(x_t)-f(x^\star)\bigr)
       +2L\eta^2\bigl(f(\tilde{x})-f(x^\star)\bigr).
\tag{16}
\]
\item \textbf{Take full expectation and sum over $t=0,\dots,m-1$.}  
Define $A_t:=\E[\norm{x_t-x^\star}^2]$ and $B_t:=\E[f(x_t)-f(x^\star)]$.  Summing (16) yields
\[
    A_m \le (1-\mu\eta)^m A_0
          -\eta\sum_{t=0}^{m-1} B_t
          +2L\eta^2 m\bigl(f(\tilde{x})-f(x^\star)\bigr).
\tag{17}
\]
\item \textbf{Relate the average inner‑loop suboptimality to the snapshot.}  
Because $f$ is convex,
\[
    f(\tilde{x}^{\text{new}})=f(x_m)
    \le \frac{1}{m}\sum_{t=0}^{m-1} f(x_t).
\tag{18}
\]
Taking expectations,
\[
    \E\!\bigl[f(\tilde{x}^{\text{new}})-f(x^\star)\bigr]
    \le \frac{1}{m}\sum_{t=0}^{m-1} B_t.
\tag{19}
\]
\item \textbf{Combine (17) and (19).}  Using $A_0=\norm{\tilde{x}-x^\star}^2$ and Lemma~\ref{lem:strong} ($A_0\le\frac{2}{\mu}(f(\tilde{x})-f(x^\star))$),
\begin{align}
    \E\!\bigl[f(\tilde{x}^{\text{new}})-f(x^\star)\bigr]
    &\le \frac{1}{\eta m}
           \Bigl[(1-\mu\eta)^m A_0
               +2L\eta^2 m\bigl(f(\tilde{x})-f(x^\star)\bigr)\Bigr] \nonumber\\
    &\le \frac{(1-\mu\eta)^m}{\eta m}\cdot\frac{2}{\mu}\bigl(f(\tilde{x})-f(x^\star)\bigr)
        +2L\eta\bigl(f(\tilde{x})-f(x^\star)\bigr).
\tag{20}
\end{align}
\item \textbf{Choose $\eta$ such that $1-2L\eta>0$ and simplify.}  
Since $0<\eta\le\frac{1}{2L}$, we have $1-2L\eta\ge0$ and
\[
    (1-\mu\eta)^m \le \frac{1}{1+\mu\eta(1-2L\eta)m}.
\tag{21}
\]
Using (21) in (20) gives
\[
    \E\!\bigl[f(\tilde{x}^{\text{new}})-f(x^\star)\bigr]
    \le
    \underbrace{\frac{1}{1+\mu\eta(1-2L\eta)m}}_{\displaystyle =\,\rho}
    \bigl(f(\tilde{x})-f(x^\star)\bigr).
\tag{22}
\]
\item \textbf{Iterate the epoch‑wise contraction.}  Applying (22) recursively over epochs $s=0,1,\dots$ yields exactly the linear rate (4) of Theorem~\ref{thm:linear}.
\end{enumerate}
All steps rely only on Assumptions \textbf{A1}–\textbf{A6}, completing the proof. \qed

\section*{Rate Derivation (With Constants)}
From (22) we identified the contraction factor
\[
    \rho = \frac{1}{1+\mu\eta(1-2L\eta)m}.
\]
If we set $\eta=\frac{1}{3L}$ (a common practical choice satisfying $\eta<\frac1{2L}$) then
\[
    1-2L\eta = 1-\frac{2}{3}= \frac{1}{3},
\qquad
    \rho = \frac{1}{1+\frac{\mu}{3L}m}.
\]
Thus after $s$ epochs,
\[
    \E\!\bigl[f(\tilde{x}^{\,s})-f(x^\star)\bigr]
    \le \Bigl(\frac{1}{1+\frac{\mu}{3L}m}\Bigr)^{\!s}
          \bigl(f(\tilde{x}^{\,0})-f(x^\star)\bigr).
\]
If we further pick $m=2n$ (twice the dataset size) the factor becomes
\[
    \rho = \frac{1}{1+\frac{2\mu n}{3L}}.
\]
Consequently the total number of stochastic gradient evaluations needed to achieve
$\E[f(\tilde{x})-f(x^\star)]\le\varepsilon$ is
\[
    \mathcal{O}\!\bigl((n+m)\log(1/\varepsilon)\bigr)
    \;=\;
    \mathcal{O}\!\bigl(n\log(1/\varepsilon)\bigr),
\]
matching the optimal complexity for strongly convex finite‑sum problems.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Pure gradient descent.} If we take $m=1$ and use the full gradient estimator (i.e., $v_t=\nabla f(\tilde{x})$), then $\rho = \frac{1}{1+\mu\eta(1-2L\eta)}$, which is exactly the classical gradient‑descent contraction.
\item \textbf{Large inner loop ($m\to\infty$).} As $m\to\infty$, $\rho\to0$ and the algorithm converges in a single epoch; this corresponds to repeatedly applying the exact gradient step with stepsize $\eta$.
\item \textbf{No variance reduction.} If we drop the correction term and set $v_t=\nabla f_{i_t}(x_t)$, the variance bound (A4) no longer holds; the proof collapses and we revert to the $O(1/t)$ rate of standard SGD.
\end{enumerate}
