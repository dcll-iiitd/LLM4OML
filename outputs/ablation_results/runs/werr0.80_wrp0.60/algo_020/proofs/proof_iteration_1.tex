\section*{Algorithm Name}
\textbf{Proximal Gradient Method (Forward–Backward Splitting)}  

The algorithm solves the composite convex problem  

\[
\min_{x\in\R^{d}} \; F(x)\;\stackrel{\text{def}}{=}\; f(x)+g(x),
\]

where  

\begin{itemize}[nosep]
  \item $f:\R^{d}\to\R$ is convex, differentiable and has an $L$‑Lipschitz continuous gradient,
  \item $g:\R^{d}\to\R\cup\{+\infty\}$ is convex (possibly nonsmooth) and its proximal operator
        $\prox_{\eta g}$ can be evaluated efficiently.
\end{itemize}

The method alternates a gradient step on $f$ with a proximal step on $g$.

\section*{Mathematical Formulation}
Given a stepsize $\eta>0$, the iterate $x_{k+1}$ is defined by  

\[
\boxed{
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\,\grad f(x_{k})\bigr)},
\qquad k=0,1,2,\dots
\tag{1}
\]

The proximal operator is  

\[
\prox_{\eta g}(u)\;\stackrel{\text{def}}{=}\;
\arg\min_{z\in\R^{d}}\Bigl\{ g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Bigr\}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$.
\For{$k=0,1,2,\dots$}
    \State $u_{k}=x_{k}-\eta\,\grad f(x_{k})$ \Comment{gradient step}
    \State $x_{k+1}= \prox_{\eta g}(u_{k})$ \Comment{proximal step}
\EndFor
\end{algorithmic}
\end{algorithm}

The algorithm stops when a user‑specified criterion is satisfied (e.g. 
$\|x_{k+1}-x_{k}\|\leq\varepsilon$ or a maximum number of iterations).

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so $\prox_{\eta g}= \operatorname{id}$.
Take $w_{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\grad f(w_{k})$ & $w_{k+1}=w_{k}-\eta\grad f(w_{k})$ \\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}

Objective values: $F(w_{0})=9$, $F(w_{1})=(0.6-3)^{2}=5.76$, $F(w_{2})\approx4.32$, $F(w_{3})\approx3.54$.
The iterates move toward the minimiser $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{A:smooth}
$f$ is convex and differentiable with $L$‑Lipschitz gradient, i.e.  

\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]

Equivalently, for all $x,y$  

\[
f(y)\le f(x)+\langle\grad f(x),\,y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Convexity of $g$]\label{A:convex}
$g$ is proper, closed and convex. Its proximal operator (2) is therefore
well‑defined for any $\eta>0$.
\tag{A3}
\end{assumption}

\begin{assumption}[Stepsize]\label{A:stepsize}
The stepsize satisfies  

\[
0<\eta\le\frac{1}{L}.
\tag{A4}
\]
\end{assumption}

All results below are derived under \ref{A:smooth}–\ref{A:stepsize}.

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear $O(1/k)$ convergence]\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{1} with a stepsize $\eta\in(0,1/L]$.
For any optimal solution $x^{\star}$ of $\min_{x}F(x)$ we have  

\[
F(x_{k})-F(x^{\star})
\;\le\;
\frac{1}{2\eta k}\,\|x_{0}-x^{\star}\|^{2},
\qquad\forall k\ge 1.
\tag{3}
\]

Consequently $F(x_{k})\to F(x^{\star})$ at the rate $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
  \item \textbf{Stability.} The algorithm is non‑expansive in the sense that
        $\|x_{k+1}-x^{\star}\|\le\|x_{k}-x^{\star}\|$ for any $x^{\star}\in\arg\min F$.
  \item \textbf{Hyper‑parameter sensitivity.}
        The stepsize bound $\eta\le 1/L$ is tight; larger $\eta$ may violate the
        descent inequality (see Lemma~\ref{lem:descent}).
  \item \textbf{Relation to baselines.}
        \begin{enumerate}[nosep]
          \item If $g\equiv0$, \eqref{1} reduces to classical gradient descent
                and Theorem~\ref{thm:rate} recovers the standard $O(1/k)$ rate.
          \item If $f\equiv0$, \eqref{1} becomes the proximal point method,
                which also satisfies \eqref{3} with $\eta$ playing the role of
                a regularisation parameter.
        \end{enumerate}
\end{itemize}

\section*{Discussion}
The proof hinges on two elementary facts:

\begin{enumerate}[label=\arabic*.]
  \item \emph{Descent Lemma} for $f$ (smoothness) – inequality (A2).
  \item \emph{Non‑expansiveness} of the proximal operator:
        \[
        \|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|
        \le \|u-v\|,\qquad\forall u,v\in\R^{d}.
        \tag{NE}
        \]
\end{enumerate}
Combining (NE) with the optimality condition of the proximal step yields a
one‑step inequality that telescopes after summation, giving the $O(1/k)$ bound.
The analysis is deterministic; no stochasticity or expectation is involved.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Proximal optimality condition]\label{lem:prox-opt}
For any $u\in\R^{d}$ and $\eta>0$, $z^{+}= \prox_{\eta g}(u)$ satisfies  

\[
0\in \partial g(z^{+})+\frac{1}{\eta}(z^{+}-u),
\tag{4}
\]
or equivalently  

\[
\langle u-z^{+},\,z-z^{+}\rangle\;\ge\;\eta\bigl[g(z)-g(z^{+})\bigr],
\qquad\forall z\in\R^{d}.
\tag{5}
\]
\end{lemma}
\begin{proof}
The definition \eqref{2} is a strongly convex minimisation problem; the
first‑order optimality condition yields (4). Multiplying (4) by $\eta$ and
rearranging gives (5) after using the subgradient inequality for $g$.
\end{proof}

\begin{lemma}[Descent inequality for the proximal gradient step]\label{lem:descent}
Let $x_{k+1}$ be defined by \eqref{1} with $\eta\le 1/L$. Then for any $x\in\R^{d}$  

\[
F(x_{k+1})-F(x)
\;\le\;
\frac{1}{2\eta}\Bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}\Bigr).
\tag{6}
\]
\end{lemma}
\begin{proof}
Set $u_{k}=x_{k}-\eta\grad f(x_{k})$ so that $x_{k+1}= \prox_{\eta g}(u_{k})$.
Apply Lemma~\ref{lem:prox-opt} with $z^{+}=x_{k+1}$, $u=u_{k}$ and arbitrary $x$:

\[
\langle u_{k}-x_{k+1},\,x-x_{k+1}\rangle
\;\ge\;
\eta\bigl[g(x)-g(x_{k+1})\bigr].
\tag{7}
\]

Rewrite $u_{k}-x_{k+1}=x_{k}-\eta\grad f(x_{k})-x_{k+1}$ and expand the inner product:

\[
\begin{aligned}
\langle x_{k}-x_{k+1},\,x-x_{k+1}\rangle
&-\eta\langle\grad f(x_{k}),\,x-x_{k+1}\rangle\\
&\ge\eta\bigl[g(x)-g(x_{k+1})\bigr].
\end{aligned}
\tag{8}
\]

Using the identity $\langle a-b,c-b\rangle
=\frac{1}{2}\bigl(\|a-b\|^{2}+\|c-b\|^{2}-\|a-c\|^{2}\bigr)$ with
$a=x_{k}$, $b=x_{k+1}$, $c=x$, we obtain  

\[
\frac{1}{2}\bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr)
-\eta\langle\grad f(x_{k}),\,x-x_{k+1}\rangle
\ge\eta\bigl[g(x)-g(x_{k+1})\bigr].
\tag{9}
\]

Rearrange (9) to isolate $g(x)-g(x_{k+1})$ and add $f(x)-f(x_{k+1})$ on both sides:

\[
\begin{aligned}
&f(x)-f(x_{k+1})+g(x)-g(x_{k+1})\\
&\qquad\le
\frac{1}{2\eta}\bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}\bigr)
-\frac{1}{2\eta}\|x_{k}-x_{k+1}\|^{2}
+\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\rangle .
\end{aligned}
\tag{10}
\]

Now invoke the smoothness inequality (A2) with $y=x_{k+1}$ and $x=x_{k}$:

\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{11}
\]

Rearrange (11) to get  

\[
\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\rangle
\ge f(x_{k+1})-f(x_{k})-\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{12}
\]

Insert (12) into (10) and collect the terms containing $\|x_{k+1}-x_{k}\|^{2}$:

\[
\begin{aligned}
F(x)-F(x_{k+1})
&\le
\frac{1}{2\eta}\bigl(\|x_{k}-x\|^{2}-\|x_{k+1}-x\|^{2}\bigr)\\
&\quad-\Bigl(\frac{1}{2\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{13}
\]

Because of the stepsize condition $\eta\le 1/L$, the coefficient of
$\|x_{k+1}-x_{k}\|^{2}$ is non‑negative, so we can drop that (non‑negative)
term to obtain the desired inequality (6). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:rate} using Lemma~\ref{lem:descent}.  
All non‑trivial manipulations are numbered for easy reference.

\begin{proof}
Let $x^{\star}$ be any optimal solution of $\min F$.  
Apply inequality \eqref{6} of Lemma~\ref{lem:descent} with $x=x^{\star}$:

\[
F(x_{k+1})-F(x^{\star})
\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}
      -\|x_{k+1}-x^{\star}\|^{2}\bigr).
\tag{Step 1}
\]

Summing (Step 1) from $k=0$ to $k=K-1$ telescopes the right‑hand side:

\[
\sum_{k=0}^{K-1}\!\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le
\frac{1}{2\eta}\bigl(\|x_{0}-x^{\star}\|^{2}
      -\|x_{K}-x^{\star}\|^{2}\bigr)
\le
\frac{1}{2\eta}\|x_{0}-x^{\star}\|^{2}.
\tag{Step 2}
\]

Since $F(x_{k})\ge F(x^{\star})$ for all $k$, each term in the sum is non‑negative,
hence the smallest term is bounded by the average:

\[
\min_{1\le k\le K}\bigl(F(x_{k})-F(x^{\star})\bigr)
\le
\frac{1}{K}\sum_{k=1}^{K}\!\bigl(F(x_{k})-F(x^{\star})\bigr)
\le
\frac{1}{2\eta K}\|x_{0}-x^{\star}\|^{2}.
\tag{Step 3}
\]

Because the sequence $\{F(x_{k})\}$ is non‑increasing (a direct consequence of
(6) with $x=x^{\star}$), we have for every $k\ge 1$

\[
F(x_{k})-F(x^{\star})
\le
\frac{1}{2\eta k}\|x_{0}-x^{\star}\|^{2},
\tag{Step 4}
\]

which is precisely the bound \eqref{3}. ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{3} we can read off the explicit constants:

\[
\boxed{
F(x_{k})-F(x^{\star})
\le
\frac{L}{2k}\,\|x_{0}-x^{\star}\|^{2}
\qquad\text{since }\eta\le\frac{1}{L}.
}
\tag{14}
\]

Thus the method attains an $O(L\|x_{0}-x^{\star}\|^{2}/k)$ suboptimality bound.

\section*{Special Cases}
\begin{enumerate}[label=\alph*.]
  \item \textbf{Pure gradient descent} ($g\equiv0$).  
        $\prox_{\eta g}=\operatorname{id}$, and \eqref{1} reduces to
        $x_{k+1}=x_{k}-\eta\grad f(x_{k})$.  
        Theorem~\ref{thm:rate} coincides with the classic $O(1/k)$ rate for smooth convex $f$.

  \item \textbf{Proximal point method} ($f\equiv0$).  
        The update becomes $x_{k+1}= \prox_{\eta g}(x_{k})$, and (6) simplifies to
        $g(x_{k+1})-g(x^{\star})\le\frac{1}{2\eta}
        (\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2})$,
        yielding the same $O(1/k)$ bound for nonsmooth convex $g$.

  \item \textbf{Accelerated variant}.  
        If Nesterov‑type extrapolation is added, the rate improves to $O(1/k^{2})$,
        but the proof then requires additional Lyapunov functions and is omitted here.
\end{enumerate}
