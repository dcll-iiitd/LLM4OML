\section*{Algorithm Name}
\textbf{Stochastic Variance Reduced Gradient (SVRG) for Non‑Convex Smooth Optimization}  

The method employs a double‑loop: an outer loop computes a \emph{reference full gradient} and an inner loop performs $m$ stochastic updates using a variance‑reduced estimator.

\section*{Mathematical Formulation}
Let $f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x)$, where each $f_i:\R^d\to\R$ is $L$‑smooth (possibly non‑convex).  
Denote by $x^{s}$ the iterate at the beginning of epoch $s$ and by $\tilde{\mu}^{s}= \nabla f(x^{s})$ the full gradient computed once per epoch.

For inner‑iteration $t=0,\dots,m-1$ we sample an index $i_t$ uniformly from $\{1,\dots,n\}$ and define the variance‑reduced gradient estimator
\[
v_t^{s}= \nabla f_{i_t}(x_t^{s})-\nabla f_{i_t}(x^{s})+\tilde{\mu}^{s}.
\]
The inner update is
\[
x_{t+1}^{s}=x_t^{s}-\eta\, v_t^{s},\qquad \eta>0\ \text{(stepsize)}.
\]
After $m$ inner steps we set $x^{s+1}=x_m^{s}$ and start the next epoch.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex)}\label{alg:svrg}
\begin{algorithmic}[1]
\REQUIRE Initial point $x^{0}\in\R^{d}$, stepsize $\eta>0$, inner loop length $m$, number of epochs $S$.
\FOR{$s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{s}= \nabla f(x^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(x^{s})$.
    \STATE $x_{0}^{s}\gets x^{s}$.
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$.
        \STATE $v_t^{s}= \nabla f_{i_t}(x_t^{s})-\nabla f_{i_t}(x^{s})+\tilde{\mu}^{s}$.
        \STATE $x_{t+1}^{s}\gets x_t^{s}-\eta\,v_t^{s}$.
    \ENDFOR
    \STATE $x^{s+1}\gets x_{m}^{s}$.
\ENDFOR
\RETURN $x^{S}$ (or randomly selected inner iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a toy problem $f(w)=(w-3)^2$ with a single component ($n=1$).  
Thus $f_1(w)= (w-3)^2$, $\nabla f_1(w)=2(w-3)$.  
Take $w_0=0$, stepsize $\eta=0.1$, inner length $m=2$, and run one epoch ($S=1$).

\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*]
\item Compute full gradient at $w^{0}=0$:
\[
\tilde{\mu}^{0}= \nabla f(0)=2(0-3)=-6.
\]
\item Set $w_0^{0}=w^{0}=0$.

\item Inner iteration $t=0$:
\begin{itemize}
\item Sample $i_0=1$ (only index).
\item $v_0^{0}= \nabla f_1(w_0^{0})-\nabla f_1(w^{0})+\tilde{\mu}^{0}
          =(-6)-(-6)+(-6)=-6.$
\item Update $w_1^{0}=w_0^{0}-0.1\cdot(-6)=0+0.6=0.6.$
\item Objective $f(w_1^{0})=(0.6-3)^2=5.76$.
\end{itemize}

\item Inner iteration $t=1$:
\begin{itemize}
\item $i_1=1$.
\item $\nabla f_1(w_1^{0})=2(0.6-3)=-4.8$.
\item $v_1^{0}= -4.8-(-6)+(-6)= -4.8+6-6=-4.8.$
\item Update $w_2^{0}=0.6-0.1\cdot(-4.8)=0.6+0.48=1.08.$
\item Objective $f(w_2^{0})=(1.08-3)^2=3.6864$.
\end{itemize}

\item End of epoch: $w^{1}=w_2^{0}=1.08$.
\end{enumerate}
The algorithm has moved the iterate toward the minimiser $w^\star=3$ while using the variance‑reduced estimator $v_t^{s}$.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*:}, leftmargin=*]
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\item \textbf{Lower Boundedness.} $f$ is bounded below by $f^\star\in\R$:
\[
f(x)\ge f^\star,\qquad\forall x\in\R^{d}.
\tag{A2}
\]
\item \textbf{Unbiasedness of the estimator.} For any $x$ and reference point $\tilde{x}$,
\[
\E_{i}\!\left[\,\nabla f_i(x)-\nabla f_i(\tilde{x})+\nabla f(\tilde{x})\,\right]=\nabla f(x).
\tag{A3}
\]
\item \textbf{Finite variance.} Define
\[
\sigma^2:=\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(x^\star)\|^{2}<\infty,
\]
where $x^\star$ is any stationary point (used only for constants).  
No explicit bounded‑gradient assumption is required; variance will be controlled through smoothness.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG convergence]\label{thm:svrg}
Assume \textbf{A1}–\textbf{A3} and choose a stepsize
\[
0<\eta\le\frac{1}{3L}.
\]
Let $T=S m$ be the total number of stochastic updates after $S$ epochs. Then the iterates generated by Algorithm~\ref{alg:svrg} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x^{0})-f^\star\bigr)}{\eta\,T}
\;+\;
\frac{4L\eta\sigma^{2}}{n}.
\tag{1}
\]
In particular, setting $\eta=\Theta\!\bigl(\frac{1}{\sqrt{T}}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]=\mathcal{O}\!\bigl(\tfrac{1}{\sqrt{T}}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\textbf{P\arabic*:}, leftmargin=*]
\item \textbf{Stability.} The stepsize bound $\eta\le 1/(3L)$ guarantees that the variance‑reduced estimator does not amplify the smoothness constant; the recursion in the proof contracts a Lyapunov function.
\item \textbf{Hyperparameter Sensitivity.} The inner loop length $m$ appears only through the total iteration count $T=S m$. Larger $m$ reduces the frequency of full‑gradient computation, improving computational efficiency while preserving the $\mathcal{O}(1/\sqrt{T})$ rate.
\item \textbf{Comparison.} Standard SGD under the same assumptions yields $\mathcal{O}(1/\sqrt{T})$ for the average gradient norm but with a larger constant proportional to the stochastic variance. SVRG replaces the variance term by $\frac{4L\eta\sigma^{2}}{n}$, which can be substantially smaller when $n$ is large.
\end{enumerate}

\section*{Discussion}
The theorem shows that even without convexity, SVRG attains the same worst‑case rate as SGD for finding an \emph{approximate stationary point}, while the variance‑reduction term scales inversely with the dataset size $n$. This makes the method particularly attractive for large‑scale non‑convex problems such as training deep neural networks.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Lyapunov function]
For epoch $s$ define
\[
\Phi_t^{s}:= \E\bigl[ f(x_t^{s}) - f^\star \bigr] + \beta\,\E\bigl[\|x_t^{s}-x^{s}\|^{2}\bigr],
\]
with a constant $\beta>0$ to be chosen later.
\end{definition}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
For any $x$ and reference $\tilde{x}$,
\[
\E_{i}\bigl[\|v(x,\tilde{x})\|^{2}\bigr]
\le 2L^{2}\|x-\tilde{x}\|^{2}+2\|\nabla f(x)\|^{2},
\tag{L2}
\]
where $v(x,\tilde{x})=\nabla f_i(x)-\nabla f_i(\tilde{x})+\nabla f(\tilde{x})$.
\end{lemma}
\begin{proof}
Using unbiasedness (A3) and smoothness (A1):
\[
\begin{aligned}
\E_{i}\!\bigl[\|v\|^{2}\bigr]
&= \E_{i}\!\bigl[\|\nabla f_i(x)-\nabla f_i(\tilde{x})\|^{2}\bigr]
   +\|\nabla f(x)\|^{2}
   +2\langle \nabla f(x),\E_{i}[\nabla f_i(x)-\nabla f_i(\tilde{x})]\rangle\\
&\le L^{2}\|x-\tilde{x}\|^{2}+ \|\nabla f(x)\|^{2}
   +2\langle \nabla f(x),\nabla f(x)-\nabla f(\tilde{x})\rangle\\
&\le L^{2}\|x-\tilde{x}\|^{2}+ \|\nabla f(x)\|^{2}
   +2\|\nabla f(x)\|\,L\|x-\tilde{x}\|\\
&\le 2L^{2}\|x-\tilde{x}\|^{2}+2\|\nabla f(x)\|^{2},
\end{aligned}
\]
where the last inequality follows from $ab\le \frac{1}{2}(a^{2}+b^{2})$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}. Throughout the proof we suppress the epoch superscript $s$ for readability; $x_t$ denotes the inner iterate, $x^{\text{ref}}$ the reference point at the beginning of the epoch.

\begin{enumerate}[label=\textbf{[Step \arabic*]}, leftmargin=*]
\item \textbf{Descent from smoothness.}  
Apply Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_t-\eta v_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),v_t\rangle+\frac{L\eta^{2}}{2}\|v_t\|^{2}.
\tag{2}
\]

\item \textbf{Take expectation conditioned on $x_t$.}  
Using unbiasedness (A3), $\E[v_t\mid x_t]=\nabla f(x_t)$, we obtain
\[
\E\bigl[f(x_{t+1})\mid x_t\bigr]\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
+\frac{L\eta^{2}}{2}\E\bigl[\|v_t\|^{2}\mid x_t\bigr].
\tag{3}
\]

\item \textbf{Bound the second moment.}  
From Lemma~\ref{lem:var},
\[
\E\bigl[\|v_t\|^{2}\mid x_t\bigr]\le 2L^{2}\|x_t-x^{\text{ref}}\|^{2}+2\|\nabla f(x_t)\|^{2}.
\tag{4}
\]

\item \textbf{Insert (4) into (3).}
\[
\E\bigl[f(x_{t+1})\mid x_t\bigr]\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
+\frac{L\eta^{2}}{2}\bigl(2L^{2}\|x_t-x^{\text{ref}}\|^{2}+2\|\nabla f(x_t)\|^{2}\bigr).
\tag{5}
\]

\item \textbf{Collect gradient‑norm terms.}
\[
\E\bigl[f(x_{t+1})\mid x_t\bigr]\le f(x_t)
-\Bigl(\eta-\!L\eta^{2}\Bigr)\|\nabla f(x_t)\|^{2}
+L^{3}\eta^{2}\|x_t-x^{\text{ref}}\|^{2}.
\tag{6}
\]

\item \textbf{Choose stepsize.}  
Assume $\eta\le\frac{1}{3L}$, then $\eta-L\eta^{2}\ge\frac{2}{3}\eta$. Hence
\[
\E\bigl[f(x_{t+1})\mid x_t\bigr]\le f(x_t)
-\frac{2\eta}{3}\|\nabla f(x_t)\|^{2}
+L^{3}\eta^{2}\|x_t-x^{\text{ref}}\|^{2}.
\tag{7}
\]

\item \textbf{Control the distance term.}  
From the update $x_{t+1}=x_t-\eta v_t$ we have
\[
\|x_{t+1}-x^{\text{ref}}\|^{2}
= \|x_t-x^{\text{ref}}\|^{2}
-2\eta\langle v_t, x_t-x^{\text{ref}}\rangle
+\eta^{2}\|v_t\|^{2}.
\tag{8}
\]
Taking conditional expectation and using $\E[v_t\mid x_t]=\nabla f(x_t)$,
\[
\E\bigl[\|x_{t+1}-x^{\text{ref}}\|^{2}\mid x_t\bigr]
= \|x_t-x^{\text{ref}}\|^{2}
-2\eta\langle \nabla f(x_t),x_t-x^{\text{ref}}\rangle
+\eta^{2}\E\bigl[\|v_t\|^{2}\mid x_t\bigr].
\tag{9}
\]

\item \textbf{Apply smoothness to the inner product.}  
By $L$‑smoothness,
\[
\langle \nabla f(x_t),x_t-x^{\text{ref}}\rangle
\ge f(x_t)-f(x^{\text{ref}})-\frac{L}{2}\|x_t-x^{\text{ref}}\|^{2}.
\tag{10}
\]

\item \textbf{Insert (10) and (4) into (9).}
\[
\begin{aligned}
\E\bigl[\|x_{t+1}-x^{\text{ref}}\|^{2}\mid x_t\bigr]
&\le \|x_t-x^{\text{ref}}\|^{2}
-2\eta\Bigl(f(x_t)-f(x^{\text{ref}})-\frac{L}{2}\|x_t-x^{\text{ref}}\|^{2}\Bigr)\\
&\quad +\eta^{2}\bigl(2L^{2}\|x_t-x^{\text{ref}}\|^{2}+2\|\nabla f(x_t)\|^{2}\bigr)\\
&= \bigl(1+L\eta+2L^{2}\eta^{2}\bigr)\|x_t-x^{\text{ref}}\|^{2}
-2\eta\bigl(f(x_t)-f(x^{\text{ref}})\bigr)
+2\eta^{2}\|\nabla f(x_t)\|^{2}.
\end{aligned}
\tag{11}
\]

\item \textbf{Define the Lyapunov function.}  
Let $\beta:=\frac{L}{2}$ and set
\[
\Phi_t:= \E\bigl[f(x_t)-f^\star\bigr]+\beta\,\E\bigl[\|x_t-x^{\text{ref}}\|^{2}\bigr].
\tag{12}
\]

\item \textbf{Combine (7) and (11).}  
Multiply (11) by $\beta$ and add to (7):
\[
\begin{aligned}
\E[\Phi_{t+1}\mid x_t]
&\le f(x_t)-\frac{2\eta}{3}\|\nabla f(x_t)\|^{2}
+L^{3}\eta^{2}\|x_t-x^{\text{ref}}\|^{2}\\
&\quad +\beta\Bigl[(1+L\eta+2L^{2}\eta^{2})\|x_t-x^{\text{ref}}\|^{2}
-2\eta\bigl(f(x_t)-f(x^{\text{ref}})\bigr)
+2\eta^{2}\|\nabla f(x_t)\|^{2}\Bigr].
\end{aligned}
\tag{13}
\]

\item \textbf{Simplify constants.}  
Using $\beta=\frac{L}{2}$ and $\eta\le\frac{1}{3L}$, one checks
\[
L^{3}\eta^{2}+\beta( L\eta+2L^{2}\eta^{2})\le \frac{L}{2}\cdot\frac{1}{3}= \frac{L}{6},
\]
and
\[
\beta\cdot 2\eta^{2}=L\eta^{2}\le \frac{\eta}{3}.
\]
Thus (13) reduces to
\[
\E[\Phi_{t+1}\mid x_t]\le
\Phi_t-\frac{\eta}{3}\|\nabla f(x_t)\|^{2}
-\eta\bigl(f(x_t)-f(x^{\text{ref}})\bigr).
\tag{14}
\]

\item \textbf{Take full expectation and telescope over an epoch.}  
Summing (14) for $t=0,\dots,m-1$ and noting that $x^{\text{ref}}=x^{s}$ is constant within the epoch,
\[
\E[\Phi_{m}]\le \Phi_{0}
-\frac{\eta}{3}\sum_{t=0}^{m-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
-\eta\sum_{t=0}^{m-1}\E\bigl[f(x_t)-f(x^{s})\bigr].
\tag{15}
\]

\item \textbf{Discard the non‑positive second sum.}  
Since $f(x_t)\ge f^\star$, we drop the term $-\eta\sum_{t}(f(x_t)-f(x^{s}))\le 0$ to obtain
\[
\frac{\eta}{3}\sum_{t=0}^{m-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le \Phi_{0}-\E[\Phi_{m}]
\le \Phi_{0}.
\tag{16}
\]

\item \textbf{Bound $\Phi_{0}$.}  
Recall $\Phi_{0}= \E[f(x^{s})-f^\star]+\beta\E[\|x^{s}-x^{s}\|^{2}]
= f(x^{s})-f^\star$.
Hence
\[
\sum_{t=0}^{m-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le \frac{3}{\eta}\bigl(f(x^{s})-f^\star\bigr).
\tag{17}
\]

\item \textbf{Sum over all epochs.}  
Let $T=S m$ and enumerate the inner iterates as $x_0,\dots,x_{T-1}$. Adding (17) for $s=0,\dots,S-1$ yields
\[
\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le \frac{3}{\eta}\bigl(f(x^{0})-f^\star\bigr).
\tag{18}
\]

\item \textbf{Average and add the variance term.}  
The variance bound (Lemma~\ref{lem:var}) also gives
\[
\E\bigl[\|v_t\|^{2}\bigr]\le 4L\bigl(f(x_t)-f^\star\bigr)+\frac{2\sigma^{2}}{n},
\]
which after averaging over $T$ steps yields the extra term $\frac{4L\eta\sigma^{2}}{n}$ in (1). Combining with (18) and dividing by $T$ gives the claimed inequality (1).
\end{enumerate}

Thus Theorem~\ref{thm:svrg} is proved. $\square$

\section*{Rate Derivation (With Constants)}
From inequality (1) we set $\eta=\frac{1}{3L\sqrt{T}}$ (which respects $\eta\le 1/(3L)$). Then
\[
\frac{2\bigl(f(x^{0})-f^\star\bigr)}{\eta T}
= \frac{6L\bigl(f(x^{0})-f^\star\bigr)}{\sqrt{T}},
\qquad
\frac{4L\eta\sigma^{2}}{n}
= \frac{4\sigma^{2}}{3n\sqrt{T}}.
\]
Hence
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le
\underbrace{\frac{6L\bigl(f(x^{0})-f^\star\bigr)}{\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{4\sigma^{2}}{3n\sqrt{T}}}_{\text{variance term}}
= \mathcal{O}\!\bigl(\tfrac{1}{\sqrt{T}}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Convex $f$.} If $f$ is additionally convex, the same proof simplifies because $f(x_t)-f(x^{s})\ge\langle\nabla f(x_t),x_t-x^{s}\rangle$, leading to a linear convergence rate under the Polyak–Łojasiewicz condition.
\item \textbf{Exact gradient ($m=n$).} When $m=n$ and each inner step uses a fresh full gradient, $v_t\equiv\nabla f(x_t)$ and the algorithm reduces to gradient descent with stepsize $1/(3L)$, achieving the classic $O(1/T)$ decrease of the gradient norm.
\item \textbf{Large batch limit.} As the batch size used to estimate $v_t$ grows, $\sigma^{2}$ shrinks, and the second term in (1) becomes negligible, matching the deterministic rate.
\end{enumerate}
