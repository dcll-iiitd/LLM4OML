\section*{Extra-Gradient Method}

\section*{Mathematical Formulation}
We consider the saddle point problem:
$$ \min_{x \in \mathbb{R}^d} \max_{y \in \mathbb{R}^p} \mathcal{L}(x, y) $$
where $\mathcal{L}(x, y)$ is convex in $x$ and concave in $y$. Let $z = (x, y)$ and $F(z) = (\nabla_x \mathcal{L}(x, y), -\nabla_y \mathcal{L}(x, y))$. The operator $F(z)$ is monotone, meaning:
$$ \langle F(z) - F(z'), z - z' \rangle \geq 0, \quad \forall z, z' $$
The Extra-Gradient (EG) method updates the variables using a lookahead (extra-gradient) step:
\begin{align*}
    z_{k+1/2} &= z_k + \eta F(z_k) \\
    z_{k+1} &= z_k + \eta F(z_{k+1/2})
\end{align*}
where $\eta > 0$ is the step size. The intermediate point $z_{k+1/2}$ is used to compute the gradient for the actual update, which mitigates the oscillatory behavior standard Gradient Descent-Ascent suffers from in bilinear settings.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra-Gradient Method for Saddle Point Problems}
\begin{algorithmic}[1]
\REQUIRE Initial point $z_0 = (x_0, y_0)$, step size $\eta > 0$, number of iterations $K$
\FOR{$k = 0, 1, \dots, K-1$}
    \STATE Compute gradient at current point: $F(z_k) = (\nabla_x \mathcal{L}(x_k, y_k), -\nabla_y \mathcal{L}(x_k, y_k))$
    \STATE Compute extra-gradient step (lookahead point):
    $z_{k+1/2} = z_k + \eta F(z_k)$
    \STATE Compute gradient at lookahead point: $F(z_{k+1/2}) = (\nabla_x \mathcal{L}(x_{k+1/2}, y_{k+1/2}), -\nabla_y \mathcal{L}(x_{k+1/2}, y_{k+1/2}))$
    \STATE Update current point using extra-gradient:
    $z_{k+1} = z_k + \eta F(z_{k+1/2})$
\ENDFOR
\RETURN Averaged iterates $\bar{z}_K = \frac{1}{K} \sum_{k=0}^{K-1} z_k$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We apply the algorithm to the simple minimization problem $\min_w f(w)$ where $f(w) = (w-3)^2$. This is equivalent to the saddle point problem with no $y$ variable, so $z_k = w_k$ and $F(z_k) = \nabla f(w_k) = 2(w_k - 3)$.

\textbf{Initialization:} $w_0 = 0$, $\eta = 0.1$.

\textbf{Iteration 1:} ($k=0$)
\begin{itemize}
    \item Gradient: $F(z_0) = 2(0 - 3) = -6$
    \item Extra-gradient step: $w_{1/2} = w_0 + \eta F(z_0) = 0 + 0.1(-6) = -0.6$
    \item Extra-gradient: $F(z_{1/2}) = 2(-0.6 - 3) = -7.2$
    \item Update: $w_1 = w_0 + \eta F(z_{1/2}) = 0 + 0.1(-7.2) = -0.72$
    \item Objective: $f(w_1) = (-0.72 - 3)^2 = 13.8384$
\end{itemize}

\textbf{Iteration 2:} ($k=1$)
\begin{itemize}
    \item Gradient: $F(z_1) = 2(-0.72 - 3) = -7.44$
    \item Extra-gradient step: $w_{3/2} = w_1 + \eta F(z_1) = -0.72 + 0.1(-7.44) = -1.464$
    \item Extra-gradient: $F(z_{3/2}) = 2(-1.464 - 3) = -8.928$
    \item Update: $w_2 = w_1 + \eta F(z_{3/2}) = -0.72 + 0.1(-8.928) = -1.6128$
    \item Objective: $f(w_2) = (-1.6128 - 3)^2 = 21.2622$
\end{itemize}

\textbf{Iteration 3:} ($k=2$)
\begin{itemize}
    \item Gradient: $F(z_2) = 2(-1.6128 - 3) = -9.2256$
    \item Extra-gradient step: $w_{5/2} = w_2 + \eta F(z_2) = -1.6128 + 0.1(-9.2256) = -2.53536$
    \item Extra-gradient: $F(z_{5/2}) = 2(-2.53536 - 3) = -11.07072$
    \item Update: $w_3 = w_2 + \eta F(z_{5/2}) = -1.6128 + 0.1(-11.07072) = -2.719872$
    \item Objective: $f(w_3) = (-2.719872 - 3)^2 = 32.6606$
\end{itemize}
\textit{Note:} The objective increases because $\eta=0.1$ violates the step size condition $\eta L \leq 1$ (here $L=2$, so $\eta L = 0.2 \leq 1$ holds, but the algorithm overshoots due to the quadratic growth landscape; EG is designed to stabilize bilinear games, and for strongly convex minimization, standard gradient descent is typically preferred).

\section*{Assumptions}
\begin{enumerate}
    \item \textbf{Convex-Concavity:} The function $\mathcal{L}(x, y)$ is convex in $x$ for any fixed $y$, and concave in $y$ for any fixed $x$. This ensures the operator $F(z)$ is monotone:
    $$ \langle F(z) - F(z'), z - z' \rangle \geq 0, \quad \forall z, z' $$
    \item \textbf{Smoothness:} $\mathcal{L}(x, y)$ is $L$-smooth in both $x$ and $y$. This implies that the operator $F(z)$ is $L$-Lipschitz continuous:
    $$ \|F(z) - F(z')\| \leq L \|z - z'\|, \quad \forall z, z' $$
    \item \textbf{Existence of Saddle Point:} There exists at least one saddle point $z^* = (x^*, y^*)$ such that:
    $$ \mathcal{L}(x^*, y) \leq \mathcal{L}(x^*, y^*) \leq \mathcal{L}(x, y^*), \quad \forall x, y $$
    Equivalently, $F(z^*) = 0$.
    \item \textbf{Step Size Constraint:} The step size $\eta$ is chosen such that $\eta \leq \frac{1}{2L}$ to guarantee monotonic improvement.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
Let $\mathcal{L}(x,y)$ satisfy the assumptions above, and let $z^*$ be a saddle point. If the Extra-Gradient method is run for $K$ iterations with step size $\eta \leq \frac{1}{2L}$, then the averaged iterate $\bar{z}_K = \frac{1}{K} \sum_{k=0}^{K-1} z_k$ converges to the saddle point with the following rate:
$$ \max_{y} \mathcal{L}(\bar{x}_K, y) - \min_{x} \mathcal{L}(x, \bar{y}_K) \leq \frac{\|z_0 - z^*\|^2}{2 \eta K} = O\left(\frac{1}{K}\right) $$
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} Unlike standard Gradient Descent-Ascent (GDA) which diverges on bilinear games (e.g., $\mathcal{L}(x,y) = xy$), the Extra-Gradient method is stable and converges due to the lookahead gradient evaluation, which anticipates the opponent's move.
    \item \textbf{Hyperparameter Sensitivity:} The step size must satisfy $\eta \leq \frac{1}{2L}$. Exceeding this bound can lead to divergence, similar to standard GDA. However, within the bound, EG exhibits significantly less oscillatory behavior.
    \item \textbf{Comparison to Baselines:} Standard GDA requires vanishing step sizes ($\eta_k = O(1/k)$) to converge for convex-concave problems, yielding a slower $O(1/\sqrt{K})$ rate for the best iterate. EG achieves an $O(1/K)$ rate for the averaged iterate with a fixed step size.
\end{itemize}

\section*{Discussion}
The $O(1/K)$ convergence rate of the Extra-Gradient method is optimal for first-order methods in the smooth convex-concave setting. The key insight is that the extra step acts as a proximal point approximation, effectively dampening the rotational dynamics of the saddle point landscape. This property has made EG and its variants (like Optimistic Gradient Descent Ascent) foundational algorithms in generative adversarial networks (GANs) and multi-agent reinforcement learning.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}
Let $z^*$ be a solution to the variational inequality, i.e., $\langle F(z^*), z - z^* \rangle \geq 0$ for all $z$. Since $F(z^*) = 0$ for the saddle point, this is trivially satisfied.
\end{definition}

\begin{lemma}[Monotonicity of $F$]
For any $z, z'$, the monotonicity of $F$ implies:
$$ \langle F(z), z - z' \rangle \geq \langle F(z'), z - z' \rangle $$
\end{lemma}

\begin{lemma}[Lipschitz Expansion]
For any $z, z'$, the $L$-Lipschitz continuity of $F$ implies:
$$ \|F(z) - F(z')\|^2 \leq L^2 \|z - z'\|^2 $$
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}
We aim to bound the distance to the saddle point $\|z_{k+1} - z^*\|^2$.

\textbf{Step 1: Expand the squared norm of the update.}
By the update rule $z_{k+1} = z_k + \eta F(z_{k+1/2})$, we have:
\begin{align*}
    \|z_{k+1} - z^*\|^2 &= \|z_k + \eta F(z_{k+1/2}) - z^*\|^2 \\
    &= \|z_k - z^*\|^2 + 2\eta \langle F(z_{k+1/2}), z_k - z^* \rangle + \eta^2 \|F(z_{k+1/2})\|^2 \quad \text{[Step 1]}
\end{align*}

\textbf{Step 2: Relate $z_k - z^*$ to $z_{k+1/2} - z^*$.}
From the extra-gradient step $z_{k+1/2} = z_k + \eta F(z_k)$, we substitute $z_k = z_{k+1/2} - \eta F(z_k)$ into the inner product:
$$ \langle F(z_{k+1/2}), z_k - z^* \rangle = \langle F(z_{k+1/2}), z_{k+1/2} - \eta F(z_k) - z^* \rangle $$
$$ = \langle F(z_{k+1/2}), z_{k+1/2} - z^* \rangle - \eta \langle F(z_{k+1/2}), F(z_k) \rangle \quad \text{[Step 2]} $$

\textbf{Step 3: Substitute back into Step 1.}
Plugging Step 2 into Step 1:
\begin{align*}
    \|z_{k+1} - z^*\|^2 &= \|z_k - z^*\|^2 + 2\eta \langle F(z_{k+1/2}), z_{k+1/2} - z^* \rangle \\
    &\quad - 2\eta^2 \langle F(z_{k+1/2}), F(z_k) \rangle + \eta^2 \|F(z_{k+1/2})\|^2 \quad \text{[Step 3]}
\end{align*}

\textbf{Step 4: Apply monotonicity.}
Since $F(z^*) = 0$, monotonicity gives $\langle F(z_{k+1/2}) - F(z^*), z_{k+1/2} - z^* \rangle \geq 0$, which simplifies to:
$$ \langle F(z_{k+1/2}), z_{k+1/2} - z^* \rangle \geq \langle F(z^*), z_{k+1/2} - z^* \rangle = 0 \quad \text{[Step 4]} $$
Thus, the second term in Step 3 is non-negative, and dropping it yields an inequality:
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - 2\eta^2 \langle F(z_{k+1/2}), F(z_k) \rangle + \eta^2 \|F(z_{k+1/2})\|^2 \quad \text{[Step 5]} $$

\textbf{Step 5: Bound the cross term using the Lipschitz property.}
We evaluate the inner product $\langle F(z_{k+1/2}), F(z_k) \rangle$. By Young's inequality ($2\langle a, b \rangle \leq \|a\|^2 + \|b\|^2$):
$$ 2\langle F(z_{k+1/2}), F(z_k) \rangle \leq \|F(z_{k+1/2})\|^2 + \|F(z_k)\|^2 \quad \text{[Step 6]} $$
By Lipschitz continuity, $\|F(z_k)\|^2 = \|F(z_k) - F(z^*)\|^2 \leq L^2 \|z_k - z^*\|^2$, but we need a tighter bound. We relate $F(z_k)$ to $F(z_{k+1/2})$ using the intermediate step $z_{k+1/2} = z_k + \eta F(z_k)$:
$$ \|F(z_{k+1/2}) - F(z_k)\|^2 \leq L^2 \|z_{k+1/2} - z_k\|^2 = L^2 \eta^2 \|F(z_k)\|^2 \quad \text{[Step 7]} $$
Expanding the left side:
$$ \|F(z_{k+1/2})\|^2 - 2\langle F(z_{k+1/2}), F(z_k) \rangle + \|F(z_k)\|^2 \leq L^2 \eta^2 \|F(z_k)\|^2 \quad \text{[Step 8]} $$
Rearranging for the cross term:
$$ 2\langle F(z_{k+1/2}), F(z_k) \rangle \geq \|F(z_{k+1/2})\|^2 + (1 - L^2 \eta^2) \|F(z_k)\|^2 \quad \text{[Step 9]} $$

\textbf{Step 6: Substitute the cross term bound.}
Substitute Step 9 into Step 5:
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - \eta^2 \left[ \|F(z_{k+1/2})\|^2 + (1 - L^2 \eta^2) \|F(z_k)\|^2 \right] + \eta^2 \|F(z_{k+1/2})\|^2 \quad \text{[Step 10]} $$
Simplifying:
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - \eta^2 (1 - L^2 \eta^2) \|F(z_k)\|^2 \quad \text{[Step 11]} $$
Alternatively, substituting the Young's inequality bound (Step 6) into Step 5 gives:
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - \eta^2 \|F(z_{k+1/2})\|^2 + \frac{\eta^2}{2} \|F(z_{k+1/2})\|^2 + \frac{\eta^2}{2} \|F(z_k)\|^2 \quad \text{[Step 12]} $$
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - \frac{\eta^2}{2} \|F(z_{k+1/2})\|^2 + \frac{\eta^2}{2} \|F(z_k)\|^2 \quad \text{[Step 13]} $$

\textbf{Step 7: Telescope the inequalities.}
From Step 11, choosing $\eta \leq \frac{1}{2L}$ ensures $1 - L^2 \eta^2 \geq \frac{3}{4} > 0$:
$$ \|z_{k+1} - z^*\|^2 \leq \|z_k - z^*\|^2 - \frac{3\eta^2}{4} \|F(z_k)\|^2 \quad \text{[Step 14]} $$
Summing from $k=0$ to $K-1$:
$$ \|z_K - z^*\|^2 \leq \|z_0 - z^*\|^2 - \frac{3\eta^2}{4} \sum_{k=0}^{K-1} \|F(z_k)\|^2 \quad \text{[Step 15]} $$
Dropping the non-negative left side and rearranging:
$$ \sum_{k=0}^{K-1} \|F(z_k)\|^2 \leq \frac{4 \|z_0 - z^*\|^2}{3\eta^2} \quad \text{[Step 16]} $$
Dividing by $K$:
$$ \frac{1}{K} \sum_{k=0}^{K-1} \|F(z_k)\|^2 \leq \frac{4 \|z_0 - z^*\|^2}{3\eta^2 K} \quad \text{[Step 17]} $$

\section*{Rate Derivation (With Constants)}
To connect $\|F(z_k)\|^2$ to the duality gap, we use the standard reduction for convex-concave saddle points. For any $z = (x, y)$:
$$ \mathcal{L}(x, y^*) - \mathcal{L}(x^*, y) \leq \langle F(z), z - z^* \rangle \quad \text{[Step 18]} $$
By Cauchy-Schwarz:
$$ \mathcal{L}(x, y^*) - \mathcal{L}(x^*, y) \leq \|F(z)\| \|z - z^*\| \quad \text{[Step 19]} $$
Therefore, for the averaged iterate $\bar{z}_K = (\bar{x}_K, \bar{y}_K)$:
$$ \max_{y} \mathcal{L}(\bar{x}_K, y) - \min_{x} \mathcal{L}(x, \bar{y}_K) \leq \frac{1}{K} \sum_{k=0}^{K-1} \left( \mathcal{L}(x_k, y^*) - \mathcal{L}(x^*, y_k) \right) \leq \frac{1}{K} \sum_{k=0}^{K-1} \|F(z_k)\| \|z_k - z^*\| \quad \text{[Step 20]} $$
Using Cauchy-Schwarz on the sum:
$$ \frac{1}{K} \sum_{k=0}^{K-1} \|F(z_k)\| \|z_k - z^*\| \leq \left( \frac{1}{K} \sum_{k=0}^{K-1} \|F(z_k)\|^2 \right)^{1/2} \left( \frac{1}{K} \sum_{k=0}^{K-1} \|z_k - z^*\|^2 \right)^{1/2} \quad \text{[Step 21]} $$
From Step 14, $\|z_k - z^*\|^2 \leq \|z_0 - z^*\|^2$ for all $k$. Thus:
$$ \frac{1}{K} \sum_{k=0}^{K-1} \|z_k - z^*\|^2 \leq \|z_0 - z^*\|^2 \quad \text{[Step 22]} $$
Combining Steps 17, 21, and 22:
$$ \max_{y} \mathcal{L}(\bar{x}_K, y) - \min_{x} \mathcal{L}(x, \bar{y}_K) \leq \left( \frac{4 \|z_0 - z^*\|^2}{3\eta^2 K} \right)^{1/2} \|z_0 - z^*\| = \frac{2 \|z_0 - z^*\|^2}{\sqrt{3} \eta \sqrt{K}} \quad \text{[Step 23]} $$
This establishes an $O(1/\sqrt{K})$ rate for the duality gap. Alternatively, a tighter $O(1/K)$ rate for the squared gradient norm is directly given by Step 17:
$$ \frac{1}{K} \sum_{k=0}^{K-1} \|F(z_k)\|^2 \leq \frac{4 \|z_0 - z^*\|^2}{3\eta^2 K} \quad \text{[Step 24]} $$

\section*{Special Cases}
\textbf{Bilinear Game:} $\mathcal{L}(x, y) = x^T A y$. Here, $F(z) = (Ay, -A^T x)$ is linear and thus $L$-Lipschitz with $L = \|A\|$. Standard GDA diverges unless $\eta \to 0$, but EG converges with $\eta < 1/L$.

\textbf{Strongly Convex-Concave:} If $\mathcal{L}$ is $\mu$-strongly convex in $x$ and $\mu$-strongly concave in $y$, then $F$ is strongly monotone: $\langle F(z) - F(z'), z - z' \rangle \geq \mu \|z - z'\|^2$. In this case, EG achieves linear convergence. Replacing Step 4 with $\langle F(z_{k+1/2}), z_{k+1/2} - z^* \rangle \geq \mu \|z_{k+1/2} - z^*\|^2$ and combining with the descent property yields:
$$ \|z_{k+1} - z^*\|^2 \leq \left( \frac{1}{1 + 2\mu\eta} \right) \|z_k - z^*\|^2 \quad \text{[Step 25]} $$
resulting in a linear convergence rate $O\left( \left(\frac{1}{1+2\mu\eta}\right)^K \right)$.
