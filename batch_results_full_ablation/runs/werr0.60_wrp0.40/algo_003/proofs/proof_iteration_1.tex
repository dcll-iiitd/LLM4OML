\section*{Algorithm Name}
\textbf{Mirror Descent with Bregman Divergence (MD)}  

The method minimizes a convex differentiable objective $f:\R^n\to\R$ by projecting a gradient step onto the geometry induced by a strongly convex prox‑function $h$ via its Bregman divergence.

\section*{Mathematical Formulation}
Let $h:\R^n\rightarrow\R$ be $\alpha$‑strongly convex w.r.t. a norm $\|\cdot\|$,
\[
\Breg(x,y)=h(x)-h(y)-\langle\nabla h(y),x-y\rangle\ge \frac{\alpha}{2}\|x-y\|^{2},\qquad\forall x,y.
\]

Given a stepsize sequence $\{\eta_k\}_{k\ge0}$, the MD iterate is
\begin{equation}
\boxed{
x_{k+1}= \arg\min_{x\in\R^{n}}\Big\{ \langle \nabla f(x_k),\,x\rangle
      +\frac{1}{\eta_k}\,\Breg(x,x_k) \Big\}
}
\label{eq:md-update}
\end{equation}
Equivalently, using the optimality condition,
\[
\nabla h(x_{k+1}) = \nabla h(x_k)-\eta_k \nabla f(x_k).
\tag{1}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent (MD)}
\begin{algorithmic}[1]
\State \textbf{Input:} convex $f$, prox‑function $h$, stepsizes $\{\eta_k\}_{k=0}^{T-1}$, initial point $x_0\in\operatorname{int}(\operatorname{dom}h)$
\State Compute $\nabla h(x_0)$
\For{$k=0$ \textbf{to} $T-1$}
    \State Compute gradient $g_k=\nabla f(x_k)$
    \State Update dual variable: $\theta_{k+1}= \nabla h(x_k)-\eta_k g_k$
    \State Map back to primal: $x_{k+1}= (\nabla h)^{-1}(\theta_{k+1})$
\EndFor
\State \textbf{Return} $x_T$ (or averaged iterate $\bar x_T=\frac1T\sum_{k=1}^{T}x_k$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ on $\R$. Choose Euclidean prox $h(w)=\frac12 w^2$ (so $\Breg(w,w_k)=\frac12 (w-w_k)^2$) and constant stepsize $\eta_k=\eta=0.1$.  
Initial point $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_k=\nabla f(w_k)$ & $w_{k+1}=w_k-\eta g_k$ & $f(w_{k+1})$\\ \hline
0 & $2(w_0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.366\\
\end{tabular}
\end{center}

The iterates move toward the minimizer $w^\star=3$ and the objective value decreases at each step.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]
$f$ is convex and $L$‑smooth w.r.t. the norm $\|\cdot\|$, i.e.
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{n}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity of $h$]
$h$ is $\alpha$‑strongly convex w.r.t. the same norm:
\[
\Breg(x,y)\ge\frac{\alpha}{2}\|x-y\|^{2},\qquad\forall x,y\in\R^{n}. 
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Stepsize]
The stepsizes are constant or non‑increasing and satisfy
\[
0<\eta_k\le \frac{\alpha}{L},\qquad\forall k\ge0.
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Descent]\label{thm:md}
Under Assumptions (A1)–(A3), let $\{x_k\}$ be generated by \eqref{eq:md-update}. For any $x^\star\in\arg\min f$,
\[
f(\bar x_T)-f(x^\star)\;\le\;
\frac{1}{\sum_{k=0}^{T-1}\eta_k}\,
\Breg(x^\star,x_0),
\qquad\text{where }\bar x_T:=\frac{\sum_{k=0}^{T-1}\eta_k x_{k+1}}{\sum_{k=0}^{T-1}\eta_k}.
\tag{2}
\]
Consequently, with a constant stepsize $\eta_k=\eta\le\alpha/L$,
\[
f(\bar x_T)-f(x^\star)\le \frac{\Breg(x^\star,x_0)}{\eta T}=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update \eqref{eq:md-update} is a proximal step in the dual space; strong convexity of $h$ guarantees that the mapping $(\nabla h)^{-1}$ is Lipschitz, preventing exploding iterates.
\item \textbf{Hyper‑parameter sensitivity.} The stepsize bound $\eta\le\alpha/L$ is tight: if $\eta>\alpha/L$, the smoothness term may dominate the strong convexity term and the descent guarantee fails.
\item \textbf{Comparison to Gradient Descent.} When $h(w)=\frac12\|w\|^{2}$, $\Breg$ reduces to Euclidean distance and MD becomes standard gradient descent with identical $O(1/T)$ rate. For non‑Euclidean $h$, MD adapts to the geometry of the feasible set, often yielding better practical performance (e.g., KL‑divergence for probability simplices).
\end{itemize}

\section*{Discussion}
The theorem shows that Mirror Descent attains the optimal $O(1/T)$ rate for convex smooth problems without requiring projection onto a constraint set (the geometry is encoded in $h$). The bound depends only on the Bregman distance between the start point and an optimal solution, illustrating that a good choice of $h$ (small $\Breg(x^\star,x_0)$) can accelerate convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:three-point}
For any $x,y,z\in\operatorname{int}(\operatorname{dom}h)$,
\[
\Breg(x,z)-\Breg(y,z)-\Breg(x,y)=\langle\nabla h(y)-\nabla h(z),\,x-y\rangle .
\tag{3}
\]
\end{lemma}
\begin{proof}
By definition,
\[
\Breg(x,z)=h(x)-h(z)-\langle\nabla h(z),x-z\rangle,
\]
\[
\Breg(y,z)=h(y)-h(z)-\langle\nabla h(z),y-z\rangle,
\]
\[
\Breg(x,y)=h(x)-h(y)-\langle\nabla h(y),x-y\rangle .
\]
Subtracting the second and third expressions from the first gives (3). ∎
\end{proof}

\begin{lemma}[Descent Lemma w.r.t. Bregman Divergence]\label{lem:descent}
If $f$ satisfies (A1) and $\eta_k\le\alpha/L$, then for any $x\in\R^{n}$,
\[
f(x_{k+1})-f(x)\le \frac{1}{\eta_k}\bigl[\Breg(x,x_k)-\Breg(x,x_{k+1})-\Breg(x_{k+1},x_k)\bigr].
\tag{4}
\]
\end{lemma}
\begin{proof}
From convexity of $f$,
\[
f(x_{k+1})-f(x)\le\langle\nabla f(x_k),x_{k+1}-x\rangle .\tag{Step 1}
\]
Using optimality condition (1), $\nabla h(x_{k+1})-\nabla h(x_k)=-\eta_k\nabla f(x_k)$. Hence
\[
\langle\nabla f(x_k),x_{k+1}-x\rangle
= -\frac{1}{\eta_k}\langle\nabla h(x_{k+1})-\nabla h(x_k),x_{k+1}-x\rangle .\tag{Step 2}
\]
Apply Lemma~\ref{lem:three-point} with $(x,y,z)=(x,x_{k+1},x_k)$:
\[
\langle\nabla h(x_{k+1})-\nabla h(x_k),x-x_{k+1}\rangle
= \Breg(x,x_k)-\Breg(x,x_{k+1})-\Breg(x_{k+1},x_k).\tag{Step 3}
\]
Insert (Step 3) into (Step 2) and reverse the sign:
\[
\langle\nabla f(x_k),x_{k+1}-x\rangle
= \frac{1}{\eta_k}\bigl[\Breg(x,x_k)-\Breg(x,x_{k+1})-\Breg(x_{k+1},x_k)\bigr].\tag{Step 4}
\]
Combine with (Step 1) to obtain (4). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:md}.  

Let $x^\star\in\arg\min f$ be any optimal solution.

\begin{enumerate}
\item[Step 5.] Apply Lemma~\ref{lem:descent} with $x=x^\star$:
\[
f(x_{k+1})-f(x^\star)\le
\frac{1}{\eta_k}\bigl[\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})-\Breg(x_{k+1},x_k)\bigr].
\tag{5}
\]

\item[Step 6.] Drop the non‑negative term $\Breg(x_{k+1},x_k)\ge0$ (by (A2)):
\[
f(x_{k+1})-f(x^\star)\le
\frac{1}{\eta_k}\bigl[\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr].
\tag{6}
\]

\item[Step 7.] Multiply both sides of (6) by $\eta_k$ and sum from $k=0$ to $T-1$:
\[
\sum_{k=0}^{T-1}\eta_k\bigl[f(x_{k+1})-f(x^\star)\bigr]
\le
\sum_{k=0}^{T-1}\bigl[\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr].
\tag{7}
\]

\item[Step 8.] The right‑hand side telescopes:
\[
\sum_{k=0}^{T-1}\bigl[\Breg(x^\star,x_k)-\Breg(x^\star,x_{k+1})\bigr]
= \Breg(x^\star,x_0)-\Breg(x^\star,x_T)\le \Breg(x^\star,x_0).
\tag{8}
\]

\item[Step 9.] Combine (7) and (8):
\[
\sum_{k=0}^{T-1}\eta_k\bigl[f(x_{k+1})-f(x^\star)\bigr]\le \Breg(x^\star,x_0).
\tag{9}
\]

\item[Step 10.] Define the weighted average iterate
\[
\bar x_T:=\frac{\sum_{k=0}^{T-1}\eta_k x_{k+1}}{\sum_{k=0}^{T-1}\eta_k}.
\]
Convexity of $f$ yields
\[
f(\bar x_T)\le\frac{\sum_{k=0}^{T-1}\eta_k f(x_{k+1})}{\sum_{k=0}^{T-1}\eta_k}.
\tag{10}
\]

\item[Step 11.] Subtract $f(x^\star)$ from both sides of (10) and use (9):
\[
f(\bar x_T)-f(x^\star)
\le
\frac{\Breg(x^\star,x_0)}{\sum_{k=0}^{T-1}\eta_k}.
\tag{11}
\]

\item[Step 12.] When $\eta_k\equiv\eta\le\alpha/L$, the denominator equals $\eta T$, giving the explicit rate
\[
f(\bar x_T)-f(x^\star)\le\frac{\Breg(x^\star,x_0)}{\eta T}=O\!\left(\frac{1}{T}\right).
\tag{12}
\]
\end{enumerate}

Thus (11) proves inequality (2) of Theorem~\ref{thm:md}, and (12) yields the $O(1/T)$ convergence rate. ∎

\section*{Rate Derivation (With Constants)}
From (12) and the strong convexity bound $\Breg(x^\star,x_0)\le\frac{1}{2\alpha}\|x^\star-x_0\|^{2}$,
\[
f(\bar x_T)-f(x^\star)\le \frac{1}{2\alpha\eta}\,\frac{\|x^\star-x_0\|^{2}}{T}.
\]
Choosing the maximal admissible stepsize $\eta=\alpha/L$ gives the tightest constant:
\[
f(\bar x_T)-f(x^\star)\le \frac{L}{2\alpha^{2}}\frac{\|x^\star-x_0\|^{2}}{T}.
\tag{13}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean setup.} $h(w)=\frac12\|w\|_{2}^{2}$, $\alpha=1$, $\Breg$ becomes $\frac12\|x-y\|_{2}^{2}$. MD coincides with Gradient Descent and (13) reduces to the classic $L\|x_0-x^\star\|^{2}/(2T)$ bound.
\item \textbf{Entropy prox on the simplex.} $h(p)=\sum_{i}p_i\log p_i$, $\alpha=1$ w.r.t. $\ell_1$ norm, $\Breg$ is KL‑divergence. The bound (13) is expressed in terms of KL distance, which can be much smaller than Euclidean distance for probability vectors.
\end{itemize}
