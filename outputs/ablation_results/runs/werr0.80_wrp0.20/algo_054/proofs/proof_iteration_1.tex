\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (possibly non‑convex) objective and let 
\[
g_t \;:=\; \nabla f(w_t;\,\xi_t)
\]
be a stochastic gradient evaluated at iterate $w_t\in\RR^{d}$ using a random sample
$\xi_t$.  
The learning rate follows a \emph{triangular cyclic schedule} with period $2M$:
\[
\boxed{
\eta_t \;=\; \eta_{\min}
        + (\eta_{\max}-\eta_{\min})
          \Bigl(1-\Bigl|\frac{(t \bmod 2M)}{M}-1\Bigr|\Bigr),\qquad t=0,1,2,\dots
}
\tag{1}
\]
Thus $\eta_t\in[\eta_{\min},\eta_{\max}]$, increases linearly from $\eta_{\min}$ to 
$\eta_{\max}$ in the first $M$ steps and then decreases back to $\eta_{\min}$ in the next $M$ steps, repeating thereafter.

The update rule is
\[
\boxed{
w_{t+1}=w_t-\eta_t\,g_t ,\qquad t=0,1,2,\dots
}
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic Stochastic Gradient Descent}
\begin{algorithmic}[1]
\Require Initial point $w_0$, period $M$, bounds $\eta_{\min},\eta_{\max}>0$, 
      total iterations $T$.
\State Initialize $t\gets0$.
\While{$t<T$}
    \State Sample $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State Compute cyclic step size
    \[
    \eta_t\gets\eta_{\min}+(\eta_{\max}-\eta_{\min})
               \Bigl(1-\Bigl|\frac{(t\bmod 2M)}{M}-1\Bigr|\Bigr).
    \]
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
    \State $t\gets t+1$.
\EndWhile
\State \textbf{return} $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$, whose gradient is
$\nabla f(w)=2(w-3)$.  
Take $w_0=0$, $\eta_{\min}=0.05$, $\eta_{\max}=0.15$, period $M=2$ (so the schedule
is $[0.05,0.15,0.15,0.05,\dots]$).  We perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\eta_t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & $0.05$ & $2(0-3)=-6$ & $0-0.05(-6)=0.30$ \\
1 & $0.15$ & $2(0.30-3)=-5.40$ & $0.30-0.15(-5.40)=1.11$ \\
2 & $0.15$ & $2(1.11-3)=-3.78$ & $1.11-0.15(-3.78)=1.68$
\end{tabular}
\end{center}
Corresponding objective values are $f(0)=9$, $f(0.30)=7.29$, $f(1.11)=3.61$, $f(1.68)=1.74$,
showing descent.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. for all $w,u\in\RR^{d}$,
\[
f(u)\le f(w)+\inner{\nabla f(w)}{u-w}+\frac{L}{2}\norm{u-w}^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Gradient]\label{as:unbiased}
For every $t$, conditional on $w_t$,
\[
\EE_{\xi_t}[\,g_t\mid w_t\,]=\nabla f(w_t).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^2\ge0$ such that
\[
\EE_{\xi_t}\!\bigl[\norm{g_t-\nabla f(w_t)}^2\mid w_t\bigr]\le\sigma^2,\qquad\forall t.
\]
\end{assumption}
\begin{assumption}[Cyclic Step Size]\label{as:step}
The learning rates $\{\eta_t\}$ are generated by \eqref{1} with
$0<\eta_{\min}\le\eta_{\max}<\frac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:step}, after $T$ iterations of
Cyclic SGD the averaged squared gradient norm satisfies
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}
\EE\!\Bigl[\sum_{t=0}^{T-1}\eta_t\,\norm{\nabla f(w_t)}^2\Bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_t}
\;+\;
L\eta_{\max}\sigma^2 ,
\tag{3}
\]
where $f^{\star}=\inf_{w}f(w)$.  Consequently,
\[
\EE\!\bigl[\norm{\nabla f(\bar w_T)}^2\bigr]
\;=\;O\!\Bigl(\frac{1}{T}\Bigr)
\quad\text{with }\bar w_T \text{ chosen as a random iterate with
probability proportional to }\eta_t .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because $\eta_{\max}<1/L$, the descent lemma (Step 4) guarantees that the expected function value never increases beyond a controllable bound.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{3} depends linearly on $\eta_{\max}$ (through the variance term) and inversely on the total \emph{effective stepsize} $\sum_{t}\eta_t$.  Larger $\eta_{\max}$ accelerates progress but amplifies the noise term $L\eta_{\max}\sigma^2$.
\item \textbf{Comparison to constant‑step SGD.} Setting $\eta_t\equiv\eta$ reduces \eqref{3} to the classic $O(1/(\eta T)+L\eta\sigma^2)$ bound.  The cyclic schedule effectively replaces $\eta$ by the average $\bar\eta=\frac{1}{T}\sum_{t}\eta_t$, while still allowing occasional larger steps that can help escape shallow local minima.
\item \textbf{Special Cases.}
  \begin{enumerate}
  \item If $\sigma=0$ (deterministic gradients) the bound simplifies to
        $\frac{2(f(w_0)-f^{\star})}{\sum_{t}\eta_t}$, i.e. the method converges to a stationary point at rate $O(1/T)$.
  \item If $\eta_{\min}=\eta_{\max}=\eta$, Theorem~\ref{thm:main} recovers the standard SGD bound.
  \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the non‑convexity and the non‑monotone step‑size schedule, Cyclic SGD retains the same $O(1/T)$ convergence rate (in expectation) as classical SGD with a constant step size.  The periodic increase to $\eta_{\max}$ can be interpreted as a controlled exploration phase, while the subsequent decrease stabilises the iterates.  The proof below follows the classical smooth‑function descent argument but carefully tracks the time‑varying $\eta_t$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $w$ and any vector $d$,
\[
f(w-d)\le f(w)-\inner{\nabla f(w)}{d}+\frac{L}{2}\norm{d}^2 .
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{as:smooth} with $u=w-d$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken over the randomness
in $\{\xi_t\}_{t=0}^{T-1}$ and we write $\EE_t[\cdot]=\EE[\cdot\mid w_t]$.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=\eta_t g_t$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\inner{\nabla f(w_t)}{g_t}
                +\frac{L\eta_t^{2}}{2}\norm{g_t}^{2}. \tag{4}
\]

\item[Step 2.] Take conditional expectation $\EE_t[\cdot]$ on both sides of \eqref{4}.
Using Assumption~\ref{as:unbiased} we obtain
\[
\EE_t\!\bigl[f(w_{t+1})\bigr]
\le f(w_t)-\eta_t \norm{\nabla f(w_t)}^{2}
   +\frac{L\eta_t^{2}}{2}\EE_t\!\bigl[\norm{g_t}^{2}\bigr]. \tag{5}
\]

\item[Step 3.] Expand $\EE_t[\norm{g_t}^{2}]$:
\[
\EE_t\!\bigl[\norm{g_t}^{2}\bigr]
= \EE_t\!\bigl[\norm{g_t-\nabla f(w_t)}^{2}\bigr]
  +\norm{\nabla f(w_t)}^{2}
\le \sigma^{2}+\norm{\nabla f(w_t)}^{2}, \tag{6}
\]
where the inequality follows from Assumption~\ref{as:variance}.

\item[Step 4.] Substitute \eqref{6} into \eqref{5}:
\[
\EE_t\!\bigl[f(w_{t+1})\bigr]
\le f(w_t)-\eta_t \norm{\nabla f(w_t)}^{2}
   +\frac{L\eta_t^{2}}{2}\bigl(\norm{\nabla f(w_t)}^{2}+\sigma^{2}\bigr). \tag{7}
\]

\item[Step 5.] Rearrange \eqref{7} to isolate the gradient term:
\[
\bigl(\eta_t-\tfrac{L}{2}\eta_t^{2}\bigr)\norm{\nabla f(w_t)}^{2}
\le f(w_t)-\EE_t\!\bigl[f(w_{t+1})\bigr]
   +\tfrac{L}{2}\eta_t^{2}\sigma^{2}. \tag{8}
\]

\item[Step 6.] Because $\eta_t\le\eta_{\max}<1/L$, we have
$\eta_t-\frac{L}{2}\eta_t^{2}\ge \frac{\eta_t}{2}$.  Hence
\[
\frac{\eta_t}{2}\norm{\nabla f(w_t)}^{2}
\le f(w_t)-\EE_t\!\bigl[f(w_{t+1})\bigr]
   +\frac{L}{2}\eta_t^{2}\sigma^{2}. \tag{9}
\]

\item[Step 7.] Take full expectation $\EE[\cdot]$ and sum \eqref{9} over $t=0,\dots,T-1$:
\[
\frac12\sum_{t=0}^{T-1}\eta_t\,
      \EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-\EE\!\bigl[f(w_T)\bigr]
   +\frac{L\sigma^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}. \tag{10}
\]

\item[Step 8.] Since $f(w_T)\ge f^{\star}$, replace $-\EE[f(w_T)]$ by $-f^{\star}$ and bound
$\eta_t^{2}\le\eta_{\max}^{2}$:
\[
\frac12\sum_{t=0}^{T-1}\eta_t\,
      \EE\!\bigl[\norm{\nabla f(w_t)}^{2}\bigr]
\le f(w_0)-f^{\star}
   +\frac{L\sigma^{2}\eta_{\max}}{2}\sum_{t=0}^{T-1}\eta_t . \tag{11}
\]

\item[Step 9.] Divide both sides of \eqref{11} by $\sum_{t=0}^{T-1}\eta_t$ and multiply by $2$:
\[
\EE\!\Bigl[\frac{\sum_{t=0}^{T-1}\eta_t\norm{\nabla f(w_t)}^{2}}
                {\sum_{t=0}^{T-1}\eta_t}\Bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_t}
   +L\eta_{\max}\sigma^{2}. \tag{12}
\]
Equation \eqref{12} is exactly the bound \eqref{3} stated in Theorem~\ref{thm:main}.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Define the random index $I$ drawn from $\{0,\dots,T-1\}$ with
$\Pr(I=t)=\eta_t/\sum_{k=0}^{T-1}\eta_k$.  Then
\[
\EE\!\bigl[\norm{\nabla f(w_I)}^{2}\bigr]
=
\EE\!\Bigl[\frac{\sum_{t=0}^{T-1}\eta_t\norm{\nabla f(w_t)}^{2}}
                {\sum_{t=0}^{T-1}\eta_t}\Bigr].
\]
Applying \eqref{12} yields
\[
\EE\!\bigl[\norm{\nabla f(w_I)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\sum_{t=0}^{T-1}\eta_t}
   +L\eta_{\max}\sigma^{2}. \tag{13}
\]
Because the schedule is periodic with period $2M$,
\[
\sum_{t=0}^{T-1}\eta_t
\;=\;
\bigl\lfloor\tfrac{T}{2M}\bigr\rfloor\,
  \underbrace{\sum_{t=0}^{2M-1}\eta_t}_{=M(\eta_{\min}+\eta_{\max})}
\;+\;
\text{remainder}\;\ge\;
\frac{T}{2M}\,M(\eta_{\min}+\eta_{\max})
\;=\;
\frac{T}{2}\,(\eta_{\min}+\eta_{\max}). \tag{14}
\]
Plugging \eqref{14} into \eqref{13} gives the explicit $O(1/T)$ rate:
\[
\EE\!\bigl[\norm{\nabla f(w_I)}^{2}\bigr]
\;\le\;
\frac{4\bigl(f(w_0)-f^{\star}\bigr)}{(\eta_{\min}+\eta_{\max})\,T}
   +L\eta_{\max}\sigma^{2}. \tag{15}
\]

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Deterministic gradients ($\sigma=0$).}  
Equation \eqref{15} reduces to
\[
\EE\!\bigl[\norm{\nabla f(w_I)}^{2}\bigr]
\le \frac{4\bigl(f(w_0)-f^{\star}\bigr)}
           {(\eta_{\min}+\eta_{\max})\,T},
\]
showing pure $O(1/T)$ decay to a stationary point.

\item \textbf{Constant step size.}  
If $\eta_{\min}=\eta_{\max}=\eta$, then $(\eta_{\min}+\eta_{\max})=2\eta$ and
\eqref{15} becomes the classic SGD bound
\[
\EE\!\bigl[\norm{\nabla f(w_I)}^{2}\bigr]
\le \frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
   +L\eta\sigma^{2}.
\]
\end{enumerate}

\section*{Conclusion}
We have presented a fully detailed analysis of Cyclic SGD for smooth,
non‑convex objectives.  The proof rigorously follows from the smoothness
descent lemma, the unbiased‑variance gradient model, and the explicit
periodic step‑size bound.  The resulting guarantee matches that of standard
SGD while allowing the learning rate to periodically explore larger values,
a feature that can be beneficial in practice.
