\section*{Random Block Coordinate Descent (RBCD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
Partition the coordinate set $\{1,\dots ,d\}$ into $B$ disjoint blocks
$\mathcal{B}_{1},\dots ,\mathcal{B}_{B}$ (for simplicity we assume equal
size, but any partition works).  
Denote by $\nabla_{i}f(w)$ the subvector of the gradient corresponding to
block $\mathcal{B}_{i}$ and by $e_{i}$ the block‑selection operator
(which zeroes out all coordinates except those in $\mathcal{B}_{i}$).

At iteration $t$ a block $i_{t}\in\{1,\dots ,B\}$ is drawn
independently and uniformly:
\[
\Pr(i_{t}=i)=\frac{1}{B},\qquad i=1,\dots ,B .
\]

Given a stepsize $\eta>0$, the update reads
\begin{equation}
\label{eq:update}
w_{t+1}=w_{t}-\eta\,e_{i_{t}}\nabla f(w_{t})\;=\; w_{t}
-\eta\,\nabla_{i_{t}}f(w_{t}) .
\end{equation}
Only the coordinates belonging to the chosen block are modified;
all other coordinates remain unchanged.

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{Random Block Coordinate Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsize $\eta>0$,
number of iterations $T$.
\For{$t=0,\dots ,T-1$}
    \State Sample block $i_{t}\sim\text{Uniform}\{1,\dots ,B\}$.
    \State Compute block gradient $g_{t}= \nabla_{i_{t}}f(w_{t})$.
    \State Update only block $i_{t}$:
    \[
        w_{t+1}= w_{t}-\eta\,e_{i_{t}} g_{t}.
    \]
\EndFor
\State \textbf{Output:} $w_{T}$ (or the average $\bar w_{T}= \frac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\subsection*{Dry Run Example}
Consider the one‑dimensional problem
\[
f(w)=(w-3)^{2},\qquad w\in\R .
\]
Here there is a single block ($B=1$), thus $i_{t}=1$ deterministically.
Take $w_{0}=0$ and stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})=2(w_{t}-3)$ & $f(w_{t})$ \\\hline
0 & $0$ & $-6$ & $9$ \\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$ \\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$ \\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3689$
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the
mechanism of RBCD.

\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{assumption}[Convexity]
\label{as:convex}
$f$ is convex, i.e.
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}

\begin{assumption}[Block‑wise Lipschitz smoothness]
\label{as:smooth}
For each block $i\in\{1,\dots ,B\}$ there exists $L_{i}>0$ such that
\[
\bigl\|\nabla_{i}f(x)-\nabla_{i}f(y)\bigr\|
\le L_{i}\,\|e_{i}(x-y)\|,\qquad\forall x,y\in\R^{d}.
\]
Define $L_{\max}:=\max_{i}L_{i}$.
\end{assumption}

\begin{assumption}[Stepsize]
\label{as:stepsize}
The stepsize satisfies
\[
0<\eta\le\frac{1}{L_{\max}} .
\]
\end{assumption}

\begin{assumption}[Existence of a minimiser]
\label{as:opt}
Let $w^{\star}\in\arg\min_{w}f(w)$ and denote $D:=\|w_{0}-w^{\star}\|^{2}<\infty$.
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{theorem}[Expected suboptimality of RBCD]
\label{thm:conv}
Under Assumptions \ref{as:convex}–\ref{as:opt} and using the stepsize
condition \ref{as:stepsize}, the iterates generated by
\eqref{eq:update} satisfy for every $T\ge 1$
\[
\E\!\left[f(w_{T})-f(w^{\star})\right]\;\le\;
\frac{2L_{\max}\,D}{T}.
\]
Consequently the method attains an $O(1/T)$ convergence rate in
expectation.
\end{theorem}

\begin{theorem}[Convergence of the averaged iterate]
\label{thm:average}
Define $\bar w_{T}:=\frac{1}{T}\sum_{t=0}^{T-1} w_{t}$.
Then
\[
\E\!\left[f(\bar w_{T})-f(w^{\star})\right]\;\le\;
\frac{2L_{\max}\,D}{T}.
\]
\end{theorem}

\subsection*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound \eqref{thm:conv} holds for any
initial point $w_{0}$; the dependence on $w_{0}$ appears only through
$D=\|w_{0}-w^{\star}\|^{2}$.
\item \textbf{Hyperparameter sensitivity.}
The stepsize must respect $\eta\le 1/L_{\max}$.
Choosing $\eta=1/L_{\max}$ yields the tightest constant
$2L_{\max}D$ in the rate.
\item \textbf{Comparison to full‑gradient GD.}
Full‑gradient gradient descent with stepsize $1/L_{\max}$ also enjoys an
$O(1/T)$ rate but each iteration costs $d$ gradient components,
whereas RBCD costs only $\frac{d}{B}$ components on average.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{lemma}[Block‑wise smoothness inequality]
\label{lem:smooth}
For any block $i$ and any $w\in\R^{d}$,
\[
f\!\bigl(w-\eta e_{i}\nabla f(w)\bigr)
\le f(w)-\eta\|\nabla_{i}f(w)\|^{2}
+\frac{L_{i}\eta^{2}}{2}\|\nabla_{i}f(w)\|^{2}.
\]
\end{lemma}
\begin{proof}
By the definition of $L_{i}$-smoothness on block $i$,
\[
f\!\bigl(w-\eta e_{i}\nabla f(w)\bigr)
\le f(w)-\eta\langle\nabla_{i}f(w),\nabla_{i}f(w)\rangle
+\frac{L_{i}\eta^{2}}{2}\|\nabla_{i}f(w)\|^{2},
\]
which is exactly the claimed inequality. ∎
\end{proof}

\begin{lemma}[One‑step expected descent]
\label{lem:expdescent}
For the random block $i_{t}$,
\[
\E_{i_{t}}\!\left[f(w_{t+1})\mid w_{t}\right]
\le f(w_{t})-\frac{\eta}{B}\|\nabla f(w_{t})\|^{2}
+\frac{\eta^{2}L_{\max}}{2B}\|\nabla f(w_{t})\|^{2}.
\]
\end{lemma}
\begin{proof}
Condition on $w_{t}$. Applying Lemma \ref{lem:smooth} and then taking
expectation over the uniformly drawn block,
\[
\begin{aligned}
\E_{i_{t}}\!\left[f(w_{t+1})\mid w_{t}\right]
&= \frac1B\sum_{i=1}^{B}
f\!\bigl(w_{t}-\eta e_{i}\nabla f(w_{t})\bigr)\\[4pt]
&\stackrel{[Step 1]}{\le}
\frac1B\sum_{i=1}^{B}\Bigl(
f(w_{t})-\eta\|\nabla_{i}f(w_{t})\|^{2}
+\frac{L_{i}\eta^{2}}{2}\|\nabla_{i}f(w_{t})\|^{2}
\Bigr)\\[4pt]
&= f(w_{t})-\frac{\eta}{B}\sum_{i=1}^{B}\|\nabla_{i}f(w_{t})\|^{2}
+\frac{\eta^{2}}{2B}\sum_{i=1}^{B}L_{i}\|\nabla_{i}f(w_{t})\|^{2}\\[4pt]
&\stackrel{[Step 2]}{\le}
f(w_{t})-\frac{\eta}{B}\|\nabla f(w_{t})\|^{2}
+\frac{\eta^{2}L_{\max}}{2B}\|\nabla f(w_{t})\|^{2}.
\end{aligned}
\]
[Step 1] uses Lemma~\ref{lem:smooth}; [Step 2] uses $L_{i}\le L_{\max}$
and $\|\nabla f\|^{2}= \sum_{i}\|\nabla_{i}f\|^{2}$. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

We now prove Theorem~\ref{thm:conv}.  
All expectations are taken over the randomness of the block selections
$\{i_{0},\dots ,i_{T-1}\}$.

\subsection*{Step 1 – One‑step expected decrease}
From Lemma~\ref{lem:expdescent} and the stepsize condition
$\eta\le 1/L_{\max}$ we have
\begin{align}
\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
&\le f(w_{t})-\frac{\eta}{B}\|\nabla f(w_{t})\|^{2}
+\frac{\eta^{2}L_{\max}}{2B}\|\nabla f(w_{t})\|^{2}\nonumber\\
&= f(w_{t})-\frac{\eta}{2B}\|\nabla f(w_{t})\|^{2}
\qquad\text{[Step 3]}.
\end{align}
[Step 3] follows because $\eta\le 1/L_{\max}$ implies
$\frac{\eta^{2}L_{\max}}{2}\le\frac{\eta}{2}$.

\subsection*{Step 2 – Relating gradient norm to function suboptimality}
Convexity (Assumption~\ref{as:convex}) together with the
first‑order optimality condition at $w^{\star}$ yields
\[
f(w_{t})-f(w^{\star})\le\langle\nabla f(w_{t}),\,w_{t}-w^{\star}\rangle
\le \|\nabla f(w_{t})\|\,\|w_{t}-w^{\star}\|.
\]
Applying Cauchy–Schwarz and squaring,
\[
\|\nabla f(w_{t})\|^{2}\ge
\frac{\bigl(f(w_{t})-f(w^{\star})\bigr)^{2}}{\|w_{t}-w^{\star}\|^{2}}
\ge \frac{\bigl(f(w_{t})-f(w^{\star})\bigr)^{2}}{D},
\qquad\text{[Step 4]}
\]
where $D=\|w_{0}-w^{\star}\|^{2}$ and we used $\|w_{t}-w^{\star}\|^{2}\le D$
by the non‑expansiveness shown next.

\subsection*{Step 3 – Non‑expansiveness of the iterates}
From the update rule \eqref{eq:update} and the fact that only a block
is changed,
\[
\begin{aligned}
\|w_{t+1}-w^{\star}\|^{2}
&= \|w_{t}-w^{\star}\|^{2}
-2\eta\langle e_{i_{t}}\nabla f(w_{t}),\,w_{t}-w^{\star}\rangle
+\eta^{2}\|e_{i_{t}}\nabla f(w_{t})\|^{2}\\
&\le \|w_{t}-w^{\star}\|^{2}
-2\eta\langle \nabla f(w_{t}),\,w_{t}-w^{\star}\rangle
+\eta^{2}\|\nabla f(w_{t})\|^{2}\\
&\le \|w_{t}-w^{\star}\|^{2}
-2\eta\bigl(f(w_{t})-f(w^{\star})\bigr)
+\eta^{2}L_{\max}\bigl(f(w_{t})-f(w^{\star})\bigr)\\
&\le \|w_{t}-w^{\star}\|^{2},
\qquad\text{[Step 5]}
\end{aligned}
\]
where the second inequality uses convexity
$\langle \nabla f(w_{t}),w_{t}-w^{\star}\rangle\ge
f(w_{t})-f(w^{\star})$ and the smoothness bound
$\|\nabla f(w_{t})\|^{2}\le 2L_{\max}\bigl(f(w_{t})-f(w^{\star})\bigr)$.
[Step 5] follows from $\eta\le 1/L_{\max}$, making the last two terms
non‑positive. Hence $\|w_{t}-w^{\star}\|^{2}\le D$ for all $t$.

\subsection*{Step 4 – Putting the pieces together}
Insert the bound \eqref{Step 4} into the expected decrease \eqref{Step 3}:
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\mid w_{t}\bigr]
&\le f(w_{t})-\frac{\eta}{2B}
\cdot\frac{\bigl(f(w_{t})-f(w^{\star})\bigr)^{2}}{D}\\
&= f(w_{t})-\frac{\beta}{B}\bigl(f(w_{t})-f(w^{\star})\bigr)^{2},
\qquad\text{[Step 6]}
\end{aligned}
\]
where we defined $\beta:=\dfrac{\eta}{2D}$ (a positive constant).

\subsection*{Step 5 – Recurrence on the expected suboptimality}
Take total expectation and denote $\Delta_{t}:=\E\bigl[f(w_{t})-f(w^{\star})\bigr]$.
From \eqref{Step 6},
\[
\Delta_{t+1}\le \Delta_{t}-\frac{\beta}{B}\Delta_{t}^{2}.
\tag{1}
\label{recurrence}
\]
Inequality \eqref{recurrence} is a deterministic scalar recursion.
Rearranging,
\[
\frac{1}{\Delta_{t+1}} \ge \frac{1}{\Delta_{t}}+\frac{\beta}{B}.
\]
Iterating from $t=0$ to $T-1$ yields
\[
\frac{1}{\Delta_{T}} \ge \frac{1}{\Delta_{0}}+\frac{\beta T}{B}
\;\Longrightarrow\;
\Delta_{T}\le\frac{1}{\frac{1}{\Delta_{0}}+\frac{\beta T}{B}}
\le\frac{B}{\beta T}
= \frac{2BD}{\eta\,T}.
\tag{2}
\label{bound}
\]
Since $\eta\le 1/L_{\max}$ we may replace $\eta$ by $1/L_{\max}$ to obtain
the cleanest constant,
\[
\Delta_{T}\le\frac{2L_{\max}D}{T}.
\]
This is precisely the claim of Theorem~\ref{thm:conv}. ∎

\subsection*{Step 6 – Convergence of the averaged iterate}
For convex functions, Jensen’s inequality gives
\[
f(\bar w_{T})\le\frac1T\sum_{t=0}^{T-1} f(w_{t}).
\]
Taking expectations and applying the bound \eqref{bound} to each term,
\[
\E\!\bigl[f(\bar w_{T})-f(w^{\star})\bigr]
\le\frac1T\sum_{t=0}^{T-1}\Delta_{t}
\le\frac{1}{T}\sum_{t=0}^{T-1}\frac{2L_{\max}D}{t+1}
\le\frac{2L_{\max}D}{T},
\]
where we used that the sequence $\Delta_{t}$ is non‑increasing.
Thus Theorem~\ref{thm:average} follows. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

From the final bound
\[
\E\!\bigl[f(w_{T})-f(w^{\star})\bigr]\le
\underbrace{\frac{2L_{\max}}{1}}_{\text{smoothness}}
\underbrace{\,\|w_{0}-w^{\star}\|^{2}}_{\text{initial distance}}
\frac{1}{T},
\]
the convergence rate is $O(1/T)$.  
If $L_{\max}$ is replaced by a tighter block‑specific bound
$\frac{1}{B}\sum_{i}L_{i}$ the constant improves accordingly.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{itemize}[leftmargin=*]
\item \textbf{Full‑gradient gradient descent.}
When $B=1$ the algorithm reduces to classical gradient descent; the bound
becomes $\frac{2L_{1}\|w_{0}-w^{\star}\|^{2}}{T}$, matching the standard
rate.
\item \textbf{Coordinate descent ($B=d$).}
If each block contains a single coordinate, $L_{\max}=\max_{j}L_{j}^{\text{coord}}$,
and the per‑iteration cost is $O(1)$. The same $O(1/T)$ rate holds,
explaining why random coordinate descent is competitive for large $d$.
\item \textbf{Strongly convex $f$.}
If additionally $f$ is $\mu$‑strongly convex, the same proof yields a
linear rate:
\[
\E\!\bigl[f(w_{T})-f(w^{\star})\bigr]\le
\bigl(1-\tfrac{\eta\mu}{B}\bigr)^{T}\bigl(f(w_{0})-f(w^{\star})\bigr).
\]
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The analysis shows that random block coordinate descent inherits the
$O(1/T)$ convergence guarantee of full‑gradient methods while
dramatically reducing per‑iteration computational effort. The key
ingredients are block‑wise smoothness (ensuring a quadratic upper
bound on the decrease) and convexity (relating gradient norm to
function suboptimality). The proof is fully constructive: every
inequality is derived from the assumptions, and each non‑trivial
manipulation is labelled with a step number for easy verification.
