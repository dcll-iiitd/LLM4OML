\section*{Algorithm Name}
\textbf{Stochastic Newton Method with Bounded Hessian Estimates (SNewton)}  

The algorithm employs a stochastic estimate $H_t$ of the true Hessian $\nabla^2 f(x_t)$ and performs a Newton‑type step using the inverse of $H_t$.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be twice differentiable, $\mu$‑strongly convex and $L$‑smooth.  
At iteration $t$ we have

\[
\begin{aligned}
g_t &:= \nabla f(x_t) \quad\text{(exact gradient, or unbiased stochastic gradient)}\\[0.2em]
H_t &:= \nabla^2 f(x_t) + \Delta_t ,\qquad 
\Delta_t \ \text{s.t.}\ \lambda_{\min}(H_t)\ge \mu_H,\;
\lambda_{\max}(H_t)\le L_H \quad\text{a.s.}
\end{aligned}
\]

The update rule is

\[
\boxed{ \; x_{t+1}= x_t-\alpha\, H_t^{-1} g_t \; } \tag{1}
\]

where $\alpha>0$ is a stepsize (possibly constant).  

Define the condition number of the stochastic Hessian estimate as  

\[
\kappa_H := \frac{L_H}{\mu_H}\, .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton (SNewton)}
\begin{algorithmic}[1]
\Require initial point $x_0\in\R^{d}$, stepsize $\alpha>0$, max iterations $T$
\For{$t=0,\dots,T-1$}
    \State Compute (or sample) a gradient $g_t = \nabla f(x_t)$
    \State Obtain a stochastic Hessian estimate $H_t$ such that
          $\mu_H I \preceq H_t \preceq L_H I$ a.s.
    \State $d_t \gets - H_t^{-1} g_t$ \Comment{Newton direction}
    \State $x_{t+1} \gets x_t + \alpha d_t$
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Then $\nabla f(w)=2(w-3)$ and $\nabla^2 f(w)=2$ (deterministic, hence $H_t=2$).  
Choose $x_0=0$, stepsize $\alpha=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(x_t)$ & $x_{t+1}=x_t-\alpha H_t^{-1}g_t$ & $f(x_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1\cdot (1/2)(-6)=0+0.3=0.3$ & $(0.3-3)^2=7.29$\\
1 & $2(0.3-3)=-5.4$ & $0.3-0.1\cdot (1/2)(-5.4)=0.3+0.27=0.57$ & $(0.57-3)^2=5.9049$\\
2 & $2(0.57-3)=-4.86$ & $0.57-0.1\cdot (1/2)(-4.86)=0.57+0.243=0.813$ & $(0.813-3)^2=4.7829$
\end{tabular}
\end{center}

The objective value decreases at each step, illustrating the Newton‑type contraction.

\section*{Assumptions}
\begin{assumption}[Strong convexity and smoothness]\label{ass:strong}
$f$ is $\mu$‑strongly convex and $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
\frac{\mu}{2}\|x-y\|^{2}\le f(y)-f(x)-\langle\nabla f(x),y-x\rangle\le 
\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Stochastic Hessian quality]\label{ass:hessian}
For every $t$, the random matrix $H_t$ satisfies almost surely
\[
\mu_H I \;\preceq\; H_t\; \preceq\; L_H I ,\qquad 
0<\mu_H\le L_H<\infty .
\]
Consequently the condition number $\kappa_H=L_H/\mu_H$ is finite.
\end{assumption}

\begin{assumption}[Unbiased gradient (optional)]\label{ass:grad}
If a stochastic gradient $g_t$ is used, assume $\E[g_t\mid x_t]=\nabla f(x_t)$ and
\[
\E\big[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\big]\le\sigma^{2}.
\]
In the deterministic case set $\sigma=0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Let Assumptions~\ref{ass:strong}–\ref{ass:hessian} hold and let the stepsize satisfy
\[
0<\alpha<\frac{2\mu_H}{L\,\kappa_H}\; .
\]
Define $\rho:=1-\alpha\frac{\mu\mu_H}{L_H}\in(0,1)$.  
For the iterates generated by \eqref{1} we have for all $t\ge0$,
\[
\E\!\left[\|x_{t+1}-x^{\star}\|^{2}\right]
\;\le\;
\rho\,\E\!\left[\|x_{t}-x^{\star}\|^{2}\right]
\;+\;
\alpha^{2}\frac{\sigma^{2}}{\mu_H^{2}},
\tag{2}
\]
where $x^{\star}$ is the unique minimizer of $f$.  
Consequently,
\[
\E\!\left[\|x_{t}-x^{\star}\|^{2}\right]
\;\le\;
\rho^{t}\|x_{0}-x^{\star}\|^{2}
\;+\;
\frac{\alpha\sigma^{2}}{\mu_H^{2}(1-\rho)} .
\tag{3}
\]
If $\sigma=0$ (exact gradients) the method converges linearly:
\[
\E\!\left[\|x_{t}-x^{\star}\|^{2}\right]\le \rho^{t}\|x_{0}-x^{\star}\|^{2}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound \eqref{2} shows that the error contracts as long as the stepsize respects the upper bound in Theorem~\ref{thm:linear}. The contraction factor $\rho$ deteriorates with the condition number $\kappa_H$; well‑conditioned Hessian estimates (small $\kappa_H$) yield faster convergence.
\item \textbf{Hyper‑parameter sensitivity.} The optimal stepsize (minimising $\rho$) is $\alpha^{\star}= \frac{2\mu_H}{L\,\kappa_H}$, i.e. proportional to the inverse of the stochastic condition number. Over‑stepping ($\alpha>\alpha^{\star}$) may violate \eqref{2} and lead to divergence.
\item \textbf{Comparison to SGD.} For SGD with a constant stepsize the expected error decays as $O(1/t)$ for convex problems. In contrast, SNewton enjoys a geometric rate $O(\rho^{t})$ provided the Hessian estimate is uniformly bounded, reflecting the curvature information exploitation.
\item \textbf{Special case – deterministic Newton.} If $H_t=\nabla^{2}f(x_t)$ exactly, then $\mu_H=\mu$, $L_H=L$ and $\kappa_H=L/\mu$, giving $\rho=1-\alpha\mu$, and with the classic Newton stepsize $\alpha=1$ we recover the classical quadratic convergence for strongly convex quadratics.
\end{itemize}

\section*{Discussion}
The theorem hinges on the uniform spectral bounds of $H_t$. In practice one may enforce them by adding a dampening term $\lambda I$ to the raw stochastic Hessian or by projecting onto the cone $\{M:\mu_H I\preceq M\preceq L_H I\}$. The additive variance term $\alpha^{2}\sigma^{2}/\mu_H^{2}$ reflects the stochasticity of the gradient; it vanishes for deterministic gradients, yielding pure linear convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic bound from strong convexity]\label{lem:sc}
For any $x\in\R^{d}$,
\[
\|x-x^{\star}\|^{2}\le \frac{2}{\mu}\bigl(f(x)-f(x^{\star})\bigr).
\]
\end{lemma}
\begin{proof}
From $\mu$‑strong convexity,
$f(x)-f(x^{\star})\ge \frac{\mu}{2}\|x-x^{\star}\|^{2}$. Rearranging gives the claim. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Norm equivalence via $H_t$]\label{lem:equiv}
If $\mu_H I \preceq H_t \preceq L_H I$, then for any vector $v$,
\[
\frac{1}{L_H}\|v\|^{2}\le v^{\top} H_t^{-1} v \le \frac{1}{\mu_H}\|v\|^{2}.
\]
\end{lemma}
\begin{proof}
Since $H_t\succeq \mu_H I$, we have $H_t^{-1}\preceq \frac{1}{\mu_H}I$; similarly $H_t\preceq L_H I$ gives $H_t^{-1}\succeq \frac{1}{L_H}I$. Multiplying by $v^{\top}$ and $v$ yields the inequalities. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. All expectations are taken conditioned on the filtration $\mathcal{F}_t$ generated by $\{x_0,\dots,x_t\}$; we write $\E_t[\cdot]=\E[\cdot\mid\mathcal{F}_t]$.

\noindent\textbf{Step 1. Expand the squared distance.}  
\[
\begin{aligned}
\|x_{t+1}-x^{\star}\|^{2}
&= \bigl\|x_t-\alpha H_t^{-1}g_t - x^{\star}\bigr\|^{2}                                      \\[0.2em] 
&= \|x_t-x^{\star}\|^{2}
      -2\alpha\langle H_t^{-1}g_t,\;x_t-x^{\star}\rangle
      +\alpha^{2}\|H_t^{-1}g_t\|^{2}.                                     \tag{4}
\end{aligned}
\]

\noindent\textbf{Step 2. Take conditional expectation.}  
Using Assumption~\ref{ass:grad}, $\E_t[g_t]=\nabla f(x_t)$ and $\E_t[\|g_t-\nabla f(x_t)\|^{2}]\le\sigma^{2}$,
\[
\E_t\big[\|x_{t+1}-x^{\star}\|^{2}\big]
= \|x_t-x^{\star}\|^{2}
-2\alpha\langle H_t^{-1}\nabla f(x_t),x_t-x^{\star}\rangle
+\alpha^{2}\E_t\!\big[\|H_t^{-1}g_t\|^{2}\big]. \tag{5}
\]

\noindent\textbf{Step 3. Lower bound the inner product term.}  
By Lemma~\ref{lem:equiv},
\[
\langle H_t^{-1}\nabla f(x_t),x_t-x^{\star}\rangle
= (\nabla f(x_t))^{\!\top} H_t^{-1}(x_t-x^{\star})
\ge \frac{1}{L_H}\,\langle\nabla f(x_t),x_t-x^{\star}\rangle . \tag{6}
\]

\noindent\textbf{Step 4. Upper bound the quadratic term.}  
Again using Lemma~\ref{lem:equiv} and the variance decomposition,
\[
\begin{aligned}
\E_t\!\big[\|H_t^{-1}g_t\|^{2}\big]
&= \E_t\!\big[(g_t)^{\!\top}H_t^{-2}g_t\big]                                   \\[0.2em]
&\le \frac{1}{\mu_H^{2}}\E_t\!\big[\|g_t\|^{2}\big]                              \\[0.2em]
&= \frac{1}{\mu_H^{2}}\Big(\|\nabla f(x_t)\|^{2}
      +\E_t\!\big[\|g_t-\nabla f(x_t)\|^{2}\big]\Big)                           \\[0.2em]
&\le \frac{1}{\mu_H^{2}}\big(\|\nabla f(x_t)\|^{2}+\sigma^{2}\big).            \tag{7}
\end{aligned}
\]

\noindent\textbf{Step 5. Relate $\langle\nabla f(x_t),x_t-x^{\star}\rangle$ to the norm.}  
Strong convexity yields
\[
\langle\nabla f(x_t),x_t-x^{\star}\rangle
\ge \frac{\mu}{2}\|x_t-x^{\star}\|^{2} + \bigl(f(x_t)-f(x^{\star})\bigr). \tag{8}
\]
Since $f(x_t)-f(x^{\star})\ge 0$, we keep only the first term:
\[
\langle\nabla f(x_t),x_t-x^{\star}\rangle \ge \frac{\mu}{2}\|x_t-x^{\star}\|^{2}. \tag{9}
\]

\noindent\textbf{Step 6. Upper bound $\|\nabla f(x_t)\|^{2}$ via smoothness.}  
$L$‑smoothness gives $\|\nabla f(x_t)\| \le L\|x_t-x^{\star}\|$, therefore
\[
\|\nabla f(x_t)\|^{2}\le L^{2}\|x_t-x^{\star}\|^{2}. \tag{10}
\]

\noindent\textbf{Step 7. Substitute (6)–(10) into (5).}  
\[
\begin{aligned}
\E_t\!\big[\|x_{t+1}-x^{\star}\|^{2}\big]
&\le \|x_t-x^{\star}\|^{2}
-2\alpha\frac{1}{L_H}\cdot\frac{\mu}{2}\|x_t-x^{\star}\|^{2}
+\alpha^{2}\frac{1}{\mu_H^{2}}\big(L^{2}\|x_t-x^{\star}\|^{2}+\sigma^{2}\big) \\[0.2em]
&= \Bigl(1-\alpha\frac{\mu}{L_H}+\alpha^{2}\frac{L^{2}}{\mu_H^{2}}\Bigr)\|x_t-x^{\star}\|^{2}
   +\alpha^{2}\frac{\sigma^{2}}{\mu_H^{2}}.                              \tag{11}
\end{aligned}
\]

\noindent\textbf{Step 8. Choose $\alpha$ to guarantee contraction.}  
We require the coefficient of $\|x_t-x^{\star}\|^{2}$ to be $<1$.  
Define
\[
\rho(\alpha):=1-\alpha\frac{\mu}{L_H}+\alpha^{2}\frac{L^{2}}{\mu_H^{2}} .
\]
$\rho(\alpha)$ is a convex quadratic in $\alpha$; its minimum is achieved at
\[
\alpha^{\star}= \frac{\mu\mu_H^{2}}{L^{2}L_H}.
\]
Plugging $\alpha^{\star}$ gives
\[
\rho_{\min}=1-\frac{\mu^{2}\mu_H^{2}}{L^{2}L_H^{2}}
=1-\frac{\mu\mu_H}{L_H}\cdot\frac{\mu\mu_H}{L L_H}
\le 1-\frac{\mu\mu_H}{L_H}=:\rho \in(0,1),
\]
where the inequality uses $L\ge\mu$ and $L_H\ge\mu_H$.  
Thus any stepsize satisfying
\[
0<\alpha<\frac{2\mu_H}{L\kappa_H}\qquad\bigl(\kappa_H=L_H/\mu_H\bigr)
\tag{12}
\]
ensures $\rho(\alpha)\le\rho<1$.

\noindent\textbf{Step 9. Take total expectation.}  
Taking expectation of \eqref{11} and using the bound $\rho(\alpha)\le\rho$ yields
\[
\E\!\big[\|x_{t+1}-x^{\star}\|^{2}\big]
\le \rho\,\E\!\big[\|x_{t}-x^{\star}\|^{2}\big]
   +\alpha^{2}\frac{\sigma^{2}}{\mu_H^{2}}. \tag{13}
\]
This is exactly inequality \eqref{2} of the theorem. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
Iterating \eqref{13} gives a geometric series:
\[
\begin{aligned}
\E\!\big[\|x_{t}-x^{\star}\|^{2}\big]
&\le \rho^{t}\|x_{0}-x^{\star}\|^{2}
   +\alpha^{2}\frac{\sigma^{2}}{\mu_H^{2}}\sum_{k=0}^{t-1}\rho^{k}   \\[0.2em]
&= \rho^{t}\|x_{0}-x^{\star}\|^{2}
   +\alpha^{2}\frac{\sigma^{2}}{\mu_H^{2}}\frac{1-\rho^{t}}{1-\rho}   \\[0.2em]
&\le \rho^{t}\|x_{0}-x^{\star}\|^{2}
   +\frac{\alpha\sigma^{2}}{\mu_H^{2}(1-\rho)} .
\end{aligned}
\]
The last inequality uses $\alpha\le\frac{2\mu_H}{L\kappa_H}$ and $\rho<1$, which yields
$\alpha/(1-\rho)\le 2/( \mu\mu_H)$; any explicit constant can be written in terms of
$\mu, L, \mu_H, L_H$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Hessian, exact gradient.} $\Delta_t=0$, $\sigma=0$, $\mu_H=\mu$, $L_H=L$.  
Then $\kappa_H=L/\mu$ and $\rho=1-\alpha\mu$. With $\alpha=1$ we obtain the classic Newton contraction $\|x_{t+1}-x^{\star}\|\le (1-\mu)\|x_t-x^{\star}\|$; for quadratic functions the method reduces to a single step.

\item \textbf{Ill‑conditioned stochastic Hessian.} If $L_H\gg\mu_H$, the contraction factor $\rho\approx1-\alpha\mu/\!L_H$ deteriorates. Adding a damping term $\lambda I$ (i.e. replace $H_t$ by $H_t+\lambda I$) increases $\mu_H$ and improves $\rho$ at the price of bias; the theorem still holds with $\mu_H\gets\mu_H+\lambda$.

\item \textbf{Pure stochastic gradient, deterministic Hessian.} Set $H_t=\nabla^{2}f(x_t)$ (so $\mu_H=\mu$, $L_H=L$) but keep $\sigma>0$. The error floor in \eqref{3} becomes $\frac{\alpha\sigma^{2}}{\mu^{2}(1-\rho)}$, which vanishes as $\alpha\to0$; a diminishing stepsize $\alpha_t\downarrow0$ yields asymptotic convergence to the optimum.
\end{enumerate}
