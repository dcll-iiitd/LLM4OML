\section*{Algorithm Name}
\textbf{Minibatch Stochastic Gradient Descent (Minibatch SGD)} with fixed step size.

\section*{Mathematical Formulation}

The algorithm minimizes an objective of the form
\[
f(w) = \mathbb{E}_{z \sim \mathcal{D}}[F(w; z)],
\]
where $w \in \mathbb{R}^d$ are the parameters and $z$ is a data sample drawn from distribution $\mathcal{D}$.

\textbf{Update rule.} At each iteration $t$, a minibatch $\mathcal{B}_t = \{z_{t,1}, \dots, z_{t,b}\}$ of size $b$ is sampled i.i.d. from $\mathcal{D}$. The minibatch stochastic gradient is
\[
g_t = \frac{1}{b} \sum_{i=1}^{b} \nabla F(w_t; z_{t,i}),
\]
and the parameters are updated as
\[
w_{t+1} = w_t - \eta\, g_t,
\]
where $\eta > 0$ is a fixed step size (learning rate).

\textbf{Hyperparameters.}
\begin{itemize}
    \item $\eta > 0$: fixed learning rate.
    \item $b \in \mathbb{N}$: minibatch size.
    \item $T \in \mathbb{N}$: total number of iterations.
    \item $w_0 \in \mathbb{R}^d$: initialization.
\end{itemize}

\section*{Pseudocode}

\begin{verbatim}
Input: initial point w_0, step size eta, minibatch size b, iterations T
for t = 0, 1, 2, ..., T-1 do
    Sample minibatch B_t = {z_{t,1}, ..., z_{t,b}} i.i.d. from D
    Compute stochastic gradient:
        g_t = (1/b) * sum_{i=1}^{b} grad F(w_t; z_{t,i})
    Update parameters:
        w_{t+1} = w_t - eta * g_t
end for
Termination: stop after T iterations (or when ||g_t|| < tolerance)
Output: w_R, where R is chosen uniformly at random from {0, ..., T-1}
\end{verbatim}

\section*{Dry Run Example}

Consider $f(w) = (w-3)^2$ with $w_0 = 0$, $\eta = 0.1$. Here $\nabla f(w) = 2(w-3)$.
We treat the gradient as exact (deterministic) for illustration.

\textbf{Iteration 1 ($t=0$):}
\begin{align*}
g_0 &= \nabla f(w_0) = 2(0-3) = -6, \\
w_1 &= w_0 - \eta g_0 = 0 - 0.1 \cdot (-6) = 0.6, \\
f(w_1) &= (0.6 - 3)^2 = (-2.4)^2 = 5.76.
\end{align*}

\textbf{Iteration 2 ($t=1$):}
\begin{align*}
g_1 &= \nabla f(w_1) = 2(0.6-3) = -4.8, \\
w_2 &= w_1 - \eta g_1 = 0.6 - 0.1 \cdot (-4.8) = 1.08, \\
f(w_2) &= (1.08 - 3)^2 = (-1.92)^2 = 3.6864.
\end{align*}

\textbf{Iteration 3 ($t=2$):}
\begin{align*}
g_2 &= \nabla f(w_2) = 2(1.08-3) = -3.84, \\
w_3 &= w_2 - \eta g_2 = 1.08 - 0.1 \cdot (-3.84) = 1.464, \\
f(w_3) &= (1.464 - 3)^2 = (-1.536)^2 = 2.359296.
\end{align*}

The objective value decreases monotonically: $9 \to 5.76 \to 3.6864 \to 2.3593$, approaching the minimum $f(3)=0$.

\section*{Assumptions}

\begin{assumption}[Lower boundedness]\label{ass:lb}
The objective $f$ is bounded below: there exists $f^* \in \mathbb{R}$ such that
\[
f(w) \geq f^* \quad \text{for all } w \in \mathbb{R}^d.
\]
\end{assumption}

\begin{assumption}[$L$-smoothness]\label{ass:smooth}
$f$ is continuously differentiable and its gradient is $L$-Lipschitz: there exists $L > 0$ such that
\[
\|\nabla f(w) - \nabla f(v)\| \leq L \|w - v\| \quad \text{for all } w, v \in \mathbb{R}^d.
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradients]\label{ass:unbiased}
At each iteration $t$, conditioned on $w_t$,
\[
\mathbb{E}[g_t \mid w_t] = \nabla f(w_t).
\]
\end{assumption}

\begin{assumption}[Bounded variance]\label{ass:var}
There exists $\sigma^2 \geq 0$ such that, conditioned on $w_t$,
\[
\mathbb{E}\big[\|g_t - \nabla f(w_t)\|^2 \mid w_t\big] \leq \frac{\sigma^2}{b},
\]
where $b$ is the minibatch size. (The $1/b$ factor follows from averaging $b$ i.i.d. samples each with per-sample variance bounded by $\sigma^2$.)
\end{assumption}

\section*{Main Convergence Theorem}

\begin{theorem}\label{thm:main}
Suppose Assumptions \ref{ass:lb}--\ref{ass:var} hold, and the step size satisfies $0 < \eta \leq \frac{1}{L}$. Run Minibatch SGD for $T$ iterations from $w_0$. Then
\[
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\leq \frac{2\big(f(w_0) - f^*\big)}{\eta T} + \frac{L \eta \sigma^2}{b}.
\]
In particular, if $R$ is sampled uniformly from $\{0,\dots,T-1\}$,
\[
\mathbb{E}\big[\|\nabla f(w_R)\|^2\big]
\leq \frac{2\big(f(w_0) - f^*\big)}{\eta T} + \frac{L \eta \sigma^2}{b}.
\]
\end{theorem}

\begin{corollary}[Rate with tuned step size]\label{cor:rate}
Choosing $\eta = \min\left\{ \frac{1}{L}, \sqrt{\frac{b(f(w_0)-f^*)}{L\sigma^2 T}} \right\}$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1} \mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
= O\!\left( \frac{1}{T} + \sqrt{\frac{L\sigma^2 (f(w_0)-f^*)}{b\,T}} \right)
= O\!\left( \frac{1}{\sqrt{bT}} \right).
\]
\end{corollary}

\section*{Theoretical Properties}

\begin{itemize}
    \item \textbf{Stationarity guarantee.} For non-convex objectives, global optimality cannot be guaranteed; the theorem certifies convergence to an approximate \emph{stationary point} ($\|\nabla f\| \to 0$).
    \item \textbf{Variance reduction via batch size.} Increasing $b$ reduces the noise floor term $\frac{L\eta\sigma^2}{b}$ linearly, improving the asymptotic accuracy.
    \item \textbf{Step-size sensitivity.} A fixed $\eta$ produces a non-vanishing residual error $\frac{L\eta\sigma^2}{b}$; only by shrinking $\eta$ (or $\eta\propto 1/\sqrt{T}$) does the error vanish.
    \item \textbf{Comparison to deterministic GD.} If $\sigma^2 = 0$ (full-batch), the bound reduces to $O(1/T)$ for the minimum gradient norm, matching deterministic gradient descent.
\end{itemize}

\section*{Discussion}

The result exhibits the canonical decomposition into an \emph{optimization} term $\tfrac{2(f(w_0)-f^*)}{\eta T}$ (decaying with iterations) and a \emph{statistical noise} term $\tfrac{L\eta\sigma^2}{b}$ (controlled by step size and batch size). The trade-off explains the practical preference for moderately large batches and decaying learning rates. The $O(1/\sqrt{bT})$ rate is known to be optimal for first-order methods on non-convex smooth stochastic problems under bounded-variance noise.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$-smooth (Assumption \ref{ass:smooth}), then for all $w, v \in \mathbb{R}^d$,
\[
f(v) \leq f(w) + \langle \nabla f(w), v - w \rangle + \frac{L}{2}\|v - w\|^2.
\]
\end{lemma}

\begin{proof}
By the fundamental theorem of calculus along the segment $w + \tau(v-w)$,
\[
f(v) - f(w) = \int_0^1 \langle \nabla f(w + \tau(v-w)), v - w\rangle \, d\tau.
\]
Subtracting $\langle \nabla f(w), v-w\rangle$,
\begin{align*}
f(v) - f(w) - \langle \nabla f(w), v-w\rangle
&= \int_0^1 \langle \nabla f(w+\tau(v-w)) - \nabla f(w),\, v-w\rangle\, d\tau.
\end{align*}
By Cauchy--Schwarz and $L$-Lipschitzness,
\begin{align*}
&\leq \int_0^1 \|\nabla f(w+\tau(v-w)) - \nabla f(w)\| \, \|v-w\| \, d\tau \\
&\leq \int_0^1 L\,\tau\,\|v-w\|^2 \, d\tau = \frac{L}{2}\|v-w\|^2.
\end{align*}
Rearranging gives the claim.
\end{proof}

\begin{lemma}[Variance--Mean Decomposition]\label{lem:decomp}
For any random vector $X$ with $\mathbb{E}[X] = \mu$,
\[
\mathbb{E}[\|X\|^2] = \|\mu\|^2 + \mathbb{E}[\|X - \mu\|^2].
\]
\end{lemma}

\begin{proof}
$\mathbb{E}[\|X\|^2] = \mathbb{E}[\|X-\mu+\mu\|^2] = \mathbb{E}[\|X-\mu\|^2] + 2\langle \mathbb{E}[X-\mu],\mu\rangle + \|\mu\|^2$.
Since $\mathbb{E}[X-\mu]=0$, the cross term vanishes.
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}

\begin{proof}[Proof of Theorem \ref{thm:main}]

\textbf{[Step 1]} Apply the Descent Lemma \ref{lem:descent} with $w = w_t$ and $v = w_{t+1} = w_t - \eta g_t$:
\[
f(w_{t+1}) \leq f(w_t) + \langle \nabla f(w_t),\, w_{t+1} - w_t\rangle + \frac{L}{2}\|w_{t+1} - w_t\|^2.
\]

\textbf{[Step 2]} Substitute $w_{t+1} - w_t = -\eta g_t$:
\[
f(w_{t+1}) \leq f(w_t) - \eta \langle \nabla f(w_t),\, g_t\rangle + \frac{L\eta^2}{2}\|g_t\|^2.
\]

\textbf{[Step 3]} Take the conditional expectation given $w_t$ (over the random minibatch $\mathcal{B}_t$). By linearity:
\[
\mathbb{E}[f(w_{t+1}) \mid w_t]
\leq f(w_t) - \eta \langle \nabla f(w_t),\, \mathbb{E}[g_t \mid w_t]\rangle + \frac{L\eta^2}{2}\,\mathbb{E}[\|g_t\|^2 \mid w_t].
\]
Here $\nabla f(w_t)$ and $f(w_t)$ are deterministic given $w_t$.

\textbf{[Step 4]} Use unbiasedness (Assumption \ref{ass:unbiased}), $\mathbb{E}[g_t \mid w_t] = \nabla f(w_t)$, in the inner-product term:
\[
\langle \nabla f(w_t),\, \mathbb{E}[g_t \mid w_t]\rangle = \langle \nabla f(w_t),\, \nabla f(w_t)\rangle = \|\nabla f(w_t)\|^2.
\]

\textbf{[Step 5]} Bound the second-moment term via Lemma \ref{lem:decomp} with $X=g_t$, $\mu = \nabla f(w_t)$:
\[
\mathbb{E}[\|g_t\|^2 \mid w_t] = \|\nabla f(w_t)\|^2 + \mathbb{E}[\|g_t - \nabla f(w_t)\|^2 \mid w_t]
\leq \|\nabla f(w_t)\|^2 + \frac{\sigma^2}{b},
\]
where the inequality uses the bounded variance Assumption \ref{ass:var}.

\textbf{[Step 6]} Substitute Steps 4 and 5 into Step 3:
\begin{align*}
\mathbb{E}[f(w_{t+1}) \mid w_t]
&\leq f(w_t) - \eta \|\nabla f(w_t)\|^2 + \frac{L\eta^2}{2}\left(\|\nabla f(w_t)\|^2 + \frac{\sigma^2}{b}\right) \\
&= f(w_t) - \eta\left(1 - \frac{L\eta}{2}\right)\|\nabla f(w_t)\|^2 + \frac{L\eta^2 \sigma^2}{2b}.
\end{align*}

\textbf{[Step 7]} Use the step-size condition $\eta \leq \frac{1}{L}$, which implies $\frac{L\eta}{2} \leq \frac{1}{2}$, hence
\[
1 - \frac{L\eta}{2} \geq \frac{1}{2}.
\]
Therefore $-\eta\left(1 - \frac{L\eta}{2}\right) \leq -\frac{\eta}{2}$, giving
\[
\mathbb{E}[f(w_{t+1}) \mid w_t] \leq f(w_t) - \frac{\eta}{2}\|\nabla f(w_t)\|^2 + \frac{L\eta^2 \sigma^2}{2b}.
\]

\textbf{[Step 8]} Take total expectation over all randomness up to iteration $t$ (tower property $\mathbb{E}[\mathbb{E}[\cdot\mid w_t]] = \mathbb{E}[\cdot]$):
\[
\mathbb{E}[f(w_{t+1})] \leq \mathbb{E}[f(w_t)] - \frac{\eta}{2}\mathbb{E}[\|\nabla f(w_t)\|^2] + \frac{L\eta^2 \sigma^2}{2b}.
\]

\textbf{[Step 9]} Rearrange to isolate the gradient-norm term:
\[
\frac{\eta}{2}\mathbb{E}[\|\nabla f(w_t)\|^2] \leq \mathbb{E}[f(w_t)] - \mathbb{E}[f(w_{t+1})] + \frac{L\eta^2 \sigma^2}{2b}.
\]

\textbf{[Step 10]} Sum over $t = 0, 1, \dots, T-1$. The right-hand side telescopes:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2]
\leq \big(f(w_0) - \mathbb{E}[f(w_T)]\big) + T \cdot \frac{L\eta^2 \sigma^2}{2b}.
\]

\textbf{[Step 11]} Apply the lower bound $\mathbb{E}[f(w_T)] \geq f^*$ (Assumption \ref{ass:lb}), so that $f(w_0) - \mathbb{E}[f(w_T)] \leq f(w_0) - f^*$:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2]
\leq \big(f(w_0) - f^*\big) + \frac{T L\eta^2 \sigma^2}{2b}.
\]

\textbf{[Step 12]} Multiply both sides by $\frac{2}{\eta T}$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2]
\leq \frac{2\big(f(w_0) - f^*\big)}{\eta T} + \frac{L\eta \sigma^2}{b}.
\]

\textbf{[Step 13]} Since $R$ is uniform on $\{0,\dots,T-1\}$, by definition
\[
\mathbb{E}[\|\nabla f(w_R)\|^2] = \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2],
\]
which establishes both stated inequalities and completes the proof.
\end{proof}

\section*{Rate Derivation (With Constants)}

\textbf{[Step 14]} Define $\Delta := f(w_0) - f^*$. From Step 12 the bound is
\[
\Phi(\eta) := \frac{2\Delta}{\eta T} + \frac{L\sigma^2}{b}\eta.
\]

\textbf{[Step 15]} Ignoring the constraint $\eta \le 1/L$, minimize $\Phi$ over $\eta>0$. Setting $\Phi'(\eta) = -\frac{2\Delta}{\eta^2 T} + \frac{L\sigma^2}{b} = 0$ gives
\[
\eta^\star = \sqrt{\frac{2\Delta b}{L\sigma^2 T}}.
\]

\textbf{[Step 16]} The constrained optimum is $\eta = \min\{1/L,\ \eta^\star\}$. Substituting $\eta^\star$ into $\Phi$:
\[
\Phi(\eta^\star) = 2\sqrt{\frac{2\Delta \cdot L\sigma^2}{bT}} = 2\sqrt{\frac{2 L \sigma^2 \Delta}{b\,T}}.
\]

\textbf{[Step 17]} When $\eta = 1/L$ is active (small $T$ regime), the first term dominates:
\[
\Phi(1/L) = \frac{2L\Delta}{T} + \frac{\sigma^2}{b}.
\]
Combining both regimes:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2]
\leq \frac{2L\Delta}{T} + 2\sqrt{\frac{2L\sigma^2 \Delta}{bT}}
= O\!\left(\frac{1}{T} + \frac{1}{\sqrt{bT}}\right),
\]
matching Corollary \ref{cor:rate}.

\section*{Special Cases}

\textbf{Case 1: Deterministic (full-batch) gradient descent ($\sigma^2 = 0$).}
The noise term vanishes and Step 12 yields
\[
\min_{0\le t < T}\|\nabla f(w_t)\|^2 \le \frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^2
\leq \frac{2(f(w_0)-f^*)}{\eta T} = O\!\left(\frac{1}{T}\right),
\]
the classical non-convex GD rate, so $\min_t\|\nabla f(w_t)\| = O(1/\sqrt{T})$.

\textbf{Case 2: Large batch limit ($b \to \infty$).}
The statistical floor $\frac{L\eta\sigma^2}{b} \to 0$, recovering the deterministic $O(1/T)$ behavior.

\textbf{Case 3: Fixed step size, $T \to \infty$.}
The optimization term vanishes but a residual remains:
\[
\limsup_{T\to\infty}\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}[\|\nabla f(w_t)\|^2] \leq \frac{L\eta\sigma^2}{b},
\]
i.e., fixed-step SGD converges to a neighborhood of a stationary point whose size scales with $\eta$ and $1/b$.

\textbf{Case 4: Diminishing step size $\eta_t = \frac{c}{\sqrt{t+1}}$.}
A standard extension (replacing the fixed $\eta$ by $\eta_t$ in Steps 6--12) yields
\[
\min_{0\le t<T}\mathbb{E}[\|\nabla f(w_t)\|^2] = O\!\left(\frac{\log T}{\sqrt{T}}\right),
\]
ensuring exact convergence of the minimum expected gradient norm to zero.
