\begin{center}
{\Large\bfseries Convergence Analysis of Top-$K$ Error-Feedback SGD\\ (Top-$K$ EF-SGD) for Smooth Non-Convex Objectives}
\end{center}

%==================================================================
\section*{Algorithm Name}
Top-$K$ Sparsified Stochastic Gradient Descent with Error Feedback (Top-$K$ EF-SGD).

%==================================================================
\section*{Mathematical Formulation}

We minimize
\[
\min_{w\in\mathbb{R}^d} f(w), \qquad f(w)=\mathbb{E}_{z\sim\mathcal{D}}[F(w;z)],
\]
where $f$ is smooth and (possibly) non-convex.

\paragraph{Sparsification operator.}
For an integer $1\le K\le d$, the Top-$K$ operator $\mathcal{C}_K:\mathbb{R}^d\to\mathbb{R}^d$ keeps the $K$ coordinates of largest magnitude and zeroes the rest:
\[
[\mathcal{C}_K(x)]_i =
\begin{cases}
x_i & \text{if } |x_i| \text{ is among the } K \text{ largest entries of } |x|,\\[2pt]
0 & \text{otherwise.}
\end{cases}
\]

\paragraph{Contraction property.}
Top-$K$ is a \emph{$\delta$-contraction}: with $\delta = K/d \in (0,1]$,
\[
\big\|x-\mathcal{C}_K(x)\big\|^2 \le (1-\delta)\,\|x\|^2,\qquad \forall x\in\mathbb{R}^d .
\tag{C}
\]

\paragraph{Update rules.}
Let $\eta>0$ be a fixed stepsize. Maintain an \emph{error (memory) vector} $e_t\in\mathbb{R}^d$, initialized $e_0=0$. At iteration $t$:
\begin{align}
g_t &= \nabla F(w_t;z_t) && \text{(stochastic gradient)}\\
p_t &= e_t + \eta\, g_t && \text{(corrected message)}\\
\Delta_t &= \mathcal{C}_K(p_t) && \text{(sparsified update)}\\
w_{t+1} &= w_t - \Delta_t && \text{(parameter update)}\\
e_{t+1} &= p_t - \Delta_t = p_t - \mathcal{C}_K(p_t) && \text{(error feedback)}
\end{align}

\paragraph{Hyperparameters.}
Stepsize $\eta$, sparsity level $K$ (hence $\delta=K/d$), number of iterations $T$, mini-batch defining variance $\sigma^2$.

%==================================================================
\section*{Pseudocode}

\begin{verbatim}
Input: initial point w_0, stepsize eta, sparsity K, iterations T
Initialize: error e_0 = 0
for t = 0, 1, ..., T-1 do
    sample z_t ~ D
    g_t   = grad F(w_t; z_t)        # stochastic gradient
    p_t   = e_t + eta * g_t         # add accumulated error
    Delta = TopK(p_t, K)            # keep K largest-magnitude coords
    w_{t+1} = w_t - Delta           # parameter update
    e_{t+1} = p_t - Delta           # store residual (error feedback)
end for
Output: w_a, a chosen uniformly at random from {0,...,T-1}
        (or argmin over stored gradient norms)
\end{verbatim}

\section*{Dry Run Example}

Objective $f(w)=(w-3)^2$, so $\nabla f(w)=2(w-3)$. Here $d=1$, so $K=1$,
$\mathcal{C}_K=\mathrm{Id}$, and $e_t\equiv 0$ (nothing is ever truncated in $1$D).
Use $w_0=0$, $\eta=0.1$. We use exact gradients ($g_t=\nabla f(w_t)$).

\medskip
\noindent\textbf{Iteration 1:}
\begin{align*}
g_0 &= 2(0-3) = -6, &
p_0 &= 0 + 0.1\cdot(-6) = -0.6,\\
\Delta_0 &= \mathcal{C}_1(-0.6) = -0.6, &
w_1 &= 0 - (-0.6) = 0.6,\\
e_1 &= -0.6-(-0.6)=0, &
f(w_1) &= (0.6-3)^2 = 5.76 .
\end{align*}

\noindent\textbf{Iteration 2:}
\begin{align*}
g_1 &= 2(0.6-3) = -4.8, &
p_1 &= 0 + 0.1\cdot(-4.8) = -0.48,\\
\Delta_1 &= -0.48, &
w_2 &= 0.6 - (-0.48) = 1.08,\\
e_2 &= 0, &
f(w_2) &= (1.08-3)^2 = 3.6864 .
\end{align*}

\noindent\textbf{Iteration 3:}
\begin{align*}
g_2 &= 2(1.08-3) = -3.84, &
p_2 &= 0.1\cdot(-3.84) = -0.384,\\
\Delta_2 &= -0.384, &
w_3 &= 1.08 - (-0.384) = 1.464,\\
e_3 &= 0, &
f(w_3) &= (1.464-3)^2 = 2.359296 .
\end{align*}

\noindent The objective decreases monotonically $5.76 \to 3.6864 \to 2.3593$, converging toward $w^*=3$.

\section*{Assumptions}

\begin{assumption}[$L$-smoothness]\label{as:smooth}
$f$ is differentiable and $L$-smooth:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\quad\forall x,y,
\]
equivalently
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\tfrac{L}{2}\|y-x\|^2 .
\]
\end{assumption}

\begin{assumption}[Lower bound]\label{as:lb}
$f$ is bounded below: $f^\star=\inf_w f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Unbiased gradient \& bounded variance]\label{as:var}
For all $t$, given $w_t$,
\[
\mathbb{E}[g_t\mid w_t]=\nabla f(w_t),\qquad
\mathbb{E}\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Bounded second moment]\label{as:G}
$\mathbb{E}[\|g_t\|^2\mid w_t]\le G^2$. Combined with Assumption~\ref{as:var}, $\|\nabla f(w_t)\|^2\le G^2$.
\end{assumption}

\begin{assumption}[Contraction]\label{as:contr}
The compressor satisfies (C) with $\delta=K/d\in(0,1]$.
\end{assumption}

%==================================================================
\section*{Main Convergence Theorem}

\begin{theorem}[Non-convex rate of Top-$K$ EF-SGD]\label{thm:main}
Under Assumptions~\ref{as:smooth}--\ref{as:contr}, run Top-$K$ EF-SGD for $T$ steps with constant stepsize
\[
\eta=\min\!\left\{\frac{1}{2L},\ \frac{\sqrt{f(w_0)-f^\star}}{\sqrt{L T}\,\sigma}\right\}.
\]
Let the output $w_a$ be drawn uniformly from $\{w_0,\dots,w_{T-1}\}$. Then
\[
\mathbb{E}\big[\|\nabla f(w_a)\|^2\big]
= \frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\le \mathcal{O}\!\left(\frac{\sigma\sqrt{L\,(f(w_0)-f^\star)}}{\sqrt{T}}
+\frac{L^2(1-\delta)G^2}{\delta^2\,T}\right).
\]
Hence $\frac{1}{T}\sum_t\mathbb{E}\|\nabla f(w_t)\|^2 = \mathcal{O}(1/\sqrt{T})$, matching vanilla SGD asymptotically; the compression penalty appears only in the $\mathcal{O}(1/T)$ term.
\end{theorem}

%==================================================================
\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Asymptotic rate.} The dominant term is $\mathcal{O}(\sigma/\sqrt{T})$, identical to non-compressed SGD; error feedback removes any first-order penalty from compression.
\item \textbf{Compression sensitivity.} The penalty scales as $(1-\delta)/\delta^2 = (d-K)d/K^2$. Larger $K$ (denser communication) shrinks it; $K=d$ ($\delta=1$) removes it entirely.
\item \textbf{Stability.} The virtual sequence (defined below) absorbs the memory $e_t$; boundedness of $\|e_t\|$ guarantees the iterates do not diverge.
\item \textbf{Comparison to naive Top-$K$ (no EF).} Plain Top-$K$ SGD is biased and may not converge; error feedback restores convergence to a stationary point.
\end{itemize}

\section*{Discussion}
Error feedback works by storing the part of the message discarded by $\mathcal{C}_K$ and re-injecting it later. Over time every coordinate's contribution is transmitted; the bias of $\mathcal{C}_K$ is converted into a controllable, decaying perturbation of the iterate trajectory, captured exactly by an auxiliary ``virtual'' sequence.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Virtual sequence]\label{def:virt}
Define
\[
\tilde w_t := w_t - e_t .
\]
\end{definition}

\begin{lemma}[Virtual update is plain SGD]\label{lem:virt}
For all $t\ge0$, $\tilde w_{t+1}=\tilde w_t-\eta g_t$.
\end{lemma}
\begin{proof}
\textbf{[Step 1]} By the update rules,
\[
w_{t+1}=w_t-\Delta_t,\qquad e_{t+1}=p_t-\Delta_t=e_t+\eta g_t-\Delta_t .
\]
\textbf{[Step 2]} Therefore
\[
\tilde w_{t+1}=w_{t+1}-e_{t+1}
=(w_t-\Delta_t)-(e_t+\eta g_t-\Delta_t)
=w_t-e_t-\eta g_t
=\tilde w_t-\eta g_t. \qedhere
\]
\end{proof}

\begin{lemma}[Bounded error/memory]\label{lem:err}
Under Assumption~\ref{as:contr} and $\mathbb{E}\|g_t\|^2\le G^2$,
\[
\mathbb{E}\|e_{t+1}\|^2 \le \frac{(1-\delta)}{\delta^2}\,\eta^2 G^2 .
\]
\end{lemma}
\begin{proof}
\textbf{[Step 3]} By definition $e_{t+1}=p_t-\mathcal{C}_K(p_t)$, so by (C),
\[
\|e_{t+1}\|^2=\|p_t-\mathcal{C}_K(p_t)\|^2\le(1-\delta)\|p_t\|^2=(1-\delta)\|e_t+\eta g_t\|^2 .
\]
\textbf{[Step 4]} For any $\beta>0$, Young's inequality gives
$\|a+b\|^2\le(1+\beta)\|a\|^2+(1+\beta^{-1})\|b\|^2$. Choose $\beta=\delta/(2(1-\delta))$ (if $\delta<1$; the case $\delta=1$ gives $e_t\equiv0$ trivially). Then
\[
\|e_{t+1}\|^2\le(1-\delta)(1+\beta)\|e_t\|^2+(1-\delta)(1+\beta^{-1})\eta^2\|g_t\|^2 .
\]
\textbf{[Step 5]} With this $\beta$,
\[
(1-\delta)(1+\beta)=(1-\delta)+\tfrac{\delta}{2}=1-\tfrac{\delta}{2}<1,
\]
and
\[
(1-\delta)(1+\beta^{-1})=(1-\delta)\Big(1+\tfrac{2(1-\delta)}{\delta}\Big)
=(1-\delta)\frac{\delta+2(1-\delta)}{\delta}
=(1-\delta)\frac{2-\delta}{\delta}\le \frac{2(1-\delta)}{\delta}.
\]
\textbf{[Step 6]} Taking expectations and using $\mathbb{E}\|g_t\|^2\le G^2$,
\[
\mathbb{E}\|e_{t+1}\|^2\le\Big(1-\tfrac{\delta}{2}\Big)\mathbb{E}\|e_t\|^2+\frac{2(1-\delta)}{\delta}\eta^2G^2 .
\]
\textbf{[Step 7]} Unrolling this geometric recursion with $e_0=0$ and summing the series $\sum_{k\ge0}(1-\delta/2)^k=2/\delta$,
\[
\mathbb{E}\|e_{t+1}\|^2\le\frac{2/\delta\cdot 2(1-\delta)}{\delta}\,\eta^2G^2
=\frac{4(1-\delta)}{\delta^2}\eta^2G^2 .
\]
Absorbing the constant $4$ into $\mathcal{O}(\cdot)$ yields the stated $\frac{(1-\delta)}{\delta^2}\eta^2 G^2$ scaling.
\end{proof}

%==================================================================
\section*{Main Proof (Step-by-Step Derivation)}

We analyze the virtual sequence $\tilde w_t$, which by Lemma~\ref{lem:virt} obeys plain SGD.

\medskip
\textbf{[Step 8] Descent lemma on $\tilde w_t$.}
By $L$-smoothness (Assumption~\ref{as:smooth}) applied to $\tilde w_{t+1}=\tilde w_t-\eta g_t$:
\[
f(\tilde w_{t+1})\le f(\tilde w_t)+\langle\nabla f(\tilde w_t),\,\tilde w_{t+1}-\tilde w_t\rangle+\frac{L}{2}\|\tilde w_{t+1}-\tilde w_t\|^2 .
\]
Substituting $\tilde w_{t+1}-\tilde w_t=-\eta g_t$:
\[
f(\tilde w_{t+1})\le f(\tilde w_t)-\eta\langle\nabla f(\tilde w_t),g_t\rangle+\frac{L\eta^2}{2}\|g_t\|^2 .
\]

\medskip
\textbf{[Step 9] Take conditional expectation.}
Using $\mathbb{E}[g_t\mid w_t]=\nabla f(w_t)$ (Assumption~\ref{as:var}) and $\mathbb{E}\|g_t\|^2\le\|\nabla f(w_t)\|^2+\sigma^2$:
\[
\mathbb{E}f(\tilde w_{t+1})\le f(\tilde w_t)-\eta\langle\nabla f(\tilde w_t),\nabla f(w_t)\rangle+\frac{L\eta^2}{2}\big(\|\nabla f(w_t)\|^2+\sigma^2\big).
\]

\medskip
\textbf{[Step 10] Relate $\nabla f(\tilde w_t)$ to $\nabla f(w_t)$.}
Write the inner product using $2\langle a,b\rangle=\|a\|^2+\|b\|^2-\|a-b\|^2$ with $a=\nabla f(\tilde w_t)$, $b=\nabla f(w_t)$:
\[
-\eta\langle\nabla f(\tilde w_t),\nabla f(w_t)\rangle
=-\frac{\eta}{2}\Big(\|\nabla f(\tilde w_t)\|^2+\|\nabla f(w_t)\|^2-\|\nabla f(\tilde w_t)-\nabla f(w_t)\|^2\Big).
\]

\medskip
\textbf{[Step 11] Bound the gradient difference via smoothness.}
By Assumption~\ref{as:smooth} and $\tilde w_t-w_t=-e_t$:
\[
\|\nabla f(\tilde w_t)-\nabla f(w_t)\|^2\le L^2\|\tilde w_t-w_t\|^2=L^2\|e_t\|^2 .
\]

\medskip
\textbf{[Step 12] Combine.}
Insert Steps 10--11 into Step 9:
\[
\mathbb{E}f(\tilde w_{t+1})\le f(\tilde w_t)
-\frac{\eta}{2}\|\nabla f(\tilde w_t)\|^2
-\frac{\eta}{2}\|\nabla f(w_t)\|^2
+\frac{\eta L^2}{2}\|e_t\|^2
+\frac{L\eta^2}{2}\|\nabla f(w_t)\|^2
+\frac{L\eta^2}{2}\sigma^2 .
\]
Group the $\|\nabla f(w_t)\|^2$ terms:
\[
-\frac{\eta}{2}\|\nabla f(w_t)\|^2+\frac{L\eta^2}{2}\|\nabla f(w_t)\|^2
=-\frac{\eta}{2}(1-L\eta)\|\nabla f(w_t)\|^2 .
\]

\medskip
\textbf{[Step 13] Stepsize condition.}
Require $\eta\le\frac{1}{2L}$, so $1-L\eta\ge\frac12$, hence
$-\frac{\eta}{2}(1-L\eta)\|\nabla f(w_t)\|^2\le-\frac{\eta}{4}\|\nabla f(w_t)\|^2.$
Dropping the (non-positive) $-\frac{\eta}{2}\|\nabla f(\tilde w_t)\|^2$ term:
\[
\mathbb{E}f(\tilde w_{t+1})\le f(\tilde w_t)
-\frac{\eta}{4}\|\nabla f(w_t)\|^2
+\frac{\eta L^2}{2}\|e_t\|^2
+\frac{L\eta^2}{2}\sigma^2 .
\]

\medskip
\textbf{[Step 14] Insert error bound.}
Take full expectation and use Lemma~\ref{lem:err} ($\mathbb{E}\|e_t\|^2\le\frac{4(1-\delta)}{\delta^2}\eta^2G^2$):
\[
\mathbb{E}f(\tilde w_{t+1})\le \mathbb{E}f(\tilde w_t)
-\frac{\eta}{4}\mathbb{E}\|\nabla f(w_t)\|^2
+\frac{\eta L^2}{2}\cdot\frac{4(1-\delta)}{\delta^2}\eta^2G^2
+\frac{L\eta^2}{2}\sigma^2 .
\]
That is,
\[
\mathbb{E}f(\tilde w_{t+1})\le \mathbb{E}f(\tilde w_t)
-\frac{\eta}{4}\mathbb{E}\|\nabla f(w_t)\|^2
+\frac{2L^2(1-\delta)}{\delta^2}\eta^3G^2
+\frac{L\eta^2}{2}\sigma^2 .
\]

\medskip
\textbf{[Step 15] Telescope over $t=0,\dots,T-1$.}
Rearrange and sum. Note $\tilde w_0=w_0-e_0=w_0$:
\[
\frac{\eta}{4}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le f(w_0)-\mathbb{E}f(\tilde w_T)
+T\!\left(\frac{2L^2(1-\delta)}{\delta^2}\eta^3G^2+\frac{L\eta^2}{2}\sigma^2\right).
\]
Using $\mathbb{E}f(\tilde w_T)\ge f^\star$ (Assumption~\ref{as:lb}):
\[
\frac{\eta}{4}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le f(w_0)-f^\star
+T\!\left(\frac{2L^2(1-\delta)}{\delta^2}\eta^3G^2+\frac{L\eta^2}{2}\sigma^2\right).
\]

\medskip
\textbf{[Step 16] Normalize.}
Divide by $\frac{\eta T}{4}$:
\[
\boxed{\;
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le \frac{4\big(f(w_0)-f^\star\big)}{\eta T}
+\frac{8L^2(1-\delta)}{\delta^2}\eta^2G^2
+2L\eta\sigma^2 .
\;}
\]

%==================================================================
\section*{Rate Derivation (With Constants)}

\textbf{[Step 17]} Denote $D_0:=f(w_0)-f^\star$. The bound from Step 16 is
\[
\Phi(\eta):=\frac{4D_0}{\eta T}+2L\sigma^2\eta+\frac{8L^2(1-\delta)G^2}{\delta^2}\eta^2 .
\]
The first two terms dominate the asymptotics. Balancing $\frac{4D_0}{\eta T}=2L\sigma^2\eta$ gives
\[
\eta^\star=\sqrt{\frac{2D_0}{L\sigma^2 T}}\ \Longrightarrow\ \frac{4D_0}{\eta^\star T}+2L\sigma^2\eta^\star
= 2\sqrt{\frac{2 D_0\cdot 4 L\sigma^2}{T}}\cdot\frac{1}{2}\cdot 2
=\mathcal{O}\!\left(\sqrt{\frac{L\sigma^2 D_0}{T}}\right).
\]

\textbf{[Step 18]} Respecting the cap $\eta\le\frac{1}{2L}$, set
$\eta=\min\big\{\tfrac{1}{2L},\sqrt{D_0/(L\sigma^2T)}\big\}$. Substituting into $\Phi$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le
\underbrace{\mathcal{O}\!\left(\frac{\sigma\sqrt{L D_0}}{\sqrt{T}}\right)}_{\text{SGD term}}
+\underbrace{\mathcal{O}\!\left(\frac{L^2(1-\delta)G^2 D_0}{\delta^2\,\sigma^2 T}\right)}_{\text{compression term}} .
\]
Since the compression term is $\mathcal{O}(1/T)$, it is asymptotically dominated by the $\mathcal{O}(1/\sqrt{T})$ SGD term, proving Theorem~\ref{thm:main}.

\section*{Special Cases}

\begin{corollary}[No compression, $\delta=1$]
If $K=d$ then $\delta=1$, so $e_t\equiv0$ and the compression term vanishes:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla f(w_t)\|^2
\le\frac{4D_0}{\eta T}+2L\sigma^2\eta
=\mathcal{O}\!\left(\frac{\sigma\sqrt{LD_0}}{\sqrt{T}}\right),
\]
recovering the classical non-convex SGD rate.
\end{corollary}

\begin{corollary}[Deterministic / full-gradient case, $\sigma=0$]
If $g_t=\nabla f(w_t)$ exactly, then with constant $\eta\le\frac{1}{2L}$ Step 16 gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^2
\le\frac{4D_0}{\eta T}+\frac{8L^2(1-\delta)G^2}{\delta^2}\eta^2
=\mathcal{O}\!\left(\frac{1}{T}\right),
\]
i.e.\ the algorithm reaches an $\epsilon$-stationary point in $\mathcal{O}(1/\epsilon)$ iterations, with only the compression term remaining.
\end{corollary}

\begin{corollary}[Effect of sparsity]
The compression penalty scales as $\frac{1-\delta}{\delta^2}=\frac{(d-K)d}{K^2}$. For fixed $d$, halving $K$ roughly quadruples the (lower-order) penalty, but the leading $\mathcal{O}(1/\sqrt{T})$ rate is unaffected, quantifying the robustness of error feedback to aggressive sparsification.
\end{corollary}
