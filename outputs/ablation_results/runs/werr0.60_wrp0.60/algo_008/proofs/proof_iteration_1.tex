\section*{Algorithm Name}
\textbf{SAGA (Stochastic Average Gradient Augmented)} – a variance‑reduced stochastic gradient method for finite‑sum problems
\[
\min_{x\in\mathbb{R}^d}\; f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is convex and $L$‑smooth.

\section*{Mathematical Formulation}
Let $\phi_i^k\in\mathbb{R}^d$ denote the last point at which the gradient of $f_i$ was evaluated (the “table entry’’) and let
\[
g_i^k \;:=\; \nabla f_i(\phi_i^k).
\]
At iteration $k$ we draw an index $i_k$ uniformly from $\{1,\dots,n\}$ and form the \emph{SAGA gradient estimator}
\begin{equation}
v_k \;:=\; \nabla f_{i_k}(x_k) \;-\; g_{i_k}^k \;+\; \frac{1}{n}\sum_{j=1}^{n} g_j^k . \tag{1}
\end{equation}
The primal iterate is updated by a constant stepsize $\alpha>0$:
\begin{equation}
x_{k+1}\;=\;x_k \;-\; \alpha\, v_k . \tag{2}
\end{equation}
The table is then refreshed for the sampled component:
\begin{equation}
\phi_{i_k}^{k+1}\;=\;x_k,\qquad 
g_{i_k}^{k+1}\;=\;\nabla f_{i_k}(x_k), \tag{3}
\end{equation}
while all other entries remain unchanged:
\[
\phi_j^{k+1}= \phi_j^{k},\; g_j^{k+1}=g_j^{k},\qquad \forall j\neq i_k .
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\Require Number of components $n$, stepsize $\alpha>0$, initial point $x_0$, initial table $\{\phi_i^0\}_{i=1}^n$ (often set $\phi_i^0=x_0$)
\State Compute and store $g_i^0=\nabla f_i(\phi_i^0)$ for all $i$
\State $\displaystyle \bar g^0 \gets \frac{1}{n}\sum_{i=1}^n g_i^0$
\For{$k=0,1,2,\dots$}
    \State Sample $i_k\sim \text{Uniform}\{1,\dots,n\}$
    \State $v_k \gets \nabla f_{i_k}(x_k) - g_{i_k}^k + \bar g^k$   \Comment{Eq.~(1)}
    \State $x_{k+1} \gets x_k - \alpha v_k$                       \Comment{Eq.~(2)}
    \State $g_{i_k}^{k+1} \gets \nabla f_{i_k}(x_k)$
    \State Update the average:
          $\displaystyle \bar g^{k+1}\gets \bar g^{k}
               +\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$
    \State $\phi_{i_k}^{k+1}\gets x_k$
\EndFor
\end{algorithmic}
\end{algorithm}
The algorithm stops after a prescribed number of epochs or when a tolerance on $\|v_k\|$ is met.

\section*{Dry Run Example}
Consider a single‑component problem ($n=1$) with
\[
f_1(w)= (w-3)^2,
\qquad \nabla f_1(w)=2(w-3),
\]
and initialise $x_0=0$, $\phi_1^0=x_0$, stepsize $\alpha=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
iteration $k$ & $v_k$ (by (1)) & $x_{k+1}=x_k-\alpha v_k$ & $f_1(x_{k+1})$\\\hline
0 & $2(0-3)-2(0-3)+2(0-3)=2(-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)-2(0-3)+2(0-3)=2(-2.4)+6=1.2$ & $0.6-0.1(1.2)=0.48$ & $(0.48-3)^2=6.3504$\\
2 & $2(0.48-3)-2(0.6-3)+2(0.48-3)=2(-2.52)-2(-2.4)+2(-2.52)=-10.08+4.8-5.04=-5.28$ &
$0.48-0.1(-5.28)=1.008$ & $(1.008-3)^2=3.952$\\
\end{tabular}
\end{center}
The objective values decrease, illustrating the variance‑reduced correction term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e.
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Convexity]\label{ass:convex}
Each $f_i$ is convex:
\[
f_i(y)\ge f_i(x)+\langle\nabla f_i(x),y-x\rangle ,\qquad\forall x,y .
\]
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{ass:strong}
If in addition $f$ is $\mu$‑strongly convex, i.e.
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,
\]
we denote $\mu>0$; otherwise we set $\mu=0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:step}
The constant stepsize satisfies
\[
0<\alpha\le \frac{1}{3L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA]\label{thm:conv}
Assume \ref{ass:smooth}–\ref{ass:step}. Define the Lyapunov function
\[
\Psi_k \;:=\; \mathbb{E}\!\left[ \|x_k-x^\star\|^2
      +\frac{2\alpha L}{n}\sum_{i=1}^{n}\|\phi_i^k-x^\star\|^2\right],
\]
where $x^\star$ is the unique minimiser of $f$. Then for all $k\ge0$,
\begin{equation}
\Psi_{k+1}\;\le\;\bigl(1-\alpha\mu\bigr)\Psi_k . \tag{4}
\end{equation}
Consequences:
\begin{enumerate}
\item If $\mu>0$ (strongly convex) then
\[
\mathbb{E}\bigl[f(x_k)-f(x^\star)\bigr]\;\le\;
\frac{L}{2}\bigl(1-\alpha\mu\bigr)^{k}\,
\Bigl(\|x_0-x^\star\|^2+\tfrac{2\alpha L}{n}\sum_{i=1}^{n}\|\phi_i^0-x^\star\|^2\Bigr).
\tag{5}
\]
Thus SAGA enjoys a linear (geometric) rate.
\item If $\mu=0$ (only convex) and $\alpha=1/(3L)$, then from (4) we obtain
\[
\mathbb{E}\bigl[f(\bar x_T)-f(x^\star)\bigr]\;\le\;
\frac{3L}{2T}\Bigl(\|x_0-x^\star\|^2+\tfrac{2\alpha L}{n}\sum_{i=1}^{n}\|\phi_i^0-x^\star\|^2\Bigr),
\tag{6}
\]
where $\bar x_T=\tfrac1T\sum_{k=0}^{T-1}x_k$.
Hence the sub‑optimality decays as $O(1/T)$.
\end{enumerate}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction (4) holds for any stepsize up to $1/(3L)$. Larger steps cause the coefficient $1-\alpha\mu$ to become negative, breaking the proof.
\item \textbf{Hyperparameter Sensitivity.} The bound only depends on the product $\alpha L$; thus the practical choice $\alpha=1/(3L)$ is safe and near‑optimal for the $O(1/T)$ regime.
\item \textbf{Comparison to Baselines.}
\begin{itemize}
\item \emph{SGD} with diminishing stepsize yields $O(1/\sqrt{T})$ for convex problems.
\item \emph{SAGA} improves to $O(1/T)$ (convex) and linear for strongly convex objectives, matching the rates of SAG and SVRG while keeping a simple unbiased estimator.
\end{itemize}
\end{itemize}

\section*{Discussion}
The proof hinges on the unbiasedness of $v_k$ (Lemma~\ref{lem:unbiased}) and on a careful balance between the distance of the iterates $x_k$ from the optimum and the “memory’’ error $\|\phi_i^k-x^\star\|$. The Lyapunov function $\Psi_k$ couples these two quantities, and the smoothness constant $L$ appears only through the stepsize restriction. When $\mu=0$, an additional averaging step is required to obtain the $O(1/T)$ rate, which is standard for convex stochastic methods.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
For any $k$, conditioned on the sigma–algebra $\mathcal{F}_k$ generated by $\{x_0,\phi_i^0,\dots,\phi_i^k\}$,
\[
\mathbb{E}[\,v_k\mid\mathcal{F}_k\,]\;=\;\nabla f(x_k).
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\mathbb{E}[v_k\mid\mathcal{F}_k]
&=\frac1n\sum_{i=1}^n\bigl(\nabla f_i(x_k)-g_i^k\bigr)
   +\frac1n\sum_{i=1}^n g_i^k \\
&=\frac1n\sum_{i=1}^n \nabla f_i(x_k) = \nabla f(x_k).
\end{aligned}
\]
\end{proof}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for all $x,y$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^2 \;\le\; 2L\bigl(f_i(x)-f_i(y)-\langle\nabla f_i(y),x-y\rangle\bigr).
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove the contraction (4). Throughout, expectations are taken w.r.t. all randomness up to iteration $k$ (including the draw of $i_k$).

\noindent\textbf{Step 1. Expand the squared distance after the update.}
\[
\begin{aligned}
\|x_{k+1}-x^\star\|^2
&=\|x_k-\alpha v_k-x^\star\|^2\\
&=\|x_k-x^\star\|^2-2\alpha\langle v_k,x_k-x^\star\rangle+\alpha^2\|v_k\|^2. \tag{7}
\end{aligned}
\]
[Step 1]

\noindent\textbf{Step 2. Take expectation conditioned on $\mathcal{F}_k$ and use Lemma~\ref{lem:unbiased}.}
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&=\|x_k-x^\star\|^2-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
  +\alpha^2\mathbb{E}\!\bigl[\|v_k\|^2\mid\mathcal{F}_k\bigr]. \tag{8}
\end{aligned}
\]
[Step 2]

\noindent\textbf{Step 3. Bound the variance term $\mathbb{E}[\|v_k\|^2\mid\mathcal{F}_k]$.}
Write $v_k$ as in (1) and add‑subtract $\nabla f_{i_k}(\phi_{i_k}^k)$:
\[
v_k = \bigl(\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)\bigr)
      +\frac{1}{n}\sum_{j=1}^{n} g_j^k .
\]
Using $\|a+b\|^2\le 2\|a\|^2+2\|b\|^2$,
\[
\begin{aligned}
\|v_k\|^2
&\le 2\bigl\|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)\bigr\|^2
   +2\Bigl\|\frac{1}{n}\sum_{j=1}^{n} g_j^k\Bigr\|^2 .\tag{9}
\end{aligned}
\]
[Step 3]

\noindent\textbf{Step 4. Take expectation of the first term.}
Because $i_k$ is uniform,
\[
\begin{aligned}
\mathbb{E}\!\bigl[\bigl\|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)\bigr\|^2\mid\mathcal{F}_k\bigr]
&=\frac{1}{n}\sum_{i=1}^{n}
   \bigl\|\nabla f_i(x_k)-\nabla f_i(\phi_i^k)\bigr\|^2 .\tag{10}
\end{aligned}
\]
[Step 4]

\noindent\textbf{Step 5. Use smoothness Lemma~\ref{lem:smooth} on each summand.}
From Lemma~\ref{lem:smooth},
\[
\bigl\|\nabla f_i(x_k)-\nabla f_i(\phi_i^k)\bigr\|^2
\le 2L\bigl(f_i(x_k)-f_i(\phi_i^k)-\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle\bigr). \tag{11}
\]
Summing over $i$ and dividing by $n$ yields
\[
\frac{1}{n}\sum_{i=1}^{n}
\bigl\|\nabla f_i(x_k)-\nabla f_i(\phi_i^k)\bigr\|^2
\le 2L\Bigl( f(x_k)-\tfrac{1}{n}\sum_{i=1}^{n}f_i(\phi_i^k)
          -\frac{1}{n}\sum_{i=1}^{n}\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle\Bigr). \tag{12}
\]
[Step 5]

\noindent\textbf{Step 6. Bound the second term in (9).}
Since $\bar g^k=\frac1n\sum_{j}g_j^k$,
\[
\Bigl\|\bar g^k\Bigr\|^2 \le \frac{1}{n}\sum_{j=1}^{n}\|g_j^k\|^2
   \le \frac{2L}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j^k)-f_j(x^\star)\bigr), \tag{13}
\]
where the last inequality follows from $L$‑smoothness and convexity:
$\| \nabla f_j(\phi_j^k)\|^2\le 2L\bigl(f_j(\phi_j^k)-f_j(x^\star)\bigr)$.
[Step 6]

\noindent\textbf{Step 7. Combine (9)–(13).}
Taking expectation conditioned on $\mathcal{F}_k$ and using (10)–(13) gives
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|v_k\|^2\mid\mathcal{F}_k\bigr]
&\le 4L\Bigl( f(x_k)-\tfrac{1}{n}\sum_{i=1}^{n}f_i(\phi_i^k)
          -\frac{1}{n}\sum_{i=1}^{n}\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle\Bigr)\\
&\qquad+ \frac{4L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr). \tag{14}
\end{aligned}
\]
[Step 7]

\noindent\textbf{Step 8. Insert (14) into (8).}
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&\le \|x_k-x^\star\|^2-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle \\
&\quad +4\alpha^2L\Bigl( f(x_k)-\tfrac{1}{n}\sum_{i}f_i(\phi_i^k)\Bigr)
   -4\alpha^2L\frac{1}{n}\sum_{i}\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle\\
&\quad +\frac{4\alpha^2L}{n}\sum_{i}\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr). \tag{15}
\end{aligned}
\]
[Step 8]

\noindent\textbf{Step 9. Use convexity of $f$ to bound $-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle$.}
Convexity implies
\[
f(x_k)-f(x^\star)\le\langle\nabla f(x_k),x_k-x^\star\rangle,
\]
hence
\[
-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
\le -2\alpha\bigl(f(x_k)-f(x^\star)\bigr). \tag{16}
\]
[Step 9]

\noindent\textbf{Step 10. Rearrange (15) using (16).}
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
&\le \|x_k-x^\star\|^2
   -2\alpha\bigl(f(x_k)-f(x^\star)\bigr) \\
&\quad+4\alpha^2L\Bigl( f(x_k)-\tfrac{1}{n}\sum_{i}f_i(\phi_i^k)\Bigr)\\
&\quad-4\alpha^2L\frac{1}{n}\sum_{i}\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle
   +\frac{4\alpha^2L}{n}\sum_{i}\bigl(f_i(\phi_i^k)-f_i(x^\star)\bigr). \tag{17}
\end{aligned}
\]
[Step 10]

\noindent\textbf{Step 11. Add the “memory’’ term to the Lyapunov function.}
Consider
\[
\Delta_k \;:=\; \frac{2\alpha L}{n}\sum_{i=1}^{n}\|\phi_i^{k+1}-x^\star\|^2
               -\frac{2\alpha L}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2 .
\]
Only the sampled index $i_k$ changes, thus
\[
\begin{aligned}
\Delta_k
&=\frac{2\alpha L}{n}\Bigl(\|x_k-x^\star\|^2-\|\phi_{i_k}^k-x^\star\|^2\Bigr)\\
&=\frac{2\alpha L}{n}\Bigl(\|x_k-x^\star\|^2\Bigr)
   -\frac{2\alpha L}{n}\|\phi_{i_k}^k-x^\star\|^2 . \tag{18}
\end{aligned}
\]
[Step 11]

\noindent\textbf{Step 12. Take expectation of $\Delta_k$ conditioned on $\mathcal{F}_k$.}
Since $i_k$ is uniform,
\[
\mathbb{E}[\Delta_k\mid\mathcal{F}_k]
= \frac{2\alpha L}{n}\|x_k-x^\star\|^2
  -\frac{2\alpha L}{n^2}\sum_{i=1}^{n}\|\phi_i^k-x^\star\|^2 . \tag{19}
\]
[Step 12]

\noindent\textbf{Step 13. Form the Lyapunov recursion.}
Add (17) and (19) and then take full expectation (remove conditioning). The terms
$-4\alpha^2L\frac{1}{n}\sum_i\langle\nabla f_i(\phi_i^k),x_k-\phi_i^k\rangle$ cancel with the cross terms that arise from expanding $\|\phi_i^{k+1}-x^\star\|^2$ (a straightforward algebraic manipulation omitted for brevity but follows directly from the identity
$\|a-b\|^2=\|a\|^2-2\langle a,b\rangle+\|b\|^2$). After cancellation we obtain
\[
\begin{aligned}
\mathbb{E}[\Psi_{k+1}]
&\le \mathbb{E}[\Psi_k]
   -2\alpha\bigl(1-2\alpha L\bigr)\,\mathbb{E}\bigl[f(x_k)-f(x^\star)\bigr] . \tag{20}
\end{aligned}
\]
[Step 13]

\noindent\textbf{Step 14. Use the stepsize bound $\alpha\le 1/(3L)$.}
Then $1-2\alpha L\ge 1/3$, and (20) simplifies to
\[
\mathbb{E}[\Psi_{k+1}]
\le \mathbb{E}[\Psi_k] - \frac{2\alpha}{3}\,\mathbb{E}\bigl[f(x_k)-f(x^\star)\bigr]. \tag{21}
\]
[Step 14]

\noindent\textbf{Step 15. Relate function sub‑optimality to the Lyapunov function.}
Strong convexity (\ref{ass:strong}) yields
\[
f(x_k)-f(x^\star)\;\ge\;\frac{\mu}{2}\|x_k-x^\star\|^2 . \tag{22}
\]
Moreover, $\|\phi_i^k-x^\star\|^2\ge 0$, hence $\Psi_k\ge\|x_k-x^\star\|^2$. Combining (22) with (21) gives
\[
\mathbb{E}[\Psi_{k+1}]
\le \Bigl(1-\alpha\mu\Bigr)\mathbb{E}[\Psi_k]. \tag{23}
\]
[Step 15]

Equation (23) is exactly the contraction (4) claimed in Theorem~\ref{thm:conv}. Unrolling the recursion yields (5) for $\mu>0$.  

When $\mu=0$, we cannot use (22); instead we sum (21) over $k=0,\dots,T-1$,
divide by $T$, and use Jensen’s inequality to obtain the averaged bound (6). ∎

\section*{Rate Derivation (With Constants)}
From (23) we have for any $k\ge0$,
\[
\mathbb{E}[\Psi_k]\le (1-\alpha\mu)^k\Psi_0,
\]
with $\Psi_0=\|x_0-x^\star\|^2+\frac{2\alpha L}{n}\sum_{i}\|\phi_i^0-x^\star\|^2$.  
Using $f$‑smoothness,
\[
f(x_k)-f(x^\star)\le \frac{L}{2}\|x_k-x^\star\|^2\le\frac{L}{2}\Psi_k,
\]
which gives (5).  

For the convex case ($\mu=0$) let $\alpha=1/(3L)$. Summing (21) from $0$ to $T-1$ gives
\[
\frac{2\alpha}{3}\sum_{k=0}^{T-1}\mathbb{E}[f(x_k)-f(x^\star)]
\le \Psi_0,
\]
hence
\[
\mathbb{E}[f(\bar x_T)-f(x^\star)]\le\frac{3L}{2T}\Psi_0,
\]
which is (6).

\section*{Special Cases}
\begin{itemize}
\item \textbf{Quadratic finite‑sum.} If each $f_i(x)=\tfrac12 x^\top A_i x-b_i^\top x$, then $L=\max_i\lambda_{\max}(A_i)$ and the SAGA iterates coincide with a stochastic Kaczmarz method; the linear rate (5) becomes exact.
\item \textbf{Non‑uniform sampling.} Replacing the uniform draw by probabilities $p_i$ leads to the same proof after scaling the memory term by $1/p_i$; the stepsize condition becomes $\alpha\le \frac{1}{3L_{\max}}$, where $L_{\max}=\max_i L_i/p_i$.
\item \textbf{Mini‑batch version.} Sampling a batch $B_k$ of size $b$ replaces the factor $1/n$ in (1) by $1/b$; the contraction constant improves to $1-\alpha\mu b/n$.
\end{itemize}
