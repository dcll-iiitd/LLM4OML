\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

We consider a distributed learning setting where each worker computes a stochastic
gradient $g_t\in\RR^d$ of a smooth non‑convex loss $f:\RR^d\rightarrow\RR$ and
communicates only the $K$ components of largest magnitude.  The operator
$\TopK_K(\cdot)$ keeps these $K$ entries and sets the remaining $d-K$ entries
to zero.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{itemize}
  \item \textbf{Top‑K operator.} For any vector $v\in\RR^d$,
  \[
    \TopK_K(v)_i=
    \begin{cases}
      v_i, & \text{if } i\in\mathcal{I}_K(v),\\[2mm]
      0,   & \text{otherwise},
    \end{cases}
  \]
  where $\mathcal{I}_K(v)\subset\{1,\dots,d\}$ denotes the index set of the
  $K$ entries of $v$ with largest absolute values $|v_i|$.
  \item \textbf{Stochastic gradient.} At iteration $t$ a worker draws a random
  sample $z_t$ and computes an unbiased stochastic gradient
  \[
    g_t:=\nabla f(w_t;z_t),\qquad 
    \EE\!\left[g_t\mid w_t\right]=\nabla f(w_t).
  \]
  \item \textbf{Update rule.}
  \[
    w_{t+1}=w_t-\eta_t\,\TopK_K(g_t),
  \]
  where $\eta_t>0$ is the learning rate.  The algorithm reduces to ordinary
  SGD when $K=d$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\RR^d$, learning‑rate schedule
$\{\eta_t\}_{t\ge0}$, sparsity level $K\in\{1,\dots,d\}$.
\For{$t=0,1,\dots,T-1$}
    \State Sample data $z_t$ (or a minibatch) and compute
    $g_t\gets\nabla f(w_t;z_t)$.
    \State $\displaystyle \tilde g_t \gets \TopK_K(g_t)$
    \State $w_{t+1}\gets w_t-\eta_t\,\tilde g_t$
\EndFor
\State \textbf{Return} $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Dry Run Example}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We take $d=1$, $K=1$ (so the Top‑K operator is the identity), step size
$\eta=0.1$, and start from $w_0=0$.
Assume we have exact gradients (i.e. $g_t=\nabla f(w_t)$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta g_t$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}

Corresponding objective values:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.464)=2.149\, .
\]
The iterates move toward the minimiser $w^\star=3$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{enumerate}
  \item[\textbf{A1}] \textbf{Smoothness.} $f$ is $L$‑smooth:
  \[
    \forall u,v\in\RR^d,\qquad 
    f(u)\le f(v)+\langle\nabla f(v),u-v\rangle+\frac{L}{2}\|u-v\|^2 .
  \]
  \item[\textbf{A2}] \textbf{Bounded variance.} For all $w$,
  \[
    \EE\!\left[\|g-\nabla f(w)\|^2\mid w\right]\le\sigma^2 .
  \]
  \item[\textbf{A3}] \textbf{Top‑K contraction.} There exists a constant
  $\rho\in[0,1)$ such that for any $v\in\RR^d$,
  \[
    \EE\!\left[\big\|\TopK_K(v)-v\big\|^2\right]\le\rho\,\EE\!\left[\|v\|^2\right].
  \]
  Intuitively, keeping the $K$ largest coordinates discards at most a
  $\rho$‑fraction of the squared norm.
  \item[\textbf{A4}] \textbf{Learning‑rate schedule.}
  \[
    \eta_t=\eta=\frac{c}{\sqrt{T}},\qquad c>0\text{ constant}.
  \]
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{theorem}[Convergence of Top‑K SGD]\label{thm:main}
Under Assumptions A1–A4 and for any $T\ge 1$, the iterates generated by
Top‑K SGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^\star\big)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{L c}{\sqrt{T}}\big(1+\rho\big)\sigma^2}_{\text{stochastic \& sparsification term}} .
\]
Consequently,
\[
\min_{0\le t<T}\EE\!\big[\|\nabla f(w_t)\|^2\big]=O\!\big(T^{-1/2}\big).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{itemize}
  \item \textbf{Stability.} The bound depends on $\rho$, the loss induced by the
  Top‑K compression.  When $K$ is large, $\rho\approx0$ and the rate matches
  that of vanilla SGD.
  \item \textbf{Hyper‑parameter sensitivity.}
  \begin{itemize}
    \item Larger $c$ (larger step size) reduces the optimisation term but
    enlarges the stochastic term; the optimal $c$ balances the two.
    \item The sparsity level $K$ appears only through $\rho$; a modest $K$ that
    yields $\rho\ll1$ retains a near‑optimal rate while drastically reducing
    communication.
  \end{itemize}
  \item \textbf{Comparison to baselines.}
  \begin{itemize}
    \item $K=d$ $\Rightarrow$ $\rho=0$ and Theorem~\ref{thm:main} reduces to the
    classical $O(T^{-1/2})$ bound for non‑convex SGD.
    \item For any fixed $K<d$, the additional factor $(1+\rho)$ is the only
    penalty, confirming that sparse communication does not change the asymptotic
    order.
  \end{itemize}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The theorem shows that Top‑K SGD inherits the optimal
$O(T^{-1/2})$ convergence rate of stochastic gradient methods for smooth
non‑convex objectives, up to a multiplicative constant that quantifies the
information loss caused by keeping only $K$ coordinates.  The proof follows the
standard descent argument for smooth functions, with a careful treatment of
the compression error via Assumption~A3.  The result justifies the empirical
success of gradient sparsification in large‑scale deep learning.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^2 .
\]
\end{lemma}
\begin{proof}
Apply the definition of $L$‑smoothness with $u=w-u$ and $v=w$.
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

We prove Theorem~\ref{thm:main}.  Throughout we denote
$\tilde g_t:=\TopK_K(g_t)$ and $e_t:=\tilde g_t-g_t$ (the compression error).

\begin{enumerate}
\item[{\bf Step 1}] \textbf{Smoothness descent.}  
Using Lemma~\ref{lem:smooth} with $u=\eta\tilde g_t$,
\[
f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),\tilde g_t\rangle
+\frac{L\eta^2}{2}\|\tilde g_t\|^2 .\tag{1}
\]

\item[{\bf Step 2}] \textbf{Decompose $\tilde g_t$.}  
Write
\[
\tilde g_t = g_t + e_t .
\]
Insert into (1):
\[
f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),g_t\rangle
-\eta\langle\nabla f(w_t),e_t\rangle
+\frac{L\eta^2}{2}\|g_t+e_t\|^2 .\tag{2}
\]

\item[{\bf Step 3}] \textbf{Expand the squared norm.}
\[
\|g_t+e_t\|^2 = \|g_t\|^2+2\langle g_t,e_t\rangle+\|e_t\|^2 .\tag{3}
\]

\item[{\bf Step 4}] \textbf{Take conditional expectation.}  
Condition on $w_t$ and use $\EE[g_t\mid w_t]=\nabla f(w_t)$ (A2):
\[
\EE\!\big[f(w_{t+1})\mid w_t\big]\le
f(w_t)-\eta\|\nabla f(w_t)\|^2
-\eta\EE\!\big[\langle\nabla f(w_t),e_t\rangle\mid w_t\big]
+\frac{L\eta^2}{2}\EE\!\big[\|g_t\|^2+2\langle g_t,e_t\rangle+\|e_t\|^2\mid w_t\big].\tag{4}
\]

\item[{\bf Step 5}] \textbf{Bound the cross terms involving $e_t$.}  
Because $e_t$ is a deterministic function of $g_t$, we have
$\langle g_t,e_t\rangle=\|e_t\|^2$ (the dropped components are exactly those
added to $e_t$).  Hence
\[
\EE\!\big[\langle g_t,e_t\rangle\mid w_t\big]=\EE\!\big[\|e_t\|^2\mid w_t\big].
\]
Moreover, by Cauchy–Schwarz and Assumption~A3,
\[
\big|\EE\!\big[\langle\nabla f(w_t),e_t\rangle\mid w_t\big]\big|
\le\|\nabla f(w_t)\|\,
\EE\!\big[\|e_t\|\mid w_t\big]
\le\|\nabla f(w_t)\|\,
\sqrt{\EE\!\big[\|e_t\|^2\mid w_t\big]}
\le\|\nabla f(w_t)\|\sqrt{\rho\,\EE\!\big[\|g_t\|^2\mid w_t\big]} .
\tag{5}
\]

\item[{\bf Step 6}] \textbf{Bound $\EE[\|g_t\|^2]$.}  
From the variance bound (A2),
\[
\EE\!\big[\|g_t\|^2\mid w_t\big]
= \|\nabla f(w_t)\|^2 + \EE\!\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]
\le \|\nabla f(w_t)\|^2+\sigma^2 .\tag{6}
\]

\item[{\bf Step 7}] \textbf{Plug (5)–(6) into (4).}  
Using $\EE[\|e_t\|^2]\le\rho\,\EE[\|g_t\|^2]$ (A3) we obtain
\[
\begin{aligned}
\EE\!\big[f(w_{t+1})\mid w_t\big]
&\le f(w_t)-\eta\|\nabla f(w_t)\|^2
+ \eta\|\nabla f(w_t)\|\sqrt{\rho\,(\|\nabla f(w_t)\|^2+\sigma^2)}\\
&\quad+\frac{L\eta^2}{2}\Big[(\|\nabla f(w_t)\|^2+\sigma^2)
+2\rho(\|\nabla f(w_t)\|^2+\sigma^2)
+\rho(\|\nabla f(w_t)\|^2+\sigma^2)\Big]\\
&= f(w_t)-\eta\|\nabla f(w_t)\|^2
+ \eta\|\nabla f(w_t)\|\sqrt{\rho\,(\|\nabla f(w_t)\|^2+\sigma^2)}\\
&\quad+\frac{L\eta^2}{2}(1+3\rho)(\|\nabla f(w_t)\|^2+\sigma^2).
\end{aligned}\tag{7}
\]

\item[{\bf Step 8}] \textbf{Simplify the mixed term.}  
For any $a,b\ge0$ and $\theta\in(0,1)$,
$a\sqrt{b}\le \frac{\theta}{2}a^2+\frac{1}{2\theta}b$.
Choose $\theta=\frac{\eta}{2}$ (allowed because $\eta\le1$).  Applying this
inequality to the second term of (7) yields
\[
\eta\|\nabla f(w_t)\|\sqrt{\rho(\|\nabla f(w_t)\|^2+\sigma^2)}
\le \frac{\eta^2\rho}{4}\|\nabla f(w_t)\|^2
     +\rho\big(\|\nabla f(w_t)\|^2+\sigma^2\big).\tag{8}
\]

\item[{\bf Step 9}] \textbf{Collect coefficients of $\|\nabla f(w_t)\|^2$.}
Insert (8) into (7):
\[
\EE\!\big[f(w_{t+1})\mid w_t\big]
\le f(w_t)-\Big(\eta-\frac{\eta^2\rho}{4}
-\frac{L\eta^2}{2}(1+3\rho)\Big)\|\nabla f(w_t)\|^2
+\Big(\rho+\frac{L\eta^2}{2}(1+3\rho)\Big)\sigma^2 .\tag{9}
\]

\item[{\bf Step 10}] \textbf{Choose a stepsize small enough.}  
Assume $\eta\le \frac{1}{L(1+3\rho)}$; then the coefficient of $\|\nabla f(w_t)\|^2$
is at least $\frac{\eta}{2}$ (because the negative quadratic terms are bounded
by $\frac{\eta}{2}$).  Hence
\[
\EE\!\big[f(w_{t+1})\mid w_t\big]
\le f(w_t)-\frac{\eta}{2}\|\nabla f(w_t)\|^2
+ C_1\eta^2\sigma^2,\qquad
C_1:=\frac{L}{2}(1+3\rho)+\frac{\rho}{\eta}. \tag{10}
\]

\item[{\bf Step 11}] \textbf{Uncondition and sum.}  
Taking total expectation and summing $t=0,\dots,T-1$ gives
\[
\EE[f(w_T)]\le f(w_0)-\frac{\eta}{2}\sum_{t=0}^{T-1}
\EE\!\big[\|\nabla f(w_t)\|^2\big]
+ C_1\eta^2\sigma^2 T .\tag{11}
\]

Since $f(w_T)\ge f^\star$, rearrange (11):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_t)\|^2\big]
\le \frac{2\big(f(w_0)-f^\star\big)}{\eta T}
+2C_1\eta\sigma^2 .\tag{12}
\]

\item[{\bf Step 12}] \textbf{Insert the schedule $\eta=c/\sqrt{T}$.}  
With $\eta=c/\sqrt{T}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_t)\|^2\big]
\le
\underbrace{\frac{2\big(f(w_0)-f^\star\big)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{2C_1c\frac{\sigma^2}{\sqrt{T}}}_{\text{stochastic\,+\;compression term}} .
\]
Recall $C_1\le L(1+3\rho)/2$ for the chosen $\eta$, which yields the bound
stated in Theorem~\ref{thm:main}.  ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

From (12) with $\eta=c/\sqrt{T}$ we obtain the explicit constant form
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2\Delta_f}{c\sqrt{T}}
+\frac{Lc}{\sqrt{T}}(1+3\rho)\sigma^2,
\qquad
\Delta_f:=f(w_0)-f^\star .
\]
Optimising $c$ (e.g. $c=\sqrt{2\Delta_f/(L\sigma^2(1+3\rho))}$) gives the clean
asymptotic rate
\[
\min_{0\le t<T}\EE\!\big[\|\nabla f(w_t)\|^2\big]=O\!\big(T^{-1/2}\big).
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{itemize}
  \item \textbf{Full communication ($K=d$).}  Then $\TopK_K$ is the identity,
  $\rho=0$, and the theorem reduces to the classical SGD bound
  $\frac{2\Delta_f}{c\sqrt{T}}+\frac{Lc\sigma^2}{\sqrt{T}}$.
  \item \textbf{Extreme sparsity ($K=1$).}  In the worst case $\rho$ may be close
  to $1$, inflating the constant factor but leaving the $T^{-1/2}$ order
  unchanged.
  \item \textbf{Deterministic gradients.}  If $\sigma=0$, the bound becomes
  $\frac{2\Delta_f}{c\sqrt{T}}$, i.e. the same rate as gradient descent with a
  diminishing stepsize.
\end{itemize}
