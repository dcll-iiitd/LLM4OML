\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\begin{center}
\textbf{Heavy‑Ball (HB) Gradient Method}
\end{center}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  Starting from two initial points 
$x_{-1},x_{0}\in\R^{d}$, the Heavy‑Ball iteration is
\begin{equation}
\boxed{
x_{k+1}=x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha \nabla f(x_{k}),\qquad k\ge 0,
}
\label{eq:HB}
\end{equation}
where
\begin{itemize}
    \item $\alpha>0$ is the \emph{stepsize} (learning rate);
    \item $\beta\in[0,1)$ is the \emph{momentum parameter}.
\end{itemize}
The term $\beta(x_{k}-x_{k-1})$ adds a velocity component based on the previous
step, while the gradient term $-\alpha\nabla f(x_{k})$ pulls the iterate towards a
stationary point.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Gradient Method}
\begin{algorithmic}[1]
\Require Stepsize $\alpha>0$, momentum $\beta\in[0,1)$, 
initial points $x_{-1}=x_{0}\in\R^{d}$, maximum iterations $K$,
tolerance $\varepsilon>0$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$.
    \State Update 
    \[
    x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha g_{k}.
    \]
    \If{$\|g_{k}\|\le \varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},
\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, initialise $w_{-1}=w_{0}=0$ and run three
iterations of \eqref{eq:HB}.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iter.} $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}+\beta(w_{k}-w_{k-1})-\alpha\nabla f(w_{k})$ & $f(w_{k})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $0+0.5(0-0)-0.1(-6)=0.6$ & $9$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ & $5.76$\\
2 & $1.38$ & $2(1.38-3)=-3.24$ & $1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ & $3.284$\\
\end{tabular}
\end{center}
The objective values decrease from $9$ to $3.284$ after three steps,
illustrating the combined effect of momentum and gradient descent.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
$f$ is bounded below: $\displaystyle f_{\inf}:=\inf_{x\in\R^{d}}f(x)>-\infty$.
\end{assumption}
\begin{assumption}[Parameter Range]\label{ass:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha\le\frac{1-\beta}{L},\qquad 0\le\beta<1.
\tag{A2}
\]
\end{assumption}
All results below are derived under Assumptions~\ref{ass:smooth}–\ref{ass:param}.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of Heavy‑Ball]\label{thm:HB}
Let $\{x_{k}\}_{k\ge0}$ be generated by \eqref{eq:HB} under
Assumptions~\ref{ass:smooth}–\ref{ass:param}.
Define the Lyapunov sequence
\[
\Phi_{k}\;:=\;f(x_{k})+\frac{\beta}{2\alpha}\,\|x_{k}-x_{k-1}\|^{2},
\qquad k\ge0,
\]
with the convention $x_{-1}=x_{0}$.
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\big\|\nabla f(x_{k})\big\|^{2}
\;\le\;
\frac{2\big(f(x_{0})-f_{\inf}\big)}{\alpha(1-\beta)T}.
\tag{T1}
\]
Consequently,
\[
\min_{0\le k<T}\|\nabla f(x_{k})\|^{2}
\;=\;O\!\left(\frac{1}{T}\right).
\]
\end{theorem}
The bound matches the optimal $O(1/T)$ rate for first‑order methods on
smooth non‑convex functions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.}  The Lyapunov function $\Phi_{k}$ is non‑increasing:
    $\Phi_{k+1}\le\Phi_{k}$ for all $k$, which follows directly from the proof
    of Theorem~\ref{thm:HB}.  Hence the iterates remain in a bounded level set.
    \item \textbf{Hyper‑parameter sensitivity.}
    Condition \eqref{ass:param} forces $\alpha$ to shrink as $\beta$ grows.
    In the extreme case $\beta\to1^{-}$, the admissible stepsize tends to $0$,
    reflecting the well‑known instability of heavy‑ball with large momentum.
    \item \textbf{Comparison to Gradient Descent (GD).}
    Setting $\beta=0$ reduces \eqref{eq:HB} to vanilla GD.
    The bound \eqref{T1} becomes
    $\frac{2(f(x_{0})-f_{\inf})}{\alpha T}$, which is the classical $O(1/T)$
    rate.  For $\beta>0$ the same $O(1/T)$ rate holds, but the constant
    improves by the factor $1-\beta$ in the denominator, reflecting the
    acceleration effect in practice.
\end{itemize}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The Heavy‑Ball method enjoys a simple implementation and, under the
smoothness and stepsize restriction \eqref{ass:param}, guarantees that the
average squared gradient norm decays at the optimal $O(1/T)$ rate even
without convexity.  The proof relies on a carefully crafted Lyapunov function
that couples function values with the kinetic energy $\|x_{k}-x_{k-1}\|^{2}$.
Extensions to stochastic gradients, adaptive stepsizes, or
non‑Euclidean geometries follow the same template with additional variance
control lemmas.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of the Lipschitz gradient assumption; see
Assumption~\ref{ass:smooth}.  $\square$
\end{proof}
\begin{lemma}[Three‑point identity]\label{lem:3pt}
For any $a,b,c\in\R^{d}$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+ \|b-c\|^{2}+2\langle a-b,b-c\rangle .
\tag{L2}
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^{2}=\|(a-b)+(b-c)\|^{2}$ and use bilinearity of the inner product.
$\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:HB}.  All derivations are numbered for easy reference.

\medskip
\noindent\textbf{Step 1 (Lyapunov decrease).}  
Consider $\Phi_{k+1}-\Phi_{k}$:
\begin{align}
\Phi_{k+1}-\Phi_{k}
&= f(x_{k+1})-f(x_{k})+\frac{\beta}{2\alpha}
   \bigl(\|x_{k+1}-x_{k}\|^{2}-\|x_{k}-x_{k-1}\|^{2}\bigr). \label{eq:phi_diff}
\end{align}
\underline{[Step 1]}

\medskip
\noindent\textbf{Step 2 (Apply smoothness).}  
Using Lemma~\ref{lem:smooth} with $x=x_{k}$, $y=x_{k+1}$,
\begin{align}
f(x_{k+1})-f(x_{k})
&\le \langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \label{eq:smooth_sub}
\end{align}
\underline{[Step 2]}

\medskip
\noindent\textbf{Step 3 (Insert the update).}  
From \eqref{eq:HB},
\[
x_{k+1}-x_{k}= \beta(x_{k}-x_{k-1})-\alpha\nabla f(x_{k}).
\tag{U1}
\]
\underline{[Step 3]}

\medskip
\noindent\textbf{Step 4 (Expand the inner product).}
\begin{align}
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
&= \beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   -\alpha\|\nabla f(x_{k})\|^{2}. \label{eq:inner_expand}
\end{align}
\underline{[Step 4]}

\medskip
\noindent\textbf{Step 5 (Expand the squared norm).}
\begin{align}
\|x_{k+1}-x_{k}\|^{2}
&= \|\beta(x_{k}-x_{k-1})-\alpha\nabla f(x_{k})\|^{2}\nonumber\\
&= \beta^{2}\|x_{k}-x_{k-1}\|^{2}
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\beta\langle x_{k}-x_{k-1},\nabla f(x_{k})\rangle.
\tag{U2}
\end{align}
\underline{[Step 5]}

\medskip
\noindent\textbf{Step 6 (Plug \eqref{eq:inner_expand} and \eqref{U2} into \eqref{eq:smooth_sub}).}
\begin{align}
f(x_{k+1})-f(x_{k})
&\le \beta\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle
   -\alpha\|\nabla f(x_{k})\|^{2} \nonumber\\
&\quad+\frac{L}{2}\Bigl[
   \beta^{2}\|x_{k}-x_{k-1}\|^{2}
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\beta\langle x_{k}-x_{k-1},\nabla f(x_{k})\rangle
   \Bigr]. \label{eq:fsplit}
\end{align}
\underline{[Step 6]}

\medskip
\noindent\textbf{Step 7 (Combine with the kinetic term in \eqref{eq:phi_diff}).}
Using Lemma~\ref{lem:3pt} with $a=x_{k+1}$, $b=x_{k}$, $c=x_{k-1}$,
\[
\|x_{k+1}-x_{k}\|^{2}-\|x_{k}-x_{k-1}\|^{2}
 =\|x_{k+1}-x_{k-1}\|^{2}
   -2\langle x_{k+1}-x_{k},x_{k}-x_{k-1}\rangle .
\tag{U3}
\]
However we do not need the first term; we directly substitute
$\|x_{k+1}-x_{k}\|^{2}$ from \eqref{U2}:
\begin{align}
\frac{\beta}{2\alpha}
\bigl(\|x_{k+1}-x_{k}\|^{2}-\|x_{k}-x_{k-1}\|^{2}\bigr)
&= \frac{\beta}{2\alpha}\Bigl[
    \beta^{2}\|x_{k}-x_{k-1}\|^{2}
   +\alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\beta\langle x_{k}-x_{k-1},\nabla f(x_{k})\rangle \nonumber\\
&\qquad -\|x_{k}-x_{k-1}\|^{2}\Bigr] \nonumber\\
&= \frac{\beta(\beta^{2}-1)}{2\alpha}\|x_{k}-x_{k-1}\|^{2}
   +\frac{\beta\alpha}{2}\|\nabla f(x_{k})\|^{2}
   -\beta^{2}\langle x_{k}-x_{k-1},\nabla f(x_{k})\rangle .
\tag{U4}
\end{align}
\underline{[Step 7]}

\medskip
\noindent\textbf{Step 8 (Add \eqref{eq:fsplit} and \eqref{U4}).}
Summing the RHS of \eqref{eq:fsplit} and \eqref{U4} yields
\begin{align}
\Phi_{k+1}-\Phi_{k}
&\le
\underbrace{\Bigl[\beta-\beta^{2}L\Bigr]\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle}_{\text{(I)}}
\;+\;
\underbrace{\Bigl[-\alpha+\frac{L\alpha^{2}}{2}+\frac{\beta\alpha}{2}\Bigr]\|\nabla f(x_{k})\|^{2}}_{\text{(II)}}\nonumber\\
&\quad
\;+\;
\underbrace{\Bigl[\frac{L\beta^{2}}{2}+\frac{\beta(\beta^{2}-1)}{2\alpha}\Bigr]\|x_{k}-x_{k-1}\|^{2}}_{\text{(III)}} .
\label{eq:phi_intermediate}
\end{align}
\underline{[Step 8]}

\medskip
\noindent\textbf{Step 9 (Eliminate the mixed term (I)).}  
Observe that term (I) can be bounded using Cauchy–Schwarz and the inequality
$2ab\le \eta a^{2}+b^{2}/\eta$ with a convenient $\eta>0$.
Choose $\eta=\frac{L\alpha}{\beta}$ (which is admissible because $\beta>0$).
Then
\begin{align}
\bigl|\beta-\beta^{2}L\bigr|\,
|\langle\nabla f(x_{k}),x_{k}-x_{k-1}\rangle|
&\le
\bigl|\beta-\beta^{2}L\bigr|
\Bigl(\frac{L\alpha}{2\beta}\|x_{k}-x_{k-1}\|^{2}
      +\frac{1}{2L\alpha}\|\nabla f(x_{k})\|^{2}\Bigr). \label{eq:mix_bound}
\end{align}
Since $\beta\le 1$ and $L\alpha\le 1-\beta$ by \eqref{ass:param},
one can verify that $\bigl|\beta-\beta^{2}L\bigr|\le\beta$.
Hence
\[
\text{(I)}\le
\frac{\beta L\alpha}{2}\|x_{k}-x_{k-1}\|^{2}
+\frac{\beta}{2L\alpha}\|\nabla f(x_{k})\|^{2}.
\tag{B1}
\]
\underline{[Step 9]}

\medskip
\noindent\textbf{Step 10 (Collect coefficients).}
Insert bound (B1) into \eqref{eq:phi_intermediate}.  The coefficient of
$\|\nabla f(x_{k})\|^{2}$ becomes
\[
c_{g}:=
-\alpha+\frac{L\alpha^{2}}{2}+\frac{\beta\alpha}{2}
+\frac{\beta}{2L\alpha}.
\]
Using $L\alpha\le 1-\beta$ we have
\[
c_{g}\le -\frac{\alpha(1-\beta)}{2}.
\tag{B2}
\]
Similarly, the coefficient of $\|x_{k}-x_{k-1}\|^{2}$ is
\[
c_{v}:=
\frac{L\beta^{2}}{2}
+\frac{\beta(\beta^{2}-1)}{2\alpha}
+\frac{\beta L\alpha}{2}
\;=\;
-\frac{\beta(1-\beta)}{2\alpha}\;+\;\frac{L\beta^{2}}{2}
\;\le\;0,
\tag{B3}
\]
where the last inequality follows again from $L\alpha\le 1-\beta$.
Therefore,
\begin{align}
\Phi_{k+1}-\Phi_{k}
\le -\frac{\alpha(1-\beta)}{2}\,\|\nabla f(x_{k})\|^{2}.
\tag{D1}
\end{align}
\underline{[Step 10]}

\medskip
\noindent\textbf{Step 11 (Telescoping sum).}
Summing \eqref{D1} from $k=0$ to $T-1$ gives
\[
\Phi_{T}-\Phi_{0}
\le -\frac{\alpha(1-\beta)}{2}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}.
\]
Since $\Phi_{T}\ge f_{\inf}$ and $\Phi_{0}=f(x_{0})$ (because $x_{-1}=x_{0}$),
\[
\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\le \frac{2\big(f(x_{0})-f_{\inf}\big)}{\alpha(1-\beta)}.
\tag{D2}
\]
Dividing by $T$ yields exactly the bound \eqref{T1} of Theorem~\ref{thm:HB}.
\underline{[Step 11]}

\medskip
\noindent\textbf{Step 12 (Deriving the min‑gradient bound).}
From \eqref{T1},
\[
\min_{0\le k<T}\|\nabla f(x_{k})\|^{2}
\;\le\;
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\;=\;O\!\left(\frac{1}{T}\right).
\]
\underline{[Step 12]}

Thus every step of the proof has been justified, completing the convergence analysis.

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
The explicit constant appearing in \eqref{T1} is
\[
C(\alpha,\beta):=
\frac{2}{\alpha(1-\beta)}\bigl(f(x_{0})-f_{\inf}\bigr).
\]
Choosing the largest admissible stepsize
$\alpha^{\star}=\frac{1-\beta}{L}$ (the boundary of \eqref{ass:param})
optimises the constant:
\[
C(\alpha^{\star},\beta)=\frac{2L}{(1-\beta)^{2}}\bigl(f(x_{0})-f_{\inf}\bigr).
\]
Hence, for the “optimal” Heavy‑Ball parameters,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|\nabla f(x_{k})\|^{2}
\le
\frac{2L}{(1-\beta)^{2}}\frac{f(x_{0})-f_{\inf}}{T}.
\]

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Gradient Descent ($\beta=0$).}  The Lyapunov function reduces to
$\Phi_{k}=f(x_{k})$ and \eqref{D1} becomes
$f(x_{k+1})\le f(x_{k})-\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}$,
recovering the classic $O(1/T)$ bound.

\item \textbf{Vanishing momentum ($\beta\downarrow0$).}  The constant
$C(\alpha,\beta)$ approaches $\frac{2}{\alpha}\bigl(f(x_{0})-f_{\inf}\bigr)$,
the same rate as GD with stepsize $\alpha$.

\item \textbf{Heavy‑ball with maximal admissible stepsize.}
Setting $\alpha=\frac{1-\beta}{L}$ yields the smallest possible denominator
$\alpha(1-\beta)=\frac{(1-\beta)^{2}}{L}$, i.e. the fastest theoretical
convergence among all feasible $(\alpha,\beta)$ pairs.
\end{enumerate}

\section*{Conclusion}
\addcontentsline{toc}{section}{Conclusion}
We have presented a self‑contained exposition of the Heavy‑Ball method for
smooth non‑convex optimization, including a concrete algorithmic description,
a numerical dry run, and a rigorous convergence proof that achieves the optimal
$O(1/T)$ rate for the average squared gradient norm.  The analysis hinges on a
Lyapunov function that captures both objective decrease and kinetic energy,
and the derived constants explicitly reveal the trade‑off between stepsize
and momentum.
