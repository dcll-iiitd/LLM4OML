\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let  
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d},
\]
where  

\begin{itemize}[noitemsep]
\item $f:\R^{d}\!\to\!\R$ is convex and differentiable with $L$‑Lipschitz continuous gradient, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d};
\]
\item $g:\R^{d}\!\to\!\R\cup\{+\infty\}$ is convex, proper and lower–semicontinuous (possibly non‑smooth).
\end{itemize}
For a stepsize $\eta>0$, the \emph{proximal operator} of $\eta g$ is
\[
\prox_{\eta g}(u)\;:=\;\arg\min_{z\in\R^{d}}\Big\{ g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Big\}.
\]

The Proximal Gradient iteration reads
\begin{equation}
\boxed{\;
x_{k+1}\;=\;\prox_{\eta g}\bigl(x_{k}-\eta\,\grad f(x_{k})\bigr),\qquad k=0,1,2,\dots
\;}
\label{eq:pgm}
\end{equation}
The stepsize is chosen such that $0<\eta\le 1/L$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Convex $f$, convex $g$, stepsize $\eta\in(0,1/L]$, initial point $x_{0}\in\R^{d}$, max iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $u_{k}\gets x_{k}-\eta\,\grad f(x_{k})$ \hfill\Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(u_{k})$ \hfill\Comment{proximal mapping}
\EndFor
\State \Return $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  
\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so that $\prox_{\eta g}= \text{id}$.
\[
\grad f(w)=2(w-3),\qquad L=2.
\]
Choose $\eta=0.1\le 1/L=0.5$, $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\grad f(w_{k})$ & $w_{k+1}=w_{k}-\eta\grad f(w_{k})$\\\hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Objective values $F(w_{k})=(w_{k}-3)^{2}$ decrease:
$F(w_{0})=9$, $F(w_{1})=5.76$, $F(w_{2})\approx3.54$.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*:},noitemsep]
\item \textbf{Smoothness of $f$.} $f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\quad\forall x,y\in\R^{d}.
\tag{A1}
\]
\item \textbf{Convexity of $g$.} $g$ is convex, proper and lower‑semicontinuous.
\tag{A2}
\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\eta\le \frac{1}{L}.
\tag{A3}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:pgm}
Let $\{x_{k}\}$ be generated by \eqref{eq:pgm} under Assumptions A1–A3 and let $x^{\star}\in\arg\min_{x}F(x)$. Then for every $K\ge1$,
\[
F(x_{K})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,K}.
\tag{T1}
\]
Consequently $F(x_{K})\to F(x^{\star})$ at the rate $O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
\item \textbf{Monotone descent.} The sequence $\{F(x_{k})\}$ is non‑increasing.
\item \textbf{Stability w.r.t.\ stepsize.} Any $\eta\in(0,1/L]$ guarantees the descent property; larger $\eta$ (up to $1/L$) yields a tighter bound in \eqref{T1}.
\item \textbf{Comparison.} When $g\equiv0$, the method reduces to gradient descent and \eqref{T1} coincides with the classical $O(1/K)$ rate. Adding a non‑smooth convex $g$ incurs no penalty in the rate thanks to the non‑expansiveness of $\prox$.
\end{itemize}

\section*{Discussion}
The theorem is deterministic; no stochasticity is involved. The proof hinges on two elementary tools:
\begin{enumerate}[label={\arabic*.},noitemsep]
\item The \emph{descent lemma} for $L$‑smooth functions.
\item The \emph{non‑expansiveness} of the proximal operator:
\[
\big\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\big\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{P}
\]
These are combined with the optimality condition of the proximal step to obtain a telescoping inequality that yields the $O(1/K)$ bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Gradient mapping]
For a stepsize $\eta>0$, define the gradient mapping at $x$ as
\[
G_{\eta}(x)\;:=\;\frac{1}{\eta}\Bigl(x-\prox_{\eta g}\bigl(x-\eta\,\grad f(x)\bigr)\Bigr).
\tag{D1}
\]
When $g\equiv0$, $G_{\eta}(x)=\grad f(x)$.
\end{definition}

\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ has $L$‑Lipschitz gradient, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard; follows from integration of the Lipschitz condition. ∎
\end{proof}

\begin{lemma}[Non‑expansiveness of $\prox$]\label{lem:nonexp}
For any $u,v\in\R^{d}$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
\tag{L2}
\]
\end{proof}
\begin{proof}
Proximal operator is the resolvent of the sub‑differential of $g$, which is firmly non‑expansive; see e.g. \cite{rockafellar}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:pgm}.  Throughout, let $x^{\star}\in\arg\min F$.

\noindent\textbf{Step 1. Optimality condition of the proximal step.}\\
From the definition of $\prox_{\eta g}$,
\[
x_{k+1}= \prox_{\eta g}(x_{k}-\eta\grad f(x_{k}))
\;\Longleftrightarrow\;
\frac{1}{\eta}\bigl(x_{k}-\eta\grad f(x_{k})-x_{k+1}\bigr)\in\partial g(x_{k+1}).
\tag{Step 1}
\]

\noindent\textbf{Step 2. Sub‑gradient inequality for $g$.}\\
Since $g$ is convex,
\[
g(x^{\star})\ge g(x_{k+1})+\big\langle s_{k+1},\,x^{\star}-x_{k+1}\big\rangle,
\quad\text{with }s_{k+1}\in\partial g(x_{k+1}).
\tag{Step 2}
\]

\noindent\textbf{Step 3. Substitute $s_{k+1}$ from Step 1.}\\
Using $s_{k+1}= \frac{1}{\eta}\bigl(x_{k}-\eta\grad f(x_{k})-x_{k+1}\bigr)$,
\[
g(x^{\star})\ge g(x_{k+1})+\frac{1}{\eta}\big\langle x_{k}-\eta\grad f(x_{k})-x_{k+1},\,x^{\star}-x_{k+1}\big\rangle.
\tag{Step 3}
\]

\noindent\textbf{Step 4. Rearrange the inner product.}\\
Expand the inner product:
\[
\big\langle x_{k}-x_{k+1},\,x^{\star}-x_{k+1}\big\rangle
= \frac{1}{2}\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{Step 4}
\]

\noindent\textbf{Step 5. Combine Step 3 and Step 4.}\\
\[
g(x^{\star})\ge g(x_{k+1})+\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr)
-\big\langle\grad f(x_{k}),\,x^{\star}-x_{k+1}\big\rangle.
\tag{Step 5}
\]

\noindent\textbf{Step 6. Apply convexity of $f$.}\\
Convexity yields
\[
f(x^{\star})\ge f(x_{k})+\big\langle\grad f(x_{k}),\,x^{\star}-x_{k}\big\rangle.
\tag{Step 6}
\]

\noindent\textbf{Step 7. Add inequalities from Step 5 and Step 6.}\\
\[
\begin{aligned}
F(x^{\star})&=f(x^{\star})+g(x^{\star})\\
&\ge f(x_{k})+g(x_{k+1})
+\big\langle\grad f(x_{k}),\,x^{\star}-x_{k}\big\rangle
-\big\langle\grad f(x_{k}),\,x^{\star}-x_{k+1}\big\rangle\\
&\quad+\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\end{aligned}
\tag{Step 7}
\]

\noindent\textbf{Step 8. Simplify the gradient inner products.}\\
The two inner products combine to
\[
\big\langle\grad f(x_{k}),\,x^{\star}-x_{k}\big\rangle
-\big\langle\grad f(x_{k}),\,x^{\star}-x_{k+1}\big\rangle
= \big\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\big\rangle.
\tag{Step 8}
\]

\noindent\textbf{Step 9. Use the descent lemma (Lemma~\ref{lem:descent}) with $y=x_{k+1}$.}\\
Since $\eta\le 1/L$, we have $L\eta\le1$ and
\[
f(x_{k+1})\le f(x_{k})+\big\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\big\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
\le f(x_{k})+\big\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\big\rangle
+\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{Step 9}
\]

\noindent\textbf{Step 10. Substitute the bound from Step 9 into Step 7.}\\
Replace $f(x_{k})+\big\langle\grad f(x_{k}),\,x_{k+1}-x_{k}\big\rangle$ by $f(x_{k+1})-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}$:
\[
\begin{aligned}
F(x^{\star}) &\ge f(x_{k+1})+g(x_{k+1})
+\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr)\\
&\quad-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{Step 10}
\]

\noindent\textbf{Step 11. Cancel the duplicated terms.}\\
The two $-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}$ terms combine:
\[
F(x^{\star})\ge F(x_{k+1})+\frac{1}{2\eta}
\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
-2\|x_{k+1}-x_{k}\|^{2}\Bigr).
\tag{Step 11}
\]

\noindent\textbf{Step 12. Rearrange to obtain a one‑step descent inequality.}\\
\[
F(x_{k+1})-F(x^{\star})
\le \frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}\Bigr)
-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{Step 12}
\]

\noindent\textbf{Step 13. Drop the non‑negative term $-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}$.}\\
Thus we obtain the fundamental recursion
\[
F(x_{k+1})-F(x^{\star})
\le \frac{1}{2\eta}\Bigl(\|x_{k}-x^{\star}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\tag{Step 13}
\]

\noindent\textbf{Step 14. Telescope the inequality from $k=0$ to $k=K-1$.}\\
Summing \eqref{Step 13} yields
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \frac{1}{2\eta}\Bigl(\|x_{0}-x^{\star}\|^{2}
-\|x_{K}-x^{\star}\|^{2}\Bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\tag{Step 14}
\]

\noindent\textbf{Step 15. Use the monotonicity of $F(x_{k})$.}\\
Since $F(x_{k})$ is non‑increasing (from Step 12), each term in the left‑hand sum is at least $F(x_{K})-F(x^{\star})$. Hence,
\[
K\bigl(F(x_{K})-F(x^{\star})\bigr)
\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}.
\tag{Step 15}
\]

\noindent\textbf{Step 16. Isolate $F(x_{K})-F(x^{\star})$.}\\
Dividing by $K$ gives exactly the bound claimed in Theorem~\ref{thm:pgm}:
\[
F(x_{K})-F(x^{\star})\le \frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,K}.
\tag{Step 16}
\]
\qed

\section*{Rate Derivation (With Constants)}
The constant in the $O(1/K)$ rate is explicitly $\displaystyle C=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta}$.  
If the stepsize is chosen as the maximal admissible value $\eta=1/L$, the bound becomes
\[
F(x_{K})-F(x^{\star})\le \frac{L\,\|x_{0}-x^{\star}\|^{2}}{2K}.
\]

\section*{Special Cases}
\begin{itemize}[noitemsep]
\item \textbf{Pure gradient descent.} When $g\equiv0$, $\prox_{\eta g}=\text{id}$ and the gradient mapping $G_{\eta}(x)=\grad f(x)$. The theorem reduces to the classical $O(1/K)$ rate for smooth convex minimization.
\item \textbf{Indicator of a convex set $C$.} If $g=\iota_{C}$, then $\prox_{\eta g}$ is the Euclidean projection onto $C$. The same bound holds for projected gradient descent.
\item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, a standard modification of Step 13 yields a linear rate:
\[
F(x_{k})-F(x^{\star})\le \bigl(1-\eta\mu\bigr)^{k}\,\bigl(F(x_{0})-F(x^{\star})\bigr).
\]
\end{itemize}
