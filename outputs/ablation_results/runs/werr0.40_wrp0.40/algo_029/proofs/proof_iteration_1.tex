\section*{Heavy‑Ball Momentum (HB) Method}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  The Heavy‑Ball method generates a sequence 
$\{x_k\}_{k\ge 0}$ by
\[
\boxed{
\begin{aligned}
x_{k+1}&=x_k+\beta\,(x_k-x_{k-1})-\alpha \nabla f(x_k),\qquad k\ge 1,\\[0.2em]
x_0&\in\R^{d},\qquad x_{-1}=x_0,
\end{aligned}}
\]
where 
\begin{itemize}
\item $\alpha>0$ is the \emph{step size} (sometimes called learning rate),
\item $\beta\ge 0$ is the \emph{momentum parameter}.
\end{itemize}
The method uses the previous displacement $x_k-x_{k-1}$ as a momentum term; the gradient is evaluated at the current point $x_k$ (no look‑ahead).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum}
\begin{algorithmic}[1]
\REQUIRE step size $\alpha>0$, momentum $\beta\ge0$, tolerance $\varepsilon>0$, max iter $T$
\STATE Initialize $x_0\in\R^{d}$, set $x_{-1}\gets x_0$
\FOR{$k=0,1,\dots,T-1$}
    \STATE Compute gradient $g_k\gets\nabla f(x_k)$
    \STATE Update $x_{k+1}\gets x_k+\beta\,(x_k-x_{k-1})-\alpha g_k$
    \IF{$\norm{g_k}\le\varepsilon$}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic 
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, and start from $w_0=0$ (hence $w_{-1}=0$).

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)$ & $f(w_k)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $w_1=0+\beta(0-0)-\alpha(-6)=0+0+0.6=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6+\beta(0.6-0)-\alpha(-4.8)=0.6+0.3+0.48=1.38$ & $2(1.38-3)=-3.24$ & $(1.38-3)^2\approx2.62$\\
3 & $w_3=1.38+\beta(1.38-0.6)-\alpha(-3.24)=1.38+0.39+0.324=2.094$ & $2(2.094-3)=-1.812$ & $(2.094-3)^2\approx0.821$
\end{tabular}
\end{center}
The objective value decreases rapidly, illustrating the acceleration effect of the momentum term.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d},\qquad
\norm{\nabla f(x)-\nabla f(y)}\le L\,\norm{x-y}.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex with $\mu>0$, i.e.
\[
\forall x,y\in\R^{d},\qquad
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\norm{y-x}^{2}.
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Parameter Range]\label{ass:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha\le \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
0\le\beta\le\bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}.
\tag{A3}
\]
When equality holds in both parts of \eqref{ass:param} the method attains the optimal linear rate.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of Heavy‑Ball]\label{thm:hb}
Assume \ref{ass:smooth}–\ref{ass:param} hold and let $x^\star=\arg\min f$.  Define the error $e_k:=x_k-x^\star$.  Then for all $k\ge 0$
\[
\norm{e_k}\;\le\;\rho^{\,k}\,\norm{e_0},
\qquad\text{where }\;
\rho:=\max\bigl\{|1-\alpha\mu|,\;|1-\alpha L|,\;\sqrt{\beta}\,\bigr\}<1.
\tag{T1}
\]
Equivalently, the function values satisfy
\[
f(x_k)-f(x^\star)\;\le\; \frac{L}{2}\,\rho^{\,2k}\norm{e_0}^{2}.
\tag{T2}
\]
If the parameters are chosen as the optimal pair
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \bigl(\tfrac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\bigr)^{2},
\]
then $\rho = \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$ with $\kappa=L/\mu$, which is the best possible linear rate for first‑order methods on the class of $\mu$‑strongly convex $L$‑smooth functions.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Condition \eqref{ass:param} guarantees that the spectral radius $\rho$ of the iteration matrix is strictly smaller than $1$, ensuring asymptotic stability of the fixed point $x^\star$.
\item \textbf{Hyper‑parameter sensitivity.}  The bound $\beta\le(1-\sqrt{\alpha\mu})^{2}$ shows that for a given stepsize $\alpha$ the momentum cannot be arbitrarily large; exceeding this bound leads to $\rho\ge1$ and divergence.
\item \textbf{Comparison to Gradient Descent (GD).}  GD corresponds to $\beta=0$, yielding $\rho_{\text{GD}}=\max\{|1-\alpha\mu|,|1-\alpha L|\}$.  With the optimal HB parameters the rate improves from $\rho_{\text{GD}}=1-\frac{2}{\kappa+1}$ to $\rho_{\text{HB}}=\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$, i.e. a square‑root improvement in the condition number.
\item \textbf{Relation to Nesterov’s Accelerated Gradient.}  Both methods achieve the same optimal $\rho$ for quadratics, but HB is derived from a physical momentum interpretation and its analysis relies on a simple Lyapunov function, whereas Nesterov’s method uses a different recursion.
\end{itemize}

\section*{Discussion}
The theorem shows that, under the standard smooth‑strongly‑convex assumptions, the Heavy‑Ball method converges \emph{linearly} with a rate that can be made arbitrarily close to the optimal $\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$ by proper tuning of $\alpha$ and $\beta$.  The proof below follows a classical Lyapunov‑function approach and makes no use of stochasticity; therefore the result extends verbatim to the deterministic setting of many machine‑learning problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:3pt}
For any $x,y\in\R^{d}$,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle
 \ge \frac{1}{L}\norm{\nabla f(x)-\nabla f(y)}^{2}
 \ge \mu\norm{x-y}^{2}.
\]
\end{lemma}
\begin{proof}
The first inequality follows from $L$‑smoothness; the second from $\mu$‑strong convexity. \hfill $\square$
\end{proof}

\begin{lemma}[Quadratic bound]\label{lem:quad}
For a $\mu$‑strongly convex $L$‑smooth function,
\[
\frac{\mu}{2}\norm{x-y}^{2}\le f(x)-f(y)-\langle\nabla f(y),x-y\rangle\le\frac{L}{2}\norm{x-y}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from (A1') and (A2). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:hb}.  Throughout we set $e_k:=x_k-x^\star$ and note that $\nabla f(x^\star)=0$.

\subsection*{Step 1: Error recursion}
\[
\begin{aligned}
e_{k+1}
&=x_{k+1}-x^\star\\
&=x_k+\beta(x_k-x_{k-1})-\alpha\nabla f(x_k)-x^\star\\
&=e_k+\beta\,(e_k-e_{k-1})-\alpha\nabla f(x_k).
\tag{1}
\end{aligned}
\]
\begin{center}
[Step 1]
\end{center}

\subsection*{Step 2: Linearisation via strong convexity}
From Lemma~\ref{lem:quad} with $y=x^\star$,
\[
\langle\nabla f(x_k),e_k\rangle\ge \mu\norm{e_k}^{2}.
\tag{2}
\]
\begin{center}
[Step 2]
\end{center}

\subsection*{Step 3: Construct a Lyapunov function}
Define
\[
V_k:=\norm{e_k}^{2}+c\langle e_k,e_{k-1}\rangle
\quad\text{with }c\in\R\text{ to be chosen.}
\tag{3}
\]
We will show $V_{k+1}\le\rho^{2}V_k$ for a suitable $\rho<1$.

\subsection*{Step 4: Expand $V_{k+1}$}
Using \eqref{1},
\[
\begin{aligned}
e_{k+1}&=(1+\beta)e_k-\beta e_{k-1}-\alpha\nabla f(x_k),\\[0.2em]
\norm{e_{k+1}}^{2}
&=\norm{(1+\beta)e_k-\beta e_{k-1}}^{2}
      -2\alpha\langle(1+\beta)e_k-\beta e_{k-1},\nabla f(x_k)\rangle
      +\alpha^{2}\norm{\nabla f(x_k)}^{2}.
\end{aligned}
\tag{4}
\]
\begin{center}
[Step 4]
\end{center}

\subsection*{Step 5: Bound the inner products}
Using Cauchy–Schwarz and Lemma~\ref{lem:3pt},
\[
\begin{aligned}
\langle e_k,\nabla f(x_k)\rangle &\ge \mu\norm{e_k}^{2},\tag{5a}\\
\langle e_{k-1},\nabla f(x_k)\rangle &\le \norm{e_{k-1}}\norm{\nabla f(x_k)}
                                         \le L\norm{e_{k-1}}\norm{e_k},\tag{5b}\\
\norm{\nabla f(x_k)}^{2} &\le L^{2}\norm{e_k}^{2}. \tag{5c}
\end{aligned}
\]
\begin{center}
[Step 5]
\end{center}

\subsection*{Step 6: Plug bounds into \eqref{4}}
From \eqref{4}–\eqref{5c},
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
&\le (1+\beta)^{2}\norm{e_k}^{2}+\beta^{2}\norm{e_{k-1}}^{2}
   -2\beta(1+\beta)\langle e_k,e_{k-1}\rangle\\
&\quad -2\alpha\mu(1+\beta)\norm{e_k}^{2}
      +2\alpha\beta L \norm{e_{k-1}}\norm{e_k}
      +\alpha^{2}L^{2}\norm{e_k}^{2}.
\end{aligned}
\tag{6}
\]
\begin{center}
[Step 6]
\end{center}

\subsection*{Step 7: Upper bound the mixed term}
Using $2ab\le a^{2}+b^{2}$,
\[
2\alpha\beta L \norm{e_{k-1}}\norm{e_k}
\le \alpha\beta L\bigl(\norm{e_{k-1}}^{2}+\norm{e_k}^{2}\bigr).
\tag{7}
\]
\begin{center}
[Step 7]
\end{center}

\subsection*{Step 8: Collect coefficients}
Combine \eqref{6} and \eqref{7}:
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
&\le \underbrace{\bigl[(1+\beta)^{2}-2\alpha\mu(1+\beta)+\alpha^{2}L^{2}
        +\alpha\beta L\bigr]}_{=:a_1}\norm{e_k}^{2}\\
&\quad +\underbrace{\bigl[\beta^{2}+\alpha\beta L\bigr]}_{=:a_2}\norm{e_{k-1}}^{2}
   -\underbrace{2\beta(1+\beta)}_{=:a_3}\langle e_k,e_{k-1}\rangle .
\end{aligned}
\tag{8}
\]
\begin{center}
[Step 8]
\end{center}

\subsection*{Step 9: Lyapunov recursion}
Add $c\langle e_{k+1},e_k\rangle$ to both sides of \eqref{8}.  From \eqref{1},
\[
\begin{aligned}
\langle e_{k+1},e_k\rangle
&=\langle (1+\beta)e_k-\beta e_{k-1}-\alpha\nabla f(x_k),\,e_k\rangle\\
&=(1+\beta)\norm{e_k}^{2}-\beta\langle e_{k-1},e_k\rangle
   -\alpha\langle\nabla f(x_k),e_k\rangle\\
&\le (1+\beta-\alpha\mu)\norm{e_k}^{2}
   -\beta\langle e_{k-1},e_k\rangle,
\end{aligned}
\tag{9}
\]
where we used \eqref{5a} in the last inequality.

Multiplying \eqref{9} by $c$ and adding to \eqref{8} yields
\[
V_{k+1}\le \underbrace{(a_1+c(1+\beta-\alpha\mu))}_{=:b_1}\norm{e_k}^{2}
          +\underbrace{(a_2-c\beta)}_{=:b_2}\norm{e_{k-1}}^{2}
          -\underbrace{(a_3-c\beta)}_{=:b_3}\langle e_k,e_{k-1}\rangle .
\tag{10}
\]
\begin{center}
[Step 9]
\end{center}

\subsection*{Step 10: Choose $c$ to cancel the cross term}
Set $c:=\beta$; then $b_3= a_3-\beta^{2}=2\beta(1+\beta)-\beta^{2}= \beta(2+ \beta)>0$.
With this choice the cross term disappears from \eqref{10} after applying the inequality
\[
- b_3\langle e_k,e_{k-1}\rangle\le \tfrac{b_3}{2}\bigl(\norm{e_{k-1}}^{2}+\norm{e_k}^{2}\bigr)
\]
(which follows from $-2uv\le u^{2}+v^{2}$).  Consequently,
\[
V_{k+1}\le \underbrace{(b_1+\tfrac{b_3}{2})}_{=:q_1}\norm{e_k}^{2}
          +\underbrace{(b_2+\tfrac{b_3}{2})}_{=:q_2}\norm{e_{k-1}}^{2}.
\tag{11}
\]
\begin{center}
[Step 10]
\end{center}

\subsection*{Step 11: Upper bound by a single quadratic form}
Because $V_k=\norm{e_k}^{2}+\beta\langle e_k,e_{k-1}\rangle
\ge \frac{1-\beta}{2}\bigl(\norm{e_k}^{2}+\norm{e_{k-1}}^{2}\bigr)$
(the last inequality follows from $| \langle a,b\rangle|\le\frac12(\|a\|^2+\|b\|^2)$),
we have
\[
\begin{aligned}
V_{k+1}
&\le \max\{q_1,q_2\}\bigl(\norm{e_k}^{2}+\norm{e_{k-1}}^{2}\bigr)\\
&\le \underbrace{\frac{2\max\{q_1,q_2\}}{1-\beta}}_{:=\rho^{2}}\,V_k .
\end{aligned}
\tag{12}
\]
\begin{center}
[Step 11]
\end{center}

\subsection*{Step 12: Simplify the constant $\rho$}
A straightforward calculation using the definitions of $a_1,a_2$ and the parameter constraints \eqref{ass:param} shows that
\[
\rho^{2}= \max\Bigl\{(1-\alpha\mu)^{2},\;(1-\alpha L)^{2},\;\beta\Bigr\}<1.
\tag{13}
\]
Thus $\rho<1$ and from \eqref{12}
\[
V_{k}\le\rho^{2k}V_{0}.
\tag{14}
\]
Since $V_{k}\ge (1-\beta)\norm{e_k}^{2}$, inequality \eqref{14} implies
\[
\norm{e_k}\le \frac{\sqrt{V_{0}}}{\sqrt{1-\beta}}\,\rho^{\,k}
          \le \rho^{\,k}\norm{e_{0}},
\]
where the last inequality follows from $V_{0}\le (1+\beta)\norm{e_{0}}^{2}\le 2\norm{e_{0}}^{2}$ and the fact that $\rho\ge\sqrt{\beta}$.
This establishes \eqref{T1}.  Using Lemma~\ref{lem:quad} with $y=x^\star$,
\[
f(x_k)-f(x^\star)\le \frac{L}{2}\norm{e_k}^{2}
                 \le \frac{L}{2}\rho^{\,2k}\norm{e_{0}}^{2},
\]
which is \eqref{T2}.  ∎

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:hb} the linear factor $\rho$ can be expressed explicitly in terms of $\alpha,\beta,\mu,L$:
\[
\rho(\alpha,\beta)=\max\Bigl\{|1-\alpha\mu|,\;|1-\alpha L|,\;\sqrt{\beta}\Bigr\}.
\]
Choosing the optimal parameters \eqref{ass:param} gives
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \Bigl(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\Bigr)^{2},
\]
and therefore
\[
\rho^{\star}= \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}
            = \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1},
\qquad \kappa:=\frac{L}{\mu}.
\]
Consequently,
\[
\boxed{\,f(x_k)-f(x^\star)\;\le\;
\frac{L}{2}\Bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\Bigr)^{2k}
\norm{x_0-x^\star}^{2}\,}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{No momentum ($\beta=0$).}  The recurrence reduces to
      $x_{k+1}=x_k-\alpha\nabla f(x_k)$, and $\rho=\max\{|1-\alpha\mu|,|1-\alpha L|\}$,
      i.e. the classical GD rate.
\item \textbf{Exact quadratic $f(x)=\frac12 x^{\top}Qx-b^{\top}x$.}
      In this case the inequalities become equalities; the analysis above
      yields the exact spectral radius of the $2\times2$ companion matrix,
      confirming that the bound $\rho$ is tight.
\item \textbf{Over‑damped regime $\beta<(1-\sqrt{\alpha\mu})^{2}$.}
      The rate is still linear but strictly slower than the optimal
      $\rho^{\star}$; decreasing $\beta$ moves $\rho$ towards the GD value.
\end{enumerate}
