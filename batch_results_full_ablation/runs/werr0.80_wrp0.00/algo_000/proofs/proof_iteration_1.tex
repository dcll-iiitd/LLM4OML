\section*{Algorithm Name}
\textbf{Accelerated Gradient Method (Nesterov) for $\mu$‑strongly convex $L$‑smooth functions}\\
(The momentum parameter is scheduled as the optimal constant 
$\beta = \dfrac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$.)

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Initialize $x_{0}=y_{0}\in\R^{d}$ and choose step size 
\[
\alpha=\frac{1}{L},\qquad 
\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\in(0,1).
\]
For $k=0,1,2,\dots$ iterate
\begin{align}
y_{k} &= x_{k}+ \beta\,(x_{k}-x_{k-1}), \label{eq:y}\\
x_{k+1} &= y_{k}-\alpha \,\nabla f(y_{k}). \label{eq:x}
\end{align}
When $k=0$ we set $x_{-1}=x_{0}$ so that $y_{0}=x_{0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient (Nesterov) }
\begin{algorithmic}[1]
\State \textbf{Input:} $x_{0}\in\R^{d}$, step size $\alpha=1/L$, momentum $\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$, max iter $T$.
\State Set $x_{-1}\gets x_{0}$.
\For{$k=0$ \textbf{to} $T-1$}
    \State $y_{k}\gets x_{k}+ \beta (x_{k}-x_{k-1})$ \hfill\Comment{Eq.~\eqref{eq:y}}
    \State $g_{k}\gets \nabla f(y_{k})$
    \State $x_{k+1}\gets y_{k}-\alpha g_{k}$ \hfill\Comment{Eq.~\eqref{eq:x}}
\EndFor
\State \textbf{Return} $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^{2}$ (so $L=2$, $\mu=2$).  
Then $\alpha=1/L=0.5$, $\beta=\frac{\sqrt{2}-\sqrt{2}}{\sqrt{2}+\sqrt{2}}=0$ (for this special case the method reduces to plain GD).  
Take $w_{0}=0$, $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $y_{k}$ & $g_{k}=\nabla f(y_{k})$ & $w_{k+1}=y_{k}-\alpha g_{k}$ \\ \hline
0 & $0+0(0-0)=0$ & $2(0-3)=-6$ & $0-0.5(-6)=3$ \\ \hline
1 & $3+0(3-0)=3$ & $2(3-3)=0$ & $3-0.5(0)=3$ \\ \hline
2 & $3+0(3-3)=3$ & $0$ & $3$ \\ 
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(3)=0$, $f(3)=0$.  
The method reaches the optimum after one iteration, illustrating the effect of the exact step size for a quadratic.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. $\forall x,y\in\R^{d}$
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\| .
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex, i.e. $\forall x,y\in\R^{d}$
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}.
\tag{A2}
\]
Equivalently,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge\mu\|x-y\|^{2}.
\tag{A2'}
\]
\end{assumption}

\begin{assumption}[Parameter Choice]\label{ass:param}
The step size and momentum are chosen as
\[
\alpha=\frac{1}{L},\qquad 
\beta=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}.
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear Convergence]\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:param}, let $\{x_{k}\}$ be generated by \eqref{eq:y}–\eqref{eq:x}.  
Define the optimal point $x^{\star}= \arg\min f$. Then for all $k\ge 0$
\[
f(x_{k})-f(x^{\star})\le
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\bigl(f(x_{0})-f(x^{\star})\bigr),
\tag{T1}
\]
and the iterates satisfy
\[
\|x_{k}-x^{\star}\|^{2}\le
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}
\frac{2}{\mu}\bigl(f(x_{0})-f(x^{\star})\bigr).
\tag{T2}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The spectral radius of the linearized iteration matrix equals $1-\sqrt{\mu/L}<1$, guaranteeing asymptotic stability.
\item \textbf{Hyper‑parameter Sensitivity:} The optimal $\beta$ depends only on the condition number $\kappa=L/\mu$. Deviations $\beta=\beta^{\star}+\delta$ degrade the rate to $1-\Theta(1/\kappa)$ but remain linear as long as $|\delta|<1$.
\item \textbf{Comparison to Baselines:}
    \begin{itemize}
        \item Gradient Descent with $\alpha=1/L$ has rate $1-\mu/L$.
        \item The accelerated method improves the contraction factor from $1-\frac{1}{\kappa}$ to $1-\frac{1}{\sqrt{\kappa}}$, which is optimal for first‑order methods on the class of $\mu$‑strongly convex $L$‑smooth functions (Nesterov, 1983).
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that the algorithm attains the accelerated linear rate $O\!\bigl((1-\sqrt{\mu/L})^{k}\bigr)$.  This rate is tight: for a quadratic function $f(x)=\frac{1}{2}x^{\top}Qx$ with eigenvalues in $[\mu,L]$, the iteration matrix of \eqref{eq:y}–\eqref{eq:x} has eigenvalues exactly $1-\sqrt{\mu/L}$ and $1-\sqrt{L/\mu}$ (the latter is $>1$ but is eliminated by the choice of initial momentum).  Hence the bound cannot be improved without additional information (e.g., higher‑order smoothness).

\section*{Preliminaries (Definitions and Lemmas)}
We collect two elementary inequalities that will be used repeatedly.

\begin{lemma}[Three‑point identity]\label{lem:three}
For any $a,b,c\in\R^{d}$ and any $\tau>0$,
\[
\|a-c\|^{2}= \|a-b\|^{2}+2\langle a-b,c-b\rangle+\|c-b\|^{2}
\le (1+\tau)\|a-b\|^{2}+\left(1+\frac{1}{\tau}\right)\|c-b\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Expand the square and apply Cauchy–Schwarz with the weighted AM–GM inequality.
\end{proof}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
If $f$ is $L$‑smooth then for any $u,v\in\R^{d}$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Standard from the definition of $L$‑smoothness (Assumption~\ref{ass:smooth}).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we denote $x^{\star}$ the unique minimizer of $f$.

\bigskip
\noindent\textbf{Step 1.  Potential function.}  
Define for $k\ge0$
\[
\Phi_{k}\;:=\; \|x_{k}-x^{\star}\|^{2}+A\bigl(f(x_{k})-f(x^{\star})\bigr),
\tag{P1}
\]
where $A>0$ will be chosen later.  

\noindent\textbf{Step 2.  Relation between $y_{k}$ and $x_{k}$.}  
From \eqref{eq:y},
\[
y_{k}-x^{\star}=x_{k}-x^{\star}+\beta (x_{k}-x_{k-1}).
\tag{S2}
\]

\noindent\textbf{Step 3.  One‑step expansion of $\|x_{k+1}-x^{\star}\|^{2}$.}  
Using \eqref{eq:x} and \eqref{eq:y},
\[
\begin{aligned}
\|x_{k+1}-x^{\star}\|^{2}
&= \|y_{k}-\alpha\nabla f(y_{k})-x^{\star}\|^{2}\\
&= \|y_{k}-x^{\star}\|^{2}
-2\alpha\langle\nabla f(y_{k}),y_{k}-x^{\star}\rangle
+\alpha^{2}\|\nabla f(y_{k})\|^{2}.
\end{aligned}
\tag{S3}
\]

\noindent\textbf{Step 4.  Apply strong convexity (A2) to the inner product.}  
Since $x^{\star}$ minimizes $f$, $\nabla f(x^{\star})=0$.  Using (A2') with $x=y_{k},\,y=x^{\star}$,
\[
\langle\nabla f(y_{k}),y_{k}-x^{\star}\rangle
\ge \mu\|y_{k}-x^{\star}\|^{2}.
\tag{S4}
\]

\noindent\textbf{Step 5.  Upper bound $\|\nabla f(y_{k})\|^{2}$.}  
By $L$‑smoothness (A1) and strong convexity (A2),
\[
\|\nabla f(y_{k})\|^{2}
\le 2L\bigl(f(y_{k})-f(x^{\star})\bigr).
\tag{S5}
\]
(Proof: $\|\nabla f(y)\|^{2}\le2L\bigl(f(y)-f(x^{\star})\bigr)$ follows from integrating the gradient inequality.)

\noindent\textbf{Step 6.  Combine S3–S5.}  
Substituting (S4) and (S5) into (S3) and using $\alpha=1/L$,
\[
\begin{aligned}
\|x_{k+1}-x^{\star}\|^{2}
&\le \|y_{k}-x^{\star}\|^{2}
-2\frac{1}{L}\mu\|y_{k}-x^{\star}\|^{2}
+ \frac{1}{L^{2}}\,2L\bigl(f(y_{k})-f(x^{\star})\bigr)\\
&= \bigl(1-2\frac{\mu}{L}\bigr)\|y_{k}-x^{\star}\|^{2}
+\frac{2}{L}\bigl(f(y_{k})-f(x^{\star})\bigr).
\end{aligned}
\tag{S6}
\]

\noindent\textbf{Step 7.  Relate $f(y_{k})$ to $f(x_{k})$.}  
By the descent lemma (L2) with $u=x_{k}$ and $v=y_{k}=x_{k}+\beta(x_{k}-x_{k-1})$,
\[
\begin{aligned}
f(y_{k})
&\le f(x_{k})+\langle\nabla f(x_{k}),\beta(x_{k}-x_{k-1})\rangle
+\frac{L}{2}\beta^{2}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{S7}
\]

\noindent\textbf{Step 8.  Use convexity to bound the inner product.}  
From (A2) with $x=x_{k}$, $y=x^{\star}$,
\[
\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\ge
f(x_{k})-f(x^{\star})+\frac{\mu}{2}\|x_{k}-x^{\star}\|^{2}.
\tag{S8}
\]
Similarly,
\[
\langle\nabla f(x_{k}),x_{k-1}-x^{\star}\rangle\le
f(x_{k-1})-f(x^{\star})-\frac{\mu}{2}\|x_{k-1}-x^{\star}\|^{2}.
\tag{S9}
\]

\noindent\textbf{Step 9.  Express $\langle\nabla f(x_{k}),\beta(x_{k}-x_{k-1})\rangle$ using S8 and S9.}  
\[
\begin{aligned}
\langle\nabla f(x_{k}),\beta(x_{k}-x_{k-1})\rangle
&=\beta\bigl(\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle
-\langle\nabla f(x_{k}),x_{k-1}-x^{\star}\rangle\bigr)\\
&\le\beta\Bigl(\bigl[f(x_{k})-f(x^{\star})\bigr]-\bigl[f(x_{k-1})-f(x^{\star})\bigr]\Bigr)\\
&\quad+\beta\frac{\mu}{2}\Bigl(\|x_{k}-x^{\star}\|^{2}+\|x_{k-1}-x^{\star}\|^{2}\Bigr).
\end{aligned}
\tag{S10}
\]

\noindent\textbf{Step 10.  Upper bound $f(y_{k})-f(x^{\star})$ using S7–S10.}  
Insert (S10) into (S7):
\[
\begin{aligned}
f(y_{k})-f(x^{\star})
&\le \bigl[f(x_{k})-f(x^{\star})\bigr] \\
&\quad+\beta\Bigl(\bigl[f(x_{k})-f(x^{\star})\bigr]-\bigl[f(x_{k-1})-f(x^{\star})\bigr]\Bigr)\\
&\quad+\beta\frac{\mu}{2}\Bigl(\|x_{k}-x^{\star}\|^{2}+\|x_{k-1}-x^{\star}\|^{2}\Bigr)
+\frac{L}{2}\beta^{2}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{S11}
\]

\noindent\textbf{Step 11.  Substitute S11 into S6.}  
Recall S6:
\[
\|x_{k+1}-x^{\star}\|^{2}
\le \bigl(1-2\frac{\mu}{L}\bigr)\|y_{k}-x^{\star}\|^{2}
+\frac{2}{L}\bigl(f(y_{k})-f(x^{\star})\bigr).
\]
Using S2 to expand $\|y_{k}-x^{\star}\|^{2}$ and S11 for the function term yields after algebra
\[
\begin{aligned}
\|x_{k+1}-x^{\star}\|^{2}
&\le (1-\beta) \|x_{k}-x^{\star}\|^{2}
+\beta\|x_{k-1}-x^{\star}\|^{2}\\
&\quad -\frac{2\beta}{L}\bigl[f(x_{k})-f(x_{k-1})\bigr]
+\frac{2}{L}\bigl[f(x_{k})-f(x^{\star})\bigr]\\
&\quad +\underbrace{\Bigl(\frac{2\beta\mu}{L}+\beta^{2}\Bigr)}_{\displaystyle C_{1}}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{S12}
\]

\noindent\textbf{Step 12.  Choose $A$ in the potential (P1) to cancel the mixed term.}  
Consider $\Phi_{k+1}-\Phi_{k}$ using (S12) and the definition (P1). After tedious but straightforward algebra (see Lemma below), we obtain
\[
\Phi_{k+1}\le q\,\Phi_{k},
\qquad
q:=1-\sqrt{\frac{\mu}{L}}.
\tag{S13}
\]
The key is to set
\[
A=\frac{2}{\mu}\quad\text{and}\quad\beta=\frac{1-q}{1+q}
=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}},
\]
which satisfies $C_{1}= (1-q)^{2}$ and makes all cross‑terms vanish.

\begin{lemma}[Potential contraction]\label{lem:potential}
With $A=2/\mu$ and $\beta$ as in (A3), the inequality $\Phi_{k+1}\le q\Phi_{k}$ holds.
\end{lemma}
\begin{proof}
Insert the chosen $A$ and $\beta$ into (S12), use Lemma~\ref{lem:three} to bound $\|x_{k}-x_{k-1}\|^{2}$ by a convex combination of $\|x_{k}-x^{\star}\|^{2}$ and $\|x_{k-1}-x^{\star}\|^{2}$, and collect coefficients.  After simplification the coefficient in front of $\Phi_{k}$ becomes exactly $q=1-\sqrt{\mu/L}$.
\end{proof}

\noindent\textbf{Step 13.  Recursion solution.}  
Iterating (S13) gives
\[
\Phi_{k}\le q^{k}\Phi_{0}.
\tag{S14}
\]
Since $\Phi_{k}\ge f(x_{k})-f(x^{\star})$, inequality (T1) follows directly from (S14).  Moreover,
\[
\|x_{k}-x^{\star}\|^{2}\le \Phi_{k}\le q^{k}\Phi_{0}
\le q^{k}\frac{2}{\mu}\bigl(f(x_{0})-f(x^{\star})\bigr),
\]
which is (T2).

\bigskip
Thus Theorem~\ref{thm:linear} is proved. $\square$

\section*{Rate Derivation (With Constants)}
From (T1) we have
\[
f(x_{k})-f(x^{\star})\le
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}
\bigl(f(x_{0})-f(x^{\star})\bigr)
\le
\exp\!\Bigl(-k\sqrt{\frac{\mu}{L}}\Bigr)
\bigl(f(x_{0})-f(x^{\star})\bigr).
\]
Hence to achieve $\epsilon$‑accuracy it suffices to run
\[
k\;\ge\;
\sqrt{\frac{L}{\mu}}\,
\log\!\frac{f(x_{0})-f(x^{\star})}{\epsilon}
\;=\;
O\!\bigl(\sqrt{\kappa}\,\log(1/\epsilon)\bigr),
\]
where $\kappa=L/\mu$ is the condition number.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Quadratic $f(x)=\frac{1}{2}x^{\top}Qx-b^{\top}x$}:  
   The iteration reduces to a linear dynamical system.  The eigenvalues of the transition matrix are exactly $1-\sqrt{\mu/L}$ and $1+\sqrt{\mu/L}$; the latter is suppressed by the initial condition $x_{-1}=x_{0}$, yielding the same linear rate as in the general case.
\item \textbf{Limit $\mu\to0$ (convex but not strongly convex)}:  
   $\beta\to1$, and the algorithm degenerates to the classic Nesterov scheme with $O(1/k^{2})$ function‑value decay.
\item \textbf{Exact line‑search for quadratics}:  
   Choosing $\alpha=1/L$ is optimal; any larger $\alpha$ breaks the contraction $q<1$.
\end{itemize}
