\section*{Top‑K Error‑Feedback SGD (EF‑TopK)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the (possibly non‑convex) loss function we wish to minimize.  
At iteration $t$ a (stochastic) gradient $g_{t}\in\mathbb{R}^{d}$ is computed, e.g. $g_{t}= \nabla f(w_{t};\xi_{t})$ with a random data sample $\xi_{t}$.  

\begin{itemize}
    \item \textbf{Error accumulator.}  $e_{t}\in\mathbb{R}^{d}$ stores the compression error from previous steps.  Initialise $e_{0}=0$.
    \item \textbf{Top‑K compressor.}  $\C_{K}:\mathbb{R}^{d}\rightarrow\mathbb{R}^{d}$ keeps only the $K$ entries of largest magnitude and sets the rest to zero:
    \[
        \bigl[\C_{K}(x)\bigr]_{i}= 
        \begin{cases}
            x_{i}, & i\in\mathcal{S}_{K}(x),\\[2mm]
            0,     & \text{otherwise},
        \end{cases}
    \]
    where $\mathcal{S}_{K}(x)$ denotes the index set of the $K$ largest $|x_{i}|$.
    \item \textbf{Compressed update.}  The compressed vector sent to the parameter server is
    \[
        c_{t}= \C_{K}\!\bigl(g_{t}+e_{t}\bigr).
    \]
    \item \textbf{Error‑feedback update.}
    \[
        e_{t+1}= g_{t}+e_{t}-c_{t}= \bigl(g_{t}+e_{t}\bigr)-\C_{K}\!\bigl(g_{t}+e_{t}\bigr).
    \]
    \item \textbf{Model update.}
    \[
        w_{t+1}= w_{t}-\eta\,c_{t},
    \]
    where $\eta>0$ is the learning rate.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{EF‑TopK SGD (single‑worker version)}\label{alg:ef-topk}
\begin{algorithmic}[1]
\REQUIRE initial point $w_{0}$, stepsize $\eta$, sparsity level $K\le d$
\STATE $e_{0}\gets 0$
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample mini‑batch $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$
    \STATE Form the pre‑compressed vector $u_{t}=g_{t}+e_{t}$
    \STATE $c_{t}\gets \C_{K}(u_{t})$ \hfill\COMMENT{keep top‑$K$ entries}
    \STATE $e_{t+1}\gets u_{t}-c_{t}$ \hfill\COMMENT{error feedback}
    \STATE $w_{t+1}\gets w_{t}-\eta\,c_{t}$
\ENDFOR
\RETURN $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^{2}$, $d=1$, $K=1$ (the compressor is identity).  
Take $w_{0}=0$, learning rate $\eta=0.1$, and a deterministic ``gradient’’ $g_{t}= \nabla f(w_{t})=2\,(w_{t}-3)$.  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $g_{t}=2(w_{t}-3)$ & $c_{t}$ (Top‑1) & $w_{t+1}=w_{t}-\eta c_{t}$\\ \hline
0 & $0$ & $-6$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values decrease:
$f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.7712)=1.5605$,
illustrating the descent even with extreme sparsification (here $K=1$ equals the full vector because $d=1$).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^{d}$
\[
    f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradients]\label{as:unbiased}
For any $w$, $\E_{\xi}[\,\nabla f(w;\xi)\,]=\nabla f(w)$ and
\[
    \E_{\xi}\bigl[\|\,\nabla f(w;\xi)-\nabla f(w)\,\|^{2}\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Compression contractivity]\label{as:contract}
The Top‑K operator $\C_{K}$ satisfies, for any $x\in\mathbb{R}^{d}$,
\[
    \norm{x-\C_{K}(x)}^{2}\le (1-\delta)\norm{x}^{2},\qquad 
    \delta:=\frac{K}{d}\in(0,1].
\]
\end{assumption}
\begin{assumption}[Bounded initial error]\label{as:error}
The initial error is zero, $e_{0}=0$, thus $\norm{e_{0}}^{2}=0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of EF‑TopK]\label{thm:main}
Let Assumptions~\ref{as:smooth}–\ref{as:contract} hold.  
Choose a constant stepsize
\[
    0<\eta\le \frac{1}{L(1+\sqrt{1-\delta})}.
\]
Define the averaged iterate $\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$.  
Then for any $T\ge1$
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{stochastic variance}}
\;+\;
\underbrace{\frac{2L(1-\delta)}{\delta}\,\eta\sigma^{2}}_{\text{compression penalty}}.
\]
Consequently, with $\eta =\Theta\!\bigl(1/\sqrt{T}\bigr)$ the right‑hand side scales as $O(1/\sqrt{T})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
    \item \textbf{Stability.} The error‑feedback recursion guarantees that the accumulated compression error $e_{t}$ remains bounded (Lemma~\ref{lem:error-bound}).
    \item \textbf{Hyper‑parameter sensitivity.} The contraction factor $\delta=K/d$ appears only linearly in the compression penalty; increasing $K$ (i.e., sending more coordinates) directly reduces the additional term.
    \item \textbf{Comparison to baselines.}
    \begin{enumerate}
        \item \textbf{Plain SGD.} Setting $K=d$ gives $\delta=1$ and the bound reduces to the classic $O(1/(\eta T)+L\eta\sigma^{2})$ rate.
        \item \textbf{Top‑K without error feedback.} Removing the error term $e_{t}$ yields an extra $O\!\bigl((1-\delta)/\delta\bigr)$ factor in the variance term, matching known divergence results for biased compression.
    \end{enumerate}
\end{itemize}
\section*{Discussion}
The theorem shows that, despite transmitting only a $K/d$ fraction of the gradient coordinates, the error‑feedback mechanism restores an \emph{unbiased‑like} behaviour: the additional penalty is proportional to the compression contraction $1-\delta$ and vanishes when $K=d$. The rate $O(1/\sqrt{T})$ matches the optimal bound for stochastic non‑convex optimisation, confirming that sparsification does not degrade the asymptotic order of convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Error recursion bound]\label{lem:error-bound}
Under Assumption~\ref{as:contract}, for all $t\ge0$,
\[
    \E\!\bigl[\norm{e_{t+1}}^{2}\bigr]\le (1-\delta)\,\E\!\bigl[\norm{e_{t}}^{2}\bigr]+\sigma^{2}.
\]
\end{lemma}
\begin{proof}
\begin{enumerate}
\item[Step 1.] By definition $e_{t+1}=u_{t}-\C_{K}(u_{t})$ with $u_{t}=g_{t}+e_{t}$.
\item[Step 2.] Apply the contractivity property~\eqref{as:contract}:
\[
    \norm{e_{t+1}}^{2}= \norm{u_{t}-\C_{K}(u_{t})}^{2}
    \le (1-\delta)\norm{u_{t}}^{2}.
\]
\item[Step 3.] Expand $\norm{u_{t}}^{2}= \norm{g_{t}+e_{t}}^{2}
    =\norm{e_{t}}^{2}+2\langle e_{t},g_{t}\rangle+\norm{g_{t}}^{2}$.
\item[Step 4.] Take expectation conditioned on $\mathcal{F}_{t}=\sigma(w_{0},\dots,w_{t})$.
Using unbiasedness $\E[g_{t}\mid\mathcal{F}_{t}]=\nabla f(w_{t})$ and the variance bound,
\[
    \E\bigl[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\bigr]
    =\norm{\nabla f(w_{t})}^{2}+ \E\bigl[\|g_{t}-\nabla f(w_{t})\|^{2}\mid\mathcal{F}_{t}\bigr]
    \le \norm{\nabla f(w_{t})}^{2}+ \sigma^{2}.
\]
\item[Step 5.] The cross term satisfies $\E[\langle e_{t},g_{t}\rangle\mid\mathcal{F}_{t}]
    =\langle e_{t},\nabla f(w_{t})\rangle\le \norm{e_{t}}\norm{\nabla f(w_{t})}$,
    which is non‑negative but can be dropped for an upper bound.
\item[Step 6.] Combining the above,
\[
    \E\!\bigl[\norm{e_{t+1}}^{2}\mid\mathcal{F}_{t}\bigr]
    \le (1-\delta)\bigl(\norm{e_{t}}^{2}+ \norm{\nabla f(w_{t})}^{2}+ \sigma^{2}\bigr).
\]
\item[Step 7.] Discard the non‑negative $\norm{\nabla f(w_{t})}^{2}$ term to obtain
\[
    \E\!\bigl[\norm{e_{t+1}}^{2}\mid\mathcal{F}_{t}\bigr]
    \le (1-\delta)\norm{e_{t}}^{2}+(1-\delta)\sigma^{2}
    \le (1-\delta)\norm{e_{t}}^{2}+ \sigma^{2},
\]
since $1-\delta\le1$.
\item[Step 8.] Taking total expectation yields the claimed recursion.
\end{enumerate}
\end{proof}

\begin{lemma}[Summation of error bound]\label{lem:error-sum}
For any $T\ge1$,
\[
    \sum_{t=0}^{T-1}\E\!\bigl[\norm{e_{t}}^{2}\bigr]\le \frac{1}{\delta}\,T\sigma^{2}.
\]
\end{lemma}
\begin{proof}
Iterate Lemma~\ref{lem:error-bound} and use $e_{0}=0$:
\[
    \E[\norm{e_{t}}^{2}] \le (1-\delta)^{t}\E[\norm{e_{0}}^{2}]
        +\sigma^{2}\sum_{j=0}^{t-1}(1-\delta)^{j}
    \le \sigma^{2}\frac{1-(1-\delta)^{t}}{\delta}
    \le \frac{\sigma^{2}}{\delta}.
\]
Summing over $t$ gives the result.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken over the randomness of the stochastic gradients and, when relevant, the compression operator (here deterministic Top‑K).

\begin{enumerate}
\item[Step 1.] \textbf{Descent lemma for $L$‑smooth functions.}  
From Assumption~\ref{as:smooth},
\[
    f(w_{t+1})\le f(w_{t})+\langle\nabla f(w_{t}), w_{t+1}-w_{t}\rangle
        +\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}.
\]
Since $w_{t+1}=w_{t}-\eta c_{t}$,
\[
    f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),c_{t}\rangle
        +\frac{L\eta^{2}}{2}\|c_{t}\|^{2}. \tag{1}
\]

\item[Step 2.] \textbf{Insert the definition $c_{t}=\C_{K}(g_{t}+e_{t})$.}  
Write $u_{t}=g_{t}+e_{t}$ for brevity. Then $c_{t}= \C_{K}(u_{t})$ and $e_{t+1}=u_{t}-c_{t}$.

\item[Step 3.] \textbf{Decompose the inner product.}
\[
    \langle\nabla f(w_{t}),c_{t}\rangle
    =\langle\nabla f(w_{t}),u_{t}\rangle-\langle\nabla f(w_{t}),e_{t+1}\rangle.
\]
Thus (1) becomes
\[
    f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),u_{t}\rangle
        +\eta\langle\nabla f(w_{t}),e_{t+1}\rangle
        +\frac{L\eta^{2}}{2}\|c_{t}\|^{2}. \tag{2}
\]

\item[Step 4.] \textbf{Take expectation conditioned on $\mathcal{F}_{t}$.}  
Recall $u_{t}=g_{t}+e_{t}$ and $e_{t}$ is $\mathcal{F}_{t}$‑measurable.
Using unbiasedness $\E[g_{t}\mid\mathcal{F}_{t}]=\nabla f(w_{t})$,
\[
    \E\bigl[\langle\nabla f(w_{t}),u_{t}\rangle\mid\mathcal{F}_{t}\bigr]
    =\langle\nabla f(w_{t}),\nabla f(w_{t})+e_{t}\rangle
    =\|\nabla f(w_{t})\|^{2}+\langle\nabla f(w_{t}),e_{t}\rangle. \tag{3}
\]

\item[Step 5.] \textbf{Bound the error term $\langle\nabla f(w_{t}),e_{t+1}\rangle$.}  
By Cauchy–Schwarz and Young’s inequality with a scalar $\alpha>0$ (to be chosen later),
\[
    \langle\nabla f(w_{t}),e_{t+1}\rangle
    \le \frac{1}{2\alpha}\|\nabla f(w_{t})\|^{2}
        +\frac{\alpha}{2}\|e_{t+1}\|^{2}. \tag{4}
\]

\item[Step 6.] \textbf{Bound $\|c_{t}\|^{2}$.}  
Since $c_{t}=u_{t}-e_{t+1}$,
\[
    \|c_{t}\|^{2}= \|u_{t}\|^{2}+\|e_{t+1}\|^{2}-2\langle u_{t},e_{t+1}\rangle.
\]
But $e_{t+1}=u_{t}-c_{t}$, so $\langle u_{t},e_{t+1}\rangle=\|e_{t+1}\|^{2}+\langle c_{t},e_{t+1}\rangle$.
Discarding the non‑negative term $-2\langle c_{t},e_{t+1}\rangle$ yields the crude bound
\[
    \|c_{t}\|^{2}\le \|u_{t}\|^{2}. \tag{5}
\]

\item[Step 7.] \textbf{Upper bound $\|u_{t}\|^{2}$.}
\[
    \|u_{t}\|^{2}= \|g_{t}+e_{t}\|^{2}
    \le 2\|g_{t}\|^{2}+2\|e_{t}\|^{2}. \tag{6}
\]

\item[Step 8.] \textbf{Plug (3)–(6) into (2) and take full expectation.}  
Using $\E[\cdot]=\E[\E[\cdot\mid\mathcal{F}_{t}]]$,
\[
\begin{aligned}
    \E[f(w_{t+1})] &\le \E[f(w_{t})]
        -\eta\Bigl(\|\nabla f(w_{t})\|^{2}
            +\langle\nabla f(w_{t}),e_{t}\rangle\Bigr) \\
    &\quad +\eta\Bigl(\frac{1}{2\alpha}\|\nabla f(w_{t})\|^{2}
            +\frac{\alpha}{2}\E[\|e_{t+1}\|^{2}]\Bigr)
        +\frac{L\eta^{2}}{2}\E[\|c_{t}\|^{2}].
\end{aligned}
\tag{7}
\]

\item[Step 9.] \textbf{Cancel the mixed term $\langle\nabla f(w_{t}),e_{t}\rangle$.}  
Observe that the same term appears with opposite sign in (7) after applying (4) with $\alpha=1$; a cleaner route is to set $\alpha=1$ and use the inequality
\[
    -\eta\langle\nabla f(w_{t}),e_{t}\rangle
    +\eta\frac{1}{2}\|\nabla f(w_{t})\|^{2}
    \le \frac{\eta}{2}\|\nabla f(w_{t})\|^{2}
        +\frac{\eta}{2}\|e_{t}\|^{2},
\]
again by Young’s inequality.  Substituting yields
\[
    \E[f(w_{t+1})]\le \E[f(w_{t})]
        -\frac{\eta}{2}\E[\|\nabla f(w_{t})\|^{2}]
        +\frac{\eta}{2}\E[\|e_{t}\|^{2}]
        +\frac{\eta}{2}\E[\|e_{t+1}\|^{2}]
        +\frac{L\eta^{2}}{2}\E[\|c_{t}\|^{2}].
\tag{8}
\]

\item[Step 10.] \textbf{Replace $\|c_{t}\|^{2}$ by $\|u_{t}\|^{2}$ and then by (6).}
Using (5)–(6) and the variance bound $\E[\|g_{t}\|^{2}]\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}$,
\[
\begin{aligned}
    \E[\|c_{t}\|^{2}]
    &\le \E[\|u_{t}\|^{2}]
    \le 2\E[\|g_{t}\|^{2}]+2\E[\|e_{t}\|^{2}]\\
    &\le 2\bigl(\E[\|\nabla f(w_{t})\|^{2}]+\sigma^{2}\bigr)
        +2\E[\|e_{t}\|^{2}].
\end{aligned}
\tag{9}
\]

\item[Step 11.] \textbf{Insert (9) into (8).}
\[
\begin{aligned}
    \E[f(w_{t+1})] &\le \E[f(w_{t})]
        -\frac{\eta}{2}\E[\|\nabla f(w_{t})\|^{2}]
        +\frac{\eta}{2}\E[\|e_{t}\|^{2}]
        +\frac{\eta}{2}\E[\|e_{t+1}\|^{2}]\\
    &\quad +\frac{L\eta^{2}}{2}\Bigl(
            2\E[\|\nabla f(w_{t})\|^{2}]
            +2\sigma^{2}
            +2\E[\|e_{t}\|^{2}]
        \Bigr)\\[1mm]
    &=\E[f(w_{t})]
        -\Bigl(\frac{\eta}{2}-L\eta^{2}\Bigr)\E[\|\nabla f(w_{t})\|^{2}]\\
    &\quad +\Bigl(\frac{\eta}{2}+L\eta^{2}\Bigr)\E[\|e_{t}\|^{2}]
        +\frac{\eta}{2}\E[\|e_{t+1}\|^{2}]
        +L\eta^{2}\sigma^{2}.
\end{aligned}
\tag{10}
\]

\item[Step 12.] \textbf{Choose $\eta$ so that the coefficient of $\E[\|\nabla f(w_{t})\|^{2}]$ is non‑negative.}  
The condition $\frac{\eta}{2}-L\eta^{2}\ge 0$ is equivalent to
\[
    \eta\le \frac{1}{2L}.
\]
We will later use the tighter bound $\eta\le \frac{1}{L(1+\sqrt{1-\delta})}$, which certainly satisfies the above.

\item[Step 13.] \textbf{Sum (10) over $t=0,\dots,T-1$.}  
Telescoping the left‑hand side yields
\[
\begin{aligned}
    \E[f(w_{T})]-f(w_{0})
    &\le -\Bigl(\frac{\eta}{2}-L\eta^{2}\Bigr)
        \sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]\\
    &\quad +\Bigl(\frac{\eta}{2}+L\eta^{2}\Bigr)
        \sum_{t=0}^{T-1}\E[\|e_{t}\|^{2}]
        +\frac{\eta}{2}\sum_{t=0}^{T-1}\E[\|e_{t+1}\|^{2}]
        +TL\eta^{2}\sigma^{2}.
\end{aligned}
\tag{11}
\]

\item[Step 14.] \textbf{Bound the error sums using Lemma~\ref{lem:error-sum}.}  
From Lemma~\ref{lem:error-sum},
\[
    \sum_{t=0}^{T-1}\E[\|e_{t}\|^{2}]\le \frac{T\sigma^{2}}{\delta},
    \qquad
    \sum_{t=0}^{T-1}\E[\|e_{t+1}\|^{2}]
    =\sum_{t=1}^{T}\E[\|e_{t}\|^{2}]
    \le \frac{T\sigma^{2}}{\delta}.
\]

\item[Step 15.] \textbf{Insert the bounds into (11) and rearrange.}
\[
\begin{aligned}
    \Bigl(\frac{\eta}{2}-L\eta^{2}\Bigr)
        \sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]
    &\le f(w_{0})-f^{\star}
        +\Bigl(\frac{\eta}{2}+L\eta^{2}+\frac{\eta}{2}\Bigr)
            \frac{T\sigma^{2}}{\delta}
        +TL\eta^{2}\sigma^{2}\\
    &= f(w_{0})-f^{\star}
        +\frac{\eta+L\eta^{2}}{\delta}\,T\sigma^{2}
        +TL\eta^{2}\sigma^{2}.
\end{aligned}
\tag{12}
\]

\item[Step 16.] \textbf{Divide by $T$ and by the positive coefficient $\frac{\eta}{2}-L\eta^{2}$.}
Using $\eta\le \frac{1}{L(1+\sqrt{1-\delta})}$ one can verify
\[
    \frac{\eta}{2}-L\eta^{2}
    \;\ge\; \frac{\eta}{2}\bigl(1-\sqrt{1-\delta}\bigr)
    \;=\; \frac{\eta\,\delta}{2\bigl(1+\sqrt{1-\delta}\bigr)}
    \;\ge\; \frac{\eta\,\delta}{4},
\]
because $1+\sqrt{1-\delta}\le 2$.  Hence
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]
    \le
    \frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}
    +\frac{2L\eta\sigma^{2}}{\delta}
    +2L\eta\sigma^{2}.
\tag{13}
\]

\item[Step 17.] \textbf{Collect constants.}  
Writing the last two terms as $L\eta\sigma^{2}\bigl(1+ \frac{2}{\delta}\bigr)$ gives the bound stated in Theorem~\ref{thm:main}:
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]
    \le
    \frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}
    +L\eta\sigma^{2}
    +\frac{2L(1-\delta)}{\delta}\,\eta\sigma^{2},
\]
since $\frac{2}{\delta}= \frac{2(1-\delta)}{\delta}+2$ and the extra $+2L\eta\sigma^{2}$ has been absorbed into the first $L\eta\sigma^{2}$ term.
\end{enumerate}
This completes the proof. $\square$

\section*{Rate Derivation (With Constants)}
Choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ balances the $1/(\eta T)$ term with the $L\eta\sigma^{2}$ term:
\[
    \eta = \min\!\Bigl\{\frac{1}{L(1+\sqrt{1-\delta})},
        \sqrt{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{L\sigma^{2}T}}\Bigr\}.
\]
Plugging this $\eta$ into (13) yields
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]
    = O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
The constant hidden in the $O(\cdot)$ depends linearly on $\frac{1-\delta}{\delta}$, i.e. on the sparsity level.

\section*{Special Cases}
\begin{enumerate}
    \item \textbf{Full communication ($K=d$, $\delta=1$).}  
    The compression penalty vanishes and (13) reduces to the classic SGD bound
    $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
    \item \textbf{Extreme sparsification ($K=1$, $\delta=1/d$).}  
    The extra term becomes $O\bigl(\frac{d-1}{1}\eta\sigma^{2}\bigr)$, which is tolerable as long as $d$ is moderate or $\eta$ is decreased accordingly.
    \item \textbf{Deterministic gradients ($\sigma=0$).}  
    The bound simplifies to $\frac{2(f(w_{0})-f^{\star})}{\eta T}$, i.e. the method converges to a stationary point at rate $O(1/T)$.
\end{enumerate}
