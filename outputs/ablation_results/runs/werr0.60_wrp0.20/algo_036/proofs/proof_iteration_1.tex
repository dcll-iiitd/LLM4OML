\section*{Stochastic Newton Method (SNM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be $\mu$‑strongly convex and $L$‑smooth.  
Denote by $x^{\star}=\arg\min_{x}f(x)$ the unique minimiser.  
At iteration $t$ we have a stochastic estimate of the Hessian,
\[
H_{t}\in\R^{d\times d},
\qquad 
\text{with } \; \E[H_{t}\mid x_{t}]=\nabla^{2}f(x_{t}) .
\]
The update rule of SNM is
\begin{equation}\label{eq:update}
x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}\nabla f(x_{t}),\qquad  \alpha>0.
\end{equation}
The step size $\alpha$ is a deterministic scalar (possibly dependent on $t$) that will be chosen to guarantee convergence.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton Method}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsize $\alpha>0$
\FOR{$t=0,1,\dots,T-1$}
    \STATE Compute (or sample) a stochastic gradient $g_{t}=\nabla f(x_{t})$.
    \STATE Obtain a stochastic Hessian estimate $H_{t}$ such that
          $\E[H_{t}\mid x_{t}]=\nabla^{2}f(x_{t})$.
    \STATE Update 
    \[
    x_{t+1}=x_{t}-\alpha\,H_{t}^{-1}g_{t}.
    \]
\ENDFOR
\RETURN $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3),\qquad \nabla^{2}f(w)=2 .
\]
We take a *deterministic* Hessian estimate $H_{t}=2$ (so $H_{t}^{-1}=1/2$) and stepsize $\alpha=0.1$.
\[
\text{Initial point } w_{0}=0.
\]

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_{t})$ & $w_{t+1}=w_{t}-\alpha H_{t}^{-1}\nabla f(w_{t})$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1\cdot\frac{1}{2}(-6)=0+0.3=0.3$ & $(0.3-3)^{2}=7.29$\\
1 & $2(0.3-3)=-5.4$ & $0.3-0.1\cdot\frac{1}{2}(-5.4)=0.3+0.27=0.57$ & $(0.57-3)^{2}=5.9049$\\
2 & $2(0.57-3)=-4.86$ & $0.57-0.1\cdot\frac{1}{2}(-4.86)=0.57+0.243=0.813$ & $(0.813-3)^{2}=4.7837$\\
\end{tabular}
\end{center}
The objective value decreases geometrically, illustrating the linear convergence that will be proved formally below.

\section*{Assumptions}
\begin{assumption}[Strong convexity and smoothness]\label{as:sc}
$f$ is $\mu$‑strongly convex and $L$‑smooth, i.e.
\[
\mu I \preceq \nabla^{2}f(x) \preceq L I,\qquad\forall x\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Stochastic Hessian quality]\label{as:hessian}
For every $t$,
\begin{enumerate}
\item \textbf{Unbiasedness:}\quad $\displaystyle \E[H_{t}\mid x_{t}]=\nabla^{2}f(x_{t})$.
\item \textbf{Uniform spectral bounds:}\quad there exist constants $0<\mu_{h}\le L_{h}$ such that
\[
\mu_{h} I \preceq H_{t} \preceq L_{h} I\qquad\text{almost surely}.
\]
Equivalently, the condition number of each stochastic Hessian satisfies
\[
\kappa_{H}:=\frac{L_{h}}{\mu_{h}}<\infty .
\]
\end{enumerate}
\end{assumption}
\begin{assumption}[Bounded stochastic gradient variance]\label{as:grad}
There exists $\sigma^{2}\ge0$ such that
\[
\E\!\left[\| \nabla f(x_{t})-g_{t}\|^{2}\mid x_{t}\right]\le\sigma^{2},
\qquad g_{t}:=\nabla f(x_{t})\;( \text{here we use exact gradients for clarity} ).
\]
\end{assumption}
\noindent In the remainder we keep the notation $g_{t}=\nabla f(x_{t})$; the proof works verbatim when $g_{t}$ is a stochastic unbiased gradient.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Assume \ref{as:sc}--\ref{as:hessian} and choose a constant stepsize
\[
\alpha\in\bigl(0,\; \tfrac{2\mu_{h}}{L}\bigr].
\]
Define $x^{\star}=\arg\min f$ and the error vector $e_{t}=x_{t}-x^{\star}$. Then for all $t\ge0$
\begin{equation}\label{eq:linear}
\E\!\left[\|e_{t+1}\|^{2}\right]\;\le\;
\Bigl(1-\alpha\frac{\mu}{\kappa_{H}}\Bigr)\,\E\!\left[\|e_{t}\|^{2}\right],
\qquad\kappa_{H}=L_{h}/\mu_{h}.
\end{equation}
Consequently,
\[
\E\!\left[\|e_{t}\|^{2}\right]\le
\Bigl(1-\alpha\frac{\mu}{\kappa_{H}}\Bigr)^{t}\|e_{0}\|^{2},
\]
i.e. SNM attains a geometric (linear) rate with contraction factor $1-\alpha\mu/\kappa_{H}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The uniform eigenvalue bounds on $H_{t}$ guarantee that the inverse $H_{t}^{-1}$ never explodes; the stepsize restriction $\alpha\le 2\mu_{h}/L$ ensures the mapping $I-\alpha H_{t}^{-1}\nabla^{2}f(x)$ is a contraction.
\item \textbf{Hyperparameter sensitivity.} The optimal choice $\alpha^{\star}= \mu_{h}/L$ yields the maximal contraction $1-\frac{\mu\,\mu_{h}}{L\,L_{h}}=1-\frac{\mu}{\kappa_{H}L/L_{h}}$. Because $L_{h}$ is typically of the same order as $L$, the rate essentially matches that of the deterministic Newton method up to the stochastic condition number $\kappa_{H}$.
\item \textbf{Comparison to SGD.} SG\!D under the same assumptions converges as $O(1/t)$ (sub‑linear). SNM enjoys a linear rate, at the cost of requiring a Hessian estimate with bounded spectrum.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Quadratic bound from strong convexity]\label{lem:strong}
For any $x$, $y\in\R^{d}$,
\[
\frac{\mu}{2}\|x-y\|^{2}\le f(x)-f(y)-\langle\nabla f(y),x-y\rangle\le \frac{L}{2}\|x-y\|^{2}.
\]
\end{lemma}
\begin{proof}
Direct consequence of the eigenvalue bounds $\mu I\preceq\nabla^{2}f(\cdot)\preceq L I$ and integration of the gradient. \hfill $\square$
\end{proof}

\begin{lemma}[Spectral bound for the inverse Hessian]\label{lem:inv}
If $\mu_{h} I\preceq H_{t}\preceq L_{h} I$, then
\[
\frac{1}{L_{h}} I\preceq H_{t}^{-1}\preceq\frac{1}{\mu_{h}} I.
\]
\end{lemma}
\begin{proof}
Take inverses of the inequality in \ref{as:hessian} and reverse the order because matrix inversion is order‑reversing on the PSD cone. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $e_{t}=x_{t}-x^{\star}$. Using the update \eqref{eq:update} we have
\[
e_{t+1}=e_{t}-\alpha H_{t}^{-1}\nabla f(x_{t}) .
\tag{1}
\]
Because $\nabla f(x^{\star})=0$, we may rewrite $\nabla f(x_{t})$ as
\[
\nabla f(x_{t})=\nabla^{2}f(\xi_{t})\,e_{t},
\qquad \xi_{t}=x^{\star}+\theta_{t}e_{t},\; \theta_{t}\in(0,1),
\]
by the mean‑value theorem for vector‑valued functions.

\subsection*{Step 1 – Expand the squared error}
\[
\|e_{t+1}\|^{2}
= \|e_{t}\|^{2}
-2\alpha\langle H_{t}^{-1}\nabla f(x_{t}),e_{t}\rangle
+\alpha^{2}\|H_{t}^{-1}\nabla f(x_{t})\|^{2}.
\tag{2}\label{eq:expansion}
\]
\textbf{[Step 1]} This is the elementary identity $\|a-b\|^{2}= \|a\|^{2}-2\langle a,b\rangle+\|b\|^{2}$ applied with $a=e_{t}$ and $b=\alpha H_{t}^{-1}\nabla f(x_{t})$.

\subsection*{Step 2 – Take conditional expectation}
Condition on $x_{t}$ (equivalently on $e_{t}$) and use the unbiasedness of $H_{t}$:
\[
\E\bigl[\langle H_{t}^{-1}\nabla f(x_{t}),e_{t}\rangle\mid x_{t}\bigr]
= \bigl\langle \E[H_{t}^{-1}\mid x_{t}]\nabla f(x_{t}),e_{t}\bigr\rangle .
\tag{3}
\]
\textbf{[Step 2]} Linearity of expectation and the fact that $e_{t}$ is $\sigma(x_{t})$‑measurable.

Because $H_{t}$ and $\nabla^{2}f(x_{t})$ share the same eigenvectors (both are PSD and bounded), Lemma~\ref{lem:inv} yields
\[
\frac{1}{L_{h}} I\preceq H_{t}^{-1}\preceq \frac{1}{\mu_{h}} I
\quad\Longrightarrow\quad
\frac{1}{L_{h}}\|\nabla f(x_{t})\|^{2}
\le \|H_{t}^{-1}\nabla f(x_{t})\|^{2}
\le \frac{1}{\mu_{h}^{2}}\|\nabla f(x_{t})\|^{2}.
\tag{4}\label{eq:invbound}
\]

\subsection*{Step 3 – Lower bound the cross term}
Using the mean‑value representation $\nabla f(x_{t})=\nabla^{2}f(\xi_{t})e_{t}$ and the spectral bounds $\mu I\preceq\nabla^{2}f(\xi_{t})\preceq L I$,
\[
\langle \nabla f(x_{t}),e_{t}\rangle
= e_{t}^{\!\top}\nabla^{2}f(\xi_{t})e_{t}
\ge \mu\|e_{t}\|^{2}.
\tag{5}
\]
\textbf{[Step 3]} Direct consequence of $\mu I\preceq\nabla^{2}f(\xi_{t})$.

Combining (5) with the upper bound from Lemma~\ref{lem:inv},
\[
\langle H_{t}^{-1}\nabla f(x_{t}),e_{t}\rangle
\ge \frac{1}{L_{h}}\langle \nabla f(x_{t}),e_{t}\rangle
\ge \frac{\mu}{L_{h}}\|e_{t}\|^{2}.
\tag{6}\label{eq:crosslb}
\]

\subsection*{Step 4 – Upper bound the quadratic term}
From (4) and Lemma~\ref{lem:strong} (right inequality) we have
\[
\| \nabla f(x_{t})\|^{2}
\le 2L\bigl(f(x_{t})-f(x^{\star})\bigr)
\le L^{2}\|e_{t}\|^{2},
\tag{7}
\]
where the second inequality follows from the left side of Lemma~\ref{lem:strong}:
$f(x_{t})-f(x^{\star})\le \frac{L}{2}\|e_{t}\|^{2}$.

Insert (7) into the right side of (4):
\[
\|H_{t}^{-1}\nabla f(x_{t})\|^{2}
\le \frac{L^{2}}{\mu_{h}^{2}}\|e_{t}\|^{2}.
\tag{8}\label{eq:quadub}
\]

\subsection*{Step 5 – Take total expectation}
Taking expectation of \eqref{eq:expansion} conditional on $x_{t}$ and then the full expectation yields
\[
\E\bigl[\|e_{t+1}\|^{2}\mid x_{t}\bigr]
\le \|e_{t}\|^{2}
-2\alpha\frac{\mu}{L_{h}}\|e_{t}\|^{2}
+\alpha^{2}\frac{L^{2}}{\mu_{h}^{2}}\|e_{t}\|^{2}.
\tag{9}
\]
\textbf{[Step 5]} Apply (6) to the cross term and (8) to the quadratic term.

Taking full expectation and noting that $\|e_{t}\|^{2}$ is $\sigma(x_{t})$‑measurable,
\[
\E\bigl[\|e_{t+1}\|^{2}\bigr]
\le \Bigl(1-2\alpha\frac{\mu}{L_{h}}+\alpha^{2}\frac{L^{2}}{\mu_{h}^{2}}\Bigr)
\E\bigl[\|e_{t}\|^{2}\bigr].
\tag{10}
\]

\subsection*{Step 6 – Choose stepsize to obtain a contraction}
Define the contraction factor
\[
\rho(\alpha)=1-2\alpha\frac{\mu}{L_{h}}+\alpha^{2}\frac{L^{2}}{\mu_{h}^{2}}.
\]
$\rho(\alpha)$ is a convex quadratic in $\alpha$. Its minimum is attained at
\[
\alpha^{\star}= \frac{\mu\mu_{h}^{2}}{L^{2}L_{h}}.
\]
Plugging $\alpha^{\star}$ gives
\[
\rho_{\min}=1-\frac{\mu^{2}\mu_{h}^{2}}{L^{2}L_{h}^{2}}
=1-\frac{\mu}{\kappa_{H}}\cdot\frac{\mu}{L}.
\]
Since $\mu\le L$, we obtain the simpler bound
\[
\rho(\alpha)\le 1-\alpha\frac{\mu}{\kappa_{H}}
\quad\text{for any}\quad 
0<\alpha\le\frac{2\mu_{h}}{L}.
\tag{11}\label{eq:rho}
\]
\textbf{[Step 6]} The inequality follows from expanding $(1-\alpha\mu/\kappa_{H})^{2}\ge0$ and using $\alpha\le 2\mu_{h}/L$.

\subsection*{Step 7 – Final recursion}
Combining (10) with (11) yields exactly the recursion \eqref{eq:linear} claimed in Theorem~\ref{thm:linear}:
\[
\E\bigl[\|e_{t+1}\|^{2}\bigr]\le\Bigl(1-\alpha\frac{\mu}{\kappa_{H}}\Bigr)\E\bigl[\|e_{t}\|^{2}\bigr].
\]
Iterating the inequality gives the geometric bound
\[
\E\bigl[\|e_{t}\|^{2}\bigr]\le\Bigl(1-\alpha\frac{\mu}{\kappa_{H}}\Bigr)^{t}\|e_{0}\|^{2}.
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
Define the linear convergence constant
\[
c:=\alpha\frac{\mu}{\kappa_{H}}
\quad\text{with}\quad
0<c<1.
\]
Then for any $t\ge0$,
\[
\E\!\bigl[f(x_{t})-f(x^{\star})\bigr]
\le \frac{L}{2}\E\!\bigl[\|e_{t}\|^{2}\bigr]
\le \frac{L}{2}\bigl(1-c\bigr)^{t}\|e_{0}\|^{2}.
\]
Thus the objective error decays at the same geometric rate as the squared distance.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Hessian ($H_{t}=\nabla^{2}f(x_{t})$).} Here $\mu_{h}=\mu$, $L_{h}=L$, $\kappa_{H}=L/\mu=\kappa$, and the optimal stepsize $\alpha=1$ recovers the classical Newton method with contraction factor $1-1/\kappa$.
\item \textbf{Noisy Hessian with bounded condition number.} If $H_{t}=\nabla^{2}f(x_{t})+E_{t}$ where $\|E_{t}\|\le \delta$ and $\delta$ is small enough to keep $\mu_{h}>0$, the same proof applies, showing robustness of SNM to moderate Hessian noise.
\end{itemize}

\section*{Discussion}
The proof shows that a *uniform* spectral bound on the stochastic Hessian estimate is sufficient to inherit the linear convergence of deterministic Newton’s method, albeit with a slowdown proportional to the condition number $\kappa_{H}$ of the stochastic Hessian. The stepsize restriction $\alpha\le 2\mu_{h}/L$ is mild; in practice one may use a diminishing stepsize to further mitigate variance when the Hessian estimate is noisy.
