\section*{Algorithm Name}
\textbf{Differentially Private Stochastic Gradient Descent (DP‑SGD) with Gaussian Noise}\\
(Non‑convex, $L$‑smooth objective)

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss.  
Given a stepsize sequence $\{\eta_t\}_{t\ge 0}$ and noise variance $\sigma^{2}>0$, DP‑SGD generates a sequence $\{x_t\}$ by  
\[
\boxed{
\begin{aligned}
g_t &:= \nabla f(x_t) + \xi_t,\\
\xi_t &\sim \mathcal{N}(0,\sigma^{2}I_d),\\
x_{t+1} &:= x_t - \eta_t\, g_t .
\end{aligned}}
\tag{1}
\]
The privacy accountant tracks the cumulative $(\varepsilon,\delta)$‑budget using the moments accountant (or R\'enyi DP) for the Gaussian mechanism; the exact bookkeeping is omitted here as it does not affect the convergence analysis.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^{d}$, stepsize $\eta>0$, noise std.\ $\sigma$, total iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute (full‑batch) gradient $\nabla f(x_t)$.
    \STATE Sample noise $\xi_t\sim\mathcal{N}(0,\sigma^{2}I_d)$.
    \STATE Form noisy gradient $g_t\gets \nabla f(x_t)+\xi_t$.
    \STATE Update $x_{t+1}\gets x_t-\eta\, g_t$.
    \STATE (Optional) update privacy accountant.
\ENDFOR
\STATE \textbf{Output:} $\{x_t\}_{t=0}^{T}$ (or the average $\bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Choose $w_0=0$, stepsize $\eta=0.1$, noise variance $\sigma^{2}=0.01$ (i.e. $\sigma=0.1$).  
We perform three iterations; the random draws of $\xi_t$ are shown for reproducibility.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $\nabla f(w_t)$ & $\xi_t$ (sample) & $w_{t+1}=w_t-\eta(\nabla f(w_t)+\xi_t)$ \\ \hline
0 & $2(0-3)=-6$ & $\xi_0=+0.05$ & $w_1=0-0.1(-6+0.05)=0.595$ \\
1 & $2(0.595-3)=-4.81$ & $\xi_1=-0.08$ & $w_2=0.595-0.1(-4.81-0.08)=1.083$ \\
2 & $2(1.083-3)=-3.834$ & $\xi_2=+0.12$ & $w_3=1.083-0.1(-3.834+0.12)=1.455$ \\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(w_0)=9,\; f(w_1)\approx 5.79,\; f(w_2)\approx 3.70,\; f(w_3)\approx 2.40.
\]
The example illustrates that noisy updates still make progress toward the minimiser $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]
There exists $G>0$ such that $\|\nabla f(x)\|\le G$ for all $x$ visited by the algorithm.
\tag{3}
\end{assumption}
\begin{assumption}[Stepsize]
The constant stepsize satisfies $0<\eta\le \frac{1}{L}$.
\tag{4}
\end{assumption}
\begin{assumption}[Noise]
The added noise is independent across iterations and satisfies $\E[\xi_t]=0$, $\E[\xi_t\xi_t^{\top}]=\sigma^{2}I_{d}$.
\tag{5}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Under Assumptions~1–4, for any $T\ge 1$ the iterates generated by (1) satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(x_0)-f^{*}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\big(G^{2}+\sigma^{2}d\big)}_{\text{variance term}} .
\tag{6}
\]
Consequently, choosing $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]=
O\!\Big(\frac{1}{\sqrt{T}}\Big)+O\!\big(\sigma^{2}d\,T^{-1/2}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound (6) shows that the algorithm is stable for any $\eta\le 1/L$; the term $L\eta\sigma^{2}d$ is the price paid for privacy.
    \item \textbf{Hyper‑parameter Sensitivity.} Increasing $\eta$ accelerates the decay of the optimisation term but inflates the variance term linearly. The optimal $\eta$ balances the two, giving $\eta^{\star}= \sqrt{\frac{2(f(x_0)-f^{*})}{L\big(G^{2}+\sigma^{2}d\big)T}}$.
    \item \textbf{Comparison to Non‑private SGD.} Setting $\sigma=0$ recovers the classic non‑convex SGD rate $\frac{2(f(x_0)-f^{*})}{\eta T}+L\eta G^{2}$.
    \item \textbf{Effect of Dimensionality.} The noise contribution scales linearly with $d$, a well‑known phenomenon in DP‑SGD.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that, despite the injected Gaussian noise required for differential privacy, DP‑SGD retains the same $O(1/\sqrt{T})$ decay of the average squared gradient norm as standard SGD for smooth non‑convex objectives. The extra variance term is unavoidable: it directly reflects the privacy‑induced noise magnitude $\sigma^{2}$ and the problem dimension $d$. In practice, one selects $\sigma$ according to the desired $(\varepsilon,\delta)$‑budget (via the moments accountant) and then tunes $\eta$ according to the trade‑off highlighted above.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]
\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz continuity of the gradient; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Quadratic Expansion of the Update]
\label{lem:expansion}
For the update $x_{t+1}=x_t-\eta g_t$ we have
\[
\|x_{t+1}-x^{*}\|^{2}
= \|x_t-x^{*}\|^{2}
-2\eta\langle g_t, x_t-x^{*}\rangle
+\eta^{2}\|g_t\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Direct expansion of the Euclidean norm. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, $x^{*}$ denotes a global minimiser (or any point with $f(x^{*})=f^{*}$).

\bigskip
\noindent\textbf{Step 1 (Apply smoothness).}  
Using Lemma~\ref{lem:smooth} with $y=x_{t+1}$ and $x=x_t$,
\[
f(x_{t+1})\le f(x_t)+\langle\nabla f(x_t),x_{t+1}-x_t\rangle+\frac{L}{2}\|x_{t+1}-x_t\|^{2}.
\tag{9}
\]
Since $x_{t+1}-x_t=-\eta g_t$, (9) becomes
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),g_t\rangle+\frac{L\eta^{2}}{2}\|g_t\|^{2}.
\tag{10}
\]

\noindent\textbf{Step 2 (Insert noisy gradient).}  
Recall $g_t=\nabla f(x_t)+\xi_t$. Hence
\[
\langle\nabla f(x_t),g_t\rangle
= \|\nabla f(x_t)\|^{2}+\langle\nabla f(x_t),\xi_t\rangle,
\qquad
\|g_t\|^{2}
= \|\nabla f(x_t)\|^{2}+2\langle\nabla f(x_t),\xi_t\rangle+\|\xi_t\|^{2}.
\tag{11}
\]
Plugging (11) into (10) yields
\[
\begin{aligned}
f(x_{t+1})\le &\, f(x_t)-\eta\|\nabla f(x_t)\|^{2}
-\eta\langle\nabla f(x_t),\xi_t\rangle\\
&\;+\frac{L\eta^{2}}{2}\Big(\|\nabla f(x_t)\|^{2}
+2\langle\nabla f(x_t),\xi_t\rangle+\|\xi_t\|^{2}\Big).
\end{aligned}
\tag{12}
\]

\noindent\textbf{Step 3 (Take expectation w.r.t. noise).}  
Condition on $x_t$ and use $\E[\xi_t]=0$, $\E[\|\xi_t\|^{2}]=\sigma^{2}d$:
\[
\begin{aligned}
\E_{\xi_t}\!\big[f(x_{t+1})\mid x_t\big]
\le &\, f(x_t)-\eta\|\nabla f(x_t)\|^{2}
+\frac{L\eta^{2}}{2}\|\nabla f(x_t)\|^{2}
+\frac{L\eta^{2}}{2}\sigma^{2}d .
\end{aligned}
\tag{13}
\]
(The cross terms $\langle\nabla f(x_t),\xi_t\rangle$ vanish by unbiasedness.)

\noindent\textbf{Step 4 (Rearrange).}  
Move the gradient term to the left:
\[
\eta\Big(1-\frac{L\eta}{2}\Big)\|\nabla f(x_t)\|^{2}
\le f(x_t)-\E_{\xi_t}[f(x_{t+1})\mid x_t]
+\frac{L\eta^{2}}{2}\sigma^{2}d .
\tag{14}
\]

\noindent\textbf{Step 5 (Use stepsize bound).}  
Assumption~(4) gives $0<\eta\le 1/L$, hence $1-\frac{L\eta}{2}\ge \frac12$. Multiplying (14) by $2$,
\[
\eta\|\nabla f(x_t)\|^{2}
\le 2\!\big(f(x_t)-\E_{\xi_t}[f(x_{t+1})\mid x_t]\big)
+L\eta^{2}\sigma^{2}d .
\tag{15}
\]

\noindent\textbf{Step 6 (Take full expectation).}  
Denote the total expectation by $\E[\cdot]$. Since the RHS of (15) depends only on $x_t$, we can drop the conditioning:
\[
\eta\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le 2\!\big(\E[f(x_t)]-\E[f(x_{t+1})]\big)
+L\eta^{2}\sigma^{2}d .
\tag{16}
\]

\noindent\textbf{Step 7 (Sum over $t=0,\dots,T-1$).}  
Summing (16) telescopes the objective terms:
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le 2\big(f(x_0)-\E[f(x_T)]\big)
+L\eta^{2}\sigma^{2}d\,T .
\tag{17}
\]

\noindent\textbf{Step 8 (Upper‑bound $f(x_T)$).}  
Since $f^{*}\le f(x_T)$, we replace $\E[f(x_T)]$ by $f^{*}$ to obtain a looser but cleaner bound:
\[
\eta\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le 2\big(f(x_0)-f^{*}\big)
+L\eta^{2}\sigma^{2}d\,T .
\tag{18}
\]

\noindent\textbf{Step 9 (Divide by $\eta T$).}  
Dividing (18) by $\eta T$ gives the average gradient norm bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le \frac{2\big(f(x_0)-f^{*}\big)}{\eta T}
+L\eta\sigma^{2}d .
\tag{19}
\]

\noindent\textbf{Step 10 (Insert bounded true gradient).}  
To incorporate the deterministic gradient bound $G$ from Assumption~(3), we observe that $\|\nabla f(x_t)\|^{2}\le G^{2}$, thus the variance term in (19) can be enlarged to $L\eta(G^{2}+\sigma^{2}d)$ without affecting the inequality:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le \frac{2\big(f(x_0)-f^{*}\big)}{\eta T}
+L\eta\big(G^{2}+\sigma^{2}d\big).
\tag{20}
\]
Equation (20) is exactly the statement (6) of Theorem~\ref{thm:main}. $\square$

\section*{Rate Derivation (With Constants)}
From (20) choose $\eta = \sqrt{\frac{2(f(x_0)-f^{*})}{L\big(G^{2}+\sigma^{2}d\big)T}}$, which satisfies $\eta\le 1/L$ for all sufficiently large $T$. Substituting,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
\le 2\sqrt{\frac{L\big(G^{2}+\sigma^{2}d\big)\big(f(x_0)-f^{*}\big)}{T}}
=O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
Thus the algorithm attains the optimal $O(T^{-1/2})$ rate up to the multiplicative factor $\sqrt{G^{2}+\sigma^{2}d}$.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No privacy noise ($\sigma=0$).} Bound (20) reduces to the classical non‑convex SGD rate
    $\frac{2(f(x_0)-f^{*})}{\eta T}+L\eta G^{2}$.
    \item \textbf{Very small stepsize $\eta\to 0$.} The optimisation term dominates, yielding a $O(1/(\eta T))$ decay but the variance term vanishes.
    \item \textbf{High dimensionality.} If $d$ is large, the term $L\eta\sigma^{2}d$ may dominate; in practice one reduces $\sigma$ (i.e. spends a larger privacy budget) or employs gradient clipping to keep $G$ small.
\end{itemize}
