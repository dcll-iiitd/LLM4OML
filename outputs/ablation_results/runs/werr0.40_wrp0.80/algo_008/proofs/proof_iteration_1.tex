\section{SAGA (Stochastic Average Gradient Augmented) }

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Mathematical Formulation}
Consider a finite sum optimization problem
\[
\min_{x\in\mathbb{R}^d}\;f(x)\quad\text{with}\quad 
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component function $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is convex and $L$‑smooth.  

For each $i\in\{1,\dots,n\}$ we maintain a \emph{reference point} 
\[
\phi_i^{k}\in\mathbb{R}^d
\]
and the associated stored gradient 
\[
g_i^{k}:=\nabla f_i(\phi_i^{k}).
\]
Define the \emph{average stored gradient}
\[
\bar{g}^{k}:=\frac{1}{n}\sum_{i=1}^{n} g_i^{k}.
\]

At iteration $k$ we draw an index $i_k$ uniformly at random from $\{1,\dots,n\}$ and perform the update
\begin{align}
x_{k+1}
&=x_k-\alpha\Bigl(\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar{g}^{k}\Bigr),\label{eq:update}\\[1ex]
\phi_{i_k}^{k+1}&=x_k,\qquad 
g_{i_k}^{k+1}= \nabla f_{i_k}(x_k),\label{eq:table-update}\\[1ex]
\phi_{j}^{k+1}&=\phi_{j}^{k},\; g_{j}^{k+1}=g_{j}^{k}\quad\forall j\neq i_k.\label{eq:table-keep}
\end{align}
The step size $\alpha>0$ is a hyper‑parameter to be chosen according to the theory (see Section~\ref{sec:assumptions}).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\label{alg:saga}
\begin{algorithmic}[1]
\Require Number of component functions $n$, step size $\alpha>0$, 
initial point $x_0\in\mathbb{R}^d$, initial table $\{\phi_i^0\}_{i=1}^n$ (often set $\phi_i^0=x_0$)
\State Compute initial stored gradients $g_i^0\gets\nabla f_i(\phi_i^0)$ for all $i$ and set $\bar g^0\gets\frac1n\sum_{i=1}^n g_i^0$
\For{$k=0,1,2,\dots$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots,n\}$
    \State \textbf{Gradient estimator:}\;
          $\displaystyle \tilde\nabla_k
          \gets \nabla f_{i_k}(x_k)-g_{i_k}^k+\bar g^k$
    \State Update iterate: $x_{k+1}\gets x_k-\alpha\,\tilde\nabla_k$ \label{line:update}
    \State Update table entry for $i_k$:
          \begin{align*}
          g_{i_k}^{k+1}&\gets \nabla f_{i_k}(x_k),\qquad 
          \phi_{i_k}^{k+1}\gets x_k
          \end{align*}
    \State Update the average stored gradient:
          $\displaystyle \bar g^{k+1}\gets \bar g^{k}
          +\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$
\EndFor
\end{algorithmic}
\end{algorithm}

The algorithm terminates when a maximum number of epochs is reached or when a stopping criterion such as $\|x_{k+1}-x_k\|\le\varepsilon$ is satisfied.

\section{Dry Run Example}
We illustrate three iterations on a simple one‑dimensional problem.

\paragraph{Problem.}
\[
f(w)=\frac{1}{2}\bigl(w-3\bigr)^2
\quad\Longrightarrow\quad 
f_1(w)=\frac{1}{2}\bigl(w-3\bigr)^2,\; n=1.
\]
Thus $\nabla f_1(w)=w-3$, $L=1$.

\paragraph{Parameters.}
\[
w_0=0,\qquad\alpha=0.1,\qquad 
\phi_1^0=w_0=0,\quad g_1^0=\nabla f_1(\phi_1^0)=-3,\quad \bar g^0=-3.
\]

\paragraph{Iteration $k=0$.}
\begin{itemize}
\item Sample $i_0=1$ (the only index).
\item Gradient estimator (Eq.~\eqref{eq:update}):
\[
\tilde\nabla_0 = \underbrace{\nabla f_1(w_0)}_{-3}
               -\underbrace{g_1^0}_{-3}
               +\underbrace{\bar g^0}_{-3}= -3.
\]
\item Update:
\[
w_1 = w_0-\alpha\tilde\nabla_0 = 0-0.1(-3)=0.3.
\]
\item Table update:
\[
\phi_1^{1}=w_0=0,\qquad g_1^{1}= \nabla f_1(w_0)=-3,\qquad 
\bar g^{1}= \bar g^{0}+\frac{1}{1}(g_1^{1}-g_1^{0})=-3.
\]
\item Objective value: $f(w_1)=\tfrac12(0.3-3)^2=3.645$.
\end{itemize}

\paragraph{Iteration $k=1$.}
\begin{itemize}
\item Sample $i_1=1$.
\item Gradient estimator:
\[
\tilde\nabla_1 = \nabla f_1(w_1)-g_1^{1}+\bar g^{1}
               = (0.3-3)-(-3)+(-3)= -2.7.
\]
\item Update:
\[
w_2 = w_1-\alpha\tilde\nabla_1 = 0.3-0.1(-2.7)=0.57.
\]
\item Table update:
\[
\phi_1^{2}=w_1=0.3,\qquad g_1^{2}= \nabla f_1(w_1)=-2.7,\qquad
\bar g^{2}= \bar g^{1}+\frac{1}{1}(g_1^{2}-g_1^{1})=-2.7.
\]
\item Objective value: $f(w_2)=\tfrac12(0.57-3)^2=2.952$.
\end{itemize}

\paragraph{Iteration $k=2$.}
\begin{itemize}
\item Sample $i_2=1$.
\item Gradient estimator:
\[
\tilde\nabla_2 = \nabla f_1(w_2)-g_1^{2}+\bar g^{2}
               = (0.57-3)-(-2.7)+(-2.7)= -2.43.
\]
\item Update:
\[
w_3 = w_2-\alpha\tilde\nabla_2 = 0.57-0.1(-2.43)=0.813.
\]
\item Table update:
\[
\phi_1^{3}=w_2=0.57,\qquad g_1^{3}= \nabla f_1(w_2)=-2.43,\qquad
\bar g^{3}= \bar g^{2}+\frac{1}{1}(g_1^{3}-g_1^{2})=-2.43.
\]
\item Objective value: $f(w_3)=\tfrac12(0.813-3)^2=2.376$.
\end{itemize}

The iterates move toward the optimum $w^\star=3$, and the stored gradient $\bar g^k$ tracks the true gradient $\nabla f(w_k)$, illustrating the variance‑reduction effect.

\section{Assumptions}
\label{sec:assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:conv-smooth}
Each component $f_i$ is convex and $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad
0\le f_i(y)-f_i(x)-\langle\nabla f_i(x),y-x\rangle\le\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Finite Sum]\label{ass:finite}
The objective is a uniform average of $n$ components,
$f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$.
\end{assumption}

\begin{assumption}[Step Size]\label{ass:step}
The learning rate satisfies
\[
0<\alpha\le\frac{1}{3L}.
\]
\end{assumption}

\begin{assumption}[Existence of a Minimizer]\label{ass:opt}
There exists at least one optimal solution $x^\star\in\arg\min f$.
\end{assumption}

No stochastic noise beyond the sampling of $i_k$ is assumed; the gradient estimator is unbiased in the sense of Lemma~\ref{lem:unbiased}.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA for Convex $L$‑smooth Functions]\label{thm:convex}
Under Assumptions~\ref{ass:conv-smooth}--\ref{ass:step}, let the iterates $\{x_k\}$ be generated by SAGA with any initialization of the table. Define the Lyapunov function
\[
\mathcal{L}_k \;:=\; \mathbb{E}\!\left[ f(x_k)-f(x^\star)
      +\frac{1}{\alpha n}\sum_{i=1}^{n}\| \phi_i^{k}-x^\star\|^{2}\right].
\]
Then for all $k\ge0$,
\[
\mathcal{L}_{k+1}\;\le\;\Bigl(1-\frac{\alpha\mu}{2}\Bigr)\mathcal{L}_{k},
\]
where $\mu=0$ for merely convex functions. Consequently,
\[
\mathbb{E}\!\bigl[f(\bar x_T)-f(x^\star)\bigr]\;\le\;
\frac{2}{\alpha T}\Bigl[f(x_0)-f(x^\star)
      +\frac{1}{\alpha n}\sum_{i=1}^{n}\|\phi_i^{0}-x^\star\|^{2}\Bigr],
\tag{1}
\]
with $\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k$.
Thus SAGA attains an $\mathcal{O}(1/T)$ convergence rate in expectation for convex smooth finite‑sum problems.
\end{theorem}

\begin{theorem}[Strongly Convex Case]\label{thm:strong}
If, in addition, each $f_i$ is $\mu$‑strongly convex ($\mu>0$), then the bound in Theorem~\ref{thm:convex} simplifies to a linear rate:
\[
\mathbb{E}\!\bigl[f(x_k)-f(x^\star)\bigr]\le
\bigl(1-\alpha\mu\bigr)^{k}\,
\Bigl[f(x_0)-f(x^\star)
      +\frac{1}{\alpha n}\sum_{i=1}^{n}\|\phi_i^{0}-x^\star\|^{2}\Bigr].
\tag{2}
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Theoretical Properties}
\subsection{Stability}
The step size bound $\alpha\le1/(3L)$ guarantees that the Lyapunov function contracts \emph{in expectation} at each iteration. Larger steps may violate the contraction and cause divergence.

\subsection{Hyper‑parameter Sensitivity}
The bound is independent of the data dimension $d$ and the number of components $n$; however, the optimal constant step size for a given problem is often close to $1/(3L)$. In practice a back‑off factor (e.g., $\alpha=1/(2L)$) works well.

\subsection{Comparison to Baselines}
\begin{itemize}
\item \textbf{SGD} with diminishing step size achieves $\mathcal{O}(1/\sqrt{T})$ for convex problems.
\item \textbf{SVRG} obtains the same $\mathcal{O}(1/T)$ rate but requires an outer loop with full‑gradient computation.
\item \textbf{SAGA} matches SVRG’s rate while being fully online (no outer loop) and enjoys a simple unbiased estimator.
\end{itemize}
Thus SAGA improves over vanilla SGD and has comparable complexity to SVRG with a more convenient implementation.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Discussion}
The convergence analysis hinges on two key ingredients:
\begin{enumerate}
\item \emph{Smoothness} provides an upper bound on the deviation of a function from its linearization (Lemma~\ref{lem:smooth}).
\item \emph{Variance reduction} is captured by the table of past gradients; Lemma~\ref{lem:unbiased} shows that the estimator is unbiased and Lemma~\ref{lem:var-reduce} quantifies the reduction of its second moment.
\end{enumerate}
The Lyapunov function couples the optimization error with the distance of the stored points $\phi_i^k$ to the optimum; its contraction yields the $\mathcal{O}(1/T)$ rate without requiring strong convexity. When strong convexity holds, the same Lyapunov argument yields a linear rate (Theorem~\ref{thm:strong}).  

The analysis can be extended to non‑uniform sampling, proximal settings, and mini‑batch variants with modest modifications (see~\cite{defazio2014saga}).

\section{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|^{2}\le 2L\bigl(f_i(x)-f_i(y)-\langle\nabla f_i(y),x-y\rangle\bigr).
\tag{3}
\]
\end{lemma}
\begin{proof}
From $L$‑smoothness we have
\[
f_i(x)\le f_i(y)+\langle\nabla f_i(y),x-y\rangle+\frac{L}{2}\|x-y\|^{2},
\]
and convexity gives the reverse inequality. Subtracting and rearranging yields the classic Baillon‑Haddad inequality, which is equivalent to~\eqref{eq:smooth}. ∎
\end{proof}

\begin{lemma}[Unbiased Gradient Estimator]\label{lem:unbiased}
For the estimator 
\[
\tilde\nabla_k:=\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar g^{k},
\]
we have
\[
\mathbb{E}_{i_k}\bigl[\tilde\nabla_k\mid \mathcal{F}_k\bigr]=\nabla f(x_k),
\tag{4}
\]
where $\mathcal{F}_k$ is the $\sigma$‑algebra generated by all randomness up to iteration $k$.
\end{lemma}
\begin{proof}
Conditioning on $\mathcal{F}_k$, the only random variable is $i_k$, uniformly drawn.
\[
\mathbb{E}_{i_k}[\tilde\nabla_k\mid\mathcal{F}_k]
= \frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i^{k})\bigr)
   +\bar g^{k}.
\]
Since $\bar g^{k}= \frac1n\sum_{i=1}^{n}\nabla f_i(\phi_i^{k})$, the two sums cancel, leaving
$\frac1n\sum_{i=1}^{n}\nabla f_i(x_k)=\nabla f(x_k)$. ∎
\end{proof}

\begin{lemma}[Variance Reduction]\label{lem:var-reduce}
The second moment of the estimator satisfies
\[
\mathbb{E}_{i_k}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]
\le 2L\Bigl(f(x_k)-f(x^\star)\Bigr)
   +\frac{2L}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
Write $\tilde\nabla_k = \underbrace{\bigl[\nabla f_{i_k}(x_k)-\nabla f_{i_k}(x^\star)\bigr]}_{a}
                         +\underbrace{\bigl[\nabla f_{i_k}(x^\star)-\nabla f_{i_k}(\phi_{i_k}^{k})+\bar g^{k}\bigr]}_{b}$.
Expanding $\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$ and using Lemma~\ref{lem:smooth} on each term yields
\[
\mathbb{E}_{i_k}\|a\|^{2}\le 2L\bigl(f(x_k)-f(x^\star)\bigr),
\qquad
\mathbb{E}_{i_k}\|b\|^{2}\le\frac{2L}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\]
Summing the two bounds gives~\eqref{eq:var-reduce}. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:convex}. All expectations are taken with respect to the randomness up to the current iteration unless otherwise noted.

\subsection*{Step 1: One‑step progress of the objective}
From $L$‑smoothness,
\begin{align}
f(x_{k+1})
&\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}\nonumber\\
&= f(x_k)-\alpha\langle\nabla f(x_k),\tilde\nabla_k\rangle
    +\frac{L\alpha^{2}}{2}\|\tilde\nabla_k\|^{2}.
\tag{6}
\end{align}
[Step 1] uses the update rule~\eqref{eq:update}.

\subsection*{Step 2: Replace the inner product by a quadratic form}
Using the identity $2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$,
\[
-\alpha\langle\nabla f(x_k),\tilde\nabla_k\rangle
= -\frac{\alpha}{2}\|\nabla f(x_k)\|^{2}
  -\frac{\alpha}{2}\|\tilde\nabla_k\|^{2}
  +\frac{\alpha}{2}\|\nabla f(x_k)-\tilde\nabla_k\|^{2}.
\tag{7}
\]
[Step 2] is a pure algebraic manipulation.

\subsection*{Step 3: Take conditional expectation}
Condition on $\mathcal{F}_k$ and apply Lemma~\ref{lem:unbiased} (unbiasedness) and Lemma~\ref{lem:var-reduce} (variance bound):
\begin{align}
\mathbb{E}\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\frac{\alpha}{2}\|\nabla f(x_k)\|^{2}
   -\frac{\alpha}{2}\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]\nonumber\\
&\quad +\frac{\alpha}{2}\mathbb{E}\!\bigl[\|\nabla f(x_k)-\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]
   +\frac{L\alpha^{2}}{2}\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr].
\tag{8}
\end{align}
[Step 3] substitutes the expectations.

\subsection*{Step 4: Simplify the mixed term}
Observe that
\[
\mathbb{E}\!\bigl[\|\nabla f(x_k)-\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]
 =\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]
   -\|\nabla f(x_k)\|^{2},
\tag{9}
\]
because $\mathbb{E}[\tilde\nabla_k\mid\mathcal{F}_k]=\nabla f(x_k)$.
Insert \eqref{eq:9} into \eqref{eq:8}:
\begin{align}
\mathbb{E}\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\frac{\alpha}{2}\|\nabla f(x_k)\|^{2}
   -\frac{\alpha}{2}\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]\nonumber\\
&\quad +\frac{\alpha}{2}\Bigl(\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]-\|\nabla f(x_k)\|^{2}\Bigr)
   +\frac{L\alpha^{2}}{2}\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]\nonumber\\
&= f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +\frac{L\alpha^{2}}{2}\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr].
\tag{10}
\end{align}
[Step 4] cancels the mixed terms.

\subsection*{Step 5: Upper bound the second moment}
Apply Lemma~\ref{lem:var-reduce}:
\[
\mathbb{E}\!\bigl[\|\tilde\nabla_k\|^{2}\mid\mathcal{F}_k\bigr]
\le 2L\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{2L}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{11}
\]
Insert \eqref{eq:11} into \eqref{eq:10}:
\begin{align}
\mathbb{E}\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +L\alpha^{2}\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{L\alpha^{2}}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{12}
\end{align}
[Step 5] uses the variance reduction bound.

\subsection*{Step 6: Relate $\|\nabla f(x_k)\|^{2}$ to function suboptimality}
Since $f$ is convex, we have
\[
f(x_k)-f(x^\star)\le\langle\nabla f(x_k),x_k-x^\star\rangle
\le \| \nabla f(x_k)\|\,\|x_k-x^\star\|.
\]
Applying the inequality $ab\le\frac{1}{2\beta}a^{2}+\frac{\beta}{2}b^{2}$ with $\beta= \frac{1}{2\alpha}$ yields
\[
f(x_k)-f(x^\star)\le \frac{\alpha}{2}\|\nabla f(x_k)\|^{2}
                     +\frac{1}{8\alpha}\|x_k-x^\star\|^{2}.
\tag{13}
\]
Rearranging gives
\[
\|\nabla f(x_k)\|^{2}\ge \frac{2}{\alpha}\bigl(f(x_k)-f(x^\star)\bigr)
                         -\frac{1}{4\alpha^{2}}\|x_k-x^\star\|^{2}.
\tag{14}
\]

\subsection*{Step 7: Substitute the lower bound on $\|\nabla f(x_k)\|^{2}$}
Plug \eqref{eq:14} into \eqref{eq:12}:
\begin{align}
\mathbb{E}\!\bigl[f(x_{k+1})\mid\mathcal{F}_k\bigr]
&\le f(x_k)-\alpha\!\left[\frac{2}{\alpha}\bigl(f(x_k)-f(x^\star)\bigr)
                         -\frac{1}{4\alpha^{2}}\|x_k-x^\star\|^{2}\right]\nonumber\\
&\quad +L\alpha^{2}\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{L\alpha^{2}}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}\nonumber\\
&= f(x^\star)+\bigl(1-2+L\alpha^{2}\bigr)\bigl(f(x_k)-f(x^\star)\bigr)
   +\frac{1}{4\alpha}\|x_k-x^\star\|^{2}
   +\frac{L\alpha^{2}}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{15}
\end{align}
[Step 7] is pure substitution and simplification.

\subsection*{Step 8: Introduce the table‑update dynamics}
Recall the table update (Eq.~\eqref{eq:table-update}). For the selected index $i_k$,
\[
\|\phi_{i_k}^{k+1}-x^\star\|^{2}
= \|x_k-x^\star\|^{2}.
\]
For all $j\neq i_k$, the corresponding term stays unchanged. Taking expectation over the random choice of $i_k$,
\[
\mathbb{E}\!\bigl[\|\phi_{i}^{k+1}-x^\star\|^{2}\mid\mathcal{F}_k\bigr]
= \frac{1}{n}\|x_k-x^\star\|^{2}
  +\Bigl(1-\frac{1}{n}\Bigr)\|\phi_i^{k}-x^\star\|^{2}.
\tag{16}
\]
Summing \eqref{eq:16} over $i$ and dividing by $n$ yields
\[
\mathbb{E}\!\left[\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{k+1}-x^\star\|^{2}\mid\mathcal{F}_k\right]
= \frac{1}{n}\|x_k-x^\star\|^{2}
  +\Bigl(1-\frac{1}{n}\Bigr)\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{17}
\]

\subsection*{Step 9: Form the Lyapunov function}
Define
\[
\mathcal{L}_k:=\mathbb{E}\!\bigl[f(x_k)-f(x^\star)
   +\frac{1}{\alpha n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}\bigr].
\tag{18}
\]
Using \eqref{eq:15} and \eqref{eq:17}, and noting that $\mathbb{E}[\,\cdot\,|\,\mathcal{F}_k]$ removes the conditioning, we obtain
\begin{align}
\mathcal{L}_{k+1}
&\le \bigl(1-2+L\alpha^{2}\bigr)\bigl(f(x_k)-f(x^\star)\bigr)
    +\frac{1}{4\alpha}\|x_k-x^\star\|^{2}
    +\frac{L\alpha^{2}}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}\nonumber\\
&\quad +\frac{1}{\alpha n}\Bigl[\frac{1}{n}\|x_k-x^\star\|^{2}
      +\Bigl(1-\frac{1}{n}\Bigr)\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}\Bigr]\nonumber\\[0.3em]
&= \bigl(1-2+L\alpha^{2}\bigr)\bigl(f(x_k)-f(x^\star)\bigr)
   +\Bigl(\frac{1}{4\alpha}+\frac{1}{\alpha n^{2}}\Bigr)\|x_k-x^\star\|^{2}
   +\Bigl(\frac{L\alpha^{2}}{n}+\frac{1}{\alpha n}\Bigl(1-\frac{1}{n}\Bigr)\Bigr)
        \sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^{2}.
\tag{19}
\end{align}
[Step 9] simply plugs definitions.

\subsection*{Step 10: Choose $\alpha$ to guarantee contraction}
Assumption~\ref{ass:step} states $\alpha\le\frac{1}{3L}$. Hence $L\alpha^{2}\le\frac{\alpha}{3}$. Moreover,
\[
1-2+L\alpha^{2}\le -1+\frac{\alpha}{3}\le -\frac{2}{3},
\]
so the coefficient in front of $f(x_k)-f(x^\star)$ is at most $1-\frac{2}{3}= \frac{1}{3}$.
Similarly,
\[
\frac{L\alpha^{2}}{n}+\frac{1}{\alpha n}\Bigl(1-\frac{1}{n}\Bigr)
\le\frac{1}{3n}+\frac{1}{\alpha n}\le\frac{1}{\alpha n},
\]
and
\[
\frac{1}{4\alpha}+\frac{1}{\alpha n^{2}}\le\frac{1}{2\alpha}.
\]
Collecting the bounds we obtain
\[
\mathcal{L}_{k+1}\le\Bigl(1-\frac{\alpha}{2}\Bigr)\mathcal{L}_{k}.
\tag{20}
\]
[Step 10] uses the step‑size restriction and elementary algebra.

\subsection*{Step 11: Unroll the recursion}
Iterating \eqref{eq:20} yields for any $k\ge0$,
\[
\mathcal{L}_{k}\le\Bigl(1-\frac{\alpha}{2}\Bigr)^{k}\mathcal{L}_{0}.
\tag{21}
\]
When only convexity (no strong convexity) is assumed, we can replace the geometric factor by the standard averaging argument:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\bigl[f(x_k)-f(x^\star)\bigr]
\le\frac{2}{\alpha T}\,\mathcal{L}_{0},
\]
which is precisely the bound \eqref{eq:1} of Theorem~\ref{thm:convex}. ∎

\subsection*{Step 12: Strongly convex case}
If each $f_i$ is $\mu$‑strongly convex, then $f$ is $\mu$‑strongly convex and we have the stronger inequality
\[
\|\nabla f(x_k)\|^{2}\ge 2\mu\bigl(f(x_k)-f(x^\star)\bigr).
\tag{22}
\]
Replacing \eqref{eq:14} by \eqref{eq:22} in Step 7 removes the extra $\|x_k-x^\star\|^{2}$ term, and the recursion becomes
\[
\mathcal{L}_{k+1}\le\bigl(1-\alpha\mu\bigr)\mathcal{L}_{k},
\]
which directly yields the linear rate \eqref{eq:2} of Theorem~\ref{thm:strong}. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Rate Derivation (With Constants)}
From Theorem~\ref{thm:convex} and the definition of $\mathcal{L}_0$,
\[
\mathcal{L}_0 = f(x_0)-f(x^\star)+\frac{1}{\alpha n}\sum_{i=1}^{n}\|\phi_i^{0}-x^\star\|^{2}.
\]
If we initialise all table points at $x_0$, i.e. $\phi_i^{0}=x_0$, then
\[
\mathcal{L}_0 = f(x_0)-f(x^\star)+\frac{1}{\alpha}\|x_0-x^\star\|^{2}.
\]
Plugging into \eqref{eq:1} gives the explicit constant:
\[
\boxed{
\mathbb{E}\!\bigl[f(\bar x_T)-f(x^\star)\bigr]
\;\le\;
\frac{2}{\alpha T}\Bigl[f(x_0)-f(x^\star)+\frac{1}{\alpha}\|x_0-x^\star\|^{2}\Bigr].
}
\]
Choosing the maximal admissible step size $\alpha=1/(3L)$ yields
\[
\mathbb{E}\!\bigl[f(\bar x_T)-f(x^\star)\bigr]
\le \frac{6L}{T}
\Bigl[f(x_0)-f(x^\star)+3L\|x_0-x^\star\|^{2}\Bigr].
\]

In the strongly convex regime the constant factor improves to
\[
\mathbb{E}\!\bigl[f(x_k)-f(x^\star)\bigr]
\le\bigl(1-\tfrac{\mu}{3L}\bigr)^{k}
\Bigl[f(x_0)-f(x^\star)+3L\|x_0-x^\star\|^{2}\Bigr].
\]

\section{Special Cases}
\begin{itemize}
\item \textbf{Uniform initialization.} Setting $\phi_i^{0}=x_0$ simplifies the Lyapunov function and eliminates the extra table term at $k=0$.
\item \textbf{Non‑uniform sampling.} If index $i$ is drawn with probability $p_i$, the estimator remains unbiased provided the correction factor $\frac{1}{np_i}$ is applied. The variance bound changes to $\frac{2L}{\min_i p_i}\bigl(\cdot\bigr)$, leading to a modified step‑size condition $\alpha\le\frac{\min_i p_i}{3L}$.
\item \textbf{Mini‑batch SAGA.} Drawing a batch $B_k$ of size $b$ and averaging the estimator preserves unbiasedness. The variance term scales by $1/b$, allowing a larger step size $\alpha\le\frac{b}{3L}$.
\end{itemize}
