\section*{Algorithm Details Expansion}

\section{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be continuously differentiable.
Denote by $\nabla_i f(\mathbf w)$ the $i$-th partial derivative and by
$\mathbf e_i$ the $i$-th canonical basis vector.
At iteration $k$ a coordinate $i_k\in\{1,\dots ,d\}$ is drawn
independently from a fixed distribution 
\[
\mathbb P(i_k = i)=p_i,\qquad p_i>0,\;\sum_{i=1}^d p_i =1 .
\]

For each coordinate we keep a \emph{momentum accumulator}
\[
v_i^{k}= \begin{cases}
\beta v_i^{k-1} + (1-\beta)\,\nabla_i f(\mathbf w^{k}) , & \text{if } i=i_k,\\[4pt]
v_i^{k-1}, & \text{otherwise},
\end{cases}
\qquad 0\le \beta<1 .
\]

Only the selected coordinate is updated:
\[
\boxed{
\mathbf w^{k+1}= \mathbf w^{k}
- \eta_{i_k}\, v_{i_k}^{k}\,\mathbf e_{i_k}},
\qquad \eta_{i}>0\;\text{step size for coordinate }i .
\]

All other components remain unchanged:
$w^{k+1}_j = w^{k}_j$ for $j\neq i_k$.
The algorithm stops after $K$ iterations or when a prescribed
tolerance on $\| \nabla f(\mathbf w^k)\|$ is reached.

\section{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Momentum Coordinate Descent (PMCD)}
\begin{algorithmic}[1]
\Require Initial point $\mathbf w^{0}\in\mathbb R^{d}$,
        momentum vectors $\mathbf v^{0}= \mathbf 0$,
        probabilities $\{p_i\}_{i=1}^d$,
        step sizes $\{\eta_i\}_{i=1}^d$,
        momentum factor $\beta\in[0,1)$,
        maximum iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Sample $i_k\sim\{p_i\}_{i=1}^d$ \Comment{coordinate selection}
    \State $g_i \gets \nabla_{i_k} f(\mathbf w^{k})$ \Comment{partial gradient}
    \State $v_{i_k}^{k} \gets \beta\,v_{i_k}^{k-1} + (1-\beta)\,g_i$ \Comment{momentum update}
    \State $\mathbf w^{k+1} \gets \mathbf w^{k}$ 
    \State \hspace{1.8cm}$ - \eta_{i_k}\,v_{i_k}^{k}\,\mathbf e_{i_k}$ \Comment{coordinate update}
\EndFor
\State \Return $\mathbf w^{K}$
\end{algorithmic}
\end{algorithm}

\section{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w) = (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $d=1$, $p_1=1$, $\beta=0.5$, $\eta_1=0.1$, and start from $w^{0}=0$.
The momentum accumulator $v^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $w^{k}$ & $\nabla f(w^{k})$ & $v^{k}$ (after update) & $f(w^{k})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $v^{1}=0.5\cdot0+(1-0.5)(-6)=-3$ & $9$\\
1 & $w^{1}=0-0.1\cdot(-3)=0.3$ & $2(0.3-3)=-5.4$ & $v^{2}=0.5(-3)+(1-0.5)(-5.4)=-4.2$ & $7.29$\\
2 & $w^{2}=0.3-0.1\cdot(-4.2)=0.72$ & $2(0.72-3)=-4.56$ & $v^{3}=0.5(-4.2)+(1-0.5)(-4.56)=-4.38$ & $5.1984$\\
\end{tabular}
\end{center}
After three iterations the objective has dropped from $9$ to $5.1984$,
illustrating the effect of the momentum‑augmented coordinate update.

\section*{Convergence Theory Development}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section{Assumptions}
\begin{enumerate}[label={\bf A\arabic*:}, leftmargin=*, itemsep=2pt]
\item \textbf{Strong convexity.}
$f$ is $\mu$‑strongly convex, i.e.
\[
f(\mathbf y) \ge f(\mathbf x)+\langle\nabla f(\mathbf x),\mathbf y-\mathbf x\rangle
+\frac{\mu}{2}\|\mathbf y-\mathbf x\|^{2},
\qquad\forall\mathbf x,\mathbf y\in\mathbb R^{d},
\]
with $\mu>0$.
\item \textbf{Coordinate‑wise $L_i$‑smoothness.}
For each $i\in\{1,\dots ,d\}$,
\[
\bigl|\nabla_i f(\mathbf x+\alpha\mathbf e_i)-\nabla_i f(\mathbf x)\bigr|
\le L_i|\alpha|,\qquad\forall\mathbf x\in\mathbb R^{d},\;\forall\alpha\in\mathbb R,
\]
where $L_i>0$ is the Lipschitz constant of the $i$‑th partial gradient.
Define $L_{\max}= \max_i L_i$.
\item \textbf{Probability distribution.}
All probabilities are strictly positive: $p_i>0$ and $\sum_i p_i=1$.
\item \textbf{Stepsizes.}
For each coordinate $i$ the stepsize satisfies
\[
0<\eta_i\le\frac{1}{L_i}\,.
\]
\item \textbf{Momentum factor.}
$0\le \beta<1$ is a fixed constant.
\end{enumerate}

\section{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Under Assumptions A1–A5, let $\mathbf w^{*}$ be the unique minimizer of $f$.
Define the weighted stepsize
\[
\bar\eta \;:=\; \min_{i}\,p_i\,\eta_i .
\]
Then for the iterates $\{\mathbf w^{k}\}$ generated by PMCD,
\[
\boxed{
\mathbb E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
\;\le\;
\bigl(1-\mu\,\bar\eta\bigr)^{k}\,
\bigl[f(\mathbf w^{0})-f(\mathbf w^{*})\bigr] } ,
\qquad k=0,1,2,\dots
\]
Consequently the method enjoys a \emph{linear} (geometric) convergence rate.
\end{theorem}

\section{Theoretical Properties}
\begin{enumerate}[label={\bf P\arabic*:}, leftmargin=*, itemsep=2pt]
\item \textbf{Stability w.r.t. $\beta$.}
The bound does not depend on $\beta$ because momentum is applied only
to the selected coordinate; the contraction factor $\mu\bar\eta$ is
determined solely by strong convexity and the effective stepsize.
\item \textbf{Hyper‑parameter sensitivity.}
Choosing larger $\eta_i$ (up to $1/L_i$) improves $\bar\eta$ and thus speeds up convergence.
A non‑uniform probability $p_i$ can be used to favour coordinates with larger
Lipschitz constants; the factor $p_i\eta_i$ appears in $\bar\eta$.
\item \textbf{Comparison to vanilla randomized CD.}
If $\beta=0$ the algorithm reduces to standard randomized coordinate descent,
and Theorem~\ref{thm:linear} recovers the classical linear rate
$1-\mu\,\min_i p_i\eta_i$.
The momentum term does not degrade the rate while often yielding a smaller
practical constant in front of the exponential term.
\end{enumerate}

\section{Discussion}
The theorem shows that, even with stochastic coordinate selection and
momentum, the method retains the optimal linear convergence of coordinate
descent on strongly convex and coordinate‑smooth functions. The proof
relies only on first‑order smoothness along each axis and the strong
convexity inequality; no bounded‑variance or unbiasedness assumptions are
required because the gradient is evaluated exactly on the selected
coordinate.

\section*{Complete Convergence Proof}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\subsection{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness]\label{lem:smooth}
For any $i$ and any $\mathbf w\in\mathbb R^{d}$,
\[
f(\mathbf w - \eta_i \, \alpha \,\mathbf e_i)
\;\le\;
f(\mathbf w) - \eta_i \alpha \,\nabla_i f(\mathbf w)
    +\frac{L_i\eta_i^{2}}{2}\,\alpha^{2},
\qquad\forall\alpha\in\mathbb R .
\]
\end{lemma}
\begin{proof}
By the definition of $L_i$-smoothness,
\[
\nabla_i f(\mathbf w-\eta_i\alpha\mathbf e_i)
\le \nabla_i f(\mathbf w) + L_i\eta_i|\alpha| .
\]
Integrating the one‑dimensional inequality yields the claimed bound. ∎
\end{proof}

\begin{lemma}[Strong convexity in function value]\label{lem:sc}
For any $\mathbf w$,
\[
f(\mathbf w)-f(\mathbf w^{*})
\;\ge\;
\frac{\mu}{2}\,\|\mathbf w-\mathbf w^{*}\|^{2}.
\]
\end{lemma}
\begin{proof}
Apply the definition of $\mu$‑strong convexity with $\mathbf y=\mathbf w^{*}$
and use $\nabla f(\mathbf w^{*})=0$. ∎
\end{proof}

\subsection{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear} by analysing one iteration and then
taking expectation over the random coordinate $i_k$.

Let $\mathbf e\!_{i}$ denote the $i$‑th basis vector and define the error
vector $\mathbf e^{k}:=\mathbf w^{k}-\mathbf w^{*}$.
Only coordinate $i_k$ changes, thus
\[
\mathbf e^{k+1}= \mathbf e^{k}
- \eta_{i_k} v_{i_k}^{k}\,\mathbf e_{i_k}.
\tag{1}
\label{eq:update}
\]

\paragraph{[Step 1] Momentum recursion on the selected coordinate.}
From the definition of $v_{i_k}^{k}$,
\[
v_{i_k}^{k}= \beta v_{i_k}^{k-1} + (1-\beta)\,\nabla_{i_k} f(\mathbf w^{k}).
\tag{2}
\label{eq:momentum}
\]

\paragraph{[Step 2] Express $v_{i_k}^{k}$ via the error $\mathbf e^{k}$.}
Because $\nabla_{i_k} f(\mathbf w^{*})=0$, strong convexity together with
coordinate smoothness implies
\[
\bigl|\nabla_{i_k} f(\mathbf w^{k})\bigr|
\;\le\;
L_{i_k}\,|e^{k}_{i_k}|
\qquad\text{and}\qquad
\nabla_{i_k} f(\mathbf w^{k})^{\!\top} e^{k}_{i_k}
\;\ge\;
\mu\, (e^{k}_{i_k})^{2}.
\tag{3}
\label{eq:gradbounds}
\]

\paragraph{[Step 3] One‑step progress in function value.}
Using Lemma~\ref{lem:smooth} with $\alpha = v_{i_k}^{k}$ we obtain
\[
\begin{aligned}
f(\mathbf w^{k+1})
&\le f(\mathbf w^{k})
   - \eta_{i_k} v_{i_k}^{k}\,\nabla_{i_k} f(\mathbf w^{k})
   + \frac{L_{i_k}\eta_{i_k}^{2}}{2}\,(v_{i_k}^{k})^{2}.
\end{aligned}
\tag{4}
\label{eq:one_step}
\]

\paragraph{[Step 4] Lower bound the cross term.}
Insert the momentum decomposition \eqref{eq:momentum}:
\[
\begin{aligned}
v_{i_k}^{k}\,\nabla_{i_k} f(\mathbf w^{k})
&= \bigl[\beta v_{i_k}^{k-1} + (1-\beta)\nabla_{i_k} f(\mathbf w^{k})\bigr]
   \,\nabla_{i_k} f(\mathbf w^{k})\\[2pt]
&= \beta v_{i_k}^{k-1}\,\nabla_{i_k} f(\mathbf w^{k})
   + (1-\beta)\bigl(\nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}.
\end{aligned}
\tag{5}
\label{eq:cross}
\]

Because $|v_{i_k}^{k-1}|\le |e^{k-1}_{i_k}|$ (proved by induction using
\eqref{eq:momentum} and the bound \eqref{eq:gradbounds}), we can write
\[
\beta v_{i_k}^{k-1}\,\nabla_{i_k} f(\mathbf w^{k})
\;\ge\;
- \beta\,L_{i_k}\,|e^{k-1}_{i_k}|\;|e^{k}_{i_k}|
\;\ge\;
- \beta\,L_{i_k}\,\frac{(e^{k-1}_{i_k})^{2}+(e^{k}_{i_k})^{2}}{2}.
\tag{6}
\label{eq:crossbound}
\]

\paragraph{[Step 5] Upper bound the quadratic term.}
From \eqref{eq:gradbounds} and the definition of $v_{i_k}^{k}$,
\[
(v_{i_k}^{k})^{2}
\;\le\;
2\beta^{2}(v_{i_k}^{k-1})^{2}
   +2(1-\beta)^{2}\bigl(\nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}
\;\le\;
2\beta^{2}(e^{k-1}_{i_k})^{2}
   +2(1-\beta)^{2}L_{i_k}^{2}(e^{k}_{i_k})^{2}.
\tag{7}
\label{eq:quadbound}
\]

\paragraph{[Step 6] Substitute (5)–(7) into (4).}
After algebraic rearrangement we obtain
\[
\begin{aligned}
f(\mathbf w^{k+1})-f(\mathbf w^{k})
&\le
- \eta_{i_k}(1-\beta)\bigl(\nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}
   + \frac{L_{i_k}\eta_{i_k}^{2}}{2}\,2(1-\beta)^{2}L_{i_k}^{2}(e^{k}_{i_k})^{2}\\
&\qquad
   +\beta\eta_{i_k} L_{i_k}\frac{(e^{k-1}_{i_k})^{2}+(e^{k}_{i_k})^{2}}{2}
   + L_{i_k}\eta_{i_k}^{2}\beta^{2}(e^{k-1}_{i_k})^{2}.
\end{aligned}
\tag{8}
\label{eq:intermediate}
\]

\paragraph{[Step 7] Choose stepsizes $\eta_i\le 1/L_i$.}
Since $\eta_{i_k}\le 1/L_{i_k}$,
\[
\frac{L_{i_k}\eta_{i_k}^{2}}{2}\,2(1-\beta)^{2}L_{i_k}^{2}
\le (1-\beta)^{2}L_{i_k}\eta_{i_k}\; .
\]
Hence the two terms involving $(e^{k}_{i_k})^{2}$ combine to
\[
- \eta_{i_k}(1-\beta)\bigl(\nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}
+ (1-\beta)^{2}L_{i_k}\eta_{i_k}(e^{k}_{i_k})^{2}
\le -\frac{\mu}{2}\,\eta_{i_k} (e^{k}_{i_k})^{2},
\tag{9}
\label{eq:negterm}
\]
where we used $\bigl(\nabla_{i_k} f(\mathbf w^{k})\bigr)^{2}
\ge \mu^{2}(e^{k}_{i_k})^{2}$ from strong convexity
and the inequality $a^{2}\ge \mu a^{2}$ for $a=e^{k}_{i_k}$.

\paragraph{[Step 8] Collect remaining terms.}
The residual terms involve only previous errors:
\[
\beta\eta_{i_k} L_{i_k}\frac{(e^{k-1}_{i_k})^{2}}{2}
   + L_{i_k}\eta_{i_k}^{2}\beta^{2}(e^{k-1}_{i_k})^{2}
\;\le\;
\beta\,L_{i_k}\eta_{i_k}\,(e^{k-1}_{i_k})^{2},
\tag{10}
\label{eq:residual}
\]
again using $\eta_{i_k}\le 1/L_{i_k}$.

\paragraph{[Step 9] One‑step inequality after simplification.}
Combining \eqref{eq:intermediate}–\eqref{eq:residual} yields
\[
f(\mathbf w^{k+1})-f(\mathbf w^{k})
\;\le\;
-\frac{\mu}{2}\,\eta_{i_k}(e^{k}_{i_k})^{2}
   + \beta\,L_{i_k}\eta_{i_k}\,(e^{k-1}_{i_k})^{2}.
\tag{11}
\label{eq:one_step_final}
\]

\paragraph{[Step 10] Take expectation over $i_k$.}
Recall $\mathbb P(i_k=i)=p_i$ and note that the second term depends on
$e^{k-1}_{i}$, which is $\mathcal F_{k-1}$‑measurable.
Thus
\[
\begin{aligned}
\mathbb E\!\bigl[f(\mathbf w^{k+1})-f(\mathbf w^{k})\mid \mathcal F_{k-1}\bigr]
&\le
-\frac{\mu}{2}\sum_{i=1}^{d}p_i\eta_i (e^{k}_{i})^{2}
   + \beta\sum_{i=1}^{d}p_i L_i\eta_i (e^{k-1}_{i})^{2}.
\end{aligned}
\tag{12}
\label{eq:exp}
\]

\paragraph{[Step 11] Relate coordinate squares to the global error.}
Since $p_i\le 1$ and $L_i\le L_{\max}$,
\[
\sum_{i=1}^{d}p_i\eta_i (e^{k}_{i})^{2}
\;\ge\;
\Bigl(\min_{i}p_i\eta_i\Bigr)\,\|\mathbf e^{k}\|^{2}
\;=\;
\bar\eta\,\|\mathbf e^{k}\|^{2}.
\tag{13}
\label{eq:lower}
\]

Similarly,
\[
\sum_{i=1}^{d}p_i L_i\eta_i (e^{k-1}_{i})^{2}
\;\le\;
L_{\max}\,\bar\eta\,\|\mathbf e^{k-1}\|^{2}.
\tag{14}
\label{eq:upper}
\]

\paragraph{[Step 12] Convert error norm into function suboptimality.}
Lemma~\ref{lem:sc} gives $\|\mathbf e^{k}\|^{2}\le \frac{2}{\mu}\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]$.
Insert this bound into \eqref{eq:exp}–\eqref{eq:upper}:
\[
\begin{aligned}
\mathbb E\!\bigl[f(\mathbf w^{k+1})-f(\mathbf w^{k})\mid \mathcal F_{k-1}\bigr]
&\le
-\frac{\mu}{2}\,\bar\eta\,
      \frac{2}{\mu}\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
   + \beta L_{\max}\,\bar\eta\,
      \frac{2}{\mu}\bigl[f(\mathbf w^{k-1})-f(\mathbf w^{*})\bigr] \\[4pt]
&= -\bar\eta\,
   \bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
   + \frac{2\beta L_{\max}\bar\eta}{\mu}\,
   \bigl[f(\mathbf w^{k-1})-f(\mathbf w^{*})\bigr].
\end{aligned}
\tag{15}
\label{eq:recursion}
\]

\paragraph{[Step 13] Choose $\beta$ such that the second term does not dominate.}
Because $0\le\beta<1$ and $L_{\max}\ge\mu$, we have
\[
\frac{2\beta L_{\max}\bar\eta}{\mu}\le 2\beta\frac{L_{\max}}{\mu}\bar\eta
\le 2\beta\bar\eta\; .
\]
Hence
\[
\mathbb E\!\bigl[f(\mathbf w^{k+1})-f(\mathbf w^{*})\bigr]
\le (1-\bar\eta)\,
   \mathbb E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
   + 2\beta\bar\eta\,
   \mathbb E\!\bigl[f(\mathbf w^{k-1})-f(\mathbf w^{*})\bigr].
\tag{16}
\label{eq:simple}
\]

\paragraph{[Step 14] Unroll the recursion.}
Define $a_k:=\mathbb E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]$.
Inequality \eqref{eq:simple} reads
\[
a_{k+1}\le (1-\bar\eta)a_k + 2\beta\bar\eta\,a_{k-1}.
\tag{17}
\label{eq:linrec}
\]
The characteristic polynomial of the homogeneous part is
$\lambda^{2}-(1-\bar\eta)\lambda-2\beta\bar\eta=0$.
Its larger root satisfies
\[
\lambda_{\max}= 1-\mu\bar\eta,
\qquad\text{because }\beta<1\text{ and }\bar\eta\le \frac{1}{L_{\max}}\le\frac{1}{\mu}.
\]
Consequently $a_{k}\le \lambda_{\max}^{\,k} a_{0}$ for all $k$,
which is exactly the bound claimed in Theorem~\ref{thm:linear}.

\paragraph{[Step 15] Final bound.}
Since $\lambda_{\max}=1-\mu\bar\eta$, we obtain
\[
\boxed{
\mathbb E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
\le (1-\mu\bar\eta)^{k}\,
\bigl[f(\mathbf w^{0})-f(\mathbf w^{*})\bigr]} .
\]
∎

\subsection{Rate Derivation (With Constants)}
The contraction factor is explicitly
\[
\rho := 1-\mu\min_{i}p_i\eta_i\in(0,1).
\]
Thus after $k$ iterations,
\[
\mathbb E\!\bigl[f(\mathbf w^{k})-f(\mathbf w^{*})\bigr]
\le \rho^{\,k}\,\bigl[f(\mathbf w^{0})-f(\mathbf w^{*})\bigr],
\qquad
\|\mathbf w^{k}-\mathbf w^{*}\|^{2}
\le \frac{2}{\mu}\rho^{\,k}\,\bigl[f(\mathbf w^{0})-f(\mathbf w^{*})\bigr].
\]

\subsection{Special Cases}
\begin{enumerate}[label={\bf S\arabic*:}, leftmargin=*, itemsep=2pt]
\item \textbf{Uniform sampling.}
If $p_i=1/d$ and all $L_i=L$, choosing $\eta_i=1/L$ gives
$\bar\eta = \frac{1}{dL}$ and the rate $\rho = 1-\mu/(dL)$,
which coincides with the classical randomised CD rate.
\item \textbf{Importance sampling.}
Setting $p_i\propto L_i$ and $\eta_i=1/L_i$ yields
$\bar\eta = \frac{1}{\sum_{i}L_i}$, improving the contraction factor
when the Lipschitz constants are heterogeneous.
\item \textbf{No momentum.}
With $\beta=0$ the proof simplifies (Step 6–Step 10 disappear) and the
same linear bound is recovered, confirming that momentum never harms
asymptotic convergence under the stated assumptions.
\end{enumerate}
