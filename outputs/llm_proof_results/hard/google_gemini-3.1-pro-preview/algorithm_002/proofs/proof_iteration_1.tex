\section*{Algorithm Name}
Importance-Weighted Randomized Coordinate Descent (IW-RCD)

\section*{Mathematical Formulation}
Let $f: \mathbb{R}^d \to \mathbb{R}$ be a differentiable objective function. At each iteration $k$, the algorithm selects a single coordinate $i_k \in \{1, 2, \dots, d\}$ randomly according to a categorical distribution defined by probabilities $p_1, p_2, \dots, p_d$, where $p_i > 0$ for all $i$ and $\sum_{i=1}^d p_i = 1$.

To ensure the update direction is an unbiased estimator of the full gradient, the algorithm scales the partial derivative by the inverse of the selection probability, $1/p_{i_k}$. The stochastic gradient estimator is defined as:
\begin{equation}
    g_k = \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k) e_{i_k}
\end{equation}
where $e_{i_k}$ is the standard basis vector for the $i_k$-th dimension, and $\nabla_{i_k} f(w_k)$ is the partial derivative of $f$ with respect to the $i_k$-th coordinate.

The parameter update rule follows standard stochastic gradient descent using this estimator:
\begin{equation}
    w_{k+1} = w_k - \eta g_k = w_k - \eta \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k) e_{i_k}
\end{equation}
where $\eta > 0$ is the learning rate. Note that this update only affects the $i_k$-th coordinate of $w_k$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Importance-Weighted Randomized Coordinate Descent}
\begin{algorithmic}[1]
\REQUIRE Objective function $f$, initial weights $w_0 \in \mathbb{R}^d$, learning rate $\eta > 0$, probabilities $\{p_i\}_{i=1}^d$, total iterations $T$.
\FOR{$k = 0$ to $T-1$}
    \STATE Sample coordinate index $i_k \in \{1, \dots, d\}$ with probability $\mathbb{P}(i_k = i) = p_i$.
    \STATE Compute the partial derivative: $d_k = \nabla_{i_k} f(w_k)$.
    \STATE Compute the importance-weighted update scalar: $\Delta = \frac{\eta}{p_{i_k}} d_k$.
    \STATE Update only the selected coordinate: $w_{k+1}^{(i_k)} = w_k^{(i_k)} - \Delta$.
    \STATE Keep other coordinates unchanged: $w_{k+1}^{(j)} = w_k^{(j)}$ for all $j \neq i_k$.
\ENDFOR
\RETURN $w_T$ (or a randomly chosen iterate $\bar{w}_T$ from $\{w_0, \dots, w_{T-1}\}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We apply the algorithm to the 1-dimensional function $f(w) = (w-3)^2$. 
\begin{itemize}
    \item \textbf{Parameters:} $d=1$, $w_0=0$, $\eta=0.1$. 
    \item \textbf{Probabilities:} Since $d=1$, the only valid probability is $p_1 = 1$. The algorithm reduces to standard Gradient Descent, but we trace it using the formal IW-RCD notation.
\end{itemize}

\textbf{Iteration $k=0$:}
\begin{itemize}
    \item Current state: $w_0 = 0$. Objective: $f(w_0) = (0-3)^2 = 9$.
    \item Sample coordinate: $i_0 = 1$ (with probability $p_1=1$).
    \item Partial derivative: $\nabla f(0) = 2(0-3) = -6$.
    \item Estimator: $g_0 = \frac{1}{1} (-6) = -6$.
    \item Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6$.
    \item New Objective: $f(w_1) = (0.6-3)^2 = (-2.4)^2 = 5.76$.
\end{itemize}

\textbf{Iteration $k=1$:}
\begin{itemize}
    \item Current state: $w_1 = 0.6$.
    \item Sample coordinate: $i_1 = 1$.
    \item Partial derivative: $\nabla f(0.6) = 2(0.6-3) = -4.8$.
    \item Estimator: $g_1 = \frac{1}{1} (-4.8) = -4.8$.
    \item Update: $w_2 = w_1 - \eta g_1 = 0.6 - 0.1(-4.8) = 1.08$.
    \item New Objective: $f(w_2) = (1.08-3)^2 = (-1.92)^2 = 3.6864$.
\end{itemize}

\textbf{Iteration $k=2$:}
\begin{itemize}
    \item Current state: $w_2 = 1.08$.
    \item Sample coordinate: $i_2 = 1$.
    \item Partial derivative: $\nabla f(1.08) = 2(1.08-3) = -3.84$.
    \item Estimator: $g_2 = -3.84$.
    \item Update: $w_3 = w_2 - \eta g_2 = 1.08 - 0.1(-3.84) = 1.464$.
    \item New Objective: $f(w_3) = (1.464-3)^2 = (-1.536)^2 \approx 2.359$.
\end{itemize}

\section*{Assumptions}
The convergence analysis relies on the following standard assumptions for coordinate descent methods:
\begin{enumerate}
    \item \textbf{Bounded Below:} The objective function $f$ is bounded below, meaning there exists a finite $f^*$ such that $f(w) \geq f^*$ for all $w \in \mathbb{R}^d$.
    \item \textbf{Coordinate-wise Smoothness:} The function $f$ is continuously differentiable, and its gradient is coordinate-wise Lipschitz continuous. Specifically, for each coordinate $i \in \{1, \dots, d\}$, there exists a constant $L_i > 0$ such that for all $w \in \mathbb{R}^d$ and $h \in \mathbb{R}$:
    \begin{equation}
        |\nabla_i f(w + h e_i) - \nabla_i f(w)| \leq L_i |h|
    \end{equation}
\end{enumerate}

\section*{Main Convergence Theorem}
\textbf{Theorem 1.} Under Assumptions 1 and 2, let $M = \max_{i} \frac{L_i}{p_i}$. If the learning rate is chosen such that $\eta \leq \frac{1}{M}$, the Importance-Weighted Randomized Coordinate Descent algorithm satisfies:
\begin{equation}
    \frac{1}{T} \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2 (f(w_0) - f^*)}{\eta T} = \mathcal{O}\left(\frac{1}{T}\right)
\end{equation}
where the expectation is taken over all random coordinate selections up to iteration $T-1$.

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Convergence Rate:} The algorithm achieves an $\mathcal{O}(1/T)$ convergence rate for the average squared gradient norm, which matches the standard rate of Gradient Descent for non-convex smooth functions.
    \item \textbf{Hyperparameter Sensitivity (Probabilities):} The maximum effective smoothness constant $M = \max_i \frac{L_i}{p_i}$ determines the allowable step size. If probabilities $p_i$ are chosen poorly (e.g., very small $p_i$ for a coordinate with large $L_i$), $M$ becomes excessively large, forcing $\eta$ to be minuscule and severely slowing down convergence.
    \item \textbf{Optimal Sampling:} To maximize the allowable learning rate (i.e., minimize $M$), one should choose $p_i$ proportional to $L_i$. This balances the $L_i/p_i$ terms across all coordinates.
\end{itemize}

\section*{Discussion}
The key advantage of IW-RCD over standard RCD is the explicit unbiasedness of the gradient estimator, which allows the algorithm to be analyzed exactly like SGD. The $1/p_{i_k}$ weighting ensures that coordinates sampled less frequently take proportionally larger steps. However, this introduces variance. If a coordinate has a very small $p_i$, its occasional update will be massive, which can lead to instability unless the global learning rate $\eta$ is aggressively reduced. This mathematically manifests in the dependence of the step size on $M = \max_i (L_i / p_i)$.

\section*{Preliminaries (Definitions and Lemmas)}

\textbf{Lemma 1 (Unbiased Estimator).} The importance-weighted coordinate gradient $g_k$ is an unbiased estimator of the full gradient $\nabla f(w_k)$.
\begin{proof}
Let $\mathbb{E}_{i_k}$ denote the conditional expectation with respect to the random choice of coordinate $i_k$ at iteration $k$, given $w_k$.
\begin{equation}
    \mathbb{E}_{i_k}[g_k] = \sum_{i=1}^d \mathbb{P}(i_k = i) \left( \frac{1}{p_i} \nabla_i f(w_k) e_i \right) = \sum_{i=1}^d p_i \frac{1}{p_i} \nabla_i f(w_k) e_i = \sum_{i=1}^d \nabla_i f(w_k) e_i = \nabla f(w_k)
\end{equation}
\end{equation*}
\end{proof}

\textbf{Lemma 2 (Coordinate Descent Lemma).} For any $w \in \mathbb{R}^d$, coordinate $i$, and scalar step $h$:
\begin{equation}
    f(w + h e_i) \leq f(w) + \nabla_i f(w) h + \frac{L_i}{2} h^2
\end{equation}
\begin{proof}
By the Fundamental Theorem of Calculus:
\begin{align*}
    f(w + h e_i) - f(w) &= \int_0^1 \langle \nabla f(w + \tau h e_i), h e_i \rangle d\tau \\
    &= \int_0^1 \nabla_i f(w + \tau h e_i) h d\tau \\
    &= \nabla_i f(w) h + \int_0^1 (\nabla_i f(w + \tau h e_i) - \nabla_i f(w)) h d\tau \\
    &\leq \nabla_i f(w) h + \int_0^1 |\nabla_i f(w + \tau h e_i) - \nabla_i f(w)| |h| d\tau \\
    &\leq \nabla_i f(w) h + \int_0^1 L_i \tau |h|^2 d\tau \quad \text{(by Assumption 2)} \\
    &= \nabla_i f(w) h + \frac{L_i}{2} h^2
\end{align*}
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}

We now derive the expected one-step descent bound for IW-RCD. 

\begin{align}
    \text{[Step 1]} \quad & \text{Apply Lemma 2 to the update } w_{k+1} = w_k + h e_{i_k} \text{ where } h = -\eta \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k): \notag \\
    & f(w_{k+1}) \leq f(w_k) + \nabla_{i_k} f(w_k) \left(-\eta \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k)\right) + \frac{L_{i_k}}{2} \left(-\eta \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k)\right)^2 \\
    \text{[Step 2]} \quad & \text{Simplify the linear term by multiplying the partial derivatives:} \notag \\
    & f(w_{k+1}) \leq f(w_k) - \eta \frac{1}{p_{i_k}} (\nabla_{i_k} f(w_k))^2 + \frac{L_{i_k}}{2} \left(-\eta \frac{1}{p_{i_k}} \nabla_{i_k} f(w_k)\right)^2 \\
    \text{[Step 3]} \quad & \text{Expand the square in the quadratic term:} \notag \\
    & f(w_{k+1}) \leq f(w_k) - \eta \frac{1}{p_{i_k}} (\nabla_{i_k} f(w_k))^2 + \frac{L_{i_k}}{2} \eta^2 \frac{1}{p_{i_k}^2} (\nabla_{i_k} f(w_k))^2 \\
    \text{[Step 4]} \quad & \text{Take the conditional expectation } \mathbb{E}_{i_k}[\cdot | w_k] \text{ on both sides:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) + \mathbb{E}_{i_k}\left[ - \eta \frac{1}{p_{i_k}} (\nabla_{i_k} f(w_k))^2 + \frac{L_{i_k} \eta^2}{2 p_{i_k}^2} (\nabla_{i_k} f(w_k))^2 \right] \\
    \text{[Step 5]} \quad & \text{Expand the expectation as a sum over all coordinates } i=1 \dots d, \text{ weighted by } p_i: \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) + \sum_{i=1}^d p_i \left( - \eta \frac{1}{p_i} (\nabla_i f(w_k))^2 + \frac{L_i \eta^2}{2 p_i^2} (\nabla_i f(w_k))^2 \right) \\
    \text{[Step 6]} \quad & \text{Distribute } p_i \text{ and cancel it with } 1/p_i \text{ in the first term:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \sum_{i=1}^d (\nabla_i f(w_k))^2 + \sum_{i=1}^d p_i \left( \frac{L_i \eta^2}{2 p_i^2} (\nabla_i f(w_k))^2 \right) \\
    \text{[Step 7]} \quad & \text{Recognize that } \sum_{i=1}^d (\nabla_i f(w_k))^2 = \|\nabla f(w_k)\|^2: \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \|\nabla f(w_k)\|^2 + \sum_{i=1}^d p_i \left( \frac{L_i \eta^2}{2 p_i^2} (\nabla_i f(w_k))^2 \right) \\
    \text{[Step 8]} \quad & \text{Cancel one } p_i \text{ in the denominator of the quadratic term:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \|\nabla f(w_k)\|^2 + \frac{\eta^2}{2} \sum_{i=1}^d \frac{L_i}{p_i} (\nabla_i f(w_k))^2 \\
    \text{[Step 9]} \quad & \text{Define } M = \max_{j} \frac{L_j}{p_j}. \text{ Thus, } \frac{L_i}{p_i} \leq M \text{ for all } i: \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \|\nabla f(w_k)\|^2 + \frac{\eta^2}{2} \sum_{i=1}^d M (\nabla_i f(w_k))^2 \\
    \text{[Step 10]} \quad & \text{Pull the constant } M \text{ out of the summation:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \|\nabla f(w_k)\|^2 + \frac{\eta^2 M}{2} \sum_{i=1}^d (\nabla_i f(w_k))^2 \\
    \text{[Step 11]} \quad & \text{Substitute } \sum_{i=1}^d (\nabla_i f(w_k))^2 = \|\nabla f(w_k)\|^2 \text{ into the quadratic term:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \|\nabla f(w_k)\|^2 + \frac{\eta^2 M}{2} \|\nabla f(w_k)\|^2 \\
    \text{[Step 12]} \quad & \text{Factor out } -\eta \|\nabla f(w_k)\|^2 \text{ from the last two terms:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \eta \left( 1 - \frac{\eta M}{2} \right) \|\nabla f(w_k)\|^2 \\
    \text{[Step 13]} \quad & \text{Require the learning rate to satisfy } \eta \leq \frac{1}{M}, \text{ which implies } \frac{\eta M}{2} \leq \frac{1}{2}: \notag \\
    & 1 - \frac{\eta M}{2} \geq \frac{1}{2} \\
    \text{[Step 14]} \quad & \text{Apply the bound from [Step 13] to the inequality in [Step 12]:} \notag \\
    & \mathbb{E}_{i_k}[f(w_{k+1})] \leq f(w_k) - \frac{\eta}{2} \|\nabla f(w_k)\|^2
\end{align}

\section*{Rate Derivation (With Constants)}

We now extend the one-step conditional bound to the full trajectory over $T$ iterations.

\begin{align}
    \text{[Step 15]} \quad & \text{Rearrange the inequality from [Step 14] to isolate the squared gradient norm:} \notag \\
    & \frac{\eta}{2} \|\nabla f(w_k)\|^2 \leq f(w_k) - \mathbb{E}_{i_k}[f(w_{k+1})] \\
    \text{[Step 16]} \quad & \text{Multiply by } \frac{2}{\eta}: \notag \\
    & \|\nabla f(w_k)\|^2 \leq \frac{2}{\eta} \left( f(w_k) - \mathbb{E}_{i_k}[f(w_{k+1})] \right) \\
    \text{[Step 17]} \quad & \text{Take the total expectation } \mathbb{E} \text{ over all randomness up to iteration } k: \notag \\
    & \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2}{\eta} \left( \mathbb{E}[f(w_k)] - \mathbb{E}[f(w_{k+1})] \right) \\
    \text{[Step 18]} \quad & \text{Sum the inequality from } k=0 \text{ to } T-1: \notag \\
    & \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \sum_{k=0}^{T-1} \frac{2}{\eta} \left( \mathbb{E}[f(w_k)] - \mathbb{E}[f(w_{k+1})] \right) \\
    \text{[Step 19]} \quad & \text{Evaluate the telescoping sum on the right-hand side:} \notag \\
    & \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2}{\eta} \left( f(w_0) - \mathbb{E}[f(w_T)] \right) \\
    \text{[Step 20]} \quad & \text{Apply Assumption 1, which states } f(w_T) \geq f^*, \text{ meaning } -f(w_T) \leq -f^*: \notag \\
    & \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2}{\eta} \left( f(w_0) - f^* \right) \\
    \text{[Step 21]} \quad & \text{Divide both sides by } T \text{ to obtain the final average expected gradient norm:} \notag \\
    & \frac{1}{T} \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2(f(w_0) - f^*)}{\eta T}
\end{align}

\section*{Special Cases}

The maximum allowable step size depends crucially on $M = \max_i \frac{L_i}{p_i}$. We analyze two common choices for the sampling probabilities $p_i$.

\subsection*{Case 1: Uniform Sampling}
If coordinates are sampled uniformly, $p_i = \frac{1}{d}$ for all $i$.
\begin{equation}
    M_{\text{uniform}} = \max_i \frac{L_i}{1/d} = d \max_i L_i
\end{equation}
The maximum allowable step size is $\eta = \frac{1}{d \max_i L_i}$. The convergence rate becomes:
\begin{equation}
    \frac{1}{T} \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2 d (\max_i L_i) (f(w_0) - f^*)}{T}
\end{equation}
This rate is bottlenecked by the worst-case (least smooth) coordinate across all dimensions.

\subsection*{Case 2: Importance Sampling (Lipschitz-Proportional)}
If probabilities are chosen proportional to the coordinate-wise Lipschitz constants, $p_i = \frac{L_i}{\sum_{j=1}^d L_j}$.
\begin{equation}
    M_{\text{importance}} = \max_i \frac{L_i}{L_i / \sum_{j=1}^d L_j} = \sum_{j=1}^d L_j
\end{equation}
The maximum allowable step size is $\eta = \frac{1}{\sum_{j=1}^d L_j}$. The convergence rate becomes:
\begin{equation}
    \frac{1}{T} \sum_{k=0}^{T-1} \mathbb{E}[\|\nabla f(w_k)\|^2] \leq \frac{2 (\sum_{j=1}^d L_j) (f(w_0) - f^*)}{T}
\end{equation}
Because $\sum_{j=1}^d L_j \leq d \max_i L_i$, importance sampling always yields a larger maximum allowable step size (and thus a strictly tighter theoretical convergence bound) than uniform sampling, particularly when the smoothness constants $L_i$ are highly heterogeneous.
