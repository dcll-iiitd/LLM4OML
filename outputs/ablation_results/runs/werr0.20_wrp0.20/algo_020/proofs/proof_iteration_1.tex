\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be convex and $L$‑smooth (i.e., $\nabla f$ is $L$‑Lipschitz) and let $g:\mathbb{R}^d\to\mathbb{R}\cup\{+\infty\}$ be a proper closed convex function for which the proximal operator
\[
\prox_{\eta g} (v)\;:=\;\arg\min_{x\in\mathbb{R}^d}\Big\{ g(x)+\frac{1}{2\eta}\|x-v\|^2\Big\}
\]
is efficiently computable.  
Given a stepsize $\eta>0$, the iterates of the proximal gradient method are
\begin{equation}\label{eq:update}
x_{k+1}= \prox_{\eta g}\!\bigl(x_k-\eta\nabla f(x_k)\bigr),\qquad k=0,1,2,\dots
\end{equation}
The objective to minimise is the composite convex function
\[
F(x):=f(x)+g(x).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, maximum iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute gradient $g_k:=\nabla f(x_k)$.
    \STATE Take a gradient step $y_k:=x_k-\eta g_k$.
    \STATE Apply proximal map $x_{k+1}:=\prox_{\eta g}(y_k)$.
\ENDFOR
\RETURN $x_K$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple scalar problem
\[
f(w)=(w-3)^2,\qquad g\equiv 0,
\]
so that $\prox_{\eta g}= \operatorname{id}$.  
Take $w_0=0$ and stepsize $\eta=0.1$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $k=0$  
    \[
    \nabla f(w_0)=2(w_0-3)=-6,\qquad 
    y_0=w_0-\eta\nabla f(w_0)=0-0.1(-6)=0.6,
    \]
    \[
    w_1=\prox_{\eta g}(y_0)=y_0=0.6,\qquad 
    F(w_1)= (0.6-3)^2=5.76.
    \]
\item $k=1$  
    \[
    \nabla f(w_1)=2(0.6-3)=-4.8,\qquad 
    y_1=0.6-0.1(-4.8)=1.08,
    \]
    \[
    w_2=1.08,\qquad 
    F(w_2)=(1.08-3)^2=3.6864.
    \]
\item $k=2$  
    \[
    \nabla f(w_2)=2(1.08-3)=-3.84,\qquad 
    y_2=1.08-0.1(-3.84)=1.464,
    \]
    \[
    w_3=1.464,\qquad 
    F(w_3)=(1.464-3)^2=2.365.
    \]
\end{enumerate}
The objective value decreases monotonically, illustrating the descent property of \eqref{eq:update}.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*:}, leftmargin=*]
\item \textbf{Convexity of $f$ and $g$:} $f$ and $g$ are convex; consequently $F=f+g$ is convex.
\item \textbf{Smoothness of $f$:} $\nabla f$ is $L$‑Lipschitz, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\tag{A2}
\]
Equivalently, for all $x,y$,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2.
\tag{A2'}
\]
\item \textbf{Stepsize bound:} The constant stepsize satisfies
\[
0<\eta\le \frac{1}{L}.
\tag{A3}
\]
\item \textbf{Existence of a minimiser:} The set of minimisers
\[
\mathcal{X}^\star :=\arg\min_{x\in\mathbb{R}^d}F(x)
\]
is non‑empty; pick any $x^\star\in\mathcal{X}^\star$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Under Assumptions A1–A4 and for a constant stepsize $\eta\in(0,1/L]$, the iterates generated by \eqref{eq:update} satisfy for all $k\ge 1$
\begin{equation}\label{eq:gap}
F(x_k)-F(x^\star)\;\le\;\frac{\|x_0-x^\star\|^2}{2\eta\,k}.
\end{equation}
Consequently $F(x_k)\to F(x^\star)$ at the rate $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\textbf{P\arabic*:}, leftmargin=*]
\item \textbf{Stability:} The method is a non‑expansive map composed of a gradient step (which is $\tfrac{1}{1-\eta L}$‑Lipschitz for $\eta\le1/L$) followed by the firmly non‑expansive proximal operator. Hence the whole iteration is contractive in the sense of Lemma~\ref{lem:nonexp}.
\item \textbf{Hyperparameter Sensitivity:} The bound \eqref{eq:gap} is monotone decreasing in $\eta$ up to $1/L$. Choosing $\eta=1/L$ yields the smallest constant in the $O(1/k)$ rate.
\item \textbf{Comparison with Gradient Descent:} When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and \eqref{eq:gap} coincides with the classic $O(1/k)$ rate for gradient descent on smooth convex functions.
\item \textbf{Extension to Varying Stepsizes:} If a diminishing stepsize $\eta_k$ satisfies $\sum_{k}\eta_k=\infty$ and $\sum_{k}\eta_k^2<\infty$, the same $O(1/k)$ bound holds with $\eta$ replaced by the harmonic mean of $\{\eta_k\}$.
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that the proximal gradient method inherits the optimal $O(1/k)$ convergence rate for the class of convex, $L$‑smooth $f$ plus a (possibly nonsmooth) convex regulariser $g$. The proof hinges on two elementary facts:
\begin{itemize}
\item the \emph{descent lemma} for $L$‑smooth functions;
\item the \emph{non‑expansiveness} (firmly non‑expansive property) of the proximal operator.
\end{itemize}
Both are used explicitly in the algebraic derivation below.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{lemma}[Firm non‑expansiveness of $\prox_{\eta g}$]\label{lem:nonexp}
For any $\eta>0$ and any $u,v\in\mathbb{R}^d$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|^2
\le \langle u-v,\;\prox_{\eta g}(u)-\prox_{\eta g}(v)\rangle .
\tag{2}
\]
\end{lemma}
\begin{proof}
The proximal operator is the resolvent of the sub‑differential $\partial g$, which is firmly non‑expansive; (2) follows directly from the definition (see e.g. \cite{Parikh2014}).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^\star\in\mathcal{X}^\star$ be a fixed optimal solution. Define the auxiliary point
\[
y_k := x_k-\eta\nabla f(x_k).
\]
From the update rule \eqref{eq:update},
\[
x_{k+1}= \prox_{\eta g}(y_k).
\]

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:nonexp} with $u=y_k$ and $v=x^\star$ (note that $x^\star=\prox_{\eta g}(x^\star)$ because $0\in\partial g(x^\star)$):
\[
\|x_{k+1}-x^\star\|^2
\le \langle y_k-x^\star,\;x_{k+1}-x^\star\rangle .
\tag{3}
\]

\noindent\textbf{[Step 2]} Expand $y_k-x^\star$:
\[
y_k-x^\star = (x_k-x^\star)-\eta\bigl(\nabla f(x_k)-\nabla f(x^\star)\bigr)
\]
because $\nabla f(x^\star)$ exists (convex differentiable) and $x^\star$ minimizes $F$, implying $0\in \nabla f(x^\star)+\partial g(x^\star)$, hence $\nabla f(x^\star) = -s^\star$ for some $s^\star\in\partial g(x^\star)$. However we do not need the explicit $s^\star$; we only use the identity above.

Insert this decomposition into (3):
\[
\|x_{k+1}-x^\star\|^2
\le \langle x_k-x^\star,\;x_{k+1}-x^\star\rangle
-\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x^\star\rangle .
\tag{4}
\]

\noindent\textbf{[Step 3]} Add and subtract $x_k$ inside the second inner product:
\[
\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x^\star\rangle
=
\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x_k\rangle
+
\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\tag{5}
\]

\noindent\textbf{[Step 4]} Substitute (5) into (4) and rearrange:
\[
\begin{aligned}
\|x_{k+1}-x^\star\|^2
&\le \langle x_k-x^\star,\;x_{k+1}-x^\star\rangle  \\
&\quad -\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x_k\rangle
-\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 5]} Use the elementary identity
\[
\|a\|^2 = \|b\|^2 + 2\langle a-b, b\rangle + \|a-b\|^2,
\]
with $a=x_{k+1}-x^\star$ and $b=x_k-x^\star$, which yields
\[
\langle x_k-x^\star,\;x_{k+1}-x^\star\rangle
= \frac{1}{2}\bigl(\|x_k-x^\star\|^2+\|x_{k+1}-x^\star\|^2-\|x_{k+1}-x_k\|^2\bigr).
\tag{7}
\]

\noindent\textbf{[Step 6]} Plug (7) into (6) and cancel the term $\|x_{k+1}-x^\star\|^2$ from both sides:
\[
\begin{aligned}
0 &\le \frac{1}{2}\bigl(\|x_k-x^\star\|^2-\|x_{k+1}-x_k\|^2\bigr) \\
&\qquad -\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x_k\rangle
-\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 7]} Move the last two inner products to the right‑hand side:
\[
\begin{aligned}
\frac{1}{2}\|x_k-x^\star\|^2
&\ge \frac{1}{2}\|x_{k+1}-x_k\|^2
+\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x_k\rangle \\
&\quad +\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 8]} Apply Cauchy–Schwarz and the $L$‑Lipschitz property (A2) to bound the first inner product:
\[
\langle \nabla f(x_k)-\nabla f(x^\star),\;x_{k+1}-x_k\rangle
\ge -\|\nabla f(x_k)-\nabla f(x^\star)\|\,\|x_{k+1}-x_k\|
\ge -L\|x_k-x^\star\|\,\|x_{k+1}-x_k\|.
\tag{10}
\]
Insert (10) into (9):
\[
\frac{1}{2}\|x_k-x^\star\|^2
\ge \frac{1}{2}\|x_{k+1}-x_k\|^2
-\eta L\|x_k-x^\star\|\|x_{k+1}-x_k\|
+\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\tag{11}
\]

\noindent\textbf{[Step 9]} Complete the square for the first two terms on the right:
\[
\frac{1}{2}\|x_{k+1}-x_k\|^2-\eta L\|x_k-x^\star\|\|x_{k+1}-x_k\|
\ge -\frac{(\eta L)^2}{2}\|x_k-x^\star\|^2 .
\tag{12}
\]
(The inequality follows from $ab\le\frac{a^2}{2}+\frac{b^2}{2}$ with $a=\|x_{k+1}-x_k\|$ and $b=\eta L\|x_k-x^\star\|$.)

Thus (11) becomes
\[
\frac{1}{2}\|x_k-x^\star\|^2
\ge -\frac{(\eta L)^2}{2}\|x_k-x^\star\|^2
+\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle .
\tag{13}
\]

\noindent\textbf{[Step 10]} Rearrange (13):
\[
\eta\langle \nabla f(x_k)-\nabla f(x^\star),\;x_k-x^\star\rangle
\le \frac{1}{2}\bigl(1+(\eta L)^2\bigr)\|x_k-x^\star\|^2 .
\tag{14}
\]

\noindent\textbf{[Step 11]} By convexity of $f$,
\[
f(x_k)-f(x^\star)\le \langle \nabla f(x_k),\;x_k-x^\star\rangle .
\tag{15}
\]
Similarly, convexity of $g$ together with the optimality condition $0\in \nabla f(x^\star)+\partial g(x^\star)$ yields
\[
g(x_{k+1})-g(x^\star)\le \langle s_{k+1},\;x_{k+1}-x^\star\rangle,
\qquad s_{k+1}\in\partial g(x_{k+1}).
\tag{16}
\]

\noindent\textbf{[Step 12]} From the definition of the proximal operator (optimality condition)
\[
\frac{1}{\eta}(x_k-\eta\nabla f(x_k)-x_{k+1})\in\partial g(x_{k+1}),
\]
i.e.
\[
s_{k+1}= \frac{1}{\eta}\bigl(x_k-x_{k+1}\bigr)-\nabla f(x_k).
\tag{17}
\]

\noindent\textbf{[Step 13]} Plug (17) into (16) and use $x^\star=\prox_{\eta g}(x^\star)$ (so $s^\star\in\partial g(x^\star)$ and $s^\star=-\nabla f(x^\star)$):
\[
\begin{aligned}
g(x_{k+1})-g(x^\star)
&\le \Big\langle \frac{1}{\eta}(x_k-x_{k+1})-\nabla f(x_k),\;x_{k+1}-x^\star\Big\rangle \\
&= \frac{1}{\eta}\langle x_k-x_{k+1},x_{k+1}-x^\star\rangle
-\langle \nabla f(x_k),x_{k+1}-x^\star\rangle .
\end{aligned}
\tag{18}
\]

\noindent\textbf{[Step 14]} Add (15) and (18) to obtain a bound on the total objective gap:
\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&= \bigl[f(x_{k+1})-f(x^\star)\bigr]+\bigl[g(x_{k+1})-g(x^\star)\bigr] \\
&\le \langle \nabla f(x_{k+1}),x_{k+1}-x^\star\rangle \\
&\quad +\frac{1}{\eta}\langle x_k-x_{k+1},x_{k+1}-x^\star\rangle
-\langle \nabla f(x_k),x_{k+1}-x^\star\rangle .
\end{aligned}
\tag{19}
\]

\noindent\textbf{[Step 15]} Use the descent lemma (Lemma~\ref{lem:descent}) with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\langle \nabla f(x_k),x_{k+1}-x_k\rangle
+\frac{L}{2}\|x_{k+1}-x_k\|^2 .
\tag{20}
\]
Rearranging gives
\[
\langle \nabla f(x_{k+1}),x_{k+1}-x_k\rangle
\le f(x_{k+1})-f(x_k)-\frac{L}{2}\|x_{k+1}-x_k\|^2 .
\tag{21}
\]

\noindent\textbf{[Step 16]} Substitute (21) into (19) after adding and subtracting $\langle \nabla f(x_k),x_{k+1}-x_k\rangle$:
\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&\le \frac{1}{\eta}\langle x_k-x_{k+1},x_{k+1}-x^\star\rangle \\
&\quad +\langle \nabla f(x_{k+1})-\nabla f(x_k),x_{k+1}-x^\star\rangle .
\end{aligned}
\tag{22}
\]

\noindent\textbf{[Step 17]} Apply Cauchy–Schwarz and the Lipschitz property to the second term:
\[
\langle \nabla f(x_{k+1})-\nabla f(x_k),x_{k+1}-x^\star\rangle
\le \|\nabla f(x_{k+1})-\nabla f(x_k)\|\,\|x_{k+1}-x^\star\|
\le L\|x_{k+1}-x_k\|\,\|x_{k+1}-x^\star\|.
\tag{23}
\]

\noindent\textbf{[Step 18]} Use the elementary inequality $ab\le\frac{a^2}{2\eta}+ \frac{\eta b^2}{2}$ with $a=\|x_{k+1}-x_k\|$, $b=L\|x_{k+1}-x^\star\|$:
\[
L\|x_{k+1}-x_k\|\,\|x_{k+1}-x^\star\|
\le \frac{1}{2\eta}\|x_{k+1}-x_k\|^2+\frac{\eta L^{2}}{2}\|x_{k+1}-x^\star\|^2 .
\tag{24}
\]

\noindent\textbf{[Step 19]} Combine (22), (23) and (24):
\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&\le \frac{1}{\eta}\langle x_k-x_{k+1},x_{k+1}-x^\star\rangle
+\frac{1}{2\eta}\|x_{k+1}-x_k\|^2
+\frac{\eta L^{2}}{2}\|x_{k+1}-x^\star\|^2 .
\end{aligned}
\tag{25}
\]

\noindent\textbf{[Step 20]} Expand the first inner product:
\[
\langle x_k-x_{k+1},x_{k+1}-x^\star\rangle
= \frac{1}{2}\Bigl(\|x_k-x^\star\|^2-\|x_{k+1}-x^\star\|^2-\|x_{k+1}-x_k\|^2\Bigr).
\tag{26}
\]

\noindent\textbf{[Step 21]} Insert (26) into (25) and simplify:
\[
\begin{aligned}
F(x_{k+1})-F(x^\star)
&\le \frac{1}{2\eta}\Bigl(\|x_k-x^\star\|^2-\|x_{k+1}-x^\star\|^2-\|x_{k+1}-x_k\|^2\Bigr)\\
&\quad +\frac{1}{2\eta}\|x_{k+1}-x_k\|^2
+\frac{\eta L^{2}}{2}\|x_{k+1}-x^\star\|^2 .
\end{aligned}
\tag{27}
\]

The two terms $\pm\frac{1}{2\eta}\|x_{k+1}-x_k\|^2$ cancel, leaving
\[
F(x_{k+1})-F(x^\star)
\le \frac{1}{2\eta}\bigl(\|x_k-x^\star\|^2-\|x_{k+1}-x^\star\|^2\bigr)
+\frac{\eta L^{2}}{2}\|x_{k+1}-x^\star\|^2 .
\tag{28}
\]

\noindent\textbf{[Step 22]} Using the stepsize bound $\eta\le 1/L$ (Assumption A3) we have $\eta L^{2}\le L$, and more importantly
\[
\frac{\eta L^{2}}{2}\|x_{k+1}-x^\star\|^2\le \frac{1}{2\eta}\|x_{k+1}-x^\star\|^2 .
\tag{29}
\]

\noindent\textbf{[Step 23]} Substitute (29) into (28):
\[
F(x_{k+1})-F(x^\star)
\le \frac{1}{2\eta}\bigl(\|x_k-x^\star\|^2-\|x_{k+1}-x^\star\|^2\bigr)
+\frac{1}{2\eta}\|x_{k+1}-x^\star\|^2
= \frac{1}{2\eta}\|x_k-x^\star\|^2 .
\tag{30}
\]

\noindent\textbf{[Step 24]} Rearranging (30) gives the fundamental recursion
\[
\|x_{k+1}-x^\star\|^2 \;\le\; \|x_k-x^\star\|^2 -2\eta\bigl(F(x_{k+1})-F(x^\star)\bigr).
\tag{31}
\]

\noindent\textbf{[Step 25]} Sum (31) from $t=0$ to $k-1$:
\[
\|x_k-x^\star\|^2 \le \|x_0-x^\star\|^2 -2\eta\sum_{t=0}^{k-1}\bigl(F(x_{t+1})-F(x^\star)\bigr).
\tag{32}
\]

Since each term $F(x_{t+1})-F(x^\star)\ge 0$, dropping all but the last yields
\[
2\eta\,k\bigl(F(x_k)-F(x^\star)\bigr)\le \|x_0-x^\star\|^2 .
\tag{33}
\]

\noindent\textbf{[Step 26]} Divide by $2\eta k$ to obtain the $O(1/k)$ bound:
\[
F(x_k)-F(x^\star)\;\le\;\frac{\|x_0-x^\star\|^2}{2\eta\,k},
\]
which is exactly \eqref{eq:gap}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{eq:gap} depends only on the initial distance to the solution set and on the stepsize $\eta$. Choosing the largest admissible stepsize $\eta=1/L$ minimises the bound:
\[
F(x_k)-F(x^\star)\le \frac{L\,\|x_0-x^\star\|^2}{2k}.
\]

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal step reduces to the identity, and the above proof collapses to the classical $O(1/k)$ rate for smooth convex minimisation.
\item \textbf{Indicator function of a convex set $C$ (projection).} If $g=\iota_C$, then $\prox_{\eta g}$ is the Euclidean projection $\Pi_C$. The algorithm becomes projected gradient descent, and the same $O(1/k)$ guarantee holds.
\item \textbf{Lasso regularisation $g(x)=\lambda\|x\|_1$.} The proximal map is soft‑thresholding; the bound \eqref{eq:gap} provides a rate for the composite objective $f+\lambda\|x\|_1$.
\end{enumerate}
