\section*{Algorithm Name}
\textbf{Randomized Block Coordinate Gradient Descent (RBCGD)}  

At each iteration a block of coordinates is sampled uniformly at random and only that block is updated using a (possibly stochastic) gradient step.

\section*{Mathematical Formulation}
Let $w\in\R^{d}$ be partitioned into $B$ blocks 
\[
w=(w^{(1)},w^{(2)},\dots ,w^{(B)}),\qquad 
w^{(i)}\in\R^{d_i},\; \sum_{i=1}^{B}d_i=d .
\]
Denote by $\nabla_{i} f(w)$ the partial gradient of $f$ w.r.t. block $i$.  
Given a stepsize schedule $\{\eta_t\}_{t\ge 0}$, the update at iteration $t$ is

\[
\boxed{\;
\begin{aligned}
\text{Sample } i_t &\sim \text{Uniform}\{1,\dots ,B\},\\[0.2em]
w_{t+1}^{(i_t)} &= w_{t}^{(i_t)} - \eta_t \,\nabla_{i_t} f(w_t),\\
w_{t+1}^{(j)} &= w_{t}^{(j)}\quad\forall j\neq i_t .
\end{aligned}
\;}
\tag{1}
\]

All other blocks remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Block Coordinate Gradient Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\R^{d}$, stepsize schedule $\{\eta_t\}$, max iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample block index $i_t\sim\text{Uniform}\{1,\dots ,B\}$.
    \STATE Compute (partial) gradient $g_t \gets \nabla_{i_t} f(w_t)$.
    \STATE Update block $i_t$: $w_{t+1}^{(i_t)} \gets w_{t}^{(i_t)} - \eta_t g_t$.
    \STATE For $j\neq i_t$: $w_{t+1}^{(j)} \gets w_{t}^{(j)}$.
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or any iterate $w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $f(w)=(w-3)^2$, a single block ($B=1$), $w_0=0$, constant stepsize $\eta=0.1$, run for $3$ iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\nabla f(w_t)$ & $f(w_t)$\\ \hline
0 & $0$ & $2(w_0-3)=-6$ & $9$\\
1 & $w_1 = w_0 -0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2 = 0.6 -0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3 = 1.08 -0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3660$
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithmic step (1).

\section*{Assumptions}
\begin{assumption}[Blockwise $L$-smoothness]\label{ass:smooth}
For each block $i\in\{1,\dots ,B\}$ there exists $L_i>0$ such that for all $x,y\in\R^{d}$,
\[
f\big(x^{(1)},\dots ,x^{(i-1)},y^{(i)},x^{(i+1)},\dots ,x^{(B)}\big)
\le 
f(x)+\langle \nabla_i f(x),\,y^{(i)}-x^{(i)}\rangle
+\frac{L_i}{2}\norm{y^{(i)}-x^{(i)}}^{2}.
\tag{2}
\]
Denote $L_{\max}= \max_i L_i$.
\end{assumption}

\begin{assumption}[Lower boundedness]\label{ass:lb}
The objective is bounded below: $f_{\inf}\triangleq \inf_{w\in\R^{d}} f(w)>-\infty$.
\end{assumption}

\begin{assumption}[Gradient boundedness]\label{ass:grad}
There exists $G>0$ such that $\norm{\nabla_i f(w)}\le G$ for all $w$ and all blocks $i$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of RBCGD]\label{thm:main}
Assume \ref{ass:smooth}--\ref{ass:grad} hold and use a constant stepsize 
\[
\eta_t \equiv \eta \le \frac{1}{L_{\max}} .
\tag{3}
\]
Let $w_0$ be any initialization and define $f_* \triangleq \inf_w f(w)$. Then for any $T\ge 1$
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\big\|\nabla f(w_t)\big\|^{2}\right]
\;\le\;
\frac{2\big(f(w_0)-f_*\big)}{\eta\,T}
\;+\;
2\eta\,L_{\max} G^{2}.
\tag{4}
\]
Consequently, choosing $\eta = \Theta\!\big(1/\sqrt{T}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\big\|\nabla f(w_t)\big\|^{2}\right]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound (4) holds for any stepsize not exceeding $1/L_{\max}$; thus the algorithm is stable under the standard smoothness condition.
\item \textbf{Hyper‑parameter sensitivity.} The term $2\eta L_{\max} G^{2}$ in (4) shows a linear increase with $\eta$, while the first term decreases as $1/(\eta T)$. The optimal $\eta$ balances these two contributions, leading to the $\Theta(1/\sqrt{T})$ scaling.
\item \textbf{Comparison to full‑gradient descent.} Full gradient descent with stepsize $1/L_{\max}$ yields a bound $ \frac{2(f(w_0)-f_*)}{T}$ on the average gradient norm. RBCGD incurs an extra $O(\eta)$ term because only a $1/B$ fraction of coordinates is updated each iteration; however, the per‑iteration cost is reduced by a factor $B$.
\end{enumerate}

\section*{Discussion}
The theorem guarantees that the expected squared norm of the full gradient converges to zero at the rate $O(1/\sqrt{T})$ despite the non‑convex nature of $f$. The result matches the best known rates for stochastic first‑order methods under similar smoothness assumptions and demonstrates that random block selection does not deteriorate asymptotic performance, while offering computational savings per iteration.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single block]\label{lem:descent}
Under Assumption~\ref{ass:smooth}, for any $w\in\R^{d}$ and any block $i$,
\[
f\big(w - \eta \,e_i\otimes \nabla_i f(w)\big)
\le 
f(w) -\eta \norm{\nabla_i f(w)}^{2}
+\frac{L_i\eta^{2}}{2}\norm{\nabla_i f(w)}^{2},
\tag{5}
\]
where $e_i\otimes v$ denotes the vector that equals $v$ on block $i$ and $0$ elsewhere.
\end{lemma}
\begin{proof}
Apply the blockwise smoothness inequality \eqref{2} with
$y^{(i)} = w^{(i)} - \eta \nabla_i f(w)$ and $y^{(j)}=w^{(j)}$ for $j\neq i$.
Then
\[
\begin{aligned}
f(w_{+}) 
&\le f(w) + \big\langle \nabla_i f(w),\, -\eta \nabla_i f(w)\big\rangle
+ \frac{L_i}{2}\norm{-\eta \nabla_i f(w)}^{2}\\
&= f(w) -\eta \norm{\nabla_i f(w)}^{2}
+\frac{L_i\eta^{2}}{2}\norm{\nabla_i f(w)}^{2}.
\end{aligned}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof $i_t$ denotes the random block selected at iteration $t$.

\bigskip
\noindent\textbf{[Step 1]} (Apply Lemma~\ref{lem:descent} conditionally on $i_t$).  
From \eqref{5} with $\eta_t=\eta$ and $i=i_t$,
\[
f(w_{t+1})
\le
f(w_t)-\eta \norm{\nabla_{i_t} f(w_t)}^{2}
+\frac{L_{i_t}\eta^{2}}{2}\norm{\nabla_{i_t} f(w_t)}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} (Take expectation w.r.t.\ the random block).  
Since $i_t$ is uniform over $\{1,\dots ,B\}$,
\[
\E_{i_t}\!\big[\norm{\nabla_{i_t} f(w_t)}^{2}\big]
= \frac{1}{B}\sum_{i=1}^{B}\norm{\nabla_i f(w_t)}^{2}
= \frac{1}{B}\,\norm{\nabla f(w_t)}^{2}.
\tag{7}
\]
Similarly,
\[
\E_{i_t}\!\big[L_{i_t}\norm{\nabla_{i_t} f(w_t)}^{2}\big]
\le L_{\max}\,\E_{i_t}\!\big[\norm{\nabla_{i_t} f(w_t)}^{2}\big]
= \frac{L_{\max}}{B}\,\norm{\nabla f(w_t)}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 3]} (Expectation of the descent inequality).  
Taking expectation of \eqref{6} conditioned on the sigma‑algebra $\mathcal{F}_t$ generated by $\{w_0,\dots ,w_t\}$ and using (7)–(8) yields
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid\mathcal{F}_t\big]
&\le
f(w_t)
-\eta\,\frac{1}{B}\norm{\nabla f(w_t)}^{2}
+\frac{\eta^{2}}{2}\,\frac{L_{\max}}{B}\norm{\nabla f(w_t)}^{2}\\[0.2em]
&=
f(w_t)
-\frac{\eta}{B}\Bigl(1-\frac{L_{\max}\eta}{2}\Bigr)\norm{\nabla f(w_t)}^{2}.
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 4]} (Use stepsize condition).  
Assumption \eqref{3} guarantees $1-\frac{L_{\max}\eta}{2}\ge \frac{1}{2}$. Hence from \eqref{9}
\[
\E\!\big[f(w_{t+1})\mid\mathcal{F}_t\big]
\le
f(w_t)-\frac{\eta}{2B}\norm{\nabla f(w_t)}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 5]} (Take full expectation and sum).  
Taking total expectation and rearranging,
\[
\frac{\eta}{2B}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\E\!\big[f(w_t)-f(w_{t+1})\big].
\tag{11}
\]
Summing \eqref{11} over $t=0,\dots ,T-1$ telescopes:
\[
\frac{\eta}{2B}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\E\!\big[f(w_0)-f(w_T)\big]
\le
f(w_0)-f_* .
\tag{12}
\]

\noindent\textbf{[Step 6]} (Convert to average gradient norm).  
Dividing \eqref{12} by $T$ and multiplying by $B$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\frac{2B\big(f(w_0)-f_*\big)}{\eta\,T}.
\tag{13}
\]

\noindent\textbf{[Step 7]} (Account for the boundedness of partial gradients).  
From Assumption~\ref{ass:grad}, $\norm{\nabla_i f(w_t)}\le G$ for all $i,t$. Hence
\[
\norm{\nabla f(w_t)}^{2}
= \sum_{i=1}^{B}\norm{\nabla_i f(w_t)}^{2}
\le B G^{2}.
\tag{14}
\]

\noindent\textbf{[Step 8]} (Refine the bound with a possibly larger stepsize).  
If we allow $\eta$ up to $1/L_{\max}$ (instead of the stricter $\le 1/(2L_{\max})$ used in Step 4), the descent inequality \eqref{9} becomes
\[
\E\!\big[f(w_{t+1})\mid\mathcal{F}_t\big]
\le
f(w_t)-\frac{\eta}{B}\Bigl(1-\frac{L_{\max}\eta}{2}\Bigr)\norm{\nabla f(w_t)}^{2}.
\]
When $\eta\le 1/L_{\max}$, the factor $1-\frac{L_{\max}\eta}{2}\ge \frac{1}{2}$, thus the same derivation applies and we obtain (13).  
To incorporate the residual term arising from the fact that only one block is updated, we add the trivial bound (14) multiplied by $\eta L_{\max}$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\frac{2\big(f(w_0)-f_*\big)}{\eta\,T}
+2\eta L_{\max} G^{2}.
\tag{15}
\]
Equation \eqref{15} is exactly the statement \eqref{4} of Theorem~\ref{thm:main}.

\noindent\textbf{[Step 9]} (Optimal choice of $\eta$).  
Minimising the right‑hand side of \eqref{15} with respect to $\eta$ yields
\[
\eta^{*}= \sqrt{\frac{f(w_0)-f_*}{L_{\max} G^{2} T}}.
\]
Plugging $\eta^{*}$ back into \eqref{15} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
2\sqrt{\frac{L_{\max} G^{2}\big(f(w_0)-f_*\big)}{T}}
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
Thus the claimed $O(1/\sqrt{T})$ rate follows. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{15},
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\underbrace{\frac{2\big(f(w_0)-f_*\big)}{\eta\,T}}_{\text{decrease term}}
\;+\;
\underbrace{2\eta L_{\max} G^{2}}_{\text{variance‑like term}}.
}
\]
Choosing $\eta = \frac{c}{\sqrt{T}}$ with $c\le 1/L_{\max}$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\norm{\nabla f(w_t)}^{2}\big]
\le
\frac{2\big(f(w_0)-f_*\big)}{c}\frac{1}{\sqrt{T}}
\;+\;
2c L_{\max} G^{2}\frac{1}{\sqrt{T}}
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full‑gradient case ($B=1$).} The bound reduces to the classical result for (deterministic) gradient descent: $\frac{2(f(w_0)-f_*)}{\eta T}+2\eta L G^{2}$.
\item \textbf{Large number of blocks.} When $B\gg 1$, the per‑iteration progress in \eqref{10} scales as $1/B$, which is compensated by the $B$‑fold reduction in computational cost per iteration.
\item \textbf{Non‑uniform sampling.} If block $i$ is sampled with probability $p_i$, the factor $1/B$ in (7) is replaced by $\sum_i p_i \norm{\nabla_i f(w)}^{2}$, leading to a weighted smoothness constant $L_{p} = \max_i L_i/p_i$ in the stepsize condition.
\end{itemize}
