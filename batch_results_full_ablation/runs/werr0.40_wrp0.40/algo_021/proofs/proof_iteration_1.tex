\section*{Proximal Gradient Method (Forward–Backward Splitting)}

\section*{Mathematical Formulation}
Let 
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\mathbb{R}^{d},
\]
where  

\begin{itemize}[noitemsep]
\item $f:\mathbb{R}^{d}\to\mathbb{R}$ is convex and differentiable with $L$‑Lipschitz continuous gradient,
\item $g:\mathbb{R}^{d}\to\mathbb{R}\cup\{+\infty\}$ is proper, closed, convex (possibly non‑smooth).
\end{itemize}
Given a step size $\eta>0$, the proximal gradient iteration is
\begin{equation}
\boxed{
x_{k+1}= \operatorname{prox}_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr)},
\qquad k=0,1,2,\dots
\label{eq:update}
\end{equation}
where the proximal operator of $g$ is defined by
\[
\operatorname{prox}_{\eta g}(v):=
\arg\min_{u\in\mathbb{R}^{d}}\Bigl\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}\in\mathbb{R}^{d}$, step size $\eta\in(0,1/L]$, max iter $K$.
\FOR{$k=0$ \textbf{to} $K-1$}
    \STATE Compute gradient $g_{k}\gets\nabla f(x_{k})$.
    \STATE Take a gradient step $y_{k}\gets x_{k}-\eta g_{k}$.
    \STATE Apply proximal map $x_{k+1}\gets \operatorname{prox}_{\eta g}(y_{k})$.
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or the best iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^{2},\qquad g\equiv0,
\]
so $\operatorname{prox}_{\eta g}= \operatorname{id}$.  
Take $w_{0}=0$, step size $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\nabla f(w_{k})=2(w_{k}-3)$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values $F(w_{k})=f(w_{k})$ decrease:
\[
f(0)=9,\; f(0.6)=5.76,\; f(1.08)=3.6864,\; f(1.464)=2.230\;(\text{approx}).
\]

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is convex and its gradient is $L$‑Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{ass:gconv}
$g$ is proper, closed, convex (no smoothness required).
\end{assumption}
\begin{assumption}[Step size]\label{ass:stepsize}
The constant step size satisfies $0<\eta\le 1/L$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{eq:update} under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}.  
For any optimal solution $x^{\star}\in\arg\min F$, the iterates satisfy
\begin{equation}
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,k},
\qquad\forall k\ge 1.
\label{eq:rate}
\end{equation}
Consequently $F(x_{k})\to F(x^{\star})$ and the worst‑case convergence rate is $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
\item \textbf{Stability.} The update is a composition of a non‑expansive map (prox) and a $1-\eta L$ contractive gradient step; thus the whole iteration is non‑expansive for $\eta\le 1/L$.
\item \textbf{Hyper‑parameter sensitivity.} If $\eta$ is chosen smaller than $1/L$, the bound \eqref{eq:rate} holds with a larger denominator, yielding slower progress; the optimal choice $\eta=1/L$ gives the smallest constant.
\item \textbf{Comparison.} When $g\equiv0$, the method reduces to gradient descent with the classic $O(1/k)$ rate. When $f\equiv0$, the iteration is a pure proximal point algorithm, which converges in a single step for many simple $g$ (e.g., $\ell_{1}$ penalty).
\end{itemize}

\section*{Discussion}
The theorem follows from a simple telescoping argument that exploits two key ingredients:
\begin{enumerate}[noitemsep]
\item the \emph{descent lemma} for $L$‑smooth $f$,
\item the \emph{non‑expansiveness} of the proximal operator.
\end{enumerate}
Both are stated and proved below, after which we assemble them into the full convergence proof.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient and the Lipschitz bound. $\square$
\end{proof}

\begin{lemma}[Non‑expansiveness of $\operatorname{prox}_{\eta g}$]\label{lem:prox}
For any $u,v\in\mathbb{R}^{d}$,
\[
\|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\|\le\|u-v\|.
\]
\end{lemma}
\begin{proof}
From the optimality condition of the proximal definition,
\[
\frac{1}{\eta}(u-\operatorname{prox}_{\eta g}(u))\in\partial g\bigl(\operatorname{prox}_{\eta g}(u)\bigr),
\quad
\frac{1}{\eta}(v-\operatorname{prox}_{\eta g}(v))\in\partial g\bigl(\operatorname{prox}_{\eta g}(v)\bigr).
\]
Monotonicity of the sub‑differential of a convex function gives
\[
\Big\langle\frac{1}{\eta}(u-\operatorname{prox}_{\eta g}(u))-
\frac{1}{\eta}(v-\operatorname{prox}_{\eta g}(v)),
\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\Big\rangle\ge0,
\]
which after rearranging yields the claimed inequality. $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{\star}\in\arg\min F$ be arbitrary but fixed.  Define the gradient step
\[
y_{k}=x_{k}-\eta\nabla f(x_{k}).
\]
By definition of the proximal map,
\[
x_{k+1}= \operatorname{prox}_{\eta g}(y_{k})\quad\Longleftrightarrow\quad
\frac{1}{\eta}\bigl(y_{k}-x_{k+1}\bigr)\in\partial g(x_{k+1}).
\tag{1}
\]

\noindent\textbf{[Step 1] (Optimality inequality).}  
For any $z\in\mathbb{R}^{d}$, convexity of $g$ and (1) give
\[
g(z)\ge g(x_{k+1})+\Big\langle\frac{1}{\eta}\bigl(y_{k}-x_{k+1}\bigr),\,z-x_{k+1}\Big\rangle.
\tag{2}
\]

\noindent\textbf{[Step 2] (Combine with smoothness of $f$).}  
Apply Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=z$:
\[
f(z)\le f(x_{k})+\langle\nabla f(x_{k}),z-x_{k}\rangle+\frac{L}{2}\|z-x_{k}\|^{2}.
\tag{3}
\]

\noindent\textbf{[Step 3] (Sum (2) and (3)).}  
Adding (2) and (3) yields for any $z$,
\[
\begin{aligned}
F(z)&=f(z)+g(z)\\
&\le f(x_{k})+g(x_{k+1})
+\langle\nabla f(x_{k}),z-x_{k}\rangle
+\Big\langle\frac{1}{\eta}(y_{k}-x_{k+1}),z-x_{k+1}\Big\rangle
+\frac{L}{2}\|z-x_{k}\|^{2}.
\end{aligned}
\tag{4}
\]

\noindent\textbf{[Step 4] (Insert $y_{k}=x_{k}-\eta\nabla f(x_{k})$).}  
Since $y_{k}-x_{k+1}=x_{k}-\eta\nabla f(x_{k})-x_{k+1}$,
\[
\frac{1}{\eta}(y_{k}-x_{k+1})=
\frac{1}{\eta}(x_{k}-x_{k+1})-\nabla f(x_{k}).
\tag{5}
\]

\noindent\textbf{[Step 5] (Cancel gradient terms).}  
Insert (5) into (4). The inner products containing $\nabla f(x_{k})$ cancel:
\[
\begin{aligned}
F(z)&\le f(x_{k})+g(x_{k+1})
+\Big\langle\frac{1}{\eta}(x_{k}-x_{k+1}),\,z-x_{k+1}\Big\rangle
+\frac{L}{2}\|z-x_{k}\|^{2}.
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 6] (Expand the quadratic term).}  
Write $\|z-x_{k}\|^{2}= \|z-x_{k+1}+x_{k+1}-x_{k}\|^{2}$ and expand:
\[
\|z-x_{k}\|^{2}= \|z-x_{k+1}\|^{2}+2\langle z-x_{k+1},x_{k+1}-x_{k}\rangle+\|x_{k+1}-x_{k}\|^{2}.
\tag{7}
\]

\noindent\textbf{[Step 7] (Plug (7) into (6)).}  
After substitution and regrouping,
\[
\begin{aligned}
F(z)&\le f(x_{k})+g(x_{k+1})
+\frac{1}{\eta}\langle x_{k}-x_{k+1},z-x_{k+1}\rangle\\
&\qquad+\frac{L}{2}\|z-x_{k+1}\|^{2}
+L\langle z-x_{k+1},x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 8] (Collect the inner products).}  
Combine the two inner‑product terms:
\[
\Big(\frac{1}{\eta}-L\Big)\langle x_{k}-x_{k+1},z-x_{k+1}\rangle.
\tag{9}
\]

\noindent\textbf{[Step 9] (Use $\eta\le 1/L$).}  
Because $0<\eta\le 1/L$, we have $1/\eta-L\ge0$, therefore
\[
\Big(\frac{1}{\eta}-L\Big)\langle x_{k}-x_{k+1},z-x_{k+1}\rangle
\le \Big(\frac{1}{\eta}-L\Big)\|x_{k}-x_{k+1}\|\|z-x_{k+1}\|
\le \frac{1}{2\eta}\|x_{k}-x_{k+1}\|^{2}
+\frac{1}{2}\Big(\frac{1}{\eta}-L\Big)^{2}\|z-x_{k+1}\|^{2},
\]
where the first inequality is Cauchy–Schwarz and the second follows from $2ab\le a^{2}+b^{2}$.

\noindent\textbf{[Step 10] (Upper bound for $F(z)$).}  
Insert the bound from Step 9 into (8) and simplify, noting that $f(x_{k})+g(x_{k+1})\le F(x_{k+1})$ by definition of $F$:
\[
F(z)\le F(x_{k+1})+\frac{1}{2\eta}\|x_{k}-x_{k+1}\|^{2}
+\frac{L}{2}\|z-x_{k+1}\|^{2}
+\frac{1}{2}\Big(\frac{1}{\eta}-L\Big)^{2}\|z-x_{k+1}\|^{2}
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{10}
\]

\noindent\textbf{[Step 11] (Cancel the duplicate quadratic terms).}  
Observe that
\[
\frac{1}{2\eta}\|x_{k}-x_{k+1}\|^{2}+ \frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
=\frac{1}{2}\Big(\frac{1}{\eta}+L\Big)\|x_{k+1}-x_{k}\|^{2}.
\]
Since $\eta\le 1/L$, we have $\frac{1}{\eta}+L\le \frac{2}{\eta}$, whence
\[
\frac{1}{2}\Big(\frac{1}{\eta}+L\Big)\|x_{k+1}-x_{k}\|^{2}\le \frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 12] (Choose $z=x^{\star}$).}  
Setting $z=x^{\star}$ in (10) and using (11) gives
\[
F(x^{\star})\le F(x_{k+1})+\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2}\Big(\frac{1}{\eta}-L\Big)^{2}\|x^{\star}-x_{k+1}\|^{2}
+\frac{L}{2}\|x^{\star}-x_{k+1}\|^{2}.
\tag{12}
\]

\noindent\textbf{[Step 13] (Rearrange to obtain a descent inequality).}  
Move $F(x^{\star})$ to the left and collect the norm terms:
\[
F(x_{k+1})-F(x^{\star})
\le -\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2}\Big(L-\frac{1}{\eta}\Big)\|x^{\star}-x_{k+1}\|^{2}
-\frac{L}{2}\|x^{\star}-x_{k+1}\|^{2}.
\]
Because $L\le 1/\eta$, the coefficient of $\|x^{\star}-x_{k+1}\|^{2}$ is non‑positive, thus we can drop it and obtain the clean inequality
\[
F(x_{k+1})-F(x^{\star})\le -\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{13}
\]

\noindent\textbf{[Step 14] (Relate the distance to the optimal point).}  
From Lemma~\ref{lem:prox} (non‑expansiveness) applied to $u=y_{k}$ and $v=x^{\star}$,
\[
\|x_{k+1}-x^{\star}\|\le\|y_{k}-x^{\star}\|
 =\|x_{k}-\eta\nabla f(x_{k})-x^{\star}\|.
\]
Expanding the square and using the convexity of $f$ gives
\[
\|x_{k+1}-x^{\star}\|^{2}\le\|x_{k}-x^{\star}\|^{2}
-2\eta\bigl(F(x_{k})-F(x^{\star})\bigr)+\eta^{2}L^{2}\|x_{k}-x^{\star}\|^{2}.
\]
Since $\eta L\le1$, we obtain
\[
\|x_{k+1}-x^{\star}\|^{2}\le\|x_{k}-x^{\star}\|^{2}
-2\eta\bigl(F(x_{k})-F(x^{\star})\bigr).
\tag{14}
\]

\noindent\textbf{[Step 15] (Telescoping sum).}  
Rearrange (14):
\[
2\eta\bigl(F(x_{k})-F(x^{\star})\bigr)\le
\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}.
\tag{15}
\]
Summing (15) from $k=0$ to $k=K-1$ yields
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{\star})\bigr)
\le\|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}
\le\|x_{0}-x^{\star}\|^{2}.
\tag{16}
\]

\noindent\textbf{[Step 16] (Derive the $O(1/k)$ rate).}  
Since $F(x_{k})$ is non‑increasing (from (13)), the average of the first $K$ values bounds each term:
\[
F(x_{K})-F(x^{\star})\le\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k})-F(x^{\star})\bigr)
\le\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{17}
\]
Inequality \eqref{eq:rate} follows, completing the proof. $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:rate} we have the explicit constant
\[
C:=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta},
\]
so that
\[
F(x_{k})-F(x^{\star})\le \frac{C}{k},\qquad k\ge1.
\]
If the optimal step size $\eta=1/L$ is used, the constant simplifies to $C= \frac{L}{2}\|x_{0}-x^{\star}\|^{2}$.

\section*{Special Cases}
\begin{enumerate}[noitemsep]
\item \textbf{Pure gradient descent ($g\equiv0$).}  The proximal operator becomes the identity, and the proof reduces to the classic $O(1/k)$ result for smooth convex $f$.
\item \textbf{Pure proximal point ($f\equiv0$).}  The update is $x_{k+1}= \operatorname{prox}_{\eta g}(x_{k})$, which is a non‑expansive map; the sequence converges weakly to a minimizer of $g$.
\item \textbf{Strongly convex $F$.}  If $F$ is $\mu$‑strongly convex, a similar derivation yields a linear rate
\[
F(x_{k})-F(x^{\star})\le \bigl(1-\eta\mu\bigr)^{k}\bigl(F(x_{0})-F(x^{\star})\bigr).
\]
\end{enumerate}
