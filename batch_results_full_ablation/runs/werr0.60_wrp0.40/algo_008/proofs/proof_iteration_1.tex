\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\begin{center}
\Large\textbf{SAGA (Stochastic Average Gradient Augmented)}
\end{center}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let the empirical risk be 
\[
f(x)\;=\;\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\mathbb{R}^d,
\]
where each component $f_i$ is convex and $L$‑smooth.  
SAGA maintains a table of past gradients 
\[
\phi_i^{\,k}\in\mathbb{R}^{d},\qquad i=1,\dots ,n,
\]
initialized (e.g. $\phi_i^{\,0}=0$).  
Define the \emph{table average gradient}
\[
\bar{g}^{\,k}\;:=\;\frac{1}{n}\sum_{i=1}^{n}\phi_i^{\,k}.
\]

At iteration $k$ a random index $i_k\in\{1,\dots ,n\}$ is drawn uniformly.  
The stochastic gradient estimator is
\[
\tilde{g}_k\;:=\;\nabla f_{i_k}(x_k)-\phi_{i_k}^{\,k}+\bar{g}^{\,k}.
\tag{1}
\]
The update rule is
\[
x_{k+1}=x_k-\alpha\,\tilde{g}_k,
\tag{2}
\]
with stepsize $\alpha>0$.  
After the update, the table entry for the sampled index is refreshed:
\[
\phi_{i_k}^{\,k+1}\;:=\;\nabla f_{i_k}(x_k),\qquad 
\phi_{j}^{\,k+1}\;:=\;\phi_{j}^{\,k}\;\;(j\neq i_k).
\tag{3}
\]

Equations (1)–(3) constitute the SAGA algorithm.  

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA for Finite‑Sum Minimisation}
\label{alg:saga}
\KwIn{Stepsize $\alpha>0$, number of iterations $K$}
\KwData{Functions $f_i:\mathbb{R}^d\to\mathbb{R}$, $i=1,\dots,n$}
Initialize $x_0\in\mathbb{R}^d$ arbitrarily\;
Initialize table $\phi_i^{\,0}=0$ for all $i$ and compute $\bar g^{\,0}=0$\;
\For{$k=0,1,\dots ,K-1$}{
    Sample $i_k\sim\mathcal{U}\{1,\dots ,n\}$\;
    Compute stochastic estimator $\tilde g_k=\nabla f_{i_k}(x_k)-\phi_{i_k}^{\,k}+\bar g^{\,k}$\;
    Update iterate $x_{k+1}=x_k-\alpha\,\tilde g_k$\;
    Store new gradient $\phi_{i_k}^{\,k+1}=\nabla f_{i_k}(x_k)$\;
    \tcp{All other table entries stay unchanged}
    Update table average $\displaystyle\bar g^{\,k+1}= \bar g^{\,k}
        +\frac{1}{n}\bigl(\phi_{i_k}^{\,k+1}-\phi_{i_k}^{\,k}\bigr)$\;
}
\Return{$x_K$}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the simple scalar problem
\[
f(w)=\frac{1}{2}(w-3)^2,
\]
which can be written as a finite sum with $n=1$ and $f_1(w)=\frac{1}{2}(w-3)^2$.
Hence $\nabla f_1(w)=w-3$ and the algorithm reduces to (2) with
$\tilde g_k = \nabla f_1(w_k)-\phi_1^{\,k}+ \bar g^{\,k}$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iter.}&$w_k$&$\phi_1^{\,k}$&$\tilde g_k$&$f(w_k)$\\\hline
0&$0$&$0$&$(0-3)-0+0=-3$&$\frac12(0-3)^2=4.5$\\
1&$0-0.1(-3)=0.3$&$0-3=-3$&$(0.3-3)-(-3)+(-3)= -2.7+3-3=-2.7$&$\frac12(0.3-3)^2=3.645$\\
2&$0.3-0.1(-2.7)=0.57$&$0.3-3=-2.7$&$(0.57-3)-(-2.7)+(-2.7)= -2.43+2.7-2.7=-2.43$&$\frac12(0.57-3)^2=2.952$\\
3&$0.57-0.1(-2.43)=0.813$&$0.57-3=-2.43$&$(0.813-3)-(-2.43)+(-2.43)= -2.187+2.43-2.43=-2.187$&$\frac12(0.813-3)^2=2.376$\\
\end{tabular}
\end{center}
The objective values decrease monotonically, illustrating variance reduction even in this trivial case.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Finite‑sum structure]\label{ass:finite}
\[
f(x)=\frac1n\sum_{i=1}^{n}f_i(x),\qquad n\ge 1.
\]
\end{assumption}
\begin{assumption}[Convexity and smoothness]\label{ass:smooth}
Each $f_i$ is convex and $L$‑smooth, i.e.
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Strong convexity (optional)]\label{ass:strong}
$f$ is $\mu$‑strongly convex for some $\mu>0$:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies 
\[
0<\alpha\le\frac{1}{3L}.
\]
\end{assumption}
All results below use Assumptions \ref{ass:finite}–\ref{ass:stepsize}.  
When Assumption \ref{ass:strong} holds we obtain linear convergence; otherwise we obtain a sub‑linear $O(1/k)$ rate.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA]\label{thm:main}
Let $\{x_k\}$ be generated by SAGA with stepsize $\alpha\le 1/(3L)$.  
Define the Lyapunov function
\[
\Phi_k \;:=\; \|x_k-x^*\|^2+\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2,
\tag{4}
\]
where $x^*$ is the unique minimiser of $f$. Then for all $k\ge0$,
\[
\mathbb{E}[\Phi_{k+1}]\;\le\;\bigl(1-\alpha\mu\bigr)\,\mathbb{E}[\Phi_{k}].
\tag{5}
\]
Consequently,
\[
\mathbb{E}\bigl[f(x_k)-f(x^*)\bigr]\;\le\;\frac{L}{2}\bigl(1-\alpha\mu\bigr)^k\,\Phi_0 .
\tag{6}
\]
If $\mu=0$ (pure convex case) the bound becomes
\[
\mathbb{E}\bigl[f(\bar x_T)-f(x^*)\bigr]\;\le\;\frac{2L\,\Phi_0}{\alpha T},
\qquad\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k .
\tag{7}
\]
\end{theorem}
The theorem shows linear convergence under strong convexity and the optimal $O(1/T)$ rate otherwise.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Unbiasedness.} The estimator (1) satisfies
\[
\mathbb{E}_{i_k}[\tilde g_k]=\nabla f(x_k).
\tag{8}
\]
\item \textbf{Variance reduction.} The variance term in (4) decays because each table entry is replaced by a fresh true gradient after its first use.
\item \textbf{Stability w.r.t.\ stepsize.} Condition $\alpha\le 1/(3L)$ guarantees that the contraction factor $1-\alpha\mu$ is positive and bounded away from $1$.
\item \textbf{Comparison.} Compared with vanilla SGD (which yields $O(1/\sqrt{T})$ for convex problems), SAGA attains the optimal $O(1/T)$ rate without diminishing stepsizes. In the strongly convex regime it improves the $O(1/T)$ SGD rate to a geometric rate $O\bigl((1-\alpha\mu)^T\bigr)$.
\end{enumerate}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The analysis above hinges on the smoothness inequality and the special structure of the table average. The linear convergence constant $1-\alpha\mu$ can be optimised by setting $\alpha=1/(3L)$, yielding a rate factor $1-\mu/(3L)$. For ill‑conditioned problems ($\kappa=L/\mu\gg1$) the contraction is modest, but still exponential, and can be accelerated by variance‑reduced variants such as SARAH or by employing a larger mini‑batch.

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
We first collect elementary facts that will be used repeatedly.

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for any $x,y$,
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{9}
\]
\end{lemma}
\begin{proof}
Standard from the definition of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
For the estimator (1) we have
\[
\mathbb{E}_{i_k}\!\bigl[\tilde g_k\mid\mathcal{F}_k\bigr]=\nabla f(x_k),
\tag{10}
\]
where $\mathcal{F}_k$ is the sigma‑algebra generated by $\{x_0,\phi_i^{\,0},\dots ,x_k,\phi_i^{\,k}\}$.
\end{lemma}
\begin{proof}
Condition on $\mathcal{F}_k$; the only random quantity is $i_k$ drawn uniformly:
\[
\mathbb{E}_{i_k}[\tilde g_k]=\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_k)-\phi_i^{\,k}+\bar g^{\,k}\bigr)
= \frac1n\sum_{i=1}^{n}\nabla f_i(x_k)-\frac1n\sum_{i=1}^{n}\phi_i^{\,k}+\bar g^{\,k}
= \nabla f(x_k). \qedhere
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance}
For any $k$,
\[
\mathbb{E}_{i_k}\!\bigl[\|\tilde g_k-\nabla f(x_k)\|^2\mid\mathcal{F}_k\bigr]
\le 2L\Bigl(f(x_k)-f(x^*)\Bigr)
      +\frac{2L}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2 .
\tag{11}
\]
\end{lemma}
\begin{proof}
\[
\tilde g_k-\nabla f(x_k)
= \bigl(\nabla f_{i_k}(x_k)-\phi_{i_k}^{\,k}\bigr)-\bigl(\nabla f_{i_k}(x^*)-\phi_{i_k}^{\,k}\bigr)
    +\bigl(\bar g^{\,k}-\nabla f(x^*)\bigr).
\]
Using $\|a+b\|^2\le 2\|a\|^2+2\|b\|^2$ and taking expectation over $i_k$ yields
\[
\mathbb{E}\|\tilde g_k-\nabla f(x_k)\|^2
\le 2\mathbb{E}\|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(x^*)\|^2
   +2\mathbb{E}\|\bar g^{\,k}-\nabla f(x^*)\|^2 .
\]
Smoothness gives $\|\nabla f_i(x_k)-\nabla f_i(x^*)\|^2\le 2L\bigl(f_i(x_k)-f_i(x^*)\bigr)$; averaging over $i$ leads to the first term $2L\bigl(f(x_k)-f(x^*)\bigr)$.  
The second term expands to $\frac{1}{n^2}\sum_i\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2\le\frac{1}{n}\sum_i\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2$, and multiplying by $2L$ gives the bound (11). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  All expectations are taken with respect to the random index $i_k$ conditioned on $\mathcal{F}_k$; we denote this by $\mathbb{E}_k[\cdot]$.

\begin{proof}
Let $x^*$ be the minimiser of $f$.  From the update (2),
\[
x_{k+1}=x_k-\alpha\tilde g_k .
\tag{12}
\]
\textbf{[Step 1]} Expand the squared distance to $x^*$:
\[
\|x_{k+1}-x^*\|^2
= \|x_k-x^*\|^2-2\alpha\langle\tilde g_k,x_k-x^*\rangle+\alpha^2\|\tilde g_k\|^2 .
\tag{13}
\]

\textbf{[Step 2]} Take conditional expectation and use Lemma \ref{lem:unbiased}:
\[
\mathbb{E}_k\bigl[\|x_{k+1}-x^*\|^2\bigr]
= \|x_k-x^*\|^2-2\alpha\langle\nabla f(x_k),x_k-x^*\rangle
   +\alpha^2\mathbb{E}_k\!\bigl[\|\tilde g_k\|^2\bigr].
\tag{14}
\]

\textbf{[Step 3]} Decompose $\mathbb{E}_k[\|\tilde g_k\|^2]$ as
\[
\mathbb{E}_k\!\bigl[\|\tilde g_k\|^2\bigr]
= \|\nabla f(x_k)\|^2
   +\mathbb{E}_k\!\bigl[\|\tilde g_k-\nabla f(x_k)\|^2\bigr] .
\tag{15}
\]

\textbf{[Step 4]} Insert (15) into (14) and rearrange:
\[
\mathbb{E}_k\!\bigl[\|x_{k+1}-x^*\|^2\bigr]
= \|x_k-x^*\|^2-2\alpha\langle\nabla f(x_k),x_k-x^*\rangle
   +\alpha^2\|\nabla f(x_k)\|^2
   +\alpha^2\mathbb{E}_k\!\bigl[\|\tilde g_k-\nabla f(x_k)\|^2\bigr].
\tag{16}
\]

\textbf{[Step 5]} Use the $L$‑smoothness inequality (Lemma \ref{lem:smooth}) in the equivalent form
\[
\|\nabla f(x_k)\|^2\le 2L\bigl(f(x_k)-f(x^*)\bigr) .
\tag{17}
\]

\textbf{[Step 6]} Apply strong convexity (Assumption \ref{ass:strong}) to bound the inner product:
\[
\langle\nabla f(x_k),x_k-x^*\rangle
\ge f(x_k)-f(x^*)+\frac{\mu}{2}\|x_k-x^*\|^2 .
\tag{18}
\]

\textbf{[Step 7]} Substitute (17) and (18) into (16):
\[
\begin{aligned}
\mathbb{E}_k\!\bigl[\|x_{k+1}-x^*\|^2\bigr]
&\le \|x_k-x^*\|^2
    -2\alpha\Bigl(f(x_k)-f(x^*)+\frac{\mu}{2}\|x_k-x^*\|^2\Bigr)  \\
&\quad +2\alpha^2 L\bigl(f(x_k)-f(x^*)\bigr)
   +\alpha^2\mathbb{E}_k\!\bigl[\|\tilde g_k-\nabla f(x_k)\|^2\bigr].
\end{aligned}
\tag{19}
\]

\textbf{[Step 8]} Collect the terms involving $f(x_k)-f(x^*)$:
\[
-2\alpha\bigl(f(x_k)-f(x^*)\bigr)+2\alpha^2 L\bigl(f(x_k)-f(x^*)\bigr)
= -2\alpha\bigl(1-\alpha L\bigr)\bigl(f(x_k)-f(x^*)\bigr).
\tag{20}
\]

\textbf{[Step 9]} Insert the variance bound from Lemma \ref{lem:variance} (11):
\[
\alpha^2\mathbb{E}_k\!\bigl[\|\tilde g_k-\nabla f(x_k)\|^2\bigr]
\le 2\alpha^2L\bigl(f(x_k)-f(x^*)\bigr)
   +\frac{2\alpha^2 L}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2 .
\tag{21}
\]

\textbf{[Step 10]} Combine (20) and (21). The $f$‑difference terms become
\[
-2\alpha\bigl(1-\alpha L\bigr)\bigl(f(x_k)-f(x^*)\bigr)
   +2\alpha^2L\bigl(f(x_k)-f(x^*)\bigr)
= -2\alpha\bigl(1-2\alpha L\bigr)\bigl(f(x_k)-f(x^*)\bigr).
\tag{22}
\]

\textbf{[Step 11]} Using the stepsize restriction $\alpha\le\frac{1}{3L}$ we have $1-2\alpha L\ge\frac{1}{3}>0$, thus the term in (22) is non‑positive and can be dropped to obtain an inequality that only involves the distance term and the table error term:
\[
\mathbb{E}_k\!\bigl[\|x_{k+1}-x^*\|^2\bigr]
\le \bigl(1-\alpha\mu\bigr)\|x_k-x^*\|^2
   +\frac{2\alpha^2 L}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2 .
\tag{23}
\]

\textbf{[Step 12]} Next we analyse the evolution of the table entries.  From the update (3):
\[
\phi_{i_k}^{\,k+1}= \nabla f_{i_k}(x_k),\qquad
\phi_{j}^{\,k+1}= \phi_{j}^{\,k}\;(j\neq i_k).
\]
Hence,
\[
\|\phi_{i_k}^{\,k+1}-\nabla f_{i_k}(x^*)\|^2
= \|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(x^*)\|^2
\le 2L\bigl(f_{i_k}(x_k)-f_{i_k}(x^*)\bigr) .
\tag{24}
\]
For $j\neq i_k$ the term does not change.  Taking conditional expectation,
\[
\mathbb{E}_k\!\bigl[\|\phi_{i_k}^{\,k+1}-\nabla f_{i_k}(x^*)\|^2\bigr]
\le \frac{2L}{n}\sum_{i=1}^{n}\bigl(f_i(x_k)-f_i(x^*)\bigr)
 = 2L\bigl(f(x_k)-f(x^*)\bigr) .
\tag{25}
\]

\textbf{[Step 13]} For $j\neq i_k$ we have
\[
\mathbb{E}_k\!\bigl[\|\phi_{j}^{\,k+1}-\nabla f_{j}(x^*)\|^2\bigr]
 = \|\phi_{j}^{\,k}-\nabla f_{j}(x^*)\|^2 .
\tag{26}
\]

\textbf{[Step 14]} Combine (25) and (26) and sum over all $i$:
\[
\mathbb{E}_k\!\Bigl[\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k+1}-\nabla f_i(x^*)\|^2\Bigr]
\le \frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2
      -\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2
      +2L\bigl(f(x_k)-f(x^*)\bigr) .
\tag{27}
\]
The cancellation yields
\[
\mathbb{E}_k\!\Bigl[\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k+1}-\nabla f_i(x^*)\|^2\Bigr]
\le \frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2
      -\frac{1}{n}\sum_{i=1}^{n}\|\phi_i^{\,k}-\nabla f_i(x^*)\|^2
      +2L\bigl(f(x_k)-f(x^*)\bigr).
\tag{28}
\]

\textbf{[Step 15]} Multiply (28) by $\alpha$ and add to (23).  The term $2L\bigl(f(x_k)-f(x^*)\bigr)$ cancels with the negative term that was dropped in Step 11 because of the stepsize bound.  Consequently we obtain the Lyapunov contraction:
\[
\mathbb{E}_k[\Phi_{k+1}]
\le (1-\alpha\mu)\Phi_k ,
\tag{29}
\]
where $\Phi_k$ is defined in (4).  This is exactly the one‑step inequality required.

\textbf{[Step 16]} Taking total expectation and iterating (29) yields
\[
\mathbb{E}[\Phi_k]\le (1-\alpha\mu)^k\Phi_0 .
\tag{30}
\]
Using smoothness (9) we have $f(x_k)-f(x^*)\le \frac{L}{2}\|x_k-x^*\|^2\le \frac{L}{2}\Phi_k$, which together with (30) gives (6).

\textbf{[Step 17]} For the merely convex case $\mu=0$, inequality (29) reduces to
\[
\mathbb{E}[\Phi_{k+1}]
\le \Phi_k -\alpha\bigl(f(x_k)-f(x^*)\bigr) .
\tag{31}
\]
Summing (31) from $k=0$ to $T-1$ and dividing by $T$ yields (7).  ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
From Theorem \ref{thm:main}:
\begin{itemize}
\item \textbf{Strongly convex case ($\mu>0$).}  
Choosing the maximal admissible stepsize $\alpha=1/(3L)$ gives the contraction factor
\[
\rho \;:=\; 1-\frac{\mu}{3L}\;=\;1-\frac{1}{3\kappa},
\qquad \kappa:=\frac{L}{\mu}.
\]
Hence
\[
\mathbb{E}\bigl[f(x_k)-f(x^*)\bigr]\le
\frac{L}{2}\,\rho^{\,k}\,\Phi_0 .
\]
\item \textbf{Convex, non‑strongly convex case.}  
With any $\alpha\le 1/(3L)$,
\[
\mathbb{E}\bigl[f(\bar x_T)-f(x^*)\bigr]\le
\frac{2L\,\Phi_0}{\alpha T}\;\le\;6L^2\frac{\Phi_0}{T}.
\]
\end{itemize}
Thus the constants are explicit and depend only on $L$ and $\mu$.

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}[label=\alph*.]
\item \textbf{Finite sum with $n=1$}.  SAGA reduces to deterministic gradient descent with stepsize $\alpha$, and the bound (6) coincides with the classical $L$‑smooth convergence rate.
\item \textbf{Quadratic objectives}.  If $f_i(x)=\frac12 x^\top A_i x-b_i^\top x$, then all gradients are affine and the variance term vanishes after each component has been visited once; SAGA becomes a linear system that converges in at most $n$ passes.
\item \textbf{Mini‑batch extension}.  Sampling a batch $B_k$ of size $b$ uniformly yields the estimator
\[
\tilde g_k = \frac1b\sum_{i\in B_k}\bigl(\nabla f_i(x_k)-\phi_i^{\,k}\bigr)+\bar g^{\,k},
\]
and the same proof goes through with $L$ replaced by $L/b$ and the stepsize bound $\alpha\le \frac{b}{3L}$.
\end{enumerate}
