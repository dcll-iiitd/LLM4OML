\section*{Algorithm Name}
\textbf{Randomized Coordinate Descent with Importance Sampling (RCD‑IS)}  

The algorithm selects at each iteration a coordinate $i_k\in\{1,\dots,n\}$ according to a fixed probability vector $p=(p_1,\dots,p_n)$, $p_i>0$, $\sum_{i=1}^n p_i=1$, and performs a scaled coordinate step.

\section*{Mathematical Formulation}
Let $w_k\in\RR^{n}$ be the iterate at step $k$.
\begin{enumerate}
\item Sample $i_k\sim\mathcal{P}$ where $\mathbb{P}(i_k=i)=p_i$.
\item Compute the \emph{partial gradient} $\displaystyle g_{i_k}(w_k)=\frac{\partial f}{\partial w_{i_k}}(w_k)$.
\item Update only the selected coordinate:
\[
w_{k+1}=w_k-\frac{\eta}{p_{i_k}}\,g_{i_k}(w_k)\,e_{i_k},
\tag{1}
\]
where $e_{i}$ is the $i$‑th canonical basis vector and $\eta>0$ is a stepsize (learning rate).
\end{enumerate}
The factor $1/p_{i_k}$ compensates for the non‑uniform sampling and will appear in the expectation analysis.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD‑IS}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\RR^n$, stepsize $\eta>0$, probability vector $p\in\RR^n$ with $p_i>0$, $\sum_i p_i=1$, maximal iteration $T$.
\FOR{$k=0,\dots,T-1$}
    \STATE Sample $i_k\in\{1,\dots,n\}$ with $\Pr(i_k=i)=p_i$.
    \STATE Compute partial derivative $g_{i_k}(w_k)=\partial_i f(w_k)$.
    \STATE $w_{k+1}=w_k-\dfrac{\eta}{p_{i_k}}\;g_{i_k}(w_k)\;e_{i_k}$.
\ENDFOR
\RETURN $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $n=1$,
\[
f(w)=(w-3)^2,\qquad w_0=0,
\]
with probability $p_1=1$, stepsize $\eta=0.1$.

\begin{tabular}{c|c|c|c}
$k$ & $g_{1}(w_k)=\partial f(w_k)$ & $w_{k+1}=w_k-\eta g_{1}(w_k)$ & $f(w_{k+1})$\\\hline
0 & $2(w_0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3689$\\
\end{tabular}
The objective value decreases at each iteration, illustrating the descent property.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L_i$‑smoothness]\label{ass:smooth}
For each coordinate $i\in\{1,\dots,n\}$ there exists $L_i>0$ such that for all $x\in\RR^n$ and any scalar $\alpha$,
\[
\Bigl| \frac{\partial f}{\partial w_i}(x+\alpha e_i)-\frac{\partial f}{\partial w_i}(x) \Bigr|
\le L_i |\alpha|.
\tag{2}
\]
Equivalently,
\[
f(x+\alpha e_i)\le f(x)+\alpha\,\frac{\partial f}{\partial w_i}(x)+\frac{L_i}{2}\alpha^2 .
\tag{3}
\]
\end{assumption}
\begin{assumption}[Bounded below]\label{ass:lower}
The objective has a finite infimum $f^{\inf}>-\infty$ and we denote $f^{*}= \inf_{w} f(w)$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies 
\[
0<\eta\le \frac{1}{\max_{i}L_i}.
\tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected gradient norm decay]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize}, let $\{w_k\}_{k\ge0}$ be generated by the RCD‑IS algorithm. Define the averaged iterate
\[
\bar w_T \;\triangleq\; \frac{1}{T}\sum_{k=0}^{T-1} w_k .
\]
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\norm{\nabla f(w_k)}^{2}\right]
\;\le\;
\frac{2\bigl(f(w_0)-f^{*}\bigr)}{\eta\,T}
\;+\;
\eta\;\sum_{i=1}^{n}\frac{L_i}{p_i}\; .
\tag{5}
\]
Consequently, if $\eta$ is chosen as $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$, the right‑hand side becomes $O\!\bigl(1/\sqrt{T}\bigr)$, i.e.
\[
\min_{0\le k<T}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr]=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition \eqref{ass:stepsize} guarantees that the term multiplying $\norm{\nabla f(w_k)}^{2}$ in the descent inequality is non‑negative, preventing divergence.
\item \textbf{Hyperparameter Sensitivity.} The bound \eqref{5} depends linearly on $\eta$ and on the weighted smoothness constants $\frac{L_i}{p_i}$. Larger sampling probabilities $p_i$ for coordinates with large $L_i$ reduce the second term.
\item \textbf{Comparison to Uniform Sampling.} If $p_i=1/n$, the bound recovers the classical $O\!\bigl((L_{\max}n)/T\bigr)$ rate for uniform coordinate descent. Importance sampling can strictly improve the constant by aligning $p_i\propto \sqrt{L_i}$ (see discussion below).
\end{itemize}

\section*{Discussion}
The theorem shows that, despite non‑convexity, RCD‑IS yields a sublinear decay of the expected squared gradient norm, which is the standard stationarity criterion for non‑convex optimization. The presence of the factor $1/p_i$ in the update (1) is crucial: it exactly cancels the bias introduced by non‑uniform sampling, as can be seen in the expectation calculations of the proof.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate Descent Lemma]\label{lem:descent}
For any $w\in\RR^{n}$ and any coordinate $i$, the $L_i$‑smoothness \eqref{ass:smooth} implies
\[
f\!\bigl(w-\alpha e_i\bigr)
\le f(w)-\alpha\,\frac{\partial f}{\partial w_i}(w)+\frac{L_i}{2}\alpha^{2},
\qquad\forall\alpha\in\RR .
\tag{6}
\]
\end{lemma}
\begin{proof}
Immediate from the equivalent form \eqref{3} of Assumption \ref{ass:smooth}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken with respect to the random choice of $i_k$ conditioned on the history up to $w_k$; we denote this conditional expectation by $\EE_k[\cdot]=\EE[\cdot\mid w_k]$.

\subsection*{Step 1 – One‑step descent inequality}
Using Lemma \ref{lem:descent} with $\alpha = \eta g_{i_k}(w_k)/p_{i_k}$ we obtain
\[
\begin{aligned}
f(w_{k+1})
&= f\!\Bigl(w_k-\frac{\eta}{p_{i_k}}g_{i_k}(w_k)\,e_{i_k}\Bigr) \\
&\le f(w_k)-\frac{\eta}{p_{i_k}}g_{i_k}(w_k)^{2}
      +\frac{L_{i_k}}{2}\Bigl(\frac{\eta}{p_{i_k}}g_{i_k}(w_k)\Bigr)^{2}.
\end{aligned}
\tag{7}
\]
\textbf{[Step 1]}

\subsection*{Step 2 – Conditional expectation}
Take $\EE_k[\cdot]$ on both sides of \eqref{7}. Since $i_k$ is the only random quantity we have
\[
\EE_k\!\bigl[f(w_{k+1})\bigr]\le
f(w_k)-\eta\sum_{i=1}^{n}\frac{1}{p_i}p_i g_i(w_k)^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{n}\frac{L_i}{p_i^{2}}p_i g_i(w_k)^{2}.
\tag{8}
\]
Simplifying the probabilities gives
\[
\EE_k\!\bigl[f(w_{k+1})\bigr]\le
f(w_k)-\eta\sum_{i=1}^{n} g_i(w_k)^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{n}\frac{L_i}{p_i} g_i(w_k)^{2}.
\tag{9}
\]
\textbf{[Step 2]}

\subsection*{Step 3 – Relating to full gradient norm}
Recall $\norm{\nabla f(w_k)}^{2}= \sum_{i=1}^{n} g_i(w_k)^{2}$. Hence \eqref{9} can be written compactly as
\[
\EE_k\!\bigl[f(w_{k+1})\bigr]\le
f(w_k)-\Bigl(\eta-\frac{\eta^{2}}{2}\max_{i}\frac{L_i}{p_i}\Bigr)\norm{\nabla f(w_k)}^{2}.
\tag{10}
\]
\textbf{[Step 3]}

\subsection*{Step 4 – Positive descent coefficient}
By Assumption \ref{ass:stepsize}, $\eta\le 1/\max_i L_i$. Since $p_i\le 1$, we have
\[
\frac{L_i}{p_i}\le \frac{L_{\max}}{\min_j p_j}\le \frac{1}{\eta\,\min_j p_j},
\]
which implies
\[
\eta-\frac{\eta^{2}}{2}\max_i\frac{L_i}{p_i}\;\ge\;\frac{\eta}{2}.
\tag{11}
\]
Thus the coefficient in front of $\norm{\nabla f(w_k)}^{2}$ in \eqref{10} is at least $\eta/2>0$.
\textbf{[Step 4]}

\subsection*{Step 5 – One‑step expected descent}
Insert \eqref{11} into \eqref{10}:
\[
\EE_k\!\bigl[f(w_{k+1})\bigr]\le
f(w_k)-\frac{\eta}{2}\norm{\nabla f(w_k)}^{2}.
\tag{12}
\]
\textbf{[Step 5]}

\subsection*{Step 6 – Unconditional expectation and telescoping sum}
Take full expectation and sum \eqref{12} for $k=0,\dots,T-1$:
\[
\EE\!\bigl[f(w_T)\bigr]\le f(w_0)-\frac{\eta}{2}\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr].
\tag{13}
\]
Rearranging yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr]
\le \frac{2\bigl(f(w_0)-\EE[f(w_T)]\bigr)}{\eta\,T}
\le \frac{2\bigl(f(w_0)-f^{*}\bigr)}{\eta\,T}.
\tag{14}
\]
\textbf{[Step 6]}

\subsection*{Step 7 – Adding the bias term from the scaling $1/p_i$}
The inequality \eqref{12} ignored the second term in \eqref{9} that does not cancel completely when the bound \eqref{11} is loose. Keeping that term gives a refined bound:
\[
\EE_k\!\bigl[f(w_{k+1})\bigr]\le
f(w_k)-\eta\norm{\nabla f(w_k)}^{2}
+\frac{\eta^{2}}{2}\sum_{i=1}^{n}\frac{L_i}{p_i} g_i(w_k)^{2}.
\tag{15}
\]
Taking full expectation, summing over $k$, and using $\norm{g(w_k)}^{2}=\sum_i g_i(w_k)^{2}$,
\[
\EE\!\bigl[f(w_T)\bigr]\le f(w_0)-\eta\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr]
+\frac{\eta^{2}}{2}\sum_{i=1}^{n}\frac{L_i}{p_i}\sum_{k=0}^{T-1}\EE\!\bigl[g_i(w_k)^{2}\bigr].
\tag{16}
\]
Since $g_i(w_k)^{2}\le \norm{\nabla f(w_k)}^{2}$ for every $i$, we can bound the double sum by $\sum_{k}\EE[\norm{\nabla f(w_k)}^{2}]$ and obtain
\[
\EE\!\bigl[f(w_T)\bigr]\le f(w_0)-\Bigl(\eta-\frac{\eta^{2}}{2}\sum_{i=1}^{n}\frac{L_i}{p_i}\Bigr)
\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr].
\tag{17}
\]
Rearranging,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^{*}\bigr)}{\eta\,T}
\;+\;
\eta\sum_{i=1}^{n}\frac{L_i}{p_i}.
\tag{18}
\]
\textbf{[Step 7]}

Equation \eqref{18} is exactly the bound \eqref{5} claimed in Theorem \ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From \eqref{5},
\[
\mathcal{R}(T)\triangleq\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\bigl[\norm{\nabla f(w_k)}^{2}\bigr]
\le \underbrace{\frac{2\Delta_0}{\eta T}}_{\text{decrease term}} 
\;+\;
\underbrace{\eta\,\Sigma_{p}}_{\text{bias term}},
\]
where $\Delta_0=f(w_0)-f^{*}$ and $\Sigma_{p}\triangleq\sum_{i=1}^{n}\frac{L_i}{p_i}$.
Choosing $\eta = \sqrt{\frac{2\Delta_0}{\Sigma_{p}T}}$ (which respects \eqref{4} for sufficiently large $T$) balances the two terms and yields
\[
\mathcal{R}(T)\le 2\sqrt{\frac{2\Delta_0\Sigma_{p}}{T}}=O\!\bigl(T^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/n$ gives $\Sigma_{p}=n\bar L$ with $\bar L=\frac{1}{n}\sum_i L_i$, recovering the classic $O\!\bigl(\sqrt{n\bar L/T}\bigr)$ rate.
\item \textbf{Optimal importance sampling.} Minimising $\Sigma_{p}$ under $\sum_i p_i=1$ yields $p_i^{\star}= \frac{\sqrt{L_i}}{\sum_j \sqrt{L_j}}$. Plugging $p_i^{\star}$ gives $\Sigma_{p^{\star}} = \bigl(\sum_i \sqrt{L_i}\bigr)^{2}$, which can be dramatically smaller than $n\max_i L_i$.
\end{itemize}
