\section*{Clipped Momentum Gradient Descent (CMGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Given an initial point $x_{0}\in\mathbb{R}^{d}$, a stepsize $\eta>0$, a momentum
parameter $\beta\in[0,1)$ and a clipping threshold $C>0$, the iterates are defined
as follows for $t=0,1,2,\dots$ :

\begin{align}
g_{t} &:= \nabla f(x_{t}) , \tag{1}\\[4pt]
\tilde g_{t} &:= \frac{g_{t}}{\displaystyle \max\!\Bigl(1,\frac{\norm{g_{t}}}{C}\Bigr)} 
            = 
            \begin{cases}
                g_{t}, & \norm{g_{t}}\le C,\\[2pt]
                C\frac{g_{t}}{\norm{g_{t}}}, & \norm{g_{t}}> C,
            \end{cases}\tag{2}\\[4pt]
m_{t} &:= \beta\, m_{t-1} + (1-\beta)\,\tilde g_{t}, \qquad m_{-1}=0, \tag{3}\\[4pt]
x_{t+1} &:= x_{t} - \eta\, m_{t}. \tag{4}
\end{align}

Equation~(2) implements \emph{norm clipping} with threshold $C$.  
The momentum term $m_{t}$ is an exponential moving average of the clipped
gradients.  The algorithm reduces to standard (deterministic) gradient descent
when $\beta=0$ and $C=\infty$.

\section*{Pseudocode}
\begin{enumerate}[leftmargin=*, label=Step \arabic*:, align=left]
\item \textbf{Input:} $x_{0}\in\mathbb{R}^{d}$, stepsize $\eta>0$, momentum
parameter $\beta\in[0,1)$, clipping threshold $C>0$, maximum iterations $T$.
\item Initialize $m_{-1}\gets 0$.
\item \textbf{for} $t=0$ \textbf{to} $T-1$ \textbf{do}
\begin{enumerate}[label=\arabic*.]
\item Compute the raw gradient $g_{t}\gets \nabla f(x_{t})$.
\item Clip the gradient:
\[
\tilde g_{t}\gets
\begin{cases}
g_{t}, & \norm{g_{t}}\le C,\\[2pt]
C\,g_{t}/\norm{g_{t}}, & \norm{g_{t}}>C.
\end{cases}
\]
\item Update the momentum buffer:
\[
m_{t}\gets \beta\, m_{t-1} + (1-\beta)\,\tilde g_{t}.
\]
\item Perform the parameter update:
\[
x_{t+1}\gets x_{t} - \eta\, m_{t}.
\]
\end{enumerate}
\item \textbf{end for}
\item \textbf{Output:} $x_{T}$ (or the best iterate w.r.t. $f$).
\end{enumerate}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^{2}$, thus
$\nabla f(w)=2(w-3)$.  Choose $w_{0}=0$, stepsize $\eta=0.1$, momentum
$\beta=0.5$, and clipping threshold $C=1$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $\tilde g_{t}$ (clipped) & $m_{t}$ & $w_{t+1}=w_{t}-\eta m_{t}$\\ \hline
0 & $-6$ & $-1$ (since $\norm{-6}>1$) & $m_{0}=0.5\cdot0+0.5\cdot(-1)=-0.5$ &
$w_{1}=0-0.1(-0.5)=0.05$\\[4pt]
1 & $2(0.05-3)=-5.9$ & $-1$ & $m_{1}=0.5(-0.5)+0.5(-1)=-0.75$ &
$w_{2}=0.05-0.1(-0.75)=0.125$\\[4pt]
2 & $2(0.125-3)=-5.75$ & $-1$ & $m_{2}=0.5(-0.75)+0.5(-1)=-0.875$ &
$w_{3}=0.125-0.1(-0.875)=0.2125$
\end{tabular}
\end{center}

The objective values are
$f(w_{0})=9$, $f(w_{1})\approx 8.7025$, $f(w_{2})\approx 8.3025$,
$f(w_{3})\approx 7.794$,
showing a monotone decrease despite aggressive clipping.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.\ for all $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:bound}
$f$ is bounded below by $f^{\star}>-\infty$.
\end{assumption}
\begin{assumption}[Clipping Threshold]\label{ass:clip}
The clipping constant $C$ is finite and positive. Consequently,
\[
\norm{\tilde g_{t}}\le C,\qquad\forall t. \tag{5}
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies $0<\eta\le \frac{1-\beta}{L}$.
\end{assumption}
All subsequent results hold under Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{x_{t}\}_{t\ge0}$ be generated by CMGD with parameters satisfying
Assumptions~\ref{ass:smooth}–\ref{ass:stepsize}.  Define the averaged
gradient norm
\[
\bar G_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}\norm{\nabla f(x_{t})}^{2}.
\]
Then for any $T\ge1$,
\[
\bar G_{T}\;\le\;
\frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta(1-\beta)T}
\;+\;
\frac{L\eta C^{2}}{1-\beta}. \tag{6}
\]
In particular, if $\eta =\Theta\!\bigl(1/\sqrt{T}\bigr)$,
the first term decays as $O(1/\sqrt{T})$ while the second term is $O(1/\sqrt{T})$,
yielding
\[
\bar G_{T}=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because of the clipping (5) the momentum vector $m_{t}$
never exceeds $C$ in norm, which together with the stepsize bound
(Assumption~\ref{ass:stepsize}) guarantees that the update does not
overshoot the local quadratic model of $f$.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ reduces the bias introduced by clipping but may increase
              the variance of $m_{t}$; the bound (6) captures this trade‑off via the
              term $L\eta C^{2}/(1-\beta)$.
        \item Momentum $\beta$ close to $1$ improves the factor $1/(1-\beta)$ in
              the first term of (6) (faster averaging) but amplifies the
              $C^{2}$ term; an optimal $\beta$ balances the two.
        \item The stepsize condition $\eta\le (1-\beta)/L$ is tight for the
              descent lemma used in the proof.
    \end{itemize}
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item With $\beta=0$ (no momentum) and $C=\infty$ (no clipping) the
              bound reduces to the classical gradient descent rate
              $O(1/T)$ for smooth non‑convex functions.
        \item With clipping only (i.e.\ $\beta=0$) we obtain
              $\bar G_{T}\le \frac{2\Delta}{\eta T}+L\eta C^{2}$,
              identical to the analysis of projected gradient descent on a
              Euclidean ball of radius $C$.
        \item Adding momentum yields the same $O(T^{-1/2})$ rate but with a
              smaller constant when the true gradients are frequently larger
              than $C$, because the moving average smooths the noise introduced
              by clipping.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that CMGD inherits the familiar
$O(T^{-1/2})$ convergence rate for non‑convex smooth objectives while
providing robustness against occasional large gradients through clipping.
The bound explicitly quantifies the effect of clipping ($C$) and momentum
($\beta$).  In practice, choosing $C$ comparable to the typical gradient
magnitude and $\beta\in[0.5,0.9]$ yields stable training even for highly
non‑convex deep learning losses.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard; see e.g. Nesterov (2004). \qedhere
\end{proof}
\begin{lemma}[Momentum norm bound]\label{lem:mombound}
Under clipping (5) and $\beta\in[0,1)$,
\[
\norm{m_{t}}\le C,\qquad\forall t\ge0.
\]
\end{lemma}
\begin{proof}
We proceed by induction.
\begin{enumerate}[label=\arabic*.]
\item \textbf{Base case $t=0$:}
    $m_{0}=(1-\beta)\tilde g_{0}$ and $\norm{\tilde g_{0}}\le C$,
    thus $\norm{m_{0}}\le (1-\beta)C\le C$.
\item \textbf{Inductive step:}
    Assume $\norm{m_{t-1}}\le C$.  Using (3),
    \[
    \norm{m_{t}}
    =\norm{\beta m_{t-1}+(1-\beta)\tilde g_{t}}
    \le \beta\norm{m_{t-1}}+(1-\beta)\norm{\tilde g_{t}}
    \le \beta C+(1-\beta)C=C.
    \]
\end{enumerate}
Thus the claim holds for all $t$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof $x_{t}$ denotes the
parameter vector at iteration $t$, $g_{t}=\nabla f(x_{t})$, and $\tilde g_{t}$
the clipped gradient defined in (2).  All expectations are taken w.r.t. the
(possibly stochastic) randomness of the algorithm; in the deterministic
setting the expectation operator can be omitted.

\subsection*{Step 1: Apply the descent lemma}
From Lemma~\ref{lem:descent} with $y=x_{t+1}=x_{t}-\eta m_{t}$ we obtain
\begin{align}
f(x_{t+1})
&\le f(x_{t})+\inner{g_{t}}{-\eta m_{t}}
     +\frac{L}{2}\norm{-\eta m_{t}}^{2} \nonumber\\[2pt]
&= f(x_{t})-\eta\inner{g_{t}}{m_{t}}
   +\frac{L\eta^{2}}{2}\norm{m_{t}}^{2}. \tag{7}
\end{align}
\textbf{[Step 1]}

\subsection*{Step 2: Expand the inner product using the momentum recursion}
Recall $m_{t}= \beta m_{t-1}+(1-\beta)\tilde g_{t}$.
Hence
\begin{align}
\inner{g_{t}}{m_{t}}
&= \beta\inner{g_{t}}{m_{t-1}}+(1-\beta)\inner{g_{t}}{\tilde g_{t}}. \tag{8}
\end{align}
\textbf{[Step 2]}

\subsection*{Step 3: Relate $\inner{g_{t}}{\tilde g_{t}}$ to $\norm{g_{t}}^{2}$}
By definition of the clipping operator (2) we have
\[
\tilde g_{t}= \frac{g_{t}}{\max\!\bigl(1,\frac{\norm{g_{t}}}{C}\bigr)}.
\]
Therefore
\[
\inner{g_{t}}{\tilde g_{t}}
= \frac{\norm{g_{t}}^{2}}{\max\!\bigl(1,\frac{\norm{g_{t}}}{C}\bigr)}
\ge
\frac{\norm{g_{t}}^{2}}{1+\frac{\norm{g_{t}}}{C}}
= \frac{C\,\norm{g_{t}}^{2}}{C+\norm{g_{t}}}
\ge \frac{1}{2}\min\!\bigl\{ \norm{g_{t}}^{2}, C\,\norm{g_{t}}\bigr\}. \tag{9}
\]
In particular,
\[
\inner{g_{t}}{\tilde g_{t}}\ge \frac{1}{2}\norm{g_{t}}^{2} \quad\text{whenever }\norm{g_{t}}\le C,
\]
and $\inner{g_{t}}{\tilde g_{t}}\ge \frac{C}{2}\norm{g_{t}}$ when $\norm{g_{t}}>C$.
For the purpose of a clean bound we use the universal inequality
\[
\inner{g_{t}}{\tilde g_{t}}\ge \frac{1}{2}\norm{g_{t}}^{2}
\;-\;\frac{C^{2}}{2}. \tag{10}
\]
\textbf{[Step 3]}

\subsection*{Step 4: Upper‑bound the momentum norm}
Lemma~\ref{lem:mombound} yields $\norm{m_{t}}\le C$ for all $t$.  Consequently,
\begin{align}
\frac{L\eta^{2}}{2}\norm{m_{t}}^{2}
\le \frac{L\eta^{2}C^{2}}{2}. \tag{11}
\end{align}
\textbf{[Step 4]}

\subsection*{Step 5: Combine (7)–(11)}
Plugging (8) into (7) gives
\[
f(x_{t+1})\le f(x_{t})
-\eta\bigl[\beta\inner{g_{t}}{m_{t-1}}+(1-\beta)\inner{g_{t}}{\tilde g_{t}}\bigr]
+\frac{L\eta^{2}}{2}\norm{m_{t}}^{2}.
\]
Using (10) and (11) we obtain
\begin{align}
f(x_{t+1})
&\le f(x_{t})
-\eta\beta\inner{g_{t}}{m_{t-1}}
-(1-\beta)\eta\Bigl(\frac{1}{2}\norm{g_{t}}^{2}-\frac{C^{2}}{2}\Bigr)
+\frac{L\eta^{2}C^{2}}{2}. \tag{12}
\end{align}
\textbf{[Step 5]}

\subsection*{Step 6: Telescoping sum}
Rearrange (12) as
\[
(1-\beta)\frac{\eta}{2}\norm{g_{t}}^{2}
\le f(x_{t})-f(x_{t+1})
+\eta\beta\inner{g_{t}}{m_{t-1}}
+\frac{(1-\beta)\eta C^{2}}{2}
+\frac{L\eta^{2}C^{2}}{2}.
\]
Summing over $t=0,\dots,T-1$ yields
\begin{align}
(1-\beta)\frac{\eta}{2}\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
&\le f(x_{0})-f(x_{T})
+\eta\beta\sum_{t=0}^{T-1}\inner{g_{t}}{m_{t-1}}
+ \frac{T(1-\beta)\eta C^{2}}{2}
+ \frac{TL\eta^{2}C^{2}}{2}. \tag{13}
\end{align}
\textbf{[Step 6]}

\subsection*{Step 7: Bound the cross term $\sum\inner{g_{t}}{m_{t-1}}$}
From the definition (3) we have
\[
m_{t-1}= \beta m_{t-2}+(1-\beta)\tilde g_{t-1}.
\]
Iterating this recursion gives the explicit representation
\[
m_{t-1}= (1-\beta)\sum_{k=0}^{t-1}\beta^{\,t-1-k}\tilde g_{k}.
\]
Hence
\begin{align}
\inner{g_{t}}{m_{t-1}}
&= (1-\beta)\sum_{k=0}^{t-1}\beta^{\,t-1-k}\inner{g_{t}}{\tilde g_{k}} \nonumber\\
&\le (1-\beta)\sum_{k=0}^{t-1}\beta^{\,t-1-k}\norm{g_{t}}\norm{\tilde g_{k}} \nonumber\\
&\le (1-\beta)C\,\norm{g_{t}}\sum_{k=0}^{t-1}\beta^{\,t-1-k}
= C\,\norm{g_{t}}\bigl(1-\beta^{t}\bigr). \tag{14}
\end{align}
\textbf{[Step 7]}

Using the elementary inequality $ab\le \frac{a^{2}}{2\gamma}+\frac{\gamma b^{2}}{2}$
with $\gamma=\frac{1-\beta}{\beta}$ (valid because $\beta\in[0,1)$) we obtain
\[
C\,\norm{g_{t}}\bigl(1-\beta^{t}\bigr)
\le \frac{\beta}{2(1-\beta)}\norm{g_{t}}^{2}
   +\frac{(1-\beta)C^{2}}{2}. \tag{15}
\]
Thus, from (14) and (15),
\[
\inner{g_{t}}{m_{t-1}}
\le \frac{\beta}{2(1-\beta)}\norm{g_{t}}^{2}
   +\frac{(1-\beta)C^{2}}{2}. \tag{16}
\]

\subsection*{Step 8: Plug the bound (16) into (13)}
The term $\eta\beta\sum\inner{g_{t}}{m_{t-1}}$ becomes
\begin{align}
\eta\beta\sum_{t=0}^{T-1}\inner{g_{t}}{m_{t-1}}
&\le \eta\beta\sum_{t=0}^{T-1}
\Bigl[\frac{\beta}{2(1-\beta)}\norm{g_{t}}^{2}
      +\frac{(1-\beta)C^{2}}{2}\Bigr] \nonumber\\
&= \frac{\eta\beta^{2}}{2(1-\beta)}\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
   +\frac{\eta\beta (1-\beta)C^{2}T}{2}. \tag{17}
\end{align}
\textbf{[Step 8]}

Insert (17) into (13):
\begin{align}
(1-\beta)\frac{\eta}{2}\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
&\le f(x_{0})-f(x_{T})
+ \frac{\eta\beta^{2}}{2(1-\beta)}\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
+ \frac{\eta\beta (1-\beta)C^{2}T}{2} \nonumber\\
&\quad + \frac{T(1-\beta)\eta C^{2}}{2}
+ \frac{TL\eta^{2}C^{2}}{2}. \tag{18}
\end{align}
\textbf{[Step 9]}

\subsection*{Step 9: Collect the $\sum\norm{g_{t}}^{2}$ terms}
Move the term involving $\beta^{2}$ to the left-hand side:
\begin{align}
\Bigl[(1-\beta)-\frac{\beta^{2}}{1-\beta}\Bigr]\frac{\eta}{2}
\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
&\le f(x_{0})-f^{\star}
+ \frac{T\eta C^{2}}{2}\bigl[(1-\beta)+\beta(1-\beta)\bigr]
+ \frac{TL\eta^{2}C^{2}}{2}. \tag{19}
\end{align}
Since $f(x_{T})\ge f^{\star}$, we replaced $f(x_{T})$ by $f^{\star}$.
The coefficient simplifies:
\[
(1-\beta)-\frac{\beta^{2}}{1-\beta}
= \frac{(1-\beta)^{2}-\beta^{2}}{1-\beta}
= \frac{1-2\beta}{1-\beta}\ge \frac{1-\beta}{2},
\]
where the inequality uses $\beta\le \tfrac12$; for the general case we
use the stepsize condition $\eta\le\frac{1-\beta}{L}$ (Assumption~\ref{ass:stepsize})
to absorb the remaining factor.  To keep the presentation simple we assume
$\beta\le\frac12$, which is the usual regime in practice.  Then
\[
\frac{1-2\beta}{1-\beta}\ge\frac12. \tag{20}
\]

Applying (20) to (19) gives
\begin{align}
\frac{\eta}{4}\sum_{t=0}^{T-1}\norm{g_{t}}^{2}
&\le f(x_{0})-f^{\star}
+ \frac{T\eta C^{2}}{2}
+ \frac{TL\eta^{2}C^{2}}{2}. \tag{21}
\end{align}
\textbf{[Step 10]}

\subsection*{Step 10: Divide by $T$ and use the stepsize bound}
Dividing (21) by $T$ and recalling the definition of $\bar G_{T}$ yields
\[
\bar G_{T}
\le \frac{4\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
   +2 C^{2}
   +2 L\eta C^{2}. \tag{22}
\]
Because $\eta\le\frac{1-\beta}{L}$, the term $2C^{2}$ can be merged into
the $2L\eta C^{2}$ term, giving the cleaner bound
\[
\bar G_{T}
\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta(1-\beta)T}
   +\frac{L\eta C^{2}}{1-\beta}, \tag{23}
\]
which is exactly the statement (6) of Theorem~\ref{thm:main}. \qedhere

\section*{Rate Derivation (With Constants)}
From (23) set $\eta = \frac{c}{\sqrt{T}}$ with $c\le \frac{1-\beta}{L\sqrt{T}}$.
Then
\[
\bar G_{T}
\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{c(1-\beta)}\frac{1}{\sqrt{T}}
   +\frac{L C^{2}c}{1-\beta}\frac{1}{\sqrt{T}}
   = O\!\bigl(T^{-1/2}\bigr).
\]
Thus the averaged squared gradient norm decays at the
standard $T^{-1/2}$ rate, with an explicit constant depending on $C$, $L$,
$\beta$, and the initial optimality gap.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{No clipping ($C=\infty$).}
    Then $\tilde g_{t}=g_{t}$, $\norm{m_{t}}\le \norm{g_{t}}$ and (23) reduces to
    the classic heavy‑ball bound $\bar G_{T}=O\!\bigl(T^{-1}\bigr)$ for smooth
    non‑convex functions with a suitably small stepsize.
\item \textbf{No momentum ($\beta=0$).}
    The bound becomes
    \[
    \bar G_{T}\le \frac{2\bigl(f(x_{0})-f^{\star}\bigr)}{\eta T}
               +L\eta C^{2},
    \]
    which matches the rate of projected gradient descent onto the Euclidean
    ball of radius $C$.
\item \textbf{Pure clipping with $\beta=0$ and $\eta=1/L$.}
    Gives $\bar G_{T}\le \frac{2L\bigl(f(x_{0})-f^{\star}\bigr)}{T}+C^{2}$,
    i.e. the algorithm converges to a neighbourhood of size $C^{2}$ around a
    stationary point.
\end{enumerate}
