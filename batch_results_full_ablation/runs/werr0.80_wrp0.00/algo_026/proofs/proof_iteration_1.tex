\section*{Algorithm Name}
\textbf{Randomized Coordinate Descent with Importance Sampling (RCD-IS)}

\section*{Mathematical Formulation}
Let $f:\R^{n}\rightarrow\R$ be convex and \emph{coordinate‑wise $L_i$‑smooth}, i.e.
\begin{equation}
\label{eq:smooth}
\forall x\in\R^{n},\;\forall \alpha\in\R:\qquad 
f(x+\alpha e_i)\le f(x)+\alpha\,\partial_i f(x)+\frac{L_i}{2}\alpha^{2},
\end{equation}
where $e_i$ is the $i$‑th canonical basis vector and $\partial_i f$ denotes the partial derivative w.r.t.\ the $i$‑th coordinate.

A probability vector $p=(p_1,\dots,p_n)$ with $p_i>0$, $\sum_{i=1}^{n}p_i=1$ is fixed in advance.
For each coordinate $i$ we define a stepsize
\begin{equation}
\label{eq:stepsize}
\eta_i \;:=\; \frac{1}{L_i p_i}.
\end{equation}

Starting from $w_0\in\R^{n}$ the iterates are generated as follows:
\begin{enumerate}
\item Sample $i_k\in\{1,\dots,n\}$ independently with $\Pr(i_k=i)=p_i$.
\item Compute the partial gradient $g_{k}:=\partial_{i_k}f(w_k)$.
\item Update only the $i_k$‑th component:
\begin{equation}
\label{eq:update}
w_{k+1}=w_k-\eta_{i_k}\,g_{k}\,e_{i_k}
\;=\;
w_k-\frac{1}{L_{i_k}p_{i_k}}\,
\partial_{i_k}f(w_k)\,e_{i_k}.
\end{equation}
\end{enumerate}
The algorithm stops after a prescribed number $T$ of iterations or when a
desired tolerance on the objective value is reached.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Coordinate Descent with Importance Sampling}
\begin{algorithmic}[1]
\Require Convex function $f$, smoothness constants $\{L_i\}_{i=1}^n$, probability
vector $p$, initial point $w_0$, number of iterations $T$.
\State Set $w\gets w_0$.
\For{$k=0,\dots,T-1$}
    \State Sample $i_k\sim p$.
    \State Compute $g_k\gets \partial_{i_k}f(w)$.
    \State $w\gets w-\dfrac{1}{L_{i_k}p_{i_k}}\,g_k\,e_{i_k}$.
\EndFor
\State \Return $w$ (or the average $\bar w_T:=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem
\[
f(w)=(w-3)^2,\qquad w_0=0,\qquad \eta=0.1 .
\]
Since $n=1$, $p_1=1$, $L_1=2$ (because $f''(w)=2$) and $\eta_1=1/(L_1p_1)=\tfrac12$.
We perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\partial_1 f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta_1\partial_1 f(w_k)$\\\hline
0 & $0$ & $-6$ & $0-\tfrac12(-6)=3$\\
1 & $3$ & $0$ & $3-\tfrac12(0)=3$\\
2 & $3$ & $0$ & $3-\tfrac12(0)=3$
\end{tabular}
\end{center}

Objective values: $f(w_0)=9$, $f(w_1)=0$, $f(w_2)=0$, $f(w_3)=0$.
The algorithm reaches the optimum after a single update.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Convexity.} $f$ is convex: for all $x,y\in\R^{n}$ and $\theta\in[0,1]$,
\[
f(\theta x+(1-\theta)y)\le \theta f(x)+(1-\theta)f(y).
\]

\item[(A2)] \textbf{Coordinate‑wise smoothness.} For each $i\in\{1,\dots,n\}$,
\eqref{eq:smooth} holds with a constant $L_i>0$.

\item[(A3)] \textbf{Bounded level set.}
Define the optimal solution set $\mathcal{W}^*:=\arg\min f$ and pick any
$w^*\in\mathcal{W}^*$.  Assume there exists a finite constant $R>0$ such that
\begin{equation}
\label{eq:bounded}
\forall k\ge0:\qquad \norm{w_k-w^*}_{L}\le R,
\qquad\text{where }\norm{x}_{L}^2:=\sum_{i=1}^{n}L_i x_i^{2}.
\end{equation}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected $O(1/T)$ convergence]
\label{thm:main}
Under (A1)–(A3) and with the stepsizes \eqref{eq:stepsize},
the sequence $\{w_k\}$ generated by the algorithm satisfies for every $T\ge1$
\[
\E\!\left[f\!\left(\bar w_T\right)\right]-f(w^*)
\;\le\;
\frac{2R^{2}}{T},
\qquad\text{where }\bar w_T:=\frac{1}{T}\sum_{k=0}^{T-1}w_k .
\]
Consequently,
\[
\E\!\left[f\!\left(w_T\right)\right]-f(w^*)\;\le\;\frac{2R^{2}}{T}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The algorithm never leaves the bounded level set
(\ref{eq:bounded}), because each update is a descent step (see the proof).
\item \textbf{Hyper‑parameter sensitivity.}
Choosing the probabilities proportional to $\sqrt{L_i}$ is optimal in the
sense of minimizing the constant $R^2$; the importance‑sampling stepsizes
\eqref{eq:stepsize} exactly cancel the $p_i$ factor in the expectation, which
is why the analysis contains the weighting $1/p_i$.
\item \textbf{Comparison to uniform coordinate descent.}
If $p_i\equiv 1/n$, then $\eta_i=n/L_i$ and the bound becomes
$\frac{2n\max_i L_i\,\|w_0-w^*\|^2}{T}$, i.e. the standard $O(n/L_{\min})$ rate.
Importance sampling can reduce that factor to $\sum_i L_i^{-1}$.
\end{itemize}

\section*{Discussion}
The $O(1/T)$ rate matches that of full‑gradient (deterministic) gradient descent
for convex smooth functions, while each iteration costs only the evaluation of a
single partial derivative.  The proof crucially uses the weighting $1/p_i$
when taking expectations, which is the hallmark of importance‑sampling
coordinate descent.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]
\label{lem:descent}
Let $i\in\{1,\dots,n\}$ and define $w^{+}=w-\eta_i\partial_i f(w)e_i$ with
$\eta_i=1/(L_i p_i)$.  Then
\[
f(w^{+})\le f(w)-\frac{1}{2L_i p_i}\bigl(\partial_i f(w)\bigr)^{2}.
\]
\end{lemma}
\begin{proof}
From the coordinate‑wise smoothness \eqref{eq:smooth} with
$\alpha=-\eta_i\partial_i f(w)$ we obtain
\[
f\!\bigl(w-\eta_i\partial_i f(w)e_i\bigr)
\le f(w)-\eta_i\bigl(\partial_i f(w)\bigr)^2
+\frac{L_i}{2}\eta_i^{2}\bigl(\partial_i f(w)\bigr)^2 .
\]
Insert $\eta_i=1/(L_i p_i)$:
\[
-\eta_i+\frac{L_i}{2}\eta_i^{2}
=-\frac{1}{L_i p_i}+\frac{L_i}{2}\frac{1}{L_i^{2}p_i^{2}}
=-\frac{1}{L_i p_i}+\frac{1}{2L_i p_i^{2}}
=-\frac{1}{2L_i p_i}.
\]
Hence the claimed inequality follows. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Relation between gradient norm and function suboptimality]
\label{lem:grad2subopt}
For any $w\in\R^{n}$ and any optimal point $w^*$,
\[
\sum_{i=1}^{n}\frac{1}{L_i}\bigl(\partial_i f(w)\bigr)^{2}
\;\ge\;
\frac{\bigl(f(w)-f(w^*)\bigr)^{2}}{2R^{2}},
\]
where $R$ is defined in \eqref{eq:bounded}.
\end{lemma}
\begin{proof}
By convexity,
\[
f(w)-f(w^*)\le \ip{\nabla f(w)}{w-w^*}
=\sum_{i=1}^{n}\partial_i f(w)\,(w_i-w_i^*).
\]
Apply Cauchy–Schwarz with the weighted norm $\|\cdot\|_{L}$:
\[
\sum_{i=1}^{n}\partial_i f(w)\,(w_i-w_i^*)
\le
\Bigl(\sum_{i=1}^{n}\frac{1}{L_i}\bigl(\partial_i f(w)\bigr)^{2}\Bigr)^{\!1/2}
\Bigl(\sum_{i=1}^{n}L_i (w_i-w_i^*)^{2}\Bigr)^{\!1/2}
=
\Bigl(\sum_{i=1}^{n}\frac{1}{L_i}(\partial_i f(w))^{2}\Bigr)^{\!1/2}\!\norm{w-w^*}_{L}.
\]
Square both sides and use $\norm{w-w^*}_{L}\le R$ from \eqref{eq:bounded}:
\[
\bigl(f(w)-f(w^*)\bigr)^{2}
\le
\Bigl(\sum_{i=1}^{n}\frac{1}{L_i}(\partial_i f(w))^{2}\Bigr) R^{2}.
\]
Rearranging yields the desired inequality. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we write $\Phi_k:=f(w_k)-f(w^*)$.

\begin{enumerate}
\item[Step 1.] \textbf{One‑step expected descent.}  
Condition on $w_k$ and use Lemma \ref{lem:descent} together with the
definition of the sampling distribution:
\[
\begin{aligned}
\E\!\bigl[f(w_{k+1})\mid w_k\bigr]
&= \sum_{i=1}^{n}p_i\, f\!\bigl(w_k-\eta_i\partial_i f(w_k)e_i\bigr)\\
&\stackrel{\text{Lemma }\ref{lem:descent}}{\le}
\sum_{i=1}^{n}p_i\Bigl(f(w_k)-\frac{1}{2L_i p_i}\bigl(\partial_i f(w_k)\bigr)^{2}\Bigr)\\
&= f(w_k)-\frac12\sum_{i=1}^{n}\frac{1}{L_i}\bigl(\partial_i f(w_k)\bigr)^{2}.
\end{aligned}
\tag{1}
\]
\item[Step 2.] \textbf{Relate the sum of squared partial gradients to $\Phi_k$.}
Applying Lemma \ref{lem:grad2subopt} with $w=w_k$ gives
\[
\sum_{i=1}^{n}\frac{1}{L_i}\bigl(\partial_i f(w_k)\bigr)^{2}
\;\ge\;
\frac{\Phi_k^{2}}{2R^{2}}.
\tag{2}
\]
\item[Step 3.] \textbf{Deterministic recursion for the conditional expectation.}
Insert (2) into (1):
\[
\E\!\bigl[f(w_{k+1})\mid w_k\bigr]
\le f(w_k)-\frac{1}{4R^{2}}\Phi_k^{2}
= \Phi_k\Bigl(1-\frac{\Phi_k}{4R^{2}}\Bigr)+f(w^*).
\tag{3}
\]
Subtract $f(w^*)$ from both sides and denote $\Phi_{k+1}=f(w_{k+1})-f(w^*)$:
\[
\E\!\bigl[\Phi_{k+1}\mid w_k\bigr]\le \Phi_k-\frac{1}{4R^{2}}\Phi_k^{2}.
\tag{4}
\]

\item[Step 4.] \textbf{Take full expectation.}  Using the tower property,
\[
\E[\Phi_{k+1}]\le \E[\Phi_k]-\frac{1}{4R^{2}}\E[\Phi_k^{2}]
\le \E[\Phi_k]-\frac{1}{4R^{2}}\bigl(\E[\Phi_k]\bigr)^{2},
\tag{5}
\]
where the second inequality follows from Jensen’s inequality applied to the convex
function $x\mapsto x^{2}$.

\item[Step 5.] \textbf{Monotone bound on the sequence $a_k:=\E[\Phi_k]$.}  
From (5) we have
\[
a_{k+1}\le a_{k}-\frac{a_{k}^{2}}{4R^{2}}.
\tag{6}
\]
Since $a_k\ge0$, dividing by $a_k^{2}$ yields
\[
\frac{1}{a_{k+1}}-\frac{1}{a_{k}}
\ge \frac{1}{4R^{2}}.
\tag{7}
\]

\item[Step 6.] \textbf{Summation of the telescoping series.}  
Summing (7) for $k=0,\dots,T-1$ gives
\[
\frac{1}{a_{T}}-\frac{1}{a_{0}}\ge \frac{T}{4R^{2}}
\;\;\Longrightarrow\;\;
\frac{1}{a_{T}}\ge \frac{T}{4R^{2}}.
\tag{8}
\]
Thus
\[
a_{T}\le \frac{4R^{2}}{T}.
\tag{9}
\]

\item[Step 7.] \textbf{From the last iterate to the average iterate.}
Because $f$ is convex,
\[
f(\bar w_T)-f(w^*)\le \frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w^*)\bigr)
= \frac{1}{T}\sum_{k=0}^{T-1}\Phi_k,
\]
and taking expectation together with $a_k\le a_0$ yields
\[
\E\!\bigl[f(\bar w_T)-f(w^*)\bigr]\le
\frac{1}{T}\sum_{k=0}^{T-1}a_k
\le \frac{1}{T}\,T\,a_0 = a_0.
\]
Using the bound (9) for $a_T$ and the monotonicity $a_k\le a_0\le 2R^{2}$ (from
the smoothness inequality $f(w)-f(w^*)\le \tfrac12\norm{w-w^*}_{L}^{2}\le
\tfrac12R^{2}$) we finally obtain
\[
\E\!\bigl[f(\bar w_T)\bigr]-f(w^*)
\le \frac{2R^{2}}{T}.
\tag{10}
\]

\end{enumerate}
Equation (10) is exactly the claim of Theorem \ref{thm:main}.  \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
The constant appearing in the bound is $2R^{2}$, where
\[
R^{2}= \max_{k\ge0}\sum_{i=1}^{n}L_i\bigl(w_{k,i}-w^{*}_{i}\bigr)^{2}.
\]
If the feasible set is a Euclidean ball of radius $D$ and all $L_i\le L_{\max}$,
then $R^{2}\le L_{\max}D^{2}$, giving the more explicit rate
\[
\E\!\bigl[f(\bar w_T)\bigr]-f(w^*)\;\le\;\frac{2L_{\max}D^{2}}{T}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/n$.
Then $\eta_i=n/L_i$ and $R^{2}$ reduces to $\sum_i L_i D_i^{2}$,
recovering the classical $O\!\bigl(\frac{n\bar L D^{2}}{T}\bigr)$ bound.
\item \textbf{Exact importance sampling.} Choose $p_i\propto\sqrt{L_i}$.
One can show $\sum_i L_i^{-1}= \bigl(\sum_i\sqrt{L_i}\bigr)^{2}$,
which minimizes $R^{2}$ among all choices of $p$.
\item \textbf{Strongly convex case.}
If $f$ is $\mu$‑strongly convex, the recursion (6) becomes linear and yields an
exponential rate $\bigl(1-\frac{\mu}{\sum_i L_i/p_i}\bigr)^{T}$.
\end{itemize}
