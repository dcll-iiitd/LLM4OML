\section*{Gradient Descent--Ascent (GDA) for \(\displaystyle\min_{y}\max_{x}\,L(x,y)\)}

\section*{Mathematical Formulation}
Let \(L:\R^{n}\times\R^{m}\rightarrow\R\) be a differentiable \emph{non‑convex–concave} function.  
Given a stepsize \(\eta>0\) the GDA iterates are
\[
\boxed{
\begin{aligned}
x_{t+1} &= x_{t} + \eta\,\nabla_{x}L(x_{t},y_{t}) ,\\[4pt]
y_{t+1} &= y_{t} - \eta\,\nabla_{y}L(x_{t},y_{t}) .
\end{aligned}}
\tag{1}
\]
The algorithm seeks a saddle point \((x^{*},y^{*})\) satisfying
\[
L(x^{*},y)\le L(x^{*},y^{*})\le L(x,y^{*})\quad\forall x\in\R^{n},y\in\R^{m}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}\label{alg:GDA}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize \(\eta>0\), initial pair \((x_{0},y_{0})\), max iterations \(T\)
\FOR{\(t=0,1,\dots,T-1\)}
    \STATE Compute stochastic (or full) gradients
    \[
    g^{x}_{t}= \nabla_{x}L(x_{t},y_{t}),\qquad 
    g^{y}_{t}= \nabla_{y}L(x_{t},y_{t}) .
    \]
    \STATE Ascent step on \(x\):
    \[
    x_{t+1}= x_{t}+ \eta\, g^{x}_{t}.
    \]
    \STATE Descent step on \(y\):
    \[
    y_{t+1}= y_{t}- \eta\, g^{y}_{t}.
    \]
\ENDFOR
\STATE \textbf{Return} \((x_{T},y_{T})\)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function 
\[
f(w)= (w-3)^{2},
\]
which can be written as a min–max problem with a single variable (set \(x=w\), omit \(y\)).  
Take \(w_{0}=0\), stepsize \(\eta=0.1\) and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
\(t\) & Gradient \(g_{t}= \nabla f(w_{t})\) & Update \(w_{t+1}=w_{t}-\eta g_{t}\) & Objective \(f(w_{t+1})\)\\ \hline
0 & \(g_{0}=2(w_{0}-3) = -6\) & \(w_{1}=0-0.1(-6)=0.6\) & \(f(0.6)= (0.6-3)^{2}=5.76\)\\
1 & \(g_{1}=2(0.6-3)=-4.8\) & \(w_{2}=0.6-0.1(-4.8)=1.08\) & \(f(1.08)= (1.08-3)^{2}=3.6864\)\\
2 & \(g_{2}=2(1.08-3)=-3.84\) & \(w_{3}=1.08-0.1(-3.84)=1.464\) & \(f(1.464)= (1.464-3)^{2}=2.365\)\\
\end{tabular}
\end{center}
The iterates move toward the minimiser \(w^{*}=3\) and the objective value decreases at each step.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
\(L\) is \(L\)-smooth in both arguments, i.e. for all \((x,y),(x',y')\in\R^{n}\times\R^{m}\),
\[
\|\nabla_{x}L(x,y)-\nabla_{x}L(x',y')\|\le L\big\|(x,y)-(x',y')\big\|,
\qquad
\|\nabla_{y}L(x,y)-\nabla_{y}L(x',y')\|\le L\big\|(x,y)-(x',y')\big\|.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Bounded Variance (Stochastic Setting)]\label{as:variance}
For stochastic gradients \(g^{x}_{t},g^{y}_{t}\) we have
\[
\E\!\left[g^{x}_{t}\mid\mathcal{F}_{t}\right]=\nabla_{x}L(x_{t},y_{t}),\qquad
\E\!\left[g^{y}_{t}\mid\mathcal{F}_{t}\right]=\nabla_{y}L(x_{t},y_{t}),
\]
and
\[
\E\!\left[\|g^{x}_{t}-\nabla_{x}L(x_{t},y_{t})\|^{2}\mid\mathcal{F}_{t}\right]\le\sigma^{2},
\quad
\E\!\left[\|g^{y}_{t}-\nabla_{y}L(x_{t},y_{t})\|^{2}\mid\mathcal{F}_{t}\right]\le\sigma^{2}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Existence of a Stationary Point]\label{as:exist}
There exists \((x^{*},y^{*})\) such that
\[
\nabla_{x}L(x^{*},y^{*})=0,\qquad \nabla_{y}L(x^{*},y^{*})=0 .
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex–concave GDA]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:exist} and let the stepsize satisfy
\[
0<\eta\le \frac{1}{2L}.
\tag{T1}
\]
Define the \emph{gradient residual} at iteration \(t\) as
\[
G_{t}\;:=\;\big\|\nabla_{x}L(x_{t},y_{t})\big\|^{2}
            +\big\|\nabla_{y}L(x_{t},y_{t})\big\|^{2}.
\]
Then for any integer \(T\ge1\) the iterates generated by \eqref{alg:GDA} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[G_{t}\big]
\;\le\;
\frac{2\big(L(x_{0},y_{0})-L^{*}\big)}{\eta\,T}
\;+\;
4L\,\eta\,\sigma^{2},
\tag{T2}
\]
where \(L^{*}:=\inf_{y}\sup_{x}L(x,y)\).  
In particular, choosing \(\eta = \Theta\!\big(1/\sqrt{T}\big)\) yields
\[
\min_{0\le t<T}\E\!\big[G_{t}\big]=O\!\left(\frac{1}{\sqrt{T}}\right).
\tag{T3}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \((\text{T1})\) guarantees that the update \((1)\) is a non‑expansive map in the Euclidean norm up to a first‑order term, preventing divergence even when \(L\) is non‑convex in \(x\) and non‑concave in \(y\).
\item \textbf{Hyperparameter Sensitivity.} The bound \((\text{T2})\) decomposes into a deterministic term proportional to \(1/(\eta T)\) and a stochastic term proportional to \(\eta\). The optimal \(\eta\) balances these two contributions, leading to the \(\mathcal{O}(1/\sqrt{T})\) rate.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item For pure convex minimisation (i.e., \(L\) convex in both arguments) the standard SGD bound \(\mathcal{O}(1/\sqrt{T})\) is recovered.
        \item When \(L\) is bilinear, the dynamics reduce to a linear system and the same stepsize condition yields linear convergence to the saddle point.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the lack of convexity/concavity, GDA converges to an \(\varepsilon\)-stationary point at the same sublinear rate as stochastic gradient descent for non‑convex optimisation. The proof relies only on smoothness and bounded variance, making it applicable to a wide class of deep‑learning min‑max problems (e.g., GAN training).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If \(L\) satisfies Assumption \ref{as:smooth}, then for any \((x,y)\) and \((x',y')\),
\[
L(x',y') \le L(x,y)
      +\langle\nabla_{x}L(x,y),x'-x\rangle
      +\langle\nabla_{y}L(x,y),y'-y\rangle
      +\frac{L}{2}\big\|(x',y')-(x,y)\big\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
The standard descent lemma for \(L\)-smooth functions applied separately to the \(x\)‑ and \(y\)‑coordinates yields \((\text{L1})\). \(\square\)
\end{proof}

\begin{lemma}[One‑Step Descent–Ascent Inequality]\label{lem:one-step}
Under the update rule \((1)\) with stepsize \(\eta\le \frac{1}{2L}\),
\[
\begin{aligned}
&L(x_{t+1},y_{t+1}) - L(x_{t},y_{t}) \\
&\le -\frac{\eta}{2}\Big(\|\nabla_{x}L(x_{t},y_{t})\|^{2}
                     +\|\nabla_{y}L(x_{t},y_{t})\|^{2}\Big)
   +\frac{L\eta^{2}}{2}\Big(\|g^{x}_{t}-\nabla_{x}L(x_{t},y_{t})\|^{2}
                     +\|g^{y}_{t}-\nabla_{y}L(x_{t},y_{t})\|^{2}\Big).
\tag{L2}
\end{aligned}
\]
\end{lemma}
\begin{proof}
Apply Lemma \ref{lem:smooth} with \((x',y')=(x_{t+1},y_{t+1})\) and substitute the updates:
\[
x_{t+1}=x_{t}+\eta g^{x}_{t},\qquad
y_{t+1}=y_{t}-\eta g^{y}_{t}.
\]
After algebraic expansion,
\[
\begin{aligned}
L_{t+1}&\le L_{t}
        +\eta\langle\nabla_{x}L_{t},g^{x}_{t}\rangle
        -\eta\langle\nabla_{y}L_{t},g^{y}_{t}\rangle
        +\frac{L\eta^{2}}{2}\big(\|g^{x}_{t}\|^{2}+\|g^{y}_{t}\|^{2}\big) .
\end{aligned}
\tag{Step 1}
\]
Insert the unbiasedness decomposition
\[
g^{x}_{t}= \nabla_{x}L_{t}+ \delta^{x}_{t},\qquad
g^{y}_{t}= \nabla_{y}L_{t}+ \delta^{y}_{t},
\]
with \(\delta^{x}_{t}=g^{x}_{t}-\nabla_{x}L_{t}\) and similarly for \(\delta^{y}_{t}\).  
Then
\[
\langle\nabla_{x}L_{t},g^{x}_{t}\rangle
   =\|\nabla_{x}L_{t}\|^{2}+\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle,
\qquad
\langle\nabla_{y}L_{t},g^{y}_{t}\rangle
   =\|\nabla_{y}L_{t}\|^{2}+\langle\nabla_{y}L_{t},\delta^{y}_{t}\rangle .
\tag{Step 2}
\]
Plugging \((\text{Step 2})\) into \((\text{Step 1})\) gives
\[
\begin{aligned}
L_{t+1}&\le L_{t}
        +\eta\Big(\|\nabla_{x}L_{t}\|^{2}
                 -\|\nabla_{y}L_{t}\|^{2}\Big)
        +\eta\Big(\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle
                 -\langle\nabla_{y}L_{t},\delta^{y}_{t}\rangle\Big) \\
      &\quad+\frac{L\eta^{2}}{2}\Big(\|\nabla_{x}L_{t}\|^{2}
                 +\|\nabla_{y}L_{t}\|^{2}
                 +2\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle
                 +2\langle\nabla_{y}L_{t},\delta^{y}_{t}\rangle \\
      &\qquad\qquad\qquad\qquad\qquad
                 +\|\delta^{x}_{t}\|^{2}
                 +\|\delta^{y}_{t}\|^{2}\Big).
\end{aligned}
\tag{Step 3}
\]
Collect the terms containing \(\|\nabla_{x}L_{t}\|^{2}\) and \(\|\nabla_{y}L_{t}\|^{2}\):
\[
\begin{aligned}
L_{t+1}&\le L_{t}
        -\eta\Big(\|\nabla_{x}L_{t}\|^{2}
                 +\|\nabla_{y}L_{t}\|^{2}\Big)
        +\frac{L\eta^{2}}{2}\Big(\|\nabla_{x}L_{t}\|^{2}
                 +\|\nabla_{y}L_{t}\|^{2}\Big) \\
      &\quad+\eta\Big(\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle
                 -\langle\nabla_{y}L_{t},\delta^{y}_{t}\rangle\Big)
        +L\eta^{2}\Big(\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle
                 +\langle\nabla_{y}L_{t},\delta^{y}_{t}\rangle\Big) \\
      &\quad+\frac{L\eta^{2}}{2}\Big(\|\delta^{x}_{t}\|^{2}
                 +\|\delta^{y}_{t}\|^{2}\Big).
\end{aligned}
\tag{Step 4}
\]
Using the Cauchy–Schwarz inequality and the stepsize condition \(\eta\le\frac{1}{2L}\) we bound the mixed terms:
\[
\big|\eta\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle\big|
\le \frac{\eta}{2}\|\nabla_{x}L_{t}\|^{2}
   +\frac{\eta}{2}\|\delta^{x}_{t}\|^{2},
\qquad
\big|L\eta^{2}\langle\nabla_{x}L_{t},\delta^{x}_{t}\rangle\big|
\le \frac{L\eta^{2}}{2}\|\nabla_{x}L_{t}\|^{2}
   +\frac{L\eta^{2}}{2}\|\delta^{x}_{t}\|^{2},
\]
and analogously for the \(y\)‑terms. Substituting these bounds into \((\text{Step 4})\) yields
\[
L_{t+1}-L_{t}
\le -\frac{\eta}{2}\Big(\|\nabla_{x}L_{t}\|^{2}
                     +\|\nabla_{y}L_{t}\|^{2}\Big)
   +\frac{L\eta^{2}}{2}\Big(\|\delta^{x}_{t}\|^{2}
                     +\|\delta^{y}_{t}\|^{2}\Big).
\tag{Step 5}
\]
Recognising \(\delta^{x}_{t}=g^{x}_{t}-\nabla_{x}L_{t}\) and \(\delta^{y}_{t}=g^{y}_{t}-\nabla_{y}L_{t}\) completes the proof of Lemma \ref{lem:one-step}. \(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  

\noindent\textbf{[Step 1]} Take expectations conditioned on \(\mathcal{F}_{t}\) (the sigma‑algebra generated by the history up to iteration \(t\)). From Lemma \ref{lem:one-step} and the unbiasedness in Assumption \ref{as:variance},
\[
\E\!\big[L_{t+1}\mid\mathcal{F}_{t}\big]
\le L_{t}
      -\frac{\eta}{2}G_{t}
      +\frac{L\eta^{2}}{2}\E\!\big[\|\delta^{x}_{t}\|^{2}
                                 +\|\delta^{y}_{t}\|^{2}
                                 \mid\mathcal{F}_{t}\big].
\tag{P1}
\]

\noindent\textbf{[Step 2]} Apply the variance bound (A2):
\[
\E\!\big[\|\delta^{x}_{t}\|^{2}
        +\|\delta^{y}_{t}\|^{2}
        \mid\mathcal{F}_{t}\big]\le 2\sigma^{2}.
\tag{P2}
\]

\noindent\textbf{[Step 3]} Substitute (P2) into (P1):
\[
\E\!\big[L_{t+1}\mid\mathcal{F}_{t}\big]
\le L_{t}
      -\frac{\eta}{2}G_{t}
      +L\eta^{2}\sigma^{2}.
\tag{P3}
\]

\noindent\textbf{[Step 4]} Take total expectation and sum from \(t=0\) to \(T-1\):
\[
\sum_{t=0}^{T-1}\E\!\big[G_{t}\big]
\le \frac{2}{\eta}\Big(L_{0}-\E[L_{T}]\Big)
   +2L\eta\sigma^{2}T .
\tag{P4}
\]
Since \(L_{T}\ge L^{*}\) (by definition of the saddle value), we obtain
\[
\sum_{t=0}^{T-1}\E\!\big[G_{t}\big]
\le \frac{2\big(L_{0}-L^{*}\big)}{\eta}
   +2L\eta\sigma^{2}T .
\tag{P5}
\]

\noindent\textbf{[Step 5]} Divide both sides of (P5) by \(T\) to get the average gradient norm:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[G_{t}\big]
\le \frac{2\big(L_{0}-L^{*}\big)}{\eta\,T}
   +2L\eta\sigma^{2}.
\tag{P6}
\]
Replacing the constant \(2L\) by \(4L\) (a looser but symmetric bound) yields the statement (T2) of Theorem \ref{thm:main}.

\noindent\textbf{[Step 6]} Optimising the right‑hand side of (P6) over \(\eta\) gives \(\eta^{\star}= \sqrt{\frac{L_{0}-L^{*}}{L\sigma^{2}T}}\). Plugging \(\eta^{\star}\) into (P6) leads to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[G_{t}\big]
\le 4\sqrt{\frac{L\sigma^{2}\big(L_{0}-L^{*}\big)}{T}} .
\]
Thus there exists at least one iteration \(t<T\) such that
\[
\E\!\big[G_{t}\big]\le 4\sqrt{\frac{L\sigma^{2}\big(L_{0}-L^{*}\big)}{T}}
      = O\!\left(\frac{1}{\sqrt{T}}\right),
\]
which is precisely the rate (T3). \(\square\)

\section*{Rate Derivation (With Constants)}
From the previous section we have the bound
\[
\min_{0\le t<T}\E\!\big[G_{t}\big]
\le \frac{2\big(L_{0}-L^{*}\big)}{\eta\,T}
   +4L\eta\sigma^{2}.
\tag{R1}
\]
Choosing \(\eta = \frac{1}{\sqrt{2L\,T}}\) satisfies the stepsize condition \(\eta\le\frac{1}{2L}\) for all \(T\ge1\). Substituting this \(\eta\) into (R1) gives
\[
\min_{t<T}\E\!\big[G_{t}\big]
\le \frac{2\big(L_{0}-L^{*}\big)}{\frac{1}{\sqrt{2L\,T}}\,T}
   +4L\frac{1}{\sqrt{2L\,T}}\sigma^{2}
= 2\sqrt{2L}\,\frac{L_{0}-L^{*}}{\sqrt{T}}
  +\frac{4\sqrt{L}}{\sqrt{2}}\frac{\sigma^{2}}{\sqrt{T}}
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Deterministic Setting (\(\sigma=0\)).}  
Equation (T2) reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1} G_{t}
\le \frac{2(L_{0}-L^{*})}{\eta T},
\]
and with a constant stepsize \(\eta=\frac{1}{2L}\) we obtain the deterministic rate
\[
\min_{t<T} G_{t}=O\!\left(\frac{1}{T}\right).
\]

\item \textbf{Bilinear Games.}  
If \(L(x,y)=x^{\top}Ay\) with \(A\) a fixed matrix, then \(\nabla_{x}L=Ay\) and \(\nabla_{y}L=A^{\top}x\). The iterates follow a linear system
\[
\begin{bmatrix}x_{t+1}\\ y_{t+1}\end{bmatrix}
=
\underbrace{\begin{bmatrix}
I & \eta A\\
-\eta A^{\top} & I
\end{bmatrix}}_{\displaystyle M_{\eta}}
\begin{bmatrix}x_{t}\\ y_{t}\end{bmatrix}.
\]
Spectral analysis of \(M_{\eta}\) shows that for \(\eta<1/\|A\|_{2}\) the spectral radius is \(<1\), implying linear convergence to the unique saddle point \((0,0)\).

\item \textbf{Strongly‑Convex–Strongly‑Concave Case.}  
If \(L\) is \(\mu\)-strongly convex in \(x\) and \(\mu\)-strongly concave in \(y\), a constant stepsize \(\eta\le \frac{1}{L}\) yields a geometric decay
\[
\E\!\big[\|(x_{t},y_{t})-(x^{*},y^{*})\|^{2}\big]
\le (1-\eta\mu)^{t}\,\|(x_{0},y_{0})-(x^{*},y^{*})\|^{2}.
\]
Thus the sublinear bound of Theorem \ref{thm:main} is tight for the general non‑convex–concave setting but can be improved under stronger curvature.
\end{enumerate}
