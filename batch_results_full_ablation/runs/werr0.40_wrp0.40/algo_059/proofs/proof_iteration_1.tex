\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be the (possibly non‑convex) loss function.  
At iteration $t$ we draw an i.i.d.\ data sample $\xi_{t}$ and form the stochastic gradient
\[
g_{t}\;:=\;\nabla f(w_{t};\xi_{t})\in\RR^{d}.
\]
A (random) quantization operator $Q:\RR^{d}\rightarrow\RR^{d}$ satisfies  

\[
\EE[\,Q(g_{t})\mid g_{t}\,]=g_{t},\qquad
\EE\!\big[\|Q(g_{t})-g_{t}\|^{2}\mid g_{t}\big]\leq \omega\|g_{t}\|^{2},
\tag{1}
\]
where $\omega\ge 0$ is the quantization variance factor.  
The update rule of QSGD is
\[
w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t}),\qquad t=0,1,\dots,T-1,
\tag{2}
\]
with a (possibly diminishing) stepsize $\{\eta_{t}\}_{t\ge0}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\RR^{d}$, stepsizes $\{\eta_{t}\}$, quantizer $Q(\cdot)$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a data point $\xi_{t}$ (or mini‑batch)
    \State Compute stochastic gradient $g_{t}\gets\nabla f(w_{t};\xi_{t})$
    \State Quantize: $\tilde g_{t}\gets Q(g_{t})$
    \State Update: $w_{t+1}\gets w_{t}-\eta_{t}\tilde g_{t}$
\EndFor
\State \textbf{Output:} $w_{T}$ (or averaged iterate $\bar w_{T}=\frac{1}{T}\sum_{t=0}^{T-1}w_{t}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
We use a deterministic “identity” quantizer $Q(g)=g$ (so $\omega=0$), stepsize $\eta=0.1$, and $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\

\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the mechanics of (2).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f(\cdot)$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\RR^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For any $w$, $\EE_{\xi}[\,\nabla f(w;\xi)\mid w\,]=\nabla f(w)$ and
\[
\EE_{\xi}\!\big[\|\nabla f(w;\xi)-\nabla f(w)\|^{2}\mid w\big]\le\sigma^{2},
\tag{A2}
\]
with $\sigma^{2}<\infty$.
\end{assumption}
\begin{assumption}[Quantizer]\label{as:quant}
The quantizer $Q$ satisfies (1) with a known $\omega\ge0$.
\tag{A3}
\end{assumption}
\begin{assumption}[Bounded Second Moment]\label{as:bound}
There exists $G>0$ such that $\EE_{\xi}\!\big[\| \nabla f(w;\xi)\|^{2}\mid w\big]\le G^{2}$ for all $w$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:bound} hold and let the stepsize be constant $\eta_{t}\equiv\eta\le \frac{1}{L(1+\omega)}$. Then after $T$ iterations,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(w_{0})-f^{\star}\big)}{\eta T}
\;+\;
2\eta L\,(1+\omega) \,G^{2},
\tag{3}
\]
where $f^{\star}=\inf_{w}f(w)$. Consequently, choosing $\eta=\Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\min_{0\le t<T}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The admissible stepsize shrinks linearly with the quantization factor $(1+\omega)$. When $\omega=0$ we recover the classic SGD bound.
\item \textbf{Hyper‑parameter Sensitivity:} The bound (3) shows a trade‑off: larger $\eta$ accelerates the $1/(\eta T)$ term but inflates the variance term $2\eta L(1+\omega)G^{2}$.
\item \textbf{Comparison:} For unbiased quantizers, QSGD incurs only a multiplicative penalty $(1+\omega)$ compared with vanilla SGD. Deterministic low‑bit quantizers (e.g., stochastic rounding) typically satisfy $\omega=O(1/2^{b})$, where $b$ is the number of bits.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\RR^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof expectations are taken over all randomness up to iteration $t$ (including the quantizer).

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $x=w_{t}$ and $y=w_{t+1}=w_{t}-\eta Q(g_{t})$:
\[
f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),Q(g_{t})\rangle+\frac{L\eta^{2}}{2}\|Q(g_{t})\|^{2}.
\tag{5}
\]

\noindent\textbf{[Step 2]} Take conditional expectation $\EE[\cdot\mid w_{t}]$.
Using unbiasedness of $Q$ (Assumption~\ref{as:quant}) and of $g_{t}$ (Assumption~\ref{as:unbiased}) we obtain
\[
\EE\!\big[\langle\nabla f(w_{t}),Q(g_{t})\rangle\mid w_{t}\big]
= \langle\nabla f(w_{t}),\EE[Q(g_{t})\mid w_{t}]\rangle
= \|\nabla f(w_{t})\|^{2}.
\tag{6}
\]

\noindent\textbf{[Step 3]} Bound the second moment of the quantized gradient.
From (1) and the law of total expectation,
\[
\EE\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]
= \EE\!\big[\|g_{t}\|^{2}\mid w_{t}\big] + \EE\!\big[\|Q(g_{t})-g_{t}\|^{2}\mid w_{t}\big]
\le (1+\omega)\EE\!\big[\|g_{t}\|^{2}\mid w_{t}\big].
\tag{7}
\]

\noindent\textbf{[Step 4]} Use the bounded second moment (Assumption~\ref{as:bound}) to replace $\EE[\|g_{t}\|^{2}\mid w_{t}]\le G^{2}$:
\[
\EE\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]\le (1+\omega)G^{2}.
\tag{8}
\]

\noindent\textbf{[Step 5]} Substitute (6) and (8) into (5) and then take the full expectation:
\[
\EE[f(w_{t+1})]\le \EE[f(w_{t})]-\eta\,\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\frac{L\eta^{2}}{2}(1+\omega)G^{2}.
\tag{9}
\]

\noindent\textbf{[Step 6]} Rearrange (9) to isolate the gradient term:
\[
\eta\,\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \EE[f(w_{t})]-\EE[f(w_{t+1})]
+\frac{L\eta^{2}}{2}(1+\omega)G^{2}.
\tag{10}
\]

\noindent\textbf{[Step 7]} Sum (10) over $t=0,\dots,T-1$ telescopically:
\[
\eta\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le f(w_{0})- \EE[f(w_{T})] + \frac{L\eta^{2}}{2}T(1+\omega)G^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8]} Since $f(w_{T})\ge f^{\star}$, replace $\EE[f(w_{T})]\ge f^{\star}$ and divide by $\eta T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{f(w_{0})-f^{\star}}{\eta T}
+\frac{L\eta}{2}(1+\omega)G^{2}.
\tag{12}
\]

\noindent\textbf{[Step 9]} Multiply numerator and denominator of the first term by $2$ to obtain the cleaner bound (3):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2\big(f(w_{0})-f^{\star}\big)}{\eta T}
+2\eta L(1+\omega)G^{2}.
\tag{13}
\]

\noindent\textbf{[Step 10]} Choice of stepsize. Setting $\eta = \sqrt{\frac{2\big(f(w_{0})-f^{\star}\big)}{L(1+\omega)G^{2}T}}$ (which respects $\eta\le 1/[L(1+\omega)]$ for all $T\ge1$) makes the two RHS terms equal, yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\tag{14}
\]
Since the average of non‑negative numbers is at least their minimum, we obtain the stated rate for $\min_{t<T}\EE[\|\nabla f(w_{t})\|^{2}]$.

Thus Theorem~\ref{thm:main} is proved. \qed

\section*{Rate Derivation (With Constants)}
From (13) with the optimal constant stepsize $\eta^{\star}= \sqrt{\frac{2\Delta}{L(1+\omega)G^{2}T}}$ where $\Delta:=f(w_{0})-f^{\star}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le 2\sqrt{\frac{2L(1+\omega)G^{2}\Delta}{T}}
\;=\; 2\sqrt{2}\,G\sqrt{\frac{L(1+\omega)\Delta}{T}}.
\tag{15}
\]
Hence the explicit constant in the $O(1/\sqrt{T})$ bound is $2\sqrt{2}\,G\sqrt{L(1+\omega)\Delta}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No quantization} ($\omega=0$): (13) reduces to the classical SGD bound.
\item \textbf{Deterministic rounding with $b$ bits}: typical analysis yields $\omega\le \frac{d}{2^{2b}}$, so the penalty scales linearly with dimension $d$ and exponentially decays with $b$.
\item \textbf{Diminishing stepsize} $\eta_{t}=\frac{c}{\sqrt{t+1}}$: a similar telescoping argument gives $\EE[\|\nabla f(w_{t})\|^{2}]=O(1/\sqrt{t})$ without the need to know $T$ beforehand.
\end{itemize}

\section*{Discussion}
The proof demonstrates that unbiased quantization only inflates the variance term by a factor $(1+\omega)$. Consequently, the convergence rate for non‑convex smooth objectives remains $O(1/\sqrt{T})$, matching that of unquantized SGD. The stepsize restriction $\eta\le 1/[L(1+\omega)]$ guarantees that the smoothness term does not dominate the descent, ensuring stability even when coarse quantization (large $\omega$) is used. This theoretical insight justifies the empirical success of low‑precision communication in distributed deep learning.
