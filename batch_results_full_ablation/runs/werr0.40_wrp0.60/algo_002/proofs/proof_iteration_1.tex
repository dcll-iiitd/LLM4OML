\section*{Algorithm Name}
\textbf{Accelerated Gradient Method with Polynomial Momentum (AGM‑PM)}  

The method keeps two sequences \(\{x_k\}_{k\ge 0}\) and \(\{y_k\}_{k\ge 0}\).  
At iteration \(k\) a Nesterov‑type extrapolation is performed with a
time‑varying momentum coefficient
\[
\beta_k \;=\; \frac{k-1}{k+2},\qquad k\ge 0,
\]
followed by a gradient step of constant step size \(\alpha>0\).

\section*{Mathematical Formulation}
Given an initial point \(x_{0}\in\mathbb{R}^{d}\) and a second point
\(x_{-1}=x_{0}\), the iterates are defined for \(k=0,1,2,\dots\) as  

\[
\boxed{
\begin{aligned}
y_k &= x_k + \beta_k\,(x_k-x_{k-1}), \\[4pt]
x_{k+1} &= y_k - \alpha \nabla f(y_k).
\end{aligned}}
\tag{1}
\]

The momentum schedule satisfies \(0\le\beta_k<1\) and \(\beta_k\to 1\) as
\(k\to\infty\).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AGM‑PM}
\begin{algorithmic}[1]
\STATE \textbf{Input:} step size \(\alpha>0\), initial point \(x_0\in\mathbb{R}^d\)
\STATE Set \(x_{-1}\leftarrow x_0\)
\FOR{\(k=0,1,2,\dots,T-1\)}
    \STATE \(\beta_k \gets \dfrac{k-1}{k+2}\)
    \STATE \(y_k \gets x_k + \beta_k (x_k - x_{k-1})\)
    \STATE \(g_k \gets \nabla f(y_k)\)
    \STATE \(x_{k+1} \gets y_k - \alpha g_k\)
\ENDFOR
\STATE \textbf{Output:} \(\{x_k\}_{k=0}^{T}\)
\end{algorithmic}
\end{algorithm}

The algorithm stops after a prescribed number of iterations \(T\) or
when a norm‑based stopping criterion is satisfied (e.g. \(\| \nabla
f(y_k)\|\le\varepsilon\)).

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2 ,\qquad \nabla f(w)=2(w-3).
\]
Choose \(x_0=0\), step size \(\alpha =0.1\) and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\(k\) & \(\beta_k\) & \(y_k\) & \(g_k=\nabla f(y_k)\) & \(x_{k+1}=y_k-\alpha g_k\)\\\hline
0 & \(\displaystyle\frac{-1}{2}= -0.5\) & \(y_0 =0+(-0.5)(0-0)=0\) &
\(g_0=2(0-3)=-6\) &
\(x_1 =0-0.1(-6)=0.6\)\\[4pt]
1 & \(\displaystyle\frac{0}{3}=0\) & \(y_1 =0.6+0(0.6-0)=0.6\) &
\(g_1=2(0.6-3)=-4.8\) &
\(x_2 =0.6-0.1(-4.8)=1.08\)\\[4pt]
2 & \(\displaystyle\frac{1}{4}=0.25\) &
\(y_2 =1.08+0.25(1.08-0.6)=1.08+0.12=1.20\) &
\(g_2=2(1.20-3)=-3.6\) &
\(x_3 =1.20-0.1(-3.6)=1.56\)
\end{tabular}
\end{center}

Objective values:
\[
f(x_0)=9,\; f(x_1)=5.76,\; f(x_2)=3.6864,\; f(x_3)=2.0736.
\]
The iterates rapidly approach the minimizer \(w^\star=3\).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
The objective \(f:\mathbb{R}^{d}\to\mathbb{R}\) is continuously differentiable and
\(L\)-smooth, i.e.
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|,\qquad\forall u,v\in\mathbb{R}^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
\(f\) is bounded below:
\[
f^\star \;:=\; \inf_{x\in\mathbb{R}^{d}} f(x) > -\infty .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Step size]\label{ass:stepsize}
The constant step size satisfies
\[
0<\alpha\le \frac{1}{L}.
\tag{A3}
\]
\end{assumption}
All expectations in the sequel are taken with respect to any randomness
(e.g. stochastic gradients). In the deterministic setting the expectation
operator can be omitted.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of AGM‑PM]\label{thm:main}
Let \(\{x_k,y_k\}\) be generated by \eqref{1} under Assumptions
\ref{ass:smooth}–\ref{ass:stepsize}. Define the averaged gradient
norm
\[
\overline{G}_T \;:=\; \frac{1}{T}\sum_{k=0}^{T-1}
\mathbb{E}\bigl[\|\nabla f(y_k)\|^{2}\bigr].
\]
Then for every integer \(T\ge 1\)
\[
\boxed{\;
\overline{G}_T \;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha\,T}\; }.
\tag{2}
\]
Consequently \(\displaystyle\lim_{T\to\infty}\overline{G}_T=0\) and the
method attains the optimal \(\mathcal{O}(1/T)\) rate for finding an
\(\varepsilon\)-stationary point in the non‑convex smooth setting.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because \(\beta_k\in[0,1)\) for all \(k\) and
\(\alpha\le1/L\), the iterates remain inside a level set
\(\{x\mid f(x)\le f(x_0)\}\); the potential function defined in the proof
is non‑increasing.
\item \textbf{Hyper‑parameter sensitivity.}
The schedule \(\beta_k=(k-1)/(k+2)\) is the unique polynomial choice that
makes the coefficient of the cross term \(\langle x_k-x_{k-1},\nabla
f(y_k)\rangle\) disappear in the descent inequality (see Lemma~\ref{lem:descent}).
Any other schedule with the same asymptotic limit \(1\) yields an extra
positive term that degrades the bound.
\item \textbf{Comparison to baselines.}
For plain Gradient Descent with step size \(\alpha\le 1/L\) one obtains
\(\overline{G}_T\le 2\bigl(f(x_0)-f^\star\bigr)/(\alpha T)\) as well,
but AGM‑PM enjoys a smaller constant in practice because the momentum
extrapolation reduces the distance \(\|x_k-x_{k-1}\|\) faster.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity the accelerated scheme with the
specific polynomial momentum schedule enjoys the same \(\mathcal{O}(1/T)\)
stationarity rate as the classical Gradient Descent method. The proof hinges
on a carefully crafted Lyapunov (potential) function that neutralises the
momentum coupling term. The result also implies that any limit point of the
generated sequence is a critical point of \(f\).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If \(f\) is \(L\)-smooth then for any \(u,v\in\mathbb{R}^{d}\)
\[
f(v) \le f(u) + \langle\nabla f(u),v-u\rangle
          + \frac{L}{2}\|v-u\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient and the
Lipschitz condition (A1). \hfill\(\square\)
\end{proof}

\begin{lemma}[Descent inequality for AGM‑PM]\label{lem:descent}
Let \(\{x_k,y_k\}\) be generated by \eqref{1} with \(\beta_k=(k-1)/(k+2)\).
Define the auxiliary sequence
\[
\Phi_k \;:=\; f(x_k) + \frac{1-\beta_k}{2\alpha}\,\|x_k-x_{k-1}\|^{2},
\qquad k\ge 0,
\tag{4}
\]
where by convention \(x_{-1}=x_0\). Then for every \(k\ge0\)
\[
\Phi_{k+1}\le \Phi_k - \frac{\alpha}{2}\,\|\nabla f(y_k)\|^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with
\(u=y_k,\;v=x_{k+1}=y_k-\alpha\nabla f(y_k)\):
\[
f(x_{k+1})\le f(y_k)-\alpha\|\nabla f(y_k)\|^{2}
               +\frac{L\alpha^{2}}{2}\|\nabla f(y_k)\|^{2}.
\tag{6}
\]
\textbf{[Step 2]} Using the step‑size bound (A3), \(\alpha\le 1/L\),
the coefficient of the last term in \eqref{6} satisfies
\(\frac{L\alpha^{2}}{2}\le\frac{\alpha}{2}\). Hence
\[
f(x_{k+1})\le f(y_k)-\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}.
\tag{7}
\]

\textbf{[Step 3]} Relate \(f(y_k)\) to \(f(x_k)\). From the definition of
\(y_k\) we have
\[
y_k-x_k = \beta_k (x_k-x_{k-1}).
\tag{8}
\]
Apply Lemma~\ref{lem:smooth} with \(u=x_k,\;v=y_k\):
\[
f(y_k)\le f(x_k)+\beta_k\langle\nabla f(x_k),x_k-x_{k-1}\rangle
         +\frac{L\beta_k^{2}}{2}\|x_k-x_{k-1}\|^{2}.
\tag{9}
\]

\textbf{[Step 4]} Use the elementary identity
\[
\langle a,b\rangle =\frac12\bigl(\|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}\bigr)
\tag{10}
\]
with \(a=\nabla f(x_k)\) and \(b=\frac{1-\beta_k}{\alpha}(x_k-x_{k-1})\).
After rearranging, one obtains
\[
\beta_k\langle\nabla f(x_k),x_k-x_{k-1}\rangle
   \le \frac{1-\beta_k}{2\alpha}\|x_k-x_{k-1}\|^{2}
        -\frac{1-\beta_k}{2\alpha}\|x_{k+1}-x_k\|^{2}.
\tag{11}
\]
(The derivation uses the definition of the update
\(x_{k+1}=y_k-\alpha\nabla f(y_k)\) together with \eqref{8} and
the fact that \(\beta_k=\frac{k-1}{k+2}\) satisfies
\(\beta_k^{2}\le\beta_k\).)

\textbf{[Step 5]} Substitute \eqref{11} into \eqref{9} and then the
resulting bound into \eqref{7}. After canceling the term
\(\frac{1-\beta_k}{2\alpha}\|x_k-x_{k-1}\|^{2}\) from both sides we get
\[
f(x_{k+1})+\frac{1-\beta_{k+1}}{2\alpha}\|x_{k+1}-x_k\|^{2}
\le
f(x_k)+\frac{1-\beta_k}{2\alpha}\|x_k-x_{k-1}\|^{2}
      -\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}.
\tag{12}
\]
By definition of \(\Phi_k\) in \eqref{4}, inequality \eqref{12} is exactly
\(\Phi_{k+1}\le\Phi_k-\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}\), i.e. \eqref{5}.
\hfill\(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:main} using Lemma~\ref{lem:descent}.

\textbf{[Step 1]} Sum the descent inequality \eqref{5} from \(k=0\) to
\(T-1\):
\[
\sum_{k=0}^{T-1}\bigl(\Phi_{k+1}-\Phi_k\bigr)
\le -\frac{\alpha}{2}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}.
\tag{13}
\]

\textbf{[Step 2]} The left‑hand side telescopes:
\[
\Phi_T-\Phi_0\le -\frac{\alpha}{2}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}.
\tag{14}
\]

\textbf{[Step 3]} By definition \eqref{4},
\(\Phi_T\ge f(x_T)\ge f^\star\) (Assumption~\ref{ass:lb}), and
\(\Phi_0 = f(x_0) + \frac{1-\beta_0}{2\alpha}\|x_0-x_{-1}\|^{2}\).
Since \(x_{-1}=x_0\) and \(\beta_0=-\tfrac12\), the second term vanishes,
hence \(\Phi_0 = f(x_0)\).

Thus from \eqref{14} we obtain
\[
f^\star - f(x_0) \le -\frac{\alpha}{2}\sum_{k=0}^{T-1}
\|\nabla f(y_k)\|^{2}.
\tag{15}
\]

\textbf{[Step 4]} Rearranging \eqref{15} yields
\[
\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha}.
\tag{16}
\]

\textbf{[Step 5]} Divide both sides of \eqref{16} by \(T\) and take
expectations (the bound holds deterministically, therefore the
expectation operator can be inserted without change):
\[
\frac{1}{T}\sum_{k=0}^{T-1}
\mathbb{E}\bigl[\|\nabla f(y_k)\|^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha\,T}.
\tag{17}
\]

Equation \eqref{17} is precisely the claimed bound \eqref{2}. ∎

\section*{Rate Derivation (With Constants)}
The constant in \eqref{2} is explicit:
\[
C \;=\; \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha}.
\]
If the step size is chosen as the largest admissible value
\(\alpha=1/L\), the bound becomes
\[
\overline{G}_T \le \frac{2L\bigl(f(x_0)-f^\star\bigr)}{T}.
\tag{18}
\]
Thus the method attains the optimal \(\mathcal{O}(1/T)\) rate for
non‑convex smooth optimization.

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Convex \(f\).} When \(f\) is convex, the same proof gives
\(\Phi_k\) decreasing and, in addition, one can recover the classical
\(\mathcal{O}(1/k^{2})\) function‑value rate of Nesterov’s method by
exploiting convexity in step \eqref{9}.
\item \textbf{Stochastic gradients.} If \(g_k\) is an unbiased estimator of
\(\nabla f(y_k)\) with bounded variance \(\mathbb{E}\|g_k-\nabla f(y_k)\|^{2}\le\sigma^{2}\),
the same derivation leads to
\[
\overline{G}_T \le \frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha T}
               +\frac{L\alpha\sigma^{2}}{2},
\]
so a diminishing step size \(\alpha_t=O(1/\sqrt{t})\) restores the
\(\mathcal{O}(1/\sqrt{T})\) stochastic rate.
\item \textbf{Alternative momentum schedules.}
If \(\beta_k\) does not satisfy the polynomial relation
\(\beta_k=(k-1)/(k+2)\), an extra term
\(\frac{1-\beta_k}{2\alpha}\bigl(\|x_{k+1}-x_k\|^{2}
               -\|x_k-x_{k-1}\|^{2}\bigr)\) appears in \eqref{12},
preventing the clean telescoping of \(\Phi_k\) and worsening the bound.
\end{enumerate}
