\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics (SGLD) with Decreasing Noise}\\
The method performs a stochastic gradient step on a smooth non‑convex function and adds isotropic Gaussian noise whose variance decays with the iteration index.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
Given an initial point $x_{0}\in\R^{d}$, a constant stepsize $\alpha>0$, and a noise schedule
\[
\sigma_{k}^{2}= \frac{c}{k+1},\qquad c>0,
\]
the iteration is
\begin{equation}
\boxed{
x_{k+1}=x_{k}-\alpha\,\nabla f(x_{k})+\xi_{k},\qquad 
\xi_{k}\sim\mathcal N\!\bigl(0,\sigma_{k}^{2}I_{d}\bigr),\;k=0,1,\dots
}
\label{eq:update}
\end{equation}
All random variables $\{\xi_{k}\}$ are independent of each other and of $x_{0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD with Decreasing Noise}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha>0$, noise constant $c>0$, max iterations $K$.
\State Initialize $x_{0}\in\R^{d}$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute deterministic gradient $g_{k}\gets\nabla f(x_{k})$.
    \State Sample noise $\xi_{k}\sim\mathcal N\!\bigl(0,\frac{c}{k+1}I_{d}\bigr)$.
    \State Update $x_{k+1}\gets x_{k}-\alpha g_{k}+ \xi_{k}$.
\EndFor
\State \textbf{Output:} $\{x_{k}\}_{k=0}^{K}$ (or the average $\bar x_{K}=\frac1K\sum_{k=0}^{K-1}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, stepsize $\alpha=0.1$, and **ignore noise** (i.e. $\xi_{k}=0$) for illustration.

\begin{enumerate}[label=\textbf{Iter \arabic*:}, leftmargin=*]
\item $k=0$: $\nabla f(w_{0})=2(0-3)=-6$.
\[
w_{1}=w_{0}-\alpha\nabla f(w_{0})=0-0.1(-6)=0.6,\qquad 
f(w_{1})=(0.6-3)^{2}=5.76.
\]

\item $k=1$: $\nabla f(w_{1})=2(0.6-3)=-4.8$.
\[
w_{2}=0.6-0.1(-4.8)=1.08,\qquad 
f(w_{2})=(1.08-3)^{2}=3.6864.
\]

\item $k=2$: $\nabla f(w_{2})=2(1.08-3)=-3.84$.
\[
w_{3}=1.08-0.1(-3.84)=1.464,\qquad 
f(w_{3})=(1.464-3)^{2}=2.3665.
\]
\end{enumerate}
The iterates move toward the minimizer $w^{*}=3$ and the objective values decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\qquad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{as:lb}
There exists $f^{*}>-\infty$ such that $\forall x,\; f(x)\ge f^{*}$.
\end{assumption}

\begin{assumption}[Gradient Boundedness at Optimum]\label{as:gradopt}
$\|\nabla f(x^{*})\|=0$ for any (global) stationary point $x^{*}$.
\end{assumption}

\begin{assumption}[Noise Schedule]\label{as:noise}
The noise $\xi_{k}$ satisfies
\[
\E[\xi_{k}]=0,\qquad 
\E[\xi_{k}\xi_{k}^{\top}]=\sigma_{k}^{2}I_{d},\qquad 
\sigma_{k}^{2}= \frac{c}{k+1},\;c>0.
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{as:stepsize}
The constant stepsize obeys $0<\alpha\le\frac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:stepsize}. Let $\{x_{k}\}$ be generated by \eqref{eq:update}. Define the averaged gradient norm
\[
\mathcal G_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[\|\nabla f(x_{k})\|^{2}\bigr].
\]
Then for any $T\ge 1$,
\begin{equation}
\mathcal G_{T}\;\le\;
\underbrace{\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T}}_{\text{deterministic decay}}
\;+\;
\underbrace{\frac{L\alpha c\bigl(1+\log T\bigr)}{T}}_{\text{noise contribution}}.
\label{eq:rate}
\end{equation}
Consequently $\mathcal G_{T}=O\!\bigl(\tfrac{\log T}{T}\bigr)$, i.e. the expected squared gradient norm vanishes at the rate $\tfrac{\log T}{T}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because $\alpha\le1/L$, the deterministic part of the update is a non‑expansive map; the added isotropic noise does not destabilise the iteration as its variance decays like $1/k$.

\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item Larger $\alpha$ speeds up the deterministic decay term $\frac{2(f(x_{0})-f^{*})}{\alpha T}$ but enlarges the noise term $L\alpha c(1+\log T)/T$.
        \item The constant $c$ controls the magnitude of injected noise; a too‑large $c$ may dominate the bound for moderate $T$.
    \end{itemize}
    The optimal trade‑off (ignoring constants) is obtained by $\alpha\asymp\sqrt{c/T}$, yielding $\mathcal G_{T}=O(1/\sqrt{T})$, which matches known SGLD rates.

\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item For pure (deterministic) gradient descent on smooth non‑convex $f$, one obtains $\frac{1}{T}\sum_{k}\|\nabla f(x_{k})\|^{2}\le \frac{2(f(x_{0})-f^{*})}{\alpha T}$.
        \item Adding decreasing Langevin noise adds the extra $O(\frac{\log T}{T})$ term, but enables exploration of the landscape and convergence to a stationary distribution.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that despite the non‑convexity of $f$, the algorithm drives the expected gradient norm to zero at a near‑optimal rate. The proof follows the classic descent lemma, augmented with careful handling of the stochastic noise variance. The $\log T$ factor originates from the harmonic series $\sum_{k=0}^{T-1}\tfrac{1}{k+1}$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{Nesterov2004}. \qedhere
\end{proof}

\begin{lemma}[Noise Moment]\label{lem:noise}
For $\xi_{k}\sim\mathcal N(0,\sigma_{k}^{2}I_{d})$,
\[
\E[\|\xi_{k}\|^{2}]=d\,\sigma_{k}^{2},\qquad
\E[\langle a,\xi_{k}\rangle]=0\;\;\forall a\in\R^{d}.
\]
\end{lemma}
\begin{proof}
Direct computation of Gaussian moments. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, expectations are taken w.r.t. the randomness up to iteration $k$.

\noindent\textbf{Step 1. Apply the descent lemma to the update \eqref{eq:update}.}  
Using Lemma~\ref{lem:descent} with $x=x_{k}$, $y=x_{k+1}=x_{k}-\alpha\nabla f(x_{k})+\xi_{k}$,
\begin{align}
f(x_{k+1})
&\le f(x_{k})+\big\langle\nabla f(x_{k}),-\,\alpha\nabla f(x_{k})+\xi_{k}\big\rangle
      +\frac{L}{2}\big\|-\alpha\nabla f(x_{k})+\xi_{k}\big\|^{2}. \tag{1}
\end{align}
\noindent\textbf{[Step 1]}

\noindent\textbf{Step 2. Expand the inner product and the squared norm.}
\begin{align}
\big\langle\nabla f(x_{k}),-\,\alpha\nabla f(x_{k})+\xi_{k}\big\rangle
   &=-\alpha\|\nabla f(x_{k})\|^{2}+\langle\nabla f(x_{k}),\xi_{k}\rangle,\\[4pt]
\big\|-\alpha\nabla f(x_{k})+\xi_{k}\big\|^{2}
   &=\alpha^{2}\|\nabla f(x_{k})\|^{2}
     -2\alpha\langle\nabla f(x_{k}),\xi_{k}\rangle
     +\|\xi_{k}\|^{2}. \tag{2}
\end{align}
\noindent\textbf{[Step 2]}

\noindent\textbf{Step 3. Substitute (2) into (1).}
\begin{align}
f(x_{k+1})
 &\le f(x_{k})
   -\alpha\|\nabla f(x_{k})\|^{2}
   +\langle\nabla f(x_{k}),\xi_{k}\rangle  \nonumber\\
 &\quad +\frac{L}{2}\Bigl(\alpha^{2}\|\nabla f(x_{k})\|^{2}
      -2\alpha\langle\nabla f(x_{k}),\xi_{k}\rangle
      +\|\xi_{k}\|^{2}\Bigr). \tag{3}
\end{align}
\noindent\textbf{[Step 3]}

\noindent\textbf{Step 4. Rearrange terms.}
\begin{align}
f(x_{k+1})-f(x_{k})
 &\le -\Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\|\nabla f(x_{k})\|^{2}
   +\Bigl(1-\!L\alpha\Bigr)\langle\nabla f(x_{k}),\xi_{k}\rangle
   +\frac{L}{2}\|\xi_{k}\|^{2}. \tag{4}
\end{align}
\noindent\textbf{[Step 4]}

\noindent\textbf{Step 5. Take full expectation $\E[\cdot]$.}  
Using Lemma~\ref{lem:noise}, $\E[\langle\nabla f(x_{k}),\xi_{k}\rangle]=0$, and $\E[\|\xi_{k}\|^{2}]=d\sigma_{k}^{2}$. Thus,
\begin{align}
\E\!\bigl[f(x_{k+1})\bigr]-\E\!\bigl[f(x_{k})\bigr]
 &\le -\Bigl(\alpha-\frac{L\alpha^{2}}{2}\Bigr)\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
   +\frac{L d}{2}\sigma_{k}^{2}. \tag{5}
\end{align}
Because $\alpha\le 1/L$, we have $ \alpha-\frac{L\alpha^{2}}{2}\ge \frac{\alpha}{2}$. Hence,
\begin{align}
\E\!\bigl[f(x_{k+1})\bigr]-\E\!\bigl[f(x_{k})\bigr]
 &\le -\frac{\alpha}{2}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
   +\frac{L d}{2}\sigma_{k}^{2}. \tag{6}
\end{align}
\noindent\textbf{[Step 5]}

\noindent\textbf{Step 6. Sum (6) from $k=0$ to $T-1$.}
\begin{align}
\E\!\bigl[f(x_{T})\bigr]-f(x_{0})
 &\le -\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
   +\frac{L d}{2}\sum_{k=0}^{T-1}\sigma_{k}^{2}. \tag{7}
\end{align}
Rearranging,
\begin{align}
\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
 &\le \frac{2\bigl(f(x_{0})- \E[f(x_{T})]\bigr)}{\alpha}
   +\frac{L d}{\alpha}\sum_{k=0}^{T-1}\sigma_{k}^{2}. \tag{8}
\end{align}
Since $f(x_{T})\ge f^{*}$ (Assumption~\ref{as:lb}), $\E[f(x_{T})]\ge f^{*}$, giving
\begin{align}
\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
 &\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha}
   +\frac{L d}{\alpha}\sum_{k=0}^{T-1}\sigma_{k}^{2}. \tag{9}
\end{align}
\noindent\textbf{[Step 6]}

\noindent\textbf{Step 7. Insert the noise schedule $\sigma_{k}^{2}=c/(k+1)$.}
\[
\sum_{k=0}^{T-1}\sigma_{k}^{2}=c\sum_{k=0}^{T-1}\frac{1}{k+1}
   =c\bigl(1+H_{T-1}\bigr)\le c\bigl(1+\log T\bigr),
\]
where $H_{n}$ is the $n$‑th harmonic number and $H_{n}\le 1+\log n$. Therefore,
\begin{align}
\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
 &\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha}
   +\frac{L d c}{\alpha}\bigl(1+\log T\bigr). \tag{10}
\end{align}
Dividing both sides by $T$ yields exactly \eqref{eq:rate} (with $d$ absorbed into the constant $c$ for notational simplicity). \qedhere
\noindent\textbf{[Step 7]}

\section*{Rate Derivation (With Constants)}
From \eqref{eq:rate},
\[
\mathcal G_{T}
\;\le\;
\underbrace{\frac{2(f(x_{0})-f^{*})}{\alpha T}}_{\text{deterministic term}}
\;+\;
\underbrace{\frac{L\alpha c (1+\log T)}{T}}_{\text{noise term}}.
\]
If one chooses $\alpha = \sqrt{\frac{2(f(x_{0})-f^{*})}{L c\, (1+\log T)}}$, the two terms balance and
\[
\mathcal G_{T}=O\!\Bigl(\frac{\log T}{T^{1/2}}\Bigr).
\]
For a fixed $\alpha$, the dominant asymptotic behaviour is $O\bigl(\frac{\log T}{T}\bigr)$ as shown in the theorem.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{No Noise ($c=0$).} The bound reduces to
\[
\mathcal G_{T}\le \frac{2(f(x_{0})-f^{*})}{\alpha T},
\]
the classic rate for deterministic gradient descent on smooth non‑convex functions.

\item \textbf{Constant Noise ($\sigma_{k}^{2}\equiv\sigma^{2}$).} The sum in Step~7 becomes $T\sigma^{2}$, leading to
\[
\mathcal G_{T}\le \frac{2(f(x_{0})-f^{*})}{\alpha T}+L\alpha\sigma^{2},
\]
i.e. the noise term does **not** vanish with $T$, reflecting a stationary distribution around critical points.

\item \textbf{High‑Dimensional Scaling.} The factor $d$ from Lemma~\ref{lem:noise} appears explicitly; in practice one often scales $c$ as $c\propto 1/d$ to keep the per‑coordinate variance constant.
\end{enumerate}
