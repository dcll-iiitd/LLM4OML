\section*{Top‑K Stochastic Gradient Descent (TopK‑SGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) loss function.  
At iteration $t$ a stochastic gradient $g_{t}$ is sampled such that  

\[
\E[g_{t}\mid w_{t}] = \nabla f(w_{t}),\qquad 
\E\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le \sigma^{2}.
\]

Define the \emph{top‑$K$ sparsification operator}
\[
\topk(v)\;:=\; \arg\min_{s\in\R^{d}}\big\{\|v-s\|^{2}\;:\; \|s\|_{0}\le K,\; s_{i}=v_{i}\;\text{if}\;|v_{i}|\text{ is among the $K$ largest magnitudes of }v\big\}.
\]
Thus $\topk(v)$ keeps the $K$ coordinates of $v$ with largest absolute value and sets the rest to zero.
The update rule of TopK‑SGD is

\[
\boxed{ \; w_{t+1}= w_{t} - \eta_{t}\,\topk(g_{t}) \; } \tag{1}
\]

with learning‑rate sequence $\{\eta_{t}\}_{t\ge0}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}$, sparsity $K\in\{1,\dots,d\}$
\For{$t=0,1,\dots,T-1$}
    \State Sample a stochastic gradient $g_{t}$ (e.g. minibatch)
    \State $s_{t}\gets \topk(g_{t})$ \Comment{keep $K$ largest magnitudes}
    \State $w_{t+1}\gets w_{t}-\eta_{t}s_{t}$ \Comment{parameter update}
\EndFor
\State \textbf{Output:} $w_{T}$ (or an average $\bar w_T$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, $d=1$, $K=1$, learning rate $\eta=0.1$, and start $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $s_{t}= \topk(g_{t})$ & $w_{t+1}=w_{t}-\eta s_{t}$\\ \hline
0 & $2(w_{0}-3)=-6$ & $-6$ & $w_{1}=0-0.1(-6)=0.6$\\
1 & $2(0.6-3)=-4.8$ & $-4.8$ & $w_{2}=0.6-0.1(-4.8)=1.08$\\
2 & $2(1.08-3)=-3.84$ & $-3.84$ & $w_{3}=1.08-0.1(-3.84)=1.464$\\
\end{tabular}
\end{center}
Corresponding objective values: $f(w_{0})=9$, $f(w_{1})\approx5.76$, $f(w_{2})\approx3.686$, $f(w_{3})\approx2.283$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\] 
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For each $t$, conditional on $w_{t}$,
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad 
\E\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2}.
\]
\end{assumption}

\begin{assumption}[Top‑$K$ Approximation Error]\label{as:topk}
For any vector $v\in\R^{d}$,
\[
\|v-\topk(v)\|^{2}\le\Bigl(1-\frac{K}{d}\Bigr)\|v\|^{2}.
\tag{2}
\]
\end{assumption}

\begin{assumption}[Bounded Second Moment]\label{as:bounded}
There exists $G>0$ such that $\E[\|g_{t}\|^{2}]\le G^{2}$ for all $t$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex TopK‑SGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:bounded} hold and choose a constant stepsize
\[
\eta_{t}\equiv \eta=\frac{c}{\sqrt{T}}\quad\text{with }c\le\frac{1}{L}.
\]
Then the iterates generated by \eqref{1} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_{0})-f^{\star}\big)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{2L\Bigl(1-\frac{K}{d}\Bigr)c\,\frac{G^{2}}{\sqrt{T}}}_{\text{sparsification penalty}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{variance term}} .
\]
Consequently,
\[
\min_{0\le t<T}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]=O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The recursion \eqref{1} is a contraction in expectation provided $\eta L\le1$, guaranteeing that the iterates remain in a bounded region.
\item \textbf{Hyper‑parameter Sensitivity.} The bound degrades linearly with the factor $1-\frac{K}{d}$; larger $K$ (more coordinates) reduces the sparsification penalty.
\item \textbf{Comparison to Full‑SGD.} Setting $K=d$ makes $\topk$ the identity operator, the bound collapses to the classic SGD rate $O(1/\sqrt{T})$.
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item If $K=d$ the term $1-\frac{K}{d}=0$ and Theorem~\ref{thm:main} recovers the standard non‑convex SGD guarantee.
        \item If $K=1$ (extreme sparsity) the penalty becomes $O\big((1-1/d) \frac{cLG^{2}}{\sqrt{T}}\big)$, i.e. the same order but with a larger constant.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that TopK‑SGD retains the optimal $O(1/\sqrt{T})$ decay of the expected squared gradient norm up to an additive constant that scales with the fraction of dropped coordinates. Hence, communication can be reduced dramatically without sacrificing asymptotic convergence speed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For an $L$‑smooth function $f$ and any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Top‑$K$ Error]\label{lem:topk}
For any $v\in\R^{d}$, inequality \eqref{2} holds.
\end{lemma}
\begin{proof}
Let $\mathcal{I}$ be the index set of the $K$ largest magnitudes.
Then $\|v-\topk(v)\|^{2}= \sum_{i\notin\mathcal{I}}v_{i}^{2}$
and $\sum_{i\notin\mathcal{I}}v_{i}^{2}\le\frac{d-K}{d}\sum_{i=1}^{d}v_{i}^{2}$ because the $K$ kept components are at least as large (in absolute value) as the average magnitude of all components. Hence \eqref{2}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. the randomness up to time $t$ unless otherwise noted.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:smooth} with $x=w_{t}$ and $y=w_{t+1}=w_{t}-\eta\topk(g_{t})$:
\[
f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),\topk(g_{t})\rangle+\frac{L\eta^{2}}{2}\|\topk(g_{t})\|^{2}. \tag{3}
\]

\item[Step 2.] Decompose the inner product using unbiasedness:
\[
\langle\nabla f(w_{t}),\topk(g_{t})\rangle
= \langle\nabla f(w_{t}),g_{t}\rangle-\langle\nabla f(w_{t}),g_{t}-\topk(g_{t})\rangle.
\]
Take conditional expectation $\E[\cdot\mid w_{t}]$ and use $\E[g_{t}\mid w_{t}]=\nabla f(w_{t})$:
\[
\E\big[\langle\nabla f(w_{t}),g_{t}\rangle\mid w_{t}\big]=\|\nabla f(w_{t})\|^{2}. \tag{4}
\]

\item[Step 3.] Bound the cross term with Cauchy–Schwarz and Lemma~\ref{lem:topk}:
\[
\big|\langle\nabla f(w_{t}),g_{t}-\topk(g_{t})\rangle\big|
\le \|\nabla f(w_{t})\|\;\|g_{t}-\topk(g_{t})\|
\le \|\nabla f(w_{t})\|\sqrt{1-\frac{K}{d}}\;\|g_{t}\|. \tag{5}
\]

\item[Step 4.] Square the norm of the sparse gradient and bound it via Lemma~\ref{lem:topk}:
\[
\|\topk(g_{t})\|^{2}= \|g_{t}-(g_{t}-\topk(g_{t}))\|^{2}
\le \|g_{t}\|^{2}. \tag{6}
\]

\item[Step 5.] Substitute (4)–(6) into (3) and then take full expectation:
\[
\begin{aligned}
\E[f(w_{t+1})]
&\le \E[f(w_{t})]
-\eta\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\eta\E\!\big[\|\nabla f(w_{t})\|\sqrt{1-\tfrac{K}{d}}\|g_{t}\|\big]
+\frac{L\eta^{2}}{2}\E\!\big[\|g_{t}\|^{2}\big].
\end{aligned} \tag{7}
\]

\item[Step 6.] Apply Cauchy–Schwarz to the mixed term in (7):
\[
\E\!\big[\|\nabla f(w_{t})\|\|g_{t}\|\big]
\le \sqrt{\E\!\big[\|\nabla f(w_{t})\|^{2}\big]\;\E\!\big[\|g_{t}\|^{2}\big]}.
\tag{8}
\]

\item[Step 7.] Use the bounded second‑moment assumption $\E[\|g_{t}\|^{2}]\le G^{2}$:
\[
\begin{aligned}
\E[f(w_{t+1})]
&\le \E[f(w_{t})]
-\eta\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\eta\sqrt{1-\tfrac{K}{d}}\;G\sqrt{\E\!\big[\|\nabla f(w_{t})\|^{2}\big]}
+\frac{L\eta^{2}}{2}G^{2}.
\end{aligned} \tag{9}
\]

\item[Step 8.] Rearrange (9) and complete the square. For any $a\ge0$ and $b\ge0$,
$-a + cb\sqrt{a}\le -\frac{a}{2}+ \frac{c^{2}b^{2}}{2}$.
Set $a=\E[\|\nabla f(w_{t})\|^{2}]$, $c=\eta\sqrt{1-\frac{K}{d}}$, $b=G$.
Then
\[
-\eta a + \eta\sqrt{1-\tfrac{K}{d}}\,G\sqrt{a}
\le -\frac{\eta}{2}a + \frac{\eta}{2}\Bigl(1-\tfrac{K}{d}\Bigr)G^{2}. \tag{10}
\]

\item[Step 9.] Plug (10) into (9):
\[
\E[f(w_{t+1})]\le \E[f(w_{t})]
-\frac{\eta}{2}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
+\frac{\eta}{2}\Bigl(1-\tfrac{K}{d}\Bigr)G^{2}
+\frac{L\eta^{2}}{2}G^{2}. \tag{11}
\]

\item[Step 10.] Sum (11) over $t=0,\dots,T-1$ and telescope the left‑hand side:
\[
\E[f(w_{T})]\le f(w_{0})
-\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
+ \frac{T\eta}{2}\Bigl(1-\tfrac{K}{d}\Bigr)G^{2}
+\frac{LT\eta^{2}}{2}G^{2}. \tag{12}
\]

\item[Step 11.] Drop the non‑negative term $\E[f(w_{T})]$ and solve for the average gradient norm:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le \frac{2\big(f(w_{0})-f^{\star}\big)}{T\eta}
+ \Bigl(1-\tfrac{K}{d}\Bigr)G^{2}
+ L\eta G^{2}. \tag{13}
\]

\item[Step 12.] Choose $\eta = c/\sqrt{T}$ with $c\le 1/L$.
Then $L\eta\le c/L\cdot L = c\le 1$, and (13) becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\underbrace{\frac{2\big(f(w_{0})-f^{\star}\big)}{c\sqrt{T}}}_{\text{optimization term}}
\;+\;
\underbrace{2L\Bigl(1-\tfrac{K}{d}\Bigr)c\,\frac{G^{2}}{\sqrt{T}}}_{\text{sparsification penalty}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{variance term}} .
\]
The variance term follows from the standard decomposition
$\E[\|g_{t}\|^{2}]=\|\nabla f(w_{t})\|^{2}+\E[\|g_{t}-\nabla f(w_{t})\|^{2}]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}$,
which adds $L\eta\sigma^{2}$ to the right‑hand side.
This completes the proof of Theorem~\ref{thm:main}. \qed
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Collecting the dominant terms for large $T$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_{t})\|^{2}\big]
\le
\underbrace{\frac{2\Delta_{0}}{c}}_{\Delta_{0}=f(w_{0})-f^{\star}}
\frac{1}{\sqrt{T}}
\;+\;
\underbrace{2Lc\Bigl(1-\tfrac{K}{d}\Bigr)G^{2}}_{\text{sparsification constant}}
\frac{1}{\sqrt{T}}
\;+\;
\underbrace{L\frac{c}{\sqrt{T}}\sigma^{2}}_{\text{variance}} .
\]
Hence $\min_{t<T}\E[\|\nabla f(w_{t})\|^{2}]=O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full Gradient ($K=d$).} The factor $1-\frac{K}{d}=0$, the bound reduces to the classical SGD guarantee.
\item \textbf{Extreme Sparsity ($K=1$).} The constant $1-\frac{K}{d}=1-\frac{1}{d}\approx1$, giving a larger sparsification penalty but still $O(1/\sqrt{T})$.
\item \textbf{Deterministic Gradients.} If $\sigma=0$, the variance term disappears and the rate improves to $O(1/\sqrt{T})$ with a smaller constant.
\end{itemize}
