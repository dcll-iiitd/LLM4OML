\section*{Algorithm Name}
\textbf{Federated Local Stochastic Gradient Descent (Local SGD)}  

The method performs $E$ local SGD steps on each of $K$ clients before a synchronous averaging (global aggregation) step. A client‐drift term $\delta_{k}^{t}$ captures the deviation of the local objective from the global one.

\section*{Mathematical Formulation}
Let $f_k:\R^{d}\to\R$ be the convex smooth loss on client $k\;(k=1,\dots,K)$ and define the global objective
\[
F(w)=\frac{1}{K}\sum_{k=1}^{K}f_k(w).
\]
For round $r=0,1,\dots,R-1$ the algorithm proceeds as follows.

\begin{enumerate}[label=\arabic*.]
\item The server broadcasts the current global model $w^{r}$ to all clients.
\item On client $k$, initialize $w_{k}^{r,0}=w^{r}$. For local step $e=0,\dots,E-1$,
\[
g_{k}^{r,e}= \nabla f_k\!\bigl(w_{k}^{r,e};\xi_{k}^{r,e}\bigr),\qquad
w_{k}^{r,e+1}= w_{k}^{r,e}-\eta\, g_{k}^{r,e},
\]
where $\xi_{k}^{r,e}$ denotes a random data sample (or mini‑batch) drawn on client $k$.
\item After $E$ local updates, client $k$ sends $w_{k}^{r,E}$ to the server.
\item The server aggregates
\[
w^{r+1}= \frac{1}{K}\sum_{k=1}^{K} w_{k}^{r,E}.
\]
\end{enumerate}
Define the \emph{client drift} at round $r$,
\[
\delta_{k}^{r}= \frac{1}{E}\sum_{e=0}^{E-1}\!\bigl(\nabla f_k(w_{k}^{r,e})-\nabla F(w^{r})\bigr).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Local SGD (Federated Averaging)}
\begin{algorithmic}[1]
\Require Initial model $w^{0}$, learning rate $\eta>0$, local steps $E$, number of clients $K$
\For{$r=0$ \textbf{to} $R-1$}
    \State Server broadcasts $w^{r}$ to all clients
    \For{each client $k\in\{1,\dots,K\}$ \textbf{in parallel}}
        \State $w_{k}^{r,0}\gets w^{r}$
        \For{$e=0$ \textbf{to} $E-1$}
            \State Sample $\xi_{k}^{r,e}$ locally
            \State $g_{k}^{r,e}\gets \nabla f_k\!\bigl(w_{k}^{r,e};\xi_{k}^{r,e}\bigr)$
            \State $w_{k}^{r,e+1}\gets w_{k}^{r,e}-\eta\, g_{k}^{r,e}$
        \EndFor
        \State Send $w_{k}^{r,E}$ to server
    \EndFor
    \State $w^{r+1}\gets \frac{1}{K}\sum_{k=1}^{K} w_{k}^{r,E}$
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $f_k\equiv f$, $K=1$, $E=2$, $\eta=0.1$, $w^{0}=0$.

\begin{tabular}{c|c|c|c}
Step & $w$ & $g=\nabla f(w)$ & $f(w)$\\\hline
0 & $w^{0}=0$ & $-6$ & $9$\\
1 & $w^{0,1}=0-0.1(-6)=0.6$ & $-4.8$ & $(0.6-3)^{2}=5.76$\\
2 & $w^{0,2}=0.6-0.1(-4.8)=1.08$ & $-3.84$ & $(1.08-3)^{2}=3.6864$\\
\hline
Aggregation (only one client) $w^{1}=w^{0,2}=1.08$.
\end{tabular}

After the first global round the objective has decreased from $9$ to $3.6864$, illustrating the effect of two local SGD steps before averaging.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{assump:convex}
Each local loss $f_k$ is convex and $L$‑smooth:
\[
f_k(u)\ge f_k(v)+\inner{\nabla f_k(v)}{u-v},\qquad
\norm{\nabla f_k(u)-\nabla f_k(v)}\le L\norm{u-v},\quad\forall u,v\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradients]\label{assump:unbiased}
For every client $k$, round $r$, local step $e$,
\[
\E_{\xi_{k}^{r,e}}\!\bigl[g_{k}^{r,e}\mid w_{k}^{r,e}\bigr]=\nabla f_k\!\bigl(w_{k}^{r,e}\bigr),
\]
and the variance is uniformly bounded,
\[
\E\!\bigl[\norm{g_{k}^{r,e}-\nabla f_k(w_{k}^{r,e})}^{2}\mid w_{k}^{r,e}\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Drift]\label{assump:drift}
Define the drift $\delta_{k}^{r}$ as above. There exists a constant $\Delta\ge0$ such that
\[
\frac{1}{K}\sum_{k=1}^{K}\norm{\delta_{k}^{r}}^{2}\le\Delta^{2},\qquad\forall r\ge0.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]\label{assump:gradbound}
For all $k$ and $w$, $\norm{\nabla f_k(w)}\le G$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Local SGD]\label{thm:main}
Under Assumptions~\ref{assump:convex}–\ref{assump:gradbound}, choose a constant stepsize
\[
\eta = \frac{1}{L\sqrt{E\,R}}.
\]
Then the global iterates $\{w^{r}\}_{r=0}^{R}$ satisfy
\[
\frac{1}{R}\sum_{r=0}^{R-1}\E\!\bigl[F(w^{r})-F(w^{\star})\bigr]
\;\le\;
\underbrace{\frac{L\norm{w^{0}-w^{\star}}^{2}}{2\sqrt{E\,R}}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{\eta L G^{2}E}{2}}_{\text{stochastic term}}
\;+\;
\underbrace{\frac{L\eta^{2}E^{2}\Delta^{2}}{2}}_{\text{drift term}}.
\]
Consequently, for $R\ge 1$,
\[
\frac{1}{R}\sum_{r=0}^{R-1}\E\!\bigl[F(w^{r})-F(w^{\star})\bigr]=
O\!\left(\frac{1}{\sqrt{E\,R}}+\eta EG^{2}+ \eta^{2}E^{2}\Delta^{2}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound holds for any $E\ge1$; larger $E$ reduces communication but inflates the drift term $\propto \eta^{2}E^{2}\Delta^{2}$.
\item \textbf{Hyperparameter Sensitivity.} The optimal $\eta$ balances the three terms; setting $\eta\asymp 1/(L\sqrt{E R})$ yields the $O(1/\sqrt{E R})$ rate typical of centralized SGD.
\item \textbf{Comparison.} When $E=1$ (i.e., no local steps) the drift term vanishes and Theorem~\ref{thm:main} recovers the classic SGD rate $O(1/\sqrt{R})$. For $E>1$ the rate degrades gracefully depending on $\Delta$.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that Local SGD retains the $\mathcal{O}(1/\sqrt{T})$ convergence speed of centralized SGD up to an additive penalty caused by client drift. When data across clients are homogeneous ($\Delta=0$), the drift term disappears and the method enjoys a linear speed‑up in $E$ (fewer communication rounds). In heterogeneous settings the penalty grows quadratically with $E$, suggesting a trade‑off between communication efficiency and statistical error.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For any $L$‑smooth function $h$ and any $u,v\in\R^{d}$,
\[
h(u)\le h(v)+\inner{\nabla h(v)}{u-v}+\frac{L}{2}\norm{u-v}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g., Nesterov (2004). \qedhere
\end{proof}
\begin{lemma}[Variance Decomposition]\label{lem:var}
For a random vector $X$ with $\E[X]=\mu$,
\[
\E\norm{X}^{2}= \norm{\mu}^{2}+ \E\norm{X-\mu}^{2}.
\]
\end{lemma}
\begin{proof}
Expand $\norm{X}^{2}=\norm{\mu+(X-\mu)}^{2}$ and take expectations. \qedhere
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken w.r.t. all randomness up to the indicated iteration.

\noindent\textbf{Notation.}
\[
w^{\star}\in\arg\min_{w}F(w),\qquad 
\bar w^{r}:=\frac{1}{K}\sum_{k=1}^{K} w_{k}^{r,E}=w^{r+1}.
\]

\noindent\textbf{Step 1.}  Distance recursion for a single client after $E$ local steps.  
For client $k$, using the update $w_{k}^{r,e+1}=w_{k}^{r,e}-\eta g_{k}^{r,e}$,
\begin{align}
\norm{w_{k}^{r,E}-w^{\star}}^{2}
&= \norm{w_{k}^{r,0}-\eta\sum_{e=0}^{E-1} g_{k}^{r,e}-w^{\star}}^{2}\nonumber\\
&= \norm{w^{r}-w^{\star}}^{2}
   -2\eta\inner{\sum_{e=0}^{E-1} g_{k}^{r,e}}{w^{r}-w^{\star}}
   +\eta^{2}\norm{\sum_{e=0}^{E-1} g_{k}^{r,e}}^{2}.
\tag{Step 1}
\end{align}

\noindent\textbf{Step 2.}  Take expectation conditioned on $w^{r}$. Using unbiasedness (Assumption~\ref{assump:unbiased}) and Lemma~\ref{lem:var},
\begin{align}
\E\!\left[\norm{w_{k}^{r,E}-w^{\star}}^{2}\mid w^{r}\right]
&= \norm{w^{r}-w^{\star}}^{2}
   -2\eta\inner{\sum_{e=0}^{E-1}\nabla f_k(w_{k}^{r,e})}{w^{r}-w^{\star}}
   +\eta^{2}\E\!\left[\norm{\sum_{e=0}^{E-1} g_{k}^{r,e}}^{2}\mid w^{r}\right]\nonumber\\
&\le \norm{w^{r}-w^{\star}}^{2}
   -2\eta\sum_{e=0}^{E-1}\inner{\nabla f_k(w_{k}^{r,e})}{w^{r}-w^{\star}}
   +\eta^{2}\!\left(E\sum_{e=0}^{E-1}\E\!\left[\norm{g_{k}^{r,e}}^{2}\mid w^{r}\right]\right).
\tag{Step 2}
\end{align}
The inequality follows from Cauchy–Schwarz and Jensen:
$\norm{\sum_{e}x_e}^{2}\le E\sum_{e}\norm{x_e}^{2}$.

\noindent\textbf{Step 3.}  Bound the stochastic term. By Lemma~\ref{lem:var} and Assumption~\ref{assump:gradbound},
\begin{align}
\E\!\left[\norm{g_{k}^{r,e}}^{2}\mid w_{k}^{r,e}\right]
&= \norm{\nabla f_k(w_{k}^{r,e})}^{2}+ \E\!\left[\norm{g_{k}^{r,e}-\nabla f_k(w_{k}^{r,e})}^{2}\mid w_{k}^{r,e}\right]\nonumber\\
&\le G^{2}+\sigma^{2}\le 2G^{2},
\tag{Step 3}
\end{align}
where we used $G\ge\sigma$ w.l.o.g. (otherwise replace $2G^{2}$ by $G^{2}+\sigma^{2}$).

\noindent\textbf{Step 4.}  Insert Step~3 into Step~2:
\begin{align}
\E\!\left[\norm{w_{k}^{r,E}-w^{\star}}^{2}\mid w^{r}\right]
&\le \norm{w^{r}-w^{\star}}^{2}
   -2\eta\sum_{e=0}^{E-1}\inner{\nabla f_k(w_{k}^{r,e})}{w^{r}-w^{\star}}
   + 2\eta^{2}E^{2}G^{2}.
\tag{Step 4}
\end{align}

\noindent\textbf{Step 5.}  Average over $k$ and use the definition of drift.  
Define $A^{r}:=\frac{1}{K}\sum_{k=1}^{K}\sum_{e=0}^{E-1}\nabla f_k(w_{k}^{r,e})$. Then
\[
\frac{1}{K}\sum_{k=1}^{K}\E\!\left[\norm{w_{k}^{r,E}-w^{\star}}^{2}\mid w^{r}\right]
\le \norm{w^{r}-w^{\star}}^{2}
   -2\eta\inner{A^{r}}{w^{r}-w^{\star}}
   + 2\eta^{2}E^{2}G^{2}.
\tag{Step 5}
\]
Decompose $A^{r}$ as
\[
A^{r}=E\,\nabla F(w^{r})+E\,\delta^{r},
\qquad \delta^{r}:=\frac{1}{K}\sum_{k=1}^{K}\delta_{k}^{r}.
\tag{Step 5a}
\]

\noindent\textbf{Step 6.}  Plug (5a) into (5) and expand the inner product:
\begin{align}
&\norm{w^{r}-w^{\star}}^{2}
   -2\eta E\inner{\nabla F(w^{r})}{w^{r}-w^{\star}}
   -2\eta E\inner{\delta^{r}}{w^{r}-w^{\star}}
   +2\eta^{2}E^{2}G^{2}.
\tag{Step 6}
\end{align}
By convexity of $F$,
\[
F(w^{r})-F(w^{\star})\le\inner{\nabla F(w^{r})}{w^{r}-w^{\star}}.
\tag{Step 6a}
\]

\noindent\textbf{Step 7.}  Apply (6a) to (6) and rearrange:
\begin{align}
2\eta E\bigl(F(w^{r})-F(w^{\star})\bigr)
&\le \norm{w^{r}-w^{\star}}^{2}
   -\E\!\left[\norm{w^{r+1}-w^{\star}}^{2}\mid w^{r}\right]
   +2\eta E\inner{\delta^{r}}{w^{r}-w^{\star}}
   +2\eta^{2}E^{2}G^{2}.
\tag{Step 7}
\end{align}
The term $\E[\norm{w^{r+1}-w^{\star}}^{2}\mid w^{r}]$ appears because $w^{r+1}= \frac{1}{K}\sum_{k} w_{k}^{r,E}$ and Jensen’s inequality yields
\[
\E\!\left[\norm{w^{r+1}-w^{\star}}^{2}\mid w^{r}\right]\le \frac{1}{K}\sum_{k}\E\!\left[\norm{w_{k}^{r,E}-w^{\star}}^{2}\mid w^{r}\right].
\tag{Step 7a}
\]

\noindent\textbf{Step 8.}  Bound the drift inner product using Cauchy–Schwarz and Assumption~\ref{assump:drift}:
\[
2\eta E\inner{\delta^{r}}{w^{r}-w^{\star}}
\le 2\eta E\norm{\delta^{r}}\norm{w^{r}-w^{\star}}
\le \eta^{2}E^{2}\norm{\delta^{r}}^{2}+ \norm{w^{r}-w^{\star}}^{2}.
\tag{Step 8}
\]
Taking expectation and using $\E\norm{\delta^{r}}^{2}\le\Delta^{2}$ gives
\[
2\eta E\inner{\delta^{r}}{w^{r}-w^{\star}}
\le \eta^{2}E^{2}\Delta^{2}+ \norm{w^{r}-w^{\star}}^{2}.
\tag{Step 8a}
\]

\noindent\textbf{Step 9.}  Substitute (8a) into (7) and simplify:
\begin{align}
2\eta E\bigl(F(w^{r})-F(w^{\star})\bigr)
&\le 2\norm{w^{r}-w^{\star}}^{2}
   -\E\!\left[\norm{w^{r+1}-w^{\star}}^{2}\mid w^{r}\right]
   +\eta^{2}E^{2}\Delta^{2}
   +2\eta^{2}E^{2}G^{2}.
\tag{Step 9}
\end{align}
Rearrange to isolate the expected suboptimality:
\begin{align}
\E\!\left[F(w^{r})-F(w^{\star})\mid w^{r}\right]
&\le \frac{1}{2\eta E}\Bigl(2\norm{w^{r}-w^{\star}}^{2}
   -\E\!\left[\norm{w^{r+1}-w^{\star}}^{2}\mid w^{r}\right]\Bigr)
   +\frac{\eta E}{2}\bigl(\Delta^{2}+2G^{2}\bigr).
\tag{Step 10}
\end{align}

\noindent\textbf{Step 11.}  Take total expectation and sum over $r=0,\dots,R-1$.
Telescoping the distance terms yields
\[
\sum_{r=0}^{R-1}\E\!\bigl[F(w^{r})-F(w^{\star})\bigr]
\le \frac{\norm{w^{0}-w^{\star}}^{2}}{\eta E}
   +\frac{R\eta E}{2}\bigl(\Delta^{2}+2G^{2}\bigr).
\tag{Step 11}
\]

\noindent\textbf{Step 12.}  Divide by $R$ and plug the concrete stepsize $\eta=\frac{1}{L\sqrt{E R}}$.
Using $G^{2}\ge \Delta^{2}$ for a cleaner expression,
\[
\frac{1}{R}\sum_{r=0}^{R-1}\E\!\bigl[F(w^{r})-F(w^{\star})\bigr]
\le \frac{L\norm{w^{0}-w^{\star}}^{2}}{2\sqrt{E R}}
   +\frac{\eta L G^{2}E}{2}
   +\frac{L\eta^{2}E^{2}\Delta^{2}}{2},
\]
which is exactly the bound stated in Theorem~\ref{thm:main}. \qed

\section*{Rate Derivation (With Constants)}
From Step~12 we identify three contributions:
\begin{enumerate}[label=\arabic*.]
\item \emph{Optimization term} $ \displaystyle \frac{L\norm{w^{0}-w^{\star}}^{2}}{2\sqrt{E R}}$ – decays as $1/\sqrt{ER}$.
\item \emph{Stochastic term} $ \displaystyle \frac{\eta L G^{2}E}{2}= \frac{G^{2}}{2\sqrt{E R}}$ – same order as the optimization term.
\item \emph{Drift term} $ \displaystyle \frac{L\eta^{2}E^{2}\Delta^{2}}{2}= \frac{\Delta^{2}}{2L\,ER}$ – decays as $1/(ER)$, which is faster when $\Delta$ is small but dominates if $\Delta$ is large relative to $G$.
\end{enumerate}
Hence the overall rate is
\[
\mathcal{O}\!\Bigl(\frac{1}{\sqrt{E R}}+\frac{\Delta^{2}}{ER}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Homogeneous data} ($\delta_{k}^{r}=0$ for all $k,r$).  
The drift term disappears and the bound reduces to the classic SGD rate $O\bigl(1/\sqrt{E R}\bigr)$. Setting $E=R$ yields $O(1/R)$, matching the deterministic gradient method.
\item \textbf{Single local step} ($E=1$).  
Local SGD coincides with standard distributed SGD; the bound simplifies to
\[
\frac{1}{R}\sum_{r}\E[F(w^{r})-F(w^{\star})]\le
\frac{L\norm{w^{0}-w^{\star}}^{2}}{2\sqrt{R}}+\frac{G^{2}}{2\sqrt{R}}+\frac{\Delta^{2}}{2L R},
\]
showing only a $1/R$ drift penalty.
\item \textbf{Large $E$ regime}.  
If $E\gg \sqrt{R}$, the optimization term dominates while the drift term may become the bottleneck unless data are nearly identical across clients ($\Delta\approx0$).
\end{itemize}
