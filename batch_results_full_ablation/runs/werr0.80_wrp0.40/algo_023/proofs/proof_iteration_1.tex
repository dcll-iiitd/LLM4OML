\section*{Proximal Gradient Method (Forward–Backward Splitting)}

\section*{Mathematical Formulation}
We consider the composite convex optimization problem  

\[
\min_{x\in\R^{d}} \; F(x):=f(x)+g(x),
\tag{P}
\]

where  

\begin{itemize}
    \item $f:\R^{d}\to\R$ is convex, differentiable and its gradient is $L$–Lipschitz, i.e.  
    \[
    \|\grad f(u)-\grad f(v)\|\le L\|u-v\|,\qquad\forall u,v\in\R^{d}.
    \tag{A1}
    \]
    \item $g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).  
    \item $\prox_{\eta g}:\R^{d}\to\R^{d}$ denotes the proximal operator
    \[
    \prox_{\eta g}(z)\;:=\;\arg\min_{x\in\R^{d}}\Big\{ g(x)+\frac1{2\eta}\|x-z\|^{2}\Big\}.
    \tag{A2}
    \]
\end{itemize}

Given a stepsize $\eta>0$, the proximal gradient iteration is  

\[
\boxed{\;x_{k+1}= \prox_{\eta g}\bigl(x_{k}-\eta\,\grad f(x_{k})\bigr)\;},\qquad k=0,1,2,\dots
\tag{1}
\]

Throughout we assume that a minimiser $x^{\star}\in\arg\min F$ exists.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$.
\For{$k=0,1,\dots,T-1$}
    \State $g_{k}\gets \grad f(x_{k})$ \Comment{gradient of the smooth part}
    \State $y_{k}\gets x_{k}-\eta\,g_{k}$ \Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(y_{k})$ \Comment{proximal mapping}
\EndFor
\State \Return $x_{T}$
\end{algorithmic}
\end{algorithm}

A typical termination condition is $\|x_{k+1}-x_{k}\|\le \epsilon$ or $F(x_{k})-F(x_{k-1})\le\epsilon$.

\section*{Dry Run Example}
Minimise the simple smooth function $f(w)=(w-3)^{2}$ (so $g\equiv0$) using $\eta=0.1$ and $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_{k}$ & $\grad f(w_{k})=2(w_{k}-3)$ & $F(w_{k})$ \\
\hline
0 & $0$ & $-6$ & $9$ \\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$ \\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$ \\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3665$
\end{tabular}
\end{center}
Since $g=0$, $\prox_{\eta g}$ is the identity, so the iteration reduces to plain gradient descent with the chosen stepsize.

\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smooth part.} $f$ is convex and differentiable with $L$‑Lipschitz gradient (A1).  
\item \textbf{Nonsmooth part.} $g$ is proper, closed, convex (A2).  
\item \textbf{Stepsize.} $\eta\in(0,1/L]$.  
\item \textbf{Existence of solution.} The set $\mathcal{X}^{\star}:=\arg\min_{x}F(x)$ is non‑empty and we denote any element by $x^{\star}$.  
\item \textbf{Bounded initial distance.} $D_{0}:=\|x_{0}-x^{\star}\|<\infty$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{1} with a stepsize $\eta\in(0,1/L]$ under Assumptions (A1)–(A5). Then for every $k\ge1$
\[
F(x_{k})-F(x^{\star})\;\le\;\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta\,k}.
\tag{2}
\]
Consequently $F(x_{k})\to F(x^{\star})$ as $k\to\infty$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The iteration is a non‑expansive mapping in the metric induced by the proximal operator; thus the sequence $\{x_{k}\}$ remains bounded (by $D_{0}$).  
\item \textbf{Stepsize sensitivity.} Any $\eta\le 1/L$ guarantees the descent inequality (see Lemma~\ref{lem:descent}); larger $\eta$ may violate the Lipschitz condition and destroy convergence.  
\item \textbf{Relation to baselines.} When $g\equiv0$, \eqref{1} reduces to gradient descent and the bound \eqref{2} recovers the classical $O(1/k)$ rate. When $f\equiv0$, \eqref{1} becomes the proximal point algorithm with the same $O(1/k)$ guarantee.  
\item \textbf{Extension.} Under additional strong convexity of $f+g$, the rate improves to linear convergence.
\end{itemize}

\section*{Discussion}
The proof hinges on two elementary facts:
\begin{enumerate}
\item The \emph{descent lemma} for $L$‑smooth $f$.
\item The \emph{non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le \|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{3}
\]
Both are invoked explicitly in the step‑by‑step derivation below.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\grad f$ is $L$‑Lipschitz, then for any $u,v\in\R^{d}$
\[
f(v)\le f(u)+\langle\grad f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on $\grad f$.
\end{proof}

\begin{lemma}[Optimality condition of the proximal operator]\label{lem:prox_opt}
For any $z\in\R^{d}$ and $\eta>0$,
\[
x^{+}= \prox_{\eta g}(z) \iff \frac{1}{\eta}(z-x^{+})\in\partial g(x^{+}).
\tag{5}
\]
\end{lemma}
\begin{proof}
By definition of $\prox_{\eta g}$, $x^{+}$ minimises $g(x)+\frac1{2\eta}\|x-z\|^{2}$, so $0\in\partial g(x^{+})+\frac1\eta (x^{+}-z)$, which is equivalent to \eqref{5}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x_{k+1}$ be defined by \eqref{1}. Set 
\[
y_{k}=x_{k}-\eta\grad f(x_{k}),\qquad x_{k+1}= \prox_{\eta g}(y_{k}).
\]

\begin{enumerate}
\item[Step 1.] From Lemma~\ref{lem:prox_opt} applied to $z=y_{k}$ we have  
\[
\frac{1}{\eta}(y_{k}-x_{k+1})\in\partial g(x_{k+1}).
\tag{6}
\]

\item[Step 2.] By convexity of $g$ and \eqref{6},
\[
g(x_{k+1})-g(x^{\star})\le \big\langle \tfrac{1}{\eta}(y_{k}-x_{k+1}),\,x_{k+1}-x^{\star}\big\rangle .
\tag{7}
\]

\item[Step 3.] Expand $y_{k}=x_{k}-\eta\grad f(x_{k})$ in \eqref{7}:
\[
g(x_{k+1})-g(x^{\star})\le
\frac{1}{\eta}\langle x_{k}-x_{k+1},\,x_{k+1}-x^{\star}\rangle
-\langle\grad f(x_{k}),\,x_{k+1}-x^{\star}\rangle .
\tag{8}
\]

\item[Step 4.] Using the identity $2\langle a-b,c-b\rangle =\|a-b\|^{2}+\|c-b\|^{2}-\|a-c\|^{2}$ with $a=x_{k}$, $b=x_{k+1}$, $c=x^{\star}$, we obtain
\[
\frac{2}{\eta}\langle x_{k}-x_{k+1},\,x_{k+1}-x^{\star}\rangle
= \frac{1}{\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr).
\tag{9}
\]
Dividing \eqref{9} by $2$ and substituting into \eqref{8} gives
\[
g(x_{k+1})-g(x^{\star})\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\bigr)
-\langle\grad f(x_{k}),\,x_{k+1}-x^{\star}\rangle .
\tag{10}
\]

\item[Step 5.] Apply the descent lemma \eqref{4} with $u=x_{k}$, $v=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{11}
\]

\item[Step 6.] Add \eqref{10} and \eqref{11}. The terms $\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle$ cancel, and we obtain
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star}) 
&\le f(x_{k})-f(x^{\star})
+\frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr)  \\
&\quad +\Bigl(\frac{L}{2}-\frac{1}{2\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{12}
\]

\item[Step 7.] Since $\eta\le 1/L$, the coefficient of $\|x_{k+1}-x_{k}\|^{2}$ in \eqref{12} is non‑positive:
\[
\frac{L}{2}-\frac{1}{2\eta}\le 0.
\tag{13}
\]
Thus we can drop the last term, yielding the fundamental descent inequality
\[
F(x_{k+1})-F(x^{\star})\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr).
\tag{14}
\]

\item[Step 8.] Rearranging \eqref{14} gives
\[
2\eta\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}.
\tag{15}
\]

\item[Step 9.] Sum \eqref{15} from $k=0$ to $k=K-1$ (telescoping):
\[
2\eta\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\|x_{0}-x^{\star}\|^{2}-\|x_{K}-x^{\star}\|^{2}
\le\|x_{0}-x^{\star}\|^{2}.
\tag{16}
\]

\item[Step 10.] Since each term $F(x_{k+1})-F(x^{\star})\ge0$, we can bound the last iterate by the average:
\[
F(x_{K})-F(x^{\star})\le\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta K}.
\tag{17}
\]

\item[Step 11.] Inequality \eqref{17} is exactly the claim \eqref{2} of Theorem~\ref{thm:rate}.
\end{enumerate}

Thus the proximal gradient method enjoys an $O(1/k)$ convergence rate in function value under the stated assumptions. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{2} we can read off the explicit constant:
\[
C:=\frac{\|x_{0}-x^{\star}\|^{2}}{2\eta},
\qquad\text{so that}\qquad
F(x_{k})-F(x^{\star})\le \frac{C}{k}.
\]
If $L$ is known, the maximal admissible stepsize is $\eta=1/L$, giving
\[
C=\frac{L}{2}\,\|x_{0}-x^{\star}\|^{2}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal step reduces to identity; the proof collapses to the classical gradient‑descent analysis.
\item \textbf{Proximal point algorithm ($f\equiv0$).} The inequality \eqref{14} becomes
\[
g(x_{k+1})-g(x^{\star})\le\frac{1}{2\eta}\bigl(\|x_{k}-x^{\star}\|^{2}-\|x_{k+1}-x^{\star}\|^{2}\bigr),
\]
again yielding $O(1/k)$.
\item \textbf{Strong convexity.} If $F$ is $\mu$‑strongly convex, a refined argument replaces \eqref{14} by
\[
F(x_{k+1})-F(x^{\star})\le\bigl(1-\eta\mu\bigr)\bigl(F(x_{k})-F(x^{\star})\bigr),
\]
implying linear convergence $F(x_{k})-F(x^{\star})\le (1-\eta\mu)^{k}\, (F(x_{0})-F(x^{\star}))$.
\end{itemize}
