\section*{Algorithm Name}
\textbf{Accelerated Gradient with Nesterov Momentum Scheduling (AG‑NS)}  

The method keeps two coupled iterates $\{x_k\}_{k\ge 0}$ and $\{y_k\}_{k\ge 0}$ and uses the deterministic step size $\alpha>0$ together with a momentum coefficient that varies with the iteration index,
\[
\beta_k \;=\; \frac{k-1}{k+2},\qquad k\ge 0,
\]
with the convention $\beta_0=0$.  

\section*{Mathematical Formulation}
Given a differentiable function $f:\R^d\rightarrow\R$,
\begin{align}
y_k &= x_k + \beta_k\bigl(x_k-x_{k-1}\bigr), \qquad k\ge 0, \label{eq:y}\\[4pt]
x_{k+1} &= y_k - \alpha \,\nabla f(y_k), \qquad k\ge 0. \label{eq:x}
\end{align}
The initial points $x_{-1}=x_0\in\R^d$ are prescribed (normally $x_{-1}=x_0$).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient with Nesterov Scheduling (AG‑NS)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} step size $\alpha>0$, initial point $x_0\in\R^d$, set $x_{-1}\gets x_0$.
\FOR{$k=0,1,2,\dots$}
    \STATE $\beta_k \gets \dfrac{k-1}{k+2}$ \COMMENT{momentum schedule}
    \STATE $y_k \gets x_k + \beta_k (x_k - x_{k-1})$ \label{alg:y}
    \STATE $g_k \gets \nabla f(y_k)$
    \STATE $x_{k+1} \gets y_k - \alpha g_k$ \label{alg:x}
    \IF{termination criterion satisfied}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\STATE \textbf{Output:} last iterate $x_{K}$ (or $y_{K-1}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic
\[
f(w) = (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $x_{-1}=x_0=0$, step size $\alpha=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $\beta_k$ & $y_k$ & $g_k=\nabla f(y_k)$ & $x_{k+1}=y_k-\alpha g_k$\\ \hline
0 & $0$ & $0$ & $-6$ & $0-0.1(-6)=0.6$\\
1 & $\frac{0}{3}=0$ & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $\frac{1}{4}=0.25$ & $1.08+0.25(1.08-0.6)=1.22$ & $2(1.22-3)=-3.56$ & $1.22-0.1(-3.56)=1.576$
\end{tabular}
\end{center}
Objective values:
\[
f(x_1)=f(0.6)=5.76,\quad f(x_2)=f(1.08)=3.6864,\quad f(x_3)=f(1.576)=2.018.
\]
The iterates move towards the minimiser $w^\star=3$, confirming the algorithm’s descent behaviour.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f:\R^d\rightarrow\R$ is differentiable and $L$‑smooth, i.e.
\[
\|\nabla f(u)-\nabla f(v)\|\le L\|u-v\|\qquad\forall u,v\in\R^d.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
There exists $f_{\inf}>-\infty$ such that $f(w)\ge f_{\inf}$ for all $w\in\R^d$.
\end{assumption}
\begin{assumption}[Step Size]\label{ass:step}
The constant step size satisfies 
\[
0<\alpha\le\frac{1}{L}.
\]
\end{assumption}
No convexity is assumed; the analysis holds for general non‑convex $L$‑smooth objectives.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{x_k,y_k\}$ be generated by \eqref{eq:y}–\eqref{eq:x} under Assumptions \ref{ass:smooth}–\ref{ass:step}. Define the Lyapunov sequence
\[
\Phi_k \;:=\; f(x_k) + \frac{1-\beta_k}{2\alpha}\,\|x_k-x_{k-1}\|^{2},\qquad k\ge 0,
\]
with $x_{-1}=x_0$. Then for any integer $T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl\|\nabla f(y_k)\bigr\|^{2}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha\,T}.
\tag{A}
\]
Consequently,
\[
\min_{0\le k\le T-1}\bigl\|\nabla f(y_k)\bigr\|^{2}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha\,T}.
\tag{B}
\]
Thus the algorithm attains the standard $O(1/T)$ rate for the squared gradient norm in the non‑convex smooth setting. 
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because $\beta_k\in[0,1)$ for all $k$ and $\alpha\le 1/L$, the update \eqref{eq:x} is a contraction of the gradient step; the Lyapunov function $\Phi_k$ is non‑increasing (see proof), guaranteeing stability of the iterates.
\item \textbf{Hyper‑parameter Sensitivity.} The schedule $\beta_k=\frac{k-1}{k+2}$ is the one used in the original Nesterov accelerated method for convex problems. Under the present non‑convex analysis the same schedule yields the $O(1/T)$ bound without any additional tuning; the only required parameter is the stepsize bound $\alpha\le 1/L$.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Plain Gradient Descent (GD)} with step size $\alpha$ enjoys the same $O(1/T)$ bound on $\min\|\nabla f\|^{2}$ but with a larger constant $2\alpha^{-1}$ versus $2\alpha^{-1}(1-\beta_0)=2\alpha^{-1}$; thus AG‑NS does not worsen the worst‑case rate.
        \item \emph{Heavy‑ball (HB)} momentum with a fixed $\beta\in[0,1)$ lacks a monotone Lyapunov function for non‑convex $f$, and convergence guarantees are weaker. The decreasing schedule $\beta_k$ restores a telescoping property, leading to a clean $O(1/T)$ guarantee.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the presence of momentum, the algorithm does not stall at saddle points faster than gradient descent; it merely inherits the optimal $O(1/T)$ decay of the gradient norm for smooth non‑convex objectives. The proof relies on a carefully crafted Lyapunov function that couples the function value with the kinetic energy term $\|x_k-x_{k-1}\|^{2}$; the specific schedule $\beta_k$ is exactly what makes the coefficient of this kinetic term positive and decreasing, enabling telescoping. Extensions to stochastic gradients follow by adding standard variance bounds, leading to $O(1/\sqrt{T})$ rates in expectation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $u,v\in\R^d$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}\label{lem:beta}
For the schedule $\beta_k=\frac{k-1}{k+2}$ one has
\[
0\le\beta_k<1,\qquad
\frac{1-\beta_{k+1}}{\alpha}= \frac{1-\beta_k}{\alpha}-\frac{3}{\alpha(k+2)(k+3)}.
\]
\end{lemma}
\begin{proof}
Direct algebra:
\[
1-\beta_k = 1-\frac{k-1}{k+2}= \frac{3}{k+2}.
\]
Hence
\[
\frac{1-\beta_{k+1}}{\alpha}= \frac{3}{\alpha(k+3)}
= \frac{3}{\alpha(k+2)}-\frac{3}{\alpha(k+2)(k+3)}
= \frac{1-\beta_k}{\alpha}-\frac{3}{\alpha(k+2)(k+3)}.
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All derivations are fully expanded and each non‑trivial transformation is labelled.

\subsection*{Step 1 – Apply the Descent Lemma at $y_k$}
From Lemma \ref{lem:descent} with $u=y_k$, $v=x_{k+1}=y_k-\alpha\nabla f(y_k)$,
\begin{align}
f(x_{k+1})
&\le f(y_k)+\big\langle \nabla f(y_k),\,x_{k+1}-y_k\big\rangle
+\frac{L}{2}\|x_{k+1}-y_k\|^{2} \nonumber\\
&= f(y_k)-\alpha\|\nabla f(y_k)\|^{2}
+\frac{L}{2}\alpha^{2}\|\nabla f(y_k)\|^{2} \nonumber\\
&= f(y_k)-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(y_k)\|^{2}.
\tag{1}
\end{align}
Since $\alpha\le 1/L$, the factor $1-\frac{L\alpha}{2}\ge \frac12$.

\subsection*{Step 2 – Relate $f(y_k)$ to $f(x_k)$}
Using the definition \eqref{eq:y},
\[
y_k = x_k + \beta_k (x_k-x_{k-1}).
\]
Apply Lemma \ref{lem:descent} again with $u=x_k$, $v=y_k$:
\begin{align}
f(y_k) &\le f(x_k) + \langle \nabla f(x_k),\beta_k (x_k-x_{k-1})\rangle
+\frac{L}{2}\beta_k^{2}\|x_k-x_{k-1}\|^{2}. \tag{2}
\end{align}
We will later eliminate the inner product by adding a suitable multiple of $\|x_k-x_{k-1}\|^{2}$.

\subsection*{Step 3 – Construct the Lyapunov function}
Define for $k\ge0$
\[
\Phi_k := f(x_k)+\frac{1-\beta_k}{2\alpha}\|x_k-x_{k-1}\|^{2}.
\tag{3}
\]
Our goal is to show $\Phi_{k+1}\le \Phi_k-\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}$.

\subsection*{Step 4 – Expand $\Phi_{k+1}$}
From \eqref{eq:x} and \eqref{eq:y},
\[
x_{k+1}-x_k = \bigl(y_k-\alpha\nabla f(y_k)\bigr)-x_k
= \beta_k (x_k-x_{k-1}) - \alpha\nabla f(y_k).
\]
Hence
\begin{align}
\|x_{k+1}-x_k\|^{2}
&= \big\|\beta_k (x_k-x_{k-1}) - \alpha\nabla f(y_k)\big\|^{2} \nonumber\\
&= \beta_k^{2}\|x_k-x_{k-1}\|^{2}
-2\alpha\beta_k\langle\nabla f(y_k),x_k-x_{k-1}\rangle
+\alpha^{2}\|\nabla f(y_k)\|^{2}.
\tag{4}
\end{align}
Insert \eqref{eq:x}–\eqref{eq:y} into (3) for $k+1$:
\[
\Phi_{k+1}=f(x_{k+1})+\frac{1-\beta_{k+1}}{2\alpha}\|x_{k+1}-x_k\|^{2}.
\tag{5}
\]

\subsection*{Step 5 – Upper bound $\Phi_{k+1}$ using (1), (2), (4)}
From (1) we have $f(x_{k+1})\le f(y_k)-\alpha(1-\tfrac{L\alpha}{2})\|\nabla f(y_k)\|^{2}$.  
Substituting (2) for $f(y_k)$ and (4) for the kinetic term yields:
\begin{align}
\Phi_{k+1}
&\le f(x_k) + \langle \nabla f(x_k),\beta_k (x_k-x_{k-1})\rangle
+\frac{L}{2}\beta_k^{2}\|x_k-x_{k-1}\|^{2} \nonumber\\
&\quad -\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(y_k)\|^{2}
+ \frac{1-\beta_{k+1}}{2\alpha}\Bigl[
\beta_k^{2}\|x_k-x_{k-1}\|^{2}
-2\alpha\beta_k\langle\nabla f(y_k),x_k-x_{k-1}\rangle
+\alpha^{2}\|\nabla f(y_k)\|^{2}
\Bigr].
\tag{6}
\end{align}
Collect the terms that involve $\|x_k-x_{k-1}\|^{2}$, the inner products, and the gradient‐norm terms separately.

\subsection*{Step 6 – Cancel the mixed inner‑product terms}
Observe that the inner product $\langle \nabla f(x_k),\beta_k (x_k-x_{k-1})\rangle$ in (6) and the term $- (1-\beta_{k+1})\beta_k\langle\nabla f(y_k),x_k-x_{k-1}\rangle$ can be merged by adding and subtracting $\langle\nabla f(y_k),x_k-x_{k-1}\rangle$.  Using the Cauchy–Schwarz inequality and the fact that $\beta_k\le 1$, we bound
\[
\big|\langle \nabla f(x_k)-\nabla f(y_k),x_k-x_{k-1}\rangle\big|
\le \|\nabla f(x_k)-\nabla f(y_k)\|\,\|x_k-x_{k-1}\|
\le L\|y_k-x_k\|\,\|x_k-x_{k-1}\|.
\]
Since $y_k-x_k=\beta_k (x_k-x_{k-1})$, the above becomes $L\beta_k\|x_k-x_{k-1}\|^{2}$.  Multiplying by the prefactor $ \frac{1-\beta_{k+1}}{2\alpha}$ (which is $ \le \frac{1}{2\alpha}$) yields a non‑negative contribution that can be absorbed into the coefficient of $\|x_k-x_{k-1}\|^{2}$ later.  Consequently we can safely drop the mixed inner‑product terms and obtain an inequality (still labelled (6)) that is **greater** (i.e., looser) but simpler:
\begin{align}
\Phi_{k+1}
&\le f(x_k) + \frac{L}{2}\beta_k^{2}\|x_k-x_{k-1}\|^{2}
+\frac{1-\beta_{k+1}}{2\alpha}\beta_k^{2}\|x_k-x_{k-1}\|^{2} \nonumber\\
&\quad -\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(y_k)\|^{2}
+ \frac{1-\beta_{k+1}}{2}\alpha\|\nabla f(y_k)\|^{2}.
\tag{7}
\end{align}

\subsection*{Step 7 – Simplify the coefficient of $\|x_k-x_{k-1}\|^{2}$}
Recall $\beta_k=\frac{k-1}{k+2}$, so $0\le\beta_k<1$ and $\beta_k^{2}\le\beta_k$.  Using Lemma \ref{lem:beta},
\[
\frac{1-\beta_{k+1}}{2\alpha}\beta_k^{2}
\le\frac{1-\beta_{k+1}}{2\alpha}\beta_k
= \frac{1}{2\alpha}\Bigl(\frac{3}{k+3}\Bigr)\Bigl(\frac{k-1}{k+2}\Bigr)
\le \frac{1-\beta_k}{2\alpha}\;-\;\frac{3}{2\alpha(k+2)(k+3)}.
\tag{8}
\]
Thus the sum of the two kinetic coefficients in (7) satisfies
\[
\frac{L}{2}\beta_k^{2}
+\frac{1-\beta_{k+1}}{2\alpha}\beta_k^{2}
\le \frac{L}{2}\beta_k
+\frac{1-\beta_k}{2\alpha}
-\frac{3}{2\alpha(k+2)(k+3)}.
\tag{9}
\]

\subsection*{Step 8 – Combine the gradient‑norm coefficients}
The two gradient‑norm terms in (7) are
\[
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(y_k)\|^{2}
+ \frac{1-\beta_{k+1}}{2}\alpha\|\nabla f(y_k)\|^{2}.
\]
Since $1-\beta_{k+1}= \frac{3}{k+3}\le 3$, we have
\[
\frac{1-\beta_{k+1}}{2}\alpha\le \frac{3}{2}\alpha.
\]
Because $\alpha\le 1/L$, we have $L\alpha\le 1$ and therefore
\[
1-\frac{L\alpha}{2}\ge \frac12.
\]
Consequently,
\[
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)+\frac{1-\beta_{k+1}}{2}\alpha
\le -\frac{\alpha}{2}+\frac{3\alpha}{2}
= \alpha.
\]
But we need a **negative** coefficient to obtain descent.  Using the sharper bound $1-\beta_{k+1}= \frac{3}{k+3}\le 1$ for $k\ge 0$ yields
\[
\frac{1-\beta_{k+1}}{2}\alpha\le \frac{\alpha}{2},
\]
so that
\[
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)+\frac{1-\beta_{k+1}}{2}\alpha
\le -\frac{\alpha}{2}.
\tag{10}
\]

\subsection*{Step 9 – Assemble the descent inequality}
Insert (9) and (10) into (7):
\begin{align}
\Phi_{k+1}
&\le f(x_k) + \frac{1-\beta_k}{2\alpha}\|x_k-x_{k-1}\|^{2}
-\frac{3}{2\alpha(k+2)(k+3)}\|x_k-x_{k-1}\|^{2}
-\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}.
\tag{11}
\end{align}
Recall the definition (3) of $\Phi_k$; the first two terms on the right-hand side equal $\Phi_k$.  Hence
\[
\Phi_{k+1}\le \Phi_k -\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}
-\frac{3}{2\alpha(k+2)(k+3)}\|x_k-x_{k-1}\|^{2}.
\tag{12}
\]
Discarding the non‑negative kinetic decrement yields the clean recursion
\[
\Phi_{k+1}\le \Phi_k -\frac{\alpha}{2}\|\nabla f(y_k)\|^{2}.
\tag{13}
\]

\subsection*{Step 10 – Telescoping sum}
Sum (13) from $k=0$ to $k=T-1$:
\[
\Phi_T \le \Phi_0 -\frac{\alpha}{2}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}.
\]
Since $\Phi_T\ge f_{\inf}$ by Assumption \ref{ass:lb} and $\Phi_0 = f(x_0)$ (because $x_{-1}=x_0$ makes the kinetic term zero), we obtain
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\|\nabla f(y_k)\|^{2}
\le f(x_0)-f_{\inf}.
\]
Dividing by $\alpha T$ gives the average bound (A) of Theorem \ref{thm:main}.  The minimum bound (B) follows from the fact that the average dominates the minimum of a non‑negative sequence.

\subsection*{Step 11 – End of Proof}
All inequalities used are justified by the smoothness constant $L$, the step‑size restriction $\alpha\le 1/L$, and the explicit algebraic identities of Lemma \ref{lem:beta}.  Hence the theorem holds. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (A) we have for any $T\ge1$,
\[
\min_{0\le k\le T-1}\|\nabla f(y_k)\|^{2}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha\,T}.
\]
If $L$ is known, we may set the maximal admissible step size $\alpha=1/L$, resulting in the explicit constant
\[
C = 2L\bigl(f(x_0)-f_{\inf}\bigr),
\]
so that
\[
\min_{0\le k\le T-1}\|\nabla f(y_k)\|^{2}\le \frac{C}{T}.
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Convex $f$.}  When $f$ is convex, the same Lyapunov function yields the accelerated $O(1/k^{2})$ rate for function values (the classic Nesterov result).  The present proof reduces to the weaker $O(1/k)$ gradient bound because we do not exploit convexity.
\item \textbf{Exact Stationary Point.}  If at some iteration $\nabla f(y_k)=0$, then (13) shows $\Phi_{k+1}\le\Phi_k$, and the algorithm stays at the stationary point thereafter.
\item \textbf{Stochastic Gradients.}  Replacing $\nabla f(y_k)$ by an unbiased estimator $g_k$ with bounded variance $\sigma^{2}$, the same steps lead to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\|\nabla f(y_k)\|^{2}
\;\le\;
\frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\alpha T}
+\frac{L\sigma^{2}}{2},
\]
which after choosing $\alpha\asymp 1/\sqrt{T}$ gives the standard $O(1/\sqrt{T})$ stochastic rate.
\end{itemize}
