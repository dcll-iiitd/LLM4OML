\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimization problem  
\[
\min_{x\in\mathbb{R}^d}\; \Phi(x)\;:=\;f(x)+g(x),
\]
where  

\begin{itemize}[noitemsep]
    \item $f:\mathbb{R}^d\to\mathbb{R}$ is convex, differentiable and $L$‑smooth:
          \[
          \|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
          \]
    \item $g:\mathbb{R}^d\to\mathbb{R}\cup\{+\infty\}$ is convex, possibly non‑smooth, but \emph{prox‑computable}.
\end{itemize}
For a stepsize $\eta>0$ the proximal operator of $g$ is defined by
\[
\prox_{\eta g}(v):=\arg\min_{u\in\mathbb{R}^d}\Big\{ g(u)+\frac{1}{2\eta}\|u-v\|^2\Big\}.
\]
The \textbf{Proximal Gradient Method} iterates
\begin{equation}\label{eq:update}
x_{k+1}= \prox_{\eta g}\!\big(x_k-\eta\grad f(x_k)\big),\qquad k=0,1,\dots .
\end{equation}
The scalar $\eta$ is the (constant) stepsize; later we also discuss the diminishing stepsize case.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0\in\mathbb{R}^d$, stepsize $\eta>0$, max iter $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_k \gets \grad f(x_k)$ \Comment{gradient of smooth part}
    \State $v_k \gets x_k-\eta g_k$ \Comment{gradient step}
    \State $x_{k+1} \gets \prox_{\eta g}(v_k)$ \Comment{proximal mapping}
\EndFor
\State \textbf{Output:} $x_K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ with $g\equiv0$ (so $\prox_{\eta g}=I$).  
Take $w_0=0$, stepsize $\eta=0.1$, run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_k$ & $\grad f(w_k)=2(w_k-3)$ & $w_{k+1}=w_k-\eta\grad f(w_k)$\\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}
The objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.7712)=1.555$,
showing monotone decrease toward the optimum $w^\star=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{ass:smooth}
$f$ is convex and $L$‑smooth, i.e.
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{ass:convex}
$g$ is proper, closed and convex. Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:step}
The constant stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Existence of a solution]\label{ass:solution}
The set of minimizers 
\[
\mathcal{X}^\star:=\arg\min_{x}\Phi(x)
\]
is non‑empty; pick any $x^\star\in\mathcal{X}^\star$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:ergodic}
Let $\{x_k\}$ be generated by \eqref{eq:update} under Assumptions \ref{ass:smooth}–\ref{ass:step}.
Define the averaged iterate $\bar{x}_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k$.
Then for all $K\ge1$,
\[
\Phi(\bar{x}_K)-\Phi(x^\star)\;\le\;\frac{\|x_0-x^\star\|^2}{2\eta K}.
\]
Consequently $\Phi(\bar{x}_K)\to\Phi(x^\star)$ at a rate $O(1/K)$.
\end{theorem}
A stronger pointwise rate $O(1/k)$ holds when $g$ is additionally \emph{strongly convex}.

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
    \item \textbf{Non‑expansiveness of $\prox$}: for any $u,v$,
          \[
          \|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
          \tag{NE}
          \]
          This property is the cornerstone of the proof.
    \item \textbf{Stability w.r.t.\ stepsize}: the bound $0<\eta\le1/L$ guarantees that the descent inequality
          \[
          \Phi(x_{k+1})\le \Phi(x_k)-\frac{1}{2\eta}\|x_{k+1}-x_k\|^2
          \]
          holds, preventing divergence.
    \item \textbf{Comparison to Gradient Descent}: when $g\equiv0$, $\prox_{\eta g}=I$ and the method reduces to standard gradient descent with identical $O(1/k)$ ergodic rate.
    \item \textbf{Hyperparameter sensitivity}: If $\eta>1/L$, the smoothness inequality may be violated, and the proof breaks; empirically the algorithm may oscillate.
\end{itemize}

\section*{Discussion}
The theorem shows that the proximal gradient method inherits the $O(1/k)$ convergence of gradient descent for composite convex problems. The proof relies only on convexity, $L$‑smoothness of $f$, and the firmly non‑expansive nature of the proximal map. Extensions to stochastic gradients, accelerated schemes (FISTA), or varying stepsizes follow the same template with additional technical lemmas.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for the proximal operator]\label{lem:threepoint}
For any $x$, $v$ and any $u\in\mathbb{R}^d$,
\[
\langle u-\prox_{\eta g}(v),\,\prox_{\eta g}(v)-v\rangle
\ge \eta\big(g(\prox_{\eta g}(v))-g(u)\big).
\]
\end{lemma}
\begin{proof}
By definition of $\prox_{\eta g}(v)$ we have
\[
0\in \partial g(\prox_{\eta g}(v))+\frac{1}{\eta}\big(\prox_{\eta g}(v)-v\big),
\]
hence $\frac{1}{\eta}(v-\prox_{\eta g}(v))\in\partial g(\prox_{\eta g}(v))$.
Convexity of $g$ gives
\[
g(u)\ge g(\prox_{\eta g}(v))+\Big\langle \frac{1}{\eta}(v-\prox_{\eta g}(v)),\,u-\prox_{\eta g}(v)\Big\rangle,
\]
which rearranges to the claimed inequality. ∎
\end{proof}

\begin{lemma}[Descent Lemma for $L$‑smooth $f$]\label{lem:descent}
For any $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^\star\in\mathcal{X}^\star$ be any optimal solution.  
Denote $v_k:=x_k-\eta\grad f(x_k)$ and $x_{k+1}:=\prox_{\eta g}(v_k)$.  

\begin{enumerate}
\item[Step 1.] \textbf{Apply the three‑point identity (Lemma~\ref{lem:threepoint})} with $u=x^\star$, $v=v_k$:
\[
\langle x^\star-x_{k+1},\,x_{k+1}-v_k\rangle
\ge \eta\big(g(x_{k+1})-g(x^\star)\big).
\tag{1}
\]
\item[Step 2.] \textbf{Rewrite the left‑hand side of (1)} using $v_k=x_k-\eta\grad f(x_k)$:
\[
\langle x^\star-x_{k+1},\,x_{k+1}-x_k+\eta\grad f(x_k)\rangle
= \langle x^\star-x_{k+1},\,x_{k+1}-x_k\rangle
+\eta\langle x^\star-x_{k+1},\,\grad f(x_k)\rangle .
\tag{2}
\]
\item[Step 3.] \textbf{Use the identity}
\[
\|a-b\|^2 = \|a\|^2+\|b\|^2-2\langle a,b\rangle
\]
to express $\langle x^\star-x_{k+1},\,x_{k+1}-x_k\rangle$:
\[
\langle x^\star-x_{k+1},\,x_{k+1}-x_k\rangle
= \tfrac12\Big(\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2-\|x_{k+1}-x_k\|^2\Big).
\tag{3}
\]
\item[Step 4.] \textbf{Insert (3) into (2) and then into (1)}:
\[
\frac12\Big(\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2-\|x_{k+1}-x_k\|^2\Big)
+\eta\langle x^\star-x_{k+1},\grad f(x_k)\rangle
\ge \eta\big(g(x_{k+1})-g(x^\star)\big).\tag{4}
\]
\item[Step 5.] \textbf{Add and subtract $f(x_k)$ inside the inner product}. By convexity of $f$,
\[
f(x^\star)\ge f(x_k)+\langle\grad f(x_k),x^\star-x_k\rangle .
\tag{5}
\]
Rearrange (5) to obtain
\[
\langle x^\star-x_k,\grad f(x_k)\rangle \ge f(x^\star)-f(x_k).
\tag{6}
\]
\item[Step 6.] \textbf{Decompose $\langle x^\star-x_{k+1},\grad f(x_k)\rangle$} as
\[
\langle x^\star-x_{k+1},\grad f(x_k)\rangle
= \langle x^\star-x_k,\grad f(x_k)\rangle
+ \langle x_k-x_{k+1},\grad f(x_k)\rangle .
\tag{7}
\]
\item[Step 7.] \textbf{Insert (6) and (7) into (4)} and multiply the whole inequality by $2$:
\[
\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2-\|x_{k+1}-x_k\|^2
+2\eta\big(f(x^\star)-f(x_k)\big)
+2\eta\langle x_k-x_{k+1},\grad f(x_k)\rangle
\ge 2\eta\big(g(x_{k+1})-g(x^\star)\big).\tag{8}
\]
\item[Step 8.] \textbf{Group the terms involving $f$ and $g$ into $\Phi$}:
\[
2\eta\big(\Phi(x^\star)-f(x_k)-g(x_{k+1})\big)
=2\eta\big(\Phi(x^\star)-\Phi(x_k)+g(x_k)-g(x_{k+1})\big).
\]
Since $g$ is convex, $g(x_k)-g(x_{k+1})\le\langle s_k, x_k-x_{k+1}\rangle$ for some $s_k\in\partial g(x_{k+1})$,
but we do not need this bound because the term will be absorbed later.  
Rewrite (8) as
\[
\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2
\ge \|x_{k+1}-x_k\|^2
+2\eta\big(\Phi(x_k)-\Phi(x^\star)\big)
-2\eta\langle x_k-x_{k+1},\grad f(x_k)\rangle .
\tag{9}
\]
\item[Step 9.] \textbf{Upper‑bound the inner product using Cauchy–Schwarz and the stepsize restriction}.  
From the definition $v_k=x_k-\eta\grad f(x_k)$ we have
\[
x_k-x_{k+1}=v_k-\prox_{\eta g}(v_k)=v_k-x_{k+1}.
\]
Using the non‑expansiveness (NE) with $u=v_k$ and $v=x_{k+1}$,
\[
\|x_k-x_{k+1}\| = \|v_k-x_{k+1}\|\le\|v_k-\prox_{\eta g}(v_k)\| = \|x_k-\eta\grad f(x_k)-x_{k+1}\|.
\]
A more direct bound is obtained from the optimality condition of the proximal step:
\[
\frac{1}{\eta}(v_k-x_{k+1})\in\partial g(x_{k+1}),
\]
hence $\|v_k-x_{k+1}\| = \eta\|s_{k+1}\|$ for some $s_{k+1}\in\partial g(x_{k+1})$.
Nevertheless, the crucial inequality comes from the smoothness of $f$ (Lemma~\ref{lem:descent}) applied with $y=x_{k+1}$ and $x=x_k$:
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^2 .
\tag{10}
\]
Rearranging (10) gives
\[
\langle x_k-x_{k+1},\grad f(x_k)\rangle
\ge f(x_k)-f(x_{k+1})-\frac{L}{2}\|x_{k+1}-x_k\|^2 .
\tag{11}
\]
\item[Step 10.] \textbf{Plug (11) into (9)}:
\[
\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2
\ge \|x_{k+1}-x_k\|^2
+2\eta\big(\Phi(x_k)-\Phi(x^\star)\big)
-2\eta\big(f(x_k)-f(x_{k+1})\big)
+L\eta\|x_{k+1}-x_k\|^2 .
\tag{12}
\]
\item[Step 11.] \textbf{Cancel the $f$ terms and collect the norm terms}.  
Since $\Phi(x_k)=f(x_k)+g(x_k)$,
\[
2\eta\big(\Phi(x_k)-\Phi(x^\star)\big)-2\eta\big(f(x_k)-f(x_{k+1})\big)
=2\eta\big(g(x_k)-g(x^\star)\big)+2\eta\big(f(x_{k+1})-f(x^\star)\big).
\]
Thus (12) becomes
\[
\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2
\ge (1+L\eta)\|x_{k+1}-x_k\|^2
+2\eta\big(\Phi(x_{k+1})-\Phi(x^\star)\big).
\tag{13}
\]
\item[Step 12.] \textbf{Use the stepsize bound $\eta\le 1/L$}.  
Because $0<\eta\le 1/L$, we have $1+L\eta\ge 1$ and, more importantly,
\[
(1+L\eta)\|x_{k+1}-x_k\|^2 \ge \|x_{k+1}-x_k\|^2 .
\]
Dropping this non‑negative term from the right‑hand side of (13) yields the simpler inequality
\[
\|x^\star-x_k\|^2-\|x^\star-x_{k+1}\|^2
\ge 2\eta\big(\Phi(x_{k+1})-\Phi(x^\star)\big).
\tag{14}
\]
\item[Step 13.] \textbf{Telescoping sum}.  
Sum (14) over $k=0,\dots,K-1$:
\[
\|x^\star-x_0\|^2-\|x^\star-x_{K}\|^2
\ge 2\eta\sum_{k=0}^{K-1}\big(\Phi(x_{k+1})-\Phi(x^\star)\big).
\tag{15}
\]
Since the leftmost norm is non‑negative, we obtain
\[
\sum_{k=1}^{K}\big(\Phi(x_{k})-\Phi(x^\star)\big)
\le \frac{\|x_0-x^\star\|^2}{2\eta}.
\tag{16}
\]
\item[Step 14.] \textbf{Convert to an ergodic bound}.  
Define the average iterate $\bar{x}_K:=\frac{1}{K}\sum_{k=1}^{K}x_k$.
Convexity of $\Phi$ gives
\[
\Phi(\bar{x}_K)\le\frac{1}{K}\sum_{k=1}^{K}\Phi(x_k).
\]
Combining with (16) yields
\[
\Phi(\bar{x}_K)-\Phi(x^\star)
\le \frac{\|x_0-x^\star\|^2}{2\eta K}.
\tag{17}
\]
This completes the proof of Theorem~\ref{thm:ergodic}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From inequality (17) we read off the explicit constant:
\[
C:=\frac{\|x_0-x^\star\|^2}{2\eta},
\qquad
\Phi(\bar{x}_K)-\Phi(x^\star)\le \frac{C}{K}.
\]
If $g$ is $\mu$‑strongly convex ($\mu>0$), a similar derivation (using the strong convexity inequality) leads to a linear rate
\[
\Phi(x_k)-\Phi(x^\star)\le\big(1-\eta\mu\big)^k\big(\Phi(x_0)-\Phi(x^\star)\big),
\]
provided $\eta\le 1/(L+\mu)$.

\section*{Special Cases}
\begin{itemize}[noitemsep]
    \item \textbf{Pure gradient descent}: $g\equiv0\;\Rightarrow\;\prox_{\eta g}=I$, and (17) reduces to the classic $O(1/k)$ bound for smooth convex $f$.
    \item \textbf{Indicator of a convex set $C$}: $g=\iota_C$, then $\prox_{\eta g}$ is the Euclidean projection onto $C$, and the method becomes the projected gradient method with the same rate.
    \item \textbf{Lasso regularization}: $g(x)=\lambda\|x\|_1$, the proximal step is soft‑thresholding; the theorem guarantees convergence of the iterative soft‑thresholding algorithm.
\end{itemize}
