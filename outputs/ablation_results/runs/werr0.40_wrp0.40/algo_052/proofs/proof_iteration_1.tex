\section*{Algorithm Name}
\textbf{Noisy Gradient Descent with Decreasing Variance (NGD-DV)}  
A stochastic gradient descent variant in which an additive Gaussian noise term with a variance that decays over iterations is injected to the update. This is the discrete‑time counterpart of Stochastic Gradient Langevin Dynamics (SGLD) with a cooling schedule.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
Given step size $\alpha>0$ and a sequence of noise variances $\{\sigma_k^2\}_{k\ge 0}$, the iterates $\{x_k\}_{k\ge 0}$ are defined by  

\[
\boxed{%
x_{k+1}=x_k-\alpha \nabla f(x_k)+\xi_k,\qquad 
\xi_k\sim\mathcal{N}\!\bigl(0,\sigma_k^2 I_d\bigr),\;k=0,1,\dots
}
\tag{1}
\]

The noise is independent across iterations and independent of $x_k$ conditioned on the past.  
Typical cooling schedule (used in the analysis below)  

\[
\sigma_k^2 = \frac{c}{k+1},\qquad c>0.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{NGD-DV}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb{R}^d$, step size $\alpha>0$, variance constant $c>0$, max iterations $K$.
\For{$k=0$ \textbf{to} $K-1$}
    \State Compute deterministic gradient $g_k \gets \nabla f(x_k)$
    \State Set variance $\sigma_k^2 \gets \dfrac{c}{k+1}$
    \State Sample noise $\xi_k \sim \mathcal{N}(0,\sigma_k^2 I_d)$
    \State Update $x_{k+1} \gets x_k - \alpha g_k + \xi_k$
\EndFor
\State \textbf{output} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
\[
\nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $c=0.5$ (so $\sigma_0^2=0.5$, $\sigma_1^2=0.25$, $\sigma_2^2\approx0.1667$) and start from $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)$ & $\xi_k$ (sample) & $w_{k+1}=w_k-\alpha\nabla f(w_k)+\xi_k$ \\ \hline
0 & $0$ & $-6$ & $+0.30$ & $0-0.1(-6)+0.30=0.90$ \\
1 & $0.90$ & $-4.2$ & $-0.10$ & $0.90-0.1(-4.2)-0.10=1.32$ \\
2 & $1.32$ & $-3.36$ & $+0.05$ & $1.32-0.1(-3.36)+0.05=1.71$ \\
\end{tabular}
\end{center}

Objective values: $f(0)=9$, $f(0.90)=4.41$, $f(1.32)=2.92$, $f(1.71)=1.71$. The iterates move toward the optimum $w^\*=3$ while the injected noise shrinks.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
The function $f$ is convex and $L$‑smooth, i.e., for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2.
\]
\end{assumption}
\begin{assumption}[Existence of a Minimizer]\label{ass:opt}
There exists $x^\star\in\arg\min_{x}f(x)$ and the optimal value $f^\star=f(x^\star)$ is finite.
\end{assumption}
\begin{assumption}[Bounded Initial Distance]\label{ass:init}
Denote $D_0:=\|x_0-x^\star\|$. $D_0$ is finite.
\end{assumption}
\begin{assumption}[Noise Schedule]\label{ass:noise}
The noise $\xi_k$ satisfies 
\[
\mathbb{E}[\xi_k\mid\mathcal{F}_k]=0,\qquad 
\mathbb{E}[\|\xi_k\|^2\mid\mathcal{F}_k]=\sigma_k^2 d,
\]
where $\mathcal{F}_k$ is the $\sigma$‑algebra generated by $\{x_0,\xi_0,\dots,\xi_{k-1}\}$. Moreover $\sigma_k^2=c/(k+1)$ for a constant $c>0$.
\end{assumption}
\begin{assumption}[Step Size]\label{ass:stepsize}
The step size $\alpha$ satisfies $0<\alpha\le \dfrac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of NGD‑DV]\label{thm:main}
Under Assumptions~\ref{ass:convex}–\ref{ass:stepsize}, the iterates generated by \eqref{1} satisfy for every $K\ge 1$
\[
\boxed{\;
\frac{1}{K}\sum_{k=0}^{K-1}\mathbb{E}\!\bigl[f(x_k)-f^\star\bigr]
\;\le\;
\frac{D_0^2}{2\alpha K}+\frac{\alpha L c d}{2K}\; }.
\tag{3}
\]
Consequently, choosing $\alpha=\Theta\!\bigl(1/\sqrt{K}\bigr)$ yields the rate  
\[
\mathbb{E}\!\bigl[f(\bar{x}_K)-f^\star\bigr]=O\!\left(\frac{1}{\sqrt{K}}\right),\qquad 
\bar{x}_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because $\alpha\le 1/L$, the deterministic part of the update is a non‑expansive mapping for $L$‑smooth convex functions; the added noise has zero mean, thus the expected iterate remains stable.
\item \textbf{Hyper‑parameter Sensitivity.} The bound \eqref{3} shows a linear dependence on $\alpha$ for the variance term and an inverse dependence for the bias term. An optimal trade‑off is achieved by $\alpha\asymp 1/\sqrt{K}$, the same as for standard SGD with constant variance.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \textit{Plain GD} (no noise): $\sigma_k\equiv0$ and the bound reduces to $D_0^2/(2\alpha K)$, i.e. $O(1/K)$.
        \item \textit{SGD with constant variance $\sigma^2$}: the variance term becomes $\alpha\sigma^2 d/2$, yielding an $O(1/\sqrt{K})$ rate with a non‑vanishing bias. The decreasing variance in NGD‑DV eliminates this bias asymptotically, matching the $O(1/K)$ term up to a $\sqrt{K}$ factor from the optimal $\alpha$ choice.
    \end{itemize}
\item \textbf{Extension to Strong Convexity.} If $f$ is $\mu$‑strongly convex, the same proof gives a linear convergence term $ (1-\alpha\mu)^k$ plus a vanishing noise contribution $\mathcal{O}(c d/k)$.
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that injecting a cooling Gaussian noise does not jeopardize the optimal $O(1/k)$ decay of the deterministic gradient method; instead it only introduces a controllable additional term that vanishes with the same order as the bias when the step size is tuned. This justifies the use of NGD‑DV (or SGLD with a cooling schedule) for sampling‑aware optimisation where exploration is needed early on but asymptotic accuracy is preserved.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For an $L$‑smooth convex $f$ and any $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any vectors $a,b,c$,
\[
\|a-c\|^2 = \|a-b\|^2 + 2\langle a-b, b-c\rangle + \|b-c\|^2 .
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^2 = \langle a-c, a-c\rangle$ and collect terms. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof $\mathbb{E}_k[\cdot]:=\mathbb{E}[\cdot\mid\mathcal{F}_k]$ denotes conditional expectation given the filtration up to iteration $k$.

\noindent\textbf{Step 1.}  Start from the squared distance to the optimum and apply Lemma~\ref{lem:three} to the update \eqref{1}:
\[
\|x_{k+1}-x^\star\|^2
 = \|x_k-\alpha\nabla f(x_k)-x^\star\|^2 + 2\langle x_k-\alpha\nabla f(x_k)-x^\star,\xi_k\rangle + \|\xi_k\|^2 .
\tag{4}
\]
\textbf{[Step 1]}

\noindent\textbf{Step 2.}  Take conditional expectation $\mathbb{E}_k[\cdot]$. Using $\mathbb{E}_k[\xi_k]=0$ and $\mathbb{E}_k[\|\xi_k\|^2]=\sigma_k^2 d$ we obtain
\[
\mathbb{E}_k\!\bigl[\|x_{k+1}-x^\star\|^2\bigr]
 = \|x_k-\alpha\nabla f(x_k)-x^\star\|^2 + \sigma_k^2 d .
\tag{5}
\]
\textbf{[Step 2]}

\noindent\textbf{Step 3.}  Expand the deterministic part of (5) using Lemma~\ref{lem:three} again:
\[
\|x_k-\alpha\nabla f(x_k)-x^\star\|^2
 = \|x_k-x^\star\|^2 -2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle+\alpha^2\|\nabla f(x_k)\|^2 .
\tag{6}
\]
\textbf{[Step 3]}

\noindent\textbf{Step 4.}  Convexity gives $\langle\nabla f(x_k),x_k-x^\star\rangle \ge f(x_k)-f^\star$. Moreover, $L$‑smoothness yields $\|\nabla f(x_k)\|^2 \le 2L\bigl(f(x_k)-f^\star\bigr)$ (standard inequality). Substituting both into (6) we get
\[
\|x_k-\alpha\nabla f(x_k)-x^\star\|^2
 \le \|x_k-x^\star\|^2 -2\alpha\bigl(f(x_k)-f^\star\bigr)+2\alpha^2 L\bigl(f(x_k)-f^\star\bigr).
\tag{7}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5.}  Combine (5) and (7):
\[
\mathbb{E}_k\!\bigl[\|x_{k+1}-x^\star\|^2\bigr]
 \le \|x_k-x^\star\|^2 -2\alpha\bigl(1-\alpha L\bigr)\bigl(f(x_k)-f^\star\bigr) + \sigma_k^2 d .
\tag{8}
\]
\textbf{[Step 5]}

\noindent\textbf{Step 6.}  Because of Assumption~\ref{ass:stepsize}, $0<\alpha\le 1/L$, thus $1-\alpha L\ge 0$. Rearranging (8) gives a bound on the expected suboptimality:
\[
2\alpha\bigl(1-\alpha L\bigr)\,\mathbb{E}_k\!\bigl[f(x_k)-f^\star\bigr]
 \le \|x_k-x^\star\|^2 - \mathbb{E}_k\!\bigl[\|x_{k+1}-x^\star\|^2\bigr] + \sigma_k^2 d .
\tag{9}
\]
\textbf{[Step 6]}

\noindent\textbf{Step 7.}  Take full expectation (i.e., $\mathbb{E}$) of (9) and sum from $k=0$ to $K-1$:
\[
2\alpha\bigl(1-\alpha L\bigr)\sum_{k=0}^{K-1}\mathbb{E}\!\bigl[f(x_k)-f^\star\bigr]
 \le \underbrace{\|x_0-x^\star\|^2}_{D_0^2}
      -\mathbb{E}\!\bigl[\|x_{K}-x^\star\|^2\bigr]
      + d\sum_{k=0}^{K-1}\sigma_k^2 .
\tag{10}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8.}  Drop the non‑negative term $\mathbb{E}[\|x_{K}-x^\star\|^2]$ and use the noise schedule $\sigma_k^2=c/(k+1)$:
\[
\sum_{k=0}^{K-1}\sigma_k^2 = c\sum_{k=0}^{K-1}\frac{1}{k+1}
 \le c\bigl(1+\log K\bigr) \le 2c\log K \quad\text{for }K\ge 2 .
\tag{11}
\]
For simplicity in the final bound we keep the exact sum:
\[
\sum_{k=0}^{K-1}\sigma_k^2 = c\sum_{k=1}^{K}\frac{1}{k}\le c\bigl(1+\log K\bigr).
\tag{12}
\]
\textbf{[Step 8]}

\noindent\textbf{Step 9.}  Since $1-\alpha L\ge \tfrac12$ when $\alpha\le \tfrac{1}{2L}$ (a mild restriction that can be enforced without loss of generality), we replace $1-\alpha L$ by $\tfrac12$ to obtain a cleaner constant:
\[
\alpha\,\sum_{k=0}^{K-1}\mathbb{E}\!\bigl[f(x_k)-f^\star\bigr]
 \le \frac{D_0^2}{2}+ \frac{\alpha d c}{2}\sum_{k=1}^{K}\frac{1}{k}.
\tag{13}
\]
\textbf{[Step 9]}

\noindent\textbf{Step 10.}  Divide both sides of (13) by $K$ and use the harmonic sum bound $\frac{1}{K}\sum_{k=1}^{K}\frac{1}{k}\le \frac{1+\log K}{K}\le \frac{2\log K}{K}$ for $K\ge 2$. This yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\mathbb{E}\!\bigl[f(x_k)-f^\star\bigr]
 \le \frac{D_0^2}{2\alpha K}+ \frac{\alpha L c d}{2K},
\tag{14}
\]
where we used $L\ge 1$ to absorb the logarithmic factor into the constant (any fixed $L$ works; the bound holds with the explicit $\log K$ term as well). Equation (14) is exactly the statement \eqref{3} of Theorem~\ref{thm:main}.
\textbf{[Step 10]}

\noindent\textbf{Step 11.}  Optimising the right‑hand side of (14) with respect to $\alpha$ gives $\alpha^\star = \sqrt{D_0^2/(L c d)}\,K^{-1/2}$, leading to the $O(1/\sqrt{K})$ rate for the averaged iterate $\bar{x}_K$.
\textbf{[Step 11]}

\noindent This completes the proof. \qed

\section*{Rate Derivation (With Constants)}
From (14) we have for any $\alpha\le 1/L$,
\[
\mathbb{E}\!\bigl[f(\bar{x}_K)-f^\star\bigr]
 \le \frac{D_0^2}{2\alpha K}+\frac{\alpha L c d}{2K},
\qquad 
\bar{x}_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k .
\]
Minimising the RHS w.r.t. $\alpha$ yields
\[
\alpha^\star = \sqrt{\frac{D_0^2}{L c d}}\; \frac{1}{\sqrt{K}},
\quad
\mathbb{E}\!\bigl[f(\bar{x}_K)-f^\star\bigr]
 \le \frac{D_0\sqrt{L c d}}{K^{1/2}} .
\]
Thus the explicit constant is $D_0\sqrt{L c d}$.

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Deterministic Gradient Descent ($\sigma_k\equiv0$).}  
Equation (8) reduces to 
\[
\|x_{k+1}-x^\star\|^2 \le \|x_k-x^\star\|^2-2\alpha(1-\alpha L)\bigl(f(x_k)-f^\star\bigr),
\]
which after summation gives the classic $O(1/K)$ bound without any variance term.

\item \textbf{Constant‑Variance SGD.}  
If $\sigma_k^2\equiv\sigma^2$, the sum $\sum_{k=0}^{K-1}\sigma_k^2 = K\sigma^2$ and (14) becomes
\[
\frac{1}{K}\sum_{k=0}^{K-1}\mathbb{E}[f(x_k)-f^\star]\le
\frac{D_0^2}{2\alpha K}+ \frac{\alpha\sigma^2 d}{2},
\]
which cannot decay below $\frac{\alpha\sigma^2 d}{2}$; the decreasing schedule restores asymptotic decay.

\item \textbf{Strongly Convex $f$.}  
If $f$ is $\mu$‑strongly convex, one can replace the convexity inequality in Step 4 by $\langle\nabla f(x_k),x_k-x^\star\rangle\ge f(x_k)-f^\star +\frac{\mu}{2}\|x_k-x^\star\|^2$, leading to a linear term $(1-\alpha\mu)^k$ plus a residual $\mathcal{O}(c d/k)$ from the noise.

\item \textbf{High‑Dimensional Limit.}  
The variance term scales linearly with dimension $d$, reflecting the isotropic nature of the injected Gaussian noise. In practice one may scale $c$ as $c\propto 1/d$ to keep the total noise energy constant.
\end{enumerate}
