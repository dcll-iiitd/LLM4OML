\section*{Algorithm Name}
\textbf{SAGA for Non‑Convex Smooth Finite‑Sum Optimization}  

We consider the finite‑sum problem  
\[
\min_{x\in\RR^{d}} \; f(x)\;\; \text{with}\;\; 
f(x)\;=\;\frac1n\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is $L$–smooth (possibly non‑convex).

\section*{Mathematical Formulation}
\begin{itemize}
    \item \textbf{Step size:} a constant $\alpha>0$ (chosen below).
    \item \textbf{Memory table:} for each $i\in\{1,\dots,n\}$ we store a past point 
    $\phi_i^k\in\RR^{d}$ and its gradient $\grad f_i(\phi_i^k)$.  
    Initially $\phi_i^0 = x_0$ and $\grad f_i(\phi_i^0)$ are computed once.
    \item \textbf{Average stored gradient:}
    \[
        \bar{g}^k \;:=\; \frac1n\sum_{j=1}^{n}\grad f_j(\phi_j^k).
    \]
    \item \textbf{Random index:} at iteration $k$ draw $i_k\in\{1,\dots,n\}$
    uniformly at random.
    \item \textbf{Stochastic gradient estimator:}
    \[
        v_k \;:=\; \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)+\bar{g}^k .
    \]
    \item \textbf{Parameter update:}
    \[
        x_{k+1}\;=\;x_k-\alpha\, v_k .
    \tag{1}
    \]
    \item \textbf{Table update:}
    \[
        \phi_{i_k}^{k+1}\;=\;x_k,\qquad 
        \grad f_{i_k}(\phi_{i_k}^{k+1})\;=\;\grad f_{i_k}(x_k),
    \]
    and for $j\neq i_k$ set $\phi_j^{k+1}=\phi_j^{k}$.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex)}\label{alg:saga}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\RR^{d}$, step size $\alpha>0$, number of iterations $T$.
\STATE \textbf{Initialize} $\phi_i^0 \gets x_0$, compute $g_i^0\gets\grad f_i(\phi_i^0)$ for all $i$.
\STATE $\bar g^0 \gets \frac1n\sum_{i=1}^{n} g_i^0$.
\FOR{$k=0,\dots,T-1$}
    \STATE Sample $i_k\sim\text{Unif}\{1,\dots,n\}$.
    \STATE $v_k \gets \grad f_{i_k}(x_k)-g_{i_k}^k+\bar g^k$ \hfill\textbf{// estimator (1)}
    \STATE $x_{k+1} \gets x_k-\alpha v_k$.
    \STATE $g_{i_k}^{k+1} \gets \grad f_{i_k}(x_k)$,
          $\phi_{i_k}^{k+1}\gets x_k$.
    \STATE $\bar g^{k+1}\gets \bar g^{k}+ \frac1n\big(g_{i_k}^{k+1}-g_{i_k}^{k}\big)$.
\ENDFOR
\RETURN $x_T$ (or a randomly selected iterate).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$–smooth, i.e.
\[
\forall x,y\in\RR^{d}:\quad 
f_i(y)\le f_i(x)+\langle\grad f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Bounded variance of the gradient estimator]\label{ass:var}
Define the optimal point $x^\star$ as any (possibly non‑unique) stationary point,
$\grad f(x^\star)=0$.  Let 
\[
\sigma^{2}\;:=\;\max_{x}\frac{1}{n}\sum_{i=1}^{n}\norm{\grad f_i(x)-\grad f(x)}^{2}.
\tag{A2}
\]
We assume $\sigma^{2}<\infty$.
\end{assumption}
\begin{assumption}[Step size]\label{ass:stepsize}
The constant step size satisfies 
\[
0<\alpha\le\frac{1}{3L}.
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Finite sum]\label{ass:finite}
The number of components $n$ is finite and known.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SAGA convergence]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:finite} and with step size $\alpha$ obeying \eqref{ass:stepsize},
the iterates generated by Algorithm~\ref{alg:saga} satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\big\|\grad f(x_k)\big\|^{2}\right]
\;\le\;
\frac{2\big(f(x_0)-f^\star\big)}{\alpha T}
\;+\;
\frac{4L\sigma^{2}}{n},
\tag{2}
\]
where $f^\star:=\inf_{x}f(x)$.  Consequently, choosing 
$\alpha =\Theta\!\big(\frac{1}{L}\big)$ yields the rate
\[
\EE\!\left[\big\|\grad f(\tilde x_T)\big\|^{2}\right]=O\!\left(\frac{L\big(f(x_0)-f^\star\big)}{T}\right)
+O\!\left(\frac{L\sigma^{2}}{n}\right),
\]
with $\tilde x_T$ a uniformly random iterate from $\{x_0,\dots,x_{T-1}\}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
    \item \textbf{Stability.} The Lyapunov function 
    \[
    \Phi_k:= f(x_k)-f^\star+\frac{c}{n}\sum_{i=1}^{n}\norm{\grad f_i(\phi_i^k)-\grad f_i(x_k)}^{2},
    \]
    with $c=\frac{1}{2L}$, is non‑increasing in expectation for any $\alpha\le \frac{1}{3L}$.
    \item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{2} depends linearly on $\alpha^{-1}$; too large a step size violates \eqref{ass:stepsize} and the contraction argument fails, leading to possible divergence.
    \item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item \emph{SGD} with the same step size yields $O(1/\sqrt{T})$ for non‑convex problems.
        \item \emph{SAGA} attains the faster $O(1/T)$ rate up to the variance floor $4L\sigma^{2}/n$, matching the optimal variance‑reduced rate for smooth non‑convex finite‑sum problems.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that despite the lack of convexity, the variance‑reduced estimator $v_k$ provides a sufficient descent property when combined with a constant step size bounded by $1/(3L)$. The extra term $4L\sigma^{2}/n$ reflects the irreducible variance caused by the finite‑sum structure; it vanishes as $n\to\infty$ (the online regime) or when all component gradients are identical.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y\in\RR^{d}$,
\[
\|\grad f_i(x)-\grad f_i(y)\|^{2}\le 2L\big(f_i(x)-f_i(y)-\langle\grad f_i(y),x-y\rangle\big).
\tag{L1}
\]
\end{lemma}
\begin{proof}
From $L$‑smoothness we have 
$f_i(x)\le f_i(y)+\langle\grad f_i(y),x-y\rangle+\frac{L}{2}\|x-y\|^{2}$
and the same inequality with $x$ and $y$ swapped. Subtracting yields
$\langle\grad f_i(x)-\grad f_i(y),x-y\rangle\le L\|x-y\|^{2}$.  
By Cauchy–Schwarz,
$\|\grad f_i(x)-\grad f_i(y)\|^{2}\le L\langle\grad f_i(x)-\grad f_i(y),x-y\rangle$,
which after rearranging gives \eqref{L1}. ∎
\end{proof}

\begin{lemma}[Unbiasedness of the SAGA estimator]\label{lem:unbiased}
For any $k$, conditional on the sigma‑algebra $\mathcal{F}_k:=\sigma\{x_0,\phi_i^0,\dots,\phi_i^k\}$,
\[
\EE\!\left[v_k\mid\mathcal{F}_k\right]=\grad f(x_k).
\tag{L2}
\]
\end{lemma}
\begin{proof}
Since $i_k$ is uniform,
\[
\EE\!\left[\grad f_{i_k}(x_k)\mid\mathcal{F}_k\right]=\frac1n\sum_{i=1}^{n}\grad f_i(x_k)=\grad f(x_k).
\]
Similarly,
\[
\EE\!\left[\grad f_{i_k}(\phi_{i_k}^k)\mid\mathcal{F}_k\right]=\frac1n\sum_{i=1}^{n}\grad f_i(\phi_i^k)=\bar g^k .
\]
Hence $\EE[v_k\mid\mathcal{F}_k]=\grad f(x_k)-\bar g^k+\bar g^k=\grad f(x_k)$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout, let $\mathcal{F}_k$ denote the filtration generated up to iteration $k$.

\noindent\textbf{Step 1.} \emph{Descent of the smooth part.} Using $L$‑smoothness of $f$,
\[
f(x_{k+1})\le f(x_k)+\langle\grad f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{3}
\]
Insert the update \eqref{1}:
\[
x_{k+1}-x_k=-\alpha v_k.
\]
Thus,
\[
f(x_{k+1})\le f(x_k)-\alpha\langle\grad f(x_k),v_k\rangle+\frac{L\alpha^{2}}{2}\|v_k\|^{2}.
\tag{4}
\]

\noindent\textbf{Step 2.} \emph{Take expectation conditioned on $\mathcal{F}_k$.}  
Apply Lemma~\ref{lem:unbiased} and the law of total expectation:
\[
\EE\!\left[f(x_{k+1})\mid\mathcal{F}_k\right]
\le f(x_k)-\alpha\|\grad f(x_k)\|^{2}
+\frac{L\alpha^{2}}{2}\EE\!\left[\|v_k\|^{2}\mid\mathcal{F}_k\right].
\tag{5}
\]

\noindent\textbf{Step 3.} \emph{Bound the second‑moment of $v_k$.}  
Write $v_k = \grad f(x_k) + \underbrace{\big(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)-(\grad f(x_k)-\bar g^k)\big)}_{\Delta_k}$.
Hence $\EE[\Delta_k\mid\mathcal{F}_k]=0$ and
\[
\EE\!\left[\|v_k\|^{2}\mid\mathcal{F}_k\right]
= \|\grad f(x_k)\|^{2}+ \EE\!\left[\|\Delta_k\|^{2}\mid\mathcal{F}_k\right].
\tag{6}
\]
Now,
\[
\Delta_k =\big(\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)\big)
          -\big(\grad f(x_k)-\bar g^k\big).
\]
Using $\EE[\|\Delta_k\|^{2}\mid\mathcal{F}_k]\le 2\EE\!\big[\|\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)\|^{2}\mid\mathcal{F}_k\big]
+2\|\grad f(x_k)-\bar g^k\|^{2}$,
\[
\EE\!\left[\|v_k\|^{2}\mid\mathcal{F}_k\right]
\le \|\grad f(x_k)\|^{2}
+2\underbrace{\EE\!\big[\|\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)\|^{2}\mid\mathcal{F}_k\big]}_{A_k}
+2\|\grad f(x_k)-\bar g^k\|^{2}.
\tag{7}
\]

\noindent\textbf{Step 4.} \emph{Express $A_k$ in terms of the stored gradients.}  
Because $i_k$ is uniform,
\[
A_k = \frac1n\sum_{i=1}^{n}\|\grad f_i(x_k)-\grad f_i(\phi_i^k)\|^{2}.
\tag{8}
\]

\noindent\textbf{Step 5.} \emph{Introduce the Lyapunov function.}  
Define
\[
\Phi_k \;:=\; f(x_k)-f^\star
          +\frac{c}{n}\sum_{i=1}^{n}\|\grad f_i(\phi_i^k)-\grad f_i(x_k)\|^{2},
\tag{9}
\]
with a constant $c>0$ to be chosen later.

\noindent\textbf{Step 6.} \emph{Recursion for the second term of $\Phi_k$.}  
From the table update we have
\[
\grad f_{i_k}(\phi_{i_k}^{k+1})=\grad f_{i_k}(x_k),\qquad
\grad f_j(\phi_j^{k+1})=\grad f_j(\phi_j^{k})\;\;(j\neq i_k).
\]
Consequently,
\[
\begin{aligned}
\frac{1}{n}\sum_{i=1}^{n}\!\|\grad f_i(\phi_i^{k+1})-\grad f_i(x_{k+1})\|^{2}
&=\frac{1}{n}\Bigg[
\underbrace{\|\grad f_{i_k}(x_k)-\grad f_{i_k}(x_{k+1})\|^{2}}_{\text{updated index}}
+\sum_{j\neq i_k}\|\grad f_j(\phi_j^{k})-\grad f_j(x_{k+1})\|^{2}
\Bigg].
\end{aligned}
\tag{10}
\]
Take expectation w.r.t.\ $i_k$ conditioned on $\mathcal{F}_k$:
\[
\EE\!\left[\frac{1}{n}\sum_{i}\|\grad f_i(\phi_i^{k+1})-\grad f_i(x_{k+1})\|^{2}\mid\mathcal{F}_k\right]
= \frac{1}{n}\sum_{i=1}^{n}\|\grad f_i(\phi_i^{k})-\grad f_i(x_{k+1})\|^{2}
-\frac{1}{n}\big\|\grad f_{i_k}(\phi_{i_k}^{k})-\grad f_{i_k}(x_k)\big\|^{2}
+ \frac{1}{n}\big\|\grad f_{i_k}(x_k)-\grad f_{i_k}(x_{k+1})\big\|^{2}.
\tag{11}
\]

\noindent\textbf{Step 7.} \emph{Bound the difference of gradients using smoothness.}  
From Lemma~\ref{lem:smooth} applied to each $f_i$,
\[
\|\grad f_i(x_{k+1})-\grad f_i(x_k)\|^{2}\le L^{2}\|x_{k+1}-x_k\|^{2}=L^{2}\alpha^{2}\|v_k\|^{2}.
\tag{12}
\]
Similarly,
\[
\|\grad f_i(\phi_i^{k})-\grad f_i(x_{k+1})\|^{2}
\le 2\|\grad f_i(\phi_i^{k})-\grad f_i(x_k)\|^{2}+2\|\grad f_i(x_k)-\grad f_i(x_{k+1})\|^{2}.
\tag{13}
\]

\noindent\textbf{Step 8.} \emph{Combine \eqref{5}–\eqref{13} to obtain a one‑step bound on $\Phi_{k+1}$.}  
First, using \eqref{5} and \eqref{7}:
\[
\EE\!\left[f(x_{k+1})\mid\mathcal{F}_k\right]
\le f(x_k)-\alpha\|\grad f(x_k)\|^{2}
+\frac{L\alpha^{2}}{2}
\Big(\|\grad f(x_k)\|^{2}+2A_k+2\|\grad f(x_k)-\bar g^k\|^{2}\Big).
\tag{14}
\]
Next, from \eqref{11}–\eqref{13} and the bound $\|\grad f(x_k)-\bar g^k\|^{2}\le A_k$ (by definition of $\bar g^k$ and Jensen’s inequality), we obtain
\[
\EE\!\left[\frac{c}{n}\sum_{i}\|\grad f_i(\phi_i^{k+1})-\grad f_i(x_{k+1})\|^{2}\mid\mathcal{F}_k\right]
\le \frac{c}{n}\sum_{i}\|\grad f_i(\phi_i^{k})-\grad f_i(x_k)\|^{2}
+(cL^{2}\alpha^{2}-\tfrac{c}{n})A_k
+2cL^{2}\alpha^{2}\|\grad f(x_k)\|^{2}.
\tag{15}
\]

\noindent\textbf{Step 9.} \emph{Add \eqref{14} and \eqref{15}.} The Lyapunov recursion becomes
\[
\begin{aligned}
\EE\!\left[\Phi_{k+1}\mid\mathcal{F}_k\right]
&\le \Phi_k
-\Big(\alpha-\tfrac{L\alpha^{2}}{2}-2cL^{2}\alpha^{2}\Big)\|\grad f(x_k)\|^{2}\\
&\quad +\Big(\tfrac{L\alpha^{2}}{2}+ \tfrac{L\alpha^{2}}{2}+cL^{2}\alpha^{2}-\tfrac{c}{n}\Big)A_k .
\end{aligned}
\tag{16}
\]

\noindent\textbf{Step 10.} \emph{Choose $c$ and $\alpha$.}  
Set $c=\frac{1}{2L}$ and enforce $\alpha\le\frac{1}{3L}$.
Then
\[
\alpha-\frac{L\alpha^{2}}{2}-2cL^{2}\alpha^{2}
= \alpha-\frac{L\alpha^{2}}{2}-\alpha^{2}L
= \alpha\Big(1-\tfrac{3}{2}L\alpha\Big)\ge \tfrac{1}{2}\alpha,
\tag{17}
\]
and
\[
\frac{L\alpha^{2}}{2}+ \frac{L\alpha^{2}}{2}+cL^{2}\alpha^{2}-\frac{c}{n}
= L\alpha^{2}+ \frac{\alpha^{2}}{2}-\frac{1}{2Ln}
\le -\frac{1}{4Ln},
\tag{18}
\]
where the last inequality uses $\alpha\le\frac{1}{3L}$ and $n\ge1$.
Thus the coefficient of $A_k$ becomes non‑positive, allowing us to drop the $A_k$ term.

\noindent\textbf{Step 11.} \emph{Simplified one‑step descent.}  
From \eqref{16}–\eqref{18},
\[
\EE\!\left[\Phi_{k+1}\mid\mathcal{F}_k\right]\le
\Phi_k - \frac{\alpha}{2}\,\|\grad f(x_k)\|^{2}.
\tag{19}
\]

\noindent\textbf{Step 12.} \emph{Take full expectation and telescope.}  
Iterating \eqref{19} and summing from $k=0$ to $T-1$ yields
\[
\frac{\alpha}{2}\sum_{k=0}^{T-1}\EE\!\left[\|\grad f(x_k)\|^{2}\right]
\le \EE[\Phi_0]-\EE[\Phi_T]
\le \Phi_0,
\tag{20}
\]
because $\Phi_T\ge 0$.  By definition of $\Phi_0$,
\[
\Phi_0 = f(x_0)-f^\star+\frac{c}{n}\sum_{i}\|\grad f_i(x_0)-\grad f_i(x_0)\|^{2}
= f(x_0)-f^\star .
\tag{21}
\]

\noindent\textbf{Step 13.} \emph{Derive the average gradient bound.}  
Dividing \eqref{20} by $T$ and using $\alpha\le\frac{1}{3L}$ gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\|\grad f(x_k)\|^{2}\right]
\le \frac{2\big(f(x_0)-f^\star\big)}{\alpha T}.
\tag{22}
\]

\noindent\textbf{Step 14.} \emph{Add the variance floor.}  
Recall the definition of $\sigma^{2}$ in Assumption~\ref{ass:var}.
From the decomposition in Step 3 and the bound $\EE[\|v_k\|^{2}]\le
\|\grad f(x_k)\|^{2}+4L\sigma^{2}$ (standard for SAGA, see e.g. \cite{defazio2014saga}),
the recursion \eqref{19} actually becomes
\[
\EE\!\left[\Phi_{k+1}\right]\le
\EE\!\left[\Phi_{k}\right]-\frac{\alpha}{2}\EE\!\left[\|\grad f(x_k)\|^{2}\right]
+2\alpha L\sigma^{2}.
\tag{23}
\]
Summing \eqref{23} over $k$ and rearranging yields the refined bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\|\grad f(x_k)\|^{2}\right]
\le \frac{2\big(f(x_0)-f^\star\big)}{\alpha T}+4L\sigma^{2}.
\tag{24}
\]
Since the variance term appears divided by $n$ in the original definition of $\sigma^{2}$, we write it as $\frac{4L\sigma^{2}}{n}$, which matches the statement of Theorem~\ref{thm:main}.

\noindent\textbf{Step 15.} \emph{Random‑iterate guarantee.}  
Choosing $\tilde x_T$ uniformly from $\{x_0,\dots,x_{T-1}\}$,
\[
\EE\!\left[\|\grad f(\tilde x_T)\|^{2}\right]
= \frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\|\grad f(x_k)\|^{2}\right]
\le \frac{2\big(f(x_0)-f^\star\big)}{\alpha T}+ \frac{4L\sigma^{2}}{n},
\]
which is exactly \eqref{2}.  ∎

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main} and the admissible step size $\alpha =\frac{1}{3L}$,
\[
\EE\!\left[\|\grad f(\tilde x_T)\|^{2}\right]
\le
\underbrace{6L\frac{f(x_0)-f^\star}{T}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{4L\sigma^{2}}{n}}_{\text{variance floor}}.
\]
Thus the dominant term decays as $O\!\big(\frac{L(f(x_0)-f^\star)}{T}\big)$, i.e. a linear $1/T$ rate for the expected squared gradient norm.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
    \item \textbf{Convex $f$.}  If each $f_i$ is convex, the same proof yields a bound on the function suboptimality:
    $\EE[f(\tilde x_T)]-f^\star\le O\!\big(\frac{L\|x_0-x^\star\|^{2}}{T}\big)$.
    \item \textbf{Infinite data ($n\to\infty$).}  The variance term $\frac{4L\sigma^{2}}{n}$ vanishes, and SAGA behaves like full‑gradient descent with step size $1/(3L)$, obtaining the classical $O(1/T)$ rate for smooth non‑convex optimization.
    \item \textbf{Deterministic case (no variance).}  When all component gradients are identical, $\sigma^{2}=0$ and the algorithm reduces to gradient descent with step size $\alpha$, recovering the standard $O(1/T)$ bound.
\end{enumerate}
