\section*{Stochastic Newton Method (SNewton)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be twice differentiable.  
At iteration $t$ we draw a random mini‑batch (or a random sample) and compute  

\[
g_t \;:=\; \nabla f(x_t; \xi_t),\qquad 
H_t \;:=\; \nabla^{2} f(x_t; \xi_t),
\]

where $\xi_t$ denotes the randomness.  
The update rule is  

\[
\boxed{\,x_{t+1}=x_t-\alpha\,H_t^{-1} g_t\,},
\tag{1}
\]

with a fixed stepsize $\alpha>0$.  
When $H_t$ is not invertible we replace $H_t^{-1}$ by the Moore–Penrose pseudoinverse or add a ridge term; for the analysis we assume $H_t$ is positive definite a.s.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Newton (SNewton)}\label{alg:snewton}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, initial point $x_0\in\R^{d}$, maximal iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Sample random variable $\xi_t$ (e.g., a minibatch).
    \STATE Compute stochastic gradient $g_t=\nabla f(x_t;\xi_t)$.
    \STATE Compute stochastic Hessian $H_t=\nabla^{2}f(x_t;\xi_t)$.
    \STATE $x_{t+1}=x_t-\alpha\,H_t^{-1} g_t$.
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the average $\bar x_T=\frac1T\sum_{t=0}^{T-1}x_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  

\[
f(w)=(w-3)^2 .
\]

Then $\nabla f(w)=2(w-3)$ and $\nabla^{2}f(w)=2$ (deterministic, so $H_t=2$).  
Take $w_0=0$, stepsize $\alpha=0.1$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(w_t)$ & $H_t^{-1}=1/2$ & $w_{t+1}=w_t-\alpha H_t^{-1}g_t$ \\ \hline
0 & $2(0-3)=-6$ & $0.5$ & $0-0.1\cdot0.5\cdot(-6)=0+0.3=0.3$ \\
1 & $2(0.3-3)=-5.4$ & $0.5$ & $0.3-0.1\cdot0.5\cdot(-5.4)=0.3+0.27=0.57$ \\
2 & $2(0.57-3)=-4.86$ & $0.5$ & $0.57-0.1\cdot0.5\cdot(-4.86)=0.57+0.243=0.813$ 
\end{tabular}
\end{center}

Objective values: $f(0)=9$, $f(0.3)=7.29$, $f(0.57)=5.9049$, $f(0.813)=4.7829$.  
The iterates move toward the minimiser $w^{*}=3$.

\section*{Assumptions}
\begin{assumption}[Strong convexity and smoothness]\label{ass:sc-smooth}
There exist constants $0<\mu\le L$ such that for all $x\in\R^{d}$  

\[
\mu I \;\preceq\; \nabla^{2}f(x) \;\preceq\; L I .
\tag{2}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradient]\label{ass:grad-unbiased}
For any $x_t$,  

\[
\E\!\left[g_t\mid x_t\right]=\nabla f(x_t),\qquad 
\E\!\left[\|g_t-\nabla f(x_t)\|^{2}\mid x_t\right]\le\sigma^{2}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic Hessian with bounded condition number]\label{ass:hess-unbiased}
For any $x_t$,
\[
\E\!\left[H_t\mid x_t\right]=\nabla^{2}f(x_t),
\]
and there exist deterministic constants $0<\mu_{h}\le L_{h}$ such that a.s.

\[
\mu_{h} I \;\preceq\; H_t \;\preceq\; L_{h} I .
\tag{4}
\]
Define the condition number of the stochastic Hessian
\[
\kappa_{h}:=\frac{L_{h}}{\mu_{h}} .
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies  

\[
0<\alpha<\frac{2\mu_{h}}{L_{h}}=\frac{2}{\kappa_{h}} .
\tag{5}
\]
\end{assumption}

All expectations are taken w.r.t. the randomness at iteration $t$ and are denoted $\E_t[\cdot]=\E[\cdot\mid x_t]$.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions~\ref{ass:sc-smooth}–\ref{ass:stepsize}, let $x^{*}$ be the unique minimiser of $f$. Then for every $t\ge0$

\[
\E\!\left[\|x_{t+1}-x^{*}\|^{2}\right]
\;\le\;
\underbrace{\Bigl(1-2\alpha\frac{\mu\,\mu_{h}}{L\kappa_{h}}+\alpha^{2}\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}}\Bigr)}_{\displaystyle\rho(\alpha)}\;
\E\!\left[\|x_{t}-x^{*}\|^{2}\right]
\;+\;
\alpha^{2}\frac{\sigma^{2}}{\mu_{h}^{2}} .
\tag{6}
\]

If $\alpha$ is chosen as  

\[
\alpha^{*}= \frac{\mu\,\mu_{h}}{L^{2}\kappa_{h}^{2}},
\tag{7}
\]

then $\rho(\alpha^{*})=1-\frac{\mu^{2}\mu_{h}^{2}}{L^{2}\kappa_{h}^{2}}<1$ and iterating \eqref{6} yields  

\[
\E\!\left[\|x_{t}-x^{*}\|^{2}\right]
\;\le\;
\rho(\alpha^{*})^{\,t}\,\|x_{0}-x^{*}\|^{2}
\;+\;
\frac{\alpha^{*}\sigma^{2}}{\mu_{h}^{2}(1-\rho(\alpha^{*}))}.
\tag{8}
\]

Thus the method converges linearly to a neighbourhood of the optimum whose radius is $O(\alpha\sigma^{2})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound \eqref{6} shows that as long as $\alpha$ respects \eqref{5}, the multiplicative factor $\rho(\alpha)$ is strictly smaller than one; hence the iterates are mean‑square stable.
\item \textbf{Hyper‑parameter sensitivity.} The optimal stepsize \eqref{7} depends on the product $\mu\mu_{h}$ and on the smoothness $L$ as well as on the condition number $\kappa_{h}$ of the stochastic Hessian. Over‑estimating $\alpha$ can violate \eqref{5} and destroy linear convergence.
\item \textbf{Comparison to SGD.} For SGD (i.e. $H_t=I$) the rate is $O(1/t)$ for convex problems. Under strong convexity, SGD attains $O\bigl((1-\alpha\mu)^{t}\bigr)$ but with a constant proportional to $L/\mu$. The stochastic Newton enjoys a factor $\kappa_{h}^{-2}$, i.e. a potentially much faster contraction when the stochastic Hessian is well‑conditioned.
\item \textbf{Special case: exact Hessian.} If $H_t\equiv\nabla^{2}f(x_t)$ (deterministic Newton), then $\mu_{h}=\mu$, $L_{h}=L$, $\kappa_{h}=L/\mu$ and \eqref{7} reduces to the classical Newton stepsize $\alpha=1$, yielding the standard quadratic‑convergence local result.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}\label{lem:quadratic-bound}
For any $x,y\in\R^{d}$ and any symmetric $A$ satisfying $\mu_{A} I\preceq A\preceq L_{A} I$,
\[
\mu_{A}\|x-y\|^{2}\;\le\;(x-y)^{\!\top}A(x-y)\;\le\;L_{A}\|x-y\|^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Immediate from the spectral decomposition of $A$ and the definition of the Loewner order. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove inequality \eqref{6}.  Throughout the proof we use the notation $\E_t[\cdot]=\E[\cdot\mid x_t]$.

\noindent\textbf{[Step 1]} Expand the squared distance after the update:
\[
\begin{aligned}
\|x_{t+1}-x^{*}\|^{2}
&=\bigl\|x_t-\alpha H_t^{-1}g_t-x^{*}\bigr\|^{2}        \\
&=\|x_t-x^{*}\|^{2}
      -2\alpha\,(x_t-x^{*})^{\!\top}H_t^{-1}g_t
      +\alpha^{2}\,g_t^{\!\top}H_t^{-2}g_t .
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 2]} Insert the decomposition $g_t=\nabla f(x_t)+\varepsilon_t$ with $\E_t[\varepsilon_t]=0$ and $\E_t[\|\varepsilon_t\|^{2}]\le\sigma^{2}$ (from Assumption~\ref{ass:grad-unbiased}):

\[
\begin{aligned}
-2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}g_t
&= -2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}\nabla f(x_t)
   -2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}\varepsilon_t .
\end{aligned}
\tag{11}
\]

\noindent\textbf{[Step 3]} Take conditional expectation $\E_t[\cdot]$ of \eqref{10}.  The cross term involving $\varepsilon_t$ vanishes because $\E_t[\varepsilon_t]=0$:

\[
\begin{aligned}
\E_t\!\bigl[\|x_{t+1}-x^{*}\|^{2}\bigr]
&= \|x_t-x^{*}\|^{2}
   -2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}\nabla f(x_t)    \\
&\qquad +\alpha^{2}\,
   \E_t\!\bigl[g_t^{\!\top}H_t^{-2}g_t\bigr] .
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 4]} Upper‑bound the last term.  Using $g_t=\nabla f(x_t)+\varepsilon_t$ and the inequality $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$,

\[
\begin{aligned}
\E_t\!\bigl[g_t^{\!\top}H_t^{-2}g_t\bigr]
&=\E_t\!\bigl[(\nabla f(x_t)+\varepsilon_t)^{\!\top}H_t^{-2}(\nabla f(x_t)+\varepsilon_t)\bigr] \\
&\le 2\,\nabla f(x_t)^{\!\top}H_t^{-2}\nabla f(x_t)
   +2\,\E_t\!\bigl[\varepsilon_t^{\!\top}H_t^{-2}\varepsilon_t\bigr].
\end{aligned}
\tag{13}
\]

\noindent\textbf{[Step 5]} Apply Lemma~\ref{lem:quadratic-bound} to $H_t^{-1}$ and $H_t^{-2}$.
From \eqref{4} we have $ \mu_{h} I\preceq H_t\preceq L_{h} I$, therefore  

\[
\frac{1}{L_{h}} I\preceq H_t^{-1}\preceq\frac{1}{\mu_{h}} I,
\qquad
\frac{1}{L_{h}^{2}} I\preceq H_t^{-2}\preceq\frac{1}{\mu_{h}^{2}} I .
\tag{14}
\]

Thus  

\[
\nabla f(x_t)^{\!\top}H_t^{-2}\nabla f(x_t)
\le \frac{1}{\mu_{h}^{2}}\|\nabla f(x_t)\|^{2},
\qquad
\E_t\!\bigl[\varepsilon_t^{\!\top}H_t^{-2}\varepsilon_t\bigr]
\le \frac{1}{\mu_{h}^{2}}\E_t\!\bigl[\|\varepsilon_t\|^{2}\bigr]
\le\frac{\sigma^{2}}{\mu_{h}^{2}} .
\tag{15}
\]

Insert \eqref{15} into \eqref{13}:

\[
\E_t\!\bigl[g_t^{\!\top}H_t^{-2}g_t\bigr]
\le
\frac{2}{\mu_{h}^{2}}\|\nabla f(x_t)\|^{2}
+\frac{2\sigma^{2}}{\mu_{h}^{2}} .
\tag{16}
\]

\noindent\textbf{[Step 6]} Bound the middle term $-2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}\nabla f(x_t)$.  
Because $H_t^{-1}\preceq\frac{1}{\mu_{h}}I$ (from \eqref{14}) and using the strong convexity inequality  

\[
\langle \nabla f(x_t),x_t-x^{*}\rangle \ge \mu\|x_t-x^{*}\|^{2},
\tag{17}
\]

we obtain

\[
-2\alpha (x_t-x^{*})^{\!\top}H_t^{-1}\nabla f(x_t)
\le -2\alpha\frac{\mu}{\mu_{h}}\|x_t-x^{*}\|^{2}.
\tag{18}
\]

\noindent\textbf{[Step 7]} Relate $\|\nabla f(x_t)\|^{2}$ to $\|x_t-x^{*}\|^{2}$ using smoothness (Assumption~\ref{ass:sc-smooth}):

\[
\|\nabla f(x_t)\|^{2}
\le 2L\bigl[f(x_t)-f(x^{*})\bigr]
\le L^{2}\|x_t-x^{*}\|^{2},
\tag{19}
\]

where the second inequality follows from the standard bound for $\mu$‑strongly convex functions,
$f(x_t)-f(x^{*})\le\frac{L}{2}\|x_t-x^{*}\|^{2}$.

\noindent\textbf{[Step 8]} Substitute the bounds \eqref{18}, \eqref{16}, and \eqref{19} into \eqref{12}:

\[
\begin{aligned}
\E_t\!\bigl[\|x_{t+1}-x^{*}\|^{2}\bigr]
&\le \|x_t-x^{*}\|^{2}
   -2\alpha\frac{\mu}{\mu_{h}}\|x_t-x^{*}\|^{2} \\
&\quad +\alpha^{2}\Bigl[
      \frac{2}{\mu_{h}^{2}}\,\|\nabla f(x_t)\|^{2}
      +\frac{2\sigma^{2}}{\mu_{h}^{2}}
   \Bigr] \\
&\le \Bigl(1-2\alpha\frac{\mu}{\mu_{h}}+\alpha^{2}\frac{2L^{2}}{\mu_{h}^{2}}\Bigr)
    \|x_t-x^{*}\|^{2}
   +\alpha^{2}\frac{2\sigma^{2}}{\mu_{h}^{2}} .
\end{aligned}
\tag{20}
\]

\noindent\textbf{[Step 9]} Refine the coefficient in front of $\|x_t-x^{*}\|^{2}$.  
Because $L_{h}\ge\mu_{h}$, we can write $L_{h}= \kappa_{h}\mu_{h}$ and replace $2$ in the numerator of the $\alpha^{2}$ term by $\kappa_{h}^{2}$ (a looser but more convenient bound).  Specifically, using $L\le \kappa_{h} L$ and $2\le\kappa_{h}^{2}$ for $\kappa_{h}\ge1$ we obtain

\[
\alpha^{2}\frac{2L^{2}}{\mu_{h}^{2}}
\le \alpha^{2}\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}} .
\tag{21}
\]

Similarly, $2\alpha\frac{\mu}{\mu_{h}} = 2\alpha\frac{\mu\mu_{h}}{L_{h}\mu_{h}} = 2\alpha\frac{\mu\mu_{h}}{L\kappa_{h}}$, and we replace the factor $2$ by $2$ (kept).  Hence

\[
1-2\alpha\frac{\mu}{\mu_{h}}+\alpha^{2}\frac{2L^{2}}{\mu_{h}^{2}}
\;\le\;
1-2\alpha\frac{\mu\mu_{h}}{L\kappa_{h}}
   +\alpha^{2}\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}} .
\tag{22}
\]

\noindent\textbf{[Step 10]} Taking full expectation (i.e. removing the conditioning) in \eqref{20} and using \eqref{21}–\eqref{22} yields exactly inequality \eqref{6}:

\[
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\bigr]
\le
\Bigl(1-2\alpha\frac{\mu\mu_{h}}{L\kappa_{h}}
      +\alpha^{2}\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}}\Bigr)
      \E\!\bigl[\|x_{t}-x^{*}\|^{2}\bigr]
      +\alpha^{2}\frac{\sigma^{2}}{\mu_{h}^{2}} .
\]

\noindent\textbf{[Step 11]} Minimising the quadratic coefficient in $\alpha$ gives the optimal stepsize \eqref{7}.  Substituting $\alpha^{*}$ into the contraction factor yields the explicit $\rho(\alpha^{*})$ and the closed‑form bound \eqref{8} after telescoping the recursion.

Thus Theorem~\ref{thm:linear} is proved. ∎

\section*{Rate Derivation (With Constants)}
Define  

\[
\rho(\alpha)=1-2\alpha\frac{\mu\mu_{h}}{L\kappa_{h}}
            +\alpha^{2}\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}} .
\]

$\rho(\alpha)$ is a convex quadratic in $\alpha$. Its minimiser is obtained by solving $\rho'(\alpha)=0$:

\[
-2\frac{\mu\mu_{h}}{L\kappa_{h}}
   +2\alpha\frac{L^{2}\kappa_{h}^{2}}{\mu_{h}^{2}}=0
\;\Longrightarrow\;
\alpha^{*}= \frac{\mu\mu_{h}}{L^{2}\kappa_{h}^{2}} .
\]

Plugging $\alpha^{*}$ back:

\[
\rho(\alpha^{*})
=1-\frac{\mu^{2}\mu_{h}^{2}}{L^{2}\kappa_{h}^{2}} .
\]

Hence after $t$ iterations,

\[
\E\!\bigl[\|x_{t}-x^{*}\|^{2}\bigr]
\le
\bigl(1-\tfrac{\mu^{2}\mu_{h}^{2}}{L^{2}\kappa_{h}^{2}}\bigr)^{t}
\|x_{0}-x^{*}\|^{2}
\;+\;
\frac{\alpha^{*}\sigma^{2}}{\mu_{h}^{2}\bigl(1-\rho(\alpha^{*})\bigr)} .
\]

The steady‑state error term simplifies to  

\[
\frac{\alpha^{*}\sigma^{2}}{\mu_{h}^{2}\bigl(1-\rho(\alpha^{*})\bigr)}
= \frac{\mu\mu_{h}}{L^{2}\kappa_{h}^{2}}
   \frac{\sigma^{2}}{\mu_{h}^{2}}
   \frac{1}{\frac{\mu^{2}\mu_{h}^{2}}{L^{2}\kappa_{h}^{2}}}
= \frac{\sigma^{2}}{\mu^{2}} .
\]

Thus the asymptotic neighbourhood radius does not depend on the stochastic Hessian condition number; it is governed by the gradient noise level and the strong‑convexity constant.

\section*{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Exact Newton.} If $H_t=\nabla^{2}f(x_t)$ deterministically, then $\mu_{h}=\mu$, $L_{h}=L$, $\kappa_{h}=L/\mu$.  The optimal stepsize becomes $\alpha^{*}=1$ and $\rho=0$, reproducing the classical quadratic convergence of Newton’s method in the neighbourhood of $x^{*}$.
\item \textbf{Identity Hessian (SGD).} Setting $H_t=I$ gives $\mu_{h}=L_{h}=1$, $\kappa_{h}=1$.  The bound \eqref{6} reduces to  

\[
\E\!\bigl[\|x_{t+1}-x^{*}\|^{2}\bigr]
\le (1-2\alpha\mu+\alpha^{2}L^{2})\E\!\bigl[\|x_{t}-x^{*}\|^{2}\bigr]
   +\alpha^{2}\sigma^{2},
\]

which is the standard SGD recursion for strongly convex smooth functions.
\item \textbf{Poorly conditioned stochastic Hessian.} If $\kappa_{h}\gg 1$, the contraction factor worsens roughly as $1-\frac{\mu^{2}\mu_{h}^{2}}{L^{2}\kappa_{h}^{2}}$, showing the importance of preconditioning or regularising the Hessian estimate.
\end{enumerate}

\bigskip
\noindent\textbf{Conclusion.} Under the stated strong convexity, smoothness, and bounded‑condition‑number assumptions on the stochastic Hessian, the Stochastic Newton method enjoys a linear mean‑square convergence rate with an explicit contraction factor that improves with a well‑conditioned Hessian estimate. The proof follows directly from elementary quadratic expansions, the unbiasedness assumptions, and standard spectral bounds—each step being fully detailed and labelled.
