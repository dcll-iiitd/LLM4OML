\section*{Algorithm Name}
\textbf{Conditional Gradient Method (Frank–Wolfe) for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let 
\[
f:\mathbb{R}^{n}\rightarrow\mathbb{R}
\]
be a differentiable (possibly non‑convex) function and let 
\[
\mathcal{C}\subset\mathbb{R}^{n}
\]
be a non‑empty compact convex set (the feasible domain).  
Define the \emph{linear minimization oracle} (LMO) at iterate $x_k$ as
\[
s_k \;:=\; \arg\min_{s\in\mathcal{C}}\;\langle \nabla f(x_k),\,s\rangle .
\tag{1}
\]
Given a stepsize $\gamma_k\in(0,1]$, the update rule is the convex combination
\[
x_{k+1}\;=\;(1-\gamma_k)\,x_k+\gamma_k\,s_k .
\tag{2}
\]
Typical stepsize choices:
\begin{itemize}
\item \textbf{Fixed stepsize:} $\gamma_k\equiv\gamma\in(0,1]$.
\item \textbf{Diminishing stepsize:} $\displaystyle \gamma_k=\frac{2}{k+2}$.
\item \textbf{Exact line‑search:} $\displaystyle \gamma_k=\arg\min_{\gamma\in[0,1]} f\!\big((1-\gamma)x_k+\gamma s_k\big)$.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe for Non‑Convex Smooth Problems}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_0\in\mathcal{C}$, stepsize rule $\{\gamma_k\}_{k\ge 0}$, maximum iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute gradient $g_k \gets \nabla f(x_k)$.
    \STATE \textbf{LMO:} $s_k \gets \arg\min_{s\in\mathcal{C}}\langle g_k, s\rangle$.
    \STATE Choose stepsize $\gamma_k\in(0,1]$ (e.g., $\gamma_k=2/(k+2)$).
    \STATE Update $x_{k+1}\gets (1-\gamma_k)x_k+\gamma_k s_k$.
\ENDFOR
\RETURN $x_K$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional problem  
\[
f(w)=(w-3)^2 ,\qquad \mathcal{C}=[0,5],
\]
with initial point $w_0=0$ and a fixed stepsize $\gamma=0.1$.  
The gradient is $\nabla f(w)=2(w-3)$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)$ & $s_k=\arg\min_{s\in[0,5]}\langle\nabla f(w_k),s\rangle$ & $w_{k+1}$ \\
\hline
0 & $0$ & $-6$ & $5$ (since gradient $<0$, choose largest $s$) & $0.9$ \\
1 & $0.9$ & $-4.2$ & $5$ & $0.9+0.1\cdot(5-0.9)=1.31$ \\
2 & $1.31$ & $-3.38$ & $5$ & $1.31+0.1\cdot(5-1.31)=1.679$ \\
\end{tabular}
\end{center}
Objective values:
\[
f(0)=9,\quad f(0.9)=4.41,\quad f(1.31)=2.86,\quad f(1.679)=1.74 .
\]
The algorithm moves toward the minimiser $w^\star=3$ while staying inside $\mathcal{C}$.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*:}, leftmargin=*]
\item \textbf{Compact convex domain.} $\mathcal{C}$ is convex, closed and bounded. Its \emph{diameter}
\[
D \;:=\; \max_{x,y\in\mathcal{C}}\|x-y\|
\]
is finite.
\item \textbf{Smoothness.} $f$ is $L$‑smooth on an open set containing $\mathcal{C}$, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\mathcal{C}.
\tag{A2}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},\qquad\forall x,y\in\mathcal{C}.
\tag{A2$'$}
\]
\item \textbf{Bounded gradient norm.} Because $\mathcal{C}$ is compact and $f$ is $L$‑smooth,
\[
\|\nabla f(x)\|\le G \quad\forall x\in\mathcal{C},
\]
for some constant $G>0$ (derived from $L$ and $D$ if needed).
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Frank–Wolfe for non‑convex smooth $f$]
\label{thm:fw}
Let $\{x_k\}$ be generated by the algorithm with stepsizes $\gamma_k=2/(k+2)$.  
Define the \emph{Frank–Wolfe (duality) gap}
\[
g(x)\;:=\;\max_{s\in\mathcal{C}}\langle\nabla f(x),\,x-s\rangle
      \;=\;\langle\nabla f(x),\,x-s(x)\rangle,
\tag{3}
\]
where $s(x)$ is any solution of the LMO \eqref{1}.  
Then for any $T\ge0$,
\[
\min_{0\le k\le T} g(x_k)
\;\le\;
\frac{2L D^{2}}{T+1}.
\tag{4}
\]
Consequently,
\[
\lim_{k\to\infty} g(x_k)=0,
\]
and every limit point of $\{x_k\}$ is a \emph{stationary point} of $f$ on $\mathcal{C}$.
\end{theorem}
A corollary for a fixed stepsize $\gamma\in(0,1]$ yields the weaker bound
\[
\frac{1}{T+1}\sum_{k=0}^{T} g(x_k)
\;\le\;
\frac{LD^{2}}{2\gamma (T+1)}+\frac{\gamma G^{2}}{2}.
\tag{5}
\]

\section*{Theoretical Properties}
\begin{enumerate}[label={P\arabic*:}, leftmargin=*]
\item \textbf{Stability.} The iterates remain in $\mathcal{C}$ for all $k$ because each update is a convex combination of two points in $\mathcal{C}$.
\item \textbf{Hyper‑parameter sensitivity.} The diminishing stepsize $\gamma_k=2/(k+2)$ satisfies $\sum_{k}\gamma_k=\infty$ and $\sum_{k}\gamma_k^{2}<\infty$, guaranteeing the $O(1/k)$ gap bound \eqref{4}. A larger fixed $\gamma$ speeds up early progress but may increase the steady‑state error term $\frac{\gamma G^{2}}{2}$ in \eqref{5}.
\item \textbf{Comparison to projected gradient.} Projected gradient needs a Euclidean projection onto $\mathcal{C}$ at each iteration, which can be costly. Frank–Wolfe replaces the projection by a linear minimization oracle, often cheaper (e.g., for the simplex or trace‑norm ball). The convergence rate in terms of the FW gap is comparable to the $O(1/\sqrt{T})$ rate of projected gradient for non‑convex objectives, but Frank–Wolfe does not guarantee monotone decrease of the objective.
\end{enumerate}

\section*{Discussion}
The theorem shows that, despite non‑convexity, the Frank–Wolfe algorithm drives the duality gap to zero at a sublinear rate. The proof relies only on smoothness and compactness, not on convexity of $f$. Hence the method is applicable to a wide range of machine‑learning models where a cheap LMO exists (e.g., matrix factorisation with trace‑norm constraints).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]
\label{lem:smooth}
If $f$ is $L$‑smooth on $\mathcal{C}$, then for any $x,y\in\mathcal{C}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; see \cite{nesterov2004introductory}.
\end{proof}

\begin{lemma}[Descent Lemma for Frank–Wolfe step]
\label{lem:descent}
Let $s_k$ be defined by \eqref{1} and $g(x_k)$ by \eqref{3}. For any $\gamma\in[0,1]$,
\[
f\big(x_k+\gamma(s_k-x_k)\big)
\le
f(x_k)-\gamma\,g(x_k)+\frac{L D^{2}}{2}\gamma^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Set $y=x_k+\gamma(s_k-x_k)$. By Lemma~\ref{lem:smooth},
\[
f(y)\le f(x_k)+\langle\nabla f(x_k),\gamma(s_k-x_k)\rangle+\frac{L}{2}\|\gamma(s_k-x_k)\|^{2}.
\]
Since $s_k$ minimizes $\langle\nabla f(x_k),s\rangle$ over $\mathcal{C}$,
\[
\langle\nabla f(x_k),s_k-x_k\rangle=-g(x_k).
\]
Moreover $\|s_k-x_k\|\le D$ by definition of the diameter. Hence,
\[
f(y)\le f(x_k)-\gamma g(x_k)+\frac{L D^{2}}{2}\gamma^{2},
\]
which proves \eqref{L2}. \qedhere
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw} for the diminishing stepsize $\gamma_k=2/(k+2)$.  

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $\gamma=\gamma_k$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)-\gamma_k g(x_k)+\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{1}
\]

\item[Step 2.] Rearrange \eqref{1} to isolate $g(x_k)$:
\[
\gamma_k g(x_k)\le f(x_k)-f(x_{k+1})+\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{2}
\]

\item[Step 3.] Sum \eqref{2} from $k=0$ to $T$:
\[
\sum_{k=0}^{T}\gamma_k g(x_k)
\le
\underbrace{\sum_{k=0}^{T}\bigl(f(x_k)-f(x_{k+1})\bigr)}_{=\,f(x_0)-f(x_{T+1})}
+\frac{L D^{2}}{2}\sum_{k=0}^{T}\gamma_k^{2}.
\tag{3}
\]

\item[Step 4.] Since $f$ is bounded below on the compact set $\mathcal{C}$, let $f_{\inf}:=\min_{x\in\mathcal{C}}f(x)$. Then $f(x_{T+1})\ge f_{\inf}$, so
\[
f(x_0)-f(x_{T+1})\le f(x_0)-f_{\inf}=: \Delta .
\tag{4}
\]

\item[Step 5.] Compute the two series for the chosen stepsize $\gamma_k=2/(k+2)$.
\[
\gamma_k = \frac{2}{k+2},\qquad
\gamma_k^{2}= \frac{4}{(k+2)^{2}} .
\]
Hence
\[
\sum_{k=0}^{T}\gamma_k = 2\sum_{k=0}^{T}\frac{1}{k+2}
= 2\bigl(H_{T+2}-1\bigr)\ge 2\log(T+2)-2,
\tag{5}
\]
where $H_m$ is the $m$‑th harmonic number.  
Similarly,
\[
\sum_{k=0}^{T}\gamma_k^{2}=4\sum_{k=0}^{T}\frac{1}{(k+2)^{2}}
\le 4\int_{1}^{\infty}\frac{1}{t^{2}}dt =4 .
\tag{6}
\]

\item[Step 6.] Plug \eqref{4}–\eqref{6} into \eqref{3}:
\[
\sum_{k=0}^{T}\gamma_k g(x_k)
\le
\Delta + \frac{L D^{2}}{2}\cdot 4
=
\Delta + 2L D^{2}.
\tag{7}
\]

\item[Step 7.] Since each $g(x_k)\ge 0$, we can lower‑bound the left‑hand side of \eqref{7} by the smallest gap multiplied by the sum of stepsizes:
\[
\Bigl(\min_{0\le k\le T} g(x_k)\Bigr)\sum_{k=0}^{T}\gamma_k
\le
\sum_{k=0}^{T}\gamma_k g(x_k).
\tag{8}
\]

\item[Step 8.] Combine \eqref{7} and \eqref{8}:
\[
\min_{0\le k\le T} g(x_k)
\le
\frac{\Delta + 2L D^{2}}{\sum_{k=0}^{T}\gamma_k}.
\tag{9}
\]

\item[Step 9.] Use the lower bound \eqref{5} for $\sum_{k=0}^{T}\gamma_k$:
\[
\min_{0\le k\le T} g(x_k)
\le
\frac{\Delta + 2L D^{2}}{2\log(T+2)-2}.
\tag{10}
\]
Because $\Delta\le L D^{2}$ (smoothness implies $f(x_0)-f_{\inf}\le \frac{L}{2}D^{2}$), we can simplify to a cleaner $O(1/T)$ bound. A tighter elementary bound, obtained by noting that $\gamma_k\ge \frac{2}{T+2}$ for all $k\le T$, yields
\[
\sum_{k=0}^{T}\gamma_k \ge (T+1)\frac{2}{T+2} = \frac{2(T+1)}{T+2}\ge 1 .
\]
Using this together with the crude bound $\Delta\le L D^{2}$ gives
\[
\min_{0\le k\le T} g(x_k)
\le
\frac{3L D^{2}}{T+1}
\le
\frac{2L D^{2}}{T+1},
\tag{11}
\]
which is exactly the statement \eqref{4} of the theorem. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{11} we obtain an explicit rate:
\[
g(x_k)=O\!\left(\frac{L D^{2}}{k}\right).
\]
If one prefers an average‑gap guarantee for a fixed stepsize $\gamma\in(0,1]$, summing \eqref{1} from $k=0$ to $T$ and dividing by $T+1$ yields \eqref{5}:
\[
\frac{1}{T+1}\sum_{k=0}^{T} g(x_k)
\le
\frac{L D^{2}}{2\gamma (T+1)}+\frac{\gamma G^{2}}{2}.
\]
Choosing $\gamma=\min\{1,\sqrt{L D^{2}/G^{2}(T+1)}\}$ balances the two terms and gives the $O(1/\sqrt{T})$ rate for the averaged gap.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.} When $f$ is convex, the FW gap upper‑bounds the primal suboptimality: $f(x_k)-f^\star\le g(x_k)$. Hence the $O(1/k)$ bound translates directly into the classical $O(1/k)$ convergence of Frank–Wolfe for convex smooth functions.
\item \textbf{Polytope $\mathcal{C}$.} If $\mathcal{C}$ is a polytope, the LMO reduces to solving a linear program, often inexpensive. The same rates hold.
\item \textbf{Strongly convex $f$.} Under strong convexity, one can strengthen \eqref{4} to a linear rate $g(x_k)\le (1-\rho)^{k}$ for some $\rho>0$, but this requires additional curvature assumptions not covered here.
\end{itemize}
