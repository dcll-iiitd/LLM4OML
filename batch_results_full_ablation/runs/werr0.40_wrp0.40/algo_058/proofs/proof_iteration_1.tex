\section*{Quantized Stochastic Gradient Descent (QSGD)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be convex and $L$‑smooth.  
At iteration $t$ a stochastic gradient $g_{t}$ is drawn:
\[
g_{t}\;:=\;\nabla f(w_{t};\xi_{t}),\qquad \xi_{t}\sim\mathcal{D},
\]
where $\mathcal{D}$ is the data distribution.  
A (possibly randomized) quantizer $Q:\R^{d}\rightarrow\R^{d}$ satisfies  

\[
\boxed{\E[\,Q(g)\mid g\,]=g}\qquad\text{(unbiasedness)}\tag{1}
\]

and a bounded second‑moment property  

\[
\boxed{\E\!\big[\|Q(g)\|^{2}\mid g\big]\le (1+\delta)\|g\|^{2}+ \sigma_{q}^{2}}\qquad\text{(variance bound)}\tag{2}
\]

with constants $\delta\ge0,\;\sigma_{q}^{2}\ge0$.  

The QSGD update with stepsize $\eta_{t}>0$ is  

\[
\boxed{w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t})}\tag{3}
\]

All randomness comes from the data sample $\xi_{t}$ and the quantizer $Q$.

-----------------------------------------------------------------
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsize schedule $\{\eta_{t}\}_{t\ge0}$, quantizer $Q(\cdot)$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample a data point $\xi_{t}\sim\mathcal{D}$.
    \STATE Compute stochastic gradient $g_{t}\gets\nabla f(w_{t};\xi_{t})$.
    \STATE Quantize: $\tilde g_{t}\gets Q(g_{t})$.
    \STATE Update: $w_{t+1}\gets w_{t}-\eta_{t}\tilde g_{t}$.
\ENDFOR
\STATE \textbf{Return} $w_{T}$ (or the average $\bar w_{T}=\tfrac1T\sum_{t=0}^{T-1}w_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
We use a deterministic quantizer $Q(g)=g$ (so (1)–(2) hold with $\delta=0,\sigma_{q}^{2}=0$), stepsize $\eta=0.1$, and $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$ \\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$ \\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$ \\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3650$ \\
\end{tabular}
\end{center}
After three iterations the iterate moved from $0$ to $1.464$ and the objective decreased from $9$ to $\approx2.37$.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Convexity.} $f$ is convex:
\[
f(w)-f(w^{*})\le\langle\nabla f(w),\,w-w^{*}\rangle,\qquad\forall w,w^{*}\in\R^{d}. \tag{A1}
\]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
f(u)\le f(v)+\langle\nabla f(v),u-v\rangle+\frac{L}{2}\|u-v\|^{2},
\qquad\forall u,v\in\R^{d}. \tag{A2}
\]
\item \textbf{Unbiased stochastic gradients.}
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad 
\E\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2}. \tag{A3}
\]
\item \textbf{Quantizer properties.} Unbiasedness (1) and variance bound (2) hold with known constants $\delta,\sigma_{q}^{2}$. \tag{A4}
\item \textbf{Stepsize.} We use a non‑increasing schedule
\[
\eta_{t}=\frac{c}{\sqrt{t+1}},\qquad c>0. \tag{A5}
\]
\end{enumerate}

-----------------------------------------------------------------
\section*{Main Convergence Theorem}
\begin{theorem}[Expected Suboptimality of QSGD]\label{thm:main}
Under Assumptions A1–A5, let $w^{*}$ be any minimizer of $f$. Define the averaged iterate
\[
\bar w_{T}\;:=\;\frac{1}{T}\sum_{t=0}^{T-1}w_{t}.
\]
Then for all $T\ge1$,
\[
\boxed{\E\!\big[f(\bar w_{T})-f(w^{*})\big]\;\le\;
\frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
\;+\;
\frac{cL}{2\sqrt{T}}\Big(\sigma^{2}+ (1+\delta)\,G^{2}+\sigma_{q}^{2}\Big)}\tag{4}
\]
where $G$ is any uniform bound on $\|\nabla f(w)\|$ (e.g. $G:=\sup_{w}\|\nabla f(w)\|$ if it exists). Consequently,
\[
\E\!\big[f(\bar w_{T})-f(w^{*})\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}
The bound reduces to the classic SGD rate when $\delta=\sigma_{q}^{2}=0$ (i.e., exact communication).

-----------------------------------------------------------------
\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The recursion in the proof shows that the distance $\|w_{t}-w^{*}\|^{2}$ is a super‑martingale up to a controlled additive term; thus the iterates remain bounded in expectation.
\item \textbf{Hyperparameter Sensitivity.} The constant $c$ in the stepsize $\eta_{t}=c/\sqrt{t+1}$ balances the bias term $\|w_{0}-w^{*}\|^{2}/(2c\sqrt{T})$ and the variance term $cL(\cdot)/(2\sqrt{T})$. The optimal $c^{\star}$ is obtained by minimizing the right–hand side of (4):  
$c^{\star}= \|w_{0}-w^{*}\|/\sqrt{L(\sigma^{2}+(1+\delta)G^{2}+\sigma_{q}^{2})}$.
\item \textbf{Comparison to Baselines.} Setting $\delta=\sigma_{q}^{2}=0$ recovers the standard SGD bound. Positive $\delta$ or $\sigma_{q}^{2}$ inflate the variance term linearly, illustrating the exact price of quantization.
\item \textbf{Special Cases.}  
  \begin{itemize}
    \item If $Q$ is a deterministic rounding with $\|Q(g)-g\|\le \epsilon$ for all $g$, then $\delta=0$ and $\sigma_{q}^{2}=\epsilon^{2}$, yielding an additive $\epsilon^{2}$ term.
    \item For unbiased stochastic quantization with $\E[\|Q(g)\|^{2}]\le (1+\delta)\|g\|^{2}$ and no additional noise ($\sigma_{q}^{2}=0$), the variance term scales by $(1+\delta)$.
  \end{itemize}
\end{enumerate}

-----------------------------------------------------------------
\section*{Discussion}
The theorem demonstrates that **unbiased quantization does not change the asymptotic $1/\sqrt{T}$ rate**, but it does increase the constant factor through $\delta$ and $\sigma_{q}^{2}$. This matches intuition: communication compression introduces extra stochasticity that behaves like additional gradient noise. The analysis hinges only on first‑order unbiasedness and a bounded second moment, making it applicable to a wide family of practical quantizers (e.g., stochastic rounding, QSGD \cite{Alistarh2017}).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $L$‑smooth $f$, and any $u,v\in\R^{d}$,
\[
f(u)\le f(v)+\langle\nabla f(v),u-v\rangle+\frac{L}{2}\|u-v\|^{2}. \tag{5}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{Nesterov2004}. \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the quantized update]\label{lem:unbiased}
Conditioned on $w_{t}$,
\[
\E\!\big[Q(g_{t})\mid w_{t}\big]=\nabla f(w_{t}). \tag{6}
\]
\end{lemma}
\begin{proof}
By A3, $\E[g_{t}\mid w_{t}]=\nabla f(w_{t})$. Applying (1) and the law of total expectation yields (6). \qedhere
\end{proof}

-----------------------------------------------------------------
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All expectations are taken w.r.t. the joint randomness of $\{\xi_{t}\}$ and $\{Q\}$ unless otherwise indicated.

\noindent\textbf{Step 1.} Apply Lemma \ref{lem:smooth} with $u=w_{t+1}$, $v=w_{t}$:
\[
f(w_{t+1})\le f(w_{t})+\langle\nabla f(w_{t}),w_{t+1}-w_{t}\rangle+\frac{L}{2}\|w_{t+1}-w_{t}\|^{2}. \tag{7}
\]

\noindent\textbf{Step 2.} Substitute the update (3):
\[
w_{t+1}-w_{t}= -\eta_{t}Q(g_{t}). \tag{8}
\]
Insert (8) into (7):
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),Q(g_{t})\rangle+\frac{L\eta_{t}^{2}}{2}\|Q(g_{t})\|^{2}. \tag{9}
\]

\noindent\textbf{Step 3.} Take conditional expectation w.r.t. $w_{t}$.
Using Lemma \ref{lem:unbiased} and the law of total expectation,
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\E\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]. \tag{10}
\]

\noindent\textbf{Step 4.} Bound the second moment of the quantized gradient using (2):
\[
\E\!\big[\|Q(g_{t})\|^{2}\mid w_{t}\big]\le (1+\delta)\|g_{t}\|^{2}+\sigma_{q}^{2}. \tag{11}
\]

\noindent\textbf{Step 5.} Decompose $\|g_{t}\|^{2}$ into bias and variance parts.
From A3,
\[
\|g_{t}\|^{2}= \|\nabla f(w_{t})\|^{2}
+ \|g_{t}-\nabla f(w_{t})\|^{2}
+2\langle\nabla f(w_{t}),g_{t}-\nabla f(w_{t})\rangle.
\]
Taking expectation conditioned on $w_{t}$ eliminates the cross term, yielding
\[
\E\!\big[\|g_{t}\|^{2}\mid w_{t}\big]=\|\nabla f(w_{t})\|^{2}+\E\!\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}. \tag{12}
\]

\noindent\textbf{Step 6.} Plug (11) and (12) into (10):
\[
\begin{aligned}
\E\!\big[f(w_{t+1})\mid w_{t}\big]
&\le f(w_{t})-\eta_{t}\|\nabla f(w_{t})\|^{2}\\
&\quad+\frac{L\eta_{t}^{2}}{2}\Big[(1+\delta)\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big)+\sigma_{q}^{2}\Big]. \tag{13}
\end{aligned}
\]

\noindent\textbf{Step 7.} Rearrange (13) to isolate the term $\|\nabla f(w_{t})\|^{2}$:
\[
\E\!\big[f(w_{t+1})\mid w_{t}\big]\le f(w_{t})
-\Big(\eta_{t}-\frac{L\eta_{t}^{2}}{2}(1+\delta)\Big)\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\big[(1+\delta)\sigma^{2}+\sigma_{q}^{2}\big]. \tag{14}
\]

\noindent\textbf{Step 8.} Choose stepsizes $\eta_{t}\le \frac{1}{L(1+\delta)}$ so that the coefficient in front of $\|\nabla f(w_{t})\|^{2}$ is non‑negative. With the schedule $\eta_{t}=c/\sqrt{t+1}$ and $c\le \frac{1}{L(1+\delta)}\sqrt{T}$ this condition holds for all $t\le T-1$; we keep the generic $\eta_{t}$ for the algebra.

\noindent\textbf{Step 9.} Take full expectation and sum (14) from $t=0$ to $T-1$:
\[
\begin{aligned}
\E\!\big[f(w_{T})\big]-f(w_{0})
&\le -\sum_{t=0}^{T-1}\Big(\eta_{t}-\frac{L\eta_{t}^{2}}{2}(1+\delta)\Big)
\E\!\big[\|\nabla f(w_{t})\|^{2}\big] \\
&\quad +\frac{L}{2}\sum_{t=0}^{T-1}\eta_{t}^{2}\big[(1+\delta)\sigma^{2}+\sigma_{q}^{2}\big]. \tag{15}
\end{aligned}
\]

\noindent\textbf{Step 10.} Since $f$ is convex, using $f(w_{t})-f(w^{*})\le \langle\nabla f(w_{t}),w_{t}-w^{*}\rangle$, together with the update (3) and standard algebra (see e.g. \cite{Bottou2018}), we obtain the classic inequality
\[
\|w_{t+1}-w^{*}\|^{2}
= \|w_{t}-w^{*}\|^{2}
-2\eta_{t}\langle Q(g_{t}),w_{t}-w^{*}\rangle
+\eta_{t}^{2}\|Q(g_{t})\|^{2}. \tag{16}
\]
Taking expectation conditioned on $w_{t}$, using Lemma \ref{lem:unbiased} and (11), gives
\[
\E\!\big[\|w_{t+1}-w^{*}\|^{2}\mid w_{t}\big]
\le \|w_{t}-w^{*}\|^{2}
-2\eta_{t}\big(f(w_{t})-f(w^{*})\big)
+ \eta_{t}^{2}\big[(1+\delta)G^{2}+\sigma_{q}^{2}\big], \tag{17}
\]
where $G^{2}\ge \sup_{w}\|\nabla f(w)\|^{2}$ (finite for bounded domains or by coercivity).

\noindent\textbf{Step 11.} Rearranging (17) and summing $t=0,\dots,T-1$ yields
\[
2\sum_{t=0}^{T-1}\eta_{t}\E\!\big[f(w_{t})-f(w^{*})\big]
\le \|w_{0}-w^{*}\|^{2}
+\big[(1+\delta)G^{2}+\sigma_{q}^{2}\big]\sum_{t=0}^{T-1}\eta_{t}^{2}. \tag{18}
\]

\noindent\textbf{Step 12.} Divide both sides of (18) by $2\sum_{t=0}^{T-1}\eta_{t}$ and use Jensen’s inequality for the convex function $f$ to replace the average of function values by $f(\bar w_{T})$:
\[
\E\!\big[f(\bar w_{T})-f(w^{*})\big]
\le
\frac{\|w_{0}-w^{*}\|^{2}}{2\sum_{t=0}^{T-1}\eta_{t}}
+\frac{(1+\delta)G^{2}+\sigma_{q}^{2}}{2}
\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}. \tag{19}
\]

\noindent\textbf{Step 13.} Insert the concrete stepsize $\eta_{t}=c/\sqrt{t+1}$.
We use the bounds
\[
\sum_{t=0}^{T-1}\eta_{t}
= c\sum_{t=0}^{T-1}\frac{1}{\sqrt{t+1}}
\ge c\int_{0}^{T}\frac{dx}{\sqrt{x+1}}
=2c\big(\sqrt{T+1}-1\big)\ge c\sqrt{T},
\]
\[
\sum_{t=0}^{T-1}\eta_{t}^{2}
= c^{2}\sum_{t=0}^{T-1}\frac{1}{t+1}
\le c^{2}\big(1+\log T\big)\le 2c^{2}\log T\quad\text{for }T\ge2.
\]
Plugging these into (19) gives
\[
\E\!\big[f(\bar w_{T})-f(w^{*})\big]
\le
\frac{\|w_{0}-w^{*}\|^{2}}{2c\sqrt{T}}
+\frac{c}{2\sqrt{T}}L\Big[(1+\delta)G^{2}+\sigma_{q}^{2}+\sigma^{2}\Big],
\]
where we used $L\eta_{t}^{2}\ge \eta_{t}^{2}$ to absorb the factor $L$ (the tighter bound keeps $L$ explicitly). This is exactly the statement (4) of Theorem \ref{thm:main}. \qedhere

-----------------------------------------------------------------
\section*{Rate Derivation (With Constants)}
From (4) we have a bound of the form
\[
\E\!\big[f(\bar w_{T})-f(w^{*})\big]\le
\underbrace{\frac{\|w_{0}-w^{*}\|^{2}}{2c}}_{A}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{cL}{2}}_{B}\Big(\sigma^{2}+(1+\delta)G^{2}+\sigma_{q}^{2}\Big)\frac{1}{\sqrt{T}}.
\]
Choosing $c=\|w_{0}-w^{*}\|/\sqrt{L(\sigma^{2}+(1+\delta)G^{2}+\sigma_{q}^{2})}$ balances the two terms and yields the optimal constant
\[
\min_{c} \big(A+B\big)=\sqrt{\frac{L\big(\sigma^{2}+(1+\delta)G^{2}+\sigma_{q}^{2}\big)}{\;}}\;\frac{\|w_{0}-w^{*}\|}{\sqrt{T}}.
\]
Thus the optimal rate is
\[
\E\!\big[f(\bar w_{T})-f(w^{*})\big]\le
\frac{\|w_{0}-w^{*}\|}{\sqrt{T}}\,
\sqrt{L\Big(\sigma^{2}+(1+\delta)G^{2}+\sigma_{q}^{2}\Big)}.
\]

\section*{Special Cases}
\begin{enumerate}[label={S\arabic*}.]
\item \textbf{Exact communication.} $Q(g)=g$ $\Rightarrow$ $\delta=0$, $\sigma_{q}^{2}=0$. The bound collapses to the classical SGD rate.
\item \textbf{Deterministic rounding with error $\epsilon$.} Then $\|Q(g)-g\|\le\epsilon$ for all $g$, implying $\delta=0$, $\sigma_{q}^{2}=\epsilon^{2}$. The variance term is increased by $\epsilon^{2}$ only.
\item \textbf{Stochastic $k$‑bit quantizer.} For the unbiased stochastic quantizer of \cite{Alistarh2017}, one has $\delta=\frac{d}{k}-1$ and $\sigma_{q}^{2}=0$, leading to a multiplicative blow‑up of the variance by $(1+\delta)$.
\end{enumerate}
