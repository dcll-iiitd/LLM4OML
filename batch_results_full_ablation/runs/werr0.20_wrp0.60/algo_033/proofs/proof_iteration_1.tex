\section*{Algorithm Name}
\textbf{Away–Step Frank–Wolfe (AFW)} for minimizing a smooth convex function over a polytope.

\section*{Mathematical Formulation}
Let 
\[
\min_{\bm{x}\in \mathcal{C}} \; f(\bm{x}),
\qquad 
\mathcal{C}= \conv(\mathcal{V})\subset\R^{d},
\]
where $f$ is convex and $L$‑smooth on $\mathcal{C}$.  
At iteration $k$ we have an \emph{active set} $\mathcal{S}_{k}\subseteq\mathcal{V}$ and a representation
\[
\bm{x}_{k}= \sum_{\bm{v}\in\mathcal{S}_{k}} \alpha_{k}^{\bm{v}} \bm{v},
\qquad 
\alpha_{k}^{\bm{v}}\ge 0,\;\sum_{\bm{v}\in\mathcal{S}_{k}}\alpha_{k}^{\bm{v}}=1 .
\]

\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*]
  \item \textbf{Linear minimization oracle (LMO)}  
  \[
  \bm{s}_{k}= \argmin_{\bm{s}\in\mathcal{C}} \langle \nabla f(\bm{x}_{k}),\bm{s}\rangle .
  \]
  \item \textbf{Away vertex}  
  \[
  \bm{v}_{k}= \argmax_{\bm{v}\in\mathcal{S}_{k}} \langle \nabla f(\bm{x}_{k}),\bm{v}\rangle .
  \]
  \item \textbf{Direction choice}  
  \[
  \begin{aligned}
  \bm{d}^{\text{FW}}_{k}&:=\bm{s}_{k}-\bm{x}_{k}, &
  \text{FW\,gap}&:= -\langle \nabla f(\bm{x}_{k}),\bm{d}^{\text{FW}}_{k}\rangle,\\
  \bm{d}^{\text{A}}_{k}&:=\bm{x}_{k}-\bm{v}_{k}, &
  \text{A\,gap}&:= -\langle \nabla f(\bm{x}_{k}),\bm{d}^{\text{A}}_{k}\rangle .
  \end{aligned}
  \]
  If $\text{FW\,gap}\ge\text{A\,gap}$ set $\bm{d}_{k}= \bm{d}^{\text{FW}}_{k}$ (a \emph{FW step}); else $\bm{d}_{k}= \bm{d}^{\text{A}}_{k}$ (an \emph{away step}).
  \item \textbf{Step–size}  
  \[
  \gamma_{\max}=
  \begin{cases}
  1 & \text{if }\bm{d}_{k}= \bm{d}^{\text{FW}}_{k},\\[4pt]
  \displaystyle\frac{\alpha_{k}^{\bm{v}_{k}}}{1-\alpha_{k}^{\bm{v}_{k}}} & \text{if }\bm{d}_{k}= \bm{d}^{\text{A}}_{k},
  \end{cases}
  \qquad
  \gamma_{k}= \min\Bigl\{\gamma_{\max},\; \frac{-\langle \nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle}{L\|\bm{d}_{k}\|^{2}}\Big\}.
  \]
  (The second term is the exact line‑search step for an $L$‑smooth function; a fixed $\eta\in(0,1]$ can be used as a simplification.)
  \item \textbf{Update}  
  \[
  \bm{x}_{k+1}= \bm{x}_{k}+ \gamma_{k}\bm{d}_{k}.
  \]
  The coefficients $\alpha_{k}^{\bm{v}}$ are updated accordingly; if an away step removes all mass from $\bm{v}_{k}$ then $\bm{v}_{k}$ is deleted from $\mathcal{S}_{k}$, otherwise its weight is decreased by $\gamma_{k}$.
  \item \textbf{Termination}  
  Stop when the \emph{dual gap}
  \[
  g_{k}:= \langle \nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{s}_{k}\rangle
  \]
  satisfies $g_{k}\le\varepsilon$ for a prescribed tolerance $\varepsilon>0$.
\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Away–Step Frank–Wolfe (AFW)}\label{alg:afw}
\begin{algorithmic}[1]
\Require Convex $L$‑smooth $f$, polytope $\mathcal{C}=\conv(\mathcal{V})$, tolerance $\varepsilon$, max. iter $T$.
\State Choose an initial vertex $\bm{v}_{0}\in\mathcal{V}$, set $\bm{x}_{0}\gets\bm{v}_{0}$, $\mathcal{S}_{0}\gets\{\bm{v}_{0}\}$, $\alpha^{\bm{v}_{0}}_{0}=1$.
\For{$k=0$ \textbf{to} $T-1$}
    \State $\bm{s}_{k}\gets \argmin_{\bm{s}\in\mathcal{C}}\langle\nabla f(\bm{x}_{k}),\bm{s}\rangle$ \Comment{LMO}
    \State $\bm{v}_{k}\gets \argmax_{\bm{v}\in\mathcal{S}_{k}}\langle\nabla f(\bm{x}_{k}),\bm{v}\rangle$
    \State $\bm{d}^{\text{FW}}_{k}\gets \bm{s}_{k}-\bm{x}_{k}$,\;
          $\bm{d}^{\text{A}}_{k}\gets \bm{x}_{k}-\bm{v}_{k}$
    \State $\text{gap}^{\text{FW}}_{k}\gets -\langle\nabla f(\bm{x}_{k}),\bm{d}^{\text{FW}}_{k}\rangle$,
          $\text{gap}^{\text{A}}_{k}\gets -\langle\nabla f(\bm{x}_{k}),\bm{d}^{\text{A}}_{k}\rangle$
    \If{$\text{gap}^{\text{FW}}_{k}\ge \text{gap}^{\text{A}}_{k}$}
        \State $\bm{d}_{k}\gets \bm{d}^{\text{FW}}_{k}$,\;
              $\gamma_{\max}\gets 1$
    \Else
        \State $\bm{d}_{k}\gets \bm{d}^{\text{A}}_{k}$,\;
              $\gamma_{\max}\gets \dfrac{\alpha^{\bm{v}_{k}}_{k}}{1-\alpha^{\bm{v}_{k}}_{k}}$
    \EndIf
    \State $\gamma_{k}\gets \min\!\Bigl\{\gamma_{\max},\;
          \dfrac{-\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle}{L\|\bm{d}_{k}\|^{2}}\Bigr\}$ \Comment{Exact line search}
    \State $\bm{x}_{k+1}\gets \bm{x}_{k}+ \gamma_{k}\bm{d}_{k}$
    \State Update coefficients $\alpha^{\bm{v}}_{k+1}$ (add $\bm{s}_{k}$ if FW step, decrease $\alpha^{\bm{v}_{k}}$ if away step, drop vertex if its weight becomes $0$)
    \State $g_{k}\gets \langle\nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{s}_{k}\rangle$
    \If{$g_{k}\le\varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $\bm{x}_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^{2}$ over the 1‑dimensional polytope $\mathcal{C}= [0,5]$.
The vertex set is $\mathcal{V}=\{0,5\}$.  
Take a fixed step size $\eta=0.1$ (instead of the line‑search above) for illustration.

\begin{center}
\begin{tabular}{c|c|c|c|c}
\textbf{Iter} & $w_{k}$ & $\nabla f(w_{k})$ & $\bm{s}_{k}$ (LMO) & $w_{k+1}=w_{k}+\eta d_{k}$ \\
\hline
0 & $0$ & $-6$ & $\displaystyle\argmin_{s\in\{0,5\}}(-6)s = 5$ & $0+\eta(5-0)=0.5$ \\
1 & $0.5$ & $-5$ & $\displaystyle\argmin_{s}(-5)s =5$ & $0.5+\eta(5-0.5)=0.95$ \\
2 & $0.95$ & $-4.1$ & $\displaystyle\argmin_{s}(-4.1)s =5$ & $0.95+\eta(5-0.95)=1.355$ \\
\end{tabular}
\end{center}
The objective values are $f(0)=9$, $f(0.5)=6.25$, $f(0.95)=4.2025$, $f(1.355)=2.949\ldots$, showing monotone decrease.  Because the away step never activates (the active set always contains only the vertex $5$), the algorithm behaves as the classic FW with a constant step size.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*:}, leftmargin=*]
  \item \textbf{Convexity and Smoothness.} $f:\R^{d}\to\R$ is convex and $L$‑smooth on $\mathcal{C}$:
  \[
  f(\bm{y})\le f(\bm{x})+\langle\nabla f(\bm{x}),\bm{y}-\bm{x}\rangle+\frac{L}{2}\|\bm{y}-\bm{x}\|^{2},
  \qquad\forall \bm{x},\bm{y}\in\mathcal{C}. \tag{A1}
  \]
  \item \textbf{Strong Convexity (optional for linear rate).} There exists $\mu>0$ such that
  \[
  f(\bm{y})\ge f(\bm{x})+\langle\nabla f(\bm{x}),\bm{y}-\bm{x}\rangle+\frac{\mu}{2}\|\bm{y}-\bm{x}\|^{2},
  \qquad\forall \bm{x},\bm{y}\in\mathcal{C}. \tag{A2}
  \]
  \item \textbf{Polytope Geometry.} $\mathcal{C}=\conv(\mathcal{V})$ is a compact polytope with \emph{pyramidal width} $\delta>0$ (see Lemma~2 below).  The diameter $\diam(\mathcal{C})\triangleq\max_{\bm{u},\bm{v}\|\bm{u}-\bm{v}\|$ is finite. \tag{A3}
  \item \textbf{Bounded Gradient.} $\|\nabla f(\bm{x})\|\le G$ for all $\bm{x}\in\mathcal{C}$. \tag{A4}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AFW]\label{thm:linear}
Assume \textbf{A1}–\textbf{A3} and that $f$ is $\mu$‑strongly convex (\textbf{A2}).  
Let $\{ \bm{x}_{k}\}$ be the sequence generated by Algorithm~\ref{alg:afw} with exact line‑search step size.  
Define the optimal value $f^{*}\triangleq\min_{\bm{x}\in\mathcal{C}}f(\bm{x})$ and the dual gap $g_{k}= \langle\nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{s}_{k}\rangle$.  
Then for every iteration $k$,
\[
f(\bm{x}_{k})-f^{*}\;\le\;(1-\rho)^{k}\bigl(f(\bm{x}_{0})-f^{*}\bigr),
\qquad\text{with}\quad 
\rho\;:=\;\frac{\mu\,\delta^{2}}{L\,\diam(\mathcal{C})^{2}}\in(0,1). \tag{1}
\]
Consequently $g_{k}\le 2L\diam(\mathcal{C})^{2}(1-\rho)^{k}$ and the algorithm terminates after at most
\[
k\;\ge\; \frac{1}{\rho}\log\!\Bigl(\frac{f(\bm{x}_{0})-f^{*}}{\varepsilon}\Bigr)
\]
iterations to achieve $f(\bm{x}_{k})-f^{*}\le\varepsilon$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\textbf{P\arabic*:}, leftmargin=*]
  \item \textbf{Stability.} The step size $\gamma_{k}$ is always bounded by $\gamma_{\max}\le 1$, guaranteeing that all iterates remain inside $\mathcal{C}$.
  \item \textbf{Hyperparameter Sensitivity.} The only algorithmic parameter is the smoothness constant $L$ (used in the line‑search). Over‑estimating $L$ yields smaller steps but does not affect the linear rate; under‑estimating $L$ may violate the sufficient decrease condition.
  \item \textbf{Comparison to Classic FW.} Classic Frank–Wolfe enjoys a sublinear $O(1/k)$ rate even under strong convexity. The away‑step mechanism eliminates ``zig‑zag'' behavior on polytopes, upgrading the rate to linear as quantified in \eqref{1}.
\end{enumerate}

\section*{Discussion}
The linear factor $\rho$ contains the ratio $\mu/L$ (condition number) and the squared pyramidal width $\delta^{2}$, which reflects how well the polytope is “pointed”.  For simplices $\delta=1$, while for poorly conditioned polytopes $\delta$ may be very small, slowing convergence.  Nevertheless, the bound is dimension‑free and matches the best known rates for conditional gradient methods with away steps.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Pyramidal Width]\label{def:pyr}
For a polytope $\mathcal{C}=\conv(\mathcal{V})$, the pyramidal width is
\[
\delta \;:=\; \min_{\substack{\bm{x}\in\mathcal{C}\\\bm{s}\in\mathcal{C}}}
\;\max_{\bm{v}\in\mathcal{V}}
\frac{\langle \bm{s}-\bm{x},\bm{v}-\bm{x}\rangle}{\|\bm{s}-\bm{x}\|^{2}}.
\]
It satisfies $0<\delta\le 1$ for any non‑degenerate polytope. \hfill\qedsymbol
\end{definition}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $\bm{x},\bm{d}\in\R^{d}$ and $\gamma\in[0,1]$,
\[
f(\bm{x}+\gamma\bm{d})\le f(\bm{x})+\gamma\langle\nabla f(\bm{x}),\bm{d}\rangle+\frac{L\gamma^{2}}{2}\|\bm{d}\|^{2}. \tag{2}
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness (A1) with $\bm{y}=\bm{x}+\gamma\bm{d}$ and rearranging terms. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Strong convexity implies quadratic lower bound]\label{lem:strong}
If $f$ is $\mu$‑strongly convex (A2) then for any $\bm{x},\bm{y}\in\mathcal{C}$,
\[
\frac{\mu}{2}\|\bm{x}-\bm{y}\|^{2}\le f(\bm{x})-f(\bm{y})-\langle\nabla f(\bm{y}),\bm{x}-\bm{y}\rangle. \tag{3}
\]
\end{lemma}
\begin{proof}
Subtract the right‑hand side of (A2) from the left‑hand side and rearrange. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we denote by $g_{k}=\langle\nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{s}_{k}\rangle$ the FW dual gap.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{Descent after one iteration.}  
Using Lemma~\ref{lem:descent} with $\bm{d}_{k}$ and $\gamma_{k}$,
\[
f(\bm{x}_{k+1})\le f(\bm{x}_{k})+\gamma_{k}\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle+\frac{L\gamma_{k}^{2}}{2}\|\bm{d}_{k}\|^{2}. \tag{4}
\]
\item \textbf{Sufficient decrease from the line‑search choice.}  
By construction of $\gamma_{k}$,
\[
\gamma_{k}= \min\Bigl\{\gamma_{\max},
\frac{-\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle}{L\|\bm{d}_{k}\|^{2}}\Bigr\}.
\]
Hence either $\gamma_{k}=\gamma_{\max}$ (case A) or $\gamma_{k}= -\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle/(L\|\bm{d}_{k}\|^{2})$ (case B).  
In case B, plugging the expression of $\gamma_{k}$ into (4) yields
\[
f(\bm{x}_{k+1})\le f(\bm{x}_{k})-\frac{\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle^{2}}{2L\|\bm{d}_{k}\|^{2}}. \tag{5}
\]
\item \textbf{Relating the chosen direction to the dual gap.}  
By definition of the away direction,
\[
\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle = 
\begin{cases}
-\;g_{k} & \text{if a FW step is taken},\\[4pt]
-\;\bigl(g_{k}^{\text{A}}\bigr) & \text{if an away step is taken},
\end{cases}
\]
where $g_{k}^{\text{A}}=-\langle\nabla f(\bm{x}_{k}),\bm{d}^{\text{A}}_{k}\rangle\ge g_{k}$ because the algorithm picks the larger of the two gaps.  Thus, in both cases,
\[
\langle\nabla f(\bm{x}_{k}),\bm{d}_{k}\rangle\le -g_{k}. \tag{6}
\]
\item \textbf{Lower bounding $\|\bm{d}_{k}\|$ via pyramidal width.}  
Let $\bm{x}^{*}$ be the optimal point.  Since $\bm{s}_{k}$ solves the LMO,
\[
\langle\nabla f(\bm{x}_{k}),\bm{s}_{k}-\bm{x}^{*}\rangle\le 0.
\]
Rearranging gives
\[
\langle\nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{s}_{k}\rangle \ge 
\langle\nabla f(\bm{x}_{k}),\bm{x}_{k}-\bm{x}^{*}\rangle .
\]
Using the definition of pyramidal width $\delta$ (Definition~\ref{def:pyr}) we obtain
\[
\|\bm{d}_{k}\|\;\ge\; \delta\,\|\bm{x}_{k}-\bm{x}^{*}\| . \tag{7}
\]
\item \textbf{Combining (5)–(7).}  
From (5) and (6),
\[
f(\bm{x}_{k+1})\le f(\bm{x}_{k})-\frac{g_{k}^{2}}{2L\|\bm{d}_{k}\|^{2}}.
\]
Insert (7):
\[
f(\bm{x}_{k+1})\le f(\bm{x}_{k})-\frac{g_{k}^{2}}{2L\delta^{2}\|\bm{x}_{k}-\bm{x}^{*}\|^{2}}. \tag{8}
\]
\item \textbf{Relating the gap to suboptimality via strong convexity.}  
Lemma~\ref{lem:strong} with $\bm{y}=\bm{x}^{*}$ gives
\[
\frac{\mu}{2}\|\bm{x}_{k}-\bm{x}^{*}\|^{2}\le f(\bm{x}_{k})-f^{*}-\langle\nabla f(\bm{x}^{*}),\bm{x}_{k}-\bm{x}^{*}\rangle
= f(\bm{x}_{k})-f^{*},
\]
since $\nabla f(\bm{x}^{*})=0$.  Moreover, convexity yields $g_{k}\ge f(\bm{x}_{k})-f^{*}$ (the dual gap upper bounds suboptimality).  Hence,
\[
g_{k}\ge f(\bm{x}_{k})-f^{*}. \tag{9}
\]
\item \textbf{Deriving the linear recursion.}  
Insert (9) into (8) and use the bound $\|\bm{x}_{k}-\bm{x}^{*}\|^{2}\le \dfrac{2}{\mu}\bigl(f(\bm{x}_{k})-f^{*}\bigr)$ from Lemma~\ref{lem:strong}:
\[
\begin{aligned}
f(\bm{x}_{k+1})-f^{*}
&\le f(\bm{x}_{k})-f^{*}
-\frac{(f(\bm{x}_{k})-f^{*})^{2}}{2L\delta^{2}\,\frac{2}{\mu}\,(f(\bm{x}_{k})-f^{*})}\\[4pt]
&= \Bigl(1-\frac{\mu\delta^{2}}{L\,\diam(\mathcal{C})^{2}}\Bigr)\bigl(f(\bm{x}_{k})-f^{*}\bigr),
\end{aligned}
\tag{10}
\]
where we used $\|\bm{x}_{k}-\bm{x}^{*}\|\le \diam(\mathcal{C})$ to replace the denominator by the worst‑case bound $\diam(\mathcal{C})^{2}$, thus defining
\[
\rho:=\frac{\mu\delta^{2}}{L\,\diam(\mathcal{C})^{2}}.
\]
Equation (10) is exactly the linear recursion claimed in \eqref{1}.
\item \textbf{Unrolling the recursion.}  
Iterating (10) yields
\[
f(\bm{x}_{k})-f^{*}\le (1-\rho)^{k}\bigl(f(\bm{x}_{0})-f^{*}\bigr),\qquad k\ge0,
\]
which completes the proof of the first part of Theorem~\ref{thm:linear}.
\item \textbf{Dual‑gap bound.}  
From convexity $g_{k}\le 2L\diam(\mathcal{C})^{2}(1-\rho)^{k}$ follows by combining $g_{k}\le L\|\bm{x}_{k}-\bm{s}_{k}\|^{2}$ (by smoothness) and the linear decay of $f(\bm{x}_{k})-f^{*}$.
\item \textbf{Complexity bound.}  
Solving $(1-\rho)^{k}\le \varepsilon/(f(\bm{x}_{0})-f^{*})$ for $k$ gives the iteration bound stated in the theorem.
\end{enumerate}
All steps rely exclusively on assumptions \textbf{A1}–\textbf{A3} and elementary algebra, which concludes the proof. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
Summarising constants:
\[
\boxed{
\rho = \frac{\mu\,\delta^{2}}{L\,\diam(\mathcal{C})^{2}},\qquad
C_{f}=L\,\diam(\mathcal{C})^{2},
}
\]
so that the linear rate can be written as
\[
f(\bm{x}_{k})-f^{*}\le \bigl(1-\tfrac{\mu\delta^{2}}{C_{f}}\bigr)^{k}\bigl(f(\bm{x}_{0})-f^{*}\bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\textbf{S\arabic*:}, leftmargin=*]
  \item \textbf{Simplex.} For $\mathcal{C}=\Delta_{n}$, $\delta=1$ and $\diam(\Delta_{n})=\sqrt{2}$, so $\rho=\mu/(2L)$. The AFW reduces to the classic ``pairwise FW'' with the same linear factor.
  \item \textbf{Strongly convex but no away steps.} If we omit away steps, the recursion (10) collapses to $f(\bm{x}_{k+1})-f^{*}\le (1-\tfrac{\mu}{L})\bigl(f(\bm{x}_{k})-f^{*}\bigr)$ only when $\mathcal{C}$ is a Euclidean ball; otherwise we retain the sublinear $O(1/k)$ rate.
  \item \textbf{Non‑strongly convex $f$.} When $\mu=0$, the linear factor $\rho$ vanishes and the proof yields the standard $O(1/k)$ convergence of the vanilla Frank–Wolfe method.
\end{enumerate}
