\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) with Double Loop}

\section*{Mathematical Formulation}
Consider a finite‑sum objective
\[
F(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad f_i:\mathbb{R}^{d}\rightarrow\mathbb{R},
\]
where each component $f_i$ is convex, $L$‑smooth and $F$ is $\mu$‑strongly convex.  
SVRG uses two loops:

\begin{itemize}
\item \textbf{Outer loop} $s=0,1,2,\dots$ constructs a \emph{snapshot} point $\tilde{x}^{\,s}$ and computes the \emph{full gradient}
\[
\tilde{\mu}^{\,s}:=\nabla F(\tilde{x}^{\,s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s}).
\]

\item \textbf{Inner loop} $t=0,1,\dots,m-1$ performs $m$ stochastic updates.  
Let $x_{0}^{\,s}:=\tilde{x}^{\,s}$. For each inner iteration draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and define the variance‑reduced estimator
\[
v_t^{\,s}= \nabla f_{i_t}\!\bigl(x_{t}^{\,s}\bigr)-\nabla f_{i_t}\!\bigl(\tilde{x}^{\,s}\bigr)+\tilde{\mu}^{\,s}.
\]
The iterate is updated by a standard SGD step with stepsize $\eta>0$:
\[
x_{t+1}^{\,s}=x_{t}^{\,s}-\eta\,v_t^{\,s}.
\]

After the inner loop, the next snapshot is set to the last inner iterate,
\[
\tilde{x}^{\,s+1}=x_{m}^{\,s}.
\]
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG}
\begin{algorithmic}[1]
\Require Number of outer epochs $S$, inner loop length $m$, stepsize $\eta>0$, initial point $\tilde{x}^{\,0}$
\For{$s=0$ \textbf{to} $S-1$}
    \State Compute full gradient $\displaystyle \tilde{\mu}^{\,s}\gets\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})$
    \State $x_{0}^{\,s}\gets\tilde{x}^{\,s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t^{\,s}\gets \nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$
        \State $x_{t+1}^{\,s}\gets x_{t}^{\,s}-\eta\,v_t^{\,s}$
    \EndFor
    \State $\tilde{x}^{\,s+1}\gets x_{m}^{\,s}$
\EndFor
\State \textbf{output} $\tilde{x}^{\,S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ (here $n=1$, $f_1=f$).  
Parameters: $\eta=0.1$, $m=3$, $S=1$, initial point $\tilde{x}^{\,0}=0$.

\begin{enumerate}[label=Step \arabic*:, leftmargin=*]
\item Compute full gradient at snapshot:
\[
\tilde{\mu}^{\,0}= \nabla f(0)=2(0-3)=-6.
\]

\item Inner iteration $t=0$ (sample $i_0=1$):
\[
v_0^{\,0}= \nabla f(x_{0}^{\,0})-\nabla f(\tilde{x}^{\,0})+\tilde{\mu}^{\,0}
           =(-6)-(-6)+(-6)=-6.
\]
Update:
\[
x_{1}^{\,0}=x_{0}^{\,0}-\eta v_0^{\,0}=0-0.1(-6)=0.6.
\]
Objective value $f(0.6)=(0.6-3)^2=5.76$.

\item Inner iteration $t=1$:
\[
\nabla f(x_{1}^{\,0})=2(0.6-3)=-4.8,\qquad
v_1^{\,0}= -4.8-(-6)+(-6)=-4.8+6-6=-4.8.
\]
Update:
\[
x_{2}^{\,0}=0.6-0.1(-4.8)=1.08,\qquad f(1.08)=3.6864.
\]

\item Inner iteration $t=2$:
\[
\nabla f(x_{2}^{\,0})=2(1.08-3)=-3.84,\qquad
v_2^{\,0}= -3.84-(-6)+(-6)=-3.84+6-6=-3.84.
\]
Update:
\[
x_{3}^{\,0}=1.08-0.1(-3.84)=1.464,\qquad f(1.464)=2.366.
\]

\item End of inner loop: set $\tilde{x}^{\,1}=x_{3}^{\,0}=1.464$.
\end{enumerate}
The algorithm has moved the iterate towards the optimum $w^\star=3$ while the objective decreased from $9$ to $\approx2.37$ in three stochastic steps.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$‑smooth, i.e.
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
Equivalently,
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
The objective $F$ is $\mu$‑strongly convex:
\[
F(y)\ge F(x)+\langle\nabla F(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,\qquad\forall x,y .
\]
\end{assumption}

\begin{assumption}[Sampling]\label{ass:sampling}
At every inner iteration an index $i_t$ is drawn i.i.d. uniformly from $\{1,\dots,n\}$ and is independent of the past randomness conditioned on the current iterate.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies
\[
0<\eta\le\frac{1}{3L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize} and for any inner loop length $m\ge 1$, the sequence of snapshot points $\{\tilde{x}^{\,s}\}_{s\ge0}$ generated by SVRG satisfies
\[
\mathbb{E}\!\left[F(\tilde{x}^{\,s})-F(x^\star)\right]
\le\rho^{\,s}\,\bigl(F(\tilde{x}^{\,0})-F(x^\star)\bigr),
\qquad
\rho:=\frac{1}{\mu\eta(1-2L\eta)m}+ \frac{2L\eta}{1-2L\eta}\;<1 .
\]
Consequently,
\[
\mathbb{E}\!\left[F(\tilde{x}^{\,S})-F(x^\star)\right]=O\!\bigl(\rho^{\,S}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound on $\eta$ guarantees that the variance‑reduced estimator $v_t^{\,s}$ is an unbiased estimator of $\nabla F(x_{t}^{\,s})$ and its second moment is uniformly bounded (see Lemma~\ref{lem:varbound}). This prevents divergence even when $m$ is large.

\item \textbf{Hyperparameter sensitivity.} The rate $\rho$ is monotone decreasing in both $\eta$ (up to the $1/(3L)$ limit) and $m$. Choosing $m\approx\frac{2}{\mu\eta}$ yields the optimal contraction factor $\rho\approx\frac{2}{3}$.

\item \textbf{Comparison to baselines.}
\begin{enumerate}
\item \emph{Plain SGD} with diminishing stepsizes attains $O(1/T)$ sub‑linear convergence for strongly convex problems.
\item \emph{Full‑gradient GD} obtains the optimal linear rate $1-\mu/L$, but costs $O(n)$ gradient evaluations per iteration.
\item \emph{SVRG} interpolates between the two: each outer epoch costs $O(n+m)$ gradient evaluations and achieves a linear rate with a constant $\rho<1$ that can be made arbitrarily close to $1-\mu/L$ by proper tuning of $\eta$ and $m$.
\end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that SVRG enjoys \emph{geometric decay} of the expected optimality gap while requiring only a single full gradient per epoch. The proof hinges on the smoothness‑strong convexity interplay and a careful bound on the variance of the gradient estimator. The result holds for any finite sum; extensions to non‑convex settings require additional assumptions and lead to different rates (see recent literature).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness]\label{lem:unbiased}
For any $t$ and $s$, the estimator $v_t^{\,s}$ satisfies
\[
\mathbb{E}_{i_t}\!\bigl[\,v_t^{\,s}\mid x_{t}^{\,s},\tilde{x}^{\,s}\bigr]=\nabla F(x_{t}^{\,s}).
\]
\end{lemma}
\begin{proof}
Conditioning on the current iterates,
\[
\mathbb{E}_{i_t}[v_t^{\,s}]
= \frac{1}{n}\sum_{i=1}^{n}\bigl(\nabla f_i(x_{t}^{\,s})-\nabla f_i(\tilde{x}^{\,s})\bigr)+\tilde{\mu}^{\,s}
= \nabla F(x_{t}^{\,s})-\nabla F(\tilde{x}^{\,s})+\nabla F(\tilde{x}^{\,s})
= \nabla F(x_{t}^{\,s}).\qedhere
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:varbound}
For any $t,s$,
\[
\mathbb{E}_{i_t}\!\bigl[\|v_t^{\,s}\|^{2}\mid x_{t}^{\,s},\tilde{x}^{\,s}\bigr]
\le 4L\bigl(F(x_{t}^{\,s})-F(x^\star)\bigr) + 4L\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr).
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the identity $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$,
\[
\begin{aligned}
\|v_t^{\,s}\|^{2}
&= \bigl\|\nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(x^\star)
      -\bigl(\nabla f_{i_t}(\tilde{x}^{\,s})-\nabla f_{i_t}(x^\star)\bigr)
      +\tilde{\mu}^{\,s}\bigr\|^{2}\\
&\le 2\bigl\|\nabla f_{i_t}(x_{t}^{\,s})-\nabla f_{i_t}(x^\star)\bigr\|^{2}
    +2\bigl\|\nabla f_{i_t}(\tilde{x}^{\,s})-\nabla f_{i_t}(x^\star)-\tilde{\mu}^{\,s}\bigr\|^{2}.
\end{aligned}
\]
Taking expectation over $i_t$ and applying $L$‑smoothness gives
\[
\mathbb{E}_{i_t}\!\bigl[\|\nabla f_{i_t}(x)-\nabla f_{i_t}(x^\star)\|^{2}\bigr]
\le 2L\bigl(F(x)-F(x^\star)\bigr),
\]
and similarly for the second term. Summing the two bounds yields the claim. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}. Let $x^\star$ denote the unique minimizer of $F$.

\begin{enumerate}[label=[Step \arabic*], leftmargin=*]
\item \textbf{One‑step descent inequality.}  
From the update $x_{t+1}^{\,s}=x_{t}^{\,s}-\eta v_t^{\,s}$,
\[
\|x_{t+1}^{\,s}-x^\star\|^{2}
= \|x_{t}^{\,s}-x^\star\|^{2}
-2\eta\langle v_t^{\,s},x_{t}^{\,s}-x^\star\rangle
+\eta^{2}\|v_t^{\,s}\|^{2}. \tag{1}
\]

\item \textbf{Take conditional expectation.}  
Condition on the sigma‑algebra $\mathcal{F}_{t}^{\,s}$ generated by all randomness up to iteration $(t,s)$. Using Lemma~\ref{lem:unbiased},
\[
\mathbb{E}\!\bigl[\langle v_t^{\,s},x_{t}^{\,s}-x^\star\rangle\mid\mathcal{F}_{t}^{\,s}\bigr]
= \langle\nabla F(x_{t}^{\,s}),x_{t}^{\,s}-x^\star\rangle . \tag{2}
\]

\item \textbf{Apply strong convexity.}  
By Assumption~\ref{ass:strong},
\[
\langle\nabla F(x_{t}^{\,s}),x_{t}^{\,s}-x^\star\rangle
\ge F(x_{t}^{\,s})-F(x^\star)+\frac{\mu}{2}\|x_{t}^{\,s}-x^\star\|^{2}. \tag{3}
\]

\item \textbf{Bound the variance term.}  
From Lemma~\ref{lem:varbound},
\[
\mathbb{E}\!\bigl[\|v_t^{\,s}\|^{2}\mid\mathcal{F}_{t}^{\,s}\bigr]
\le 4L\bigl(F(x_{t}^{\,s})-F(x^\star)\bigr)
   +4L\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr). \tag{4}
\]

\item \textbf{Insert (2)–(4) into (1) and take expectation.}  
Using linearity of expectation and the tower property,
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|x_{t+1}^{\,s}-x^\star\|^{2}\mid\mathcal{F}_{t}^{\,s}\bigr]
&\le \|x_{t}^{\,s}-x^\star\|^{2}
-2\eta\Bigl(F(x_{t}^{\,s})-F(x^\star)\Bigr)
-\eta\mu\|x_{t}^{\,s}-x^\star\|^{2} \\
&\quad +\eta^{2}\Bigl[4L\bigl(F(x_{t}^{\,s})-F(x^\star)\bigr)
+4L\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr)\Bigr].
\end{aligned} \tag{5}
\]

\item \textbf{Collect like terms.}  
Rearrange (5) as
\[
\begin{aligned}
\mathbb{E}\!\bigl[\|x_{t+1}^{\,s}-x^\star\|^{2}\mid\mathcal{F}_{t}^{\,s}\bigr]
&\le \bigl(1-\eta\mu\bigr)\|x_{t}^{\,s}-x^\star\|^{2} \\
&\quad -2\eta\bigl(1-2L\eta\bigr)\bigl(F(x_{t}^{\,s})-F(x^\star)\bigr)
+4L\eta^{2}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr).
\end{aligned} \tag{6}
\]

\item \textbf{Define a potential function.}  
Let
\[
\Phi_{t}^{\,s}:= \mathbb{E}\!\bigl[\|x_{t}^{\,s}-x^\star\|^{2}\bigr]
+ \alpha\,\mathbb{E}\!\bigl[F(x_{t}^{\,s})-F(x^\star)\bigr],
\]
where $\alpha>0$ will be chosen later.

\item \textbf{Derive a recursion for $\Phi_{t}^{\,s}$.}  
Taking total expectation in (6) and adding $\alpha$ times the expectation of $F(x_{t}^{\,s})-F(x^\star)$ yields
\[
\Phi_{t+1}^{\,s}
\le (1-\eta\mu)\Phi_{t}^{\,s}
-\bigl[2\eta(1-2L\eta)-\alpha\bigr]\mathbb{E}[F(x_{t}^{\,s})-F(x^\star)]
+4L\eta^{2}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr). \tag{7}
\]

\item \textbf{Choose $\alpha$ to cancel the middle term.}  
Set
\[
\alpha = 2\eta(1-2L\eta). \tag{8}
\]
Since $\eta\le \frac{1}{3L}$, we have $1-2L\eta>0$ and $\alpha>0$.

With this choice, the bracket in (7) vanishes and we obtain
\[
\Phi_{t+1}^{\,s}
\le (1-\eta\mu)\Phi_{t}^{\,s}
+4L\eta^{2}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr). \tag{9}
\]

\item \textbf{Unroll the inner recursion for $t=0,\dots,m-1$.}  
Iterating (9) yields
\[
\Phi_{m}^{\,s}
\le (1-\eta\mu)^{m}\Phi_{0}^{\,s}
+4L\eta^{2}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr)
\sum_{k=0}^{m-1}(1-\eta\mu)^{k}. \tag{10}
\]

The geometric sum satisfies
\[
\sum_{k=0}^{m-1}(1-\eta\mu)^{k}
=\frac{1-(1-\eta\mu)^{m}}{\eta\mu}
\le\frac{1}{\eta\mu}. \tag{11}
\]

Plugging (11) into (10) gives
\[
\Phi_{m}^{\,s}
\le (1-\eta\mu)^{m}\Phi_{0}^{\,s}
+\frac{4L\eta}{\mu}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr). \tag{12}
\]

\item \textbf{Relate $\Phi_{0}^{\,s}$ and $\Phi_{m}^{\,s}$ to snapshots.}  
Recall $x_{0}^{\,s}=\tilde{x}^{\,s}$ and $\tilde{x}^{\,s+1}=x_{m}^{\,s}$. Moreover, by definition of $\Phi$ and (8),
\[
\Phi_{0}^{\,s}= \mathbb{E}\!\bigl[\|\tilde{x}^{\,s}-x^\star\|^{2}\bigr]
+2\eta(1-2L\eta)\,\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr],
\]
\[
\Phi_{m}^{\,s}= \mathbb{E}\!\bigl[\|\tilde{x}^{\,s+1}-x^\star\|^{2}\bigr]
+2\eta(1-2L\eta)\,\mathbb{E}\!\bigl[F(\tilde{x}^{\,s+1})-F(x^\star)\bigr].
\]

\item \textbf{Discard the non‑negative distance term.}  
Since $\|\,\cdot\,\|^{2}\ge0$, inequality (12) implies
\[
2\eta(1-2L\eta)\,\mathbb{E}\!\bigl[F(\tilde{x}^{\,s+1})-F(x^\star)\bigr]
\le (1-\eta\mu)^{m}\Phi_{0}^{\,s}
+\frac{4L\eta}{\mu}\,\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr]. \tag{13}
\]

\item \textbf{Upper bound $\Phi_{0}^{\,s}$ by a multiple of the optimality gap.}  
Using smoothness and strong convexity,
\[
\|\tilde{x}^{\,s}-x^\star\|^{2}
\le \frac{2}{\mu}\bigl(F(\tilde{x}^{\,s})-F(x^\star)\bigr).
\]
Thus,
\[
\Phi_{0}^{\,s}
\le \left(\frac{2}{\mu}+2\eta(1-2L\eta)\right)
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr]. \tag{14}
\]

\item \textbf{Combine (13) and (14).}  
Dividing both sides of (13) by $2\eta(1-2L\eta)$ and substituting (14) gives
\[
\begin{aligned}
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s+1})-F(x^\star)\bigr]
&\le \frac{(1-\eta\mu)^{m}}{2\eta(1-2L\eta)}
\left(\frac{2}{\mu}+2\eta(1-2L\eta)\right)
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr] \\
&\quad +\frac{2L}{\mu(1-2L\eta)}\,
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr].
\end{aligned}
\]

Simplifying,
\[
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s+1})-F(x^\star)\bigr]
\le \underbrace{\left[\frac{1}{\mu\eta(1-2L\eta)m}
      +\frac{2L\eta}{1-2L\eta}\right]}_{\displaystyle\rho}
      \,\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr],
\]
where we used the elementary bound $(1-\eta\mu)^{m}\le 1/(\mu\eta m)$ (valid for $0<\eta\mu\le1$). This is exactly the contraction factor stated in Theorem~\ref{thm:linear}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From the recurrence
\[
\mathbb{E}\!\bigl[F(\tilde{x}^{\,s})-F(x^\star)\bigr]\le\rho^{\,s}\bigl(F(\tilde{x}^{\,0})-F(x^\star)\bigr),
\]
and the definition of $\rho$, we obtain an explicit bound:
\[
\rho
= \frac{1}{\mu\eta(1-2L\eta)m}+ \frac{2L\eta}{1-2L\eta}
\le \frac{1}{\mu\eta m}+2L\eta,
\]
where the last inequality follows from $1-2L\eta\ge\frac13$ (recall $\eta\le\frac{1}{3L}$). Choosing
\[
\eta=\frac{1}{3L},\qquad m= \left\lceil\frac{6L}{\mu}\right\rceil
\]
yields $\rho\le\frac{2}{3}$, i.e. a geometric decay factor of $0.666\ldots$ per outer epoch.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact gradient descent.} When $m=1$ and $n=1$, SVRG reduces to full‑gradient GD and $\rho=1-\mu\eta$, reproducing the classical linear rate.
\item \textbf{Large batch limit.} If $m\to\infty$, the term $\frac{1}{\mu\eta(1-2L\eta)m}\to0$ and $\rho\to\frac{2L\eta}{1-2L\eta}$. Selecting $\eta\approx\frac{1}{2L}$ yields $\rho\approx1$, reflecting that an excessively long inner loop erodes the benefit of variance reduction.
\item \textbf{Quadratic objectives.} For $F(x)=\frac12x^{\top}Ax-b^{\top}x$ with $A$ SPD and eigenvalues in $[\mu,L]$, all inequalities become equalities, and the bound above is tight.
\end{enumerate}
