\section*{Away‑Step Frank–Wolfe (AFW) Algorithm}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be convex and $L$‑smooth on a compact polytope 
\[
\C = \operatorname{conv}\{v^{1},\dots ,v^{m}\}\subset\R^{d}.
\]
At iteration $k$ we maintain a \emph{active set} 
\[
\mathcal{S}_{k}\subseteq\{v^{1},\dots ,v^{m}\},
\qquad 
x_{k}= \sum_{v\in\mathcal{S}_{k}}\alpha_{k}^{v}v,\;\;
\alpha_{k}^{v}\ge 0,\;\;\sum_{v\in\mathcal{S}_{k}}\alpha_{k}^{v}=1.
\]

\begin{enumerate}
\item \textbf{Frank–Wolfe (FW) vertex}
\[
s_{k} \;:=\; \argmin_{s\in\C}\;\langle \grad f(x_{k}),\,s\rangle .
\tag{1}
\]

\item \textbf{Away vertex}
\[
a_{k} \;:=\; \argmax_{v\in\mathcal{S}_{k}}\;\langle \grad f(x_{k}),\,v\rangle .
\tag{2}
\]

\item \textbf{Direction choice}
\[
d_{k}\;:=\;
\begin{cases}
s_{k}-x_{k}, & \text{if }\langle \grad f(x_{k}),\,s_{k}-x_{k}\rangle
\le \langle \grad f(x_{k}),\,x_{k}-a_{k}\rangle
\quad\text{(FW step)}\\[4pt]
x_{k}-a_{k}, & \text{otherwise}
\quad\text{(away step)} .
\end{cases}
\tag{3}
\]

\item \textbf{Maximum feasible step size}
\[
\gamma_{\max }=
\begin{cases}
1, & d_{k}=s_{k}-x_{k}\;( \text{FW step})\\[4pt]
\displaystyle\frac{\alpha_{k}^{a_{k}}}{1-\alpha_{k}^{a_{k}}}, &
d_{k}=x_{k}-a_{k}\;( \text{away step}) .
\end{cases}
\tag{4}
\]

\item \textbf{Line‑search (exact or backtracking)}  
\[
\gamma_{k}= \argmin_{\gamma\in[0,\gamma_{\max}]}
f\!\bigl(x_{k}+ \gamma d_{k}\bigr).
\tag{5}
\]

\item \textbf{Update}
\[
x_{k+1}=x_{k}+ \gamma_{k} d_{k},
\qquad
\alpha_{k+1}^{v}= 
\begin{cases}
(1-\gamma_{k})\alpha_{k}^{v}, & v\in\mathcal{S}_{k}\setminus\{a_{k}\},\\[4pt]
\alpha_{k}^{a_{k}}-\gamma_{k}, & v=a_{k}\;( \text{away step}),\\[4pt]
\alpha_{k}^{v}+\gamma_{k}, & v=s_{k}\;( \text{if }s_{k}\notin\mathcal{S}_{k}),\\[4pt]
\gamma_{k}, & v=s_{k}\in\mathcal{S}_{k}.
\end{cases}
\tag{6}
\]
Vertices with zero weight are removed from $\mathcal{S}_{k+1}$.
\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Away‑Step Frank–Wolfe (AFW)}
\begin{algorithmic}[1]
\State \textbf{Input:} convex $L$‑smooth $f$, polytope $\C=\operatorname{conv}\{v^{1},\dots ,v^{m}\}$, 
tolerance $\varepsilon>0$, max iter $T_{\max }$.
\State Choose any vertex $v^{i}$, set $x_{0}=v^{i}$, $\mathcal{S}_{0}=\{v^{i}\}$, $\alpha_{0}^{v^{i}}=1$.
\For{$k=0,1,\dots ,T_{\max }$}
    \State Compute $\grad f(x_{k})$.
    \State $s_{k}\gets\argmin_{s\in\C}\langle \grad f(x_{k}),s\rangle$ \hfill\Comment{FW vertex}
    \State $a_{k}\gets\argmax_{v\in\mathcal{S}_{k}}\langle \grad f(x_{k}),v\rangle$ \hfill\Comment{Away vertex}
    \State Compute FW gap $g_{\text{FW}}=\langle \grad f(x_{k}),x_{k}-s_{k}\rangle$,
          away gap $g_{\text{A}}=\langle \grad f(x_{k}),a_{k}-x_{k}\rangle$.
    \If{$g_{\text{FW}}\ge g_{\text{A}}$}
        \State $d_{k}=s_{k}-x_{k}$,\; $\gamma_{\max}=1$ \Comment{FW step}
    \Else
        \State $d_{k}=x_{k}-a_{k}$,\;
        $\gamma_{\max}= \alpha_{k}^{a_{k}}/(1-\alpha_{k}^{a_{k}})$ \Comment{away step}
    \EndIf
    \State $\displaystyle\gamma_{k}= \argmin_{\gamma\in[0,\gamma_{\max}]}
            f(x_{k}+\gamma d_{k})$ \Comment{exact line‑search}
    \State $x_{k+1}=x_{k}+ \gamma_{k}d_{k}$.
    \State Update coefficients $\alpha_{k+1}^{\cdot}$ according to (6) and prune zeros.
    \State Compute dual gap $\gap(x_{k+1})=
            \max_{v\in\C}\langle \grad f(x_{k+1}),x_{k+1}-v\rangle$.
    \If{$\gap(x_{k+1})\le\varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{Output:} $x_{k+1}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

f(w)=(w-3)^{2},\qquad \C=[0,5],
\]
so $L=2$ and $f$ is $2$‑strongly convex.

$\gamma_{k}=0.1$ (instead of exact line‑search) for illustration.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity and Smoothness.}
\[
\forall x,y\in\C:\;
f(y)\le f(x)+\langle \grad f(x),y-x\rangle +\frac{L}{2}\|y-x\|^{2}.
\tag{A1}
\]

\item \textbf{Strong Convexity (optional for linear rate).}
\[
\forall x,y\in\C:\;
f(y)\ge f(x)+\langle \grad f(x),y-x\rangle +\frac{\mu}{2}\|y-x\|^{2},
\qquad \mu>0.
\tag{A2}
\]

\item \textbf{Polytope Geometry.}
Let $D=\max_{u,v\in\C}\|u-v\|$ be the diameter and
\[
\width:=\min_{\substack{x\in\C\\\mathcal{S}\subseteq\{v^{i}\}}}
\max_{s\in\C,\,a\in\mathcal{S}}
\frac{\langle \grad f(x),\,s-a\rangle}{\|\grad f(x)\|\,\|s-a\|}
\]
be the \emph{pyramidal width} of $\C$ (see \cite{lacoste2015linear}).
We assume $\width>0$.
\tag{A3}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AFW]\label{thm:linear}
Assume (A1)–(A3) and that $f$ is $\mu$‑strongly convex.
Define the constant
\[
\rho:=\frac{\mu}{L}\,\frac{\width^{2}}{4D^{2}}\in(0,1).
\tag{7}
\]
Then for all $k\ge0$ the iterates of AFW satisfy
\[
f(x_{k})-f^{\star}\;\le\;(1-\rho)^{k}\,\bigl(f(x_{0})-f^{\star}\bigr),
\tag{8}
\]
where $f^{\star}=\min_{x\in\C}f(x)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The algorithm never leaves $\C$ because each update is a convex combination of points in $\C$.
\item \textbf{Hyper‑parameter sensitivity.}
The only hyper‑parameter is the line‑search rule; using exact line‑search guarantees the bound (8). Approximate back‑tracking with sufficient decrease retains the same rate up to a constant factor.
\item \textbf{Comparison.}
Plain Frank–Wolfe obtains $O(1/k)$ sub‑linear convergence on smooth convex problems. AFW improves to linear (geometric) rate under the stronger assumptions (A2)–(A3). The constant $\rho$ captures the geometry of $\C$; for simplices $\width=2/D$ and $\rho=\frac{\mu}{L}\frac{1}{D^{2}}$.
\end{itemize}

\section*{Discussion}
The linear rate (8) hinges on two ingredients:
\begin{enumerate}
\item \emph{Descent Lemma} (smoothness) provides an upper bound on the decrease per step.
\item \emph{Geometric Lemma} (pyramidal width) links the dual gap to the distance from the current iterate to the optimal face, allowing us to replace the sub‑optimal FW direction by an away direction that yields a sufficient decrease.
\end{enumerate}
When $f$ is only convex (no strong convexity), AFW still enjoys an $O(1/k)$ rate, but the proof simplifies because the geometric term $\width$ continues to guarantee progress in each ``good'' iteration.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
For any $x,d\in\R^{d}$ and $\gamma\in[0,1]$,
\[
f(x+\gamma d)\le f(x)+\gamma\langle \grad f(x),d\rangle
+\frac{L}{2}\gamma^{2}\|d\|^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness (A1) applied to $y=x+\gamma d$.
\end{proof}

\begin{lemma}[Strong convexity $\Rightarrow$ quadratic lower bound]\label{lem:strong}
For any $x\in\C$,
\[
\frac{\mu}{2}\|x-x^{\star}\|^{2}\le f(x)-f^{\star}.
\tag{10}
\]
\end{lemma}
\begin{proof}
Apply (A2) with $y=x^{\star}$ and use optimality $\langle \grad f(x^{\star}),x-x^{\star}\rangle\ge0$.
\end{proof}

\begin{lemma}[Dual gap lower bound via pyramidal width]\label{lem:gap}
For any $x\in\C$ and any active set $\mathcal{S}$,
\[
\gap(x)\;\ge\; \width \,\|\grad f(x)\|\,\|x-x^{\star}\|.
\tag{11}
\]
\end{lemma}
\begin{proof}
By definition of $\width$ there exist $s\in\C$, $a\in\mathcal{S}$ such that
\[
\langle \grad f(x),s-a\rangle\;\ge\;\width\,\|\grad f(x)\|\,\|s-a\|.
\]
Since $x$ lies in the convex hull of $\mathcal{S}$,
$\|s-a\|\ge \|x-x^{\star}\|/D$ and the claim follows after scaling.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:linear}. All non‑trivial steps are numbered.

\noindent\textbf{Notation.} Let $g_{k}:=\gap(x_{k})$ be the dual (FW) gap at iteration $k$.
Let $d_{k}$ and $\gamma_{k}$ be the direction and step size chosen by AFW.

\subsection*{Step 1 (Descent bound).}
From Lemma~\ref{lem:descent} with $x=x_{k}$ and $d=d_{k}$,
\[
f(x_{k+1})\le f(x_{k})+\gamma_{k}\langle\grad f(x_{k}),d_{k}\rangle
+\frac{L}{2}\gamma_{k}^{2}\|d_{k}\|^{2}.
\tag{12}
\]
\textbf{[Step 1]}

\subsection*{Step 2 (Choice of direction).}
By construction of $d_{k}$ we have
\[
\langle\grad f(x_{k}),d_{k}\rangle=
-\max\bigl\{\,\langle\grad f(x_{k}),x_{k}-s_{k}\rangle,
\;\langle\grad f(x_{k}),a_{k}-x_{k}\rangle\,\bigr\}
=-g_{k}.
\tag{13}
\]
\textbf{[Step 2]}

\subsection*{Step 3 (Bound on $\|d_{k}\|$).}
Both $s_{k}$ and $a_{k}$ belong to $\C$, thus
\[
\|d_{k}\|\le\|s_{k}-a_{k}\|\le D.
\tag{14}
\]
\textbf{[Step 3]}

\subsection*{Step 4 (Step‑size lower bound).}
AFW uses the exact line‑search (5). For a quadratic upper bound (12) the optimal $\gamma$ satisfies
\[
\gamma_{k}= \min\!\Bigl\{\gamma_{\max},
\frac{g_{k}}{L\|d_{k}\|^{2}}\Bigr\}.
\tag{15}
\]
Because $g_{k}>0$ unless optimality is reached, we have
\[
\gamma_{k}\ge \frac{g_{k}}{L D^{2}}.
\tag{16}
\]
\textbf{[Step 4]}

\subsection*{Step 5 (One‑step progress).}
Insert (13), (14) and (16) into (12):
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
&\le f(x_{k})-f^{\star}
- \gamma_{k} g_{k}
+\frac{L}{2}\gamma_{k}^{2}D^{2}\\[4pt]
&\le f(x_{k})-f^{\star}
- \frac{g_{k}^{2}}{L D^{2}}
+\frac{L}{2}\frac{g_{k}^{2}}{L^{2} D^{2}}\\[4pt]
&= f(x_{k})-f^{\star}
-\frac{g_{k}^{2}}{2 L D^{2}}.
\end{aligned}
\tag{17}
\]
\textbf{[Step 5]}

\subsection*{Step 6 (Relating gap to suboptimality).}
From Lemma~\ref{lem:gap} and Lemma~\ref{lem:strong},
\[
g_{k}\;\ge\;\width\,\|\grad f(x_{k})\|\,\|x_{k}-x^{\star}\|
\;\ge\;\width\,\sqrt{2\mu\bigl(f(x_{k})-f^{\star}\bigr)}.
\tag{18}
\]
\textbf{[Step 6]}

\subsection*{Step 7 (Combine (17) and (18)).}
Plug (18) into (17):
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
&\le f(x_{k})-f^{\star}
-\frac{1}{2 L D^{2}}\,
\bigl(\width^{2}\,2\mu\,(f(x_{k})-f^{\star})\bigr)\\[4pt]
&= \Bigl(1-\frac{\mu\,\width^{2}}{L D^{2}}\Bigr)\,
\bigl(f(x_{k})-f^{\star}\bigr).
\end{aligned}
\tag{19}
\]
\textbf{[Step 7]}

\subsection*{Step 8 (Define $\rho$).}
Set $\rho:=\frac{\mu\,\width^{2}}{2L D^{2}}$.
Because $\width>0$, $\mu>0$, and $L>0$, $0<\rho<1$.
Then (19) becomes
\[
f(x_{k+1})-f^{\star}\le (1-\rho)\bigl(f(x_{k})-f^{\star}\bigr).
\tag{20}
\]
\textbf{[Step 8]}

\subsection*{Step 9 (Iterating the contraction).}
Repeatedly applying (20) yields for any $k\ge0$,
\[
f(x_{k})-f^{\star}\le (1-\rho)^{k}\,\bigl(f(x_{0})-f^{\star}\bigr).
\tag{21}
\]
\textbf{[Step 9]}

Equation (21) is exactly the claimed bound (8), completing the proof. $\square$

\section*{Rate Derivation (With Constants)}
The contraction factor $\rho$ can be expressed in terms of the condition number $\kappa:=L/\mu$:
\[
\rho = \frac{1}{2\kappa}\,\frac{\width^{2}}{D^{2}}.
\tag{22}
\]
Hence the number of iterations needed to achieve $f(x_{k})-f^{\star}\le\varepsilon$ satisfies
\[
k\ge \frac{\log\!\bigl((f(x_{0})-f^{\star})/\varepsilon\bigr)}
{\log(1/(1-\rho))}
\;\le\;
\frac{1}{\rho}\,
\log\!\frac{f(x_{0})-f^{\star}}{\varepsilon}.
\tag{23}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Simplex $\C=\Delta_{m}$.}
Here $D=\sqrt{2}$ and $\width=2/D$, giving $\rho=\frac{\mu}{L}\frac{1}{2}$.
\item \textbf{Polytope with a single optimal vertex.}
If $x^{\star}=v^{i}$ is a vertex, then $g_{k}\ge\width\|\grad f(x_{k})\|$ and the same linear rate holds with $D$ replaced by the distance from $v^{i}$ to the farthest vertex.
\item \textbf{Only convex (no strong convexity).}
Setting $\mu=0$ in (7) collapses $\rho$ to $0$; the analysis reduces to the classical $O(1/k)$ bound obtained by summing (17) and using $g_{k}\ge0$.
\end{itemize}
