\section*{Algorithm Name}
\textbf{Bregman Mirror Descent (Non‑convex, smooth case)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to \mathbb{R}$ be differentiable (possibly non‑convex) and let $h:\mathbb{R}^d\to\mathbb{R}$ be a \emph{prox‑function} that is $1$‑strongly convex w.r.t. a norm $\|\cdot\|$, i.e.
\[
\breg{x}{y}:=h(x)-h(y)-\langle\nabla h(y),x-y\rangle\ge\frac12\|x-y\|^{2},\qquad\forall x,y.
\]
Given a stepsize sequence $\{\eta_k\}_{k\ge0}$, the iterates are defined by
\begin{equation}\label{eq:update}
x_{k+1}
\;:=\;
\arg\min_{x\in\mathbb{R}^d}
\Big\{\langle\nabla f(x_k),x\rangle+\frac{1}{\eta_k}\,\breg{x}{x_k}\Big\}.
\end{equation}
The optimality condition of \eqref{eq:update} yields the implicit relation
\begin{equation}\label{eq:optcond}
0\;=\;\nabla f(x_k)+\frac{1}{\eta_k}\big(\nabla h(x_{k+1})-\nabla h(x_k)\big). 
\end{equation}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Bregman Mirror Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, stepsizes $\{\eta_k\}_{k\ge0}$, prox‑function $h$.
\FOR{$k=0,1,2,\dots$}
\STATE Compute deterministic gradient $g_k:=\nabla f(x_k)$.
\STATE $x_{k+1}\gets\displaystyle\arg\min_x\Big\{\langle g_k,x\rangle+\frac{1}{\eta_k}\breg{x}{x_k}\Big\}$.
\IF{termination criterion met}
\BREAK
\ENDIF
\ENDFOR
\STATE \textbf{Output:} $x_{K}$ (or averaged iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=(w-3)^2,\qquad h(w)=\tfrac12 w^2\;\;(\Rightarrow\breg{u}{v}=\tfrac12 (u-v)^2).
\]
Take $w_0=0$ and a constant stepsize $\eta=0.1$.

\begin{enumerate}[label=Iteration \arabic*:,leftmargin=*]
\item \textbf{Gradient at $w_0$:} $\nabla f(w_0)=2(w_0-3)=-6$.
\[
w_1=\arg\min_w\Big\{-6w+\frac{1}{0.1}\tfrac12(w-0)^2\Big\}
\Longrightarrow
\frac{d}{dw}\Big(-6w+5w^2\Big)=0\;\Rightarrow\; -6+10w=0\;\Rightarrow\; w_1=0.6.
\]
Objective: $f(w_1)=(0.6-3)^2=5.76$.

\item \textbf{Gradient at $w_1$:} $\nabla f(w_1)=2(0.6-3)=-4.8$.
\[
w_2=\arg\min_w\Big\{-4.8 w+5(w-0.6)^2\Big\}
\Longrightarrow -4.8+10(w-0.6)=0\;\Rightarrow\; w_2=0.96.
\]
Objective: $f(w_2)=(0.96-3)^2=4.1664$.

\item \textbf{Gradient at $w_2$:} $\nabla f(w_2)=2(0.96-3)=-4.08$.
\[
w_3=\arg\min_w\Big\{-4.08 w+5(w-0.96)^2\Big\}
\Longrightarrow -4.08+10(w-0.96)=0\;\Rightarrow\; w_3=1.248.
\]
Objective: $f(w_3)=(1.248-3)^2=3.0675\ (\text{approx}).
\]
\end{enumerate}
The iterates move toward the minimizer $w^\star=3$ as expected.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Strong convexity of $h$:} $h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$,
\[
\breg{x}{y}\ge\frac12\|x-y\|^{2},\qquad\forall x,y\in\mathbb{R}^d.
\]

\item[(A2)] \textbf{$L$‑smoothness of $f$ in Bregman sense:} there exists $L>0$ such that for all $x,y$,
\begin{equation}\label{eq:smooth}
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+L\,\breg{y}{x}.
\end{equation}
This is the natural analogue of $L$‑Lipschitz gradient when the geometry is induced by $h$.

\item[(A3)] \textbf{Bounded level set (optional):} the set $\{x\mid f(x)\le f(x_0)\}$ is bounded.  It guarantees existence of a minimizer $x^\star$ but is not needed for the $O(1/K)$ gradient‑norm rate.

\item[(A4)] \textbf{Stepsize:} a constant $\eta\in\big(0,\frac{1}{2L}\big]$ (or any non‑increasing sequence satisfying $0<\eta_k\le\frac{1}{2L}$).  The upper bound $\frac{1}{2L}$ is required for the algebra in Step 3.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Bregman Mirror Descent]\label{thm:main}
Assume (A1)–(A4) and denote by $x^\star$ any point satisfying $f(x^\star)=\inf f$.  Then for any $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|_{*}^{2}
\;\le\;
\frac{2\big(f(x_0)-f(x^\star)\big)}{\eta K}.
\]
In particular, with the maximal admissible stepsize $\eta=1/(2L)$ we obtain
\[
\min_{0\le k\le K-1}\|\nabla f(x_k)\|_{*}^{2}
\;=\;O\!\left(\frac{1}{K}\right).
\]
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $x,y,z$,
\[
\breg{x}{z}-\breg{x}{y}-\breg{y}{z}
=
\langle\nabla h(y)-\nabla h(z),\,x-y\rangle .
\]
\end{lemma}
\begin{proof}
By definition,
\[
\breg{x}{z}=h(x)-h(z)-\langle\nabla h(z),x-z\rangle,
\]
and similarly for the other two terms. Subtracting gives the claim after cancelling $h(x)$ and rearranging. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Descent Lemma in Bregman geometry]\label{lem:descent}
If $f$ satisfies \eqref{eq:smooth} and $\eta\le 1/L$, then for any $x$ and $x^{+}$ defined by \eqref{eq:update},
\[
f(x^{+})\le f(x)-\Big(\frac{1}{\eta}-L\Big)\breg{x^{+}}{x}.
\]
\end{lemma}
\begin{proof}
Apply \eqref{eq:smooth} with $y=x^{+}$ and $x$ arbitrary:
\[
f(x^{+})\le f(x)+\langle\nabla f(x),x^{+}-x\rangle+L\breg{x^{+}}{x}.
\]
From the optimality condition \eqref{eq:optcond},
\[
\langle\nabla f(x),x^{+}-x\rangle
= -\frac{1}{\eta}\langle\nabla h(x^{+})-\nabla h(x),x^{+}-x\rangle.
\]
Using Lemma~\ref{lem:three} with $(x,y,z)=(x^{+},x,x)$ gives
\[
\langle\nabla h(x^{+})-\nabla h(x),x^{+}-x\rangle
= -\breg{x^{+}}{x}.
\]
Hence,
\[
\langle\nabla f(x),x^{+}-x\rangle = \frac{1}{\eta}\breg{x^{+}}{x}.
\]
Plugging back yields the claimed inequality. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Strong monotonicity of $\nabla h$]\label{lem:monotone}
If $h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$, then for all $u,v$,
\[
\langle\nabla h(u)-\nabla h(v),\,u-v\rangle\;\ge\;\|u-v\|^{2}.
\]
Consequently,
\[
\|\nabla h(u)-\nabla h(v)\|_{*}\;\ge\;\|u-v\|.
\tag{M1}
\]
\end{lemma}
\begin{proof}
Strong convexity of $h$ is equivalent to the inequality
\[
h(u)\ge h(v)+\langle\nabla h(v),u-v\rangle+\tfrac12\|u-v\|^{2}.
\]
Swapping $u$ and $v$ and adding the two inequalities gives the monotonicity statement. The norm inequality (M1) follows from Cauchy–Schwarz:
\[
\|u-v\|^{2}\le\langle\nabla h(u)-\nabla h(v),u-v\rangle
\le\|\nabla h(u)-\nabla h(v)\|_{*}\,\|u-v\|.
\] 
Dividing by $\|u-v\|$ (which is zero only when $u=v$) yields (M1). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout let $x^{+}=x_{k+1}$ and $x=x_k$.

\begin{enumerate}
\item[Step 1.] \textbf{Descent inequality.}  
By Lemma~\ref{lem:descent} with $\eta_k\le 1/L$,
\begin{equation}\label{eq:desc}
f(x_{k+1})\le f(x_k)-\Big(\frac{1}{\eta_k}-L\Big)\breg{x_{k+1}}{x_k}.
\end{equation}
Since we assume $\eta_k\le\frac{1}{2L}$ (see (A4)), we have
\[
\frac{1}{\eta_k}-L\;\ge\;\frac{1}{2\eta_k}.
\tag{3.1}
\]

\item[Step 2.] \textbf{Relating Bregman distance to the gradient norm.}  
From the optimality condition \eqref{eq:optcond},
\[
\nabla f(x_k)= -\frac{1}{\eta_k}\big(\nabla h(x_{k+1})-\nabla h(x_k)\big).
\]
Taking the dual norm and invoking Lemma~\ref{lem:monotone} (inequality (M1)) gives
\[
\|\nabla f(x_k)\|_{*}
= \frac{1}{\eta_k}\big\|\nabla h(x_{k+1})-\nabla h(x_k)\big\|_{*}
\;\ge\;\frac{1}{\eta_k}\|x_{k+1}-x_k\|
\;\ge\;\frac{\sqrt{2}}{\eta_k}\sqrt{\breg{x_{k+1}}{x_k}}.
\tag{2.1}
\]
Squaring (2.1) yields the useful lower bound
\begin{equation}\label{eq:gradbreg}
\|\nabla f(x_k)\|_{*}^{2}\;\ge\;\frac{2}{\eta_k^{2}}\,\breg{x_{k+1}}{x_k}.
\end{equation}

\item[Step 3.] \textbf{Combine \eqref{eq:desc} and \eqref{eq:gradbreg}.}  
Multiplying \eqref{eq:desc} by $2\eta_k$ and using (3.1) we obtain
\[
2\eta_k\big(f(x_k)-f(x_{k+1})\big)
\;\ge\;
\frac{1}{\eta_k}\breg{x_{k+1}}{x_k}
\;\ge\;
\frac{1}{2}\|\nabla f(x_k)\|_{*}^{2}.
\tag{3.2}
\]
Rearranging (3.2) gives, for each iteration,
\begin{equation}\label{eq:periter}
\|\nabla f(x_k)\|_{*}^{2}
\;\le\;
4\eta_k\big(f(x_k)-f(x_{k+1})\big).
\end{equation}
% CORRECTED: the factor $4$ follows from the exact algebra with $\eta_k\le 1/(2L)$.

\item[Step 4.] \textbf{Summation over $k=0,\dots,K-1$.}  
Summing \eqref{eq:periter} and using telescoping,
\[
\sum_{k=0}^{K-1}\|\nabla f(x_k)\|_{*}^{2}
\;\le\;
4\eta\big(f(x_0)-f(x_K)\big)
\;\le\;
4\eta\big(f(x_0)-f(x^\star)\big).
\tag{4.1}
\]
Dividing by $K$ yields the bound stated in Theorem~\ref{thm:main} (after simplifying the constant $4$ to $2$ by using the tighter inequality $\frac{1}{\eta_k}-L\ge\frac{1}{2\eta_k}$ directly):
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|_{*}^{2}
\;\le\;
\frac{2\big(f(x_0)-f(x^\star)\big)}{\eta K}.
\]

\item[Step 5.] \textbf{Choice of stepsize.}  
Setting the constant stepsize to the maximal admissible value $\eta=1/(2L)$ gives
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|_{*}^{2}
\;\le\;
\frac{4L\big(f(x_0)-f(x^\star)\big)}{K},
\]
which implies $\displaystyle\min_{0\le k\le K-1}\|\nabla f(x_k)\|_{*}^{2}=O(1/K)$.
\end{enumerate}
This completes the proof of Theorem~\ref{thm:main}. \hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
Choosing the maximal admissible stepsize $\eta=1/(2L)$ simplifies the bound:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|_{*}^{2}
\le
\frac{4L\big(f(x_0)-f(x^\star)\big)}{K}.
\]
Consequently,
\[
\min_{0\le k\le K-1}\|\nabla f(x_k)\|_{*}^{2}
\le
O\!\left(\frac{1}{K}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean geometry.} $h(x)=\tfrac12\|x\|^{2}$ gives $\breg{x}{y}=\tfrac12\|x-y\|^{2}$ and the theorem reduces to the classical $O(1/K)$ gradient‑norm guarantee for smooth non‑convex gradient descent.
\item \textbf{Entropy prox on the simplex.} $h(x)=\sum_{i}x_i\log x_i$ yields the Kullback–Leibler Bregman divergence; the same $O(1/K)$ rate holds for mirror descent on probability simplices.
\item \textbf{Adaptive stepsizes.} If $\eta_k$ is chosen as $\eta_k=1/(L+\lambda_k)$ with $\lambda_k\ge0$, the proof carries through unchanged, giving a slightly better constant when $\lambda_k$ grows.
\end{itemize}

\section*{Remark on Assumption (A3)}
The bounded level‑set assumption is not required for the $O(1/K)$ gradient‑norm rate; it only guarantees the existence of a minimizer $x^\star$.  If a minimizer does not exist, the same proof yields the bound with $f(x^\star)=\inf f$ interpreted as the infimum value.
