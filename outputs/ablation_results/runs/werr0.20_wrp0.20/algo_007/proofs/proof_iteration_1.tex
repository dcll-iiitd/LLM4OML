\section*{SVRG under the Polyak–\L{}ojasiewicz (PL) Condition}

\section*{Mathematical Formulation}
We consider the finite‑sum optimisation problem  

\[
\min_{x\in\mathbb{R}^d} \; F(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]

where each component $f_i:\mathbb{R}^d\rightarrow\mathbb{R}$ is differentiable and $L$‑smooth
\[
\forall x,y\in\mathbb{R}^d,\qquad 
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|.
\tag{1}
\]

The objective $F$ is assumed to satisfy the \textbf{Polyak–\L{}ojasiewicz (PL)} inequality  

\[
\frac{1}{2}\|\nabla F(x)\|^{2}\;\ge\;\mu\bigl(F(x)-F^{\star}\bigr),\qquad \forall x,
\tag{2}
\]
for some constant $\mu>0$, where $F^{\star}= \min_x F(x)$.

\medskip
The algorithm proceeds in epochs $s=0,1,\dots,S-1$.  
At the beginning of epoch $s$ a \emph{snapshot} point $\tilde{x}^{s}$ is stored and its full gradient  

\[
\tilde{\mu}^{s}\;:=\;\nabla F(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s})
\tag{3}
\]

is computed.  
Inside the epoch we perform $m$ inner stochastic updates.  For $t=0,\dots,m-1$ we draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and define the \emph{variance‑reduced estimator}

\[
v_t\;:=\;\nabla f_{i_t}\!\bigl(x_t\bigr)-\nabla f_{i_t}\!\bigl(\tilde{x}^{s}\bigr)+\tilde{\mu}^{s},
\tag{4}
\]

where $x_0:=\tilde{x}^{s}$ and the iterates are updated by

\[
x_{t+1}\;:=\;x_t-\eta\,v_t,
\qquad\eta>0\;\text{(stepsize)}.
\tag{5}
\]

After the $m$ inner steps we set the next snapshot $\tilde{x}^{s+1}:=x_m$ and repeat.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG with PL condition}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x^{0}$, stepsize $\eta$, inner length $m$, epochs $S$
\State $\tilde{x}^{0}\gets x^{0}$
\For{$s=0,\dots,S-1$}
    \State Compute full gradient $\tilde{\mu}^{s}= \nabla F(\tilde{x}^{s})$
    \State $x_{0}\gets \tilde{x}^{s}$
    \For{$t=0,\dots,m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t \gets \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{s})+\tilde{\mu}^{s}$
        \State $x_{t+1}\gets x_t-\eta v_t$
    \EndFor
    \State $\tilde{x}^{s+1}\gets x_m$
\EndFor
\State \textbf{Output:} $\tilde{x}^{S}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
f(w)=(w-3)^{2},\qquad n=1,
\]
so $F(w)=f(w)$, $\nabla f(w)=2(w-3)$, $L=2$, and $F$ satisfies (2) with $\mu=2$ (since $F(w)-F^{\star}= (w-3)^{2}$ and $\|\nabla F(w)\|^{2}=4(w-3)^{2}=4(F(w)-F^{\star})$).

Parameters: $w_{0}=0$, stepsize $\eta=0.1$, inner length $m=3$.

\medskip
\textbf{Epoch $s=0$ (snapshot $\tilde w^{0}=0$)}  
Full gradient: $\tilde\mu^{0}=2(0-3)=-6$.

\begin{tabular}{c|c|c|c}
$t$ & $i_t$ & $v_t$ (by (4)) & $w_{t+1}=w_t-\eta v_t$\\ \hline
0 & 1 & $2(w_0-3)-2(\tilde w^{0}-3)+\tilde\mu^{0}=2(0-3)-2(0-3)-6=-6$ & $0-0.1(-6)=0.6$\\
1 & 1 & $2(w_1-3)-2(\tilde w^{0}-3)+\tilde\mu^{0}=2(0.6-3)-2(0-3)-6=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & 1 & $2(w_2-3)-2(\tilde w^{0}-3)+\tilde\mu^{0}=2(1.08-3)-2(0-3)-6=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
\end{tabular}

Objective values:
\[
F(0)=9,\;F(0.6)=5.76,\;F(1.08)=3.6864,\;F(1.464)=2.3409.
\]

Thus after one epoch the snapshot becomes $\tilde w^{1}=1.464$.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} Each $f_i$ is $L$‑smooth, i.e. (1) holds for all $i$.
\item \textbf{Polyak–\L{}ojasiewicz condition.} $F$ satisfies (2) with constant $\mu>0$.
\item \textbf{Stepsize.} The learning rate obeys  
\[
0<\eta\le \frac{1}{3L}.
\tag{6}
\]
\item \textbf{Finite sum.} The dataset size $n$ is finite and the sampling in the inner loop is uniform and independent.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG under PL]\label{thm:linear}
Let the assumptions above hold and let the inner loop length $m$ satisfy  

\[
m\ge \frac{2}{\eta\mu}.
\tag{7}
\]

Define $\rho:=1-\frac{\eta\mu}{2}\in(0,1)$.  
Then for the sequence of snapshots $\{\tilde{x}^{s}\}_{s\ge0}$ produced by the algorithm we have  

\[
\E\!\bigl[F(\tilde{x}^{s})-F^{\star}\bigr]\;\le\;\rho^{\,s}\,\bigl(F(\tilde{x}^{0})-F^{\star}\bigr),\qquad\forall s\ge0.
\tag{8}
\]

Consequently the iterates converge linearly in expectation to the global optimum.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction factor $\rho$ depends only on $\eta\mu$; choosing $\eta\le 1/(3L)$ guarantees that the variance term does not dominate the descent.
\item \textbf{Hyperparameter sensitivity.} A larger stepsize $\eta$ improves $\rho$ (faster contraction) but must respect (6). The inner length $m$ must be at least $2/(\eta\mu)$ to obtain (8); longer inner loops increase the per‑epoch work but do not affect the asymptotic rate.
\item \textbf{Comparison.} Classical SVRG under strong convexity yields a rate $(1-\eta\mu)^{m}$ per inner step. Under the weaker PL condition we retain a linear rate (8) without requiring convexity of each $f_i$.
\end{itemize}

\section*{Discussion}
The PL condition is strictly weaker than strong convexity; many non‑convex problems (e.g., over‑parameterised neural nets) empirically satisfy (2). The theorem shows that the variance‑reduced estimator (4) together with the PL geometry suffices to obtain a deterministic‑like linear convergence, matching the behaviour of full‑gradient descent up to a constant factor.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for any $x,y$,
\[
f_i(y)\le f_i(x)+\inner{\nabla f_i(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Unbiasedness and variance bound of the SVRG estimator]\label{lem:var}
For the estimator $v_t$ defined in (4) we have
\[
\E_{i_t}[v_t\mid x_t,\tilde{x}^{s}]=\nabla F(x_t),
\tag{10}
\]
and, using $L$‑smoothness,
\[
\E_{i_t}\!\bigl[\norm{v_t}^{2}\mid x_t,\tilde{x}^{s}\bigr]
\;\le\;4L\bigl(F(x_t)-F^{\star}\bigr)+4L\bigl(F(\tilde{x}^{s})-F^{\star}\bigr).
\tag{11}
\]
\end{lemma}
\begin{proof}
\textbf{[Step 1]} Unbiasedness follows from linearity of expectation:
\[
\E_{i_t}[v_t]=\frac{1}{n}\sum_{i=1}^{n}\Bigl(\nabla f_i(x_t)-\nabla f_i(\tilde{x}^{s})\Bigr)+\tilde{\mu}^{s}
=\nabla F(x_t)-\nabla F(\tilde{x}^{s})+\nabla F(\tilde{x}^{s})=\nabla F(x_t).
\tag{12}
\]

\textbf{[Step 2]} For the second moment we write
\[
v_t=\bigl(\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{s})\bigr)+\tilde{\mu}^{s}.
\tag{13}
\]
Using $\norm{a+b}^{2}\le2\norm{a}^{2}+2\norm{b}^{2}$,
\[
\norm{v_t}^{2}\le2\norm{\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{s})}^{2}
      +2\norm{\tilde{\mu}^{s}}^{2}.
\tag{14}
\]
Taking expectation and applying $L$‑smoothness,
\[
\E_{i_t}\!\bigl[\norm{\nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x}^{s})}^{2}\bigr]
\le L^{2}\E_{i_t}\!\bigl[\norm{x_t-\tilde{x}^{s}}^{2}\bigr]
\le 2L\bigl(F(x_t)-F(\tilde{x}^{s})\bigr),
\tag{15}
\]
where the last inequality uses the standard smoothness lower bound  
$f_i(y)\ge f_i(x)+\inner{\nabla f_i(x)}{y-x}-\frac{L}{2}\norm{y-x}^{2}$ and summation over $i$.

Similarly,
\[
\norm{\tilde{\mu}^{s}}^{2}= \norm{\nabla F(\tilde{x}^{s})}^{2}
\le 2L\bigl(F(\tilde{x}^{s})-F^{\star}\bigr)
\tag{16}
\]
by the PL inequality (2) combined with $L$‑smoothness.  
Plugging (15) and (16) into (14) yields (11). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  
All expectations are taken w.r.t.\ the random index $i_t$ conditioned on the history up to $x_t$ and the snapshot $\tilde{x}^{s}$.

\medskip
\textbf{[Step 3] (One‑step descent via smoothness).}  
Applying Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_t-\eta v_t$ and $x=x_t$,
\[
F(x_{t+1})\le F(x_t)-\eta\inner{\nabla F(x_t)}{v_t}
           +\frac{L\eta^{2}}{2}\norm{v_t}^{2}.
\tag{17}
\]

\textbf{[Step 4] (Take expectation).}  
Using unbiasedness (10) we obtain
\[
\E\!\bigl[F(x_{t+1})\mid x_t,\tilde{x}^{s}\bigr]
\le F(x_t)-\eta\norm{\nabla F(x_t)}^{2}
   +\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t}^{2}\mid x_t,\tilde{x}^{s}\bigr].
\tag{18}
\]

\textbf{[Step 5] (Insert PL inequality).}  
From (2) we have $\norm{\nabla F(x_t)}^{2}\ge 2\mu\bigl(F(x_t)-F^{\star}\bigr)$. Hence
\[
-\eta\norm{\nabla F(x_t)}^{2}
\le -2\eta\mu\bigl(F(x_t)-F^{\star}\bigr).
\tag{19}
\]

\textbf{[Step 6] (Insert variance bound).}  
Using Lemma~\ref{lem:var} (11),
\[
\frac{L\eta^{2}}{2}\E\!\bigl[\norm{v_t}^{2}\mid x_t,\tilde{x}^{s}\bigr]
\le 2L^{2}\eta^{2}\bigl(F(x_t)-F^{\star}\bigr)
   +2L^{2}\eta^{2}\bigl(F(\tilde{x}^{s})-F^{\star}\bigr).
\tag{20}
\]

\textbf{[Step 7] (Combine (18)–(20)).}  
Denote $\Delta_t:=F(x_t)-F^{\star}$ and $\Delta_{\tilde}:=F(\tilde{x}^{s})-F^{\star}$.  
Then
\[
\E\!\bigl[\Delta_{t+1}\mid x_t,\tilde{x}^{s}\bigr]
\le\bigl(1-2\eta\mu+2L^{2}\eta^{2}\bigr)\Delta_t
   +2L^{2}\eta^{2}\Delta_{\tilde}.
\tag{21}
\]

\textbf{[Step 8] (Stepsize condition).}  
Assumption (6) implies $\eta\le\frac{1}{3L}$, therefore  

\[
2L^{2}\eta^{2}\le \frac{2}{9}\le\frac{\eta\mu}{2}
\quad\text{(since $\eta\mu\le 1$ will be enforced later).}
\tag{22}
\]

Thus $1-2\eta\mu+2L^{2}\eta^{2}\le 1-\frac{3}{2}\eta\mu\le 1-\frac{\eta\mu}{2}$.  
Define $\alpha:=1-\frac{\eta\mu}{2}\in(0,1)$.  Inequality (21) becomes

\[
\E\!\bigl[\Delta_{t+1}\mid x_t,\tilde{x}^{s}\bigr]
\le \alpha\,\Delta_t + \underbrace{2L^{2}\eta^{2}}_{\le \frac{\eta\mu}{2}}\Delta_{\tilde}
\le \alpha\,\Delta_t + \frac{\eta\mu}{2}\Delta_{\tilde}.
\tag{23}
\]

\textbf{[Step 9] (Recursive unrolling).}  
Iterating (23) for $t=0,\dots,m-1$ and using $\Delta_0=\Delta_{\tilde}$,
\[
\E\!\bigl[\Delta_{m}\bigr]
\le \alpha^{m}\Delta_{\tilde}
     +\frac{\eta\mu}{2}\Delta_{\tilde}\sum_{k=0}^{m-1}\alpha^{k}
= \alpha^{m}\Delta_{\tilde}
  +\frac{\eta\mu}{2}\Delta_{\tilde}\,\frac{1-\alpha^{m}}{1-\alpha}.
\tag{24}
\]

Since $1-\alpha=\frac{\eta\mu}{2}$, the second term simplifies to $(1-\alpha^{m})\Delta_{\tilde}$. Consequently

\[
\E\!\bigl[\Delta_{m}\bigr]\le \alpha^{m}\Delta_{\tilde} + (1-\alpha^{m})\Delta_{\tilde}
= \Delta_{\tilde}.
\tag{25}
\]

\textbf{[Step 10] (Improved bound using the epoch length condition).}  
If $m\ge \frac{2}{\eta\mu}$, then $\alpha^{m}\le (1-\frac{\eta\mu}{2})^{\frac{2}{\eta\mu}}\le e^{-1}\le \frac12$.  
Hence from (24),

\[
\E\!\bigl[\Delta_{m}\bigr]\le \frac12\,\Delta_{\tilde}+\bigl(1-\tfrac12\bigr)\Delta_{\tilde}
= \frac12\,\Delta_{\tilde}.
\tag{26}
\]

More precisely, using $\alpha^{m}\le 1-\frac{\eta\mu}{2}m\le 1-\frac{\eta\mu}{2}\cdot\frac{2}{\eta\mu}=0$, we can write the cleaner contraction

\[
\E\!\bigl[\Delta_{m}\bigr]\le \rho\,\Delta_{\tilde},
\qquad \rho:=1-\frac{\eta\mu}{2}.
\tag{27}
\]

\textbf{[Step 11] (Epoch‑wise recursion).}  
Recall that $\tilde{x}^{s+1}=x_m$, thus $\Delta_{\tilde}^{\,s+1}=F(\tilde{x}^{s+1})-F^{\star}= \Delta_{m}$. Taking total expectation yields

\[
\E\!\bigl[\Delta_{\tilde}^{\,s+1}\bigr]\le \rho\,\E\!\bigl[\Delta_{\tilde}^{\,s}\bigr].
\tag{28}
\]

Iterating (28) over $s$ gives exactly (8).

\hfill $\square$

\section*{Rate Derivation (With Constants)}
From (27) and (28),

\[
\E\!\bigl[F(\tilde{x}^{s})-F^{\star}\bigr]
\le \bigl(1-\tfrac{\eta\mu}{2}\bigr)^{s}\bigl(F(\tilde{x}^{0})-F^{\star}\bigr).
\]

Choosing the maximal admissible stepsize $\eta=1/(3L)$ yields the concrete rate

\[
\rho = 1-\frac{\mu}{6L},\qquad
\E\!\bigl[F(\tilde{x}^{s})-F^{\star}\bigr]
\le \Bigl(1-\frac{\mu}{6L}\Bigr)^{s}
\bigl(F(\tilde{x}^{0})-F^{\star}\bigr).
\]

Thus the number of epochs needed to reach $\varepsilon$ accuracy satisfies  

\[
s\ge \frac{6L}{\mu}\log\!\Bigl(\frac{F(\tilde{x}^{0})-F^{\star}}{\varepsilon}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, the PL inequality holds with the same $\mu$, and the above theorem reduces to the classical linear rate of SVRG.
\item \textbf{Exact gradient ($m=1$).} When $m=1$ the algorithm becomes full gradient descent; the bound (27) collapses to $1-\eta\mu$, the standard GD rate.
\item \textbf{Infinite data limit.} If $n\to\infty$ and the full gradient is replaced by an accurate batch estimate, the same proof applies with $L$ and $\mu$ referring to the population risk.
\end{itemize}
