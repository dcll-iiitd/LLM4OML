\section*{Algorithm Name}
Top-$K$ Error-Feedback Stochastic Gradient Descent (Top-$K$ EF-SGD)

\section*{Mathematical Formulation}
We aim to minimize a possibly non-convex, $L$-smooth objective
\[
\min_{w \in \R^d} f(w) = \E_{z \sim \mathcal{D}}[F(w;z)].
\]
At iteration $t$, we draw a sample $z_t \sim \mathcal{D}$, compute a stochastic gradient $g_t = \nabla F(w_t; z_t)$, and apply a sparse compressor $\mathcal{C}$ that keeps only the $K$ largest-magnitude coordinates (Top-$K$) together with error-feedback.

- Error-feedback state $e_t \in \R^d$ accumulates past compression errors.
- Step-size (learning rate) $\eta_t > 0$.

Update equations:
\[
\begin{aligned}
u_t &= e_t + \eta_t g_t,\\
s_t &= \mathcal{C}(u_t) \quad \text{(Top-$K$)},\\
w_{t+1} &= w_t - s_t,\\
e_{t+1} &= u_t - s_t.
\end{aligned}
\]
Equivalently, $s_t = \eta_t g_t + e_t - e_{t+1}$ and
\[
w_{t+1} = w_t - \eta_t g_t - e_t + e_{t+1}.
\]

Top-$K$ compressor:
- For $x \in \R^d$, $\mathrm{TopK}(x)$ keeps the $K$ coordinates of $x$ with largest absolute values and sets others to zero.

Contractive property of Top-$K$ (deterministic):
\[
\|x - \mathrm{TopK}(x)\|^2 \le (1-\tau)\|x\|^2,\quad \tau := \frac{K}{d} \in (0,1]. 
\]

\section*{Pseudocode}
\noindent
Inputs: initial $w_1 \in \R^d$, horizon $T$, step sizes $\{\eta_t\}_{t=1}^T$, sparsity $K$.

\noindent
Initialize: $e_1 \gets 0 \in \R^d$.

\noindent
For $t = 1,2,\dots,T$:
\begin{enumerate}
\item Sample data point $z_t \sim \mathcal{D}$ and compute $g_t \gets \nabla F(w_t; z_t)$.
\item Form accumulator $u_t \gets e_t + \eta_t g_t$.
\item Compress $s_t \gets \mathrm{TopK}(u_t)$ (keep $K$ largest $|u_t|$ entries).
\item Update parameters $w_{t+1} \gets w_t - s_t$.
\item Update error $e_{t+1} \gets u_t - s_t$.
\end{enumerate}
Output: (optionally) a random iterate $w_R$ with $R$ uniform in $\{1,\dots,T\}$, or the average $\bar{w}_T = \frac{1}{T}\sum_{t=1}^T w_t$.

g_2 = 2(0.6-3) = -4.8,\quad u_2 = 0 + 0.1(-4.8) = -0.48,\quad s_2 = -0.48,\quad
w_3 = 0.6 - (-0.48) = 1.08,\quad e_3 = 0.
\]
Objective: $f(w_3) = (1.08-3)^2 = 3.6864$.

\section*{Assumptions}
Let $\{\mathcal{F}_t\}$ be the filtration generated by the algorithmic randomness up to time $t$ (including $w_t, e_t$).
\begin{itemize}
\item[A1] Smoothness: $f$ is $L$-smooth: for all $x,y$, $f(y) \le f(x) + \langle \nabla f(x), y-x\rangle + \frac{L}{2}\|y-x\|^2$.
\item[A2] Bounded below: $f^* := \inf_w f(w) > -\infty$ and $\Delta_f := f(w_1)-f^* < \infty$.
\item[A3] Unbiased stochastic gradients: $\E[g_t \mid \mathcal{F}_t] = \nabla f(w_t)$.
\item[A4] Variance bound: $\E[\|g_t - \nabla f(w_t)\|^2 \mid \mathcal{F}_t] \le \sigma^2$. Consequently,
\[
\E[\|g_t\|^2 \mid \mathcal{F}_t] \le \|\nabla f(w_t)\|^2 + \sigma^2.
\]
\item[A5] Compressor contractivity (Top-$K$): for all $x \in \R^d$,
\[
\|x - \mathcal{C}(x)\|^2 \le (1-\tau)\|x\|^2,\quad \tau = K/d \in (0,1].
\]
\end{itemize}

\section*{Main Convergence Theorem}
% CORRECTED: Updated step-size condition and constants to resolve [Step 25] and [Step 28] issues.
Let Assumptions A1--A5 hold and let $\eta_t \equiv \eta$ satisfy
\[
0 < \eta \le \min\left\{ \frac{1}{6L},\ \frac{1}{2L\bigl(3 + 15\, c_0(\tau)/\tau\bigr)}\right\},\quad
c_0(\tau) := (1-\tau)\left(1 + \frac{2}{\tau}\right).
\]
Then the iterates of Top-$K$ EF-SGD satisfy
\[
\frac{1}{T}\sum_{t=1}^T \E\bigl[\|\nabla f(w_t)\|^2\bigr]
\ \le\ 
\frac{2\Delta_f}{\eta T}
\ +\ 2L\eta \left(3 + \frac{15\,c_0(\tau)}{\tau}\right)\sigma^2
\ +\ \frac{1}{2LT}\,\E\bigl[\|\nabla f(w_{T+1})\|^2\bigr].
\]
In particular, choosing $\eta = \Theta(1/\sqrt{T})$ obeying the step-size condition gives
\[
\frac{1}{T}\sum_{t=1}^T \E\bigl[\|\nabla f(w_t)\|^2\bigr]
= O\!\left(\frac{1}{\sqrt{T}}\left[\Delta_f L + \left(3 + \frac{15\,c_0(\tau)}{\tau}\right)\sigma^2\right]\right).
\]
Thus Top-$K$ EF-SGD attains the standard nonconvex $O(1/\sqrt{T})$ rate with explicit dependence on the sparsity via $\tau = K/d$.

\section*{Theoretical Properties}
- Stability and error damping: Error-feedback state satisfies a contraction-type recursion with factor $1-\tau/2$, ensuring bounded accumulation even for biased compressors.
- Sensitivity to $K$ (via $\tau$): Constants scale as $1/\tau$ (and $c_0(\tau)= (1-\tau)\left(1+\frac{2}{\tau}\right) = \frac{2}{\tau} - 1 - \tau \le \frac{2}{\tau}$), capturing the intuition that more aggressive sparsification (smaller $K$) slows convergence through larger constants.
- Comparison to SGD: With $K=d$ ($\tau=1$), EF-SGD reduces to standard SGD and the bound matches classical results up to constant factors.
- Variance-free case: If $\sigma^2=0$, the rate simplifies to $\frac{1}{T}\sum_{t=1}^T \E\|\nabla f(w_t)\|^2 \le \frac{2\Delta_f}{\eta T}$ (up to a single-iterate boundary term) with the same step-size condition.

\section*{Preliminaries (Definitions and Lemmas)}
We compile properties used in the proof.

\noindent
Definition (Contractive compressor):
A mapping $\mathcal{C}:\R^d \to \R^d$ is $\tau$-contractive if
\[
\|x - \mathcal{C}(x)\|^2 \le (1-\tau)\|x\|^2\quad \forall x \in \R^d.
\]
Top-$K$ satisfies this with $\tau = K/d$.

\noindent
Lemma 1 (Top-$K$ contractivity).
For $\mathcal{C}(x) = \mathrm{TopK}(x)$, for any $x \in \R^d$,
\[
\|x - \mathcal{C}(x)\|^2 \le (1-\tau)\|x\|^2,\quad \tau = K/d.
\]
Proof sketch: the $K$ largest-magnitude coordinates of $x$ contain at least a fraction $K/d$ of the squared $\ell_2$ norm (worst case is equal magnitudes), hence the residual energy is at most $(1-\tau)$ times the total.

\noindent
Lemma 2 (Error recursion).
Let $u_t = e_t + \eta g_t$, $e_{t+1} = u_t - \mathcal{C}(u_t)$ and assume $\mathcal{C}$ is $\tau$-contractive. Then, for any $\alpha>0$,
\[
\|e_{t+1}\|^2 \le (1-\tau)\|u_t\|^2 \le (1-\tau)\Bigl((1+\alpha)\|e_t\|^2 + \Bigl(1+\tfrac{1}{\alpha}\Bigr)\eta^2\|g_t\|^2\Bigr).
\]
In particular, with $\alpha = \tau/2$,
\[
\|e_{t+1}\|^2 \le \Bigl(1-\frac{\tau}{2}\Bigr)\|e_t\|^2 + c_0(\tau)\,\eta^2 \|g_t\|^2,
\]
where $c_0(\tau) := (1-\tau)\Bigl(1 + \frac{2}{\tau}\Bigr)$.

\noindent
Lemma 3 (Summation bound for error state).
Let $x_{t+1} \le (1-\rho)x_t + a_t$ with $x_1=0$, $\rho \in (0,1]$, $a_t \ge 0$. Then
\[
\sum_{t=1}^{T+1} x_t \le \frac{1}{\rho}\sum_{t=1}^{T} a_t.
\]
Applying with $x_t=\E\|e_t\|^2$, $\rho=\tau/2$, and $a_t = c_0(\tau)\,\eta^2\,\E\|g_t\|^2$, we obtain
\[
\sum_{t=1}^{T+1} \E\|e_t\|^2 \le \frac{2 c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^{T} \E\|g_t\|^2.
\]

\section*{Main Proof (Step-by-Step Derivation)}
We prove the main theorem with full detail.

Notation: $d_t := s_t = \mathcal{C}(e_t + \eta g_t)$ and recall $d_t = \eta g_t + e_t - e_{t+1}$.

Start from $L$-smoothness [Step 1]:
\[
f(w_{t+1}) \le f(w_t) + \langle \nabla f(w_t), w_{t+1}-w_t\rangle + \frac{L}{2}\|w_{t+1}-w_t\|^2.
\]
Substitute $w_{t+1}-w_t = -d_t$ [Step 2]:
\begin{equation}\label{eq:smooth1}
f(w_{t+1}) \le f(w_t) - \langle \nabla f(w_t), d_t\rangle + \frac{L}{2}\|d_t\|^2.
\end{equation}
Express $d_t = \eta g_t + e_t - e_{t+1}$ [Step 3] to obtain
\begin{align}
- \langle \nabla f(w_t), d_t\rangle
&= -\eta \langle \nabla f(w_t), g_t\rangle - \langle \nabla f(w_t), e_t - e_{t+1}\rangle. \label{eq:decomp}
\end{align}
We now sum \eqref{eq:smooth1} over $t=1$ to $T$ and take total expectation. Using \eqref{eq:decomp}, we get
\begin{align}
\E[f(w_{T+1})]
&\le \E[f(w_1)] - \sum_{t=1}^T \eta\, \E\langle \nabla f(w_t), g_t\rangle - \sum_{t=1}^T \E\langle \nabla f(w_t), e_t - e_{t+1}\rangle + \frac{L}{2}\sum_{t=1}^T \E\|d_t\|^2. \label{eq:master1}
\end{align}
Handle the stochastic gradient term via unbiasedness [Step 4]:
\[
\E\bigl[\langle \nabla f(w_t), g_t\rangle \mid \mathcal{F}_t\bigr] = \langle \nabla f(w_t), \E[g_t \mid \mathcal{F}_t]\rangle = \|\nabla f(w_t)\|^2.
\]
Thus
\begin{equation}\label{eq:sgd-term}
\sum_{t=1}^T \eta\, \E\langle \nabla f(w_t), g_t\rangle = \eta \sum_{t=1}^T \E\|\nabla f(w_t)\|^2.
\end{equation}
For the error cross-term, write a telescoping decomposition [Step 5]:
\[
\langle \nabla f(w_t), e_t - e_{t+1}\rangle
= \langle \nabla f(w_t), e_t\rangle - \langle \nabla f(w_t), e_{t+1}\rangle.
\]
Add and subtract $\nabla f(w_{t+1})$ in the second inner product [Step 6]:
\begin{align*}
- \langle \nabla f(w_t), e_{t+1}\rangle
&= - \langle \nabla f(w_{t+1}), e_{t+1}\rangle - \langle \nabla f(w_t) - \nabla f(w_{t+1}), e_{t+1}\rangle.
\end{align*}
Summing over $t$ and telescoping [Step 7]:
\begin{align}
\sum_{t=1}^T \langle \nabla f(w_t), e_t - e_{t+1}\rangle
&= \underbrace{\langle \nabla f(w_1), e_1\rangle}_{=0} - \langle \nabla f(w_{T+1}), e_{T+1}\rangle - \sum_{t=1}^T \langle \nabla f(w_t) - \nabla f(w_{t+1}), e_{t+1}\rangle. \label{eq:telescoping}
\end{align}
Use Lipschitz gradients to bound the correction term [Step 8]:
\[
\|\nabla f(w_t) - \nabla f(w_{t+1})\| \le L\|w_t - w_{t+1}\| = L\|d_t\|.
\]
Apply Cauchy-Schwarz and $2ab \le a^2+b^2$ [Step 9]:
\begin{align}
\bigl|\langle \nabla f(w_t) - \nabla f(w_{t+1}), e_{t+1}\rangle\bigr|
&\le \|\nabla f(w_t) - \nabla f(w_{t+1})\|\, \|e_{t+1}\|
\le L\|d_t\|\, \|e_{t+1}\| \nonumber \\
&\le \frac{L}{2}\|d_t\|^2 + \frac{L}{2}\|e_{t+1}\|^2. \label{eq:correction-bound}
\end{align}
Substitute \eqref{eq:sgd-term} and \eqref{eq:telescoping}--\eqref{eq:correction-bound} into \eqref{eq:master1} [Step 10]:
\begin{align}
\E[f(w_{T+1})]
&\le \E[f(w_1)] - \eta \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 
+ \E\langle \nabla f(w_{T+1}), e_{T+1}\rangle
+ L \sum_{t=1}^T \E\|d_t\|^2
+ \frac{L}{2}\sum_{t=1}^T \E\|e_{t+1}\|^2.
\label{eq:master2}
\end{align}
We next bound $\|d_t\|^2$ using $(a+b+c)^2 \le 3(\|a\|^2+\|b\|^2+\|c\|^2)$ [Step 11]:
\begin{align}
\|d_t\|^2 
= \|\eta g_t + e_t - e_{t+1}\|^2
\le 3\eta^2 \|g_t\|^2 + 3\|e_t\|^2 + 3\|e_{t+1}\|^2. \label{eq:dt-bound}
\end{align}
Combining \eqref{eq:master2} and \eqref{eq:dt-bound} [Step 12]:
\begin{align}
\E[f(w_{T+1})]
&\le \E[f(w_1)] - \eta \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 
+ \E\langle \nabla f(w_{T+1}), e_{T+1}\rangle \nonumber\\
&\quad + L \sum_{t=1}^T \E\bigl[3\eta^2 \|g_t\|^2 + 3\|e_t\|^2 + 3\|e_{t+1}\|^2\bigr]
+ \frac{L}{2}\sum_{t=1}^T \E\|e_{t+1}\|^2. \label{eq:master3}
\end{align}
Group like terms [Step 13]:
\begin{align}
\E[f(w_{T+1})]
&\le \E[f(w_1)] - \eta \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 
+ \E\langle \nabla f(w_{T+1}), e_{T+1}\rangle \nonumber\\
&\quad + 3L\eta^2 \sum_{t=1}^T \E\|g_t\|^2 
+ 3L \sum_{t=1}^T \E\|e_t\|^2 
+ \frac{7L}{2}\sum_{t=1}^T \E\|e_{t+1}\|^2. \label{eq:master4}
\end{align}
Invoke the second-moment bound from A4: $\E[\|g_t\|^2] \le \E\|\nabla f(w_t)\|^2 + \sigma^2$ [Step 14], which yields
\begin{align}
3L\eta^2 \sum_{t=1}^T \E\|g_t\|^2 
&\le 3L\eta^2 \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 + 3L\eta^2 T \sigma^2. \label{eq:g2-split}
\end{align}
Next, control the error sums via Lemma 3 [Step 15]. From Lemma 3 with $\rho=\tau/2$:
\begin{align}
\sum_{t=1}^{T+1} \E\|e_t\|^2 
&\le \frac{2 c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^{T} \E\|g_t\|^2 \nonumber\\
&\le \frac{2 c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^{T}\Bigl(\E\|\nabla f(w_t)\|^2 + \sigma^2\Bigr). \label{eq:e-sum}
\end{align}
Use \eqref{eq:e-sum} to bound the two error sums in \eqref{eq:master4} [Step 16]. Note that
\[
\sum_{t=1}^{T} \E\|e_t\|^2 \le \sum_{t=1}^{T+1} \E\|e_t\|^2,\quad
\sum_{t=1}^{T} \E\|e_{t+1}\|^2 = \sum_{t=2}^{T+1} \E\|e_t\|^2 \le \sum_{t=1}^{T+1} \E\|e_t\|^2.
\]
Hence
\begin{align}
3L \sum_{t=1}^{T} \E\|e_t\|^2 + \frac{7L}{2}\sum_{t=1}^{T} \E\|e_{t+1}\|^2 
&\le \frac{13L}{2} \sum_{t=1}^{T+1} \E\|e_t\|^2 \nonumber\\
&\le \frac{13L c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^T \Bigl(\E\|\nabla f(w_t)\|^2 + \sigma^2\Bigr) \nonumber\\
&= \frac{13L c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 + \frac{13L c_0(\tau)}{\tau}\,\eta^2 T \sigma^2. \label{eq:error-sum-final}
\end{align}
Substitute \eqref{eq:g2-split} and \eqref{eq:error-sum-final} into \eqref{eq:master4} [Step 17]:
\begin{align}
\E[f(w_{T+1})]
&\le \E[f(w_1)] 
- \eta \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 
+ \E\langle \nabla f(w_{T+1}), e_{T+1}\rangle \nonumber\\
&\quad + 3L\eta^2 \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 + 3L\eta^2 T \sigma^2 \nonumber\\
&\quad + \frac{13L c_0(\tau)}{\tau}\,\eta^2 \sum_{t=1}^T \E\|\nabla f(w_t)\|^2 + \frac{13L c_0(\tau)}{\tau}\,\eta^2 T \sigma^2. \label{eq:master5}
\end{align}
Rearrange gradient-norm terms to the left [Step 18]:
\begin{align}
\Bigl(\eta - 3L\eta^2 - \frac{13L c_0(\tau)}{\tau}\eta^2\Bigr) \sum_{t=1}^T \E\|\nabla f(w_t)\|^2
&\le \E[f(w_1)] - \E[f(w_{T+1})] + \E\langle \nabla f(w_{T+1}), e_{T+1}\rangle \nonumber\\
&\quad + \Bigl(3L + \frac{13L c_0(\tau)}{\tau}\Bigr)\eta^2 T \sigma^2. \label{eq:pre-final}
\end{align}
Since $f(w_{T+1}) \ge f^*$ [Step 19], we have
\[
\E[f(w_1)] - \E[f(w_{T+1})] \le \Delta_f.
\]

\section*{Rate Derivation (With Constants)}
Set $\eta = \min\left\{\eta_0,\ \frac{c}{\sqrt{T}}\right\}$ with
\[
\eta_0 := \min\left\{ \frac{1}{6L},\ \frac{1}{2L\bigl(3 + 15 c_0(\tau)/\tau\bigr)}\right\},\quad c>0.
\]
Then from \eqref{eq:avg-grad-corrected}:
\[
\frac{1}{T}\sum_{t=1}^T \E\|\nabla f(w_t)\|^2
\le \frac{2\Delta_f}{\eta T}
+ \frac{1}{2LT}\E\|\nabla f(w_{T+1})\|^2
+ 2L\eta \left(3 + \frac{15 c_0(\tau)}{\tau}\right)\sigma^2.
\]
If $\eta = c/\sqrt{T}$ (and $T$ large so that $\eta\le \eta_0$), we obtain
\[
\frac{1}{T}\sum_{t=1}^T \E\|\nabla f(w_t)\|^2
\le \frac{2\Delta_f}{c}\cdot \frac{1}{\sqrt{T}}
+ O\!\left(\frac{1}{T}\right)
+ \left[2Lc\left(3 + \frac{15 c_0(\tau)}{\tau}\right)\right]\frac{\sigma^2}{\sqrt{T}}
= O\!\left(\frac{1}{\sqrt{T}}\right),
\]
with explicit constants depending on $L,\sigma^2,\Delta_f,\tau$.

\section*{Special Cases}
- No compression ($K=d$, $\tau=1$): $c_0(1)=0$. The bound reduces to
\[
\frac{1}{T}\sum_{t=1}^T \E\|\nabla f(w_t)\|^2
\le \frac{2\Delta_f}{\eta T} + 6L\eta\,\sigma^2 + \frac{1}{2LT}\E\|\nabla f(w_{T+1})\|^2,
\]
matching the classical SGD rate (up to constants).
- Variance-free gradients ($\sigma^2=0$): The bound simplifies to
\[
\frac{1}{T}\sum_{t=1}^T \E\|\nabla f(w_t)\|^2
\le \frac{2\Delta_f}{\eta T} + \frac{1}{2LT}\E\|\nabla f(w_{T+1})\|^2,
\]
so with $\eta=\eta_0$ fixed we get $O(1/T)$ decay in the average squared gradient norm up to a single-iterate boundary term.
- Extreme sparsity (small $\tau$): Constants scale like $1/\tau$ through $c_0(\tau)$ and $A(\tau):=3 + 15 c_0(\tau)/\tau$, reflecting the increased error due to aggressive sparsification. Error-feedback ensures stability via the $(1-\tau/2)$ contraction in Lemma 2.
