\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be a (possibly non‑convex) loss function.  
At iteration $t$ we draw a random mini‑batch $\xi_t$ and compute a stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t)\in\mathbb{R}^d .
\]
A (possibly random) quantization operator $Q:\mathbb{R}^d\rightarrow\mathbb{R}^d$ satisfies  

\[
\mathbb{E}\!\left[\,Q(g_t)\mid g_t\right]=g_t ,\qquad 
\mathbb{E}\!\left[\|Q(g_t)\|^2\mid g_t\right]\le (1+\delta)\|g_t\|^2,
\tag{1}
\]
where $\delta\ge0$ quantifies the additional variance introduced by quantization.  

With a stepsize (learning rate) $\eta_t>0$, the QSGD update is
\[
\boxed{\,w_{t+1}=w_t-\eta_t\,Q(g_t)\,}. \tag{2}
\]

Hyper‑parameters:
\begin{itemize}
    \item Stepsize schedule $\{\eta_t\}_{t\ge0}$ (e.g. constant $\eta_t\equiv\eta$ or $\eta_t=\frac{c}{\sqrt{t+1}}$).
    \item Quantization level (controls $\delta$); we treat $\delta$ as a given constant.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, stepsize schedule $\{\eta_t\}$, quantizer $Q$.
\FOR{$t=0,1,\dots,T-1$}
    \STATE Sample mini‑batch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \STATE Quantize: $q_t = Q(g_t)$.
    \STATE Update: $w_{t+1}= w_t - \eta_t q_t$.
\ENDFOR
\STATE \textbf{Output:} $\{w_t\}_{t=0}^{T}$ (or the averaged iterate $\bar w_T$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic function
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We take a deterministic setting (no mini‑batch noise) and an identity quantizer $Q(g)=g$ (hence $\delta=0$).  
Parameters: $w_0=0$, constant stepsize $\eta=0.1$, $T=3$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.371\\
\end{tabular}
\end{center}
The objective value is decreasing, illustrating the descent property of (2).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Bounded Variance of Stochastic Gradient]\label{ass:var}
There exists $\sigma^2\ge0$ such that
\[
\mathbb{E}\!\left[\|g_t-\nabla f(w_t)\|^2\mid w_t\right]\le\sigma^2,\qquad\forall t .
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Unbiased Quantization]\label{ass:quant}
The quantizer $Q$ satisfies (1):
\[
\mathbb{E}[Q(g_t)\mid g_t]=g_t,\qquad
\mathbb{E}[\|Q(g_t)\|^2\mid g_t]\le (1+\delta)\|g_t\|^2 ,
\]
for a known constant $\delta\ge0$.
\tag{A3}
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize is constant $\eta_t\equiv\eta$ with $0<\eta\le \frac{1}{L(1+\delta)}$.
\tag{A4}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of QSGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize} and after $T$ iterations,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\left[\|\nabla f(w_t)\|^2\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}}_{\text{initial gap term}}
\;+\;
\underbrace{\eta L(1+\delta)\sigma^2}_{\text{variance term}} .
\tag{3}
\]
Choosing the constant stepsize
\[
\eta^\star \;=\; \sqrt{\frac{2\bigl(f(w_0)-f^\star\bigr)}{L(1+\delta)\sigma^2\,T}}
\tag{4}
\]
(which respects (A4) for all sufficiently large $T$) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\left[\|\nabla f(w_t)\|^2\right]
\;\le\;
\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\tag{5}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound (3) contains a term $\eta L(1+\delta)\sigma^2$; larger quantization error $\delta$ or gradient noise $\sigma^2$ requires a smaller stepsize to retain stability.
    \item \textbf{Hyper‑parameter Sensitivity.} The optimal stepsize (4) balances the initial‑error term $O(1/(\eta T))$ against the variance term $O(\eta)$. Over‑aggressive $\eta$ inflates the variance term, while too small $\eta$ slows down progress.
    \item \textbf{Comparison to vanilla SGD.} Setting $\delta=0$ recovers the classic non‑convex SGD bound. Hence quantization incurs at most a multiplicative factor $(1+\delta)$ in the variance term.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Filtration]
Let $\mathcal{F}_t:=\sigma\{w_0,\xi_0,\dots,\xi_{t-1}\}$ be the $\sigma$‑algebra generated by all randomness up to iteration $t$.
\end{definition}
All expectations below are taken with respect to the joint distribution of the stochastic gradients and the quantizer, conditioned on the appropriate filtration.

\begin{lemma}[Smoothness Descent Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from (A1'). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  
All expectations are taken w.r.t. the randomness of $g_t$ and $Q(g_t)$ conditioned on $\mathcal{F}_t$.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $u=\eta Q(g_t)$:
\[
f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),Q(g_t)\rangle+\frac{L\eta^2}{2}\|Q(g_t)\|^2 .
\tag{6}
\]

\noindent\textbf{[Step 2]} Take conditional expectation $\mathbb{E}[\cdot\mid\mathcal{F}_t]$ on both sides of (6).  
Using unbiasedness of $Q$ (Assumption \ref{ass:quant}) we obtain
\[
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le f(w_t)-\eta\|\nabla f(w_t)\|^2
      +\frac{L\eta^2}{2}\,\mathbb{E}\!\left[\|Q(g_t)\|^2\mid\mathcal{F}_t\right].
\tag{7}
\]

\noindent\textbf{[Step 3]} Decompose $g_t$ as $\nabla f(w_t)+\underbrace{(g_t-\nabla f(w_t))}_{\xi_t}$.
Using the second‑moment bound of $Q$ (Assumption \ref{ass:quant}) and the variance bound (Assumption \ref{ass:var}):
\[
\begin{aligned}
\mathbb{E}\!\left[\|Q(g_t)\|^2\mid\mathcal{F}_t\right]
&\le (1+\delta)\,\mathbb{E}\!\left[\|g_t\|^2\mid\mathcal{F}_t\right] \\
&= (1+\delta)\,\Bigl(\|\nabla f(w_t)\|^2
      +\mathbb{E}\!\left[\|g_t-\nabla f(w_t)\|^2\mid\mathcal{F}_t\right]\Bigr) \\
&\le (1+\delta)\,\bigl(\|\nabla f(w_t)\|^2+\sigma^2\bigr).
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 4]} Substitute (8) into (7):
\[
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le f(w_t)-\eta\|\nabla f(w_t)\|^2
   +\frac{L\eta^2}{2}(1+\delta)\bigl(\|\nabla f(w_t)\|^2+\sigma^2\bigr).
\tag{9}
\]

\noindent\textbf{[Step 5]} Rearrange (9) to isolate $\|\nabla f(w_t)\|^2$:
\[
\bigl(\eta-\tfrac{L\eta^2}{2}(1+\delta)\bigr)\|\nabla f(w_t)\|^2
\le f(w_t)-\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
   +\tfrac{L\eta^2}{2}(1+\delta)\sigma^2 .
\tag{10}
\]

\noindent\textbf{[Step 6]} By Assumption \ref{ass:stepsize}, $0<\eta\le\frac{1}{L(1+\delta)}$, which guarantees
\[
\eta-\frac{L\eta^2}{2}(1+\delta)\;\ge\;\frac{\eta}{2}.
\tag{11}
\]
Thus (10) yields
\[
\frac{\eta}{2}\,\|\nabla f(w_t)\|^2
\le f(w_t)-\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
   +\frac{L\eta^2}{2}(1+\delta)\sigma^2 .
\tag{12}
\]

\noindent\textbf{[Step 7]} Take total expectation and sum (12) over $t=0,\dots,T-1$:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\le \mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]
   +\frac{L\eta^2}{2}(1+\delta)\sigma^2 T .
\tag{13}
\]

\noindent\textbf{[Step 8]} Since $f(w_T)\ge f^\star$, where $f^\star:=\inf_w f(w)$,
\[
\mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]\le f(w_0)-f^\star .
\tag{14}
\]

\noindent\textbf{[Step 9]} Insert (14) into (13) and divide by $\eta T/2$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T}
   +\eta L(1+\delta)\sigma^2 .
\tag{15}
\]
Equation (15) is exactly the bound (3) stated in Theorem \ref{thm:main}.

\noindent\textbf{[Step 10]} Optimizing the right‑hand side of (15) over a constant $\eta$ yields the choice (4) and the rate (5). \qed

\section*{Rate Derivation (With Constants)}
Plugging $\eta^\star$ from (4) into (15) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\le 
2\sqrt{\frac{L(1+\delta)\sigma^2\bigl(f(w_0)-f^\star\bigr)}{T}}
\;=\;
\mathcal{O}\!\bigl(T^{-1/2}\bigr).
\]
All constants are explicit:
\[
C:=2\sqrt{L(1+\delta)\sigma^2\bigl(f(w_0)-f^\star\bigr)} .
\]

\section*{Special Cases}
\begin{enumerate}[label=\arabic*.]
\item \textbf{No quantization.} $\delta=0$ reduces (15) to the classical SGD bound.
\item \textbf{Deterministic gradients.} $\sigma^2=0$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\le \frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta T},
\]
i.e. a $O(1/T)$ rate for the average gradient norm.
\item \textbf{Large quantization variance.} If $\delta\gg1$, the variance term dominates; a smaller stepsize $\eta\propto 1/\sqrt{1+\delta}$ is required to keep the same $O(1/\sqrt{T})$ rate.
\end{enumerate}

\section*{Discussion}
The proof shows that, despite the additional noise introduced by unbiased quantization, the descent mechanism of SGD is preserved up to a multiplicative factor $(1+\delta)$ in the variance term. Consequently, QSGD enjoys the same $O(T^{-1/2})$ convergence rate for non‑convex smooth objectives as vanilla SGD, provided the stepsize is chosen according to (4). This theoretical guarantee justifies the practical use of aggressive gradient compression in distributed learning settings.
