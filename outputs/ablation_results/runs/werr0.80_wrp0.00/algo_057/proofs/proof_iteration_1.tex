\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\textbf{Randomized Block‑Coordinate Gradient Descent (R‑BCGD)}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Let $w\in\mathbb{R}^d$ be partitioned into $B$ (disjoint) blocks
\[
w =\big( w^{(1)},w^{(2)},\dots ,w^{(B)}\big),\qquad 
w^{(i)}\in\mathbb{R}^{d_i},\; \sum_{i=1}^{B}d_i=d .
\]

At iteration $t\ge 0$:
\begin{enumerate}[label=\arabic*)]
\item Sample a block index $i_t\in\{1,\dots ,B\}$ independently and uniformly:
\[
\mathbb{P}(i_t=i)=\frac1B,\qquad \forall i.
\]
\item Compute the partial (block) gradient of $f$ at the current point:
\[
g_t \;:=\; \nabla_{i_t}f(w_t)\in\mathbb{R}^{d_{i_t}} .
\]
\item Perform a gradient step only on the selected block:
\[
w_{t+1}^{(i_t)} \;=\; w_t^{(i_t)}- \eta_t\, g_t ,\qquad 
w_{t+1}^{(j)} = w_t^{(j)}\;\; \forall j\neq i_t .
\]
\end{enumerate}
The step size $\eta_t>0$ may be constant ($\eta_t\equiv\eta$) or follow a diminishing schedule; the analysis below assumes a constant $\eta$ satisfying
\[
0<\eta\le \frac{1}{L_{\max}},\qquad 
L_{\max}:=\max_{i\in[B]}L_i,
\]
where $L_i$ is the block‑wise Lipschitz constant defined in the assumptions.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Block‑Coordinate Gradient Descent (R‑BCGD)}
\label{alg:rbcgd}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0$, step size $\eta$, number of iterations $T$.
\FOR{$t=0,\dots ,T-1$}
    \STATE Sample $i_t\sim \mathrm{Uniform}\{1,\dots ,B\}$.
    \STATE Compute block gradient $g_t=\nabla_{i_t}f(w_t)$.
    \STATE Update the selected block:
    \[
    w_{t+1}^{(i_t)} = w_t^{(i_t)}-\eta\, g_t .
    \]
    \STATE Keep other blocks unchanged:
    \[
    w_{t+1}^{(j)} = w_t^{(j)},\quad\forall j\neq i_t .
    \]
\ENDFOR
\STATE \textbf{Output:} $w_T$ (or optionally the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
Consider the scalar function (single block) 
\[
f(w)=(w-3)^2,\qquad w\in\mathbb{R}.
\]
Here $B=1$, $L_1=2$, and the full gradient coincides with the block gradient:
$\nabla f(w)=2(w-3)$.  
Take $w_0=0$, step size $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=\nabla f(w_t)$ & $f(w_t)$\\ \hline

1 & $w_1=0-0.1\cdot(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\

The objective value decreases at each step, illustrating the basic descent mechanism of the algorithm.

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*}]
\item \label{A1} \textbf{(Block‑wise smoothness)}  
For each block $i\in[B]$ there exists $L_i>0$ such that for all $x,y\in\mathbb{R}^d$,
\[
\big\|\nabla_i f(x)-\nabla_i f(y)\big\|
   \le L_i\big\|x-y\big\|.
\]
Define $L_{\max}:=\max_i L_i$ and $L_{\mathrm{avg}}:=\frac1B\sum_{i=1}^B L_i$.

\item \label{A2} \textbf{(Lower boundedness)}  
The objective $f$ is bounded below:
\[
f_{\inf}:=\inf_{w\in\mathbb{R}^d} f(w) > -\infty .
\]

\item \label{A3} \textbf{(Unbiased block gradient)}  
Conditioned on $w_t$, the random block gradient satisfies
\[
\mathbb{E}\!\big[\,\nabla_{i_t} f(w_t)\mid w_t\,\big]=\frac1B\,\nabla f(w_t).
\]

\item \label{A4} \textbf{(Finite variance of block gradient)}  
There exists $\sigma^2\ge0$ such that
\[
\mathbb{E}\!\big[\,\|\nabla_{i_t} f(w_t)-\tfrac1B\nabla f(w_t)\|^2\mid w_t\,\big]
\le \frac{\sigma^2}{B}.
\]
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Expected Gradient Norm Convergence]\label{thm:main}
Assume \ref{A1}–\ref{A4} and choose a constant step size
\[
0<\eta\le \frac{1}{L_{\max}} .
\]
Let $\{w_t\}_{t=0}^T$ be generated by Algorithm~\ref{alg:rbcgd}.  
Define the averaged iterate
\[
\bar w_T:=\frac1T\sum_{t=0}^{T-1} w_t .
\]
Then the following bound holds:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{2B\big(f(w_0)-f_{\inf}\big)}{\eta T}
\;+\;
2\eta\,L_{\max}\,\sigma^2 .
\tag{1}
\]
Consequently, by selecting $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ we obtain
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
= \mathcal{O}\!\big(\tfrac{1}{\sqrt{T}}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The step‑size restriction $\eta\le1/L_{\max}$ guarantees that each stochastic block update does not increase the objective in expectation (see the proof).

\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{eq:1} shows a linear dependence on $\eta$ in the variance term and an inverse dependence on $\eta$ in the deterministic descent term. The optimal $\eta$ balances these two effects, yielding the $\mathcal{O}(1/\sqrt{T})$ rate.

\item \textbf{Comparison to full‑gradient GD.} For $B=1$ the algorithm reduces to standard (stochastic) gradient descent and the bound coincides with the classical non‑convex SGD rate. For $B>1$, the factor $B$ appears only in the first term, reflecting the fact that only a $1/B$ fraction of the gradient is used per iteration.

\item \textbf{Special case $ \sigma =0$.} If the block gradients are deterministic (e.g., exact coordinate gradients), the variance term vanishes and we obtain the deterministic rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \frac{2B\big(f(w_0)-f_{\inf}\big)}{\eta T},
\]
i.e. $\mathcal{O}(1/T)$.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Blockwise smoothness inequality]\label{lem:smooth}
For any block $i$ and any $x,y\in\mathbb{R}^d$,
\[
f\big(x+U_i (y_i-x_i)\big)
\le
f(x) + \langle \nabla_i f(x),\, y_i-x_i\rangle
+ \frac{L_i}{2}\|y_i-x_i\|^2,
\]
where $U_i$ is the $d\times d_i$ selector matrix that injects a block vector into the full space.
\end{lemma}
\begin{proof}
The lemma is a direct consequence of the $L_i$‑Lipschitz continuity of $\nabla_i f$ and the fundamental theorem of calculus applied only along block $i$. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof (Step‑by‑Step Derivation)}
\renewcommand{\theequation}{\arabic{equation}}
\setcounter{equation}{0}
\begin{proof}
We prove inequality \eqref{eq:1} by analysing one iteration and then telescoping.

\bigskip
\noindent\textbf{[Step 1]} \emph{Descent inequality for the selected block.}  
Condition on $w_t$ and apply Lemma~\ref{lem:smooth} with
\[
x = w_t,\qquad y_i = w_t^{(i_t)} - \eta\,\nabla_{i_t}f(w_t),\qquad y_j = w_t^{(j)}\;(j\neq i_t).
\]
Since only block $i_t$ changes,
\[
f(w_{t+1}) \le f(w_t) - \eta\|\nabla_{i_t}f(w_t)\|^2 + \frac{L_{i_t}\eta^2}{2}\|\nabla_{i_t}f(w_t)\|^2 .
\tag{2}
\]

\noindent\textbf{[Step 2]} \emph{Simplify the coefficient.}  
Because $\eta\le 1/L_{\max}\le 1/L_{i_t}$,
\[
-\eta + \frac{L_{i_t}\eta^2}{2}
= -\frac{\eta}{2}\Big(1 - L_{i_t}\eta\Big)
\le -\frac{\eta}{2}.
\]
Plugging into \eqref{2} gives
\[
f(w_{t+1}) \le f(w_t) - \frac{\eta}{2}\,\|\nabla_{i_t}f(w_t)\|^2 .
\tag{3}
\]

\noindent\textbf{[Step 3]} \emph{Take expectation over the random block $i_t$.}  
Using the uniform sampling and the unbiasedness property \ref{A3},
\[
\mathbb{E}_{i_t}\!\big[\|\nabla_{i_t}f(w_t)\|^2\mid w_t\big]
= \frac1B\sum_{i=1}^B\|\nabla_i f(w_t)\|^2 .
\]
Define the \emph{full} gradient norm squared as
\[
\|\nabla f(w_t)\|^2 = \sum_{i=1}^B\|\nabla_i f(w_t)\|^2 .
\]
Thus
\[
\mathbb{E}_{i_t}\!\big[\|\nabla_{i_t}f(w_t)\|^2\mid w_t\big]
= \frac{1}{B}\,\|\nabla f(w_t)\|^2 .
\tag{4}
\]

\noindent\textbf{[Step 4]} \emph{Conditional expectation of the descent inequality.}  
Taking $\mathbb{E}_{i_t}[\cdot\mid w_t]$ on \eqref{3} and using \eqref{4} yields
\[
\mathbb{E}_{i_t}\!\big[f(w_{t+1})\mid w_t\big]
\le
f(w_t) - \frac{\eta}{2B}\,\|\nabla f(w_t)\|^2 .
\tag{5}
\]

\noindent\textbf{[Step 5]} \emph{Incorporate variance term.}  
When the block gradient is stochastic (e.g., due to noisy evaluations) we replace $g_t$ by an unbiased estimator $\tilde g_t$ satisfying \ref{A4}. The same algebra as in Steps 1–4 leads to an extra term $\frac{\eta L_{\max}}{2}\sigma^2$ (see derivation below). For clarity we first present the deterministic case; the stochastic extension is added after the telescoping argument.

\noindent\textbf{[Step 6]} \emph{Take total expectation and telescope.}  
Define $\mathcal{F}_t$ as the sigma‑algebra generated by $\{i_0,\dots,i_{t-1}\}$. Taking full expectation in \eqref{5} gives
\[
\mathbb{E}\big[f(w_{t+1})\big]
\le
\mathbb{E}\big[f(w_t)\big]
- \frac{\eta}{2B}\,\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big] .
\tag{6}
\]
Summing \eqref{6} over $t=0,\dots,T-1$ telescopes the left‑hand side:
\[
\mathbb{E}[f(w_T)] - f(w_0)
\le
- \frac{\eta}{2B}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big] .
\tag{7}
\]
Rearranging,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2B\big(f(w_0)-\mathbb{E}[f(w_T)]\big)}{\eta T}
\le
\frac{2B\big(f(w_0)-f_{\inf}\big)}{\eta T} .
\tag{8}
\]

\noindent\textbf{[Step 7]} \emph{Add the stochastic variance contribution.}  
When $g_t$ is a noisy estimator, the smoothness inequality \eqref{2} becomes
\[
f(w_{t+1}) \le f(w_t) - \eta\langle \nabla_{i_t}f(w_t),\tilde g_t\rangle
+ \frac{L_{i_t}\eta^2}{2}\|\tilde g_t\|^2 .
\]
Using $\mathbb{E}[\tilde g_t\mid w_t,i_t]=\nabla_{i_t}f(w_t)$ and \ref{A4},
\[
\mathbb{E}\!\big[f(w_{t+1})\mid \mathcal{F}_t\big]
\le
f(w_t) - \frac{\eta}{2}\|\nabla_{i_t}f(w_t)\|^2
+ \frac{L_{i_t}\eta^2}{2}\,\sigma^2/B .
\]
Taking expectation over $i_t$ and using $\eta\le 1/L_{\max}$,
\[
\mathbb{E}\!\big[f(w_{t+1})\mid \mathcal{F}_t\big]
\le
f(w_t) - \frac{\eta}{2B}\|\nabla f(w_t)\|^2
+ \frac{\eta L_{\max}\sigma^2}{2B}.
\tag{9}
\]
Proceeding as in Steps 5–6 and summing over $t$ yields the full bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2B\big(f(w_0)-f_{\inf}\big)}{\eta T}
+ 2\eta L_{\max}\sigma^2 .
\tag{10}
\]
Equation \eqref{eq:1} is precisely \eqref{10}, completing the proof. ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main},
\[
\Phi(T):=
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{C_1}{\eta T}+C_2\eta,
\quad
C_1:=2B\big(f(w_0)-f_{\inf}\big),\;
C_2:=2L_{\max}\sigma^2 .
\]
Minimising the right‑hand side w.r.t. $\eta$ gives
\[
\eta^\star = \sqrt{\frac{C_1}{C_2\,T}}
\quad\Longrightarrow\quad
\Phi(T)\le 2\sqrt{C_1C_2}\,\frac{1}{\sqrt{T}} .
\]
Thus the algorithm attains the optimal $\mathcal{O}(1/\sqrt{T})$ rate for non‑convex smooth objectives under the stated assumptions.

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}[label=\arabic*)]
\item \textbf{Deterministic block gradients ($\sigma=0$).}  
The bound reduces to $\Phi(T)\le \frac{2B\big(f(w_0)-f_{\inf}\big)}{\eta T}$, i.e. $\mathcal{O}(1/T)$.

\item \textbf{Single block ($B=1$).}  
Algorithm becomes standard (stochastic) gradient descent and Theorem~\ref{thm:main} recovers the classic non‑convex SGD bound.

\item \textbf{Increasing block size.}  
If each block contains $d_i\approx d/B$ coordinates and $L_i\approx L_{\max}$, the constant $C_1$ grows linearly with $B$, reflecting the reduced amount of information per iteration. Choosing $\eta\propto 1/\sqrt{B}$ restores the same asymptotic rate.

\end{enumerate}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The analysis hinges only on block‑wise smoothness and a simple unbiasedness condition; no convexity is required. Consequently, R‑BCGD is applicable to a broad class of deep learning and signal‑processing problems where full‑gradient computation is expensive but coordinate‑wise (or block‑wise) gradients are cheap to obtain. The explicit constants in \eqref{eq:1} enable practitioners to tune $\eta$ according to problem‑specific $L_{\max}$ and variance estimates, guaranteeing a provable descent in expectation.
