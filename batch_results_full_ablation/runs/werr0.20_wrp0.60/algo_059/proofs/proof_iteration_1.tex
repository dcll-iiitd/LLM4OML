\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (possibly non‑convex) objective.  
At iteration $t$ we draw a stochastic sample $\xi_{t}$ and compute a stochastic gradient
\[
g_{t}\;\stackrel{\text{def}}{=}\;\nabla f(w_{t};\xi_{t})\in\R^{d}.
\]
A (random) quantization operator $Q:\R^{d}\to\R^{d}$ satisfies  
\[
\boxed{\;\E[\,Q(g)\mid g\,]=g\;}\tag{1}
\]
and will be communicated to the parameter server.  
The update rule of QSGD is
\[
\boxed{\;w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t})\;}\tag{2}
\]
where $\{\eta_{t}\}_{t\ge0}$ is a prescribed stepsize schedule.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}$, quantizer $Q(\cdot)$
\For{$t=0,1,\dots,T-1$}
    \State Sample $\xi_{t}\sim\mathcal{D}$ 
    \State Compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$
    \State Quantize: $\tilde g_{t}= Q(g_{t})$   \Comment{communicate $\tilde g_{t}$}
    \State Update: $w_{t+1}= w_{t}-\eta_{t}\tilde g_{t}$
\EndFor
\State \textbf{Output:} $\{w_{t}\}_{t=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Take a deterministic setting (no stochasticity) and an identity quantizer $Q(g)=g$.  
Initialize $w_{0}=0$, stepsize $\eta=0.1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3650$\\
\end{tabular}
\end{center}
The objective value decreases from $9$ to $2.365$ in three iterations.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{as:lb}
There exists $f_{\inf}>-\infty$ such that $f(w)\ge f_{\inf}$ for all $w$.
\tag{A2}
\end{assumption}

\begin{assumption}[Stochastic Gradient]\label{as:sg}
For each $t$, conditional on $w_{t}$,
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad 
\E\big[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\big]\le\sigma^{2}.
\tag{A3}
\]
\end{assumption}

\begin{assumption}[Unbiased Quantizer]\label{as:quant}
$Q$ satisfies (1) and has bounded relative variance:
\[
\E\big[\|Q(g)-g\|^{2}\mid g\big]\le \delta^{2}\|g\|^{2}
\quad\text{for all }g\in\R^{d},
\tag{A4}
\]
with a constant $\delta\ge0$ (e.g. $\delta=0$ for exact communication).
\end{assumption}

\begin{assumption}[Stepsize]\label{as:step}
The stepsizes are non‑increasing and satisfy
\[
0<\eta_{t}\le\frac{1}{L(1+\delta^{2})}\quad\forall t.
\tag{A5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:step}. Let $\{w_{t}\}$ be generated by QSGD \eqref{2}. Define
\[
\bar\eta_{T}\;\stackrel{\text{def}}{=}\;\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}.
\]
Then
\[
\boxed{
\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}
\sum_{t=0}^{T-1}\eta_{t}\,
\E\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(w_{0})-f_{\inf}\big)}{(1-\beta)L}
\;+\;
\frac{L\,\sigma^{2}}{1-\beta}\,
\bar\eta_{T},
}
\tag{3}
\]
where $\beta\stackrel{\text{def}}{=}\frac{L\delta^{2}}{2}\,\max_{t}\eta_{t}\in[0,1)$.  
Consequently, for a constant stepsize $\eta_{t}\equiv\eta\le\frac{1}{L(1+\delta^{2})}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(w_{t})\|^{2}\big]
\;\le\;
\frac{2\big(f(w_{0})-f_{\inf}\big)}{(1-\beta)L\,\eta\,T}
\;+\;
\frac{L\sigma^{2}}{1-\beta}\,\frac{1}{T}.
\tag{4}
\]
Thus $\displaystyle\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]=\mathcal O\!\left(\frac{1}{T}\right)$
when $\sigma=0$ (deterministic case) and $\mathcal O\!\left(\frac{1}{\sqrt{T}}\right)$ with the standard diminishing schedule $\eta_{t}=c/\sqrt{t+1}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \ref{as:step} guarantees that the term $1-\beta-\frac{L\eta_{t}}{2}>0$, preventing divergence even in the presence of quantization noise.
\item \textbf{Hyper‑parameter Sensitivity.} The admissible stepsize upper bound shrinks as $\delta$ grows; high quantization variance forces smaller $\eta_{t}$.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item With $\delta=0$ (no quantization) Theorem~\ref{thm:main} reduces to the classic SGD bound for smooth non‑convex functions.
        \item With $\sigma=0$ (full‑batch) the rate improves to $O(1/T)$, matching deterministic gradient descent.
    \end{itemize}
\item \textbf{Special Cases.}
    \begin{enumerate}
        \item \emph{Exact Quantizer} ($\delta=0$): the bound coincides with standard SGD.
        \item \emph{Unbiased Random Sparsifier} where each coordinate is kept with probability $p$: $\delta^{2}= \frac{1-p}{p}$.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that unbiased quantization only inflates the effective stepsize denominator by the factor $1-\beta$, where $\beta=O(L\delta^{2}\eta)$. Hence, for modest quantization (small $\delta$) and stepsizes respecting \ref{as:step}, the asymptotic convergence rate remains unchanged up to a constant factor. This aligns with empirical observations that low‑bit communication does not deteriorate the speed of training deep non‑convex models.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step Smoothness]\label{lem:smooth}
For any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Directly from \ref{as:smooth} (A1') with $y=w-u$.
\end{proof}

\begin{lemma}[Quantization Error Decomposition]\label{lem:quant}
Let $Q$ satisfy \ref{as:quant}. Define $\varepsilon_{t}=Q(g_{t})-g_{t}$. Then
\[
\E[\varepsilon_{t}\mid g_{t}]=0,\qquad
\E[\|\varepsilon_{t}\|^{2}\mid g_{t}]\le\delta^{2}\|g_{t}\|^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Immediate from unbiasedness (1) and the variance bound (A4).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness up to iteration $t$, unless otherwise noted.

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $u=\eta_{t}Q(g_{t})$:
\[
f(w_{t+1})\le f(w_{t})-\eta_{t}\langle\nabla f(w_{t}),Q(g_{t})\rangle
+\frac{L\eta_{t}^{2}}{2}\|Q(g_{t})\|^{2}.
\tag{5}
\]

\noindent\textbf{[Step 2]} Decompose $Q(g_{t})=g_{t}+\varepsilon_{t}$:
\[
\langle\nabla f(w_{t}),Q(g_{t})\rangle
= \langle\nabla f(w_{t}),g_{t}\rangle
+ \langle\nabla f(w_{t}),\varepsilon_{t}\rangle .
\tag{6}
\]

\noindent\textbf{[Step 3]} Expand $\|Q(g_{t})\|^{2}$:
\[
\|Q(g_{t})\|^{2}= \|g_{t}\|^{2}+2\langle g_{t},\varepsilon_{t}\rangle+\|\varepsilon_{t}\|^{2}.
\tag{7}
\]

\noindent\textbf{[Step 4]} Insert (6)–(7) into (5):
\[
\begin{aligned}
f(w_{t+1})\le f(w_{t}) 
&-\eta_{t}\langle\nabla f(w_{t}),g_{t}\rangle
-\eta_{t}\langle\nabla f(w_{t}),\varepsilon_{t}\rangle \\
&+\frac{L\eta_{t}^{2}}{2}\|g_{t}\|^{2}
+L\eta_{t}^{2}\langle g_{t},\varepsilon_{t}\rangle
+\frac{L\eta_{t}^{2}}{2}\|\varepsilon_{t}\|^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 5]} Take conditional expectation $\E[\cdot\mid w_{t}]$.
Using Lemma~\ref{lem:quant} ($\E[\varepsilon_{t}\mid g_{t}]=0$) and the law of total expectation,
\[
\E[\langle\nabla f(w_{t}),\varepsilon_{t}\rangle\mid w_{t}]
=0,\qquad
\E[\langle g_{t},\varepsilon_{t}\rangle\mid w_{t}]=0 .
\tag{9}
\]

\noindent\textbf{[Step 6]} Apply (9) to (8):
\[
\E[f(w_{t+1})\mid w_{t}]
\le f(w_{t})-\eta_{t}\E[\langle\nabla f(w_{t}),g_{t}\rangle\mid w_{t}]
+\frac{L\eta_{t}^{2}}{2}\E[\|g_{t}\|^{2}\mid w_{t}]
+\frac{L\eta_{t}^{2}}{2}\E[\|\varepsilon_{t}\|^{2}\mid w_{t}].
\tag{10}
\]

\noindent\textbf{[Step 7]} Use unbiased stochastic gradient (A3):
\[
\E[\langle\nabla f(w_{t}),g_{t}\rangle\mid w_{t}]
= \|\nabla f(w_{t})\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8]} Decompose $\|g_{t}\|^{2}$ using variance bound (A3):
\[
\E[\|g_{t}\|^{2}\mid w_{t}]
= \|\nabla f(w_{t})\|^{2}+\E[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 9]} Bound the quantization error term using Lemma~\ref{lem:quant} and (12):
\[
\E[\|\varepsilon_{t}\|^{2}\mid w_{t}]
\le \delta^{2}\E[\|g_{t}\|^{2}\mid w_{t}]
\le \delta^{2}\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big).
\tag{13}
\]

\noindent\textbf{[Step 10]} Substitute (11)–(13) into (10):
\[
\begin{aligned}
\E[f(w_{t+1})\mid w_{t}]
\le\;& f(w_{t})
-\eta_{t}\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big) \\
&+\frac{L\eta_{t}^{2}}{2}\,\delta^{2}\big(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\big).
\end{aligned}
\tag{14}
\]

\noindent\textbf{[Step 11]} Collect the coefficients of $\|\nabla f(w_{t})\|^{2}$.
Define
\[
\alpha_{t}\;\stackrel{\text{def}}{=}\;
\eta_{t}
-\frac{L\eta_{t}^{2}}{2}\big(1+\delta^{2}\big).
\tag{15}
\]
Then (14) becomes
\[
\E[f(w_{t+1})\mid w_{t}]
\le f(w_{t})-\alpha_{t}\,\|\nabla f(w_{t})\|^{2}
+\frac{L\eta_{t}^{2}}{2}\big(1+\delta^{2}\big)\sigma^{2}.
\tag{16}
\]

\noindent\textbf{[Step 12]} By assumption \ref{as:step}, $\eta_{t}\le\frac{1}{L(1+\delta^{2})}$, hence
\[
\alpha_{t}\ge\frac{\eta_{t}}{2}\;>\;0.
\tag{17}
\]

\noindent\textbf{[Step 13]} Take full expectation and sum (16) over $t=0,\dots,T-1$:
\[
\begin{aligned}
\E[f(w_{T})]
\le\;& f(w_{0})
-\sum_{t=0}^{T-1}\alpha_{t}\,\E[\|\nabla f(w_{t})\|^{2}]
+\frac{L}{2}\big(1+\delta^{2}\big)\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.
\end{aligned}
\tag{18}
\]

\noindent\textbf{[Step 14]} Drop the non‑negative term $\E[f(w_{T})]\ge f_{\inf}$ and rearrange:
\[
\sum_{t=0}^{T-1}\alpha_{t}\,\E[\|\nabla f(w_{t})\|^{2}]
\le f(w_{0})-f_{\inf}
+\frac{L}{2}\big(1+\delta^{2}\big)\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{19}
\]

\noindent\textbf{[Step 15]} Using $\alpha_{t}\ge\frac{\eta_{t}}{2}$ from (17):
\[
\frac{1}{2}\sum_{t=0}^{T-1}\eta_{t}\,\E[\|\nabla f(w_{t})\|^{2}]
\le f(w_{0})-f_{\inf}
+\frac{L}{2}\big(1+\delta^{2}\big)\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}.
\tag{20}
\]

\noindent\textbf{[Step 16]} Divide both sides of (20) by $\sum_{t=0}^{T-1}\eta_{t}$ and multiply by $2$:
\[
\frac{\sum_{t=0}^{T-1}\eta_{t}\,\E[\|\nabla f(w_{t})\|^{2}]}
     {\sum_{t=0}^{T-1}\eta_{t}}
\le
\frac{2\big(f(w_{0})-f_{\inf}\big)}{\sum_{t=0}^{T-1}\eta_{t}}
+L\big(1+\delta^{2}\big)\sigma^{2}
\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}.
\tag{21}
\]

\noindent\textbf{[Step 17]} Define $\beta\stackrel{\text{def}}{=}\frac{L\delta^{2}}{2}\max_{t}\eta_{t}\in[0,1)$.
Since $\eta_{t}\le\frac{1}{L(1+\delta^{2})}$, we have
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}
\le\frac{1}{(1-\beta)L}\,\bar\eta_{T},
\qquad
\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}
\le\frac{\max_{t}\eta_{t}}{1-\beta}\;\le\;\frac{1}{(1-\beta)L}.
\tag{22}
\]

\noindent\textbf{[Step 18]} Plug (22) into (21) and simplify, obtaining exactly bound (3) of Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
For a constant stepsize $\eta_{t}\equiv\eta$ satisfying $\eta\le\frac{1}{L(1+\delta^{2})}$,
\[
\sum_{t=0}^{T-1}\eta_{t}= \eta T,\qquad
\sum_{t=0}^{T-1}\eta_{t}^{2}= \eta^{2}T .
\]
Insert into (21):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_{t})\|^{2}]
\le
\frac{2\big(f(w_{0})-f_{\inf}\big)}{(1-\beta)L\,\eta\,T}
+\frac{L\big(1+\delta^{2}\big)\sigma^{2}}{1-\beta}\,\frac{\eta}{T}.
\]
Choosing $\eta=c/\sqrt{T}$ yields the standard $\mathcal O(1/\sqrt{T})$ rate; choosing $\eta=c/T$ gives $\mathcal O(1/T)$ when $\sigma=0$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact Gradient (no stochasticity, $\sigma=0$).}  
Bound (4) reduces to $\displaystyle\frac{2(f(w_{0})-f_{\inf})}{(1-\beta)L\,\eta\,T}$, i.e. $O(1/T)$.
\item \textbf{Deterministic Quantization ($\sigma=0$, $\delta>0$).}  
Same $O(1/T)$ rate with an enlarged constant $1/(1-\beta)$.
\item \textbf{High‑Precision Quantization ($\delta\to0$).}  
$\beta\to0$ and the bound coincides with the classical SGD result.
\end{itemize}
