\section*{Mirror Descent with Bregman Divergence (Non‑Convex Setting)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable (possibly non‑convex) and let $h:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a $1$‑strongly convex \emph{prox‑function} with respect to a norm $\|\cdot\|$.  
Its Bregman divergence is
\[
\Breg(x,y)=h(x)-h(y)-\inner{\grad h(y)}{x-y},\qquad x,y\in\mathbb{R}^{d}.
\]

Given a stepsize $\eta_{k}>0$, the \emph{mirror‑descent} iterate is
\begin{equation}\label{eq:update}
x_{k+1}
\;=\;\arg\min_{x\in\mathbb{R}^{d}}\Big\{\,\inner{\grad f(x_{k})}{x}\;+\;\frac{1}{\eta_{k}}\Breg(x,x_{k})\Big\}.
\end{equation}
Equivalently, using the first‑order optimality condition,
\begin{equation}\label{eq:optcond}
\grad h(x_{k+1}) \;=\; \grad h(x_{k}) - \eta_{k}\,\grad f(x_{k}). 
\end{equation}
Define the \emph{gradient mapping} (mirror‑gradient) 
\[
G_{\eta_{k}}(x_{k}) \;:=\; \frac{1}{\eta_{k}}\bigl(\grad h(x_{k})-\grad h(x_{k+1})\bigr)
\;=\; \grad f(x_{k}).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent for Non‑Convex Smooth $f$}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_{0}\in\mathbb{R}^{d}$, stepsize schedule $\{\eta_{k}\}_{k\ge 0}$
\State \textbf{Initialize:} compute $u_{0}\gets\grad h(x_{0})$
\For{$k=0,1,\dots,T-1$}
    \State Compute stochastic (here deterministic) gradient $g_{k}\gets \grad f(x_{k})$
    \State Update dual variable $u_{k+1}\gets u_{k}-\eta_{k} g_{k}$
    \State Map back to primal space $x_{k+1}\gets (\grad h)^{-1}(u_{k+1})$
\EndFor
\State \textbf{Output:} $\{x_{k}\}_{k=0}^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, choose $h(w)=\tfrac12 w^{2}$ (so that $\Breg(w,w_{k})=\tfrac12(w-w_{k})^{2}$ and $\grad h(w)=w$).  
Set $w_{0}=0$, constant stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_{k}=\grad f(w_{k})$ & $w_{k+1}=w_{k}-\eta g_{k}$ & $f(w_{k+1})$\\ \hline
0 & $g_{0}= -6$ & $w_{1}=0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $g_{1}= 2(0.6-3)=-4.8$ & $w_{2}=0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $g_{2}= 2(1.08-3)=-3.84$ & $w_{3}=1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.366\\
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the algorithmic step \eqref{eq:update} for this simple case.

\section*{Assumptions}
\begin{assumption}[Strong Convexity of $h$]\label{ass:h_sc}
$h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$, i.e.
\[
\Breg(x,y)\;\ge\;\tfrac12\|x-y\|^{2},\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}

\begin{assumption}[Smoothness of $f$ w.r.t. Bregman geometry]\label{ass:f_smooth}
There exists $L>0$ such that for all $x,y\in\mathbb{R}^{d}$,
\[
f(x)\;\le\; f(y)+\inner{\grad f(y)}{x-y}+\frac{L}{2}\|x-y\|^{2}.
\]
Equivalently,
\[
\|\grad f(x)-\grad f(y)\|\;\le\;L\|x-y\|.
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsizes satisfy $0<\eta_{k}\le\frac{1}{L}$ for all $k$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{x_{k}\}$ be generated by \eqref{eq:update} under Assumptions~\ref{ass:h_sc}--\ref{ass:stepsize}. Define the gradient mapping
\[
G_{\eta_{k}}(x_{k})\;:=\;\frac{1}{\eta_{k}}\bigl(\grad h(x_{k})-\grad h(x_{k+1})\bigr).
\]
Then for any $T\ge1$,
\[
\frac{1}{\sum_{k=0}^{T-1}\eta_{k}}\sum_{k=0}^{T-1}\eta_{k}\,\EE\!\left[\|G_{\eta_{k}}(x_{k})\|^{2}\right]
\;\le\;
\frac{2L\bigl(f(x_{0})-f^{\inf}\bigr)}{\sum_{k=0}^{T-1}\eta_{k}}.
\]
In particular, with a constant stepsize $\eta_{k}\equiv\eta\le1/L$,
\[
\min_{0\le k\le T-1}\EE\!\left[\|G_{\eta}(x_{k})\|^{2}\right]
\;\le\;
\frac{2L\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta\,T}=O\!\left(\frac{1}{T}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stationarity Measure.} $G_{\eta}(x_{k})$ coincides with the Euclidean gradient when $h(w)=\tfrac12\|w\|^{2}$, thus the bound recovers the classic $O(1/T)$ rate for (non‑convex) gradient descent.
\item \textbf{Stability w.r.t. $\eta$.} The condition $\eta\le1/L$ guarantees that the descent lemma (Lemma~\ref{lem:descent}) holds; larger stepsizes may violate the inequality and cause divergence.
\item \textbf{Comparison to SGD.} When stochastic gradients are used, the same proof yields an extra variance term $\sigma^{2}$, leading to the familiar $O(1/\sqrt{T})$ rate. The deterministic bound above is thus tighter.
\item \textbf{Effect of the prox‑function.} Choosing $h$ adapted to the geometry of the feasible set (e.g., negative entropy for the simplex) can dramatically improve practical performance while preserving the $O(1/T)$ stationarity guarantee.
\end{itemize}

\section*{Discussion}
The theorem shows that even without convexity, mirror descent drives the Bregman gradient mapping to zero at the optimal $1/T$ rate, provided the objective is smooth in the Euclidean sense and the prox‑function is strongly convex. The proof relies solely on three‑point identities of Bregman divergences and the smoothness inequality; no convexity of $f$ is required. Hence the method is applicable to a broad class of non‑convex problems (e.g., training of deep neural networks with appropriate mirror maps).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:3pt}
For any $x,y,z\in\mathbb{R}^{d}$,
\[
\Breg(x,z)-\Breg(y,z)-\Breg(x,y)=\inner{\grad h(y)-\grad h(z)}{x-y}.
\]
\end{lemma}
\begin{proof}
Direct expansion of the definition of $\Breg(\cdot,\cdot)$ yields the identity. \hfill $\square$
\end{proof}

\begin{lemma}[Descent Lemma for $f$]\label{lem:descent}
If $f$ is $L$‑smooth (Assumption~\ref{ass:f_smooth}), then for any $x,y$,
\[
f(x)\le f(y)+\inner{\grad f(y)}{x-y}+\frac{L}{2}\|x-y\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz continuity of $\grad f$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof all expectations are taken w.r.t. the randomness (if any) up to iteration $k$; in the deterministic setting they disappear.

\noindent\textbf{Step 1 (Optimality condition).}  
From the definition \eqref{eq:update} and the first‑order optimality condition we have for the minimiser $x_{k+1}$:
\begin{equation}\label{eq:opt}
\inner{\grad f(x_{k})}{x-x_{k+1}}+\frac{1}{\eta_{k}}\inner{\grad h(x_{k+1})-\grad h(x_{k})}{x-x_{k+1}}\ge0,\qquad\forall x.
\end{equation}
\hfill[Step 1]

\noindent\textbf{Step 2 (Choose $x=x_{k}$).}  
Plugging $x=x_{k}$ into \eqref{eq:opt} gives
\[
\inner{\grad f(x_{k})}{x_{k}-x_{k+1}}+\frac{1}{\eta_{k}}\inner{\grad h(x_{k+1})-\grad h(x_{k})}{x_{k}-x_{k+1}}\ge0.
\]
Rearranging,
\begin{equation}\label{eq:inner}
\inner{\grad f(x_{k})}{x_{k+1}-x_{k}}\le\frac{1}{\eta_{k}}\inner{\grad h(x_{k})-\grad h(x_{k+1})}{x_{k+1}-x_{k}}.
\end{equation}
\hfill[Step 2]

\noindent\textbf{Step 3 (Apply three‑point identity).}  
Using Lemma~\ref{lem:3pt} with $(x,y,z)=(x_{k+1},x_{k},x_{k})$ yields
\[
\Breg(x_{k+1},x_{k})-\Breg(x_{k},x_{k})-\Breg(x_{k+1},x_{k})=
\inner{\grad h(x_{k})-\grad h(x_{k})}{x_{k+1}-x_{k}}=0.
\]
More useful is the identity with $z=x_{k-1}$; however, the needed relation is directly given by the strong convexity of $h$:
\[
\inner{\grad h(x_{k})-\grad h(x_{k+1})}{x_{k+1}-x_{k}}\;=\; \|\grad h(x_{k})-\grad h(x_{k+1})\|_{*}^{2},
\]
where $\|\cdot\|_{*}$ denotes the dual norm. By the $1$‑strong convexity of $h$ (Assumption~\ref{ass:h_sc}),
\[
\|\grad h(x_{k})-\grad h(x_{k+1})\|_{*}^{2}\;\ge\;\|x_{k+1}-x_{k}\|^{2}.
\]
Hence,
\begin{equation}\label{eq:bound_inner}
\inner{\grad f(x_{k})}{x_{k+1}-x_{k}}\;\le\; -\frac{1}{\eta_{k}}\|x_{k+1}-x_{k}\|^{2}.
\end{equation}
\hfill[Step 3]

\noindent\textbf{Step 4 (Smoothness of $f$).}  
Apply Lemma~\ref{lem:descent} with $y=x_{k}$ and $x=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\inner{\grad f(x_{k})}{x_{k+1}-x_{k}}+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\]
Insert the bound \eqref{eq:bound_inner}:
\begin{equation}\label{eq:descent}
f(x_{k+1})\le f(x_{k})-\Bigl(\frac{1}{\eta_{k}}-\frac{L}{2}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\end{equation}
\hfill[Step 4]

\noindent\textbf{Step 5 (Step size condition).}  
By Assumption~\ref{ass:stepsize}, $\eta_{k}\le 1/L$, which implies $\frac{1}{\eta_{k}}-\frac{L}{2}\ge\frac{1}{2\eta_{k}}$. Hence \eqref{eq:descent} simplifies to
\begin{equation}\label{eq:descent2}
f(x_{k+1})\le f(x_{k})-\frac{1}{2\eta_{k}}\|x_{k+1}-x_{k}\|^{2}.
\end{equation}
\hfill[Step 5]

\noindent\textbf{Step 6 (Relate distance to gradient mapping).}  
From the definition of $G_{\eta_{k}}(x_{k})$ and \eqref{eq:optcond},
\[
G_{\eta_{k}}(x_{k})=\frac{1}{\eta_{k}}\bigl(\grad h(x_{k})-\grad h(x_{k+1})\bigr)=\grad f(x_{k}).
\]
Strong convexity of $h$ yields
\[
\|x_{k+1}-x_{k}\|\;\le\;\|\grad h(x_{k})-\grad h(x_{k+1})\|_{*}
= \eta_{k}\,\|G_{\eta_{k}}(x_{k})\|.
\]
Squaring,
\begin{equation}\label{eq:relate}
\|x_{k+1}-x_{k}\|^{2}\;\le\;\eta_{k}^{2}\,\|G_{\eta_{k}}(x_{k})\|^{2}.
\end{equation}
\hfill[Step 6]

\noindent\textbf{Step 7 (Combine \eqref{eq:descent2} and \eqref{eq:relate}).}  
Substituting \eqref{eq:relate} into \eqref{eq:descent2} gives
\[
f(x_{k+1})\le f(x_{k})-\frac{\eta_{k}}{2}\,\|G_{\eta_{k}}(x_{k})\|^{2}.
\]
Rearranging,
\begin{equation}\label{eq:one_step}
\eta_{k}\,\|G_{\eta_{k}}(x_{k})\|^{2}\;\le\;2\bigl(f(x_{k})-f(x_{k+1})\bigr).
\end{equation}
\hfill[Step 7]

\noindent\textbf{Step 8 (Summation over iterations).}  
Sum \eqref{eq:one_step} from $k=0$ to $T-1$:
\[
\sum_{k=0}^{T-1}\eta_{k}\,\|G_{\eta_{k}}(x_{k})\|^{2}\;\le\;2\bigl(f(x_{0})-f(x_{T})\bigr)
\;\le\;2\bigl(f(x_{0})-f^{\inf}\bigr),
\]
where $f^{\inf}:=\inf_{x}f(x)$. Divide both sides by $\sum_{k=0}^{T-1}\eta_{k}$ to obtain
\[
\frac{1}{\sum_{k=0}^{T-1}\eta_{k}}\sum_{k=0}^{T-1}\eta_{k}\,\|G_{\eta_{k}}(x_{k})\|^{2}
\;\le\;\frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{\sum_{k=0}^{T-1}\eta_{k}}.
\]
Taking expectations (trivial in the deterministic case) yields the bound of Theorem~\ref{thm:main}.
\hfill[Step 8]

\noindent\textbf{Step 9 (Constant stepsize).}  
If $\eta_{k}\equiv\eta\le1/L$, then $\sum_{k=0}^{T-1}\eta_{k}= \eta T$ and the previous inequality becomes
\[
\min_{0\le k\le T-1}\EE\!\left[\|G_{\eta}(x_{k})\|^{2}\right]
\;\le\;\frac{1}{T}\sum_{k=0}^{T-1}\EE\!\left[\|G_{\eta}(x_{k})\|^{2}\right]
\;\le\;\frac{2L\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta T},
\]
which is the claimed $O(1/T)$ stationarity rate.
\hfill[Step 9]

\section*{Rate Derivation (With Constants)}
Collecting the constants from the proof:
\begin{itemize}
\item Smoothness constant $L$ (Assumption~\ref{ass:f_smooth}).
\item Strong‑convexivity constant of $h$ (taken as $1$ without loss of generality).
\item Stepsize bound $\eta\le1/L$.
\end{itemize}
Consequently,
\[
\boxed{\displaystyle
\min_{0\le k<T}\EE\!\bigl[\|G_{\eta}(x_{k})\|^{2}\bigr]
\le
\frac{2L\bigl(f(x_{0})-f^{\inf}\bigr)}{\eta\,T}
}
\]
If one prefers the Euclidean gradient $\|\grad f(x_{k})\|$, note that $G_{\eta}(x_{k})=\grad f(x_{k})$ by \eqref{eq:optcond}, so the same bound applies.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Euclidean Mirror Map.} $h(w)=\tfrac12\|w\|^{2}$ gives $\Breg$ equal to the squared Euclidean distance. The algorithm reduces to classical gradient descent, and Theorem~\ref{thm:main} recovers the standard $O(1/T)$ convergence to a stationary point.
\item \textbf{Negative Entropy on the Simplex.} For $h(x)=\sum_{i}x_{i}\log x_{i}$ (defined on the probability simplex), $\Breg$ is the Kullback–Leibler divergence. The same theorem holds, providing an $O(1/T)$ guarantee for mirror descent on the simplex.
\item \textbf{Stochastic Gradients.} If $g_{k}$ is an unbiased estimator of $\grad f(x_{k})$ with bounded variance $\sigma^{2}$, the inequality \eqref{eq:one_step} acquires an additional term $\eta_{k}\sigma^{2}$, leading to the well‑known $O(1/\sqrt{T})$ rate for non‑convex stochastic mirror descent.
\end{enumerate}
