\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimisation problem  
\[
\min_{x\in\R^{d}} \;F(x):=f(x)+g(x),
\]
where  

\begin{itemize}[noitemsep]
\item $f:\R^{d}\!\to\!\R$ is convex and $L$‑smooth, i.e.  
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\qquad\forall x,y\in\R^{d},
\]
\item $g:\R^{d}\!\to\!\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).
\end{itemize}

For a stepsize $\eta>0$, the proximal gradient iteration reads
\begin{equation}
\boxed{\;x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta \nabla f(x_{k})\bigr)\;},
\qquad k=0,1,2,\dots
\tag{1}
\end{equation}
where the proximal operator of $g$ is defined by  
\[
\prox_{\eta g}(v):=\arg\min_{u\in\R^{d}}
\Bigl\{ g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$
\For{$k=0,1,\dots,T-1$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$
    \State $x_{k+1}\gets \prox_{\eta g}\!\bigl(x_{k}-\eta g_{k}\bigr)$
\EndFor
\State \textbf{output} $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimise $f(w)=(w-3)^{2}$, take $g\equiv0$, $w_{0}=0$, stepsize $\eta=0.1$, and perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ & $f(w_{k+1})$\\\hline
0 & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3677$\\
\end{tabular}
\end{center}
Since $g\equiv0$, the proximal step reduces to the ordinary gradient step, confirming (1).

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]
\label{ass:convexsmooth}
$f$ is convex and $L$‑smooth on $\R^{d}$.
\end{assumption}
\begin{assumption}[Convexity of $g$]
\label{ass:convexg}
$g$ is proper, closed and convex (no smoothness required).
\end{assumption}
\begin{assumption}[Stepsize]
\label{ass:stepsize}
The stepsize satisfies $0<\eta\le \frac{1}{L}$.
\end{assumption}
All subsequent results are derived under Assumptions \ref{ass:convexsmooth}–\ref{ass:stepsize}.

\section*{Main Convergence Theorem}
\begin{theorem}[O(1/k) convergence in function value]\label{thm:rate}
Let $\{x_{k}\}$ be generated by \eqref{1} with a stepsize $\eta\in(0,1/L]$.  
Denote $x^{\star}\in\arg\min F$ and $R:=\|x_{0}-x^{\star}\|$.  
Then for every $k\ge 1$,
\[
F(x_{k})-F(x^{\star})\;\le\;\frac{R^{2}}{2\eta\,k}.
\tag{2}
\]
\end{theorem}
Thus the proximal gradient method attains the optimal $O(1/k)$ rate for convex composite optimisation.

\section*{Theoretical Properties}
\begin{itemize}[noitemsep]
\item \textbf{Stability:} The iteration is a non‑expansive map when $\eta\le 1/L$, guaranteeing that the sequence $\{x_{k}\}$ remains bounded.
\item \textbf{Hyperparameter Sensitivity:} Choosing $\eta$ larger than $1/L$ may violate the descent property; the bound (2) deteriorates or even diverges.
\item \textbf{Comparison to Baselines:}
    \begin{enumerate}
    \item Compared with pure gradient descent on $f+g$ (when $g$ is smooth), PGM inherits the same $O(1/k)$ rate while allowing nonsmooth $g$.
    \item When $g\equiv0$, \eqref{1} reduces to classical gradient descent and Theorem \ref{thm:rate} recovers its standard rate.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The proof below relies crucially on the \emph{firm non‑expansiveness} of the proximal operator (Lemma~\ref{lem:firm}) and the smoothness inequality for $f$ (Lemma~\ref{lem:smooth}). No stochasticity or strong convexity is required; the result holds for any convex composite problem satisfying the listed assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Firm non‑expansiveness of $\prox_{\eta g}$]\label{lem:firm}
For any $u,v\in\R^{d}$,
\[
\bigl\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\bigr\|^{2}
\le \bigl\langle \prox_{\eta g}(u)-\prox_{\eta g}(v),\,u-v\bigr\rangle .
\tag{3}
\]
\end{lemma}
\begin{proof}
The proximal operator is the resolvent of the maximal monotone sub‑differential $\partial g$.  
Standard monotone‑operator theory yields the firm non‑expansiveness property (see, e.g., \cite{rockafellar1976}). ∎
\end{proof}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
The inequality follows from integrating the Lipschitz condition on $\nabla f$ (see \cite{nesterov2004}). ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout we fix an optimal point $x^{\star}\in\arg\min F$.

\noindent\textbf{Notation.} For brevity set 
\[
y_{k}=x_{k}-\eta\nabla f(x_{k}),\qquad
x_{k+1}= \prox_{\eta g}(y_{k}) .
\]

\noindent\textbf{Step 1 (Optimality condition of the proximal step).}\\
By definition of the proximal operator,
\[
0\in \partial g(x_{k+1})+\frac{1}{\eta}(x_{k+1}-y_{k}),
\]
hence there exists $s_{k+1}\in\partial g(x_{k+1})$ such that
\begin{equation}
\frac{1}{\eta}(y_{k}-x_{k+1})=s_{k+1}.
\tag{5}
\end{equation}
\textbf{[Step 1]}

\noindent\textbf{Step 2 (Convexity of $g$).}\\
Convexity of $g$ gives
\[
g(x^{\star})\ge g(x_{k+1})+\langle s_{k+1},x^{\star}-x_{k+1}\rangle .
\]
Using (5) we rewrite the inner product:
\[
\langle s_{k+1},x^{\star}-x_{k+1}\rangle
= \frac{1}{\eta}\langle y_{k}-x_{k+1},x^{\star}-x_{k+1}\rangle .
\tag{6}
\]
\textbf{[Step 2]}

\noindent\textbf{Step 3 (Expand the right‑hand side of (6)).}\\
Observe that
\[
\langle y_{k}-x_{k+1},x^{\star}-x_{k+1}\rangle
= \frac{1}{2}\Bigl(\|y_{k}-x^{\star}\|^{2}
-\|y_{k}-x_{k+1}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\tag{7}
\]
\textbf{[Step 3]}

\noindent\textbf{Step 4 (Combine convexity of $g$ with (6)–(7)).}\\
Multiplying (7) by $1/\eta$ and inserting into the inequality from Step 2 yields
\[
g(x_{k+1})-g(x^{\star})
\le \frac{1}{2\eta}\Bigl(
\|y_{k}-x^{\star}\|^{2}
-\|y_{k}-x_{k+1}\|^{2}
-\|x_{k+1}-x^{\star}\|^{2}
\Bigr).
\tag{8}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5 (Smoothness of $f$).}\\
Apply Lemma~\ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{9}
\]
\textbf{[Step 5]}

\noindent\textbf{Step 6 (Express $x_{k+1}-x_{k}$ via $y_{k}$).}\\
Since $y_{k}=x_{k}-\eta\nabla f(x_{k})$, we have
\[
x_{k+1}-x_{k}=x_{k+1}-y_{k}+\eta\nabla f(x_{k}).
\tag{10}
\]
\textbf{[Step 6]}

\noindent\textbf{Step 7 (Plug (10) into (9) and simplify).}\\
Using $\langle\nabla f(x_{k}),\eta\nabla f(x_{k})\rangle=\eta\|\nabla f(x_{k})\|^{2}$ and
$\|x_{k+1}-x_{k}\|^{2}
=\|x_{k+1}-y_{k}\|^{2}+2\eta\langle x_{k+1}-y_{k},\nabla f(x_{k})\rangle+\eta^{2}\|\nabla f(x_{k})\|^{2}$,
after elementary algebra we obtain
\[
f(x_{k+1})\le f(x_{k})
+\frac{1}{\eta}\langle y_{k}-x_{k+1},y_{k}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-y_{k}\|^{2}
-\Bigl(\frac{1}{\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-y_{k}\|^{2}.
\tag{11}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8 (Use identity $\langle y_{k}-x_{k+1},y_{k}-x_{k}\rangle
= \frac{1}{2}\bigl(\|y_{k}-x_{k}\|^{2}
-\|y_{k}-x_{k+1}\|^{2}
-\|x_{k+1}-x_{k}\|^{2}\bigr)$).}\\
Substituting this identity into (11) gives
\[
f(x_{k+1})\le f(x_{k})
+\frac{1}{2\eta}\Bigl(
\|y_{k}-x_{k}\|^{2}
-\|y_{k}-x_{k+1}\|^{2}
-\|x_{k+1}-x_{k}\|^{2}
\Bigr)
+\frac{L}{2}\|x_{k+1}-y_{k}\|^{2}
-\Bigl(\frac{1}{\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-y_{k}\|^{2}.
\tag{12}
\]
\textbf{[Step 8]}

\noindent\textbf{Step 9 (Cancel terms).}\\
Notice that $\|x_{k+1}-x_{k}\|^{2}= \|x_{k+1}-y_{k}+\eta\nabla f(x_{k})\|^{2}$, but the term
$-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}$ together with the last two terms simplifies to
\[
-\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
+\frac{L}{2}\|x_{k+1}-y_{k}\|^{2}
-\Bigl(\frac{1}{\eta}-\frac{L}{2}\Bigr)\|x_{k+1}-y_{k}\|^{2}
= -\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}.
\tag{13}
\]
Thus (12) reduces to
\[
f(x_{k+1})\le f(x_{k})
+\frac{1}{2\eta}\bigl(\|y_{k}-x_{k}\|^{2}
-\|y_{k}-x_{k+1}\|^{2}\bigr)
-\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}.
\tag{14}
\]
\textbf{[Step 9]}

\noindent\textbf{Step 10 (Combine $f$ and $g$).}\\
Add inequality (8) (Step 4) and (14). The terms $-\frac{1}{2\eta}\|x_{k+1}-y_{k}\|^{2}$ cancel, leaving
\[
\begin{aligned}
F(x_{k+1})-F(x^{\star})
&= \bigl[f(x_{k+1})-f(x^{\star})\bigr]
   +\bigl[g(x_{k+1})-g(x^{\star})\bigr]\\[2mm]
&\le \frac{1}{2\eta}
   \Bigl(\|y_{k}-x_{k}\|^{2}
        -\|y_{k}-x_{k+1}\|^{2}
        +\|y_{k}-x^{\star}\|^{2}
        -\|x_{k+1}-x^{\star}\|^{2}\Bigr).
\end{aligned}
\tag{15}
\]
\textbf{[Step 10]}

\noindent\textbf{Step 11 (Express $y_{k}$ in terms of $x_{k}$).}\\
Recall $y_{k}=x_{k}-\eta\nabla f(x_{k})$. Hence
\[
\|y_{k}-x_{k}\|^{2}= \eta^{2}\|\nabla f(x_{k})\|^{2}\ge 0,
\]
and more importantly,
\[
\|y_{k}-x^{\star}\|^{2}
= \|x_{k}-x^{\star}\|^{2}
-2\eta\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle
+\eta^{2}\|\nabla f(x_{k})\|^{2}.
\tag{16}
\]
\textbf{[Step 11]}

\noindent\textbf{Step 12 (Upper bound the mixed term).}\\
Using convexity of $f$,
\[
f(x_{k})-f(x^{\star})\le \langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle .
\tag{17}
\]
Thus $-2\eta\langle\nabla f(x_{k}),x_{k}-x^{\star}\rangle\le -2\eta\bigl(f(x_{k})-f(x^{\star})\bigr)$.
Insert (16) and (17) into (15). After cancelling the non‑negative $\eta^{2}\|\nabla f(x_{k})\|^{2}$ terms we obtain
\[
F(x_{k+1})-F(x^{\star})
\le \frac{1}{2\eta}
\bigl(\|x_{k}-x^{\star}\|^{2}
      -\|x_{k+1}-x^{\star}\|^{2}\bigr).
\tag{18}
\]
\textbf{[Step 12]}

\noindent\textbf{Step 13 (Telescoping sum).}\\
Sum inequality (18) for $k=0,\dots, K-1$:
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{\star})\bigr)
\le \frac{1}{2\eta}\bigl(\|x_{0}-x^{\star}\|^{2}
      -\|x_{K}-x^{\star}\|^{2}\bigr)
\le \frac{R^{2}}{2\eta},
\tag{19}
\]
where $R:=\|x_{0}-x^{\star}\|$.  
Since the left‑hand side contains $K$ non‑negative terms, the smallest of them is bounded by the average:
\[
K\bigl(F(x_{K})-F(x^{\star})\bigr)
\le \frac{R^{2}}{2\eta}.
\tag{20}
\]
Dividing by $K$ yields exactly the rate (2):
\[
F(x_{K})-F(x^{\star})\le\frac{R^{2}}{2\eta\,K}.
\qquad\blacksquare
\]
\textbf{[Step 13]}

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:rate}, choosing the maximal admissible stepsize $\eta=1/L$ gives the explicit bound
\[
F(x_{k})-F(x^{\star})\le \frac{L\,R^{2}}{2k}.
\]
If a smaller stepsize $\eta\in(0,1/L]$ is used, the constant scales as $1/(2\eta)$.

\section*{Special Cases}
\begin{enumerate}[noitemsep]
\item \textbf{Pure Gradient Descent ($g\equiv0$).}  
The proximal operator reduces to the identity, and the proof collapses to the classical $O(1/k)$ analysis for smooth convex $f$.
\item \textbf{Indicator Function $g=\iota_{C}$ of a convex set $C$.}  
Then $\prox_{\eta g}$ is the Euclidean projection onto $C$, and the method becomes the projected gradient method with the same $O(1/k)$ guarantee.
\item \textbf{Strongly Convex $F$.}  
If $F$ is $\mu$‑strongly convex, a standard refinement of inequality (18) yields a linear rate $O\bigl((1-\eta\mu)^{k}\bigr)$.
\end{enumerate}
