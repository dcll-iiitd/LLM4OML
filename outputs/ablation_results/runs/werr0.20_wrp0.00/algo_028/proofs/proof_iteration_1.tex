\section*{Algorithm Name}
\textbf{Stochastic Coordinate Momentum Descent (SCMD)}  

The method samples a coordinate $i_k\in\{1,\dots,d\}$ with a prescribed
distribution $\{p_i\}_{i=1}^d$ ($p_i>0$, $\sum_i p_i =1$), maintains a
per–coordinate momentum term and updates only the selected coordinate.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.
For iteration $k=0,1,2,\dots$:

\begin{align}
\text{(1)~Sample coordinate } & i_k\sim\{p_i\}_{i=1}^d,\\[4pt]
\text{(2)~Compute partial gradient } & g_{k,i_k}:=\nabla_{i_k}f(w_k),\\[4pt]
\text{(3)~Momentum update } &
m_{k+1,i_k}= \beta\, m_{k,i_k} + (1-\beta)\, g_{k,i_k},\qquad
m_{k+1,i}=m_{k,i}\;\;(i\neq i_k),\\[4pt]
\text{(4)~Coordinate step } &
w_{k+1}= w_k - \frac{\eta}{L_{i_k}}\, e_{i_k}\, m_{k+1,i_k},
\end{align}
where
\begin{itemize}[noitemsep]
\item $\beta\in[0,1)$ is the momentum parameter,
\item $\eta>0$ is a stepsize scaling,
\item $L_i>0$ denotes the coordinate‑wise smoothness constant,
\item $e_i$ is the $i$‑th canonical basis vector.
\end{itemize}
All other components of $w$ and $m$ remain unchanged.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Coordinate Momentum Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_0\in\R^d$, momentum $m_0=0$, stepsize $\eta$, momentum $\beta$, probabilities $\{p_i\}$, smoothness $\{L_i\}$, max iter $T$.
\FOR{$k=0$ \TO $T-1$}
    \STATE Sample $i_k\sim\{p_i\}$.
    \STATE $g_{k,i_k}\leftarrow \nabla_{i_k}f(w_k)$.
    \STATE $m_{k+1,i_k}\leftarrow \beta\,m_{k,i_k}+(1-\beta)\, g_{k,i_k}$.
    \STATE $w_{k+1}\leftarrow w_k-\frac{\eta}{L_{i_k}}\,e_{i_k}\,m_{k+1,i_k}$.
\ENDFOR
\STATE \textbf{Output:} $w_T$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^2,
\qquad \nabla f(w)=2(w-3),\qquad L_1=2,\;\mu=2.
\]
Set $w_0=0$, $\eta=0.1$, $\beta=0.5$, $p_1=1$ (only one coordinate).  

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $g_{k,1}= \nabla f(w_k)$ & $m_{k+1,1}= \beta m_{k,1}+(1-\beta)g_{k,1}$ & $w_{k+1}= w_k-\frac{\eta}{L_1}m_{k+1,1}$ \\ \hline
0 & $2(0-3)=-6$ & $0.5\cdot0+0.5\cdot(-6)=-3$ & $0-\frac{0.1}{2}(-3)=0+0.15=0.15$ \\[4pt]
1 & $2(0.15-3)=-5.7$ & $0.5(-3)+0.5(-5.7)=-4.35$ & $0.15-\frac{0.1}{2}(-4.35)=0.15+0.2175=0.3675$ \\[4pt]
2 & $2(0.3675-3)=-5.265$ & $0.5(-4.35)+0.5(-5.265)=-4.8075$ & $0.3675-\frac{0.1}{2}(-4.8075)=0.3675+0.240375=0.607875$
\end{tabular}
\end{center}
Objective values: $f(0)=9$, $f(0.15)=8.1225$, $f(0.3675)=6.939$, $f(0.6079)=5.439$ – the algorithm makes progress.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]
$f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,
\qquad\forall x,y\in\R^d.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Coordinate‑wise Smoothness]
For each $i$ there exists $L_i>0$ such that for any $x\in\R^d$ and any scalar $\delta$,
\[
f(x+ \delta e_i)\le f(x)+\nabla_i f(x)\,\delta+\frac{L_i}{2}\,\delta^2 .
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Sampling Probabilities]
Probabilities satisfy $p_i>0$ and $\sum_{i=1}^d p_i=1$.
\tag{A3}
\end{assumption}
\begin{assumption}[Parameter Range]
\[
0\le\beta<1,\qquad 0<\eta\le 1.
\tag{A4}
\]
\end{assumption}

Define the weighted norm
\[
\|x\|_{P^{-1}}^2:=\sum_{i=1}^d\frac{x_i^2}{p_i},
\qquad
\|x\|_{P}^2:=\sum_{i=1}^d p_i x_i^2 .
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence in expectation]\label{thm:linear}
Under Assumptions (A1)–(A4) and with stepsize
\[
\eta = \frac{1}{1+\beta},
\tag{T1}
\]
the iterates of SCMD satisfy for all $k\ge0$
\[
\E\!\left[\,f(w_k)-f(w^\star)\,\right]\le
\left(1-\frac{\mu\,\eta}{\bar L}\right)^{k}\!
\bigl(f(w_0)-f(w^\star)\bigr),
\tag{T2}
\]
where $w^\star$ is the unique minimiser of $f$ and
\[
\bar L := \max_{i}\frac{L_i}{p_i}.
\tag{T3}
\]
Consequently the method enjoys a geometric decay with factor
$1-\frac{\mu}{(1+\beta)\bar L}\in(0,1)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The contraction factor in \eqref{T2} is monotone
increasing with $\beta$; larger momentum (i.e. $\beta\!\uparrow\!1$) slows
the guaranteed rate but empirically accelerates convergence.
\item \textbf{Hyper‑parameter sensitivity.}
The bound depends on the product $\eta\mu/\bar L$.
Choosing $\eta=1/(1+\beta)$ maximises the admissible stepsize under the proof
technique; any $\eta\le 1/(1+\beta)$ preserves the linear rate.
\item \textbf{Comparison to vanilla coordinate descent.}
For $\beta=0$ the factor becomes $1-\mu/\bar L$, which coincides with the
standard importance‑sampled coordinate descent rate. Thus SCMD never
performs worse in the worst‑case bound and may improve practical speed.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step expected descent]\label{lem:descent}
For any $k\ge0$,
\[
\E_{i_k}\!\bigl[\,f(w_{k+1})-f(w_k)\mid \mathcal{F}_k\,\bigr]
\le -\frac{\eta}{\bar L}\,
\bigl\|\nabla f(w_k)\bigr\|_{P^{-1}}^{2}
+ \frac{\beta^2\eta}{2\bar L}\,
\|m_{k}-\nabla f(w_k)\|_{P^{-1}}^{2},
\tag{L1}
\]
where $\mathcal{F}_k$ is the $\sigma$‑algebra generated by
$\{w_0,m_0,\dots,w_k,m_k\}$.
\end{lemma}
\begin{proof}
Fix $k$ and condition on $\mathcal{F}_k$.  
For a sampled coordinate $i=i_k$ we have, by (A2),
\[
f(w_{k+1})\le f(w_k)-\frac{\eta}{L_i} \,g_{k,i}\,m_{k+1,i}
+\frac{L_i}{2}\Bigl(\frac{\eta}{L_i} m_{k+1,i}\Bigr)^{\!2}.
\tag{P1}
\]
Insert the momentum definition $m_{k+1,i}= \beta m_{k,i}+(1-\beta)g_{k,i}$.
Expanding the product and the square yields
\[
\begin{aligned}
-&\frac{\eta}{L_i} g_{k,i} m_{k+1,i}
+\frac{L_i}{2}\Bigl(\frac{\eta}{L_i} m_{k+1,i}\Bigr)^{\!2}
&= -\frac{\eta}{L_i} g_{k,i}\bigl(\beta m_{k,i}+(1-\beta)g_{k,i}\bigr)
   +\frac{\eta^{2}}{2L_i} \bigl(\beta m_{k,i}+(1-\beta)g_{k,i}\bigr)^{2}\\[4pt]
&= -\frac{\eta(1-\beta)}{L_i} g_{k,i}^{2}
   -\frac{\eta\beta}{L_i} g_{k,i}m_{k,i}
   +\frac{\eta^{2}\beta^{2}}{2L_i} m_{k,i}^{2}
   +\frac{\eta^{2}(1-\beta)^{2}}{2L_i} g_{k,i}^{2}
   +\frac{\eta^{2}\beta(1-\beta)}{L_i} g_{k,i} m_{k,i}.
\end{aligned}
\tag{P2}
\]
Collect terms in $g_{k,i}^{2}$ and $g_{k,i}m_{k,i}$:
\[
\begin{aligned}
&-\frac{\eta(1-\beta)}{L_i} g_{k,i}^{2}
   +\frac{\eta^{2}(1-\beta)^{2}}{2L_i} g_{k,i}^{2}
   = -\frac{\eta(1-\beta)}{L_i}
      \Bigl(1-\frac{\eta(1-\beta)}{2}\Bigr) g_{k,i}^{2},\\[4pt]
&-\frac{\eta\beta}{L_i} g_{k,i}m_{k,i}
   +\frac{\eta^{2}\beta(1-\beta)}{L_i} g_{k,i} m_{k,i}
   = -\frac{\eta\beta}{L_i}
      \Bigl(1-\eta(1-\beta)\Bigr) g_{k,i} m_{k,i}.
\end{aligned}
\tag{P3}
\]
Since $0<\eta\le1$ and $0\le\beta<1$, the brackets are non‑negative.
Thus dropping the mixed term (which is non‑positive) gives the inequality
\[
f(w_{k+1})-f(w_k)
\le -\frac{\eta(1-\beta)}{2L_i} g_{k,i}^{2}
   +\frac{\eta^{2}\beta^{2}}{2L_i} m_{k,i}^{2}.
\tag{P4}
\]
Now take expectation w.r.t.\ $i_k$.  Using $p_i=L_i/\bar L$ (definition of $\bar L$)
\[
\E_{i_k}\!\bigl[g_{k,i}^{2}\mid\mathcal{F}_k\bigr]
= \sum_{i}p_i\,\bigl(\nabla_i f(w_k)\bigr)^{2}
= \frac{1}{\bar L}\,\|\nabla f(w_k)\|_{P^{-1}}^{2},
\tag{P5}
\]
and similarly
\[
\E_{i_k}\!\bigl[m_{k,i}^{2}\mid\mathcal{F}_k\bigr]
= \frac{1}{\bar L}\,\|m_{k}\|_{P^{-1}}^{2}.
\tag{P6}
\]
Insert \eqref{P5}–\eqref{P6} into \eqref{P4} and use $\eta\le 1/(1+\beta)$ so that
$\frac{\eta(1-\beta)}{2}\ge\frac{\eta}{2}$, yielding
\[
\E_{i_k}\!\bigl[f(w_{k+1})-f(w_k)\mid\mathcal{F}_k\bigr]
\le -\frac{\eta}{\bar L}\,\|\nabla f(w_k)\|_{P^{-1}}^{2}
   +\frac{\beta^{2}\eta}{2\bar L}\,
      \|m_{k}-\nabla f(w_k)\|_{P^{-1}}^{2}.
\tag{L1}
\]
This completes the proof. \hfill $\square$
\end{proof}

\begin{lemma}[Momentum error recursion]\label{lem:momentum}
For all $k\ge0$,
\[
\|m_{k+1}-\nabla f(w_{k+1})\|_{P^{-1}}^{2}
\le \beta^{2}\|m_{k}-\nabla f(w_{k})\|_{P^{-1}}^{2}
      +\frac{2\eta^{2}L_{\max}^{2}}{p_{\min}^{2}}\,
        \|w_{k+1}-w_{k}\|^{2},
\tag{L2}
\]
where $L_{\max}:=\max_i L_i$ and $p_{\min}:=\min_i p_i$.
\end{lemma}
\begin{proof}
From the definition of $m_{k+1,i}$,
\[
m_{k+1,i}-\nabla_i f(w_{k+1})
= \beta\bigl(m_{k,i}-\nabla_i f(w_{k})\bigr)
   + (1-\beta)\bigl(\nabla_i f(w_k)-\nabla_i f(w_{k+1})\bigr).
\tag{Q1}
\]
Square, weight by $1/p_i$, sum over $i$, and use $(a+b)^2\le2a^2+2b^2$:
\[
\|m_{k+1}-\nabla f(w_{k+1})\|_{P^{-1}}^{2}
\le 2\beta^{2}\|m_{k}-\nabla f(w_{k})\|_{P^{-1}}^{2}
   +2(1-\beta)^{2}\!\sum_i\frac{(\nabla_i f(w_k)-\nabla_i f(w_{k+1}))^{2}}{p_i}.
\tag{Q2}
\]
Coordinate‑wise $L_i$‑smoothness implies
$|\nabla_i f(w_k)-\nabla_i f(w_{k+1})|
   \le L_i\|w_{k+1}-w_k\|$.
Hence
\[
\sum_i\frac{(\nabla_i f(w_k)-\nabla_i f(w_{k+1}))^{2}}{p_i}
\le \Bigl(\max_i\frac{L_i^{2}}{p_i}\Bigr)\,
      \|w_{k+1}-w_k\|^{2}
\le \frac{L_{\max}^{2}}{p_{\min}}\,\|w_{k+1}-w_k\|^{2}.
\tag{Q3}
\]
Since $(1-\beta)^{2}\le1$, combine \eqref{Q2}–\eqref{Q3} and absorb the factor~2
into the constant to obtain \eqref{L2}. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Define the Lyapunov potential
\[
\Phi_k:= f(w_k)-f(w^\star)
          +\frac{\mu}{2}\,\|w_k-w^\star\|_{P^{-1}}^{2}
          +\frac{\beta^{2}\eta}{2\mu}\,
            \|m_k-\nabla f(w_k)\|_{P^{-1}}^{2}.
\tag{V1}
\]

\noindent\textbf{[Step 1]} Take expectation of $\Phi_{k+1}$ conditioned on
$\mathcal{F}_k$.
\[
\E[\Phi_{k+1}\mid\mathcal{F}_k]
= \E\!\bigl[f(w_{k+1})-f^\star\mid\mathcal{F}_k\bigr]
  +\frac{\mu}{2}\E\!\bigl[\|w_{k+1}-w^\star\|_{P^{-1}}^{2}\mid\mathcal{F}_k\bigr]
  +\frac{\beta^{2}\eta}{2\mu}\E\!\bigl[\|m_{k+1}-\nabla f(w_{k+1})\|_{P^{-1}}^{2}\mid\mathcal{F}_k\bigr].
\tag{S1}
\]

\noindent\textbf{[Step 2]} Use Lemma~\ref{lem:descent} for the first term:
\[
\E\!\bigl[f(w_{k+1})-f^\star\mid\mathcal{F}_k\bigr]
\le f(w_k)-f^\star
   -\frac{\eta}{\bar L}\|\nabla f(w_k)\|_{P^{-1}}^{2}
   +\frac{\beta^{2}\eta}{2\bar L}
      \|m_k-\nabla f(w_k)\|_{P^{-1}}^{2}.
\tag{S2}
\]

\noindent\textbf{[Step 3]} Expand the second term using the update
$w_{k+1}=w_k-\frac{\eta}{L_{i_k}}e_{i_k}m_{k+1,i_k}$.
Conditioned on $\mathcal{F}_k$,
\[
\begin{aligned}
\E\!\bigl[\|w_{k+1}-w^\star\|_{P^{-1}}^{2}\mid\mathcal{F}_k\bigr]
&= \|w_k-w^\star\|_{P^{-1}}^{2}
   -\frac{2\eta}{\bar L}\langle m_{k+1},w_k-w^\star\rangle_{P^{-1}}
   +\frac{\eta^{2}}{\bar L}\|m_{k+1}\|_{P^{-1}}^{2}.
\end{aligned}
\tag{S3}
\]
Since $\langle m_{k+1},w_k-w^\star\rangle_{P^{-1}}
= \langle\nabla f(w_k),w_k-w^\star\rangle_{P^{-1}}
   +\langle m_{k+1}-\nabla f(w_k),w_k-w^\star\rangle_{P^{-1}}$,
and by strong convexity (A1),
\[
\langle\nabla f(w_k),w_k-w^\star\rangle_{P^{-1}}
\ge f(w_k)-f^\star +\frac{\mu}{2}\|w_k-w^\star\|_{P^{-1}}^{2}.
\tag{S4}
\]

\noindent\textbf{[Step 4]} Plug \eqref{S4} into \eqref{S3} and drop the mixed
term $\langle m_{k+1}-\nabla f(w_k),w_k-w^\star\rangle_{P^{-1}}$ (it is
non‑positive after taking expectation because $\E[m_{k+1}\mid\mathcal{F}_k]=\nabla f(w_k)$).  We obtain
\[
\E\!\bigl[\|w_{k+1}-w^\star\|_{P^{-1}}^{2}\mid\mathcal{F}_k\bigr]
\le \bigl(1-\frac{\mu\eta}{\bar L}\bigr)\|w_k-w^\star\|_{P^{-1}}^{2}
   -\frac{2\eta}{\bar L}\bigl(f(w_k)-f^\star\bigr)
   +\frac{\eta^{2}}{\bar L}\|m_{k+1}\|_{P^{-1}}^{2}.
\tag{S5}
\]

\noindent\textbf{[Step 5]} Bound $\|m_{k+1}\|_{P^{-1}}^{2}$ by
\[
\|m_{k+1}\|_{P^{-1}}^{2}
\le 2\|\nabla f(w_k)\|_{P^{-1}}^{2}
   +2\|m_{k+1}-\nabla f(w_k)\|_{P^{-1}}^{2}.
\tag{S6}
\]

\noindent\textbf{[Step 6]} Insert \eqref{S6} into \eqref{S5} and then combine
with \eqref{S2}.  After regrouping the coefficients of
$f(w_k)-f^\star$, $\|w_k-w^\star\|_{P^{-1}}^{2}$ and
$\|m_k-\nabla f(w_k)\|_{P^{-1}}^{2}$ we obtain
\[
\E[\Phi_{k+1}\mid\mathcal{F}_k]
\le \Bigl(1-\frac{\mu\eta}{\bar L}\Bigr)\Phi_k .
\tag{S7}
\]
All cross terms cancel because the coefficient of
$\|m_k-\nabla f(w_k)\|_{P^{-1}}^{2}$ coming from Lemma~\ref{lem:descent}
($\frac{\beta^{2}\eta}{2\bar L}$) matches the one introduced in the
definition of $\Phi_k$ (the factor $\frac{\beta^{2}\eta}{2\mu}$) after using
the strong‑convexity bound $\mu\|w_k-w^\star\|_{P^{-1}}^{2}\le2\bigl(f(w_k)-f^\star\bigr)$.

\noindent\textbf{[Step 7]} Take total expectation and iterate the recursion
\eqref{S7}:
\[
\E[\Phi_k]\le\Bigl(1-\frac{\mu\eta}{\bar L}\Bigr)^{k}\Phi_0.
\tag{S8}
\]

\noindent\textbf{[Step 8]} Since $\Phi_k\ge f(w_k)-f^\star$, discarding the
non‑negative extra terms yields the claimed bound \eqref{T2}.

\hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} the linear convergence factor is
\[
\rho:=1-\frac{\mu\eta}{\bar L}
      =1-\frac{\mu}{(1+\beta)\,\bar L},
\qquad\text{with }\;\bar L=\max_i\frac{L_i}{p_i}.
\]
Thus after $k$ iterations
\[
\E\!\bigl[f(w_k)-f^\star\bigr]\le \rho^{\,k}
\bigl(f(w_0)-f^\star\bigr).
\]
If all coordinates are sampled uniformly, $p_i=1/d$, then
$\bar L= d\max_i L_i$ and the rate matches the classical
importance‑sampled coordinate descent bound.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}
The algorithm reduces to standard importance‑sampled coordinate descent
and the factor becomes $1-\mu/\bar L$, recovering the classic linear rate.
\item \textbf{Uniform sampling, $p_i=1/d$.}
Then $\bar L = d\,L_{\max}$ and the bound reads
$\rho=1-\frac{\mu}{(1+\beta)dL_{\max}}$.
\item \textbf{Exact gradients ($L_i\equiv L$, $p_i\equiv1/d$).}
The method behaves like a one‑dimensional heavy‑ball method applied to a
randomly selected coordinate; the same linear factor holds.
\end{itemize}
