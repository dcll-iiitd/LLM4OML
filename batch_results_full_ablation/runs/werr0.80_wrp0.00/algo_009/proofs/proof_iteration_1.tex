\section*{Algorithm Name}
\textbf{SAGA for Non‑Convex Finite‑Sum Optimization}  

We consider the finite‑sum problem  

\[
\min_{x\in\R^{d}} \; f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]

where each component $f_i:\R^{d}\to\R$ is $L$–smooth (possibly non‑convex).

--------------------------------------------------------------------

\section*{Mathematical Formulation}
Let $\phi_i\in\R^{d}$ denote the last point at which the gradient of $f_i$ was stored.  
Define the \emph{control variate}  

\[
\bar{g}_k \;:=\; \frac{1}{n}\sum_{j=1}^{n}\nabla f_j(\phi_j^k).
\]

At iteration $k$ we sample an index $i_k$ uniformly from $\{1,\dots ,n\}$ and form the unbiased gradient estimator  

\[
g_k \;:=\; \nabla f_{i_k}(x_k)\;-\;\nabla f_{i_k}(\phi_{i_k}^k)\;+\;\bar{g}_k .
\tag{1}
\]

The primal iterate is updated by a constant stepsize $\alpha>0$:

\[
x_{k+1}\;=\;x_k\;-\;\alpha\, g_k .
\tag{2}
\]

Finally the table entry for the sampled component is refreshed:

\[
\phi_{i_k}^{k+1}\;=\;x_k,\qquad
\phi_{j}^{k+1}\;=\;\phi_{j}^{k}\;\;(j\neq i_k).
\tag{3}
\]

The algorithm is completely specified by the tuple $(\alpha,n)$ and the
initialization $x_0$ together with $\phi_i^0=x_0$ for all $i$.

--------------------------------------------------------------------

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA (non‑convex version)}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\alpha>0$, number of components $n$, max. iterations $T$
\State Initialize $x_0\in\R^{d}$, set $\phi_i^0\gets x_0$ for $i=1,\dots ,n$
\State Compute $\bar g_0\gets \frac1n\sum_{i=1}^{n}\nabla f_i(\phi_i^0)$
\For{$k=0,\dots ,T-1$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots ,n\}$
    \State $g_k \gets \nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)+\bar g_k$
    \State $x_{k+1}\gets x_k-\alpha\,g_k$ \Comment{primal update}
    \State $\phi_{i_k}^{k+1}\gets x_k$ \Comment{refresh table entry}
    \State Update the control variate:
          \[
          \bar g_{k+1}\gets \bar g_k
                 -\frac{1}{n}\bigl(\nabla f_{i_k}(\phi_{i_k}^k)-\nabla f_{i_k}(\phi_{i_k}^{k+1})\bigr)
          \]
\EndFor
\State \textbf{Output:} $\{x_k\}_{k=0}^{T}$ (or a randomly selected iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate the algorithm on a single‑component quadratic  

\[
f(w)=\bigl(w-3\bigr)^2,\qquad n=1,
\]

so that $f_1\equiv f$ and $\phi_1^k$ is the only table entry.
The gradient is $\nabla f(w)=2(w-3)$.  
Choose stepsize $\alpha=0.1$ and initialise $w_0=0$; consequently $\phi_1^0=w_0=0$ and  

\[
\bar g_0=\nabla f(\phi_1^0)=2(0-3)=-6 .
\]

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $k=0$ (the only index $i_0=1$ is selected):
\[
g_0 = \nabla f(w_0)-\nabla f(\phi_1^0)+\bar g_0
    = (-6)-(-6)+(-6) = -6 .
\]
Update:
\[
w_1 = w_0-\alpha g_0 = 0-0.1(-6)=0.6 .
\]
Refresh table: $\phi_1^{1}=w_0=0$ (unchanged).  
Control‑variate update:
\[
\bar g_{1}= \bar g_0-\frac{1}{1}\bigl(\nabla f(\phi_1^0)-\nabla f(\phi_1^{1})\bigr)
          = -6-(-6+6)= -6 .
\]
Objective value: $f(w_1) = (0.6-3)^2 = 5.76$.

\item $k=1$:
\[
g_1 = \nabla f(w_1)-\nabla f(\phi_1^1)+\bar g_1
    = 2(0.6-3)-(-6)+(-6)= -4.8+6-6 = -4.8 .
\]
Update:
\[
w_2 = w_1-\alpha g_1 = 0.6-0.1(-4.8)=1.08 .
\]
Refresh: $\phi_1^{2}=w_1=0.6$.  
Control‑variate:
\[
\bar g_{2}= \bar g_1-\bigl(\nabla f(\phi_1^1)-\nabla f(\phi_1^{2})\bigr)
          = -6- \bigl((-6)-2(0.6-3)\bigr)
          = -6-(-6+4.8)= -4.8 .
\]
Objective: $f(w_2) = (1.08-3)^2 = 3.6864$.

\item $k=2$:
\[
g_2 = \nabla f(w_2)-\nabla f(\phi_1^{2})+\bar g_2
    = 2(1.08-3)-2(0.6-3)+(-4.8)
    = -3.84+4.8-4.8 = -3.84 .
\]
Update:
\[
w_3 = w_2-\alpha g_2 = 1.08-0.1(-3.84)=1.464 .
\]
Refresh: $\phi_1^{3}=w_2=1.08$.  
Control‑variate:
\[
\bar g_{3}= \bar g_2-\bigl(\nabla f(\phi_1^{2})-\nabla f(\phi_1^{3})\bigr)
          = -4.8-\bigl(2(0.6-3)-2(1.08-3)\bigr)
          = -4.8-(-4.8+3.84)= -3.84 .
\]
Objective: $f(w_3) = (1.464-3)^2 \approx 2.366 .
\]

The iterates move steadily toward the minimiser $w^{*}=3$, and the
gradient estimator $g_k$ tracks the true gradient with decreasing variance. 
\end{enumerate}

\section*{Assumptions}
\begin{assumption}[Smoothness]
Each component $f_i$ is $L$–smooth, i.e., for all $x,y\in\R^{d}$,
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\;\le\;L\|x-y\|.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Finite Sum]
The objective is a finite average $f(x)=\frac1n\sum_{i=1}^{n}f_i(x)$.
\tag{A2}
\end{assumption}
\begin{assumption}[Bounded Below]
There exists $f^{*}>-\infty$ such that $f(x)\ge f^{*}$ for all $x$.
\tag{A3}
\end{assumption}
\begin{assumption}[Stepsize]
The constant stepsize satisfies 
\[
0<\alpha\le \frac{1}{3L}.
\tag{A4}
\]
\end{assumption}

--------------------------------------------------------------------

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SAGA convergence]\label{thm:saga}
Under Assumptions (A1)–(A4) and with the iterate sequence $\{x_k\}$ generated by
the algorithm, let $x^{*}$ be any global minimiser of $f$. Then for any
$T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^{*}\bigr)}{\alpha T}.
\tag{4}
\]
Consequently $\min_{0\le k<T}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]=O(1/T)$.
\end{theorem}

--------------------------------------------------------------------

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Unbiasedness.} By construction,
\[
\E_{i_k}[g_k\mid x_k,\{\phi_i^k\}]=\nabla f(x_k).
\tag{P1}
\]
\item \textbf{Variance control.} Using $L$‑smoothness we obtain
\[
\E_{i_k}\!\bigl[\|g_k\|^{2}\mid x_k,\{\phi_i^k\}\bigr]
\;\le\;
4L\bigl(f(x_k)-f^{*}\bigr)
   +\frac{4L}{n}\sum_{i=1}^{n}\bigl(f(\phi_i^k)-f^{*}\bigr).
\tag{P2}
\]
\item \textbf{Lyapunov function.} Define
\[
\Psi_k \;:=\; f(x_k)-f^{*}
   +\frac{1}{\alpha n}\sum_{i=1}^{n}\bigl(f(\phi_i^k)-f^{*}\bigr).
\tag{P3}
\]
Under (A4) the recursion
\[
\E[\Psi_{k+1}]\;\le\;\E[\Psi_{k}]
          -\frac{\alpha}{2}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\tag{P4}
\]
holds, which yields the $O(1/T)$ bound after telescoping.

\item \textbf{Comparison.} Compared with plain SGD, SAGA enjoys a
variance reduction term that eliminates the $\sigma^{2}$ floor, achieving
the same $O(1/T)$ rate without diminishing stepsizes. In the convex case it
recovers the linear convergence rate of the original SAGA analysis.

\item \textbf{Hyperparameter sensitivity.} The stepsize bound $\alpha\le 1/(3L)$
is tight for the proof technique; in practice larger steps are often safe,
but violating (A4) may break the descent inequality (P4).
\end{enumerate}

\section*{Discussion}
The theorem shows that, despite the non‑convexity of the finite‑sum,
the SAGA estimator drives the expected squared gradient norm to zero at a
rate comparable to the optimal stochastic first‑order rate for smooth
non‑convex optimisation. The key is the control‑variate term that keeps the
variance of $g_k$ bounded by a quantity proportional to the current suboptimality
of both the primal iterate and the stored table points. The Lyapunov function
$\Psi_k$ couples these two sources of error and yields a clean linear
recursion.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for any $x,y$,
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{L1}
\]
\end{lemma}
\begin{lemma}[Descent lemma for unbiased estimator]\label{lem:descent}
Let $g_k$ be any unbiased estimator of $\nabla f(x_k)$, i.e.
$\E[g_k]=\nabla f(x_k)$. Then for $L$‑smooth $f$,
\[
\E\!\bigl[f(x_{k+1})\mid x_k\bigr]
\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
      +\frac{L\alpha^{2}}{2}\E\!\bigl[\|g_k\|^{2}\mid x_k\bigr].
\tag{L2}
\]
\end{lemma}
\begin{proof}
From Lemma~\ref{lem:smooth} with $y=x_{k+1}=x_k-\alpha g_k$ we have
\[
f(x_{k+1})\le f(x_k)-\alpha\langle\nabla f(x_k),g_k\rangle
               +\frac{L\alpha^{2}}{2}\|g_k\|^{2}.
\]
Taking conditional expectation and using $\E[g_k]=\nabla f(x_k)$ yields (L2).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:saga}. Throughout the proof the expectation
$\E[\cdot]$ is taken over all random choices up to the current iteration.

\noindent\textbf{Step 1. Unbiasedness of the estimator.}  
From the definition (1) and uniform sampling,
\[
\E_{i_k}[g_k\mid x_k,\{\phi_i^k\}]
 =\frac{1}{n}\sum_{i=1}^{n}
   \bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i^k)+\bar g_k\bigr)
 =\nabla f(x_k). \tag{S1}
\]

\noindent\textbf{Step 2. Bounding the second moment of $g_k$.}  
Using $(a+b)^{2}\le 2a^{2}+2b^{2}$,
\[
\|g_k\|^{2}
\le 2\|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)\|^{2}
   +2\|\bar g_k\|^{2}. \tag{S2}
\]
By $L$‑smoothness,
\[
\|\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k}^k)\|^{2}
\le L^{2}\|x_k-\phi_{i_k}^k\|^{2}. \tag{S3}
\]
Moreover, using the convexity of the squared norm and the definition of $\bar g_k$,
\[
\|\bar g_k\|^{2}
= \Bigl\|\frac{1}{n}\sum_{j=1}^{n}\nabla f_j(\phi_j^k)\Bigr\|^{2}
\le \frac{1}{n}\sum_{j=1}^{n}\|\nabla f_j(\phi_j^k)\|^{2}. \tag{S4}
\]
Applying smoothness again,
\[
\|\nabla f_j(\phi_j^k)\|^{2}
\le 2L\bigl(f_j(\phi_j^k)-f_j^{*}\bigr)
     +2\|\nabla f_j(x^{*})\|^{2}
\le 2L\bigl(f_j(\phi_j^k)-f^{*}\bigr), \tag{S5}
\]
where $f^{*}=\min_x f(x)$ and we used $f_j^{*}\ge f^{*}$.
Combining (S2)–(S5) and taking expectation over $i_k$ yields
\[
\E_{i_k}\!\bigl[\|g_k\|^{2}\mid x_k,\{\phi_i^k\}\bigr]
\le 4L\bigl(f(x_k)-f^{*}\bigr)
   +\frac{4L}{n}\sum_{i=1}^{n}\bigl(f(\phi_i^k)-f^{*}\bigr). \tag{S6}
\]

\noindent\textbf{Step 3. One‑step descent of the primal objective.}  
Apply Lemma~\ref{lem:descent} with $g_k$ and use (S6):
\[
\begin{aligned}
\E\!\bigl[f(x_{k+1})\mid x_k,\{\phi_i^k\}\bigr]
&\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +\frac{L\alpha^{2}}{2}\E\!\bigl[\|g_k\|^{2}\mid\cdot\bigr] \\
&\le f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
   +2L\alpha^{2}\Bigl(f(x_k)-f^{*}\Bigr)
   +\frac{2L\alpha^{2}}{n}\sum_{i=1}^{n}\bigl(f(\phi_i^k)-f^{*}\bigr).
\end{aligned}
\tag{S7}
\]

\noindent\textbf{Step 4. Evolution of the table entries.}  
Recall that only $\phi_{i_k}$ is replaced by $x_k$.
Hence for any $i$,
\[
\E\!\bigl[f(\phi_i^{k+1})\mid x_k,\{\phi_j^k\}\bigr]
=
\begin{cases}
f(x_k), & i=i_k,\\[2mm]
f(\phi_i^k), & i\neq i_k .
\end{cases}
\]
Taking expectation over $i_k$,
\[
\E\!\bigl[f(\phi_i^{k+1})\mid x_k,\{\phi_j^k\}\bigr]
= \frac{1}{n}f(x_k)+\Bigl(1-\frac{1}{n}\Bigr)f(\phi_i^k). \tag{S8}
\]

Summing (S8) over $i$ and dividing by $n$ gives the recursion for the
average table loss:
\[
\frac{1}{n}\sum_{i=1}^{n}\E\!\bigl[f(\phi_i^{k+1})\mid\cdot\bigr]
= \frac{1}{n}f(x_k)+\Bigl(1-\frac{1}{n}\Bigr)
   \frac{1}{n}\sum_{i=1}^{n}f(\phi_i^k). \tag{S9}
\]

\noindent\textbf{Step 5. Lyapunov function descent.}  
Define the Lyapunov quantity (P3):
\[
\Psi_k := f(x_k)-f^{*}
          +\frac{1}{\alpha n}\sum_{i=1}^{n}\bigl(f(\phi_i^k)-f^{*}\bigr).
\tag{S10}
\]

Take total expectation of $\Psi_{k+1}$ conditioned on the history.
Using (S7) for the first term and (S9) for the second term,
\[
\begin{aligned}
\E[\Psi_{k+1}]
&\le \E\!\Bigl[
        f(x_k)-\alpha\|\nabla f(x_k)\|^{2}
        +2L\alpha^{2}\bigl(f(x_k)-f^{*}\bigr)
        +\frac{2L\alpha^{2}}{n}\sum_{i} (f(\phi_i^k)-f^{*}) \\
&\qquad\qquad\qquad
        +\frac{1}{\alpha n}\Bigl(
                \frac{1}{n}f(x_k)
                +\bigl(1-\frac{1}{n}\bigr)\sum_{i}f(\phi_i^k)
                -n f^{*}
              \Bigr)
      \Bigr] .
\end{aligned}
\tag{S11}
\]

Collect the coefficients of $f(x_k)-f^{*}$ and $\sum_i(f(\phi_i^k)-f^{*})$.
The coefficient in front of $f(x_k)-f^{*}$ is
\[
1 + \frac{1}{\alpha n^{2}} + 2L\alpha^{2}
= 1 - \alpha\; \underbrace{\bigl(\,1 - 2L\alpha - \tfrac{1}{\alpha n^{2}}\,\bigr)}_{\ge 0\;\text{by (A4)}} .
\tag{S12}
\]
Similarly, the coefficient of $\frac{1}{n}\sum_i(f(\phi_i^k)-f^{*})$ becomes
\[
\frac{1}{\alpha n}\Bigl(1-\frac{1}{n}\Bigr) + \frac{2L\alpha^{2}}{n}
= \frac{1}{\alpha n} -\underbrace{\Bigl(\frac{1}{\alpha n^{2}}-2L\alpha^{2}\Bigr)}_{\ge 0\;\text{by (A4)}} .
\tag{S13}
\]

Because $\alpha\le 1/(3L)$, both subtractions are non‑negative, thus
\[
\E[\Psi_{k+1}]
\le \E[\Psi_{k}]
     -\alpha\,\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr].
\tag{S14}
\]

\noindent\textbf{Step 6. Telescoping sum.}  
Summing (S14) over $k=0,\dots ,T-1$ yields
\[
\sum_{k=0}^{T-1}\alpha\,\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le \Psi_{0}-\E[\Psi_{T}]
\le \Psi_{0}.
\tag{S15}
\]

Recall $\Psi_{0}=f(x_{0})-f^{*}+\frac{1}{\alpha n}\sum_{i}\bigl(f(\phi_i^{0})-f^{*}\bigr)
= \frac{1}{\alpha}\bigl(f(x_{0})-f^{*}\bigr)$ because $\phi_i^{0}=x_{0}$.
Thus (S15) becomes
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha T},
\tag{S16}
\]
which is exactly the bound (4) claimed in Theorem~\ref{thm:saga}.
\qed

--------------------------------------------------------------------

\section*{Rate Derivation (With Constants)}
From (S16) we read the explicit constant $C=2/\alpha$:
\[
\boxed{\;
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le \frac{2}{\alpha T}\bigl(f(x_{0})-f^{*}\bigr)
\;}
\]
Choosing the maximal admissible stepsize $\alpha=1/(3L)$ gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_k)\|^{2}\bigr]
\le 6L\,\frac{f(x_{0})-f^{*}}{T}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f_i$.} When each $f_i$ is convex, the same Lyapunov
argument yields a linear convergence rate for strongly convex $f$,
recovering the classic SAGA result.
\item \textbf{Single component ($n=1$).} The algorithm reduces to ordinary
gradient descent with stepsize $\alpha$, and (S14) becomes the standard
smooth‑non‑convex descent inequality.
\item \textbf{Large minibatch.} If a minibatch of size $b$ is sampled,
the variance term in (S6) is divided by $b$, allowing a proportionally
larger stepsize $\alpha\le \frac{b}{3L}$ while preserving (S14).
\end{itemize}
