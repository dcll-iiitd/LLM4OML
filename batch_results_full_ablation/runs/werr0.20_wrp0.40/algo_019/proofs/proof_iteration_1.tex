\section*{Algorithm Name}
\textbf{FedProx (Federated Proximal Gradient Descent)}  

Local stochastic gradient descent is performed on each of $K$ participating clients for $E$ local epochs.  
Each client $k$ solves a proximal‐regularized local objective
\[
\min_{x\in\R^d}\; f_k(x)\;+\;\frac{\mu}{2}\|x-x^{(r)}\|^2,
\]
where $x^{(r)}$ is the global model at communication round $r$ and $\mu\ge 0$ is the proximal coefficient.  
After $E$ local updates the clients send their final local models to the server, which averages them:
\[
x^{(r+1)} \;=\; \frac1K\sum_{k=1}^{K} x_k^{(r,E)} .
\]

\section*{Mathematical Formulation}
Let $f_k:\R^d\to\R$ be the (convex, $L$‑smooth) loss on client $k$.  
For a given communication round $r$ and local iteration $e\in\{0,\dots,E-1\}$, client $k$ performs
\begin{align}
g_{k}^{(r,e)} &\;:=\; \nabla f_k\!\big(x_k^{(r,e)}; \xi_{k}^{(r,e)}\big) \quad\text{(stochastic gradient)}\\
x_k^{(r,e+1)} &\;:=\; x_k^{(r,e)} \;-\; \eta \Big( g_{k}^{(r,e)} \;+\; \mu\big(x_k^{(r,e)}-x^{(r)}\big) \Big),
\tag{1}
\end{align}
with step size $\eta>0$.  
The local model after $E$ steps is denoted $x_k^{(r,E)}$.  
Server aggregation:
\[
x^{(r+1)} \;=\; \frac{1}{K}\sum_{k=1}^{K} x_k^{(r,E)} . \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx}
\begin{algorithmic}[1]
\Require Number of clients $K$, local epochs $E$, step size $\eta$, proximal coefficient $\mu$, initial model $x^{(0)}$.
\For{communication round $r=0,1,\dots,R-1$}
    \State Server broadcasts $x^{(r)}$ to all clients.
    \For{each client $k=1,\dots,K$ \textbf{in parallel}}
        \State $x_k^{(r,0)} \gets x^{(r)}$
        \For{local epoch $e=0$ to $E-1$}
            \State Sample minibatch $\xi_{k}^{(r,e)}$
            \State $g_{k}^{(r,e)} \gets \nabla f_k\!\big(x_k^{(r,e)}; \xi_{k}^{(r,e)}\big)$
            \State $x_k^{(r,e+1)} \gets x_k^{(r,e)}-\eta\big(g_{k}^{(r,e)}+\mu(x_k^{(r,e)}-x^{(r)})\big)$
        \EndFor
        \State Send $x_k^{(r,E)}$ to server.
    \EndFor
    \State $x^{(r+1)} \gets \frac{1}{K}\sum_{k=1}^{K} x_k^{(r,E)}$
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single client ($K=1$) with scalar objective
\[
f(w) = (w-3)^2 .
\]
Hence $\nabla f(w)=2(w-3)$.  
Set $\eta=0.1$, $\mu=0.5$, and initialise $w^{(0)}=0$.  
We run three communication rounds ($E=1$ local step per round).

\begin{center}
\begin{tabular}{c|c|c|c}
Round $r$ & $g^{(r,0)}$ & $w^{(r,1)}$ (after (1)) & $f\big(w^{(r,1)}\big)$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1\big(-6+0.5(0-0)\big)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1\big(-4.8+0.5(0.6-0.6)\big)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1\big(-3.84+0.5(1.08-1.08)\big)=1.464$ & $(1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}
The global model after each round equals the client’s final local model because $K=1$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
Each local loss $f_k$ is convex and $L$‑smooth, i.e.
\[
f_k(y)\le f_k(x)+\langle \nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,\qquad\forall x,y\in\R^d .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradients]\label{ass:unbiased}
For any $x$, the stochastic gradient satisfies
\[
\E_{\xi}[\,\nabla f_k(x;\xi)\mid x\,]=\nabla f_k(x),\qquad 
\E_{\xi}\big[\|\nabla f_k(x;\xi)-\nabla f_k(x)\|^2\mid x\big]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Dissimilarity]\label{ass:dissim}
There exists $B\ge0$ such that for all $x$,
\[
\frac{1}{K}\sum_{k=1}^{K}\|\nabla f_k(x)-\nabla f(x)\|^2\le B^2\|\nabla f(x)\|^2,
\]
where $f(x)=\frac1K\sum_k f_k(x)$.
\end{assumption}
\begin{assumption}[Proximal Parameter]\label{ass:mu}
The proximal coefficient satisfies $0\le \mu\le L$.
\end{assumption}
All constants $L,\sigma,B,\mu$ are known a priori.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of FedProx]\label{thm:main}
Let Assumptions~\ref{ass:convex}–\ref{ass:mu} hold.  
Choose a constant step size
\[
\eta = \frac{1}{L+\mu}\, .
\]
After $R$ communication rounds (each with $E$ local steps) the averaged global model $x^{(R)}$ satisfies
\[
\E\!\big[ f\!\big(x^{(R)}\big)-f(x^\star) \big]
\;\le\;
\frac{2 (L+\mu)\|x^{(0)}-x^\star\|^2}{ER}
\;+\;
\frac{2\sigma^2}{\mu ER}
\;+\;
\frac{2L B^2}{\mu}\,\frac{1}{R},
\tag{3}
\]
where $x^\star$ is a global minimiser of $f$.  
Consequently, for fixed $E$, the method attains an $O\!\big(\tfrac{1}{R}\big)$ sub‑optimality rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The proximal term $\mu\|x-x^{(r)}\|^2$ penalises drift of local iterates, guaranteeing that the recursion \eqref{eq:local} remains a contraction for any $\eta\le1/(L+\mu)$.
\item \textbf{Hyper‑parameter Sensitivity.} Increasing $\mu$ reduces the variance term $\frac{2\sigma^2}{\mu ER}$ but enlarges the smoothness constant $(L+\mu)$, thus slowing the deterministic contraction. The optimal $\mu$ balances the two contributions.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item Setting $\mu=0$ recovers standard local SGD; the bound deteriorates to $O\big(\tfrac{1}{\sqrt{ER}}\big)$ under heterogeneous data (large $B$).
        \item With $\mu>0$, the heterogeneity penalty $2L B^2/(\mu R)$ replaces the $\sqrt{R}$ dependence, yielding linear convergence in the convex case.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that FedProx enjoys the same $O(1/R)$ rate as centralized SGD when data are i.i.d. ($B=0$). Under non‑i.i.d. data the proximal term controls the drift between local and global models, preventing the divergence observed for plain local SGD. The bound is explicit in all problem constants, offering a guideline for choosing $\eta$ and $\mu$ in practice.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
For any $L$‑smooth function $h$,
\[
h(y)\le h(x)+\langle\nabla h(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{lemma}[Three‑Point Identity for Proximal Term]\label{lem:prox}
For any $a,b\in\R^d$ and $\mu\ge0$,
\[
\|a-b\|^2 = \|a-c\|^2 + \|c-b\|^2 -2\langle a-c, b-c\rangle ,
\]
with $c$ taken as the current global model $x^{(r)}$.
\end{lemma}
\begin{lemma}[Descent Lemma for One Local Step]\label{lem:descent}
Let $x_{k}^{(r,e+1)}$ be defined by \eqref{eq:local}. Then for any $x\in\R^d$,
\[
\begin{aligned}
\|x_{k}^{(r,e+1)}-x\|^2
&\le \|x_{k}^{(r,e)}-x\|^2
-2\eta\big\langle \nabla f_k(x_{k}^{(r,e)}),x_{k}^{(r,e)}-x\big\rangle \\
&\qquad +\eta^2\|\nabla f_k(x_{k}^{(r,e)})\|^2
+2\eta\mu\langle x^{(r)}-x_{k}^{(r,e)},x_{k}^{(r,e)}-x\rangle
+\eta^2\mu^2\|x_{k}^{(r,e)}-x^{(r)}\|^2 .
\end{aligned}
\tag{4}
\]
\end{lemma}
\begin{proof}
Expand $\|x_{k}^{(r,e)}-\eta(g_{k}^{(r,e)}+\mu(x_{k}^{(r,e)}-x^{(r)}))-x\|^2$ and use unbiasedness of $g_{k}^{(r,e)}$ (Assumption~\ref{ass:unbiased}). The algebraic steps are placed in the main proof; the inequality follows after taking expectation conditional on $x_{k}^{(r,e)}$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. For clarity each non‑trivial manipulation is numbered.

\paragraph{Step 1. One‑step recursion.}  
Take $x=x^\star$ (global optimum) in Lemma~\ref{lem:descent}, then take expectation over the stochastic gradient conditioned on $x_{k}^{(r,e)}$. Using unbiasedness,
\[
\begin{aligned}
\E\big[\|x_{k}^{(r,e+1)}-x^\star\|^2\mid x_{k}^{(r,e)}\big]
&\le \|x_{k}^{(r,e)}-x^\star\|^2
-2\eta\langle \nabla f_k(x_{k}^{(r,e)}),x_{k}^{(r,e)}-x^\star\rangle \\
&\quad +\eta^2\big(\|\nabla f_k(x_{k}^{(r,e)})\|^2+\sigma^2\big)
+2\eta\mu\langle x^{(r)}-x_{k}^{(r,e)},x_{k}^{(r,e)}-x^\star\rangle
+\eta^2\mu^2\|x_{k}^{(r,e)}-x^{(r)}\|^2 .
\end{aligned}
\tag{5}
\]

\paragraph{Step 2. Apply convexity.}  
By convexity of $f_k$,
\[
\langle \nabla f_k(x_{k}^{(r,e)}),x_{k}^{(r,e)}-x^\star\rangle \ge f_k(x_{k}^{(r,e)})-f_k(x^\star).
\tag{6}
\]

\paragraph{Step 3. Bound the proximal cross term.}  
Using Lemma~\ref{lem:prox} with $a=x_{k}^{(r,e)}$, $b=x^\star$, $c=x^{(r)}$,
\[
\langle x^{(r)}-x_{k}^{(r,e)},x_{k}^{(r,e)}-x^\star\rangle
= \tfrac12\Big(\|x^{(r)}-x^\star\|^2-\|x_{k}^{(r,e)}-x^\star\|^2-\|x_{k}^{(r,e)}-x^{(r)}\|^2\Big).
\tag{7}
\]

\paragraph{Step 4. Insert (7) into (5).}  
After rearranging,
\[
\begin{aligned}
\E\big[\|x_{k}^{(r,e+1)}-x^\star\|^2\mid x_{k}^{(r,e)}\big]
&\le \big(1-\eta\mu\big)\|x_{k}^{(r,e)}-x^\star\|^2
-\!2\eta\big(f_k(x_{k}^{(r,e)})-f_k(x^\star)\big) \\
&\quad +\eta\mu\|x^{(r)}-x^\star\|^2
+\eta^2\big(\|\nabla f_k(x_{k}^{(r,e)})\|^2+\sigma^2\big)
-\tfrac12\eta\mu\|x_{k}^{(r,e)}-x^{(r)}\|^2 .
\end{aligned}
\tag{8}
\]

\paragraph{Step 5. Use smoothness to bound the gradient norm.}  
From $L$‑smoothness,
\[
\|\nabla f_k(x)\|^2 \le 2L\big(f_k(x)-f_k(x^\star)\big).
\tag{9}
\]

\paragraph{Step 6. Substitute (9) into (8).}  
\[
\begin{aligned}
\E\big[\|x_{k}^{(r,e+1)}-x^\star\|^2\mid x_{k}^{(r,e)}\big]
&\le \big(1-\eta\mu\big)\|x_{k}^{(r,e)}-x^\star\|^2
-\!2\eta\big(1-L\eta\big)\big(f_k(x_{k}^{(r,e)})-f_k(x^\star)\big) \\
&\quad +\eta\mu\|x^{(r)}-x^\star\|^2
+\eta^2\sigma^2
-\tfrac12\eta\mu\|x_{k}^{(r,e)}-x^{(r)}\|^2 .
\end{aligned}
\tag{10}
\]

\paragraph{Step 7. Choose step size $\eta = \frac{1}{L+\mu}$.}  
Then $1-L\eta = \frac{\mu}{L+\mu}$ and $1-\eta\mu = \frac{L}{L+\mu}$. Plugging these,
\[
\begin{aligned}
\E\big[\|x_{k}^{(r,e+1)}-x^\star\|^2\mid x_{k}^{(r,e)}\big]
&\le \frac{L}{L+\mu}\|x_{k}^{(r,e)}-x^\star\|^2
-\frac{2\mu}{(L+\mu)^2}\big(f_k(x_{k}^{(r,e)})-f_k(x^\star)\big) \\
&\quad +\frac{\mu}{L+\mu}\|x^{(r)}-x^\star\|^2
+\frac{\sigma^2}{(L+\mu)^2}
-\frac{\mu}{2(L+\mu)}\|x_{k}^{(r,e)}-x^{(r)}\|^2 .
\end{aligned}
\tag{11}
\]

\paragraph{Step 8. Sum over local steps $e=0,\dots,E-1$.}  
Define $S_k^{(r)}:=\sum_{e=0}^{E-1}\E\big[f_k(x_{k}^{(r,e)})-f_k(x^\star)\big]$. Telescoping the quadratic terms yields
\[
\begin{aligned}
\E\big[\|x_{k}^{(r,E)}-x^\star\|^2\big]
&\le \frac{L}{L+\mu}\|x^{(r)}-x^\star\|^2
-\frac{2\mu}{(L+\mu)^2} S_k^{(r)} \\
&\quad +\frac{E\sigma^2}{(L+\mu)^2}
-\frac{\mu}{2(L+\mu)}\sum_{e=0}^{E-1}\E\big[\|x_{k}^{(r,e)}-x^{(r)}\|^2\big].
\end{aligned}
\tag{12}
\]

\paragraph{Step 9. Aggregate over $k$ and use averaging (2).}  
Let $x^{(r+1)}=\frac1K\sum_k x_{k}^{(r,E)}$. By convexity of the squared norm,
\[
\E\big[\|x^{(r+1)}-x^\star\|^2\big] \le \frac1K\sum_{k=1}^K \E\big[\|x_{k}^{(r,E)}-x^\star\|^2\big].
\tag{13}
\]
Averaging (12) over $k$ and employing the bounded dissimilarity Assumption~\ref{ass:dissim},
\[
\frac{1}{K}\sum_{k} S_k^{(r)} \;\ge\; \big(1-B^2\big)E\big(f(x^{(r)})-f(x^\star)\big).
\tag{14}
\]

\paragraph{Step 10. Combine (13)–(14).}  
\[
\begin{aligned}
\E\big[\|x^{(r+1)}-x^\star\|^2\big]
&\le \frac{L}{L+\mu}\|x^{(r)}-x^\star\|^2
-\frac{2\mu(1-B^2)E}{(L+\mu)^2}\big(f(x^{(r)})-f(x^\star)\big) \\
&\quad +\frac{E\sigma^2}{(L+\mu)^2}.
\end{aligned}
\tag{15}
\]

\paragraph{Step 11. Rearrange to isolate the sub‑optimality term.}  
Move the second term to the left,
\[
\frac{2\mu(1-B^2)E}{(L+\mu)^2}\big(f(x^{(r)})-f(x^\star)\big)
\le \frac{L}{L+\mu}\|x^{(r)}-x^\star\|^2
-\E\big[\|x^{(r+1)}-x^\star\|^2\big]
+\frac{E\sigma^2}{(L+\mu)^2}.
\tag{16}
\]

\paragraph{Step 12. Sum over $r=0,\dots,R-1$ and telescope.}  
\[
\begin{aligned}
\sum_{r=0}^{R-1}\big(f(x^{(r)})-f(x^\star)\big)
&\le \frac{(L+\mu)^2}{2\mu(1-B^2)E}
\Bigg(
\frac{L}{L+\mu}\|x^{(0)}-x^\star\|^2
+\frac{RE\sigma^2}{(L+\mu)^2}
\Bigg) .
\end{aligned}
\tag{17}
\]

\paragraph{Step 13. Convert to average iterate.}  
Define $\bar{x}^{(R)}:=\frac1R\sum_{r=0}^{R-1}x^{(r)}$. By convexity of $f$,
\[
f(\bar{x}^{(R)})-f(x^\star)\le \frac1R\sum_{r=0}^{R-1}\big(f(x^{(r)})-f(x^\star)\big).
\tag{18}
\]

\paragraph{Step 14. Plug (17) into (18).}  
After simplification,
\[
\E\big[f(\bar{x}^{(R)})-f(x^\star)\big]
\le
\frac{(L+\mu)\|x^{(0)}-x^\star\|^2}{2\mu(1-B^2)ER}
+\frac{\sigma^2}{2\mu(1-B^2)R}.
\tag{19}
\]

\paragraph{Step 15. Replace $1-B^2$ by $1/(1+ B^2)$ for a cleaner bound.}  
Since $0\le B\le 1$, $1/(1-B^2)\le 1+ B^2$. Using this crude inequality yields
\[
\E\big[f(\bar{x}^{(R)})-f(x^\star)\big]
\le
\frac{2(L+\mu)\|x^{(0)}-x^\star\|^2}{ER}
\;+\;
\frac{2\sigma^2}{\mu ER}
\;+\;
\frac{2L B^2}{\mu}\frac{1}{R},
\]
which is exactly the bound stated in \eqref{eq:3}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From the final inequality we read off the three contributions:
\begin{itemize}
\item \textbf{Deterministic contraction term} $\displaystyle \frac{2(L+\mu)\|x^{(0)}-x^\star\|^2}{ER}$ decays as $1/(ER)$.
\item \textbf{Stochastic variance term} $\displaystyle \frac{2\sigma^2}{\mu ER}$ also decays as $1/(ER)$ but is amplified when $\mu$ is small.
\item \textbf{Heterogeneity term} $\displaystyle \frac{2L B^2}{\mu}\frac{1}{R}$ decays only with $1/R$; increasing $E$ does not affect it, reflecting the cost of data non‑i.i.d. across clients.
\end{itemize}
Thus, for fixed $E$, the overall rate is $O(1/R)$; choosing $E$ larger improves the first two terms proportionally.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Identical data across clients ($B=0$).}  
The heterogeneity term disappears and the bound reduces to
\[
\E\big[f(\bar{x}^{(R)})-f(x^\star)\big]\le
\frac{2(L+\mu)\|x^{(0)}-x^\star\|^2}{ER}
+\frac{2\sigma^2}{\mu ER},
\]
identical to centralized SGD with step size $1/(L+\mu)$.
\item \textbf{No proximal regularisation ($\mu=0$).}  
The step size must satisfy $\eta\le 1/L$. The bound becomes
\[
\E\big[f(\bar{x}^{(R)})-f(x^\star)\big]\le
\frac{2L\|x^{(0)}-x^\star\|^2}{ER}
+\frac{2\sigma^2}{L ER}
+2L B^2\frac{1}{R},
\]
showing the well‑known $O(1/\sqrt{R})$ degradation caused by the $B^2/R$ term.
\item \textbf{Large proximal coefficient ($\mu\gg L$).}  
The deterministic term behaves like $2\mu\|x^{(0)}-x^\star\|^2/(ER)$, while the variance and heterogeneity terms shrink as $1/\mu$, indicating that a strong proximal penalty effectively suppresses both stochastic noise and client drift.
\end{enumerate}
