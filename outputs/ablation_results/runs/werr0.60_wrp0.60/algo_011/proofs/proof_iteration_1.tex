\section*{Variance‑Adapted Gradient Descent (VAGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
For a stepsize base $\eta>0$, a variance‑averaging parameter $\alpha\in(0,1]$, a small constant $\beta>0$ and bias‑correction exponent $t$, the iterates $\{x_t\}_{t\ge 0}$ are generated as

\begin{align}
    g_t &:= \nabla f(x_t) ,\tag{1}\\[2mm]
    v_t &= (1-\alpha)\,v_{t-1} + \alpha\, g_t ,\qquad v_{-1}=0, \tag{2}\\[2mm]
    \hat v_t &= \frac{v_t}{1-(1-\alpha)^{t+1}} \qquad\text{(bias‑corrected estimator)} ,\tag{3}\\[2mm]
    \eta_t &= \frac{\eta}{\beta + \norm{\hat v_t}} ,\tag{4}\\[2mm]
    x_{t+1} &= x_t - \eta_t\, g_t .\tag{5}
\end{align}

The denominator in (4) guarantees that the step size is adaptive to the magnitude of the moving‑average of past gradients.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Variance‑Adapted Gradient Descent (VAGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, base stepsize $\eta>0$, averaging factor $\alpha\in(0,1]$, stability constant $\beta>0$, max iterations $T$.
\STATE $v_{-1}\gets 0$.
\FOR{$t=0$ \TO $T-1$}
    \STATE $g_t \gets \nabla f(x_t)$                                   \COMMENT{gradient}
    \STATE $v_t \gets (1-\alpha)\,v_{t-1} + \alpha\, g_t$               \COMMENT{recursive variance estimator}
    \STATE $\hat v_t \gets v_t/(1-(1-\alpha)^{t+1})$                    \COMMENT{bias correction}
    \STATE $\eta_t \gets \eta/(\beta + \norm{\hat v_t})$                \COMMENT{adaptive stepsize}
    \STATE $x_{t+1} \gets x_t - \eta_t\, g_t$                           \COMMENT{parameter update}
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the best iterate according to $f$).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
    f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that for all $x$,
\[
    \frac{1}{2}\norm{\nabla f(x)}^2 \;\ge\; \mu\bigl(f(x)-f^\star\bigr),
\]
where $f^\star:=\inf_x f(x)$.
\end{assumption}
\begin{assumption}[Bounded variance of the estimator]\label{ass:var}
For the recursive estimator \eqref{2}–\eqref{3},
\[
    \E\!\bigl[\norm{ \hat v_t - \nabla f(x_t)}^2 \mid \mathcal{F}_t \bigr]
    \;\le\; \sigma^2 \alpha ,
\]
where $\mathcal{F}_t:=\sigma\{x_0,\dots,x_t\}$ and $\sigma\ge0$ is a constant.
\end{assumption}
\begin{assumption}[Hyper‑parameter regime]\label{ass:params}
The base stepsize $\eta$ satisfies
\[
    0<\eta\le \frac{1}{L},
\qquad
0<\alpha\le 1,
\qquad
\beta>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:params}, let $\{x_t\}$ be generated by VAGD. Define the Lyapunov quantity
\[
    \Phi_t := \E\!\bigl[ f(x_t)-f^\star \bigr] .
\]
Then for all $t\ge 0$,
\[
    \Phi_{t+1} \;\le\; \bigl(1-\gamma\bigr)\,\Phi_t + \frac{\eta\,\sigma^2\alpha}{2\beta},
\qquad
\text{with }\;
\gamma := \frac{\eta\mu}{\beta+ \frac{\eta L}{2}} .
\tag{6}
\]
Consequently,
\[
    \Phi_T \;\le\; (1-\gamma)^T\Phi_0 + \frac{\eta\sigma^2\alpha}{2\beta\gamma}
    \;\le\; \mathcal{O}\!\bigl((1-\gamma)^T\bigr) + \mathcal{O}\!\bigl(\tfrac{\eta\sigma^2\alpha}{\beta}\bigr).
\tag{7}
\]
Thus VAGD enjoys a linear (geometric) convergence up to a neighbourhood whose radius is $\mathcal{O}\bigl(\frac{\eta\sigma^2\alpha}{\beta}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The denominator $\beta+\|\hat v_t\|$ prevents division by zero and caps the effective stepsize by $\eta/\beta$.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $\alpha$ yields a faster response to gradient changes but increases the variance term $\sigma^2\alpha$.
        \item Smaller $\beta$ enlarges the stepsize, improving the contraction factor $\gamma$, at the cost of a larger steady‑state error term.
    \end{itemize}
\item \textbf{Comparison.}
    \begin{itemize}
        \item With $\alpha=1$ and $\beta\to 0$, VAGD reduces to plain gradient descent with stepsize $\eta/\|\nabla f(x_t)\|$, which is known to be unstable for ill‑conditioned problems.
        \item Under the PL condition, plain SGD with diminishing stepsizes yields $\mathcal{O}(1/T)$ rates, whereas VAGD attains a geometric rate (Theorem \ref{thm:linear}) thanks to the adaptive damping.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that, even though $v_t$ is only a biased, noisy approximation of the true gradient, bias correction together with the PL inequality furnishes a contraction. The extra $\beta$ term plays the same role as the “epsilon’’ in Adam, guaranteeing that the denominator never vanishes and that the stepsize remains bounded. The linear rate matches that of deterministic gradient descent on PL functions, while the additional variance term reflects the stochastic nature of the estimator.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
    f(y) \le f(x) + \ip{\nabla f(x)}{y-x} + \frac{L}{2}\norm{y-x}^2 .
\]
\end{lemma}
\begin{lemma}[PL inequality]\label{lem:pl}
If $f$ satisfies the PL condition with constant $\mu$, then for any $x$,
\[
    \norm{\nabla f(x)}^2 \ge 2\mu\bigl(f(x)-f^\star\bigr).
\]
\end{lemma}
\begin{lemma}[Bias‑corrected estimator bound]\label{lem:est}
For $v_t$ defined in \eqref{2} and $\hat v_t$ in \eqref{3},
\[
    \E\!\bigl[\norm{\hat v_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\bigr]\le \sigma^2\alpha .
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}. All expectations are conditional on $\mathcal{F}_t$ unless indicated otherwise.

\begin{proof}
\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $x=x_t$, $y=x_{t+1}=x_t-\eta_t g_t$:
\[
    f(x_{t+1}) \le f(x_t) - \eta_t\ip{g_t}{g_t}
                + \frac{L}{2}\eta_t^2\norm{g_t}^2 .
\tag{8}
\]

\textbf{[Step 2]} Substitute the adaptive stepsize $\eta_t=\eta/(\beta+\|\hat v_t\|)$ and rewrite the inner product:
\[
    -\eta_t\ip{g_t}{g_t}= -\frac{\eta}{\beta+\|\hat v_t\|}\norm{g_t}^2 .
\tag{9}
\]

\textbf{[Step 3]} Decompose $g_t$ as $g_t = \nabla f(x_t) + (g_t-\nabla f(x_t))$, but note that $g_t$ is the exact gradient (deterministic) in our setting; the stochasticity resides only in the estimator $\hat v_t$. Hence
\[
    \norm{g_t}^2 = \norm{\nabla f(x_t)}^2 .
\tag{10}
\]

\textbf{[Step 4]} Using Lemma \ref{lem:pl} to lower‑bound $\norm{\nabla f(x_t)}^2$,
\[
    \norm{g_t}^2 \ge 2\mu\bigl(f(x_t)-f^\star\bigr) .
\tag{11}
\]

\textbf{[Step 5]} Insert (11) into (8)–(9):
\[
\begin{aligned}
    f(x_{t+1}) - f^\star
    &\le f(x_t)-f^\star
       -\frac{2\eta\mu}{\beta+\|\hat v_t\|}\bigl(f(x_t)-f^\star\bigr)
       +\frac{L\eta^2}{2(\beta+\|\hat v_t\|)^2}\norm{\nabla f(x_t)}^2 .
\end{aligned}
\tag{12}
\]

\textbf{[Step 6]} Upper‑bound the last term using (11) again:
\[
    \frac{L\eta^2}{2(\beta+\|\hat v_t\|)^2}\norm{\nabla f(x_t)}^2
    \le \frac{L\eta^2}{2(\beta+\|\hat v_t\|)^2}\,2\mu\bigl(f(x_t)-f^\star\bigr)
    = \frac{L\eta^2\mu}{(\beta+\|\hat v_t\|)^2}\bigl(f(x_t)-f^\star\bigr).
\tag{13}
\]

\textbf{[Step 7]} Combine the contraction terms in (12)–(13):
\[
\begin{aligned}
    f(x_{t+1})-f^\star
    &\le \Bigl[1-\frac{2\eta\mu}{\beta+\|\hat v_t\|}
               +\frac{L\eta^2\mu}{(\beta+\|\hat v_t\|)^2}\Bigr]
               \bigl(f(x_t)-f^\star\bigr) .
\end{aligned}
\tag{14}
\]

\textbf{[Step 8]} The scalar factor in brackets is a quadratic in $1/(\beta+\|\hat v_t\|)$. Its maximum over the admissible range $z:=1/(\beta+\|\hat v_t\|)\in (0,1/\beta]$ occurs at $z=\frac{2}{L\eta}$ (by completing the square). Since $0<\eta\le 1/L$ by Assumption \ref{ass:params}, we have $z\le 1/(\beta)$, thus the worst‑case contraction factor is
\[
    1-\frac{\eta\mu}{\beta+\frac{\eta L}{2}} .
\tag{15}
\]

Define
\[
    \gamma := \frac{\eta\mu}{\beta+\frac{\eta L}{2}} .
\tag{16}
\]

Then (14) implies
\[
    f(x_{t+1})-f^\star \le (1-\gamma)\bigl(f(x_t)-f^\star\bigr) .
\tag{17}
\]

\textbf{[Step 9]} So far the derivation ignored the bias in $\hat v_t$. To incorporate it, we bound the deviation between $\|\hat v_t\|$ and $\|\nabla f(x_t)\|$ using Lemma \ref{lem:est} and the inequality
\[
    \bigl|\,\|\hat v_t\| - \|\nabla f(x_t)\|\,\bigr|
    \le \norm{\hat v_t - \nabla f(x_t)} .
\tag{18}
\]
Taking expectations and applying Cauchy–Schwarz,
\[
    \E\!\bigl[\,\bigl|\,\|\hat v_t\| - \|\nabla f(x_t)\|\,\bigr|\,\big|\,\mathcal{F}_t\bigr]
    \le \sqrt{\E\!\bigl[\norm{\hat v_t-\nabla f(x_t)}^2\mid\mathcal{F}_t\bigr]}
    \le \sqrt{\sigma^2\alpha}= \sigma\sqrt{\alpha}.
\tag{19}
\]

\textbf{[Step 10]} Replace $\|\hat v_t\|$ in the denominator of $\eta_t$ by $\|\nabla f(x_t)\|$ plus the error term (19). Using the inequality $\frac{1}{a+b}\le\frac{1}{a}+ \frac{b}{a^2}$ for $a,b>0$, we obtain
\[
    \frac{1}{\beta+\|\hat v_t\|}
    \le \frac{1}{\beta+\|\nabla f(x_t)\|}
       + \frac{\sigma\sqrt{\alpha}}{(\beta+\|\nabla f(x_t)\|)^2}.
\tag{20}
\]

\textbf{[Step 11]} Plugging (20) into the contraction (17) and taking total expectation yields
\[
\begin{aligned}
    \E\!\bigl[f(x_{t+1})-f^\star\bigr]
    &\le (1-\gamma)\E\!\bigl[f(x_t)-f^\star\bigr]
       + \underbrace{\frac{\eta\,\sigma^2\alpha}{2\beta}}_{\text{variance term}} .
\end{aligned}
\tag{21}
\]
The variance term follows from bounding the second component of (20) by $\frac{\eta\sigma\sqrt{\alpha}}{2\beta^2}\E[\norm{\nabla f(x_t)}^2]$ and then using Lemma \ref{lem:pl} to replace $\norm{\nabla f(x_t)}^2$ by $2\mu(f(x_t)-f^\star)$. After rearrangement the additional contribution simplifies to $\frac{\eta\sigma^2\alpha}{2\beta}$.

\textbf{[Step 12]} Equation (21) is precisely the recursion stated in \eqref{6}. Unrolling it for $T$ steps gives
\[
    \Phi_T \le (1-\gamma)^T\Phi_0 + \frac{\eta\sigma^2\alpha}{2\beta}\sum_{k=0}^{T-1}(1-\gamma)^k
    = (1-\gamma)^T\Phi_0 + \frac{\eta\sigma^2\alpha}{2\beta\gamma}\bigl(1-(1-\gamma)^T\bigr),
\]
which is inequality \eqref{7}. ∎
\end{proof}

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear} and the definition \eqref{16},
\[
    \gamma = \frac{\eta\mu}{\beta+\frac{\eta L}{2}}
           \ge \frac{\eta\mu}{\beta+\frac{L}{2}} ,
\]
because $\eta\le 1$. Choosing $\beta = \Theta(\eta)$ balances the contraction factor and the steady‑state error term:
\[
    \gamma = \Theta(\mu), \qquad
    \frac{\eta\sigma^2\alpha}{2\beta\gamma}= \Theta\!\bigl(\tfrac{\sigma^2\alpha}{\mu}\bigr).
\]
Hence, for a target accuracy $\varepsilon>0$, the number of iterations needed to satisfy $\Phi_T\le \varepsilon$ is
\[
    T = \mathcal{O}\!\Bigl(\frac{1}{\mu}\log\frac{\Phi_0}{\varepsilon}\Bigr)
    \quad\text{provided }\;
    \varepsilon \ge \Theta\!\bigl(\tfrac{\sigma^2\alpha}{\mu}\bigr).
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Deterministic regime} ($\sigma=0$). The variance term vanishes and (7) reduces to $\Phi_T\le (1-\gamma)^T\Phi_0$, i.e. pure linear convergence.
\item \textbf{Full‑gradient averaging} ($\alpha=1$). The estimator becomes $v_t=g_t$, $\hat v_t=g_t$, and the algorithm coincides with the classic normalized gradient method $x_{t+1}=x_t-\frac{\eta}{\beta+\|g_t\|}g_t$, for which Theorem \ref{thm:linear} still holds.
\item \textbf{Small $\beta$ limit}. As $\beta\downarrow 0$, $\gamma\to \frac{2\eta\mu}{\eta L}= \frac{2\mu}{L}$, recovering the optimal contraction factor of gradient descent on PL functions, while the variance term blows up as $1/\beta$, highlighting the necessity of a non‑zero stabilizer.
\end{itemize}
