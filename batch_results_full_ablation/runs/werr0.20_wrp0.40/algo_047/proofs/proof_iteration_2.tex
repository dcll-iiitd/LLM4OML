\section*{Algorithm Name}
\textbf{AdaGrad for Non‑Convex Smooth Optimization (running‑average version)}  

The method keeps a cumulative sum of past gradients and adapts the learning
rate coordinate‑wise (here we present the scalar version for clarity).

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be differentiable.  
Initialize $w_0\in\R^d$ and set $A_0=0\in\R^d$.  
For $k=0,1,2,\dots$ perform  

\[
\begin{aligned}
g_k &:= \nabla f(w_k) \in\R^d,\\[4pt]
A_{k+1} &:= A_k + g_k \qquad\text{(cumulative gradient sum)},\\[4pt]
\eta_{k+1} &:= \frac{\alpha}{\sqrt{\displaystyle\sum_{i=0}^{k} g_i^{\odot 2}+\epsilon}}\;,
\qquad\text{(scalar step size)}\\[4pt]
w_{k+1} &:= w_k - \eta_{k+1}\, g_k .
\end{aligned}
\]

Here $g_i^{\odot 2}$ denotes the element‑wise square, and the denominator is
interpreted component‑wise; the scalar version uses the Euclidean norm,
\[
\eta_{k+1}= \frac{\alpha}{\sqrt{\|A_k\|_2^{2}+\epsilon}} ,
\]
with hyper‑parameters $\alpha>0$ (base learning rate) and $\epsilon>0$ (stability).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (scalar step size)}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\R^d$, base step $\alpha>0$, stability $\epsilon>0$,
        max iterations $T$.
\State $A_0\gets 0\in\R^d$
\For{$k=0$ \textbf{to} $T-1$}
    \State $g_k \gets \nabla f(w_k)$
    \State $A_{k+1} \gets A_k + g_k$
    \State $\eta_{k+1} \gets \dfrac{\alpha}{\sqrt{\|A_k\|_2^{2}+\epsilon}}$
    \State $w_{k+1} \gets w_k - \eta_{k+1}\, g_k$
\EndFor
\State \Return $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: $\forall x,y\in\R^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{ass:bound}
There exists $G>0$ such that $\forall k$, $\|g_k\|=\|\nabla f(w_k)\|\le G$.
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
$f$ is bounded below by $f_{\inf}>-\infty$.
\end{assumption}

All constants $L,G,\alpha,\epsilon$ are known a priori and $\alpha,\epsilon>0$.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{w_k\}_{k\ge0}$ be generated by the algorithm under
Assumptions~\ref{ass:smooth}–\ref{ass:lb}.
Define $S_T:=\sum_{k=0}^{T-1}\|g_k\|$.
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\|\nabla f(w_k)\|^2\right]
\;\le\;
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,
\frac{G}{\sqrt{T}}
\;+\;
\frac{L\alpha^{2}G^{2}}{2\epsilon\,T}
\;+\;
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)\sqrt{\epsilon}}{\alpha\,T}.
\tag{1}
\]
Consequently, the dominant term scales as $O(1/\sqrt{T})$.
\end{theorem}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^2 .
\]
\end{lemma}
\begin{proof}
Directly the statement of Assumption~\ref{ass:smooth}. \qedhere
\end{proof}

% ADDED: Lemma bounding the cumulative gradient norm using the bounded‑gradient
% assumption. This lemma is needed in Step 4–5 to obtain the correct
% $\mathcal{O}(1/\sqrt{T})$ rate.
\begin{lemma}\label{lem:cum}
For every $k\ge0$,
\[
\|A_k\|\;\le\; G\sqrt{k+1}.
\]
\end{lemma}
\begin{proof}
By definition $A_k=\sum_{i=0}^{k-1}g_i$ (with $A_0=0$). Using the Cauchy–Schwarz
inequality,
\[
\|A_k\|^2 = \Big\|\sum_{i=0}^{k-1}g_i\Big\|^2
          \le \Big(\sum_{i=0}^{k-1}1^2\Big)\Big(\sum_{i=0}^{k-1}\|g_i\|^2\Big)
          \le (k)\,G^{2},
\]
where we used $\|g_i\|\le G$ from Assumption~\ref{ass:bound}. Adding the
current gradient $g_k$ gives the claimed bound with $k+1$ in place of $k$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
All expectations are taken with respect to any randomness in the gradient
oracle; the derivation below is deterministic, hence taking expectations
preserves every inequality.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=-\eta_{k+1}g_k$:
\[
f(w_{k+1})\le f(w_k)-\eta_{k+1}\langle g_k,g_k\rangle
+\frac{L}{2}\eta_{k+1}^{2}\|g_k\|^{2}.
\tag{2}
\]

\item[Step 2.] Substitute the definition of $\eta_{k+1}$.
Recall $\eta_{k+1}= \alpha/ \sqrt{\|A_k\|^{2}+\epsilon}$, thus
\[
\eta_{k+1}\|g_k\|^{2}
= \frac{\alpha\|g_k\|^{2}}{\sqrt{\|A_k\|^{2}+\epsilon}},
\qquad
\eta_{k+1}^{2}\|g_k\|^{2}
= \frac{\alpha^{2}\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{3}
\]

\item[Step 3.] Insert (3) into (2) and rearrange:
\[
f(w_{k+1})\le f(w_k)-\frac{\alpha\|g_k\|^{2}}
{\sqrt{\|A_k\|^{2}+\epsilon}}
+\frac{L\alpha^{2}}{2}\,
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{4}
\]

\item[Step 4.] % ADDED: bound on the denominator
Using Lemma~\ref{lem:cum} and the fact that $\epsilon>0$,
\[
\sqrt{\|A_k\|^{2}+\epsilon}
\;\le\; \sqrt{G^{2}(k+1)+\epsilon}
\;\le\; G\sqrt{k+1}+\sqrt{\epsilon}
\;\le\; G\sqrt{T}+\sqrt{\epsilon},
\qquad\forall\,k\le T-1.
\tag{5}
\]

\item[Step 5.] From (4) we obtain
\[
\frac{\alpha\|g_k\|^{2}}{\sqrt{\|A_k\|^{2}+\epsilon}}
\le
f(w_k)-f(w_{k+1})
+\frac{L\alpha^{2}}{2}\,
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{6}
\]

\item[Step 6.] Sum inequality (6) over $k=0,\dots,T-1$:
\[
\frac{\alpha}{G\sqrt{T}+\sqrt{\epsilon}}
\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\underbrace{\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w_{k+1})\bigr)}_{=\,f(w_0)-f(w_T)}
\;+\;
\frac{L\alpha^{2}}{2}\sum_{k=0}^{T-1}
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{7}
\]

\item[Step 7.] The telescoping sum satisfies
$f(w_0)-f(w_T)\le f(w_0)-f_{\inf}$.
Moreover, because $\|A_k\|^{2}+\epsilon\ge\epsilon$ and $\|g_k\|^{2}\le G^{2}$,
\[
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}
\le \frac{G^{2}}{\epsilon}.
\]
Hence
\[
\frac{L\alpha^{2}}{2}\sum_{k=0}^{T-1}
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}
\le
\frac{L\alpha^{2}G^{2}}{2\epsilon}\,T.
\tag{8}
\]

\item[Step 8.] Plugging the bounds from Step 7 into (7) yields
\[
\frac{\alpha}{G\sqrt{T}+\sqrt{\epsilon}}
\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\bigl(f(w_0)-f_{\inf}\bigr)
\;+\;
\frac{L\alpha^{2}G^{2}}{2\epsilon}\,T .
\tag{9}
\]

\item[Step 9.] Rearranging (9) gives a deterministic bound on the
average squared gradient:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,
\frac{G\sqrt{T}+\sqrt{\epsilon}}{T}
\;+\;
\frac{L\alpha^{2}G^{2}}{2\epsilon\,T}.
\tag{10}
\]

\item[Step 10.] % ADDED: expectation step
Since the right‑hand side of (10) is deterministic, taking expectation on
both sides preserves the inequality:
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|g_k\|^{2}\bigr]
=
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(w_k)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,
\frac{G}{\sqrt{T}}
\;+\;
\frac{L\alpha^{2}G^{2}}{2\epsilon\,T}
\;+\;
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)\sqrt{\epsilon}}{\alpha\,T}.
\tag{11}
\]

\item[Step 11.] The bound (11) is exactly the statement of Theorem~\ref{thm:main}.
\qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From (11) we see that the leading term behaves as
$\displaystyle \frac{2G\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,T^{-1/2}$,
while the remaining terms decay as $T^{-1}$. Hence the dominant
asymptotic rate is $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Gradient (deterministic) case.}
   The expectation operator disappears; the same $O(1/\sqrt{T})$ bound holds.
\item \textbf{Sparse gradients.}
   If only $s\ll d$ coordinates are non‑zero per iteration,
   Lemma~\ref{lem:cum} yields $\|A_k\|\le \sqrt{s}\,G\sqrt{k+1}$,
   leading to a tighter bound $O\!\bigl(1/\sqrt{sT}\bigr)$.
\item \textbf{Convex $f$.}
   When $f$ is convex, the same analysis yields a regret bound
   $O(\sqrt{T})$, reproducing the classic AdaGrad guarantee.
\end{enumerate}
