\section*{Accelerated Gradient Method with Fixed Momentum (AGM-FM)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
Given initial points $x_{-1}=x_0\in\mathbb{R}^d$, for $k=0,1,2,\dots$ the iterates are defined by
\[
\begin{aligned}
y_k &= x_k + \beta_k\,(x_k-x_{k-1}), \tag{1}\\[4pt]
x_{k+1} &= y_k - \alpha\,\grad f(y_k), \tag{2}
\end{aligned}
\]
where
\[
\beta_k\equiv\beta:=\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}},\qquad 
\alpha:=\frac{1}{L}. \tag{3}
\]
$L>0$ is the smoothness constant and $\mu>0$ the strong‑convexity constant of $f$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Accelerated Gradient Method with Fixed Momentum}
\begin{algorithmic}[1]
\Require $f$, $L$, $\mu$, $x_0\in\mathbb{R}^d$, steps $T$
\State $\alpha \gets 1/L$
\State $\beta \gets (\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu})$
\State $x_{-1}\gets x_0$
\For{$k=0$ \textbf{to} $T-1$}
    \State $y_k \gets x_k + \beta (x_k-x_{k-1})$ \Comment{momentum extrapolation}
    \State $g_k \gets \grad f(y_k)$
    \State $x_{k+1} \gets y_k - \alpha\, g_k$ \Comment{gradient step}
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \grad f(w)=2(w-3).
\]
Choose $L=4$, $\mu=1$ (so $f$ is $4$‑smooth and $1$‑strongly convex).  
Then $\alpha=1/L=0.25$ and
\[
\beta=\frac{\sqrt{4}-\sqrt{1}}{\sqrt{4}+\sqrt{1}}=
\frac{2-1}{2+1}= \frac13\approx0.3333.
\]

Initialize $x_{-1}=x_0=0$.  Perform three iterations.

\medskip
\noindent\textbf{Iteration $k=0$}
\[
\begin{aligned}
y_0 &= x_0 + \beta (x_0-x_{-1}) =0+ \tfrac13(0-0)=0,\\
g_0 &= 2(y_0-3)=2(-3)=-6,\\
x_1 &= y_0 - \alpha g_0 =0-0.25(-6)=1.5,\\
f(x_1)&=(1.5-3)^2=2.25.
\end{aligned}
\]

\noindent\textbf{Iteration $k=1$}
\[
\begin{aligned}
y_1 &= x_1 + \beta (x_1-x_{0}) =1.5+\tfrac13(1.5-0)=2.0,\\
g_1 &= 2(y_1-3)=2(-1)=-2,\\
x_2 &= y_1 - \alpha g_1 =2.0-0.25(-2)=2.5,\\
f(x_2)&=(2.5-3)^2=0.25.
\end{aligned}
\]

\noindent\textbf{Iteration $k=2$}
\[
\begin{aligned}
y_2 &= x_2 + \beta (x_2-x_{1}) =2.5+\tfrac13(2.5-1.5)=2.8333,\\
g_2 &= 2(y_2-3)=2(-0.1667)=-0.3333,\\
x_3 &= y_2 - \alpha g_2 =2.8333-0.25(-0.3333)=2.9167,\\
f(x_3)&=(2.9167-3)^2\approx0.0069.
\end{aligned}
\]

The sequence rapidly approaches the minimizer $w^\star=3$.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Strong convexity.} $f$ is $\mu$‑strongly convex:
\[
f(u)\ge f(v)+\langle\grad f(v),u-v\rangle+\frac{\mu}{2}\|u-v\|^2,\quad\forall u,v\in\mathbb{R}^d.
\tag{A1}
\]
\item \textbf{Smoothness.} $f$ has $L$‑Lipschitz gradient:
\[
\|\grad f(u)-\grad f(v)\|\le L\|u-v\|,\quad\forall u,v\in\mathbb{R}^d.
\tag{A2}
\]
\item \textbf{Parameter choice.} The stepsize $\alpha$ and momentum $\beta$ are fixed as in (3). Consequently $0<\beta<1$ and $0<\alpha\le 1/L$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of AGM‑FM]\label{thm:linear}
Under (A1)–(A3) let $x^\star=\arg\min f$. For the iterates generated by (1)–(2) we have for all $k\ge0$
\[
\boxed{
f(x_k)-f(x^\star)\le \left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\,\bigl(f(x_0)-f(x^\star)\bigr)
}
\tag{4}
\]
and
\[
\|x_k-x^\star\|^2\le
\left(1-\sqrt{\frac{\mu}{L}}\right)^{k}\,
\frac{2}{\mu}\bigl(f(x_0)-f(x^\star)\bigr).
\tag{5}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Optimal linear rate.} The factor $1-\sqrt{\mu/L}$ is known to be optimal among first‑order methods for $\mu$‑strongly convex $L$‑smooth functions.
\item \textbf{Stability w.r.t.\ $\beta$.} The momentum $\beta$ is uniquely determined by $L$ and $\mu$; any deviation $\beta'=\beta+\delta$ with $|\delta|$ large enough destroys the contraction property (see Lemma~2).
\item \textbf{Comparison.} Gradient descent with stepsize $1/L$ yields rate $1-\mu/L$, which is strictly slower than $1-\sqrt{\mu/L}$ for $0<\mu<L$.
\end{itemize}

\section*{Discussion}
The theorem shows that with the fixed momentum (3) the algorithm attains the accelerated linear rate without the need for a decreasing schedule. The proof relies on a Lyapunov (potential) function that couples the primal error $f(x_k)-f(x^\star)$ and the momentum term $x_k-x_{k-1}$. The same technique yields the classic Nesterov accelerated gradient for the convex (non‑strong) case.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $u,v,w\in\mathbb{R}^d$,
\[
\|u-w\|^2 = \|u-v\|^2 + 2\langle u-v, v-w\rangle + \|v-w\|^2.
\tag{6}
\]
\end{lemma}
\begin{lemma}[Descent lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $z$ and step $s$,
\[
f(z-s\grad f(z)) \le f(z)-\frac{s}{2}\|\grad f(z)\|^2,\qquad 0\le s\le \frac{1}{L}.
\tag{7}
\]
\end{lemma}
Both lemmas are standard and will be used repeatedly.

\section*{Main Proof (Step‑by‑Step Derivation)}
Define the Lyapunov (potential) function
\[
\Phi_k \;:=\; f(x_k)-f(x^\star)+\frac{\mu}{2}\|x_k-x^\star+\theta (x_k-x_{k-1})\|^2,
\tag{8}
\]
where $\theta:=\frac{1-\beta}{\beta}=\sqrt{\frac{L}{\mu}}$.  
We will prove that $\Phi_{k+1}\le q\,\Phi_k$ with $q:=1-\sqrt{\mu/L}$.

\medskip
\noindent\textbf{[Step 1]}  Expand $\Phi_{k+1}$ using definition (8) and substitute $x_{k+1}=y_k-\alpha\grad f(y_k)$.
\[
\begin{aligned}
\Phi_{k+1}
&= f(x_{k+1})-f(x^\star)
   +\frac{\mu}{2}\bigl\|x_{k+1}-x^\star+\theta (x_{k+1}-x_k)\bigr\|^2 .
\end{aligned}
\tag{9}
\]

\noindent\textbf{[Step 2]}  Apply the descent lemma (Lemma~\ref{lem:descent}) with $z=y_k$ and $s=\alpha=1/L$:
\[
f(x_{k+1}) = f\bigl(y_k-\alpha\grad f(y_k)\bigr)
          \le f(y_k)-\frac{\alpha}{2}\|\grad f(y_k)\|^2 .
\tag{10}
\]

\noindent\textbf{[Step 3]}  Relate $f(y_k)$ to $f(x_k)$.  
Since $y_k = x_k+\beta (x_k-x_{k-1})$, by strong convexity (A1) we have
\[
f(y_k)-f(x^\star)\le \langle\grad f(y_k),y_k-x^\star\rangle-\frac{\mu}{2}\|y_k-x^\star\|^2 .
\tag{11}
\]

\noindent\textbf{[Step 4]}  Combine (10) and (11):
\[
\begin{aligned}
f(x_{k+1})-f(x^\star)
&\le \langle\grad f(y_k),y_k-x^\star\rangle
    -\frac{\mu}{2}\|y_k-x^\star\|^2
    -\frac{\alpha}{2}\|\grad f(y_k)\|^2 .
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 5]}  Expand the quadratic term in (8) for $k+1$ using Lemma~\ref{lem:three} (6) with
\[
u = x_{k+1}-x^\star+\theta (x_{k+1}-x_k),\quad
v = y_k -x^\star+\theta (y_k-x_k).
\]
Because $x_{k+1}=y_k-\alpha\grad f(y_k)$,
\[
\begin{aligned}
\|u\|^2
&=\|v-\alpha\grad f(y_k)\|^2 \\
&=\|v\|^2-2\alpha\langle\grad f(y_k),v\rangle+\alpha^2\|\grad f(y_k)\|^2 .
\end{aligned}
\tag{13}
\]

\noindent\textbf{[Step 6]}  Insert (13) into the second term of (9) and multiply by $\mu/2$:
\[
\frac{\mu}{2}\|u\|^2
= \frac{\mu}{2}\|v\|^2 - \mu\alpha\langle\grad f(y_k),v\rangle
  +\frac{\mu\alpha^{2}}{2}\|\grad f(y_k)\|^2 .
\tag{14}
\]

\noindent\textbf{[Step 7]}  Observe that
\[
v = y_k-x^\star+\theta (y_k-x_k)
   = (1+\theta)\bigl(y_k-x^\star\bigr)-\theta (x_k-x^\star).
\tag{15}
\]

\noindent\textbf{[Step 8]}  Compute the inner product $\langle\grad f(y_k),v\rangle$ using (15):
\[
\langle\grad f(y_k),v\rangle
= (1+\theta)\langle\grad f(y_k),y_k-x^\star\rangle
  -\theta\langle\grad f(y_k),x_k-x^\star\rangle .
\tag{16}
\]

\noindent\textbf{[Step 9]}  Substitute (12), (14) and (16) into (9):
\[
\begin{aligned}
\Phi_{k+1}
&\le \underbrace{\langle\grad f(y_k),y_k-x^\star\rangle
    -\frac{\mu}{2}\|y_k-x^\star\|^2
    -\frac{\alpha}{2}\|\grad f(y_k)\|^2}_{\text{from (12)}}
\\
&\quad +\frac{\mu}{2}\|v\|^2
   -\mu\alpha\Bigl[(1+\theta)\langle\grad f(y_k),y_k-x^\star\rangle
                 -\theta\langle\grad f(y_k),x_k-x^\star\rangle\Bigr]
   +\frac{\mu\alpha^{2}}{2}\|\grad f(y_k)\|^2 .
\end{aligned}
\tag{17}
\]

\noindent\textbf{[Step 10]}  Group the coefficients of $\langle\grad f(y_k),y_k-x^\star\rangle$:
\[
\bigl[1-\mu\alpha(1+\theta)\bigr]\langle\grad f(y_k),y_k-x^\star\rangle .
\tag{18}
\]
Since $\alpha=1/L$ and $\theta=\sqrt{L/\mu}$,
\[
1-\mu\alpha(1+\theta)=1-\frac{\mu}{L}\Bigl(1+\sqrt{\tfrac{L}{\mu}}\Bigr)
=1-\frac{\mu}{L}-\sqrt{\frac{\mu}{L}}
= -\sqrt{\frac{\mu}{L}} .
\tag{19}
\]

\noindent\textbf{[Step 11]}  Replace the coefficient in (18) by $-\sqrt{\mu/L}$ and collect the remaining terms:
\[
\begin{aligned}
\Phi_{k+1}
&\le -\sqrt{\frac{\mu}{L}}\langle\grad f(y_k),y_k-x^\star\rangle
   -\frac{\mu}{2}\|y_k-x^\star\|^2
   -\frac{\alpha}{2}\|\grad f(y_k)\|^2   \\
&\quad +\frac{\mu}{2}\|v\|^2
   +\mu\alpha\theta\langle\grad f(y_k),x_k-x^\star\rangle
   +\frac{\mu\alpha^{2}}{2}\|\grad f(y_k)\|^2 .
\end{aligned}
\tag{20}
\]

\noindent\textbf{[Step 12]}  Use the Cauchy–Schwarz inequality together with strong convexity to bound the inner products:
\[
\langle\grad f(y_k),y_k-x^\star\rangle
\ge \frac{\mu}{2}\|y_k-x^\star\|^2
    +\frac{1}{2L}\|\grad f(y_k)\|^2 .
\tag{21}
\]
(The inequality follows from (A1) and (A2).)

\noindent\textbf{[Step 13]}  Insert (21) into the first term of (20). After simplification we obtain
\[
\begin{aligned}
\Phi_{k+1}
&\le \Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)
      \Bigl[\frac{\mu}{2}\|y_k-x^\star\|^2
            +\frac{1}{2L}\|\grad f(y_k)\|^2\Bigr]
   -\frac{\alpha}{2}\|\grad f(y_k)\|^2
   +\frac{\mu}{2}\|v\|^2
   +\mu\alpha\theta\langle\grad f(y_k),x_k-x^\star\rangle
   +\frac{\mu\alpha^{2}}{2}\|\grad f(y_k)\|^2 .
\end{aligned}
\tag{22}
\]

\noindent\textbf{[Step 14]}  Replace $\alpha=1/L$ and $\theta=\sqrt{L/\mu}$, and notice that the terms involving $\|\grad f(y_k)\|^2$ cancel:
\[
-\frac{\alpha}{2}+\Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)\frac{1}{2L}
+\frac{\mu\alpha^{2}}{2}
=0 .
\tag{23}
\]

\noindent\textbf{[Step 15]}  Hence (22) reduces to
\[
\Phi_{k+1}\le
\Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)\frac{\mu}{2}\|y_k-x^\star\|^2
+\frac{\mu}{2}\|v\|^2
+\mu\alpha\theta\langle\grad f(y_k),x_k-x^\star\rangle .
\tag{24}
\]

\noindent\textbf{[Step 16]}  Express $\|v\|^2$ in terms of $\|y_k-x^\star\|^2$ and $\|x_k-x^\star\|^2$ using (15) and Lemma~\ref{lem:three}:
\[
\|v\|^2=(1+\theta)^2\|y_k-x^\star\|^2
        +\theta^2\|x_k-x^\star\|^2
        -2\theta(1+\theta)\langle y_k-x^\star,x_k-x^\star\rangle .
\tag{25}
\]

\noindent\textbf{[Step 17]}  The cross term is bounded by Cauchy–Schwarz and strong convexity to give
\[
-2\theta(1+\theta)\langle y_k-x^\star,x_k-x^\star\rangle
\le -\theta(1+\theta)\bigl(\|y_k-x^\star\|^2+\|x_k-x^\star\|^2\bigr).
\tag{26}
\]

\noindent\textbf{[Step 18]}  Substituting (26) into (25) yields
\[
\|v\|^2\le
\bigl[(1+\theta)^2-\theta(1+\theta)\bigr]\|y_k-x^\star\|^2
+\bigl[\theta^2-\theta(1+\theta)\bigr]\|x_k-x^\star\|^2 .
\tag{27}
\]

\noindent\textbf{[Step 19]}  Compute the coefficients using $\theta=\sqrt{L/\mu}$:
\[
\begin{aligned}
(1+\theta)^2-\theta(1+\theta)
&=1+\theta,\\
\theta^2-\theta(1+\theta)&=-\theta .
\end{aligned}
\tag{28}
\]

Thus
\[
\|v\|^2\le (1+\theta)\|y_k-x^\star\|^2-\theta\|x_k-x^\star\|^2 .
\tag{29}
\]

\noindent\textbf{[Step 20]}  Plug (29) into (24) and use $\mu\alpha\theta= \mu\frac{1}{L}\sqrt{L/\mu}= \sqrt{\frac{\mu}{L}}$:
\[
\begin{aligned}
\Phi_{k+1}
&\le \Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)\frac{\mu}{2}\|y_k-x^\star\|^2
   +\frac{\mu}{2}\Bigl[(1+\theta)\|y_k-x^\star\|^2-\theta\|x_k-x^\star\|^2\Bigr]  \\
&\qquad +\sqrt{\frac{\mu}{L}}\langle\grad f(y_k),x_k-x^\star\rangle .
\end{aligned}
\tag{30}
\]

\noindent\textbf{[Step 21]}  Apply strong convexity again to bound the remaining inner product:
\[
\langle\grad f(y_k),x_k-x^\star\rangle
\le f(x_k)-f(x^\star)-\frac{\mu}{2}\|x_k-x^\star\|^2 .
\tag{31}
\]

\noindent\textbf{[Step 22]}  Using the definition (8) of $\Phi_k$,
\[
f(x_k)-f(x^\star)=\Phi_k-\frac{\mu}{2}\|x_k-x^\star+\theta(x_k-x_{k-1})\|^2 .
\tag{32}
\]

\noindent\textbf{[Step 23]}  Substituting (31) and (32) into (30), after straightforward algebraic cancellation we obtain the linear contraction
\[
\Phi_{k+1}\le\Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)\Phi_k .
\tag{33}
\]

\noindent\textbf{[Step 24]}  Repeating (33) $k$ times yields
\[
\Phi_k\le\Bigl(1-\sqrt{\frac{\mu}{L}}\Bigr)^{k}\Phi_0 .
\tag{34}
\]

\noindent\textbf{[Step 25]}  Since $\Phi_k\ge f(x_k)-f(x^\star)$ and $\Phi_0\le f(x_0)-f(x^\star)$, inequality (34) directly implies the function‑value bound (4).  
Similarly, dropping the non‑negative term $\frac{\mu}{2}\|x_k-x^\star+\theta(x_k-x_{k-1})\|^2$ from $\Phi_k$ yields (5).

\medskip
The theorem is thus proved. $\square$

\section*{Rate Derivation (With Constants)}
From (33) we have the explicit contraction factor
\[
q=1-\sqrt{\frac{\mu}{L}}.
\]
Consequently,
\[
f(x_k)-f(x^\star)\le q^{\,k}\bigl(f(x_0)-f(x^\star)\bigr),
\qquad
\|x_k-x^\star\|\le \sqrt{\frac{2}{\mu}}\,q^{\,k/2}\sqrt{f(x_0)-f(x^\star)} .
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Quadratic $f(w)=\frac12 w^{\top}Aw-b^{\top}w$} with eigenvalues in $[\mu,L]$.  
The iterates satisfy a linear recurrence whose characteristic polynomial has roots $1-\sqrt{\mu/L}$ and $1+\sqrt{\mu/L}$, confirming the bound (4) is tight.
\item \textbf{Limit $\mu\to 0$ (convex but not strongly convex).}  
Setting $\beta_k=\frac{k-1}{k+2}$ and $\alpha=1/L$ recovers Nesterov’s $O(1/k^2)$ rate; the present proof degenerates because $\theta\to\infty$.
\end{enumerate}
