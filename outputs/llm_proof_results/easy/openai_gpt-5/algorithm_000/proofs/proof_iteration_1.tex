\section*{Algorithm Name}
Minibatch Stochastic Gradient Descent with Fixed Stepsize (MB-SGD-Fixed)

\section*{Mathematical Formulation}
We consider minimizing a (possibly nonconvex) smooth objective
\[
f(w) \triangleq \mathbb{E}_{z\sim\mathcal{D}}[F(w;z)], \quad w\in\R^d,
\]
where $F(\cdot;z)$ is differentiable for each random sample $z$ drawn i.i.d.\ from distribution $\mathcal{D}$.

At iteration $t$, draw a minibatch $\{z_{t,i}\}_{i=1}^B$ i.i.d.\ from $\mathcal{D}$, form the unbiased stochastic gradient estimator
\[
g_t \triangleq \frac{1}{B}\sum_{i=1}^B \nabla F(w_t; z_{t,i}),
\]
and update with a fixed stepsize $\eta>0$:
\[
w_{t+1} = w_t - \eta\, g_t.
\]

Hyperparameters:
- Stepsize (learning rate): $\eta>0$ (fixed across iterations)
- Minibatch size: $B\in\mathbb{N}$
- Total iterations: $T\in\mathbb{N}$

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{MB-SGD-Fixed}
\begin{algorithmic}[1]
\STATE Input: initial point $w_0\in\R^d$, stepsize $\eta>0$, minibatch size $B$, iterations $T$
\FOR{$t=0,1,2,\dots,T-1$}
  \STATE Sample minibatch $\{z_{t,i}\}_{i=1}^B$ i.i.d.\ from $\mathcal{D}$
  \STATE Compute $g_t \leftarrow \frac{1}{B}\sum_{i=1}^B \nabla F(w_t; z_{t,i})$
  \STATE Update $w_{t+1} \leftarrow w_t - \eta\, g_t$
\ENDFOR
\STATE Output: final iterate $w_T$ or a random iterate $w_R$ with $R\sim\text{Unif}\{0,\dots,T-1\}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider $f(w)=(w-3)^2$ (deterministic, $d=1$), gradient $\nabla f(w)=2(w-3)$. Take $w_0=0$, $\eta=0.1$, and run 3 iterations ($B=1$ and no noise).

- Iteration $t=0$:
  - Gradient: $g_0 = 2(w_0-3) = 2(0-3) = -6$
  - Update: $w_1 = w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6$
  - Objective: $f(w_1) = (0.6-3)^2 = 5.76$

- Iteration $t=1$:
  - Gradient: $g_1 = 2(w_1-3) = 2(0.6-3) = -4.8$
  - Update: $w_2 = 0.6 - 0.1(-4.8) = 1.08$
  - Objective: $f(w_2) = (1.08-3)^2 = 3.6864$

- Iteration $t=2$:
  - Gradient: $g_2 = 2(w_2-3) = 2(1.08-3) = -3.84$
  - Update: $w_3 = 1.08 - 0.1(-3.84) = 1.464$
  - Objective: $f(w_3) = (1.464-3)^2 = ( -1.536 )^2 = 2.359296$

\section*{Assumptions}
We work under the following standard conditions.

\begin{assumption}[Smoothness]
\label{assump:smooth}
The function $f$ is differentiable and has $L$-Lipschitz continuous gradient:
\[
\|\nabla f(x) - \nabla f(y)\| \le L \|x-y\|, \quad \forall x,y\in\R^d.
\]
\end{assumption}

\begin{assumption}[Unbiasedness and Bounded Variance]
\label{assump:unbiased-var}
For the minibatch estimator $g_t$,
\[
\E[g_t \mid w_t] = \nabla f(w_t), \quad \text{and}\quad \E\big[\|g_t - \nabla f(w_t)\|^2 \mid w_t\big] \le \frac{\sigma^2}{B},
\]
for some $\sigma\ge 0$. Samples across iterations and within minibatches are i.i.d.\ and independent of the past given $w_t$.
\end{assumption}

\begin{assumption}[Lower Boundedness]
\label{assump:lower}
There exists $f_\star>-\infty$ such that $f(w)\ge f_\star$ for all $w\in\R^d$.
\end{assumption}

\begin{assumption}[Stepsize]
\label{assump:stepsize}
The stepsize $\eta$ satisfies $0<\eta<\frac{2}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity bound with fixed stepsize]
\label{thm:main}
Under Assumptions \ref{assump:smooth}--\ref{assump:stepsize}, the iterates of MB-SGD-Fixed satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{f(w_0)-f_\star}{\eta\left(1-\frac{L\eta}{2}\right)T}
\;+\;
\frac{L\eta}{2\left(1-\frac{L\eta}{2}\right)}\cdot \frac{\sigma^2}{B}.
\]
Equivalently, if $R$ is uniform on $\{0,1,\dots,T-1\}$ and independent of the algorithm,
\[
\E\big[\|\nabla f(w_R)\|^2\big]
\;\le\;
\frac{f(w_0)-f_\star}{\eta\left(1-\frac{L\eta}{2}\right)T}
\;+\;
\frac{L\eta}{2\left(1-\frac{L\eta}{2}\right)}\cdot \frac{\sigma^2}{B}.
\]
\end{theorem}

\section*{Theoretical Properties}
- Expected descent and noise floor:
  - The bound exhibits an $O(1/T)$ optimization term and an $O(\eta/B)$ noise floor.
- Hyperparameter sensitivity:
  - Smaller $\eta$ and larger $B$ reduce the noise floor but may slow initial progress.
  - Any $\eta \in (0,2/L)$ ensures a positive descent coefficient $1-\frac{L\eta}{2}$.
- Comparison to full-batch gradient descent:
  - When $\sigma=0$ (or $B\to\infty$ so variance vanishes), the noise floor vanishes and the averaged stationarity scales as $O(1/T)$ with a fixed stepsize.
- Stationarity vs. optimality:
  - For nonconvex $f$, the guarantee is on small gradients (approximate stationarity), not on global optimality.

\section*{Discussion}
- To achieve $\E[\|\nabla f(w_R)\|^2]\le \varepsilon$ with fixed $\eta$, one needs both $T=\Theta(1/\varepsilon)$ and $\eta/B=\tilde{O}(\varepsilon)$ so that the noise floor does not dominate.
- Alternatively, decreasing stepsizes can remove the fixed noise floor and yield the classical $O(1/\sqrt{T})$ rate for the minimum expected gradient norm, but this is outside the fixed-stepsize setting.

\section*{Preliminaries (Definitions and Lemmas)}
We first derive two standard tools from first principles.

\begin{lemma}[Descent lemma from $L$-smoothness]
\label{lem:descent}
Under Assumption \ref{assump:smooth}, for all $x,y\in\R^d$,
\[
f(y) \le f(x) + \langle \nabla f(x), y-x\rangle + \frac{L}{2}\|y-x\|^2.
\]
\end{lemma}

\begin{proof}
Let $\phi(\tau)=f\big(x+\tau(y-x)\big)$ for $\tau\in[0,1]$. Then
\[
f(y)-f(x) = \int_0^1 \phi'(\tau)\,d\tau \quad \text{with}\quad \phi'(\tau)=\left\langle \nabla f\big(x+\tau(y-x)\big),\, y-x\right\rangle. \quad \text{[Step 1]}
\]
Add and subtract $\nabla f(x)$ inside the integrand:
\begin{align*}
f(y)-f(x)
&= \int_0^1 \left\langle \nabla f(x), y-x\right\rangle d\tau
  + \int_0^1 \left\langle \nabla f\big(x+\tau(y-x)\big)-\nabla f(x),\, y-x\right\rangle d\tau \quad \text{[Step 2]}\\
&= \left\langle \nabla f(x), y-x\right\rangle
  + \int_0^1 \left\langle \nabla f\big(x+\tau(y-x)\big)-\nabla f(x),\, y-x\right\rangle d\tau. \quad \text{[Step 3]}
\end{align*}
Apply Cauchy--Schwarz and Lipschitz gradient:
\begin{align*}
\left\langle \nabla f\big(x+\tau(y-x)\big)-\nabla f(x),\, y-x\right\rangle
&\le \left\|\nabla f\big(x+\tau(y-x)\big)-\nabla f(x)\right\|\,\|y-x\| \quad \text{[Step 4a]}\\
&\le L\| \tau(y-x)\|\,\|y-x\| = L\tau \|y-x\|^2. \quad \text{[Step 4b]}
\end{align*}
Integrating w.r.t.\ $\tau$ gives
\[
\int_0^1 \left\langle \nabla f\big(x+\tau(y-x)\big)-\nabla f(x),\, y-x\right\rangle d\tau
\le \int_0^1 L\tau \|y-x\|^2 d\tau = \frac{L}{2}\|y-x\|^2. \quad \text{[Step 5]}
\]
Combining, we obtain $f(y)\le f(x)+\langle \nabla f(x), y-x\rangle + \frac{L}{2}\|y-x\|^2$.
\end{proof}

\begin{lemma}[Second-moment decomposition for unbiased estimators]
\label{lem:second-moment}
Let $g_t=\nabla f(w_t)+\xi_t$ with $\E[\xi_t\mid w_t]=0$ and $\E[\|\xi_t\|^2\mid w_t]\le \frac{\sigma^2}{B}$. Then
\[
\E\big[\langle \nabla f(w_t), g_t\rangle\big]=\E\big[\|\nabla f(w_t)\|^2\big], \quad
\E\big[\|g_t\|^2\big]=\E\big[\|\nabla f(w_t)\|^2\big]+\E\big[\|\xi_t\|^2\big].
\]
\end{lemma}

\begin{proof}
Using the tower property of conditional expectation,
\begin{align*}
\E\big[\langle \nabla f(w_t), g_t\rangle\big]
&= \E\Big[\E\big[\langle \nabla f(w_t), g_t\rangle \mid w_t\big]\Big] \quad \text{[Step 6a]}\\
&= \E\Big[\langle \nabla f(w_t), \E[g_t\mid w_t]\rangle\Big] \quad \text{[Step 6b]}\\
&= \E\big[\langle \nabla f(w_t), \nabla f(w_t)\rangle\big] = \E\big[\|\nabla f(w_t)\|^2\big], \quad \text{[Step 6c]}
\end{align*}
where [Step 6b] uses linearity and measurability of $\nabla f(w_t)$ given $w_t$, and [Step 6c] uses $\E[g_t\mid w_t]=\nabla f(w_t)$.

For the second moment,
\begin{align*}
\|g_t\|^2 &= \|\nabla f(w_t)+\xi_t\|^2
= \|\nabla f(w_t)\|^2 + 2\langle \nabla f(w_t), \xi_t\rangle + \|\xi_t\|^2. \quad \text{[Step 7]}
\end{align*}
Take expectations and use zero-mean noise,
\begin{align*}
\E[\|g_t\|^2]
&= \E[\|\nabla f(w_t)\|^2] + 2\,\E\big[\E[\langle \nabla f(w_t), \xi_t\rangle \mid w_t]\big] + \E[\|\xi_t\|^2] \quad \text{[Step 8a]}\\
&= \E[\|\nabla f(w_t)\|^2] + 2\,\E\big[\langle \nabla f(w_t), \E[\xi_t\mid w_t]\rangle\big] + \E[\|\xi_t\|^2] \quad \text{[Step 8b]}\\
&= \E[\|\nabla f(w_t)\|^2] + \E[\|\xi_t\|^2], \quad \text{[Step 8c]}
\end{align*}
since $\E[\xi_t\mid w_t]=0$.
\end{proof}

\section*{Main Proof (Step-by-Step Derivation)}
We now prove Theorem \ref{thm:main}.

\begin{proof}[Proof of Theorem \ref{thm:main}]
Apply Lemma \ref{lem:descent} with $x=w_t$ and $y=w_{t+1}=w_t-\eta g_t$:
\begin{align*}
f(w_{t+1})
&\le f(w_t) + \left\langle \nabla f(w_t), w_{t+1}-w_t\right\rangle + \frac{L}{2}\|w_{t+1}-w_t\|^2 \quad \text{[Step 9a]}\\
&= f(w_t) + \left\langle \nabla f(w_t), -\eta g_t\right\rangle + \frac{L}{2}\|\,-\eta g_t\|^2 \quad \text{[Step 9b]}\\
&= f(w_t) - \eta \langle \nabla f(w_t), g_t\rangle + \frac{L\eta^2}{2}\|g_t\|^2. \quad \text{[Step 9c]}
\end{align*}
Take total expectation on both sides and use linearity of expectation:
\begin{align*}
\E[f(w_{t+1})]
&\le \E[f(w_t)] - \eta \E[\langle \nabla f(w_t), g_t\rangle] + \frac{L\eta^2}{2}\E[\|g_t\|^2]. \quad \text{[Step 10]}
\end{align*}
Invoke Lemma \ref{lem:second-moment} to replace the inner product term and the second moment:
\begin{align*}
\E[f(w_{t+1})]
&\le \E[f(w_t)] - \eta \E[\|\nabla f(w_t)\|^2] + \frac{L\eta^2}{2}\Big(\E[\|\nabla f(w_t)\|^2] + \E[\|\xi_t\|^2]\Big) \quad \text{[Step 11]}\\
&= \E[f(w_t)] - \eta\!\left(1-\frac{L\eta}{2}\right)\!\E[\|\nabla f(w_t)\|^2] + \frac{L\eta^2}{2}\E[\|\xi_t\|^2]. \quad \text{[Step 12]}
\end{align*}
Use the conditional variance bound from Assumption \ref{assump:unbiased-var} and the tower property:
\begin{align*}
\E[\|\xi_t\|^2] = \E\big[\E[\|\xi_t\|^2\mid w_t]\big] \le \E\Big[\frac{\sigma^2}{B}\Big] = \frac{\sigma^2}{B}. \quad \text{[Step 13]}
\end{align*}
Therefore,
\begin{align*}
\E[f(w_{t+1})]
&\le \E[f(w_t)] - \eta\!\left(1-\frac{L\eta}{2}\right)\!\E[\|\nabla f(w_t)\|^2] + \frac{L\eta^2}{2}\cdot \frac{\sigma^2}{B}. \quad \text{[Step 14]}
\end{align*}
Sum the inequality over $t=0,1,\dots,T-1$:
\begin{align*}
\E[f(w_T)]
&\le \E[f(w_0)] - \eta\!\left(1-\frac{L\eta}{2}\right)\!\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2] + T\cdot \frac{L\eta^2}{2}\cdot \frac{\sigma^2}{B}. \quad \text{[Step 15]}
\end{align*}
By Assumption \ref{assump:lower}, $f(w_T)\ge f_\star$ and so $\E[f(w_T)]\ge f_\star$:
\begin{align*}
\eta\!\left(1-\frac{L\eta}{2}\right)\!\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
&\le \E[f(w_0)] - f_\star + T\cdot \frac{L\eta^2}{2}\cdot \frac{\sigma^2}{B}. \quad \text{[Step 16]}
\end{align*}
Divide both sides by $\eta\left(1-\frac{L\eta}{2}\right)T$ (positive by Assumption \ref{assump:stepsize}):
\begin{align*}
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
&\le \frac{\E[f(w_0)] - f_\star}{\eta\left(1-\frac{L\eta}{2}\right)T}
+ \frac{L\eta}{2\left(1-\frac{L\eta}{2}\right)}\cdot \frac{\sigma^2}{B}. \quad \text{[Step 17]}
\end{align*}
Since $w_0$ is deterministic in the algorithm, we may replace $\E[f(w_0)]$ by $f(w_0)$. For a uniform random index $R$ independent of the iterates,
\[
\E[\|\nabla f(w_R)\|^2] = \frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2], \quad \text{by linearity and uniformity. \ [Step 18]}
\]
This proves the claimed bounds.
\end{proof}

\section*{Rate Derivation (With Constants)}
Define the positive constant $A(\eta)\triangleq \eta\left(1-\frac{L\eta}{2}\right)$ for $\eta\in(0,2/L)$. Then Theorem \ref{thm:main} reads
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{f(w_0)-f_\star}{A(\eta)\,T}
\;+\;
\frac{L\eta}{2\left(1-\frac{L\eta}{2}\right)}\cdot \frac{\sigma^2}{B}.
\]
Consequences:
- To ensure $\E[\|\nabla f(w_R)\|^2]\le \varepsilon$ with fixed $\eta$, it suffices that both
  \[
  T \;\ge\; \frac{f(w_0)-f_\star}{A(\eta)\,\varepsilon}
  \quad\text{and}\quad
  \eta \;\le\; \frac{2\varepsilon}{L(\sigma^2/B)+\varepsilon L}\ \text{(sufficient to make the floor $\le \varepsilon/2$).}
  \]
- A simple choice is $\eta = \min\left\{\frac{1}{L},\, \frac{\varepsilon B}{L\sigma^2}\right\}$, yielding
  \[
  \E[\|\nabla f(w_R)\|^2] \le \frac{2(f(w_0)-f_\star)}{\eta T} + \varepsilon \quad \text{(using $1-\frac{L\eta}{2}\ge \tfrac{1}{2}$ for $\eta\le \tfrac{1}{L}$).}
  \]

\section*{Special Cases}
\paragraph{Deterministic or full-batch case ($\sigma=0$).}
When $\sigma=0$ (equivalently, $g_t=\nabla f(w_t)$), inequality [Step 14] becomes
\[
f(w_{t+1}) \le f(w_t) - \eta\!\left(1-\frac{L\eta}{2}\right)\!\|\nabla f(w_t)\|^2. \quad \text{[Step 19]}
\]
Summing from $t=0$ to $T-1$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^2 \le \frac{f(w_0)-f_\star}{\eta\left(1-\frac{L\eta}{2}\right)T}. \quad \text{[Step 20]}
\]
Thus the averaged stationarity decays as $O(1/T)$ with fixed stepsize.

\paragraph{PL condition with $\sigma=0$ (linear convergence to optimum).}
Suppose in addition $f$ satisfies the Polyak--{\L}ojasiewicz (PL) inequality: there exists $\mu>0$ such that
\[
\|\nabla f(w)\|^2 \ge 2\mu\big(f(w)-f_\star\big), \quad \forall w\in\R^d.
\]
Using [Step 19] and the PL inequality,
\begin{align*}
f(w_{t+1}) - f_\star
&\le f(w_t) - f_\star - \eta\!\left(1-\frac{L\eta}{2}\right)\!\|\nabla f(w_t)\|^2 \quad \text{[Step 21a]}\\
&\le \left(1 - 2\mu\,\eta\!\left(1-\frac{L\eta}{2}\right)\right)\,\big(f(w_t)-f_\star\big). \quad \text{[Step 21b]}
\end{align*}
Let $\rho(\eta)\triangleq 1 - 2\mu\,\eta\left(1-\frac{L\eta}{2}\right) = 1 - 2\mu\eta + \mu L\eta^2$. For $0<\eta\le \tfrac{1}{L}$,
\[
\rho(\eta) \le 1 - \mu\eta \quad \text{since } \mu L\eta^2 - \mu\eta \le 0. \quad \text{[Step 22]}
\]
Hence $f(w_t)-f_\star \le \rho(\eta)^t \big(f(w_0)-f_\star\big) \le (1-\mu\eta)^t\big(f(w_0)-f_\star\big)$, i.e., linear convergence.
