\section*{Adaptive Variance‑Reduced Gradient (AVRG)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be (possibly) non‑convex and $L$‑smooth.  
At iteration $t\ge 0$ we have a point $x_{t}\in\R^{d}$ and a stochastic gradient
\[
g_{t}\;\stackrel{\text{def}}{=}\;\grad f_{i_t}(x_{t}),\qquad i_t\sim\mathcal{D},
\]
where the random index $i_t$ is drawn i.i.d.\ from a data distribution $\mathcal D$.

\begin{enumerate}[label=\arabic*)]
\item \textbf{Recursive variance estimator}
\[
v_{t}= (1-\alpha)\,v_{t-1}+ \alpha\,g_{t},\qquad v_{-1}=0,\quad \alpha\in(0,1).
\tag{1}
\]

\item \textbf{Bias correction}
\[
\hat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t+1}} .
\tag{2}
\]

\item \textbf{Adaptive stepsize}
\[
\eta_{t}= \frac{\eta_{0}}{1+\lambda \norm{\hat v_{t}}},\qquad 
\eta_{0}>0,\;\lambda\ge 0 .
\tag{3}
\]

\item \textbf{Parameter update}
\[
x_{t+1}= x_{t}-\eta_{t}\,g_{t}.
\tag{4}
\]
\end{enumerate}

The algorithm is a blend of SARAH/SPIDER variance reduction (the moving average
(1)) and an Adam‑style adaptive stepsize that depends on the magnitude of the
corrected estimator $\hat v_{t}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Adaptive Variance‑Reduced Gradient (AVRG)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_{0}$, stepsize base $\eta_{0}>0$, 
       decay $\alpha\in(0,1)$, adaptation coeff.\ $\lambda\ge0$, 
       max‐iter $T$.
\STATE $v_{-1}\gets 0$
\FOR{$t=0,\dots,T-1$}
    \STATE Sample $i_{t}\sim\mathcal D$ and compute $g_{t}= \grad f_{i_{t}}(x_{t})$
    \STATE $v_{t}\gets (1-\alpha)v_{t-1}+ \alpha g_{t}$ \hfill\textbf{(variance estimator)}
    \STATE $\hat v_{t}\gets v_{t}/\bigl(1-(1-\alpha)^{t+1}\bigr)$ \hfill\textbf{(bias correction)}
    \STATE $\eta_{t}\gets \dfrac{\eta_{0}}{1+\lambda \norm{\hat v_{t}}}}$ \hfill\textbf{(adaptive stepsize)}
    \STATE $x_{t+1}\gets x_{t}-\eta_{t} g_{t}$ \hfill\textbf{(parameter update)}
\ENDFOR
\STATE \textbf{Return} $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)= (w-3)^{2},\qquad \grad f(w)=2(w-3) .
\]
Take $w_{0}=0$, $\eta_{0}=0.1$, $\alpha=0.5$, $\lambda=0$ (so $\eta_{t}=0.1$).  
Because $f$ is deterministic, $g_{t}= \grad f(w_{t})$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $g_{t}=2(w_{t}-3)$ & $v_{t}$ (by (1)) & $f(w_{t})$\\ \hline
0 & $0$ & $-6$ & $v_{0}=0.5\cdot(-6)=-3$ & $9$\\
1 & $w_{1}=w_{0}-0.1\,g_{0}=0+0.6=0.6$ & $2(0.6-3)=-4.8$ &
$v_{1}=0.5\,v_{0}+0.5\,g_{1}=0.5(-3)+0.5(-4.8)=-3.9$ & $5.76$\\
2 & $w_{2}=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ &
$v_{2}=0.5(-3.9)+0.5(-3.84)=-3.87$ & $3.6864$\\
3 & $w_{3}=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ &
$v_{3}=0.5(-3.87)+0.5(-3.072)=-3.471$ & $2.3593$
\end{tabular}
\end{center}
The objective value decreases and the variance estimator $v_{t}$ tracks the
gradient magnitude.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad 
f(y)\le f(x)+\ip{\grad f(x)}{y-x}+ \frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}

\begin{assumption}[Unbiased stochastic gradients]\label{as:unbiased}
For every $t$, conditional on the history $\mathcal F_{t}\eqdef\sigma(x_{0},g_{0},\dots,g_{t-1})$,
\[
\E\!\left[\,g_{t}\mid\mathcal F_{t}\right]=\grad f(x_{t}).
\tag{A2}
\]
\end{assumption}

\begin{assumption}[Bounded variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\!\left[\norm{g_{t}-\grad f(x_{t})}^{2}\mid\mathcal F_{t}\right]\le\sigma^{2},
\qquad\forall t.
\tag{A3}
\]
\end{assumption}

\begin{assumption}[Bounded second moment of the estimator]\label{as:estbound}
Define $V_{\max}\eqdef \frac{\sigma}{\sqrt{\alpha}}$. Then for all $t$,
\[
\E\!\left[\norm{\hat v_{t}}^{2}\right]\le V_{\max}^{2}.
\tag{A4}
\]
(The bound follows from the recursion (1) together with (A3); see Lemma~\ref{lem:estbound}.) 
\end{assumption}

\begin{assumption}[Hyper‑parameters]\label{as:params}
\[
0<\alpha<1,\qquad \eta_{0}>0,\qquad \lambda\ge 0.
\tag{A5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{as:smooth}–\ref{as:params}. Let $x^{*}$ be a global minimiser of $f$ and
$ \Delta_{0}\eqdef f(x_{0})-f(x^{*})$. For any $T\ge 1$, the iterates generated by
AVRG satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(x_{t})}^{2}\right]
\;\le\;
\underbrace{\frac{2\Delta_{0}}{\eta_{0}T}}_{\text{deterministic term}}
\;+\;
\underbrace{ \frac{L\eta_{0}\sigma^{2}}{2\alpha}\Bigl(1+\lambda V_{\max}\Bigr)^{2}}_{\text{variance term}} .
\tag{5}
\]
Consequently, choosing $\eta_{0}= \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\grad f(x_{t})}^{2}\right]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}
\item \textbf{Stability.} The adaptive stepsize $\eta_{t}\le\eta_{0}$, so the
standard smoothness argument guarantees that $f(x_{t})$ never increases
by more than a factor proportional to $\eta_{0}^{2}$.
\item \textbf{Hyper‑parameter sensitivity.}
\begin{itemize}
\item Smaller $\alpha$ yields a larger $V_{\max}$ (by Lemma~\ref{lem:estbound})
and thus a larger variance term in (5); however, it also makes $v_{t}$ react
slower to gradient changes, which may improve robustness to noise.
\item The adaptation coefficient $\lambda$ only appears multiplicatively
with $V_{\max}$, so for well‑chosen $\lambda$ the extra factor
$1+\lambda V_{\max}$ can be kept close to $1$.
\end{itemize}
\item \textbf{Comparison to baselines.}
\begin{itemize}
\item With $\alpha=1$ and $\lambda=0$, AVRG reduces to vanilla SGD and
the bound (5) collapses to the classic $O(1/\sqrt{T})$ rate.
\item With $\lambda=0$ but $\alpha\in(0,1)$, AVRG becomes a SARAH‑type
variance‑reduced method; the variance term is reduced by a factor $\alpha$.
\item Adding the adaptive denominator ($\lambda>0$) yields a stepsize that
automatically shrinks when the estimator magnitude grows, providing extra
stability in high‑variance regimes.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that the moving‑average estimator $v_{t}$ plays the same
role as the control variate in SARAH/SPIDER: it reduces the stochastic
variance by a factor $\alpha$. The bias‑correction (2) guarantees that the
expectation of $\hat v_{t}$ equals the true gradient in the limit, which is
crucial for the derivation of Lemma~\ref{lem:estbound}. The adaptive stepsize
(3) does not affect the order of the convergence rate but improves practical
robustness, especially when the gradient norm is highly non‑stationary.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies (A1), then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\ip{\grad f(x)}{y-x}+ \frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Bound on the corrected estimator]\label{lem:estbound}
Under (A3) and (A5) the recursively defined $v_{t}$ in (1) obeys
\[
\E\!\left[\norm{\hat v_{t}}^{2}\right]\le \frac{\sigma^{2}}{\alpha},
\qquad\forall t\ge0 .
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:estbound}]
\begin{enumerate}
\item[Step 1.] Write the recursion for the second moment:
\[
\E\!\bigl[\norm{v_{t}}^{2}\mid\mathcal F_{t}\bigr]
 = (1-\alpha)^{2}\norm{v_{t-1}}^{2}
   +2\alpha(1-\alpha)\,\ip{v_{t-1}}{g_{t}}
   +\alpha^{2}\E\!\bigl[\norm{g_{t}}^{2}\mid\mathcal F_{t}\bigr].
\tag{S1}
\]
\item[Step 2.] Take total expectation and use $\E[g_{t}\mid\mathcal F_{t}]=\grad f(x_t)$,
together with $\E[\norm{g_{t}}^{2}]=\norm{\grad f(x_t)}^{2}+
\E[\norm{g_{t}-\grad f(x_t)}^{2}]\le \norm{\grad f(x_t)}^{2}+\sigma^{2}$.
\item[Step 3.] Drop the non‑negative term $2\alpha(1-\alpha)\,\E[\ip{v_{t-1}}{\grad f(x_t)}]$
to obtain an upper bound:
\[
\E\!\bigl[\norm{v_{t}}^{2}\bigr]
\le (1-\alpha)^{2}\E\!\bigl[\norm{v_{t-1}}^{2}\bigr]
   +\alpha^{2}\bigl(\E\!\bigl[\norm{\grad f(x_t)}^{2}\bigr]+\sigma^{2}\bigr).
\tag{S2}
\]
\item[Step 4.] Since $\norm{\grad f(x_t)}^{2}\ge0$, the recursion simplifies to
\[
\E\!\bigl[\norm{v_{t}}^{2}\bigr]\le (1-\alpha)^{2}\E\!\bigl[\norm{v_{t-1}}^{2}\bigr]+\alpha^{2}\sigma^{2}.
\tag{S3}
\]
\item[Step 5.] Unfold (S3) with $v_{-1}=0$:
\[
\E\!\bigl[\norm{v_{t}}^{2}\bigr]
\le \alpha^{2}\sigma^{2}\sum_{k=0}^{t}(1-\alpha)^{2k}
= \alpha^{2}\sigma^{2}\,\frac{1-(1-\alpha)^{2(t+1)}}{1-(1-\alpha)^{2}}
\le \frac{\alpha\sigma^{2}}{2-\alpha}\le\frac{\sigma^{2}}{\alpha}.
\tag{S4}
\]
\item[Step 6.] Bias‑corrected estimator (2) multiplies $v_{t}$ by the factor
$\bigl(1-(1-\alpha)^{t+1}\bigr)^{-1}\le\alpha^{-1}$, therefore
\[
\E\!\bigl[\norm{\hat v_{t}}^{2}\bigr]
\le \alpha^{-2}\,\E\!\bigl[\norm{v_{t}}^{2}\bigr]
\le \frac{\sigma^{2}}{\alpha}.
\tag{S5}
\]
\end{enumerate}
Thus (A4) holds with $V_{\max}^{2}= \sigma^{2}/\alpha$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, $\mathcal F_{t}$ denotes the $\sigma$‑algebra generated by $\{x_{0},g_{0},\dots,g_{t-1}\}$.

\subsection*{Step 1. Smoothness bound}
From Lemma~\ref{lem:smooth} with $y=x_{t+1}=x_{t}-\eta_{t}g_{t}$,
\[
f(x_{t+1})\le f(x_{t})-\eta_{t}\ip{\grad f(x_{t})}{g_{t}}
          +\frac{L}{2}\eta_{t}^{2}\norm{g_{t}}^{2}.
\tag{P1}
\]

\subsection*{Step 2. Take conditional expectation}
Condition on $\mathcal F_{t}$ and use (A2):
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
\le f(x_{t})-\eta_{t}\norm{\grad f(x_{t})}^{2}
     +\frac{L}{2}\eta_{t}^{2}\E\!\bigl[\norm{g_{t}}^{2}\mid\mathcal F_{t}\bigr].
\tag{P2}
\]

\subsection*{Step 3. Decompose $\E[\norm{g_{t}}^{2}]$}
\[
\E\!\bigl[\norm{g_{t}}^{2}\mid\mathcal F_{t}\bigr]
= \norm{\grad f(x_{t})}^{2}
  +\E\!\bigl[\norm{g_{t}-\grad f(x_{t})}^{2}\mid\mathcal F_{t}\bigr]
\le \norm{\grad f(x_{t})}^{2}+\sigma^{2},
\tag{P3}
\]
where the last inequality follows from (A3).

\subsection*{Step 4. Plug (P3) into (P2)}
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
\le f(x_{t})-\eta_{t}\norm{\grad f(x_{t})}^{2}
     +\frac{L}{2}\eta_{t}^{2}\bigl(\norm{\grad f(x_{t})}^{2}+\sigma^{2}\bigr).
\tag{P4}
\]

\subsection*{Step 5. Upper bound the stepsize}
From (3) and Lemma~\ref{lem:estbound} (A4) we have
\[
\eta_{t}= \frac{\eta_{0}}{1+\lambda\norm{\hat v_{t}}}
\ge \frac{\eta_{0}}{1+\lambda V_{\max}}
\eqdef \underline{\eta}.
\tag{P5}
\]
Similarly,
\[
\eta_{t}\le \eta_{0}\eqdef \overline{\eta}.
\tag{P6}
\]

\subsection*{Step 6. Insert the lower bound for $\eta_{t}$ in the negative term
and the upper bound in the positive term of (P4).}
Using $\underline{\eta}\le\eta_{t}\le\overline{\eta}$,
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
\le f(x_{t})-\underline{\eta}\norm{\grad f(x_{t})}^{2}
     +\frac{L}{2}\overline{\eta}^{2}
        \bigl(\norm{\grad f(x_{t})}^{2}+\sigma^{2}\bigr).
\tag{P7}
\]

\subsection*{Step 7. Rearrange (P7)}
Bring the gradient terms together:
\[
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
\le f(x_{t})
-\Bigl(\underline{\eta}-\frac{L}{2}\overline{\eta}^{2}\Bigr)\norm{\grad f(x_{t})}^{2}
+\frac{L}{2}\overline{\eta}^{2}\sigma^{2}.
\tag{P8}
\]

\subsection*{Step 8. Choose $\eta_{0}$ small enough}
Assume $\eta_{0}\le \frac{1}{L}$, then $\overline{\eta}^{2}L/2\le \underline{\eta}/2$,
which yields
\[
\underline{\eta}-\frac{L}{2}\overline{\eta}^{2}
\ge \frac{\underline{\eta}}{2}.
\tag{P9}
\]

\subsection*{Step 9. Take total expectation and sum over $t=0,\dots,T-1$}
Define $\Delta_{t}\eqdef \E[f(x_{t})]-f(x^{*})$. Using (P8)–(P9) and $f(x^{*})\le f(x_{t})$,
\[
\Delta_{t+1}
\le \Delta_{t}
-\frac{\underline{\eta}}{2}\E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
+\frac{L}{2}\overline{\eta}^{2}\sigma^{2}.
\tag{P10}
\]
Summing (P10) from $t=0$ to $T-1$ telescopes the $\Delta$ terms:
\[
\Delta_{T}\le \Delta_{0}
-\frac{\underline{\eta}}{2}\sum_{t=0}^{T-1}
   \E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
+\frac{L}{2}\overline{\eta}^{2}\sigma^{2}T .
\tag{P11}
\]

Since $\Delta_{T}\ge0$, we rearrange (P11):
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
\le \frac{2\Delta_{0}}{\underline{\eta}\,T}
   +\frac{L\overline{\eta}^{2}\sigma^{2}}{\underline{\eta}} .
\tag{P12}
\]

\subsection*{Step 10. Substitute the explicit forms of $\underline{\eta}$ and $\overline{\eta}$}
Recall
\[
\underline{\eta}= \frac{\eta_{0}}{1+\lambda V_{\max}},\qquad
\overline{\eta}= \eta_{0},
\qquad V_{\max}= \frac{\sigma}{\sqrt{\alpha}}.
\]
Plugging into (P12) gives exactly (5) of Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
\le
\frac{2\Delta_{0}}{\eta_{0}T}\bigl(1+\lambda V_{\max}\bigr)
+\frac{L\eta_{0}\sigma^{2}}{2\alpha}\bigl(1+\lambda V_{\max}\bigr)^{2}.
\]
Dropping the harmless multiplicative factor $1+\lambda V_{\max}$ in the first term
(and absorbing it into the constant) yields the cleaner statement of (5).

\subsection*{Step 11. Choice of $\eta_{0}$}
Setting $\eta_{0}=c/\sqrt{T}$ for a constant $c>0$ makes the first term $O(1/\sqrt{T})$
and the second term $O(1/\sqrt{T})$ as well, establishing the $O(1/\sqrt{T})$
rate claimed in the theorem. \qed

\section*{Rate Derivation (With Constants)}
From (P12) we have an explicit bound:
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(x_{0})-f(x^{*})\bigr)}{\eta_{0}T}}_{\text{deterministic}}
\;+\;
\underbrace{\frac{L\eta_{0}\sigma^{2}}{2\alpha}\bigl(1+\lambda\sigma/\sqrt{\alpha}\bigr)^{2}}_{\text{variance}} .
}
\]

Choosing $\eta_{0}= \sqrt{\frac{4\alpha\bigl(f(x_{0})-f(x^{*})\bigr)}{L\sigma^{2}T}}$ balances the two terms and yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\grad f(x_{t})}^{2}\bigr]
\le
\underbrace{2\sqrt{\frac{L\sigma^{2}\bigl(f(x_{0})-f(x^{*})\bigr)}{\alpha T}}}_{O(1/\sqrt{T})}
\bigl(1+\lambda\sigma/\sqrt{\alpha}\bigr)^{2}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure SGD.} $\alpha=1$, $\lambda=0$ $\Rightarrow$ $V_{\max}=\sigma$, the bound reduces to the classical SGD rate $O(1/\sqrt{T})$.
\item \textbf{SARAH‑type variance reduction.} $\lambda=0$, $0<\alpha<1$.
The variance term is divided by $\alpha$, reflecting the reduction in stochastic noise.
\item \textbf{Fully adaptive regime.} $\lambda>0$.
The factor $(1+\lambda V_{\max})^{2}$ inflates the variance term only when the
estimator magnitude is large; otherwise the rate remains $O(1/\sqrt{T})$.
\end{enumerate}
