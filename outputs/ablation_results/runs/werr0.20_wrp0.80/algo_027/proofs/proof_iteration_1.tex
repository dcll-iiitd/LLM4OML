\section*{Probabilistic Coordinate Descent (PCD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable (possibly non‑convex).  
For each coordinate $i\in\{1,\dots,d\}$ let a probability $p_i>0$ be assigned such that 
\[
\sum_{i=1}^{d}p_i =1 .
\]
At iteration $t$ a random index $i_t$ is drawn according to $\{p_i\}$.  
Define the (scaled) stochastic partial gradient
\[
g_t \;:=\; \frac{1}{p_{i_t}}\;e_{i_t}\,\partial_{i_t}f(w_t),
\]
where $e_i$ is the $i$‑th canonical basis vector and $\partial_i f$ denotes the partial derivative w.r.t.\ $i$.
The update is
\begin{equation}
\boxed{ \; w_{t+1}= w_t - \eta\, g_t \;}
\label{eq:update}
\end{equation}
with a stepsize $\eta>0$.  
Because of the $1/p_{i_t}$ factor,
\[
\E[g_t\mid w_t]=\sum_{i=1}^{d}p_i\frac{1}{p_i}e_i\partial_i f(w_t)=\grad f(w_t),
\]
so the expected update coincides with a full gradient step.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Probabilistic Coordinate Descent (PCD)}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\R^{d}$, stepsize $\eta>0$, probability vector $p\in\Delta^{d-1}$.
\For{$t=0,1,2,\dots,T-1$}
    \State Sample $i_t\in\{1,\dots,d\}$ with $\Pr(i_t=i)=p_i$.
    \State Compute partial derivative $g^{(i_t)}_t:=\partial_{i_t}f(w_t)$.
    \State Form scaled stochastic gradient 
           $g_t:=\frac{1}{p_{i_t}}\,e_{i_t}\,g^{(i_t)}_t$.
    \State Update the selected coordinate 
           $w_{t+1}= w_t-\eta\,g_t$.
\EndFor
\State \Return $w_T$ (or optionally the averaged iterate $\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem $f(w)=(w-3)^2$, $d=1$, $p_1=1$, stepsize $\eta=0.1$, $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $\partial f(w_t)$ & $f(w_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $w_1=0-0.1\cdot\frac{1}{1}(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.366\,$
\end{tabular}
\end{center}
The algorithm follows the exact gradient descent trajectory because $p_1=1$.

\section*{Assumptions}
\begin{assumption}[Coordinate‑wise $L$‑smoothness]\label{ass:smooth}
For each $i\in\{1,\dots,d\}$ there exists $L_i>0$ such that for all $w\in\R^{d}$ and any scalar $\alpha$,
\[
\bigl|\,\partial_i f(w+ \alpha e_i)-\partial_i f(w)\,\bigr|\le L_i |\alpha|.
\]
Equivalently,
\[
f(w+ \alpha e_i)\le f(w)+\alpha\,\partial_i f(w)+\tfrac{L_i}{2}\alpha^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradient]\label{ass:var}
There exists $\sigma^{2}<\infty$ such that for all $t$,
\[
\E\!\left[\|g_t-\grad f(w_t)\|^{2}\mid w_t\right]\le \sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize $\eta$ is chosen so that $0<\eta\le \min_{i}\frac{1}{L_i}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:stepsize}. Let $\{w_t\}$ be generated by PCD with any $T\ge 1$. Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\grad f(w_t)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta\,T}
\;+\;
2\eta\,\sigma^{2},
\]
where $f^{\star}= \inf_{w}f(w)$.
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[\|\grad f(w_t)\|^{2}\bigr]=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Unbiasedness.} By construction $\E[g_t\mid w_t]=\grad f(w_t)$, hence the method is an unbiased estimator of full‑gradient descent.
\item \textbf{Stability.} The stepsize restriction \ref{ass:stepsize} guarantees that each coordinate update does not increase the objective by more than a quadratic term, yielding a descent in expectation.
\item \textbf{Hyperparameter sensitivity.} The bound depends linearly on $\eta$ and on $\sigma^{2}$; a larger stepsize accelerates progress but amplifies the variance term $2\eta\sigma^{2}$.
\item \textbf{Comparison.} Compared with uniform coordinate descent ($p_i=1/d$) the scaling $1/p_i$ removes the bias caused by non‑uniform sampling, leading to the same $O(T^{-1/2})$ rate as standard stochastic gradient for non‑convex objectives.
\end{itemize}

\section*{Discussion}
The theorem establishes that PCD attains the canonical $O(T^{-1/2})$ rate for finding an approximate stationary point in the non‑convex, coordinate‑wise smooth setting. The proof follows the standard descent‑lemma argument but requires an extra weighting by $1/p_i$ when taking expectations; this step is highlighted in the detailed proof below.

\newpage
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]\label{lem:descent}
Under Assumption \ref{ass:smooth}, for any $w\in\R^{d}$ and any $i$,
\[
f\bigl(w- \eta\,\tfrac{1}{p_i}e_i\partial_i f(w)\bigr)
\le f(w)-\frac{\eta}{p_i}\,\partial_i f(w)^{2}
+\frac{L_i\eta^{2}}{2p_i^{2}}\partial_i f(w)^{2}.
\]
\end{lemma}
\begin{proof}
Apply the coordinate‑wise smoothness inequality with $\alpha=-\eta\,\frac{1}{p_i}\partial_i f(w)$:
\[
f(w+\alpha e_i)\le f(w)+\alpha\partial_i f(w)+\tfrac{L_i}{2}\alpha^{2}.
\]
Insert $\alpha$ and simplify to obtain the claimed bound. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof we condition on the filtration $\mathcal{F}_t:=\sigma(w_0,\dots,w_t)$.

\begin{align}
\text{[Step 1]} \quad 
&\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
 = \sum_{i=1}^{d}p_i\,
    f\!\bigl(w_t-\eta\,\tfrac{1}{p_i}e_i\partial_i f(w_t)\bigr) .
\end{align}
Using Lemma \ref{lem:descent} for each term,
\begin{align}
\text{[Step 2]} \quad 
&\le \sum_{i=1}^{d}p_i\Bigl(
      f(w_t)-\frac{\eta}{p_i}\partial_i f(w_t)^{2}
      +\frac{L_i\eta^{2}}{2p_i^{2}}\partial_i f(w_t)^{2}
      \Bigr) \nonumber\\
&= f(w_t)-\eta\sum_{i=1}^{d}\partial_i f(w_t)^{2}
   +\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}\partial_i f(w_t)^{2}.
\end{align}
Define $L_{\max}:=\max_i L_i$ and note $\frac{L_i}{p_i}\le \frac{L_{\max}}{\min_j p_j}$.
Under Assumption \ref{ass:stepsize} we have $\eta\le 1/L_{\max}$, hence
\begin{align}
\text{[Step 3]} \quad 
&\frac{\eta^{2}}{2}\sum_{i=1}^{d}\frac{L_i}{p_i}\partial_i f(w_t)^{2}
 \le \frac{\eta}{2}\sum_{i=1}^{d}\partial_i f(w_t)^{2}.
\end{align}
Insert \eqref{Step 3} into \eqref{Step 2}:
\begin{align}
\text{[Step 4]} \quad 
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\frac{\eta}{2}\,\|\grad f(w_t)\|^{2}.
\end{align}
Now we relate the stochastic gradient $g_t$ to the true gradient.
From the definition of $g_t$,
\[
g_t = \grad f(w_t)+\bigl(g_t-\grad f(w_t)\bigr).
\]
Hence
\begin{align}
\text{[Step 5]} \quad 
\|\grad f(w_t)\|^{2}
&= \|g_t\|^{2}
   -2\ip{g_t}{g_t-\grad f(w_t)}
   +\|g_t-\grad f(w_t)\|^{2}.
\end{align}
Taking conditional expectation and using $\E[g_t\mid\mathcal{F}_t]=\grad f(w_t)$ yields
\begin{align}
\text{[Step 6]} \quad 
\E\!\bigl[\|\grad f(w_t)\|^{2}\mid\mathcal{F}_t\bigr]
&= \E\!\bigl[\|g_t\|^{2}\mid\mathcal{F}_t\bigr]
   -\E\!\bigl[\|g_t-\grad f(w_t)\|^{2}\mid\mathcal{F}_t\bigr].
\end{align}
By Assumption \ref{ass:var},
\[
\E\!\bigl[\|g_t-\grad f(w_t)\|^{2}\mid\mathcal{F}_t\bigr]\le\sigma^{2},
\]
so from \eqref{Step 6}
\begin{align}
\text{[Step 7]} \quad 
\E\!\bigl[\|\grad f(w_t)\|^{2}\mid\mathcal{F}_t\bigr]
\ge \E\!\bigl[\|g_t\|^{2}\mid\mathcal{F}_t\bigr]-\sigma^{2}.
\end{align}
Combine \eqref{Step 4} with the tower property of expectation:
\begin{align}
\text{[Step 8]} \quad 
\E[f(w_{t+1})]
&\le \E[f(w_t)]-\frac{\eta}{2}\,\E\!\bigl[\|\grad f(w_t)\|^{2}\bigr] \nonumber\\
&\le \E[f(w_t)]-\frac{\eta}{2}\,
      \bigl(\E[\|g_t\|^{2}]-\sigma^{2}\bigr).
\end{align}
Rearranging,
\begin{align}
\text{[Step 9]} \quad 
\frac{\eta}{2}\,\E[\|g_t\|^{2}]
\le \E[f(w_t)]-\E[f(w_{t+1})] +\frac{\eta\sigma^{2}}{2}.
\end{align}
Summing \eqref{Step 9} over $t=0,\dots,T-1$ telescopes the objective terms:
\begin{align}
\text{[Step 10]} \quad 
\frac{\eta}{2}\sum_{t=0}^{T-1}\E[\|g_t\|^{2}]
&\le f(w_0)-\E[f(w_T)] +\frac{\eta\sigma^{2}T}{2} \nonumber\\
&\le f(w_0)-f^{\star} +\frac{\eta\sigma^{2}T}{2}.
\end{align}
Divide by $T$ and use \eqref{Step 7} (inverse inequality) to replace $\E[\|g_t\|^{2}]$ by $\E[\|\grad f(w_t)\|^{2}]+\sigma^{2}$:
\begin{align}
\text{[Step 11]} \quad 
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\grad f(w_t)\|^{2}]
&\le \frac{2\bigl(f(w_0)-f^{\star}\bigr)}{\eta T}
   +2\eta\sigma^{2}.
\end{align}
\eqref{Step 11} is exactly the bound claimed in Theorem \ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main} set $\eta = \sqrt{\dfrac{2\bigl(f(w_0)-f^{\star}\bigr)}{\sigma^{2}T}}$, which satisfies Assumption \ref{ass:stepsize} for sufficiently large $T$ because $\eta=O(T^{-1/2})$. Substituting yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\grad f(w_t)\|^{2}]
\le 2\sqrt{\frac{2\sigma^{2}\bigl(f(w_0)-f^{\star}\bigr)}{T}}
= O\!\bigl(T^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Uniform sampling.} If $p_i=1/d$ for all $i$, the update reduces to the classic randomized coordinate descent with stepsize $\eta d$; the bound remains $O(T^{-1/2})$.
\item \textbf{Deterministic cyclic coordinate descent.} The analysis does not directly apply because the unbiasedness property (\ref{Step 1}) fails; however, with $p_i=1/d$ and a deterministic schedule the same $L$‑smoothness constants give a deterministic descent guarantee.
\end{itemize}
