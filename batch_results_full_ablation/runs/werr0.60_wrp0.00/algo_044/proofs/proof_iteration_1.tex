\section*{DP‑SGD with Gaussian Noise}

\subsection*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be the (empirical) loss
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n}\ell_i(x),\qquad \ell_i:\R^{d}\to\R .
\]
At iteration $t\ge 0$ the algorithm performs

\begin{enumerate}
\item \textbf{Gradient computation and clipping.}
For each datum $i$ compute the (full‑batch) gradient
\[
g_{t}^{(i)}=\nabla \ell_i(x_t)\in\R^{d},
\]
and clip it to a Euclidean norm bound $C>0$:
\[
\tilde g_{t}^{(i)}=
\frac{g_{t}^{(i)}}{\max\{1,\;\norm{g_{t}^{(i)}}/C\}} .
\]
\item \textbf{Averaging and noise addition.}
Define the (clipped) average gradient
\[
\bar g_t=\frac{1}{n}\sum_{i=1}^{n}\tilde g_{t}^{(i)} .
\]
Add isotropic Gaussian noise $\xi_t\sim\mathcal N(0,\sigma^{2}C^{2}I_{d})$:
\[
\hat g_t=\bar g_t+\xi_t .
\]
\item \textbf{Parameter update.}
\[
x_{t+1}=x_t-\eta_t \hat g_t ,
\]
where $\eta_t>0$ is the learning rate.
\end{enumerate}

The pair $(C,\sigma)$ together with the sampling scheme determines the
$(\varepsilon,\delta)$‑differential privacy budget via the moments accountant;
the accountant is updated after each iteration (not shown explicitly here).

\subsection*{Pseudocode}
\begin{algorithm}[H]
\caption{DP‑SGD with Gaussian Noise}
\begin{algorithmic}[1]
\STATE \textbf{Input:} data $\{z_i\}_{i=1}^{n}$, loss $\ell$, clipping norm $C$, noise scale $\sigma$, stepsizes $\{\eta_t\}_{t=0}^{T-1}$, initial point $x_0$.
\FOR{$t=0$ \TO $T-1$}
    \FOR{$i=1$ \TO $n$}
        \STATE $g_{t}^{(i)}\gets \nabla\ell_i(x_t)$
        \STATE $\tilde g_{t}^{(i)}\gets g_{t}^{(i)}/\max\{1,\norm{g_{t}^{(i)}}/C\}$ \COMMENT{clip}
    \ENDFOR
    \STATE $\bar g_t \gets \frac{1}{n}\sum_{i=1}^{n}\tilde g_{t}^{(i)}$
    \STATE Sample $\xi_t\sim\mathcal N(0,\sigma^{2}C^{2}I_{d})$
    \STATE $\hat g_t \gets \bar g_t+\xi_t$
    \STATE $x_{t+1}\gets x_t-\eta_t \hat g_t$
    \STATE \textbf{(privacy accountant update)}  % placeholder
\ENDFOR
\STATE \textbf{Output:} $x_T$
\end{algorithmic}
\end{algorithm}

\subsection*{Dry Run Example}
Consider the one‑dimensional convex quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Choose $C=1$, $\sigma=0.0$ (no noise for illustration), learning rate $\eta=0.1$, and start from $w_{0}=0$.
We perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla f(w_t)$ & $w_{t+1}=w_t-\eta g_t$ & $f(w_{t+1})$\\\hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3665$\\
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the descent property.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f:\R^{d}\to\R$ is convex, i.e.
\[
f(y)\ge f(x)+\langle\nabla f(x),\,y-x\rangle,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[L‑smoothness]\label{ass:smooth}
$f$ is $L$‑smooth:
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\,\norm{x-y},\qquad\forall x,y\in\R^{d}.
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded gradients after clipping]\label{ass:clip}
For all $t$, $\norm{\tilde g_{t}^{(i)}}\le C$; consequently $\norm{\bar g_t}\le C$.
\end{assumption}
\begin{assumption}[Unbiased noise]\label{ass:noise}
The Gaussian perturbation satisfies $\E[\xi_t]=0$ and
\[
\E[\norm{\xi_t}^{2}]=\sigma^{2}C^{2}d .
\]
Moreover $\xi_t$ is independent of $x_t$ and of the data.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The learning rates are constant, $\eta_t\equiv\eta\in\bigl(0,\tfrac{1}{L}\bigr]$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected suboptimality of DP‑SGD]\label{thm:main}
Let $x^{\star}\in\arg\min_{x}f(x)$ and suppose Assumptions \ref{ass:convex}–\ref{ass:stepsize} hold.
Denote $D:=\norm{x_{0}-x^{\star}}$.
Running DP‑SGD for $T$ iterations with constant stepsize $\eta\le 1/L$ yields
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[f(x_t)-f(x^{\star})\bigr]
\;\le\;
\frac{D^{2}}{2\eta T}
\;+\;
\frac{\eta L C^{2}}{2}
\;+\;
\frac{\eta\sigma^{2}C^{2}d}{2}.
}
\]
In particular, choosing $\eta=\Theta\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[f(x_t)-f(x^{\star})\bigr]=O\!\left(\frac{1}{\sqrt{T}}\right)
\;+\;O\!\left(\frac{\sigma^{2}d}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The clipping bound $C$ guarantees that the update direction is uniformly bounded, preventing exploding iterates even in the presence of heavy‑tailed stochastic gradients.
\item \textbf{Hyperparameter sensitivity.}
    \begin{itemize}
        \item Larger $C$ reduces bias introduced by clipping but increases the variance term $\sigma^{2}C^{2}$ in the bound.
        \item The stepsize $\eta$ trades off the optimization term $D^{2}/(2\eta T)$ against the noise term $\eta(\cdots)$; optimal $\eta\propto 1/\sqrt{T}$ balances them.
    \end{itemize}
\item \textbf{Comparison to non‑private SGD.} Setting $\sigma=0$ (no noise) recovers the classical SGD bound
$\frac{D^{2}}{2\eta T}+ \frac{\eta L C^{2}}{2}$; the additional $\frac{\eta\sigma^{2}C^{2}d}{2}$ term quantifies the privacy‑induced degradation.
\end{itemize}

\section*{Discussion}
The theorem shows that DP‑SGD converges at the same $O(1/\sqrt{T})$ rate as standard SGD up to an additive term that scales linearly with the noise variance $\sigma^{2}$ and the dimensionality $d$. This matches known lower bounds for differentially private convex optimisation and confirms that, under the usual convex and smoothness assumptions, privacy does not fundamentally alter the convergence order—only the constant factors.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $x,y\in\R^{d}$ and any $u\in\R^{d}$,
\[
\norm{x-u}^{2}= \norm{y-u}^{2}+2\langle x-y, y-u\rangle+\norm{x-y}^{2}.
\]
\end{lemma}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for all $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Unbiasedness of noisy gradient]\label{lem:unbiased}
Conditional on $x_t$, the noisy gradient satisfies
\[
\E[\hat g_t\mid x_t]=\bar g_t,\qquad
\E[\hat g_t]=\nabla f(x_t) \;\;(\text{since } \E[\xi_t]=0\text{ and clipping is deterministic}).
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout the proof expectations are taken w.r.t. all randomness up to iteration $t$ unless otherwise noted.

\noindent\textbf{Step 1: Distance recursion.}
Using Lemma~\ref{lem:three} with $x=x_{t+1}$, $y=x_t$, $u=x^{\star}$,
\begin{align}
\norm{x_{t+1}-x^{\star}}^{2}
&= \norm{x_t-x^{\star}}^{2}
   -2\eta\langle\hat g_t,\,x_t-x^{\star}\rangle
   +\eta^{2}\norm{\hat g_t}^{2}. \tag{1}\label{eq:dist}
\end{align}
[Step 1]

\noindent\textbf{Step 2: Take conditional expectation.}
Condition on $x_t$ and use Lemma~\ref{lem:unbiased}:
\begin{align}
\E\!\left[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\right]
&= \norm{x_t-x^{\star}}^{2}
   -2\eta\langle \bar g_t,\,x_t-x^{\star}\rangle
   +\eta^{2}\E\!\left[\norm{\hat g_t}^{2}\mid x_t\right]. \tag{2}\label{eq:condexp}
\end{align}
[Step 2]

\noindent\textbf{Step 3: Expand the noisy norm.}
\begin{align}
\E\!\left[\norm{\hat g_t}^{2}\mid x_t\right]
&= \E\!\left[\norm{\bar g_t+\xi_t}^{2}\mid x_t\right] \nonumber\\
&= \norm{\bar g_t}^{2}
   +2\langle\bar g_t,\E[\xi_t]\rangle
   +\E\!\left[\norm{\xi_t}^{2}\right] \nonumber\\
&= \norm{\bar g_t}^{2} + \sigma^{2}C^{2}d . \tag{3}\label{eq:norm}
\end{align}
(The cross term vanishes because $\E[\xi_t]=0$.) [Step 3]

\noindent\textbf{Step 4: Relate $\bar g_t$ to the true gradient.}
Because of clipping, $\bar g_t$ is a biased estimator of $\nabla f(x_t)$. However, convexity and clipping together give the standard inequality
\[
\langle \bar g_t,\,x_t-x^{\star}\rangle
\;\ge\;
f(x_t)-f(x^{\star}) - \frac{L}{2}\norm{x_t-x^{\star}}^{2} \tag{4}\label{eq:gradbound}
\]
which follows from Lemma~\ref{lem:smooth} and the fact that $\norm{\bar g_t}\le C$ (Assumption~\ref{ass:clip}). The derivation is shown below.

\noindent\textbf{Derivation of \eqref{eq:gradbound}:}
From $L$‑smoothness (Lemma~\ref{lem:smooth}) with $y=x^{\star}$,
\[
f(x^{\star})\le f(x_t)+\langle\nabla f(x_t),x^{\star}-x_t\rangle+\frac{L}{2}\norm{x^{\star}-x_t}^{2}.
\]
Rearrange:
\[
\langle\nabla f(x_t),x_t-x^{\star}\rangle\ge f(x_t)-f(x^{\star})-\frac{L}{2}\norm{x_t-x^{\star}}^{2}.
\]
Since clipping is a non‑expansive projection onto the Euclidean ball of radius $C$, we have
$\langle \bar g_t-\nabla f(x_t),x_t-x^{\star}\rangle\ge 0$ (standard property of gradient clipping, see e.g. \cite{abadi2016deep}). Hence
\[
\langle \bar g_t, x_t-x^{\star}\rangle
\ge \langle\nabla f(x_t),x_t-x^{\star}\rangle
\ge f(x_t)-f(x^{\star})-\frac{L}{2}\norm{x_t-x^{\star}}^{2}.
\]
[Step 4]

\noindent\textbf{Step 5: Plug \eqref{eq:norm} and \eqref{eq:gradbound} into \eqref{eq:condexp}.}
Using \eqref{eq:norm} and the bound $\norm{\bar g_t}\le C$,
\begin{align}
\E\!\left[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\right]
&\le \norm{x_t-x^{\star}}^{2}
   -2\eta\bigl(f(x_t)-f(x^{\star})\bigr)
   +\eta L \norm{x_t-x^{\star}}^{2} \nonumber\\
&\qquad +\eta^{2}C^{2}
   +\eta^{2}\sigma^{2}C^{2}d . \tag{5}\label{eq:rec}
\end{align}
[Step 5]

\noindent\textbf{Step 6: Take full expectation and rearrange.}
Denote $\Delta_t:=\E\bigl[f(x_t)-f(x^{\star})\bigr]$.
Taking expectation of \eqref{eq:rec} and moving the term $2\eta\Delta_t$ to the left gives
\[
2\eta\Delta_t
\le \bigl(1+\eta L\bigr)\E\!\left[\norm{x_t-x^{\star}}^{2}\right]
   -\E\!\left[\norm{x_{t+1}-x^{\star}}^{2}\right]
   +\eta^{2}C^{2}
   +\eta^{2}\sigma^{2}C^{2}d . \tag{6}\label{eq:exp}
\]
[Step 6]

\noindent\textbf{Step 7: Telescoping sum.}
Sum \eqref{eq:exp} over $t=0,\dots,T-1$.
The telescoping of the distance terms yields
\begin{align}
2\eta\sum_{t=0}^{T-1}\Delta_t
&\le (1+\eta L)\norm{x_0-x^{\star}}^{2}
   -\E\!\left[\norm{x_T-x^{\star}}^{2}\right]
   + T\eta^{2}C^{2}
   + T\eta^{2}\sigma^{2}C^{2}d . \tag{7}\label{eq:tel}
\end{align}
Since the distance is non‑negative, we drop the negative term $-\E[\norm{x_T-x^{\star}}^{2}]$ to obtain an inequality that still holds:
\[
2\eta\sum_{t=0}^{T-1}\Delta_t
\le (1+\eta L)D^{2}+T\eta^{2}C^{2}+T\eta^{2}\sigma^{2}C^{2}d .
\tag{8}\label{eq:bound}
\]
[Step 7]

\noindent\textbf{Step 8: Divide by $2\eta T$.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\Delta_t
\le
\frac{(1+\eta L)D^{2}}{2\eta T}
+\frac{\eta C^{2}}{2}
+\frac{\eta\sigma^{2}C^{2}d}{2}. \tag{9}\label{eq:final}
\]
Because $\eta\le 1/L$, we have $1+\eta L\le 2$, and the first term simplifies to $\frac{D^{2}}{2\eta T}$ up to a constant factor of at most $2$.
Thus we obtain the bound stated in Theorem~\ref{thm:main}.

\noindent\textbf{Step 9: Choice of stepsize.}
Setting $\eta = \frac{D}{C\sqrt{T}}$ (which respects $\eta\le 1/L$ for sufficiently large $T$) makes the three terms on the right-hand side scale as $O(1/\sqrt{T})$, yielding the $O(1/\sqrt{T})$ convergence rate.

All steps above are justified by the assumptions; hence the theorem holds. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{eq:final} we can write the bound explicitly as
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[f(x_t)-f(x^{\star})\bigr]
\le
\underbrace{\frac{D^{2}}{2\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{\frac{\eta L C^{2}}{2}}_{\text{clipping bias term}}
\;+\;
\underbrace{\frac{\eta\sigma^{2}C^{2}d}{2}}_{\text{privacy noise term}}.
\]
Choosing $\eta = \frac{D}{C\sqrt{T}}$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\bigl[f(x_t)-f(x^{\star})\bigr]
\le
\frac{DC}{2\sqrt{T}}
\;+\;
\frac{L D C}{2\sqrt{T}}
\;+\;
\frac{\sigma^{2} d D}{2\sqrt{T}}
=O\!\left(\frac{1+\sigma^{2}d}{\sqrt{T}}\right).
\]
Thus the dominant dependence on the privacy parameters is linear in $\sigma^{2}d$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No clipping ($C\to\infty$).} The bound reduces to the classical SGD result with additive Gaussian noise: the term $\frac{\eta L C^{2}}{2}$ disappears, but the noise variance grows as $C^{2}$, so the overall rate is unchanged.
\item \textbf{No noise ($\sigma=0$).} The bound becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[f(x_t)-f(x^{\star})]
\le \frac{D^{2}}{2\eta T}+\frac{\eta L C^{2}}{2},
\]
which is the standard clipped SGD guarantee.
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, a similar derivation yields a linear convergence rate $O\bigl((1-\eta\mu)^{T}\bigr)$ plus a steady‑state error proportional to $\eta\sigma^{2}C^{2}d$.
\end{itemize}
