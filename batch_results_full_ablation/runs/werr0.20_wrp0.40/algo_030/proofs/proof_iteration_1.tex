\section*{Heavy–Ball Momentum (HB) for Smooth Non‑Convex Optimization}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be differentiable.  
Given an initial pair $(x_{0},x_{-1})\in\mathbb{R}^d\times\mathbb{R}^d$, a stepsize $\alpha>0$ and a momentum parameter $\beta\in[0,1)$, the Heavy–Ball (HB) iteration is  

\[
\boxed{%
x_{k+1}=x_{k}+ \beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha \nabla f(x_{k}),\qquad k=0,1,2,\dots
}
\tag{HB}
\]

The term $\beta(x_{k}-x_{k-1})$ is a \emph{velocity} (momentum) built from the previous displacement, while the gradient term uses the \emph{current} gradient (no look‑ahead).

\section*{Pseudocode}
\begin{verbatim}
Input: stepsize α>0, momentum β∈[0,1), 
       initial points x0∈R^d, x_{-1}=x0 (or any choice)
for k = 0,1,2,… do
    gk = ∇f(xk)                     # current gradient
    vk = β (xk - x_{k-1})           # momentum term
    x_{k+1} = xk + vk - α * gk      # Heavy‑Ball update
    if stopping criterion satisfied then break
end for
return x_{k+1}
\end{verbatim}

\section*{Dry Run Example}
Consider the one–dimensional quadratic  

\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]

Choose $\alpha=0.1$, $\beta=0.5$, $x_{-1}=x_0=0$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $x_k$ & $\nabla f(x_k)$ & $f(x_k)$\\\hline
0 & $0$ & $-6$ & $9$\\
1 & $0.6$ & $-4.8$ & $5.76$\\
2 & $1.38$ & $-3.24$ & $2.6244$\\
3 & $2.094$ & $-1.812$ & $0.822$\\
\end{tabular}
\end{center}

The updates are obtained as follows  

\[
\begin{aligned}
x_{1}&=0+0.5(0-0)-0.1(-6)=0.6,\\
x_{2}&=0.6+0.5(0.6-0)-0.1(-4.8)=1.38,\\
x_{3}&=1.38+0.5(1.38-0.6)-0.1(-3.24)=2.094.
\end{aligned}
\]

The objective value decreases monotonically, illustrating the descent property of HB on a simple non‑convex (here convex) function.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.\ for all $x,y\in\mathbb{R}^d$,
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\|y-x\|^2.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Lower Boundedness]\label{ass:lower}
$f$ is bounded below: $f^\star:=\inf_{x}f(x)>-\infty$.
\tag{A2}
\end{assumption}

\begin{assumption}[Stepsize and Momentum]\label{ass:params}
The parameters satisfy  
\[
0<\alpha\le\frac{1}{L},\qquad
0\le\beta\le\frac{1-\alpha L}{1+\alpha L}\;(<1).
\tag{A3}
\]
\end{assumption}

All results below are derived under \ref{ass:smooth}–\ref{ass:params}.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of Heavy‑Ball]\label{thm:hb}
Let $\{x_k\}$ be generated by \eqref{HB} under Assumptions \ref{ass:smooth}–\ref{ass:params}.  
Define the Lyapunov function  

\[
\Phi_k\;:=\;f(x_k)+\frac{\beta}{2\alpha}\,\|x_k-x_{k-1}\|^2,\qquad k\ge0 .
\tag{T1}
\]

Then for every $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha\,K}.
\tag{T2}
\]

Consequently $\displaystyle\lim_{K\to\infty}\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2 =0$, i.e. the average squared gradient norm converges to zero at the rate $O(1/K)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Descent of a Lyapunov function.}  The quantity $\Phi_k$ combines objective value and kinetic energy $\|x_k-x_{k-1}\|^2$. Under \ref{ass:params} it is non‑increasing (Lemma~\ref{lem:descent}).
\item \textbf{Stability w.r.t.\ $\beta$.}  The admissible range \eqref{A3} guarantees that the momentum does not amplify the iterates; as $\beta\to0$ the method reduces to gradient descent with stepsize $\alpha$.
\item \textbf{Comparison to Gradient Descent.}  For the same stepsize $\alpha\le 1/L$, GD satisfies
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k^{\text{GD}})\|^2\le\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha K},
\]
the same $O(1/K)$ rate.  HB may exhibit faster practical progress because the momentum term can accelerate the decrease of $f$ while preserving the same worst‑case bound.
\item \textbf{Hyper‑parameter sensitivity.}  If $\beta$ exceeds the bound in \ref{ass:params}, the descent property can be lost and the iterates may diverge (illustrated in many numerical studies).  The bound is tight for the worst‑case $L$‑smooth quadratic.
\end{enumerate}

\section*{Discussion}
The theorem guarantees that any limit point of $\{x_k\}$ is a \emph{stationary point} (since a subsequence with vanishing gradient exists).  No assumption of convexity is required, making the result applicable to a broad class of deep‑learning loss landscapes.  The proof technique—Lyapunov descent—mirrors the analysis of Nesterov’s accelerated method, yet the heavy‑ball dynamics are simpler because the momentum is built from the previous displacement rather than a gradient‑based extrapolation.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent of the Lyapunov function]\label{lem:descent}
Under Assumptions \ref{ass:smooth}–\ref{ass:params}, the sequence $\{\Phi_k\}$ defined in \eqref{T1} satisfies for all $k\ge0$
\[
\Phi_{k+1}\le \Phi_k-\frac{\alpha}{2}\,\|\nabla f(x_k)\|^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
We start from the smoothness inequality (A1') with $x=x_k$, $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\inner{\nabla f(x_k)}{x_{k+1}-x_k}
+\frac{L}{2}\|x_{k+1}-x_k\|^2 .\tag{S1}
\]

From the HB update,
\[
x_{k+1}-x_k = \beta (x_k-x_{k-1})-\alpha\nabla f(x_k). \tag{U1}
\]

Insert \eqref{U1} into \eqref{S1}:

\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)+\inner{\nabla f(x_k)}{\beta (x_k-x_{k-1})-\alpha\nabla f(x_k)}
+\frac{L}{2}\|\beta (x_k-x_{k-1})-\alpha\nabla f(x_k)\|^2\\
&= f(x_k) -\alpha\|\nabla f(x_k)\|^2
+ \beta\inner{\nabla f(x_k)}{x_k-x_{k-1}}
+\frac{L}{2}\bigl(\beta^2\|x_k-x_{k-1}\|^2
-2\alpha\beta\inner{\nabla f(x_k)}{x_k-x_{k-1}}
+\alpha^2\|\nabla f(x_k)\|^2\bigr). \tag{S2}
\end{aligned}
\]

Collect the terms that involve $\inner{\nabla f(x_k)}{x_k-x_{k-1}}$:

\[
\beta\inner{\nabla f(x_k)}{x_k-x_{k-1}}
-\frac{L\alpha\beta}{1}\inner{\nabla f(x_k)}{x_k-x_{k-1}}
= \beta\Bigl(1-\alpha L\Bigr)\inner{\nabla f(x_k)}{x_k-x_{k-1}} .\tag{S3}
\]

Collect the squared‑gradient terms:

\[
-\alpha\|\nabla f(x_k)\|^2+\frac{L\alpha^{2}}{2}\|\nabla f(x_k)\|^2
= -\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_k)\|^2 .\tag{S4}
\]

Collect the kinetic terms:

\[
\frac{L\beta^{2}}{2}\|x_k-x_{k-1}\|^2 .\tag{S5}
\]

Thus

\[
f(x_{k+1})\le f(x_k)
+\beta\bigl(1-\alpha L\bigr)\inner{\nabla f(x_k)}{x_k-x_{k-1}}
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_k)\|^2
+\frac{L\beta^{2}}{2}\|x_k-x_{k-1}\|^2 .\tag{S6}
\]

Now add $\dfrac{\beta}{2\alpha}\|x_{k+1}-x_k\|^2$ to both sides.  
Using \eqref{U1} again,

\[
\|x_{k+1}-x_k\|^2
= \beta^{2}\|x_k-x_{k-1}\|^2
-2\alpha\beta\inner{\nabla f(x_k)}{x_k-x_{k-1}}
+\alpha^{2}\|\nabla f(x_k)\|^2 .\tag{S7}
\]

Hence

\[
\begin{aligned}
\Phi_{k+1}
&= f(x_{k+1})+\frac{\beta}{2\alpha}\|x_{k+1}-x_k\|^2\\
&\le f(x_k)+\frac{\beta}{2\alpha}\|x_k-x_{k-1}\|^2 \\
&\quad +\beta\bigl(1-\alpha L\bigr)\inner{\nabla f(x_k)}{x_k-x_{k-1}}
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_k)\|^2
+\frac{L\beta^{2}}{2}\|x_k-x_{k-1}\|^2 \\
&\quad -\beta^{2}\frac{\beta}{2\alpha}\|x_k-x_{k-1}\|^2
+\beta^{2}\frac{\beta}{\alpha}\inner{\nabla f(x_k)}{x_k-x_{k-1}}
-\frac{\beta\alpha}{2}\|\nabla f(x_k)\|^2 .
\end{aligned}
\tag{S8}
\]

Group the kinetic terms.  The coefficient of $\|x_k-x_{k-1}\|^2$ becomes  

\[
\frac{\beta}{2\alpha}
+\frac{L\beta^{2}}{2}
-\frac{\beta^{3}}{2\alpha}
= \frac{\beta}{2\alpha}\bigl(1-\beta^{2}\bigr)+\frac{L\beta^{2}}{2}. \tag{S9}
\]

Group the cross terms $\inner{\nabla f(x_k)}{x_k-x_{k-1}}$:

\[
\beta\bigl(1-\alpha L\bigr)
+\frac{2\beta^{2}}{\alpha}
= \beta\Bigl(1-\alpha L+\frac{2\beta}{\alpha}\Bigr). \tag{S10}
\]

Group the squared‑gradient terms:

\[
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)-\frac{\beta\alpha}{2}
= -\alpha\Bigl(1-\frac{L\alpha}{2}+\frac{\beta}{2}\Bigr). \tag{S11}
\]

Now impose the parameter restriction \eqref{A3}.  
Because $0\le\beta\le\frac{1-\alpha L}{1+\alpha L}$, we have  

\[
\beta\Bigl(1-\alpha L+\frac{2\beta}{\alpha}\Bigr)\le 0, \qquad
\frac{\beta}{2\alpha}\bigl(1-\beta^{2}\bigr)+\frac{L\beta^{2}}{2}\le \frac{\beta}{2\alpha}.
\tag{S12}
\]

Using \eqref{S12} in \eqref{S8} yields the clean descent inequality  

\[
\Phi_{k+1}\le \Phi_k
-\alpha\Bigl(1-\frac{L\alpha}{2}+\frac{\beta}{2}\Bigr)\|\nabla f(x_k)\|^2 .
\tag{S13}
\]

Finally, by the stepsize bound $\alpha\le 1/L$ we have $1-\frac{L\alpha}{2}\ge\frac12$, and together with $\beta\le1$ we obtain  

\[
1-\frac{L\alpha}{2}+\frac{\beta}{2}\;\ge\;\frac12 .
\]

Thus

\[
\Phi_{k+1}\le \Phi_k-\frac{\alpha}{2}\|\nabla f(x_k)\|^2,
\]

which is exactly \eqref{L1}.  ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We now prove Theorem~\ref{thm:hb} using Lemma~\ref{lem:descent}.

\begin{enumerate}[label={[\arabic*]},leftmargin=*]
\item \textbf{Summation of the descent inequality.}  
From Lemma~\ref{lem:descent} we have for every $k\ge0$
\[
\Phi_{k+1}\le\Phi_k-\frac{\alpha}{2}\|\nabla f(x_k)\|^2 .
\tag{P1}
\]
Summing \eqref{P1} from $k=0$ to $K-1$ gives
\[
\Phi_K\le\Phi_0-\frac{\alpha}{2}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2 .
\tag{P2}
\]

\item \textbf{Bounding $\Phi_K$ from below.}  
By definition \eqref{T1},
\[
\Phi_K = f(x_K)+\frac{\beta}{2\alpha}\|x_K-x_{K-1}\|^2\ge f(x_K)\ge f^\star .
\tag{P3}
\]

\item \textbf{Rearranging to obtain a bound on the gradient sum.}  
Insert \eqref{P3} into \eqref{P2}:
\[
f^\star\le\Phi_0-\frac{\alpha}{2}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2 .
\tag{P4}
\]
Hence
\[
\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
\le\frac{2}{\alpha}\bigl(\Phi_0-f^\star\bigr).
\tag{P5}
\]

\item \textbf{Express $\Phi_0$ in terms of the initial data.}  
Since $x_{-1}=x_0$ (usual initialization), the kinetic term vanishes and $\Phi_0=f(x_0)$.  
Thus \eqref{P5} simplifies to
\[
\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
\le\frac{2}{\alpha}\bigl(f(x_0)-f^\star\bigr).
\tag{P6}
\]

\item \textbf{Deriving the average‑gradient bound.}  
Dividing \eqref{P6} by $K$ yields exactly the statement \eqref{T2} of Theorem~\ref{thm:hb}:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2
\le\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha K}.
\tag{P7}
\]

\item \textbf{Concluding the rate.}  
Since the right‑hand side decays as $O(1/K)$, we have
\[
\lim_{K\to\infty}\frac{1}{K}\sum_{k=0}^{K-1}\|\nabla f(x_k)\|^2 =0 .
\tag{P8}
\]
Thus the iterates admit at least one accumulation point that is a stationary point of $f$.
\end{enumerate}
The theorem is proved. ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
The bound \eqref{T2} can be rewritten as  

\[
\boxed{
\min_{0\le k\le K-1}\|\nabla f(x_k)\|^2
\;\le\;
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha K}
}
\tag{R1}
\]

so that after $K$ iterations the smallest gradient norm encountered is $O(1/K)$.  
All constants are explicit:
\begin{itemize}
\item $L$: smoothness constant from \ref{ass:smooth};
\item $\alpha$: stepsize, required to satisfy $0<\alpha\le1/L$;
\item $\beta$: momentum, required to satisfy $0\le\beta\le\frac{1-\alpha L}{1+\alpha L}$.
\end{itemize}
If one chooses the maximal admissible $\beta$, the bound on $\beta$ becomes
\[
\beta_{\max}= \frac{1-\alpha L}{1+\alpha L}\,,
\]
which yields the largest possible inertial effect while preserving the $O(1/K)$ guarantee.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Pure Gradient Descent ($\beta=0$).}  
The Lyapunov function reduces to $\Phi_k=f(x_k)$ and Lemma~\ref{lem:descent} becomes the classical smooth‑function descent lemma, recovering the standard $O(1/K)$ gradient‑norm bound for GD.

\item \textbf{Quadratic Objective $f(x)=\frac12 x^\top Q x - b^\top x$ with $Q\succeq0$ and $\lambda_{\max}(Q)=L$.}  
For this case the HB iteration is linear, and the admissible region \eqref{A3} coincides with the stability region of the corresponding second‑order difference equation.  The bound \eqref{T2} is tight for the worst‑case eigenvalue $L$.

\item \textbf{Stochastic Gradient Approximation.}  
If $g_k$ is an unbiased estimator of $\nabla f(x_k)$ with bounded variance, the same Lyapunov analysis yields an expected bound 
\[
\mathbb{E}\Bigl[\frac{1}{K}\sum_{k=0}^{K-1}\| \nabla f(x_k)\|^2\Bigr]
\le
\frac{2\bigl(f(x_0)-f^\star\bigr)}{\alpha K}
+\frac{C\sigma^2}{K},
\]
where $\sigma^2$ bounds the variance and $C$ depends on $\alpha,\beta,L$.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Conclusion}
We have presented a complete, step‑by‑step convergence analysis of the Heavy‑Ball momentum method for smooth non‑convex objectives.  The proof is entirely elementary, relying only on the $L$‑smoothness of $f$ and a carefully chosen Lyapunov function.  Under the standard stepsize–momentum restriction \eqref{A3}, the method attains the optimal $O(1/K)$ rate for the average squared gradient norm, matching gradient descent while often delivering faster practical convergence due to the inertial term.
