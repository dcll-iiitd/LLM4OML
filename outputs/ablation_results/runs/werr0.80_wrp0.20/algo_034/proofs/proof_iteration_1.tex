\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a possibly non‑convex, $L$‑smooth objective.  
At iteration $t$ we draw a random sample $\xi_{t}$ (e.g. a minibatch) and compute a stochastic gradient
\[
g^{\text{raw}}_{t}\;:=\;\nabla f(x_{t};\xi_{t}) .
\]
The **gradient clipping operator** with threshold $C>0$ is defined component‑wise by
\[
\clip_{C}(u)\;:=\;
\begin{cases}
u, & \norm{u}\le C,\\[4pt]
C\,\dfrac{u}{\norm{u}}, & \norm{u}>C .
\end{cases}
\]
The **clipped gradient** is
\[
g_{t}\;:=\;\clip_{C}\!\bigl(g^{\text{raw}}_{t}\bigr)
          \;=\;\frac{g^{\text{raw}}_{t}}{\max\!\bigl\{1,\;\norm{g^{\text{raw}}_{t}}/C\bigr\}} .
\]
The update rule of Clipped SGD is
\[
\boxed{\,x_{t+1}=x_{t}-\eta_{t}\,g_{t}\,},\qquad t=0,1,\dots,T-1,
\]
where $\{\eta_{t}\}_{t\ge0}$ is a (deterministic) stepsize schedule.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Stochastic Gradient Descent}
\begin{algorithmic}[1]
\STATE {\bf Input:} initial point $x_{0}\in\mathbb{R}^{d}$, stepsizes $\{\eta_{t}\}$, clipping threshold $C>0$.
\FOR{$t=0$ {\bf to} $T-1$}
    \STATE Sample mini‑batch $\xi_{t}$ and compute raw stochastic gradient 
          $g^{\text{raw}}_{t}\gets \nabla f(x_{t};\xi_{t})$.
    \STATE Compute clipping factor 
          $\displaystyle \alpha_{t}\gets\frac{1}{\max\{1,\,\norm{g^{\text{raw}}_{t}}/C\}}$.
    \STATE Form clipped gradient $g_{t}\gets \alpha_{t}\,g^{\text{raw}}_{t}$.
    \STATE Update $x_{t+1}\gets x_{t}-\eta_{t}\,g_{t}$.
\ENDFOR
\STATE {\bf Output:} $\{x_{t}\}_{t=0}^{T}$ (or the average $\bar x_{T}:=\frac1T\sum_{t=0}^{T-1}x_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Set $w_{0}=0$, stepsize $\eta=0.1$, clipping threshold $C=1$ and use the exact gradient (no stochasticity).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})$ & $g_{t}=\clip_{C}(\nabla f(w_{t}))$ & $f(w_{t})$\\ \hline
0 & $0$ & $-6$ & $-1$ (clipped) & $9$\\
1 & $0-0.1(-1)=0.1$ & $-5.8$ & $-1$ (clipped) & $8.41$\\
2 & $0.1-0.1(-1)=0.2$ & $-5.6$ & $-1$ (clipped) & $7.84$\\
3 & $0.2-0.1(-1)=0.3$ & $-5.4$ & $-1$ (clipped) & $7.29$
\end{tabular}
\end{center}
The clipping prevents the update magnitude from exceeding $0.1$, stabilising the iterates.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$, conditioned on $x_{t}$,
\[
\EE\bigl[g^{\text{raw}}_{t}\mid x_{t}\bigr]=\nabla f(x_{t}),\qquad
\EE\!\left[\norm{g^{\text{raw}}_{t}-\nabla f(x_{t})}^{2}\mid x_{t}\right]\le\sigma^{2},
\]
where $\sigma^{2}<\infty$.
\end{assumption}
\begin{assumption}[Bounded Gradient Norm]\label{ass:bounded}
There exists $G>0$ such that $\norm{\nabla f(x)}\le G$ for all $x$ visited by the algorithm.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize is constant $\eta_{t}\equiv\eta\le\frac{1}{L}$.
\end{assumption}
All constants $L,G,\sigma,C,\eta$ are known a priori.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Clipped SGD]\label{thm:convergence}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize} define the **effective contraction factor**
\[
\alpha:=\frac{C}{G}\in(0,1].
\]
Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\,\eta\,T}}_{\text{deterministic decay}}
\;+\;
\underbrace{2L\eta\sigma^{2}}_{\text{variance term}}
\;+\;
\underbrace{2L\alpha^{-1}C^{2}}_{\text{clipping bias}} .
\]
Choosing $\eta=\Theta\!\bigl(1/\sqrt{T}\bigr)$ yields the rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]=\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because $\norm{g_{t}}\le C$, the update step size is effectively bounded by $\eta C$, preventing exploding iterates even when raw stochastic gradients have heavy tails.
\item \textbf{Hyper‑parameter Sensitivity.} The contraction factor $\alpha=C/G$ quantifies the loss of information due to clipping. Larger $C$ (i.e. $\alpha\!\uparrow\!1$) reduces the bias term $2L\alpha^{-1}C^{2}$ and improves the constant in front of the deterministic decay.
\item \textbf{Comparison to vanilla SGD.} If $C\ge G$, then $\alpha=1$ and $g_{t}=g^{\text{raw}}_{t}$, so Theorem~\ref{thm:convergence} collapses to the standard SGD bound $\mathcal{O}(1/\sqrt{T})$ for non‑convex smooth objectives.
\item \textbf{Special Cases.}
\begin{itemize}
\item \emph{Deterministic setting} ($\sigma=0$): the variance term disappears and the bound reduces to $ \mathcal{O}\!\bigl(1/T\bigr)$ after choosing $\eta\propto 1/T$.
\item \emph{Large clipping} ($C\to G$): $\alpha\to1$ and the clipping‑bias term behaves like $2LC^{2}$, which is negligible compared with the $1/(\eta T)$ term for the usual choice $\eta\asymp 1/\sqrt{T}$.
\end{itemize}
\end{enumerate}

\section*{Discussion}
Clipping introduces a deterministic bias because $g_{t}\neq\nabla f(x_{t})$ when the raw gradient norm exceeds $C$. The analysis above isolates this bias into the term $2L\alpha^{-1}C^{2}$, which is independent of $T$ and therefore does not affect the asymptotic $\mathcal{O}(1/\sqrt{T})$ rate. The price paid is a (potentially) larger constant factor, controlled by the ratio $C/G$. In practice, $C$ is chosen to balance the trade‑off between stability (small $C$) and bias (large $C$).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x$ and any vector $u$,
\[
f(x-u)\le f(x)-\langle\nabla f(x),u\rangle+\frac{L}{2}\norm{u}^{2}.
\]
\end{lemma}
\begin{proof}
Apply Assumption~\ref{ass:smooth} with $y=x-u$.
\end{proof}
\begin{lemma}[Bias decomposition]\label{lem:bias}
Define the **bias vector**
\[
b_{t}:=\nabla f(x_{t})-g_{t}.
\]
Then for every $t$,
\[
\langle\nabla f(x_{t}),g_{t}\rangle
= \norm{\nabla f(x_{t})}^{2}-\langle\nabla f(x_{t}),b_{t}\rangle.
\]
Moreover,
\[
\langle\nabla f(x_{t}),b_{t}\rangle\le\frac{1}{2}\norm{\nabla f(x_{t})}^{2}
+\frac{1}{2}\norm{b_{t}}^{2}
\qquad\text{(Cauchy--Schwarz and }2ab\le a^{2}+b^{2}\text{)}.
\]
\end{lemma}
\begin{proof}
The identity follows from the definition of $b_{t}$. The inequality is a standard application of $2ab\le a^{2}+b^{2}$ with $a=\norm{\nabla f(x_{t})}$ and $b=\norm{b_{t}}$.
\end{proof}
\begin{lemma}[Bound on the bias norm]\label{lem:biasnorm}
Because of the clipping rule,
\[
\norm{b_{t}}\le\bigl(\norm{\nabla f(x_{t})}-C\bigr)_{+}\le G-C,
\]
where $(\cdot)_{+}$ denotes $\max\{\,\cdot\,,0\}$.
\end{lemma}
\begin{proof}
If $\norm{\nabla f(x_{t})}\le C$ then $g_{t}=\nabla f(x_{t})$ and $b_{t}=0$.  
If $\norm{\nabla f(x_{t})}>C$, then $g_{t}=C\,\nabla f(x_{t})/\norm{\nabla f(x_{t})}$, hence
\[
b_{t}= \nabla f(x_{t})-C\frac{\nabla f(x_{t})}{\norm{\nabla f(x_{t})}}
      =\Bigl(1-\frac{C}{\norm{\nabla f(x_{t})}}\Bigr)\nabla f(x_{t}),
\]
so $\norm{b_{t}}=\norm{\nabla f(x_{t})}-C$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convergence}. All expectations are taken over the randomness of the stochastic gradients $\{g^{\text{raw}}_{t}\}$ conditioned on the history up to time $t$.

\bigskip
\noindent\textbf{[Step 1] Smoothness bound.}  
From Lemma~\ref{lem:smooth} with $u=\eta g_{t}$,
\[
f(x_{t+1})\le f(x_{t})-\eta\langle\nabla f(x_{t}),g_{t}\rangle
               +\frac{L\eta^{2}}{2}\norm{g_{t}}^{2}.
\tag{1}
\]

\noindent\textbf{[Step 2] Insert bias decomposition.}  
Using Lemma~\ref{lem:bias},
\[
\langle\nabla f(x_{t}),g_{t}\rangle
= \norm{\nabla f(x_{t})}^{2}
  -\langle\nabla f(x_{t}),b_{t}\rangle
\ge \norm{\nabla f(x_{t})}^{2}
   -\frac{1}{2}\norm{\nabla f(x_{t})}^{2}
   -\frac{1}{2}\norm{b_{t}}^{2}
= \frac{1}{2}\norm{\nabla f(x_{t})}^{2}
  -\frac{1}{2}\norm{b_{t}}^{2}.
\tag{2}
\]

\noindent\textbf{[Step 3] Upper‑bound $\norm{g_{t}}^{2}$.}  
Because of clipping, $\norm{g_{t}}\le C$. Hence
\[
\norm{g_{t}}^{2}\le C^{2}.
\tag{3}
\]

\noindent\textbf{[Step 4] Combine (1)–(3).}  
Plugging (2) and (3) into (1) gives
\[
f(x_{t+1})\le f(x_{t})
          -\frac{\eta}{2}\norm{\nabla f(x_{t})}^{2}
          +\frac{\eta}{2}\norm{b_{t}}^{2}
          +\frac{L\eta^{2}}{2}C^{2}.
\tag{4}
\]

\noindent\textbf{[Step 5] Take conditional expectation.}  
Condition on $x_{t}$ and use Assumption~\ref{ass:unbiased} (which guarantees that $b_{t}$ depends only on $g^{\text{raw}}_{t}$ and hence is measurable w.r.t.\ the same sigma‑algebra). The only random term left in (4) is $b_{t}$ via $g^{\text{raw}}_{t}$, but Lemma~\ref{lem:biasnorm} provides a deterministic bound. Thus
\[
\EE\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
   -\frac{\eta}{2}\norm{\nabla f(x_{t})}^{2}
   +\frac{\eta}{2}\underbrace{\EE\!\bigl[\norm{b_{t}}^{2}\mid x_{t}\bigr]}_{\le (G-C)^{2}}
   +\frac{L\eta^{2}}{2}C^{2}.
\tag{5}
\]

\noindent\textbf{[Step 6] Relate $\norm{b_{t}}^{2}$ to $\norm{\nabla f(x_{t})}^{2}$.}  
From Lemma~\ref{lem:biasnorm},
\[
\norm{b_{t}}^{2}\le\bigl(\norm{\nabla f(x_{t})}-C\bigr)_{+}^{2}
               \le \norm{\nabla f(x_{t})}^{2}\,\underbrace{\frac{(G-C)^{2}}{G^{2}}}_{=:1-\alpha}.
\]
Define $\alpha:=C/G\in(0,1]$. Then $1-\alpha=(G-C)/G$ and
\[
\norm{b_{t}}^{2}\le (1-\alpha)^{2}\norm{\nabla f(x_{t})}^{2}\le (1-\alpha)\norm{\nabla f(x_{t})}^{2},
\tag{6}
\]
where the last inequality uses $1-\alpha\le1$.

\noindent\textbf{[Step 7] Substitute (6) into (5).}
\[
\EE\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})
   -\frac{\eta}{2}\norm{\nabla f(x_{t})}^{2}
   +\frac{\eta}{2}(1-\alpha)\norm{\nabla f(x_{t})}^{2}
   +\frac{L\eta^{2}}{2}C^{2}.
\]
Simplify the coefficients of $\norm{\nabla f(x_{t})}^{2}$:
\[
-\frac{\eta}{2}+\frac{\eta}{2}(1-\alpha)
= -\frac{\eta\alpha}{2}.
\]
Hence
\[
\EE\!\bigl[f(x_{t+1})\mid x_{t}\bigr]
\le f(x_{t})-\frac{\eta\alpha}{2}\norm{\nabla f(x_{t})}^{2}
     +\frac{L\eta^{2}}{2}C^{2}.
\tag{7}
\]

\noindent\textbf{[Step 8] Take total expectation and telescope.}  
Iterating (7) from $t=0$ to $T-1$ and using the law of total expectation yields
\[
\EE\!\bigl[f(x_{T})\bigr]
\le f(x_{0})
   -\frac{\eta\alpha}{2}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
   +\frac{L\eta^{2}}{2}C^{2}T.
\tag{8}
\]

\noindent\textbf{[Step 9] Rearrange to isolate the gradient sum.}
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-\EE[f(x_{T})]\bigr)}{\alpha\eta T}
     +L\eta C^{2}.
\tag{9}
\]

\noindent\textbf{[Step 10] Lower bound $\EE[f(x_{T})]$ by $f^{*}$.}  
Since $f^{*}:=\inf_{x}f(x)\le f(x)$ for any $x$, we have $\EE[f(x_{T})]\ge f^{*}$. Inserting this bound into (9) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\eta T}
     +L\eta C^{2}.
\tag{10}
\]

\noindent\textbf{[Step 11] Add the stochastic variance term.}  
So far we have used only the deterministic part of the stochastic gradient. To incorporate the variance $\sigma^{2}$, we refine the bound on $\norm{g_{t}}^{2}$:
\[
\EE\!\bigl[\norm{g_{t}}^{2}\mid x_{t}\bigr]
= \EE\!\bigl[\norm{\clip_{C}(g^{\text{raw}}_{t})}^{2}\mid x_{t}\bigr]
\le \EE\!\bigl[\norm{g^{\text{raw}}_{t}}^{2}\mid x_{t}\bigr]
\le \norm{\nabla f(x_{t})}^{2}+\sigma^{2},
\]
where the first inequality follows from the fact that clipping can only reduce the norm. Re‑doing Step 4 with this refined bound yields an extra term $ \frac{L\eta^{2}}{2}\sigma^{2}$ on the right‑hand side of (7). Propagating it through Steps 8–10 adds $L\eta\sigma^{2}$ to the final bound. Consequently,
\[
\boxed{
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\eta T}
     +L\eta\sigma^{2}
     +L\eta C^{2}.
}
\tag{11}
\]

\noindent\textbf{[Step 12] Simplify constants.}  
Factor $L\eta$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\eta T}
     +L\eta\bigl(\sigma^{2}+C^{2}\bigr).
\tag{12}
\]

\noindent\textbf{[Step 13] Choose stepsize.}  
Set $\eta = \sqrt{\dfrac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha L\bigl(\sigma^{2}+C^{2}\bigr)T}}$, which respects $\eta\le 1/L$ for all sufficiently large $T$ because $C\le G$ and $G$ is finite. Substituting this $\eta$ into (12) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le 2\sqrt{\frac{L\bigl(\sigma^{2}+C^{2}\bigr)\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\,T}}
      =\mathcal{O}\!\Bigl(\frac{1}{\sqrt{T}}\Bigr).
\]
This completes the proof of Theorem~\ref{thm:convergence}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
Collecting the constants appearing in (12):
\[
\begin{aligned}
\text{Deterministic decay term} &:\; \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\eta T},\\[2pt]
\text{Variance term}            &:\; L\eta\sigma^{2},\\[2pt]
\text{Clipping‑bias term}       &:\; L\eta C^{2}.
\end{aligned}
\]
If one wishes to separate the bias caused solely by clipping (i.e. set $\sigma=0$) the bound simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\alpha\eta T}+L\eta C^{2},
\]
which attains the optimal $\mathcal{O}(1/\sqrt{T})$ rate with $\eta\asymp 1/\sqrt{T}$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{No clipping} ($C\ge G$). Then $\alpha=1$, $b_{t}=0$, and (12) reduces to the classic non‑convex SGD bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\norm{\nabla f(x_{t})}^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f^{*}\bigr)}{\eta T}+L\eta\sigma^{2}.
\]
\item \textbf{Deterministic gradients} ($\sigma=0$). The variance term disappears and the rate improves to $\mathcal{O}(1/T)$ when $\eta\propto 1/T$.
\item \textbf{Very aggressive clipping} ($C\ll G$). Then $\alpha\approx C/G$ is small; the deterministic decay term is amplified by $1/\alpha$, reflecting the loss of information due to heavy clipping. Nevertheless the $\mathcal{O}(1/\sqrt{T})$ asymptotic rate is preserved.
\end{enumerate}
