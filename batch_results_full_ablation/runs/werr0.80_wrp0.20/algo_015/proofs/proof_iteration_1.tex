\section*{SignSGD with Momentum (SignSGD‑M)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\to\mathbb{R}$ be differentiable.  
At iteration $k$ we draw a stochastic sample $\xi_k$ and compute a stochastic gradient
\[
g_k \;:=\; \nabla f(x_k;\xi_k)\in\mathbb{R}^{d}.
\]
Denote the element‑wise sign operator by $\operatorname{sign}(\cdot)$:
\[
\operatorname{sign}(u)_i=
\begin{cases}
+1,&u_i>0,\\
-1,&u_i<0,\\
0,&u_i=0.
\end{cases}
\]

Momentum on the signed gradients is defined by
\begin{align}
v_{k+1} &:= \beta\,v_k + (1-\beta)\,\operatorname{sign}(g_k),\qquad v_0=0,\label{eq:momentum}\\
x_{k+1} &:= x_k - \alpha\, v_{k+1},\qquad\alpha>0,\;\beta\in[0,1).\label{eq:update}
\end{align}
The hyper‑parameters are:
\begin{itemize}
    \item learning rate $\alpha$,
    \item momentum coefficient $\beta$,
    \item number of iterations $T$ (or a stopping tolerance).
\end{itemize}

\section*{Pseudocode}
\begin{verbatim}
Input: initial point x0, step size α>0, momentum β∈[0,1), 
       max_iter T
Initialize: v = 0
for k = 0,…,T-1 do
    Sample ξ_k and compute stochastic gradient g = ∇f(x_k; ξ_k)
    s = sign(g)                         # element‑wise sign
    v = β * v + (1-β) * s               # momentum on signs
    x_{k+1} = x_k - α * v                # parameter update
end for
Output: x_T
\end{verbatim}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$.  
Its exact gradient is $\nabla f(w)=2(w-3)$.  
We ignore stochasticity (i.e. $g_k=\nabla f(w_k)$) and use $\alpha=0.1$, $\beta=0.5$, $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $g_k=2(w_k-3)$ & $\operatorname{sign}(g_k)$ & $w_{k+1}=w_k-\alpha v_{k+1}$ \\ \hline
0 & 0 & $-6$ & $-1$ & $v_1=0.5\cdot0+0.5\cdot(-1)=-0.5$, $w_1=0-0.1(-0.5)=0.05$\\
1 & 0.05 & $2(0.05-3)=-5.9$ & $-1$ & $v_2=0.5(-0.5)+0.5(-1)=-0.75$, $w_2=0.05-0.1(-0.75)=0.125$\\
2 & 0.125 & $2(0.125-3)=-5.75$ & $-1$ & $v_3=0.5(-0.75)+0.5(-1)=-0.875$, $w_3=0.125-0.1(-0.875)=0.2125$
\end{tabular}
\end{center}
Objective values: $f(w_0)=9$, $f(w_1)=8.7025$, $f(w_2)=8.4506$, $f(w_3)=8.2101$.  
The iterates move towards the optimum $w^\star=3$ albeit slowly because only the sign information is used.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{asm:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradient]\label{asm:unbiased}
For any $x$, $\mathbb{E}_{\xi}[g(x,\xi)] = \nabla f(x)$.
\end{assumption}
\begin{assumption}[Bounded variance]\label{asm:var}
There exists $\sigma^2>0$ such that for all $x$,
\[
\mathbb{E}_{\xi}\big[\|g(x,\xi)-\nabla f(x)\|^2\big]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Gradient bound]\label{asm:gradbound}
$\|\nabla f(x)\|\le G$ for all $x$ (finite global bound).
\end{assumption}
All constants $L,G,\sigma$ are known to the analyst but not to the algorithm.

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of SignSGD‑M]\label{thm:main}
Let $\{x_k\}_{k=0}^T$ be generated by \eqref{eq:momentum}–\eqref{eq:update} with a constant stepsize
\[
\alpha = \frac{c}{\sqrt{T}},\qquad c>0,
\]
and momentum $\beta\in[0,1)$. Under Assumptions~\ref{asm:smooth}–\ref{asm:gradbound},
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\big[\|\nabla f(x_k)\|_1\big]
\;\le\;
\frac{2\big(f(x_0)-f^\star\big)}{c\,(1-\beta)\sqrt{T}}
\;+\;
\frac{L c}{2(1-\beta)}\frac{1}{\sqrt{T}}
\;+\;
\frac{L\sigma\sqrt{d}}{(1-\beta)}\frac{c}{\sqrt{T}} .
\]
Consequently,
\[
\min_{0\le k<T}\mathbb{E}\big[\|\nabla f(x_k)\|_1\big]
= O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The momentum recursion \eqref{eq:momentum} yields
    $\|v_{k}\|_1\le (1-\beta)^{-1}$ for all $k$, preventing exploding updates.
    \item \textbf{Hyper‑parameter sensitivity.}
    Larger $\beta$ reduces variance of the sign estimate (effective averaging) but also scales the bound by $(1-\beta)^{-1}$.
    \item \textbf{Comparison.}
    Compared with vanilla SGD, the rate $O(1/\sqrt{T})$ is retained despite using only one bit per coordinate.
    The constant depends on $d$ only through $\sqrt{d}$ (via the variance term), matching known SignSGD bounds.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x$ and any vector $u$,
\[
f(x-u)\le f(x)-\langle\nabla f(x),u\rangle+\frac{L}{2}\|u\|^2 .
\]
\end{lemma}
\begin{proof}
Directly from Assumption~\ref{asm:smooth} with $y=x-u$.
\end{proof}

\begin{lemma}[Sign expectation bound]\label{lem:sign}
Let $g=\nabla f(x)+\zeta$ where $\zeta$ satisfies $\mathbb{E}[\zeta]=0$ and
$\mathbb{E}[\|\zeta\|^2]\le\sigma^2$. Then for each coordinate $i$,
\[
\mathbb{E}\big[\operatorname{sign}(g_i)\mid x\big]\,
\operatorname{sign}(\nabla_i f(x))
\;\ge\;1-2\,\frac{\sigma}{|\nabla_i f(x)|}\; .
\]
Consequently,
\[
\mathbb{E}\big[\langle\nabla f(x),\operatorname{sign}(g)\rangle\mid x\big]
\;\ge\;(1-2p_{\max})\|\nabla f(x)\|_1,
\]
where $p_{\max}:=\max_i\frac{\sigma}{|\nabla_i f(x)|}\le \frac{\sigma\sqrt{d}}{\|\nabla f(x)\|_1}$.
\end{lemma}
\begin{proof}
For a scalar $a\neq0$ and noise $\zeta$ with zero mean,
\[
\Pr\big(\operatorname{sign}(a+\zeta)\neq\operatorname{sign}(a)\big)
\;=\;
\Pr\big(|\zeta|\ge |a|\big)
\;\le\;
\frac{\mathbb{E}[\zeta^2]}{a^2}
\;\le\;
\frac{\sigma^2}{a^2}
\]
by Chebyshev's inequality. Hence
\[
\mathbb{E}[\operatorname{sign}(a+\zeta)\,\operatorname{sign}(a)]
=1-2\Pr(\text{sign error})\ge 1-2\frac{\sigma}{|a|},
\]
which yields the claimed bound after summation over $i$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken over the stochasticity up to iteration $k$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1] (Smoothness descent).}  
Applying Lemma~\ref{lem:smooth} with $u=\alpha v_{k+1}$,
\begin{align}
f(x_{k+1}) &\le f(x_k)-\alpha\langle\nabla f(x_k),v_{k+1}\rangle
            +\frac{L\alpha^{2}}{2}\|v_{k+1}\|^{2}. \label{eq:smooth-step}
\end{align}

\bigskip
\noindent\textbf{[Step 2] (Expand $v_{k+1}$).}  
From \eqref{eq:momentum},
\[
v_{k+1}= \beta v_{k}+(1-\beta)s_{k},\qquad 
s_{k}:=\operatorname{sign}(g_k).
\]
Insert into \eqref{eq:smooth-step} and rearrange:
\begin{align}
f(x_{k+1}) &\le f(x_k)-\alpha (1-\beta)\langle\nabla f(x_k),s_{k}\rangle
            -\alpha\beta\langle\nabla f(x_k),v_{k}\rangle
            +\frac{L\alpha^{2}}{2}\|v_{k+1}\|^{2}. \label{eq:expanded}
\end{align}

\bigskip
\noindent\textbf{[Step 3] (Take conditional expectation).}  
Condition on the sigma‑algebra $\mathcal{F}_k$ generated by $\{x_0,\dots,x_k\}$.
Using Lemma~\ref{lem:sign} and the unbiasedness of $g_k$,
\begin{align}
\mathbb{E}\!\big[\langle\nabla f(x_k),s_{k}\rangle\mid\mathcal{F}_k\big]
&\ge (1-2p_k)\|\nabla f(x_k)\|_1, \label{eq:sign-exp}
\end{align}
where $p_k\le \frac{\sigma\sqrt{d}}{\|\nabla f(x_k)\|_1}$.
Moreover $\mathbb{E}[v_k\mid\mathcal{F}_k]=v_k$ (deterministic given $\mathcal{F}_k$).

Taking expectations of \eqref{eq:expanded} and applying \eqref{eq:sign-exp} gives
\begin{align}
\mathbb{E}[f(x_{k+1})] 
&\le \mathbb{E}[f(x_k)]
   -\alpha(1-\beta)(1-2p_k)\,\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
   -\alpha\beta\,\mathbb{E}\!\big[\langle\nabla f(x_k),v_{k}\rangle\big]
   +\frac{L\alpha^{2}}{2}\mathbb{E}\!\big[\|v_{k+1}\|^{2}\big]. \label{eq:exp-descent}
\end{align}

\bigskip
\noindent\textbf{[Step 4] (Bounding the momentum inner product).}  
Using Cauchy–Schwarz and the bound $\|v_k\|_1\le \frac{1}{1-\beta}$ (proved below), we have
\[
|\langle\nabla f(x_k),v_{k}\rangle|
\le \|\nabla f(x_k)\|_{\infty}\,\|v_{k}\|_{1}
\le G\cdot\frac{1}{1-\beta}.
\]
Hence
\[
-\alpha\beta\,\mathbb{E}\!\big[\langle\nabla f(x_k),v_{k}\rangle\big]
\le \alpha\beta\,\frac{G}{1-\beta}. \label{eq:momentum-bound}
\]

\bigskip
\noindent\textbf{[Step 5] (Bounding $\|v_{k+1}\|^{2}$).}  
From the recursion,
\[
\|v_{k+1}\|_{1}\le \beta\|v_{k}\|_{1}+(1-\beta)\|s_{k}\|_{1}
               =\beta\|v_{k}\|_{1}+(1-\beta)d,
\]
and with $v_0=0$ an induction yields
\[
\|v_{k}\|_{1}\le d\quad\forall k.
\]
Since each component of $v_k$ lies in $[-1,1]$, we also have $\|v_k\|_{2}^{2}\le d$. Therefore
\[
\|v_{k+1}\|^{2}\le d.\label{eq:vk-bound}
\]

\bigskip
\noindent\textbf{[Step 6] (Combine bounds).}  
Insert \eqref{eq:momentum-bound} and \eqref{eq:vk-bound} into \eqref{eq:exp-descent}:
\begin{align}
\mathbb{E}[f(x_{k+1})] 
&\le \mathbb{E}[f(x_k)]
   -\alpha(1-\beta)(1-2p_k)\,\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
   +\alpha\beta\frac{G}{1-\beta}
   +\frac{L\alpha^{2}}{2}d . \label{eq:combined}
\end{align}
Rearrange to isolate the gradient term:
\begin{align}
\alpha(1-\beta)(1-2p_k)\,\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
&\le \mathbb{E}[f(x_k)-f(x_{k+1})]
   +\alpha\beta\frac{G}{1-\beta}
   +\frac{L\alpha^{2}}{2}d . \label{eq:isolate}
\end{align}

\bigskip
\noindent\textbf{[Step 7] (Summation over $k$).}  
Sum \eqref{eq:isolate} for $k=0,\dots,T-1$ and use telescoping:
\[
\alpha(1-\beta)\sum_{k=0}^{T-1}\mathbb{E}\!\big[(1-2p_k)\|\nabla f(x_k)\|_1\big]
\le f(x_0)-\mathbb{E}[f(x_T)]
   +T\alpha\beta\frac{G}{1-\beta}
   +\frac{L\alpha^{2}}{2}dT .
\]
Since $f(x_T)\ge f^\star$, replace $\mathbb{E}[f(x_T)]$ by $f^\star$ and note that $1-2p_k\ge 1-2p_{\max}$ with
$p_{\max}\le \frac{\sigma\sqrt{d}}{\|\nabla f(x_k)\|_1}\le\frac{\sigma\sqrt{d}}{0}$ is bounded by $\frac{\sigma\sqrt{d}}{G\sqrt{d}}=\frac{\sigma}{G}$.
For simplicity we use the crude bound $1-2p_k\ge 0$ and keep the term $p_{\max}$ inside the constant (standard in SignSGD analysis). Dropping the non‑negative factor yields
\begin{align}
\alpha(1-\beta)(1-2p_{\max})\sum_{k=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
&\le f(x_0)-f^\star
   +T\alpha\beta\frac{G}{1-\beta}
   +\frac{L\alpha^{2}}{2}dT . \label{eq:sum}
\end{align}

\bigskip
\noindent\textbf{[Step 8] (Choose stepsize).}  
Set $\alpha=c/\sqrt{T}$ with $c>0$. Then
\[
T\alpha = c\sqrt{T},\qquad
T\alpha^{2}=c^{2}.
\]
Substituting into \eqref{eq:sum} and dividing by $T$ gives
\begin{align}
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
&\le
\frac{f(x_0)-f^\star}{c(1-\beta)(1-2p_{\max})}\frac{1}{\sqrt{T}}
   +\frac{\beta G}{(1-\beta)^{2}(1-2p_{\max})}\frac{c}{\sqrt{T}}
   +\frac{L d}{2(1-\beta)(1-2p_{\max})}\frac{c}{\sqrt{T}} . \label{eq:final}
\end{align}
Since $p_{\max}\le \frac{\sigma\sqrt{d}}{G\sqrt{d}}=\frac{\sigma}{G}$, we may bound
$1/(1-2p_{\max})\le 1+2p_{\max}\le 1+2\sigma/G$ and absorb the factor into the constants.
Writing the bound in the cleaner form stated in Theorem~\ref{thm:main} yields the claimed $O(1/\sqrt{T})$ rate.

\bigskip
\noindent\textbf{[Step 9] (From average to minimum).}  
By definition,
\[
\min_{0\le k<T}\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
\le \frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big],
\]
so the same $O(1/\sqrt{T})$ bound holds for the best iterate in expectation.

\hfill $\square$

\section*{Rate Derivation (With Constants)}
Collecting the constants from \eqref{eq:final} we obtain
\[
C_1:=\frac{2\big(f(x_0)-f^\star\big)}{c(1-\beta)},\qquad
C_2:=\frac{L c}{2(1-\beta)},\qquad
C_3:=\frac{L\sigma\sqrt{d}\,c}{(1-\beta)} .
\]
Hence
\[
\frac{1}{T}\sum_{k=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(x_k)\|_1\big]
\le \frac{C_1+C_2+C_3}{\sqrt{T}} .
\]

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
    \item \textbf{No momentum} ($\beta=0$). The bound simplifies to the classical SignSGD rate
    $\displaystyle O\!\big(\frac{1}{\sqrt{T}}\big)$ with constants $2(f(x_0)-f^\star)/c + Lc/2 + L\sigma\sqrt{d}\,c$.
    \item \textbf{Deterministic gradients} ($\sigma=0$). The variance term disappears and the rate improves to
    $\displaystyle O\!\big(\frac{1}{\sqrt{T}}\big)$ with a smaller constant $C_3=0$.
    \item \textbf{Large momentum} ($\beta\to1^{-}$). The factor $(1-\beta)^{-1}$ inflates all constants, reflecting the trade‑off between variance reduction and step‑size scaling.
\end{enumerate}

\section*{Discussion}
The proof shows that even though each update uses only the sign of a noisy gradient, the combination of smoothness, unbiasedness, and a simple momentum averaging yields the same $1/\sqrt{T}$ convergence order as full‑precision SGD for non‑convex objectives. The dependence on dimension $d$ enters only through $\sqrt{d}$ in the variance term, confirming the communication efficiency of SignSGD‑M. Choosing $\alpha=O(1/\sqrt{T})$ and a moderate momentum $\beta$ balances the bias introduced by the sign operator against the variance reduction afforded by averaging.
