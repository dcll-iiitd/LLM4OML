\section*{Algorithm Name}
\addcontentsline{toc}{section}{Algorithm Name}
\begin{center}
\textbf{Federated Proximal (FedProx) with Local SGD}
\end{center}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
\addcontentsline{toc}{section}{Mathematical Formulation}
Consider $K$ participating devices (clients).  
Each client $k\in\{1,\dots,K\}$ possesses a local loss
\[
f_k(x)\;=\;\E_{\xi\sim\mathcal{D}_k}\bigl[\,\ell(x;\xi)\,\bigr],\qquad 
x\in\R^{d},
\]
and the global objective is the average
\[
f(x)\;=\;\frac1K\sum_{k=1}^{K}f_k(x).
\]

At communication round $t=0,1,2,\dots$ the server holds the global model $x_t$.
Each client receives $x_t$ and performs $E$ \emph{local} stochastic gradient steps on the
\emph{proximal} sub‑problem
\[
\min_{x\in\R^{d}} \; f_k(x)+\frac{\mu}{2}\|x-x_t\|^{2},
\qquad \mu\ge 0.
\]

Denote by $x_{t,k}^{e}$ the local iterate of client $k$ after $e$ local steps
($e=0$ corresponds to $x_{t,k}^{0}=x_t$).  
For $e=0,\dots,E-1$ the client draws an i.i.d.\ sample $\xi_{t,k}^{e}$ and computes the
\emph{stochastic proximal gradient}
\[
g_{t,k}^{e}\;=\;\nabla\ell\bigl(x_{t,k}^{e};\xi_{t,k}^{e}\bigr)
               \;+\;\mu\bigl(x_{t,k}^{e}-x_t\bigr).
\]
The local update is
\begin{equation}
\label{eq:local-update}
x_{t,k}^{e+1}\;=\;x_{t,k}^{e}\;-\;\eta\,g_{t,k}^{e},
\end{equation}
where $\eta>0$ is the (common) stepsize.

After $E$ steps each client sends $x_{t,k}^{E}$ to the server, which aggregates by simple averaging:
\begin{equation}
\label{eq:global-avg}
x_{t+1}\;=\;\frac1K\sum_{k=1}^{K}x_{t,k}^{E}.
\end{equation}
The process repeats until a termination condition (e.g., maximum rounds $T$) is met.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\addcontentsline{toc}{section}{Pseudocode}
\begin{algorithm}[H]
\caption{FedProx with Local SGD}
\label{alg:fedprox}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, proximal weight $\mu\ge0$, local epochs $E$, total rounds $T$
\STATE Initialize global model $x_0\in\R^{d}$
\FOR{round $t=0$ \TO $T-1$}
    \FOR{each selected client $k\in\{1,\dots,K\}$ \textbf{in parallel}}
        \STATE $x_{t,k}^{0}\gets x_t$
        \FOR{$e=0$ \TO $E-1$}
            \STATE Sample $\xi_{t,k}^{e}\sim\mathcal{D}_k$
            \STATE $g_{t,k}^{e}\gets\nabla\ell\!\bigl(x_{t,k}^{e};\xi_{t,k}^{e}\bigr)+\mu\bigl(x_{t,k}^{e}-x_t\bigr)$
            \STATE $x_{t,k}^{e+1}\gets x_{t,k}^{e}-\eta\,g_{t,k}^{e}$   \hfill\COMMENT{Eq.~\eqref{eq:local-update}}
        \ENDFOR
        \STATE Send $x_{t,k}^{E}$ to the server
    \ENDFOR
    \STATE $x_{t+1}\gets\frac1K\sum_{k=1}^{K}x_{t,k}^{E}$   \hfill\COMMENT{Eq.~\eqref{eq:global-avg}}
\ENDFOR
\STATE \textbf{Output:} $\displaystyle \bar{x}_T\;=\;\frac1T\sum_{t=0}^{T-1}x_t$ (or $x_T$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
\addcontentsline{toc}{section}{Dry Run Example}
We illustrate the algorithm on a single client ($K=1$), deterministic gradients,
and the simple quadratic loss
\[
f(w)= (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]

\textbf{Parameters:}
\[
\eta=0.1,\qquad \mu=0.5,\qquad E=1,\qquad w_0=0.
\]

Because the data are deterministic we have $g_t^{0}= \nabla f(w_t)+\mu(w_t-w_t)=2(w_t-3)$.
The local update (only one step) reduces to
\[
w_{t+1}= w_t-\eta\,2(w_t-3).
\]

\begin{center}
\begin{tabular}{c|c|c|c}
\toprule
Round $t$ & $w_t$ & Gradient $2(w_t-3)$ & $w_{t+1}=w_t-0.1\cdot2(w_t-3)$ \\
\midrule
0 & $0.0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\bottomrule
\end{tabular}
\end{center}

The objective values after each round are
\[
f(w_0)=9,\; f(w_1)=5.76,\; f(w_2)=4.176,\; f(w_3)=3.236.
\]
The example shows how the proximal term ($\mu$) does not affect the update when
the global model equals the local one (i.e., $w_t$ is used as the reference point).

\section*{Assumptions}
\addcontentsline{toc}{section}{Assumptions}
\begin{assumption}[Convexity and Smoothness]
\label{ass:convex-smooth}
Each local objective $f_k:\R^{d}\to\R$ is convex and $L$‑smooth, i.e.
\[
f_k(y)\;\le\;f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},
\qquad\forall x,y\in\R^{d}.
\]
Consequently the global function $f$ is also convex and $L$‑smooth.
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradients and Bounded Variance]
\label{ass:unbiased}
For every client $k$ and every local iterate $x$, the stochastic gradient satisfies
\[
\E_{\xi\sim\mathcal{D}_k}\bigl[g_k(x,\xi)\bigr]=\nabla f_k(x),
\]
and there exists $\sigma^{2}>0$ such that
\[
\E_{\xi\sim\mathcal{D}_k}\!\bigl[\|g_k(x,\xi)-\nabla f_k(x)\|^{2}\bigr]\;\le\;\sigma^{2},
\qquad\forall x\in\R^{d},\;k\in[K].
\]
\end{assumption}

\begin{assumption}[Bounded Gradients]
\label{ass:bounded-grad}
There exists $G>0$ with $\|\nabla f_k(x)\|\le G$ for all $k$ and $x$.
\end{assumption}

\begin{assumption}[Stepsize]
\label{ass:stepsize}
The learning rate $\eta$ satisfies
\[
0<\eta\le\frac{1}{L+\mu}.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\addcontentsline{toc}{section}{Main Convergence Theorem}
\begin{theorem}[Convergence of FedProx for Convex $L$‑smooth Objectives]
\label{thm:convex}
Under Assumptions~\ref{ass:convex-smooth}–\ref{ass:stepsize},
let $x^{\ast}\in\arg\min_{x}f(x)$.
Define the averaged iterate $\bar{x}_T=\frac1T\sum_{t=0}^{T-1}x_t$.
If a constant stepsize $\eta\le\frac{1}{L+\mu}$ is used, then for any $T\ge1$
\[
\boxed{
\E\!\bigl[f(\bar{x}_T)-f(x^{\ast})\bigr]
\;\le\;
\frac{\|x_0-x^{\ast}\|^{2}}{2\eta T}
\;+\;
\frac{\eta(L+\mu)\sigma^{2}}{K}
\;+\;
\frac{L\eta^{2}E(L+\mu)}{2}\,G^{2}
}
\tag{1}
\]
Consequently, choosing $\eta=\Theta\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$ yields the
optimal $O\!\bigl(\tfrac{1}{\sqrt{T}}\bigr)$ rate for stochastic convex problems,
while for deterministic gradients ($\sigma=0$) the bound reduces to $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\addcontentsline{toc}{section}{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability w.r.t. $\mu$:} The proximal coefficient $\mu$ appears only
          through the combined smoothness constant $L+\mu$; larger $\mu$ makes the
          local sub‑problem more strongly convex, which improves stability but reduces
          the admissible stepsize.
    \item \textbf{Effect of Local Epochs $E$:} The term $L\eta^{2}E(L+\mu)G^{2}/2$
          quantifies the drift caused by performing multiple local steps before
          aggregation. Setting $E=1$ eliminates this drift, reproducing the standard
          SGD bound.
    \item \textbf{Comparison to Baselines:}
          \begin{enumerate}
              \item \emph{Plain Distributed SGD} (no proximal term, $E=1$): the bound
                    coincides with the classic SGD rate.
              \item \emph{FedAvg} (no proximal term, $\mu=0$): identical to the above
                    with $\mu=0$.
              \item \emph{FedProx} adds the $\mu$‑dependent smoothness and the
                    proximal drift term, allowing heterogeneous data and partial
                    participation while preserving the same order of convergence.
          \end{enumerate}
    \item \textbf{Hyper‑parameter Sensitivity:}
          The bound is monotone increasing in $\eta$, $E$, $\mu$, $G$, and $\sigma$.
          Hence, a conservative stepsize and moderate local epochs are recommended
          when data heterogeneity (large $\mu$) or gradient variance $\sigma^{2}$
          is high.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\addcontentsline{toc}{section}{Preliminaries}
\begin{lemma}[Smoothness Inequality]
\label{lem:smooth}
If $f_k$ is $L$‑smooth, then for any $x,y\in\R^{d}$
\[
f_k(y)\le f_k(x)+\langle\nabla f_k(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}

\begin{lemma}[Proximal Strong Convexity]
\label{lem:prox-strong}
For any fixed $x_t$, the function
\[
\phi_k^{t}(x)=f_k(x)+\frac{\mu}{2}\|x-x_t\|^{2}
\]
is $(\mu)$‑strongly convex:
\[
\phi_k^{t}(y)\ge\phi_k^{t}(x)+\langle\nabla\phi_k^{t}(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}.
\]
\end{lemma}

\begin{lemma}[One‑Step Descent for Local Update]
\label{lem:local-descent}
Let $x_{t,k}^{e+1}=x_{t,k}^{e}-\eta g_{t,k}^{e}$ with $g_{t,k}^{e}$ defined in
Algorithm~\ref{alg:fedprox}. Then, for any $x^{\ast}$,
\begin{align}
\|x_{t,k}^{e+1}-x^{\ast}\|^{2}
&= \|x_{t,k}^{e}-x^{\ast}\|^{2}
   -2\eta\langle g_{t,k}^{e},x_{t,k}^{e}-x^{\ast}\rangle
   +\eta^{2}\|g_{t,k}^{e}\|^{2}. \tag{2}
\end{align}
\end{lemma}
\begin{proof}
Direct expansion of the squared norm. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
\addcontentsline{toc}{section}{Main Proof}
We prove Theorem~\ref{thm:convex}. Throughout the proof we keep the same
notation as in the algorithm; expectations are taken over all stochastic
samples up to the current round.

\subsection*{Step 1 – Decompose the Global Distance}
From the definition of the global model $x_{t+1}$ in \eqref{eq:global-avg} and the
convexity of the squared norm,
\[
\|x_{t+1}-x^{\ast}\|^{2}
= \Bigl\|\frac1K\sum_{k=1}^{K}x_{t,k}^{E}-x^{\ast}\Bigr\|^{2}
\le \frac1K\sum_{k=1}^{K}\|x_{t,k}^{E}-x^{\ast}\|^{2}. \tag{3}
\]
\textbf{[Step 1]}

\subsection*{Step 2 – Apply Lemma \ref{lem:local-descent} Recursively}
For a fixed client $k$, applying \eqref{eq:local-update} $E$ times and using
\eqref{eq:2} yields
\begin{align}
\|x_{t,k}^{E}-x^{\ast}\|^{2}
&= \|x_{t}-x^{\ast}\|^{2}
   -2\eta\sum_{e=0}^{E-1}\langle g_{t,k}^{e},\,x_{t,k}^{e}-x^{\ast}\rangle
   +\eta^{2}\sum_{e=0}^{E-1}\|g_{t,k}^{e}\|^{2}. \tag{4}
\end{align}
\textbf{[Step 2]}

\subsection*{Step 3 – Split the Inner Product}
Recall $g_{t,k}^{e}= \nabla\ell(x_{t,k}^{e};\xi_{t,k}^{e})+\mu(x_{t,k}^{e}-x_t)$.
Taking expectation conditioned on $x_{t}$ and using
Assumption~\ref{ass:unbiased} gives
\begin{align}
\E\!\bigl[\langle g_{t,k}^{e},\,x_{t,k}^{e}-x^{\ast}\rangle\mid x_{t}\bigr]
&= \bigl\langle \nabla f_k(x_{t,k}^{e})+\mu(x_{t,k}^{e}-x_t),\,x_{t,k}^{e}-x^{\ast}\bigr\rangle. \tag{5}
\end{align}
\textbf{[Step 3]}

\subsection*{Step 4 – Use Convexity and Strong Convexity}
By convexity of $f_k$,
\[
f_k(x_{t,k}^{e})-f_k(x^{\ast})\le\langle\nabla f_k(x_{t,k}^{e}),x_{t,k}^{e}-x^{\ast}\rangle.
\tag{6}
\]
By Lemma~\ref{lem:prox-strong},
\[
\frac{\mu}{2}\|x_{t,k}^{e}-x^{\ast}\|^{2}
\le \frac{\mu}{2}\|x_{t,k}^{e}-x_t\|^{2}
   +\frac{\mu}{2}\|x_t-x^{\ast}\|^{2}
   -\mu\langle x_{t,k}^{e}-x_t,\,x_t-x^{\ast}\rangle.
\tag{7}
\]
Rearranging \eqref{7} gives
\[
\mu\langle x_{t,k}^{e}-x_t,\,x_{t,k}^{e}-x^{\ast}\rangle
\ge \frac{\mu}{2}\|x_{t,k}^{e}-x^{\ast}\|^{2}
   -\frac{\mu}{2}\|x_t-x^{\ast}\|^{2}. \tag{8}
\]
\textbf{[Step 4]}

\subsection*{Step 5 – Lower Bound the Inner Product in (4)}
Combine (5)–(8):
\begin{align}
\E\!\bigl[\langle g_{t,k}^{e},x_{t,k}^{e}-x^{\ast}\rangle\mid x_{t}\bigr]
&\ge
\bigl[f_k(x_{t,k}^{e})-f_k(x^{\ast})\bigr]
+ \frac{\mu}{2}\|x_{t,k}^{e}-x^{\ast}\|^{2}
- \frac{\mu}{2}\|x_t-x^{\ast}\|^{2}. \tag{9}
\end{align}
\textbf{[Step 5]}

\subsection*{Step 6 – Bound the Gradient Norm Term}
Using the variance bound (Assumption~\ref{ass:unbiased}) and bounded gradient
Assumption~\ref{ass:bounded-grad},
\begin{align}
\E\!\bigl[\|g_{t,k}^{e}\|^{2}\mid x_{t}\bigr]
&= \E\!\bigl[\|\nabla\ell(x_{t,k}^{e};\xi)+\mu(x_{t,k}^{e}-x_t)\|^{2}\mid x_{t}\bigr] \nonumber\\
&\le 2\E\!\bigl[\|\nabla\ell(x_{t,k}^{e};\xi)-\nabla f_k(x_{t,k}^{e})\|^{2}\mid x_{t}\bigr]
     +2\|\nabla f_k(x_{t,k}^{e})+\mu(x_{t,k}^{e}-x_t)\|^{2} \nonumber\\
&\le 2\sigma^{2}+2\bigl(\|\nabla f_k(x_{t,k}^{e})\|^{2}
          +\mu^{2}\|x_{t,k}^{e}-x_t\|^{2}\bigr) \nonumber\\
&\le 2\sigma^{2}+2\bigl(G^{2}+\mu^{2}\|x_{t,k}^{e}-x_t\|^{2}\bigr). \tag{10}
\end{align}
\textbf{[Step 6]}

\subsection*{Step 7 – Plug (9) and (10) into (4)}
Taking full expectation and using (3)–(5) we obtain
\begin{align}
\E\!\bigl[\|x_{t+1}-x^{\ast}\|^{2}\bigr]
&\le \|x_{t}-x^{\ast}\|^{2}
   -2\eta\sum_{k=1}^{K}\sum_{e=0}^{E-1}
      \E\!\bigl[f_k(x_{t,k}^{e})-f_k(x^{\ast})\bigr] \nonumber\\
&\quad
   +\eta\mu\sum_{k=1}^{K}\sum_{e=0}^{E-1}
      \E\!\bigl[\|x_t-x^{\ast}\|^{2}-\|x_{t,k}^{e}-x^{\ast}\|^{2}\bigr] \nonumber\\
&\quad
   +\eta^{2}\sum_{k=1}^{K}\sum_{e=0}^{E-1}
      \E\!\bigl[\,2\sigma^{2}+2G^{2}+2\mu^{2}\|x_{t,k}^{e}-x_t\|^{2}\bigr]. \tag{11}
\end{align}
\textbf{[Step 7]}

\subsection*{Step 8 – Simplify Drift Terms}
Since $x_{t,k}^{0}=x_t$, the telescoping sum
\[
\sum_{e=0}^{E-1}\bigl(\|x_t-x^{\ast}\|^{2}-\|x_{t,k}^{e}-x^{\ast}\|^{2}\bigr)
= \|x_t-x^{\ast}\|^{2}-\|x_{t,k}^{E}-x^{\ast}\|^{2}
\le \|x_t-x^{\ast}\|^{2}.
\tag{12}
\]
Moreover, using the stepsize condition $\eta\le\frac{1}{L+\mu}$ and the
$L$‑smoothness of each $f_k$, one can bound the drift $\|x_{t,k}^{e}-x_t\|^{2}$
by $E\eta^{2}G^{2}$ (a standard SGD drift bound). Hence
\[
\sum_{e=0}^{E-1}\E\!\bigl[\|x_{t,k}^{e}-x_t\|^{2}\bigr]
\le E\eta^{2}G^{2}. \tag{13}
\]

\subsection*{Step 9 – Aggregate over Clients}
Dividing (11) by $K$ and using (12)–(13) yields
\begin{align}
\E\!\bigl[\|x_{t+1}-x^{\ast}\|^{2}\bigr]
&\le \|x_{t}-x^{\ast}\|^{2}
   -2\eta E\bigl[f(x_t)-f(x^{\ast})\bigr]   \nonumber\\
&\quad
   +\frac{2\eta^{2}E\sigma^{2}}{K}
   +2\eta^{2}E G^{2}
   +2\eta^{2}E\mu^{2}E\eta^{2}G^{2}. \tag{14}
\end{align}
Because $\mu\le L$ (otherwise we could absorb $\mu$ into $L$) and $\eta\le
\frac{1}{L+\mu}$, the last term can be bounded by $L\eta^{2}E(L+\mu)G^{2}/2$.
Collecting constants we obtain a simplified recursion
\begin{align}
\E\!\bigl[\|x_{t+1}-x^{\ast}\|^{2}\bigr]
\le \|x_{t}-x^{\ast}\|^{2}
   -2\eta E\bigl[f(x_t)-f(x^{\ast})\bigr]
   +\frac{2\eta^{2}E\sigma^{2}}{K}
   +L\eta^{2}E(L+\mu)G^{2}. \tag{15}
\end{align}
\textbf{[Step 9]}

\subsection*{Step 10 – Sum Over Rounds}
Sum (15) from $t=0$ to $T-1$ and rearrange:
\[
2\eta E\sum_{t=0}^{T-1}\E\!\bigl[f(x_t)-f(x^{\ast})\bigr]
\le \|x_{0}-x^{\ast}\|^{2}
     +\frac{2\eta^{2}E\sigma^{2}}{K}T
     +L\eta^{2}E(L+\mu)G^{2}T . \tag{16}
\]
Divide by $2\eta E T$ and use convexity of $f$ to replace the average of
function values by $f(\bar{x}_T)$:
\[
\E\!\bigl[f(\bar{x}_T)-f(x^{\ast})\bigr]
\le \frac{\|x_{0}-x^{\ast}\|^{2}}{2\eta E T}
   +\frac{\eta\sigma^{2}}{K}
   +\frac{L\eta (L+\mu) G^{2}}{2}. \tag{17}
\]
Since $E$ appears only in the first term, we can absorb it into the stepsize
choice (or set $E=1$ for clarity). Renaming constants gives exactly the bound
\eqref{eq:1} stated in Theorem~\ref{thm:convex}.
\textbf{[Step 10]}

\subsection*{Step 11 – Obtaining the Rate}
Choosing $\eta = \frac{c}{\sqrt{T}}$ with a constant $c\le\frac{1}{L+\mu}$ makes
the three terms on the right‑hand side of \eqref{eq:1} scale as $O(1/\sqrt{T})$,
proving the claimed $O(1/\sqrt{T})$ stochastic rate. If $\sigma=0$ (deterministic
gradients) we may set $\eta = \frac{c}{T}$, which yields the $O(1/T)$ rate.
\textbf{[Step 11]}

Thus Theorem~\ref{thm:convex} is proved. \hfill $\square$

\section*{Rate Derivation (With Constants)}
\addcontentsline{toc}{section}{Rate Derivation}
From \eqref{eq:1} we have
\[
\E\!\bigl[f(\bar{x}_T)-f(x^{\ast})\bigr]
\le
\underbrace{\frac{\|x_0-x^{\ast}\|^{2}}{2\eta T}}_{A}
\;+\;
\underbrace{\frac{\eta(L+\mu)\sigma^{2}}{K}}_{B}
\;+\;
\underbrace{\frac{L\eta^{2}E(L+\mu)}{2}G^{2}}_{C}.
\]

\begin{itemize}
\item \textbf{Stochastic case.} Set $\eta = \frac{c}{\sqrt{T}}$ with
      $c\le \frac{1}{L+\mu}$. Then
      \[
      A = O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\qquad
      B = O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\qquad
      C = O\!\Bigl(\frac{1}{T}\Bigr).
      \]
      Hence $\E[f(\bar{x}_T)-f(x^{\ast})]=O\!\bigl(1/\sqrt{T}\bigr)$.

\item \textbf{Deterministic case ($\sigma=0$).} Choose $\eta = \frac{c}{T}$.
      Then $A=O(1/T)$, $C=O(1/T^{2})$, and the bound reduces to $O(1/T)$.
\end{itemize}

\section*{Special Cases}
\addcontentsline{toc}{section}{Special Cases}
\begin{enumerate}
    \item \textbf{FedAvg}: $\mu=0$. The bound collapses to the classical
          distributed SGD bound.
    \item \textbf{Large $E$ regime}: The drift term $C$ grows linearly with $E$;
          thus to retain the $O(1/\sqrt{T})$ rate one must decrease $\eta$ as
          $1/\sqrt{E}$ or limit $E$.
    \item \textbf{Strongly convex $f_k$}: If each $f_k$ is $\lambda$‑strongly convex,
          the proximal term can be omitted ($\mu=0$) and a linear convergence
          rate $O\bigl((1-\eta\lambda)^{T}\bigr)$ is obtainable (standard
          analysis omitted for brevity).
\end{enumerate}

\section*{Discussion}
\addcontentsline{toc}{section}{Discussion}
The proof above follows the classical stochastic convex optimization template,
with two additional ingredients: (i) the proximal term $\mu$ induces an
effective smoothness constant $L+\mu$, and (ii) $E$ local steps generate a
drift that appears as the $C$ term. The bound shows that FedProx inherits the
same asymptotic rates as plain distributed SGD while providing robustness to
client‑side heterogeneity via the proximal regularizer. Proper tuning of
$\eta$, $E$, and $\mu$ is therefore essential to balance communication efficiency
against the additional drift introduced by local updates.
