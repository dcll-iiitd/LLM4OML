\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics (SGLD) with Decaying Noise}\\
The method performs a stochastic gradient step together with an injected Gaussian noise whose variance decays over iterations. It is designed for non‑convex smooth objectives.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective. For iteration $k\ge 0$ define
\[
\boxed{
x_{k+1}=x_{k}-\alpha_{k}\,g_{k}+ \sqrt{2\alpha_{k}\,\tau_{k}}\;\xi_{k}},
\tag{1}
\]
where
\begin{itemize}[nosep]
\item $\alpha_{k}>0$ is the stepsize,
\item $g_{k}$ is a stochastic gradient satisfying $\E[g_{k}\mid x_{k}]=\nabla f(x_{k})$,
\item $\tau_{k}>0$ is a temperature (noise level) that is non‑increasing,
\item $\xi_{k}\sim\mathcal N(0,\mathbf I_{d})$ are i.i.d. standard Gaussian vectors, independent of the past.
\end{itemize}
When $\tau_{k}=0$ the method reduces to plain stochastic gradient descent.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD with Decaying Noise}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_{0}\in\R^{d}$, stepsize schedule $\{\alpha_{k}\}_{k\ge0}$, temperature schedule $\{\tau_{k}\}_{k\ge0}$
\For{$k=0,1,2,\dots,T-1$}
    \State Sample a mini‑batch $B_{k}$ and compute stochastic gradient 
    \[
    g_{k}= \frac{1}{|B_{k}|}\sum_{z\in B_{k}}\nabla f(x_{k};z)
    \]
    \State Sample $\xi_{k}\sim\mathcal N(0,\mathbf I_{d})$
    \State Update 
    \[
    x_{k+1}=x_{k}-\alpha_{k} g_{k}+ \sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}
    \]
\EndFor
\State \textbf{Return} $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, deterministic gradients (so $g_{k}=\nabla f(w_{k})$) and set the noise to zero for illustration. Choose $\alpha=0.1$, $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\alpha\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
Corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.3616$, showing monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{2}
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For every $k$, $\E[g_{k}\mid x_{k}]=\nabla f(x_{k})$.
\end{assumption}
\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\E\!\left[\norm{g_{k}-\nabla f(x_{k})}^{2}\mid x_{k}\right]\le\sigma^{2},\qquad\forall k.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Stepsize and Temperature]\label{as:schedule}
The sequences $\{\alpha_{k}\}$, $\{\tau_{k}\}$ satisfy
\[
\alpha_{k}>0,\;\tau_{k}\ge0,\qquad
\sum_{k=0}^{\infty}\alpha_{k}= \infty,\;\; 
\sum_{k=0}^{\infty}\alpha_{k}^{2}<\infty,\qquad
\tau_{k}=o(\alpha_{k})\;\;(\text{i.e., }\tau_{k}/\alpha_{k}\to0).
\tag{4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:schedule}, let $\{x_{k}\}$ be generated by \eqref{eq:sgld}. Define the weighted average stepsize
\[
\bar\alpha_{K}:=\frac{\sum_{k=0}^{K-1}\alpha_{k}}{K}.
\]
Then for any $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\;\le\;
\frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{\sum_{k=0}^{K-1}\alpha_{k}}
\;+\;
\frac{L\sigma^{2}}{\sum_{k=0}^{K-1}\alpha_{k}}\sum_{k=0}^{K-1}\alpha_{k}^{2}
\;+\;
\frac{2L d}{\sum_{k=0}^{K-1}\alpha_{k}}\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{5}
\]
Consequently, if $\alpha_{k}=c/k^{a}$ with $a\in(0.5,1]$ and $\tau_{k}=c_{\tau}/k^{b}$ with $b>a$, the right‑hand side is $O(K^{-a})$, implying 
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]=O\!\left(K^{-a}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[nosep]
\item \textbf{Stability.} The $L$‑smoothness together with the decreasing stepsize guarantees that the iterates remain bounded in expectation (see Lemma~\ref{lem:bounded}).
\item \textbf{Hyperparameter Sensitivity.} The convergence bound decomposes into three contributions: (i) deterministic descent term $\propto 1/\sum\alpha_{k}$, (ii) stochastic gradient variance term $\propto \sum\alpha_{k}^{2}/\sum\alpha_{k}$, (iii) injected noise term $\propto \sum\alpha_{k}\tau_{k}/\sum\alpha_{k}$. Choosing $\tau_{k}=o(\alpha_{k})$ makes the third term negligible.
\item \textbf{Comparison to Baselines.} Setting $\tau_{k}\equiv0$ recovers the standard SGD bound (the third term disappears). Conversely, keeping $\tau_{k}$ constant yields a stationary distribution whose variance does not vanish; Theorem~\ref{thm:main} shows that decaying $\tau_{k}$ restores convergence to stationary points.
\end{itemize}

\section*{Discussion}
The theorem shows that SGLD with a temperature schedule decreasing faster than the stepsize inherits the same $O(1/\sqrt{K})$ (or $O(K^{-a})$) rate as SGD for the expected squared gradient norm, while still providing an exploration mechanism in early iterations. The bound is non‑asymptotic and holds for any non‑convex $L$‑smooth objective.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{6}
\]
\end{lemma}
\begin{lemma}[Bounded Second Moment of Noise]\label{lem:noise}
For $\xi\sim\mathcal N(0,\mathbf I_{d})$ and any $\beta>0$,
\[
\E\bigl[\norm{\sqrt{2\beta}\,\xi}^{2}\bigr]=2\beta d.
\tag{7}
\]
\end{lemma}
\begin{lemma}[Iterate Boundedness]\label{lem:bounded}
Under Assumptions \ref{as:smooth}–\ref{as:schedule} the sequence $\{x_{k}\}$ satisfies
\[
\sup_{k\ge0}\E\bigl[f(x_{k})\bigr]\le f(x_{0})+\frac{L\sigma^{2}}{2}\sum_{k=0}^{\infty}\alpha_{k}^{2}
\;+\;L d\sum_{k=0}^{\infty}\alpha_{k}\tau_{k}<\infty .
\tag{8}
\]
\end{lemma}
\begin{proof}[Proof of Lemma~\ref{lem:bounded}]
From smoothness \eqref{eq:smooth} (i.e., \eqref{6}) with $y=x_{k+1}$ and $x=x_{k}$ we obtain
\[
f(x_{k+1})\le f(x_{k})+\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}+\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}.
\tag{9}
\]
Insert the update \eqref{eq:sgld}:
\[
x_{k+1}-x_{k}= -\alpha_{k}g_{k}+ \sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}.
\tag{10}
\]
Taking expectation conditioned on $x_{k}$ and using unbiasedness \eqref{as:unbiased} yields
\[
\E\!\left[\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}\mid x_{k}\right]
= -\alpha_{k}\norm{\nabla f(x_{k})}^{2}.
\tag{11}
\]
For the quadratic term,
\[
\E\!\left[\norm{x_{k+1}-x_{k}}^{2}\mid x_{k}\right]
= \alpha_{k}^{2}\E\!\left[\norm{g_{k}}^{2}\mid x_{k}\right]
+2\alpha_{k}\tau_{k}\E\!\left[\norm{\xi_{k}}^{2}\right]
\tag{12}
\]
(the cross term vanishes because $\E[\xi_{k}]=0$). Using variance bound \eqref{as:variance} and Lemma~\ref{lem:noise}:
\[
\E\!\left[\norm{g_{k}}^{2}\mid x_{k}\right]
= \norm{\nabla f(x_{k})}^{2}+\E\!\left[\norm{g_{k}-\nabla f(x_{k})}^{2}\mid x_{k}\right]
\le \norm{\nabla f(x_{k})}^{2}+\sigma^{2},
\tag{13}
\]
\[
\E\!\left[\norm{\xi_{k}}^{2}\right]=d.
\tag{14}
\]
Plugging \eqref{11}–\eqref{14} into \eqref{9} and taking full expectation gives
\[
\E[f(x_{k+1})]\le \E[f(x_{k})]
-\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+\frac{L}{2}\alpha_{k}^{2}\bigl(\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]+\sigma^{2}\bigr)
+L\alpha_{k}\tau_{k}d .
\tag{15}
\]
Discard the negative term $-\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]$ to obtain an upper recursion
\[
\E[f(x_{k+1})]\le \E[f(x_{k})]
+\frac{L\sigma^{2}}{2}\alpha_{k}^{2}+L d\alpha_{k}\tau_{k}.
\tag{16}
\]
Iterating \eqref{16} from $0$ to $K-1$ yields \eqref{8}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, let $\mathcal F_{k}$ denote the $\sigma$‑algebra generated by $\{x_{0},\xi_{0},\dots,\xi_{k-1}\}$.

\noindent\textbf{Step 1 (Smoothness bound).}  
From Lemma~\ref{lem:smooth} with $x=x_{k}$, $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}+\frac{L}{2}\norm{x_{k+1}-x_{k}}^{2}.
\tag{17}
\]
\textit{[Step 1]}

\noindent\textbf{Step 2 (Insert the update).}  
Using \eqref{1},
\[
x_{k+1}-x_{k}= -\alpha_{k}g_{k}+ \sqrt{2\alpha_{k}\tau_{k}}\;\xi_{k}.
\tag{18}
\]
\textit{[Step 2]}

\noindent\textbf{Step 3 (Take conditional expectation of the linear term).}  
Condition on $\mathcal F_{k}$ and invoke unbiasedness \eqref{as:unbiased}:
\[
\E\!\left[\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}\mid\mathcal F_{k}\right]
= -\alpha_{k}\norm{\nabla f(x_{k})}^{2}.
\tag{19}
\]
\textit{[Step 3]}

\noindent\textbf{Step 4 (Second‑moment of the step).}  
Conditioned on $\mathcal F_{k}$,
\[
\begin{aligned}
\E\!\left[\norm{x_{k+1}-x_{k}}^{2}\mid\mathcal F_{k}\right]
&= \alpha_{k}^{2}\E\!\left[\norm{g_{k}}^{2}\mid\mathcal F_{k}\right]
+2\alpha_{k}\tau_{k}\E\!\left[\norm{\xi_{k}}^{2}\right] \\
&\overset{(a)}{=}\alpha_{k}^{2}\Bigl(\norm{\nabla f(x_{k})}^{2}
+\E\!\left[\norm{g_{k}-\nabla f(x_{k})}^{2}\mid\mathcal F_{k}\right]\Bigr)
+2\alpha_{k}\tau_{k}d,
\end{aligned}
\tag{20}
\]
where (a) uses $\E[g_{k}]=\nabla f(x_{k})$ and Lemma~\ref{lem:noise}.  
\textit{[Step 4]}

\noindent\textbf{Step 5 (Apply variance bound).}  
From Assumption~\ref{as:variance},
\[
\E\!\left[\norm{g_{k}-\nabla f(x_{k})}^{2}\mid\mathcal F_{k}\right]\le\sigma^{2}.
\tag{21}
\]
Thus,
\[
\E\!\left[\norm{x_{k+1}-x_{k}}^{2}\mid\mathcal F_{k}\right]
\le \alpha_{k}^{2}\bigl(\norm{\nabla f(x_{k})}^{2}+\sigma^{2}\bigr)
+2\alpha_{k}\tau_{k}d .
\tag{22}
\]
\textit{[Step 5]}

\noindent\textbf{Step 6 (Combine \eqref{17}, \eqref{19}, \eqref{22}).}  
Taking total expectation and using linearity,
\[
\begin{aligned}
\E[f(x_{k+1})]
&\le \E[f(x_{k})]
-\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+\frac{L}{2}\alpha_{k}^{2}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
+\frac{L\sigma^{2}}{2}\alpha_{k}^{2}
+L\alpha_{k}\tau_{k}d .
\end{aligned}
\tag{23}
\]
\textit{[Step 6]}

\noindent\textbf{Step 7 (Rearrange).}  
Move the gradient term to the left:
\[
\alpha_{k}\Bigl(1-\frac{L\alpha_{k}}{2}\Bigr)\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le \E[f(x_{k})]-\E[f(x_{k+1})] +\frac{L\sigma^{2}}{2}\alpha_{k}^{2}+L d\alpha_{k}\tau_{k}.
\tag{24}
\]
\textit{[Step 7]}

\noindent\textbf{Step 8 (Step‑size restriction).}  
Assumption \ref{as:schedule} ensures $\alpha_{k}\le 1/L$ for all sufficiently large $k$, hence $1-\tfrac{L\alpha_{k}}{2}\ge\tfrac12$. Dropping the factor $\tfrac12$ yields a simpler inequality valid for all $k$:
\[
\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le 2\bigl(\E[f(x_{k})]-\E[f(x_{k+1})]\bigr)
+L\sigma^{2}\alpha_{k}^{2}+2L d\alpha_{k}\tau_{k}.
\tag{25}
\]
\textit{[Step 8]}

\noindent\textbf{Step 9 (Sum over $k=0,\dots,K-1$).}  
Summing \eqref{25} telescopes the function values:
\[
\sum_{k=0}^{K-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le 2\bigl(f(x_{0})- \E[f(x_{K})]\bigr)
+L\sigma^{2}\sum_{k=0}^{K-1}\alpha_{k}^{2}
+2L d\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{26}
\]
Since $f(x_{K})\ge f^{\inf}$,
\[
\sum_{k=0}^{K-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le 2\bigl(f(x_{0})-f^{\inf}\bigr)
+L\sigma^{2}\sum_{k=0}^{K-1}\alpha_{k}^{2}
+2L d\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{27}
\]
\textit{[Step 9]}

\noindent\textbf{Step 10 (Divide by $\sum_{k=0}^{K-1}\alpha_{k}$).}  
Define $S_{K}:=\sum_{k=0}^{K-1}\alpha_{k}$. Dividing \eqref{27} by $S_{K}$ yields
\[
\frac{1}{S_{K}}\sum_{k=0}^{K-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le \frac{2\bigl(f(x_{0})-f^{\inf}\bigr)}{S_{K}}
+\frac{L\sigma^{2}}{S_{K}}\sum_{k=0}^{K-1}\alpha_{k}^{2}
+\frac{2L d}{S_{K}}\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}.
\tag{28}
\]
Observe that the left‑hand side is a weighted average of $\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]$ with weights $\alpha_{k}/S_{K}$. Since the weights are non‑negative and sum to $1$, the weighted average is bounded above by the simple arithmetic average:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le \frac{1}{S_{K}}\sum_{k=0}^{K-1}\alpha_{k}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right].
\tag{29}
\]
Combining \eqref{28} and \eqref{29} gives exactly the bound \eqref{5} stated in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
Assume $\alpha_{k}=c_{\alpha}k^{-a}$ with $a\in(0.5,1]$ and $\tau_{k}=c_{\tau}k^{-b}$ with $b>a$. Then
\[
S_{K}\asymp \frac{c_{\alpha}}{1-a}K^{1-a},\qquad
\sum_{k=0}^{K-1}\alpha_{k}^{2}\asymp c_{\alpha}^{2}\frac{K^{1-2a}}{1-2a},
\qquad
\sum_{k=0}^{K-1}\alpha_{k}\tau_{k}\asymp c_{\alpha}c_{\tau}\frac{K^{1-(a+b)}}{1-(a+b)}.
\]
Plugging into \eqref{5} yields
\[
\frac{1}{K}\sum_{k=0}^{K-1}\E\!\left[\norm{\nabla f(x_{k})}^{2}\right]
\le
\underbrace{\frac{2(f(x_{0})-f^{\inf})(1-a)}{c_{\alpha}}}_{\displaystyle O(K^{-a})}
+ \underbrace{\frac{L\sigma^{2}c_{\alpha}}{2(1-2a)}K^{-a}}_{\displaystyle O(K^{-a})}
+ \underbrace{\frac{2Ld c_{\tau}}{1-(a+b)}K^{-a}}_{\displaystyle O(K^{-a})},
\]
which is $O(K^{-a})$. Choosing $a=1/2$ recovers the classical $O(1/\sqrt{K})$ rate.

\section*{Special Cases}
\begin{enumerate}[nosep]
\item \textbf{Pure SGD ($\tau_{k}=0$).} The third term in \eqref{5} disappears, and the bound coincides with the standard SGD result for non‑convex smooth objectives.
\item \textbf{Constant temperature ($\tau_{k}\equiv\tau>0$).} The term $\sum\alpha_{k}\tau_{k}$ grows like $S_{K}$, so the bound stabilises at a constant $O(\tau)$, reflecting convergence to a stationary distribution rather than a point.
\item \textbf{Vanishing stepsize, fixed temperature.} If $\alpha_{k}\to0$ but $\tau_{k}\equiv\tau$, the injected noise dominates the asymptotics and the iterates converge in distribution to $\mathcal N\bigl(0, \tau I\bigr)$ around local minima (classical Langevin diffusion result).
\end{enumerate}
