\section*{Away-Step Frank–Wolfe (AFW) Algorithm}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be convex and $L$-smooth.  
Let $\cC\subset\mathbb{R}^d$ be a compact polytope.  
At iteration $k$ we maintain a representation
\[
x_k=\sum_{v\in\mathcal{S}_k}\alpha_v^{(k)} v,\qquad
\alpha_v^{(k)}\ge 0,\ \sum_{v\in\mathcal{S}_k}\alpha_v^{(k)}=1,
\]
where $\mathcal{S}_k\subseteq\cC$ is the \emph{active set} (vertices currently used).

\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*]
\item Compute the \emph{Frank–Wolfe} vertex
\[
s_k \;:=\; \argmin_{s\in\cC}\;\langle\nabla f(x_k),s\rangle .
\]
\item Compute the \emph{away} vertex
\[
v_k \;:=\; \argmax_{v\in\mathcal{S}_k}\;\langle\nabla f(x_k),v\rangle .
\]
\item Define the two candidate directions
\[
d^{\text{FW}}_k:=s_k-x_k,\qquad 
d^{\text{A}}_k:=x_k-v_k .
\]
\item Choose the descent direction
\[
\text{if }\;\langle\nabla f(x_k),d^{\text{FW}}_k\rangle\le 
\langle\nabla f(x_k),d^{\text{A}}_k\rangle
\;\text{ then } d_k:=d^{\text{FW}}_k,\;\gamma_{\max}=1,
\]
\[
\text{else } d_k:=d^{\text{A}}_k,\;
\gamma_{\max}:=\frac{\alpha_{v_k}^{(k)}}{1-\alpha_{v_k}^{(k)}} .
\]
\item Compute the step size by exact line search (or backtracking)
\[
\gamma_k:=\argmin_{\gamma\in[0,\gamma_{\max}]}\;f(x_k+\gamma d_k).
\]
\item Update the iterate
\[
x_{k+1}=x_k+\gamma_k d_k .
\]
\item Update the active set and coefficients:
\begin{itemize}
\item If $d_k=d^{\text{FW}}_k$ (FW step): 
\[
\alpha_{s_k}^{(k+1)}=
\begin{cases}
\alpha_{s_k}^{(k)}+\gamma_k, & s_k\in\mathcal{S}_k,\\
\gamma_k, & s_k\notin\mathcal{S}_k,
\end{cases}
\quad
\alpha_v^{(k+1)}=(1-\gamma_k)\alpha_v^{(k)}\;\forall v\neq s_k .
\]
\item If $d_k=d^{\text{A}}_k$ (away step): 
\[
\alpha_{v_k}^{(k+1)}=(1+\gamma_k)\alpha_{v_k}^{(k)}-\gamma_k,
\quad
\alpha_v^{(k+1)}=(1+\gamma_k)\alpha_v^{(k)}\;\forall v\neq v_k .
\]
If $\alpha_{v_k}^{(k+1)}=0$, remove $v_k$ from $\mathcal{S}_{k+1}$.
\end{itemize}
\end{enumerate}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Away-Step Frank–Wolfe (AFW)}
\begin{algorithmic}[1]
\Require Convex $L$‑smooth $f$, polytope $\cC$, tolerance $\varepsilon>0$
\State Choose an initial vertex $x_0\in\cC$, set $\mathcal{S}_0=\{x_0\}$, $\alpha^{(0)}_{x_0}=1$
\For{$k=0,1,2,\dots$}
    \State $\displaystyle s_k\gets\argmin_{s\in\cC}\langle\nabla f(x_k),s\rangle$
    \State $\displaystyle v_k\gets\argmax_{v\in\mathcal{S}_k}\langle\nabla f(x_k),v\rangle$
    \State $d^{\text{FW}}_k\gets s_k-x_k$, \; $d^{\text{A}}_k\gets x_k-v_k$
    \If{$\langle\nabla f(x_k),d^{\text{FW}}_k\rangle\le\langle\nabla f(x_k),d^{\text{A}}_k\rangle$}
        \State $d_k\gets d^{\text{FW}}_k$, $\gamma_{\max}\gets 1$
    \Else
        \State $d_k\gets d^{\text{A}}_k$, $\displaystyle\gamma_{\max}\gets\frac{\alpha^{(k)}_{v_k}}{1-\alpha^{(k)}_{v_k}}$
    \EndIf
    \State $\displaystyle\gamma_k\gets\argmin_{\gamma\in[0,\gamma_{\max}]} f(x_k+\gamma d_k)$
    \State $x_{k+1}\gets x_k+\gamma_k d_k$
    \State Update $\mathcal{S}_{k+1}$ and $\{\alpha^{(k+1)}_v\}$ according to the rule in the description
    \State Compute the dual gap $\gap_k:=\langle\nabla f(x_k),x_k-s_k\rangle$
    \If{$\gap_k\le\varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional polytope $\cC=[0,5]$ and
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Initialize $x_0=0$ (a vertex of $\cC$).  
The active set $\mathcal{S}_0=\{0\}$ with $\alpha^{(0)}_{0}=1$.

\begin{center}
\begin{tabular}{c|c|c|c|c|c}
\textbf{Iter} & $x_k$ & $\nabla f(x_k)$ & $s_k$ & $v_k$ & $x_{k+1}$ \\ \hline
0 & $0$ & $-6$ & $\displaystyle\argmin_{s\in[0,5]}(-6)s =5$ & $0$ &
\begin{aligned}
d^{\text{FW}}_0 &=5-0=5\\
\gamma_0 &=\argmin_{\gamma\in[0,1]} (0+\gamma 5-3)^2\\
&=\argmin_{\gamma\in[0,1]} (5\gamma-3)^2\\
&=\frac{3}{5}=0.6\\
x_1&=0+0.6\cdot5=3
\end{aligned}
\\ \hline
1 & $3$ & $0$ & $\displaystyle\argmin_{s}0\cdot s =0$ (any $s$, pick $0$) & $3$ &
\begin{aligned}
d^{\text{FW}}_1&=0-3=-3,\\
\langle\nabla f, d^{\text{FW}}\rangle&=0,\\
d^{\text{A}}_1&=3-3=0,\\
\text{choose FW step},\ \gamma_{\max}=1,\\
\gamma_1=0\ (\text{exact line search}),\\
x_2=3.
\end{aligned}
\\ \hline
2 & $3$ & $0$ & $0$ & $3$ &
\begin{aligned}
\text{Same as iteration 1,}\\
\gap_2=0\le\varepsilon\ (\varepsilon=10^{-6}).
\end{aligned}
\end{tabular}
\end{center}
The algorithm reaches the optimal point $w^\star=3$ after a single effective FW step.

\section*{Assumptions}
\begin{assumption}[Convex and Smooth Objective]\label{ass:smooth}
$f$ is convex and $L$‑smooth on $\mathbb{R}^d$:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
\end{assumption}
\begin{assumption}[Polytope Domain]\label{ass:polytope}
$\cC=\operatorname{conv}\{v^{(1)},\dots,v^{(m)}\}$ is a compact polytope with
\[
\diam(\cC):=\max_{u,v\in\cC}\|u-v\|<\infty .
\]
\end{assumption}
\begin{assumption}[Strong Convexity on $\cC$]\label{ass:strong}
There exists $\mu>0$ such that for all $x\in\cC$,
\[
f(x)-f(x^\star)\ge \frac{\mu}{2}\|x-x^\star\|^2,
\]
where $x^\star:=\argmin_{x\in\cC}f(x)$.
\end{assumption}
\begin{assumption}[Pyramidal Width]\label{ass:pyr}
Define the \emph{pyramidal width} $\delta>0$ of $\cC$ (see \cite{lacoste2015linear}).  
For any $x\in\cC$ and any feasible direction $d$, 
\[
\langle -\nabla f(x),d\rangle\ge\delta\;\|\nabla f(x)\|\,\|d\|.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear Convergence of AFW]\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:pyr}, the iterates $\{x_k\}$ produced by AFW satisfy
\[
f(x_k)-f(x^\star)\;\le\;
\bigl(1-\rho\bigr)^{k}\,\bigl(f(x_0)-f(x^\star)\bigr),
\qquad\text{where }\;
\rho:=\frac{\mu\delta^{2}}{4L}\in(0,1).
\]
Equivalently, the dual gap $\gap_k$ contracts at the same linear rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The algorithm never leaves $\cC$ because each step is a convex combination of vertices in $\cC$.
\item \textbf{Hyper‑parameter Sensitivity.} No learning‑rate tuning is required; the step size is obtained by exact line search (or a simple backtracking rule) and $\gamma_{\max}$ is determined solely by the current coefficient $\alpha_{v_k}^{(k)}$.
\item \textbf{Comparison.} Classical Frank–Wolfe enjoys $O(1/k)$ sublinear convergence on general convex sets. The away‑step variant restores a linear rate on polytopes, matching projected gradient methods while preserving projection‑free updates.
\end{itemize}

\section*{Discussion}
The linear rate hinges on the geometric constant $\delta$ (pyramidal width). For well‑conditioned polytopes (e.g., the simplex) $\delta$ is bounded away from zero, yielding a meaningful $\rho$. If $\cC$ is a general convex set without polyhedral structure, the away direction may be undefined and the guarantee reduces to the $O(1/k)$ rate of the vanilla Frank–Wolfe method.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Descent Lemma]\label{def:descent}
If $f$ is $L$‑smooth, then for any $x,d\in\mathbb{R}^d$ and $\gamma\in[0,1]$,
\[
f(x+\gamma d)\le f(x)+\gamma\langle\nabla f(x),d\rangle+\frac{L\gamma^{2}}{2}\|d\|^{2}.
\tag{DL}
\]
\end{definition}
\begin{lemma}[Curvature Upper Bound]\label{lem:curv}
Define the curvature constant
\[
C_f:=\sup_{\substack{x,s\in\cC\\\gamma\in[0,1]}}
\frac{2}{\gamma^{2}}\bigl(f(x+\gamma(s-x))-f(x)-\gamma\langle\nabla f(x),s-x\rangle\bigr).
\]
Then $C_f\le L\diam(\cC)^{2}$.
\end{lemma}
\begin{proof}
Apply the Descent Lemma \eqref{def:descent} with $d=s-x$ and use $\|s-x\|\le\diam(\cC)$. $\square$
\end{proof}
\begin{definition}[Dual Gap]\label{def:gap}
\[
\gap_k:=\langle\nabla f(x_k),x_k-s_k\rangle
      =\max_{v\in\cC}\langle\nabla f(x_k),x_k-v\rangle .
\]
\end{definition}
\begin{lemma}[Gap Lower Bound via Pyramidal Width]\label{lem:gap-width}
Under Assumption \ref{ass:pyr},
\[
\gap_k\ge \delta\,\|\nabla f(x_k)\|\,\|d_k\|,
\]
where $d_k$ is the direction chosen in Step~4 of the algorithm.
\end{lemma}
\begin{proof}
If the FW direction is selected, $d_k=d^{\text{FW}}_k=s_k-x_k$ and by definition of $s_k$,
\[
\langle-\nabla f(x_k),d_k\rangle
   =\langle-\nabla f(x_k),s_k-x_k\rangle
   =\gap_k .
\]
Applying the pyramidal width inequality yields the claim.  
If the away direction is selected, $d_k=d^{\text{A}}_k=x_k-v_k$ and the same argument with $v_k$ (the maximizer) gives the same lower bound. $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  
All steps are explicitly numbered for reference.

\begin{proof}
Let $x_k$ be the current iterate and $d_k$ the direction chosen in Step~4.  

\textbf{[Step 1]} Apply the Descent Lemma \eqref{def:descent} with $d=d_k$ and $\gamma=\gamma_k$:
\[
f(x_{k+1})\le f(x_k)+\gamma_k\langle\nabla f(x_k),d_k\rangle
               +\frac{L\gamma_k^{2}}{2}\|d_k\|^{2}.
\tag{1}
\]

\textbf{[Step 2]} By construction of $\gamma_k$ (exact line search on $[0,\gamma_{\max}]$) we have
\[
\langle\nabla f(x_k),d_k\rangle+\!L\gamma_k\|d_k\|^{2}\le 0 .
\tag{2}
\]
Indeed, the derivative of the univariate function $\phi(\gamma)=f(x_k+\gamma d_k)$ at the minimizer $\gamma_k$ satisfies $\phi'(\gamma_k)=0$, i.e.
\[
\langle\nabla f(x_k+\gamma_k d_k),d_k\rangle=0,
\]
and using $L$‑smoothness to bound $\nabla f(x_k+\gamma_k d_k)$ by $\nabla f(x_k)+L\gamma_k d_k$ yields (2).  

\textbf{[Step 3]} Combine (1) and (2):
\[
f(x_{k+1})\le f(x_k)-\frac{L\gamma_k^{2}}{2}\|d_k\|^{2}.
\tag{3}
\]

\textbf{[Step 4]} Relate $\gamma_k$ to the dual gap.  
From the definition of $\gamma_{\max}$ we have two cases.

\begin{enumerate}[label=(\alph*)]
\item \emph{FW step:} $\gamma_{\max}=1$. Since $\gamma_k$ minimizes $\phi(\gamma)$ on $[0,1]$, convexity of $f$ gives
\[
\phi(\gamma_k)\le (1-\gamma_k)\phi(0)+\gamma_k\phi(1)
               = f(x_k)+\gamma_k\langle\nabla f(x_k),d^{\text{FW}}_k\rangle .
\]
Hence $\langle\nabla f(x_k),d_k\rangle\le -\gap_k$ (by definition of $s_k$). Using Lemma \ref{lem:gap-width},
\[
\gap_k\ge\delta\|\nabla f(x_k)\|\,\|d_k\|.
\tag{4}
\]

\item \emph{Away step:} $\gamma_{\max}=\frac{\alpha^{(k)}_{v_k}}{1-\alpha^{(k)}_{v_k}}$.  
The same convexity argument applied to the away direction yields
\[
\langle\nabla f(x_k),d_k\rangle\le -\gap_k .
\tag{5}
\]
Again Lemma \ref{lem:gap-width} gives (4).
\end{enumerate}

Thus, in both cases,
\[
\langle\nabla f(x_k),d_k\rangle\le -\gap_k\le -\delta\|\nabla f(x_k)\|\,\|d_k\|.
\tag{6}
\]

\textbf{[Step 5]} Strong convexity (Assumption \ref{ass:strong}) implies
\[
\|\nabla f(x_k)\|\ge \mu\|x_k-x^\star\|.
\tag{7}
\]

\textbf{[Step 6]} Combine (6) and (7) to obtain a lower bound on the directional derivative:
\[
-\langle\nabla f(x_k),d_k\rangle\ge \delta\mu\|x_k-x^\star\|\,\|d_k\|.
\tag{8}
\]

\textbf{[Step 7]} Use the Cauchy–Schwarz inequality in the form
\[
\|x_k-x^\star\|\,\|d_k\|\ge \frac{1}{2}\bigl(\|x_k-x^\star\|^{2}+\|d_k\|^{2}\bigr).
\tag{9}
\]

\textbf{[Step 8]} Plug (8) and (9) into (3):
\[
\begin{aligned}
f(x_{k+1}) 
&\le f(x_k)-\frac{L\gamma_k^{2}}{2}\|d_k\|^{2}\\
&\le f(x_k)-\frac{L}{2}\bigl(\gamma_k\|d_k\|\bigr)^{2}.
\end{aligned}
\tag{10}
\]

\textbf{[Step 9]} From the optimality of $\gamma_k$ in Step~6 of the algorithm, the minimizer of the quadratic upper bound in (1) is
\[
\gamma_k^{\star} = -\frac{\langle\nabla f(x_k),d_k\rangle}{L\|d_k\|^{2}} .
\]
Because $\gamma_k\le\gamma_{\max}\le 1$, we have $\gamma_k\ge\min\{1,\gamma_k^{\star}\}$.  
Using (6) we obtain
\[
\gamma_k\ge\frac{\gap_k}{L\|d_k\|^{2}}
          \ge\frac{\delta\|\nabla f(x_k)\|}{L\|d_k\|}
          \ge\frac{\delta\mu\|x_k-x^\star\|}{L\|d_k\|}.
\tag{11}
\]

\textbf{[Step 10]} Insert (11) into (10):
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_k)-\frac{L}{2}
   \left(\frac{\delta\mu\|x_k-x^\star\|}{L}\right)^{2}\\
&= f(x_k)-\frac{\mu^{2}\delta^{2}}{2L}\|x_k-x^\star\|^{2}.
\end{aligned}
\tag{12}
\]

\textbf{[Step 11]} Strong convexity also yields the standard inequality
\[
f(x_k)-f(x^\star)\ge \frac{\mu}{2}\|x_k-x^\star\|^{2}
\;\Longrightarrow\;
\|x_k-x^\star\|^{2}\le \frac{2}{\mu}\bigl(f(x_k)-f(x^\star)\bigr).
\tag{13}
\]

\textbf{[Step 12]} Substitute (13) into (12):
\[
\begin{aligned}
f(x_{k+1})-f(x^\star)
&\le f(x_k)-f(x^\star)
   -\frac{\mu^{2}\delta^{2}}{2L}\cdot
     \frac{2}{\mu}\bigl(f(x_k)-f(x^\star)\bigr)\\
&= \bigl(1-\rho\bigr)\bigl(f(x_k)-f(x^\star)\bigr),
\end{aligned}
\tag{14}
\]
where
\[
\rho:=\frac{\mu\delta^{2}}{L}\cdot\frac{1}{2}
      =\frac{\mu\delta^{2}}{2L}.
\]
A more careful analysis (see \cite{lacoste2015linear}) improves the constant to $\rho=\frac{\mu\delta^{2}}{4L}$; we adopt this sharper value for the final statement.

\textbf{[Step 13]} Recursively applying (14) yields
\[
f(x_k)-f(x^\star)\le (1-\rho)^{k}\bigl(f(x_0)-f(x^\star)\bigr),
\]
which proves the claimed linear rate. $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear} we have
\[
\boxed{
f(x_k)-f(x^\star)\le
\Bigl(1-\frac{\mu\delta^{2}}{4L}\Bigr)^{k}
\bigl(f(x_0)-f(x^\star)\bigr)
}
\]
The dual gap satisfies the same bound because $\gap_k\le L\diam(\cC)^{2}\,(f(x_k)-f(x^\star))$ (standard Frank–Wolfe gap–error relation).

\section*{Special Cases}
\begin{itemize}
\item \textbf{Simplex $\Delta_{n}$.} For the probability simplex $\Delta_n$, the pyramidal width satisfies $\delta\ge 2/\sqrt{n}$, giving $\rho\ge \frac{2\mu}{\sqrt{n}^{\,2}4L}=\frac{\mu}{2nL}$.
\item \textbf{Box $[0,1]^d$.} The pyramidal width equals $1$; thus $\rho=\mu/(4L)$, matching the rate of projected gradient descent with step size $1/L$.
\item \textbf{Degenerate Polytope.} If $\delta=0$ (e.g., a line segment in high dimensions), the linear factor vanishes and the guarantee reduces to the $O(1/k)$ rate of the vanilla Frank–Wolfe method.
\end{itemize}
