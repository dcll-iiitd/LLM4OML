\section*{Algorithm Name}
Online Gradient Descent (OGD) for Strongly Convex Functions

\section*{Mathematical Formulation}
Online Gradient Descent is designed to sequentially update model parameters $w_t$ as new data arrives, minimizing the cumulative loss (regret) over time. 
\begin{itemize}
    \item \textbf{Update rule:} $w_{t+1} = \Pi_{\mathcal{W}} \left( w_t - \eta_t g_t \right)$
    \item \textbf{Gradient:} $g_t \in \partial f_t(w_t)$ is the subgradient of the online cost function $f_t$ evaluated at $w_t$.
    \item \textbf{Projection:} $\Pi_{\mathcal{W}}(\cdot)$ denotes the Euclidean projection onto a closed convex feasible set $\mathcal{W}$.
    \item \textbf{Learning rate schedule:} For $\mu$-strongly convex functions, the step size decays as $\eta_t = \frac{1}{\mu t}$.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Online Gradient Descent (OGD)}
\begin{algorithmic}[1]
\State \textbf{Input:} Learning rate parameter $\mu > 0$, number of iterations $T$, convex set $\mathcal{W}$
\State \textbf{Initialize:} $w_1 \in \mathcal{W}$ arbitrarily (e.g., $w_1 = \mathbf{0}$)
\For{$t = 1, 2, \dots, T$}
    \State Receive or observe function $f_t$
    \State Suffer loss $f_t(w_t)$
    \State Compute subgradient $g_t \in \partial f_t(w_t)$
    \State Set learning rate $\eta_t = \frac{1}{\mu t}$
    \State Update parameters: $w_{t+1} = \Pi_{\mathcal{W}}(w_t - \eta_t g_t)$
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let the objective function be fixed for all $t$ as $f_t(w) = (w-3)^2$. We initialize $w_1 = 0$ and use a constant learning rate $\eta = 0.1$ for simplicity over 3 iterations. No projection is needed as $\mathcal{W} = \mathbb{R}$.

\textbf{Iteration 1 ($t=1$):}
\begin{itemize}
    \item Current parameter: $w_1 = 0$
    \item Gradient computation: $g_1 = \nabla f_1(w_1) = 2(w_1 - 3) = 2(0 - 3) = -6$
    \item Parameter update: $w_2 = w_1 - \eta g_1 = 0 - 0.1(-6) = 0.6$
    \item Objective value suffered: $f_1(w_1) = (0 - 3)^2 = 9$
\end{itemize}

\textbf{Iteration 2 ($t=2$):}
\begin{itemize}
    \item Current parameter: $w_2 = 0.6$
    \item Gradient computation: $g_2 = \nabla f_2(w_2) = 2(0.6 - 3) = -4.8$
    \item Parameter update: $w_3 = w_2 - \eta g_2 = 0.6 - 0.1(-4.8) = 1.08$
    \item Objective value suffered: $f_2(w_2) = (0.6 - 3)^2 = 5.76$
\end{itemize}

\textbf{Iteration 3 ($t=3$):}
\begin{itemize}
    \item Current parameter: $w_3 = 1.08$
    \item Gradient computation: $g_3 = \nabla f_3(w_3) = 2(1.08 - 3) = -3.84$
    \item Parameter update: $w_4 = w_3 - \eta g_3 = 1.08 - 0.1(-3.84) = 1.464$
    \item Objective value suffered: $f_3(w_3) = (1.08 - 3)^2 \approx 3.6864$
\end{itemize}

\section*{Assumptions}
To prove the sublinear regret bound, we require the following formal assumptions:
\begin{enumerate}
    \item \textbf{Convex Domain:} The decision set $\mathcal{W} \subseteq \mathbb{R}^d$ is closed, convex, and non-empty.
    \item \textbf{Strong Convexity:} Every online function $f_t: \mathcal{W} \to \mathbb{R}$ is $\mu$-strongly convex. That is, for all $u, v \in \mathcal{W}$ and $g \in \partial f_t(v)$:
    $$f_t(u) \geq f_t(v) + \langle g, u - v \rangle + \frac{\mu}{2} \|u - v\|^2$$
    \item \textbf{Bounded Gradients:} The subgradients of $f_t$ are uniformly bounded over $\mathcal{W}$. There exists a constant $G > 0$ such that for all $t$ and all $w \in \mathcal{W}$:
    $$\|g_t\| \leq G, \quad \forall g_t \in \partial f_t(w)$$
\end{enumerate}

\section*{Main Convergence Theorem}
\textbf{Theorem (Regret Bound of Strongly Convex OGD):} Under Assumptions 1-3, if Online Gradient Descent is run with step sizes $\eta_t = \frac{1}{\mu t}$, then for any optimal comparator $w^* \in \mathcal{W}$, the cumulative regret $R_T$ after $T$ iterations is bounded by:
$$R_T = \sum_{t=1}^T f_t(w_t) - \sum_{t=1}^T f_t(w^*) \leq \frac{G^2}{2\mu} (1 + \log T)$$

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Logarithmic Regret:} The algorithm achieves $O(\log T)$ regret, which is exponentially better than the $O(\sqrt{T})$ regret achievable for general (non-strongly) convex functions. 
    \item \textbf{Hyperparameter Sensitivity:} The optimal theoretical learning rate requires exact knowledge of the strong convexity parameter $\mu$. Underestimating $\mu$ preserves convergence but worsens the constants, while overestimating $\mu$ can lead to divergence or linear regret.
    \item \textbf{Stability:} The strongly convex objective naturally stabilizes the iterates, meaning the distance to the optimum shrinks aggressively in early iterations, allowing for a fast decaying step size $\eta_t \propto 1/t$.
\end{itemize}

\section*{Discussion}
In the online learning paradigm, the environment can be adversarial, meaning the sequence of functions $f_t$ can be chosen arbitrarily. Despite this, OGD guarantees that the algorithm's performance will asymptotically match that of the single best fixed decision in hindsight $w^*$. The strong convexity assumption is critical for achieving the optimal $O(\log T)$ rate.

\section*{Preliminaries (Definitions and Lemmas)}
\textbf{Lemma 1 (Properties of Euclidean Projection):} For any closed convex set $\mathcal{W} \subset \mathbb{R}^d$, the projection operator $\Pi_{\mathcal{W}}(x) = \arg\min_{y \in \mathcal{W}} \|x - y\|$ is non-expansive. Specifically, for any $x \in \mathbb{R}^d$ and any $y \in \mathcal{W}$:
$$\|\Pi_{\mathcal{W}}(x) - y\|^2 \leq \|x - y\|^2$$

\section*{Main Proof (Step-by-Step Derivation)}
We wish to bound the instantaneous regret $f_t(w_t) - f_t(w^*)$ for any arbitrary fixed $w^* \in \mathcal{W}$.

\begin{itemize}
    \item \textbf{[Step 1]} By the definition of $\mu$-strong convexity applied to $f_t$ at points $w^*$ and $w_t$:
    $$f_t(w^*) \geq f_t(w_t) + \langle g_t, w^* - w_t \rangle + \frac{\mu}{2} \|w^* - w_t\|^2$$
    
    \item \textbf{[Step 2]} Rearranging the inequality to isolate the instantaneous regret $f_t(w_t) - f_t(w^*)$:
    $$f_t(w_t) - f_t(w^*) \leq \langle g_t, w_t - w^* \rangle - \frac{\mu}{2} \|w_t - w^*\|^2$$
    
    \item \textbf{[Step 3]} We analyze the distance between the unprojected next iterate and $w^*$. Expanding the squared Euclidean norm yields:
    $$\|w_t - \eta_t g_t - w^*\|^2 = \|w_t - w^*\|^2 - 2\eta_t \langle g_t, w_t - w^* \rangle + \eta_t^2 \|g_t\|^2$$
    
    \item \textbf{[Step 4]} Applying Lemma 1 (non-expansiveness of projection) since $w_{t+1} = \Pi_{\mathcal{W}}(w_t - \eta_t g_t)$ and $w^* \in \mathcal{W}$:
    $$\|w_{t+1} - w^*\|^2 \leq \|w_t - \eta_t g_t - w^*\|^2$$
    
    \item \textbf{[Step 5]} Combining [Step 3] and [Step 4] provides a bound on the distance of the next iterate:
    $$\|w_{t+1} - w^*\|^2 \leq \|w_t - w^*\|^2 - 2\eta_t \langle g_t, w_t - w^* \rangle + \eta_t^2 \|g_t\|^2$$
    
    \item \textbf{[Step 6]} Rearranging the inequality from [Step 5] to solve for the inner product $\langle g_t, w_t - w^* \rangle$:
    $$\langle g_t, w_t - w^* \rangle \leq \frac{\|w_t - w^*\|^2 - \|w_{t+1} - w^*\|^2}{2\eta_t} + \frac{\eta_t}{2} \|g_t\|^2$$
    
    \item \textbf{[Step 7]} Substituting the upper bound of the inner product from [Step 6] into the instantaneous regret bound from [Step 2]:
    $$f_t(w_t) - f_t(w^*) \leq \frac{\|w_t - w^*\|^2 - \|w_{t+1} - w^*\|^2}{2\eta_t} + \frac{\eta_t}{2} \|g_t\|^2 - \frac{\mu}{2} \|w_t - w^*\|^2$$
    
    \item \textbf{[Step 8]} Grouping the terms that share the common factor $\|w_t - w^*\|^2$:
    $$f_t(w_t) - f_t(w^*) \leq \left( \frac{1}{2\eta_t} - \frac{\mu}{2} \right) \|w_t - w^*\|^2 - \frac{1}{2\eta_t} \|w_{t+1} - w^*\|^2 + \frac{\eta_t}{2} \|g_t\|^2$$
    
    \item \textbf{[Step 9]} We substitute the specific learning rate $\eta_t = \frac{1}{\mu t}$ into the coefficient of $\|w_t - w^*\|^2$:
    $$\frac{1}{2\eta_t} - \frac{\mu}{2} = \frac{\mu t}{2} - \frac{\mu}{2} = \frac{\mu (t-1)}{2}$$
    
    \item \textbf{[Step 10]} We observe that $\frac{\mu (t-1)}{2} = \frac{1}{2\eta_{t-1}}$ for $t > 1$. For $t=1$, this coefficient is strictly $0$. We define $\frac{1}{\eta_0} = 0$ for notational convenience. The inequality becomes:
    $$f_t(w_t) - f_t(w^*) \leq \frac{1}{2\eta_{t-1}} \|w_t - w^*\|^2 - \frac{1}{2\eta_t} \|w_{t+1} - w^*\|^2 + \frac{\eta_t}{2} \|g_t\|^2$$
    
    \item \textbf{[Step 11]} We sum the instantaneous regret bounds over all iterations $t = 1$ to $T$:
    $$\sum_{t=1}^T (f_t(w_t) - f_t(w^*)) \leq \sum_{t=1}^T \left( \frac{1}{2\eta_{t-1}} \|w_t - w^*\|^2 - \frac{1}{2\eta_t} \|w_{t+1} - w^*\|^2 \right) + \sum_{t=1}^T \frac{\eta_t}{2} \|g_t\|^2$$
    
    \item \textbf{[Step 12]} The first summation on the right-hand side forms a telescoping series. Expanding it yields:
    $$\left( 0 - \frac{1}{2\eta_1} \|w_2 - w^*\|^2 \right) + \left( \frac{1}{2\eta_1} \|w_2 - w^*\|^2 - \frac{1}{2\eta_2} \|w_3 - w^*\|^2 \right) + \dots - \frac{1}{2\eta_T} \|w_{T+1} - w^*\|^2$$
    
    \item \textbf{[Step 13]} Canceling the intermediate terms in the telescoping sum leaves only the final negative term:
    $$- \frac{1}{2\eta_T} \|w_{T+1} - w^*\|^2$$
    
    \item \textbf{[Step 14]} Since $- \frac{1}{2\eta_T} \|w_{T+1} - w^*\|^2 \leq 0$, we can safely drop this term to establish an upper bound:
    $$\sum_{t=1}^T (f_t(w_t) - f_t(w^*)) \leq \sum_{t=1}^T \frac{\eta_t}{2} \|g_t\|^2$$
\end{itemize}

\section*{Rate Derivation (With Constants)}
\begin{itemize}
    \item \textbf{[Step 15]} Applying Assumption 3 ($\|g_t\| \leq G$) and substituting the learning rate $\eta_t = \frac{1}{\mu t}$ into the bound from [Step 14]:
    $$R_T = \sum_{t=1}^T (f_t(w_t) - f_t(w^*)) \leq \sum_{t=1}^T \frac{1}{2\mu t} G^2 = \frac{G^2}{2\mu} \sum_{t=1}^T \frac{1}{t}$$
    
    \item \textbf{[Step 16]} We bound the harmonic sum using the continuous integral test:
    $$\sum_{t=1}^T \frac{1}{t} \leq 1 + \int_1^T \frac{1}{x} dx = 1 + \log T$$
    
    \item \textbf{[Step 17]} Substituting the harmonic sum bound into [Step 15] provides the final deterministic regret bound:
    $$R_T \leq \frac{G^2}{2\mu} (1 + \log T)$$
\end{itemize}

\section*{Special Cases}
\subsection*{Online-to-Batch Conversion (Expected Risk Bound)}
If the online sequence is generated by sampling i.i.d. data $z_t \sim \mathcal{D}$ and letting $f_t(w) = \ell(w, z_t)$, we can convert the regret bound into an expected risk bound for the population objective $F(w) = \mathbb{E}_{z \sim \mathcal{D}}[\ell(w, z)]$.

\begin{itemize}
    \item \textbf{[Step 18]} Taking the full expectation over the randomness of the data sequence on the regret bound from [Step 17]:
    $$\sum_{t=1}^T \mathbb{E}[\ell(w_t, z_t) - \ell(w^*, z_t)] \leq \frac{G^2}{2\mu}(1 + \log T)$$
    
    \item \textbf{[Step 19]} Because $w_t$ depends only on the history $z_1, \dots, z_{t-1}$, it is independent of the current sample $z_t$. Thus, evaluating the conditional expectation gives:
    $$\mathbb{E}_{z_t}[\ell(w_t, z_t) \mid w_t] = F(w_t)$$
    
    \item \textbf{[Step 20]} Using the law of total expectation, we rewrite the sum in terms of the population risk:
    $$\mathbb{E}[\ell(w_t, z_t) - \ell(w^*, z_t)] = \mathbb{E}[F(w_t)] - F(w^*)$$
    
    \item \textbf{[Step 21]} We define the average iterate $\bar{w}_T = \frac{1}{T} \sum_{t=1}^T w_t$. Since $F$ is convex (as a sum of convex functions), Jensen's inequality implies:
    $$F(\bar{w}_T) - F(w^*) \leq \frac{1}{T} \sum_{t=1}^T (F(w_t) - F(w^*))$$
    
    \item \textbf{[Step 22]} Taking the expectation of [Step 21] and substituting the sum from [Step 20] yields the final expected convergence rate:
    $$\mathbb{E}[F(\bar{w}_T) - F(w^*)] \leq \frac{G^2 (1 + \log T)}{2\mu T} = O\left(\frac{\log T}{T}\right)$$
\end{itemize}
