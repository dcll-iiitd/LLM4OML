\section*{Algorithm Name}
Stochastic Variance Reduced Gradient (SVRG) for Non-Convex Functions

\section*{Mathematical Formulation}
SVRG is designed to minimize a finite-sum objective function:
\begin{equation}
    \min_{x \in \mathbb{R}^d} f(x) = \frac{1}{n} \sum_{i=1}^n f_i(x)
\end{equation}
The algorithm employs a double-loop structure to periodically compute a full gradient, which is then used to center the stochastic gradients in the inner loop, progressively reducing their variance.

\begin{itemize}
    \item \textbf{Outer Loop:} At epoch $s$, we maintain a snapshot weight $\tilde{x}_s$ and compute the exact full gradient $\tilde{\mu}_s = \nabla f(\tilde{x}_s) = \frac{1}{n}\sum_{i=1}^n \nabla f_i(\tilde{x}_s)$.
    \item \textbf{Inner Loop:} For $t = 0, \dots, m-1$, we sample an index $i_t \sim \text{Uniform}(1, \dots, n)$ and compute the variance-reduced stochastic gradient:
    \begin{equation}
        v_t^s = \nabla f_{i_t}(x_t^s) - \nabla f_{i_t}(\tilde{x}_s) + \tilde{\mu}_s
    \end{equation}
    \item \textbf{Update Rule:} $x_{t+1}^s = x_t^s - \eta v_t^s$
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Non-Convex SVRG}
\begin{algorithmic}[1]
\State \textbf{Input:} Initial weights $\tilde{x}_0$, learning rate $\eta$, epoch length $m$, number of epochs $S$
\For{$s = 0, 1, \dots, S-1$}
    \State Compute full gradient: $\tilde{\mu}_s = \frac{1}{n} \sum_{i=1}^n \nabla f_i(\tilde{x}_s)$
    \State Initialize inner loop: $x_0^s = \tilde{x}_s$
    \For{$t = 0, 1, \dots, m-1$}
        \State Sample index $i_t$ uniformly from $\{1, \dots, n\}$
        \State Compute variance-reduced gradient: $v_t^s = \nabla f_{i_t}(x_t^s) - \nabla f_{i_t}(\tilde{x}_s) + \tilde{\mu}_s$
        \State Update weights: $x_{t+1}^s = x_t^s - \eta v_t^s$
    \EndFor
    \State Update snapshot: $\tilde{x}_{s+1} = x_m^s$
\EndFor
\State \textbf{Output:} Iterate $x_a$ chosen uniformly at random from all $\{x_t^s\}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let $f(w) = (w-3)^2$. We treat this as a single-component function ($n=1$), meaning $f_1(w) = (w-3)^2$ and $\nabla f(w) = \nabla f_1(w) = 2(w-3)$. We use $w_0 = 0$, $\eta = 0.1$, and run 3 inner iterations ($m=3$).

\textbf{Initialization ($s=0$):}
\begin{align*}
    \tilde{w}_0 &= 0 \\
    \tilde{\mu}_0 &= \nabla f(\tilde{w}_0) = 2(0-3) = -6 \\
    w_0^0 &= \tilde{w}_0 = 0
\end{align*}

\textbf{Iteration $t=0$:}
\begin{align*}
    v_0^0 &= \nabla f_1(w_0^0) - \nabla f_1(\tilde{w}_0) + \tilde{\mu}_0 \\
          &= 2(0-3) - 2(0-3) + (-6) = -6 - (-6) - 6 = -6 \\
    w_1^0 &= w_0^0 - \eta v_0^0 = 0 - 0.1(-6) = 0.6 \\
    f(w_1^0) &= (0.6 - 3)^2 = (-2.4)^2 = 5.76
\end{align*}

\textbf{Iteration $t=1$:}
\begin{align*}
    v_1^0 &= \nabla f_1(w_1^0) - \nabla f_1(\tilde{w}_0) + \tilde{\mu}_0 \\
          &= 2(0.6-3) - 2(0-3) + (-6) = -4.8 - (-6) - 6 = -4.8 \\
    w_2^0 &= w_1^0 - \eta v_1^0 = 0.6 - 0.1(-4.8) = 1.08 \\
    f(w_2^0) &= (1.08 - 3)^2 = (-1.92)^2 = 3.6864
\end{align*}

\textbf{Iteration $t=2$:}
\begin{align*}
    v_2^0 &= \nabla f_1(w_2^0) - \nabla f_1(\tilde{w}_0) + \tilde{\mu}_0 \\
          &= 2(1.08-3) - (-6) + (-6) = -3.84 \\
    w_3^0 &= w_2^0 - \eta v_2^0 = 1.08 - 0.1(-3.84) = 1.464 \\
    f(w_3^0) &= (1.464 - 3)^2 = (-1.536)^2 \approx 2.3593
\end{align*}
\textit{Note: Because $n=1$, SVRG mathematically reduces to standard gradient descent, as the variance reduction terms perfectly cancel out the snapshot gradient.}

\section*{Assumptions}
\begin{enumerate}
    \item \textbf{Lower Boundedness:} The objective function $f(x)$ is bounded below, meaning there exists $f^*$ such that $f(x) \ge f^*$ for all $x$.
    \item \textbf{L-Smoothness:} Each individual function $f_i(x)$ is continuously differentiable and has $L$-Lipschitz continuous gradients:
    \begin{equation}
        \|\nabla f_i(x) - \nabla f_i(y)\| \le L \|x - y\| \quad \forall x, y \in \mathbb{R}^d, \forall i \in \{1, \dots, n\}
    \end{equation}
    This implies the global objective $f(x)$ is also $L$-smooth.
    \item \textbf{Unbiased Estimator:} The index $i_t$ is drawn uniformly at random from $\{1, \dots, n\}$, ensuring $\mathbb{E}[v_t^s | x_t^s] = \nabla f(x_t^s)$.
\end{enumerate}

\section*{Main Convergence Theorem}
\textbf{Theorem:} Suppose Assumptions 1-3 hold. Let $T = S \times m$ be the total number of inner iterations. If we choose the learning rate $\eta = \frac{1}{3Lm}$ and epoch length $m$, then the output $x_a$ chosen uniformly at random from all iterates $\{x_t^s\}$ satisfies:
\begin{equation}
    \mathbb{E}[\|\nabla f(x_a)\|^2] \le \frac{6Lm(f(\tilde{x}_0) - f^*)}{T} = \mathcal{O}\left(\frac{1}{T}\right)
\end{equation}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Variance Reduction:} Unlike SGD, where the variance of the gradient estimator remains strictly positive unless the learning rate decays to zero, SVRG's gradient variance approaches zero as $x_t^s \to x^*$, allowing for a constant learning rate.
    \item \textbf{Complexity:} To achieve an $\epsilon$-approximate stationary point ($\mathbb{E}[\|\nabla f(x)\|^2] \le \epsilon$), SVRG requires $\mathcal{O}(n + n^{2/3}\epsilon^{-1})$ gradient evaluations, which strictly dominates SGD's $\mathcal{O}(\epsilon^{-2})$ dependency for non-convex functions.
    \item \textbf{Hyperparameter Sensitivity:} The algorithm requires careful tuning of the epoch length $m$ (typically $\mathcal{O}(n)$) and learning rate $\eta$. If $\eta$ is too large, the Lyapunov sequence diverges.
\end{itemize}

\section*{Discussion}
SVRG beautifully bridges the gap between deterministic Gradient Descent (which has fast convergence but $\mathcal{O}(n)$ per-iteration cost) and Stochastic Gradient Descent (which has $\mathcal{O}(1)$ per-iteration cost but slow convergence due to variance). By utilizing a double-loop structure, SVRG anchors the stochastic updates to a reliable full gradient, eliminating the noise asymptotically. The proof for the non-convex case relies on a carefully constructed Lyapunov function that penalizes both the objective value and the distance of the inner iterates from the snapshot.

\section*{Preliminaries (Definitions and Lemmas)}
\textbf{Lemma 1 (Smoothness Bound):} For any $L$-smooth function $f$:
\begin{equation}
    f(y) \le f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2}\|y - x\|^2
\end{equation}

\textbf{Lemma 2 (Variance of SVRG Estimator):} 
Given $v_t^s = \nabla f_{i_t}(x_t^s) - \nabla f_{i_t}(\tilde{x}_s) + \nabla f(\tilde{x}_s)$, we have:
\begin{equation}
    \mathbb{E}[\|v_t^s - \nabla f(x_t^s)\|^2] \le L^2 \|x_t^s - \tilde{x}_s\|^2
\end{equation}

\section*{Main Proof (Step-by-Step Derivation)}
To simplify notation, we drop the epoch superscript $s$ and denote $\tilde{x}_s$ as $\tilde{x}$, $x_t^s$ as $x_t$, and $v_t^s$ as $v_t$. Let $\mathcal{F}_t$ be the filtration capturing all randomness up to iteration $t$.

\textbf{[Step 1]} By the $L$-smoothness of $f$ (Lemma 1) and the update rule $x_{t+1} = x_t - \eta v_t$, we expand the objective function at the next iteration:
\begin{equation} \tag{Step 1}
    f(x_{t+1}) \le f(x_t) - \eta \langle \nabla f(x_t), v_t \rangle + \frac{L\eta^2}{2} \|v_t\|^2
\end{equation}

\textbf{[Step 2]} We use the fundamental algebraic identity for inner products $\langle a, b \rangle = \frac{1}{2}(\|a\|^2 + \|b\|^2 - \|a - b\|^2)$ to rewrite the cross term $-\eta \langle \nabla f(x_t), v_t \rangle$:
\begin{equation} \tag{Step 2}
    -\eta \langle \nabla f(x_t), v_t \rangle = -\frac{\eta}{2} \|\nabla f(x_t)\|^2 - \frac{\eta}{2} \|v_t\|^2 + \frac{\eta}{2} \|v_t - \nabla f(x_t)\|^2
\end{equation}

\textbf{[Step 3]} Substitute the identity from Step 2 into the smoothness bound from Step 1 to decouple the gradient norm and the estimator norm:
\begin{align} 
    f(x_{t+1}) &\le f(x_t) - \frac{\eta}{2} \|\nabla f(x_t)\|^2 - \frac{\eta}{2} \|v_t\|^2 + \frac{\eta}{2} \|v_t - \nabla f(x_t)\|^2 + \frac{L\eta^2}{2} \|v_t\|^2 \nonumber \\
    &= f(x_t) - \frac{\eta}{2} \|\nabla f(x_t)\|^2 - \frac{\eta}{2}(1 - L\eta) \|v_t\|^2 + \frac{\eta}{2} \|v_t - \nabla f(x_t)\|^2 \tag{Step 3}
\end{align}

\textbf{[Step 4]} We derive the variance bound for the SVRG estimator. Notice that $v_t - \nabla f(x_t) = (\nabla f_{i_t}(x_t) - \nabla f_{i_t}(\tilde{x})) - (\nabla f(x_t) - \nabla f(\tilde{x}))$. This takes the form $X - \mathbb{E}[X]$. Since $\mathbb{E}[\|X - \mathbb{E}[X]\|^2] \le \mathbb{E}[\|X\|^2]$, we have:
\begin{align} 
    \mathbb{E}[\|v_t - \nabla f(x_t)\|^2 \mid \mathcal{F}_t] &\le \mathbb{E}_{i_t}[\|\nabla f_{i_t}(x_t) - \nabla f_{i_t}(\tilde{x})\|^2] \nonumber \\
    &\le L^2 \|x_t - \tilde{x}\|^2 \tag{Step 4}
\end{align}
where the last inequality uses the $L$-smoothness of individual components $f_i$.

\textbf{[Step 5]} Taking the expectation of the bound in Step 3 conditioned on $\mathcal{F}_t$, and substituting the variance bound from Step 4:
\begin{equation} \tag{Step 5}
    \mathbb{E}[f(x_{t+1}) \mid \mathcal{F}_t] \le f(x_t) - \frac{\eta}{2} \|\nabla f(x_t)\|^2 - \frac{\eta}{2}(1 - L\eta) \mathbb{E}[\|v_t\|^2 \mid \mathcal{F}_t] + \frac{\eta L^2}{2} \|x_t - \tilde{x}\|^2
\end{equation}

\textbf{[Step 6]} To handle the tracking distance $\|x_t - \tilde{x}\|^2$, we analyze its evolution. Expanding the squared norm for the next iteration:
\begin{align} 
    \mathbb{E}[\|x_{t+1} - \tilde{x}\|^2 \mid \mathcal{F}_t] &= \mathbb{E}[\|(x_t - \tilde{x}) - \eta v_t\|^2 \mid \mathcal{F}_t] \nonumber \\
    &= \|x_t - \tilde{x}\|^2 - 2\eta \langle x_t - \tilde{x}, \mathbb{E}[v_t \mid \mathcal{F}_t] \rangle + \eta^2 \mathbb{E}[\|v_t\|^2 \mid \mathcal{F}_t] \tag{Step 6}
\end{align}

\textbf{[Step 7]} Bounding the cross term $-2\eta \langle x_t - \tilde{x}, \nabla f(x_t) \rangle$ directly is difficult in non-convex settings. Instead, we apply Young's Inequality, $-2\langle a, b \rangle \le \beta \|a\|^2 + \frac{1}{\beta} \|b\|^2$ for any $\beta > 0$. Let $a = x_t - \tilde{x}$ and $b = \eta v_t$:
\begin{equation} \tag{Step 7}
    \mathbb{E}[\|x_{t+1} - \tilde{x}\|^2 \mid \mathcal{F}_t] \le (1 + \beta) \|x_t - \tilde{x}\|^2 + \eta^2 (1 + \beta^{-1}) \mathbb{E}[\|v_t\|^2 \mid \mathcal{F}_t]
\end{equation}

\textbf{[Step 8]} We define a Lyapunov function $R_t = \mathbb{E}[f(x_t) + c_t \|x_t - \tilde{x}\|^2]$ where $c_t > 0$ is a sequence to be determined. Taking the full expectation of $R_{t+1}$ using Steps 5 and 7:
\begin{align} 
    \mathbb{E}[R_{t+1}] &\le \mathbb{E}[f(x_t)] - \frac{\eta}{2} \mathbb{E}[\|\nabla f(x_t)\|^2] - \frac{\eta}{2}(1 - L\eta) \mathbb{E}[\|v_t\|^2] + \frac{\eta L^2}{2} \mathbb{E}[\|x_t - \tilde{x}\|^2] \nonumber \\
    &\quad + c_{t+1}(1 + \beta) \mathbb{E}[\|x_t - \tilde{x}\|^2] + c_{t+1}\eta^2 (1 + \beta^{-1}) \mathbb{E}[\|v_t\|^2] \tag{Step 8}
\end{align}

\textbf{[Step 9]} Grouping the expected terms by $\|x_t - \tilde{x}\|^2$ and $\|v_t\|^2$:
\begin{align} 
    \mathbb{E}[R_{t+1}] &\le \mathbb{E}[f(x_t)] + \left[ c_{t+1}(1 + \beta) + \frac{\eta L^2}{2} \right] \mathbb{E}[\|x_t - \tilde{x}\|^2] \nonumber \\
    &\quad - \frac{\eta}{2} \mathbb{E}[\|\nabla f(x_t)\|^2] - \left[ \frac{\eta}{2}(1 - L\eta) - c_{t+1}\eta^2(1 + \beta^{-1}) \right] \mathbb{E}[\|v_t\|^2] \tag{Step 9}
\end{align}

\textbf{[Step 10]} To form a telescoping sum with $R_t$, we set the coefficient of $\|x_t - \tilde{x}\|^2$ equal to $c_t$:
\begin{equation} \tag{Step 10}
    c_t = c_{t+1}(1 + \beta) + \frac{\eta L^2}{2}
\end{equation}
This substitution simplifies the Lyapunov inequality to:
\begin{equation*}
    \mathbb{E}[R_{t+1}] \le \mathbb{E}[R_t] - \frac{\eta}{2} \mathbb{E}[\|\nabla f(x_t)\|^2] - \left[ \frac{\eta}{2}(1 - L\eta) - c_{t+1}\eta^2(1 + \beta^{-1}) \right] \mathbb{E}[\|v_t\|^2]
\end{equation*}

\textbf{[Step 11]} We require the coefficient of $\mathbb{E}[\|v_t\|^2]$ to be non-negative (so its subtraction is non-positive). This imposes the condition:
\begin{equation} \tag{Step 11}
    c_{t+1} \le \frac{1 - L\eta}{2\eta(1 + \beta^{-1})}
\end{equation}
Since $c_t$ is a decreasing sequence, the maximum value is $c_0$. Thus, we must ensure $c_0 \le \frac{1 - L\eta}{2\eta(1 + \beta^{-1})}$.

\section*{Rate Derivation (With Constants)}

\textbf{[Step 12]} We solve the recurrence relation for $c_t$ with boundary condition $c_m = 0$.
\begin{equation*}
    c_t = \frac{\eta L^2}{2} \sum_{j=0}^{m-t-1} (1+\beta)^j = \frac{\eta L^2}{2} \frac{(1+\beta)^{m-t} - 1}{\beta}
\end{equation*}
Let $\beta = \frac{1}{m}$. Then $(1+\frac{1}{m})^m \le e$. The maximum value $c_0$ is bounded by:
\begin{equation*}
    c_0 = \frac{\eta L^2 m}{2} \left[ \left(1+\frac{1}{m}\right)^m - 1 \right] \le \frac{\eta L^2 m (e-1)}{2}
\end{equation*}
We must satisfy the condition from Step 11:
\begin{equation*}
    \frac{\eta L^2 m (e-1)}{2} \le \frac{1 - L\eta}{2\eta(m + 1)} \implies \eta^2 L^2 m(m+1)(e-1) \le 1 - L\eta
\end{equation*}
Choosing $\eta = \frac{1}{3Lm}$ guarantees $\eta L \le \frac{1}{3}$ (for $m \ge 1$). Substituting this into the inequality:
\begin{equation} \tag{Step 12}
    \frac{1}{9m^2} m(m+1)(e-1) \le \frac{2(e-1)}{9} \approx 0.38 \quad \text{and} \quad 1 - L\eta \ge 1 - \frac{1}{3} \approx 0.66
\end{equation}
Since $0.38 \le 0.66$, the step-size condition is strictly satisfied, meaning the $\|v_t\|^2$ term can be safely discarded.

\textbf{[Step 13]} Dropping the non-positive $\|v_t\|^2$ term and rearranging the Lyapunov inequality:
\begin{equation} \tag{Step 13}
    \frac{\eta}{2} \mathbb{E}[\|\nabla f(x_t^s)\|^2] \le \mathbb{E}[R_t^s] - \mathbb{E}[R_{t+1}^s]
\end{equation}
Summing this telescoping series over the inner loop $t = 0$ to $m-1$:
\begin{equation*}
    \frac{\eta}{2} \sum_{t=0}^{m-1} \mathbb{E}[\|\nabla f(x_t^s)\|^2] \le \mathbb{E}[R_0^s] - \mathbb{E}[R_m^s]
\end{equation*}
Note that $R_0^s = f(x_0^s) + c_0\|x_0^s - \tilde{x}_s\|^2 = f(\tilde{x}_s)$ because $x_0^s = \tilde{x}_s$. Also, $R_m^s = f(x_m^s) + c_m\|x_m^s - \tilde{x}_s\|^2 = f(x_m^s) = f(\tilde{x}_{s+1})$ because $c_m = 0$.

\textbf{[Step 14]} Summing over the outer loop epochs $s = 0$ to $S-1$:
\begin{equation*}
    \frac{\eta}{2} \sum_{s=0}^{S-1} \sum_{t=0}^{m-1} \mathbb{E}[\|\nabla f(x_t^s)\|^2] \le f(\tilde{x}_0) - \mathbb{E}[f(\tilde{x}_S)] \le f(\tilde{x}_0) - f^*
\end{equation*}
Dividing by the total number of inner iterations $T = Sm$, and substituting $\eta = \frac{1}{3Lm}$:
\begin{equation} \tag{Step 14}
    \frac{1}{T} \sum_{s=0}^{S-1} \sum_{t=0}^{m-1} \mathbb{E}[\|\nabla f(x_t^s)\|^2] \le \frac{2(f(\tilde{x}_0) - f^*)}{\eta T} = \frac{6Lm(f(\tilde{x}_0) - f^*)}{T} = \mathcal{O}\left(\frac{1}{T}\right)
\end{equation}
Since $x_a$ is chosen uniformly at random from all iterates, $\mathbb{E}[\|\nabla f(x_a)\|^2]$ is exactly the left-hand side, completing the proof.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Convex and Strongly Convex Settings:} If $f$ is strongly convex with parameter $\mu$, SVRG achieves a linear convergence rate $\mathcal{O}(e^{-\rho T})$. The proof relies on bounding the variance by the suboptimality $f(x) - f^*$ rather than the distance to the snapshot.
    \item \textbf{$m=1$ Limit:} If the inner loop length is 1, SVRG updates the snapshot at every step. The variance reduced gradient becomes $v_0 = \nabla f_{i_0}(x_0) - \nabla f_{i_0}(x_0) + \nabla f(x_0) = \nabla f(x_0)$. The algorithm degenerates into standard deterministic Full Gradient Descent, losing its computational advantage.
    \item \textbf{$m \to \infty$ Limit:} If the snapshot is never updated, the algorithm operates as SGD with a constant bias offset. The variance reduction mechanism fails because the snapshot $\tilde{x}$ becomes stale, and $\|x_t - \tilde{x}\|^2$ grows unbounded, violating the Lyapunov stability condition.
\end{itemize}
