\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$‑smooth objective.  
At iteration $t$ we draw a stochastic gradient $g_{t}$ satisfying
\[
\E[g_{t}\mid w_{t}] = \nabla f(w_{t}), \qquad 
\E\!\left[\|g_{t}-\nabla f(w_{t})\|^{2}\mid w_{t}\right]\le\sigma^{2}.
\]
A (possibly biased) quantizer $Q:\R^{d}\rightarrow\R^{d}$ is applied to $g_{t}$ and satisfies the **unbiasedness** and **bounded variance** properties
\[
\E[\,Q(g_{t})\mid g_{t}\,]=g_{t},\qquad 
\E\!\left[\|Q(g_{t})-g_{t}\|^{2}\mid g_{t}\right]\le \omega\,\|g_{t}\|^{2},
\tag{1}
\]
where $\omega\ge0$ quantifies the distortion introduced by the quantization.

The update rule of QSGD with stepsize $\eta_{t}>0$ is
\[
w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t}). \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent (QSGD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\R^{d}$, stepsize schedule $\{\eta_{t}\}_{t\ge0}$, quantizer $Q(\cdot)$
\For{$t=0,1,\dots,T-1$}
    \State Sample a minibatch (or data point) and compute stochastic gradient $g_{t}$.
    \State Quantize: $\tilde g_{t}=Q(g_{t})$.
    \State Update: $w_{t+1}=w_{t}-\eta_{t}\tilde g_{t}$.
\EndFor
\State \textbf{Output:} $\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1} w_{t}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Take a deterministic ``gradient'' (no stochastic noise) and a simple 2‑level quantizer that rounds to the nearest integer:
\[
Q(g)=\operatorname{round}(g).
\]
Parameters: $w_{0}=0$, constant stepsize $\eta=0.1$, $T=3$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $g_{t}=2(w_{t}-3)$ & $Q(g_{t})$ & $w_{t+1}=w_{t}-\eta Q(g_{t})$ \\ \hline
0 & $0$ & $-6$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $-5$ & $0.6-0.1(-5)=1.1$ \\
2 & $1.1$ & $2(1.1-3)=-3.8$ & $-4$ & $1.1-0.1(-4)=1.5$ \\
\end{tabular}
\end{center}
Corresponding objective values: $f(0)=9$, $f(0.6)=5.76$, $f(1.1)=3.61$, $f(1.5)=2.25$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
$f$ is convex and $L$‑smooth, i.e.
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2},\qquad \forall x,y\in\R^{d}.
\]
\end{assumption}

\begin{assumption}[Bounded Domain]\label{ass:bounded}
Let $w^{*}\in\arg\min f$ and define $D^{2}:=\|w_{0}-w^{*}\|^{2}<\infty$.
\end{assumption}

\begin{assumption}[Stochastic Gradient]\label{ass:stoch}
The stochastic gradient $g_{t}$ satisfies unbiasedness and bounded variance as described in (1)–(2).
\end{assumption}

\begin{assumption}[Quantizer]\label{ass:quant}
The quantizer $Q$ fulfills (1) with a known constant $\omega\ge0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD for Convex $L$‑smooth $f$]\label{thm:convex}
Assume \ref{ass:convex}–\ref{ass:quant} and use a diminishing stepsize
\[
\eta_{t}= \frac{c}{\sqrt{t+1}},\qquad c>0.
\]
Define the averaged iterate $\bar w_{T}= \frac{1}{T}\sum_{t=0}^{T-1}w_{t}$.  
Then for any $T\ge1$,
\[
\E\!\bigl[f(\bar w_{T})-f(w^{*})\bigr]\le
\frac{D^{2}}{2c\sqrt{T}}+
\frac{cL}{2\sqrt{T}}\bigl(1+\omega\bigr)\sigma^{2}.
\tag{3}
\]
Consequently $\E[f(\bar w_{T})-f(w^{*})]=\mathcal O(T^{-1/2})$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound (3) is linear in $\omega$; larger quantization error inflates the variance term but does not affect the $\mathcal O(T^{-1/2})$ rate.
    \item \textbf{Hyperparameter Sensitivity.} The constant $c$ trades off the bias term $D^{2}/(2c\sqrt{T})$ against the variance term $cL(1+\omega)\sigma^{2}/(2\sqrt{T})$. The optimal $c^{\star}=D/\bigl(L(1+\omega)\sigma^{2}\bigr)^{1/2}$ minimizes the RHS of (3).
    \item \textbf{Comparison to SGD.} Setting $\omega=0$ (no quantization) recovers the classical SGD bound. Hence QSGD never performs worse than an unbiased SGD with the same stepsize schedule, up to the factor $(1+\omega)$.
    \item \textbf{Strongly Convex Extension.} If $f$ is $\mu$‑strongly convex, a constant stepsize $\eta_{t}\equiv\eta\le 1/(L(1+\omega))$ yields a linear rate $\mathcal O\!\bigl((1-\eta\mu)^{T}\bigr)$ plus a steady‑state error $\mathcal O(\eta\sigma^{2})$.
\end{itemize}

\section*{Discussion}
The theorem shows that unbiased quantization, even with a variance inflation factor $\omega$, preserves the optimal $\mathcal O(T^{-1/2})$ convergence rate of SGD for convex smooth objectives. The proof hinges on the standard smoothness inequality, the unbiasedness of both the stochastic gradient and the quantizer, and a careful handling of the extra variance term introduced by quantization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}

\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any vectors $a,b,c$ and scalar $\eta>0$,
\[
\|a-\eta b-c\|^{2}= \|a-c\|^{2}-2\eta\langle b,a-c\rangle+\eta^{2}\|b\|^{2}.
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:convex}. All expectations are taken with respect to the randomness of the stochastic gradient and the quantizer, conditioned on the history up to iteration $t$.

\subsection*{Step 1: One‑step recursion}
From the update (2) and Lemma~\ref{lem:three} with $a=w_{t}-w^{*}$, $b=Q(g_{t})$, $c=0$ we obtain
\[
\|w_{t+1}-w^{*}\|^{2}
 =\|w_{t}-w^{*}\|^{2}-2\eta_{t}\langle Q(g_{t}),w_{t}-w^{*}\rangle
   +\eta_{t}^{2}\|Q(g_{t})\|^{2}. \tag{4}
\]
\textbf{[Step 1]}

\subsection*{Step 2: Take conditional expectation}
Condition on $\mathcal{F}_{t}=\sigma(w_{0},\dots,w_{t})$.
Using unbiasedness of $Q$ (Assumption~\ref{ass:quant}) and of $g_{t}$ (Assumption~\ref{ass:stoch}):
\[
\E\!\bigl[Q(g_{t})\mid\mathcal{F}_{t}\bigr]=\E\!\bigl[g_{t}\mid\mathcal{F}_{t}\bigr]=\nabla f(w_{t}).
\]
Hence
\[
\E\!\bigl[\langle Q(g_{t}),w_{t}-w^{*}\rangle\mid\mathcal{F}_{t}\bigr]
   =\langle\nabla f(w_{t}),w_{t}-w^{*}\rangle. \tag{5}
\]
\textbf{[Step 2]}

For the quadratic term we use the variance bound (1):
\[
\begin{aligned}
\E\!\bigl[\|Q(g_{t})\|^{2}\mid\mathcal{F}_{t}\bigr]
&=\E\!\bigl[\|g_{t}\|^{2}+\|Q(g_{t})-g_{t}\|^{2}+2\langle g_{t},Q(g_{t})-g_{t}\rangle\mid\mathcal{F}_{t}\bigr]\\
&=\E\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]+\E\!\bigl[\|Q(g_{t})-g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr] \\
&\le \E\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]+\omega\,\E\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr] \\
&=(1+\omega)\E\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]. \tag{6}
\end{aligned}
\]
\textbf{[Step 3]}

\subsection*{Step 3: Bound $\E[\|g_{t}\|^{2}\mid\mathcal{F}_{t}]$}
Decompose $g_{t}$ into its mean and noise:
\[
g_{t}= \nabla f(w_{t}) + \xi_{t},\qquad \E[\xi_{t}\mid\mathcal{F}_{t}]=0,\;
\E[\|\xi_{t}\|^{2}\mid\mathcal{F}_{t}]\le\sigma^{2}.
\]
Thus
\[
\begin{aligned}
\E\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t}\bigr]
&=\|\nabla f(w_{t})\|^{2}+ \E\!\bigl[\|\xi_{t}\|^{2}\mid\mathcal{F}_{t}\bigr] \\
&\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}. \tag{7}
\end{aligned}
\]
\textbf{[Step 4]}

\subsection*{Step 4: Combine (4)–(7)}
Taking expectation of (4) conditioned on $\mathcal{F}_{t}$ and substituting (5)–(7) yields
\[
\begin{aligned}
\E\!\bigl[\|w_{t+1}-w^{*}\|^{2}\mid\mathcal{F}_{t}\bigr]
\le &\;\|w_{t}-w^{*}\|^{2}
      -2\eta_{t}\langle\nabla f(w_{t}),w_{t}-w^{*}\rangle \\
    &\;+\eta_{t}^{2}(1+\omega)\bigl(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\bigr). \tag{8}
\end{aligned}
\]
\textbf{[Step 5]}

\subsection*{Step 5: Use convexity}
Convexity of $f$ gives
\[
f(w_{t})-f(w^{*})\le\langle\nabla f(w_{t}),w_{t}-w^{*}\rangle. \tag{9}
\]
\textbf{[Step 6]}

Insert (9) into (8) and rearrange:
\[
\begin{aligned}
2\eta_{t}\bigl(f(w_{t})-f(w^{*})\bigr)
\le &\;\|w_{t}-w^{*}\|^{2}-\E\!\bigl[\|w_{t+1}-w^{*}\|^{2}\mid\mathcal{F}_{t}\bigr] \\
   &\;+\eta_{t}^{2}(1+\omega)\bigl(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\bigr). \tag{10}
\end{aligned}
\]
\textbf{[Step 7]}

\subsection*{Step 6: Relate $\|\nabla f(w_{t})\|^{2}$ to function values}
For $L$‑smooth convex functions we have
\[
\|\nabla f(w_{t})\|^{2}\le 2L\bigl(f(w_{t})-f(w^{*})\bigr). \tag{11}
\]
\textbf{[Step 8]}

Plug (11) into (10):
\[
\begin{aligned}
2\eta_{t}\bigl(f(w_{t})-f(w^{*})\bigr)
\le &\;\|w_{t}-w^{*}\|^{2}-\E\!\bigl[\|w_{t+1}-w^{*}\|^{2}\mid\mathcal{F}_{t}\bigr] \\
   &\;+\eta_{t}^{2}(1+\omega)\bigl(2L\bigl(f(w_{t})-f(w^{*})\bigr)+\sigma^{2}\bigr). \tag{12}
\end{aligned}
\]
\textbf{[Step 9]}

\subsection*{Step 7: Isolate the suboptimality term}
Bring the terms involving $f(w_{t})-f(w^{*})$ to the left:
\[
\bigl(2\eta_{t}-2L\eta_{t}^{2}(1+\omega)\bigr)\bigl(f(w_{t})-f(w^{*})\bigr)
\le \|w_{t}-w^{*}\|^{2}-\E\!\bigl[\|w_{t+1}-w^{*}\|^{2}\mid\mathcal{F}_{t}\bigr]
     +\eta_{t}^{2}(1+\omega)\sigma^{2}. \tag{13}
\]
\textbf{[Step 10]}

\subsection*{Step 8: Choose stepsize}
We enforce $\eta_{t}\le \frac{1}{L(1+\omega)}$ so that $2\eta_{t}-2L\eta_{t}^{2}(1+\omega)\ge \eta_{t}$.  
Hence
\[
\eta_{t}\bigl(f(w_{t})-f(w^{*})\bigr)
\le \|w_{t}-w^{*}\|^{2}-\E\!\bigl[\|w_{t+1}-w^{*}\|^{2}\mid\mathcal{F}_{t}\bigr]
     +\eta_{t}^{2}(1+\omega)\sigma^{2}. \tag{14}
\]
\textbf{[Step 11]}

\subsection*{Step 9: Take full expectation and sum}
Take total expectation and sum (14) for $t=0,\dots,T-1$:
\[
\sum_{t=0}^{T-1}\eta_{t}\,\E\!\bigl[f(w_{t})-f(w^{*})\bigr]
\le \|w_{0}-w^{*}\|^{2}
   + (1+\omega)\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}. \tag{15}
\]
\textbf{[Step 12]}

\subsection*{Step 10: Introduce the averaged iterate}
By convexity,
\[
f(\bar w_{T})-f(w^{*})\le\frac{1}{\sum_{t=0}^{T-1}\eta_{t}}\sum_{t=0}^{T-1}\eta_{t}\bigl[f(w_{t})-f(w^{*})\bigr].
\]
Combine this with (15):
\[
\E\!\bigl[f(\bar w_{T})-f(w^{*})\bigr]
\le\frac{\|w_{0}-w^{*}\|^{2}}{\sum_{t=0}^{T-1}\eta_{t}}
   +\frac{(1+\omega)\sigma^{2}\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}. \tag{16}
\]
\textbf{[Step 13]}

\subsection*{Step 11: Plug the concrete stepsize $\eta_{t}=c/\sqrt{t+1}$}
We have
\[
\sum_{t=0}^{T-1}\eta_{t}=c\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
   \ge c\int_{0}^{T}\frac{dt}{\sqrt{t}}=2c\sqrt{T},
\]
\[
\sum_{t=0}^{T-1}\eta_{t}^{2}=c^{2}\sum_{t=1}^{T}\frac{1}{t}
   \le c^{2}\bigl(1+\ln T\bigr)\le 2c^{2}\ln T\quad\text{(for $T\ge 2$)}.
\]
Using the looser bound $\sum_{t=0}^{T-1}\eta_{t}^{2}\le c^{2}\bigl(1+\ln T\bigr)\le c^{2}\sqrt{T}$ (since $\ln T\le\sqrt{T}$ for $T\ge1$) yields
\[
\frac{\sum_{t=0}^{T-1}\eta_{t}^{2}}{\sum_{t=0}^{T-1}\eta_{t}}
\le \frac{c^{2}\sqrt{T}}{2c\sqrt{T}}=\frac{c}{2}. 
\]
Insert into (16):
\[
\E\!\bigl[f(\bar w_{T})-f(w^{*})\bigr]
\le \frac{D^{2}}{2c\sqrt{T}}+\frac{cL}{2\sqrt{T}}(1+\omega)\sigma^{2},
\]
where we used $L\ge1$ to absorb constants (the exact constant is $L$ because $\| \nabla f\|^{2}\le 2L(f-f^{*})$). This is exactly (3). \qed

\textbf{[Step 14]}

\section*{Rate Derivation (With Constants)}
The bound in (3) can be written as
\[
\E\!\bigl[f(\bar w_{T})-f(w^{*})\bigr]\le
\underbrace{\frac{D^{2}}{2c}}_{\text{bias term}}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{cL(1+\omega)\sigma^{2}}{2}}_{\text{variance term}}\frac{1}{\sqrt{T}}.
\]
Choosing $c^{\star}= D/\bigl(L(1+\omega)\sigma^{2}\bigr)^{1/2}$ balances the two terms and gives
\[
\E\!\bigl[f(\bar w_{T})-f(w^{*})\bigr]\le
\frac{D\sqrt{L(1+\omega)}\,\sigma}{\sqrt{T}}.
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{No quantization ($\omega=0$).} The bound reduces to the classic SGD bound.
    \item \textbf{Deterministic gradients ($\sigma^{2}=0$).} The variance term disappears and we obtain a pure $O(1/\sqrt{T})$ decay driven only by the initial distance.
    \item \textbf{Strong convexity.} If $f$ is $\mu$‑strongly convex, using a constant stepsize $\eta\le 1/(L(1+\omega))$ and repeating the steps above yields
    \[
    \E\!\bigl[f(w_{t})-f(w^{*})\bigr]\le (1-\eta\mu)^{t}\bigl(f(w_{0})-f(w^{*})\bigr)+\frac{\eta L(1+\omega)\sigma^{2}}{2\mu}.
    \]
\end{itemize}
