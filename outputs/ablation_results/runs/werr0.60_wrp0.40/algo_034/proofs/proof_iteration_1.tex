\section*{Clipped Stochastic Gradient Descent (Clipped SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb R^d\rightarrow\mathbb R$ be the objective function.  
At iteration $t$ a stochastic gradient $g_t$ is drawn such that  

\[
\EE[g_t\mid x_t]=\nabla f(x_t),\qquad
\EE\big[\|g_t-\nabla f(x_t)\|^2\mid x_t\big]\le\sigma^2 .
\]

Define the \emph{clipping operator} with clipping threshold $C>0$  

\[
\tilde g_t\;:=\;\clip{g_t}{C}
        =\frac{g_t}{\max\!\Big\{1,\frac{\norm{g_t}}{C}\Big\}}
        =\begin{cases}
            g_t, & \norm{g_t}\le C,\\[4pt]
            \displaystyle C\frac{g_t}{\norm{g_t}}, & \norm{g_t}> C .
          \end{cases}
\tag{1}
\]

The update rule of Clipped SGD is  

\[
x_{t+1}=x_t-\eta_t \tilde g_t ,\qquad t=0,1,\dots,T-1,
\tag{2}
\]

where $\{\eta_t\}$ is a stepsize schedule (e.g. $\eta_t=\eta/\sqrt{t+1}$).  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped SGD}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\mathbb R^d$, stepsize sequence $\{\eta_t\}_{t\ge0}$, clipping threshold $C>0$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a stochastic gradient $g_t$ (e.g. on a minibatch)
    \State $\displaystyle \tilde g_t \gets \frac{g_t}{\max\{1,\;\norm{g_t}/C\}}$ \Comment{clipping}
    \State $x_{t+1}\gets x_t-\eta_t \tilde g_t$
\EndFor
\State \Return $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic $f(w)=(w-3)^2$.
\[
\nabla f(w)=2(w-3), \qquad L=2 .
\]

Parameters: $w_0=0$, $\eta=0.1$, clipping threshold $C=1$,
deterministic gradients (so $g_t=\nabla f(w_t)$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=\nabla f(w_t)$ & $\tilde g_t$ (clipped) & $f(w_t)$\\ \hline
0 & $0$   & $-6$   & $C\frac{-6}{6}=-1$ & $9$\\
1 & $0-0.1(-1)=0.1$ & $-5.8$ & $-1$ & $8.41$\\
2 & $0.1-0.1(-1)=0.2$ & $-5.6$ & $-1$ & $7.84$\\
3 & $0.2-0.1(-1)=0.3$ & $-5.4$ & $-1$ & $7.29$
\end{tabular}
\end{center}
The clipping caps the update magnitude to $0.1$, preventing the large initial step that would occur without clipping.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$–smooth, i.e.
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2,
\qquad\forall x,y\in\mathbb R^d .
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:var}
For all $t$, the stochastic gradient satisfies
\[
\EE[g_t\mid x_t]=\nabla f(x_t),\qquad 
\EE\!\left[\|g_t-\nabla f(x_t)\|^2\mid x_t\right]\le\sigma^2 .
\]
\end{assumption}
\begin{assumption}[Clipping Threshold]\label{ass:clip}
The clipping constant $C$ is a positive scalar. No further restriction is needed; however,
the bias induced by clipping can be bounded as
\[
\big\|\EE[\tilde g_t\mid x_t]-\nabla f(x_t)\big\|
   \le \max\{0,\;\norm{\nabla f(x_t)}-C\}.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
We use a diminishing stepsize $\eta_t=\frac{\eta}{\sqrt{t+1}}$ with $\eta>0$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}--\ref{ass:stepsize}. Let $x_*$ be any point satisfying 
$\nabla f(x_*)=0$ (a stationary point). Define $G:=\max\{C,\sigma\}$.
Then for $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
   \EE\!\big[\|\nabla f(x_t)\|^2\big]
   \;\le\;
   \frac{2\big(f(x_0)-f(x_*)\big)}{\eta\sqrt{T}}
   \;+\;
   \frac{2L\eta G^2}{\sqrt{T}}
   \;+\;
   \frac{2}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|-C\big]_+^{2},
\tag{4}
\]
where $[a]_+:=\max\{a,0\}$. Consequently, if the iterates eventually satisfy $\|\nabla f(x_t)\|\le C$, the last term vanishes and we obtain the standard $O(1/\sqrt{T})$ rate for non‑convex SGD.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\|\tilde g_t\|\le C$ by construction, the update (2) never moves more than $\eta_t C$ in any iteration, guaranteeing that the iterates remain in a bounded region provided the stepsizes are summable.
\item \textbf{Hyperparameter Sensitivity.} The bound (4) shows a direct trade‑off: a larger $C$ reduces the clipping bias term but increases the magnitude of the stochastic noise term $L\eta G^2/\sqrt{T}$. Choosing $C\approx\sqrt{L}\,\eta$ often yields a balanced bound.
\item \textbf{Comparison to vanilla SGD.} Vanilla SGD enjoys the same $O(1/\sqrt{T})$ bound under identical assumptions, but without the extra bias term. When gradients are occasionally large (e.g. heavy‑tailed noise), clipping can dramatically reduce $\sigma^2$ and improve practical performance, at the cost of the bias term in (4).
\end{itemize}

\section*{Discussion}
The theorem demonstrates that clipping does not destroy the canonical $1/\sqrt{T}$ convergence rate for non‑convex smooth objectives; the price paid is the additive bias term, which disappears once the norm of true gradients falls below the threshold $C$. This aligns with empirical observations that gradient clipping stabilises training of deep networks while preserving convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{lemma}[Bias bound of clipping]\label{lem:bias}
For any random vector $u$ and any $C>0$,
\[
\big\|\EE\!\big[\clip{u}{C}\big]-\EE[u]\big\|
   \le \EE\big[\,\big(\|u\|-C\big)_+\,\big].
\]
\end{lemma}
\begin{proof}
If $\|u\|\le C$, the clipping operator is the identity and contributes zero to the difference.
If $\|u\|>C$, $\clip{u}{C}=C\,u/\|u\|$, therefore
\[
\big\|\clip{u}{C}-u\big\| = \|u\|-C .
\]
Taking expectation and using Jensen’s inequality yields the claimed bound. \(\square\)
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main} by expanding the smoothness inequality and carefully handling the clipping bias.

\noindent\textbf{Step 1 (Smoothness expansion).}  
Apply Lemma~\ref{lem:smooth} with $x=x_t$, $y=x_{t+1}=x_t-\eta_t\tilde g_t$:
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\nabla f(x_t),\tilde g_t\rangle
          +\frac{L\eta_t^{2}}{2}\|\tilde g_t\|^{2}.
\tag{5}
\]

\noindent\textbf{Step 2 (Insert expectation).}  
Take conditional expectation w.r.t.\ $\mathcal F_t:=\sigma(x_0,\dots,x_t)$:
\[
\EE\!\big[f(x_{t+1})\mid\mathcal F_t\big]
\le f(x_t)-\eta_t\EE\!\big[\langle\nabla f(x_t),\tilde g_t\rangle\mid\mathcal F_t\big]
      +\frac{L\eta_t^{2}}{2}\EE\!\big[\|\tilde g_t\|^{2}\mid\mathcal F_t\big].
\tag{6}
\]

\noindent\textbf{Step 3 (Decompose the inner product).}  
Write $\tilde g_t = \EE[\tilde g_t\mid\mathcal F_t] + (\tilde g_t-\EE[\tilde g_t\mid\mathcal F_t])$.
Since $\EE[\tilde g_t-\EE[\tilde g_t\mid\mathcal F_t]\mid\mathcal F_t]=0$, we obtain
\[
\EE\!\big[\langle\nabla f(x_t),\tilde g_t\rangle\mid\mathcal F_t\big]
   =\Big\langle\nabla f(x_t),\EE[\tilde g_t\mid\mathcal F_t]\Big\rangle .
\tag{7}
\]

\noindent\textbf{Step 4 (Introduce the bias).}  
Define the clipping bias
\[
b_t:=\EE[\tilde g_t\mid\mathcal F_t]-\nabla f(x_t).
\tag{8}
\]
Then (7) becomes
\[
\EE\!\big[\langle\nabla f(x_t),\tilde g_t\rangle\mid\mathcal F_t\big]
   =\|\nabla f(x_t)\|^{2}+\langle\nabla f(x_t),b_t\rangle .
\tag{9}
\]

\noindent\textbf{Step 5 (Bound the bias term).}  
By Cauchy–Schwarz and Lemma~\ref{lem:bias},
\[
\langle\nabla f(x_t),b_t\rangle
   \ge -\|\nabla f(x_t)\|\,\|b_t\|
   \ge -\|\nabla f(x_t)\|\,\EE\!\big[(\|g_t\|-C)_+\mid\mathcal F_t\big].
\tag{10}
\]
Moreover, because $\|g_t\|\le\|\nabla f(x_t)\|+ \|g_t-\nabla f(x_t)\|$, we have
\[
(\|g_t\|-C)_+ \le (\|\nabla f(x_t)\|-C)_+ + \|g_t-\nabla f(x_t)\|.
\tag{11}
\]
Taking expectation and using $\EE[\|g_t-\nabla f(x_t)\|\mid\mathcal F_t]\le\sigma$ yields
\[
\EE\!\big[(\|g_t\|-C)_+\mid\mathcal F_t\big]
   \le (\|\nabla f(x_t)\|-C)_+ + \sigma .
\tag{12}
\]

\noindent\textbf{Step 6 (Bound the second moment of the clipped gradient).}  
Since $\|\tilde g_t\|\le C$ by construction,
\[
\EE\!\big[\|\tilde g_t\|^{2}\mid\mathcal F_t\big]\le C^{2}.
\tag{13}
\]

\noindent\textbf{Step 7 (Plug bounds into (6)).}  
Using (9)–(13) in (6) gives
\[
\EE\!\big[f(x_{t+1})\mid\mathcal F_t\big]
\le f(x_t)-\eta_t\|\nabla f(x_t)\|^{2}
     +\eta_t\|\nabla f(x_t)\|\,\big[(\|\nabla f(x_t)\|-C)_+ +\sigma\big]
     +\frac{L\eta_t^{2}}{2}C^{2}.
\tag{14}
\]

\noindent\textbf{Step 8 (Simplify the mixed term).}  
Apply $ab\le \frac{a^{2}}{2}+ \frac{b^{2}}{2}$ with  
$a=\|\nabla f(x_t)\|$, $b=(\|\nabla f(x_t)\|-C)_+ +\sigma$:
\[
\eta_t\|\nabla f(x_t)\|\,\big[(\|\nabla f(x_t)\|-C)_+ +\sigma\big]
\le \frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
   +\frac{\eta_t}{2}\big[(\|\nabla f(x_t)\|-C)_+ +\sigma\big]^{2}.
\tag{15}
\]

\noindent\textbf{Step 9 (Collect terms).}  
Insert (15) into (14):
\[
\EE\!\big[f(x_{t+1})\mid\mathcal F_t\big]
\le f(x_t)-\frac{\eta_t}{2}\|\nabla f(x_t)\|^{2}
   +\frac{\eta_t}{2}\big[(\|\nabla f(x_t)\|-C)_+ +\sigma\big]^{2}
   +\frac{L\eta_t^{2}}{2}C^{2}.
\tag{16}
\]

\noindent\textbf{Step 10 (Unconditional expectation).}  
Take total expectation and sum from $t=0$ to $T-1$:
\[
\EE[f(x_T)]\le f(x_0)-\frac{1}{2}\sum_{t=0}^{T-1}\eta_t
                 \EE\!\big[\|\nabla f(x_t)\|^{2}\big]
   +\frac{1}{2}\sum_{t=0}^{T-1}\eta_t
                 \EE\!\big[(\|\nabla f(x_t)\|-C)_+^{2}\big]
   +\frac{L C^{2}}{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{17}
\]

\noindent\textbf{Step 11 (Rearrange).}  
Since $f(x_T)\ge f(x_*)$, move the term $f(x_*)$ to the left and divide by $\sum_{t=0}^{T-1}\eta_t$:
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}
   \sum_{t=0}^{T-1}\eta_t\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
\le
\frac{2\big(f(x_0)-f(x_*)\big)}{\sum_{t=0}^{T-1}\eta_t}
 +\frac{1}{\sum_{t=0}^{T-1}\eta_t}
   \sum_{t=0}^{T-1}\eta_t\EE\!\big[(\|\nabla f(x_t)\|-C)_+^{2}\big]
 +\frac{L C^{2}\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{18}
\]

\noindent\textbf{Step 12 (Insert the stepsize schedule).}  
For $\eta_t=\eta/\sqrt{t+1}$,
\[
\sum_{t=0}^{T-1}\eta_t
   =\eta\sum_{t=1}^{T}\frac{1}{\sqrt{t}}
   \ge \eta\int_{1}^{T}\frac{dt}{\sqrt{t}}
   =2\eta(\sqrt{T}-1)
   \ge \eta\sqrt{T},
\tag{19}
\]
and
\[
\sum_{t=0}^{T-1}\eta_t^{2}
   =\eta^{2}\sum_{t=1}^{T}\frac{1}{t}
   \le \eta^{2}(1+\log T).
\tag{20}
\]

\noindent\textbf{Step 13 (Final bound).}  
Using (19)–(20) in (18) and noting that $C\le G:=\max\{C,\sigma\}$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
\le
\frac{2\big(f(x_0)-f(x_*)\big)}{\eta\sqrt{T}}
   +\frac{2L\eta G^{2}}{\sqrt{T}}
   +\frac{2}{T}\sum_{t=0}^{T-1}
        \EE\!\big[(\|\nabla f(x_t)\|-C)_+^{2}\big],
\]
which is exactly the statement of Theorem~\ref{thm:main}. \(\square\)

\section*{Rate Derivation (With Constants)}
Define $D:=f(x_0)-f(x_*)$. Choosing $\eta = \frac{D^{1/2}}{L^{1/2}G^{1/2}}$ balances the first two terms in (4) and yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\big[\|\nabla f(x_t)\|^{2}\big]
\;\le\;
\frac{4L^{1/2}G^{1/2}D^{1/2}}{\sqrt{T}}
   +\frac{2}{T}\sum_{t=0}^{T-1}
        \EE\!\big[(\|\nabla f(x_t)\|-C)_+^{2}\big].
\]
If after a finite burn‑in period $\|\nabla f(x_t)\|\le C$, the bias term disappears and we obtain the clean $O(1/\sqrt{T})$ rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Convex $f$.} The same derivation applies; by convexity one can replace $\|\nabla f(x_t)\|^{2}$ with $2L\big(f(x_t)-f(x_*)\big)$ and recover the $O(1/\sqrt{T})$ function‑value rate.
\item \textbf{Deterministic gradients ($\sigma=0$).} The variance term vanishes, and the bound simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(x_t)\|^{2}
\le
\frac{2D}{\eta\sqrt{T}}
   +\frac{2L\eta C^{2}}{\sqrt{T}}
   +\frac{2}{T}\sum_{t=0}^{T-1}(\|\nabla f(x_t)\|-C)_+^{2}.
\]
\item \textbf{Large clipping threshold $C\ge \max_t\|\nabla f(x_t)\|$.}
The bias term is identically zero, and Clipped SGD reduces to standard SGD with the same convergence guarantee.
\end{itemize}
