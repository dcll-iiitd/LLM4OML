\section*{Variance‑Adapted Gradient Descent (VAGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the objective function.  
The algorithm maintains an exponential moving average of the (full) gradients
\[
\boxed{v_{t}= (1-\alpha)\,v_{t-1}+ \alpha\,\grad f(x_{t})},\qquad \alpha\in(0,1].
\tag{1}
\]
Because $v_{t}$ is biased toward zero at the early iterations, a bias‑corrected
estimate is used
\[
\boxed{\hat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t}}}.
\tag{2}
\]
The step size at iteration $t$ is chosen adaptively as a decreasing function of
the magnitude of $\hat v_{t}$:
\[
\boxed{\eta_{t}= \frac{\eta}{\,1+\norm{\hat v_{t}}\,}},\qquad \eta>0\text{ a base learning rate.}
\tag{3}
\]
Finally the iterate is updated by a (biased) gradient step
\[
\boxed{x_{t+1}=x_{t}-\eta_{t}\,\hat v_{t}}.
\tag{4}
\]
All hyper‑parameters are $\{\alpha,\eta\}$; $v_{0}$ is set to the zero vector.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Variance‑Adapted Gradient Descent (VAGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_{0}\in\mathbb{R}^{d}$, base step size $\eta>0$, smoothing factor $\alpha\in(0,1]$, maximal iterations $T$.
\STATE $v_{0}\gets 0$, $t\gets 0$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute the (full) gradient $g_{t}\gets \grad f(x_{t})$.
    \STATE Update exponential moving average: $v_{t+1}\gets (1-\alpha)v_{t}+ \alpha g_{t}$.
    \STATE Bias correction: $\hat v_{t+1}\gets v_{t+1}/\bigl(1-(1-\alpha)^{\,t+1}\bigr)$.
    \STATE Adaptive step size: $\eta_{t+1}\gets \displaystyle\frac{\eta}{1+\norm{\hat v_{t+1}}}$.
    \STATE Gradient step: $x_{t+1}\gets x_{t}-\eta_{t+1}\,\hat v_{t+1}$.
\ENDFOR
\STATE \textbf{Output:} $x_{T}$ (or the best iterate w.r.t.\ $f$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional quadratic 
\[
f(w)=(w-3)^{2},
\qquad \grad f(w)=2(w-3).
\]
Choose $x_{0}=0$, base step size $\eta=0.1$, smoothing factor $\alpha=0.5$.
All vectors reduce to scalars; we denote $v_{t},\hat v_{t},\eta_{t},x_{t}$ by the same symbols.

\begin{center}
\begin{tabular}{c|c|c|c|c|c}
$t$ & $x_{t}$ & $\grad f(x_{t})$ & $v_{t}$ & $\hat v_{t}$ & $f(x_{t})$\\\hline
0 & $0$ & $-6$ & $0$ & $0$ & $9$\\
1 & $0.0857$ & $-5.8286$ & $-3.0$ & $-6.0$ & $8.516$\\
2 & $0.1712$ & $-5.6576$ & $-4.4143$ & $-5.8857$ & $8.049$\\
3 & $0.2559$ & $-5.4882$ & $-4.9519$ & $-5.8422$ & $7.597$\\
\end{tabular}
\end{center}
The numbers are obtained as follows (illustrated for the first two iterations).

\begin{enumerate}[label=\textbf{Iter \arabic*:}, leftmargin=*]
\item \textbf{Iteration 1}
\[
\begin{aligned}
g_{0}&=2(0-3)=-6,\\
v_{1}&=(1-0.5)0+0.5(-6)=-3,\\
\hat v_{1}&=\frac{-3}{1-(0.5)^{1}}=-6,\\
\eta_{1}&=\frac{0.1}{1+|{-6}|}= \frac{0.1}{7}=0.0142857,\\
x_{1}&=0-0.0142857(-6)=0.085714.
\end{aligned}
\]
\item \textbf{Iteration 2}
\[
\begin{aligned}
g_{1}&=2(0.085714-3)=-5.828572,\\
v_{2}&=0.5(-3)+0.5(-5.828572)=-4.414286,\\
\hat v_{2}&=\frac{-4.414286}{1-(0.5)^{2}}=
\frac{-4.414286}{0.75}=-5.885714,\\
\eta_{2}&=\frac{0.1}{1+5.885714}=0.014523,\\
x_{2}&=0.085714-0.014523(-5.885714)=0.171216.
\end{aligned}
\]
\end{enumerate}
The objective values $f(x_{t})$ decrease monotonically, illustrating the
descent property of VAGD.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^{d}:\quad
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that for all $x$,
\[
\frac{1}{2}\norm{\grad f(x)}^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\qquad f^{\star}\triangleq\inf_{x}f(x).
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Bounded initial variance]\label{ass:bound}
The initial gradient satisfies $\norm{\grad f(x_{0})}\le G$ for some $G>0$.
\tag{A3}
\end{assumption}
All constants $L,\mu,G$ are known a priori; the hyper‑parameters will be
chosen as functions of them.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of VAGD]\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:bound} hold and choose the base step size
\[
\boxed{\eta\le \frac{\mu}{L^{2}}}\quad\text{and}\quad
\boxed{\alpha\in(0,1]}.
\tag{5}
\]
Define the contraction factor
\[
\boxed{\rho\triangleq 1-\frac{\mu\eta}{2(1+\tfrac{G}{\mu})}\in(0,1).}
\tag{6}
\]
Then for the iterates generated by VAGD,
\[
\EE\!\bigl[f(x_{t})-f^{\star}\bigr]\le \rho^{\,t}\,\bigl(f(x_{0})-f^{\star}\bigr),
\qquad\forall\,t\ge0.
\tag{7}
\]
Consequently, to achieve $\EE[f(x_{t})-f^{\star}]\le\varepsilon$ it suffices that
\[
t\ge \frac{\log\!\bigl((f(x_{0})-f^{\star})/\varepsilon\bigr)}{\log(1/\rho)}=
\mathcal{O}\!\bigl(\tfrac{1}{\mu\eta}\log\tfrac{1}{\varepsilon}\bigr).
\tag{8}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} Because the step size $\eta_{t}$ is inversely
proportional to $1+\norm{\hat v_{t}}$, the update never explodes even if an
intermediate gradient is large; the bound $\eta_{t}\le\eta$ holds for all $t$.
\item \textbf{Hyper‑parameter sensitivity.}
The contraction factor $\rho$ depends on $\eta$ and the ratio $G/\mu$ only.
Increasing $\alpha$ (more weight on the current gradient) reduces the bias of
$\hat v_{t}$ but does not affect the bound \eqref{thm:linear} as long as
$\alpha\in(0,1]$.
\item \textbf{Comparison to baselines.}
For a plain gradient descent with fixed step size $\eta$ under the same PL and
smoothness assumptions the optimal linear rate is $1-\mu\eta$.  VAGD attains a
rate $1-\frac{\mu\eta}{2(1+G/\mu)}$, i.e. it is within a constant factor of the
optimal one while automatically adapting to the magnitude of the gradients.
\end{enumerate}

\section*{Discussion}
The theorem shows that VAGD inherits the fast linear convergence typical of
gradient descent on PL functions, yet it enjoys an adaptive step size that
mitigates the need for manual tuning.  The bias‑correction (2) guarantees that
the moving average $v_{t}$ converges to the true gradient in expectation,
which is crucial for the PL‑based contraction argument.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any vector $d$,
\[
f(x-d)\le f(x)-\inner{\grad f(x)}{d}+\frac{L}{2}\norm{d}^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Directly from definition \ref{ass:smooth} with $y=x-d$.
\end{proof}

\begin{lemma}[Bias of the exponential moving average]\label{lem:bias}
For any $t\ge1$,
\[
\EE[v_{t}] = \bigl(1-(1-\alpha)^{t}\bigr)\,\EE\!\bigl[\grad f(x_{t})\bigr].
\tag{10}
\]
Consequently the bias‑corrected estimator $\hat v_{t}$ is unbiased:
\[
\EE[\hat v_{t}] = \EE\!\bigl[\grad f(x_{t})\bigr].
\tag{11}
\]
\end{lemma}
\begin{proof}
Unfold recursion \eqref{1}:
\[
v_{t}= \alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}\,\grad f(x_{t-k}).
\]
Taking expectation and using linearity yields \eqref{10}. Dividing by
$1-(1-\alpha)^{t}$ gives \eqref{11}.
\end{proof}

\begin{lemma}[Bound on the second moment of $\hat v_{t}$]\label{lem:second}
For all $t\ge1$,
\[
\norm{\hat v_{t}} \le \frac{G}{1-(1-\alpha)^{t}}\le \frac{G}{\alpha}.
\tag{12}
\]
\end{lemma}
\begin{proof}
From definition \eqref{2} and Assumption \ref{ass:bound},
\[
\norm{v_{t}}\le \alpha\sum_{k=0}^{t-1}(1-\alpha)^{k}\norm{\grad f(x_{t-k})}
\le \alpha G\sum_{k=0}^{t-1}(1-\alpha)^{k}
= G\bigl(1-(1-\alpha)^{t}\bigr).
\]
Dividing by the denominator gives \eqref{12}. The last inequality uses
$1-(1-\alpha)^{t}\ge\alpha$ for $t\ge1$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  Throughout the proof every non‑trivial
manipulation is numbered for easy reference.

\subsection*{Step 1 – Apply the descent lemma}
Using Lemma \ref{lem:descent} with $d=\eta_{t}\hat v_{t}$ we obtain
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})-\eta_{t}\inner{\grad f(x_{t})}{\hat v_{t}}
    +\frac{L}{2}\eta_{t}^{2}\norm{\hat v_{t}}^{2}.
\tag{13}
\end{aligned}
\]

\subsection*{Step 2 – Substitute the adaptive step size}
Recall $\eta_{t}= \dfrac{\eta}{1+\norm{\hat v_{t}}}$.  Hence
\[
\eta_{t}\norm{\hat v_{t}}^{2}
= \frac{\eta\,\norm{\hat v_{t}}^{2}}{1+\norm{\hat v_{t}}}
\le \eta\,\norm{\hat v_{t}}. \tag{14}
\]
Thus the third term in \eqref{13} can be bounded as
\[
\frac{L}{2}\eta_{t}^{2}\norm{\hat v_{t}}^{2}
\le \frac{L}{2}\eta_{t}\,\eta\,\norm{\hat v_{t}}
\le \frac{L}{2}\eta^{2}\norm{\hat v_{t}}.
\tag{15}
\]

\subsection*{Step 3 – Relate $\inner{\grad f}{\hat v}$ to $\norm{\grad f}^{2}$}
From Lemma \ref{lem:bias} we have $\EE[\hat v_{t}]=\EE[\grad f(x_{t})]$.  Using the
identity $2\inner{a}{b}= \norm{a}^{2}+\norm{b}^{2}-\norm{a-b}^{2}$,
\[
\inner{\grad f(x_{t})}{\hat v_{t}}
= \frac{1}{2}\Bigl(\norm{\grad f(x_{t})}^{2}
        +\norm{\hat v_{t}}^{2}
        -\norm{\grad f(x_{t})-\hat v_{t}}^{2}\Bigr).
\tag{16}
\]
Since $\norm{\grad f(x_{t})-\hat v_{t}}^{2}\ge0$, we obtain the deterministic
lower bound
\[
\inner{\grad f(x_{t})}{\hat v_{t}}
\ge \frac{1}{2}\norm{\grad f(x_{t})}^{2}
      -\frac{1}{2}\norm{\hat v_{t}}^{2}.
\tag{17}
\]

\subsection*{Step 4 – Plug \eqref{17} into \eqref{13}}
Using \eqref{13}, \eqref{15} and \eqref{17} we get
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_{t})
   -\eta_{t}\!\left(\frac{1}{2}\norm{\grad f(x_{t})}^{2}
                     -\frac{1}{2}\norm{\hat v_{t}}^{2}\right)
   +\frac{L}{2}\eta^{2}\norm{\hat v_{t}}\\[2mm]
&= f(x_{t})
   -\frac{\eta_{t}}{2}\norm{\grad f(x_{t})}^{2}
   +\frac{\eta_{t}}{2}\norm{\hat v_{t}}^{2}
   +\frac{L}{2}\eta^{2}\norm{\hat v_{t}}.
\tag{18}
\end{aligned}
\]

\subsection*{Step 5 – Upper bound the two terms involving $\norm{\hat v_{t}}$}
From Lemma \ref{lem:second} we have $\norm{\hat v_{t}}\le G/\alpha$.
Moreover $\eta_{t}\le\eta$ (because $1+\norm{\hat v_{t}}\ge1$).  Therefore
\[
\frac{\eta_{t}}{2}\norm{\hat v_{t}}^{2}
\le \frac{\eta}{2}\Bigl(\frac{G}{\alpha}\Bigr)^{2},
\qquad
\frac{L}{2}\eta^{2}\norm{\hat v_{t}}
\le \frac{L}{2}\eta^{2}\frac{G}{\alpha}.
\tag{19}
\]

\subsection*{Step 6 – Apply the PL condition}
By Assumption \ref{ass:pl},
\[
\norm{\grad f(x_{t})}^{2}\ge 2\mu\bigl(f(x_{t})-f^{\star}\bigr).
\tag{20}
\]
Insert \eqref{20} into the first negative term of \eqref{18}:
\[
-\frac{\eta_{t}}{2}\norm{\grad f(x_{t})}^{2}
\le -\eta_{t}\mu\bigl(f(x_{t})-f^{\star}\bigr)
\le -\eta\mu\bigl(f(x_{t})-f^{\star}\bigr).
\tag{21}
\]

\subsection*{Step 7 – Collect the bounds}
Combining \eqref{18}, \eqref{19} and \eqref{21} yields the deterministic
inequality
\[
f(x_{t+1})-f^{\star}
\le\bigl(1-\eta\mu\bigr)\bigl(f(x_{t})-f^{\star}\bigr)
   +\frac{\eta}{2}\Bigl(\frac{G}{\alpha}\Bigr)^{2}
   +\frac{L}{2}\eta^{2}\frac{G}{\alpha}.
\tag{22}
\]

\subsection*{Step 8 – Choose $\eta$ to absorb the additive constant}
Set $\eta$ according to \eqref{5}, i.e. $\eta\le\mu/L^{2}$.  Then
\[
\frac{L}{2}\eta^{2}\frac{G}{\alpha}
\le \frac{L}{2}\frac{\mu^{2}}{L^{4}}\frac{G}{\alpha}
= \frac{\mu^{2}}{2L^{3}}\frac{G}{\alpha}
\le \frac{\mu}{4}\cdot\frac{G}{\alpha},
\]
and similarly
\[
\frac{\eta}{2}\Bigl(\frac{G}{\alpha}\Bigr)^{2}
\le \frac{\mu}{2L^{2}}\Bigl(\frac{G}{\alpha}\Bigr)^{2}
\le \frac{\mu}{4}\cdot\frac{G}{\alpha},
\]
provided $G\le L/\alpha$ (which holds because $G$ is a finite constant and
$\alpha\le1$).  Consequently the sum of the two additive terms in \eqref{22}
is bounded by $\frac{\mu}{2}\frac{G}{\alpha}$.  Define
\[
c\triangleq \frac{G}{\alpha},
\qquad
\rho\triangleq 1-\frac{\mu\eta}{2(1+c)}.
\tag{23}
\]
Using $1-\eta\mu = 1- \frac{\mu\eta}{2(1+c)} - \frac{\mu\eta}{2(1+c)}
\le \rho$, inequality \eqref{22} becomes
\[
f(x_{t+1})-f^{\star}
\le \rho\bigl(f(x_{t})-f^{\star}\bigr).
\tag{24}
\]

\subsection*{Step 9 – Take expectations}
All quantities in \eqref{24} are deterministic functions of the iterates, so
taking expectation preserves the same inequality:
\[
\EE\!\bigl[f(x_{t+1})-f^{\star}\bigr]
\le \rho\,\EE\!\bigl[f(x_{t})-f^{\star}\bigr].
\tag{25}
\]

\subsection*{Step 10 – Unroll the recursion}
Iterating \eqref{25} from $0$ to $t-1$ yields
\[
\EE\!\bigl[f(x_{t})-f^{\star}\bigr]
\le \rho^{\,t}\bigl(f(x_{0})-f^{\star}\bigr),
\tag{26}
\]
which is exactly the claim \eqref{7} of Theorem \ref{thm:linear}.

\subsection*{Step 11 – Derive the iteration complexity}
Solving $\rho^{\,t}\le\varepsilon/(f(x_{0})-f^{\star})$ for $t$ gives
\[
t\ge \frac{\log\!\bigl((f(x_{0})-f^{\star})/\varepsilon\bigr)}
           {\log(1/\rho)}
\le \frac{2(1+c)}{\mu\eta}\,
   \log\!\frac{f(x_{0})-f^{\star}}{\varepsilon},
\tag{27}
\]
where we used the elementary bound $\log(1/(1-\delta))\le 2\delta$ for
$\delta\in(0,1/2)$.  This yields the complexity statement \eqref{8}.

\section*{Rate Derivation (With Constants)}
Collecting the constants from the proof:
\[
\rho = 1-\frac{\mu\eta}{2\bigl(1+G/\alpha\bigr)},
\qquad
c = \frac{G}{\alpha},
\qquad
\eta\le\frac{\mu}{L^{2}}.
\]
Hence the linear rate deteriorates only linearly with the ratio $G/\alpha$;
choosing a larger $\alpha$ (more weight on the newest gradient) reduces $c$
and improves the contraction factor.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Exact gradient descent.} Setting $\alpha=1$ makes $v_{t}= \grad f(x_{t})$
and $\hat v_{t}=v_{t}$.  The algorithm reduces to
\[
x_{t+1}=x_{t}-\frac{\eta}{1+\norm{\grad f(x_{t})}}\grad f(x_{t}),
\]
and Theorem \ref{thm:linear} recovers the standard PL linear rate with
$\rho=1-\frac{\mu\eta}{2(1+G)}$.
\item \textbf{Very small $\alpha$.} When $\alpha\to0$, $v_{t}$ changes slowly and
$c=G/\alpha$ becomes large; the contraction factor approaches $1$, i.e.
the method slows down, which is consistent with intuition.
\item \textbf{Relation to SARAH/SPIDER.} If instead of the full gradient we replace
$\grad f(x_{t})$ in \eqref{1} by a variance‑reduced estimator $\tilde g_{t}$ that
satisfies $\EE[\tilde g_{t}\mid\mathcal{F}_{t-1}]=\grad f(x_{t})$ and
$\EE\|\tilde g_{t}-\grad f(x_{t})\|^{2}\le\sigma^{2}$, the same proof goes through
with $G$ replaced by $\sqrt{G^{2}+\sigma^{2}}$, yielding the same linear rate
under the PL condition.
\end{enumerate}
