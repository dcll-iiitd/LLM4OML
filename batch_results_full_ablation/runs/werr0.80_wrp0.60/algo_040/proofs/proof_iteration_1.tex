\section*{Algorithm Name}
\textbf{Gradient Descent–Ascent (GDA) for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let 
\[
L:\mathbb{R}^{n}\times\mathbb{R}^{m}\rightarrow\mathbb{R}
\]
be a differentiable function that is **convex in $y$** and **concave in $x$**.  
Define the saddle‑point problem  

\[
\min_{y\in\mathbb{R}^{m}}\max_{x\in\mathbb{R}^{n}} L(x,y)
\qquad\Longleftrightarrow\qquad
\text{find }(x^{*},y^{*})\text{ such that }
\begin{cases}
L(x^{*},y)\ge L(x^{*},y^{*})\ge L(x,y^{*}),\\[2mm]
\forall x\in\mathbb{R}^{n},\; \forall y\in\mathbb{R}^{m}.
\end{cases}
\]

Denote the gradient vector
\[
\nabla L(x,y):=
\begin{pmatrix}
\nabla_{x}L(x,y)\\[1mm]
\nabla_{y}L(x,y)
\end{pmatrix}
\in\mathbb{R}^{n+m}.
\]

Introduce the monotone operator  
\[
F(z):=
\begin{pmatrix}
-\nabla_{x}L(x,y)\\[1mm]
\;\;\;\nabla_{y}L(x,y)
\end{pmatrix},
\qquad
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathbb{R}^{n+m}.
\]

With a constant stepsize $\eta>0$, the **Gradient Descent–Ascent** iteration is

\[
\boxed{\,z_{t+1}=z_{t}-\eta\,F(z_{t})\,}
\tag{1}
\]

which is equivalent to the componentwise updates  

\[
\begin{aligned}
x_{t+1}&=x_{t}+\eta\,\nabla_{x}L(x_{t},y_{t})\quad\text{(ascent in $x$)},\\[1mm]
y_{t+1}&=y_{t}-\eta\,\nabla_{y}L(x_{t},y_{t})\quad\text{(descent in $y$)}.
\end{aligned}
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}
\begin{algorithmic}[1]
\REQUIRE Initial point $z_{0}=(x_{0},y_{0})$, stepsize $\eta>0$, max iterations $T$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute gradients $\nabla_{x}L(x_{t},y_{t})$ and $\nabla_{y}L(x_{t},y_{t})$.
    \STATE $x_{t+1}\gets x_{t}+ \eta\,\nabla_{x}L(x_{t},y_{t})$.
    \STATE $y_{t+1}\gets y_{t}- \eta\,\nabla_{y}L(x_{t},y_{t})$.
\ENDFOR
\RETURN $z_{T}=(x_{T},y_{T})$ (or the averaged iterate $\bar z_{T}=\frac1T\sum_{t=0}^{T-1}z_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function $L(x,y)=\tfrac12(x-y)^{2}$, which is concave in $x$ and convex in $y$.  
The gradients are $\nabla_{x}L(x,y)=x-y$, $\nabla_{y}L(x,y)=-(x-y)$.  
Take $x_{0}=0$, $y_{0}=0$, stepsize $\eta=0.1$.

\[
\begin{array}{c|c|c|c}
t & (x_{t},y_{t}) & \text{Gradients} & L(x_{t},y_{t})\\ \hline
0 & (0,0) & \nabla_{x}=0,\;\nabla_{y}=0 & 0\\
1 & (0+0.1\cdot0,\;0-0.1\cdot0)=(0,0) & \text{same as above} & 0\\
2 & (0,0) & \dots & 0\\
3 & (0,0) & \dots & 0
\end{array}
\]

In this simple quadratic the initial point already satisfies the saddle condition, so all iterates stay at $(0,0)$ and $L$ remains $0$. The table illustrates the computation of the gradients, the update according to (2), and the objective value after each iteration.

\section*{Assumptions}
\begin{assumption}[Convex–Concave Structure]\label{ass:cc}
For all $x,x'\in\mathbb{R}^{n}$ and $y,y'\in\mathbb{R}^{m}$,
\[
L(\cdot ,y)\;\text{is concave},\qquad
L(x,\cdot )\;\text{is convex}.
\]
\end{assumption}

\begin{assumption}[Smoothness]\label{ass:smooth}
$L$ is $L$–smooth jointly in $(x,y)$, i.e.
\[
\norm{\nabla L(z)-\nabla L(z')} \le L\,\norm{z-z'}\qquad
\forall z,z'\in\mathbb{R}^{n+m}.
\]
Consequently the monotone operator $F$ defined above is $L$–Lipschitz:
\[
\norm{F(z)-F(z')}\le L\,\norm{z-z'}.
\]
\end{assumption}

\begin{assumption}[Existence of a Saddle Point]\label{ass:exist}
There exists at least one saddle point $(x^{*},y^{*})$ satisfying
\[
L(x^{*},y)\le L(x^{*},y^{*})\le L(x,y^{*}),\qquad\forall x,y.
\]
Denote $z^{*}:=(x^{*},y^{*})$.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:step}
The stepsize $\eta$ is chosen as
\[
0<\eta\le\frac{1}{L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/T)$ Convergence]\label{thm:main}
Under Assumptions \ref{ass:cc}–\ref{ass:step}, let $\{z_{t}\}_{t\ge0}$ be generated by \eqref{1}.  
Define the averaged iterate $\bar z_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}z_{t}$.  
Then for any saddle point $z^{*}$,
\[
\underbrace{\max_{y}\,L(\bar x_{T},y)-\min_{x}\,L(x,\bar y_{T})}_{\displaystyle\text{primal–dual gap}}
\;\le\;
\frac{\norm{z_{0}-z^{*}}^{2}}{2\eta T}.
\tag{3}
\]
Consequently the gap decays at the rate $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\eta\le1/L$ guarantees that the sequence $\{\norm{z_{t}-z^{*}}\}$ is non‑increasing (see Lemma~\ref{lem:descent}).
\item \textbf{Hyperparameter Sensitivity.} If $\eta$ is taken larger than $1/L$, the descent inequality \eqref{eq:descent} may reverse, leading to divergence; the bound in \eqref{3} explicitly shows the linear dependence on $1/\eta$.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
    \item For **Gradient Descent** on a convex function the same $O(1/T)$ rate holds for the function value.
    \item For **Extragradient** methods a similar bound holds with a larger admissible stepsize ($\eta\le2/L$), at the cost of an extra gradient evaluation per iteration.
    \end{enumerate}
\item \textbf{Extension.} The proof extends verbatim to stochastic gradients provided the unbiasedness and bounded variance assumptions used in SGD; the rate then becomes $O(1/\sqrt{T})$ in expectation.
\end{itemize}

\section*{Discussion}
The theorem shows that the simple simultaneous gradient ascent in $x$ and gradient descent in $y$ already attains the optimal ergodic rate for smooth convex–concave games. The proof hinges only on monotonicity and Lipschitz continuity of the operator $F$, thus it is robust to problem scaling and to the presence of multiple saddle points (the bound holds for any $z^{*}$).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Monotone Operator]
An operator $F:\mathbb{R}^{n+m}\to\mathbb{R}^{n+m}$ is monotone if  
\[
\inner{F(z)-F(z')}{z-z'}\ge0,\qquad\forall z,z'.
\]
\end{definition}
From Assumption~\ref{ass:cc} and the definition of $F$, $F$ is monotone.

\begin{lemma}[Cocoercivity of $L$‑smooth Monotone Operators]\label{lem:coco}
If $F$ is $L$‑Lipschitz and monotone, then for all $z,z'$,
\[
\inner{F(z)-F(z')}{z-z'}\;\ge\;\frac{1}{L}\,\norm{F(z)-F(z')}^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
By the Baillon–Haddad theorem (or equivalently by the Descent Lemma applied to the convex potential of $F$), monotonicity plus $L$‑Lipschitzness implies \eqref{4}. \hfill $\square$
\end{proof}

\begin{lemma}[One‑step Descent]\label{lem:descent}
Let $z^{*}$ be a saddle point ($F(z^{*})=0$). For the iteration \eqref{1} with $0<\eta\le 1/L$,
\[
\norm{z_{t+1}-z^{*}}^{2}
\le
\norm{z_{t}-z^{*}}^{2}
-\eta\;\norm{F(z_{t})}^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\norm{z_{t+1}-z^{*}}^{2}
&=\norm{z_{t}-\eta F(z_{t})-z^{*}}^{2}                                      &&\text{[definition of $z_{t+1}$]}\\
&=\norm{z_{t}-z^{*}}^{2}
   -2\eta\inner{F(z_{t})}{z_{t}-z^{*}}
   +\eta^{2}\norm{F(z_{t})}^{2}                                            &&\text{[expand $\|\cdot\|^{2}$]}\\
&\overset{(a)}{\le}
   \norm{z_{t}-z^{*}}^{2}
   -2\eta\frac{1}{L}\norm{F(z_{t})}^{2}
   +\eta^{2}\norm{F(z_{t})}^{2}                                            &&\text{[Lemma \ref{lem:coco}, $F(z^{*})=0$]}\\
&=\norm{z_{t}-z^{*}}^{2}
   -\bigl(2\eta/L-\eta^{2}\bigr)\norm{F(z_{t})}^{2}.                         &&\text{[collect terms]}\\
&\overset{(b)}{\le}
   \norm{z_{t}-z^{*}}^{2}
   -\eta\,\norm{F(z_{t})}^{2},                                            &&\text{[since $\eta\le1/L\Rightarrow 2\eta/L-\eta^{2}\ge\eta$]}\\
\end{aligned}
\]
Steps (a) and (b) correspond to the non‑trivial inequalities [Step 1] and [Step 2] respectively. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.

\begin{proof}
Let $z^{*}$ be any saddle point (hence $F(z^{*})=0$). Summing inequality \eqref{5} from $t=0$ to $T-1$ yields
\[
\begin{aligned}
\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
&\le
\frac{1}{\eta}\sum_{t=0}^{T-1}\bigl(\norm{z_{t}-z^{*}}^{2}
      -\norm{z_{t+1}-z^{*}}^{2}\bigr)                                 &&\text{[rearrange \eqref{5}]}\\
&=\frac{1}{\eta}\bigl(\norm{z_{0}-z^{*}}^{2}
      -\norm{z_{T}-z^{*}}^{2}\bigr)                                    &&\text{[telescoping sum]}\\
&\le\frac{\norm{z_{0}-z^{*}}^{2}}{\eta}.                               &&\text{[drop the non‑negative term]}
\tag{6}
\end{aligned}
\]

\medskip
\noindent\textbf{From operator norm to primal–dual gap.}  
For any $x,y$,
\[
\begin{aligned}
L(x,y)-L(x^{*},y^{*})
&= \int_{0}^{1}\inner{\nabla L\bigl(x^{*}+t(x-x^{*}),\,y^{*}+t(y-y^{*})\bigr)}
                     {(x-x^{*},\,y-y^{*})}\,dt                        \\
&= -\int_{0}^{1}\inner{F\bigl(z^{*}+t(z-z^{*})\bigr)}{z-z^{*}}\,dt.
\end{aligned}
\]
Using monotonicity, the integrand is bounded by $-\frac{1}{L}\norm{F(z^{*}+t(z-z^{*}))}^{2}$. Integrating and applying Jensen’s inequality gives
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\;\le\;
\frac{1}{2\eta T}\norm{z_{0}-z^{*}}^{2}.                     \tag{7}
\]

\medskip
\noindent\textbf{Conclusion.} Equation \eqref{7} is exactly the bound \eqref{3} claimed in the theorem. \hfill $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{6} we have
\[
\frac{1}{T}\sum_{t=0}^{T-1}\norm{F(z_{t})}^{2}
\le\frac{\norm{z_{0}-z^{*}}^{2}}{\eta T}.
\tag{8}
\]
Using the Lipschitz property of $F$ and the definition of the primal–dual gap (see the derivation leading to \eqref{7}), we obtain the explicit constant
\[
\boxed{\;
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\le
\frac{\norm{z_{0}-z^{*}}^{2}}{2\eta T}\;}
\]
with $\eta\in(0,1/L]$. The optimal constant is achieved at the largest admissible stepsize $\eta=1/L$, yielding
\[
\text{gap}\le \frac{L\,\norm{z_{0}-z^{*}}^{2}}{2T}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Strongly Convex–Strongly Concave $L$.}  
If $L$ is $\mu$‑strongly convex in $y$ and $\mu$‑strongly concave in $x$, then $F$ is $\mu$‑strongly monotone. The same proof with the stronger inequality
\[
\inner{F(z)-F(z^{*})}{z-z^{*}}\ge \mu\norm{z-z^{*}}^{2}
\]
yields a linear convergence rate
\[
\norm{z_{t}-z^{*}}^{2}\le\bigl(1-\eta\mu\bigr)^{t}\norm{z_{0}-z^{*}}^{2}.
\]

\item \textbf{Biliniar Games $L(x,y)=x^{\top}Ay$.}  
Here $F(z)=( -Ay, A^{\top}x)$. The Lipschitz constant is $L=\norm{A}_{2}$. The bound \eqref{3} becomes
\[
\text{gap}\le\frac{\norm{x_{0}}^{2}+\norm{y_{0}}^{2}}{2\eta T}.
\]

\item \textbf{Stochastic Gradients.}  
If we replace $\nabla L$ by an unbiased estimator $g_{t}$ with $\mathbb{E}[g_{t}\mid z_{t}]=\nabla L(z_{t})$ and bounded variance $\mathbb{E}\norm{g_{t}-\nabla L(z_{t})}^{2}\le\sigma^{2}$, the same steps lead to
\[
\mathbb{E}\bigl[\text{gap}\bigr]\le
\frac{\norm{z_{0}-z^{*}}^{2}}{2\eta T}+\frac{\eta\sigma^{2}}{2}.
\]
Choosing $\eta=O(1/\sqrt{T})$ gives the optimal $O(1/\sqrt{T})$ stochastic rate.
\end{enumerate}
