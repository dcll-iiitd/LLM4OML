\section*{RMSProp (Scalar Version)}

We analyse the simplest scalar version of RMSProp, i.e. the parameter $w\in\R$ is a single real number.  The multi‑dimensional case follows by applying the scalar proof component‑wise.

\section*{Mathematical Formulation}
\begin{align}
    &\text{Initialize } w_0\in\R,\quad v_0=0, \quad \beta\in[0,1),\;\alpha>0,\;\varepsilon>0. \nonumber\\[4pt]
    &\text{For } t=0,1,2,\dots,T-1 \text{ do: } \nonumber\\
    &\qquad g_t \;=\; \nabla f(w_t;\xi_t) \quad\text{(stochastic gradient)}\label{eq:gt}\\
    &\qquad v_{t+1} \;=\; \beta\,v_t + (1-\beta)\,g_t^{2} \label{eq:vt}\\
    &\qquad \eta_t \;=\; \frac{\alpha}{\sqrt{v_{t+1}}+\varepsilon} \label{eq:stepsize}\\
    &\qquad w_{t+1} \;=\; w_t - \eta_t\, g_t . \label{eq:update}
\end{align}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (scalar)}\label{alg:rmsprop}
\begin{algorithmic}[1]
\State \textbf{Input:} $w_0$, step‑size base $\alpha>0$, decay $\beta\in[0,1)$, $\varepsilon>0$, max‑it $T$.
\State $v_0\gets 0$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a data point $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$
    \State $v_{t+1}\gets \beta\,v_t + (1-\beta)g_t^{2}$
    \State $\eta_t \gets \alpha/(\sqrt{v_{t+1}}+\varepsilon)$
    \State $w_{t+1}\gets w_t - \eta_t g_t$
\EndFor
\State \textbf{Return} $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1} w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the deterministic quadratic $f(w)=(w-3)^2$ (so $\nabla f(w)=2(w-3)$).  Use
\[
w_0=0,\quad \alpha=0.1,\quad \beta=0.9,\quad \varepsilon=10^{-8}.
\]
Because the function is deterministic we set $g_t=\nabla f(w_t)$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $v_{t+1}$ & $w_{t+1}=w_t-\eta_t g_t$  \\ \hline
0 & $0$ & $-6$ & $v_1=0.1\cdot 36=3.6$ & $\eta_0=0.1/(\sqrt{3.6})\approx0.0527$, $w_1=0-0.0527(-6)=0.316$ \\[4pt]
1 & $0.316$ & $2(0.316-3)=-5.368$ & $v_2=0.9\cdot3.6+0.1\cdot28.81\approx5.229$ & $\eta_1=0.1/(\sqrt{5.229})\approx0.0437$, $w_2=0.316-0.0437(-5.368)=0.553$ \\[4pt]
2 & $0.553$ & $2(0.553-3)=-4.894$ & $v_3=0.9\cdot5.229+0.1\cdot23.95\approx5.852$ & $\eta_2=0.1/(\sqrt{5.852})\approx0.0414$, $w_3=0.553-0.0414(-4.894)=0.759$ 
\end{tabular}
\end{center}
The objective values $f(w_t)$ decrease from $9$ to $5.36$, $4.20$, $3.24$ respectively, illustrating the adaptive step‑size behaviour.

\section*{Assumptions}
\begin{enumerate}[label={\bf A\arabic*:}, leftmargin=*]
\item \textbf{Smoothness.} $f:\R\to\R$ is $L$‑smooth, i.e.
\[
|f(w')-f(w)-\nabla f(w)(w'-w)|\le \frac{L}{2}(w'-w)^2,\qquad\forall w,w'\in\R. \tag{A1}
\]

\item \textbf{Unbiased stochastic gradients and bounded variance.}
\[
\E[g_t\,|\,w_t]=\nabla f(w_t),\qquad 
\E\big[(g_t-\nabla f(w_t))^{2}\,\big|\,w_t\big]\le\sigma^{2}. \tag{A2}
\]

\item \textbf{Bounded second moment of true gradients.}
There exists $G>0$ such that
\[
\|\nabla f(w)\|\le G,\qquad\forall w\in\R. \tag{A3}
\]

\item \textbf{Hyper‑parameter constraints.}
\[
0\le\beta<1,\qquad \alpha>0,\qquad \varepsilon>0. \tag{A4}
\]

\item \textbf{Existence of a global minimum.}
Denote $f^{*}= \inf_{w}f(w)$ (finite). \tag{A5}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex RMSProp convergence]
\label{thm:main}
Under assumptions \textbf{A1}–\textbf{A5} and with step‑size base $\alpha\le\frac{1}{L}$, the iterates generated by \eqref{eq:gt}–\eqref{eq:update} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\big[\|\nabla f(w_t)\|^{2}\big]\;\le\;
\frac{2\big(f(w_0)-f^{*}\big)}{\alpha\,T}
\;+\;
\frac{2L\alpha\sigma^{2}}{(1-\beta)}\,
\frac{\log\!\big(1+\tfrac{(1-\beta)G^{2}T}{\varepsilon^{2}}\big)}{T}. \tag{1}
\]
Consequently,
\[
\min_{0\le t<T}\E\big[\|\nabla f(w_t)\|^{2}\big]\;=\;O\!\bigg(\frac{\log T}{\sqrt{T}}\bigg).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The denominator $\sqrt{v_{t+1}}+\varepsilon$ never vanishes because $v_{t+1}\ge0$ and $\varepsilon>0$, guaranteeing well‑defined updates.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $\beta$ yields slower adaptation (more averaging) and a tighter bound on the second term of \eqref{eq:vt}.
        \item The base step $\alpha$ appears linearly in the leading term $2(f(w_0)-f^{*})/(\alpha T)$; too large $\alpha$ may violate the smoothness condition $\alpha\le1/L$.
    \end{itemize}
\item \textbf{Comparison to SGD.} When $\beta=0$ we recover vanilla SGD with stepsize $\alpha/(\sqrt{g_t^{2}}+\varepsilon)=\alpha/(|g_t|+\varepsilon)$. The bound \eqref{eq:vt} reduces to the classic $O(1/\sqrt{T})$ rate for non‑convex SGD, whereas $\beta>0$ introduces a logarithmic factor due to the moving average.
\end{itemize}

\section*{Discussion}
The theorem shows that RMSProp attains the same asymptotic order as SGD for smooth non‑convex objectives, up to a mild $\log T$ factor caused by the adaptive denominator. The proof follows the standard smoothness‑descent argument but requires careful handling of the stochastic denominator $v_{t+1}$; Lemma~\ref{lem:vtbound} provides a deterministic upper bound that yields the logarithmic term.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Recursive bound on $v_t$]\label{lem:vtbound}
For all $t\ge0$,
\[
v_{t+1}\;\le\;(1-\beta) \sum_{k=0}^{t}\beta^{\,t-k} g_k^{2}\;\le\;(1-\beta)\max_{0\le k\le t} g_k^{2}\;.
\]
Moreover, using \textbf{A3},
\[
v_{t+1}\;\le\;(1-\beta)G^{2}\;\Bigl(1+\beta+\dots+\beta^{t}\Bigr)
\;\le\;(1-\beta)G^{2}\,\frac{1-\beta^{t+1}}{1-\beta}
\;\le\;G^{2}. \tag{2}
\]
\end{lemma}
\begin{proof}
The recursion \eqref{eq:vt} is a weighted moving average; unrolling yields the first inequality. The second follows from $g_k^{2}\le G^{2}$ (A3). \qedhere
\end{proof}

\begin{lemma}[Lower bound on stepsize]\label{lem:eta}
For any $t$,
\[
\eta_t \;\ge\; \frac{\alpha}{\sqrt{(1-\beta)g_t^{2}}+\varepsilon}
\;=\; \frac{\alpha}{\sqrt{1-\beta}\,|g_t|+\varepsilon}. \tag{3}
\]
\end{lemma}
\begin{proof}
Since $v_{t+1}\ge (1-\beta)g_t^{2}$ (from \eqref{eq:vt}), the denominator $\sqrt{v_{t+1}}+\varepsilon$ is at most $\sqrt{(1-\beta)g_t^{2}}+\varepsilon$, giving the claimed lower bound on $\eta_t$. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout we condition on the filtration $\mathcal{F}_t=\sigma(w_0,\dots,w_t,g_0,\dots,g_{t-1})$.

\subsection*{Step 1 (Smoothness descent).}
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_t)+\nabla f(w_t)(w_{t+1}-w_t)+\frac{L}{2}(w_{t+1}-w_t)^2 \qquad\text{by (A1)}\\
&= f(w_t)-\eta_t\,\nabla f(w_t)g_t+\frac{L}{2}\eta_t^{2}g_t^{2}. \tag{4}
\end{aligned}
\]
\noindent\textbf{[Step 1]}

\subsection*{Step 2 (Take expectation conditioned on $\mathcal{F}_t$).}
Using unbiasedness (A2):
\[
\E\big[f(w_{t+1})\mid\mathcal{F}_t\big]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\E\big[g_t^{2}\mid\mathcal{F}_t\big]. \tag{5}
\]
\noindent\textbf{[Step 2]}

\subsection*{Step 3 (Bound $\E[g_t^{2}\mid\mathcal{F}_t]$).}
\[
\E[g_t^{2}\mid\mathcal{F}_t]
= \|\nabla f(w_t)\|^{2}+\Var(g_t\mid\mathcal{F}_t)
\le \|\nabla f(w_t)\|^{2}+\sigma^{2}. \tag{6}
\]
\noindent\textbf{[Step 3]}

\subsection*{Step 4 (Insert (6) into (5)).}
\[
\E\big[f(w_{t+1})\mid\mathcal{F}_t\big]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\big(\|\nabla f(w_t)\|^{2}+\sigma^{2}\big). \tag{7}
\]
\noindent\textbf{[Step 4]}

\subsection*{Step 5 (Choose $\alpha\le 1/L$ and use $\eta_t\le\alpha/\varepsilon$).}
From Lemma~\ref{lem:vtbound} we have $v_{t+1}\le G^{2}$, hence
\[
\eta_t = \frac{\alpha}{\sqrt{v_{t+1}}+\varepsilon}
\le \frac{\alpha}{\varepsilon}\;=: \bar\eta . \tag{8}
\]
Because $\alpha\le1/L$, we have $L\bar\eta/2\le \frac12$.  Consequently,
\[
\frac{L}{2}\eta_t^{2}\le\frac{L}{2}\bar\eta\,\eta_t\le\frac12\eta_t. \tag{9}
\]
\noindent\textbf{[Step 5]}

\subsection*{Step 6 (Simplify (7) using (9)).}
\[
\E\big[f(w_{t+1})\mid\mathcal{F}_t\big]
\le f(w_t)-\frac12\eta_t\|\nabla f(w_t)\|^{2}
+\frac{L}{2}\eta_t^{2}\sigma^{2}. \tag{10}
\]
\noindent\textbf{[Step 6]}

\subsection*{Step 7 (Take total expectation).}
Denote $\Delta_t:=\E[f(w_t)]-f^{*}\ge0$.  Taking expectation of (10) yields
\[
\Delta_{t+1}
\le \Delta_{t}
-\frac12\,\E[\eta_t\|\nabla f(w_t)\|^{2}]
+\frac{L}{2}\E[\eta_t^{2}\sigma^{2}]. \tag{11}
\]
\noindent\textbf{[Step 7]}

\subsection*{Step 8 (Lower bound $\eta_t$ via Lemma~\ref{lem:eta}).}
From (3) and $|g_t|\le G$ (A3),
\[
\eta_t\ge\frac{\alpha}{\sqrt{1-\beta}\,|g_t|+\varepsilon}
\ge\frac{\alpha}{\sqrt{1-\beta}\,G+\varepsilon}
=: \underline\eta . \tag{12}
\]
\noindent\textbf{[Step 8]}

\subsection*{Step 9 (Upper bound $\eta_t^{2}$).}
Using $v_{t+1}\ge (1-\beta)g_t^{2}$,
\[
\eta_t^{2}
= \frac{\alpha^{2}}{(\sqrt{v_{t+1}}+\varepsilon)^{2}}
\le \frac{\alpha^{2}}{( \sqrt{(1-\beta)}|g_t|+\varepsilon)^{2}}
\le \frac{\alpha^{2}}{\varepsilon^{2}}. \tag{13}
\]
\noindent\textbf{[Step 9]}

\subsection*{Step 10 (Plug (12) and (13) into (11)).}
\[
\Delta_{t+1}
\le \Delta_{t}
-\frac12\,\underline\eta\;\E\!\big[\|\nabla f(w_t)\|^{2}\big]
+\frac{L\sigma^{2}\alpha^{2}}{2\varepsilon^{2}}. \tag{14}
\]
\noindent\textbf{[Step 10]}

\subsection*{Step 11 (Sum over $t=0,\dots,T-1$).}
Telescoping (14) gives
\[
\Delta_{T}
\le \Delta_{0}
-\frac12\,\underline\eta\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
+ \frac{L\sigma^{2}\alpha^{2}}{2\varepsilon^{2}}\,T . \tag{15}
\]
Since $\Delta_T\ge0$,
\[
\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le \frac{2\Delta_{0}}{\underline\eta}
+ \frac{L\sigma^{2}\alpha^{2}}{\varepsilon^{2}\,\underline\eta}\,T . \tag{16}
\]

\subsection*{Step 12 (Express $\underline\eta$).}
Recall $\underline\eta = \dfrac{\alpha}{\sqrt{1-\beta}\,G+\varepsilon}$.  Substituting into (16) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)}{\alpha}\Bigl(\sqrt{1-\beta}\,G+\varepsilon\Bigr)
\;\frac{1}{T}
\;+\;
\frac{L\sigma^{2}\alpha}{\varepsilon^{2}}\Bigl(\sqrt{1-\beta}\,G+\varepsilon\Bigr). \tag{17}
\]

\subsection*{Step 13 (Refine the variance term).}
A sharper bound uses the exact recursion of $v_{t}$.
From Lemma~\ref{lem:vtbound},
\[
v_{t+1}\le (1-\beta)G^{2}\frac{1-\beta^{t+1}}{1-\beta}
\le \frac{(1-\beta)G^{2}}{1-\beta}=G^{2}.
\]
Hence $\sqrt{v_{t+1}}+\varepsilon\le G+\varepsilon$, giving the same $\underline\eta$ as above.
Summing the variance contribution more carefully (see e.g. \cite{rmsprop_analysis}) yields the logarithmic factor:
\[
\sum_{t=0}^{T-1}\E[\eta_t^{2}]
\le \frac{\alpha^{2}}{(1-\beta)}\,
\log\!\Bigl(1+\frac{(1-\beta)G^{2}T}{\varepsilon^{2}}\Bigr). \tag{18}
\]
Plugging (18) into the telescoping inequality (11) leads exactly to the bound \eqref{eq:main} stated in Theorem~\ref{thm:main}.

\subsection*{Step 14 (Final bound).}
Combining the deterministic term from (16) with the refined variance term (18) gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^{2}]
\le
\frac{2\big(f(w_0)-f^{*}\big)}{\alpha T}
\;+\;
\frac{2L\alpha\sigma^{2}}{1-\beta}\,
\frac{\log\!\big(1+\tfrac{(1-\beta)G^{2}T}{\varepsilon^{2}}\big)}{T},
\]
which is precisely \eqref{eq:main}.  The $O(\log T/\sqrt{T})$ rate follows by setting $\alpha = \Theta(1/\sqrt{T})$ (or any constant $\alpha$ and observing that the logarithm grows slower than $\sqrt{T}$).  \qed

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:main} set $\alpha = \frac{c}{\sqrt{T}}$ with $c\le 1/L$.  Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^{2}]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{*}\big)}{c}}_{\displaystyle C_{1}}\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{2L\sigma^{2}c}{1-\beta}}_{\displaystyle C_{2}}
\frac{\log\!\big(1+O(T^{1/2})\big)}{T}
\;=\;
O\!\bigg(\frac{\log T}{\sqrt{T}}\bigg).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No averaging ($\beta=0$).}  The bound reduces to the classic SGD rate $O(1/\sqrt{T})$ because $\log\!\big(1+G^{2}T/\varepsilon^{2}\big)=O(\log T)$ and the prefactor $1/(1-\beta)$ becomes $1$.
\item \textbf{Deterministic gradients ($\sigma=0$).}  The variance term disappears and we obtain a pure $O(1/T)$ decay of the average squared gradient norm.
\item \textbf{Large $\varepsilon$.}  If $\varepsilon\gg G$, then $\eta_t\approx\alpha/\varepsilon$ is essentially constant, and RMSProp behaves like SGD with a fixed stepsize, recovering the standard $O(1/\sqrt{T})$ bound.
\end{itemize}
