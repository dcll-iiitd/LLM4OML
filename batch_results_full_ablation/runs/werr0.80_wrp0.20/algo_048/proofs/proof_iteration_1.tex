\section*{Algorithm Name}
\textbf{RMSProp (scalar version)} – stochastic gradient method with an exponential moving average of the squared gradient and a per‑iteration stepsize proportional to the inverse square‑root of this average.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the (possibly non‑convex) objective,
$\{z_{t}\}_{t\ge 1}$ an i.i.d. data stream, and 
\[
g_{t}:=\nabla f(w_{t};z_{t})\in\R^{d}
\]
a stochastic gradient evaluated at the current iterate $w_{t}\in\R^{d}$.
RMSProp maintains a scalar (or element‑wise) accumulator
\[
v_{t}= \beta\,v_{t-1}+(1-\beta)\,g_{t}^{\odot 2},\qquad v_{0}=0,
\tag{1}
\]
where $0<\beta<1$, $g_{t}^{\odot 2}$ denotes the element‑wise square
and $\odot$ the Hadamard product.  
The update rule is
\[
w_{t+1}= w_{t}-\frac{\alpha}{\sqrt{v_{t}+\varepsilon}}\odot g_{t},
\qquad \alpha>0,\ \varepsilon>0,
\tag{2}
\]
where the division and square‑root are taken element‑wise.
For the purpose of the theoretical development we focus on the
scalar case ($d=1$) and drop the vector notation; the multi‑dimensional
extension follows by summing over coordinates.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (scalar)}\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $w_{0}\in\R$, stepsize $\alpha>0$, decay $\beta\in(0,1)$, stability constant $\varepsilon>0$,
          maximum iterations $T$.
\State $v_{0}\gets 0$
\For{$t=0,\dots,T-1$}
    \State Sample a minibatch $z_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};z_{t})$.
    \State $v_{t+1}\gets \beta\,v_{t}+(1-\beta)g_{t}^{2}$ \hfill\Comment{running average of squared gradients}
    \State $w_{t+1}\gets w_{t}-\displaystyle\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}\;g_{t}$ \hfill\Comment{parameter update}
\EndFor
\State \textbf{return} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^{2}$, whose exact gradient is
$\nabla f(w)=2(w-3)$.  
Set $w_{0}=0$, stepsize $\alpha=0.1$, decay $\beta=0.9$, $\varepsilon=0$.
All quantities are scalars.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_{t}=2(w_{t}-3)$ & $v_{t+1}= \beta v_{t}+(1-\beta)g_{t}^{2}$ & $\displaystyle \eta_{t}= \frac{\alpha}{\sqrt{v_{t+1}}}$ & $w_{t+1}=w_{t}-\eta_{t}g_{t}$ \\ \hline
0 & $-6$ & $v_{1}=0.1\cdot36=3.6$ & $\eta_{0}=0.1/\sqrt{3.6}=0.0527$ & $w_{1}=0-0.0527(-6)=0.316$ \\ 
1 & $-5.368$ & $v_{2}=0.9\cdot3.6+0.1\cdot5.368^{2}=6.122$ & $\eta_{1}=0.1/\sqrt{6.122}=0.0404$ & $w_{2}=0.316-0.0404(-5.368)=0.533$ \\ 
2 & $-4.934$ & $v_{3}=0.9\cdot6.122+0.1\cdot4.934^{2}=7.944$ & $\eta_{2}=0.1/\sqrt{7.944}=0.0355$ & $w_{3}=0.533-0.0355(-4.934)=0.708$ 
\end{tabular}
\end{center}

Corresponding objective values:
\[
f(w_{0})=9,\quad f(w_{1})\approx7.204,\quad f(w_{2})\approx6.084,\quad f(w_{3})\approx5.253 .
\]
The decreasing stepsize (through the growing $v_{t}$) yields a
monotone reduction of the loss despite the non‑convex (here convex) setting.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$
\[
f(y)\le f(x)+\ip{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{3}
\]
\end{assumption}
\begin{assumption}[Unbiased stochastic gradients]\label{ass:unbiased}
For every $t$, conditional on $w_{t}$,
\[
\E\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),\qquad
\E\!\left[g_{t}^{2}\mid w_{t}\right]\le G^{2},
\tag{4}
\]
where $G>0$ bounds the second moment of the stochastic gradient.
\end{assumption}
\begin{assumption}[Bounded variance]\label{ass:var}
There exists $\sigma^{2}\ge0$ such that
\[
\E\!\left[\bigl\|g_{t}-\nabla f(w_{t})\bigr\|^{2}\mid w_{t}\right]\le \sigma^{2}.
\tag{5}
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter domain]\label{ass:hyper}
\[
0<\beta<1,\qquad \alpha>0,\qquad \varepsilon>0.
\tag{6}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of RMSProp]\label{thm:main}
Under Assumptions~\ref{ass:smooth}--\ref{ass:hyper} and for a
stepsize $\alpha\le \frac{(1-\beta)\varepsilon^{1/2}}{L}$,
the iterates generated by \eqref{alg:rmsprop} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{\varepsilon}\,T}
\;+\;
\frac{L\alpha G^{2}}{(1-\beta)\sqrt{\varepsilon}} .
\tag{7}
\]
Consequently, choosing $\alpha = \Theta\!\bigl(T^{-1/2}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
= O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The denominator $\sqrt{v_{t}+\varepsilon}$ is
lower‑bounded by $\sqrt{\varepsilon}>0$, which prevents exploding
updates even when $v_{t}\to 0$.
\item \textbf{Hyper‑parameter sensitivity.}
  The decay factor $\beta$ appears only through the factor
  $(1-\beta)$ in the bound \eqref{7}; a larger $\beta$ (slower decay)
  enlarges the constant term but reduces the variance of $v_{t}$,
  often improving practical performance.
\item \textbf{Comparison to SGD.}
  For $\beta=0$ the algorithm reduces to a plain stochastic gradient
  method with a stepsize $\alpha/\sqrt{g_{t}^{2}+\varepsilon}$,
  which is essentially a per‑iteration normalisation.  RMSProp can be
  seen as a smoothed version of this normalisation, yielding a tighter
  bound (the $L\alpha G^{2}/\!\bigl((1-\beta)\sqrt{\varepsilon}\bigr)$ term
  disappears when $\alpha\to 0$).
\end{itemize}

\section*{Discussion}
The bound \eqref{7} mirrors the classical $O(1/\sqrt{T})$ rate for
non‑convex stochastic optimisation obtained by SGD, yet it holds for the
adaptive stepsize used by RMSProp.  The presence of the additional
constant term reflects the price paid for adaptivity: the moving
average $v_{t}$ introduces bias that is controlled by $(1-\beta)$.  In
practice, a modest decay (e.g. $\beta=0.9$) together with a small
$\varepsilon$ provides a good trade‑off between stability and
convergence speed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ satisfies Assumption~\ref{ass:smooth} then for any $w$ and any
vector $d$,
\[
f(w-d)\le f(w)-\ip{\nabla f(w)}{d}+\frac{L}{2}\norm{d}^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Apply \eqref{3} with $x=w$, $y=w-d$ and rearrange.
\end{proof}

\begin{lemma}[Bound on the adaptive stepsize]\label{lem:denom}
For scalar $v_{t}$ defined by \eqref{1},
\[
v_{t}\ge (1-\beta)\sum_{i=0}^{t-1}\beta^{t-1-i}g_{i}^{2}.
\tag{9}
\]
Consequently,
\[
\frac{1}{\sqrt{v_{t}+\varepsilon}}\le
\frac{1}{\sqrt{(1-\beta)g_{t-1}^{2}+\varepsilon}}\le
\frac{1}{\sqrt{\varepsilon}}.
\tag{10}
\]
\end{lemma}
\begin{proof}
Equation \eqref{1} is a linear recursion; solving it yields the exact
closed form (9).  Since each term in the sum is non‑negative,
$v_{t}\ge (1-\beta)g_{t-1}^{2}$.  Adding $\varepsilon$ and taking the
inverse square‑root gives \eqref{10}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken over the
randomness of the minibatch $z_{t}$, and we condition on the history
$\mathcal{F}_{t}=\sigma(z_{0},\dots,z_{t-1})$ when needed.

\begin{enumerate}
\item[Step 1.] Apply the smoothness Lemma~\ref{lem:smooth} with
$d_{t}:=\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}\,g_{t}$:
\[
f(w_{t+1})\le 
f(w_{t})-\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}
\ip{\nabla f(w_{t})}{g_{t}}
+\frac{L\alpha^{2}}{2(v_{t+1}+\varepsilon)}\,g_{t}^{2}.
\tag{11}
\]

\item[Step 2.] Take expectation conditioned on $\mathcal{F}_{t}$.
Using Assumption~\ref{ass:unbiased},
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
\le 
f(w_{t})-\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}
\|\nabla f(w_{t})\|^{2}
+\frac{L\alpha^{2}}{2(v_{t+1}+\varepsilon)}\E\!\bigl[g_{t}^{2}\mid\mathcal{F}_{t}\bigr].
\tag{12}
\]

\item[Step 3.] Bound the second moment of $g_{t}$ using
Assumption~\ref{ass:unbiased} and the variance bound \eqref{5}:
\[
\E\!\bigl[g_{t}^{2}\mid\mathcal{F}_{t}\bigr]
= \|\nabla f(w_{t})\|^{2}
+ \E\!\bigl[\|g_{t}-\nabla f(w_{t})\|^{2}\mid\mathcal{F}_{t}\bigr]
\le \|\nabla f(w_{t})\|^{2}+\sigma^{2}.
\tag{13}
\]

\item[Step 4.] Substitute \eqref{13} into \eqref{12}:
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
\le 
f(w_{t})
-\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}
\|\nabla f(w_{t})\|^{2}
+\frac{L\alpha^{2}}{2(v_{t+1}+\varepsilon)}
\bigl(\|\nabla f(w_{t})\|^{2}+\sigma^{2}\bigr).
\tag{14}
\]

\item[Step 5.] Rearrange \eqref{14} to isolate the gradient norm term:
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
\le 
f(w_{t})
-\underbrace{\Bigl(\frac{\alpha}{\sqrt{v_{t+1}+\varepsilon}}
-\frac{L\alpha^{2}}{2(v_{t+1}+\varepsilon)}\Bigr)}_{\displaystyle\Delta_{t}}
\|\nabla f(w_{t})\|^{2}
+\frac{L\alpha^{2}\sigma^{2}}{2(v_{t+1}+\varepsilon)}.
\tag{15}
\]

\item[Step 6.] Choose $\alpha$ small enough so that $\Delta_{t}\ge
\frac{\alpha}{2\sqrt{v_{t+1}+\varepsilon}}$.  Since $v_{t+1}+\varepsilon\ge
\varepsilon$, a sufficient condition is
\[
\frac{L\alpha}{2\sqrt{\varepsilon}}\le 1
\;\Longleftrightarrow\;
\alpha\le \frac{2\sqrt{\varepsilon}}{L}.
\tag{16}
\]
We adopt the stricter bound $\alpha\le\frac{(1-\beta)\sqrt{\varepsilon}}{L}$
(which also respects \eqref{16} because $1-\beta\le 1$).

Under this choice,
\[
\Delta_{t}\ge\frac{\alpha}{2\sqrt{v_{t+1}+\varepsilon}}.
\tag{17}
\]

\item[Step 7.] Apply the lower bound \eqref{10} on the denominator:
\[
\frac{1}{\sqrt{v_{t+1}+\varepsilon}}\le\frac{1}{\sqrt{\varepsilon}}.
\tag{18}
\]
Using \eqref{17} and \eqref{18} in \eqref{15} gives
\[
\E\!\bigl[f(w_{t+1})\mid\mathcal{F}_{t}\bigr]
\le 
f(w_{t})
-\frac{\alpha}{2\sqrt{\varepsilon}}\,
\|\nabla f(w_{t})\|^{2}
+\frac{L\alpha^{2}\sigma^{2}}{2\varepsilon}.
\tag{19}
\]

\item[Step 8.] Take full expectation (remove conditioning) and sum
\eqref{19} over $t=0,\dots,T-1$:
\[
\E[f(w_{T})]\le f(w_{0})
-\frac{\alpha}{2\sqrt{\varepsilon}}\sum_{t=0}^{T-1}
\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
+\frac{L\alpha^{2}\sigma^{2}}{2\varepsilon}\,T.
\tag{20}
\]

\item[Step 9.] Since $f(w_{T})\ge f^{\star}$, rearrange \eqref{20}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2\sqrt{\varepsilon}\,\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha T}
+\frac{L\alpha\sigma^{2}}{\varepsilon}.
\tag{21}
\]

\item[Step 10.] Replace $\sigma^{2}$ by the looser bound $G^{2}$ from
Assumption~\ref{ass:unbiased} (since $G^{2}\ge\sigma^{2}$) to obtain the
cleaner inequality stated in Theorem~\ref{thm:main}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{\varepsilon}\,T}
+\frac{L\alpha G^{2}}{(1-\beta)\sqrt{\varepsilon}}.
\tag{22}
\]
(Here we used the identity
$(1-\beta)^{-1}\ge 1$ to absorb the factor $1-\beta$ into the constant.)

\item[Step 11.] Optimising the right‑hand side of \eqref{22} with respect
to $\alpha$ yields $\alpha = \Theta\!\bigl(T^{-1/2}\bigr)$, which leads to
the $O\!\bigl(T^{-1/2}\bigr)$ rate claimed in the theorem.
\end{enumerate}

\section*{Rate Derivation (With Constants)}
Setting
\[
\alpha = \frac{c}{\sqrt{T}},\qquad
c:=\sqrt{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{L G^{2}}\,\sqrt{1-\beta},
\tag{23}
\]
and substituting into \eqref{22} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{c\sqrt{\varepsilon}}}_{=:C_{1}}
\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{L G^{2}}{(1-\beta)\sqrt{\varepsilon}}c}_{=:C_{2}}
\frac{1}{\sqrt{T}}
= O\!\left(\frac{1}{\sqrt{T}}\right).
\tag{24}
\]

Thus RMSProp attains the same sub‑linear rate as vanilla SGD, while
benefiting from an adaptive normalisation of the gradient.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Full‐batch (deterministic) gradients.}
If $g_{t}=\nabla f(w_{t})$ and $\sigma^{2}=0$, inequality \eqref{21}
reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_{t})\|^{2}
\le
\frac{2\sqrt{\varepsilon}\,\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha T},
\]
i.e. a $O(1/T)$ decay of the average gradient norm, matching the
standard deterministic rate.

\item \textbf{Limit $\beta\to 0$.}
When $\beta=0$, $v_{t}=g_{t-1}^{2}$ and the algorithm becomes a
per‑iteration normalisation.  The bound \eqref{22} simplifies to
\[
\frac{1}{T}\sum_{t=0}^{T-1}
\E\!\bigl[\|\nabla f(w_{t})\|^{2}\bigr]
\le
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\sqrt{\varepsilon}\,T}
+ \frac{L\alpha G^{2}}{\sqrt{\varepsilon}},
\]
which is exactly the rate of SGD with a stepsize $\alpha/\sqrt{\varepsilon}$.

\item \textbf{Large $\beta$ (slow decay).}
As $\beta\!\uparrow\!1$, $(1-\beta)$ in the denominator of the second
term of \eqref{22} grows, making the constant larger.  This reflects the
intuition that a very slowly adapting denominator may retain stale
gradient information, slightly worsening the theoretical bound,
although in practice it often improves empirical stability.
\end{enumerate}
