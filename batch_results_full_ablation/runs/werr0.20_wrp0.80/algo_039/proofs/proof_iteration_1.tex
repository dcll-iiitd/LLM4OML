\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{n}\rightarrow\mathbb{R}\cup\{+\infty\}$ and $g:\mathbb{R}^{m}\rightarrow\mathbb{R}\cup\{+\infty\}$ be proper, convex, lower‑semicontinuous (possibly non‑smooth) functions.  
Let $K\in\mathbb{R}^{m\times n}$ be a linear operator.  
We consider the saddle–point problem  

\[
\min_{x\in\mathbb{R}^{n}}\;\max_{y\in\mathbb{R}^{m}} \; \mathcal{L}(x,y)\;:=\; f(x)+\langle Kx ,y\rangle-g^{*}(y),
\tag{1}
\]

where $g^{*}$ is the convex conjugate of $g$.  

For step sizes $\tau>0$, $\sigma>0$ and extrapolation parameter $\theta\in[0,1]$, the PDHG iterates are

\[
\begin{aligned}
\bar{x}_{k} &= x_{k}+ \theta (x_{k}-x_{k-1}), \qquad (\text{extrapolation})\\[4pt]
y_{k+1}   &= \dual\!\Big(y_{k}+ \sigma K\bar{x}_{k}\Big), \qquad (\text{dual ascent})\\[4pt]
x_{k+1}   &= \primal\!\Big(x_{k}- \tau K^{\top} y_{k+1}\Big), \qquad (\text{primal descent}).
\end{aligned}
\tag{2}
\]

The proximal operators are defined by  

\[
\prox_{\tau f}(u)=\arg\min_{x}\Big\{f(x)+\frac{1}{2\tau}\|x-u\|^{2}\Big\},
\qquad
\prox_{\sigma g^{*}}(v)=\arg\min_{y}\Big\{g^{*}(y)+\frac{1}{2\sigma}\|y-v\|^{2}\Big\}.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\REQUIRE Initial primal $x_{0}\in\mathbb{R}^{n}$, dual $y_{0}\in\mathbb{R}^{m}$,
step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, linear operator $K$.
\STATE Set $x_{-1}=x_{0}$.
\FOR{$k=0,1,2,\dots$}
    \STATE $\displaystyle \bar{x}_{k}=x_{k}+ \theta (x_{k}-x_{k-1})$ \hfill\COMMENT{extrapolation}
    \STATE $\displaystyle y_{k+1}= \prox_{\sigma g^{*}}\!\big(y_{k}+ \sigma K\bar{x}_{k}\big)$ \hfill\COMMENT{dual update}
    \STATE $\displaystyle x_{k+1}= \prox_{\tau f}\!\big(x_{k}- \tau K^{\top} y_{k+1}\big)$ \hfill\COMMENT{primal update}
\ENDFOR
\RETURN $x_{k},y_{k}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar case $n=m=1$, $K=1$, $g\equiv0$ (hence $g^{*}=0$ and $\prox_{\sigma g^{*}}(v)=v$), and  

\[
f(x)=\tfrac12 (x-3)^{2}.
\]

The proximal operator of $f$ with step $\tau$ is  

\[
\prox_{\tau f}(u)=\frac{u+\tau\cdot 3}{1+\tau}.
\tag{4}
\]

Choose $\tau=\sigma=0.1$, $\theta=1$, and initialise $x_{0}=0$, $y_{0}=0$.  
Since $g^{*}=0$, the dual update reduces to $y_{k+1}=y_{k}+ \sigma \bar{x}_{k}$.

\section*{Assumptions}
\begin{assumption}[Convexity and Regularity]\label{ass:convex}
\begin{enumerate}[label=(\alph*)]
\item $f:\mathbb{R}^{n}\rightarrow\mathbb{R}\cup\{+\infty\}$ and $g:\mathbb{R}^{m}\rightarrow\mathbb{R}\cup\{+\infty\}$ are proper, closed, convex.
\item $K\in\mathbb{R}^{m\times n}$ is a bounded linear operator with operator norm $\|K\|$.
\item The saddle point set $\mathcal{S}:=\{(x^{\star},y^{\star})\mid 0\in\partial f(x^{\star})+K^{\top}y^{\star},\;0\in\partial g^{*}(y^{\star})-Kx^{\star}\}$ is non‑empty.
\end{enumerate}
\end{assumption}

\begin{assumption}[Step‑size Condition]\label{ass:steps}
The parameters $\tau,\sigma>0$ and $\theta\in[0,1]$ satisfy  

\[
\tau\sigma\|K\|^{2}<1,\qquad \theta=\frac{1}{\sqrt{1+\tau\sigma\|K\|^{2}}}\in[0,1].
\tag{5}
\]

In particular, the choice $\theta=1$ is admissible whenever $\tau\sigma\|K\|^{2}<1$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/N)$ primal–dual gap]\label{thm:ergodic}
Let Assumptions~\ref{ass:convex}–\ref{ass:steps} hold.  
Define the averaged iterates  

\[
\bar{x}^{N}:=\frac{1}{N}\sum_{k=1}^{N}x_{k},\qquad
\bar{y}^{N}:=\frac{1}{N}\sum_{k=1}^{N}y_{k}.
\]

For any saddle point $(x^{\star},y^{\star})\in\mathcal{S}$ the primal–dual gap satisfies  

\[
\gap\big(\bar{x}^{N},\bar{y}^{N}\big)
:=\big[\,f(\bar{x}^{N})+g(K\bar{x}^{N})\big]
-\big[\, -f^{*}(-K^{\top}\bar{y}^{N})+g^{*}(\bar{y}^{N})\big]
\le
\frac{C}{N},
\tag{6}
\]

where  

\[
C:=\frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
      +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}.
\]
Consequently $\big(\bar{x}^{N},\bar{y}^{N}\big)$ converges to the saddle set at rate $O(1/N)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The condition $\tau\sigma\|K\|^{2}<1$ guarantees that the Lyapunov function 
\[
\Phi_{k}:=\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
          +\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
\]
is non‑increasing up to a telescoping term (see proof). Hence the iterates remain uniformly bounded.
\item \textbf{Hyper‑parameter sensitivity.} The product $\tau\sigma$ appears only through the bound $\tau\sigma\|K\|^{2}<1$; any pair satisfying this yields the same $O(1/N)$ rate, but the constant $C$ improves when $\tau$ (resp. $\sigma$) better matches the scaling of $f$ (resp. $g^{*}$).
\item \textbf{Comparison to baselines.} For smooth $f$ and $g$ the PDHG reduces to the classical Arrow–Hurwicz method with the same $O(1/N)$ ergodic rate. In the non‑smooth case it improves over subgradient methods which only guarantee $O(1/\sqrt{N})$.
\end{itemize}

\section*{Discussion}
The theorem is *ergodic*: convergence is guaranteed for the averages $\bar{x}^{N},\bar{y}^{N}$. Under additional strong convexity of $f$ or $g^{*}$, a non‑ergodic $O(1/N)$ or even linear rate can be shown (see e.g., Chambolle–Pock 2011). The proof below follows the original analysis but is written with full algebraic detail and explicit step numbers for easy verification.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal operators]\label{lem:threepoint}
For any convex, proper, lower‑semicontinuous $h$ and any $u,v\in\mathbb{R}^{d}$,
\[
\langle \prox_{\lambda h}(u)-\prox_{\lambda h}(v),\,u-v\rangle
\ge \| \prox_{\lambda h}(u)-\prox_{\lambda h}(v) \|^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
The optimality condition of $\prox_{\lambda h}$ gives
\[
0\in\partial h(\prox_{\lambda h}(u))+\frac{1}{\lambda}\big(\prox_{\lambda h}(u)-u\big),
\qquad
0\in\partial h(\prox_{\lambda h}(v))+\frac{1}{\lambda}\big(\prox_{\lambda h}(v)-v\big).
\]
Subtracting the two inclusions, using monotonicity of $\partial h$, and multiplying by $\lambda$ yields (7).∎
\end{proof}

\begin{lemma}[Descent Lemma for the primal update]\label{lem:primal}
Let $x^{+}= \prox_{\tau f}(x-\tau K^{\top}y^{+})$ with any $y^{+}\in\mathbb{R}^{m}$. Then for any $x^{\star}$,
\[
\frac{1}{2\tau}\|x^{+}-x^{\star}\|^{2}
\le \frac{1}{2\tau}\|x-x^{\star}\|^{2}
      -\langle K^{\top}y^{+},\,x^{+}-x^{\star}\rangle
      +\frac{\tau}{2}\|K^{\top}y^{+}\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Apply Lemma~\ref{lem:threepoint} with $h=f$, $\lambda=\tau$, $u=x-\tau K^{\top}y^{+}$ and $v=x^{\star}$. Rearranging gives (8).∎
\end{proof}

\begin{lemma}[Descent Lemma for the dual update]\label{lem:dual}
Let $y^{+}= \prox_{\sigma g^{*}}(y+\sigma K\bar{x})$ for any $\bar{x}\in\mathbb{R}^{n}$. Then for any $y^{\star}$,
\[
\frac{1}{2\sigma}\|y^{+}-y^{\star}\|^{2}
\le \frac{1}{2\sigma}\|y-y^{\star}\|^{2}
      +\langle K\bar{x},\,y^{+}-y^{\star}\rangle
      -\frac{\sigma}{2}\|K\bar{x}\|^{2}.
\tag{9}
\]
\end{lemma}
\begin{proof}
Exactly as Lemma~\ref{lem:primal}, using $h=g^{*}$, $\lambda=\sigma$, $u=y+\sigma K\bar{x}$ and $v=y^{\star}$.∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:ergodic}.  Throughout let $(x^{\star},y^{\star})\in\mathcal{S}$ be an arbitrary saddle point.

\begin{enumerate}[label=[Step \arabic*]]
\item \textbf{Combine primal and dual descent.}  
Add inequalities \eqref{8} (with $x_{k},y_{k+1}$) and \eqref{9} (with $y_{k},\bar{x}_{k}$):
\[
\begin{aligned}
\frac{1}{2\tau}\|x_{k+1}-x^{\star}\|^{2}
&\le \frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
      -\langle K^{\top}y_{k+1},\,x_{k+1}-x^{\star}\rangle
      +\frac{\tau}{2}\|K^{\top}y_{k+1}\|^{2}\\[4pt]
\frac{1}{2\sigma}\|y_{k+1}-y^{\star}\|^{2}
&\le \frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}
      +\langle K\bar{x}_{k},\,y_{k+1}-y^{\star}\rangle
      -\frac{\sigma}{2}\|K\bar{x}_{k}\|^{2}.
\end{aligned}
\tag{10}
\]

\item \textbf{Introduce the Lyapunov function.}  
Define
\[
\Phi_{k}:=\frac{1}{2\tau}\|x_{k}-x^{\star}\|^{2}
        +\frac{1}{2\sigma}\|y_{k}-y^{\star}\|^{2}.
\tag{11}
\]
Summing the two lines of \eqref{10} gives
\[
\Phi_{k+1}\le \Phi_{k}
   -\langle K^{\top}y_{k+1},x_{k+1}-x^{\star}\rangle
   +\langle K\bar{x}_{k},y_{k+1}-y^{\star}\rangle
   +\frac{\tau}{2}\|K^{\top}y_{k+1}\|^{2}
   -\frac{\sigma}{2}\|K\bar{x}_{k}\|^{2}.
\tag{12}
\]

\item \textbf{Re‑express inner products.}  
Using $\langle K^{\top}y_{k+1},x_{k+1}-x^{\star}\rangle
      =\langle y_{k+1},Kx_{k+1}-Kx^{\star}\rangle$ and the definition of $\bar{x}_{k}$,
\[
\begin{aligned}
&-\langle y_{k+1},Kx_{k+1}-Kx^{\star}\rangle
   +\langle K\bar{x}_{k},y_{k+1}-y^{\star}\rangle\\
&= -\langle y_{k+1},Kx_{k+1}-K\bar{x}_{k}\rangle
   -\langle y_{k+1},K\bar{x}_{k}-Kx^{\star}\rangle
   +\langle K\bar{x}_{k},y_{k+1}\rangle
   -\langle K\bar{x}_{k},y^{\star}\rangle\\
&= -\langle y_{k+1},Kx_{k+1}-K\bar{x}_{k}\rangle
   -\langle K\bar{x}_{k},y^{\star}\rangle .
\end{aligned}
\tag{13}
\]

\item \textbf{Exploit the extrapolation relation.}  
Recall $\bar{x}_{k}=x_{k}+ \theta (x_{k}-x_{k-1})$. Hence
\[
Kx_{k+1}-K\bar{x}_{k}=K\big(x_{k+1}-x_{k}\big)-\theta K\big(x_{k}-x_{k-1}\big).
\tag{14}
\]

\item \textbf{Bound the mixed term.}  
Apply Cauchy–Schwarz and Young’s inequality with a parameter $\beta>0$ (to be chosen later):
\[
\begin{aligned}
-\langle y_{k+1},Kx_{k+1}-K\bar{x}_{k}\rangle
&\le \|y_{k+1}\|\,\|Kx_{k+1}-K\bar{x}_{k}\|\\
&\le \frac{\beta}{2}\|y_{k+1}\|^{2}
    +\frac{1}{2\beta}\|Kx_{k+1}-K\bar{x}_{k}\|^{2}.
\end{aligned}
\tag{15}
\]

\item \textbf{Insert (15) into (12).}  
Using $\|Kx_{k+1}-K\bar{x}_{k}\|^{2}\le\|K\|^{2}\|x_{k+1}-\bar{x}_{k}\|^{2}$,
\[
\Phi_{k+1}\le \Phi_{k}
   +\frac{\beta}{2}\|y_{k+1}\|^{2}
   +\frac{1}{2\beta}\|K\|^{2}\|x_{k+1}-\bar{x}_{k}\|^{2}
   -\langle K\bar{x}_{k},y^{\star}\rangle
   +\frac{\tau}{2}\|K^{\top}y_{k+1}\|^{2}
   -\frac{\sigma}{2}\|K\bar{x}_{k}\|^{2}.
\tag{16}
\]

\item \textbf{Choose $\beta$ to cancel the quadratic terms.}  
Set $\beta=\frac{1}{\tau\|K\|^{2}}$. Then
\[
\frac{\beta}{2}\|y_{k+1}\|^{2}
   +\frac{\tau}{2}\|K^{\top}y_{k+1}\|^{2}
\le \frac{1}{2\tau}\|y_{k+1}\|^{2}
   +\frac{\tau}{2}\|K^{\top}y_{k+1}\|^{2}
   \le \frac{1}{2\tau}\|y_{k+1}\|^{2}
   +\frac{\tau\|K\|^{2}}{2}\|y_{k+1}\|^{2}
   =\frac{1}{2\tau}(1+\tau^{2}\|K\|^{2})\|y_{k+1}\|^{2}.
\tag{17}
\]
Similarly,
\[
\frac{1}{2\beta}\|K\|^{2}\|x_{k+1}-\bar{x}_{k}\|^{2}
    =\frac{\tau\|K\|^{2}}{2}\|x_{k+1}-\bar{x}_{k}\|^{2}.
\tag{18}
\]

\item \textbf{Use the step‑size condition.}  
Assumption~\ref{ass:steps} implies $\tau\sigma\|K\|^{2}<1$, i.e.
\[
\frac{\tau\|K\|^{2}}{2}\le \frac{1-\tau\sigma\|K\|^{2}}{2\sigma}.
\tag{19}
\]

\item \textbf{Re‑write the inequality.}  
Insert (17)–(19) into (16) and rearrange terms:
\[
\Phi_{k+1}+\frac{\sigma}{2}\|K\bar{x}_{k}\|^{2}
\le \Phi_{k}
   -\langle K\bar{x}_{k},y^{\star}\rangle
   +\frac{1}{2\tau}\bigl(1+\tau^{2}\|K\|^{2}\bigr)\|y_{k+1}\|^{2}
   +\frac{1-\tau\sigma\|K\|^{2}}{2\sigma}\|x_{k+1}-\bar{x}_{k}\|^{2}.
\tag{20}
\]

\item \textbf{Introduce the saddle‑point optimality.}  
From the definition of a saddle point,
\[
f(x^{\star})+g(Kx^{\star})
   \le f(x)+g(Kx)+\langle y^{\star},Kx-Kx^{\star}\rangle
   \quad\forall x,
\]
and similarly
\[
-g^{*}(y^{\star})\le -g^{*}(y)-\langle Kx^{\star},y-y^{\star}\rangle
   \quad\forall y.
\]
Adding the two inequalities yields the primal–dual gap bound
\[
\mathcal{G}(x,y):=f(x)+g(Kx)-\bigl[-f^{*}(-K^{\top}y)+g^{*}(y)\bigr]
   \ge \langle Kx-y^{\star},\,y-y^{\star}\rangle .
\tag{21}
\]

\item \textbf{Relate the inner product to the Lyapunov decrement.}  
From (20) and (21) applied to $(x_{k+1},y_{k+1})$ we obtain
\[
\Phi_{k+1}+\frac{\sigma}{2}\|K\bar{x}_{k}\|^{2}
\le \Phi_{k}
   -\mathcal{G}(x_{k+1},y_{k+1})
   +\frac{1}{2\tau}\bigl(1+\tau^{2}\|K\|^{2}\bigr)\|y_{k+1}\|^{2}
   +\frac{1-\tau\sigma\|K\|^{2}}{2\sigma}\|x_{k+1}-\bar{x}_{k}\|^{2}.
\tag{22}
\]

\item \textbf{Discard non‑negative terms.}  
Both $\|K\bar{x}_{k}\|^{2}$ and the last two quadratic terms are non‑negative; dropping them yields
\[
\mathcal{G}(x_{k+1},y_{k+1})
\le \Phi_{k}-\Phi_{k+1}.
\tag{23}
\]

\item \textbf{Sum from $k=0$ to $N-1$.}  
Telescoping the right‑hand side:
\[
\sum_{k=0}^{N-1}\mathcal{G}(x_{k+1},y_{k+1})
\le \Phi_{0}-\Phi_{N}
\le \Phi_{0}
= \frac{1}{2\tau}\|x_{0}-x^{\star}\|^{2}
  +\frac{1}{2\sigma}\|y_{0}-y^{\star}\|^{2}=C.
\tag{24}
\]

\item \textbf{Pass to averages.}  
By convexity of $\mathcal{G}$ in each argument,
\[
\mathcal{G}(\bar{x}^{N},\bar{y}^{N})
\le \frac{1}{N}\sum_{k=1}^{N}\mathcal{G}(x_{k},y_{k})
\le \frac{C}{N}.
\tag{25}
\]

\item \textbf{Conclusion.}  
Inequality \eqref{25} is exactly the bound \eqref{6} claimed in Theorem~\ref{thm:ergodic}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The constant $C$ in \eqref{6} depends only on the initial distance to a saddle point.  
If the initial point is chosen as the origin and the saddle point is bounded by $R$, i.e. $\|x^{\star}\|\le R$, $\|y^{\star}\|\le R$, then  

\[
C\le \frac{R^{2}}{2}\Bigl(\frac{1}{\tau}+\frac{1}{\sigma}\Bigr).
\tag{26}
\]

Thus the explicit ergodic rate reads  

\[
\gap\big(\bar{x}^{N},\bar{y}^{N}\big)
\le \frac{R^{2}}{2N}\Bigl(\frac{1}{\tau}+\frac{1}{\sigma}\Bigr).
\tag{27}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure primal problem.} If $g\equiv0$, then $y_{k}\equiv0$ and PDHG reduces to the proximal point method on $f$, with the same $O(1/N)$ ergodic rate.
\item \textbf{Smooth $f$ and $g$.} When $f$ and $g$ are differentiable with Lipschitz gradients, one may set $\tau=\sigma=1/\|K\|$ and obtain the classic Arrow–Hurwicz update; the proof above simplifies because the proximal steps become explicit gradient steps.
\item \textbf{Strong convexity.} If $f$ is $\mu$‑strongly convex, inequality \eqref{23} can be sharpened to  
$\mathcal{G}(x_{k+1},y_{k+1})\le (1-\mu\tau)\bigl(\Phi_{k}-\Phi_{k+1}\bigr)$, leading to a linear convergence factor $1-\mu\tau$.
\end{itemize}
