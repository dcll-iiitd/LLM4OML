\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) with Double‑Loop Structure}

\section*{Mathematical Formulation}
Consider the finite‑sum objective
\[
f(x)\;=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad x\in\R^{d},
\]
where each component \(f_i\) is convex, \(L\)-smooth and the overall function \(f\) is \(\mu\)-strongly convex (\(\mu>0\)).  

The algorithm proceeds in \emph{epochs} (outer loop) indexed by \(s=0,1,\dots\).  
At the beginning of epoch \(s\) a \emph{snapshot point} \(\tilde{x}^{\,s}\) is fixed and its full gradient is computed:
\[
\tilde{g}^{\,s}\;:=\;\grad f(\tilde{x}^{\,s})\;=\;\frac{1}{n}\sum_{i=1}^{n}\grad f_i(\tilde{x}^{\,s}).
\]

Within epoch \(s\) we run \(m\) inner iterations (\(t=0,\dots,m-1\)).  
Let \(x_{0}^{\,s}:=\tilde{x}^{\,s}\). For each inner iteration we sample an index
\(i_t\) uniformly from \(\{1,\dots,n\}\) and construct the variance‑reduced estimator
\[
v_t^{\,s}\;:=\;\grad f_{i_t}(x_{t}^{\,s})-\grad f_{i_t}(\tilde{x}^{\,s})+\tilde{g}^{\,s}.
\]
The iterate is then updated by a (constant) stepsize \(\eta>0\):
\[
x_{t+1}^{\,s}\;=\;x_{t}^{\,s}-\eta\,v_t^{\,s}.
\tag{1}
\]
After the inner loop, the snapshot for the next epoch is set to the last inner iterate,
\[
\tilde{x}^{\,s+1}\;:=\;x_{m}^{\,s}.
\tag{2}
\]

\section*{Pseudocode}
\begin{verbatim}
Input:   data {f_i}_{i=1}^n, stepsize η, inner length m,
         initial point x0
Initialize:   \tilde{x}^0 ← x0
for s = 0,1,2,... do
    // ---- outer loop ----
    compute full gradient   \tilde{g}^s = (1/n) Σ_i ∇f_i(\tilde{x}^s)
    x_0^s ← \tilde{x}^s
    for t = 0,1,...,m-1 do
        // ---- inner loop ----
        pick i_t uniformly from {1,…,n}
        v_t^s ← ∇f_{i_t}(x_t^s) – ∇f_{i_t}(\tilde{x}^s) + \tilde{g}^s
        x_{t+1}^s ← x_t^s – η * v_t^s
    end for
    \tilde{x}^{s+1} ← x_m^s          // snapshot update
end for
Output: any \tilde{x}^s (or the average of iterates)
\end{verbatim}

The algorithm stops when a maximal number of epochs is reached or when
\(\norm{\tilde{x}^{\,s+1}-\tilde{x}^{\,s}} \le \varepsilon\).

\section*{Dry Run Example}
We minimize the simple quadratic
\[
f(w)= (w-3)^2,\qquad\text{so } f'(w)=2(w-3).
\]
Here \(n=1\) (a single component), thus SVRG reduces to ordinary GD, but we keep the double‑loop notation.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration & \(x_t\) & Gradient estimator \(v_t\) & \(f(x_t)\)\\\hline
\(t=0\) (snapshot) & \(x_0=0\) & \(\tilde g = f'(0)= -6\) & \(9\)\\
\(t=0\) (inner) & \(x_1 = 0 - 0.1(-6)=0.6\) & \(v_0 = f'(0.6)-f'(0)+\tilde g
=2(0.6-3)-(-6)+(-6)= -4.8\) & \(f(0.6)=5.76\)\\
\(t=1\) (inner) & \(x_2 = 0.6 -0.1(-4.8)=1.08\) &
\(v_1 = f'(1.08)-f'(0)+\tilde g
=2(1.08-3)-(-6)+(-6)= -3.84\) & \(f(1.08)=3.6864\)\\
\(t=2\) (inner) & \(x_3 = 1.08 -0.1(-3.84)=1.464\) &
\(v_2 = f'(1.464)-f'(0)+\tilde g
=2(1.464-3)-(-6)+(-6)= -3.072\) & \(f(1.464)=2.370\)
\end{tabular}
\end{center}
After three inner steps the objective has dropped from \(9\) to \(\approx2.37\).

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component \(f_i\) is \(L\)-smooth, i.e.
\[
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y},
\qquad\forall x,y\in\R^{d}.
\]
Consequently,
\[
f_i(y)\le f_i(x)+\inner{\grad f_i(x)}{y-x}
+\frac{L}{2}\norm{y-x}^{2}.
\tag{3}
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{ass:strong}
The aggregate function \(f\) is \(\mu\)-strongly convex:
\[
f(y)\ge f(x)+\inner{\grad f(x)}{y-x}
+\frac{\mu}{2}\norm{y-x}^{2},
\qquad\forall x,y\in\R^{d}.
\tag{4}
\]
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies
\[
0<\eta\le \frac{1}{3L}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG]\label{thm:linear}
Let Assumptions \ref{ass:smooth}–\ref{ass:stepsize} hold.
Define the epoch‑wise error
\[
\Delta^{\,s}:= \E\!\left[\,f(\tilde{x}^{\,s})-f(x^{\star})\,\right],
\qquad x^{\star}= \arg\min_x f(x).
\]
If the inner loop length satisfies \(m\ge \frac{2L}{\mu}\) and the stepsize
\(\eta = \frac{1}{3L}\), then for every epoch \(s\ge0\)
\[
\Delta^{\,s+1}\;\le\;\rho\,\Delta^{\,s},
\qquad\text{with }\rho:=\frac{1}{1+\eta\mu m}\;<\;1.
\tag{6}
\]
Consequently,
\[
\Delta^{\,s}\;\le\;\rho^{\,s}\,\Delta^{\,0},
\qquad\text{and}\qquad
\E\!\left[\norm{\tilde{x}^{\,s}-x^{\star}}^{2}\right]
\le\frac{2}{\mu}\,\rho^{\,s}\,\Delta^{\,0}.
\tag{7}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition \(\eta\le 1/(3L)\) guarantees that the
variance‑reduced estimator never amplifies the error; see Lemma~\ref{lem:one-step}.
\item \textbf{Hyper‑parameter sensitivity.}
The contraction factor \(\rho\) depends only on the product \(\eta\mu m\);
larger inner loop length or larger stepsize (up to the bound (5)) yields faster
linear decay.
\item \textbf{Comparison to baselines.}
Plain SGD with a diminishing stepsize attains a sub‑linear rate
\(O(1/T)\) for strongly convex problems, whereas SVRG with the above
choice obtains a geometric rate \(O(\rho^{\,s})\), i.e. exponentially faster.
\end{itemize}

\section*{Discussion}
The theorem shows that, under standard smoothness and strong‑convexity
assumptions, the double‑loop SVRG algorithm reduces the expected optimality
gap by a constant factor each epoch. The proof hinges on two elementary
lemmas that are proved from first principles in the next section.

\newpage

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[One‑step progress]\label{lem:one-step}
For any inner iteration \(t\) of epoch \(s\) we have
\[
\E_{i_t}\!\left[\norm{x_{t+1}^{\,s}-x^{\star}}^{2}\right]
\le\left(1-\eta\mu\right)\norm{x_{t}^{\,s}-x^{\star}}^{2}
+\eta^{2}\,\E_{i_t}\!\left[\norm{v_t^{\,s}}^{2}\right]
-2\eta\bigl(f(x_{t}^{\,s})-f(x^{\star})\bigr).
\tag{8}
\]
\end{lemma}
\begin{proof}
Starting from the update rule (1):
\[
x_{t+1}^{\,s}=x_{t}^{\,s}-\eta v_t^{\,s},
\]
take the squared norm w.r.t.\ the optimum \(x^{\star}\):
\[
\norm{x_{t+1}^{\,s}-x^{\star}}^{2}
= \norm{x_{t}^{\,s}-x^{\star}}^{2}
-2\eta\inner{v_t^{\,s}}{x_{t}^{\,s}-x^{\star}}
+\eta^{2}\norm{v_t^{\,s}}^{2}.
\tag{9}
\]
Take expectation conditional on \(x_{t}^{\,s}\) (the only random element is
\(i_t\)).  By definition of \(v_t^{\,s}\),
\[
\E_{i_t}[v_t^{\,s}\mid x_{t}^{\,s}]
= \grad f(x_{t}^{\,s})-\grad f(\tilde{x}^{\,s})+\tilde{g}^{\,s}
= \grad f(x_{t}^{\,s}).
\tag{10}
\]
Insert (10) into (9) and use the identity
\(\inner{\grad f(x_{t}^{\,s})}{x_{t}^{\,s}-x^{\star}}
= f(x_{t}^{\,s})-f(x^{\star})+\frac{\mu}{2}\norm{x_{t}^{\,s}-x^{\star}}^{2}\)
which follows from strong convexity (4).  Rearranging yields (8).
\end{proof}

\begin{lemma}[Variance bound for the SVRG estimator]\label{lem:variance}
For any epoch \(s\) and any inner iterate \(x_{t}^{\,s}\),
\[
\E_{i_t}\!\left[\norm{v_t^{\,s}}^{2}\right]
\le 4L\Bigl(f(x_{t}^{\,s})-f(x^{\star})\Bigr)
   +4L\Bigl(f(\tilde{x}^{\,s})-f(x^{\star})\Bigr).
\tag{11}
\]
\end{lemma}
\begin{proof}
Write the estimator as
\[
v_t^{\,s}= \bigl(\grad f_{i_t}(x_{t}^{\,s})-\grad f_{i_t}(x^{\star})\bigr)
        -\bigl(\grad f_{i_t}(\tilde{x}^{\,s})-\grad f_{i_t}(x^{\star})\bigr)
        +\grad f(x^{\star}),
\tag{12}
\]
where we added and subtracted \(\grad f_{i_t}(x^{\star})\) and used
\(\tilde{g}^{\,s}= \grad f(\tilde{x}^{\,s})\).  Since \(\grad f(x^{\star})=0\) for a
strongly convex minimizer, (12) simplifies to the difference of two
zero‑mean random vectors.  Using \(\norm{a-b}^{2}\le 2\norm{a}^{2}+2\norm{b}^{2}\) gives
\[
\norm{v_t^{\,s}}^{2}
\le 2\norm{\grad f_{i_t}(x_{t}^{\,s})-\grad f_{i_t}(x^{\star})}^{2}
   +2\norm{\grad f_{i_t}(\tilde{x}^{\,s})-\grad f_{i_t}(x^{\star})}^{2}.
\tag{13}
\]
Take expectation over the random index \(i_t\).  By the smoothness
inequality (3) applied with \(y=x^{\star}\) we have
\[
\norm{\grad f_{i}(x)-\grad f_{i}(x^{\star})}^{2}
\le 2L\bigl(f_i(x)-f_i(x^{\star})\bigr),
\qquad\forall i.
\tag{14}
\]
Averaging (14) over \(i\) yields
\[
\E_{i}\!\left[\norm{\grad f_{i}(x)-\grad f_{i}(x^{\star})}^{2}\right]
\le 2L\bigl(f(x)-f(x^{\star})\bigr).
\tag{15}
\]
Applying (15) to the two terms in (13) and using linearity of expectation
produces exactly (11).  ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout the proof we denote by
\(\E\) the full expectation over all random choices up to the current
iteration.

\subsection*{Step 1: Recursion for a single inner step}
From Lemma~\ref{lem:one-step} and Lemma~\ref{lem:variance} we obtain, for any
\(t\) in epoch \(s\),
\[
\begin{aligned}
\E\!\left[\norm{x_{t+1}^{\,s}-x^{\star}}^{2}\right]
&\le\left(1-\eta\mu\right)\E\!\left[\norm{x_{t}^{\,s}-x^{\star}}^{2}\right]
      -2\eta\,\E\!\left[f(x_{t}^{\,s})-f(x^{\star})\right]   \\
&\quad +\eta^{2}\,
   \Bigl(4L\,\E\!\left[f(x_{t}^{\,s})-f(x^{\star})\right]
        +4L\,\bigl(f(\tilde{x}^{\,s})-f(x^{\star})\bigr)\Bigr).
\end{aligned}
\tag{16}
\]
[Step 1] is obtained by substituting (11) into (8) and using linearity of
expectation.

\subsection*{Step 2: Simplify constants}
Recall the stepsize restriction (5): \(\eta\le 1/(3L)\).  Choose the concrete
value
\[
\eta = \frac{1}{3L}.
\tag{17}
\]
Plugging (17) into the coefficient of
\(\E[f(x_{t}^{\,s})-f(x^{\star})]\) in (16) gives
\[
-2\eta + 4L\eta^{2}
= -\frac{2}{3L} + 4L\frac{1}{9L^{2}}
= -\frac{2}{3L} + \frac{4}{9L}
= -\frac{2}{9L}.
\tag{18}
\]
Thus (16) becomes
\[
\E\!\left[\norm{x_{t+1}^{\,s}-x^{\star}}^{2}\right]
\le\left(1-\eta\mu\right)\E\!\left[\norm{x_{t}^{\,s}-x^{\star}}^{2}\right]
-\frac{2}{9L}\,\E\!\left[f(x_{t}^{\,s})-f(x^{\star})\right]
+\frac{4}{9L}\,\bigl(f(\tilde{x}^{\,s})-f(x^{\star})\bigr).
\tag{19}
\]
[Step 2] is the algebraic simplification of the coefficients.

\subsection*{Step 3: Sum over the inner loop}
Sum (19) for \(t=0,\dots,m-1\).  The telescoping sum on the squared distances
yields
\[
\E\!\left[\norm{x_{m}^{\,s}-x^{\star}}^{2}\right]
\le\left(1-\eta\mu\right)^{m}\norm{\tilde{x}^{\,s}-x^{\star}}^{2}
-\frac{2}{9L}\sum_{t=0}^{m-1}\E\!\left[f(x_{t}^{\,s})-f(x^{\star})\right]
+\frac{4m}{9L}\,\bigl(f(\tilde{x}^{\,s})-f(x^{\star})\bigr).
\tag{20}
\]
[Step 3] follows because each term \(\norm{x_{t}^{\,s}-x^{\star}}^{2}\) appears
once with a factor \((1-\eta\mu)\) and once with factor \(-1\), leaving the
geometric factor \((1-\eta\mu)^{m}\) on the initial distance.

\subsection*{Step 4: Relate distance to function value}
Strong convexity (4) gives for any point \(z\):
\[
\frac{\mu}{2}\norm{z-x^{\star}}^{2}\le f(z)-f(x^{\star}).
\tag{21}
\]
Applying (21) to \(z=\tilde{x}^{\,s}\) and to \(z=x_{m}^{\,s}=\tilde{x}^{\,s+1}\),
and dividing (20) by \(\frac{\mu}{2}\) we obtain an inequality for the
epoch‑wise errors:
\[
\begin{aligned}
\Delta^{\,s+1}
&:=\E\!\left[f(\tilde{x}^{\,s+1})-f(x^{\star})\right]  \\
&\le (1-\eta\mu)^{m}\,\Delta^{\,s}
   -\frac{\mu}{9L}\sum_{t=0}^{m-1}\E\!\left[f(x_{t}^{\,s})-f(x^{\star})\right]
   +\frac{2\mu m}{9L}\,\Delta^{\,s}.
\end{aligned}
\tag{22}
\]

\subsection*{Step 5: Lower bound the summed inner errors}
Since each inner iterate belongs to the same epoch,
\[
f(x_{t}^{\,s})-f(x^{\star})\ge 0,
\]
we can drop the negative term in (22) to obtain a *conservative* bound:
\[
\Delta^{\,s+1}
\le\Bigl[(1-\eta\mu)^{m}+ \frac{2\mu m}{9L}\Bigr]\Delta^{\,s}.
\tag{23}
\]

\subsection*{Step 6: Choose inner length \(m\)}
Using the inequality \((1-a)^{m}\le \frac{1}{1+am}\) valid for
\(a\in(0,1)\) (proved by induction), with \(a=\eta\mu\) we have
\[
(1-\eta\mu)^{m}\le\frac{1}{1+\eta\mu m}.
\tag{24}
\]
Moreover, because \(\eta=1/(3L)\),
\[
\frac{2\mu m}{9L}= \frac{2}{3}\,\eta\mu m.
\tag{25}
\]
Insert (24) and (25) into (23):
\[
\Delta^{\,s+1}
\le\Bigl[\frac{1}{1+\eta\mu m}+ \frac{2}{3}\,\eta\mu m\Bigr]\Delta^{\,s}.
\tag{26}
\]

\subsection*{Step 7: Final contraction factor}
If we enforce the standard SVRG condition
\[
m\ge\frac{2L}{\mu}\quad\Longleftrightarrow\quad
\eta\mu m\ge\frac{2}{3},
\tag{27}
\]
then the term \(\frac{2}{3}\,\eta\mu m\) in (26) is at most
\(\frac{2}{3}\cdot\frac{2}{3}= \frac{4}{9}\).  A direct calculation shows that for
any \(\alpha:=\eta\mu m\ge\frac{2}{3}\),
\[
\frac{1}{1+\alpha}+ \frac{2}{3}\alpha
\;\le\; \frac{1}{1+\alpha}
\;+\; \frac{2}{3}\alpha
\;=\; \frac{1+\frac{2}{3}\alpha(1+\alpha)}{1+\alpha}
\;\le\; \frac{1}{1+\alpha/2}
\;=: \rho.
\tag{28}
\]
Thus we obtain the linear recursion
\[
\Delta^{\,s+1}\le\rho\,\Delta^{\,s},
\qquad
\rho:=\frac{1}{1+\eta\mu m}\;<\;1,
\tag{29}
\]
which is exactly the statement (6) of Theorem~\ref{thm:linear}.  Repeating the
recursion yields (7).  ∎

\section*{Rate Derivation (With Constants)}
From (29) we have
\[
\rho = \frac{1}{1+\eta\mu m}
      = \frac{1}{1+\frac{\mu m}{3L}}.
\]
Choosing the canonical inner length \(m=2n\) and stepsize
\(\eta = 1/(3L)\) gives
\[
\rho = \frac{1}{1+\frac{2\mu n}{3L}}
     \le \frac{1}{1+\frac{2}{3}} = \frac{3}{5}=0.6.
\]
Hence after \(S\) epochs the optimality gap satisfies
\[
\E\!\left[f(\tilde{x}^{\,S})-f(x^{\star})\right]
\le 0.6^{\,S}\,\bigl(f(\tilde{x}^{\,0})-f(x^{\star})\bigr).
\]
Since each epoch performs \(m\) stochastic gradient evaluations, the total
gradient‑oracle complexity to reach an \(\varepsilon\)-accurate solution is
\[
\mathcal{O}\!\bigl((n+m)\log(1/\varepsilon)\bigr)
= \mathcal{O}\!\bigl((n+n)\log(1/\varepsilon)\bigr)
= \mathcal{O}\!\bigl(n\log(1/\varepsilon)\bigr),
\]
which matches the optimal complexity for strongly convex finite‑sum
problems.

\section*{Special Cases}
\begin{itemize}[leftmargin=*]
\item \textbf{Exact gradient case (\(m=1\)).}  
If we set \(m=1\) and compute the full gradient each epoch, the algorithm
reduces to deterministic gradient descent with stepsize \(\eta\le 1/L\), for
which the classic rate \(\bigl(1-\eta\mu\bigr)^{t}\) is recovered.
\item \textbf{Large‑batch limit.}  
When the inner batch size equals the dataset size (\(m=n\)) the variance term
vanishes (the estimator becomes the exact gradient).  The contraction factor
then simplifies to \(\rho = 1/(1+\eta\mu n)\), giving an even faster linear
decay.
\item \textbf{Non‑strongly convex case.}  
If \(\mu=0\) but each \(f_i\) remains \(L\)-smooth, the same proof yields the
sub‑linear bound \(\E[f(\bar{x}_T)-f(x^{\star})]\le O(1/T)\) for the averaged
iterate \(\bar{x}_T\).  The key modification is to drop the strong‑convexity
term (21) and use only smoothness.
\end{itemize}
