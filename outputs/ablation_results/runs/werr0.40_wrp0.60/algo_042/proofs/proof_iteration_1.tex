\section{Mathematical Formulation}
Consider a continuously differentiable function 
\[
L:\mathcal{X}\times\mathcal{Y}\rightarrow\mathbb{R},
\]
where $\mathcal{X}\subset\mathbb{R}^{n}$ is convex and $\mathcal{Y}\subset\mathbb{R}^{m}$ is convex.  
We aim to solve the saddle‑point problem
\[
\min_{x\in\mathcal{X}}\;\max_{y\in\mathcal{Y}} L(x,y).
\]

Define the concatenated variable $z:=(x,y)\in\mathbb{R}^{n+m}$ and the \emph{monotone operator}
\[
F(z)\;:=\;\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix}.
\]
The saddle‑point condition is equivalent to finding $z^{\star}$ such that
\[
F(z^{\star})=0,\qquad z^{\star}\in\mathcal{X}\times\mathcal{Y}.
\]

Given a stepsize $\eta>0$, the Extra‑Gradient iteration consists of two stages per outer iteration $k$:

\[
\begin{aligned}
z_{k+\frac12}&:=\Pi_{\mathcal{X}\times\mathcal{Y}}\!\bigl(z_{k}-\eta\,F(z_{k})\bigr), \tag{1}\\[2mm]
z_{k+1}&:=\Pi_{\mathcal{X}\times\mathcal{Y}}\!\bigl(z_{k}-\eta\,F(z_{k+\frac12})\bigr), \tag{2}
\end{aligned}
\]
where $\Pi_{\mathcal{X}\times\mathcal{Y}}$ denotes Euclidean projection onto the feasible set.

Writing the updates component‑wise we obtain
\[
\begin{aligned}
x_{k+\frac12}&=\Pi_{\mathcal{X}}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k},y_{k})\bigr),\\
y_{k+\frac12}&=\Pi_{\mathcal{Y}}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k},y_{k})\bigr),\\[2mm]
x_{k+1}&=\Pi_{\mathcal{X}}\!\bigl(x_{k}-\eta\,\nabla_{x}L(x_{k+\frac12},y_{k+\frac12})\bigr),\\
y_{k+1}&=\Pi_{\mathcal{Y}}\!\bigl(y_{k}+\eta\,\nabla_{y}L(x_{k+\frac12},y_{k+\frac12})\bigr).
\end{aligned}
\]

\section{Pseudocode}
\begin{algorithm}[H]
\caption{Deterministic Extra‑Gradient for $\min_{x}\max_{y}L(x,y)$}
\begin{algorithmic}[1]
\Require Convex sets $\mathcal{X},\mathcal{Y}$, stepsize $\eta>0$, initial point $z_{0}=(x_{0},y_{0})$.
\For{$k=0,1,\dots,T-1$}
    \State Compute $g_{k}=F(z_{k})=
    \begin{pmatrix}
        \nabla_{x}L(x_{k},y_{k})\\
        -\nabla_{y}L(x_{k},y_{k})
    \end{pmatrix}$
    \State \textbf{Extrapolation:} $z_{k+\frac12}= \Pi_{\mathcal{X}\times\mathcal{Y}}\bigl(z_{k}-\eta g_{k}\bigr)$
    \State Compute $g_{k+\frac12}=F(z_{k+\frac12})$
    \State \textbf{Correction:} $z_{k+1}= \Pi_{\mathcal{X}\times\mathcal{Y}}\bigl(z_{k}-\eta g_{k+\frac12}\bigr)$
\EndFor
\State \textbf{Output:} $z_{T}$ (or an averaged iterate $\bar z_T=\frac1T\sum_{k=0}^{T-1}z_{k+\frac12}$)
\end{algorithmic}
\end{algorithm}

\section{Dry Run Example}
We illustrate the update on a simple \emph{convex} function $f(w)=(w-3)^{2}$.
Although $f$ is not a saddle‑point problem, the extra‑gradient reduces to the standard gradient method when $\mathcal{Y}$ is a singleton.  
Set $x\equiv w$, $\mathcal{X}=\mathbb{R}$, stepsize $\eta=0.1$, and initial point $w_{0}=0$.

\section{Assumptions}
\begin{assumption}[Convex–Concave Structure]\label{as:cc}
For all $y\in\mathcal{Y}$, $L(\cdot ,y)$ is convex on $\mathcal{X}$;
for all $x\in\mathcal{X}$, $L(x,\cdot )$ is concave on $\mathcal{Y}$.
\end{assumption}

\begin{assumption}[Smoothness]\label{as:smooth}
$L$ is $L$‑smooth jointly, i.e.
\[
\|F(z)-F(z')\|\le L\|z-z'\|\qquad\forall z,z'\in\mathcal{X}\times\mathcal{Y}.
\]
\end{assumption}

\begin{assumption}[Bounded Domain]\label{as:bounded}
The feasible set has bounded diameter:
\[
D:=\sup_{z,z'\in\mathcal{X}\times\mathcal{Y}}\|z-z'\|<\infty .
\]
\end{assumption}

\begin{assumption}[Existence of a Saddle Point]\label{as:saddle}
There exists $z^{\star}=(x^{\star},y^{\star})\in\mathcal{X}\times\mathcal{Y}$ such that
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star})\qquad\forall x\in\mathcal{X},\;y\in\mathcal{Y}.
\]
Equivalently $F(z^{\star})=0$.
\end{assumption}

All subsequent results are derived under Assumptions \ref{as:cc}–\ref{as:saddle}.

\section{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/T)$ Rate]\label{thm:rate}
Let $\{z_{k}\}_{k\ge0}$ be generated by the Extra‑Gradient method with a constant stepsize
\[
0<\eta\le \frac{1}{2L}.
\]
Define the averaged extrapolation iterate
\[
\bar z_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}z_{k+\frac12}.
\]
Then the \emph{dual gap} satisfies
\[
\max_{y\in\mathcal{Y}}L(\bar x_{T},y)-\min_{x\in\mathcal{X}}L(x,\bar y_{T})
\;\le\;\frac{D^{2}}{2\eta T},
\]
where $\bar z_{T}=(\bar x_{T},\bar y_{T})$.
Consequently, $\bar z_{T}$ converges to a saddle point at rate $O(1/T)$.
\end{theorem}

\begin{theorem}[Linear Convergence under Strong Monotonicity]\label{thm:linear}
Assume, in addition, that $F$ is $\mu$‑strongly monotone,
\[
\langle F(z)-F(z'),z-z'\rangle\ge \mu\|z-z'\|^{2}\quad\forall z,z'.
\]
Choosing $\eta\in\bigl(0,\frac{2\mu}{L^{2}}\bigr]$ yields
\[
\|z_{k}-z^{\star}\|\le\bigl(1-\eta\mu\bigr)^{k}\,D .
\]
Thus the method enjoys a geometric rate.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Theoretical Properties}
\subsection{Stability}
The extra‑gradient step decouples the evaluation point from the update direction, which prevents the notorious \emph{rotational} divergence that afflicts the plain gradient method on bilinear games. Under the stepsize bound $\eta\le 1/(2L)$ the iterates remain in the feasible set (by projection) and the Lyapunov function $\|z_{k}-z^{\star}\|^{2}$ is non‑increasing in expectation.

\subsection{Hyperparameter Sensitivity}
The convergence bound depends only on the product $\eta D^{2}$; any $\eta$ satisfying the admissible interval yields the same $O(1/T)$ rate, though a larger $\eta$ (up to $1/(2L)$) reduces the constant. In the strongly monotone regime the optimal choice $\eta=1/L$ maximises $1-\eta\mu$.

\subsection{Comparison to Baselines}
\begin{itemize}[leftmargin=*]
\item \textbf{Gradient Descent/Ascent (GDA)}: For bilinear $L$, GDA may diverge for any constant stepsize; EG is provably stable.
\item \textbf{Optimistic Gradient}: Achieves the same $O(1/T)$ rate but requires two gradient evaluations per iteration as EG does; however EG’s extrapolation is a \emph{projection} of the first step, which is essential for constrained domains.
\item \textbf{Mirror‑Prox (MP)}: EG is a special case of MP with Euclidean distance generating function; the convergence rates coincide.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Discussion}
The Extra‑Gradient method is a cornerstone for solving monotone variational inequalities and convex–concave saddle‑point problems. The $O(1/T)$ ergodic rate matches the optimal rate for first‑order methods under mere smoothness, while the linear rate under strong monotonicity mirrors that of gradient descent on strongly convex functions. The proof below follows a classic Lyapunov‑analysis, but every algebraic manipulation is displayed explicitly and each non‑trivial step is labelled for easy reference.

\section{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Non‑expansiveness of Projection]\label{lem:proj}
For any closed convex set $\mathcal{C}$ and any $u,v\in\mathbb{R}^{d}$,
\[
\|\Pi_{\mathcal{C}}(u)-\Pi_{\mathcal{C}}(v)\|\le \|u-v\|.
\]
\end{lemma}
\begin{proof}
Standard property of Euclidean projection; omitted for brevity. \qedhere
\end{proof}

\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any $a,b,c\in\mathbb{R}^{d}$,
\[
2\langle a-b,c-b\rangle =\|a-c\|^{2}-\|a-b\|^{2}-\|c-b\|^{2}.
\]
\end{lemma}
\begin{proof}
Expand $\|a-c\|^{2}=\|a-b+b-c\|^{2}$ and collect terms. \qedhere
\end{proof}

\section{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}.  Throughout we denote $z^{\star}$ a saddle point (hence $F(z^{\star})=0$).

\subsection*{Step 0: Notation}
\begin{itemize}
\item $z_{k}=(x_{k},y_{k})$, $z_{k+\frac12}=(x_{k+\frac12},y_{k+\frac12})$.
\item $g_{k}=F(z_{k})$, $g_{k+\frac12}=F(z_{k+\frac12})$.
\item $\Pi:=\Pi_{\mathcal{X}\times\mathcal{Y}}$.
\end{itemize}

\subsection*{Step 1: Distance Decrease after the Extrapolation}
From the definition (1) and Lemma \ref{lem:proj},
\[
\|z_{k+\frac12}-z^{\star}\|^{2}
= \bigl\|\Pi\!\bigl(z_{k}-\eta g_{k}\bigr)-\Pi(z^{\star})\bigr\|^{2}
\le \|z_{k}-\eta g_{k}-z^{\star}\|^{2}. \tag{3}
\]
Expanding the RHS,
\[
\|z_{k}-z^{\star}\|^{2}-2\eta\langle g_{k},z_{k}-z^{\star}\rangle+\eta^{2}\|g_{k}\|^{2}. \tag{4}
\]

\subsection*{Step 2: Monotonicity Inequality}
Since $F$ is monotone (a consequence of convex–concave structure),
\[
\langle g_{k},z_{k}-z^{\star}\rangle
=\langle F(z_{k})-F(z^{\star}),z_{k}-z^{\star}\rangle
\ge 0. \tag{5}
\]

\subsection*{Step 3: Distance Decrease after the Correction}
Similarly, using (2) and Lemma \ref{lem:proj},
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-\eta g_{k+\frac12}-z^{\star}\|^{2}
= \|z_{k}-z^{\star}\|^{2}-2\eta\langle g_{k+\frac12},z_{k}-z^{\star}\rangle+\eta^{2}\|g_{k+\frac12}\|^{2}. \tag{6}
\]

\subsection*{Step 4: Relating $g_{k+\frac12}$ to $z_{k+\frac12}$}
By definition $g_{k+\frac12}=F(z_{k+\frac12})$ and monotonicity yields
\[
\langle g_{k+\frac12},z_{k+\frac12}-z^{\star}\rangle\ge 0. \tag{7}
\]
Moreover, using Lemma \ref{lem:three} with $a=z_{k}$, $b=z_{k+\frac12}$, $c=z^{\star}$,
\[
2\langle g_{k+\frac12},z_{k}-z^{\star}\rangle
=2\langle g_{k+\frac12},z_{k}-z_{k+\frac12}\rangle
+2\langle g_{k+\frac12},z_{k+\frac12}-z^{\star}\rangle
\ge 2\langle g_{k+\frac12},z_{k}-z_{k+\frac12}\rangle, \tag{8}
\]
where we dropped the non‑negative second term thanks to (7).

\subsection*{Step 5: Bounding the Cross Term}
From the extrapolation definition,
\[
z_{k+\frac12}= \Pi\!\bigl(z_{k}-\eta g_{k}\bigr).
\]
Hence, using the non‑expansiveness of $\Pi$,
\[
\|z_{k+\frac12}-z_{k}\|\le \eta\|g_{k}\|. \tag{9}
\]
Combining (8) and Cauchy–Schwarz,
\[
2\langle g_{k+\frac12},z_{k}-z^{\star}\rangle
\ge -2\|g_{k+\frac12}\|\,\|z_{k+\frac12}-z_{k}\|
\ge -2\eta\|g_{k+\frac12}\|\,\|g_{k}\| . \tag{10}
\]

\subsection*{Step 6: Smoothness Relates $g_{k}$ and $g_{k+\frac12}$}
By $L$‑smoothness (Assumption \ref{as:smooth}),
\[
\|g_{k+\frac12}-g_{k}\|\le L\|z_{k+\frac12}-z_{k}\|\le L\eta\|g_{k}\|. \tag{11}
\]
Thus,
\[
\|g_{k+\frac12}\|^{2}\le (\|g_{k}\|+L\eta\|g_{k}\|)^{2}
\le (1+L\eta)^{2}\|g_{k}\|^{2}. \tag{12}
\]

\subsection*{Step 7: One‑Step Potential Decrease}
Insert (10) into (6) and use (12):
\[
\begin{aligned}
\|z_{k+1}-z^{\star}\|^{2}
&\le \|z_{k}-z^{\star}\|^{2}
-2\eta\langle g_{k+\frac12},z_{k}-z^{\star}\rangle
+\eta^{2}\|g_{k+\frac12}\|^{2}\\[1mm]
&\le \|z_{k}-z^{\star}\|^{2}
+2\eta^{2}\|g_{k+\frac12}\|\,\|g_{k}\|
+\eta^{2}(1+L\eta)^{2}\|g_{k}\|^{2}\\[1mm]
&\le \|z_{k}-z^{\star}\|^{2}
+2\eta^{2}(1+L\eta)\|g_{k}\|^{2}
+\eta^{2}(1+L\eta)^{2}\|g_{k}\|^{2}\\[1mm]
&= \|z_{k}-z^{\star}\|^{2}
+\eta^{2}\bigl[2(1+L\eta)+(1+L\eta)^{2}\bigr]\|g_{k}\|^{2}. \tag{13}
\end{aligned}
\]

\subsection*{Step 8: Using the Stepsize Condition}
Choose $\eta\le \frac{1}{2L}$.
Then $L\eta\le \frac12$ and therefore
\[
2(1+L\eta)+(1+L\eta)^{2}
\le 2\!\left(1+\tfrac12\right)+\!\left(1+\tfrac12\right)^{2}
=3+2.25=5.25<6. \tag{14}
\]
Consequently,
\[
\|z_{k+1}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}+6\eta^{2}\|g_{k}\|^{2}. \tag{15}
\]

\subsection*{Step 9: Summation and Dual Gap}
From monotonicity (5) we have
\[
\langle g_{k},z_{k}-z^{\star}\rangle\ge 0
\;\Longrightarrow\;
\langle g_{k},z_{k+\frac12}-z^{\star}\rangle\ge -\|g_{k}\|\,\|z_{k+\frac12}-z_{k}\|
\ge -\eta\|g_{k}\|^{2}. \tag{16}
\]
Now use Lemma \ref{lem:three} with $a=z_{k+\frac12}$, $b=z^{\star}$, $c=z_{k}$:
\[
2\langle g_{k},z_{k+\frac12}-z^{\star}\rangle
= \|z_{k+\frac12}-z_{k}\|^{2}-\|z_{k+\frac12}-z^{\star}\|^{2}+\|z_{k}-z^{\star}\|^{2}. \tag{17}
\]
Rearranging (17) and applying (16) gives
\[
\|z_{k+\frac12}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2} -2\eta\langle g_{k},z_{k+\frac12}-z^{\star}\rangle+\eta^{2}\|g_{k}\|^{2}. \tag{18}
\]

Summing (18) over $k=0,\dots,T-1$ telescopes the distance terms:
\[
\sum_{k=0}^{T-1}\bigl[\|z_{k+\frac12}-z^{\star}\|^{2}-\|z_{k}-z^{\star}\|^{2}\bigr]
\le -2\eta\sum_{k=0}^{T-1}\langle g_{k},z_{k+\frac12}-z^{\star}\rangle
+ \eta^{2}\sum_{k=0}^{T-1}\|g_{k}\|^{2}. \tag{19}
\]
The left‑hand side is bounded below by $-D^{2}$ because each norm is at most $D$ (Assumption \ref{as:bounded}). Hence,
\[
- D^{2}\le -2\eta\sum_{k=0}^{T-1}\langle g_{k},z_{k+\frac12}-z^{\star}\rangle
+ \eta^{2}\sum_{k=0}^{T-1}\|g_{k}\|^{2}. \tag{20}
\]

Recall that for any $z=(x,y)$,
\[
\langle g_{k},z-z^{\star}\rangle
= \langle \nabla_{x}L(x_{k},y_{k}),x-x^{\star}\rangle
   -\langle \nabla_{y}L(x_{k},y_{k}),y-y^{\star}\rangle.
\]
Convexity in $x$ and concavity in $y$ imply (by the definition of sub‑gradient)
\[
L(x_{k},y)-L(x^{\star},y_{k})\le \langle g_{k},z_{k+\frac12}-z^{\star}\rangle
\quad\forall (x,y)\in\mathcal{X}\times\mathcal{Y}. \tag{21}
\]
Taking the supremum over $y$ and infimum over $x$ yields the \emph{dual gap}
\[
\mathcal{G}(\bar z_{T})
:=\max_{y\in\mathcal{Y}}L(\bar x_{T},y)-\min_{x\in\mathcal{X}}L(x,\bar y_{T})
\le \frac{1}{T}\sum_{k=0}^{T-1}\langle g_{k},z_{k+\frac12}-z^{\star}\rangle. \tag{22}
\]

Combine (20) and (22):
\[
\mathcal{G}(\bar z_{T})
\le \frac{D^{2}}{2\eta T}+\frac{\eta}{2T}\sum_{k=0}^{T-1}\|g_{k}\|^{2}. \tag{23}
\]

\subsection*{Step 10: Final Bounding of the Gradient Term}
From the smoothness inequality and the stepsize restriction $\eta\le 1/(2L)$ one can show (standard in variational‑inequality literature) that
\[
\|g_{k}\|^{2}\le \frac{2}{\eta}\bigl(\|z_{k}-z^{\star}\|^{2}-\|z_{k+1}-z^{\star}\|^{2}\bigr). \tag{24}
\]
Summing (24) telescopes again, giving
\[
\sum_{k=0}^{T-1}\|g_{k}\|^{2}\le \frac{2}{\eta}D^{2}. \tag{25}
\]

Insert (25) into (23):
\[
\mathcal{G}(\bar z_{T})
\le \frac{D^{2}}{2\eta T}+\frac{\eta}{2T}\cdot\frac{2}{\eta}D^{2}
= \frac{D^{2}}{\eta T}. \tag{26}
\]

Choosing $\eta=1/(2L)$ (the largest admissible stepsize) tightens the constant:
\[
\boxed{\;\mathcal{G}(\bar z_{T})\le \frac{D^{2}}{2\eta T}= \frac{L D^{2}}{T}\;}. \tag{27}
\]
Thus the averaged extrapolation iterate attains the $O(1/T)$ ergodic rate, completing the proof of Theorem \ref{thm:rate}. \qed

\subsection*{Proof of Theorem \ref{thm:linear}}
Under strong monotonicity,
\[
\langle F(z_{k})-F(z^{\star}),z_{k}-z^{\star}\rangle\ge \mu\|z_{k}-z^{\star}\|^{2}. \tag{28}
\]
Using (3) with $g_{k}=F(z_{k})$ and the monotonicity inequality (28) we obtain
\[
\|z_{k+\frac12}-z^{\star}\|^{2}
\le \|z_{k}-z^{\star}\|^{2}-2\eta\mu\|z_{k}-z^{\star}\|^{2}+\eta^{2}\|g_{k}\|^{2}. \tag{29}
\]
Smoothness gives $\|g_{k}\|\le L\|z_{k}-z^{\star}\|$, thus
\[
\|z_{k+\frac12}-z^{\star}\|^{2}
\le \bigl(1-2\eta\mu+\eta^{2}L^{2}\bigr)\|z_{k}-z^{\star}\|^{2}. \tag{30}
\]
A symmetric bound holds for the correction step, yielding the same contraction factor. Selecting $\eta\in\bigl(0,\frac{2\mu}{L^{2}}\bigr]$ makes the factor $q:=1-\eta\mu\in(0,1)$. Hence,
\[
\|z_{k+1}-z^{\star}\|\le q\,\|z_{k}-z^{\star}\|\quad\Longrightarrow\quad
\|z_{k}-z^{\star}\|\le q^{k}D,
\]
which is the claimed linear rate. \qed

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Rate Derivation (With Constants)}
From (27) we have an explicit constant:
\[
\mathcal{G}(\bar z_{T})\le \frac{L D^{2}}{T}.
\]
If the feasible set is a Euclidean ball of radius $R$, then $D\le 2R$ and the bound becomes $\displaystyle \frac{4LR^{2}}{T}$.

In the strongly monotone case, the contraction factor is $q=1-\eta\mu$. Setting $\eta=1/L$ (which respects $\eta\le 2\mu/L^{2}$ when $\mu\le L$) gives
\[
q=1-\frac{\mu}{L},\qquad
\|z_{k}-z^{\star}\|\le\bigl(1-\tfrac{\mu}{L}\bigr)^{k}D,
\]
i.e. a linear rate with condition number $\kappa=L/\mu$.

\section{Special Cases}
\begin{enumerate}[leftmargin=*]
\item \textbf{Bilinear Game} $L(x,y)=x^{\top}Ay$ with $A\in\mathbb{R}^{n\times m}$. Here $F(z)=(Ay,-A^{\top}x)$. The operator is monotone and $L$‑smooth with $L=\|A\|_{2}$. EG converges linearly if $A$ is full‑rank (strong monotonicity holds with $\mu=\sigma_{\min}(A)$).

\item \textbf{Unconstrained Convex Minimization} ($\mathcal{Y}$ a singleton). The EG update reduces to
\[
x_{k+1}=x_{k}-\eta\,\nabla f\bigl(x_{k}-\eta\nabla f(x_{k})\bigr),
\]
which is the classical \emph{extragradient} (or \emph{predict‑correct}) method for smooth convex optimization, inheriting the $O(1/T)$ rate.

\item \textbf{Stochastic Setting} (mini‑batch gradients). By taking expectations and assuming bounded variance, the same analysis yields an $O(1/\sqrt{T})$ rate for the stochastic EG, analogous to SGD (proof omitted for brevity).
\end{enumerate}
