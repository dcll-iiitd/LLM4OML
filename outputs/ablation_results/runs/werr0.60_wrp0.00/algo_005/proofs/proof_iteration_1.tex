\section*{SVRG (Stochastic Variance Reduced Gradient)}

\section*{Mathematical Formulation}
Let the finite‑sum objective be
\[
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),\qquad x\in\mathbb{R}^{d},
\]
where each component $f_i$ is convex and $L$–smooth.  
SVRG operates in epochs $s=0,1,\dots$.
\begin{itemize}
\item \textbf{Full‑gradient snapshot.} At the beginning of epoch $s$ we set a \emph{snapshot point}
\[
\tilde{x}^{s}=x^{s}_0,
\qquad 
\mu^{s}:=\nabla f(\tilde{x}^{s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s}).
\]
\item \textbf{Inner loop.} For $t=0,\dots,m-1$ we draw an index $i_t$ uniformly from $\{1,\dots,n\}$ and compute the \emph{variance‑reduced estimator}
\[
v_t:=\nabla f_{i_t}\!\bigl(x^{s}_t\bigr)-\nabla f_{i_t}\!\bigl(\tilde{x}^{s}\bigr)+\mu^{s}.
\]
The iterate is updated by a standard SGD step
\[
x^{s}_{t+1}=x^{s}_t-\eta\,v_t .
\]
\item \textbf{Epoch output.} After $m$ inner updates we set
\[
\tilde{x}^{s+1}=x^{s}_m .
\]
\end{itemize}
The hyper‑parameters are the stepsize $\eta>0$ and the inner‑loop length $m\in\mathbb{N}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG}
\begin{algorithmic}[1]
\Require Initial point $x^{0}$, stepsize $\eta$, inner‑loop length $m$.
\State $\tilde{x}^{0}\gets x^{0}$
\For{$s=0,1,2,\dots$}
    \State $\mu^{s}\gets \displaystyle\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{s})$ \Comment{full gradient}
    \State $x^{s}_0\gets \tilde{x}^{s}$
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\text{Uniform}\{1,\dots,n\}$
        \State $v_t\gets \nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}^{s})+\mu^{s}$
        \State $x^{s}_{t+1}\gets x^{s}_t-\eta\,v_t$
    \EndFor
    \State $\tilde{x}^{s+1}\gets x^{s}_m$
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Here $n=1$, so the full gradient equals the stochastic gradient.  
Choose $w_0=0$, stepsize $\eta=0.1$, inner‑loop length $m=3$, and run one epoch.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $v_t=\nabla f(w_t)$ & $f(w_t)$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$ \\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$ \\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$ \\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3681$ \\
\end{tabular}
\end{center}
After the inner loop the snapshot is updated $\tilde{w}^{1}=w_3=1.464$ and the objective has dropped from $9$ to $\approx2.37$.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\tag{A1}
\]
\item \textbf{Strong convexity.} $f$ is $\mu$‑strongly convex:
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^2,
\qquad\forall x,y.
\tag{A2}
\]
\item \textbf{Finite sum.} $f(x)=\frac1n\sum_{i=1}^n f_i(x)$ with $n\ge 1$.
\item \textbf{Stepsize.} $\eta$ satisfies $0<\eta\le \frac{1}{3L}$.
\item \textbf{Inner‑loop length.} $m$ is a positive integer, later chosen as $m\ge \frac{2L}{\mu}$ for linear rate.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:linear}
Under (A1)–(A5) and with the choices
\[
\eta\le\frac{1}{3L},\qquad m\ge\frac{2L}{\mu},
\]
the sequence of snapshot points $\{\tilde{x}^{s}\}_{s\ge0}$ produced by SVRG satisfies
\[
\mathbb{E}\!\left[f(\tilde{x}^{s+1})-f(x^{\star})\right]
\le\rho\,
\mathbb{E}\!\left[f(\tilde{x}^{s})-f(x^{\star})\right],
\qquad
\rho:=\frac{1}{2}<1 .
\]
Consequently,
\[
\mathbb{E}\!\left[f(\tilde{x}^{s})-f(x^{\star})\right]\le\rho^{\,s}\bigl(f(\tilde{x}^{0})-f(x^{\star})\bigr),
\]
i.e. SVRG converges \emph{linearly} in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The unbiasedness of $v_t$ (Lemma~\ref{lem:unbiased}) together with the variance bound (Lemma~\ref{lem:variance}) guarantees that the algorithm behaves like SGD with a variance that vanishes as the snapshot approaches the optimum.
\item \textbf{Hyper‑parameter sensitivity.} The contraction factor $\rho$ depends only on the product $\eta\mu m$. Larger $m$ or larger $\eta$ (up to $1/3L$) improve the rate, but too large $\eta$ violates the smoothness‑based bound and may break the contraction.
\item \textbf{Comparison.} Compared with plain SGD, which attains only a sub‑linear $O(1/T)$ rate for strongly convex problems, SVRG reaches a geometric decay with the same per‑iteration computational cost (one stochastic gradient plus occasional full gradient).
\end{itemize}

\section*{Discussion}
The theorem shows that a modest inner‑loop length $m=O(L/\mu)$ suffices to obtain a factor $1/2$ reduction per epoch. In practice $m$ is often set to $2n$ (one pass over the data) which satisfies the bound when $n\ge L/\mu$. The proof technique—combining smoothness, strong convexity, and a variance‑reduction identity—is reusable for many other variance‑reduced methods (SAGA, SARAH, etc.).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness]
\label{lem:unbiased}
For any $t$ and any history up to iteration $t$,
\[
\mathbb{E}_{i_t}[\,v_t\mid x^{s}_t,\tilde{x}^{s}\,]=\nabla f(x^{s}_t).
\]
\end{lemma}
\begin{proof}
\[
\mathbb{E}_{i_t}[v_t]
=\frac1n\sum_{i=1}^n\Bigl(\nabla f_i(x^{s}_t)-\nabla f_i(\tilde{x}^{s})\Bigr)+\mu^{s}
=\nabla f(x^{s}_t)-\nabla f(\tilde{x}^{s})+\nabla f(\tilde{x}^{s})
=\nabla f(x^{s}_t).
\tag*{$\square$}
\]
\end{proof}

\begin{lemma}[Variance bound]
\label{lem:variance}
For any $t$,
\[
\mathbb{E}_{i_t}\!\bigl[\|v_t\|^{2}\mid x^{s}_t,\tilde{x}^{s}\bigr]
\le 4L\bigl(f(x^{s}_t)-f(x^{\star})\bigr)
   +4L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr).
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the inequality $\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$,
\[
\begin{aligned}
\|v_t\|^{2}
&=\bigl\|\nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}^{s})
          +\mu^{s}\bigr\|^{2}\\
&\le 2\bigl\|\nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}^{s})\bigr\|^{2}
   +2\|\mu^{s}\|^{2}.
\end{aligned}
\]
Taking expectation over $i_t$ and applying smoothness,
\[
\mathbb{E}_{i_t}\!\bigl[\|\nabla f_{i_t}(x^{s}_t)-\nabla f_{i_t}(\tilde{x}^{s})\|^{2}\bigr]
\le L^{2}\|x^{s}_t-\tilde{x}^{s}\|^{2}
\le 2L\bigl(f(x^{s}_t)-f(x^{\star})+f(\tilde{x}^{s})-f(x^{\star})\bigr),
\]
where the last inequality follows from the standard smoothness–strong convexity relation
$\frac{1}{2L}\|\nabla f(x)-\nabla f(y)\|^{2}\le f(x)-f(y)-\langle\nabla f(y),x-y\rangle$.
Similarly $\|\mu^{s}\|^{2}\le 2L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr)$. Combining the two bounds yields the claim.
\tag*{$\square$}
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  
All expectations are taken w.r.t. the random choices of the inner indices $i_t$; we write $\mathbb{E}_t[\cdot]$ for the conditional expectation given the history up to iteration $t$.

\noindent\textbf{Step 1.} Expand the squared distance after one inner update:
\[
\begin{aligned}
\|x^{s}_{t+1}-x^{\star}\|^{2}
&=\|x^{s}_{t}-\eta v_t-x^{\star}\|^{2}\\
&=\|x^{s}_{t}-x^{\star}\|^{2}
   -2\eta\langle v_t,x^{s}_{t}-x^{\star}\rangle
   +\eta^{2}\|v_t\|^{2}.
\end{aligned}
\tag{1}
\]
\noindent\textbf{Step 2.} Take conditional expectation and use Lemma~\ref{lem:unbiased}:
\[
\mathbb{E}_t\!\bigl[\langle v_t,x^{s}_{t}-x^{\star}\rangle\bigr]
= \langle\nabla f(x^{s}_{t}),x^{s}_{t}-x^{\star}\rangle .
\tag{2}
\]
\noindent\textbf{Step 3.} By $\mu$‑strong convexity,
\[
\langle\nabla f(x^{s}_{t}),x^{s}_{t}-x^{\star}\rangle
\ge f(x^{s}_{t})-f(x^{\star})+\frac{\mu}{2}\|x^{s}_{t}-x^{\star}\|^{2}.
\tag{3}
\]
\noindent\textbf{Step 4.} Apply Lemma~\ref{lem:variance} to bound the second moment:
\[
\mathbb{E}_t\!\bigl[\|v_t\|^{2}\bigr]
\le 4L\bigl(f(x^{s}_{t})-f(x^{\star})\bigr)
   +4L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr).
\tag{4}
\]
\noindent\textbf{Step 5.} Insert (2)–(4) into (1) and take expectation:
\[
\begin{aligned}
\mathbb{E}_t\!\bigl[\|x^{s}_{t+1}-x^{\star}\|^{2}\bigr]
&\le \|x^{s}_{t}-x^{\star}\|^{2}
   -2\eta\!\left(f(x^{s}_{t})-f(x^{\star})+\frac{\mu}{2}\|x^{s}_{t}-x^{\star}\|^{2}\right)\\
&\qquad +\eta^{2}\!\left(4L\bigl(f(x^{s}_{t})-f(x^{\star})\bigr)
   +4L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr)\right).
\end{aligned}
\tag{5}
\]
\noindent\textbf{Step 6.} Rearrange terms and collect coefficients of $f(x^{s}_{t})-f(x^{\star})$:
\[
\begin{aligned}
\mathbb{E}_t\!\bigl[\|x^{s}_{t+1}-x^{\star}\|^{2}\bigr]
&\le \bigl(1-\eta\mu\bigr)\|x^{s}_{t}-x^{\star}\|^{2}
   +\bigl(-2\eta+4\eta^{2}L\bigr)\bigl(f(x^{s}_{t})-f(x^{\star})\bigr)\\
&\qquad +4\eta^{2}L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr).
\end{aligned}
\tag{6}
\]
\noindent\textbf{Step 7.} Choose $\eta\le\frac{1}{3L}$, which guarantees $-2\eta+4\eta^{2}L\le -\frac{2}{3}\eta\le0$. Hence we can drop the non‑positive term:
\[
\mathbb{E}_t\!\bigl[\|x^{s}_{t+1}-x^{\star}\|^{2}\bigr]
\le (1-\eta\mu)\|x^{s}_{t}-x^{\star}\|^{2}
   +4\eta^{2}L\bigl(f(\tilde{x}^{s})-f(x^{\star})\bigr).
\tag{7}
\]
\noindent\textbf{Step 8.} Unroll (7) for $t=0,\dots,m-1$ and take full expectation (denoted simply by $\mathbb{E}$). Using $(1-\eta\mu)^{m}\le 1-\eta\mu m/2$ for $\eta\mu m\le1$ (which holds because $m\ge 2L/\mu$ and $\eta\le 1/(3L)$),
\[
\mathbb{E}\!\bigl[\|x^{s}_{m}-x^{\star}\|^{2}\bigr]
\le (1-\eta\mu m)\,\mathbb{E}\!\bigl[\|\tilde{x}^{s}-x^{\star}\|^{2}\bigr]
   +4\eta^{2}L\,m\,\mathbb{E}\!\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr].
\tag{8}
\]
\noindent\textbf{Step 9.} Relate the distance to function values via strong convexity:
\[
\frac{\mu}{2}\|z-x^{\star}\|^{2}\le f(z)-f(x^{\star}),\qquad\forall z.
\tag{9}
\]
Applying (9) to $z=\tilde{x}^{s+1}=x^{s}_{m}$ and to $z=\tilde{x}^{s}$, (8) becomes
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{s+1})-f(x^{\star})\bigr]
\le (1-\eta\mu m)\,\mathbb{E}\!\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr]
   +2\eta L\,\mathbb{E}\!\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr].
\tag{10}
\]
\noindent\textbf{Step 10.} Choose $m\ge\frac{2L}{\mu}$ and $\eta=\frac{1}{3L}$. Then
\[
1-\eta\mu m\le 1-\frac{\mu}{3L}\cdot\frac{2L}{\mu}=1-\frac{2}{3}= \frac{1}{3},
\qquad
2\eta L = \frac{2}{3}.
\]
Plugging into (10) yields
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{s+1})-f(x^{\star})\bigr]
\le\left(\frac{1}{3}+\frac{2}{3}\right)\mathbb{E}\!\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr]
= \frac{1}{2}\,\mathbb{E}\!\bigl[f(\tilde{x}^{s})-f(x^{\star})\bigr].
\tag{11}
\]
Equation (11) is precisely the contraction claimed in Theorem~\ref{thm:linear}. Repeating the argument for $s=0,1,\dots$ gives the geometric bound.
\hfill $\blacksquare$

\section*{Rate Derivation (With Constants)}
From (11) we have $\rho=1/2$. More generally, for any $\eta\le 1/(3L)$ and $m\ge 2L/\mu$,
\[
\rho = 1-\frac{\eta\mu m}{2}\; \in\;(0,1).
\]
Thus the expected suboptimality after $S$ epochs satisfies
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{S})-f(x^{\star})\bigr]
\le \rho^{\,S}\bigl(f(\tilde{x}^{0})-f(x^{\star})\bigr).
\]
Since each epoch performs $m$ stochastic updates, the total number of stochastic gradient evaluations $T=S\,m$ yields
\[
\mathbb{E}\!\bigl[f(x_T)-f(x^{\star})\bigr]
\le \exp\!\bigl(-\tfrac{\mu}{3L}T\bigr)\bigl(f(\tilde{x}^{0})-f(x^{\star})\bigr),
\]
i.e. an \emph{exponential} decay in $T$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Quadratic functions.} If each $f_i$ is a quadratic, the variance bound (Lemma~\ref{lem:variance}) holds with equality, and the contraction factor becomes exact: $\rho=1-\eta\mu m$.
\item \textbf{Full‑batch limit.} Setting $m=1$ and $\eta=1/L$ reduces SVRG to gradient descent with the classical rate $1-\mu/L$.
\item \textbf{Infinite data.} When $n\to\infty$ and a fresh full gradient is approximated by a large batch, the analysis still applies provided the batch gradient satisfies the same unbiasedness and variance properties.
\end{itemize}
