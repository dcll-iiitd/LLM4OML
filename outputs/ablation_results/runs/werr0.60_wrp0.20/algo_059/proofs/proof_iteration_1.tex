\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be the (possibly non‑convex) loss function.  
At iteration $t$ a stochastic gradient $g_{t}$ is drawn such that  

\[
\E\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}) .
\]

A (random) quantization operator $Q:\mathbb{R}^{d}\rightarrow\mathbb{R}^{d}$ satisfies  

\[
\E\!\left[Q(u)\mid u\right]=u,\qquad 
\E\!\left[\norm{Q(u)-u}^{2}\mid u\right]\leq \beta^{2}\norm{u}^{2},
\tag{1}
\]

for a constant $\beta\ge 0$ that measures the relative quantization error.  
The algorithm communicates the compressed vector $Q(g_{t})$ and updates

\[
\boxed{%
w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t})},
\qquad t=0,1,\dots,T-1,
\tag{2}
\]

where $\{\eta_{t}\}_{t\ge0}$ is a (deterministic) stepsize schedule.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent (QSGD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\mathbb{R}^{d}$, stepsizes $\{\eta_{t}\}$, quantizer $Q(\cdot)$
\FOR{$t=0$ {\bfseries to} $T-1$}
    \STATE Sample a mini‑batch (or a random data point) and compute a stochastic gradient $g_{t}$.
    \STATE Quantize: $\tilde g_{t}=Q(g_{t})$.
    \STATE Update: $w_{t+1}=w_{t}-\eta_{t}\tilde g_{t}$.
\ENDFOR
\STATE \textbf{Output:} $\{w_{t}\}_{t=0}^{T}$ (or the average $\bar w_{T}= \tfrac1T\sum_{t=0}^{T-1}w_{t}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, whose gradient is $\nabla f(w)=2(w-3)$.  
Take a *deterministic* quantizer $Q(u)=u$ (so $\beta=0$), stepsize $\eta=0.1$, and $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$ \\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3645$\\
\end{tabular}
\end{center}

The algorithm makes progress toward the minimiser $w^{*}=3$, and the objective values decrease at each step.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}:,leftmargin=*,align=left]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2},
\qquad\forall x,y\in\mathbb{R}^{d}.
\tag{A1}
\]

\item \textbf{Unbiased stochastic gradients with bounded variance.}
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad
\E\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2}.
\tag{A2}
\]

\item \textbf{Unbiased quantization with relative variance bound.}
For the quantizer $Q$,
\[
\E[Q(u)\mid u]=u,\qquad
\E\!\left[\norm{Q(u)-u}^{2}\mid u\right]\le\beta^{2}\norm{u}^{2},
\tag{A3}
\]
where $\beta\ge0$ is a constant (e.g. $\beta^{2}= \frac{d}{s}$ for $s$‑bit stochastic rounding).

\item \textbf{Stepsize.} The stepsizes are constant $\eta_{t}\equiv\eta$ and satisfy
\[
0<\eta\le\frac{1}{L(1+\beta^{2})}.
\tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD for non‑convex smooth objectives]\label{thm:main}
Under assumptions \textbf{A1}–\textbf{A4}, for any integer $T\ge1$ the iterates generated by \eqref{2} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{*}\bigr)}{\eta T}}_{\text{initial gap term}}
\;+\;
\underbrace{L\eta\bigl(\sigma^{2}+ \beta^{2}G^{2}\bigr)}_{\text{variance term}},
\tag{3}
\]
where $f^{*}\eqdef\inf_{w}f(w)$ and $G^{2}\eqdef\sup_{w}\E\!\left[\norm{g_{t}}^{2}\mid w\right]$ (finite under A2). Consequently,
\[
\min_{0\le t<T}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\;=\;O\!\left(\frac{1}{\sqrt{T}}\right)
\quad\text{when }\eta\asymp\frac{1}{\sqrt{T}}.
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label={P\arabic*}:,leftmargin=*,align=left]
\item \textbf{Stability.} The bound \eqref{3} holds for any $L$‑smooth $f$; the quantization error only inflates the variance term by the factor $\beta^{2}$.
\item \textbf{Hyper‑parameter sensitivity.} The stepsize must respect \textbf{A4}. A larger $\beta$ (coarser quantization) forces a smaller admissible $\eta$, reflecting the trade‑off between communication cost and convergence speed.
\item \textbf{Comparison to vanilla SGD.} Setting $\beta=0$ (no quantization) recovers the classical non‑convex SGD bound $\frac{2\Delta}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Special cases.} If $g_{t}=\nabla f(w_{t})$ (deterministic gradients) then $\sigma=0$ and the bound reduces to $ \frac{2\Delta}{\eta T}+L\eta\beta^{2}G^{2}$. Choosing $\eta = \Theta(1/\sqrt{T})$ yields $O(1/\sqrt{T})$ convergence.
\end{enumerate}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of the Lipschitz continuity of $\nabla f$. $\square$
\end{proof}

\begin{lemma}[Bound on the second moment of stochastic gradients]\label{lem:Gbound}
Under \textbf{A2} there exists $G^{2}<\infty$ such that
\[
\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]\le G^{2}:= \norm{\nabla f(w_{t})}^{2}+\sigma^{2}.
\]
\end{lemma}
\begin{proof}
Decompose $\norm{g_{t}}^{2}= \norm{\nabla f(w_{t})}^{2}
+2\inner{\nabla f(w_{t})}{g_{t}-\nabla f(w_{t})}
+\norm{g_{t}-\nabla f(w_{t})}^{2}$,
take conditional expectation and use unbiasedness and variance bound. $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout, expectations are taken w.r.t. all randomness up to iteration $t$ unless otherwise noted.

\medskip
\noindent\textbf{[Step 1] Smoothness applied to the update.}  
From Lemma~\ref{lem:smooth} with $x=w_{t}$ and $y=w_{t+1}=w_{t}-\eta Q(g_{t})$,
\[
f(w_{t+1})\le f(w_{t})-\eta\inner{\nabla f(w_{t})}{Q(g_{t})}
+\frac{L\eta^{2}}{2}\norm{Q(g_{t})}^{2}.
\tag{4}
\]

\noindent\textbf{[Step 2] Take conditional expectation given $w_{t}$.}  
Using the law of total expectation and the unbiasedness of $Q$ (Assumption \textbf{A3}),
\[
\E\!\left[\inner{\nabla f(w_{t})}{Q(g_{t})}\mid w_{t}\right]
= \inner{\nabla f(w_{t})}{\E[Q(g_{t})\mid w_{t}]}
= \inner{\nabla f(w_{t})}{\E[g_{t}\mid w_{t}]}
= \norm{\nabla f(w_{t})}^{2},
\tag{5}
\]
where the last equality follows from \textbf{A2}.

\noindent\textbf{[Step 3] Bound the second moment of the quantized gradient.}  
Conditioned on $w_{t}$,
\[
\begin{aligned}
\E\!\left[\norm{Q(g_{t})}^{2}\mid w_{t}\right] 
&= \E\!\left[\norm{g_{t}+ \bigl(Q(g_{t})-g_{t}\bigr)}^{2}\mid w_{t}\right] \\
&= \E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
   + 2\,\E\!\left[\inner{g_{t}}{Q(g_{t})-g_{t}}\mid w_{t}\right]
   + \E\!\left[\norm{Q(g_{t})-g_{t}}^{2}\mid w_{t}\right].
\end{aligned}
\tag{6}
\]
The cross term vanishes because $\E[Q(g_{t})-g_{t}\mid g_{t}]=0$ (unbiasedness), hence also zero after conditioning on $w_{t}$. Using \textbf{A3} and Lemma~\ref{lem:Gbound},
\[
\E\!\left[\norm{Q(g_{t})}^{2}\mid w_{t}\right]
\le (1+\beta^{2})\,\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
\le (1+\beta^{2})\,G^{2}.
\tag{7}
\]

\noindent\textbf{[Step 4] Take full expectation of (4).}  
Combining (4), (5) and (7) and then taking total expectation yields
\[
\E\!\left[f(w_{t+1})\right]
\le \E\!\left[f(w_{t})\right]
-\eta\,\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+\frac{L\eta^{2}}{2}(1+\beta^{2})\,G^{2}.
\tag{8}
\]

\noindent\textbf{[Step 5] Telescoping sum.}  
Sum \eqref{8} over $t=0,\dots,T-1$:
\[
\E\!\left[f(w_{T})\right]
\le f(w_{0})
-\eta\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+\frac{L\eta^{2}}{2}(1+\beta^{2})\,G^{2}T .
\tag{9}
\]

\noindent\textbf{[Step 6] Rearrangement.}  
Since $f(w_{T})\ge f^{*}$,
\[
\eta\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\le f(w_{0})-f^{*}
+\frac{L\eta^{2}}{2}(1+\beta^{2})\,G^{2}T .
\tag{10}
\]

\noindent\textbf{[Step 7] Divide by $\eta T$.}  
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\le
\frac{f(w_{0})-f^{*}}{\eta T}
+\frac{L\eta}{2}(1+\beta^{2})\,G^{2}.
\tag{11}
\]

\noindent\textbf{[Step 8] Substitute $G^{2}\le \sigma^{2}+ \sup_{w}\norm{\nabla f(w)}^{2}$.}  
Define $G^{2}:=\sigma^{2}+M^{2}$ where $M:=\sup_{w}\norm{\nabla f(w)}$ (finite for bounded domains or by coercivity). Then (11) becomes exactly the bound \eqref{3} after multiplying the numerator of the first term by $2$ (a harmless constant factor) and noting $L\eta(1+\beta^{2})\le L\eta+L\eta\beta^{2}$.

Thus we have proved the claimed convergence rate. $\square$

\section*{Rate Derivation (With Constants)}
Choosing a constant stepsize
\[
\eta = \min\!\Bigl\{\frac{1}{L(1+\beta^{2})},\,
\sqrt{\frac{2\bigl(f(w_{0})-f^{*}\bigr)}{L(1+\beta^{2})\,\bigl(\sigma^{2}+\beta^{2}M^{2}\bigr)T}}\Bigr\},
\tag{12}
\]
the two terms on the right‑hand side of \eqref{3} become balanced, yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\le
\underbrace{2\sqrt{\frac{L(1+\beta^{2})\bigl(\sigma^{2}+\beta^{2}M^{2}\bigr)\bigl(f(w_{0})-f^{*}\bigr)}{T}}}_{\displaystyle O\!\bigl(T^{-1/2}\bigr)} .
\tag{13}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Exact gradients, no quantization.} $\sigma=0$, $\beta=0$:
\[
\frac{1}{T}\sum_{t}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\le \frac{2\Delta}{\eta T}+ \frac{L\eta}{2}M^{2},
\]
recovering the deterministic gradient descent bound.
\item \textbf{Pure stochastic gradients, no quantization.} $\beta=0$:
\[
\frac{1}{T}\sum_{t}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\le \frac{2\Delta}{\eta T}+ \frac{L\eta}{2}\bigl(\sigma^{2}+M^{2}\bigr),
\]
the standard non‑convex SGD result.
\item \textbf{Heavy quantization.} Large $\beta$ inflates the variance term linearly in $\beta^{2}$, confirming the intuitive trade‑off between communication compression and statistical efficiency.
\end{itemize}

\section*{Discussion}
The proof shows that the only place where quantization enters the analysis is through the relative variance constant $\beta^{2}$. As long as $\beta$ is bounded (which holds for most practical stochastic rounding or sparsification schemes), QSGD enjoys the same $O(T^{-1/2})$ convergence rate as uncompressed SGD. The stepsize condition \textbf{A4} guarantees that the extra term $L\eta^{2}\beta^{2}G^{2}/2$ does not dominate the descent term $-\eta\|\nabla f(w_{t})\|^{2}$, ensuring overall descent in expectation.
