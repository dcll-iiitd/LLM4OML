\section*{Algorithm Name}
\textbf{SAGA} (Stochastic Average Gradient Augmented) – a variance‑reduced stochastic gradient method for finite‑sum problems.

\section*{Mathematical Formulation}
Consider the finite‑sum convex optimization problem  

\[
\min_{x\in\R^{d}} \; f(x)\;\; \text{with}\;\; f(x)=\frac1n\sum_{i=1}^{n}f_i(x),
\tag{1}
\]

where each component $f_i:\R^{d}\rightarrow\R$ is convex and $L$‑smooth:
\[
\norm{\nabla f_i(x)-\nabla f_i(y)}\le L\norm{x-y},\qquad\forall x,y\in\R^{d}.
\tag{2}
\]

SAGA maintains a table of past points $\{\phi_i\}_{i=1}^{n}$ and the corresponding stored gradients
\[
g_i^{\phi}:=\nabla f_i(\phi_i).
\]

At iteration $k$ a random index $i_k\in\{1,\dots,n\}$ is drawn uniformly.  
Define the \emph{control variate}  

\[
\tilde{g}_k:=\nabla f_{i_k}(x_k)-\nabla f_{i_k}(\phi_{i_k})+\frac1n\sum_{j=1}^{n}g_j^{\phi}.
\tag{3}
\]

The update rule is  

\[
x_{k+1}=x_k-\alpha\,\tilde{g}_k,
\qquad\text{with stepsize }\alpha>0.
\tag{4}
\]

Finally the table entry for the sampled index is refreshed  

\[
\phi_{i_k}\leftarrow x_k,\qquad 
g_{i_k}^{\phi}\leftarrow \nabla f_{i_k}(x_k).
\tag{5}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\alpha>0$, number of epochs $T$.
\STATE Initialize $x_0\in\R^{d}$ arbitrarily.
\FOR{$i=1,\dots,n$}
    \STATE $\phi_i\gets x_0$ \COMMENT{initial table points}
    \STATE $g_i^{\phi}\gets \nabla f_i(\phi_i)$
\ENDFOR
\STATE $\bar g\gets \frac1n\sum_{i=1}^{n} g_i^{\phi}$ \COMMENT{average stored gradient}
\FOR{$k=0$ \TO $T-1$}
    \STATE Sample $i_k\sim \mathrm{Unif}\{1,\dots,n\}$
    \STATE $\tilde g_k\gets \nabla f_{i_k}(x_k)-g_{i_k}^{\phi}+\bar g$
    \STATE $x_{k+1}\gets x_k-\alpha\,\tilde g_k$
    \STATE % update table
    \STATE $\bar g\gets \bar g -\frac{1}{n}g_{i_k}^{\phi}+ \frac{1}{n}\nabla f_{i_k}(x_k)$
    \STATE $g_{i_k}^{\phi}\gets \nabla f_{i_k}(x_k)$,\; $\phi_{i_k}\gets x_k$
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or an averaged iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single component ($n=1$) quadratic  

\[
f_1(w)=(w-3)^2,
\qquad \nabla f_1(w)=2(w-3).
\]

Take $w_0=0$, stepsize $\alpha=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f_1(w_k)$ & $f_1(w_k)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3689$
\end{tabular}
\end{center}

Since $n=1$, the table always stores the current point, and the SAGA update reduces to ordinary gradient descent, as illustrated.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex}
Each $f_i$ is convex and $L$‑smooth (cf. (2)). Consequently $f$ in (1) is convex and $L$‑smooth.
\end{assumption}
\begin{assumption}[Finite Sum]\label{ass:finite}
The number of components $n$ is finite and known.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:stepsize}
The stepsize satisfies 
\[
0<\alpha\le\frac{1}{3L}.
\tag{6}
\]
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{ass:strong}
If additionally $f$ is $\mu$‑strongly convex,
\[
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\norm{y-x}^2,\qquad\forall x,y,
\tag{7}
\]
we denote $\kappa:=L/\mu$ the condition number.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA]\label{thm:conv}
Let Assumptions \ref{ass:convex}–\ref{ass:stepsize} hold.
\begin{enumerate}[label=(\roman*)]
\item \textbf{Convex case.} For the averaged iterate $\bar x_T:=\frac1T\sum_{k=0}^{T-1}x_k$,
\[
\E\bigl[f(\bar x_T)-f(x^\star)\bigr]\;\le\;
\frac{2\bigl(\|x_0-x^\star\|^2+ \tfrac{1}{n}\sum_{i=1}^n\|\phi_i^{0}-x^\star\|^2\bigr)}{\alpha T},
\tag{8}
\]
so the expected suboptimality decays as $O(1/T)$.

\item \textbf{Strongly convex case.} If Assumption \ref{ass:strong} also holds, then for every $k\ge0$
\[
\E\bigl[f(x_k)-f(x^\star)\bigr]\le
\bigl(1-\alpha\mu\bigr)^{k}\Bigl(
f(x_0)-f(x^\star)+\tfrac{L}{2n}\sum_{i=1}^{n}\|\phi_i^{0}-x^\star\|^2\Bigr),
\tag{9}
\]
i.e. a linear (geometric) convergence rate with factor $1-\alpha\mu$.
\end{enumerate}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Variance reduction.} The control variate (3) yields an unbiased estimator,
\[
\E_{i_k}\bigl[\tilde g_k\mid\mathcal F_k\bigr]=\nabla f(x_k),
\tag{10}
\]
and its conditional variance diminishes as the table entries approach the optimum.
\item \textbf{Stability w.r.t.\ stepsize.} Condition (6) guarantees that the Lyapunov function (defined below) is contractive. Larger stepsizes break the contraction and may cause divergence.
\item \textbf{Comparison.} Compared with plain SGD, SAGA attains the optimal $O(1/T)$ rate for convex smooth problems without a diminishing stepsize. In the strongly convex regime it matches the $O\bigl((1-\alpha\mu)^T\bigr)$ rate of variance‑reduced methods such as SVRG and Finito.
\end{itemize}

\section*{Discussion}
The theorem shows that SAGA enjoys both the simplicity of stochastic updates and the fast convergence of full‑gradient methods. The linear rate (9) is attained with a constant stepsize independent of $n$, while the $O(1/T)$ rate (8) holds without any strong convexity assumption. The analysis hinges on a Lyapunov function coupling the distance to the optimum and the stored table errors; the detailed derivation is presented next.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Filtration]
Let $\mathcal F_k$ denote the $\sigma$‑algebra generated by $\{x_0,\phi_1^{0},\dots,\phi_n^{0},i_0,\dots,i_{k-1}\}$. All expectations $\E[\cdot\mid\mathcal F_k]$ are taken with respect to the random index $i_k$ conditioned on the past.
\end{definition}
\begin{lemma}[Unbiasedness]\label{lem:unbiased}
For any $k$, $\E_{i_k}[\tilde g_k\mid\mathcal F_k]=\nabla f(x_k)$.
\end{lemma}
\begin{proof}
\begin{align*}
\E_{i_k}[\tilde g_k\mid\mathcal F_k]
&=\frac1n\sum_{i=1}^{n}\Bigl(\nabla f_i(x_k)-\nabla f_i(\phi_i)+\frac1n\sum_{j=1}^{n}\nabla f_j(\phi_j)\Bigr)\\
&=\frac1n\sum_{i=1}^{n}\nabla f_i(x_k)-\frac1n\sum_{i=1}^{n}\nabla f_i(\phi_i)+\frac1n\sum_{j=1}^{n}\nabla f_j(\phi_j)\\
&=\frac1n\sum_{i=1}^{n}\nabla f_i(x_k)=\nabla f(x_k).
\end{align*}
\end{proof}
\begin{lemma}[Quadratic bound for smooth functions]\label{lem:smooth}
If each $f_i$ is $L$‑smooth then for any $x,y$,
\[
f_i(y)\le f_i(x)+\langle\nabla f_i(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2,
\tag{11}
\]
and also
\[
\|\nabla f_i(y)-\nabla f_i(x)\|^2\le 2L\bigl(f_i(y)-f_i(x)-\langle\nabla f_i(x),y-x\rangle\bigr).
\tag{12}
\]
\end{lemma}
\begin{proof}
Both statements are standard consequences of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}. Throughout the proof we use the Lyapunov function  

\[
\Psi_k:=\|x_k-x^\star\|^2+\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2.
\tag{13}
\]

\subsection*{Step 1 – Expansion of the distance term}
\begin{align}
\|x_{k+1}-x^\star\|^2
&=\|x_k-\alpha\tilde g_k-x^\star\|^2 \nonumber\\
&=\|x_k-x^\star\|^2-2\alpha\langle\tilde g_k,x_k-x^\star\rangle+\alpha^2\|\tilde g_k\|^2.
\tag{14}
\end{align}
\noindent\textbf{[Step 1]}

\subsection*{Step 2 – Take expectation conditioned on $\mathcal F_k$}
Using Lemma~\ref{lem:unbiased} and the law of total expectation,
\[
\E\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal F_k\bigr]
= \|x_k-x^\star\|^2-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
+\alpha^2\E\bigl[\|\tilde g_k\|^2\mid\mathcal F_k\bigr].
\tag{15}
\]
\textbf{[Step 2]}

\subsection*{Step 3 – Bound the second‑moment of $\tilde g_k$}
We write $\tilde g_k$ as a sum of three terms:
\[
\tilde g_k=\underbrace{\nabla f_{i_k}(x_k)-\nabla f_{i_k}(x^\star)}_{a_k}
+\underbrace{\nabla f_{i_k}(x^\star)-\nabla f_{i_k}(\phi_{i_k})}_{b_k}
+\underbrace{\frac1n\sum_{j=1}^{n}\nabla f_j(\phi_j)}_{c_k}.
\]
Hence
\[
\|\tilde g_k\|^2\le 3\bigl(\|a_k\|^2+\|b_k\|^2+\|c_k\|^2\bigr).
\tag{16}
\]
\textbf{[Step 3]}

\subsection*{Step 4 – Expectation of the three parts}
Because $i_k$ is uniform,
\[
\E_{i_k}[a_k\mid\mathcal F_k]=\frac1n\sum_{i=1}^{n}\bigl(\nabla f_i(x_k)-\nabla f_i(x^\star)\bigr)=\nabla f(x_k)-\nabla f(x^\star).
\]
Moreover, by smoothness,
\[
\|a_k\|^2\le 2L\bigl(f_{i_k}(x_k)-f_{i_k}(x^\star)-\langle\nabla f_{i_k}(x^\star),x_k-x^\star\rangle\bigr).
\tag{17}
\]
Taking expectation and summing over $i$ yields
\[
\E_{i_k}[\|a_k\|^2\mid\mathcal F_k]\le 2L\bigl(f(x_k)-f(x^\star)\bigr).
\tag{18}
\]
\textbf{[Step 4]}

Similarly,
\[
\|b_k\|^2=\|\nabla f_{i_k}(\phi_{i_k})-\nabla f_{i_k}(x^\star)\|^2
\le 2L\bigl(f_{i_k}(\phi_{i_k})-f_{i_k}(x^\star)-\langle\nabla f_{i_k}(x^\star),\phi_{i_k}-x^\star\rangle\bigr),
\]
and after expectation,
\[
\E_{i_k}[\|b_k\|^2\mid\mathcal F_k]\le
\frac{2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr).
\tag{19}
\]

For the third term $c_k$ we use the bound $\|c_k\|^2\le\frac{1}{n}\sum_{j=1}^{n}\|\nabla f_j(\phi_j)\|^2$ and smoothness again to obtain  
\[
\|c_k\|^2\le\frac{2L}{n}\sum_{j=1}^{n}\bigl(f_j(\phi_j)-f_j(x^\star)-\langle\nabla f_j(x^\star),\phi_j-x^\star\rangle\bigr).
\tag{20}
\]

Collecting (18)–(20) and using (16) gives
\[
\E_{i_k}[\|\tilde g_k\|^2\mid\mathcal F_k]\le
6L\Bigl(f(x_k)-f(x^\star)\Bigr)
+\frac{6L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr).
\tag{21}
\]
\textbf{[Step 5]}

\subsection*{Step 5 – Plug (21) into (15)}
\begin{align}
\E\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal F_k\bigr]
&\le \|x_k-x^\star\|^2-2\alpha\langle\nabla f(x_k),x_k-x^\star\rangle
+6\alpha^2L\bigl(f(x_k)-f(x^\star)\bigr) \nonumber\\
&\quad+\frac{6\alpha^2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr).
\tag{22}
\end{align}
\textbf{[Step 6]}

\subsection*{Step 6 – Use convexity to replace the inner product}
Convexity of $f$ implies
\[
f(x_k)-f(x^\star)\le\langle\nabla f(x_k),x_k-x^\star\rangle.
\tag{23}
\]
Insert (23) into (22) and rearrange:
\begin{align}
\E\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal F_k\bigr]
&\le \|x_k-x^\star\|^2-(2\alpha-6\alpha^2L)\bigl(f(x_k)-f(x^\star)\bigr) \nonumber\\
&\quad+\frac{6\alpha^2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr).
\tag{24}
\end{align}
\textbf{[Step 7]}

\subsection*{Step 7 – Update of the table error term}
Recall that only the entry $i_k$ changes:
\[
\phi_{i_k}^{k+1}=x_k,\qquad\phi_j^{k+1}=\phi_j^{k}\;(j\neq i_k).
\]
Hence
\[
\sum_{i=1}^{n}\|\phi_i^{k+1}-x^\star\|^2
= \sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2
- \|\phi_{i_k}^{k}-x^\star\|^2 +\|x_k-x^\star\|^2.
\tag{25}
\]

Taking conditional expectation and using uniform sampling,
\[
\E\Bigl[\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k+1}-x^\star\|^2\mid\mathcal F_k\Bigr]
= \frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k}-x^\star\|^2
-\frac{\alpha}{n}\bigl\|\phi_{i_k}^{k}-x^\star\bigr\|^2
+\frac{\alpha}{n}\|x_k-x^\star\|^2.
\tag{26}
\]

\subsection*{Step 8 – Combine distance and table parts}
Add (24) and (26) to obtain a recursion for the Lyapunov function (13):
\begin{align}
\E[\Psi_{k+1}\mid\mathcal F_k]
&\le \Psi_k-(2\alpha-6\alpha^2L)\bigl(f(x_k)-f(x^\star)\bigr)
+\frac{6\alpha^2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr)
\nonumber\\
&\quad -\frac{\alpha}{n}\bigl\|\phi_{i_k}^{k}-x^\star\bigr\|^2
+\frac{\alpha}{n}\|x_k-x^\star\|^2.
\tag{27}
\end{align}
\textbf{[Step 8]}

\subsection*{Step 9 – Relate function values to distances}
For each $i$, $L$‑smoothness and convexity give
\[
f_i(\phi_i)-f_i(x^\star)\le\langle\nabla f_i(\phi_i),\phi_i-x^\star\rangle
\le \frac{L}{2}\|\phi_i-x^\star\|^2,
\tag{28}
\]
where the last inequality follows from $\|\nabla f_i(\phi_i)-\nabla f_i(x^\star)\|\le L\|\phi_i-x^\star\|$ and the fact that $\nabla f_i(x^\star)$ belongs to the subgradient at the optimum (zero for differentiable case). Hence
\[
\frac{6\alpha^2L}{n}\sum_{i=1}^{n}\bigl(f_i(\phi_i)-f_i(x^\star)\bigr)
\le 3\alpha^2L^2\frac{1}{n}\sum_{i=1}^{n}\|\phi_i-x^\star\|^2.
\tag{29}
\]

Similarly,
\[
\frac{\alpha}{n}\|x_k-x^\star\|^2\le\frac{\alpha}{2}\|x_k-x^\star\|^2+\frac{\alpha}{2n}\sum_{i=1}^{n}\|\phi_i-x^\star\|^2.
\tag{30}
\]

Insert (29) and (30) into (27):
\begin{align}
\E[\Psi_{k+1}\mid\mathcal F_k]
&\le \Bigl(1-\frac{\alpha}{2}\Bigr)\|x_k-x^\star\|^2
+\Bigl(1+\frac{\alpha}{2}+3\alpha^2L^2\Bigr)\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i-x^\star\|^2 \nonumber\\
&\quad -(2\alpha-6\alpha^2L)\bigl(f(x_k)-f(x^\star)\bigr)
-\frac{\alpha}{n}\bigl\|\phi_{i_k}^{k}-x^\star\bigr\|^2.
\tag{31}
\end{align}
\textbf{[Step 9]}

\subsection*{Step 10 – Choose stepsize}
Assumption \eqref{ass:stepsize} guarantees $0<\alpha\le\frac{1}{3L}$, which implies $2\alpha-6\alpha^2L\ge\alpha$. Moreover,
\[
1+\frac{\alpha}{2}+3\alpha^2L^2\le 1+\frac{1}{6}+ \frac{1}{3}= \frac{5}{3}<2.
\]
Thus (31) simplifies to
\[
\E[\Psi_{k+1}\mid\mathcal F_k]\le \Psi_k-\alpha\bigl(f(x_k)-f(x^\star)\bigr).
\tag{32}
\]
\textbf{[Step 10]}

\subsection*{Step 11 – Telescoping sum}
Taking total expectation and summing (32) for $k=0,\dots,T-1$ yields
\[
\E[\Psi_T]\le \Psi_0-\alpha\sum_{k=0}^{T-1}\E\bigl[f(x_k)-f(x^\star)\bigr].
\tag{33}
\]
Since $\Psi_T\ge0$, we obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[f(x_k)-f(x^\star)\bigr]\le\frac{\Psi_0}{\alpha T}.
\tag{34}
\]

Using convexity of $f$, the average iterate $\bar x_T$ satisfies
\[
\E\bigl[f(\bar x_T)-f(x^\star)\bigr]\le\frac{1}{T}\sum_{k=0}^{T-1}\E\bigl[f(x_k)-f(x^\star)\bigr],
\]
and with the explicit expression of $\Psi_0$ we recover (8). This proves part (i) of Theorem \ref{thm:conv}.

\subsection*{Step 12 – Strongly convex case}
When $f$ is $\mu$‑strongly convex, (23) can be sharpened to
\[
f(x_k)-f(x^\star)\ge\frac{\mu}{2}\|x_k-x^\star\|^2.
\tag{35}
\]
Insert (35) into (32):
\[
\E[\Psi_{k+1}\mid\mathcal F_k]\le (1-\alpha\mu)\Psi_k.
\tag{36}
\]
Iterating (36) gives
\[
\E[\Psi_k]\le (1-\alpha\mu)^{k}\Psi_0.
\tag{37}
\]
Since $\Psi_k\ge\|x_k-x^\star\|^2$, and by smoothness $f(x_k)-f(x^\star)\le\frac{L}{2}\|x_k-x^\star\|^2$, we obtain (9). This concludes the proof of Theorem \ref{thm:conv}. \qed

\section*{Rate Derivation (With Constants)}
From (34) we have the explicit constant
\[
C:=\|x_0-x^\star\|^2+\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{0}-x^\star\|^2,
\]
so the convex-rate reads
\[
\E\bigl[f(\bar x_T)-f(x^\star)\bigr]\le\frac{C}{\alpha T}.
\]
Choosing $\alpha=\frac{1}{3L}$ (the largest admissible stepsize) gives
\[
\E\bigl[f(\bar x_T)-f(x^\star)\bigr]\le\frac{3L\,C}{T}.
\]

In the strongly convex regime, (37) together with $f(x_k)-f(x^\star)\le\frac{L}{2}\|x_k-x^\star\|^2$ yields
\[
\E\bigl[f(x_k)-f(x^\star)\bigr]\le
\frac{L}{2}(1-\alpha\mu)^{k}\Psi_0,
\qquad\alpha\le\frac{1}{3L}.
\]
Thus the linear convergence factor is $1-\alpha\mu\ge 1-\frac{\mu}{3L}=1-\frac{1}{3\kappa}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full gradient ($n=1$).} The table degenerates to a single stored point; SAGA coincides with standard gradient descent and the rates reduce to the classic $O(1/k)$ and $O((1-\alpha\mu)^k)$ bounds.
\item \textbf{Large $n$ limit.} When $n\gg1$, the variance reduction term $\frac{1}{n}\sum_i\|\phi_i-x^\star\|^2$ becomes negligible after a few passes over the data, explaining the empirical observation that SAGA behaves like a deterministic method after one epoch.
\item \textbf{Relation to SVRG.} If the table is reset to the current iterate every $m$ steps, the recursion (32) still holds with an extra $O(\alpha^2L^2/m)$ term, reproducing the SVRG complexity $O((n+\kappa)\log(1/\varepsilon))$.
\end{itemize}
