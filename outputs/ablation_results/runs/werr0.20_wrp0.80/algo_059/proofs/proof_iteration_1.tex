\section*{Algorithm Name}
\textbf{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the (possibly non‑convex) objective and let $\xi_t$ denote a random data sample at iteration $t$.  
Define the stochastic gradient 
\[
g_t \;:=\; \nabla f(w_t;\xi_t).
\]
A (random) quantizer $Q:\mathbb{R}^d\rightarrow\mathbb{R}^d$ satisfies the unbiasedness property
\[
\mathbb{E}\!\left[\,Q(g_t)\mid g_t\right]=g_t .
\]
The QSGD update with step size $\eta_t>0$ is
\[
\boxed{\,w_{t+1}=w_t-\eta_t\,Q(g_t)\,}.
\]
The algorithm communicates only the compressed vector $Q(g_t)$ instead of the full gradient $g_t$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\mathbb{R}^d$, step‑size schedule $\{\eta_t\}_{t\ge0}$, quantizer $Q(\cdot)$
\For{$t=0,1,\dots,T-1$}
    \State Sample a data point (or mini‑batch) $\xi_t$
    \State Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$
    \State Quantize: $\tilde g_t = Q(g_t)$
    \State Update: $w_{t+1}=w_t-\eta_t \tilde g_t$
\EndFor
\State \textbf{Output:} $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple scalar function $f(w)=(w-3)^2$.
\begin{itemize}
    \item Gradient: $\nabla f(w)=2(w-3)$.
    \item Choose a deterministic “quantizer’’ $Q(g)=g$ (so $Q$ is unbiased trivially).
    \item Step size $\eta=0.1$, initial point $w_0=0$.
\end{itemize}
\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(w_t)$ & $w_{t+1}=w_t-\eta Q(g_t)$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^2=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^2=2.3685$\\
\end{tabular}
\end{center}
After three iterations the objective value has dropped from $9$ to approximately $2.37$.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}]
    \item \textbf{Smoothness.} $f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
    \[
    f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
    \tag{A1}
    \]
    \item \textbf{Lower Boundedness.} $f^\star:=\inf_{w}f(w)>-\infty$.
    \item \textbf{Unbiased Stochastic Gradient and Bounded Variance.}
    \[
    \mathbb{E}[g_t\mid w_t]=\nabla f(w_t),\qquad
    \mathbb{E}\!\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]\le\sigma^2 .
    \tag{A2}
    \]
    \item \textbf{Unbiased Quantizer with Bounded Relative Error.} For some $\omega\ge0$,
    \[
    \mathbb{E}[Q(g_t)\mid g_t]=g_t,\qquad
    \mathbb{E}\!\big[\|Q(g_t)-g_t\|^2\mid g_t\big]\le\omega\|g_t\|^2 .
    \tag{A3}
    \]
    \item \textbf{Step‑size.} The step size is constant $\eta_t\equiv\eta$ satisfying
    \[
    0<\eta\le\frac{1}{L(1+\omega)} .
    \tag{A4}
    \]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD]\label{thm:main}
Under assumptions A1–A4, for any $T\ge1$ the iterates generated by QSGD satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{2\big(f(w_0)-f^\star\big)}{\eta T}
\;+\;
L\eta\,\sigma^2(1+\omega) .
\tag{1}
\]
Consequently, choosing $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields the rate
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
= O\!\big(T^{-1/2}\big) .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label={P\arabic*}]
    \item \textbf{Stability.} The bound (1) holds for any $L$‑smooth non‑convex $f$; the algorithm never diverges as long as $\eta$ respects (A4).
    \item \textbf{Effect of Quantization.} The term $\omega$ appears only multiplicatively with $L\eta\sigma^2$, showing that a quantizer with small relative error ($\omega\approx0$) behaves like vanilla SGD.
    \item \textbf{Hyper‑parameter Sensitivity.} The admissible step size shrinks by a factor $1/(1+\omega)$. Larger quantization error forces a more conservative learning rate.
    \item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item With $\omega=0$ (no compression) Theorem~\ref{thm:main} reduces to the classic SGD bound.
        \item For stochastic quantizers such as stochastic rounding, $\omega$ is a constant (e.g., $\omega\le \frac{d}{s^2}$ for $s$‑bit quantization) and thus the $O(T^{-1/2})$ rate is retained.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem demonstrates that unbiased compression does not alter the asymptotic convergence order for smooth non‑convex objectives. The price paid is a larger constant factor proportional to $(1+\omega)$, which can be mitigated by using finer quantization (smaller $\omega$) or by decreasing the step size.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\mathbb{R}^d$,
\[
f(y) \le f(x) + \langle\nabla f(x),y-x\rangle + \frac{L}{2}\|y-x\|^2 .
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness (integrating the Lipschitz gradient bound). \hfill $\square$
\end{proof}

\begin{lemma}[Variance Decomposition]\label{lem:var}
For any random vector $u$ with $\mathbb{E}[u]=\mu$,
\[
\mathbb{E}\|u\|^2 = \|\mu\|^2 + \mathbb{E}\|u-\mu\|^2 .
\]
\end{lemma}
\begin{proof}
Expand $\|u\|^2=\|\mu+(u-\mu)\|^2$ and take expectation. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken w.r.t. the randomness of the sample $\xi_t$ and the quantizer $Q(\cdot)$, conditioned on the history up to iteration $t$ (denoted $\mathcal{F}_t$).

\subsection*{Step 1: Apply Smoothness}
Using Lemma~\ref{lem:smooth} with $x=w_t$, $y=w_{t+1}=w_t-\eta Q(g_t)$,
\[
f(w_{t+1}) \le f(w_t) -\eta\langle\nabla f(w_t), Q(g_t)\rangle + \frac{L\eta^2}{2}\|Q(g_t)\|^2 .
\tag{2}
\]

\subsection*{Step 2: Take Conditional Expectation}
Condition on $\mathcal{F}_t$ and employ unbiasedness of $Q$ (A3) and $g_t$ (A2):
\[
\begin{aligned}
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t] 
&\le f(w_t) -\eta\langle\nabla f(w_t), \mathbb{E}[Q(g_t)\mid\mathcal{F}_t]\rangle
     + \frac{L\eta^2}{2}\mathbb{E}\!\big[\|Q(g_t)\|^2\mid\mathcal{F}_t\big] \\[4pt]
&= f(w_t) -\eta\|\nabla f(w_t)\|^2
     + \frac{L\eta^2}{2}\mathbb{E}\!\big[\|Q(g_t)\|^2\mid\mathcal{F}_t\big] .
\end{aligned}
\tag{3}
\]

\subsection*{Step 3: Bound $\mathbb{E}\|Q(g_t)\|^2$}
From Lemma~\ref{lem:var} and A3,
\[
\mathbb{E}\!\big[\|Q(g_t)\|^2\mid\mathcal{F}_t\big]
   = \|g_t\|^2 + \mathbb{E}\!\big[\|Q(g_t)-g_t\|^2\mid\mathcal{F}_t\big]
   \le \|g_t\|^2 + \omega\|g_t\|^2
   = (1+\omega)\|g_t\|^2 .
\tag{4}
\]

\subsection*{Step 4: Decompose $\|g_t\|^2$}
Using Lemma~\ref{lem:var} again with $g_t=\nabla f(w_t)+\zeta_t$, where $\zeta_t:=g_t-\nabla f(w_t)$,
\[
\mathbb{E}\!\big[\|g_t\|^2\mid\mathcal{F}_t\big]
   = \|\nabla f(w_t)\|^2 + \mathbb{E}\!\big[\|\zeta_t\|^2\mid\mathcal{F}_t\big]
   \le \|\nabla f(w_t)\|^2 + \sigma^2 ,
\tag{5}
\]
where the inequality follows from the variance bound in A2.

\subsection*{Step 5: Plug (4)–(5) into (3)}
\[
\begin{aligned}
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
&\le f(w_t) -\eta\|\nabla f(w_t)\|^2
   + \frac{L\eta^2}{2}(1+\omega)\big(\|\nabla f(w_t)\|^2+\sigma^2\big) \\[4pt]
&= f(w_t) -\underbrace{\Big(\eta - \frac{L\eta^2}{2}(1+\omega)\Big)}_{\displaystyle\alpha}
      \|\nabla f(w_t)\|^2
   + \frac{L\eta^2}{2}(1+\omega)\sigma^2 .
\end{aligned}
\tag{6}
\]

\subsection*{Step 6: Choose $\eta$ to make $\alpha\ge\frac{\eta}{2}$}
Assumption A4 guarantees $0<\eta\le\frac{1}{L(1+\omega)}$, hence
\[
\eta - \frac{L\eta^2}{2}(1+\omega) 
\;\ge\; \frac{\eta}{2} .
\tag{7}
\]
Thus (6) simplifies to
\[
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le f(w_t) - \frac{\eta}{2}\|\nabla f(w_t)\|^2
    + \frac{L\eta^2}{2}(1+\omega)\sigma^2 .
\tag{8}
\]

\subsection*{Step 7: Take Full Expectation and Telescoping}
Take expectation over $\mathcal{F}_t$ and sum from $t=0$ to $T-1$:
\[
\begin{aligned}
\mathbb{E}[f(w_T)]
&\le f(w_0) - \frac{\eta}{2}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
   + \frac{L\eta^2}{2}(1+\omega)\sigma^2 T .
\end{aligned}
\tag{9}
\]
Rearranging and using $f(w_T)\ge f^\star$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2\big(f(w_0)-f^\star\big)}{\eta T}
   + L\eta\,\sigma^2(1+\omega) .
\tag{10}
\]
Equation~(10) is exactly the bound (1) claimed in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
From (10), set $\eta = \frac{c}{\sqrt{T}}$, where $c\le \frac{1}{L(1+\omega)}$ to respect A4. Substituting,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2\big(f(w_0)-f^\star\big)}{c}\,T^{-1/2}
   + Lc\,\sigma^2(1+\omega)T^{-1/2}
= O\!\big(T^{-1/2}\big).
\]
Thus QSGD attains the standard $O(T^{-1/2})$ convergence rate for smooth non‑convex optimization.

\section*{Special Cases}
\begin{enumerate}[label={S\arabic*}]
    \item \textbf{No Quantization ($\omega=0$).} The bound collapses to the classic SGD result.
    \item \textbf{Deterministic Quantization with Zero Error.} If $Q(g)=g$ almost surely, then $\omega=0$ and the algorithm behaves identically to SGD.
    \item \textbf{High‑Precision Quantization.} For $s$‑bit stochastic rounding, one can show $\omega\le \frac{d}{2^{2s}}$. Plugging this into (10) quantifies precisely how many bits are needed to keep the extra constant term negligible.
\end{enumerate}
