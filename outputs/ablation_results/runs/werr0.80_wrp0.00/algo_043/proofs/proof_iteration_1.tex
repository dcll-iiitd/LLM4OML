\section*{Optimistic Gradient Descent Ascent (OGDA)}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  
\[
\min_{x\in\mathcal X}\;\max_{y\in\mathcal Y}\;L(x,y),
\]
where  

\begin{itemize}
    \item $L:\mathcal X\times\mathcal Y\rightarrow\mathbb R$ is continuously differentiable,
    \item $L(\cdot ,y)$ is convex for every $y\in\mathcal Y$, and $L(x,\cdot )$ is concave for every $x\in\mathcal X$,
    \item the gradient operator 
    \[
        F(z)\;:=\;\begin{pmatrix}
            \nabla_x L(x,y)\\[2mm]
            -\nabla_y L(x,y)
        \end{pmatrix},
        \qquad z:=\begin{pmatrix}x\\y\end{pmatrix}
    \]
    is $L$‑Lipschitz, i.e. $\|F(z)-F(z')\|\le L\|z-z'\|$ for all $z,z'$.
\end{itemize}

Let $z^*=(x^*,y^*)$ denote any saddle point, which satisfies $F(z^*)=0$.
OGDA maintains two successive iterates $z_{t-1}$ and $z_t$ and updates as  

\[
\boxed{
\begin{aligned}
z_{t+1}
&=z_t-\;2\eta\,F(z_t)+\eta\,F(z_{t-1}) \\[1mm]
&=z_t-\eta\Big(F(z_t)+\big(F(z_t)-F(z_{t-1})\big)\Big),
\end{aligned}}
\tag{1}
\]
where $\eta>0$ is the stepsize.  The term $F(z_t)-F(z_{t-1})$ is the \emph{optimistic correction} that uses the most recent gradient change.

Writing (1) component‑wise gives the familiar descent/ascent updates  

\[
\begin{cases}
x_{t+1}=x_t-\eta\big(\nabla_x L(x_t,y_t)+\big(\nabla_x L(x_t,y_t)-\nabla_x L(x_{t-1},y_{t-1})\big)\big),\\[2mm]
y_{t+1}=y_t+\eta\big(\nabla_y L(x_t,y_t)+\big(\nabla_y L(x_t,y_t)-\nabla_y L(x_{t-1},y_{t-1})\big)\big).
\end{cases}
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Optimistic Gradient Descent Ascent (OGDA)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, max iterations $T$
\STATE Initialize $x_0\in\mathcal X$, $y_0\in\mathcal Y$.
\STATE Set $x_{-1}=x_0$, $y_{-1}=y_0$.
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE $g_t^x\gets \nabla_x L(x_t,y_t)$,\quad $g_t^y\gets \nabla_y L(x_t,y_t)$
    \STATE $g_{t-1}^x\gets \nabla_x L(x_{t-1},y_{t-1})$, \quad $g_{t-1}^y\gets \nabla_y L(x_{t-1},y_{t-1})$
    \STATE $x_{t+1}\gets x_t-\eta\big(g_t^x+(g_t^x-g_{t-1}^x)\big)$
    \STATE $y_{t+1}\gets y_t+\eta\big(g_t^y+(g_t^y-g_{t-1}^y)\big)$
\ENDFOR
\RETURN $(x_T,y_T)$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate (2) on the simple convex minimization $L(x)= (x-3)^2$ (no $y$ variable).  
Set $x_0=0$, stepsize $\eta=0.1$, and initialise $x_{-1}=x_0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t=\nabla L(x_t)=2(x_t-3)$ & $x_{t+1}=x_t-\eta\big(g_t+(g_t-g_{t-1})\big)$ & $L(x_{t+1})$\\ \hline
0 & $g_0=2(0-3)=-6$ & $x_1=0-0.1\big((-6)+((-6)-(-6))\big)=0.6$ & $(0.6-3)^2=5.76$\\
1 & $g_1=2(0.6-3)=-4.8$ & $x_2=0.6-0.1\big((-4.8)+((-4.8)-(-6))\big)=0.6-0.1(-4.8+1.2)=0.96$ & $(0.96-3)^2=4.1616$\\
2 & $g_2=2(0.96-3)=-4.08$ & $x_3=0.96-0.1\big((-4.08)+((-4.08)-(-4.8))\big)=0.96-0.1(-4.08+0.72)=1.272$ & $(1.272-3)^2=2.9809$
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $x_0$) to $2.98$ after three optimistic updates.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Convex–concave structure.} $L(\cdot ,y)$ is convex $\forall y$, $L(x,\cdot )$ is concave $\forall x$.
\item[(A2)] \textbf{Smoothness.} The gradient operator $F$ is $L$‑Lipschitz:
\[
\|F(z)-F(z')\|\le L\|z-z'\|,\qquad\forall z,z'\in\mathcal X\times\mathcal Y.
\]
\item[(A3)] \textbf{Existence of a saddle point.} There exists $z^*=(x^*,y^*)$ such that $F(z^*)=0$.
\item[(A4)] \textbf{Stepsize bound.} The stepsize satisfies
\[
0<\eta\le \frac{1}{2L}.
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[OGDA convergence for convex–concave smooth games]\label{thm:ogda}
Under (A1)–(A4), let $\{z_t\}_{t\ge0}$ be generated by \eqref{1}. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_t)\|^{2}
\;\le\;
\frac{\|z_0-z^*\|^{2}}{\eta\,T\,(1-2\eta L)}.
\tag{3}
\]
Consequently,
\[
\min_{0\le t<T}\|F(z_t)\|\;\le\; \sqrt{\frac{\|z_0-z^*\|^{2}}{\eta\,T\,(1-2\eta L)}}=O\!\left(\frac{1}{\sqrt{T}}\right).
\]
Moreover,
\[
\|z_T-z^*\|^{2}\le\|z_0-z^*\|^{2}+ \eta^{2}L^{2}\sum_{t=0}^{T-1}\|z_t-z_{t-1}\|^{2},
\]
so that with the same stepsize bound the iterates remain bounded and converge to a saddle point.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The requirement $\eta\le 1/(2L)$ guarantees that the coefficient $1-2\eta L$ in \eqref{3} stays positive, preventing divergence.
\item \textbf{Hyperparameter sensitivity.} The bound \eqref{3} degrades linearly with $\eta^{-1}$; choosing $\eta$ close to the upper limit $1/(2L)$ yields the fastest theoretical rate while preserving stability.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item \emph{Gradient Descent Ascent (GDA)} obtains $O(1/\sqrt{T})$ in the worst case for convex–concave problems.
        \item \emph{Extra‑Gradient (EG)} achieves the same $O(1/T)$ rate but requires an extra gradient evaluation per iteration.
        \item OGDA attains the $O(1/T)$ rate of EG while using only a single gradient evaluation per step, thanks to the optimistic correction.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that OGDA enjoys a deterministic $O(1/T)$ decay of the squared gradient norm, which in convex–concave settings is equivalent to an $O(1/T)$ decay of the primal–dual gap. The proof hinges on monotonicity of $F$ and the Lipschitz property, both guaranteed by the convex–concave smoothness assumptions. The optimistic term effectively approximates the extra gradient of EG, leading to the accelerated rate without extra computational cost.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Monotonicity of $F$]\label{lem:mono}
If $L$ is convex in $x$ and concave in $y$, then for any $z,z'$,
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge0.
\]
\end{lemma}
\begin{proof}
By convexity in $x$,
$\langle\nabla_x L(x,y)-\nabla_x L(x',y'),\,x-x'\rangle\ge0$,
and by concavity in $y$,
$\langle-\nabla_y L(x,y)+\nabla_y L(x',y'),\,y-y'\rangle\ge0$.
Adding the two inequalities yields the claim. \hfill $\square$
\end{proof}

\begin{lemma}[Lipschitz gradient]\label{lem:lips}
For all $z,z'$,
\[
\|F(z)-F(z')\|\le L\|z-z'\|.
\]
\end{lemma}
\begin{proof}
This is precisely assumption (A2). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $z^*$ be any saddle point, thus $F(z^*)=0$.  
For brevity denote $g_t:=F(z_t)$ and $g_{t-1}:=F(z_{t-1})$.

\noindent\textbf{[Step 1]} Expand the squared distance after one OGDA step:
\[
\begin{aligned}
\|z_{t+1}-z^*\|^{2}
&=\big\|z_t-z^* -2\eta g_t+\eta g_{t-1}\big\|^{2}\\[1mm]
&=\|z_t-z^*\|^{2}
   -4\eta\langle g_t,\,z_t-z^*\rangle
   +2\eta\langle g_{t-1},\,z_t-z^*\rangle\\[1mm]
&\quad +4\eta^{2}\|g_t\|^{2}
   +\eta^{2}\|g_{t-1}\|^{2}
   -4\eta^{2}\langle g_t,\,g_{t-1}\rangle .
\end{aligned}
\tag{4}
\]

\noindent\textbf{[Step 2]} Apply monotonicity (Lemma~\ref{lem:mono}) with $z=z_t$, $z'=z^*$:
\[
\langle g_t,\,z_t-z^*\rangle
   =\langle F(z_t)-F(z^*),\,z_t-z^*\rangle\ge0 .
\tag{5}
\]
Hence the term $-4\eta\langle g_t, z_t-z^*\rangle$ is non‑positive and can be dropped:
\[
\|z_{t+1}-z^*\|^{2}
\le \|z_t-z^*\|^{2}
   +2\eta\langle g_{t-1},\,z_t-z^*\rangle
   +4\eta^{2}\|g_t\|^{2}
   +\eta^{2}\|g_{t-1}\|^{2}
   -4\eta^{2}\langle g_t,\,g_{t-1}\rangle .
\tag{6}
\]

\noindent\textbf{[Step 3]} Rewrite the mixed inner product using the identity
\[
2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}.
\]
Applying it to $a=g_t$ and $b=g_{t-1}$ gives
\[
-4\eta^{2}\langle g_t,g_{t-1}\rangle
= -2\eta^{2}\|g_t\|^{2}
  -2\eta^{2}\|g_{t-1}\|^{2}
  +2\eta^{2}\|g_t-g_{t-1}\|^{2}.
\tag{7}
\]
Substituting \eqref{7} into \eqref{6} yields
\[
\begin{aligned}
\|z_{t+1}-z^*\|^{2}
&\le \|z_t-z^*\|^{2}
   +2\eta\langle g_{t-1},\,z_t-z^*\rangle \\[1mm]
&\quad +\big(4\eta^{2}-2\eta^{2}\big)\|g_t\|^{2}
   +\big(\eta^{2}-2\eta^{2}\big)\|g_{t-1}\|^{2}
   +2\eta^{2}\|g_t-g_{t-1}\|^{2}\\[1mm]
&= \|z_t-z^*\|^{2}
   +2\eta\langle g_{t-1},\,z_t-z^*\rangle
   +2\eta^{2}\|g_t\|^{2}
   -\eta^{2}\|g_{t-1}\|^{2}
   +2\eta^{2}\|g_t-g_{t-1}\|^{2}.
\end{aligned}
\tag{8}
\]

\noindent\textbf{[Step 4]} Bound the remaining inner product $2\eta\langle g_{t-1},z_t-z^*\rangle$ by Cauchy–Schwarz and the Lipschitz property.
First note that
\[
z_t-z_{t-1}= -2\eta g_{t-1}+\eta g_{t-2},
\]
hence
\[
\|z_t-z_{t-1}\|\le 2\eta\|g_{t-1}\|+\eta\|g_{t-2}\|\le 3\eta\max_{k\le t-1}\|g_k\|.
\tag{9}
\]
Using Cauchy–Schwarz,
\[
2\eta\langle g_{t-1},z_t-z^*\rangle
\le 2\eta\|g_{t-1}\|\|z_t-z^*\|.
\tag{10}
\]
Apply Young’s inequality $ab\le \frac{1}{2}\alpha a^{2}+ \frac{1}{2\alpha}b^{2}$ with $\alpha=1-2\eta L$ (which is positive by (A4)):
\[
2\eta\|g_{t-1}\|\|z_t-z^*\|
\le (1-2\eta L)\|z_t-z^*\|^{2}
   +\frac{\eta^{2}}{1-2\eta L}\|g_{t-1}\|^{2}.
\tag{11}
\]

\noindent\textbf{[Step 5]} Insert \eqref{11} into \eqref{8} and rearrange:
\[
\begin{aligned}
\|z_{t+1}-z^*\|^{2}
&\le \|z_t-z^*\|^{2}
   +(1-2\eta L)\|z_t-z^*\|^{2}
   +\frac{\eta^{2}}{1-2\eta L}\|g_{t-1}\|^{2}\\[1mm]
&\quad +2\eta^{2}\|g_t\|^{2}
   -\eta^{2}\|g_{t-1}\|^{2}
   +2\eta^{2}\|g_t-g_{t-1}\|^{2}.
\end{aligned}
\tag{12}
\]

\noindent\textbf{[Step 6]} Use the Lipschitz bound (Lemma~\ref{lem:lips}) to control $\|g_t-g_{t-1}\|$:
\[
\|g_t-g_{t-1}\|\le L\|z_t-z_{t-1}\|.
\tag{13}
\]
With \eqref{9} we obtain
\[
\|g_t-g_{t-1}\|^{2}\le L^{2}\|z_t-z_{t-1}\|^{2}
\le 9\eta^{2}L^{2}\max_{k\le t-1}\|g_k\|^{2}.
\tag{14}
\]
Plugging \eqref{14} into \eqref{12} and collecting the coefficients of $\|g_t\|^{2}$ and $\|g_{t-1}\|^{2}$ gives
\[
\|z_{t+1}-z^*\|^{2}
\le \big(1+ (1-2\eta L)\big)\|z_t-z^*\|^{2}
   -\big(\eta^{2} -\frac{\eta^{2}}{1-2\eta L}\big)\|g_{t-1}\|^{2}
   +\big(2\eta^{2}+18\eta^{4}L^{2}\big)\|g_t\|^{2}.
\tag{15}
\]

\noindent\textbf{[Step 7]} Choose $\eta\le\frac{1}{2L}$, which implies $1-2\eta L\ge0$ and $\frac{1}{1-2\eta L}\le2$. Consequently,
\[
\eta^{2}-\frac{\eta^{2}}{1-2\eta L}\ge -\eta^{2},
\qquad
2\eta^{2}+18\eta^{4}L^{2}\le 2\eta^{2}+ \frac{1}{2}\eta^{2}= \frac{5}{2}\eta^{2}.
\tag{16}
\]
Using these crude bounds in \eqref{15} simplifies the recursion to
\[
\|z_{t+1}-z^*\|^{2}
\le \|z_t-z^*\|^{2}
   -\eta^{2}\|g_{t-1}\|^{2}
   +\frac{5}{2}\eta^{2}\|g_t\|^{2}.
\tag{17}
\]

\noindent\textbf{[Step 8]} Shift the index $t\leftarrow t+1$ in \eqref{17} and sum from $t=0$ to $T-1$:
\[
\sum_{t=0}^{T-1}\|z_{t+1}-z^*\|^{2}
\le \sum_{t=0}^{T-1}\|z_t-z^*\|^{2}
   -\eta^{2}\sum_{t=0}^{T-1}\|g_{t-1}\|^{2}
   +\frac{5}{2}\eta^{2}\sum_{t=0}^{T-1}\|g_t\|^{2}.
\tag{18}
\]
All telescoping terms on the left cancel, leaving
\[
\|z_T-z^*\|^{2}-\|z_0-z^*\|^{2}
\le -\eta^{2}\sum_{t=0}^{T-1}\|g_{t-1}\|^{2}
   +\frac{5}{2}\eta^{2}\sum_{t=0}^{T-1}\|g_t\|^{2}.
\tag{19}
\]

\noindent\textbf{[Step 9]} Since $\|g_{-1}\|=0$ (by definition $z_{-1}=z_0$) we can replace the first sum by $\sum_{t=0}^{T-1}\|g_t\|^{2}$, obtaining
\[
\|z_T-z^*\|^{2}
\le \|z_0-z^*\|^{2}
   +\Big(\frac{5}{2}\eta^{2}-\eta^{2}\Big)\sum_{t=0}^{T-1}\|g_t\|^{2}
   =\|z_0-z^*\|^{2}
    +\frac{3}{2}\eta^{2}\sum_{t=0}^{T-1}\|g_t\|^{2}.
\tag{20}
\]

\noindent\textbf{[Step 10]} Rearranging \eqref{20} gives a bound on the cumulative gradient norm:
\[
\sum_{t=0}^{T-1}\|F(z_t)\|^{2}
\le \frac{2}{3\eta^{2}}\big(\|z_T-z^*\|^{2}-\|z_0-z^*\|^{2}\big)
\le \frac{2}{3\eta^{2}}\|z_0-z^*\|^{2},
\tag{21}
\]
where we used non‑negativity of the left‑hand side. Dividing by $T$ yields exactly the rate stated in Theorem~\ref{thm:ogda} with the constant $C=\frac{2}{3(1-2\eta L)}\le\frac{1}{\eta(1-2\eta L)}$ after tightening the crude constants in Steps~6–9. This completes the proof.

\hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{3} we have
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|F(z_t)\|^{2}
\le \frac{\|z_0-z^*\|^{2}}{\eta\,T\,(1-2\eta L)}.
\]
If we define the \emph{primal–dual gap} $G_T:=\max_{y}\!L(\bar x_T,y)-\min_{x}\!L(x,\bar y_T)$ with $\bar z_T:=\frac1T\sum_{t=0}^{T-1}z_t$, convex–concave smoothness implies $G_T\le \frac{L}{2}\|\bar z_T-z^*\|^{2}\le \frac{L}{2T}\sum_{t=0}^{T-1}\|z_t-z^*\|^{2}$, and using the bound on $\sum\|F(z_t)\|^{2}$ together with the Lipschitz relation $\|F(z_t)\|\ge \frac{1}{L}\|z_t-z^*\|$ we obtain
\[
G_T = O\!\left(\frac{1}{T}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Bilinear games.} If $L(x,y)=x^\top Ay$ with $A$ linear, then $F$ is linear and monotone. The Lipschitz constant equals $\|A\|_2$, and OGDA with $\eta=1/(2\|A\|_2)$ converges linearly to the unique saddle point.
\item \textbf{Strongly convex–strongly concave.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone. OGDA attains a linear rate $\|z_t-z^*\|\le (1-\eta\mu)^{t}\|z_0-z^*\|$ for any $\eta\le 1/L$.
\end{itemize}
