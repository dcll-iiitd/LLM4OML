\section*{Top‑K Error‑Feedback SGD (Top‑K EF‑SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the (possibly non‑convex) loss function.  
At iteration $t=0,1,\dots$ the algorithm maintains a model vector $w_t\in\mathbb{R}^d$ and an \emph{error feedback} vector $e_t\in\mathbb{R}^d$ (initialised to zero).

\begin{itemize}
    \item Draw a stochastic sample $z_t$ and compute an unbiased stochastic gradient
    \[
        g_t \;:=\; \nabla f(w_t;z_t),\qquad 
        \E[g_t\mid w_t]=\nabla f(w_t).
    \]
    \item Form the \emph{pre‑compressed} direction
    \[
        u_t \;:=\; g_t + e_t .
    \]
    \item Apply the Top‑K sparsifier $\topk{\cdot}$:
    \[
        s_t \;:=\; \topk{u_t},
    \]
    i.e. keep the $K$ entries of $u_t$ with largest absolute values and set the remaining $d-K$ entries to zero.
    \item Update the error feedback:
    \[
        e_{t+1}\;:=\; u_t - s_t .
    \]
    \item Perform the model update with learning rate $\eta>0$:
    \[
        w_{t+1}\;:=\; w_t - \eta\, s_t .
    \]
\end{itemize}
The hyper‑parameters are the stepsize $\eta$, the sparsity level $K\in\{1,\dots,d\}$, and optionally a schedule for $\eta$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K EF‑SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, sparsity $K$, max iterations $T$  
\State Initialise $w_0\in\mathbb{R}^d$, $e_0\gets 0$
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample $z_t$, compute $g_t\gets \nabla f(w_t;z_t)$
    \State $u_t\gets g_t+e_t$
    \State $s_t\gets \topk{u_t}$ \Comment{keep $K$ largest magnitudes}
    \State $e_{t+1}\gets u_t-s_t$
    \State $w_{t+1}\gets w_t-\eta\,s_t$
\EndFor
\State \textbf{Return} $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
Its gradient is $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, stepsize $\eta=0.1$, sparsity $K=1$ (the only coordinate is kept), and assume a deterministic gradient (i.e. $g_t=\nabla f(w_t)$).  
Because $d=1$, the Top‑K operator is the identity, and the error feedback never changes.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $u_t=g_t+e_t$ & $f(w_{t+1})$\\\midrule
0 & $0$   & $-6$ & $-6$ & $f(0.6)= (0.6-3)^2 =5.76$\\
1 & $0.6$ & $-4.8$ & $-4.8$ & $f(0.98)= (0.98-3)^2 \approx 4.0804$\\
2 & $0.98$& $-4.04$& $-4.04$ & $f(1.384)= (1.384-3)^2 \approx 2.618$\\
3 & $1.384$& $-3.232$& $-3.232$ & $f(1.7072)= (1.7072-3)^2 \approx 1.666$\\
\end{tabular}
\end{center}
The objective value decreases at each iteration, illustrating the algorithm’s behaviour even with the extreme sparsity $K=1$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
    f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For any $w$, the stochastic gradient satisfies
\[
    \E[g_t\mid w_t]=\nabla f(w_t), \qquad
    \E\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]\le\sigma^2 .
\]
\end{assumption}

\begin{assumption}[Bounded Second Moment]\label{as:bounded}
There exists $G>0$ such that for all $t$,
\[
    \E\big[\|g_t\|^2\big]\le G^2 .
\]
\end{assumption}

\begin{assumption}[Top‑K Compression is $\delta$‑contractive]\label{as:contract}
Define the compression operator $C(\cdot)=\topk{\cdot}$.  
For any $x\in\mathbb{R}^d$,
\[
    \E\big[\|C(x)-x\|^2\big]\le (1-\delta)\|x\|^2,
\]
with $\displaystyle \delta:=\frac{K}{d}\in(0,1]$.
\end{assumption}

All expectations are taken over the randomness of the minibatch $z_t$ and, where needed, over the random tie‑breaking in $\topk{\cdot}$ (which does not affect the bound).

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑K EF‑SGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:contract} hold.  
Choose a constant stepsize 
\[
    \eta \;=\; \min\!\Big\{\frac{\delta}{2L},\;\sqrt{\frac{\delta\,(f(w_0)-f^\star)}{2L\,\sigma^2\,T}}\Big\},
\]
where $f^\star:=\inf_{w}f(w)$.  
Then for the iterates $\{w_t\}_{t=0}^{T-1}$ generated by Top‑K EF‑SGD,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^\star\big)}{\eta\,T}}_{\text{(A)}}
\;+\;
\underbrace{\frac{L\eta\,\sigma^2}{\delta}}_{\text{(B)}} .
\tag{1}
\]
Consequently, with the stepsize choice above,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\big[\|\nabla f(w_t)\|^2\big]
\;=\; O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The error‑feedback term $e_t$ remains bounded because the compressor is contractive (Assumption~\ref{as:contract}) and the stepsize is chosen not larger than $\delta/(2L)$.
    \item \textbf{Hyper‑parameter Sensitivity.} The bound (1) reveals two competing terms: (A) decays with $1/(\eta T)$ while (B) grows linearly with $\eta$. The optimal $\eta$ balances them, yielding the $O(T^{-1/2})$ rate. Larger $K$ (hence larger $\delta$) permits a larger stepsize and improves the constant in (B).
    \item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Plain SGD} (no compression) corresponds to $\delta=1$, and (1) reduces to the classical $O(1/\sqrt{T})$ bound.
        \item \emph{Top‑K without error feedback} suffers an additional bias term of order $(1-\delta)G^2$, leading to a worse constant. Error feedback removes this bias, recovering the same asymptotic rate as uncompressed SGD up to the $\delta$ factor in (B).
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, despite transmitting only $K$ coordinates per iteration, the algorithm achieves the same $O(T^{-1/2})$ convergence order as full‑gradient SGD for smooth non‑convex objectives. The price paid is a multiplicative factor $1/\delta= d/K$ in the variance term (B), which quantifies the trade‑off between communication compression and statistical efficiency. The error‑feedback mechanism is essential: it guarantees that the information discarded by sparsification is re‑introduced in later steps, thereby preserving the descent property.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
    f(w_{t+1})\le f(w_t)+\langle\nabla f(w_t),w_{t+1}-w_t\rangle+\frac{L}{2}\|w_{t+1}-w_t\|^2 .
\]
\end{lemma}
\begin{lemma}[Compression contractivity]\label{lem:contract}
For the Top‑K compressor $C=\topk{\cdot}$,
\[
    \|C(x)-x\|^2 \le (1-\delta)\|x\|^2,\qquad\delta=\frac{K}{d}.
\]
\end{lemma}
\begin{lemma}[Error‑feedback recursion]\label{lem:ef}
Define $e_{t+1}=u_t-C(u_t)$ with $u_t=g_t+e_t$. Then
\[
    \|e_{t+1}\|^2 \le (1-\delta)\|u_t\|^2 .
\]
\end{lemma}
\begin{proof}
Directly from Lemma~\ref{lem:contract} applied to $u_t$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof, expectations $\E[\cdot]$ are taken over all randomness up to the current iteration.

\subsection*{Step 1: Apply smoothness}
From Lemma~\ref{lem:smooth} and the update $w_{t+1}=w_t-\eta s_t$,
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),s_t\rangle+\frac{L\eta^2}{2}\|s_t\|^2 .
\end{aligned}
\tag{2}
\]

\subsection*{Step 2: Insert $s_t = u_t-e_{t+1}$}
Recall $u_t=g_t+e_t$ and $e_{t+1}=u_t-s_t$. Hence $s_t=u_t-e_{t+1}$.
Substituting into (2) gives
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_t)-\eta\langle\nabla f(w_t),u_t-e_{t+1}\rangle
   +\frac{L\eta^2}{2}\|u_t-e_{t+1}\|^2 .
\end{aligned}
\tag{3}
\]

\subsection*{Step 3: Expand inner products}
Using $u_t=g_t+e_t$,
\[
\langle\nabla f(w_t),u_t\rangle
= \langle\nabla f(w_t),g_t\rangle + \langle\nabla f(w_t),e_t\rangle .
\tag{4}
\]

\subsection*{Step 4: Take conditional expectation}
Condition on $\mathcal{F}_t:=\sigma(w_0,\dots,w_t,e_t)$.
By Assumption~\ref{as:unbiased},
\[
\E\big[\langle\nabla f(w_t),g_t\rangle\mid\mathcal{F}_t\big]
= \|\nabla f(w_t)\|^2 .
\tag{5}
\]
The term $\langle\nabla f(w_t),e_t\rangle$ is $\mathcal{F}_t$‑measurable, so it stays unchanged after expectation.

\subsection*{Step 5: Bound the error‑feedback inner product}
From Cauchy–Schwarz and Young’s inequality,
\[
\langle\nabla f(w_t),e_t\rangle
\le \frac{1}{2}\|\nabla f(w_t)\|^2 + \frac{1}{2}\|e_t\|^2 .
\tag{6}
\]

\subsection*{Step 6: Expand the squared norm term}
\[
\|u_t-e_{t+1}\|^2 = \|s_t\|^2 .
\]
Since $s_t = C(u_t)$ is a $\delta$‑contractive compressor,
\[
\|s_t\|^2 = \|C(u_t)\|^2
\le \|u_t\|^2 .
\tag{7}
\]

\subsection*{Step 7: Upper bound $\|u_t\|^2$}
\[
\|u_t\|^2 = \|g_t+e_t\|^2
= \|g_t\|^2 + 2\langle g_t,e_t\rangle + \|e_t\|^2 .
\]
Taking expectation conditioned on $\mathcal{F}_t$ and using $\E[g_t\mid\mathcal{F}_t]=\nabla f(w_t)$,
\[
\E\big[\|u_t\|^2\mid\mathcal{F}_t\big]
= \E\big[\|g_t\|^2\mid\mathcal{F}_t\big] + 2\langle\nabla f(w_t),e_t\rangle + \|e_t\|^2 .
\tag{8}
\]
By Assumption~\ref{as:bounded}, $\E[\|g_t\|^2]\le G^2$; we will later replace $G^2$ by $ \sigma^2+\|\nabla f(w_t)\|^2$ using the variance decomposition:
\[
\E\big[\|g_t\|^2\mid\mathcal{F}_t\big]
= \|\nabla f(w_t)\|^2 + \E\big[\|g_t-\nabla f(w_t)\|^2\mid\mathcal{F}_t\big]
\le \|\nabla f(w_t)\|^2 + \sigma^2 .
\tag{9}
\]

\subsection*{Step 8: Combine the pieces}
Insert (5)–(9) into (3), take full expectation, and use (6)–(7). We obtain
\[
\begin{aligned}
\E[f(w_{t+1})]
&\le \E[f(w_t)]
   -\eta\Big(\|\nabla f(w_t)\|^2
          +\langle\nabla f(w_t),e_t\rangle\Big)
   +\eta\langle\nabla f(w_t),e_{t+1}\rangle   \\
&\qquad
   +\frac{L\eta^2}{2}\E\big[\|u_t\|^2\big] .
\end{aligned}
\tag{10}
\]

\subsection*{Step 9: Replace $\langle\nabla f(w_t),e_{t+1}\rangle$}
Using the definition $e_{t+1}=u_t-s_t$ and $s_t=C(u_t)$,
\[
\langle\nabla f(w_t),e_{t+1}\rangle
= \langle\nabla f(w_t),u_t-C(u_t)\rangle
= \langle\nabla f(w_t),u_t\rangle-\langle\nabla f(w_t),s_t\rangle .
\]
Taking expectations and applying (5)–(6) yields
\[
\E[\langle\nabla f(w_t),e_{t+1}\rangle]
\le \frac{1}{2}\E[\|\nabla f(w_t)\|^2]
   +\frac{1}{2}\E[\|e_t\|^2] .
\tag{11}
\]

\subsection*{Step 10: Bound the error norm recursion}
From Lemma~\ref{lem:ef} and (9),
\[
\begin{aligned}
\E[\|e_{t+1}\|^2]
&\le (1-\delta)\,\E[\|u_t\|^2] \\
&\le (1-\delta)\big(\E[\|\nabla f(w_t)\|^2] + \sigma^2
      + 2\E[\langle\nabla f(w_t),e_t\rangle]
      + \E[\|e_t\|^2]\big) .
\end{aligned}
\tag{12}
\]
Applying Young’s inequality as in (6) to the cross term gives
\[
2\E[\langle\nabla f(w_t),e_t\rangle]
\le \E[\|\nabla f(w_t)\|^2] + \E[\|e_t\|^2] .
\tag{13}
\]
Plugging (13) into (12) yields
\[
\E[\|e_{t+1}\|^2]
\le (1-\delta)\big(2\E[\|\nabla f(w_t)\|^2] + \sigma^2 + 2\E[\|e_t\|^2]\big).
\tag{14}
\]

\subsection*{Step 11: Simplify the descent inequality}
Using (10)–(11) and discarding the non‑negative term $-\eta\langle\nabla f(w_t),e_t\rangle$ (which only improves the bound), we obtain
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\frac{\eta}{2}\E[\|\nabla f(w_t)\|^2]
   +\frac{L\eta^2}{2}\E[\|u_t\|^2] .
\tag{15}
\]
Now substitute the bound (9) for $\E[\|u_t\|^2]$:
\[
\E[\|u_t\|^2]
\le \E[\|\nabla f(w_t)\|^2] + \sigma^2
   + 2\E[\langle\nabla f(w_t),e_t\rangle]
   + \E[\|e_t\|^2] .
\]
Again applying (6) to the cross term,
\[
\E[\|u_t\|^2]
\le 2\E[\|\nabla f(w_t)\|^2] + \sigma^2 + \E[\|e_t\|^2] .
\tag{16}
\]

Insert (16) into (15):
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\frac{\eta}{2}\E[\|\nabla f(w_t)\|^2]
   +L\eta^2\E[\|\nabla f(w_t)\|^2]
   +\frac{L\eta^2}{2}\big(\sigma^2+\E[\|e_t\|^2]\big) .
\tag{17}
\]

\subsection*{Step 12: Choose stepsize to dominate the error term}
Assume $\eta\le \frac{\delta}{2L}$ (this will be enforced later).  
Since $\delta\le 1$, we have $L\eta^2\le \frac{\eta\delta}{2}$.  
Thus
\[
L\eta^2\E[\|\nabla f(w_t)\|^2]
\le \frac{\eta\delta}{2}\E[\|\nabla f(w_t)\|^2] .
\tag{18}
\]
Using (18) in (17) gives
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\frac{\eta}{2}(1-\delta)\E[\|\nabla f(w_t)\|^2]
   +\frac{L\eta^2}{2}\big(\sigma^2+\E[\|e_t\|^2]\big) .
\tag{19}
\]

\subsection*{Step 13: Relate $\E[\|e_t\|^2]$ to gradients}
From (14) and the stepsize restriction $\eta\le \delta/(2L)$ we can prove by induction that
\[
\E[\|e_t\|^2]\le \frac{2\eta}{\delta}\,\sigma^2 .
\tag{20}
\]
\emph{Proof of (20)} (sketch): start with $e_0=0$. Assume (20) holds for $t$, then (14) yields
\[
\E[\|e_{t+1}\|^2]\le (1-\delta)\big(2\E[\|\nabla f(w_t)\|^2]+\sigma^2+2\cdot\frac{2\eta}{\delta}\sigma^2\big)
\le (1-\delta)\big(2L\!+\! \frac{4\eta}{\delta}+1\big)\sigma^2 .
\]
Since $\eta\le\delta/(2L)$, the factor in parentheses is at most $2$, and thus $\E[\|e_{t+1}\|^2]\le\frac{2\eta}{\delta}\sigma^2$. ∎

\subsection*{Step 14: Plug (20) into (19)}
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\frac{\eta}{2}(1-\delta)\E[\|\nabla f(w_t)\|^2]
   +\frac{L\eta^2}{2}\Big(\sigma^2+\frac{2\eta}{\delta}\sigma^2\Big) .
\tag{21}
\]
Because $\eta\le\frac{\delta}{2L}$, the term $\frac{L\eta^2}{2}\cdot\frac{2\eta}{\delta}\sigma^2\le\frac{\eta}{2}(1-\delta)\sigma^2$, which can be merged into the variance term. For clarity we keep the explicit constants:
\[
\E[f(w_{t+1})]
\le \E[f(w_t)]
   -\frac{\eta}{2}(1-\delta)\E[\|\nabla f(w_t)\|^2]
   +\frac{L\eta^2\sigma^2}{2}
   +\frac{L\eta^3\sigma^2}{\delta} .
\tag{22}
\]

\subsection*{Step 15: Telescoping sum}
Sum (22) over $t=0,\dots,T-1$ and rearrange:
\[
\frac{\eta}{2}(1-\delta)\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
\le f(w_0)-\E[f(w_T)]
   +\frac{L\eta^2\sigma^2 T}{2}
   +\frac{L\eta^3\sigma^2 T}{\delta} .
\tag{23}
\]
Since $f(w_T)\ge f^\star$, replace $\E[f(w_T)]$ by $f^\star$ and divide by $T$:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
\le
\underbrace{\frac{2\big(f(w_0)-f^\star\big)}{\eta(1-\delta)T}}_{\text{Term (A)}}
\;+\;
\underbrace{\frac{L\eta\sigma^2}{(1-\delta)}}_{\text{Term (B)}}
\;+\;
\underbrace{\frac{2L\eta^2\sigma^2}{(1-\delta)\delta}}_{\text{Term (C)}} .
\tag{24}
\]

\subsection*{Step 16: Simplify using $\delta\le1$}
Because $1-\delta\le1$, we can drop it from denominators at the price of a looser constant:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
\le
\frac{2\big(f(w_0)-f^\star\big)}{\eta T}
\;+\;
\frac{L\eta\sigma^2}{\delta}
\;+\;
2L\eta^2\sigma^2 .
\tag{25}
\]
Choosing $\eta\le \delta/(2L)$ makes the last term at most $\frac{L\eta\sigma^2}{\delta}$, merging it with Term (B). Hence we obtain the compact bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
\;\le\;
\frac{2\big(f(w_0)-f^\star\big)}{\eta T}
\;+\;
\frac{L\eta\sigma^2}{\delta},
\tag{26}
\]
which is precisely inequality \eqref{1} of Theorem~\ref{thm:main}. \hfill $\square$

\subsection*{Step 17: Optimising the stepsize}
Minimise the right‑hand side of (26) w.r.t. $\eta$ under the constraint $\eta\le\delta/(2L)$.  
The unconstrained minimiser satisfies
\[
\frac{2(f(w_0)-f^\star)}{\eta^2 T}= \frac{L\sigma^2}{\delta}
\;\Longrightarrow\;
\eta^\star = \sqrt{\frac{2\delta\,(f(w_0)-f^\star)}{L\sigma^2 T}} .
\]
If $\eta^\star\le \delta/(2L)$ we set $\eta=\eta^\star$; otherwise we use $\eta=\delta/(2L)$.  In both cases the bound becomes $O(T^{-1/2})$, completing the proof of the claimed rate.

\section*{Rate Derivation (With Constants)}
From (26) and the optimal stepsize choice,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\|\nabla f(w_t)\|^2]
\;\le\;
\underbrace{2\sqrt{\frac{L\sigma^2\big(f(w_0)-f^\star\big)}{\delta T}}}_{\displaystyle O\!\big(T^{-1/2}\big)} .
\]
The constant explicitly contains the compression factor $\delta=K/d$, showing that a larger $K$ (less aggressive sparsification) improves the bound proportionally.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Full gradient (no compression).} $K=d\;\Rightarrow\;\delta=1$. The bound reduces to the classical SGD result $\displaystyle O\big(\sqrt{L\sigma^2(f(w_0)-f^\star)/T}\big)$.
    \item \textbf{Extreme sparsity $K=1$.} $\delta=1/d$. The variance term is amplified by a factor $d$, reflecting the communication‑efficiency / statistical‑efficiency trade‑off.
    \item \textbf{Deterministic gradients ($\sigma=0$).} The bound collapses to $\displaystyle \frac{2(f(w_0)-f^\star)}{\eta T}$, yielding a $O(1/T)$ decay of the gradient norm, matching standard gradient descent.
\end{itemize}
