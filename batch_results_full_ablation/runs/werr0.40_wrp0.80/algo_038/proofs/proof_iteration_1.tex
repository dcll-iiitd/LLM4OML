\section*{Primal--Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Consider the convex–concave saddle‑point problem  

\[
\min_{x\in\mathbb{R}^n}\max_{y\in\mathbb{R}^m}
\; \mathcal L (x,y):= f(x)+\langle Kx ,y\rangle - g^{*}(y),
\tag{1}
\]

where  

\begin{itemize}
  \item $f:\mathbb{R}^{n}\to\mathbb{R}\cup\{+\infty\}$ is proper, closed, convex and $L_f$–smooth,
  \item $g^{*}:\mathbb{R}^{m}\to\mathbb{R}\cup\{+\infty\}$ is the convex conjugate of a proper closed convex function $g$,
  \item $K\in\mathbb{R}^{m\times n}$ is a linear operator with spectral norm $\|K\|_2\le L_K$.
\end{itemize}

Define the proximal operator of a convex function $h$ with stepsize $\tau>0$ as  

\[
\operatorname{prox}_{\tau h}(v):=\arg\min_{u}\Big\{h(u)+\frac{1}{2\tau}\|u-v\|^{2}\Big\}.
\tag{2}
\]

The PDHG iterates are  

\[
\begin{aligned}
y^{k+1}      &=\operatorname{prox}_{\sigma g^{*}}\!\Big(y^{k}+ \sigma K\bar x^{k}\Big), \tag{3a}\\
x^{k+1}      &=\operatorname{prox}_{\tau f}\!\Big(x^{k}-\tau K^{\!\top} y^{k+1}\Big), \tag{3b}\\
\bar x^{k+1} &=x^{k+1}+ \theta\big(x^{k+1}-x^{k}\big), \tag{3c}
\end{aligned}
\]
with stepsizes $\tau,\sigma>0$ and extrapolation parameter $\theta\in[0,1]$.  
The initial points $x^{0},y^{0}$ are arbitrary, and we set $\bar x^{0}=x^{0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal--Dual Hybrid Gradient (PDHG)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $f,g^{*},K,\tau>0,\sigma>0,\theta\in[0,1]$, max iterations $T$.
\STATE Initialize $x^{0}\in\mathbb{R}^{n},\; y^{0}\in\mathbb{R}^{m},\; \bar x^{0}=x^{0}$.
\FOR{$k=0$ \textbf{to} $T-1$}
    \STATE $y^{k+1}\gets\operatorname{prox}_{\sigma g^{*}}\!\big(y^{k}+ \sigma K\bar x^{k}\big)$ \hfill\textit{// dual ascent}
    \STATE $x^{k+1}\gets\operatorname{prox}_{\tau f}\!\big(x^{k}-\tau K^{\!\top}y^{k+1}\big)$ \hfill\textit{// primal proximal step}
    \STATE $\bar x^{k+1}\gets x^{k+1}+ \theta\big(x^{k+1}-x^{k}\big)$
\ENDFOR
\STATE \textbf{Output:} $\{x^{k},y^{k}\}_{k=0}^{T}$ (or ergodic averages).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

\[

\]

The proximal operators are elementary:
\[
\operatorname{prox}_{\tau f}(v)=\frac{v+2\tau\cdot3}{1+2\tau},\qquad 
\operatorname{prox}_{\sigma g^{*}}(v)=\frac{v}{1+\sigma}.
\]

Choose $\tau=\sigma=0.1$, $\theta=1$, and start from $x^{0}=0$, $y^{0}=0$, $\bar x^{0}=0$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]
\label{as:convex}
\begin{enumerate}
  \item $f$ is convex and $L_f$‑Lipschitz differentiable:
  $\|\nabla f(x)-\nabla f(x')\|\le L_f\|x-x'\|,\;\forall x,x'$.
  \item $g^{*}$ is convex (possibly nonsmooth) with a tractable proximal operator.
\end{enumerate}
\end{assumption}
\begin{assumption}[Bounded Linear Operator]
\label{as:K}
The linear map $K$ satisfies $\|K\|_2\le L_K$.
\end{assumption}
\begin{assumption}[Stepsize Condition]
\label{as:steps}
The parameters $\tau,\sigma>0$ and $\theta\in[0,1]$ obey  

\[
\tau\sigma L_K^{2}\le 1,\qquad \theta\in[0,1].
\tag{4}
\]

\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ primal–dual gap]
\label{thm:conv}
Let $\{(x^{k},y^{k})\}_{k\ge0}$ be generated by PDHG under Assumptions \ref{as:convex}–\ref{as:steps}.  
Define the ergodic averages  

\[
\bar x^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}\!x^{k},\qquad
\bar y^{K}:=\frac{1}{K}\sum_{k=0}^{K-1}\!y^{k}.
\]

Then for any saddle point $(x^{\star},y^{\star})$ of \eqref{1} we have  

\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl[\mathcal L(x^{k},y^{\star})-\mathcal L(x^{\star},y^{k})\bigr]
\;\le\;
\frac{1}{2K}\Bigl(\frac{\|x^{0}-x^{\star}\|^{2}}{\tau}
      +\frac{\|y^{0}-y^{\star}\|^{2}}{\sigma}\Bigr).
\tag{5}
\]

Consequently $\mathcal L(\bar x^{K},y^{\star})-\mathcal L(x^{\star},\bar y^{K})=O(1/K)$. 
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability:} Condition \eqref{as:steps} guarantees that the combined operator is non‑expansive, yielding bounded iterates.
  \item \textbf{Hyperparameter Sensitivity:} The product $\tau\sigma$ must not exceed $1/L_K^{2}$. Larger $\tau$ (or $\sigma$) accelerates progress but risks violating \eqref{as:steps}.
  \item \textbf{Comparison:} Compared with vanilla gradient descent on the primal alone, PDHG attains the same $O(1/k)$ rate while handling nonsmooth $g^{*}$ without smoothing.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal operator]
\label{lem:three}
For any convex $h$ and $\tau>0$, and for all $u,v$,
\[
\langle u-\operatorname{prox}_{\tau h}(u),\; \operatorname{prox}_{\tau h}(u)-v\rangle
\ge \tau\bigl[h(\operatorname{prox}_{\tau h}(u))-h(v)\bigr].
\tag{6}
\]
\end{lemma}
\begin{proof}
Directly from the optimality condition of \eqref{2}; see e.g. \cite{Bauschke2011}.
\end{proof}
\begin{lemma}[Non‑expansiveness of the proximal map]
\label{lem:nonexp}
For any convex $h$, $\operatorname{prox}_{\tau h}$ is $1$‑Lipschitz:
\[
\bigl\|\operatorname{prox}_{\tau h}(u)-\operatorname{prox}_{\tau h}(v)\bigr\|\le \|u-v\|.
\tag{7}
\]
\end{lemma}
\begin{proof}
Standard; follows from firm non‑expansiveness of the proximal operator.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:conv}.  Throughout the proof, fix an arbitrary saddle point $(x^{\star},y^{\star})$.

\noindent\textbf{Notation.} Write $A^{k}:=K\bar x^{k}$ and $B^{k}:=K^{\!\top}y^{k+1}$.

\noindent\underline{[Step 1]} \textbf{Dual proximal inequality.}  
Apply Lemma \ref{lem:three} with $h=g^{*}$, $u=y^{k}+ \sigma A^{k}$, $v=y^{\star}$, and stepsize $\sigma$:

\[
\langle y^{k}+ \sigma A^{k}-y^{k+1},\; y^{k+1}-y^{\star}\rangle
\ge \sigma\bigl[g^{*}(y^{k+1})-g^{*}(y^{\star})\bigr].
\tag{8}
\]

Rearranging gives  

\[
\langle y^{k}-y^{k+1},\; y^{k+1}-y^{\star}\rangle
+ \sigma\langle A^{k}, y^{k+1}-y^{\star}\rangle
\ge \sigma\bigl[g^{*}(y^{k+1})-g^{*}(y^{\star})\bigr].
\tag{9}
\]

\noindent\underline{[Step 2]} \textbf{Primal proximal inequality.}  
Apply Lemma \ref{lem:three} with $h=f$, $u=x^{k}-\tau B^{k}$, $v=x^{\star}$, stepsize $\tau$:

\[
\langle x^{k}-\tau B^{k}-x^{k+1},\; x^{k+1}-x^{\star}\rangle
\ge \tau\bigl[f(x^{k+1})-f(x^{\star})\bigr].
\tag{10}
\]

Equivalently,

\[
\langle x^{k}-x^{k+1},\; x^{k+1}-x^{\star}\rangle
-\tau\langle B^{k},x^{k+1}-x^{\star}\rangle
\ge \tau\bigl[f(x^{k+1})-f(x^{\star})\bigr].
\tag{11}
\]

\noindent\underline{[Step 3]} \textbf{Combine the two inequalities.}  
Add \eqref{9} and \eqref{11}:

\[
\begin{aligned}
&\langle y^{k}-y^{k+1},y^{k+1}-y^{\star}\rangle
+\langle x^{k}-x^{k+1},x^{k+1}-x^{\star}\rangle \\
&\quad +\sigma\langle A^{k},y^{k+1}-y^{\star}\rangle
-\tau\langle B^{k},x^{k+1}-x^{\star}\rangle \\
&\ge \sigma\bigl[g^{*}(y^{k+1})-g^{*}(y^{\star})\bigr]
   +\tau\bigl[f(x^{k+1})-f(x^{\star})\bigr].
\end{aligned}
\tag{12}
\]

\noindent\underline{[Step 4]} \textbf{Express bilinear terms via the saddle function.}  
Recall $\mathcal L(x,y)=f(x)+\langle Kx,y\rangle-g^{*}(y)$.  Using $A^{k}=K\bar x^{k}$ and $B^{k}=K^{\!\top}y^{k+1}$,

\[
\begin{aligned}
\sigma\langle A^{k},y^{k+1}-y^{\star}\rangle
-\tau\langle B^{k},x^{k+1}-x^{\star}\rangle
&=\sigma\langle K\bar x^{k},y^{k+1}-y^{\star}\rangle
   -\tau\langle K^{\!\top}y^{k+1},x^{k+1}-x^{\star}\rangle\\
&= \sigma\langle \bar x^{k},K^{\!\top}(y^{k+1}-y^{\star})\rangle
   -\tau\langle K^{\!\top}y^{k+1},x^{k+1}-x^{\star}\rangle\\
&= \sigma\langle \bar x^{k},K^{\!\top}y^{k+1}\rangle
   -\sigma\langle \bar x^{k},K^{\!\top}y^{\star}\rangle
   -\tau\langle K^{\!\top}y^{k+1},x^{k+1}\rangle
   +\tau\langle K^{\!\top}y^{k+1},x^{\star}\rangle .
\end{aligned}
\tag{13}
\]

Re‑group the terms to obtain  

\[
\begin{aligned}
&\sigma\langle \bar x^{k},K^{\!\top}y^{k+1}\rangle
-\tau\langle K^{\!\top}y^{k+1},x^{k+1}\rangle \\
&= \langle K\bar x^{k},y^{k+1}\rangle
   -\langle Kx^{k+1},y^{k+1}\rangle
 = \langle K(\bar x^{k}-x^{k+1}),y^{k+1}\rangle .
\end{aligned}
\tag{14}
\]

Thus the sum of the bilinear pieces in \eqref{12} can be bounded using Cauchy–Schwarz and the norm bound $\|K\|\le L_K$:

\[
\begin{aligned}
\sigma\langle A^{k},y^{k+1}-y^{\star}\rangle
-\tau\langle B^{k},x^{k+1}-x^{\star}\rangle
&\le \frac{L_K^{2}}{2}\bigl(\tau\|x^{k+1}-\bar x^{k}\|^{2}
                         +\sigma\|y^{k+1}-y^{\star}\|^{2}\bigr) \\
&\quad +\langle K\bar x^{k},y^{\star}\rangle
      -\langle Kx^{\star},y^{k+1}\rangle .
\end{aligned}
\tag{15}
\]

(The inequality follows from $2ab\le a^{2}+b^{2}$ with $a=\sqrt{\tau}\|K(\bar x^{k}-x^{k+1})\|$, $b=\sqrt{\sigma}\|y^{k+1}-y^{\star}\|$.)

\noindent\underline{[Step 5]} \textbf{Quadratic expansion of the inner products.}  
For any vectors $u,v$,
\[
\langle u-v,\,u-w\rangle = \tfrac12\bigl(\|u-w\|^{2}-\|v-w\|^{2}+\|u-v\|^{2}\bigr).
\tag{16}
\]

Apply \eqref{16} to the two first inner products in \eqref{12}:

\[
\begin{aligned}
\langle y^{k}-y^{k+1},y^{k+1}-y^{\star}\rangle
&=\tfrac12\bigl(\|y^{k}-y^{\star}\|^{2}-\|y^{k+1}-y^{\star}\|^{2}
               -\|y^{k}-y^{k+1}\|^{2}\bigr),\\
\langle x^{k}-x^{k+1},x^{k+1}-x^{\star}\rangle
&=\tfrac12\bigl(\|x^{k}-x^{\star}\|^{2}-\|x^{k+1}-x^{\star}\|^{2}
               -\|x^{k}-x^{k+1}\|^{2}\bigr).
\end{aligned}
\tag{17}
\]

\noindent\underline{[Step 6]} \textbf{Insert \eqref{15}–\eqref{17} into \eqref{12}.}  
After substitution and cancelling the mixed term $\langle K\bar x^{k},y^{\star}\rangle-\langle Kx^{\star},y^{k+1}\rangle$ with the definition of $\mathcal L$, we obtain

\[
\begin{aligned}
&\frac{1}{2}\Bigl(\tfrac{\|x^{k}-x^{\star}\|^{2}}{\tau}
                 -\tfrac{\|x^{k+1}-x^{\star}\|^{2}}{\tau}
                 -\tfrac{\|x^{k}-x^{k+1}\|^{2}}{\tau}\Bigr)\\
&\;+\frac{1}{2}\Bigl(\tfrac{\|y^{k}-y^{\star}\|^{2}}{\sigma}
                 -\tfrac{\|y^{k+1}-y^{\star}\|^{2}}{\sigma}
                 -\tfrac{\|y^{k}-y^{k+1}\|^{2}}{\sigma}\Bigr)\\
&\;\le \mathcal L(x^{k+1},y^{\star})-\mathcal L(x^{\star},y^{k+1})
      +\frac{L_K^{2}}{2}\bigl(\tau\|x^{k+1}-\bar x^{k}\|^{2}
                            +\sigma\|y^{k+1}-y^{\star}\|^{2}\bigr).
\end{aligned}
\tag{18}
\]

\noindent\underline{[Step 7]} \textbf{Use the stepsize condition.}  
Because $\tau\sigma L_K^{2}\le1$, the term $\frac{L_K^{2}}{2}\tau\|x^{k+1}-\bar x^{k}\|^{2}$ can be absorbed by $-\frac{1}{2\tau}\|x^{k}-x^{k+1}\|^{2}$, and similarly for the dual part.  Hence

\[
\frac{1}{2}\Bigl(\tfrac{\|x^{k}-x^{\star}\|^{2}}{\tau}
               -\tfrac{\|x^{k+1}-x^{\star}\|^{2}}{\tau}\Bigr)
+\frac{1}{2}\Bigl(\tfrac{\|y^{k}-y^{\star}\|^{2}}{\sigma}
               -\tfrac{\|y^{k+1}-y^{\star}\|^{2}}{\sigma}\Bigr)
\le \mathcal L(x^{k+1},y^{\star})-\mathcal L(x^{\star},y^{k+1}).
\tag{19}
\]

\noindent\underline{[Step 8]} \textbf{Telescoping sum.}  
Sum \eqref{19} over $k=0,\dots,K-1$:

\[
\frac{1}{2}\Bigl(\tfrac{\|x^{0}-x^{\star}\|^{2}}{\tau}
               +\tfrac{\|y^{0}-y^{\star}\|^{2}}{\sigma}\Bigr)
\ge \sum_{k=0}^{K-1}\bigl[\mathcal L(x^{k+1},y^{\star})
                         -\mathcal L(x^{\star},y^{k+1})\bigr].
\tag{20}
\]

Divide both sides by $K$ and note that the sum on the right equals
$K$ times the average primal–dual gap, which yields exactly the bound \eqref{5}.
\hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{5} we have for any $K\ge1$  

\[
\mathcal G_{K}:=
\frac{1}{K}\sum_{k=0}^{K-1}
\bigl[\mathcal L(x^{k},y^{\star})-\mathcal L(x^{\star},y^{k})\bigr]
\le
\frac{C}{K},
\qquad 
C:=\frac{1}{2}\Bigl(\frac{\|x^{0}-x^{\star}\|^{2}}{\tau}
                  +\frac{\|y^{0}-y^{\star}\|^{2}}{\sigma}\Bigr).
\tag{21}
\]

Thus the ergodic primal–dual gap decays at rate $O(1/K)$.  If $f$ is additionally $\mu$‑strongly convex, the same proof (with the strong convexity inequality) yields a linear rate $O((1-\rho)^{K})$ where $\rho=\frac{\mu\tau}{1+\mu\tau}$.

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Pure primal gradient descent.}  
    Set $g^{*}\equiv0$, $K=I$, $\sigma=0$, $\theta=0$.  The update reduces to $x^{k+1}= \operatorname{prox}_{\tau f}(x^{k})$, i.e. gradient descent with stepsize $\tau\le 1/L_f$, and the $O(1/k)$ rate matches the classical result.
  \item \textbf{Chambolle–Pock algorithm.}  
    Choosing $\theta=1$ and $\tau\sigma L_K^{2}=1$ recovers the original Chambolle–Pock scheme; our theorem therefore reproduces their established $O(1/k)$ bound.
  \item \textbf{Strongly convex $f$.}  
    If $f$ is $\mu$‑strongly convex, replace inequality \eqref{11} by the stronger bound  
    $\langle x^{k}-x^{k+1},x^{k+1}-x^{\star}\rangle
    \ge \tau\bigl[f(x^{k+1})-f(x^{\star})\bigr]
    +\frac{\mu\tau}{2}\|x^{k+1}-x^{\star}\|^{2}$,
    which leads to a geometric convergence factor as mentioned above.
\end{itemize}
