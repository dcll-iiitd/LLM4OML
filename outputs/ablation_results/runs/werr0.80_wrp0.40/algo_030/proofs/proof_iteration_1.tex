\section*{Heavy‑Ball Momentum Gradient Descent (HB‑MGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
Given a stepsize $\alpha>0$ and a momentum parameter $\beta\in[0,1)$, the Heavy‑Ball update is
\[
\boxed{
\begin{aligned}
x_{k+1}&=x_{k}+\beta\bigl(x_{k}-x_{k-1}\bigr)-\alpha\,\grad f(x_{k}),\qquad k\ge 0,
\end{aligned}}
\]
with an initial pair $(x_{0},x_{-1})$ (often $x_{-1}=x_{0}$).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum Gradient Descent}
\begin{algorithmic}[1]
\Require stepsize $\alpha>0$, momentum $\beta\in[0,1)$, max iterations $K$, tolerance $\varepsilon$
\State Initialize $x_{0}\in\mathbb{R}^{d}$, set $x_{-1}\gets x_{0}$
\For{$k=0$ \textbf{to} $K-1$}
    \State $g_{k}\gets \grad f(x_{k})$
    \State $x_{k+1}\gets x_{k}+\beta\,(x_{k}-x_{k-1})-\alpha\,g_{k}$
    \If{$\|g_{k}\|\le\varepsilon$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate quadratic
\[
f(w)=(w-3)^{2},\qquad \grad f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, $w_{0}=0$, $w_{-1}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $g_{k}=2(w_{k}-3)$ & $w_{k+1}=w_{k}+0.5(w_{k}-w_{k-1})-0.1\,g_{k}$ & $f(w_{k+1})$\\ \hline
0 & $2(0-3)=-6$ &
$0+0.5(0-0)-0.1(-6)=0+0+0.6=0.6$ &
$(0.6-3)^{2}=5.76$\\[4pt]
1 & $2(0.6-3)=-4.8$ &
$0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38$ &
$(1.38-3)^{2}=2.6244$\\[4pt]
2 & $2(1.38-3)=-3.24$ &
$1.38+0.5(1.38-0.6)-0.1(-3.24)=1.38+0.39+0.324=2.094$ &
$(2.094-3)^{2}=0.8196$
\end{tabular}
\end{center}
The objective value decreases from $9$ (at $w_{0}=0$) to $0.82$ after three iterations.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ has $L$‑Lipschitz gradient, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
$f$ is bounded below: $\displaystyle f_{\inf}:=\inf_{x}f(x)>-\infty$.
\end{assumption}
\begin{assumption}[Parameter Range]\label{ass:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha\le\frac{1-\beta}{L},\qquad 0\le\beta<1 .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of HB‑MGD]\label{thm:main}
Under Assumptions~\ref{ass:smooth}–\ref{ass:param}, let $\{x_{k}\}$ be generated by the Heavy‑Ball update. Define the Lyapunov function
\[
\Phi_{k}:=f(x_{k})+\frac{\beta}{2\alpha}\|x_{k}-x_{k-1}\|^{2}.
\]
Then for any $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_{k})\|^{2}
\;\le\;
\frac{2\bigl(\Phi_{0}-f_{\inf}\bigr)}{\alpha K}.
\tag{1}
\]
Consequently $\displaystyle \lim_{K\to\infty}\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_{k})\|^{2}=0$ and there exists a subsequence $\{x_{k_{j}}\}$ with $\|\grad f(x_{k_{j}})\|\to0$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition \eqref{ass:param} guarantees that the Lyapunov function $\Phi_{k}$ is non‑increasing, preventing divergence even though $f$ may be non‑convex.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\beta$ accelerates progress when the curvature is benign, but the admissible stepsize $\alpha$ shrinks linearly with $(1-\beta)$; violating \eqref{ass:param} can cause $\Phi_{k}$ to increase.
\item \textbf{Comparison.} Setting $\beta=0$ recovers standard gradient descent and (1) reduces to the classic $O(1/K)$ bound. For $0<\beta<1$, the same $O(1/K)$ rate holds, but with a potentially smaller constant because the momentum term can reduce $\Phi_{0}$.
\end{itemize}

\section*{Discussion}
The theorem shows that Heavy‑Ball momentum does not improve the worst‑case asymptotic rate for non‑convex smooth problems (still $O(1/K)$), yet the constant can be better in practice. The proof hinges on a carefully chosen Lyapunov function that couples function values and the ``velocity'' $x_{k}-x_{k-1}$. The parameter restriction \eqref{ass:param} is tight for the analysis; empirical practice often uses larger stepsizes, which may still converge but lies beyond the current theory.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ has $L$‑Lipschitz gradient, then for any $x,y$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from integration of the Lipschitz condition. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All non‑trivial steps are numbered.

\medskip
\noindent\textbf{Step 1 (Expand the update).}  
From the Heavy‑Ball rule,
\[
x_{k+1}=x_{k}+\beta (x_{k}-x_{k-1})-\alpha g_{k},
\qquad g_{k}:=\grad f(x_{k}).
\tag{2}
\]

\noindent\textbf{Step 2 (Apply Descent Lemma to $f(x_{k+1})$).}  
Using Lemma~\ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle g_{k},x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{3}
\]

\noindent\textbf{Step 3 (Insert the update (2) into (3)).}  
Since $x_{k+1}-x_{k}= \beta (x_{k}-x_{k-1})-\alpha g_{k}$,
\[
\begin{aligned}
\langle g_{k},x_{k+1}-x_{k}\rangle
&=\beta\langle g_{k},x_{k}-x_{k-1}\rangle-\alpha\|g_{k}\|^{2},\\
\|x_{k+1}-x_{k}\|^{2}
&=\|\beta (x_{k}-x_{k-1})-\alpha g_{k}\|^{2}\\
&=\beta^{2}\|x_{k}-x_{k-1}\|^{2}
-2\alpha\beta\langle g_{k},x_{k}-x_{k-1}\rangle
+\alpha^{2}\|g_{k}\|^{2}.
\end{aligned}
\tag{4}
\]

\noindent\textbf{Step 4 (Plug (4) into (3)).}  
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
+\beta\langle g_{k},x_{k}-x_{k-1}\rangle-\alpha\|g_{k}\|^{2}\\
&\quad+\frac{L}{2}\Bigl[\beta^{2}\|x_{k}-x_{k-1}\|^{2}
-2\alpha\beta\langle g_{k},x_{k}-x_{k-1}\rangle
+\alpha^{2}\|g_{k}\|^{2}\Bigr].
\end{aligned}
\tag{5}
\]

\noindent\textbf{Step 5 (Collect like terms).}  
Group the inner‑product terms and the norm terms:
\[
\begin{aligned}
f(x_{k+1})
&\le f(x_{k})
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|g_{k}\|^{2}\\
&\quad+\beta\Bigl(1-L\alpha\Bigr)\langle g_{k},x_{k}-x_{k-1}\rangle
+\frac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}.
\end{aligned}
\tag{6}
\]

\noindent\textbf{Step 6 (Introduce the Lyapunov term).}  
Define
\[
\Phi_{k}=f(x_{k})+\frac{\beta}{2\alpha}\|x_{k}-x_{k-1}\|^{2}.
\tag{7}
\]
Compute $\Phi_{k+1}-\Phi_{k}$ using (6) and the definition (7):
\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&=f(x_{k+1})-f(x_{k})
+\frac{\beta}{2\alpha}\bigl(\|x_{k+1}-x_{k}\|^{2}
-\|x_{k}-x_{k-1}\|^{2}\bigr).
\end{aligned}
\tag{8}
\]

\noindent\textbf{Step 7 (Expand the difference of the velocity norms).}  
From (4),
\[
\|x_{k+1}-x_{k}\|^{2}
=\beta^{2}\|x_{k}-x_{k-1}\|^{2}
-2\alpha\beta\langle g_{k},x_{k}-x_{k-1}\rangle
+\alpha^{2}\|g_{k}\|^{2}.
\]
Hence
\[
\|x_{k+1}-x_{k}\|^{2}
-\|x_{k}-x_{k-1}\|^{2}
=(\beta^{2}-1)\|x_{k}-x_{k-1}\|^{2}
-2\alpha\beta\langle g_{k},x_{k}-x_{k-1}\rangle
+\alpha^{2}\|g_{k}\|^{2}.
\tag{9}
\]

\noindent\textbf{Step 8 (Insert (6) and (9) into (8)).}  
After substitution and simplifying coefficients,
\[
\begin{aligned}
\Phi_{k+1}-\Phi_{k}
&\le
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|g_{k}\|^{2}
+\beta\Bigl(1-L\alpha\Bigr)\langle g_{k},x_{k}-x_{k-1}\rangle\\
&\quad+\frac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}
+\frac{\beta}{2\alpha}\Bigl[(\beta^{2}-1)\|x_{k}-x_{k-1}\|^{2}
-2\alpha\beta\langle g_{k},x_{k}-x_{k-1}\rangle
+\alpha^{2}\|g_{k}\|^{2}\Bigr].
\end{aligned}
\tag{10}
\]

\noindent\textbf{Step 9 (Cancel the mixed inner‑product terms).}  
The coefficient of $\langle g_{k},x_{k}-x_{k-1}\rangle$ becomes
\[
\beta\bigl(1-L\alpha\bigr)-\beta^{2}
=\beta\bigl(1-L\alpha-\beta\bigr).
\]
Because of the parameter restriction $\alpha\le\frac{1-\beta}{L}$, we have $1-L\alpha-\beta\ge0$, thus the whole term is \emph{non‑positive}. Dropping it yields an inequality that is still valid:
\[
\Phi_{k+1}-\Phi_{k}
\le
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|g_{k}\|^{2}
+\frac{L\beta^{2}}{2}\|x_{k}-x_{k-1}\|^{2}
+\frac{\beta}{2\alpha}\Bigl[(\beta^{2}-1)\|x_{k}-x_{k-1}\|^{2}
+\alpha^{2}\|g_{k}\|^{2}\Bigr].
\tag{11}
\]

\noindent\textbf{Step 10 (Gather $\|g_{k}\|^{2}$ terms).}  
The coefficient in front of $\|g_{k}\|^{2}$ is
\[
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)+\frac{\beta\alpha}{2}
= -\alpha\Bigl(1-\frac{L\alpha}{2}-\frac{\beta}{2}\Bigr).
\tag{12}
\]

\noindent\textbf{Step 11 (Gather $\|x_{k}-x_{k-1}\|^{2}$ terms).}  
The coefficient in front of $\|x_{k}-x_{k-1}\|^{2}$ is
\[
\frac{L\beta^{2}}{2}+\frac{\beta}{2\alpha}(\beta^{2}-1)
= -\frac{\beta}{2\alpha}(1-\beta) +\frac{L\beta^{2}}{2}.
\tag{13}
\]

\noindent\textbf{Step 12 (Use the stepsize bound).}  
From Assumption~\ref{ass:param},
\[
\alpha\le\frac{1-\beta}{L}
\;\Longrightarrow\;
L\alpha\le 1-\beta.
\]
Consequently,
\[
1-\frac{L\alpha}{2}-\frac{\beta}{2}
\ge \frac{1}{2}\bigl(1-L\alpha-\beta\bigr)\ge0.
\]
Thus the coefficient in (12) is \emph{non‑positive}.  
Similarly,
\[
-\frac{\beta}{2\alpha}(1-\beta)+\frac{L\beta^{2}}{2}
\le 0,
\]
because $L\alpha\le 1-\beta$ implies $\frac{L}{2}\le\frac{1-\beta}{2\alpha}$.

\noindent\textbf{Step 13 (Final descent inequality).}  
Discarding the non‑positive terms yields
\[
\Phi_{k+1}\le\Phi_{k}-\frac{\alpha}{2}\|g_{k}\|^{2}.
\tag{14}
\]

\noindent\textbf{Step 14 (Sum over $k=0,\dots,K-1$).}  
Telescoping (14):
\[
\Phi_{K}\le\Phi_{0}-\frac{\alpha}{2}\sum_{k=0}^{K-1}\|g_{k}\|^{2}.
\tag{15}
\]

\noindent\textbf{Step 15 (Lower bound $\Phi_{K}$).}  
By definition (7) and Assumption~\ref{ass:lb},
\[
\Phi_{K}=f(x_{K})+\frac{\beta}{2\alpha}\|x_{K}-x_{K-1}\|^{2}\ge f_{\inf}.
\tag{16}
\]

\noindent\textbf{Step 16 (Derive the average gradient bound).}  
Combining (15) and (16) gives
\[
\frac{\alpha}{2}\sum_{k=0}^{K-1}\|g_{k}\|^{2}
\le \Phi_{0}-f_{\inf}.
\]
Dividing by $\alpha K$ yields the claimed inequality (1):
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|g_{k}\|^{2}
\le \frac{2\bigl(\Phi_{0}-f_{\inf}\bigr)}{\alpha K}.
\qquad\blacksquare
\]

\section*{Rate Derivation (With Constants)}
The constant in (1) can be expressed explicitly using the initialization:
\[
\Phi_{0}=f(x_{0})+\frac{\beta}{2\alpha}\|x_{0}-x_{-1}\|^{2}.
\]
If we start with $x_{-1}=x_{0}$ (common practice), the second term vanishes and
\[
\frac{1}{K}\sum_{k=0}^{K-1}\|\grad f(x_{k})\|^{2}
\le\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha K}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}  
$\Phi_{k}=f(x_{k})$, and (14) reduces to the classic gradient‑descent descent
\[
f(x_{k+1})\le f(x_{k})-\frac{\alpha}{2}\|\grad f(x_{k})\|^{2},
\]
recovering the $O(1/K)$ bound with constant $2(f(x_{0})-f_{\inf})/\alpha$.
\item \textbf{Exact quadratic $f(x)=\frac{1}{2}x^{\top}Ax$ with $A\succeq0$.}  
The Heavy‑Ball method can be analyzed via eigen‑decomposition and achieves a linear rate when $\beta$ and $\alpha$ are chosen according to the spectral radius condition; this is outside the present non‑convex $O(1/K)$ guarantee but illustrates that the bound is not tight for special convex quadratics.
\end{itemize}
