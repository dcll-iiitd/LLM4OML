\section{Mathematical Formulation}
We consider the stochastic optimisation problem
\[
\min_{x\in\R^{d}} \; f(x):=\E_{z\sim\mathcal{D}}[F(x;z)],
\]
where each realisation $F(\cdot;z)$ is differentiable.  
At iteration $k$ we have a stochastic gradient $g_k\in\R^{d}$ such that 
\[
g_k:=\nabla F(x_k;z_k),\qquad z_k\stackrel{i.i.d.}{\sim}\mathcal{D}.
\]

\medskip
\noindent\textbf{Hyper‑parameters}
\begin{itemize}
    \item Step size (learning rate) $\alpha>0$.
    \item Momentum coefficient $\beta\in[0,1)$.
\end{itemize}

\medskip
\noindent\textbf{State variables}
\begin{itemize}
    \item $x_k\in\R^{d}$ – current iterate.
    \item $v_k\in\R^{d}$ – momentum accumulator (initialized $v_{-1}=0$).
\end{itemize}

\medskip
\noindent\textbf{Update rules}
\begin{align}
    s_k &:= \sign(g_k)\in\{-1,0,1\}^{d},\label{eq:sign}\\[4pt]
    v_k &:= \beta\, v_{k-1} + (1-\beta)\, s_k,\qquad  k\ge 0,\label{eq:momentum}\\[4pt]
    x_{k+1} &:= x_k - \alpha\, v_k.\label{eq:update}
\end{align}
Equation \eqref{eq:sign} extracts the element‑wise sign of the stochastic gradient.
Equation \eqref{eq:momentum} forms an exponential moving average of the signed directions.
Equation \eqref{eq:update} moves the iterate opposite to the momentum term.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Pseudocode}
\begin{algorithm}[H]
\caption{SignSGD with Momentum}
\begin{algorithmic}[1]
\Require Initial point $x_0\in\R^{d}$, step size $\alpha>0$, momentum $\beta\in[0,1)$, max iterations $T$.
\State $v_{-1}\gets 0$.
\For{$k=0,1,\dots,T-1$}
    \State Sample $z_k\sim\mathcal{D}$ and compute stochastic gradient $g_k:=\nabla F(x_k;z_k)$.
    \State $s_k \gets \sign(g_k)$ \hfill\Comment{element‑wise sign}
    \State $v_k \gets \beta\, v_{k-1} + (1-\beta)\, s_k$.
    \State $x_{k+1} \gets x_k - \alpha\, v_k$.
\EndFor
\State \Return $x_T$ (or the average $\bar x_T:=\frac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w) = (w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
We use deterministic gradients (i.e. $g_k=\nabla f(w_k)$) for illustration,
step size $\alpha=0.1$, momentum $\beta=0.5$, and start from $w_0=0$.

\medskip
\begin{tabular}{c|c|c|c|c}
Iteration $k$ & $w_k$ & $g_k$ & $s_k=\sign(g_k)$ & $w_{k+1}=w_k-\alpha v_k$\\ \hline
0 & $0$ & $-6$ & $-1$ & $v_0=0.5\cdot0+0.5(-1)=-0.5$, $w_1=0-0.1(-0.5)=0.05$\\
1 & $0.05$ & $-5.9$ & $-1$ & $v_1=0.5(-0.5)+0.5(-1)=-0.75$, $w_2=0.05-0.1(-0.75)=0.125$\\
2 & $0.125$ & $-5.75$ & $-1$ & $v_2=0.5(-0.75)+0.5(-1)=-0.875$, $w_3=0.125-0.1(-0.875)=0.2125$
\end{tabular}

\medskip
Objective values:
\[
f(w_0)=9,\; f(w_1)=(0.05-3)^2\approx 8.7025,\; f(w_2)\approx 8.2656,\; f(w_3)\approx 7.7227.
\]
The iterates move towards the optimum $w^\star=3$, albeit slowly because only the sign information is used.

\section{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e. for all $x,y\in\R^{d}$,
\[
\|\nabla f(x)-\nabla f(y)\|_2 \le L\|x-y\|_2 .
\]
Equivalently,
\[
f(y) \le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|_2^{2}.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For any $x$, the stochastic gradient satisfies
\[
\E[g_k\mid x_k]=\nabla f(x_k).
\]
\end{assumption}

\begin{assumption}[Bounded Second Moment]\label{ass:boundedvar}
There exists $G>0$ such that for all $k$,
\[
\E\big[\|g_k\|_2^{2}\mid x_k\big]\le G^{2}.
\]
\end{assumption}

\begin{assumption}[Sign Correctness]\label{ass:sign}
For each coordinate $i\in\{1,\dots,d\}$ there exists a constant $\gamma\in(0,\tfrac12]$ such that
\[
\Pr\big(\sign(g_{k,i})=\sign(\nabla_i f(x_k))\mid x_k\big)\ge \tfrac12+\gamma .
\]
Consequently,
\[
\E\big[\sign(g_{k,i})\nabla_i f(x_k)\mid x_k\big]\ge 2\gamma\,|\nabla_i f(x_k)|.
\]
\end{assumption}

All constants $L,G,\gamma$ are assumed known and independent of $k$.

\section{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{x_k\}$ be generated by \eqref{eq:sign}–\eqref{eq:update} with step size
\[
0<\alpha\le\frac{2\gamma(1-\beta)}{L d},
\]
and momentum $\beta\in[0,1)$. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\big[\|\nabla f(x_k)\|_{1}\big]
\;\le\;
\frac{f(x_0)-f^{\star}}{\alpha\gamma(1-\beta)T}
\;+\;
\frac{L\alpha d}{2\gamma(1-\beta)} ,
\]
where $f^{\star}:=\inf_{x}f(x)$. 
Choosing $\alpha = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\big[\|\nabla f(x_k)\|_{1}\big]=\mathcal{O}\!\big(T^{-1/2}\big).
\]
\end{theorem}

\section{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The bound on $\alpha$ guarantees that the quadratic term arising from smoothness never outweighs the descent term.
    \item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
        \item Larger momentum $\beta$ reduces the effective descent coefficient $(1-\beta)$, requiring a smaller step size for stability.
        \item The constant $\gamma$ captures the reliability of the sign; larger $\gamma$ improves the convergence rate linearly.
    \end{itemize}
    \item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item Compared to vanilla SGD ($\mathcal{O}(1/\sqrt{T})$ in Euclidean norm), SignSGD with momentum achieves the same order but with a dependence on $d$ only through the factor $d$ in the second term, reflecting the reduced communication cost (1‑bit per coordinate).
        \item When $\beta=0$, Theorem~\ref{thm:main} reduces to the classical SignSGD bound (see \cite{bernstein2018signsgd}).
    \end{itemize}
\end{itemize}

\section{Discussion}
The theorem shows that, under modest statistical assumptions, the momentum‑augmented sign method converges to a stationary point of a non‑convex smooth objective. The dependence on the $\ell_{1}$ norm of the gradient is natural because the sign operator discards magnitude information; nevertheless the bound scales linearly with the dimension $d$, which is unavoidable without additional variance‑reduction tricks.

\section{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Bound on Momentum Vector]\label{lem:boundmom}
For every $k\ge0$, each coordinate of $v_k$ satisfies $|v_{k,i}|\le1$. Consequently,
\[
\|v_k\|_2^{2}\le d.
\]
\end{lemma}
\begin{proof}
By induction. $v_{-1}=0$ trivially satisfies $|v_{-1,i}|\le1$. Assume $|v_{k-1,i}|\le1$. From \eqref{eq:momentum},
\[
v_{k,i}= \beta v_{k-1,i}+(1-\beta)s_{k,i},
\]
with $s_{k,i}\in\{-1,0,1\}$. Hence $|v_{k,i}|\le \beta\cdot1+(1-\beta)\cdot1=1$. The $\ell_2$ bound follows by summing over $i$.
\end{proof}

\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x$ and any direction $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|_2^{2}.
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on $\nabla f$.
\end{proof}

\section{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken w.r.t. the randomness up to iteration $k$ unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $d=-\alpha v_k$:
\begin{equation}
f(x_{k+1})\le f(x_k)-\alpha\langle\nabla f(x_k),v_k\rangle
            +\frac{L\alpha^{2}}{2}\|v_k\|_2^{2}.
\label{eq:smooth}
\end{equation}

\noindent\textbf{[Step 2]} Expand the inner product using the definition of $v_k$ (Eq.~\eqref{eq:momentum}):
\[
\langle\nabla f(x_k),v_k\rangle
  =\beta\langle\nabla f(x_k),v_{k-1}\rangle
   +(1-\beta)\langle\nabla f(x_k),s_k\rangle .
\tag{2.1}
\]

\noindent\textbf{[Step 3]} Take conditional expectation $\E[\cdot\mid\mathcal{F}_k]$, where $\mathcal{F}_k$ is the sigma‑algebra generated by $\{x_0,s_0,\dots,s_{k-1}\}$.
Using Assumption~\ref{ass:sign},
\[
\E\big[\langle\nabla f(x_k),s_k\rangle\mid\mathcal{F}_k\big]
   =\sum_{i=1}^{d}\E\big[\sign(g_{k,i})\nabla_i f(x_k)\mid\mathcal{F}_k\big]
   \ge 2\gamma\sum_{i=1}^{d}|\nabla_i f(x_k)|
   = 2\gamma\|\nabla f(x_k)\|_{1}.
\tag{3.1}
\]

\noindent\textbf{[Step 4]} Substitute (2.1) and (3.1) into \eqref{eq:smooth} and then take full expectation:
\begin{align}
\E[f(x_{k+1})] 
&\le \E[f(x_k)] 
   -\alpha\beta\E[\langle\nabla f(x_k),v_{k-1}\rangle]
   -(1-\beta)\alpha\,2\gamma\,\E[\|\nabla f(x_k)\|_{1}]
   +\frac{L\alpha^{2}}{2}\E[\|v_k\|_2^{2}].
\label{eq:afterexp}
\end{align}

\noindent\textbf{[Step 5]} Drop the non‑positive term $-\alpha\beta\E[\langle\nabla f(x_k),v_{k-1}\rangle]$ (it can only improve the bound) to obtain a simpler inequality:
\begin{equation}
\E[f(x_{k+1})] 
\le \E[f(x_k)] 
   -2\alpha\gamma(1-\beta)\E[\|\nabla f(x_k)\|_{1}]
   +\frac{L\alpha^{2}}{2}\E[\|v_k\|_2^{2}].
\tag{5.1}
\end{equation}

\noindent\textbf{[Step 6]} Apply Lemma~\ref{lem:boundmom} to bound the last term:
\[
\E[\|v_k\|_2^{2}]\le d .
\tag{6.1}
\]

\noindent\textbf{[Step 7]} Insert (6.1) into (5.1):
\begin{equation}
\E[f(x_{k+1})] 
\le \E[f(x_k)] 
   -2\alpha\gamma(1-\beta)\E[\|\nabla f(x_k)\|_{1}]
   +\frac{L\alpha^{2}d}{2}.
\tag{7.1}
\end{equation}

\noindent\textbf{[Step 8]} Rearrange (7.1) to isolate the gradient term:
\begin{equation}
2\alpha\gamma(1-\beta)\E[\|\nabla f(x_k)\|_{1}]
\le \E[f(x_k)]-\E[f(x_{k+1})] +\frac{L\alpha^{2}d}{2}.
\tag{8.1}
\end{equation}

\noindent\textbf{[Step 9]} Sum inequality (8.1) over $k=0,\dots,T-1$:
\begin{align}
2\alpha\gamma(1-\beta)\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
&\le \E[f(x_0)]-\E[f(x_T)] + \frac{L\alpha^{2}d}{2}T.
\tag{9.1}
\end{align}
Since $f(x_T)\ge f^{\star}$,
\[
\E[f(x_0)]-\E[f(x_T)]\le f(x_0)-f^{\star}.
\tag{9.2}
\]

\noindent\textbf{[Step 10]} Divide (9.1) by $2\alpha\gamma(1-\beta)T$:
\begin{equation}
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
\le
\frac{f(x_0)-f^{\star}}{2\alpha\gamma(1-\beta)T}
+\frac{L\alpha d}{4\gamma(1-\beta)}.
\tag{10.1}
\end{equation}
Multiplying numerator and denominator of the first term by $2$ yields the statement of Theorem~\ref{thm:main}.

\noindent\textbf{[Step 11]} Choice of step size.  
Set $\alpha = \frac{c}{\sqrt{T}}$ with $c\le \frac{2\gamma(1-\beta)}{Ld}$ to satisfy the stability condition $0<\alpha\le\frac{2\gamma(1-\beta)}{Ld}$ (the bound required in the theorem). Substituting $\alpha = \Theta(T^{-1/2})$ into (10.1) gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
 = \mathcal{O}\!\big(T^{-1/2}\big).
\]
Thus the algorithm attains the claimed convergence rate.

\qed

\section{Rate Derivation (With Constants)}
From (10.1) we can write the bound explicitly as
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
\le
\underbrace{\frac{f(x_0)-f^{\star}}{2\gamma(1-\beta)}}_{\displaystyle C_1}
\frac{1}{\alpha T}
+
\underbrace{\frac{L d}{4\gamma(1-\beta)}}_{\displaystyle C_2}\alpha .
\]
Optimising the right‑hand side over $\alpha>0$ yields $\alpha^{*}= \sqrt{\frac{C_1}{C_2}}\frac{1}{\sqrt{T}}$, which coincides (up to constants) with the step size choice in Step~11. Substituting $\alpha^{*}$ gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
\le
2\sqrt{C_1 C_2}\,\frac{1}{\sqrt{T}}
=
\frac{L d}{2\gamma(1-\beta)}\sqrt{\frac{f(x_0)-f^{\star}}{L d}}\,\frac{1}{\sqrt{T}}.
\]

\section{Special Cases}
\begin{itemize}
    \item \textbf{No momentum ($\beta=0$).} The bound reduces to
    \[
    \frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|_{1}]
    \le
    \frac{f(x_0)-f^{\star}}{2\alpha\gamma T}
    +\frac{L\alpha d}{4\gamma},
    \]
    which matches the classic SignSGD result.
    \item \textbf{Perfect sign ($\gamma=\tfrac12$).} If the stochastic gradient always has the correct sign, then $\gamma=\tfrac12$ and the descent coefficient doubles, improving the constant factor by $2$.
    \item \textbf{High‑dimensional limit.} The term $\frac{L\alpha d}{4\gamma(1-\beta)}$ shows a linear dependence on $d$; however, the communication cost remains $d$ bits per iteration, independent of $G$ or $L$.
\end{itemize}
