\section*{Randomized Coordinate Descent (RCD) for Non-Convex Coordinate-wise Smooth Functions}

We analyze a Randomized Coordinate Descent algorithm where at each iteration a single
coordinate $i_k \in \{1, \dots, n\}$ is selected at random with probability $p_{i_k}$,
and only that coordinate is updated using its partial derivative. The central analytical
device is reweighting the expectation of the per-coordinate decrease by $1/p_i$ to
recover a full-gradient quantity.

\section*{Mathematical Formulation}

Let $f : \mathbb{R}^n \to \mathbb{R}$ be a (possibly non-convex) differentiable function.
We denote by $\nabla_i f(w) = \frac{\partial f}{\partial w_i}(w)$ the $i$-th partial
derivative, and by $e_i$ the $i$-th standard basis vector.

\textbf{Update rule.} At iteration $k$:
\begin{enumerate}
\item Sample a coordinate $i_k \in \{1,\dots,n\}$ with probability $p_{i_k} > 0$,
where $\sum_{i=1}^n p_i = 1$.
\item Update only coordinate $i_k$:
\[
w_{k+1} = w_k - \eta_{i_k}\, \nabla_{i_k} f(w_k)\, e_{i_k},
\]
equivalently, componentwise,
\[
(w_{k+1})_j =
\begin{cases}
(w_k)_j - \eta_{i_k}\,\nabla_{i_k} f(w_k), & j = i_k,\\[4pt]
(w_k)_j, & j \neq i_k.
\end{cases}
\]
\end{enumerate}

\textbf{Hyperparameters.}
\begin{itemize}
\item Coordinate-wise step sizes $\eta_i > 0$ (typically $\eta_i = 1/L_i$ where $L_i$
is the coordinate-wise Lipschitz constant of the gradient).
\item Sampling distribution $\{p_i\}_{i=1}^n$ with $p_i > 0$ and $\sum_i p_i = 1$
(e.g.\ uniform $p_i = 1/n$, or Lipschitz-proportional $p_i = L_i / \sum_j L_j$).
\end{itemize}

\section*{Pseudocode}

\begin{algorithm}[h]
\caption{Randomized Coordinate Descent (RCD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0 \in \mathbb{R}^n$, step sizes $\{\eta_i\}$,
probabilities $\{p_i\}$, tolerance $\epsilon$, max iterations $K$
\State \textbf{Initialize:} $k \gets 0$
\While{$k < K$ and $\|\nabla f(w_k)\| > \epsilon$}
    \State Sample $i_k \sim \{p_1, \dots, p_n\}$
    \State Compute partial derivative $g_k \gets \nabla_{i_k} f(w_k)$
    \State $w_{k+1} \gets w_k - \eta_{i_k}\, g_k\, e_{i_k}$
    \State $k \gets k + 1$
\EndWhile
\State \textbf{Output:} $w_k$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

Consider $f(w) = (w-3)^2$ with $n=1$ (single coordinate, so $p_1 = 1$, $i_k = 1$
always), $w_0 = 0$, $\eta = 0.1$. The gradient is $\nabla f(w) = 2(w-3)$.

\begin{itemize}
\item \textbf{Iteration 0:} $w_0 = 0$.
$\nabla f(w_0) = 2(0-3) = -6$.
$w_1 = 0 - 0.1 \cdot (-6) = 0.6$.
$f(w_1) = (0.6 - 3)^2 = (-2.4)^2 = 5.76$.
(Initial $f(w_0) = (0-3)^2 = 9$.)

\item \textbf{Iteration 1:} $w_1 = 0.6$.
$\nabla f(w_1) = 2(0.6-3) = -4.8$.
$w_2 = 0.6 - 0.1 \cdot (-4.8) = 0.6 + 0.48 = 1.08$.
$f(w_2) = (1.08 - 3)^2 = (-1.92)^2 = 3.6864$.

\item \textbf{Iteration 2:} $w_2 = 1.08$.
$\nabla f(w_2) = 2(1.08-3) = -3.84$.
$w_3 = 1.08 - 0.1 \cdot (-3.84) = 1.08 + 0.384 = 1.464$.
$f(w_3) = (1.464 - 3)^2 = (-1.536)^2 = 2.359296$.
\end{itemize}

The objective decreases monotonically: $9 \to 5.76 \to 3.6864 \to 2.3593$,
converging toward the minimizer $w^* = 3$.

\section*{Assumptions}

\begin{assumption}[Coordinate-wise smoothness]
\label{ass:smooth}
For each coordinate $i \in \{1,\dots,n\}$, there exists $L_i > 0$ such that for all
$w \in \mathbb{R}^n$ and all $t \in \mathbb{R}$,
\[
\big| \nabla_i f(w + t e_i) - \nabla_i f(w) \big| \le L_i\, |t|.
\]
Equivalently, the descent lemma along coordinate $i$ holds:
\[
f(w + t e_i) \le f(w) + t\, \nabla_i f(w) + \frac{L_i}{2} t^2 .
\]
\end{assumption}

\begin{assumption}[Lower boundedness]
\label{ass:lb}
$f$ is bounded below: $f^\star := \inf_{w} f(w) > -\infty$.
\end{assumption}

\begin{assumption}[Sampling distribution]
\label{ass:prob}
Coordinate $i$ is sampled with fixed probability $p_i > 0$, independently across
iterations, with $\sum_{i=1}^n p_i = 1$.
\end{assumption}

\begin{assumption}[Step size]
\label{ass:step}
The step sizes satisfy $0 < \eta_i \le 1/L_i$. The canonical choice is
$\eta_i = 1/L_i$.
\end{assumption}

\section*{Main Convergence Theorem}

\begin{theorem}[Convergence of RCD in the non-convex setting]
\label{thm:main}
Under Assumptions \ref{ass:smooth}--\ref{ass:step} with $\eta_i = 1/L_i$, the iterates
of RCD satisfy, for any $K \ge 1$,
\[
\min_{0 \le k \le K-1}\; \mathbb{E}\!\left[ \sum_{i=1}^n \frac{p_i}{L_i}
\big(\nabla_i f(w_k)\big)^2 \right]
\le \frac{1}{K} \sum_{k=0}^{K-1} \mathbb{E}\!\left[ \sum_{i=1}^n \frac{p_i}{L_i}
\big(\nabla_i f(w_k)\big)^2 \right]
\le \frac{2\big(f(w_0) - f^\star\big)}{K}.
\]
In particular, defining the weighted gradient norm
$\|\nabla f(w)\|_{P}^2 := \sum_{i=1}^n \frac{p_i}{L_i}\big(\nabla_i f(w)\big)^2$,
we have
\[
\min_{0 \le k \le K-1} \mathbb{E}\big[\|\nabla f(w_k)\|_P^2\big]
= O\!\left(\frac{1}{K}\right).
\]
With uniform sampling $p_i = 1/n$ and $L_i = L$ for all $i$,
\[
\frac{1}{K}\sum_{k=0}^{K-1} \mathbb{E}\big[\|\nabla f(w_k)\|^2\big]
\le \frac{2nL\big(f(w_0) - f^\star\big)}{K}
= O\!\left(\frac{nL}{K}\right).
\]
\end{theorem}

\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Stationarity:} The bound certifies convergence to a stationary point
($\nabla f = 0$) in the weighted-gradient sense, the natural guarantee for
non-convex problems.
\item \textbf{Monotone expected descent:} Each step yields a non-negative expected
decrease (Lemma \ref{lem:descent}), so RCD is a descent method in expectation.
\item \textbf{Sampling sensitivity:} The factor $1/p_i$ governs the weighting. Sampling
proportional to $L_i$ ($p_i = L_i/\sum_j L_j$) balances coordinate contributions
and often improves the effective constant.
\item \textbf{Comparison to full gradient descent (GD):} GD attains
$O(L/K)$ on $\frac1K\sum \mathbb E\|\nabla f(w_k)\|^2$ but costs a full gradient per
step ($n$ partials). RCD costs one partial per step; per coordinate-update the
constants are comparable, with RCD trading a factor $n$ in iterations for a factor
$n$ cheaper iterations.
\item \textbf{Hyperparameter robustness:} Any $\eta_i \le 1/L_i$ preserves descent;
$\eta_i = 1/L_i$ maximizes the guaranteed per-step decrease.
\end{itemize}

\section*{Discussion}

The proof hinges on the reweighting by $1/p_i$: the expectation over the random
coordinate produces a $p_i$-weighted sum of squared partial derivatives. By design,
this weighted sum is exactly the quantity that telescopes against the global function
decrease $f(w_0)-f^\star$, converting per-coordinate progress into a global
stationarity certificate. The non-convex analysis tracks function values rather than
distances to optimum, since no minimizer uniqueness or convexity is available.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{lemma}[Coordinate descent lemma]
\label{lem:cdl}
Under Assumption \ref{ass:smooth}, for the RCD update
$w_{k+1} = w_k - \eta_{i_k}\nabla_{i_k} f(w_k) e_{i_k}$ with $0 < \eta_{i_k} \le 1/L_{i_k}$,
\[
f(w_{k+1}) \le f(w_k) - \frac{\eta_{i_k}}{2}\big(\nabla_{i_k} f(w_k)\big)^2 .
\]
\end{lemma}

\begin{proof}
\textbf{[Step 1]} Apply the coordinate descent inequality from Assumption
\ref{ass:smooth} with $w = w_k$, $i = i_k$, and step
$t = -\eta_{i_k}\nabla_{i_k} f(w_k)$:
\[
f(w_k + t e_{i_k}) \le f(w_k) + t\,\nabla_{i_k} f(w_k) + \frac{L_{i_k}}{2} t^2 .
\]

\textbf{[Step 2]} Substitute $t = -\eta_{i_k}\nabla_{i_k} f(w_k)$. Let
$g := \nabla_{i_k} f(w_k)$ for brevity:
\[
f(w_{k+1}) \le f(w_k) + (-\eta_{i_k} g)\, g + \frac{L_{i_k}}{2}(\eta_{i_k} g)^2
= f(w_k) - \eta_{i_k} g^2 + \frac{L_{i_k}\eta_{i_k}^2}{2} g^2 .
\]

\textbf{[Step 3]} Factor the $g^2$ terms:
\[
f(w_{k+1}) \le f(w_k) - \eta_{i_k}\Big(1 - \frac{L_{i_k}\eta_{i_k}}{2}\Big) g^2 .
\]

\textbf{[Step 4]} Since $\eta_{i_k} \le 1/L_{i_k}$, we have
$L_{i_k}\eta_{i_k} \le 1$, hence
$1 - \frac{L_{i_k}\eta_{i_k}}{2} \ge 1 - \frac12 = \frac12$. Therefore
$\eta_{i_k}\big(1 - \frac{L_{i_k}\eta_{i_k}}{2}\big) \ge \frac{\eta_{i_k}}{2}$,
and since $g^2 \ge 0$,
\[
f(w_{k+1}) \le f(w_k) - \frac{\eta_{i_k}}{2} g^2
= f(w_k) - \frac{\eta_{i_k}}{2}\big(\nabla_{i_k} f(w_k)\big)^2 .
\]
\end{proof}

\begin{lemma}[Expected one-step decrease with $1/p_i$ reweighting]
\label{lem:descent}
Under Assumptions \ref{ass:smooth}--\ref{ass:step}, conditioned on $w_k$,
\[
\mathbb{E}_{i_k}\big[ f(w_{k+1}) \mid w_k \big]
\le f(w_k) - \frac12 \sum_{i=1}^n p_i\, \eta_i \big(\nabla_i f(w_k)\big)^2 .
\]
With $\eta_i = 1/L_i$:
\[
\mathbb{E}_{i_k}\big[ f(w_{k+1}) \mid w_k \big]
\le f(w_k) - \frac12 \sum_{i=1}^n \frac{p_i}{L_i}\big(\nabla_i f(w_k)\big)^2 .
\]
\end{lemma}

\begin{proof}
\textbf{[Step 5]} Take the conditional expectation over the random coordinate $i_k$
(which equals $i$ with probability $p_i$ by Assumption \ref{ass:prob}) of the bound
in Lemma \ref{lem:cdl}. Since the right-hand side is a function of the realized
$i_k$:
\[
\mathbb{E}_{i_k}\big[ f(w_{k+1}) \mid w_k \big]
\le \mathbb{E}_{i_k}\!\left[ f(w_k) - \frac{\eta_{i_k}}{2}\big(\nabla_{i_k} f(w_k)\big)^2
\,\Big|\, w_k \right].
\]

\textbf{[Step 6]} Linearity of expectation, with $f(w_k)$ deterministic given $w_k$:
\[
= f(w_k) - \frac12\, \mathbb{E}_{i_k}\!\left[
\eta_{i_k}\big(\nabla_{i_k} f(w_k)\big)^2 \,\Big|\, w_k \right].
\]

\textbf{[Step 7]} Expand the expectation as a sum over coordinates weighted by $p_i$.
Here the probability $p_i$ enters as the weight; this is precisely the step where
the per-coordinate quantity is averaged according to the sampling law:
\[
\mathbb{E}_{i_k}\!\left[ \eta_{i_k}\big(\nabla_{i_k} f(w_k)\big)^2 \,\Big|\, w_k \right]
= \sum_{i=1}^n p_i\, \eta_i \big(\nabla_i f(w_k)\big)^2 .
\]

\textbf{[Step 8]} Combine Steps 6 and 7:
\[
\mathbb{E}_{i_k}\big[ f(w_{k+1}) \mid w_k \big]
\le f(w_k) - \frac12 \sum_{i=1}^n p_i\, \eta_i \big(\nabla_i f(w_k)\big)^2 .
\]
Substituting $\eta_i = 1/L_i$ yields the second statement.
\end{proof}

\noindent\textbf{Remark on the $1/p_i$ weighting.} In RCD, only coordinate $i_k$ is
updated. To relate the \emph{per-coordinate} progress to a \emph{global} quantity, one
typically reasons as follows: the expected progress along coordinate $i$ is scaled by
$p_i$ (Step 7). Conversely, if one wishes to recover the full unweighted gradient norm
$\sum_i (\nabla_i f)^2$, one reweights individual coordinate decreases by $1/p_i$, since
$\mathbb{E}_{i_k}[\frac{1}{p_{i_k}} \cdot (\text{decrease on } i_k)]
= \sum_i p_i \cdot \frac{1}{p_i} (\text{decrease on } i) = \sum_i (\text{decrease on } i)$.
The weighted norm $\|\cdot\|_P^2$ used in Theorem \ref{thm:main} is the natural object
arising directly from Step 7 without additional reweighting.

\section*{Main Proof (Step-by-Step Derivation)}

\begin{proof}[Proof of Theorem \ref{thm:main}]
\textbf{[Step 9]} Start from the conditional bound of Lemma \ref{lem:descent} with
$\eta_i = 1/L_i$:
\[
\mathbb{E}_{i_k}\big[ f(w_{k+1}) \mid w_k \big]
\le f(w_k) - \frac12 \sum_{i=1}^n \frac{p_i}{L_i}\big(\nabla_i f(w_k)\big)^2 .
\]

\textbf{[Step 10]} Take total expectation over all randomness up to iteration $k$
(the history $\mathcal{F}_k = \sigma(i_0,\dots,i_{k-1})$), using the tower property
$\mathbb{E}[\mathbb{E}[\,\cdot \mid w_k]] = \mathbb{E}[\,\cdot\,]$:
\[
\mathbb{E}\big[ f(w_{k+1}) \big]
\le \mathbb{E}\big[ f(w_k) \big]
- \frac12\, \mathbb{E}\!\left[ \sum_{i=1}^n \frac{p_i}{L_i}\big(\nabla_i f(w_k)\big)^2 \right].
\]

\textbf{[Step 11]} Define $\Delta_k := \mathbb{E}\!\left[ \sum_{i=1}^n
\frac{p_i}{L_i}\big(\nabla_i f(w_k)\big)^2 \right]
= \mathbb{E}\big[\|\nabla f(w_k)\|_P^2\big] \ge 0$. Rearranging Step 10:
\[
\frac12 \Delta_k \le \mathbb{E}\big[ f(w_k) \big] - \mathbb{E}\big[ f(w_{k+1}) \big].
\]

\textbf{[Step 12]} Sum this inequality over $k = 0, 1, \dots, K-1$. The right-hand
side telescopes:
\[
\frac12 \sum_{k=0}^{K-1} \Delta_k
\le \sum_{k=0}^{K-1}\Big( \mathbb{E}\big[ f(w_k) \big]
- \mathbb{E}\big[ f(w_{k+1}) \big] \Big)
= \mathbb{E}\big[ f(w_0) \big] - \mathbb{E}\big[ f(w_K) \big].
\]

\textbf{[Step 13]} Since $w_0$ is deterministic, $\mathbb{E}[f(w_0)] = f(w_0)$.
By Assumption \ref{ass:lb}, $f(w_K) \ge f^\star$, hence
$-\mathbb{E}[f(w_K)] \le -f^\star$. Therefore:
\[
\frac12 \sum_{k=0}^{K-1} \Delta_k \le f(w_0) - f^\star .
\]

\textbf{[Step 14]} Multiply both sides by $2/K$:
\[
\frac{1}{K} \sum_{k=0}^{K-1} \Delta_k \le \frac{2\big(f(w_0) - f^\star\big)}{K}.
\]

\textbf{[Step 15]} Since the minimum of a finite set is at most its average:
\[
\min_{0 \le k \le K-1} \Delta_k
\le \frac{1}{K} \sum_{k=0}^{K-1} \Delta_k
\le \frac{2\big(f(w_0) - f^\star\big)}{K}.
\]
Recalling $\Delta_k = \mathbb{E}\big[\|\nabla f(w_k)\|_P^2\big]$ proves the theorem.
\end{proof}

\section*{Rate Derivation (With Constants)}

\textbf{[Step 16]} From Step 14, the explicit constant is
$C := 2\big(f(w_0) - f^\star\big)$, giving
\[
\frac{1}{K}\sum_{k=0}^{K-1} \mathbb{E}\big[\|\nabla f(w_k)\|_P^2\big] \le \frac{C}{K}.
\]

\textbf{[Step 17]} To reach $\frac1K\sum_k \mathbb{E}\|\nabla f(w_k)\|_P^2 \le \epsilon^2$,
it suffices to take
\[
K \ge \frac{2\big(f(w_0) - f^\star\big)}{\epsilon^2},
\]
i.e.\ an $O(1/\epsilon^2)$ iteration complexity to reach an
$\epsilon$-stationary point (in weighted norm).

\section*{Special Cases}

\textbf{Case 1: Uniform sampling, equal smoothness ($p_i = 1/n$, $L_i = L$).}
Then $\|\nabla f(w)\|_P^2 = \sum_i \frac{1/n}{L}(\nabla_i f(w))^2
= \frac{1}{nL}\|\nabla f(w)\|^2$. Substituting into Step 14:
\[
\frac{1}{K}\sum_{k=0}^{K-1} \frac{1}{nL}\mathbb{E}\|\nabla f(w_k)\|^2
\le \frac{2(f(w_0)-f^\star)}{K}
\;\;\Longrightarrow\;\;
\frac{1}{K}\sum_{k=0}^{K-1}\mathbb{E}\|\nabla f(w_k)\|^2
\le \frac{2nL(f(w_0)-f^\star)}{K},
\]
recovering the $O(nL/K)$ rate stated in Theorem \ref{thm:main}.

\textbf{Case 2: Lipschitz-proportional sampling ($p_i = L_i/S$, $S = \sum_j L_j$).}
Then $\frac{p_i}{L_i} = \frac{1}{S}$ and
$\|\nabla f(w)\|_P^2 = \frac{1}{S}\sum_i (\nabla_i f(w))^2 = \frac{1}{S}\|\nabla f(w)\|^2$.
Hence
\[
\frac{1}{K}\sum_{k=0}^{K-1}\mathbb{E}\|\nabla f(w_k)\|^2
\le \frac{2S(f(w_0)-f^\star)}{K}
= \frac{2\big(\sum_j L_j\big)(f(w_0)-f^\star)}{K}.
\]
When $L_i = L$ for all $i$, $S = nL$ and this coincides with Case 1; for
heterogeneous $L_i$, $S = \sum_j L_j \le n \max_j L_j$ can be substantially smaller
than $n L_{\max}$, illustrating the benefit of importance sampling.

\textbf{Case 3: Single coordinate ($n=1$, $p_1=1$).}
The algorithm reduces to deterministic gradient descent with step $1/L$, and
Step 14 gives $\frac1K\sum_k \frac{1}{L}(\nabla f(w_k))^2 \le 2(f(w_0)-f^\star)/K$,
i.e.\ the classical $O(L/K)$ non-convex GD rate, consistent with the dry run
example above.
