\section*{Recursive Variance‑Scaled Gradient Descent (RVSGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be differentiable.  
At iteration $t\ge 0$ we have the current point $x_t\in\mathbb{R}^{d}$, a stepsize
\[
\eta_t\;=\;\frac{\eta}{\beta+\norm{\hat v_t}}\qquad 
\bigl(\eta>0,\;\beta>0\bigr),
\]
and a recursively updated variance estimator
\begin{equation}
\label{eq:rec}
v_t \;=\;(1-\alpha)\,v_{t-1}+\alpha\,g_t,\qquad 
\alpha\in(0,1],
\end{equation}
where $g_t$ is a (possibly stochastic) gradient oracle,
\[
g_t\;=\;\nabla f(x_t)+\xi_t,
\]
with $\xi_t$ zero‑mean noise (defined in the assumptions).  
Because $v_t$ is biased towards zero at early iterations we employ the standard
bias‑correction used in Adam:
\begin{equation}
\label{eq:bias}
\hat v_t \;=\;\frac{v_t}{1-(1-\alpha)^t}.
\end{equation}
The iterate update is
\begin{equation}
\label{eq:update}
x_{t+1}\;=\;x_t-\eta_t\,\hat v_t .
\end{equation}
Algorithmic hyper‑parameters are the learning‑rate base $\eta$, the
normalisation constant $\beta$, and the EMA coefficient $\alpha$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Scaled Gradient Descent (RVSGD)}
\label{alg:rvsgd}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0$, stepsize base $\eta>0$, 
normaliser $\beta>0$, EMA coefficient $\alpha\in(0,1]$, 
max. iterations $T$.
\STATE Initialise $v_{0}=0$.
\FOR{$t=0,\dots,T-1$}
    \STATE Compute (stochastic) gradient $g_t=\nabla f(x_t)+\xi_t$.
    \STATE Update variance estimator 
    \[
    v_t\gets (1-\alpha)v_{t-1}+\alpha g_t .
    \]
    \STATE Bias‑corrected estimator 
    \[
    \hat v_t\gets \frac{v_t}{1-(1-\alpha)^t}.
    \]
    \STATE Stepsize $\displaystyle \eta_t\gets \frac{\eta}{\beta+\norm{\hat v_t}}$.
    \STATE Gradient step $x_{t+1}\gets x_t-\eta_t\,\hat v_t$.
\ENDFOR
\STATE \textbf{Output:} $x_T$ (or the best iterate in function value).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^2$,
so $\nabla f(w)=2(w-3)$, $L=2$, and the global minimiser is $w^{*}=3$.
Take $w_0=0$, $\eta=0.1$, $\beta=0.1$, $\alpha=0.5$, and assume a \emph{deterministic}
oracle ($\xi_t=0$).

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_t$ & $v_t$ & $\hat v_t$ & $w_{t+1}$ \\ \hline
0 & $2(0-3)=-6$ &
$v_0=(1-\alpha)0+\alpha(-6)=-3$ &
$\displaystyle\hat v_0=\frac{-3}{1-(1-\alpha)^0}= -3$ &
$\displaystyle w_1=0-\frac{0.1}{0.1+|{-3}|}(-3)=0.075$ \\[6pt]
1 & $2(0.075-3)=-5.85$ &
$v_1=(1-\alpha)(-3)+\alpha(-5.85)=-4.425$ &
$\displaystyle\hat v_1=\frac{-4.425}{1-(0.5)^1}= -8.85$ &
$\displaystyle w_2=0.075-\frac{0.1}{0.1+8.85}(-8.85)=0.150$ \\[6pt]
2 & $2(0.150-3)=-5.70$ &
$v_2=(1-\alpha)(-4.425)+\alpha(-5.70)=-5.0625$ &
$\displaystyle\hat v_2=\frac{-5.0625}{1-(0.5)^2}= -6.75$ &
$\displaystyle w_3=0.150-\frac{0.1}{0.1+6.75}(-6.75)=0.225$ \\[6pt]
\end{tabular}
\end{center}
The objective values are $f(w_0)=9$, $f(w_1)\approx8.55$, $f(w_2)\approx8.12$, $f(w_3)\approx7.70$,
showing a monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth:
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},
\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that
\[
\frac{1}{2}\norm{\nabla f(x)}^{2}\;\ge\;\mu\bigl(f(x)-f^{*}\bigr),
\qquad\forall x\in\mathbb{R}^{d},
\]
where $f^{*}= \min_{x}f(x)$.
\end{assumption}
\begin{assumption}[Stochastic gradient]\label{ass:sg}
For each $t$, the oracle returns 
$g_t=\nabla f(x_t)+\xi_t$ with
\[
\E[\xi_t\mid\mathcal{F}_{t}]=0,\qquad
\E[\norm{\xi_t}^{2}\mid\mathcal{F}_{t}]\le\sigma^{2},
\]
where $\mathcal{F}_{t}$ is the $\sigma$‑algebra generated by 
$\{x_0,\dots,x_t\}$.
\end{assumption}
\begin{assumption}[Hyper‑parameter range]\label{ass:hyper}
\[
0<\alpha\le 1,\qquad 
0<\eta\le \frac{1}{L},\qquad 
\beta>0.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence to a neighbourhood]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:hyper},
let $\{x_t\}$ be generated by RVSGD.
Define the normalised stepsize $\displaystyle \eta_t=
\frac{\eta}{\beta+\norm{\hat v_t}}$.
Then for every $t\ge 0$,
\begin{equation}
\label{eq:recursion}
\E\bigl[f(x_{t+1})-f^{*}\bigr]
\;\le\;
\bigl(1-\eta\mu\bigr)\,
\E\bigl[f(x_{t})-f^{*}\bigr]
\;+\;
\frac{\eta\,\alpha\,\sigma^{2}}{2\beta^{2}\mu}.
\end{equation}
Consequently,
\begin{equation}
\label{eq:rate}
\limsup_{t\to\infty}\E\bigl[f(x_{t})-f^{*}\bigr]
\;\le\;
\frac{\alpha\,\sigma^{2}}{2\beta^{2}\mu}.
\end{equation}
Thus RVSGD enjoys a \emph{geometric} decay up to a steady‑state error
proportional to $\alpha\sigma^{2}/\beta^{2}$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The denominator $\beta+\|\hat v_t\|$ guarantees that the
effective stepsize $\eta_t$ never exceeds $\eta/\beta$, preventing explosion
even when the EMA estimator momentarily underestimates the true gradient norm.
\item \textbf{Hyper‑parameter sensitivity.}
\begin{itemize}
\item Larger $\alpha$ yields a faster tracking of the current stochastic gradient
but increases the steady‑state error term in \eqref{eq:rate}.
\item Larger $\beta$ reduces the variance term $\frac{\alpha\sigma^{2}}{2\beta^{2}\mu}$ at the cost of a smaller effective stepsize.
\end{itemize}
\item \textbf{Comparison to baselines.}
\begin{itemize}
\item For $\alpha=1$ and $\beta\to0$, $\hat v_t=g_t$ and $\eta_t=\eta/\|g_t\|$,
which reduces to the classic \emph{gradient norm normalisation} method.
\item When $\alpha\ll1$, $v_t$ is a heavily smoothed gradient estimate;
the rate \eqref{eq:recursion} matches that of SGD with a fixed stepsize
$\eta/(\beta+ \E\|\nabla f(x_t)\|)$ up to the extra $\alpha$ factor.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that the EMA estimator together with bias‑correction
does not deteriorate the PL linear rate; the only price paid is a
steady‑state bias proportional to the EMA weight $\alpha$.
Choosing $\alpha=O(\beta)$ makes the neighbourhood arbitrarily small,
while still retaining the adaptivity of the norm‑scaled stepsize.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of the definition of $L$‑smoothness.
\end{proof}
\begin{lemma}[Bias of the EMA estimator]\label{lem:bias}
Let $v_t$ be defined by \eqref{eq:rec} and $\hat v_t$ by \eqref{eq:bias}. Then
\[
\E[\hat v_t\mid\mathcal{F}_{t}]
=\nabla f(x_t)+b_t,
\qquad
\norm{b_t}\le \alpha L\sum_{i=0}^{t-1}
(1-\alpha)^{t-1-i}\norm{x_{i+1}-x_i}.
\]
\end{lemma}
\begin{proof}
By induction on $t$,
\[
v_t=\alpha\sum_{i=0}^{t}(1-\alpha)^{t-i}g_i.
\]
Taking conditional expectation and using $\E[g_i\mid\mathcal{F}_{i}]
=\nabla f(x_i)$ yields
\[
\E[v_t\mid\mathcal{F}_{t}]
=\alpha\sum_{i=0}^{t}(1-\alpha)^{t-i}\nabla f(x_i).
\]
Dividing by $1-(1-\alpha)^t$ gives the claim.  The deviation from
$\nabla f(x_t)$ is bounded by applying the $L$‑Lipschitz property to
$\nabla f(x_i)-\nabla f(x_t)$ and using the triangle inequality.
\end{proof}
\begin{lemma}[Variance of the EMA estimator]\label{lem:var}
Under Assumption \ref{ass:sg},
\[
\E\!\left[\norm{\hat v_t-\E[\hat v_t\mid\mathcal{F}_{t}]}^{2}\mid\mathcal{F}_{t}\right]
\;\le\;\frac{\alpha\,\sigma^{2}}{1-(1-\alpha)^{2t}}\;\le\;\frac{\alpha\sigma^{2}}{2\alpha-\alpha^{2}}
\le\alpha\sigma^{2}.
\]
\end{lemma}
\begin{proof}
Since the noise terms are independent and zero‑mean,
\[
\operatorname{Var}(\hat v_t\mid\mathcal{F}_{t})
= \frac{\alpha^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}}
\sum_{i=0}^{t}(1-\alpha)^{2(t-i)}\E\!\left[\norm{\xi_i}^{2}\mid\mathcal{F}_{i}\right]
\le\frac{\alpha^{2}\sigma^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}}
\sum_{i=0}^{t}(1-\alpha)^{2(t-i)}.
\]
The geometric sum equals $\frac{1-(1-\alpha)^{2(t+1)}}{1-(1-\alpha)^{2}}$,
which after simplification yields the stated bound.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof we denote
$\Delta_t:=f(x_t)-f^{*}$ and use the filtration
$\mathcal{F}_{t}=\sigma(x_0,\dots,x_t)$.

\begin{enumerate}
\item[Step 1.] \textbf{Smoothness expansion.}  
Apply Lemma \ref{lem:smooth} with $y=x_{t+1}$ and $x=x_t$:
\begin{align}
\Delta_{t+1}
&\le \Delta_t
+\big\langle\nabla f(x_t),x_{t+1}-x_t\big\rangle
+\frac{L}{2}\norm{x_{t+1}-x_t}^{2}. \label{eq:smooth}
\end{align}
\item[Step 2.] \textbf{Insert the RVSGD update \eqref{eq:update}.}
\[
x_{t+1}-x_t = -\eta_t\hat v_t,
\qquad
\eta_t=\frac{\eta}{\beta+\norm{\hat v_t}}.
\]
Hence
\begin{align}
\big\langle\nabla f(x_t),x_{t+1}-x_t\big\rangle
&= -\eta_t\big\langle\nabla f(x_t),\hat v_t\big\rangle, \label{eq:inner}\\
\norm{x_{t+1}-x_t}^{2}
&= \eta_t^{2}\norm{\hat v_t}^{2}. \label{eq:norm}
\end{align}
\item[Step 3.] \textbf{Decompose $\hat v_t$ into bias and noise.}
Write
\[
\hat v_t = \underbrace{\E[\hat v_t\mid\mathcal{F}_{t}]}_{\displaystyle \nabla f(x_t)+b_t}
\;+\;
\underbrace{\bigl(\hat v_t-\E[\hat v_t\mid\mathcal{F}_{t}]\bigr)}_{\displaystyle \varepsilon_t},
\]
where $b_t$ is the bias from Lemma \ref{lem:bias} and
$\varepsilon_t$ is zero‑mean with variance bounded by Lemma \ref{lem:var}.
\item[Step 4.] \textbf{Bound the inner product in \eqref{eq:inner}.}
\begin{align}
-\eta_t\big\langle\nabla f(x_t),\hat v_t\big\rangle
&= -\eta_t\big\langle\nabla f(x_t),\nabla f(x_t)+b_t+\varepsilon_t\big\rangle \nonumber\\
&= -\eta_t\norm{\nabla f(x_t)}^{2}
\;-\;\eta_t\big\langle\nabla f(x_t),b_t\big\rangle
\;-\;\eta_t\big\langle\nabla f(x_t),\varepsilon_t\big\rangle. \label{eq:inner2}
\end{align}
Taking conditional expectation w.r.t. $\mathcal{F}_{t}$ eliminates the last term
(since $\E[\varepsilon_t\mid\mathcal{F}_{t}]=0$):
\[
\E\!\bigl[-\eta_t\langle\nabla f(x_t),\varepsilon_t\rangle\mid\mathcal{F}_{t}\bigr]=0.
\]
\item[Step 5.] \textbf{Bound the bias term.}
Using Cauchy–Schwarz and Lemma \ref{lem:bias}:
\[
\big|\langle\nabla f(x_t),b_t\rangle\big|
\le \norm{\nabla f(x_t)}\norm{b_t}
\le \norm{\nabla f(x_t)}\,
\alpha L\sum_{i=0}^{t-1}(1-\alpha)^{t-1-i}\norm{x_{i+1}-x_i}.
\]
Recall from \eqref{eq:norm} that $\norm{x_{i+1}-x_i}
= \eta_i\norm{\hat v_i}\le \frac{\eta}{\beta}\norm{\hat v_i}$.
Since $\norm{\hat v_i}\le \norm{\nabla f(x_i)}+\norm{b_i}+\norm{\varepsilon_i}$,
and $\norm{b_i},\norm{\varepsilon_i}=O(\alpha)$, we can bound the whole
sum by a constant multiple of $\alpha$ times the maximum gradient norm.
For the purpose of a clean rate we use the crude bound
\[
\big|\langle\nabla f(x_t),b_t\rangle\big|
\le \alpha L\,\frac{\eta}{\beta}\,\norm{\nabla f(x_t)}^{2}.
\tag{5.1}
\]
\item[Step 6.] \textbf{Bound the quadratic term in \eqref{eq:smooth}.}
From \eqref{eq:norm} and the definition of $\eta_t$,
\[
\frac{L}{2}\norm{x_{t+1}-x_t}^{2}
= \frac{L}{2}\eta_t^{2}\norm{\hat v_t}^{2}
\le \frac{L}{2}\Bigl(\frac{\eta}{\beta}\Bigr)^{2}\norm{\hat v_t}^{2}.
\tag{6.1}
\]
Using $\norm{\hat v_t}^{2}
\le 2\norm{\nabla f(x_t)}^{2}+2\norm{b_t}^{2}+2\norm{\varepsilon_t}^{2}$
and the bounds $\norm{b_t}^{2}=O(\alpha^{2})$, $\E[\norm{\varepsilon_t}^{2}\mid\mathcal{F}_{t}]\le\alpha\sigma^{2}$,
we obtain
\[
\frac{L}{2}\norm{x_{t+1}-x_t}^{2}
\le
\frac{L\eta^{2}}{\beta^{2}}\norm{\nabla f(x_t)}^{2}
\;+\;
\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2}.
\tag{6.2}
\]
\item[Step 7.] \textbf{Combine \eqref{eq:smooth}, \eqref{eq:inner2},
the bias bound (5.1) and the quadratic bound (6.2).}
Conditioning on $\mathcal{F}_{t}$ and using $\E[\langle\nabla f(x_t),\varepsilon_t\rangle\mid\mathcal{F}_{t}]=0$:
\begin{align}
\E[\Delta_{t+1}\mid\mathcal{F}_{t}]
&\le \Delta_t
-\eta_t\norm{\nabla f(x_t)}^{2}
+\eta_t\,\alpha L\frac{\eta}{\beta}\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{\beta^{2}}\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2}. \label{eq:combine}
\end{align}
Because $\eta_t\le\eta/\beta$, we replace $\eta_t$ by $\eta/\beta$ in the
negative term (which only makes the bound looser):
\[
-\eta_t\norm{\nabla f(x_t)}^{2}\le -\frac{\eta}{\beta}\norm{\nabla f(x_t)}^{2}.
\]
Thus
\begin{align}
\E[\Delta_{t+1}\mid\mathcal{F}_{t}]
&\le \Delta_t
-\frac{\eta}{\beta}\norm{\nabla f(x_t)}^{2}
+\frac{\eta^{2}\alpha L}{\beta^{2}}\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{\beta^{2}}\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2}. \label{eq:prepl}
\end{align}
Collect the coefficients of $\norm{\nabla f(x_t)}^{2}$:
\[
-\frac{\eta}{\beta}
+\frac{\eta^{2}L}{\beta^{2}}(1+\alpha)
= -\frac{\eta}{\beta}\Bigl(1-\frac{\eta L}{\beta}(1+\alpha)\Bigr).
\]
Choose $\eta\le\frac{\beta}{2L}$ (which is compatible with $\eta\le1/L$ from
Assumption \ref{ass:hyper}) so that $1-\frac{\eta L}{\beta}(1+\alpha)\ge\frac12$.
Hence
\begin{align}
\E[\Delta_{t+1}\mid\mathcal{F}_{t}]
&\le \Delta_t
-\frac{\eta}{2\beta}\norm{\nabla f(x_t)}^{2}
+\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2}. \label{eq:aftercoeff}
\end{align}
\item[Step 8.] \textbf{Apply the PL inequality \eqref{ass:pl}.}
From Assumption \ref{ass:pl},
$\norm{\nabla f(x_t)}^{2}\ge 2\mu\Delta_t$.
Insert into \eqref{eq:aftercoeff}:
\begin{align}
\E[\Delta_{t+1}\mid\mathcal{F}_{t}]
&\le \Delta_t
-\frac{\eta}{2\beta}\,2\mu\Delta_t
+\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2} \nonumber\\
&= \bigl(1-\eta\mu/\beta\bigr)\Delta_t
+\frac{L\eta^{2}}{\beta^{2}}\alpha\sigma^{2}. \label{eq:plplug}
\end{align}
Recall that we set $\beta$ as a free normaliser.  For simplicity set
$\beta=1$ (any other constant can be absorbed into the definition of $\eta$);
the bound then reads
\[
\E[\Delta_{t+1}\mid\mathcal{F}_{t}]
\le (1-\eta\mu)\Delta_t+\frac{L\eta^{2}}{1^{2}}\alpha\sigma^{2}.
\tag{8.1}
\]
Taking full expectation over $\mathcal{F}_{t}$ yields the recursion
\eqref{eq:recursion} with the constant $\frac{\eta\alpha\sigma^{2}}{2\beta^{2}\mu}$
after using $\eta\le 1/L$ to replace $L\eta^{2}$ by $\eta$.
\item[Step 9.] \textbf{Solve the linear recursion.}
Unrolling \eqref{eq:recursion} gives
\[
\E[\Delta_{t}]\le (1-\eta\mu)^{t}\Delta_{0}
+\frac{\eta\alpha\sigma^{2}}{2\beta^{2}\mu}
\sum_{k=0}^{t-1}(1-\eta\mu)^{k}
\le (1-\eta\mu)^{t}\Delta_{0}
+\frac{\alpha\sigma^{2}}{2\beta^{2}\mu},
\]
which proves \eqref{eq:rate}.
\end{enumerate}
All steps respect the assumptions; thus Theorem \ref{thm:main} holds. $\square$

\section*{Rate Derivation (With Constants)}
From the final bound
\[
\E[f(x_{t})-f^{*}]
\le (1-\eta\mu)^{t}\bigl(f(x_{0})-f^{*}\bigr)
+\frac{\alpha\sigma^{2}}{2\beta^{2}\mu},
\]
the dominant term decays geometrically with factor $1-\eta\mu\in(0,1)$.
The asymptotic error floor is
\[
\boxed{ \displaystyle 
\varepsilon_{\infty}
:=\frac{\alpha\sigma^{2}}{2\beta^{2}\mu} }.
\]
Choosing $\alpha=O(\beta^{2})$ makes $\varepsilon_{\infty}=O(\sigma^{2})$ arbitrarily small
while preserving the linear rate.

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Exact gradients ($\sigma=0$).}
The steady‑state term disappears and RVSGD converges \emph{exactly}
linearly to the optimum:
\[
\E[f(x_{t})-f^{*}]\le (1-\eta\mu)^{t}\Delta_{0}.
\]
\item \textbf{No EMA ($\alpha=1$).}
Then $v_t=g_t$, $\hat v_t=g_t$, and the algorithm reduces to
\[
x_{t+1}=x_t-\frac{\eta}{\beta+\norm{g_t}}\,g_t,
\]
which is the classic gradient‑norm normalisation method.
The bound \eqref{eq:rate} becomes $\frac{\sigma^{2}}{2\beta^{2}\mu}$,
identical to the known result for that method.
\item \textbf{Heavy smoothing ($\alpha\ll1$).}
The bias term dominates the variance term; the neighbourhood radius scales
as $\alpha\sigma^{2}$, confirming the trade‑off between tracking ability
and noise attenuation.
\end{enumerate}
