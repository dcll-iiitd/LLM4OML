\section*{Algorithm Name}
\textbf{Gradient Descent Ascent (GDA)} for Convex-Concave Saddle Point Problems.

\section*{Mathematical Formulation}
Consider a function $L: \mathcal{X} \times \mathcal{Y} \to \mathbb{R}$ where $\mathcal{X} \subseteq \mathbb{R}^{d_x}$, $\mathcal{Y} \subseteq \mathbb{R}^{d_y}$ are convex sets. We assume $L$ is \textit{convex} in $y$ and \textit{concave} in $x$. The saddle point problem is
\[
\min_{y \in \mathcal{Y}} \max_{x \in \mathcal{X}} L(x,y).
\]
A pair $(x^*, y^*)$ is a saddle point if
\[
L(x^*, y) \le L(x^*, y^*) \le L(x, y^*) \quad \forall x \in \mathcal{X}, y \in \mathcal{Y}.
\]
Define the combined variable $z = (x, y) \in \mathcal{Z} = \mathcal{X} \times \mathcal{Y}$ and the operator
\[
F(z) = \begin{pmatrix} \nabla_x L(x,y) \\ -\nabla_y L(x,y) \end{pmatrix}.
\]
The saddle point condition is equivalent to the variational inequality: find $z^* \in \mathcal{Z}$ such that
\[
\langle F(z^*), z - z^* \rangle \ge 0 \quad \forall z \in \mathcal{Z}.
\]
If $\mathcal{Z} = \mathbb{R}^{d_x+d_y}$ (unconstrained), this reduces to $F(z^*) = 0$.

\textbf{Update Rules:} Given step size $\eta_t > 0$, the Gradient Descent Ascent (GDA) algorithm performs the simultaneous updates:
\begin{align}
x_{t+1} &= x_t + \eta_t \nabla_x L(x_t, y_t), \label{eq:gda_x} \\
y_{t+1} &= y_t - \eta_t \nabla_y L(x_t, y_t). \label{eq:gda_y}
\end{align}
Equivalently, in the combined variable $z_t = (x_t, y_t)$:
\[
z_{t+1} = z_t + \eta_t G(z_t), \quad \text{where } G(z) = \begin{pmatrix} \nabla_x L(x,y) \\ -\nabla_y L(x,y) \end{pmatrix}.
\]
Note that $G(z) = F(z)$. The algorithm uses the same step size $\eta_t$ for both ascent and descent.

\textbf{Hyperparameters:}
\begin{itemize}
    \item Step size sequence $\{\eta_t\}_{t \ge 1}$ (e.g., constant $\eta_t = \eta$ or diminishing $\eta_t = \frac{\eta}{\sqrt{t}}$).
    \item Initial point $z_0 = (x_0, y_0)$.
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent Ascent (GDA)}
\begin{algorithmic}
\REQUIRE Function $L(x,y)$, initial point $(x_0, y_0)$, step size schedule $\{\eta_t\}$, max iterations $T$.
\STATE $x \gets x_0$, $y \gets y_0$
\FOR{$t = 1$ to $T$}
    \STATE Compute gradients: $g_x \gets \nabla_x L(x, y)$, $g_y \gets \nabla_y L(x, y)$
    \STATE $x \gets x + \eta_t \cdot g_x$ \quad (Gradient Ascent on $x$)
    \STATE $y \gets y - \eta_t \cdot g_y$ \quad (Gradient Descent on $y$)
    \STATE (Optional) Project onto $\mathcal{X}, \mathcal{Y}$ if constrained.
\ENDFOR
\RETURN $(x, y)$ or averaged iterates $(\bar{x}_T, \bar{y}_T) = \frac{1}{T}\sum_{t=1}^T (x_t, y_t)$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We apply GDA to the simple convex function $f(w) = (w-3)^2$. This can be viewed as a saddle point problem with $L(x,y) = (y-3)^2$ (independent of $x$). Then $\nabla_x L = 0$, $\nabla_y L = 2(y-3)$. The updates become:
\[
x_{t+1} = x_t, \qquad y_{t+1} = y_t - \eta \cdot 2(y_t - 3).
\]
Initialize $x_0 = 0$, $y_0 = 0$, step size $\eta = 0.1$. Perform 3 iterations.

\section*{Assumptions}
We make the following standard assumptions for convergence analysis.

\begin{enumerate}
    \item \textbf{Convex-Concave Structure:} $L(x,y)$ is convex in $y$ for every fixed $x$, and concave in $x$ for every fixed $y$.
    \item \textbf{Smoothness:} $L$ is $L$-smooth, i.e., its gradient $\nabla L = (\nabla_x L, \nabla_y L)$ is $L$-Lipschitz continuous:
    \[
    \|\nabla L(z) - \nabla L(z')\| \le L \|z - z'\| \quad \forall z, z' \in \mathcal{Z}.
    \]
    Equivalently, the operator $F(z) = (\nabla_x L, -\nabla_y L)$ is $L$-Lipschitz.
    \item \textbf{Existence of Saddle Point:} The solution set $\mathcal{Z}^* = \{z^* \in \mathcal{Z} : F(z^*) = 0\}$ is nonempty. (For constrained domains, $z^*$ satisfies the variational inequality.)
    \item \textbf{Bounded Domain / Iterates:} The iterates remain in a bounded set, or we assume $\|z_t - z^*\| \le R$ for all $t$ and some $R > 0$. (This holds automatically if $\mathcal{Z}$ is compact or if step sizes are chosen appropriately.)
    \item \textbf{Step Size Schedule (Diminishing):} $\eta_t = \frac{\eta}{\sqrt{t}}$ for $t \ge 1$, with $\eta > 0$ a constant. 
    \item \textbf{Step Size Schedule (Constant) -- Alternative Case:} $\eta_t = \eta$ with $\eta \le 1/L$. This is analyzed separately in Section \ref{sec:constant-step}.
\end{enumerate}

\section*{Main Convergence Theorem}
Under Assumptions 1--5, let $\{z_t\}_{t=1}^T$ be the iterates generated by GDA. Define the averaged iterate $\bar{z}_T = \frac{1}{T}\sum_{t=1}^T z_t$. Then for any $z \in \mathcal{Z}$, the saddle point gap (primal-dual gap) satisfies
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{R^2}{2\eta\sqrt{T}} + \frac{2\eta G^2}{\sqrt{T}},
\]
where $R = \|z_1 - z^*\|$ (assuming $z_1$ is the first iterate after initialization) and $G$ is a uniform bound on $\|F(z_t)\|$ (which exists due to smoothness and bounded iterates; see Lemma 3). In particular, the gap converges as $O\left(\frac{1}{\sqrt{T}}\right)$.

Optimizing the constant $\eta$ yields $\eta = \frac{R}{\sqrt{2}G}$ and the bound
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{\sqrt{2} R G}{\sqrt{T}}.
\]

If a constant step size $\eta_t = \eta \le \frac{1}{L}$ is used and $L$ is additionally \textit{strongly} convex-concave, then GDA converges linearly to the unique saddle point (see Section \ref{sec:strongly-convex}).

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} GDA with diminishing step sizes is stable for any convex-concave smooth problem. With constant step size, stability requires $\eta \le 1/L$; larger step sizes may cause divergence (e.g., cycling in bilinear games).
    \item \textbf{Hyperparameter Sensitivity:} The convergence rate depends on the choice of $\eta$. The optimal $\eta$ scales with $R/G$. Adaptive step sizes (e.g., AdaGrad) can improve practical performance.
    \item \textbf{Comparison to Baselines:}
    \begin{itemize}
        \item \textit{Extragradient (EG)} achieves $O(1/T)$ rate for smooth convex-concave problems with constant step size, but requires two gradient evaluations per iteration.
        \item \textit{Optimistic Gradient Descent Ascent (OGDA)} also achieves $O(1/T)$ with a single gradient evaluation per iteration (using past gradient).
        \item GDA is simpler but slower ($O(1/\sqrt{T})$ with diminishing step size) and may not converge with constant step size without strong convexity-concavity.
    \end{itemize}
    \item \textbf{Last-Iterate Convergence:} The theorem above guarantees convergence of the \textit{averaged} iterates. Last-iterate convergence for GDA in convex-concave settings is not guaranteed in general (it may cycle), but holds for strongly convex-concave or with appropriate modifications.
\end{itemize}

\section*{Discussion}
GDA is the most natural first-order method for saddle point problems. Its convergence theory for general convex-concave smooth functions requires diminishing step sizes to guarantee convergence, yielding a suboptimal $O(1/\sqrt{T})$ rate. This is because the operator $F$ is monotone and Lipschitz but not necessarily cocoercive. The forward method (GDA) on monotone Lipschitz variational inequalities converges only with vanishing step sizes. Stronger assumptions (strong convexity-concavity, cocoercivity, or using extragradient/optimistic methods) are needed for faster rates or constant step sizes. In practice, GDA with constant step size often works well for deep learning minimax problems (e.g., GANs) despite the lack of theoretical guarantees, possibly due to implicit regularization or the structure of the problem.

\section*{Preliminaries (Definitions and Lemmas)}

\textbf{Definition 1 (Monotone Operator).} An operator $F: \mathcal{Z} \to \mathbb{R}^d$ is monotone if
\[
\langle F(z) - F(z'), z - z' \rangle \ge 0 \quad \forall z, z' \in \mathcal{Z}.
\]
For a convex-concave $L$, the operator $F(z) = (\nabla_x L, -\nabla_y L)$ is monotone. [Step 1]

\textbf{Definition 2 (Lipschitz Continuity).} $F$ is $L$-Lipschitz if
\[
\|F(z) - F(z')\| \le L \|z - z'\| \quad \forall z, z' \in \mathcal{Z}.
\]
This is equivalent to $L$-smoothness of $L$. [Step 2]

\textbf{Lemma 1 (Key Inequality).} For any $z, z' \in \mathcal{Z}$ and step size $\eta > 0$, the GDA update $z_{t+1} = z_t + \eta F(z_t)$ satisfies
\[
\langle F(z_t), z_t - z' \rangle = \frac{1}{2\eta} \left( \|z_t - z'\|^2 + \|z_t - z_{t+1}\|^2 - \|z_{t+1} - z'\|^2 \right).
\]
\textit{Proof.} Start from the identity $\langle a - b, a - c \rangle = \frac{1}{2}(\|a-c\|^2 + \|a-b\|^2 - \|b-c\|^2)$. Set $a = z_t$, $b = z_{t+1}$, $c = z'$. Note that $z_t - z_{t+1} = -\eta F(z_t)$. Then
\[
\langle z_t - z_{t+1}, z_t - z' \rangle = \langle -\eta F(z_t), z_t - z' \rangle = -\eta \langle F(z_t), z_t - z' \rangle.
\]
Rearranging gives the lemma. [Step 3]

\textbf{Lemma 2 (Monotonicity Bound).} By monotonicity of $F$, for any $z' \in \mathcal{Z}$,
\[
\langle F(z_t), z_t - z' \rangle \ge \langle F(z'), z_t - z' \rangle.
\]
[Step 4]

\textbf{Lemma 3 (Uniform Bound on Operator Norm).} Under Assumptions 2 and 4, there exists a constant $G$ such that $\|F(z_t)\| \le G$ for all $t$. Specifically, for any $z^* \in \mathcal{Z}^*$,
\[
\|F(z_t)\| = \|F(z_t) - F(z^*)\| \le L \|z_t - z^*\| \le L R.
\]
Thus we can take $G = L R$. [Step 5]

\section*{Main Proof (Step-by-Step Derivation)}

We now prove the convergence rate for GDA with diminishing step sizes $\eta_t = \eta/\sqrt{t}$.

\textbf{Step 1: Apply Lemma 1 at each iteration.}
For each $t = 1, \dots, T$, using step size $\eta_t$, we have
\[
\langle F(z_t), z_t - z \rangle = \frac{1}{2\eta_t} \left( \|z_t - z\|^2 + \|z_t - z_{t+1}\|^2 - \|z_{t+1} - z\|^2 \right). \tag{1}
\]
[Step 6]

\textbf{Step 2: Use monotonicity (Lemma 2) to lower bound the left-hand side.}
\[
\langle F(z), z_t - z \rangle \le \langle F(z_t), z_t - z \rangle. \tag{2}
\]
[Step 7]

\textbf{Step 3: Combine (1) and (2).}
\[
\langle F(z), z_t - z \rangle \le \frac{1}{2\eta_t} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 + \|z_t - z_{t+1}\|^2 \right). \tag{3}
\]
[Step 8]

\textbf{Step 4: Bound the step size difference term.}
Since $z_{t+1} - z_t = \eta_t F(z_t)$, we have $\|z_t - z_{t+1}\|^2 = \eta_t^2 \|F(z_t)\|^2$. By Lemma 3, $\|F(z_t)\| \le G$. Thus
\[
\|z_t - z_{t+1}\|^2 \le \eta_t^2 G^2. \tag{4}
\]
[Step 9]

\textbf{Step 5: Substitute (4) into (3).}
\[
\langle F(z), z_t - z \rangle \le \frac{1}{2\eta_t} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 \right) + \frac{\eta_t G^2}{2}. \tag{5}
\]
[Step 10]

\textbf{Step 6: Sum over $t = 1$ to $T$.}
\[
\sum_{t=1}^T \langle F(z), z_t - z \rangle \le \sum_{t=1}^T \frac{1}{2\eta_t} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 \right) + \frac{G^2}{2} \sum_{t=1}^T \eta_t. \tag{6}
\]
[Step 11]

\textbf{Step 7: Handle the telescoping sum with varying $\eta_t$.}
We use the following algebraic manipulation (Abel's summation / summation by parts):
\[
\sum_{t=1}^T \frac{1}{\eta_t} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 \right)
= \frac{\|z_1 - z\|^2}{\eta_1} + \sum_{t=2}^T \|z_t - z\|^2 \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right) - \frac{\|z_{T+1} - z\|^2}{\eta_T}.
\]
Since $\frac{1}{\eta_t} = \frac{\sqrt{t}}{\eta}$ is increasing, the coefficients $\frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \ge 0$. Also $\|z_t - z\|^2 \le R^2$ for all $t$ (by boundedness assumption, where $R$ is an upper bound on $\|z_t - z\|$; we can take $z = z^*$ and $R = \|z_1 - z^*\|$). Thus
\[
\sum_{t=1}^T \frac{1}{\eta_t} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 \right)
\le \frac{R^2}{\eta_1} + R^2 \sum_{t=2}^T \left( \frac{1}{\eta_t} - \frac{1}{\eta_{t-1}} \right)
= \frac{R^2}{\eta_T} = \frac{R^2 \sqrt{T}}{\eta}. \tag{7}
\]
[Step 12]

\textbf{Step 8: Bound the sum of step sizes.}
\[
\sum_{t=1}^T \eta_t = \eta \sum_{t=1}^T \frac{1}{\sqrt{t}} \le \eta \left( 1 + \int_1^T \frac{1}{\sqrt{t}} dt \right) = \eta \left( 1 + 2(\sqrt{T} - 1) \right) \le 2\eta \sqrt{T}. \tag{8}
\]
A tighter bound is $\sum_{t=1}^T \frac{1}{\sqrt{t}} \le 2\sqrt{T} - 1$. We'll use $2\eta\sqrt{T}$ for simplicity. [Step 13]

\textbf{Step 9: Combine (7) and (8) into (6).}
\[
\sum_{t=1}^T \langle F(z), z_t - z \rangle \le \frac{1}{2} \cdot \frac{R^2 \sqrt{T}}{\eta} + \frac{G^2}{2} \cdot 2\eta \sqrt{T}
= \frac{R^2 \sqrt{T}}{2\eta} + \eta G^2 \sqrt{T}. \tag{9}
\]
[Step 14]

\textbf{Step 10: Relate to the averaged iterate $\bar{z}_T = \frac{1}{T}\sum_{t=1}^T z_t$.}
By linearity of the inner product,
\[
\sum_{t=1}^T \langle F(z), z_t - z \rangle = T \langle F(z), \bar{z}_T - z \rangle.
\]
Thus
\[
T \langle F(z), \bar{z}_T - z \rangle \le \sqrt{T} \left( \frac{R^2}{2\eta} + \eta G^2 \right).
\]
Dividing by $T$:
\[
\langle F(z), \bar{z}_T - z \rangle \le \frac{1}{\sqrt{T}} \left( \frac{R^2}{2\eta} + \eta G^2 \right). \tag{10}
\]
[Step 15]

\textbf{Step 11: Take supremum over $z \in \mathcal{Z}$.}
The left-hand side is the saddle point gap (primal-dual gap) at $\bar{z}_T$:
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{1}{\sqrt{T}} \left( \frac{R^2}{2\eta} + \eta G^2 \right). \tag{11}
\]
This is $O(1/\sqrt{T})$. [Step 16]

\textbf{Step 12: Optimize the constant $\eta$.}
The bound $\frac{R^2}{2\eta} + \eta G^2$ is minimized when $\frac{R^2}{2\eta^2} = G^2$, i.e., $\eta = \frac{R}{\sqrt{2} G}$. Substituting gives
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{\sqrt{2} R G}{\sqrt{T}}. \tag{12}
\]
[Step 17]

\section*{Rate Derivation (With Constants)}

From (11), we have the explicit bound:
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{R^2}{2\eta\sqrt{T}} + \frac{2\eta G^2}{\sqrt{T}}.
\]
Optimizing over $\eta$ gives $\eta = \frac{R}{\sqrt{2}G}$, yielding
\[
\sup_{z \in \mathcal{Z}} \langle F(z), \bar{z}_T - z \rangle \le \frac{\sqrt{2} R G}{\sqrt{T}}.
\]
Thus the saddle point gap converges as $O(1/\sqrt{T})$ with constant $\sqrt{2} R G$.

If we use a constant step size $\eta_t = \eta \le 1/L$, the telescoping sum in Step 7 becomes
\[
\sum_{t=1}^T \frac{1}{2\eta} \left( \|z_t - z\|^2 - \|z_{t+1} - z\|^2 \right) = \frac{\|z_1 - z\|^2 - \|z_{T+1} - z\|^2}{2\eta} \le \frac{R^2}{2\eta}.
\]
The sum of step sizes is $T\eta$. Then (9) becomes
\[
\sum_{t=1}^T \langle F(z), z_t - z \rangle \le \frac{R^2}{2\eta} + \frac{\eta G^2 T}{2}.
\]
Dividing by $T$:
\[
\langle F(z), \bar{z}_T - z \rangle \le \frac{R^2}{2\eta T} + \frac{\eta G^2}{2}.
\]
This does \textit{not} converge to zero as $T \to \infty$; the second term remains constant. Hence constant step size does not guarantee convergence of the gap for general convex-concave smooth problems. (It may converge to a neighborhood.)

\section*{Special Cases}

\subsection*{Strongly Convex-Concave Case} \label{sec:strongly-convex}
If $L$ is $\mu$-strongly convex in $y$ and $\mu$-strongly concave in $x$, then $F$ is $\mu$-strongly monotone:
\[
\langle F(z) - F(z'), z - z' \rangle \ge \mu \|z - z'\|^2.
\]
With constant step size $\eta \le \frac{\mu}{L^2}$, GDA converges linearly:
\[
\|z_{t+1} - z^*\|^2 \le (1 - \eta \mu) \|z_t - z^*\|^2.
\]
More generally, for any $\eta \le 1/L$, the contraction factor is $1 - 2\eta\mu + \eta^2 L^2$. The condition $\eta \le \mu/L^2$ ensures $1 - 2\eta\mu + \eta^2 L^2 \le 1 - \eta\mu$.
\textit{Proof:} Expand $\|z_{t+1} - z^*\|^2 = \|z_t - z^* + \eta F(z_t)\|^2 = \|z_t - z^*\|^2 + 2\eta \langle F(z_t), z_t - z^* \rangle + \eta^2 \|F(z_t)\|^2$. Using strong monotonicity $\langle F(z_t), z_t - z^* \rangle \ge \mu \|z_t - z^*\|^2$ and Lipschitz continuity $\|F(z_t)\| \le L \|z_t - z^*\|$, we get $\|z_{t+1} - z^*\|^2 \le (1 - 2\eta\mu + \eta^2 L^2) \|z_t - z^*\|^2$. [Step 18]

\subsection*{Bilinear Game}
$L(x,y) = x^\top A y$. Then $F(z) = (A y, -A^\top x)$. This operator is monotone and Lipschitz but not strongly monotone. GDA with constant step size cycles and does not converge. Diminishing step sizes $\eta_t = \eta/\sqrt{t}$ yield $O(1/\sqrt{T})$ convergence of the averaged iterates.

\subsection*{Constrained Domains}
If $\mathcal{X}, \mathcal{Y}$ are closed convex sets, the updates include projections:
\[
x_{t+1} = \Pi_{\mathcal{X}}(x_t + \eta_t \nabla_x L), \quad y_{t+1} = \Pi_{\mathcal{Y}}(y_t - \eta_t \nabla_y L).
\]
The same convergence proof holds using the non-expansiveness of the projection operator: $\|\Pi_{\mathcal{Z}}(u) - \Pi_{\mathcal{Z}}(v)\| \le \|u - v\|$.
