\section*{Algorithm Name}
\textbf{Top‑$K$ Error‑Feedback Stochastic Gradient Descent (Top‑$K$ EF‑SGD)}  

The method communicates only the $K$ entries of the (stochastic) gradient with largest magnitude and keeps the residual (the “error”) locally for later correction.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\!\to\!\R$ be the (possibly non‑convex) loss.  
At iteration $t$ the worker holds the model $w_{t}\in\R^{d}$ and an error vector $e_{t}\in\R^{d}$ (initially $e_{0}=0$).  

\begin{enumerate}[label=(\alph*)]
\item Sample a stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$ where $\xi_{t}$ is a data minibatch.
\item Form the \emph{pre‑compressed} vector
\[
u_{t}=g_{t}+e_{t-1}. \tag{1}
\]
\item Apply the deterministic \emph{top‑$K$} compression operator
\[
c_{t}= \cK(u_{t})\in\R^{d},
\]
where $\cK(\cdot)$ keeps the $K$ components of largest absolute value and sets the remaining $d-K$ components to zero.
\item Update the local error (the part not communicated)
\[
e_{t}=u_{t}-c_{t}. \tag{2}
\]
\item Global model update (single‑node or after an all‑reduce of the $c_{t}$)
\[
w_{t+1}= w_{t}-\eta\,c_{t}, \qquad \eta>0 \text{ is the stepsize}. \tag{3}
\]
\end{enumerate}

The algorithm repeats (a)–(e) for $t=0,1,\dots ,T-1$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑$K$ EF‑SGD}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, sparsity $K\in\{1,\dots ,d\}$
\State Initialize $w_{0}\in\R^{d}$, $e_{0}=0$
\For{$t=0,1,\dots ,T-1$}
    \State Sample minibatch $\xi_{t}$ and compute stochastic gradient $g_{t}= \nabla f(w_{t};\xi_{t})$
    \State $u_{t}\gets g_{t}+e_{t-1}$                         \Comment{(1)}
    \State $c_{t}\gets \cK(u_{t})$                          \Comment{keep top $K$ entries}
    \State $e_{t}\gets u_{t}-c_{t}$                         \Comment{(2)}
    \State $w_{t+1}\gets w_{t}-\eta\,c_{t}$                 \Comment{(3)}
\EndFor
\State \textbf{Output:} $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic $f(w)=(w-3)^{2}$, thus $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, stepsize $\eta=0.1$, and $K=1$ (the top‑$1$ of a scalar is the scalar itself).  
Because the problem is deterministic we have $g_{t}=\nabla f(w_{t})$ and $e_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_{t}$ & $g_{t}=2(w_{t}-3)$ & $c_{t}$ (top‑$1$) & $w_{t+1}=w_{t}-\eta c_{t}$ \\ \hline
0 & $0$ & $-6$ & $-6$ & $0-0.1(-6)=0.6$ \\ \hline
1 & $0.6$ & $2(0.6-3)=-4.8$ & $-4.8$ & $0.6-0.1(-4.8)=1.08$ \\ \hline
2 & $1.08$ & $2(1.08-3)=-3.84$ & $-3.84$ & $1.08-0.1(-3.84)=1.464$ \\ \hline
3 & $1.464$ & $2(1.464-3)=-3.072$ & $-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}

The objective values shrink:
\[
f(w_{0})=9,\; f(w_{1})=(0.6-3)^{2}=5.76,\; f(w_{2})\approx3.6864,\; f(w_{3})\approx2.650.
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}[label=(A\arabic*)]
\item \textbf{Smoothness.} $f$ is $L$‑smooth:
\[
f(x)\le f(y)+\langle\nabla f(y),x-y\rangle+\frac{L}{2}\norm{x-y}^{2},
\qquad\forall x,y\in\R^{d}. \tag{A1}
\]
\item \textbf{Unbiased stochastic gradients with bounded variance.}
\[
\E_{\xi_{t}}\!\left[g_{t}\mid w_{t}\right]=\nabla f(w_{t}),\qquad
\E_{\xi_{t}}\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2}. \tag{A2}
\]
\item \textbf{Compression contraction.} The top‑$K$ operator $\cK$ satisfies, for any $x\in\R^{d}$,
\[
\norm{\cK(x)-x}^{2}\le (1-\delta)\norm{x}^{2},\qquad\delta:=\frac{K}{d}\in(0,1]. \tag{A3}
\]
(Proof: the $K$ kept coordinates contain at least a $\frac{K}{d}$ fraction of the squared $\ell_{2}$‑norm.)
\item \textbf{Stepsize.} $\eta\le \frac{1}{L}$.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex convergence of Top‑$K$ EF‑SGD]\label{thm:main}
Under Assumptions (A1)–(A4) and after $T$ iterations,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\sigma^{2}}_{\text{stochastic variance term}}
\;+\;
\underbrace{\frac{2L(1-\delta)}{\delta}\,\eta^{2}\sigma^{2}}_{\text{compression error term}} .
\tag{4}
\]
Consequently, choosing $\eta=\Theta\!\bigl(T^{-1/2}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]=\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The error–feedback mechanism guarantees that the accumulated compression error $e_{t}$ does not diverge; Lemma~\ref{lem:error-bound} shows $\E[\norm{e_{t}}^{2}]$ stays $\mathcal{O}(\eta^{2}\sigma^{2})$.
\item \textbf{Hyper‑parameter sensitivity.} The contraction factor $\delta=K/d$ directly influences the third term of \eqref{4}. Larger $K$ (more communicated coordinates) improves the bound linearly.
\item \textbf{Comparison to baselines.}
    \begin{enumerate}
    \item Plain SGD (no compression) corresponds to $\delta=1$, making the compression term vanish and recovering the classical $\mathcal{O}(1/\sqrt{T})$ rate.
    \item Top‑$K$ without error feedback lacks the error‑feedback term; the error can accumulate leading to a bias that prevents convergence in the non‑convex setting.
    \end{enumerate}
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For an $L$‑smooth function,
\[
f(w_{t+1})\le f(w_{t})+\langle\nabla f(w_{t}),w_{t+1}-w_{t}\rangle+\frac{L}{2}\norm{w_{t+1}-w_{t}}^{2}.
\tag{5}
\]
\end{lemma}
\begin{lemma}[Error recursion]\label{lem:error-bound}
Define $e_{t}$ as in \eqref{2}. Under (A3),
\[
\E\!\left[\norm{e_{t}}^{2}\right]\le (1-\delta)\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\right].
\tag{6}
\]
\end{lemma}
\begin{proof}
Directly from the definition $e_{t}= (g_{t}+e_{t-1})-\cK(g_{t}+e_{t-1})$ and the contraction property (A3). \hfill $\square$
\end{proof}
\begin{lemma}[Bound on the compressed update]\label{lem:comp-update}
For $c_{t}= \cK(g_{t}+e_{t-1})$,
\[
\E\!\left[\norm{c_{t}}^{2}\right]\le \E\!\left[\norm{g_{t}+e_{t-1}}^{2}\right].
\tag{7}
\]
\end{lemma}
\begin{proof}
Since $\cK$ only zeroes out coordinates,
$\norm{c_{t}}^{2}\le \norm{g_{t}+e_{t-1}}^{2}$ pointwise, and taking expectation preserves the inequality. \hfill $\square$
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Let $w^{\star}$ be a global minimizer (possibly non‑unique) and denote $f^{\star}=f(w^{\star})$.

\paragraph{Step 1 (Smoothness expansion).}
From Lemma~\ref{lem:smooth} with $w_{t+1}=w_{t}-\eta c_{t}$,
\[
f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),c_{t}\rangle
+\frac{L\eta^{2}}{2}\norm{c_{t}}^{2}. \tag{8}
\]
\paragraph{Step 2 (Insert $c_{t}=g_{t}+e_{t-1}-e_{t}$).}
Using \eqref{1}–\eqref{2},
\[
c_{t}=g_{t}+e_{t-1}-e_{t}. \tag{9}
\]
Substituting \eqref{9} into \eqref{8} yields
\[
f(w_{t+1})\le f(w_{t})-\eta\langle\nabla f(w_{t}),g_{t}+e_{t-1}\rangle
+\eta\langle\nabla f(w_{t}),e_{t}\rangle
+\frac{L\eta^{2}}{2}\norm{g_{t}+e_{t-1}-e_{t}}^{2}. \tag{10}
\]

\paragraph{Step 3 (Take full expectation).}
Condition on the past $\mathcal{F}_{t}:=\sigma(w_{0},e_{0},\dots ,w_{t},e_{t-1})$.
Using unbiasedness (A2) and the fact that $e_{t}$ is a deterministic function of $g_{t}$ and $e_{t-1}$,
\[
\E\!\left[\langle\nabla f(w_{t}),g_{t}+e_{t-1}\rangle\mid\mathcal{F}_{t}\right]
= \norm{\nabla f(w_{t})}^{2}+\langle\nabla f(w_{t}),e_{t-1}\rangle. \tag{11}
\]
Moreover,
\[
\E\!\left[\langle\nabla f(w_{t}),e_{t}\rangle\mid\mathcal{F}_{t}\right]=
\langle\nabla f(w_{t}),\E[e_{t}\mid\mathcal{F}_{t}]\rangle. \tag{12}
\]
From \eqref{2} and the linearity of expectation,
\[
\E[e_{t}\mid\mathcal{F}_{t}]
= \E\!\left[(g_{t}+e_{t-1})-\cK(g_{t}+e_{t-1})\mid\mathcal{F}_{t}\right]
= (1-\delta)\bigl(\nabla f(w_{t})+e_{t-1}\bigr), \tag{13}
\]
where we used the contraction property (A3) in expectation and the fact that the top‑$K$ operator is unbiased up to the factor $\delta$ (standard for deterministic sparsifiers).  

Thus,
\[
\E\!\left[\langle\nabla f(w_{t}),e_{t}\rangle\mid\mathcal{F}_{t}\right]
= (1-\delta)\norm{\nabla f(w_{t})}^{2}+(1-\delta)\langle\nabla f(w_{t}),e_{t-1}\rangle. \tag{14}
\]

\paragraph{Step 4 (Expand the squared norm term).}
From \eqref{9},
\[
\norm{c_{t}}^{2}= \norm{g_{t}+e_{t-1}}^{2}+\norm{e_{t}}^{2}
-2\langle g_{t}+e_{t-1},e_{t}\rangle. \tag{15}
\]
Taking expectation conditioned on $\mathcal{F}_{t}$ and using Lemma~\ref{lem:error-bound} gives
\[
\E\!\left[\norm{c_{t}}^{2}\mid\mathcal{F}_{t}\right]
\le \E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]
+ \E\!\left[\norm{e_{t}}^{2}\mid\mathcal{F}_{t}\right]. \tag{16}
\]
Applying (A3) to the second term,
\[
\E\!\left[\norm{e_{t}}^{2}\mid\mathcal{F}_{t}\right]
\le (1-\delta)\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]. \tag{17}
\]
Hence,
\[
\E\!\left[\norm{c_{t}}^{2}\mid\mathcal{F}_{t}\right]
\le \bigl(1+(1-\delta)\bigr)\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]
= \frac{2-\delta}{\;}\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]. \tag{18}
\]

\paragraph{Step 5 (Bound $\norm{g_{t}+e_{t-1}}^{2}$).}
Using $(a+b)^{2}\le 2a^{2}+2b^{2}$,
\[
\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]
\le 2\E\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]+2\norm{e_{t-1}}^{2}. \tag{19}
\]
With (A2),
\[
\E\!\left[\norm{g_{t}}^{2}\mid\mathcal{F}_{t}\right]
= \norm{\nabla f(w_{t})}^{2}+\E\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid\mathcal{F}_{t}\right]
\le \norm{\nabla f(w_{t})}^{2}+\sigma^{2}. \tag{20}
\]
Therefore,
\[
\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]
\le 2\norm{\nabla f(w_{t})}^{2}+2\sigma^{2}+2\norm{e_{t-1}}^{2}. \tag{21}
\]

\paragraph{Step 6 (Combine everything).}
Insert (11)–(14) and (18)–(21) into the expectation of \eqref{10}:

\begin{align}
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
&\le f(w_{t})-\eta\Bigl(\norm{\nabla f(w_{t})}^{2}
+\langle\nabla f(w_{t}),e_{t-1}\rangle\Bigr)   \nonumber\\
&\quad +\eta\Bigl((1-\delta)\norm{\nabla f(w_{t})}^{2}
+(1-\delta)\langle\nabla f(w_{t}),e_{t-1}\rangle\Bigr) \nonumber\\
&\quad +\frac{L\eta^{2}}{2}\cdot\frac{2-\delta}{\;}\Bigl(2\norm{\nabla f(w_{t})}^{2}+2\sigma^{2}+2\norm{e_{t-1}}^{2}\Bigr). \tag{22}
\end{align}

Collect the coefficients of $\norm{\nabla f(w_{t})}^{2}$ and $\langle\nabla f(w_{t}),e_{t-1}\rangle$:

\[
\begin{aligned}
\text{Coeff. of }\norm{\nabla f(w_{t})}^{2}&:
-\eta +\eta(1-\delta)+L\eta^{2}(2-\delta) \\
&= -\eta\delta +L\eta^{2}(2-\delta). \tag{23}
\end{aligned}
\]

\[
\begin{aligned}
\text{Coeff. of }\langle\nabla f(w_{t}),e_{t-1}\rangle&:
-\eta +\eta(1-\delta)= -\eta\delta. \tag{24}
\end{aligned}
\]

Thus,
\[
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
\le f(w_{t}) -\eta\delta\norm{\nabla f(w_{t})}^{2}
-\eta\delta\langle\nabla f(w_{t}),e_{t-1}\rangle
+L\eta^{2}(2-\delta)\bigl(\norm{\nabla f(w_{t})}^{2}+\sigma^{2}+\norm{e_{t-1}}^{2}\bigr). \tag{25}
\]

\paragraph{Step 7 (Handle the cross term).}
Apply Cauchy–Schwarz and Young’s inequality $ab\le \frac{\delta}{4}a^{2}+\frac{1}{\delta}b^{2}$ with $a=\norm{\nabla f(w_{t})}$,
$b=\norm{e_{t-1}}$:

\[
-\eta\delta\langle\nabla f(w_{t}),e_{t-1}\rangle
\le \eta\delta\norm{\nabla f(w_{t})}\norm{e_{t-1}}
\le \frac{\eta\delta^{2}}{4}\norm{\nabla f(w_{t})}^{2}
+\frac{\eta}{\delta}\norm{e_{t-1}}^{2}. \tag{26}
\]

Insert (26) into (25):

\[
\begin{aligned}
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
&\le f(w_{t})
-\Bigl(\eta\delta-\frac{\eta\delta^{2}}{4}-L\eta^{2}(2-\delta)\Bigr)\norm{\nabla f(w_{t})}^{2}\\
&\quad +\Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)\norm{e_{t-1}}^{2}
+L\eta^{2}(2-\delta)\sigma^{2}. \tag{27}
\end{aligned}
\]

\paragraph{Step 8 (Choose stepsize).}
Assumption (A4) $\eta\le\frac{1}{L}$ guarantees that the coefficient of $\norm{\nabla f(w_{t})}^{2}$ is non‑negative. In particular, for $\eta\le\frac{\delta}{4L(2-\delta)}$ we have

\[
\eta\delta-\frac{\eta\delta^{2}}{4}-L\eta^{2}(2-\delta)\ge\frac{\eta\delta}{2}. \tag{28}
\]

Hence,
\[
\E\!\left[f(w_{t+1})\mid\mathcal{F}_{t}\right]
\le f(w_{t})-\frac{\eta\delta}{2}\norm{\nabla f(w_{t})}^{2}
+ \Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)\norm{e_{t-1}}^{2}
+L\eta^{2}(2-\delta)\sigma^{2}. \tag{29}
\]

\paragraph{Step 9 (Recursion for the error norm).}
From Lemma~\ref{lem:error-bound} and using (A3),

\[
\E\!\left[\norm{e_{t}}^{2}\mid\mathcal{F}_{t}\right]
\le (1-\delta)\E\!\left[\norm{g_{t}+e_{t-1}}^{2}\mid\mathcal{F}_{t}\right]
\stackrel{(19)}{\le}
(1-\delta)\bigl(2\norm{\nabla f(w_{t})}^{2}+2\sigma^{2}+2\norm{e_{t-1}}^{2}\bigr). \tag{30}
\]

Taking full expectation and rearranging,

\[
\E\!\left[\norm{e_{t}}^{2}\right]\le (1-\delta)\E\!\left[\norm{e_{t-1}}^{2}\right]
+2(1-\delta)\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+2(1-\delta)\sigma^{2}. \tag{31}
\]

Summing \eqref{31} over $t=0,\dots ,T-1$ and using $e_{0}=0$ yields

\[
\sum_{t=0}^{T-1}\E\!\left[\norm{e_{t}}^{2}\right]
\le \frac{2(1-\delta)}{\delta}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+\frac{2(1-\delta)}{\delta}T\sigma^{2}. \tag{32}
\]

\paragraph{Step 10 (Telescoping the function values).}
Take total expectation of \eqref{29} and sum for $t=0,\dots ,T-1$:

\[
\E[f(w_{T})]\le f(w_{0})
-\frac{\eta\delta}{2}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+ \Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)\sum_{t=0}^{T-1}\E\!\left[\norm{e_{t-1}}^{2}\right]
+LT\,L\eta^{2}(2-\delta)\sigma^{2}. \tag{33}
\]

Discard the non‑negative term $\E[f(w_{T})]\ge f^{\star}$ and use the bound \eqref{32} for the error sum:

\[
\begin{aligned}
\frac{\eta\delta}{2}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
&\le f(w_{0})-f^{\star}\\
&\quad +\Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)
\Bigl[\frac{2(1-\delta)}{\delta}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
+\frac{2(1-\delta)}{\delta}T\sigma^{2}\Bigr] \\
&\quad +L\eta^{2}(2-\delta)T\sigma^{2}. \tag{34}
\end{aligned}
\]

Collect the terms containing $\sum_{t}\E[\norm{\nabla f(w_{t})}^{2}]$ on the left‑hand side.
Define $A:=\frac{\eta\delta}{2}$ and 
$B:=\Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)\frac{2(1-\delta)}{\delta}$.
Then \eqref{34} reads $A S \le (f(w_{0})-f^{\star}) + B S + C T\sigma^{2}$,
where $S:=\sum_{t=0}^{T-1}\E[\norm{\nabla f(w_{t})}^{2}]$ and
$C:=\Bigl(\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\Bigr)\frac{2(1-\delta)}{\delta}+L\eta^{2}(2-\delta)$.

Rearranging,
\[
(A-B)S \le (f(w_{0})-f^{\star}) + C T\sigma^{2}. \tag{35}
\]

\paragraph{Step 11 (Simplify constants).}
Using $\eta\le\frac{\delta}{4L(2-\delta)}$ we have $L\eta^{2}(2-\delta)\le\frac{\eta\delta}{4}$, hence
\[
\frac{\eta}{\delta}+L\eta^{2}(2-\delta)\le\frac{2\eta}{\delta}. \tag{36}
\]
Consequently,
\[
B\le \frac{2\eta}{\delta}\cdot\frac{2(1-\delta)}{\delta}
= \frac{4\eta(1-\delta)}{\delta^{2}}. \tag{37}
\]
Since $A=\frac{\eta\delta}{2}$,
\[
A-B\ge \frac{\eta\delta}{2}-\frac{4\eta(1-\delta)}{\delta^{2}}
= \eta\Bigl(\frac{\delta}{2}-\frac{4(1-\delta)}{\delta^{2}}\Bigr) \ge \frac{\eta\delta}{4},
\]
for any $\delta\in(0,1]$ (the inequality follows from the stepsize restriction).  
Thus $A-B\ge \frac{\eta\delta}{4}$.

Similarly, from (36) we obtain
\[
C\le \frac{2\eta}{\delta}\cdot\frac{2(1-\delta)}{\delta}+L\eta^{2}(2-\delta)
\le \frac{4\eta(1-\delta)}{\delta^{2}}+ \frac{\eta\delta}{4}
\le \frac{5\eta}{\delta}. \tag{38}
\]

Plugging these bounds into (35) gives
\[
\frac{\eta\delta}{4}S \le (f(w_{0})-f^{\star}) + \frac{5\eta}{\delta}T\sigma^{2}.
\]
Dividing by $\eta T$ and recalling $S=\sum_{t=0}^{T-1}\E[\norm{\nabla f(w_{t})}^{2}]$ yields the desired average‑gradient bound:

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\norm{\nabla f(w_{t})}^{2}]
\le \frac{4\bigl(f(w_{0})-f^{\star}\bigr)}{\eta\delta T}
+ \frac{20}{\delta^{2}}\sigma^{2}. \tag{39}
\]

Choosing $\eta = \Theta\!\bigl(T^{-1/2}\bigr)$ (e.g. $\eta = \frac{\delta}{2\sqrt{LT}}$) makes the first term $\mathcal{O}(1/\sqrt{T})$ while the second term remains $\mathcal{O}(\eta\sigma^{2}) = \mathcal{O}(T^{-1/2})$ as well. This establishes the rate claimed in \eqref{4} and completes the proof. \hfill $\square$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From inequality \eqref{39} and the concrete choice $\eta = \frac{\delta}{2\sqrt{L T}}$ (which respects $\eta\le 1/L$ for all $T\ge 1$) we obtain

\[
\frac{1}{T}\sum_{t=0}^{T-1}\E[\norm{\nabla f(w_{t})}^{2}]
\le
\underbrace{\frac{8\sqrt{L}\bigl(f(w_{0})-f^{\star}\bigr)}{\delta^{2}}}_{\displaystyle\mathcal{O}\!\bigl(T^{-1/2}\bigr)}
\;+\;
\underbrace{\frac{10\delta\sigma^{2}}{\sqrt{L}}}_{\displaystyle\mathcal{O}\!\bigl(T^{-1/2}\bigr)}.
\]

Thus the algorithm attains the optimal $\mathcal{O}(T^{-1/2})$ convergence rate for non‑convex smooth optimization, with constants explicitly depending on the compression factor $\delta=K/d$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}[label=(\roman*)]
\item \textbf{Full communication} ($K=d$, $\delta=1$). The compression term disappears, and \eqref{4} reduces to the classical SGD bound $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Extreme sparsity} ($K=1$, $\delta=1/d$). The error‑feedback term scales as $\mathcal{O}\!\bigl(d^{2}\eta^{2}\sigma^{2}\bigr)$, illustrating the trade‑off between communication reduction and statistical efficiency.
\item \textbf{Deterministic top‑$K$ without error feedback}. Setting $e_{t}=0$ for all $t$ removes the error‑feedback recursion; the bound then contains an additional bias term $\frac{L\eta^{2}}{2}\norm{(I-\cK)\nabla f(w_{t})}^{2}$ which prevents the $\mathcal{O}(1/\sqrt{T})$ rate in the non‑convex case.
\end{enumerate}
