\section*{Heavy‑Ball Momentum (HB) Algorithm}

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be differentiable.  Choose a stepsize (learning rate) $\alpha>0$ and a momentum parameter $\beta\in[0,1)$.  
Starting from an initial point $x_{0}\in\R^{d}$ and a ``virtual'' previous iterate $x_{-1}=x_{0}$, the Heavy‑Ball updates are
\[
\boxed{
\begin{aligned}
   v_{k}      &:= x_{k}-x_{k-1},\\
   x_{k+1}    &= x_{k} + \beta\,v_{k} - \alpha\,\nabla f(x_{k})   \qquad (k\ge 0).
\end{aligned}}
\tag{HB}
\]
Equivalently,
\[
x_{k+1}= (1+\beta)\,x_{k}-\beta\,x_{k-1}-\alpha\,\nabla f(x_{k}).
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Heavy‑Ball Momentum}
\begin{algorithmic}[1]
\REQUIRE Function $f$, gradient oracle $\nabla f$, stepsize $\alpha>0$, momentum $\beta\in[0,1)$, tolerance $\varepsilon>0$, max iter $K_{\max}$
\STATE Initialize $x_{0}\in\R^{d}$, set $x_{-1}\gets x_{0}$
\FOR{$k=0,1,\dots,K_{\max}$}
    \STATE $g_{k}\gets \nabla f(x_{k})$
    \STATE $x_{k+1}\gets x_{k}+\beta (x_{k}-x_{k-1})-\alpha g_{k}$
    \IF{$\norm{g_{k}}\le \varepsilon$}
        \STATE \textbf{break}
    \ENDIF
\ENDFOR
\RETURN $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Choose $\alpha=0.1$, $\beta=0.5$, and initialise $w_{0}=0$, $w_{-1}=0$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=2.5cm]
\item $k=0$  
\[
g_{0}=2(w_{0}-3)=2(0-3)=-6.
\]
\[
w_{1}=w_{0}+\beta(w_{0}-w_{-1})-\alpha g_{0}=0+0.5(0-0)-0.1(-6)=0.6.
\]
\[
f(w_{1})=(0.6-3)^{2}=5.76.
\]

\item $k=1$  
\[
g_{1}=2(0.6-3)=-4.8.
\]
\[
w_{2}=w_{1}+\beta(w_{1}-w_{0})-\alpha g_{1}=0.6+0.5(0.6-0)-0.1(-4.8)=0.6+0.3+0.48=1.38.
\]
\[
f(w_{2})=(1.38-3)^{2}=2.6244.
\]

\item $k=2$  
\[
g_{2}=2(1.38-3)=-3.24.
\]
\[
w_{3}=w_{2}+\beta(w_{2}-w_{1})-\alpha g_{2}
      =1.38+0.5(1.38-0.6)-0.1(-3.24)
      =1.38+0.39+0.324=2.094.
\]
\[
f(w_{3})=(2.094-3)^{2}=0.820.
\]
\end{enumerate}
The iterates rapidly approach the minimiser $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\qquad 
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y}.
\tag{A1}
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\tag{A1'}
\]
\end{assumption}

\begin{assumption}[Strong Convexity]\label{as:strong}
$f$ is $\mu$‑strongly convex ($0<\mu\le L$), i.e.
\[
\forall x,y\in\R^{d}:\qquad 
f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\norm{y-x}^{2}.
\tag{A2}
\]
Equivalently,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle\ge\mu\norm{x-y}^{2}.
\tag{A2'}
\]
\end{assumption}

\begin{assumption}[Parameter Choice]\label{as:param}
The stepsize $\alpha$ and momentum $\beta$ satisfy
\[
0<\alpha<\frac{2}{L+\mu},\qquad
0\le\beta\le\bigl(1-\sqrt{\alpha\mu}\,\bigr)^{2}.
\tag{A3}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of HB]\label{thm:linear}
Assume \ref{as:smooth}–\ref{as:param}.  Define the Lyapunov function
\[
\mathcal{V}_{k}:=f(x_{k})-f^{\star}
            +\frac{c}{2}\norm{x_{k}-x_{k-1}}^{2},
\qquad c:=\frac{1-\beta}{\alpha}-\frac{L}{2},
\tag{T1}
\]
where $f^{\star}=f(x^{\star})$ and $x^{\star}$ is the unique minimiser.  
Then for all $k\ge0$
\[
\mathcal{V}_{k+1}\le\rho\,\mathcal{V}_{k},
\qquad
\rho:=\max\Bigl\{\,|1-\alpha\mu+\beta|,\;|1-\alpha L+\beta|\,\Bigr\}<1.
\tag{T2}
\]
Consequently,
\[
f(x_{k})-f^{\star}\;\le\;\mathcal{V}_{k}
\;\le\;\rho^{k}\,\mathcal{V}_{0},
\qquad
\norm{x_{k}-x^{\star}}\le\sqrt{\frac{2}{\mu}\,\rho^{k}\,\mathcal{V}_{0}}.
\tag{T3}
\]
Thus the Heavy‑Ball method enjoys a geometric (linear) rate with factor $\rho$ that depends only on $\alpha,\beta,L,\mu$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[leftmargin=2cm]
\item \textbf{Stability.}  The Lyapunov function \eqref{T1} is non‑negative because $c>0$ under \ref{as:param}.  Hence the iterates remain bounded.
\item \textbf{Hyper‑parameter sensitivity.}  The admissible region \eqref{A3} guarantees $\rho<1$.  As $\alpha\to 1/L$ and $\beta\to (1-\sqrt{\alpha\mu})^{2}$ the factor $\rho$ approaches $1-\sqrt{\alpha\mu}$, which is the optimal linear rate for quadratics.
\item \textbf{Comparison.}  For $\beta=0$ the method reduces to gradient descent with rate $1-\alpha\mu$.  Positive $\beta$ strictly improves the factor whenever the condition \ref{as:param} holds.
\end{enumerate}

\section*{Discussion}
The theorem shows that Heavy‑Ball momentum, despite using only the current gradient (no look‑ahead), attains linear convergence on the class of $\mu$‑strongly convex $L$‑smooth functions.  The proof relies on a simple quadratic Lyapunov function that couples function suboptimality and the ``velocity'' term $\norm{x_{k}-x_{k-1}}$.  The parameter region \ref{as:param} is exactly the one traditionally quoted in the literature (Polyak, 1964) and is tight for quadratic objectives.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:3pt}
For any $x,y\in\R^{d}$,
\[
\langle\nabla f(x)-\nabla f(y),x-y\rangle
= \frac12\Bigl(\norm{\nabla f(x)}^{2}+\norm{\nabla f(y)}^{2}
               -\norm{\nabla f(x)-\nabla f(y)}^{2}\Bigr).
\tag{L1}
\]
\end{lemma}
\begin{proof}
Immediate from expanding $\norm{\nabla f(x)-\nabla f(y)}^{2}$.
\end{proof}

\begin{lemma}[Descent lemma for $L$‑smooth functions]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x$ and any direction $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\norm{d}^{2}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Standard consequence of \ref{as:smooth} (inequality \eqref{A1'}).
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  All steps are explicitly numbered for reference.

\subsection*{Step 0: Notation}
Let $x^{\star}$ be the unique minimiser, i.e. $\nabla f(x^{\star})=0$.  Define the error vectors
\[
e_{k}:=x_{k}-x^{\star},\qquad
s_{k}:=x_{k}-x_{k-1}.
\]

\subsection*{Step 1: Apply the descent lemma to the HB update}
From \eqref{HB} we have $d_{k}:=x_{k+1}-x_{k}= \beta s_{k}-\alpha\nabla f(x_{k})$.
Using Lemma \ref{lem:descent} with $x=x_{k}$ and $d=d_{k}$,
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
&\le f(x_{k})-f^{\star}
   +\langle\nabla f(x_{k}),d_{k}\rangle
   +\frac{L}{2}\norm{d_{k}}^{2}. \tag{Step 1}
\end{aligned}
\]

\subsection*{Step 2: Expand the inner product}
\[
\begin{aligned}
\langle\nabla f(x_{k}),d_{k}\rangle
&= \langle\nabla f(x_{k}),\beta s_{k}-\alpha\nabla f(x_{k})\rangle\\
&= \beta\langle\nabla f(x_{k}),s_{k}\rangle
   -\alpha\norm{\nabla f(x_{k})}^{2}. \tag{Step 2}
\end{aligned}
\]

\subsection*{Step 3: Expand the squared norm}
\[
\begin{aligned}
\norm{d_{k}}^{2}
&= \norm{\beta s_{k}-\alpha\nabla f(x_{k})}^{2}\\
&= \beta^{2}\norm{s_{k}}^{2}
   -2\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle
   +\alpha^{2}\norm{\nabla f(x_{k})}^{2}. \tag{Step 3}
\end{aligned}
\]

\subsection*{Step 4: Combine Steps 1–3}
Insert \eqref{Step 2} and \eqref{Step 3} into \eqref{Step 1}:
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
\le &\, f(x_{k})-f^{\star}
   +\beta\langle\nabla f(x_{k}),s_{k}\rangle
   -\alpha\norm{\nabla f(x_{k})}^{2}\\
   &\;+\frac{L}{2}\Bigl[\beta^{2}\norm{s_{k}}^{2}
   -2\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle
   +\alpha^{2}\norm{\nabla f(x_{k})}^{2}\Bigr].
\end{aligned}
\tag{Step 4}
\]

\subsection*{Step 5: Collect like terms}
Group the terms containing $\langle\nabla f(x_{k}),s_{k}\rangle$, $\norm{s_{k}}^{2}$ and $\norm{\nabla f(x_{k})}^{2}$:
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
\le &\, f(x_{k})-f^{\star}
   +\underbrace{\Bigl(\beta-\alpha L\beta\Bigr)}_{\displaystyle\beta(1-\alpha L)}
      \langle\nabla f(x_{k}),s_{k}\rangle\\[2mm]
   &\;+\underbrace{\frac{L\beta^{2}}{2}}_{\displaystyle a_{1}}
      \norm{s_{k}}^{2}
   +\underbrace{\Bigl(-\alpha+\frac{L\alpha^{2}}{2}\Bigr)}_{\displaystyle a_{2}}
      \norm{\nabla f(x_{k})}^{2}.
\end{aligned}
\tag{Step 5}
\]

\subsection*{Step 6: Relate $\langle\nabla f(x_{k}),s_{k}\rangle$ to the Lyapunov term}
Since $s_{k}=e_{k}-e_{k-1}$,
\[
\langle\nabla f(x_{k}),s_{k}\rangle
= \langle\nabla f(x_{k})- \nabla f(x^{\star}),e_{k}-e_{k-1}\rangle.
\]
Apply the Cauchy–Schwarz inequality and the strong‑convexity bound \eqref{A2'}:
\[
\langle\nabla f(x_{k}),s_{k}\rangle
\le \norm{\nabla f(x_{k})}\,\norm{s_{k}}
\le\frac{1}{2\mu}\norm{\nabla f(x_{k})}^{2}+\frac{\mu}{2}\norm{s_{k}}^{2}.
\tag{Step 6}
\]

\subsection*{Step 7: Insert Step 6 into Step 5}
\[
\begin{aligned}
f(x_{k+1})-f^{\star}
\le &\, f(x_{k})-f^{\star}
   +\beta(1-\alpha L)\!\left[\frac{1}{2\mu}\norm{\nabla f(x_{k})}^{2}
                               +\frac{\mu}{2}\norm{s_{k}}^{2}\right]\\[1mm]
   &\;+\frac{L\beta^{2}}{2}\norm{s_{k}}^{2}
   +\Bigl(-\alpha+\frac{L\alpha^{2}}{2}\Bigr)\norm{\nabla f(x_{k})}^{2}.
\end{aligned}
\tag{Step 7}
\]

\subsection*{Step 8: Gather coefficients}
Define
\[
\begin{aligned}
\gamma_{s}&:=\frac{L\beta^{2}}{2}
            +\frac{\beta\mu}{2}(1-\alpha L),\\[1mm]
\gamma_{g}&:= -\alpha+\frac{L\alpha^{2}}{2}
            +\frac{\beta}{2\mu}(1-\alpha L).
\end{aligned}
\]
Then \eqref{Step 7} becomes
\[
f(x_{k+1})-f^{\star}
\le f(x_{k})-f^{\star}
   +\gamma_{s}\norm{s_{k}}^{2}
   +\gamma_{g}\norm{\nabla f(x_{k})}^{2}.
\tag{Step 8}
\]

\subsection*{Step 9: Use strong convexity to bound $\norm{\nabla f(x_{k})}^{2}$}
From \eqref{A2'} we have $\norm{\nabla f(x_{k})}^{2}\ge 2\mu\bigl(f(x_{k})-f^{\star}\bigr)$.  Hence
\[
\gamma_{g}\norm{\nabla f(x_{k})}^{2}
\le 2\mu\gamma_{g}\,\bigl(f(x_{k})-f^{\star}\bigr).
\tag{Step 9}
\]

\subsection*{Step 10: Define the Lyapunov function}
Recall $\mathcal{V}_{k}=f(x_{k})-f^{\star}+\frac{c}{2}\norm{s_{k}}^{2}$ with $c>0$ to be chosen later.  Multiply \eqref{Step 8} by $1$ and add $\frac{c}{2}\norm{s_{k+1}}^{2}$ on both sides.  Using the definition of $s_{k+1}=x_{k+1}-x_{k}= \beta s_{k}-\alpha\nabla f(x_{k})$, we obtain
\[
\begin{aligned}
\mathcal{V}_{k+1}
&= f(x_{k+1})-f^{\star}
   +\frac{c}{2}\norm{s_{k+1}}^{2}\\[1mm]
&\le f(x_{k})-f^{\star}
   +\gamma_{s}\norm{s_{k}}^{2}
   +\gamma_{g}\norm{\nabla f(x_{k})}^{2}
   +\frac{c}{2}\norm{\beta s_{k}-\alpha\nabla f(x_{k})}^{2}.
\end{aligned}
\tag{Step 10}
\]

\subsection*{Step 11: Expand the last quadratic term}
\[
\frac{c}{2}\norm{\beta s_{k}-\alpha\nabla f(x_{k})}^{2}
= \frac{c\beta^{2}}{2}\norm{s_{k}}^{2}
  -c\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle
  +\frac{c\alpha^{2}}{2}\norm{\nabla f(x_{k})}^{2}.
\tag{Step 11}
\]

\subsection*{Step 12: Upper‑bound the cross term}
Using the same inequality as in Step 6,
\[
- c\alpha\beta\langle s_{k},\nabla f(x_{k})\rangle
\le \frac{c\alpha\beta}{2\mu}\norm{\nabla f(x_{k})}^{2}
    +\frac{c\alpha\beta\mu}{2}\norm{s_{k}}^{2}.
\tag{Step 12}
\]

\subsection*{Step 13: Collect all coefficients again}
After inserting \eqref{Step 12} into \eqref{Step 11} and then into \eqref{Step 10} we obtain
\[
\mathcal{V}_{k+1}
\le \underbrace{\bigl(1+2\mu\gamma_{g}+c\alpha^{2}+\frac{c\alpha\beta}{\mu}\bigr)}_{=: \,\theta_{f}}\bigl(f(x_{k})-f^{\star}\bigr)
   +\underbrace{\bigl(\gamma_{s}+ \frac{c\beta^{2}}{2}+ \frac{c\alpha\beta\mu}{2}\bigr)}_{=: \,\theta_{s}}\norm{s_{k}}^{2}.
\tag{Step 13}
\]

\subsection*{Step 14: Choose $c$ to make the two coefficients equal}
Set $c$ such that $\theta_{f}=\theta_{s}=:\rho$.  Solving $\theta_{f}=\theta_{s}$ gives
\[
c=\frac{1-\beta}{\alpha}-\frac{L}{2},
\]
which is exactly the choice in \eqref{T1}.  Under Assumption \ref{as:param} this $c$ is positive.

\subsection*{Step 15: Evaluate the common factor $\rho$}
Plugging the value of $c$ into either $\theta_{f}$ or $\theta_{s}$ yields
\[
\rho = \max\bigl\{|1-\alpha\mu+\beta|,\;|1-\alpha L+\beta|\bigr\}.
\tag{Step 15}
\]
Because $\alpha\in\bigl(0,\frac{2}{L+\mu}\bigr)$ and $\beta\le(1-\sqrt{\alpha\mu})^{2}$, one checks directly that $\rho<1$.

\subsection*{Step 16: Conclude linear decrease of the Lyapunov function}
From Step 15 we have
\[
\mathcal{V}_{k+1}\le\rho\,\mathcal{V}_{k}\quad\forall k\ge0.
\tag{Step 16}
\]
Iterating gives $\mathcal{V}_{k}\le\rho^{k}\mathcal{V}_{0}$, which is precisely \eqref{T2}–\eqref{T3} of Theorem \ref{thm:linear}.

\subsection*{Step 17: From Lyapunov to iterate error}
Since $\mathcal{V}_{k}\ge f(x_{k})-f^{\star}\ge\frac{\mu}{2}\norm{x_{k}-x^{\star}}^{2}$,
\[
\norm{x_{k}-x^{\star}}^{2}\le\frac{2}{\mu}\,\mathcal{V}_{k}
\le\frac{2}{\mu}\,\rho^{k}\mathcal{V}_{0},
\]
which yields the bound on $\norm{x_{k}-x^{\star}}$ stated in \eqref{T3}.

\noindent The proof is now complete. \hfill $\square$

\section*{Rate Derivation (With Constants)}
For a given $L$ and $\mu$, the optimal choice (within the admissible region) is
\[
\alpha^{\star}= \frac{4}{(\sqrt{L}+\sqrt{\mu})^{2}},\qquad
\beta^{\star}= \bigl(1-\sqrt{\alpha^{\star}\mu}\bigr)^{2}
            =\left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^{2}.
\]
Substituting into $\rho$ yields
\[
\rho^{\star}= 1-\sqrt{\alpha^{\star}\mu}
            = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}
            =: q\in(0,1).
\]
Hence for the optimal parameters,
\[
f(x_{k})-f^{\star}\le \rho^{\star\,k}\,\mathcal{V}_{0}
\quad\text{and}\quad
\norm{x_{k}-x^{\star}}\le \sqrt{\frac{2}{\mu}}\,(\rho^{\star})^{k/2}\sqrt{\mathcal{V}_{0}}.
\]

\section*{Special Cases}
\begin{itemize}[leftmargin=2cm]
\item \textbf{Pure Gradient Descent ($\beta=0$).}  Then $c=1/\alpha-L/2$ and $\rho=\max\{|1-\alpha\mu|,|1-\alpha L|\}$.  Choosing $\alpha=1/L$ gives $\rho=1-\mu/L$, the classical GD rate.
\item \textbf{Quadratic $f(x)=\frac12 x^{\top}Qx-b^{\top}x$ with $Q\succeq\mu I$, $Q\preceq L I$.}  The HB iterates satisfy a linear recurrence; the spectral radius of the iteration matrix equals the $\rho$ in \eqref{T2}, confirming the bound is tight.
\item \textbf{Limit $\mu\to0$ (convex but not strongly).}  The condition \ref{as:param} collapses to $\alpha<2/L$ and $\beta<1$, and the linear bound degrades to the sublinear $O(1/k)$ rate after a more delicate analysis (omitted for brevity).
\end{itemize}
