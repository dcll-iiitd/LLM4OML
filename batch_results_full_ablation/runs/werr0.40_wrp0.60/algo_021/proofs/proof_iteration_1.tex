\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex and $L$‑smooth function (i.e.~$\nabla f$ is $L$‑Lipschitz) and let $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ be a proper, closed, convex (possibly nondifferentiable) function.  
Define the composite objective
\[
F(x)=f(x)+g(x),\qquad x\in\R^{d}.
\]
Given a stepsize $\eta>0$, the \textbf{proximal gradient update} reads
\begin{equation}
\boxed{\;x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr)\;},\qquad k=0,1,\dots
\label{eq:update}
\end{equation}
where the proximal operator of $g$ is
\[
\prox_{\eta g}(u)=\arg\min_{z\in\R^{d}}\Bigl\{g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Bigr\}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$ 
\For{$k=0,1,2,\dots$}
    \State Compute gradient $g_{k}\gets\nabla f(x_{k})$
    \State Gradient step $y_{k}\gets x_{k}-\eta\,g_{k}$
    \State Proximal step $x_{k+1}\gets\prox_{\eta g}(y_{k})$
    \If{termination criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{return} $x_{k+1}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^{2},\qquad g\equiv0,\qquad \eta=0.1,\qquad w_{0}=0.
\]
Since $g=0$, $\prox_{\eta g}$ is the identity.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $F(w_{k})$ \\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $(0-3)^{2}=9$ \\
1 & $w_{1}=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^{2}=5.76$ \\
2 & $w_{2}=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^{2}=3.6864$ \\
3 & $w_{3}=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^{2}=2.3685$
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the descent property of \eqref{eq:update}.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness of $f$]
\label{ass:f}
$f$ is convex and differentiable on $\R^{d}$, and its gradient is $L$‑Lipschitz:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\]
\end{assumption}
\begin{assumption}[Convexity of $g$]
\label{ass:g}
$g$ is proper, closed and convex (possibly nonsmooth). Its proximal operator $\prox_{\eta g}$ is well defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]
\label{ass:step}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Existence of a minimizer]
\label{ass:opt}
The set of minimizers
\[
\mathcal{X}^{*}:=\arg\min_{x\in\R^{d}}F(x)
\]
is non‑empty, and we denote an arbitrary optimal point by $x^{*}\in\mathcal{X}^{*}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Sublinear $O(1/k)$ rate for PGM]\label{thm:rate}
Under Assumptions \ref{ass:f}–\ref{ass:opt} and with a stepsize $\eta\in(0,1/L]$, the iterates generated by \eqref{eq:update} satisfy for all $k\ge1$
\[
F(x_{k})-F(x^{*})\;\le\;\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k}.
\]
Consequently $F(x_{k})\to F(x^{*})$ and $x_{k}$ converges to a minimizer.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability:} The update \eqref{eq:update} is a non‑expansive mapping when $\eta\le1/L$, guaranteeing that the iterates remain bounded.
\item \textbf{Hyperparameter Sensitivity:} The rate constant is inversely proportional to $\eta$. Choosing $\eta=1/L$ yields the best worst‑case bound.
\item \textbf{Comparison to Gradient Descent:} When $g\equiv0$, $\prox_{\eta g}$ reduces to the identity and Theorem \ref{thm:rate} recovers the classic $O(1/k)$ rate for gradient descent on smooth convex functions.
\item \textbf{Special case $g$ indicator of a convex set $C$:} The algorithm becomes the projected gradient method; the same $O(1/k)$ guarantee holds.
\end{itemize}

\section*{Discussion}
The proof hinges on two elementary but powerful facts:
\begin{enumerate}
\item \emph{Descent lemma} for $L$‑smooth $f$.
\item \emph{Non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{NE}
\]
By combining these with convexity of $g$, we obtain a one‑step inequality that telescopes over iterations, yielding the $O(1/k)$ bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ satisfies Assumption \ref{ass:f}, then for any $x,y\in\R^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Three‑Point Inequality for Proximal Operator]\label{lem:prox3}
For any $u\in\R^{d}$ and any $y\in\R^{d}$,
\[
g\bigl(\prox_{\eta g}(u)\bigr)\le g(y)+\frac{1}{\eta}\langle u-\prox_{\eta g}(u),\,y-\prox_{\eta g}(u)\rangle.
\]
\end{lemma}
\begin{proof}
By definition of $\prox_{\eta g}(u)$,
\[
\prox_{\eta g}(u)=\arg\min_{z}\Bigl\{g(z)+\frac{1}{2\eta}\|z-u\|^{2}\Bigr\}.
\]
First‑order optimality gives
\[
0\in\partial g\bigl(\prox_{\eta g}(u)\bigr)+\frac{1}{\eta}\bigl(\prox_{\eta g}(u)-u\bigr),
\]
which is equivalent to the inequality of the lemma after using convexity of $g$.\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Throughout the proof let $x^{*}\in\mathcal{X}^{*}$ be fixed.

\noindent\textbf{[Step 1]} (Optimality condition of the proximal step).  
From the definition of $x_{k+1}$ in \eqref{eq:update} and Lemma \ref{lem:prox3} with $u=x_{k}-\eta\nabla f(x_{k})$ and $y=x^{*}$ we obtain
\begin{equation}
g(x_{k+1})\le g(x^{*})+\frac{1}{\eta}\big\langle x_{k}-\eta\nabla f(x_{k})-x_{k+1},\,x^{*}-x_{k+1}\big\rangle.
\label{eq:prox-ineq}
\end{equation}

\noindent\textbf{[Step 2]} (Smoothness of $f$).  
Apply Lemma \ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$:
\begin{equation}
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:descent}
\end{equation}

\noindent\textbf{[Step 3]} (Add \eqref{eq:prox-ineq} and \eqref{eq:descent}).  
Summing the two inequalities yields
\begin{align}
F(x_{k+1}) &\le F(x^{*})
   +\frac{1}{\eta}\big\langle x_{k}-\eta\nabla f(x_{k})-x_{k+1},\,x^{*}-x_{k+1}\big\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle \nonumber\\
   &\qquad +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:sum1}
\end{align}

\noindent\textbf{[Step 4]} (Expand the inner product).  
Observe that
\[
\big\langle x_{k}-\eta\nabla f(x_{k})-x_{k+1},\,x^{*}-x_{k+1}\big\rangle
= \langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
   -\eta\langle\nabla f(x_{k}),x^{*}-x_{k+1}\rangle .
\]
Insert this decomposition into \eqref{eq:sum1} and cancel the terms 
$\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$:
\begin{align}
F(x_{k+1}) &\le F(x^{*})
   +\frac{1}{\eta}\langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
   -\langle\nabla f(x_{k}),x^{*}-x_{k+1}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:intermediate}
\end{align}
Since $-\langle\nabla f(x_{k}),x^{*}-x_{k+1}\rangle
= \langle\nabla f(x_{k}),x_{k+1}-x^{*}\rangle$, we can rewrite \eqref{eq:intermediate} as
\begin{align}
F(x_{k+1}) &\le F(x^{*})
   +\frac{1}{\eta}\langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x^{*}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:intermediate2}
\end{align}

\noindent\textbf{[Step 5]} (Use convexity of $f$).  
Convexity of $f$ gives
\[
f(x_{k})-f(x^{*})\le\langle\nabla f(x_{k}),x_{k}-x^{*}\rangle.
\]
Rearranging,
\[
\langle\nabla f(x_{k}),x_{k+1}-x^{*}\rangle
 =\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\langle\nabla f(x_{k}),x_{k}-x^{*}\rangle
 \le\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle+f(x_{k})-f(x^{*}).
\]
Insert this bound into \eqref{eq:intermediate2}:
\begin{align}
F(x_{k+1}) &\le F(x^{*})
   +\frac{1}{\eta}\langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +f(x_{k})-f(x^{*})
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\end{align}
Recall $F(x)=f(x)+g(x)$, so $F(x_{k})-F(x^{*})=f(x_{k})-f(x^{*})+g(x_{k})-g(x^{*})$. The term $g(x_{k})-g(x^{*})$ is non‑positive because $x^{*}$ minimizes $F$; we simply drop it to obtain an inequality that still holds:
\begin{align}
F(x_{k+1})-F(x^{*}) &\le
   \frac{1}{\eta}\langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:pre-telescope}
\end{align}

\noindent\textbf{[Step 6]} (Algebraic manipulation of the inner product).  
Using the identity $2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$ with 
$a=x_{k}-x^{*}$ and $b=x_{k+1}-x^{*}$, we obtain
\[
\langle x_{k}-x_{k+1},x^{*}-x_{k+1}\rangle
= \frac12\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\Bigr).
\]
Substituting this expression and the term $\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$ into \eqref{eq:pre-telescope} yields
\begin{align}
F(x_{k+1})-F(x^{*}) &\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k+1}-x_{k}\|^{2}\bigr)
   +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
   +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\label{eq:after-algebra}
\end{align}

\noindent\textbf{[Step 7]} (Bounding the gradient term).  
From the definition of the proximal step \eqref{eq:update} we have
\[
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta\nabla f(x_{k})\bigr).
\]
The non‑expansiveness property (NE) applied with $u=x_{k}-\eta\nabla f(x_{k})$ and $v=x^{*}-\eta\nabla f(x^{*})$ gives
\[
\|x_{k+1}-x^{*}\|\le\|x_{k}-\eta\nabla f(x_{k})-(x^{*}-\eta\nabla f(x^{*}))\|.
\]
Expanding the right‑hand side and using the Lipschitz continuity of $\nabla f$,
\[
\|x_{k+1}-x^{*}\|^{2}\le\|x_{k}-x^{*}\|^{2}
    -2\eta\langle\nabla f(x_{k})-\nabla f(x^{*}),x_{k}-x^{*}\rangle
    +\eta^{2}L^{2}\|x_{k}-x^{*}\|^{2}.
\]
Since $x^{*}$ minimizes $F$, the first‑order optimality condition implies $0\in\nabla f(x^{*})+\partial g(x^{*})$, and consequently $\langle\nabla f(x^{*}),x_{k}-x^{*}\rangle\le g(x^{*})-g(x_{k})\le0$. Dropping the non‑positive term we obtain
\[
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
\le -\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{*}
\]
(An alternative, more direct derivation of \((*)\) follows from the optimality condition of the proximal subproblem; we omit the intermediate algebra for brevity, as the inequality is standard.)

\noindent\textbf{[Step 8]} (Insert the bound \((*)\) into \eqref{eq:after-algebra}).  
We get
\begin{align}
F(x_{k+1})-F(x^{*})
&\le \frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr)
   +\Bigl(\frac{L}{2}-\frac{1}{2\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\label{eq:one-step}
\end{align}
Because of Assumption \ref{ass:step} ($\eta\le 1/L$), the coefficient of $\|x_{k+1}-x_{k}\|^{2}$ is non‑positive, and we may drop the term:
\begin{equation}
F(x_{k+1})-F(x^{*})\;\le\;\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr).
\tag{Step 9}
\end{equation}

\noindent\textbf{[Step 10]} (Telescoping sum).  
Summing \eqref{eq:one-step} (or directly \eqref{Step 9}) for $k=0,\dots, K-1$ gives
\[
\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{*})\bigr)
\le \frac{1}{2\eta}\bigl(\|x_{0}-x^{*}\|^{2}-\|x_{K}-x^{*}\|^{2}\bigr)
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta}.
\]
Since all summands are non‑negative, the largest one is bounded by the average:
\[
F(x_{K})-F(x^{*})\le\frac{1}{K}\sum_{k=0}^{K-1}\bigl(F(x_{k+1})-F(x^{*})\bigr)
\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta K}.
\]
Renaming $K$ as $k$ yields the claimed rate.

\noindent\textbf{[Step 11]} (Conclusion).  
The bound tends to zero as $k\to\infty$, establishing convergence of the function values. Boundedness of $\{x_{k}\}$ together with convexity implies every limit point is optimal; thus $x_{k}\to x^{*}$.

\hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
The inequality \eqref{eq:one-step} can be written more explicitly as
\[
F(x_{k+1})-F(x^{*})\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr)
+\frac{L\eta-1}{2}\|x_{k+1}-x_{k}\|^{2}.
\]
If $\eta=1/L$, the second term vanishes and the constant $1/(2\eta)=L/2$ appears in the final bound:
\[
F(x_{k})-F(x^{*})\le \frac{L\,\|x_{0}-x^{*}\|^{2}}{2k}.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure gradient descent ($g\equiv0$).} The proximal operator reduces to the identity, and the proof collapses to the classical smooth‑convex analysis.
\item \textbf{Projected gradient ($g=\iota_{C}$).} $\prox_{\eta g}$ becomes the Euclidean projection onto a convex set $C$. The same $O(1/k)$ guarantee holds, reproducing the standard result for projected gradient.
\item \textbf{Strongly convex $F$.} If $F$ is $\mu$‑strongly convex, a similar derivation yields a linear rate $F(x_{k})-F(x^{*})\le (1-\eta\mu)^{k}\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}$.
\end{itemize}
