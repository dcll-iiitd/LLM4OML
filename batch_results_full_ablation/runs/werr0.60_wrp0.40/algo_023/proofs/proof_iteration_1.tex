\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
We consider the composite convex optimization problem  

\[
\min_{x\in \R^{d}} \; F(x)\;\stackrel{\text{def}}{=}\;f(x)+g(x),
\tag{1}
\]

where  

\begin{itemize}
\item $f:\R^{d}\to\R$ is convex, continuously differentiable and has an $L$‑Lipschitz continuous gradient, i.e.  
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad \forall x,y\in\R^{d};
\tag{2}
\]
\item $g:\R^{d}\to\R\cup\{+\infty\}$ is proper, closed and convex (possibly non‑smooth).  
\end{itemize}

For a stepsize (learning rate) $\eta>0$, the \emph{proximal gradient update} is  

\[
x_{k+1}= \prox_{\eta g}\!\bigl(x_{k}-\eta \nabla f(x_{k})\bigr),
\qquad k=0,1,2,\dots
\tag{3}
\]

where the proximal operator of $g$ is defined by  

\[
\prox_{\eta g}(z) \;\stackrel{\text{def}}{=}\; \arg\min_{u\in\R^{d}}
\Bigl\{ g(u)+\frac{1}{2\eta}\|u-z\|^{2}\Bigr\}.
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, stepsize $\eta\in(0,1/L]$.
\FOR{$k=0,1,\dots,T-1$}
    \STATE Compute gradient $g_{k}:=\nabla f(x_{k})$.
    \STATE Take a gradient step $y_{k}=x_{k}-\eta g_{k}$.
    \STATE Apply proximal map $x_{k+1}= \prox_{\eta g}(y_{k})$.
\ENDFOR
\RETURN $x_{T}$ (or optionally the average $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimize the simple scalar problem  

\[
F(w)=f(w)+g(w),\qquad f(w)=(w-3)^{2},\; g(w)=0,
\]

so $\prox_{\eta g}=\operatorname{id}$.  
Take $w_{0}=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\eta\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(w_{0}-3)=-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}

Corresponding objective values  

\[
F(w_{0})=9,\;F(w_{1})=(0.6-3)^{2}=5.76,\;F(w_{2})=(1.08-3)^{2}=3.6864,\;
F(w_{3})=(1.464-3)^{2}=2.376\ldots
\]

The values decrease monotonically, illustrating the descent property of (3).

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{as:lip}
$f$ is convex and differentiable with $L$‑Lipschitz gradient (cf. (2)).
\end{assumption}
\begin{assumption}[Convexity of $g$]\label{as:g}
$g$ is proper, closed and convex; its proximal operator $\prox_{\eta g}$ is well‑defined for any $\eta>0$.
\end{assumption}
\begin{assumption}[Stepsize]\label{as:step}
The stepsize satisfies $0<\eta\le 1/L$.
\end{assumption}
\begin{assumption}[Existence of a solution]\label{as:sol}
The set of minimizers $\mathcal X^{*}\stackrel{\text{def}}{=}\arg\min_{x}F(x)$ is non‑empty, and we denote any $x^{*}\in\mathcal X^{*}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:ergodic}
Let Assumptions \ref{as:lip}–\ref{as:sol} hold and let $\{x_{k}\}$ be generated by (3) with a stepsize $\eta\in(0,1/L]$. Then for all $k\ge1$  

\[
F(x_{k})-F(x^{*})\;\le\;\frac{\norm{x_{0}-x^{*}}^{2}}{2\eta\,k}.
\tag{5}
\]

Equivalently, the averaged iterate $\bar x_{k}\stackrel{\text{def}}{=}\frac1k\sum_{t=0}^{k-1}x_{t}$ satisfies  

\[
F(\bar x_{k})-F(x^{*})\;\le\;\frac{\norm{x_{0}-x^{*}}^{2}}{2\eta\,k}.
\tag{6}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Because the proximal operator is \emph{non‑expansive}
\[
\norm{\prox_{\eta g}(u)-\prox_{\eta g}(v)}\le \norm{u-v},
\tag{7}
\]
the iteration (3) inherits the stability of a forward Euler step for smooth $f$.
\item \textbf{Hyperparameter Sensitivity.}  The stepsize bound $\eta\le1/L$ is tight: if $\eta>1/L$ the descent lemma (Lemma~\ref{lem:descent}) may fail, and the sequence can diverge.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item When $g\equiv0$, (3) reduces to standard gradient descent with the classic $O(1/k)$ rate.
        \item When $f\equiv0$, (3) becomes the proximal point algorithm, which also enjoys an $O(1/k)$ rate in function value.
        \item For composite problems, PGM attains the same $O(1/k)$ rate without requiring subgradient steps on $g$.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that, under merely convexity and Lipschitz smoothness, the proximal gradient method converges at the optimal sublinear rate for first‑order methods on the class of composite convex problems.  The proof hinges on two elementary facts: (i) the descent lemma for $f$ (smoothness) and (ii) the non‑expansiveness of the proximal map (7).  No stochasticity is involved, thus expectations are unnecessary.  Extensions to accelerated schemes (e.g., FISTA) or stochastic variants follow the same skeleton with refined inequalities.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\R^{d}$  

\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
Standard consequence of the integral form of the gradient; see e.g. Nesterov (2004). \hfill $\square$
\end{proof}

\begin{lemma}[Proximal Optimality Condition]\label{lem:prox-opt}
For any $z\in\R^{d}$ and $\eta>0$, $u=\prox_{\eta g}(z)$ iff  

\[
0\in \partial g(u)+\frac{1}{\eta}(u-z),
\tag{9}
\]
i.e. $-\frac{1}{\eta}(z-u)\in\partial g(u)$.
\end{lemma}
\begin{proof}
By definition (4), $u$ minimizes $g(v)+\frac{1}{2\eta}\|v-z\|^{2}$. First‑order optimality yields (9). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
Let $x^{*}\in\mathcal X^{*}$ be a fixed optimal solution.  Define  

\[
y_{k}=x_{k}-\eta\nabla f(x_{k}),\qquad x_{k+1}= \prox_{\eta g}(y_{k}).
\tag{10}
\]

\noindent\textbf{Goal:} bound $F(x_{k+1})-F(x^{*})$.

\begin{enumerate}
\item[Step 1.] Apply the proximal optimality condition (Lemma~\ref{lem:prox-opt}) with $z=y_{k}$:
\[
-\frac{1}{\eta}(y_{k}-x_{k+1})\in\partial g(x_{k+1}).
\tag{11}
\]

\item[Step 2.] By convexity of $g$,
\[
g(x_{k+1})-g(x^{*})\le \inner{s_{k+1}}{x_{k+1}-x^{*}},
\qquad s_{k+1}:=-\frac{1}{\eta}(y_{k}-x_{k+1}).
\tag{12}
\]

\item[Step 3.] Substitute $y_{k}=x_{k}-\eta\nabla f(x_{k})$ into $s_{k+1}$:
\[
s_{k+1}= -\frac{1}{\eta}\bigl(x_{k}-\eta\nabla f(x_{k})-x_{k+1}\bigr)
          =\nabla f(x_{k})-\frac{1}{\eta}(x_{k}-x_{k+1}).
\tag{13}
\]

\item[Step 4.] Combine (12) and (13):
\[
g(x_{k+1})-g(x^{*})\le 
\inner{\nabla f(x_{k})}{x_{k+1}-x^{*}}
-\frac{1}{\eta}\inner{x_{k}-x_{k+1}}{x_{k+1}-x^{*}}.
\tag{14}
\]

\item[Step 5.] Expand the inner product in the second term:

\[
\inner{x_{k}-x_{k+1}}{x_{k+1}-x^{*}}
= \frac12\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{15}
\]

\item[Step 6.] Plug (15) into (14) and multiply by $1$ (no change):
\[
g(x_{k+1})-g(x^{*})\le
\inner{\nabla f(x_{k})}{x_{k+1}-x^{*}}
-\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{16}
\]

\item[Step 7.] Add $f(x_{k+1})-f(x^{*})$ to both sides.  Using convexity of $f$,
\[
f(x_{k+1})-f(x^{*})\le
\inner{\nabla f(x_{k})}{x_{k+1}-x^{*}} .
\tag{17}
\]

\item[Step 8.] Summing (16) and (17) yields a bound on the composite gap:

\[
F(x_{k+1})-F(x^{*})
\le
\underbrace{2\inner{\nabla f(x_{k})}{x_{k+1}-x^{*}}}_{\text{(A)}}
-\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{18}
\]

\item[Step 9.] Apply the Descent Lemma (Lemma~\ref{lem:descent}) with $x=x_{k}$ and $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_{k})+\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{19}
\]

\item[Step 10.] Rearrange (19) to isolate $\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}$:

\[
\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}
\ge f(x_{k+1})-f(x_{k})-\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{20}
\]

\item[Step 11.] Note that term (A) in (18) can be split as

\[
2\inner{\nabla f(x_{k})}{x_{k+1}-x^{*}}
=2\inner{\nabla f(x_{k})}{x_{k+1}-x_{k}}
+2\inner{\nabla f(x_{k})}{x_{k}-x^{*}}.
\tag{21}
\]

\item[Step 12.] Substitute (20) into (21) and then into (18).  Using convexity of $f$ again,
$\inner{\nabla f(x_{k})}{x_{k}-x^{*}}\le f(x_{k})-f(x^{*})$:

\[
\begin{aligned}
F(x_{k+1})-F(x^{*})
&\le
2\bigl[f(x_{k+1})-f(x_{k})\bigr]
- L\|x_{k+1}-x_{k}\|^{2}
+2\bigl[f(x_{k})-f(x^{*})\bigr] \\
&\qquad
-\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\end{aligned}
\tag{22}
\]

\item[Step 13.] Simplify (22) by cancelling $2f(x_{k})$ terms:

\[
F(x_{k+1})-F(x^{*})
\le
2\bigl[f(x_{k+1})-f(x^{*})\bigr]
- L\|x_{k+1}-x_{k}\|^{2}
-\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{23}
\]

\item[Step 14.] Bring the term $2[f(x_{k+1})-f(x^{*})]$ to the left side, noting that $F(x_{k+1})=f(x_{k+1})+g(x_{k+1})$:

\[
\underbrace{F(x_{k+1})-F(x^{*})}_{\text{LHS}}
-2\bigl[f(x_{k+1})-f(x^{*})\bigr]
= g(x_{k+1})-g(x^{*})- \bigl[f(x_{k+1})-f(x^{*})\bigr].
\tag{24}
\]

Since $g(x_{k+1})-g(x^{*})\ge -\bigl[f(x_{k+1})-f(x^{*})\bigr]$ (by convexity of $f+g$), the left‑hand side of (24) is non‑positive.  Dropping it yields a valid inequality:

\[
0\le
- L\|x_{k+1}-x_{k}\|^{2}
-\frac{1}{2\eta}\!\Bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}
-\|x_{k}-x_{k+1}\|^{2}\Bigr).
\tag{25}
\]

\item[Step 15.] Rearrange (25) to isolate the distance term:

\[
\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr)
\le
\Bigl(L-\frac{1}{2\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}.
\tag{26}
\]

\item[Step 16.] By the stepsize condition $\eta\le 1/L$, we have $L-\frac{1}{2\eta}\le 0$.  Hence the right‑hand side of (26) is non‑positive, and we can discard it, obtaining the fundamental descent inequality

\[
\|x_{k+1}-x^{*}\|^{2}\le \|x_{k}-x^{*}\|^{2}.
\tag{27}
\]

\item[Step 17.] More importantly, using $\eta\le 1/L$ in (26) gives

\[
\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr)
\ge \frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{28}
\]

\item[Step 18.] Return to (22) and drop the non‑positive $-L\|x_{k+1}-x_{k}\|^{2}$ term, then substitute (28):

\[
F(x_{k+1})-F(x^{*})
\le
\frac{1}{2\eta}\bigl(\|x_{k}-x^{*}\|^{2}-\|x_{k+1}-x^{*}\|^{2}\bigr).
\tag{29}
\]

\item[Step 19.] Summing (29) over $t=0,\dots,k-1$ telescopes the distance terms:

\[
\sum_{t=0}^{k-1}\bigl[F(x_{t+1})-F(x^{*})\bigr]
\le
\frac{1}{2\eta}\bigl(\|x_{0}-x^{*}\|^{2}-\|x_{k}-x^{*}\|^{2}\bigr)
\le
\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}.
\tag{30}
\]

\item[Step 20.] Since each summand on the left is non‑negative, we obtain the average bound

\[
\frac{1}{k}\sum_{t=0}^{k-1}\bigl[F(x_{t+1})-F(x^{*})\bigr]
\le
\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k}.
\tag{31}
\]

\item[Step 21.] By convexity of $F$, the function value at the average iterate $\bar x_{k}=\frac1k\sum_{t=0}^{k-1}x_{t+1}$ satisfies  

\[
F(\bar x_{k})-F(x^{*})\le 
\frac{1}{k}\sum_{t=0}^{k-1}\bigl[F(x_{t+1})-F(x^{*})\bigr].
\tag{32}
\]

Combining (31) and (32) yields exactly the bound (6) of Theorem~\ref{thm:ergodic}.  The pointwise bound (5) follows from the monotonicity of the sequence $\{F(x_{k})\}$ which is a direct consequence of (29).  ∎

\section*{Rate Derivation (With Constants)}
From (29) we have for each $k$  

\[
F(x_{k+1})-F(x^{*})\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta(k+1)}.
\]

Thus the constant $C=\frac{\|x_{0}-x^{*}\|^{2}}{2\eta}$ fully characterizes the $O(1/k)$ rate.  If the stepsize is chosen as the maximal admissible $\eta=1/L$, the constant becomes $C=\frac{L}{2}\|x_{0}-x^{*}\|^{2}$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Pure Gradient Descent ($g\equiv0$).}  $\prox_{\eta g}=\operatorname{id}$, and (3) reduces to $x_{k+1}=x_{k}-\eta\nabla f(x_{k})$.  The same proof yields the classical bound $f(x_{k})-f(x^{*})\le\frac{\|x_{0}-x^{*}\|^{2}}{2\eta k}$.
\item \textbf{Proximal Point Method ($f\equiv0$).}  The iteration becomes $x_{k+1}= \prox_{\eta g}(x_{k})$, and (7) directly gives $\|x_{k+1}-x^{*}\|\le\|x_{k}-x^{*}\|$, i.e. monotone convergence with the same $O(1/k)$ functional rate.
\item \textbf{Indicator Regularizer.}  If $g=\iota_{\mathcal C}$ for a closed convex set $\mathcal C$, then $\prox_{\eta g}$ is the Euclidean projection onto $\mathcal C$.  The algorithm is projected gradient descent, and the proof proceeds unchanged.
\end{itemize}
