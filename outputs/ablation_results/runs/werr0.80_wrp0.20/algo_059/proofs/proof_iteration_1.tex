\section*{Quantized Stochastic Gradient Descent (QSGD)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable and $L$–smooth:
\[
\forall\,x,y\in\R^{d},\qquad 
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}. \tag{1}
\]

At iteration $t$ a stochastic gradient $g_{t}$ is sampled, e.g. $g_{t}=\nabla f(w_{t};\xi_{t})$ with
\[
\E[g_{t}\mid w_{t}]=\nabla f(w_{t}),\qquad 
\E\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]\le\sigma^{2}. \tag{2}
\]

A (possibly biased) \emph{quantizer} $Q:\R^{d}\rightarrow\R^{d}$ satisfies
\[
\E[Q(g)\mid g]=g,\qquad 
\E\!\left[\norm{Q(g)-g}^{2}\mid g\right]\le \omega^{2}\norm{g}^{2},\tag{3}
\]
where $\omega\ge0$ quantifies the distortion (e.g. $\omega^{2}= \frac{d}{s}$ for $s$‑bit stochastic rounding).

The update rule of QSGD is
\[
w_{t+1}=w_{t}-\eta_{t}\,Q(g_{t}),\qquad t=0,1,\dots,T-1, \tag{4}
\]
with a (possibly time‑varying) stepsize $\eta_{t}>0$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Quantized Stochastic Gradient Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_{0}\in\R^{d}$, stepsizes $\{\eta_{t}\}_{t=0}^{T-1}$, quantizer $Q(\cdot)$
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample stochastic gradient $g_{t}=\nabla f(w_{t};\xi_{t})$
    \STATE Compute quantized gradient $\tilde g_{t}=Q(g_{t})$
    \STATE $w_{t+1}\leftarrow w_{t}-\eta_{t}\tilde g_{t}$
\ENDFOR
\RETURN $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic 
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3),
\]
which is $L=2$ smooth.  Use a deterministic (identity) quantizer $Q(g)=g$, stepsize $\eta=0.1$, and start from $w_{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_{t}= \nabla f(w_{t})$ & $w_{t+1}=w_{t}-\eta g_{t}$ & $f(w_{t+1})$\\ \hline
0 & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ & $(0.6-3)^{2}=5.76$\\
1 & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ & $(1.08-3)^{2}=3.6864$\\
2 & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ & $(1.464-3)^{2}=2.3665$\\
\end{tabular}
\end{center}
The objective value decreases monotonically, illustrating the algorithm’s behaviour on a simple smooth non‑convex (here actually convex) function.

\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Smoothness.} $f$ is $L$‑smooth as in (1).
\item[(A2)] \textbf{Unbiased stochastic gradient with bounded variance.}  Equation (2) holds for some $\sigma^{2}\ge0$.
\item[(A3)] \textbf{Unbiased quantizer with bounded relative variance.}  Equation (3) holds for some $\omega^{2}\ge0$.
\item[(A4)] \textbf{Stepsize.} For the constant stepsize analysis we assume $0<\eta\le\frac{1}{L}$; for decreasing stepsizes we require $\sum_{t=0}^{\infty}\eta_{t}= \infty$ and $\sum_{t=0}^{\infty}\eta_{t}^{2}<\infty$.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of QSGD for non‑convex smooth $f$]\label{thm:main}
Under (A1)–(A4) and using a constant stepsize $\eta\le\frac{1}{L}$, the iterates generated by \eqref{4} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{L\eta\bigl(\sigma^{2}+\omega^{2}G^{2}\bigr)}_{\text{variance \& quantization term}},
\tag{5}
\]
where $f^{\star}:=\inf_{w}f(w)$ and $G^{2}\ge \sup_{t}\E\!\bigl[\norm{g_{t}}^{2}\bigr]$ (finite for bounded stochastic gradients).  
Choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
\min_{0\le t<T}\E\!\left[\norm{\nabla f(w_{t})}^{2}\right]=\mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound \eqref{5} holds for any $L$‑smooth $f$, regardless of convexity; thus the method is stable under the standard smoothness assumption.
\item \textbf{Hyper‑parameter sensitivity.} The term $L\eta(\sigma^{2}+\omega^{2}G^{2})$ reveals a linear dependence on the stepsize. Too large $\eta$ inflates the variance/quantization contribution, while too small $\eta$ slows the optimization term.
\item \textbf{Comparison to vanilla SGD.} Setting $\omega=0$ (no quantization) recovers the classical non‑convex SGD bound $\frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}$.
\item \textbf{Effect of quantization.} The extra factor $\omega^{2}G^{2}$ quantifies the loss incurred by compression; more bits $\Rightarrow$ smaller $\omega$, thus the bound approaches the uncompressed case.
\end{itemize}

\section*{Discussion}
The theorem shows that QSGD inherits the $\mathcal{O}(1/\sqrt{T})$ convergence rate of standard SGD for non‑convex smooth objectives, up to an additive penalty proportional to the quantizer’s relative variance $\omega^{2}$. By increasing the communication budget (reducing $\omega$) one can make this penalty arbitrarily small, while still enjoying the communication savings of quantization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{lemma}
\begin{lemma}[Unbiasedness of quantizer]\label{lem:unbiased}
For any random vector $g$, the quantizer $Q$ satisfies
\[
\E[Q(g)\mid g]=g.
\]
\end{lemma}
\begin{lemma}[Bounded quantization error]\label{lem:quantvar}
For any $g$,
\[
\E\!\left[\norm{Q(g)-g}^{2}\mid g\right]\le\omega^{2}\norm{g}^{2}.
\]
\end{lemma}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  All expectations are taken with respect to the randomness of the stochastic gradient $g_{t}$ and the quantizer $Q$ unless otherwise noted.

\subsection*{Step 1 – Smoothness expansion}
From Lemma~\ref{lem:smooth} with $x=w_{t}$, $y=w_{t+1}=w_{t}-\eta Q(g_{t})$,
\[
f(w_{t+1})\le f(w_{t})-\eta\inner{\nabla f(w_{t})}{Q(g_{t})}
+\frac{L\eta^{2}}{2}\norm{Q(g_{t})}^{2}. \tag{6}
\]

\subsection*{Step 2 – Take conditional expectation}
Condition on $w_{t}$ and use Lemma~\ref{lem:unbiased}:
\[
\E\!\left[\inner{\nabla f(w_{t})}{Q(g_{t})}\mid w_{t}\right]
= \inner{\nabla f(w_{t})}{\E[Q(g_{t})\mid w_{t}]}
= \inner{\nabla f(w_{t})}{\nabla f(w_{t})}
= \norm{\nabla f(w_{t})}^{2}. \tag{7}
\]

\subsection*{Step 3 – Bound the second moment of the quantized gradient}
\[
\begin{aligned}
\E\!\left[\norm{Q(g_{t})}^{2}\mid w_{t}\right]
&= \E\!\left[\norm{g_{t}+ (Q(g_{t})-g_{t})}^{2}\mid w_{t}\right] \\[1mm]
&= \E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
   +2\E\!\left[\inner{g_{t}}{Q(g_{t})-g_{t}}\mid w_{t}\right]
   +\E\!\left[\norm{Q(g_{t})-g_{t}}^{2}\mid w_{t}\right].
\end{aligned}
\tag{8}
\]
The middle term vanishes because $\E[Q(g_{t})-g_{t}\mid g_{t}]=0$ (Lemma~\ref{lem:unbiased}), hence also zero when further conditioned on $w_{t}$.  Using Lemma~\ref{lem:quantvar} and (A2):
\[
\E\!\left[\norm{Q(g_{t})}^{2}\mid w_{t}\right]
\le \E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
    +\omega^{2}\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
= (1+\omega^{2})\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]. \tag{9}
\]

\subsection*{Step 4 – Decompose stochastic gradient variance}
From (A2):
\[
\E\!\left[\norm{g_{t}}^{2}\mid w_{t}\right]
= \norm{\nabla f(w_{t})}^{2}
  +\E\!\left[\norm{g_{t}-\nabla f(w_{t})}^{2}\mid w_{t}\right]
\le \norm{\nabla f(w_{t})}^{2} + \sigma^{2}. \tag{10}
\]

\subsection*{Step 5 – Plug (7)–(10) into (6) and take full expectation}
Taking expectation $\E[\cdot]$ of (6) and using (7)–(10) yields
\[
\begin{aligned}
\E\!\bigl[f(w_{t+1})\bigr]
&\le \E\!\bigl[f(w_{t})\bigr]
   -\eta\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
   +\frac{L\eta^{2}}{2}\E\!\bigl[\norm{Q(g_{t})}^{2}\bigr] \\[1mm]
&\le \E\!\bigl[f(w_{t})\bigr]
   -\eta\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
   +\frac{L\eta^{2}}{2}(1+\omega^{2})\E\!\bigl[\norm{g_{t}}^{2}\bigr] \\[1mm]
&\le \E\!\bigl[f(w_{t})\bigr]
   -\eta\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
   +\frac{L\eta^{2}}{2}(1+\omega^{2})\Bigl(\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]+\sigma^{2}\Bigr).
\end{aligned}
\tag{11}
\]

\subsection*{Step 6 – Rearrangement}
Move the term involving $\E[\norm{\nabla f(w_{t})}^{2}]$ to the left-hand side:
\[
\Bigl(\eta-\frac{L\eta^{2}}{2}(1+\omega^{2})\Bigr)
\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \E\!\bigl[f(w_{t})-f(w_{t+1})\bigr]
      +\frac{L\eta^{2}}{2}(1+\omega^{2})\sigma^{2}. \tag{12}
\]

\subsection*{Step 7 – Choose stepsize $\eta\le\frac{1}{L}$}
Since $\eta\le\frac{1}{L}$ and $\omega^{2}\ge0$, we have
\[
\eta-\frac{L\eta^{2}}{2}(1+\omega^{2})
\ge \frac{\eta}{2}. \tag{13}
\]
(Indeed, $\frac{L\eta^{2}}{2}(1+\omega^{2})\le\frac{L\eta^{2}}{2}\cdot2=\ L\eta^{2}\le\eta$.)

Applying (13) to (12) gives
\[
\frac{\eta}{2}\,\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \E\!\bigl[f(w_{t})-f(w_{t+1})\bigr]
      +\frac{L\eta^{2}}{2}(1+\omega^{2})\sigma^{2}. \tag{14}
\]

\subsection*{Step 8 – Sum over $t=0,\dots,T-1$}
Summing (14) from $t=0$ to $T-1$ telescopes the function values:
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le f(w_{0})- \E\!\bigl[f(w_{T})\bigr] 
   + \frac{L\eta^{2}}{2}(1+\omega^{2})\sigma^{2}T. \tag{15}
\]
Since $f(w_{T})\ge f^{\star}$,
\[
\frac{\eta}{2}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le f(w_{0})-f^{\star}
   + \frac{L\eta^{2}}{2}(1+\omega^{2})\sigma^{2}T. \tag{16}
\]

\subsection*{Step 9 – Divide by $T$ and rearrange}
Dividing (16) by $T$ and multiplying by $2/\eta$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le
\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\eta T}
+ L\eta(1+\omega^{2})\sigma^{2}. \tag{17}
\]

\subsection*{Step 10 – Replace $G^{2}$}
If the stochastic gradients are uniformly bounded, i.e. $\E[\norm{g_{t}}^{2}]\le G^{2}$, then $\sigma^{2}\le G^{2}$ and the variance term can be written as $L\eta(\sigma^{2}+\omega^{2}G^{2})$, which is the form displayed in Theorem~\ref{thm:main}.  This completes the proof. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (17) set $\eta = \frac{c}{\sqrt{T}}$ with $c\le\frac{1}{L}$.
\[
\frac{2(f(w_{0})-f^{\star})}{\eta T}
= \frac{2(f(w_{0})-f^{\star})}{c}\,\frac{1}{\sqrt{T}},\qquad
L\eta(\sigma^{2}+\omega^{2}G^{2})
= \frac{Lc}{\sqrt{T}}(\sigma^{2}+\omega^{2}G^{2}).
\]
Thus
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\;\le\;
\underbrace{\frac{2(f(w_{0})-f^{\star})}{c}}_{\text{constant}}
\frac{1}{\sqrt{T}}
+\underbrace{Lc(\sigma^{2}+\omega^{2}G^{2})}_{\text{constant}}
\frac{1}{\sqrt{T}}
= \mathcal{O}\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No quantization ($\omega=0$).}  The bound reduces to the classic SGD result:
\[
\frac{1}{T}\sum_{t}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\sigma^{2}.
\]
\item \textbf{Deterministic gradients ($\sigma=0$).}  Only the quantization term remains:
\[
\frac{1}{T}\sum_{t}\E\!\bigl[\norm{\nabla f(w_{t})}^{2}\bigr]
\le \frac{2(f(w_{0})-f^{\star})}{\eta T}+L\eta\omega^{2}G^{2}.
\]
\item \textbf{Increasing bits.}  For stochastic rounding with $s$ bits per coordinate,
$\omega^{2}= \frac{d}{s}$, so the penalty decays linearly with the communication budget.
\end{itemize}
