\section*{Algorithm Name}
\textbf{Mirror Descent (MD)} with a fixed stepsize schedule $\{\eta_k\}_{k\ge 0}$ and a strongly convex prox‑function $h$.

\section*{Mathematical Formulation}
Let $f:\R^{n}\rightarrow\R$ be a convex differentiable objective and let $h:\R^{n}\rightarrow\R$ be a $\sigma$‑strongly convex function (w.r.t. a norm $\|\cdot\|$).  
Its Bregman divergence is
\[
\D_h(x,y)\;:=\;h(x)-h(y)-\inner{\grad h(y)}{x-y},\qquad x,y\in\R^{n}.
\]

Given the current iterate $x_k$, the MD update with stepsize $\eta_k>0$ is
\begin{equation}
\label{eq:md}
x_{k+1}\;=\;\argmin_{x\in\R^{n}}\Big\{ \inner{\grad f(x_k)}{x}+\frac{1}{\eta_k}\,\D_h(x,x_k)\Big\}.
\end{equation}
When $h(x)=\frac12\|x\|_2^{2}$, $\D_h$ reduces to the Euclidean distance and \eqref{eq:md} becomes the classic gradient descent step.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x_0\in\R^{n}$, stepsizes $\{\eta_k\}_{k\ge0}$, prox‑function $h$ (strongly convex).
\FOR{$k=0,1,2,\dots$}
    \STATE Compute gradient $g_k=\grad f(x_k)$.
    \STATE Solve the proximal subproblem
    \[
        x_{k+1}\;=\;\argmin_{x}\Big\{ \inner{g_k}{x}+\frac{1}{\eta_k}\,\D_h(x,x_k)\Big\}.
    \]
\ENDFOR
\STATE \textbf{Output:} $\{x_k\}_{k}$ (or the averaged iterate $\bar x_T=\frac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=(w-3)^2,\qquad w\in\R,
\]
with prox‑function $h(w)=\frac12 w^{2}$ (so $\D_h(u,v)=\frac12(u-v)^2$) and stepsize $\eta_k\equiv\eta=0.1$.  
Initialize $w_0=0$.

\medskip
\noindent\textbf{Iteration $k=0$:}
\[
g_0=\grad f(w_0)=2(w_0-3)=-6.
\]
The subproblem:
\[
w_{1}=\argmin_{w}\Big\{ g_0 w+\frac{1}{\eta}\frac12 (w-w_0)^2\Big\}
       =\argmin_{w}\Big\{-6 w+5 (w-0)^2\Big\}.
\]
Derivative $-6+10w=0\;\Rightarrow\; w_1=0.6$.
\[
f(w_1)=(0.6-3)^2=5.76.
\]

\noindent\textbf{Iteration $k=1$:}
\[
g_1=2(w_1-3)=2(0.6-3)=-4.8.
\]
Subproblem:
\[
w_{2}=\argmin_{w}\Big\{-4.8 w+5 (w-0.6)^2\Big\}
\Longrightarrow -4.8+10(w-0.6)=0\;\Rightarrow\; w_2=0.96.
\]
\[
f(w_2)=(0.96-3)^2=4.1616.
\]

\noindent\textbf{Iteration $k=2$:}
\[
g_2=2(w_2-3)=2(0.96-3)=-4.08.
\]
Subproblem:
\[
w_{3}=\argmin_{w}\Big\{-4.08 w+5 (w-0.96)^2\Big\}
\Longrightarrow -4.08+10(w-0.96)=0\;\Rightarrow\; w_3=1.248.
\]
\[
f(w_3)=(1.248-3)^2=3.0705.
\]

Thus after three iterations the objective value dropped from $9$ to $3.07$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness w.r.t. $h$]
\label{ass:convexsmooth}
The objective $f$ is convex and $L$‑smooth relative to $h$, i.e.
\[
f(x)\le f(y)+\inner{\grad f(y)}{x-y}+L\,\D_h(x,y),\qquad\forall x,y\in\R^{n}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Strong Convexity of $h$]
\label{ass:hstrong}
The prox‑function $h$ is $\sigma$‑strongly convex w.r.t. $\|\cdot\|$:
\[
\D_h(x,y)\ge\frac{\sigma}{2}\|x-y\|^{2},\qquad\forall x,y\in\R^{n}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Stepsize]
\label{ass:stepsize}
The stepsizes are constant $\eta_k=\eta\in\bigl(0,\frac{1}{L}\bigr]$.
\tag{A3}
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Descent]
\label{thm:md}
Under Assumptions~\ref{ass:convexsmooth}–\ref{ass:stepsize}, let $x^{\star}\in\arg\min f$ and define the averaged iterate
\[
\bar x_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k .
\]
Then for any $T\ge1$,
\[
f(\bar x_T)-f(x^{\star})\;\le\;\frac{1}{\eta T}\,\D_h(x^{\star},x_0).
\tag{1}
\]
Consequently, choosing $\eta=\frac{1}{L}$ yields the $O\!\bigl(\frac{1}{T}\bigr)$ rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The update is a contraction in the Bregman geometry because of the strong convexity of $h$ (Assumption~\ref{ass:hstrong}); the distance $\D_h(x^{\star},x_k)$ never increases.
\item \textbf{Hyperparameter Sensitivity.} The bound (1) is monotone decreasing in $\eta$ up to $\eta=1/L$; larger steps lead to a larger constant, while $\eta>1/L$ may violate the descent inequality.
\item \textbf{Comparison to Gradient Descent.} When $h$ is Euclidean, $\D_h$ becomes $\frac12\|x-y\|^2$ and Theorem~\ref{thm:md} reduces to the classical $O(1/T)$ guarantee for gradient descent on $L$‑smooth convex functions.
\end{itemize}

\section*{Discussion}
The theorem shows that Mirror Descent inherits the optimal $O(1/T)$ convergence for convex smooth problems, while allowing the algorithm to exploit geometry via a non‑Euclidean $h$. In settings where the feasible set or the regularizer is naturally expressed in a specific Bregman geometry (e.g., simplex with negative entropy), MD can yield dramatically better practical performance despite the same worst‑case rate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Bregman Divergence]
For a differentiable convex $h$, $\D_h(x,y)=h(x)-h(y)-\inner{\grad h(y)}{x-y}$.
\end{definition}
\begin{definition}[Convex Conjugate]
The Fenchel conjugate of $h$ is $h^{*}(s)=\sup_{x}\{\inner{s}{x}-h(x)\}$.
\end{definition}
\begin{lemma}[Three‑Point Identity]
\label{lem:3pt}
For any $x,y,z$,
\[
\inner{\grad h(z)-\grad h(y)}{x-z}= \D_h(x,y)-\D_h(x,z)-\D_h(z,y).
\tag{L1}
\]
\end{lemma}
\begin{proof}
Direct expansion of the definition of $\D_h$ yields the identity; see e.g. \cite{Beck2003}.
\end{proof}
\begin{lemma}[Optimality Condition of the MD Subproblem]
\label{lem:optcond}
The solution $x_{k+1}$ of \eqref{eq:md} satisfies
\[
\inner{\grad f(x_k)}{x-x_{k+1}}+\frac{1}{\eta_k}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x-x_{k+1}}\ge0,
\qquad\forall x\in\R^{n}.
\tag{L2}
\]
\end{lemma}
\begin{proof}
The subproblem is a convex optimization problem; the first‑order optimality condition gives exactly \eqref{L2}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:md}. Let $x^{\star}\in\arg\min f$ be arbitrary but fixed.

\begin{proof}
\textbf{[Step 1]} Apply Lemma~\ref{lem:optcond} with $x=x^{\star}$:
\[
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}+\frac{1}{\eta_k}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}\ge0.
\tag{1.1}
\]

\textbf{[Step 2]} Rearrange the inner product involving $h$ using the three‑point identity (Lemma~\ref{lem:3pt}) with $(x,y,z)=(x^{\star},x_k,x_{k+1})$:
\[
\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}
= \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})-\D_h(x_{k+1},x_k).
\tag{2.1}
\]

\textbf{[Step 3]} Substitute \eqref{2.1} into \eqref{1.1} and multiply by $\eta_k$:
\[
\eta_k\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
+ \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})-\D_h(x_{k+1},x_k)\ge0.
\tag{3.1}
\]

\textbf{[Step 4]} Use convexity of $f$:
\[
f(x^{\star})\ge f(x_k)+\inner{\grad f(x_k)}{x^{\star}-x_k}.
\tag{4.1}
\]
Add $\inner{\grad f(x_k)}{x_k-x_{k+1}}$ to both sides of \eqref{4.1} and note that
\[
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
=\inner{\grad f(x_k)}{x^{\star}-x_k}+\inner{\grad f(x_k)}{x_k-x_{k+1}}.
\tag{4.2}
\]

\textbf{[Step 5]} Combine \eqref{3.1} and \eqref{4.2}:
\[
\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
\le \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})-\D_h(x_{k+1},x_k)
-\eta_k\inner{\grad f(x_k)}{x_k-x_{k+1}}.
\tag{5.1}
\]

\textbf{[Step 6]} Apply the relative smoothness assumption (A1) with $(x,y)=(x_{k+1},x_k)$:
\[
f(x_{k+1})\le f(x_k)+\inner{\grad f(x_k)}{x_{k+1}-x_k}+L\,\D_h(x_{k+1},x_k).
\tag{6.1}
\]
Rearrange to obtain
\[
-\inner{\grad f(x_k)}{x_k-x_{k+1}}\le f(x_{k+1})-f(x_k)+L\,\D_h(x_{k+1},x_k).
\tag{6.2}
\]

\textbf{[Step 7]} Plug \eqref{6.2} into \eqref{5.1}:
\[
\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
\le \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})-\D_h(x_{k+1},x_k)
+\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr]+\eta_k L\,\D_h(x_{k+1},x_k).
\tag{7.1}
\]

\textbf{[Step 8]} Collect the terms involving $f(x_k)$:
\[
\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
-\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr]
\le \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})
+\bigl(\eta_k L-1\bigr)\D_h(x_{k+1},x_k).
\tag{8.1}
\]

\textbf{[Step 9]} Under the stepsize condition $\eta_k\le 1/L$ (Assumption~\ref{ass:stepsize}), the coefficient $\eta_k L-1\le0$, hence we can drop the non‑positive term:
\[
\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
-\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr]
\le \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1}).
\tag{9.1}
\]

\textbf{[Step 10]} Rearrange \eqref{9.1} to isolate $f(x_k)-f(x^{\star})$:
\[
\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
\le \D_h(x^{\star},x_k)-\D_h(x^{\star},x_{k+1})
+\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr].
\tag{10.1}
\]

\textbf{[Step 11]} Sum inequality \eqref{10.1} for $k=0,\dots,T-1$ telescopically. The Bregman terms collapse:
\[
\sum_{k=0}^{T-1}\eta_k\bigl[f(x_k)-f(x^{\star})\bigr]
\le \D_h(x^{\star},x_0)-\D_h(x^{\star},x_T)
+\sum_{k=0}^{T-1}\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr].
\tag{11.1}
\]

\textbf{[Step 12]} The second sum on the right side telescopes as well:
\[
\sum_{k=0}^{T-1}\eta_k\bigl[f(x_{k+1})-f(x_k)\bigr]
= \eta_{T-1}f(x_T)-\eta_0f(x_0)-\sum_{k=0}^{T-2}\bigl(\eta_{k+1}-\eta_k\bigr)f(x_{k+1}).
\]
With a \emph{constant} stepsize $\eta_k=\eta$, this reduces to $\eta\bigl[f(x_T)-f(x_0)\bigr]$ and cancels partially with the left‑hand side, yielding
\[
\eta\sum_{k=0}^{T-1}\bigl[f(x_k)-f(x^{\star})\bigr]
\le \D_h(x^{\star},x_0)-\D_h(x^{\star},x_T).
\tag{12.1}
\]

\textbf{[Step 13]} Since $\D_h(\cdot,\cdot)\ge0$, we drop the non‑negative term $-\D_h(x^{\star},x_T)$:
\[
\eta\sum_{k=0}^{T-1}\bigl[f(x_k)-f(x^{\star})\bigr]\le \D_h(x^{\star},x_0).
\tag{13.1}
\]

\textbf{[Step 14]} Divide by $T\eta$ and use convexity of $f$ to replace the average of function values by the function value at the averaged iterate $\bar x_T$ (Jensen’s inequality):
\[
f(\bar x_T)-f(x^{\star})
\le \frac{1}{T\eta}\,\D_h(x^{\star},x_0).
\tag{14.1}
\]
This is precisely the bound \eqref{1} claimed in Theorem~\ref{thm:md}. $\square$
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{14.1} and Assumption~\ref{ass:stepsize} with $\eta=1/L$ we obtain
\[
f(\bar x_T)-f(x^{\star})\;\le\;\frac{L}{T}\,\D_h(x^{\star},x_0).
\]
If $h$ is $\sigma$‑strongly convex (Assumption~\ref{ass:hstrong}), then $\D_h(x^{\star},x_0)\le\frac{1}{2\sigma}\|x^{\star}-x_0\|^2$, giving the more explicit
\[
f(\bar x_T)-f(x^{\star})\;\le\;\frac{L}{2\sigma T}\,\|x^{\star}-x_0\|^2.
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean Mirror Descent.} $h(x)=\tfrac12\|x\|_2^2$ $\Rightarrow$ $\D_h=\tfrac12\|x-y\|_2^2$, $\sigma=1$, and the bound reduces to the classic gradient descent rate $O(L\|x_0-x^{\star}\|^2/T)$.
\item \textbf{Negative Entropy on the Simplex.} $h(x)=\sum_{i}x_i\log x_i$ yields $\D_h$ equal to the Kullback–Leibler divergence. The same $O(1/T)$ guarantee holds, but the geometry respects the simplex constraints without explicit projection.
\item \textbf{Adaptive Stepsizes.} If $\eta_k=1/(L_k)$ with $L_k$ an upper bound on the local relative smoothness constant, the proof proceeds unchanged; the telescoping argument uses $\eta_k\le 1/L_k$ at each iteration.
\end{itemize}
