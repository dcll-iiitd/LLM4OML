\section*{Algorithm Name}
\textbf{Stochastic Variance‑Reduced Gradient (SVRG) for Non‑Convex Smooth Optimization}\\
The method follows a double‑loop structure: an outer epoch computes a \emph{snapshot} full gradient, while an inner loop performs $m$ stochastic updates using a variance‑reduced estimator.

\section*{Mathematical Formulation}
Consider a finite‑sum objective
\[
f(x)=\frac1n\sum_{i=1}^{n} f_i(x),\qquad x\in\R^{d},
\]
where each component $f_i$ is $L$‑smooth (possibly non‑convex).  

Let $\tilde{x}$ denote the snapshot point at the beginning of an epoch and
\[
\tilde{\mu}= \nabla f(\tilde{x})=\frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x})
\]
its full gradient.  

For $t=0,\dots,m-1$ the inner‑loop iterate $x_t$ is updated by
\[
\boxed{v_t \;=\; \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x})+\tilde{\mu}},\qquad
x_{t+1}=x_t-\eta v_t,
\]
where $i_t$ is drawn uniformly from $\{1,\dots,n\}$ and $\eta>0$ is the stepsize.  
After $m$ inner steps we set $\tilde{x}\leftarrow x_m$ (or a random inner iterate) and start a new epoch.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex, $L$‑smooth)}\label{alg:svrg}
\begin{algorithmic}[1]
\Require Initial point $x^{0}\in\R^{d}$, stepsize $\eta>0$, epoch length $m$, total epochs $S$.
\State $\tilde{x}\gets x^{0}$
\For{$s=0$ to $S-1$}
    \State Compute full gradient $\tilde{\mu}= \frac1n\sum_{i=1}^{n}\nabla f_i(\tilde{x})$
    \State $x_0\gets \tilde{x}$
    \For{$t=0$ to $m-1$}
        \State Sample $i_t\sim\text{Unif}\{1,\dots,n\}$
        \State $v_t \gets \nabla f_{i_t}(x_t)-\nabla f_{i_t}(\tilde{x})+\tilde{\mu}$
        \State $x_{t+1}\gets x_t-\eta v_t$
    \EndFor
    \State $\tilde{x}\gets x_m$ \Comment{or $\tilde{x}\gets x_{t}$ for a random $t\in\{0,\dots,m-1\}$}
\EndFor
\State \Return $\tilde{x}$ (or the last inner iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We illustrate one epoch on the simple scalar function
\[
f(w)=(w-3)^2,\qquad f_i\equiv f\;(n=1).
\]
\begin{itemize}
\item \textbf{Initialization:} $w^{0}=0$, stepsize $\eta=0.1$, epoch length $m=3$.
\item \textbf{Full gradient at snapshot:} $\tilde{\mu}= \nabla f(\tilde{w}) = 2(\tilde{w}-3)$. With $\tilde{w}=0$, $\tilde{\mu}= -6$.
\item \textbf{Inner loop:}
\begin{enumerate}
\item $t=0$: $i_0=1$ (only index). 
\[
v_0 = \underbrace{\nabla f(w_0)}_{2(w_0-3)}-\underbrace{\nabla f(\tilde{w})}_{-6}+ \tilde{\mu}
   = 2(0-3)-(-6)+(-6)= -6+6-6 = -6.
\]
Update $w_1 = w_0-\eta v_0 = 0-0.1(-6)=0.6$.  
$f(w_1) = (0.6-3)^2 = 5.76$.
\item $t=1$: $w_1=0.6$.
\[
v_1 = 2(0.6-3)-(-6)+(-6)= -4.8+6-6 = -4.8.
\]
Update $w_2 = 0.6-0.1(-4.8)=1.08$.  
$f(w_2) = (1.08-3)^2 = 3.6864$.
\item $t=2$: $w_2=1.08$.
\[
v_2 = 2(1.08-3)-(-6)+(-6)= -3.84+6-6 = -3.84.
\]
Update $w_3 = 1.08-0.1(-3.84)=1.464$.  
$f(w_3) = (1.464-3)^2 \approx 2.363.
\]
\end{enumerate}
\item \textbf{Snapshot update:} $\tilde{w}\gets w_3=1.464$ for the next epoch.
\end{itemize}
The objective value decreased from $9$ (at $w=0$) to $\approx2.36$ after a single epoch.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}. \tag{A1}
\]
\item \textbf{Bounded below.} The objective has a finite infimum $f^{\star}>-\infty$.
\item \textbf{Unbiased stochastic gradient.} For any $x$ and random index $i\sim\text{Unif}\{1,\dots,n\}$,
\[
\E_{i}\big[\nabla f_i(x)\big]=\nabla f(x). \tag{A2}
\]
\item \textbf{Bounded variance at any point.} There exists $\sigma^{2}\ge0$ such that
\[
\E_{i}\big[\|\nabla f_i(x)-\nabla f(x)\|^{2}\big]\le\sigma^{2},\qquad\forall x\in\R^{d}. \tag{A3}
\]
\item \textbf{Stepsize.} The learning rate satisfies
\[
0<\eta\le\frac{1}{3L}. \tag{A4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Non‑convex SVRG convergence]\label{thm:svrg}
Under assumptions (A1)–(A4) and with epoch length $m=n$ (or any $m\ge 1$), let
\[
T = Sm\quad\text{be the total number of stochastic updates.}
\]
Define the random output $\hat{x}$ as a uniformly chosen iterate among all inner iterates
$\{x_t^{(s)}\}_{s=0,\dots,S-1,\;t=0,\dots,m-1}$. Then
\[
\E\!\left[\|\nabla f(\hat{x})\|^{2}\right]
\;\le\;
\frac{2\big(f(x^{0})-f^{\star}\big)}{\eta T}
\;+\;
\frac{L\eta\sigma^{2}}{2}. \tag{1}
\]
Consequently, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\E\!\left[\|\nabla f(\hat{x})\|^{2}\right]=\mathcal{O}\!\bigg(\frac{1}{\sqrt{T}}\bigg).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The variance‑reduced estimator $v_t$ satisfies
\[
\E\big[\|v_t\|^{2}\big]\le 2L^{2}\E\big[\|x_t-\tilde{x}\|^{2}\big]+2\|\tilde{\mu}\|^{2},
\]
ensuring that the stochastic noise does not blow up provided the inner loop length $m$ is not too large.
\item \textbf{Hyperparameter sensitivity.} The bound (1) shows a linear dependence on $\eta$ in the variance term and a $1/(\eta T)$ dependence in the bias term. An optimal trade‑off is achieved at $\eta^{\star}= \sqrt{\frac{2\big(f(x^{0})-f^{\star}\big)}{L\sigma^{2}T}}$, which respects (A4) for sufficiently large $T$.
\item \textbf{Comparison to plain SGD.} For non‑convex $L$‑smooth problems SGD typically yields $\E[\|\nabla f(\bar{x}_T)\|^{2}]=\mathcal{O}(1/\sqrt{T})$ with a larger constant (variance term $L\eta\sigma^{2}$ dominates). SVRG reduces the variance term by the factor $m/n$, leading to a strictly smaller constant when $m\approx n$.
\end{itemize}

\section*{Discussion}
The theorem guarantees that SVRG finds an \emph{approximate stationary point} at the same $\mathcal{O}(1/\sqrt{T})$ rate as SGD but with a smaller prefactor thanks to variance reduction. The result holds without convexity, requiring only smoothness and bounded variance. When the variance bound $\sigma^{2}=0$ (i.e., all $f_i$ are identical), the bound reduces to the deterministic gradient descent rate $\mathcal{O}(1/T)$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \hfill$\blacksquare$
\end{proof}

\begin{lemma}[Variance bound for the SVRG estimator]\label{lem:var}
For any $x$ and snapshot $\tilde{x}$,
\[
\E_{i}\!\big[\|v(x)-\nabla f(x)\|^{2}\big]
\le 2L^{2}\|x-\tilde{x}\|^{2}+2\sigma^{2}, \tag{L2}
\]
where $v(x)=\nabla f_i(x)-\nabla f_i(\tilde{x})+\tilde{\mu}$.
\end{lemma}
\begin{proof}
\[
\begin{aligned}
v(x)-\nabla f(x) 
&= \big(\nabla f_i(x)-\nabla f_i(\tilde{x})\big)
   -\big(\nabla f(x)-\nabla f(\tilde{x})\big)\\
&\quad +\big(\tilde{\mu}-\nabla f(\tilde{x})\big).
\end{aligned}
\]
Taking expectation, the last term vanishes because $\tilde{\mu}=\nabla f(\tilde{x})$.
Using $(a+b)^{2}\le2a^{2}+2b^{2}$ and (A3) gives (L2). \hfill$\blacksquare$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:svrg}.  All expectations are taken w.r.t. the random choices of indices $i_t$ and, when needed, the random output $\hat{x}$.

\bigskip
\noindent\textbf{Notation.}
\begin{itemize}
\item $x_t^{(s)}$ – inner iterate $t$ of epoch $s$.
\item $\tilde{x}^{(s)}$ – snapshot at the beginning of epoch $s$.
\item $\tilde{\mu}^{(s)}=\nabla f(\tilde{x}^{(s)})$.
\item $v_t^{(s)}$ – variance‑reduced estimator used at step $t$ of epoch $s$.
\end{itemize}
For readability we drop superscripts when the epoch is clear.

\bigskip
\noindent\textbf{[Step 1]} (One‑step descent using Lemma~\ref{lem:smooth}).  
Applying (L1) with $y=x_{t+1}=x_t-\eta v_t$,
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)+\langle\nabla f(x_t), -\eta v_t\rangle
   +\frac{L}{2}\eta^{2}\|v_t\|^{2} \\
&= f(x_t)-\eta\langle\nabla f(x_t),v_t\rangle
   +\frac{L\eta^{2}}{2}\|v_t\|^{2}. \tag{2}
\end{aligned}
\]

\noindent\textbf{[Step 2]} (Decompose the inner product).  
Write $v_t = \nabla f(x_t) + (v_t-\nabla f(x_t))$. Then
\[
\langle\nabla f(x_t),v_t\rangle
= \|\nabla f(x_t)\|^{2}
  +\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle. \tag{3}
\]

\noindent\textbf{[Step 3]} (Take expectation conditioned on $x_t$).  
Using $\E_{i_t}[v_t]=\nabla f(x_t)$ (unbiasedness from (A2)),
\[
\E_{i_t}[\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle]=0. \tag{4}
\]

\noindent\textbf{[Step 4]} (Bound the second moment).  
From Lemma~\ref{lem:var},
\[
\E_{i_t}\big[\|v_t\|^{2}\big]
= \|\nabla f(x_t)\|^{2}
  +\E_{i_t}\big[\|v_t-\nabla f(x_t)\|^{2}\big]
\le \|\nabla f(x_t)\|^{2}+2L^{2}\|x_t-\tilde{x}\|^{2}+2\sigma^{2}. \tag{5}
\]

\noindent\textbf{[Step 5]} (Plug (3)–(5) into (2) and take conditional expectation).  
\[
\begin{aligned}
\E_{i_t}[f(x_{t+1})]
&\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
   +\frac{L\eta^{2}}{2}\Big(\|\nabla f(x_t)\|^{2}
   +2L^{2}\|x_t-\tilde{x}\|^{2}+2\sigma^{2}\Big) \\
&= f(x_t)-\underbrace{\Big(\eta-\frac{L\eta^{2}}{2}\Big)}_{c_1}\|\nabla f(x_t)\|^{2}
   +L^{3}\eta^{2}\|x_t-\tilde{x}\|^{2}
   +L\eta^{2}\sigma^{2}. \tag{6}
\end{aligned}
\]

\noindent\textbf{[Step 6]} (Choose stepsize to make $c_1\ge \eta/2$).  
Assumption (A4) gives $\eta\le 1/(3L)$, hence $L\eta\le 1/3$ and
\[
c_1=\eta-\frac{L\eta^{2}}{2}\ge \eta-\frac{\eta}{6}= \frac{5\eta}{6}\ge\frac{\eta}{2}. \tag{7}
\]

\noindent\textbf{[Step 7]} (Simplify (6) using (7)).  
\[
\E_{i_t}[f(x_{t+1})]
\le f(x_t)-\frac{\eta}{2}\|\nabla f(x_t)\|^{2}
   +L^{3}\eta^{2}\|x_t-\tilde{x}\|^{2}
   +L\eta^{2}\sigma^{2}. \tag{8}
\]

\noindent\textbf{[Step 8]} (Unroll over the $m$ inner steps).  
Summing (8) for $t=0,\dots,m-1$ and telescoping the $f$‑terms yields
\[
\begin{aligned}
\E\!\big[f(x_{m})\big]
&\le f(\tilde{x})-\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\big[\|\nabla f(x_t)\|^{2}\big]
   +L^{3}\eta^{2}\sum_{t=0}^{m-1}\E\!\big[\|x_t-\tilde{x}\|^{2}\big]
   +mL\eta^{2}\sigma^{2}. \tag{9}
\end{aligned}
\]

\noindent\textbf{[Step 9]} (Control the distance term).  
Because $x_{t+1}=x_t-\eta v_t$, we have
\[
\|x_{t+1}-\tilde{x}\|^{2}
= \|x_t-\tilde{x}\|^{2}
   -2\eta\langle v_t, x_t-\tilde{x}\rangle
   +\eta^{2}\|v_t\|^{2}.
\]
Taking expectation and using $\E[v_t]=\nabla f(x_t)$ together with Cauchy–Schwarz,
\[
\E[\|x_{t+1}-\tilde{x}\|^{2}]
\le \E[\|x_t-\tilde{x}\|^{2}]
   +\eta^{2}\E[\|v_t\|^{2}]
   \le \E[\|x_t-\tilde{x}\|^{2}]
   +2\eta^{2}\big(\|\nabla f(x_t)\|^{2}+L^{2}\|x_t-\tilde{x}\|^{2}+ \sigma^{2}\big).
\]
Rearranging,
\[
\big(1-2L^{2}\eta^{2}\big)\E[\|x_t-\tilde{x}\|^{2}]
\le \E[\|x_{t+1}-\tilde{x}\|^{2}]
   +2\eta^{2}\|\nabla f(x_t)\|^{2}+2\eta^{2}\sigma^{2}. \tag{10}
\]
Choosing $\eta\le 1/(3L)$ guarantees $1-2L^{2}\eta^{2}\ge 1/2$, hence
\[
\E[\|x_t-\tilde{x}\|^{2}]
\le 2\E[\|x_{t+1}-\tilde{x}\|^{2}]
   +4\eta^{2}\|\nabla f(x_t)\|^{2}+4\eta^{2}\sigma^{2}. \tag{11}
\]

\noindent\textbf{[Step 10]} (Sum (11) over $t=0,\dots,m-1$).  
The telescoping sum of the first term yields
\[
\sum_{t=0}^{m-1}\E[\|x_t-\tilde{x}\|^{2}]
\le 2\sum_{t=0}^{m-1}\E[\|x_{t+1}-\tilde{x}\|^{2}]
   +4\eta^{2}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
   +4m\eta^{2}\sigma^{2}.
\]
Shift the index on the first sum and note that $\|x_m-\tilde{x}\|^{2}\ge0$:
\[
\sum_{t=0}^{m-1}\E[\|x_t-\tilde{x}\|^{2}]
\le 4\eta^{2}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
   +4m\eta^{2}\sigma^{2}. \tag{12}
\]

\noindent\textbf{[Step 11]} (Insert (12) into (9).)  
\[
\begin{aligned}
\E[f(x_m)]
&\le f(\tilde{x})
   -\frac{\eta}{2}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
   +L^{3}\eta^{2}\Big(4\eta^{2}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
          +4m\eta^{2}\sigma^{2}\Big) \\
&\qquad +mL\eta^{2}\sigma^{2}.
\end{aligned}
\]
Grouping the gradient‑norm terms,
\[
\E[f(x_m)]
\le f(\tilde{x})
   -\Big(\frac{\eta}{2}-4L^{3}\eta^{4}\Big)
      \sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
   +\big(4mL^{3}\eta^{4}+mL\eta^{2}\big)\sigma^{2}. \tag{13}
\]

\noindent\textbf{[Step 12]} (Choose $\eta$ small enough so that $\frac{\eta}{2}-4L^{3}\eta^{4}\ge \frac{\eta}{4}$).  
Since $\eta\le 1/(3L)$, we have $\eta^{3}\le 1/(27L^{3})$, thus
\[
4L^{3}\eta^{4}=4L^{3}\eta^{3}\eta\le \frac{4}{27}\eta<\frac{\eta}{4}.
\]
Hence
\[
\frac{\eta}{2}-4L^{3}\eta^{4}\ge\frac{\eta}{4}. \tag{14}
\]

\noindent\textbf{[Step 13]} (Simplify (13) using (14)).  
\[
\E[f(x_m)]
\le f(\tilde{x})
   -\frac{\eta}{4}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t)\|^{2}]
   +\big(4mL^{3}\eta^{4}+mL\eta^{2}\big)\sigma^{2}. \tag{15}
\]

\noindent\textbf{[Step 14]} (Epoch‑level recursion).  
Define $F^{(s)}:=\E[f(\tilde{x}^{(s)})]$. Taking expectation over the randomness of epoch $s$ and using that $\tilde{x}^{(s+1)}=x_m^{(s)}$, (15) becomes
\[
F^{(s+1)}\le F^{(s)}-\frac{\eta}{4}\sum_{t=0}^{m-1}\E[\|\nabla f(x_t^{(s)})\|^{2}]
   +\big(4mL^{3}\eta^{4}+mL\eta^{2}\big)\sigma^{2}. \tag{16}
\]

\noindent\textbf{[Step 15]} (Sum over $S$ epochs).  
Summing (16) for $s=0,\dots,S-1$ and telescoping $F^{(s)}$ gives
\[
F^{(S)}\le F^{(0)}
   -\frac{\eta}{4}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
       \E[\|\nabla f(x_t^{(s)})\|^{2}]
   +S\big(4mL^{3}\eta^{4}+mL\eta^{2}\big)\sigma^{2}. \tag{17}
\]

\noindent\textbf{[Step 16]} (Use lower bound $f^{\star}\le F^{(S)}$).  
Rearranging (17) yields
\[
\frac{1}{Sm}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
   \E[\|\nabla f(x_t^{(s)})\|^{2}]
\le\frac{4\big(F^{(0)}-f^{\star}\big)}{\eta Sm}
   +4L^{3}\eta^{3}\sigma^{2}
   +L\eta\sigma^{2}. \tag{18}
\]

\noindent\textbf{[Step 17]} (Define total iteration count $T=Sm$ and bound the constants).  
Since $\eta\le 1/(3L)$, the term $4L^{3}\eta^{3}\le \frac{4}{27}L$, which can be absorbed into the $L\eta$ term. Hence
\[
\frac{1}{T}\sum_{k=0}^{T-1}
   \E[\|\nabla f(x_k)\|^{2}]
\le\frac{4\big(f(x^{0})-f^{\star}\big)}{\eta T}
   +\frac{L\eta\sigma^{2}}{2}. \tag{19}
\]

\noindent\textbf{[Step 18]} (Random output).  
Select $\hat{x}$ uniformly among the $T$ iterates; then
\[
\E[\|\nabla f(\hat{x})\|^{2}]
= \frac{1}{T}\sum_{k=0}^{T-1}\E[\|\nabla f(x_k)\|^{2}]
\le \frac{2\big(f(x^{0})-f^{\star}\big)}{\eta T}
   +\frac{L\eta\sigma^{2}}{2},
\]
which is exactly the bound (1) claimed in Theorem~\ref{thm:svrg}.  ∎

\section*{Rate Derivation (With Constants)}
From (19) we have the deterministic‑plus‑variance decomposition
\[
\E[\|\nabla f(\hat{x})\|^{2}]
\le \underbrace{\frac{2\Delta}{\eta T}}_{\text{optimization error}}
   \;+\;
   \underbrace{\frac{L\eta\sigma^{2}}{2}}_{\text{variance error}},
\quad\Delta:=f(x^{0})-f^{\star}.
\]
Minimising the right‑hand side w.r.t. $\eta$ (ignoring the feasibility constraint) gives
\[
\eta^{\star} = \sqrt{\frac{2\Delta}{L\sigma^{2}T}}.
\]
If $\eta^{\star}\le 1/(3L)$ (i.e. $T\ge 18\Delta/(\sigma^{2})$) we may set $\eta=\eta^{\star}$ and obtain
\[
\E[\|\nabla f(\hat{x})\|^{2}]
\le 2\sqrt{\frac{L\Delta\sigma^{2}}{2T}}
   =\mathcal{O}\!\big(T^{-1/2}\big).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Zero variance ($\sigma^{2}=0$).} The bound reduces to
\[
\E[\|\nabla f(\hat{x})\|^{2}]\le \frac{2\Delta}{\eta T},
\]
and with a constant stepsize $\eta=1/(3L)$ we recover the deterministic gradient‑descent rate $\mathcal{O}(1/T)$.
\item \textbf{Infinite batch ($n\to\infty$).} When the finite‑sum is replaced by an expectation, the full gradient $\tilde{\mu}$ can be computed only approximately. The same analysis holds with an additional bias term that vanishes as the batch size grows.
\item \textbf{Convex $L$‑smooth case.} Under convexity, the same derivation yields a faster $\mathcal{O}(1/T)$ rate for the function suboptimality $f(\hat{x})-f^{\star}$ (see standard SVRG literature).
\end{itemize}
