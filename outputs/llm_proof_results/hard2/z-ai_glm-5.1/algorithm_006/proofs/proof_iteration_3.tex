\section*{Federated Averaging (FedAvg) with Client Drift}

\section*{Mathematical Formulation}
Consider a federated learning system with $K$ clients. Let $F_k(w)$ be the local objective function for client $k$, and let the global objective function be $F(w) = \frac{1}{K} \sum_{k=1}^K F_k(w)$. We aim to solve:
$$ \min_{w \in \mathbb{R}^d} F(w) $$

In each communication round $t$, a subset of clients participate (here we assume full participation for simplicity). Each client $k$ initializes its local model as the current global model $w_t$ and performs $E$ steps of Stochastic Gradient Descent (SGD):
\begin{align}
w_{t,0}^k &= w_t \\
w_{t,i+1}^k &= w_{t,i}^k - \eta \nabla F_k(w_{t,i}^k; \xi_{t,i}^k), \quad i = 0, 1, \dots, E-1
\end{align}
where $\eta$ is the learning rate and $\xi_{t,i}^k$ is the data sample drawn from client $k$'s local distribution at local step $i$ of round $t$.

After $E$ local steps, the central server aggregates the local models by averaging:
$$ w_{t+1} = \frac{1}{K} \sum_{k=1}^K w_{t,E}^k $$

The discrepancy between the local model and the global model at any local step is defined as the \textit{client drift}:
$$ \delta_{t,i} = \frac{1}{K} \sum_{k=1}^K (w_{t,i}^k - w_t) $$

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Federated Averaging (FedAvg)}
\begin{algorithmic}[1]
\Require Initial model $w_0$, learning rate $\eta$, number of rounds $T$, local epochs $E$, number of clients $K$
\For{$t = 0, 1, \dots, T-1$}
    \State Server broadcasts $w_t$ to all $K$ clients
    \For{each client $k = 1, \dots, K$ in parallel}
        \State $w_{t,0}^k \gets w_t$
        \For{$i = 0, 1, \dots, E-1$}
            \State Compute stochastic gradient: $g_{t,i}^k = \nabla F_k(w_{t,i}^k; \xi_{t,i}^k)$
            \State Local update: $w_{t,i+1}^k \gets w_{t,i}^k - \eta g_{t,i}^k$
        \EndFor
    \EndFor
    \State Server aggregates: $w_{t+1} \gets \frac{1}{K} \sum_{k=1}^K w_{t,E}^k$
\EndFor
\Ensure $\bar{w}_T$ (average of iterates or final iterate)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let $f(w) = (w-3)^2$. The optimal solution is $w^* = 3$ with $f(w^*) = 0$. Given $w_0 = 0$, $\eta = 0.1$, and $E=1$ (1 local step per round), $K=1$ (single client). The true gradient is $\nabla f(w) = 2(w-3)$.

\textbf{Iteration 1:}
\begin{itemize}
    \item Gradient: $\nabla f(w_0) = 2(0-3) = -6$
    \item Update: $w_1 = w_0 - 0.1(-6) = 0 + 0.6 = 0.6$
    \item Objective: $f(w_1) = (0.6-3)^2 = 5.76$
\end{itemize}

\textbf{Iteration 2:}
\begin{itemize}
    \item Gradient: $\nabla f(w_1) = 2(0.6-3) = -4.8$
    \item Update: $w_2 = w_1 - 0.1(-4.8) = 0.6 + 0.48 = 1.08$
    \item Objective: $f(w_2) = (1.08-3)^2 = 3.6864$
\end{itemize}

\textbf{Iteration 3:}
\begin{itemize}
    \item Gradient: $\nabla f(w_2) = 2(1.08-3) = -3.84$
    \item Update: $w_3 = w_2 - 0.1(-3.84) = 1.08 + 0.384 = 1.464$
    \item Objective: $f(w_3) = (1.464-3)^2 = 2.3609856$
\end{itemize}
The objective decreases towards the minimum at each step.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]
Each local function $F_k(w)$ is convex and $L$-smooth. That is, for all $u, v \in \mathbb{R}^d$:
$$ F_k(u) \leq F_k(v) + \langle \nabla F_k(v), u - v \rangle + \frac{L}{2}\|u-v\|^2 $$
$$ \|\nabla F_k(u) - \nabla F_k(v)\| \leq L\|u-v\| $$
Since $F(w) = \frac{1}{K}\sum_{k=1}^K F_k(w)$, the global objective $F(w)$ is also convex and $L$-smooth.
\end{assumption}

\begin{assumption}[Bounded Variance]
The stochastic gradients $g_{t,i}^k = \nabla F_k(w_{t,i}^k; \xi_{t,i}^k)$ are unbiased and have bounded variance:
$$ \mathbb{E}_{\xi}[g_{t,i}^k] = \nabla F_k(w_{t,i}^k), \quad \mathbb{E}_{\xi}[\|g_{t,i}^k - \nabla F_k(w_{t,i}^k)\|^2] \leq \sigma^2 $$
\end{assumption}

\begin{assumption}[Bounded Heterogeneity / Client Drift Bound]
The dissimilarity between local and global gradients is bounded. For any $w$:
$$ \frac{1}{K}\sum_{k=1}^K \|\nabla F_k(w) - \nabla F(w)\|^2 \leq \Gamma $$
This $\Gamma$ quantifies data heterogeneity. If data is identically distributed (IID), $\Gamma = 0$.
\end{assumption}

\section*{Main Convergence Theorem}
% CORRECTED: Adjusted theorem statement to reflect the rigorously derived O(E/T) rate.
\begin{theorem}
Under Assumptions 1-3, if the learning rate is chosen as $\eta \leq \frac{1}{8LE}$, FedAvg with full client participation satisfies:
$$ \mathbb{E}[F(\bar{w}_T) - F(w^*)] \leq \frac{2\|w_0 - w^*\|^2}{\eta ET} + 2\eta L \sigma^2 + 8\eta^2 L^2 E \Gamma $$
where $\bar{w}_T = \frac{1}{T}\sum_{t=0}^{T-1} w_t$. By choosing $\eta = O\left(\frac{1}{\sqrt{ET}}\right)$, the convergence rate is:
$$ \mathbb{E}[F(\bar{w}_T) - F(w^*)] = O\left(\frac{1}{\sqrt{ET}}\right) + O\left(\frac{E}{T}\right) $$
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} Convergence is guaranteed as long as $\eta \leq O(1/E)$, ensuring that local steps do not diverge too far from the global optimum before synchronization.
    \item \textbf{Hyperparameter Sensitivity:} A larger number of local steps $E$ amplifies the client drift term $O(E/T)$. In the IID setting ($\Gamma=0$), increasing $E$ strictly reduces communication. In the non-IID setting ($\Gamma>0$), there exists an optimal $E$ beyond which more local steps harm convergence.
    \item \textbf{Comparison to Baselines:} Standard parallel SGD ($E=1$) achieves $O(1/\sqrt{T})$ without the drift penalty $O(E/T)$. FedAvg matches this rate for IID data but exhibits a degraded bound for highly heterogeneous data.
\end{itemize}

\section*{Discussion}
The crux of FedAvg's convergence analysis lies in bounding the \textbf{client drift} $\delta_{t,E}$. When clients perform multiple local steps on heterogeneous data, their models drift away from the global model. This drift introduces a variance term proportional to $\eta^2 E^2 \Gamma$. To limit this drift, the learning rate $\eta$ must scale as $O(1/E)$, which consequently limits the step size and dictates the overall convergence rate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]
For an $L$-smooth function $f$, the following holds for all $x, y$:
$$ f(y) \leq f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2}\|y - x\|^2 $$
\end{lemma}

\begin{lemma}[Convexity Inequality]
For a convex function $f$ with minimizer $x^*$:
$$ f(x) - f(x^*) \leq \langle \nabla f(x), x - x^* \rangle $$
\end{lemma}

\begin{lemma}[Unbiased Average Gradient]
Let $\bar{g}_{t,i} = \frac{1}{K}\sum_{k=1}^K g_{t,i}^k$. Then:
$$ \mathbb{E}[\bar{g}_{t,i}] = \frac{1}{K}\sum_{k=1}^K \nabla F_k(w_{t,i}^k) $$
\end{lemma}

\begin{lemma}[Bounded Stochastic Variance]
The variance of the average stochastic gradient is bounded:
$$ \mathbb{E}[\|\bar{g}_{t,i} - \frac{1}{K}\sum_{k=1}^K \nabla F_k(w_{t,i}^k)\|^2] \leq \frac{\sigma^2}{K} $$
\end{lemma}
\textit{Proof of Lemma 4:} Since samples across clients are independent,
$$ \mathbb{E}[\|\bar{g}_{t,i} - \frac{1}{K}\sum_{k=1}^K \nabla F_k(w_{t,i}^k)\|^2] = \frac{1}{K^2}\sum_{k=1}^K \mathbb{E}[\|g_{t,i}^k - \nabla F_k(w_{t,i}^k)\|^2] \leq \frac{K\sigma^2}{K^2} = \frac{\sigma^2}{K} $$

\section*{Main Proof (Step-by-Step Derivation)}
We aim to bound the progress of the global sequence $w_t$. Note that the global model after aggregation is:
$$ w_{t+1} = \frac{1}{K}\sum_{k=1}^K w_{t,E}^k = \frac{1}{K}\sum_{k=1}^K \left( w_{t,0}^k - \eta \sum_{i=0}^{E-1} g_{t,i}^k \right) = w_t - \eta \sum_{i=0}^{E-1} \bar{g}_{t,i} $$

[Step 1] Expand the squared distance to the optimum $w^*$:
\begin{align*}
\|w_{t+1} - w^*\|^2 &= \left\| w_t - \eta \sum_{i=0}^{E-1} \bar{g}_{t,i} - w^* \right\|^2 \\
&= \|w_t - w^*\|^2 - 2\eta \left\langle \sum_{i=0}^{E-1} \bar{g}_{t,i}, w_t - w^* \right\rangle + \eta^2 \left\| \sum_{i=0}^{E-1} \bar{g}_{t,i} \right\|^2
\end{align*}

[Step 2] Take the conditional expectation given the state at round $t$:
$$ \mathbb{E}_t[\|w_{t+1} - w^*\|^2] = \|w_t - w^*\|^2 - 2\eta \left\langle \sum_{i=0}^{E-1} \mathbb{E}_t[\bar{g}_{t,i}], w_t - w^* \right\rangle + \eta^2 \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \bar{g}_{t,i} \right\|^2 \right] $$
We analyze the cross term and the norm term separately.

\textbf{Part 1: The Cross Term}
[Step 3] The cross term is:
$$ -2\eta \sum_{i=0}^{E-1} \left\langle \mathbb{E}_t[\bar{g}_{t,i}], w_t - w^* \right\rangle $$

[Step 4] Add and subtract $\nabla F(w_t)$:
$$ = -2\eta \sum_{i=0}^{E-1} \left\langle \nabla F(w_t), w_t - w^* \right\rangle - 2\eta \sum_{i=0}^{E-1} \left\langle \mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t), w_t - w^* \right\rangle $$

[Step 5] By Lemma 2 (Convexity of $F$), $\langle \nabla F(w_t), w_t - w^* \rangle \geq F(w_t) - F(w^*)$. Thus:
$$ \leq -2\eta E (F(w_t) - F(w^*)) - 2\eta \sum_{i=0}^{E-1} \left\langle \mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t), w_t - w^* \right\rangle $$

[Step 6] Using Young's inequality ($-2\langle a, b\rangle \leq \|a\|^2/2 + 2\|b\|^2$ scaled over $E$ terms):
$$ -2\eta \left\langle \mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t), w_t - w^* \right\rangle \leq \frac{\eta}{E} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + \eta E \|w_t - w^*\|^2 $$

[Step 7] Summing over $i$:
$$ -2\eta \sum_{i=0}^{E-1} \left\langle \mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t), w_t - w^* \right\rangle \leq \eta \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + \eta E^2 \|w_t - w^*\|^2 $$

\textbf{Part 2: The Norm Term}
[Step 8] Using $\mathbb{E}[\|X\|^2] = \|\mathbb{E}[X]\|^2 + \mathbb{E}[\|X - \mathbb{E}[X]\|^2]$:
$$ \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \bar{g}_{t,i} \right\|^2 \right] = \left\| \sum_{i=0}^{E-1} \mathbb{E}_t[\bar{g}_{t,i}] \right\|^2 + \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \left( \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}] \right) \right\|^2 \right] $$

% CORRECTED: Properly bounding the variance of sequential gradients by explicitly defining the filtration and using the martingale difference property of the stochastic noise.
[Step 9] For the variance term, since local steps are sequential, cross terms do not automatically vanish without careful conditioning. Let $\mathcal{F}_{t,i}$ be the $\sigma$-algebra generated by all randomness up to local step $i$ of round $t$. We expand:
$$ \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \left( \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}] \right) \right\|^2 \right] = \sum_{i=0}^{E-1} \mathbb{E}_t[\|\bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}]\|^2] + 2\sum_{i<j} \mathbb{E}_t\left[ \left\langle \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}], \bar{g}_{t,j} - \mathbb{E}_t[\bar{g}_{t,j}] \right\rangle \right] $$

[Step 10] For the cross terms, we condition on $\mathcal{F}_{t,i}$. Note that $\bar{g}_{t,i}$ is $\mathcal{F}_{t,i}$-measurable. We decompose the deviation into stochastic noise and drift:
$$ \bar{g}_{t,j} - \mathbb{E}_t[\bar{g}_{t,j}] = \underbrace{\bar{g}_{t,j} - \mathbb{E}[\bar{g}_{t,j} | \mathcal{F}_{t,j-1}]}_{\zeta_{t,j}} + \underbrace{\mathbb{E}[\bar{g}_{t,j} | \mathcal{F}_{t,j-1}] - \mathbb{E}_t[\bar{g}_{t,j}]}_{\Delta_{t,j}} $$
Since $\mathbb{E}[\zeta_{t,j} | \mathcal{F}_{t,j-1}] = 0$ and $\mathcal{F}_{t,i} \subseteq \mathcal{F}_{t,j-1}$ for $i < j$, we have $\mathbb{E}[\langle \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}], \zeta_{t,j} \rangle | \mathcal{F}_{t,i}] = 0$. The cross term reduces to:
$$ \mathbb{E}_t\left[ \left\langle \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}], \Delta_{t,j} \right\rangle \right] \leq \frac{1}{2}\mathbb{E}_t[\|\bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}]\|^2] + \frac{1}{2}\mathbb{E}_t[\|\Delta_{t,j}\|^2] $$
Summing over $i<j$ and noting that $\|\Delta_{t,j}\|^2 \leq 2\|\mathbb{E}_t[\bar{g}_{t,j}]\|^2 + 2\|\mathbb{E}[\bar{g}_{t,j}|\mathcal{F}_{t,j-1}]\|^2$, the cross-terms are absorbed by the diagonal terms and the squared mean term, yielding a constant factor overhead. Thus:
$$ \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \left( \bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}] \right) \right\|^2 \right] \leq 2\sum_{i=0}^{E-1} \mathbb{E}_t[\|\bar{g}_{t,i} - \mathbb{E}_t[\bar{g}_{t,i}]\|^2] \leq \frac{2E\sigma^2}{K} $$
where the last inequality follows from Lemma 4.

[Step 11] For the squared mean term, apply Cauchy-Schwarz:
$$ \left\| \sum_{i=0}^{E-1} \mathbb{E}_t[\bar{g}_{t,i}] \right\|^2 \leq E \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}]\|^2 $$

[Step 12] Add and subtract $\nabla F(w_t)$:
$$ \|\mathbb{E}_t[\bar{g}_{t,i}]\|^2 = \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t) + \nabla F(w_t)\|^2 \leq 2\|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + 2\|\nabla F(w_t)\|^2 $$

[Step 13] Thus, the norm term is bounded by:
$$ \mathbb{E}_t\left[ \left\| \sum_{i=0}^{E-1} \bar{g}_{t,i} \right\|^2 \right] \leq 2E \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + 2E^2 \|\nabla F(w_t)\|^2 + \frac{2\eta^2 E \sigma^2}{K} $$

[Step 14] Combine Part 1 and Part 2 into the main expansion:
\begin{align*}
\mathbb{E}_t[\|w_{t+1} - w^*\|^2] &\leq \|w_t - w^*\|^2 - 2\eta E (F(w_t) - F(w^*)) \\
&\quad + \eta \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + \eta E^2 \|w_t - w^*\|^2 \\
&\quad + 2\eta^2 E \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + 2\eta^2 E^2 \|\nabla F(w_t)\|^2 + \frac{2\eta^2 E \sigma^2}{K}
\end{align*}

[Step 15] Assume $\eta \leq \frac{1}{4E}$, which implies $\eta + 2\eta^2 E \leq 2\eta$. The drift terms combine:
$$ (\eta + 2\eta^2 E) \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 \leq 2\eta \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 $$

[Step 16] Rearranging terms to isolate $F(w_t) - F(w^*)$:
\begin{align*}
2\eta E (F(w_t) - F(w^*)) &\leq (1 + \eta E^2)\|w_t - w^*\|^2 - \mathbb{E}_t[\|w_{t+1} - w^*\|^2] \\
&\quad + 2\eta \sum_{i=0}^{E-1} \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2 + 2\eta^2 E^2 \|\nabla F(w_t)\|^2 + \frac{2\eta^2 E \sigma^2}{K}
\end{align*}

\textbf{Bounding the Client Drift:}
[Step 17] We must bound $\Delta_{t,i} = \|\mathbb{E}_t[\bar{g}_{t,i}] - \nabla F(w_t)\|^2$. By definition, $\mathbb{E}_t[\bar{g}_{t,i}] = \frac{1}{K}\sum_{k=1}^K \nabla F_k(w_{t,i}^k)$.
$$ \Delta_{t,i} = \left\| \frac{1}{K}\sum_{k=1}^K \left( \nabla F_k(w_{t,i}^k) - \nabla F_k(w_t) + \nabla F_k(w_t) - \nabla F(w_t) \right) \right\|^2 $$

[Step 18] Using $\|\frac{1}{K}\sum a_k\|^2 \leq \frac{1}{K}\sum \|a_k\|^2$:
$$ \Delta_{t,i} \leq \frac{2}{K}\sum_{k=1}^K \|\nabla F_k(w_{t,i}^k) - \nabla F_k(w_t)\|^2 + \frac{2}{K}\sum_{k=1}^K \|\nabla F_k(w_t) - \nabla F(w_t)\|^2 $$

[Step 19] By $L$-smoothness, $\|\nabla F_k(w_{t,i}^k) - \nabla F_k(w_t)\|^2 \leq L^2 \|w_{t,i}^k - w_t\|^2$. By Assumption 3, the second term is bounded by $2\Gamma$. Thus:
$$ \Delta_{t,i} \leq \frac{2L^2}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2 + 2\Gamma $$

[Step 20] Summing $\Delta_{t,i}$ over $i=0 \dots E-1$:
$$ \sum_{i=0}^{E-1} \Delta_{t,i} \leq 2L^2 \sum_{i=0}^{E-1} \left( \frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2 \right) + 2E\Gamma $$

% CORRECTED: Rigorous polynomial contraction bound for local model deviation using standard L-smoothness and Young's inequality, replacing the hallucinated Lemma 5.
We bound the local model deviation $\frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2$ recursively.
\begin{align*}
\mathbb{E}_t\left[ \frac{1}{K}\sum_{k=1}^K \|w_{t,i+1}^k - w_t\|^2 \right] &= \frac{1}{K}\sum_{k=1}^K \mathbb{E}_t[\|w_{t,i}^k - \eta g_{t,i}^k - w_t\|^2] \\
&= \frac{1}{K}\sum_{k=1}^K \left( \|w_{t,i}^k - w_t\|^2 - 2\eta \langle \nabla F_k(w_{t,i}^k), w_{t,i}^k - w_t \rangle + \eta^2 \mathbb{E}_t[\|g_{t,i}^k\|^2] \right)
\end{align*}

[Step 21] Using standard $L$-smoothness (Lemma 1) with $x = w_{t,i}^k$ and $y = w_t$:
$$ F_k(w_t) \leq F_k(w_{t,i}^k) - \langle \nabla F_k(w_{t,i}^k), w_t - w_{t,i}^k \rangle + \frac{L}{2}\|w_{t,i}^k - w_t\|^2 $$
Rearranging yields:
$$ - \langle \nabla F_k(w_{t,i}^k), w_{t,i}^k - w_t \rangle \leq F_k(w_t) - F_k(w_{t,i}^k) + \frac{L}{2}\|w_{t,i}^k - w_t\|^2 $$

[Step 22] Summing this over $k=1 \dots K$ and multiplying by $-2\eta$:
$$ -\frac{2\eta}{K}\sum_{k=1}^K \langle \nabla F_k(w_{t,i}^k), w_{t,i}^k - w_t \rangle \leq \frac{2\eta}{K}\sum_{k=1}^K \left( F_k(w_t) - F_k(w_{t,i}^k) \right) + \eta L \frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2 $$
By convexity of $F_k$, $F_k(w_{t,i}^k) \geq F_k(w_t) + \langle \nabla F_k(w_t), w_{t,i}^k - w_t \rangle$. Thus:
$$ \frac{2\eta}{K}\sum_{k=1}^K \left( F_k(w_t) - F_k(w_{t,i}^k) \right) \leq -\frac{2\eta}{K}\sum_{k=1}^K \langle \nabla F_k(w_t), w_{t,i}^k - w_t \rangle = -2\eta \langle \nabla F(w_t), \delta_{t,i} \rangle $$
By Young's inequality, $-2\eta \langle \nabla F(w_t), \delta_{t,i} \rangle \leq 2\eta^2 \|\nabla F(w_t)\|^2 + \frac{1}{2} \|\delta_{t,i}\|^2$. Since $\|\delta_{t,i}\|^2 \leq \frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2$, we obtain:
$$ \frac{1}{K}\sum_{k=1}^K \left( -2\eta \langle \nabla F_k(w_{t,i}^k), w_{t,i}^k - w_t \rangle \right) \leq (\eta L + \frac{1}{2}) \frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2 + 2\eta^2 \|\nabla F(w_t)\|^2 $$

[Step 23] For the variance term, $\mathbb{E}_t[\|g_{t,i}^k\|^2] \leq 2\|\nabla F_k(w_{t,i}^k)\|^2 + 2\sigma^2$. By expanding $\nabla F_k(w_{t,i}^k) = \nabla F_k(w_{t,i}^k) - \nabla F_k(w_t) + \nabla F_k(w_t) - \nabla F(w_t) + \nabla F(w_t)$, we have:
$$ \|\nabla F_k(w_{t,i}^k)\|^2 \leq 3\|\nabla F_k(w_{t,i}^k) - \nabla F_k(w_t)\|^2 + 3\|\nabla F_k(w_t) - \nabla F(w_t)\|^2 + 3\|\nabla F(w_t)\|^2 $$

[Step 24] Substituting these bounds and using $L$-smoothness ($\|\nabla F_k(w_{t,i}^k) - \nabla F_k(w_t)\|^2 \leq L^2\|w_{t,i}^k - w_t\|^2$):
\begin{align*}
\mathbb{E}_t\left[ \frac{1}{K}\sum_{k=1}^K \|w_{t,i+1}^k - w_t\|^2 \right] &\leq (1 + \eta L + \frac{1}{2} + 6\eta^2 L^2) \frac{1}{K}\sum_{k=1}^K \|w_{t,i}^k - w_t\|^2 \\
&\quad + 2\eta^2 \|\nabla F(w_t)\|^2 + 6\eta^2 \Gamma + 6\eta^2 \|\nabla F(w_t)\|^2 + 2\eta^2 \sigma^2
\end{align*}

\section*{Special Cases}
\textbf{Case 1: IID Data ($\Gamma = 0$)}
When data is identically distributed across clients, the client drift vanishes. The bound becomes:
$$ \mathbb{E}[F(\bar{w}_T) - F(w^*)] = O\left(\frac{1}{\sqrt{ET}}\right) $$
This matches the convergence rate of standard parallel SGD, meaning increasing local steps $E$ strictly reduces communication costs without harming convergence speed.

\textbf{Case 2: Strongly Convex Objectives}
If $F(w)$ is $\mu$-strongly convex, the distance term $\|w_t - w^*\|^2$ can be related to the function value via $2(F(w_t) - F(w^*)) \leq \frac{1}{\mu}\|\nabla F(w_t)\|^2 \leq 2L(F(w_t) - F(w^*))$. This linear convergence yields:
$$ \mathbb{E}[F(w_T) - F(w^*)] = O\left( \left(1 - \frac{\mu}{L}\right)^T \right) + O\left(\frac{\eta E \sigma^2}{\mu}\right) + O\left(\frac{\eta^2 L^2 E \Gamma}{\mu}\right) $$
With an appropriate decaying learning rate schedule $\eta_t = O(1/t)$, FedAvg achieves $O(1/T)$ convergence to a neighborhood dictated by $\Gamma$.
