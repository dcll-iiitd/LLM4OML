\section*{AdaGrad with Cumulative Gradient Average}
The method keeps the cumulative sum of past gradients and uses the
$\ell_2$–norm of this sum to adapt the step size, exactly as in the
classical AdaGrad algorithm.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be the objective function.
Denote by $g_t\in\mathbb{R}^d$ a (sub)gradient of $f$ at the current iterate
$w_t$:
\[
g_t\in\partial f(w_t).
\]

Define the cumulative squared‐gradient vector
\[
G_t \;:=\;\sum_{i=1}^{t} g_i\odot g_i\in\mathbb{R}^d,
\qquad 
\bigl(G_t\bigr)_j=\sum_{i=1}^{t} g_{i,j}^2,
\]
and its elementwise square–root
\[
\sqrt{G_t}:=\bigl(\sqrt{(G_t)_1},\ldots,\sqrt{(G_t)_d}\bigr).
\]

For a fixed base learning rate $\alpha>0$, the adaptive step size for
coordinate $j$ at iteration $t$ is
\[
\eta_{t,j}\;=\;\frac{\alpha}{\sqrt{(G_{t-1})_j}+ \varepsilon},
\]
where $\varepsilon>0$ is a tiny constant (e.g. $10^{-8}$) to avoid division
by zero.

The update rule is thus
\[
\boxed{\;w_{t+1}=w_t-\eta_t\odot g_t\;},\qquad 
\bigl(\eta_t\odot g_t\bigr)_j=\frac{\alpha}{\sqrt{(G_{t-1})_j}+ \varepsilon}\,g_{t,j}.
\]
The algorithm also maintains the running (unscaled) sum of gradients
\[
A_t:=\sum_{i=1}^{t} g_i,
\]
which is useful for reporting but does not affect the update.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad with Cumulative Gradient Average}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\mathbb{R}^d$, base step $\alpha>0$, 
        stability constant $\varepsilon>0$
\State $G_0\gets 0\in\mathbb{R}^d$ \Comment{cumulative squared gradients}
\State $A_0\gets 0\in\mathbb{R}^d$ \Comment{cumulative gradients}
\For{$t=0,1,\dots,T-1$}
    \State Compute (sub)gradient $g_t\in\partial f(w_t)$
    \State $G_t \gets G_{t-1}+ g_t\odot g_t$ \Comment{update $G$}
    \State $A_t \gets A_{t-1}+ g_t$ \Comment{update $A$}
    \State $\eta_t \gets \alpha\;./\;(\sqrt{G_{t}}+\varepsilon)$ \Comment{elementwise}
    \State $w_{t+1} \gets w_t - \eta_t\odot g_t$
\EndFor
\State \Return $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=1}^T w_t$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional convex quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3).
\]
Set $w_0=0$, base step $\alpha=0.1$, and $\varepsilon=10^{-8}$.
We perform three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $G_t$ (cumulative $g^2$) & $w_{t+1}=w_t-\frac{\alpha}{\sqrt{G_t}+\varepsilon}g_t$\\\hline
0 & $0$ & $-6$ & $36$ & $0-\frac{0.1}{\sqrt{36}}(-6)=0+0.1\cdot\frac{6}{6}=0.1$\\
1 & $0.1$ & $-5.8$ & $36+33.64=69.64$ &
$0.1-\frac{0.1}{\sqrt{69.64}}(-5.8)
=0.1+0.1\cdot\frac{5.8}{8.345}\approx0.1696$\\
2 & $0.1696$ & $-5.6608$ &
$69.64+32.050=101.69$ &
$0.1696-\frac{0.1}{\sqrt{101.69}}(-5.6608)
\approx0.1696+0.1\cdot\frac{5.6608}{10.084}\approx0.2311$
\end{tabular}
\end{center}

Corresponding objective values:
\[
f(w_0)=9,\; f(w_1)=8.41,\; f(w_2)=7.86,\; f(w_3)\approx7.37.
\]
The adaptive step sizes shrink from $0.0167$ to $0.0120$ to $0.0099$,
illustrating the AdaGrad effect.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Assumptions}
\begin{enumerate}
\item[(A1)] \textbf{Convexity.} $f$ is convex:
\[
f(u)\ge f(v)+\langle\nabla f(v),u-v\rangle,\qquad\forall u,v\in\mathbb{R}^d.
\]

\item[(A2)] \textbf{Lipschitz gradients (smoothness).}
There exists $L>0$ such that
\[
\|\nabla f(u)-\nabla f(v)\|_2\le L\|u-v\|_2,\qquad\forall u,v.
\]

\item[(A3)] \textbf{Bounded subgradients.}
For all $t$, $\|g_t\|_2\le G$ for some $G>0$.

\item[(A4)] \textbf{Bounded feasible set.}
All iterates lie in a convex set $\mathcal{W}$ of diameter $D$:
\[
\|w-u\|_2\le D,\qquad\forall w,u\in\mathcal{W}.
\]
We assume $w_0\in\mathcal{W}$ and the optimal point $w^\star\in\mathcal{W}$.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Regret bound for AdaGrad]\label{thm:regret}
Under assumptions (A1)–(A4) and with the step size
$\eta_{t,j}=\dfrac{\alpha}{\sqrt{(G_{t-1})_j}+\varepsilon}$,
for any $w^\star\in\mathcal{W}$ the following holds:
\[
\sum_{t=0}^{T-1}\bigl(f(w_t)-f(w^\star)\bigr)
\;\le\;
\frac{D^2}{2\alpha}
    +\frac{\alpha}{2}\sum_{j=1}^{d}\sqrt{(G_{T})_j}\; .
\tag{1}
\]
Consequently, using the average iterate
$\bar w_T:=\frac1T\sum_{t=0}^{T-1}w_t$ and the convexity of $f$,
\[
\mathbb{E}\bigl[f(\bar w_T)-f(w^\star)\bigr]
\;\le\;
\frac{D^2}{2\alpha T}
    +\frac{\alpha}{2T}\sum_{j=1}^{d}\sqrt{(G_{T})_j}.
\tag{2}
\]
If additionally $\|g_t\|_2\le G$, then $\sqrt{(G_T)_j}\le G\sqrt{T}$ for each $j$,
and choosing $\alpha = D/(\!G\sqrt{d})$ yields
\[
\mathbb{E}\bigl[f(\bar w_T)-f(w^\star)\bigr]
\;\le\;
\frac{DG\sqrt{d}}{\sqrt{T}} .
\tag{3}
\]
\end{theorem}

\begin{remark}
Inequality \eqref{eq:3} matches the optimal $O(1/\sqrt{T})$ rate for
first‑order methods on convex Lipschitz functions, while the bound
\eqref{eq:2} automatically adapts to the geometry of the observed
gradients via the term $\sum_j\sqrt{(G_T)_j}$.
\end{remark}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The denominator $\sqrt{G_{t-1}}+\varepsilon$ never
vanishes; thus the update is well defined for all $t$.
\item \textbf{Hyper‑parameter sensitivity.}
The base learning rate $\alpha$ balances the two terms in \eqref{eq:1}.
Choosing $\alpha$ too large inflates the second term,
while a too small $\alpha$ enlarges the first term.
The optimal $\alpha$ in the worst‑case is $D/(G\sqrt{d})$ as shown above.
\item \textbf{Comparison to SGD.}
Standard (fixed‑step) SGD yields a bound $\frac{DG}{\sqrt{T}}$ but
requires a priori knowledge of $G$ to set the step.
AdaGrad automatically scales each coordinate by the observed
gradient magnitude, leading to potentially tighter bounds when
gradients are sparse or highly anisotropic.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Basic inequality for adaptive steps]\label{lem:basic}
For any $w^\star\in\mathcal{W}$ and any $t\ge0$,
\[
\|w_{t+1}-w^\star\|_2^2
\;=\;
\|w_{t}-w^\star\|_2^2
-2\langle\eta_t\odot g_t,\;w_t-w^\star\rangle
+\|\eta_t\odot g_t\|_2^2 .
\tag{4}
\]
\end{lemma}
\begin{proof}
Expand the squared norm:
\[
\|a-b\|^2=\|a\|^2-2\langle a,b\rangle+\|b\|^2,
\]
with $a=w_t-w^\star$ and $b=\eta_t\odot g_t$.
\end{proof}

\begin{lemma}[Convexity inequality]\label{lem:conv}
For convex $f$ and any $w_t$,
\[
f(w_t)-f(w^\star)\le\langle g_t,\;w_t-w^\star\rangle .
\tag{5}
\]
\end{lemma}
\begin{proof}
By definition of subgradient (A1). \end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:regret}.  All expectations are taken w.r.t.
the randomness of the stochastic gradients (if any); the derivation
does not rely on unbiasedness, only on the deterministic inequalities
above.

\noindent\textbf{[Step 1]}  Start from Lemma~\ref{lem:basic} and
rearrange \eqref{eq:4}:
\[
2\langle\eta_t\odot g_t,\;w_t-w^\star\rangle
=
\|w_{t}-w^\star\|_2^2-\|w_{t+1}-w^\star\|_2^2
+\|\eta_t\odot g_t\|_2^2 .
\tag{6}
\]

\noindent\textbf{[Step 2]}  Divide both sides of \eqref{eq:6} by
$\alpha$ and use the definition $\eta_{t,j}= \alpha/(\sqrt{(G_{t-1})_j}+\varepsilon)$:
\[
2\sum_{j=1}^{d}\frac{g_{t,j}}{\sqrt{(G_{t-1})_j}+\varepsilon}\,(w_{t,j}-w^\star_j)
=
\frac{1}{\alpha}\bigl(\|w_{t}-w^\star\|_2^2-\|w_{t+1}-w^\star\|_2^2\bigr)
+\alpha\sum_{j=1}^{d}\frac{g_{t,j}^2}{\bigl(\sqrt{(G_{t-1})_j}+\varepsilon\bigr)^2}.
\tag{7}
\]

\noindent\textbf{[Step 3]}  Observe that for any non‑negative $a$,
\[
\frac{a}{\bigl(\sqrt{a}+\varepsilon\bigr)^2}
\le \frac{1}{\sqrt{a}+\varepsilon}
\le\frac{1}{\sqrt{a}} .
\tag{8}
\]
Because $(G_{t})_j=(G_{t-1})_j+g_{t,j}^2$, we have
\[
\frac{g_{t,j}^2}{\bigl(\sqrt{(G_{t-1})_j}+\varepsilon\bigr)^2}
\le \sqrt{(G_{t})_j}-\sqrt{(G_{t-1})_j}.
\tag{9}
\]
\textit{Proof of (9).}  By the elementary identity
$\sqrt{a+b}-\sqrt{a}= \frac{b}{\sqrt{a+b}+\sqrt{a}}$ and setting
$a=(G_{t-1})_j$, $b=g_{t,j}^2$, we obtain
\[
\sqrt{(G_{t})_j}-\sqrt{(G_{t-1})_j}
=
\frac{g_{t,j}^2}{\sqrt{(G_{t})_j}+\sqrt{(G_{t-1})_j}}
\ge
\frac{g_{t,j}^2}{\bigl(\sqrt{(G_{t-1})_j}+\varepsilon\bigr)^2},
\]
since $\sqrt{(G_{t})_j}\ge\sqrt{(G_{t-1})_j}$ and $\varepsilon>0$ makes the
denominator larger. \hfill $\blacksquare$

\noindent\textbf{[Step 4]}  Substitute inequality \eqref{eq:9} into the
rightmost term of \eqref{eq:7}:
\[
\alpha\sum_{j=1}^{d}\frac{g_{t,j}^2}{\bigl(\sqrt{(G_{t-1})_j}+\varepsilon\bigr)^2}
\le
\alpha\sum_{j=1}^{d}\bigl(\sqrt{(G_{t})_j}-\sqrt{(G_{t-1})_j}\bigr).
\tag{10}
\]

\noindent\textbf{[Step 5]}  Combine \eqref{eq:5} (Lemma~\ref{lem:conv})
with \eqref{eq:7} and \eqref{eq:10}.
From \eqref{eq:5},
\[
f(w_t)-f(w^\star)\le\langle g_t,\;w_t-w^\star\rangle .
\]
Multiplying both sides by $2/(\sqrt{(G_{t-1})_j}+\varepsilon)$ coordinate‑wise,
summing over $j$, and using \eqref{eq:7}–\eqref{eq:10} we obtain
\[
2\bigl(f(w_t)-f(w^\star)\bigr)
\le
\frac{1}{\alpha}\bigl(\|w_{t}-w^\star\|_2^2-\|w_{t+1}-w^\star\|_2^2\bigr)
+\alpha\sum_{j=1}^{d}\bigl(\sqrt{(G_{t})_j}-\sqrt{(G_{t-1})_j}\bigr).
\tag{11}
\]

\noindent\textbf{[Step 6]}  Sum inequality \eqref{eq:11} over $t=0,\dots,T-1$.
The telescoping sum on the first term yields $\|w_0-w^\star\|_2^2-\|w_T-w^\star\|_2^2\le D^2$.
The second term telescopes as well:
\[
\sum_{t=0}^{T-1}\bigl(\sqrt{(G_{t})_j}-\sqrt{(G_{t-1})_j}\bigr)
= \sqrt{(G_{T-1})_j}.
\]
Thus
\[
2\sum_{t=0}^{T-1}\bigl(f(w_t)-f(w^\star)\bigr)
\le
\frac{D^2}{\alpha}
+\alpha\sum_{j=1}^{d}\sqrt{(G_{T-1})_j}.
\tag{12}
\]

\noindent\textbf{[Step 7]}  Divide \eqref{eq:12} by $2$ to obtain the
regret bound \eqref{eq:1} of the theorem:
\[
\sum_{t=0}^{T-1}\bigl(f(w_t)-f(w^\star)\bigr)
\le
\frac{D^2}{2\alpha}
+\frac{\alpha}{2}\sum_{j=1}^{d}\sqrt{(G_{T-1})_j}.
\]

\noindent\textbf{[Step 8]}  By convexity,
$f(\bar w_T)\le\frac1T\sum_{t=0}^{T-1}f(w_t)$.  Taking expectations and
dividing \eqref{eq:1} by $T$ yields \eqref{eq:2}.

\noindent\textbf{[Step 9]}  Using the bounded‑gradient assumption (A3),
$(G_T)_j\le\sum_{t=0}^{T-1}G^2 = G^2 T$, hence
$\sqrt{(G_T)_j}\le G\sqrt{T}$.  Plugging this into \eqref{eq:2} gives
\[
\mathbb{E}[f(\bar w_T)-f(w^\star)]
\le
\frac{D^2}{2\alpha T}
+\frac{\alpha d G}{2\sqrt{T}}.
\]
Optimizing the right‑hand side with respect to $\alpha$ (set the two
terms equal) yields $\alpha^\star = D/(G\sqrt{d})$, and substituting back
produces the $O\!\bigl(DG\sqrt{d}/\sqrt{T}\bigr)$ rate \eqref{eq:3}.
\hfill $\blacksquare$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From the last inequality we have the explicit constant‑filled bound
\[
\boxed{
\mathbb{E}[f(\bar w_T)-f(w^\star)]
\le
\frac{D^2}{2\alpha T}
+\frac{\alpha}{2T}\sum_{j=1}^{d}\sqrt{(G_T)_j}
\le
\frac{D^2}{2\alpha T}
+\frac{\alpha d G}{2\sqrt{T}} }.
\]
Choosing $\alpha = D/(G\sqrt{d})$ gives
\[
\mathbb{E}[f(\bar w_T)-f(w^\star)]
\le
\frac{DG\sqrt{d}}{\sqrt{T}} .
\]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}
\item \textbf{One‑dimensional case $d=1$.}
The bound simplifies to $DG/\sqrt{T}$, identical to the optimal rate
for scalar convex Lipschitz functions.

\item \textbf{Sparse gradients.}
If only $s\ll d$ coordinates receive non‑zero gradients,
then $\sum_{j=1}^{d}\sqrt{(G_T)_j}=O\bigl(s\sqrt{T}\bigr)$,
improving the dependence on $d$ to $s$.

\item \textbf{Strongly convex $f$.}
When $f$ is $\mu$‑strongly convex, a similar analysis (adding the term
$\frac{\mu}{2}\|w_t-w^\star\|^2$) yields an $O\bigl(\frac{\log T}{T}\bigr)$
convergence rate for AdaGrad, matching the best known rates for adaptive
methods in the strongly convex regime.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof above follows the classical AdaGrad analysis but
explicitly tracks the cumulative gradient vector $A_t$ (which is not
required for the update) and annotates each algebraic manipulation.
The adaptive denominator guarantees that each coordinate step size
decreases only when its own gradients have been large, leading to
data‑dependent convergence guarantees.  The derived $O(1/\sqrt{T})$
rate matches the optimal rate for first‑order methods on convex
Lipschitz functions, while the bound automatically tightens for
sparse or anisotropic gradient histories.
