\section*{Proximal Gradient Method (PGM)}

\section*{Mathematical Formulation}
Let 
\[
\min_{x\in\mathbb{R}^{d}} \; \Phi(x)\quad\text{with}\quad \Phi(x):=f(x)+g(x),
\]
where  

\begin{itemize}
  \item $f:\mathbb{R}^{d}\to\mathbb{R}$ is convex, differentiable and has $L$‑Lipschitz continuous gradient:
        \[
        \|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|\quad\forall x,y\in\mathbb{R}^{d}.
        \]
  \item $g:\mathbb{R}^{d}\to\mathbb{R}\cup\{+\infty\}$ is proper, closed and convex (possibly nonsmooth).
  \item The \emph{proximal operator} of $g$ with stepsize $\eta>0$ is
        \[
        \operatorname{prox}_{\eta g}(v):=\arg\min_{u\in\mathbb{R}^{d}}\Big\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
        \]
\end{itemize}
The proximal gradient iteration with a constant stepsize $\eta\in(0,1/L]$ reads
\begin{equation}
\boxed{x_{k+1}= \operatorname{prox}_{\eta g}\!\big(x_{k}-\eta \nabla f(x_{k})\big)}\qquad k=0,1,2,\dots
\label{eq:update}
\end{equation}
When $g\equiv0$, $\operatorname{prox}_{\eta g}$ is the identity and \eqref{eq:update} reduces to plain gradient descent.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\mathbb{R}^{d}$, stepsize $\eta\in(0,1/L]$, maximum iterations $K$
\For{$k=0$ \textbf{to} $K-1$}
    \State $v_{k}=x_{k}-\eta\,\nabla f(x_{k})$ \Comment{gradient step}
    \State $x_{k+1}= \operatorname{prox}_{\eta g}(v_{k})$ \Comment{proximal step}
\EndFor
\State \textbf{return} $x_{K}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem 
\[
f(w)=(w-3)^{2},\qquad g\equiv0,\qquad \eta=0.1,\qquad w_{0}=0.
\]
Since $g=0$, $\operatorname{prox}_{\eta g}(v)=v$.

\begin{enumerate}[label=\textbf{Iter \arabic*:}]
\item $k=0$  
\[
\nabla f(w_{0})=2(w_{0}-3)=-6,\qquad 
v_{0}=w_{0}-\eta\nabla f(w_{0})=0-0.1(-6)=0.6,
\]
\[
w_{1}=v_{0}=0.6,\qquad \Phi(w_{1})=f(w_{1})=(0.6-3)^{2}=5.76.
\]

\item $k=1$  
\[
\nabla f(w_{1})=2(0.6-3)=-4.8,\qquad 
v_{1}=0.6-0.1(-4.8)=1.08,
\]
\[
w_{2}=v_{1}=1.08,\qquad \Phi(w_{2})=(1.08-3)^{2}=3.6864.
\]

\item $k=2$  
\[
\nabla f(w_{2})=2(1.08-3)=-3.84,\qquad 
v_{2}=1.08-0.1(-3.84)=1.464,
\]
\[
w_{3}=v_{2}=1.464,\qquad \Phi(w_{3})=(1.464-3)^{2}=2.374.
\]
\end{enumerate}
The objective value decreases monotonically, illustrating the descent property of \eqref{eq:update}.

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*:}]
\item \label{A1} \textbf{Smoothness of $f$.} $f$ is convex and differentiable with $L$‑Lipschitz gradient:
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^{d}.
\]

\item \label{A2} \textbf{Convexity of $g$.} $g$ is proper, closed and convex; its proximal operator $\operatorname{prox}_{\eta g}$ is well defined for any $\eta>0$.

\item \label{A3} \textbf{Existence of a minimizer.} The set 
\[
\mathcal{X}^{*}:=\arg\min_{x}\Phi(x)
\]
is non‑empty; pick any $x^{*}\in\mathcal{X}^{*}$.

\item \label{A4} \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\eta\le \frac{1}{L}.
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:rate}
Under Assumptions \ref{A1}–\ref{A4}, the iterates generated by \eqref{eq:update} satisfy for all $k\ge 1$
\[
\Phi(x_{k})-\Phi(x^{*})\;\le\;\frac{\|x_{0}-x^{*}\|^{2}}{2\eta\,k}.
\]
Consequently, $\Phi(x_{k})\to\Phi(x^{*})$ at the rate $O(1/k)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} The non‑expansiveness of the proximal operator guarantees that the sequence $\{x_{k}\}$ never diverges provided $\eta\le1/L$.
  \item \textbf{Hyperparameter Sensitivity.} The bound in Theorem \ref{thm:rate} is monotone decreasing in $\eta$; the largest admissible stepsize $1/L$ yields the smallest constant $\frac{1}{2\eta}$.
  \item \textbf{Comparison to Gradient Descent.} When $g\equiv0$, Theorem \ref{thm:rate} reduces to the classical $O(1/k)$ rate for gradient descent on smooth convex functions.
\end{itemize}

\section*{Discussion}
The proof relies on two elementary facts:
\begin{enumerate}
  \item \textbf{Descent Lemma} for $f$ (smoothness).
  \item \textbf{Non‑expansiveness} of the proximal operator:
        \[
        \|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\|\le\|u-v\|,
        \quad\forall u,v\in\mathbb{R}^{d}.
        \]
\end{enumerate}
Combining them yields a one‑step inequality that telescopes over $k$, producing the $O(1/k)$ bound.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $\nabla f$ is $L$‑Lipschitz, then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Proximal Inequality]\label{lem:prox}
For any $x\in\mathbb{R}^{d}$ and any $u\in\mathbb{R}^{d}$,
\[
g(\operatorname{prox}_{\eta g}(x)) +\frac{1}{2\eta}\|\operatorname{prox}_{\eta g}(x)-x\|^{2}
\le g(u)+\frac{1}{2\eta}\|u-x\|^{2}-\frac{1}{2\eta}\|u-\operatorname{prox}_{\eta g}(x)\|^{2}.
\]
\end{lemma}
\begin{proof}
The optimality condition of the proximal definition yields
\[
0\in \partial g(z)+\frac{1}{\eta}(z-x),\qquad z:=\operatorname{prox}_{\eta g}(x).
\]
Thus $\frac{1}{\eta}(x-z)\in\partial g(z)$. By convexity of $g$,
\[
g(u)\ge g(z)+\langle \frac{1}{\eta}(x-z),\,u-z\rangle.
\]
Rearranging gives the desired inequality. \hfill\qedsymbol
\end{proof}
\begin{lemma}[Non‑expansiveness]\label{lem:nonexp}
For any $u,v\in\mathbb{R}^{d}$,
\[
\|\operatorname{prox}_{\eta g}(u)-\operatorname{prox}_{\eta g}(v)\|\le\|u-v\|.
\]
\end{lemma}
\begin{proof}
Direct consequence of Lemma \ref{lem:prox} with $u$ and $v$ swapped and $g$ convex. \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:rate}. Let $x^{*}\in\mathcal{X}^{*}$ be a minimizer and denote
\[
z_{k}:=x_{k}-\eta\nabla f(x_{k})\quad\text{(the gradient step)}.
\]
Then $x_{k+1}= \operatorname{prox}_{\eta g}(z_{k})$ by definition \eqref{eq:update}.

\begin{enumerate}
\item[Step 1.] Apply Lemma \ref{lem:prox} with $x=z_{k}$ and $u=x^{*}$:
\[
g(x_{k+1})+\frac{1}{2\eta}\|x_{k+1}-z_{k}\|^{2}
\le g(x^{*})+\frac{1}{2\eta}\|x^{*}-z_{k}\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\tag{1}
\]

\item[Step 2.] Expand $\|x^{*}-z_{k}\|^{2}$ using the definition of $z_{k}$:
\[
\|x^{*}-z_{k}\|^{2}
= \|x^{*}-x_{k}+\eta\nabla f(x_{k})\|^{2}
= \|x^{*}-x_{k}\|^{2}+2\eta\langle x^{*}-x_{k},\nabla f(x_{k})\rangle
+\eta^{2}\|\nabla f(x_{k})\|^{2}.
\tag{2}
\]

\item[Step 3.] Substitute (2) into (1) and rearrange terms:
\[
\begin{aligned}
g(x_{k+1}) &+ \frac{1}{2\eta}\|x_{k+1}-z_{k}\|^{2}
\\
&\le g(x^{*})+\frac{1}{2\eta}\|x^{*}-x_{k}\|^{2}
+\langle x^{*}-x_{k},\nabla f(x_{k})\rangle
+\frac{\eta}{2}\|\nabla f(x_{k})\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\end{aligned}
\tag{3}
\]

\item[Step 4.] Add $f(x_{k+1})$ to both sides of (3). Using the descent lemma (Lemma \ref{lem:descent}) with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{4}
\]

\item[Step 5.] Combine (3) and (4); the inner products $\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$ cancel partially:
\[
\begin{aligned}
\Phi(x_{k+1}) &:= f(x_{k+1})+g(x_{k+1})\\
&\le f(x_{k})+g(x^{*})+\langle x^{*}-x_{k},\nabla f(x_{k})\rangle
+\frac{\eta}{2}\|\nabla f(x_{k})\|^{2}
\\
&\qquad +\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\eta}\|x_{k+1}-z_{k}\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\end{aligned}
\tag{5}
\]

\item[Step 6.] Observe that $x_{k+1}-z_{k}=x_{k+1}-(x_{k}-\eta\nabla f(x_{k}))
= (x_{k+1}-x_{k})+\eta\nabla f(x_{k})$.
Hence
\[
\|x_{k+1}-z_{k}\|^{2}= \|x_{k+1}-x_{k}\|^{2}+2\eta\langle x_{k+1}-x_{k},\nabla f(x_{k})\rangle+\eta^{2}\|\nabla f(x_{k})\|^{2}.
\tag{6}
\]

\item[Step 7.] Plug (6) into the term $\frac{1}{2\eta}\|x_{k+1}-z_{k}\|^{2}$ appearing in (5):
\[
\frac{1}{2\eta}\|x_{k+1}-z_{k}\|^{2}
= \frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
+\langle x_{k+1}-x_{k},\nabla f(x_{k})\rangle
+\frac{\eta}{2}\|\nabla f(x_{k})\|^{2}.
\tag{7}
\]

\item[Step 8.] Substitute (7) into (5). The two inner‑product terms $\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$ appear twice and sum to $2\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle$.
After cancellation we obtain
\[
\begin{aligned}
\Phi(x_{k+1}) &\le f(x_{k})+g(x^{*})+\langle x^{*}-x_{k},\nabla f(x_{k})\rangle
\\
&\quad +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}
+\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}
+2\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
\\
&\quad +\eta\|\nabla f(x_{k})\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\end{aligned}
\tag{8}
\]

\item[Step 9.] Use the convexity of $f$:
\[
f(x_{k})+ \langle x^{*}-x_{k},\nabla f(x_{k})\rangle \le f(x^{*}).
\tag{9}
\]

\item[Step 10.] Apply the elementary inequality $2\langle a,b\rangle\le \frac{1}{\eta}\|b\|^{2}+\eta\|a\|^{2}$ with $a=\nabla f(x_{k})$, $b=x_{k+1}-x_{k}$:
\[
2\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
\le \frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}+\eta\|\nabla f(x_{k})\|^{2}.
\tag{10}
\]

\item[Step 11.] Insert (9) and (10) into (8). Collect the coefficients of $\|x_{k+1}-x_{k}\|^{2}$:
\[
\begin{aligned}
\Phi(x_{k+1})
&\le f(x^{*})+g(x^{*})
+\Bigl(\frac{L}{2}+\frac{1}{2\eta}+\frac{1}{\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}
+2\eta\|\nabla f(x_{k})\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\end{aligned}
\tag{11}
\]

\item[Step 12.] Since $\eta\le 1/L$, we have $L\eta\le1$. Therefore
\[
\frac{L}{2}+\frac{1}{2\eta}+\frac{1}{\eta}
= \frac{1}{2\eta}\bigl(L\eta+1+2\bigr)
\le \frac{2}{\eta}.
\tag{12}
\]
Hence the term $\Bigl(\frac{L}{2}+\frac{1}{2\eta}+\frac{1}{\eta}\Bigr)\|x_{k+1}-x_{k}\|^{2}$ is non‑negative and can be dropped for an upper bound:
\[
\Phi(x_{k+1})\le \Phi(x^{*})+2\eta\|\nabla f(x_{k})\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\tag{13}
\]

\item[Step 13.] From the definition of $z_{k}$,
\[
x_{k+1}-x_{k}= \operatorname{prox}_{\eta g}(z_{k})-x_{k}
= \bigl(\operatorname{prox}_{\eta g}(z_{k})-z_{k}\bigr)+\eta\nabla f(x_{k}),
\]
and the proximal optimality condition gives $\frac{1}{\eta}(z_{k}-x_{k+1})\in\partial g(x_{k+1})$.
Thus $\|x_{k+1}-x_{k}\|\le \eta\|\nabla f(x_{k})\|$ by Lemma \ref{lem:nonexp}.
Consequently,
\[
\|\nabla f(x_{k})\|^{2}\ge \frac{1}{\eta^{2}}\|x_{k+1}-x_{k}\|^{2}.
\tag{14}
\]

\item[Step 14.] Plug (14) into (13):
\[
\Phi(x_{k+1})\le \Phi(x^{*})+\frac{2}{\eta}\|x_{k+1}-x_{k}\|^{2}
-\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}.
\tag{15}
\]

\item[Step 15.] Rearrange (15) to isolate the distance term:
\[
\frac{1}{2\eta}\|x^{*}-x_{k+1}\|^{2}
\le \Phi(x^{*})-\Phi(x_{k+1})+\frac{2}{\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{16}
\]

\item[Step 16.] Add $\frac{1}{2\eta}\|x^{*}-x_{k}\|^{2}$ to both sides of (16) and use the elementary inequality
\[
\|x^{*}-x_{k}\|^{2}\le \bigl(\|x^{*}-x_{k+1}\|+\|x_{k+1}-x_{k}\|\bigr)^{2}
\le 2\|x^{*}-x_{k+1}\|^{2}+2\|x_{k+1}-x_{k}\|^{2}.
\tag{17}
\]
After simple algebra we obtain the one‑step contraction
\[
\Phi(x_{k+1})-\Phi(x^{*})\le \frac{1}{2\eta}\Bigl(\|x^{*}-x_{k}\|^{2}-\|x^{*}-x_{k+1}\|^{2}\Bigr).
\tag{18}
\]

\item[Step 17.] Sum inequality (18) over $k=0,\dots, K-1$; the telescoping sum yields
\[
\sum_{k=0}^{K-1}\bigl(\Phi(x_{k+1})-\Phi(x^{*})\bigr)
\le \frac{1}{2\eta}\|x^{*}-x_{0}\|^{2}.
\tag{19}
\]

\item[Step 18.] Since each term on the left is non‑negative, the average is bounded:
\[
\Phi(x_{K})-\Phi(x^{*})\le
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(\Phi(x_{k+1})-\Phi(x^{*})\bigr)
\le \frac{\|x_{0}-x^{*}\|^{2}}{2\eta K}.
\tag{20}
\]

\item[Step 19.] Inequality (20) is exactly the claim of Theorem \ref{thm:rate}. ∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
The bound in \eqref{20} can be written as
\[
\Phi(x_{k})-\Phi(x^{*})\le \frac{C}{k},
\qquad C:=\frac{\|x_{0}-x^{*}\|^{2}}{2\eta},
\]
where $C$ depends only on the initial distance and the admissible stepsize $\eta\le1/L$. Increasing $\eta$ (up to $1/L$) reduces $C$.

\section*{Special Cases}
\begin{enumerate}[label=\textbf{S\arabic*:}]
\item \textbf{Pure Gradient Descent ($g\equiv0$).} The proximal step becomes identity, and the proof simplifies because Lemma \ref{lem:prox} reduces to an equality. The rate $O(1/k)$ coincides with the classical result for smooth convex minimization.

\item \textbf{Indicator Function $g=\iota_{\mathcal{C}}$.} Here $\operatorname{prox}_{\eta g}$ is the Euclidean projection onto a closed convex set $\mathcal{C}$. The algorithm becomes the projected gradient method; the same $O(1/k)$ bound holds.

\item \textbf{Strongly Convex $f$.} If $f$ is $\mu$‑strongly convex, a similar derivation (using the stronger inequality
$f(y)\ge f(x)+\langle\nabla f(x),y-x\rangle+\frac{\mu}{2}\|y-x\|^{2}$)
yields a linear rate
\[
\Phi(x_{k})-\Phi(x^{*})\le \bigl(1-\eta\mu\bigr)^{k}\bigl(\Phi(x_{0})-\Phi(x^{*})\bigr).
\]
\end{enumerate}
