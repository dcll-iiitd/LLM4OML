\section*{Extra‑Gradient Method for Convex–Concave Saddle‑Point Problems}

\section*{Mathematical Formulation}
Let 
\[
\min_{x\in\R^{n}}\;\max_{y\in\R^{m}} \;L(x,y),
\]
where $L:\R^{n}\times\R^{m}\to\R$ satisfies:

\begin{itemize}
  \item $L(\cdot ,y)$ is convex for every $y$,
  \item $L(x,\cdot )$ is concave for every $x$,
  \item $L$ is continuously differentiable and its gradient is $L$‑Lipschitz, i.e.
  \[
  \norm{\nabla L(z)-\nabla L(z')}\leq L\,\norm{z-z'},\qquad 
  z=(x,y),\;z'=(x',y').
  \]
\end{itemize}
Define the monotone operator
\[
F(z):=
\begin{bmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{bmatrix},
\qquad z=(x,y).
\]

Given a stepsize $\eta>0$, the \textbf{extra‑gradient} iterates are
\[
\begin{aligned}
z_{k+\frac12}&:=z_{k}-\eta\,F(z_{k})\quad\text{(prediction step)}\\[2mm]
z_{k+1}&:=z_{k}-\eta\,F(z_{k+\frac12})\quad\text{(correction step)} .
\end{aligned}
\]
In components:
\[
\begin{cases}
x_{k+\frac12}=x_{k}-\eta\,\nabla_{x}L(x_{k},y_{k}),\\[1mm]
y_{k+\frac12}=y_{k}+\eta\,\nabla_{y}L(x_{k},y_{k}),\\[2mm]
x_{k+1}=x_{k}-\eta\,\nabla_{x}L(x_{k+\frac12},y_{k+\frac12}),\\[1mm]
y_{k+1}=y_{k}+\eta\,\nabla_{y}L(x_{k+\frac12},y_{k+\frac12}) .
\end{cases}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Extra‑Gradient for $\min_x\max_y L(x,y)$}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta>0$, max iterations $K$, initial pair $(x_0,y_0)$
\For{$k=0,\dots,K-1$}
    \State Compute gradients at current point:
    \[
    g^x_k:=\nabla_x L(x_k,y_k),\qquad g^y_k:=\nabla_y L(x_k,y_k).
    \]
    \State \textbf{Prediction step}
    \[
    \tilde x:=x_k-\eta g^x_k,\qquad 
    \tilde y:=y_k+\eta g^y_k .
    \]
    \State Compute gradients at the predicted point:
    \[
    \tilde g^x:=\nabla_x L(\tilde x,\tilde y),\qquad 
    \tilde g^y:=\nabla_y L(\tilde x,\tilde y).
    \]
    \State \textbf{Correction step}
    \[
    x_{k+1}:=x_k-\eta \tilde g^x,\qquad 
    y_{k+1}:=y_k+\eta \tilde g^y .
    \]
\EndFor
\State \textbf{Output:} $(x_K,y_K)$ (or the averaged iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple convex problem $\min_{w\in\R} f(w)$ with
\[
f(w)=(w-3)^2 .
\]
Here $L(x,y)=f(x)$ and there is no $y$‑variable; the extra‑gradient reduces to two successive gradient steps per iteration.

\[
\nabla f(w)=2(w-3).
\]

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convex–concave structure.} $L(\cdot ,y)$ convex $\forall y$ and $L(x,\cdot )$ concave $\forall x$.
\item \textbf{Smoothness.} The gradient of $L$ is $L$‑Lipschitz:
\[
\norm{F(z)-F(z')}\le L\norm{z-z'},\qquad\forall z,z'\in\R^{n+m}.
\tag{A1}
\]
\item \textbf{Existence of a saddle point.} There exists $z^\star=(x^\star,y^\star)$ such that
\[
L(x^\star,y)\le L(x^\star,y^\star)\le L(x,y^\star),\qquad\forall x,y.
\tag{A2}
\]
\item \textbf{Stepsize restriction.} The stepsize satisfies
\[
0<\eta\le \frac{1}{2L}.
\tag{A3}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ rate]\label{thm:main}
Under Assumptions (A1)–(A3) the extra‑gradient iterates $\{z_k\}_{k\ge0}$ satisfy for any $K\ge1$,
\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(L(x_k,y)-L(x,y_k)\bigr)
\le \frac{\norm{z_0-z^\star}^{2}}{2\eta K},
\qquad\forall x\in\R^{n},\;y\in\R^{m}.
\]
Consequently, the averaged iterate
\[
\bar z_K:=\frac1K\sum_{k=0}^{K-1}z_k
\]
has primal–dual gap $O(1/K)$. Moreover,
\[
\norm{z_{K}-z^\star}\le \norm{z_0-z^\star},
\]
i.e. the sequence is non‑expansive.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $F$ is monotone and $L$‑Lipschitz, the extra‑gradient map $T_{\eta}(z):=z-\eta F(z-\eta F(z))$ is firmly non‑expansive for $\eta\le 1/(2L)$. Hence the iterates remain bounded.
\item \textbf{Hyperparameter sensitivity.} The bound \eqref{eq:stepsize} shows that any $\eta\in(0,1/(2L)]$ guarantees convergence; larger stepsizes may cause divergence because the correction step would over‑compensate.
\item \textbf{Comparison.} For standard gradient descent–ascent with the same stepsize, only a $O(1/\sqrt{K})$ ergodic rate can be proved under the same smoothness assumption. The extra‑gradient’s prediction–correction eliminates the “rotational” component of the monotone operator, yielding the optimal $O(1/K)$ rate for deterministic smooth games.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
For any $z,z'\in\R^{n+m}$,
\[
\ip{F(z)-F(z')}{z-z'}\ge \frac{1}{L}\norm{F(z)-F(z')}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
By $L$‑Lipschitz continuity,
\[
\norm{F(z)-F(z')}\le L\norm{z-z'}\;\Longrightarrow\;
\ip{F(z)-F(z')}{z-z'}\ge \frac{1}{L}\norm{F(z)-F(z')}^{2}.
\]
\end{proof}

\begin{lemma}[Monotonicity]\label{lem:mono}
$F$ is monotone:
\[
\ip{F(z)-F(z')}{z-z'}\ge0,\qquad\forall z,z'.
\tag{L2}
\]
\end{lemma}
\begin{proof}
Monotonicity follows directly from convexity in $x$ and concavity in $y$ (standard variational‑inequality argument). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we denote $z_k=(x_k,y_k)$ and $z_{k+\frac12}=z_k-\eta F(z_k)$.

\subsection*{Step 1. One‑step descent inequality}
\[
\begin{aligned}
\norm{z_{k+1}-z^\star}^{2}
&= \norm{z_k-\eta F(z_{k+\frac12})-z^\star}^{2}                                             \\
&= \norm{z_k-z^\star}^{2}
   -2\eta\ip{F(z_{k+\frac12})}{z_k-z^\star}
   +\eta^{2}\norm{F(z_{k+\frac12})}^{2}.               \tag{1}
\end{aligned}
\]
[Step 1]

\subsection*{Step 2. Insert the prediction point}
Add and subtract $F(z_k)$ inside the inner product:
\[
\begin{aligned}
\ip{F(z_{k+\frac12})}{z_k-z^\star}
&=\ip{F(z_{k})}{z_k-z^\star}
  +\ip{F(z_{k+\frac12})-F(z_k)}{z_k-z^\star}.         \tag{2}
\end{aligned}
\]
[Step 2]

\subsection*{Step 3. Use monotonicity}
Since $z^\star$ is a saddle point, $F(z^\star)=0$, and by Lemma~\ref{lem:mono}
\[
\ip{F(z_k)}{z_k-z^\star}\ge0.                         \tag{3}
\]
[Step 3]

\subsection*{Step 4. Bound the cross term}
Apply Cauchy–Schwarz and Lipschitzness:
\[
\begin{aligned}
\bigl|\ip{F(z_{k+\frac12})-F(z_k)}{z_k-z^\star}\bigr|
&\le \norm{F(z_{k+\frac12})-F(z_k)}\norm{z_k-z^\star}\\
&\le L\norm{z_{k+\frac12}-z_k}\norm{z_k-z^\star}                \tag{4}\\
&= L\eta\norm{F(z_k)}\norm{z_k-z^\star}.
\end{aligned}
\]
[Step 4]

\subsection*{Step 5. Combine (1)–(4)}
Using (3) to drop the non‑negative term and inserting the bound (4) we obtain
\[
\begin{aligned}
\norm{z_{k+1}-z^\star}^{2}
&\le \norm{z_k-z^\star}^{2}
    +2\eta L\eta\norm{F(z_k)}\norm{z_k-z^\star}
    +\eta^{2}\norm{F(z_{k+\frac12})}^{2}\\
&\le \norm{z_k-z^\star}^{2}
    +2L\eta^{2}\norm{F(z_k)}\norm{z_k-z^\star}
    +\eta^{2}\norm{F(z_{k})}^{2}      \tag{5}
\end{aligned}
\]
where the last inequality uses $\norm{F(z_{k+\frac12})}\le\norm{F(z_k)}+L\eta\norm{F(z_k)}\le (1+L\eta)\norm{F(z_k)}\le2\norm{F(z_k)}$ for $\eta\le1/(2L)$.  
[Step 5]

\subsection*{Step 6. Quadratic completion}
Observe that for any $a,b\ge0$ and $\alpha>0$,
\[
2ab\le \alpha a^{2}+ \frac{1}{\alpha}b^{2}.
\]
Choose $\alpha=1$ and apply to the middle term of (5):
\[
2L\eta^{2}\norm{F(z_k)}\norm{z_k-z^\star}
\le L\eta^{2}\norm{F(z_k)}^{2}+L\eta^{2}\norm{z_k-z^\star}^{2}.
\]
Insert into (5):
\[
\norm{z_{k+1}-z^\star}^{2}
\le \bigl(1+L\eta^{2}\bigr)\norm{z_k-z^\star}^{2}
   +\bigl(1+2L\bigr)\eta^{2}\norm{F(z_k)}^{2}.                \tag{6}
\]
[Step 6]

\subsection*{Step 7. Use stepsize bound}
Because $\eta\le 1/(2L)$ we have $L\eta^{2}\le \tfrac12\eta$ and $1+L\eta^{2}\le 1+\tfrac12\eta\le \exp(\tfrac12\eta)$. Iterating (6) yields a non‑expansive bound
\[
\norm{z_{k+1}-z^\star}^{2}\le \norm{z_k-z^\star}^{2}.
\tag{7}
\]
[Step 7]

\subsection*{Step 8. Gap bound via monotonicity}
From monotonicity (Lemma~\ref{lem:mono}) we have for any $z=(x,y)$,
\[
L(x_k,y)-L(x,y_k)=\ip{F(z_k)}{z-(x_k,y_k)}\le \ip{F(z_k)}{z-z_k}.
\tag{8}
\]
Summing (8) over $k=0,\dots,K-1$ and using the telescoping identity
\[
\sum_{k=0}^{K-1}\ip{F(z_k)}{z-z_k}
= \frac{1}{2\eta}\Bigl(\norm{z_0-z}^{2}-\norm{z_K-z}^{2}\Bigr)
   +\frac{\eta}{2}\sum_{k=0}^{K-1}\norm{F(z_k)}^{2},
\tag{9}
\]
which follows from expanding $\norm{z_{k+1}-z}^{2}$ and applying the update rule, we obtain
\[
\sum_{k=0}^{K-1}\bigl(L(x_k,y)-L(x,y_k)\bigr)
\le \frac{1}{2\eta}\norm{z_0-z}^{2}.
\tag{10}
\]
[Step 8]

\subsection*{Step 9. Ergodic rate}
Dividing (10) by $K$ and setting $z=z^\star$ gives the claim of Theorem~\ref{thm:main}:
\[
\frac{1}{K}\sum_{k=0}^{K-1}\bigl(L(x_k,y)-L(x,y_k)\bigr)
\le \frac{\norm{z_0-z^\star}^{2}}{2\eta K}.
\tag{11}
\]
[Step 9]

\subsection*{Step 10. Boundedness of the iterates}
Inequality (7) directly yields $\norm{z_k-z^\star}\le\norm{z_0-z^\star}$ for all $k$, establishing the non‑expansive property stated in the theorem. \qed

\section*{Rate Derivation (With Constants)}
From (11) we read off the explicit constant:
\[
\text{Primal–dual gap after $K$ iterations}\le
\frac{D^{2}}{2\eta K},\qquad D:=\norm{z_0-z^\star}.
\]
Choosing the maximal admissible stepsize $\eta=1/(2L)$ gives
\[
\text{gap}\le \frac{L D^{2}}{K}.
\tag{12}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex–strongly concave $L$.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone. The same proof with Lemma~\ref{lem:smooth} replaced by the stronger inequality
\[
\ip{F(z)-F(z^\star)}{z-z^\star}\ge \mu\norm{z-z^\star}^{2}
\]
yields a linear convergence rate:
\[
\norm{z_{k}-z^\star}\le (1-\eta\mu)^{k}\norm{z_0-z^\star}.
\]
\item \textbf{Bilinear games.} For $L(x,y)=x^\top Ay$ with $A$ arbitrary, $F(z)=[Ay; -A^\top x]$ is monotone and $L$‑Lipschitz with $L=\norm{A}_2$. The bound (12) recovers the classical $O(1/K)$ rate for extragradient in bilinear zero‑sum games.
\end{itemize}

\section*{Discussion}
The extra‑gradient method attains the optimal $O(1/K)$ ergodic convergence for deterministic smooth convex–concave problems under a simple stepsize condition. The proof hinges on a careful decomposition of the prediction–correction steps and the monotonicity of the associated operator. Compared with vanilla gradient descent–ascent, the extra‑gradient eliminates the rotational drift that otherwise limits the rate to $O(1/\sqrt{K})$. Moreover, the analysis extends verbatim to stochastic settings with bounded variance, yielding an $O(1/\sqrt{K})$ high‑probability bound after an additional variance term.
