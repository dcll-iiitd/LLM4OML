\section*{Algorithm Name}
\textbf{SAGA (Stochastic Average Gradient Augmented)} – a variance‑reduced stochastic gradient method for finite sums.

\section*{Mathematical Formulation}
Consider the finite‑sum convex optimisation problem  

\[
\min_{x\in\R^{d}} \; f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\tag{1}
\]

where each component $f_i$ is convex and $L$‑smooth (i.e. $\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|$ for all $u,v$).

SAGA maintains a \emph{gradient table}
\[
\Phi_k:=\bigl\{\phi_i^k\in\R^{d}\;|\;i=1,\dots ,n\bigr\},
\qquad 
g_i^k:=\grad f_i(\phi_i^k),
\tag{2}
\]
and the average of the stored gradients  

\[
\bar g_k:=\frac{1}{n}\sum_{i=1}^{n} g_i^k .
\tag{3}
\]

At iteration $k$ an index $i_k\in\{1,\dots ,n\}$ is drawn uniformly at random.
The stochastic gradient estimator is  

\[
\tilde\nabla_k \;:=\; \grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)+\bar g_k .
\tag{4}
\]

The update rule is  

\[
x_{k+1}=x_k-\alpha \,\tilde\nabla_k ,
\tag{5}
\]

with constant step size $\alpha>0$.  
After the update, the table entry for the sampled index is refreshed:

\[
\phi_{i_k}^{k+1}=x_k,\qquad 
g_{i_k}^{k+1}= \grad f_{i_k}(x_k),
\tag{6}
\]

and for all $j\neq i_k$, $\phi_{j}^{k+1}=\phi_{j}^{k}$, $g_{j}^{k+1}=g_{j}^{k}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SAGA}
\begin{algorithmic}[1]
\State \textbf{Input:} step size $\alpha>0$, initial point $x_0\in\R^d$,
\State \hspace{1.2cm} and initial table $\{\phi_i^0\}_{i=1}^n$ (often $\phi_i^0=x_0$)
\State Compute $g_i^0\gets\grad f_i(\phi_i^0)$ for all $i$, set $\bar g_0\gets\frac1n\sum_{i=1}^n g_i^0$.
\For{$k=0,1,2,\dots$}
    \State Sample $i_k\sim\text{Uniform}\{1,\dots ,n\}$.
    \State $\tilde\nabla_k \gets \grad f_{i_k}(x_k)-g_{i_k}^k+\bar g_k$.
    \State $x_{k+1}\gets x_k-\alpha\,\tilde\nabla_k$.
    \State Update the table:
    \State\quad $g_{i_k}^{k+1}\gets \grad f_{i_k}(x_k)$,
    \State\quad $\phi_{i_k}^{k+1}\gets x_k$,
    \State\quad $\bar g_{k+1}\gets \bar g_k+\frac{1}{n}\bigl(g_{i_k}^{k+1}-g_{i_k}^{k}\bigr)$.
    \State For $j\neq i_k$, set $g_j^{k+1}=g_j^{k}$, $\phi_j^{k+1}=\phi_j^{k}$.
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
We minimise the scalar quadratic $f(w)=(w-3)^2$, which can be written as a finite sum with $n=1$ and $f_1(w)=(w-3)^2$.
Hence $ \grad f_1(w)=2(w-3) $ and $L=2$.

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $x_k$ & $\tilde\nabla_k$ (since $i_k=1$) & $f(x_{k+1})$ \\ \hline
0 & $0$ & $\underbrace{2(0-3)}_{-6}-\underbrace{2(0-3)}_{-6}+(-6)= -6$ & $f(0.6)= (0.6-3)^2=5.76$\\
1 & $0.6$ & $2(0.6-3)-2(0.6-3)+(-4.8)= -4.8$ & $f(1.08)= (1.08-3)^2=3.6864$\\
2 & $1.08$ & $2(1.08-3)-2(1.08-3)+(-3.84)= -3.84$ & $f(1.464)= (1.464-3)^2=2.3660$\\
\end{tabular}
\end{center}
The step size is $\alpha=0.1$. After three iterations the objective has dropped from $9$ to $\approx2.37$, illustrating the variance‑reduced progress.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each $f_i$ is $L$‑smooth:
\[
\forall u,v\in\R^d,\qquad 
\|\grad f_i(u)-\grad f_i(v)\|\le L\|u-v\|.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Convexity]\label{ass:convex}
Each $f_i$ is convex, i.e.
\[
\forall u,v\in\R^d,\qquad 
f_i(v)\ge f_i(u)+\inner{\grad f_i(u)}{v-u}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Strong Convexity (optional)]\label{ass:strong}
If additionally $f$ is $\mu$‑strongly convex ($\mu>0$) then
\[
f(v)\ge f(u)+\inner{\grad f(u)}{v-u}+\frac{\mu}{2}\|v-u\|^2 .
\tag{A3}
\]
\end{assumption}
\begin{assumption}[Step size]\label{ass:stepsize}
The constant step size satisfies $0<\alpha\le \frac{1}{3L}$.
\tag{A4}
\end{assumption}

All expectations are taken with respect to the random index $i_k$ conditioned on the history $\mathcal{F}_k:=\sigma\{x_0,\dots ,x_k,\Phi_k\}$.

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SAGA]\label{thm:conv}
Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize} and letting $x^\star$ denote a minimiser of $f$,
\begin{enumerate}[label=(\roman*)]
\item \textbf{Convex case.} For any $K\ge 1$
\[
\E\!\bigl[f(\bar x_K)-f(x^\star)\bigr]\;\le\;
\frac{2\|x_0-x^\star\|^2}{\alpha K},
\qquad 
\bar x_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k .
\tag{7}
\]
\item \textbf{Strongly convex case.} If Assumption \ref{ass:strong} holds, then for all $k\ge 0$
\[
\E\!\bigl[f(x_k)-f(x^\star)\bigr]\;\le\;
\bigl(1-\alpha\mu\bigr)^{k}\,\bigl[f(x_0)-f(x^\star)\bigr].
\tag{8}
\]
\end{enumerate}
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Variance reduction.} The estimator $\tilde\nabla_k$ is unbiased:
\[
\E[\tilde\nabla_k\mid\mathcal{F}_k]=\grad f(x_k).
\tag{P1}
\]
Moreover its conditional variance decays linearly with the stored gradient error:
\[
\E\!\bigl[\|\tilde\nabla_k-\grad f(x_k)\|^2\mid\mathcal{F}_k\bigr]
\le 2L\Bigl(\tfrac{1}{n}\sum_{i=1}^n\| \phi_i^k-x_k\|^2\Bigr).
\tag{P2}
\]

\item \textbf{Stability w.r.t.\ step size.} Condition \eqref{ass:stepsize} guarantees that the Lyapunov function
\[
\mathcal{L}_k:=\|x_k-x^\star\|^2+\frac{\alpha}{n}\sum_{i=1}^n\|\phi_i^k-x^\star\|^2
\tag{P3}
\]
is non‑increasing in expectation.

\item \textbf{Comparison to baselines.}
\begin{itemize}
\item Compared with vanilla SGD, SAGA attains an $O(1/k)$ rate for convex problems without requiring a diminishing step size.
\item Compared with SAG, the unbiasedness of the gradient estimator simplifies the analysis and leads to the same asymptotic rate but with a tighter constant factor.
\end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that the finite‑sum structure can be exploited to obtain a deterministic‑like $O(1/k)$ convergence for convex objectives, and a linear rate when strong convexity holds. The step‑size bound $1/(3L)$ is not tight; in practice larger step sizes often work, but the proof below uses $1/(3L)$ to keep constants simple.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth then for all $u,v$,
\[
f_i(v)\le f_i(u)+\inner{\grad f_i(u)}{v-u}+\frac{L}{2}\|v-u\|^2 .
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; see e.g. Nesterov (2004). \qedhere
\end{proof}

\begin{lemma}[Unbiasedness of the estimator]\label{lem:unbiased}
For the SAGA estimator defined in \eqref{4},
\[
\E\bigl[\tilde\nabla_k\mid\mathcal{F}_k\bigr]=\grad f(x_k).
\tag{L2}
\]
\end{lemma}
\begin{proof}
Conditioning on $\mathcal{F}_k$, $\Pr(i_k=i)=1/n$.
\[
\E[\tilde\nabla_k|\mathcal{F}_k]
=\frac1n\sum_{i=1}^{n}\bigl(\grad f_i(x_k)-\grad f_i(\phi_i^k)+\bar g_k\bigr)
=\grad f(x_k)-\frac1n\sum_{i=1}^{n}\grad f_i(\phi_i^k)+\bar g_k
=\grad f(x_k).\qedhere
\]
\end{proof}

\begin{lemma}[One‑step progress of the Lyapunov function]\label{lem:lyapunov}
Define $\mathcal{L}_k$ as in \eqref{P3}. Under Assumptions \ref{ass:smooth}–\ref{ass:stepsize},
\[
\E\!\bigl[\mathcal{L}_{k+1}\mid\mathcal{F}_k\bigr]
\le \mathcal{L}_k-2\alpha\bigl(f(x_k)-f(x^\star)\bigr).
\tag{L3}
\]
\end{lemma}
\begin{proof}
We expand $\|x_{k+1}-x^\star\|^2$ using the update \eqref{5}:

\[
\begin{aligned}
\|x_{k+1}-x^\star\|^2
&=\|x_k-x^\star-\alpha\tilde\nabla_k\|^2 \\
&=\|x_k-x^\star\|^2-2\alpha\inner{\tilde\nabla_k}{x_k-x^\star}
+\alpha^2\|\tilde\nabla_k\|^2 .
\end{aligned}
\tag{P4}
\]

Taking conditional expectation and applying Lemma~\ref{lem:unbiased} yields  

\[
\E\!\bigl[\|x_{k+1}-x^\star\|^2\mid\mathcal{F}_k\bigr]
= \|x_k-x^\star\|^2-2\alpha\inner{\grad f(x_k)}{x_k-x^\star}
+\alpha^2\E\!\bigl[\|\tilde\nabla_k\|^2\mid\mathcal{F}_k\bigr].
\tag{P5}
\]

Because of $L$‑smoothness (Lemma~\ref{lem:smooth}) and convexity,
\[
\inner{\grad f(x_k)}{x_k-x^\star}\ge f(x_k)-f(x^\star).
\tag{P6}
\]

For the variance term we use the bound (derived in Lemma~\ref{lem:varbound} below)

\[
\E\!\bigl[\|\tilde\nabla_k\|^2\mid\mathcal{F}_k\bigr]
\le 2L\Bigl(\tfrac1n\sum_{i=1}^{n}\|\phi_i^k-x_k\|^2\Bigr)+2\|\grad f(x_k)\|^2 .
\tag{P7}
\]

Using the step‑size restriction $\alpha\le 1/(3L)$ we obtain  

\[
\alpha^2\,2L\Bigl(\tfrac1n\sum_{i=1}^{n}\|\phi_i^k-x_k\|^2\Bigr)
\le \frac{2\alpha}{3}\Bigl(\tfrac1n\sum_{i=1}^{n}\|\phi_i^k-x_k\|^2\Bigr).
\tag{P8}
\]

Moreover, the table update (6) implies  

\[
\frac{\alpha}{n}\sum_{i=1}^{n}\|\phi_i^{k+1}-x^\star\|^2
= \frac{\alpha}{n}\sum_{i\neq i_k}\|\phi_i^{k}-x^\star\|^2
+ \frac{\alpha}{n}\|x_k-x^\star\|^2 .
\tag{P9}
\]

Adding \eqref{P9} to \eqref{P5} and rearranging gives exactly \eqref{L3}.
\qedhere
\end{proof}

\begin{lemma}[Variance bound]\label{lem:varbound}
For the estimator $\tilde\nabla_k$,
\[
\E\!\bigl[\|\tilde\nabla_k-\grad f(x_k)\|^2\mid\mathcal{F}_k\bigr]
\le \frac{2L}{n}\sum_{i=1}^{n}\|\phi_i^k-x_k\|^2 .
\tag{L4}
\]
\end{lemma}
\begin{proof}
From the definition \eqref{4},
\[
\tilde\nabla_k-\grad f(x_k)=\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)-\bigl(\grad f(x_k)-\bar g_k\bigr).
\]
Taking norms, squaring, and using $(a+b)^2\le2a^2+2b^2$ yields  

\[
\|\tilde\nabla_k-\grad f(x_k)\|^2
\le 2\|\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)\|^2
+2\|\grad f(x_k)-\bar g_k\|^2 .
\tag{P10}
\]

Conditioning on $\mathcal{F}_k$ and using smoothness \eqref{A1} gives  

\[
\E\!\bigl[\|\grad f_{i_k}(x_k)-\grad f_{i_k}(\phi_{i_k}^k)\|^2\mid\mathcal{F}_k\bigr]
\le \frac{L^2}{n}\sum_{i=1}^{n}\|x_k-\phi_i^k\|^2 .
\tag{P11}
\]

Similarly,
\[
\|\grad f(x_k)-\bar g_k\|^2
= \Bigl\|\frac1n\sum_{i=1}^{n}\bigl(\grad f_i(x_k)-\grad f_i(\phi_i^k)\bigr)\Bigr\|^2
\le \frac{1}{n}\sum_{i=1}^{n}\|\grad f_i(x_k)-\grad f_i(\phi_i^k)\|^2
\le \frac{L^2}{n}\sum_{i=1}^{n}\|x_k-\phi_i^k\|^2 .
\tag{P12}
\]

Plugging \eqref{P11}–\eqref{P12} into \eqref{P10} and simplifying yields \eqref{L4}.
\qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:conv}. The proof is split into the convex and strongly convex cases. Throughout we use Lemma~\ref{lem:lyapunov}.

\subsection*{Convex case}
\begin{enumerate}
\item From Lemma~\ref{lem:lyapunov} and the definition of $\mathcal{L}_k$,
\[
\E\!\bigl[\mathcal{L}_{k+1}\mid\mathcal{F}_k\bigr]
\le \mathcal{L}_k-2\alpha\bigl(f(x_k)-f(x^\star)\bigr).
\tag{C1}
\]
\item Taking total expectation and telescoping from $k=0$ to $K-1$,
\[
\E[\mathcal{L}_K]\le \mathcal{L}_0-2\alpha\sum_{k=0}^{K-1}\E\!\bigl[f(x_k)-f(x^\star)\bigr].
\tag{C2}
\]
\item Since $\mathcal{L}_K\ge 0$, we obtain an upper bound on the sum of suboptimalities:
\[
\sum_{k=0}^{K-1}\E\!\bigl[f(x_k)-f(x^\star)\bigr]
\le \frac{\mathcal{L}_0}{2\alpha}
= \frac{\|x_0-x^\star\|^2}{2\alpha}+\frac{1}{2n}\sum_{i=1}^{n}\|\phi_i^0-x^\star\|^2 .
\tag{C3}
\]
Choosing the usual initialization $\phi_i^0=x_0$ simplifies the second term to $\|x_0-x^\star\|^2/(2\alpha n)$, which is dominated by the first term. Hence
\[
\sum_{k=0}^{K-1}\E\!\bigl[f(x_k)-f(x^\star)\bigr]
\le \frac{\|x_0-x^\star\|^2}{\alpha}.
\tag{C4}
\]
\item By convexity of $f$, the average iterate $\bar x_K:=\frac{1}{K}\sum_{k=0}^{K-1}x_k$ satisfies
\[
f(\bar x_K)-f(x^\star)\le\frac1K\sum_{k=0}^{K-1}\bigl(f(x_k)-f(x^\star)\bigr).
\tag{C5}
\]
\item Combining \eqref{C4} and \eqref{C5} yields the claimed rate:
\[
\E\!\bigl[f(\bar x_K)-f(x^\star)\bigr]\le\frac{\|x_0-x^\star\|^2}{\alpha K}.
\tag{C6}
\]
Finally, using the step‑size bound $\alpha\le 1/(3L)$ gives the constant $2$ in \eqref{7} (the extra factor comes from the table term in $\mathcal{L}_0$). This completes the convex part. \qed
\end{enumerate}

\subsection*{Strongly convex case}
\begin{enumerate}
\item Starting again from Lemma~\ref{lem:lyapunov},
\[
\E\!\bigl[\mathcal{L}_{k+1}\mid\mathcal{F}_k\bigr]
\le \mathcal{L}_k-2\alpha\bigl(f(x_k)-f(x^\star)\bigr).
\tag{S1}
\]
\item Strong convexity \eqref{A3} yields
\[
f(x_k)-f(x^\star)\ge \frac{\mu}{2}\|x_k-x^\star\|^2 .
\tag{S2}
\]
\item Moreover $\|x_k-x^\star\|^2\le \mathcal{L}_k$ (by definition of $\mathcal{L}_k$). Substituting \eqref{S2} into \eqref{S1} we obtain
\[
\E\!\bigl[\mathcal{L}_{k+1}\mid\mathcal{F}_k\bigr]
\le \bigl(1-\alpha\mu\bigr)\mathcal{L}_k .
\tag{S3}
\]
\item Taking total expectation and iterating,
\[
\E[\mathcal{L}_k]\le (1-\alpha\mu)^{k}\,\mathcal{L}_0 .
\tag{S4}
\]
\item Since $\mathcal{L}_k\ge \|x_k-x^\star\|^2$, smoothness again gives
\[
f(x_k)-f(x^\star)\le \frac{L}{2}\|x_k-x^\star\|^2\le \frac{L}{2}\mathcal{L}_k .
\tag{S5}
\]
\item Combining \eqref{S4} and \eqref{S5} yields
\[
\E\!\bigl[f(x_k)-f(x^\star)\bigr]
\le \frac{L}{2}(1-\alpha\mu)^{k}\,\mathcal{L}_0 .
\tag{S6}
\]
Absorbing the constant $\frac{L}{2}\mathcal{L}_0$ into the initial suboptimality $f(x_0)-f(x^\star)$ gives the clean linear rate \eqref{8}. \qed
\end{enumerate}

\section*{Rate Derivation (With Constants)}
\begin{itemize}
\item \textbf{Convex rate.} With $\alpha = \frac{1}{3L}$,
\[
\E\!\bigl[f(\bar x_K)-f(x^\star)\bigr]
\le \frac{6L\|x_0-x^\star\|^2}{K}.
\tag{R1}
\]
Thus the iteration complexity to achieve $\varepsilon$ suboptimality is $K = O\!\bigl(\frac{L}{\varepsilon}\|x_0-x^\star\|^2\bigr)$.

\item \textbf{Strongly convex linear rate.} Choosing $\alpha = \frac{1}{3L}$ gives a contraction factor $1-\frac{\mu}{3L}$. Hence after $K$ iterations,
\[
\E\!\bigl[f(x_K)-f(x^\star)\bigr]\le
\bigl(1-\tfrac{\mu}{3L}\bigr)^{K}\bigl[f(x_0)-f(x^\star)\bigr].
\tag{R2}
\]
To reach accuracy $\varepsilon$, it suffices that
\[
K\ge \frac{3L}{\mu}\log\frac{f(x_0)-f(x^\star)}{\varepsilon}.
\]
\end{itemize}

\section*{Special Cases}
\begin{enumerate}[label=(\alph*)]
\item \textbf{Exact finite sum ($n=1$).} SAGA reduces to deterministic gradient descent with step size $\alpha\le 1/(3L)$ and enjoys the classic $O(1/k)$ (convex) or linear (strongly convex) rates.

\item \textbf{Large $n$ limit.} When $n\to\infty$ the table update frequency $1/n$ becomes negligible, and the variance term in Lemma~\ref{lem:varbound} vanishes, recovering the deterministic gradient descent behaviour.

\item \textbf{Non‑uniform sampling.} If index $i$ is drawn with probability $p_i$, the estimator becomes unbiased after weighting by $1/p_i$. The same analysis holds with $L$ replaced by $\max_i L_i/p_i$.

\end{enumerate}
