\section*{Algorithm Name}
\textbf{Stochastic Subsampled Newton (SSN)}  

The method uses a stochastic estimate $H_t$ of the Hessian $\nabla^{2}f(x_t)$ obtained by sampling a mini‑batch of data points.  At iteration $t$ the update is a (damped) Newton step:
\[
x_{t+1}=x_t-\alpha\,H_t^{-1}\,\nabla f(x_t),\qquad \alpha>0.
\]

\section*{Mathematical Formulation}
Let $f:\R^{d}\to\R$ be twice differentiable.  
\begin{itemize}
    \item \textbf{Gradient:} $g_t:=\nabla f(x_t)$ (exact or possibly stochastic, but for simplicity we keep the gradient exact).
    \item \textbf{Stochastic Hessian:} $H_t$ is an unbiased estimator of the true Hessian,
    \[
    \E[H_t\mid x_t]=\nabla^{2}f(x_t).
    \]
    It is constructed by uniformly sampling a subset $\mathcal{S}_t$ of the data and forming
    \[
    H_t:=\frac{1}{|\mathcal{S}_t|}\sum_{i\in\mathcal{S}_t}\nabla^{2}f_i(x_t),
    \]
    where $f(x)=\frac{1}{n}\sum_{i=1}^{n}f_i(x)$.
    \item \textbf{Step size (damping):} $\alpha\in(0,1]$.
\end{itemize}
The iteration reads
\begin{equation}
\boxed{\,x_{t+1}=x_t-\alpha H_t^{-1}g_t\,}. \tag{1}
\end{equation}
A regularisation term $\lambda I$ may be added to $H_t$ to guarantee invertibility; we absorb it in the definition of $H_t$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Stochastic Subsampled Newton (SSN)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0\in\R^{d}$, step size $\alpha\in(0,1]$, batch size $b$, max iterations $T$.
\For{$t=0,1,\dots,T-1$}
    \State Compute exact gradient $g_t=\nabla f(x_t)$.
    \State Sample a mini‑batch $\mathcal{S}_t\subset\{1,\dots,n\}$ with $|\mathcal{S}_t|=b$.
    \State Form stochastic Hessian estimate 
    \[
    H_t=\frac{1}{b}\sum_{i\in\mathcal{S}_t}\nabla^{2} f_i(x_t)+\lambda I.
    \]
    \State Solve for Newton direction $p_t= - H_t^{-1}g_t$ (e.g. by CG).
    \State Update $x_{t+1}=x_t+\alpha p_t$.
\EndFor
\State \textbf{Output:} $x_T$ (or the iterate with smallest $\|g_t\|$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^2,\qquad \nabla f(w)=2(w-3),\qquad \nabla^{2}f(w)=2.
\]
We take a deterministic “subsampled” Hessian $H_t=2+\lambda$ with $\lambda=0$ (so $H_t^{-1}=1/2$).  
Set $w_0=0$, $\alpha=0.1$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t= \nabla f(w_t)$ & $p_t=-H_t^{-1}g_t$ & $w_{t+1}=w_t+\alpha p_t$ \\ \hline
0 & $2(0-3)=-6$ & $-(-6)/2=3$ & $0+0.1\cdot 3=0.3$ \\
1 & $2(0.3-3)=-5.4$ & $-(-5.4)/2=2.7$ & $0.3+0.1\cdot 2.7=0.57$ \\
2 & $2(0.57-3)=-4.86$ & $-(-4.86)/2=2.43$ & $0.57+0.1\cdot 2.43=0.813$ 
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(0)=9,\; f(0.3)=7.29,\; f(0.57)=5.9049,\; f(0.813)=4.6937.
\]
The algorithm monotonically decreases $f$ toward the optimum $w^{\star}=3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ has $L$‑Lipschitz gradient:
\[
\forall x,y\in\R^{d},\qquad 
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y}.
\]
Equivalently,
\[
f(y)\le f(x)+\inner{\nabla f(x),y-x}+\frac{L}{2}\norm{y-x}^{2}.
\tag{2}
\]
\end{assumption}

\begin{assumption}[Hessian Boundedness]\label{ass:hessian}
For every $x$, the true Hessian satisfies
\[
\mu I \preceq \nabla^{2}f(x) \preceq L_{H} I,
\]
for constants $0<\mu\le L_{H}$.
\end{assumption}

\begin{assumption}[Stochastic Hessian Unbiasedness and Variance]\label{ass:unbiased}
Conditional on $x_t$,
\[
\E[H_t\mid x_t]=\nabla^{2}f(x_t),\qquad
\E\bigl[\norm{H_t-\nabla^{2}f(x_t)}^{2}\mid x_t\bigr]\le \sigma_{H}^{2}.
\tag{3}
\]
Moreover $H_t$ is uniformly positive definite:
\[
\lambda_{\min}(H_t)\ge \mu_{\!h}>0\quad\text{a.s.}
\tag{4}
\]
\end{assumption}

\begin{assumption}[Step Size]\label{ass:stepsize}
The damping parameter satisfies
\[
0<\alpha\le \min\Bigl\{1,\;\frac{\mu_{\!h}}{L}\Bigr\}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:stepsize}. Let $\{x_t\}_{t\ge0}$ be generated by \eqref{1}. Define $f^{\star}= \inf_{x}f(x)$. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha\,\mu_{\!h}\,T}
\;+\;
\frac{L\,\sigma_{H}^{2}}{\mu_{\!h}^{2}}\,\frac{1}{T}.
\tag{6}
\]
Consequently $\E[\norm{\nabla f(x_t)}^{2}]=\mathcal{O}(1/t)$, i.e. the method converges to a stationary point in expectation.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability:} The bound (6) holds without requiring convexity; only smoothness and Hessian boundedness are needed.
    \item \textbf{Hyper‑parameter sensitivity:} Larger $\alpha$ accelerates the first term but must respect \eqref{5}. The variance term scales linearly with $\sigma_H^{2}$ and is independent of $\alpha$.
    \item \textbf{Comparison:} For SGD with stepsize $\eta_t\asymp 1/\sqrt{t}$ the expected gradient norm decays as $\mathcal{O}(1/\sqrt{t})$. The SSN method improves this to $\mathcal{O}(1/t)$ thanks to curvature information.
\end{itemize}

\section*{Discussion}
The theorem shows that a single Newton‑type step with a stochastic Hessian yields a \emph{first‑order} convergence rate identical to that of deterministic Newton under exact Hessians, up to an additive variance penalty.  In practice the variance term can be reduced by increasing the mini‑batch size $b$, or by employing variance‑reduced Hessian estimators.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for $L$‑smooth functions]\label{lem:descent}
For any $x$ and any direction $p$,
\[
f(x+\alpha p)\le f(x)+\alpha\inner{\nabla f(x),p}+\frac{L\alpha^{2}}{2}\norm{p}^{2}.
\tag{7}
\]
\end{lemma}
\begin{proof}
Apply inequality (2) with $y=x+\alpha p$ and simplify. \hfill\qedsymbol
\end{proof}

\begin{lemma}[Quadratic bound via Hessian eigenvalues]\label{lem:hessian}
If $\mu_{\!h}I\preceq H_t\preceq L_{H}I$, then for any vector $v$,
\[
\frac{1}{L_{H}}\norm{v}^{2}\le \inner{v, H_t^{-1} v}\le \frac{1}{\mu_{\!h}}\norm{v}^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
From the spectral decomposition $H_t=Q\Lambda Q^{\top}$ with $\Lambda$ diagonal containing eigenvalues in $[\mu_{\!h},L_{H}]$, we have $H_t^{-1}=Q\Lambda^{-1}Q^{\top}$ and $\Lambda^{-1}\in[1/L_{H},1/\mu_{\!h}]$. Multiplying by $v^{\top}$ yields (8). \hfill\qedsymbol
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout the proof $x_t$ denotes the random iterate at time $t$ and all expectations are taken w.r.t. the randomness up to time $t$ (including the sampled Hessian).

\bigskip
\noindent\textbf{Step 1 (Apply Descent Lemma).}  
From Lemma \ref{lem:descent} with $p_t=-H_t^{-1}g_t$,
\begin{align}
f(x_{t+1})
&=f\bigl(x_t+\alpha p_t\bigr) \nonumber\\
&\le f(x_t)+\alpha\inner{g_t,p_t}
   +\frac{L\alpha^{2}}{2}\norm{p_t}^{2}. \tag{9}
\end{align}
\noindent\textbf{Step 2 (Insert Newton direction).}  
Since $p_t=-H_t^{-1}g_t$,
\[
\inner{g_t,p_t}= -\inner{g_t, H_t^{-1}g_t}
\quad\text{and}\quad
\norm{p_t}^{2}= \inner{g_t, H_t^{-2}g_t}.
\tag{10}
\]

\noindent\textbf{Step 3 (Bound the quadratic term).}  
Using Lemma \ref{lem:hessian} twice,
\[
\inner{g_t, H_t^{-1}g_t}\ge \frac{1}{L_{H}}\norm{g_t}^{2},
\qquad
\inner{g_t, H_t^{-2}g_t}\le \frac{1}{\mu_{\!h}^{2}}\norm{g_t}^{2}.
\tag{11}
\]

\noindent\textbf{Step 4 (Combine 9–11).}  
Plugging (10)–(11) into (9) yields
\begin{align}
f(x_{t+1})
&\le f(x_t)
   -\alpha\frac{1}{L_{H}}\norm{g_t}^{2}
   +\frac{L\alpha^{2}}{2}\frac{1}{\mu_{\!h}^{2}}\norm{g_t}^{2} \nonumber\\
&= f(x_t)-\Bigl(\frac{\alpha}{L_{H}}-\frac{L\alpha^{2}}{2\mu_{\!h}^{2}}\Bigr)\norm{g_t}^{2}. \tag{12}
\end{align}
\noindent\textbf{Step 5 (Choose $\alpha$ to keep the coefficient positive).}  
Assumption \ref{ass:stepsize} guarantees $\alpha\le \mu_{\!h}/L$, which implies
\[
\frac{\alpha}{L_{H}}-\frac{L\alpha^{2}}{2\mu_{\!h}^{2}}
\;\ge\;
\frac{\alpha}{2L_{H}}. \tag{13}
\]
Thus (12) simplifies to
\[
f(x_{t+1})\le f(x_t)-\frac{\alpha}{2L_{H}}\norm{g_t}^{2}. \tag{14}
\]

\noindent\textbf{Step 6 (Take conditional expectation).}  
Condition on $x_t$ and use the law of total expectation:
\begin{align}
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
&\le f(x_t)-\frac{\alpha}{2L_{H}}\E\!\bigl[\norm{g_t}^{2}\mid x_t\bigl].
\tag{15}
\end{align}
Note that $g_t$ is deterministic given $x_t$ (we assumed exact gradients), so $\E[\norm{g_t}^{2}\mid x_t]=\norm{\nabla f(x_t)}^{2}$.

\noindent\textbf{Step 7 (Introduce Hessian variance).}  
The previous steps used only the mean of $H_t$.  To account for its stochasticity we refine the bound on $\inner{g_t, H_t^{-1}g_t}$:
\[
\inner{g_t, H_t^{-1}g_t}
= \inner{g_t, (\nabla^{2}f(x_t))^{-1}g_t}
   +\inner{g_t,\bigl(H_t^{-1}-(\nabla^{2}f(x_t))^{-1}\bigr)g_t}.
\tag{16}
\]
Taking expectation conditioned on $x_t$ and using unbiasedness \eqref{3},
\[
\E\bigl[\inner{g_t, H_t^{-1}g_t}\mid x_t\bigr]
= \inner{g_t, (\nabla^{2}f(x_t))^{-1}g_t}
  +\E\bigl[\inner{g_t,\bigl(H_t^{-1}-(\nabla^{2}f(x_t))^{-1}\bigr)g_t}\mid x_t\bigr].
\tag{17}
\]
The second term is bounded by Cauchy–Schwarz and Lemma \ref{lem:hessian}:
\begin{align}
\Bigl|\E\bigl[\inner{g_t,\bigl(H_t^{-1}-(\nabla^{2}f(x_t))^{-1}\bigr)g_t}\mid x_t\bigr]\Bigr|
&\le \norm{g_t}^{2}\,
   \E\bigl[\norm{H_t^{-1}-(\nabla^{2}f(x_t))^{-1}}\mid x_t\bigr] \nonumber\\
&\le \norm{g_t}^{2}\,
   \frac{\sigma_{H}}{\mu_{\!h}^{2}}. \tag{18}
\end{align}
(Here we used $\norm{A^{-1}}\le 1/\mu_{\!h}$ and that $\E\norm{H_t-\nabla^{2}f(x_t)}^{2}\le\sigma_H^{2}$ implies $\E\norm{H_t^{-1}-(\nabla^{2}f)^{-1}}\le\sigma_H/\mu_{\!h}^{2}$.)

\noindent\textbf{Step 8 (Replace the deterministic bound).}  
Insert (18) into (16)–(17) and use the lower eigenvalue bound $\mu I\preceq \nabla^{2}f(x_t)$ to obtain
\[
\E\bigl[\inner{g_t, H_t^{-1}g_t}\mid x_t\bigr]
\ge \frac{1}{L_{H}}\norm{g_t}^{2}
   -\frac{\sigma_{H}}{\mu_{\!h}^{2}}\norm{g_t}^{2}. \tag{19}
\]

\noindent\textbf{Step 9 (Re‑derive descent with variance term).}  
Repeating the algebra of Steps 1–4 but now using (19) gives
\begin{align}
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
&\le f(x_t)
   -\alpha\Bigl(\frac{1}{L_{H}}-\frac{\sigma_{H}}{\mu_{\!h}^{2}}\Bigr)\norm{g_t}^{2}
   +\frac{L\alpha^{2}}{2}\frac{1}{\mu_{\!h}^{2}}\norm{g_t}^{2}. \tag{20}
\end{align}
Applying the step‑size condition \eqref{5} again yields
\[
\E\!\bigl[f(x_{t+1})\mid x_t\bigr]
\le f(x_t)-\frac{\alpha}{2L_{H}}\norm{g_t}^{2}
      +\frac{L\alpha^{2}\sigma_{H}^{2}}{2\mu_{\!h}^{2}}. \tag{21}
\]

\noindent\textbf{Step 10 (Unconditional expectation).}  
Take full expectation on both sides of (21):
\[
\E\bigl[f(x_{t+1})\bigr]
\le \E\bigl[f(x_t)\bigr]
      -\frac{\alpha}{2L_{H}}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
      +\frac{L\alpha^{2}\sigma_{H}^{2}}{2\mu_{\!h}^{2}}. \tag{22}
\]

\noindent\textbf{Step 11 (Telescoping sum).}  
Sum \eqref{22} for $t=0,\dots,T-1$:
\[
\E\bigl[f(x_T)\bigr]-f(x_0)
\le -\frac{\alpha}{2L_{H}}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
   +\frac{L\alpha^{2}\sigma_{H}^{2}}{2\mu_{\!h}^{2}}T. \tag{23}
\]
Since $f(x_T)\ge f^{\star}$,
\[
\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le \frac{2L_{H}}{\alpha}\bigl(f(x_0)-f^{\star}\bigr)
   +\frac{L\alpha\sigma_{H}^{2}}{\mu_{\!h}^{2}}T. \tag{24}
\]

\noindent\textbf{Step 12 (Divide by $T$ and simplify constants).}  
Dividing (24) by $T$ and using $\alpha\le \mu_{\!h}/L$ gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le
\frac{2L_{H}\bigl(f(x_0)-f^{\star}\bigr)}{\alpha T}
\;+\;
\frac{L\sigma_{H}^{2}}{\mu_{\!h}^{2}}\,\frac{\alpha}{T}. \tag{25}
\]
Choosing the maximal admissible $\alpha=\mu_{\!h}/L$ (which minimises the RHS) yields the bound stated in \eqref{6}:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]
\le
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha\mu_{\!h}T}
\;+\;
\frac{L\sigma_{H}^{2}}{\mu_{\!h}^{2}}\frac{1}{T}.
\tag{26}
\]
\hfill\qedsymbol

\section*{Rate Derivation (With Constants)}
From (26) we observe two contributions:
\begin{enumerate}[label=(\alph*)]
    \item A \emph{deterministic} term that decays as $\mathcal{O}\!\bigl(1/(\alpha T)\bigr)$.  Larger $\alpha$ improves this term up to the limit \eqref{5}.
    \item A \emph{variance} term $\mathcal{O}\!\bigl(\sigma_{H}^{2}/T\bigr)$ independent of $\alpha$.
\end{enumerate}
Hence the overall expected squared gradient norm satisfies
\[
\E\!\bigl[\norm{\nabla f(x_t)}^{2}\bigr]\le
\frac{C_{1}}{t}+ \frac{C_{2}\sigma_{H}^{2}}{t},
\qquad
C_{1}= \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\alpha\mu_{\!h}},\;
C_{2}= \frac{L}{\mu_{\!h}^{2}}.
\]

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Exact Hessian ($\sigma_{H}=0$).}  The variance term disappears and we recover the classic deterministic Newton rate $\mathcal{O}(1/t)$ for non‑convex smooth functions.
    \item \textbf{Large mini‑batch.}  Since $\sigma_{H}^{2}\propto 1/b$, increasing the batch size $b$ reduces the second term linearly.
    \item \textbf{Strongly convex $f$.}  If additionally $f$ is $\mu$‑strongly convex, one can combine (26) with the Polyak–Łojasiewicz inequality to obtain linear convergence of the function values.
\end{itemize}
