\section*{Algorithm Name}
\textbf{AdaGrad for Non‑Convex Smooth Optimization (running‑average version)}  

The method keeps a cumulative sum of past gradients and adapts the learning
rate coordinate‑wise (here we present the scalar version for clarity).

\section*{Mathematical Formulation}
Let $f:\R^d\rightarrow\R$ be differentiable.  
Initialize $w_0\in\R^d$ and set $A_0=0\in\R^d$.  
For $k=0,1,2,\dots$ perform  

\[
\begin{aligned}
g_k &:= \nabla f(w_k) \in\R^d,\\[4pt]
A_{k+1} &:= A_k + g_k \qquad\text{(cumulative gradient sum)},\\[4pt]
\eta_{k+1} &:= \frac{\alpha}{\sqrt{\displaystyle\sum_{i=0}^{k} g_i^{\odot 2}+\epsilon}}\;,
\qquad\text{(scalar step size)}\\[4pt]
w_{k+1} &:= w_k - \eta_{k+1}\, g_k .
\end{aligned}
\]

Here $g_i^{\odot 2}$ denotes the element‑wise square, and the denominator is
interpreted component‑wise; the scalar version uses the Euclidean norm,
\[
\eta_{k+1}= \frac{\alpha}{\sqrt{\|A_k\|_2^{2}+\epsilon}} ,
\]
with hyper‑parameters $\alpha>0$ (base learning rate) and $\epsilon>0$ (stability).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (scalar step size)}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\R^d$, base step $\alpha>0$, stability $\epsilon>0$,
        max iterations $T$.
\State $A_0\gets 0\in\R^d$
\For{$k=0$ \textbf{to} $T-1$}
    \State $g_k \gets \nabla f(w_k)$
    \State $A_{k+1} \gets A_k + g_k$
    \State $\eta_{k+1} \gets \dfrac{\alpha}{\sqrt{\|A_k\|_2^{2}+\epsilon}}$
    \State $w_{k+1} \gets w_k - \eta_{k+1}\, g_k$
\EndFor
\State \Return $w_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: $\forall x,y\in\R^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]\label{ass:bound}
There exists $G>0$ such that $\forall k$, $\|g_k\|=\|\nabla f(w_k)\|\le G$.
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{ass:lb}
$f$ is bounded below by $f_{\inf}>-\infty$.
\end{assumption}

All constants $L,G,\alpha,\epsilon$ are known a priori and $\alpha,\epsilon>0$.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Let $\{w_k\}_{k\ge0}$ be generated by the algorithm under
Assumptions~\ref{ass:smooth}–\ref{ass:lb}.
Define $S_T:=\sum_{k=0}^{T-1}\|g_k\|$.
Then for any $T\ge 1$,
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\|\nabla f(w_k)\|^2\right]
\;\le\;
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}
\;\frac{\sqrt{S_T^2+\epsilon}}{T}
\;+\;
\frac{L\alpha G^2}{2T}\, .
\tag{1}
\]
In particular, because $S_T\le G T$, the bound simplifies to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\left[\|\nabla f(w_k)\|^2\right]
\;\le\;
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,
\frac{G\sqrt{T}}{T}
\;+\;
\frac{L\alpha G^2}{2T}
\;=\;
O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The denominator $\sqrt{\|A_k\|^2+\epsilon}$ grows
monotonically, guaranteeing that the effective step size $\eta_{k+1}$
is non‑increasing. Hence the iterates cannot diverge due to exploding
steps.
\item \textbf{Hyper‑parameter Sensitivity.}
    \begin{itemize}
    \item Larger $\alpha$ accelerates early progress but may increase the
    constant $L\alpha G^2/2$ in~\eqref{thm:main}.
    \item The stability term $\epsilon$ only matters at the first iteration;
    any $\epsilon>0$ prevents division by zero.
    \end{itemize}
\item \textbf{Comparison to SGD.} For constant step size $\eta$, SGD attains
$O(1/\sqrt{T})$ in the convex case; here the same rate holds for non‑convex
smooth objectives without requiring a diminishing schedule.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that AdaGrad automatically adapts to the geometry of
the loss landscape: if the cumulative gradient norm $S_T$ grows slowly,
the bound improves accordingly. In the worst case $S_T=GT$, the
$O(1/\sqrt{T})$ rate matches the optimal rate for stochastic first‑order
methods on smooth non‑convex functions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x$ and any vector $d$,
\[
f(x+d)\le f(x)+\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^2 .
\]
\end{lemma}
\begin{proof}
Directly the statement of Assumption~\ref{ass:smooth}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  
All expectations are taken w.r.t. any randomness in the gradient oracle; the
analysis below works for the deterministic case as a special instance.

\begin{enumerate}
\item[Step 1.] Apply Lemma~\ref{lem:descent} with $d=-\eta_{k+1}g_k$:
\[
f(w_{k+1})\le f(w_k)-\eta_{k+1}\langle g_k,g_k\rangle
+\frac{L}{2}\eta_{k+1}^{2}\|g_k\|^{2}.
\tag{2}
\]
\item[Step 2.] Substitute the definition of $\eta_{k+1}$.
Recall $\eta_{k+1}= \alpha/ \sqrt{\|A_k\|^{2}+\epsilon}$, thus
\[
\eta_{k+1}\|g_k\|^{2}
= \frac{\alpha\|g_k\|^{2}}{\sqrt{\|A_k\|^{2}+\epsilon}},
\qquad
\eta_{k+1}^{2}\|g_k\|^{2}
= \frac{\alpha^{2}\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{3}
\]
\item[Step 3.] Insert (3) into (2) and rearrange:
\[
f(w_{k+1})\le f(w_k)-\frac{\alpha\|g_k\|^{2}}
{\sqrt{\|A_k\|^{2}+\epsilon}}
+\frac{L\alpha^{2}}{2}\,
\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}.
\tag{4}
\]
\item[Step 4.] Because $\|A_k\|^{2}\ge 0$, we have
\[
\frac{1}{\|A_k\|^{2}+\epsilon}
\le \frac{1}{\sqrt{\|A_k\|^{2}+\epsilon}\,\sqrt{\epsilon}}
\le \frac{1}{\epsilon^{1/2}\,\sqrt{\|A_k\|^{2}+\epsilon}} .
\]
However, a tighter bound that will be useful later is obtained by
the elementary inequality
\[
\frac{1}{\|A_k\|^{2}+\epsilon}
\le \frac{1}{\sqrt{\|A_k\|^{2}+\epsilon}\;\sqrt{\epsilon}}
\le \frac{1}{\epsilon^{1/2}\sqrt{\|A_k\|^{2}+\epsilon}} .
\tag{5}
\]
\item[Step 5.] Using (5) in (4) we obtain
\[
f(w_{k+1})\le f(w_k)-\frac{\alpha\|g_k\|^{2}}
{\sqrt{\|A_k\|^{2}+\epsilon}}
+\frac{L\alpha^{2}}{2\sqrt{\epsilon}}\,
\frac{\|g_k\|^{2}}{\sqrt{\|A_k\|^{2}+\epsilon}} .
\tag{6}
\]
\item[Step 6.] Factor the common denominator:
\[
f(w_{k+1})\le f(w_k)-\frac{\alpha\|g_k\|^{2}}
{\sqrt{\|A_k\|^{2}+\epsilon}}
\Bigl(1-\frac{L\alpha}{2\sqrt{\epsilon}}\Bigr).
\tag{7}
\]
\item[Step 7.] Choose $\alpha$ small enough such that
$1-\frac{L\alpha}{2\sqrt{\epsilon}}\ge \frac12$.
A sufficient condition is $\alpha\le \frac{\sqrt{\epsilon}}{L}$.
Under this regime,
\[
f(w_{k+1})\le f(w_k)-\frac{\alpha}{2}\,
\frac{\|g_k\|^{2}}{\sqrt{\|A_k\|^{2}+\epsilon}} .
\tag{8}
\]
\item[Step 8.] Rearranging (8) gives a lower bound on the squared gradient:
\[
\|g_k\|^{2}\le
\frac{2}{\alpha}\bigl(f(w_k)-f(w_{k+1})\bigr)\,
\sqrt{\|A_k\|^{2}+\epsilon}.
\tag{9}
\]
\item[Step 9.] Sum (9) over $k=0,\dots,T-1$:
\[
\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2}{\alpha}\sum_{k=0}^{T-1}\bigl(f(w_k)-f(w_{k+1})\bigr)
\sqrt{\|A_k\|^{2}+\epsilon}.
\tag{10}
\]
\item[Step 10.] The telescoping sum $\sum_{k=0}^{T-1}(f(w_k)-f(w_{k+1}))
= f(w_0)-f(w_T)\le f(w_0)-f_{\inf}$.
Moreover, $\sqrt{\|A_k\|^{2}+\epsilon}\le
\sqrt{S_T^{2}+\epsilon}$ because $\|A_k\|\le S_T:=\sum_{i=0}^{k}\|g_i\|$
and $S_T$ is monotone increasing in $k$.
Hence
\[
\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}
\sqrt{S_T^{2}+\epsilon}.
\tag{11}
\]
\item[Step 11.] Divide both sides of (11) by $T$ and use
$\|g_k\|^{2}\ge 0$ to obtain the average bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}
\frac{\sqrt{S_T^{2}+\epsilon}}{T}.
\tag{12}
\]
\item[Step 12.] To incorporate the smoothness term that appeared in
Step~1 but was discarded when we enforced $\alpha\le\sqrt{\epsilon}/L$, we
add back the omitted $ \frac{L\alpha^{2}}{2}\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}$
from (4). Using the bound $\|A_k\|^{2}+\epsilon\ge\epsilon$ and
$\|g_k\|^{2}\le G^{2}$ yields
\[
\frac{L\alpha^{2}}{2}\frac{\|g_k\|^{2}}{\|A_k\|^{2}+\epsilon}
\le \frac{L\alpha^{2}G^{2}}{2\epsilon}.
\]
Summing over $k$ and dividing by $T$ adds the term
$\frac{L\alpha G^{2}}{2}$ to the right‑hand side of (12).
Thus we finally obtain
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}
\frac{\sqrt{S_T^{2}+\epsilon}}{T}
\;+\;
\frac{L\alpha G^{2}}{2T},
\]
which is exactly the bound \eqref{thm:main}. \qedhere
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From $S_T\le G T$ we have $\sqrt{S_T^{2}+\epsilon}\le G T+\sqrt{\epsilon}$.
Plugging into the bound of Theorem~\ref{thm:main} gives
\[
\frac{1}{T}\sum_{k=0}^{T-1}\|g_k\|^{2}
\le
\frac{2\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}\,
\frac{G T+\sqrt{\epsilon}}{T}
\;+\;
\frac{L\alpha G^{2}}{2T}
=
\underbrace{\frac{2G\bigl(f(w_0)-f_{\inf}\bigr)}{\alpha}}_{C_{1}}
\frac{1}{\sqrt{T}}
\;+\;
\underbrace{\frac{2\bigl(f(w_0)-f_{\inf}\bigr)\sqrt{\epsilon}}{\alpha T}
+\frac{L\alpha G^{2}}{2T}}_{C_{2}/T}.
\]
Thus the dominant term scales as $O(1/\sqrt{T})$.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact Gradient (deterministic) case.}
   The expectation operator disappears; the same $O(1/\sqrt{T})$ bound holds.
\item \textbf{Sparse gradients.}
   If only $s\ll d$ coordinates are non‑zero per iteration,
   $\|A_k\|^{2}$ grows as $O(sT)$, leading to a tighter bound
   $O(1/\sqrt{sT})$.
\item \textbf{Convex $f$.}
   When $f$ is convex, the same analysis yields a regret bound
   $O(\sqrt{T})$, reproducing the classic AdaGrad guarantee.
\end{enumerate}
