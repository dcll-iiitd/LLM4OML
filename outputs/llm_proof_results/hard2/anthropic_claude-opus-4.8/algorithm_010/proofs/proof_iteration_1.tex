\section*{Algorithm Name}
\textbf{Extra-Gradient Method (EG) for Convex-Concave Saddle Point Problems}

We consider the saddle point (min-max) problem
\[
\min_{x \in \mathbb{R}^n} \max_{y \in \mathbb{R}^m} \; L(x,y),
\]
where $L$ is convex in $x$ and concave in $y$. (We treat the
$\max_x \min_y$ formulation symmetrically; by sign convention this is the
same as the standard $\min_x \max_y$ analysis.)

\section*{Mathematical Formulation}

Let $z = (x,y) \in \mathbb{R}^{n+m}$ be the joint variable. Define the
\emph{monotone operator} (saddle gradient field)
\[
F(z) = \begin{pmatrix} \nabla_x L(x,y) \\ -\nabla_y L(x,y) \end{pmatrix}.
\]
A saddle point $z^\star = (x^\star,y^\star)$ satisfies the variational
inequality
\[
\langle F(z^\star), z - z^\star \rangle \geq 0 \quad \forall z.
\]

\paragraph{Update rules (operator form).}
With step size $\eta > 0$, the Extra-Gradient method performs:
\begin{align}
z_{k+1/2} &= z_k - \eta\, F(z_k) && \text{(extrapolation / prediction step)}\\
z_{k+1}   &= z_k - \eta\, F(z_{k+1/2}) && \text{(correction step using the midpoint gradient)}.
\end{align}

\paragraph{Update rules (componentwise).}
\begin{align}
x_{k+1/2} &= x_k - \eta\, \nabla_x L(x_k,y_k), &
y_{k+1/2} &= y_k + \eta\, \nabla_y L(x_k,y_k),\\
x_{k+1}   &= x_k - \eta\, \nabla_x L(x_{k+1/2},y_{k+1/2}), &
y_{k+1}   &= y_k + \eta\, \nabla_y L(x_{k+1/2},y_{k+1/2}).
\end{align}

\paragraph{Hyperparameters.}
\begin{itemize}
\item Step size $\eta > 0$, required to satisfy $\eta \le \tfrac{1}{L_{\mathrm{Lip}}}$
where $L_{\mathrm{Lip}}$ is the Lipschitz constant of $F$.
\item Number of iterations $K$.
\item The averaged iterate $\bar z_K = \frac{1}{K}\sum_{k=0}^{K-1} z_{k+1/2}$
is returned (ergodic averaging).
\end{itemize}

\section*{Pseudocode}
\begin{verbatim}
Input: initial point z_0 = (x_0, y_0), step size eta, iterations K
Initialize: S = 0   (running sum for averaging)
for k = 0, 1, ..., K-1 do
    # Extrapolation step
    g_k       = F(z_k)                  # F(z) = (grad_x L, -grad_y L)
    z_{k+1/2} = z_k - eta * g_k
    # Correction step
    g_mid     = F(z_{k+1/2})
    z_{k+1}   = z_k - eta * g_mid
    # Accumulate for ergodic average
    S = S + z_{k+1/2}
    # Termination check
    if ||g_mid|| <= tol then break
end for
Output: averaged iterate  z_bar = S / K   (or last z_{k+1})
\end{verbatim}

\section*{Dry Run Example}

To illustrate the mechanics on a scalar example we use the convex
function $f(w)=(w-3)^2$ (a pure minimization, i.e. the $y$-block is
absent and $F(w)=f'(w)=2(w-3)$). Parameters: $w_0=0$, $\eta=0.1$, $3$ iterations.

\paragraph{Iteration $k=0$.}
\begin{align*}
g_0 &= f'(w_0) = 2(0-3) = -6,\\
w_{1/2} &= w_0 - \eta g_0 = 0 - 0.1(-6) = 0.6,\\
g_{\text{mid}} &= f'(w_{1/2}) = 2(0.6-3) = -4.8,\\
w_1 &= w_0 - \eta g_{\text{mid}} = 0 - 0.1(-4.8) = 0.48,\\
f(w_1) &= (0.48-3)^2 = 6.3504.
\end{align*}

\paragraph{Iteration $k=1$.}
\begin{align*}
g_1 &= f'(w_1) = 2(0.48-3) = -5.04,\\
w_{3/2} &= w_1 - \eta g_1 = 0.48 - 0.1(-5.04) = 0.984,\\
g_{\text{mid}} &= f'(w_{3/2}) = 2(0.984-3) = -4.032,\\
w_2 &= w_1 - \eta g_{\text{mid}} = 0.48 - 0.1(-4.032) = 0.8832,\\
f(w_2) &= (0.8832-3)^2 = 4.4808.
\end{align*}

\paragraph{Iteration $k=2$.}
\begin{align*}
g_2 &= f'(w_2) = 2(0.8832-3) = -4.2336,\\
w_{5/2} &= w_2 - \eta g_2 = 0.8832 - 0.1(-4.2336) = 1.30656,\\
g_{\text{mid}} &= f'(w_{5/2}) = 2(1.30656-3) = -3.38688,\\
w_3 &= w_2 - \eta g_{\text{mid}} = 0.8832 - 0.1(-3.38688) = 1.221888,\\
f(w_3) &= (1.221888-3)^2 \approx 3.16111.
\end{align*}

The objective decreases monotonically toward the optimum $w^\star=3$,
$f(w^\star)=0$.

\section*{Assumptions}

\begin{assumption}[Convexity-Concavity]
$L(x,y)$ is convex in $x$ for every fixed $y$, and concave in $y$ for
every fixed $x$. Equivalently the operator $F$ is \emph{monotone}:
\[
\langle F(z) - F(z'),\, z - z' \rangle \geq 0 \qquad \forall z,z'.
\]
\end{assumption}

\begin{assumption}[Lipschitz smoothness]
$F$ is $L_{\mathrm{Lip}}$-Lipschitz continuous:
\[
\| F(z) - F(z') \| \leq L_{\mathrm{Lip}} \, \| z - z' \| \qquad \forall z,z'.
\]
\end{assumption}

\begin{assumption}[Existence of a saddle point]
There exists $z^\star=(x^\star,y^\star)$ such that
$L(x^\star,y) \le L(x^\star,y^\star) \le L(x,y^\star)$ for all $x,y$.
Equivalently $\langle F(z^\star), z-z^\star\rangle \ge 0$ for all $z$.
\end{assumption}

\begin{assumption}[Step size]
The step size satisfies $0 < \eta \le \dfrac{1}{L_{\mathrm{Lip}}}$.
\end{assumption}

\noindent We measure convergence through the \emph{primal-dual gap}
\[
\mathrm{Gap}(\bar x,\bar y) \;=\;
\max_{y} L(\bar x, y) - \min_{x} L(x, \bar y).
\]

\section*{Main Convergence Theorem}

\begin{theorem}[Ergodic $O(1/K)$ rate]
\label{thm:main}
Under Assumptions 1--4, let the iterates be generated by the
Extra-Gradient method and define the ergodic average
$\bar z_K = \frac{1}{K}\sum_{k=0}^{K-1} z_{k+1/2}$, $\bar z_K=(\bar x_K,\bar y_K)$.
Then for any compact feasible set of radius $D$ around $z^\star$,
\[
\mathrm{Gap}(\bar x_K, \bar y_K)
\;\le\; \frac{\| z_0 - z^\star \|^2}{2\eta K}
\;=\; O\!\left(\frac{1}{K}\right).
\]
Moreover the iterates remain bounded:
$\|z_{k+1}-z^\star\| \le \|z_k - z^\star\|$.
\end{theorem}

\section*{Theoretical Properties}

\begin{itemize}
\item \textbf{Stability.} Each step is non-expansive toward $z^\star$
(Lemma~\ref{lem:contraction}); unlike plain Gradient Descent-Ascent (GDA),
which can diverge on bilinear games, EG is stable.
\item \textbf{Rate.} The ergodic gap converges as $O(1/K)$, optimal for
monotone Lipschitz problems with first-order oracles.
\item \textbf{Hyperparameter sensitivity.} Convergence requires
$\eta \le 1/L_{\mathrm{Lip}}$. Larger $\eta$ may violate the descent
inequality [Step 7] and break monotonicity.
\item \textbf{Comparison to baselines.} GDA achieves only $O(\eta)$
neighborhood convergence (or diverges); EG's extrapolation cancels the
rotational component of the dynamics, recovering true convergence.
\end{itemize}

\section*{Discussion}
The extra (midpoint) gradient evaluation gives EG an implicit
look-ahead, effectively approximating the proximal point method to second
order. This yields global convergence on the entire monotone class at the
cost of two gradient evaluations per iteration.

\section*{Preliminaries (Definitions and Lemmas)}

We repeatedly use the \textbf{three-point identity}: for any vectors $a,b,c$,
\begin{equation}
\label{eq:3pt}
2\langle a-b,\, a-c\rangle = \|a-b\|^2 + \|a-c\|^2 - \|b-c\|^2.
\end{equation}

\begin{lemma}[Monotonicity from convexity-concavity]
\label{lem:mono}
If $L$ is convex-concave then $F$ is monotone, i.e.
$\langle F(z)-F(z'), z-z'\rangle \ge 0$.
\end{lemma}
\begin{proof}
Write $z=(x,y)$, $z'=(x',y')$. Convexity in $x$ gives
$\langle \nabla_x L(x,y)-\nabla_x L(x',y),\, x-x'\rangle\ge0$ and similarly
along $y$ using concavity (note the sign flip in $F$'s second block makes
$-\nabla_y L$ monotone). Summing the blocks yields the claim.
\end{proof}

\begin{lemma}[Optimality of $z^\star$]
\label{lem:opt}
$\langle F(z^\star),\, z-z^\star\rangle \ge 0$ for all $z$, and by
monotonicity $\langle F(z),\, z-z^\star\rangle \ge 0$ for all $z$.
\end{lemma}

\section*{Main Proof (Step-by-Step Derivation)}

Fix an arbitrary comparison point $z$ (later set to $z^\star$ or used in the
gap argument). Denote $w_k = z_{k+1/2}$ for brevity. Recall:
\[
w_k = z_k - \eta F(z_k), \qquad z_{k+1} = z_k - \eta F(w_k).
\]

\paragraph{[Step 1] Three-point identity on the correction step.}
Apply \eqref{eq:3pt} with $a=z_{k+1}$, $b=z_k$, $c=z$. But it is cleaner to
expand the squared distance directly. Using $z_{k+1}=z_k-\eta F(w_k)$,
\begin{align}
\|z_{k+1}-z\|^2
&= \|z_k - \eta F(w_k) - z\|^2 \nonumber\\
&= \|z_k - z\|^2 - 2\eta \langle F(w_k),\, z_k - z\rangle
   + \eta^2 \|F(w_k)\|^2. \label{eq:s1}
\end{align}

\paragraph{[Step 2] Decompose the inner product through $w_k$.}
Write $z_k - z = (z_k - w_k) + (w_k - z)$, so
\begin{equation}
\label{eq:s2}
\langle F(w_k),\, z_k - z\rangle
= \langle F(w_k),\, w_k - z\rangle
+ \langle F(w_k),\, z_k - w_k\rangle.
\end{equation}

\paragraph{[Step 3] Substitute \eqref{eq:s2} into \eqref{eq:s1}.}
\begin{align}
\|z_{k+1}-z\|^2
&= \|z_k - z\|^2
   - 2\eta \langle F(w_k),\, w_k - z\rangle \nonumber\\
&\quad - 2\eta \langle F(w_k),\, z_k - w_k\rangle
   + \eta^2 \|F(w_k)\|^2. \label{eq:s3}
\end{align}

\paragraph{[Step 4] Handle the prediction step.}
From $w_k = z_k - \eta F(z_k)$ we have
$z_k - w_k = \eta F(z_k)$. Hence
\begin{equation}
\label{eq:s4}
-2\eta\langle F(w_k),\, z_k - w_k\rangle
= -2\eta\langle F(w_k),\, \eta F(z_k)\rangle
= -2\eta^2 \langle F(w_k),\, F(z_k)\rangle.
\end{equation}

\paragraph{[Step 5] Combine the two $\eta^2$ terms.}
Insert \eqref{eq:s4} into \eqref{eq:s3}:
\begin{align}
\|z_{k+1}-z\|^2
&= \|z_k - z\|^2 - 2\eta \langle F(w_k),\, w_k - z\rangle \nonumber\\
&\quad + \eta^2\|F(w_k)\|^2 - 2\eta^2\langle F(w_k),F(z_k)\rangle.
\label{eq:s5}
\end{align}
Complete the square on the last two $\eta^2$ terms by adding and
subtracting $\eta^2\|F(z_k)\|^2$:
\begin{align}
\eta^2\|F(w_k)\|^2 - 2\eta^2\langle F(w_k),F(z_k)\rangle
&= \eta^2\|F(w_k)-F(z_k)\|^2 - \eta^2\|F(z_k)\|^2. \label{eq:s5b}
\end{align}

\paragraph{[Step 6] Re-express $\eta^2\|F(z_k)\|^2$ via the prediction step.}
Since $w_k - z_k = -\eta F(z_k)$, we have
$\|w_k - z_k\|^2 = \eta^2 \|F(z_k)\|^2$. Substituting \eqref{eq:s5b}
and this identity into \eqref{eq:s5}:
\begin{align}
\|z_{k+1}-z\|^2
&= \|z_k - z\|^2 - 2\eta \langle F(w_k),\, w_k - z\rangle \nonumber\\
&\quad + \eta^2\|F(w_k)-F(z_k)\|^2 - \|w_k - z_k\|^2. \label{eq:s6}
\end{align}

\paragraph{[Step 7] Bound the error term using Lipschitzness.}
By Assumption 2, $\|F(w_k)-F(z_k)\| \le L_{\mathrm{Lip}}\|w_k - z_k\|$, so
\[
\eta^2\|F(w_k)-F(z_k)\|^2 \le \eta^2 L_{\mathrm{Lip}}^2\|w_k-z_k\|^2.
\]
Therefore the last two terms of \eqref{eq:s6} satisfy
\begin{equation}
\label{eq:s7}
\eta^2\|F(w_k)-F(z_k)\|^2 - \|w_k - z_k\|^2
\le -\big(1-\eta^2 L_{\mathrm{Lip}}^2\big)\,\|w_k - z_k\|^2 \;\le\; 0,
\end{equation}
where the final inequality uses $\eta \le 1/L_{\mathrm{Lip}}$
(Assumption 4), giving $1-\eta^2 L_{\mathrm{Lip}}^2 \ge 0$.

\paragraph{[Step 8] Key descent inequality.}
Combining \eqref{eq:s6} and \eqref{eq:s7},
\begin{equation}
\label{eq:s8}
\|z_{k+1}-z\|^2 \le \|z_k - z\|^2
- 2\eta \langle F(w_k),\, w_k - z\rangle
- \big(1-\eta^2 L_{\mathrm{Lip}}^2\big)\|w_k-z_k\|^2.
\end{equation}

\paragraph{[Step 9] Non-expansiveness (set $z=z^\star$).}
By Lemma~\ref{lem:opt} with monotonicity,
$\langle F(w_k),\, w_k - z^\star\rangle \ge \langle F(z^\star), w_k - z^\star\rangle \ge 0$.
Hence from \eqref{eq:s8},
\begin{equation}
\label{eq:s9}
\|z_{k+1}-z^\star\|^2 \le \|z_k - z^\star\|^2,
\end{equation}
proving the iterates are bounded (stability claim of Theorem~\ref{thm:main}).

\paragraph{[Step 10] Convert the operator inner product to a gap term.}
By monotonicity, for arbitrary $z$,
\[
\langle F(w_k),\, w_k - z\rangle \ge \langle F(z),\, w_k - z\rangle.
\]
For the saddle problem, the right-hand side relates to the function gap.
Specifically, with $z=(x,y)$ and $w_k=(x_{k+1/2},y_{k+1/2})$, convexity-concavity gives
\begin{equation}
\label{eq:s10}
\langle F(w_k),\, w_k - z\rangle
\;\ge\; L(x_{k+1/2},y) - L(x, y_{k+1/2}).
\end{equation}
This is the standard variational-inequality lower bound: convexity in $x$
gives $L(x_{k+1/2},y)-L(x,y)\le \langle\nabla_xL(w_k),x_{k+1/2}-x\rangle$
and concavity in $y$ gives the matching $y$-term; summing yields
\eqref{eq:s10}.

\paragraph{[Step 11] Rearrange \eqref{eq:s8} and drop the negative term.}
Using \eqref{eq:s10} and dropping the non-positive last term of
\eqref{eq:s8}:
\begin{equation}
\label{eq:s11}
2\eta\big( L(x_{k+1/2},y) - L(x, y_{k+1/2})\big)
\le \|z_k - z\|^2 - \|z_{k+1}-z\|^2.
\end{equation}

\paragraph{[Step 12] Telescoping over $k=0,\dots,K-1$.}
Sum \eqref{eq:s11}; the right side telescopes:
\begin{equation}
\label{eq:s12}
2\eta \sum_{k=0}^{K-1}\big( L(x_{k+1/2},y) - L(x, y_{k+1/2})\big)
\le \|z_0 - z\|^2 - \|z_K - z\|^2 \le \|z_0 - z\|^2.
\end{equation}

\paragraph{[Step 13] Jensen's inequality (convexity-concavity of averages).}
Let $\bar x_K = \frac1K\sum_k x_{k+1/2}$, $\bar y_K = \frac1K\sum_k y_{k+1/2}$.
By convexity in $x$ and concavity in $y$,
\[
L(\bar x_K, y) \le \frac1K\sum_k L(x_{k+1/2}, y),
\qquad
L(x,\bar y_K) \ge \frac1K\sum_k L(x, y_{k+1/2}).
\]
Therefore
\begin{equation}
\label{eq:s13}
L(\bar x_K, y) - L(x, \bar y_K)
\le \frac1K\sum_{k=0}^{K-1}\big(L(x_{k+1/2},y)-L(x,y_{k+1/2})\big).
\end{equation}

\paragraph{[Step 14] Combine \eqref{eq:s12} and \eqref{eq:s13}.}
\begin{equation}
\label{eq:s14}
L(\bar x_K, y) - L(x, \bar y_K)
\le \frac{\|z_0 - z\|^2}{2\eta K}.
\end{equation}

\paragraph{[Step 15] Take the supremum to form the gap.}
Maximizing over $y$ and minimizing over $x$ on a ball of radius $D$ about
$z^\star$ (so $\|z_0-z\|^2 \le \|z_0-z^\star\|^2 + \text{const}$; in the
unconstrained case set $z=z^\star$ for each block), we obtain
\begin{equation}
\label{eq:s15}
\mathrm{Gap}(\bar x_K, \bar y_K)
= \max_y L(\bar x_K, y) - \min_x L(x,\bar y_K)
\le \frac{\|z_0 - z^\star\|^2}{2\eta K}.
\end{equation}
This proves Theorem~\ref{thm:main}. \hfill$\square$

\section*{Rate Derivation (With Constants)}

Choosing the largest admissible step $\eta = 1/L_{\mathrm{Lip}}$ in
\eqref{eq:s15}:
\begin{equation}
\mathrm{Gap}(\bar x_K, \bar y_K)
\le \frac{L_{\mathrm{Lip}}\,\|z_0 - z^\star\|^2}{2 K}
= O\!\left(\frac{L_{\mathrm{Lip}} D_0^2}{K}\right),
\end{equation}
where $D_0 = \|z_0 - z^\star\|$. Thus to reach gap $\le \varepsilon$ one needs
\[
K \ge \frac{L_{\mathrm{Lip}} D_0^2}{2\varepsilon}
\]
iterations, i.e. $O(1/\varepsilon)$ complexity.

\section*{Special Cases}

\begin{corollary}[Pure minimization]
If $L(x,y)=f(x)$ does not depend on $y$ and $f$ is convex
$L_{\mathrm{Lip}}$-smooth, then $F=\nabla f$ and \eqref{eq:s15} reduces to
\[
f(\bar x_K) - f(x^\star) \le \frac{\|x_0-x^\star\|^2}{2\eta K} = O(1/K),
\]
recovering the standard $O(1/K)$ rate for smooth convex minimization.
\end{corollary}

\begin{corollary}[Bilinear games]
For $L(x,y)=x^\top A y$ we have $F(z)=(Ay,\,-A^\top x)$, which is monotone
($\langle F(z)-F(z'),z-z'\rangle=0$) and Lipschitz with
$L_{\mathrm{Lip}}=\|A\|$. EG converges at $O(1/K)$ on the gap, whereas plain
GDA diverges (its iterates spiral outward). This is the canonical example
showing the necessity of the extrapolation step.
\end{corollary}

\begin{corollary}[Strongly monotone case]
If in addition $F$ is $\mu$-strongly monotone
($\langle F(z)-F(z'),z-z'\rangle \ge \mu\|z-z'\|^2$), then \eqref{eq:s8}
strengthens to a contraction
\[
\|z_{k+1}-z^\star\|^2 \le (1-c)\,\|z_k-z^\star\|^2
\]
for some $c=c(\mu,\eta)\in(0,1)$, yielding linear (geometric) convergence
$\|z_K-z^\star\|^2 \le (1-c)^K\|z_0-z^\star\|^2$.
\end{corollary}
