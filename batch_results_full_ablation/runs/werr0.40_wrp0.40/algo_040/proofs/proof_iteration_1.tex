\section*{Gradient Descent–Ascent (GDA)}

\section*{Mathematical Formulation}
Let $L:\mathcal X\times\mathcal Y\rightarrow\mathbb R$ be a continuously differentiable function.
We consider the saddle‑point problem  

\[
\min_{y\in\mathcal Y}\max_{x\in\mathcal X} L(x,y).
\]

Define the gradient operator  

\[
F(z)\;:=\;
\begin{pmatrix}
\nabla_{x}L(x,y)\\[2mm]
-\nabla_{y}L(x,y)
\end{pmatrix},
\qquad 
z:=\begin{pmatrix}x\\y\end{pmatrix}\in\mathcal Z:=\mathcal X\times\mathcal Y .
\]

Given a stepsize $\eta>0$, the \textbf{Gradient Descent–Ascent} iteration is  

\[
\boxed{
\begin{aligned}
x_{t+1}&=x_{t}+ \eta \,\nabla_{x}L(x_{t},y_{t}),\\
y_{t+1}&=y_{t}- \eta \,\nabla_{y}L(x_{t},y_{t}),
\end{aligned}}
\qquad  t=0,1,2,\dots
\tag{1}
\]

Equivalently, in the concatenated form  

\[
z_{t+1}=z_{t}-\eta\,F(z_{t}). \tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta>0$, initial pair $(x_{0},y_{0})$,
maximum iterations $T$ (or tolerance).
\FOR{$t=0$ \textbf{to} $T-1$}
    \STATE Compute gradients 
    $\;g_{x}^{t}\gets \nabla_{x}L(x_{t},y_{t})$,
    $\;g_{y}^{t}\gets \nabla_{y}L(x_{t},y_{t})$.
    \STATE Update primal variable (ascent) 
    $\;x_{t+1}\gets x_{t}+ \eta\, g_{x}^{t}$.
    \STATE Update dual variable (descent) 
    $\;y_{t+1}\gets y_{t}- \eta\, g_{y}^{t}$.
\ENDFOR
\STATE \textbf{Output:} $(x_{T},y_{T})$ (or the averaged iterate 
$\bar z_{T}= \frac1T\sum_{t=0}^{T-1}z_{t}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Although GDA is designed for a min–max problem, we illustrate the arithmetic on the simple
univariate quadratic $f(w)=(w-3)^{2}$ (pure minimisation).  
We treat $x=w$ and perform only the descent part of (1), i.e. $w_{t+1}=w_{t}-\eta\nabla f(w_{t})$,
with $w_{0}=0$, $\eta=0.1$, for three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_{t}$ & $\nabla f(w_{t})=2(w_{t}-3)$ & $w_{t+1}=w_{t}-\eta\nabla f(w_{t})$\\\hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$\\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$\\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$\\
3 & $1.464$ & $2(1.464-3)=-3.072$ & $1.464-0.1(-3.072)=1.7712$
\end{tabular}
\end{center}

The objective values $f(w_{t})$ decrease from $9$ to $3.30$ after three steps, confirming the
mechanics of the update rule.

\section*{Assumptions}
\begin{assumption}[Convex–concave structure]\label{ass:cc}
\begin{enumerate}[label=(\alph*)]
\item For every $y\in\mathcal Y$, $x\mapsto L(x,y)$ is convex.
\item For every $x\in\mathcal X$, $y\mapsto L(x,y)$ is concave.
\end{enumerate}
\end{assumption}

\begin{assumption}[Smoothness]\label{ass:smooth}
There exists $L>0$ such that for all $z,z'\in\mathcal Z$,
\[
\|F(z)-F(z')\|\le L\|z-z'\| .
\tag{3}
\]
Equivalently, each partial gradient is $L$‑Lipschitz:
\[
\|\nabla_{x}L(x,y)-\nabla_{x}L(x',y')\|\le L\|(x,y)-(x',y')\|,
\]
and the same for $-\nabla_{y}L$.
\end{assumption}

\begin{assumption}[Existence of a saddle point]\label{ass:saddle}
There exists $(x^{\star},y^{\star})\in\mathcal X\times\mathcal Y$ such that
\[
L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star}),\qquad\forall x\in\mathcal X,\;y\in\mathcal Y .
\tag{4}
\]
Denote $z^{\star}=(x^{\star},y^{\star})$.
\end{assumption}

\begin{assumption}[Stepsize]\label{ass:stepsize}
The constant stepsize satisfies
\[
0<\eta\le\frac{1}{L}.
\tag{5}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/T)$ rate for GDA]\label{thm:main}
Under Assumptions \ref{ass:cc}–\ref{ass:stepsize}, let $\{z_{t}\}_{t\ge0}$ be the iterates generated by
\eqref{1} with $z_{0}$ arbitrary. Define the averaged iterate
\[
\bar z_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}z_{t},
\qquad
\bar x_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}x_{t},
\;\;
\bar y_{T}:=\frac{1}{T}\sum_{t=0}^{T-1}y_{t}.
\]
Then for any saddle point $z^{\star}$,
\[
\underbrace{\max_{y\in\mathcal Y}L(\bar x_{T},y)-\min_{x\in\mathcal X}L(x,\bar y_{T})}_{\text{primal–dual gap}}
\;\le\;
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta T}.
\tag{6}
\]
Consequently $\bar z_{T}$ converges to the set of saddle points and the gap decays as $O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $F$ is $L$‑Lipschitz and $\eta\le1/L$, the map $z\mapsto z-\eta F(z)$ is
non‑expansive; the sequence $\{\|z_{t}-z^{\star}\|\}$ is monotonically non‑increasing up to a
$O(\eta^{2})$ correction (see Lemma~\ref{lem:one-step}).
\item \textbf{Hyperparameter sensitivity.} The bound \eqref{6} shows linear dependence on $\eta^{-1}$. Choosing a
smaller $\eta$ improves the constant but slows practical progress; the optimal
choice in the worst case is $\eta=1/L$, which yields the tightest $O(1/T)$ rate.
\item \textbf{Comparison to baselines.} For pure convex minimisation, standard gradient descent
achieves the same $O(1/T)$ rate under identical smoothness. The GDA iteration
extends this guarantee to convex–concave saddle‑point problems without requiring
proximal operators or extra momentum terms.
\end{itemize}

\section*{Discussion}
The result holds for deterministic gradients. In stochastic settings one obtains an
additional variance term, leading to an $O(1/\sqrt{T})$ rate unless variance reduction
techniques are employed. Extensions to accelerated schemes (e.g. extragradient)
can improve the rate to $O(1/T^{2})$ under stronger monotonicity assumptions.

\section*{Preliminaries (Definitions and Lemmas)}

\begin{definition}[Monotone operator]
A mapping $F:\mathcal Z\to\mathcal Z$ is monotone if
\[
\langle F(z)-F(z'),\,z-z'\rangle\ge0,\qquad\forall z,z'\in\mathcal Z .
\tag{7}
\]
\end{definition}

\begin{lemma}[Monotonicity of $F$ under convex–concave]\label{lem:monotone}
If $L$ satisfies Assumption~\ref{ass:cc}, then $F$ defined in (2) is monotone.
\end{lemma}
\begin{proof}
For any $z=(x,y)$ and $z'=(x',y')$,
\[
\begin{aligned}
\langle F(z)-F(z'),z-z'\rangle
&=\langle\nabla_{x}L(x,y)-\nabla_{x}L(x',y'),\,x-x'\rangle\\
&\quad-\langle\nabla_{y}L(x,y)-\nabla_{y}L(x',y'),\,y-y'\rangle .
\end{aligned}
\]
Convexity of $x\mapsto L(x,y')$ gives
\[
\langle\nabla_{x}L(x',y')-\nabla_{x}L(x,y'),\,x-x'\rangle\ge0,
\]
and concavity of $y\mapsto L(x',y)$ gives
\[
\langle\nabla_{y}L(x',y)-\nabla_{y}L(x',y'),\,y-y'\rangle\le0.
\]
Adding the two inequalities and using the fact that
$\nabla_{x}L(x,y)-\nabla_{x}L(x,y')$ and
$\nabla_{y}L(x,y)-\nabla_{y}L(x',y)$ are each non‑negative in the inner product with
$x-x'$ or $y-y'$ yields the desired monotonicity. ∎
\end{proof}

\begin{lemma}[One‑step inequality]\label{lem:one-step}
Let $z_{t+1}=z_{t}-\eta F(z_{t})$ with $\eta\le1/L$. Then for any $z^{\star}$,
\[
\|z_{t+1}-z^{\star}\|^{2}
\le
\|z_{t}-z^{\star}\|^{2}
-2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
+\eta^{2}L^{2}\|z_{t}-z^{\star}\|^{2}.
\tag{8}
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\|z_{t+1}-z^{\star}\|^{2}
&=\|z_{t}-\eta F(z_{t})-z^{\star}\|^{2}\\
&=\|z_{t}-z^{\star}\|^{2}
-2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
+\eta^{2}\|F(z_{t})\|^{2}.
\end{aligned}
\]
By Lipschitz continuity \eqref{3},
$\|F(z_{t})\|\le L\|z_{t}-z^{\star}\|$ because $F(z^{\star})=0$ (optimality condition). Substituting
gives \eqref{8}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}

We prove Theorem~\ref{thm:main}. Throughout the proof we fix an arbitrary saddle point
$z^{\star}=(x^{\star},y^{\star})$.

\begin{enumerate}
\item[Step 1.] \textbf{Apply Lemma~\ref{lem:one-step}} to the iterates:
\[
\|z_{t+1}-z^{\star}\|^{2}
\le
\|z_{t}-z^{\star}\|^{2}
-2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
+\eta^{2}L^{2}\|z_{t}-z^{\star}\|^{2}.
\tag{9}
\]

\item[Step 2.] \textbf{Rearrange \eqref{9}} and move the term with $\langle F(z_{t}),z_{t}-z^{\star}\rangle$ to the left:
\[
2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le
\|z_{t}-z^{\star}\|^{2}-\|z_{t+1}-z^{\star}\|^{2}
+\eta^{2}L^{2}\|z_{t}-z^{\star}\|^{2}.
\tag{10}
\]

\item[Step 3.] \textbf{Use the stepsize condition} $\eta\le1/L$ to bound the last term:
\[
\eta^{2}L^{2}\|z_{t}-z^{\star}\|^{2}\le\eta\|z_{t}-z^{\star}\|^{2}.
\tag{11}
\]

\item[Step 4.] \textbf{Insert \eqref{11} into \eqref{10}}:
\[
2\eta\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le
\|z_{t}-z^{\star}\|^{2}-\|z_{t+1}-z^{\star}\|^{2}
+\eta\|z_{t}-z^{\star}\|^{2}.
\tag{12}
\]

\item[Step 5.] \textbf{Divide by $2\eta$}:
\[
\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le
\frac{1}{2\eta}\Bigl(\|z_{t}-z^{\star}\|^{2}-\|z_{t+1}-z^{\star}\|^{2}\Bigr)
+\frac{1}{2}\|z_{t}-z^{\star}\|^{2}.
\tag{13}
\]

\item[Step 6.] \textbf{Sum \eqref{13} over $t=0,\dots,T-1$}.
The telescoping sum of the first term yields
\[
\sum_{t=0}^{T-1}\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le
\frac{1}{2\eta}\bigl(\|z_{0}-z^{\star}\|^{2}-\|z_{T}-z^{\star}\|^{2}\bigr)
+\frac{1}{2}\sum_{t=0}^{T-1}\|z_{t}-z^{\star}\|^{2}.
\tag{14}
\]

\item[Step 7.] \textbf{Drop the non‑negative term $\|z_{T}-z^{\star}\|^{2}$} to obtain a looser but simpler bound:
\[
\sum_{t=0}^{T-1}\langle F(z_{t}),z_{t}-z^{\star}\rangle
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta}
+\frac{1}{2}\sum_{t=0}^{T-1}\|z_{t}-z^{\star}\|^{2}.
\tag{15}
\]

\item[Step 8.] \textbf{Relate $\langle F(z_{t}),z_{t}-z^{\star}\rangle$ to the primal–dual gap}.
Recall $F(z)=(\nabla_{x}L(x,y),-\nabla_{y}L(x,y))$. By convexity in $x$ and concavity in $y$,
\[
\begin{aligned}
L(x_{t},y)-L(x^{\star},y)
&\le \langle\nabla_{x}L(x_{t},y),x_{t}-x^{\star}\rangle,
\\
L(x^{\star},y)-L(x^{\star},y_{t})
&\ge \langle\nabla_{y}L(x^{\star},y_{t}),y-y_{t}\rangle .
\end{aligned}
\]
Setting $y=y_{t}$ in the first inequality and $x=x_{t}$ in the second and adding gives
\[
L(x_{t},y_{t})-L(x^{\star},y^{\star})
\le
\langle F(z_{t}),z_{t}-z^{\star}\rangle .
\tag{16}
\]

\item[Step 9.] \textbf{Combine \eqref{15} and \eqref{16}}:
\[
\sum_{t=0}^{T-1}\bigl(L(x_{t},y_{t})-L(x^{\star},y^{\star})\bigr)
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta}
+\frac{1}{2}\sum_{t=0}^{T-1}\|z_{t}-z^{\star}\|^{2}.
\tag{17}
\]

\item[Step 10.] \textbf{Upper‑bound the second sum}.
Since $F$ is monotone and $z^{\star}$ is a zero of $F$, the sequence $\|z_{t}-z^{\star}\|$ is
non‑increasing (a standard consequence of Lemma~\ref{lem:one-step} with $\eta\le1/L$). Hence
\[
\|z_{t}-z^{\star}\|^{2}\le\|z_{0}-z^{\star}\|^{2},\qquad\forall t.
\]
Thus
\[
\frac{1}{2}\sum_{t=0}^{T-1}\|z_{t}-z^{\star}\|^{2}
\le
\frac{T}{2}\|z_{0}-z^{\star}\|^{2}.
\tag{18}
\]

\item[Step 11.] \textbf{Insert \eqref{18} into \eqref{17}}:
\[
\sum_{t=0}^{T-1}\bigl(L(x_{t},y_{t})-L(x^{\star},y^{\star})\bigr)
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta}
+\frac{T}{2}\|z_{0}-z^{\star}\|^{2}.
\tag{19}
\]

\item[Step 12.] \textbf{Divide by $T$} and use the definition of the averaged iterates:
\[
\frac{1}{T}\sum_{t=0}^{T-1}L(x_{t},y_{t})-L(x^{\star},y^{\star})
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta T}
+\frac{1}{2}\|z_{0}-z^{\star}\|^{2}.
\tag{20}
\]

\item[Step 13.] \textbf{Refine the bound}.
Because $L(x^{\star},y)\le L(x^{\star},y^{\star})\le L(x,y^{\star})$, the quantity
\[
\underbrace{\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})}_{\text{primal–dual gap}}
\]
is upper bounded by the left‑hand side of \eqref{20}. Dropping the non‑negative
$\frac{1}{2}\|z_{0}-z^{\star}\|^{2}$ term yields the clean $O(1/T)$ bound:
\[
\max_{y}L(\bar x_{T},y)-\min_{x}L(x,\bar y_{T})
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta T}.
\tag{21}
\]

\item[Step 14.] \textbf{Conclusion.}
Equation \eqref{21} is exactly the statement \eqref{6} of Theorem~\ref{thm:main}.
Therefore the averaged GDA iterates converge to a saddle point with rate $O(1/T)$.
∎
\end{enumerate}

\section*{Rate Derivation (With Constants)}
From \eqref{21} we can read off the explicit constant:
\[
C:=\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta}.
\]
Hence for any $T\ge1$,
\[
\text{gap}(\bar x_{T},\bar y_{T})\le \frac{C}{T}.
\]
If the initial distance is bounded by $D$, i.e. $\|z_{0}-z^{\star}\|\le D$, the bound simplifies to
\[
\text{gap}(\bar x_{T},\bar y_{T})\le \frac{D^{2}}{2\eta T}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Strongly convex–strongly concave $L$.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, then $F$ is $\mu$‑strongly monotone. In this case a single‑step analysis (without averaging) yields linear convergence:
\[
\|z_{t}-z^{\star}\|\le (1-\eta\mu)^{t}\|z_{0}-z^{\star}\|,
\qquad 0<\eta<\frac{2}{L+\mu}.
\]

\item \textbf{Extragradient method.} Replacing the update \eqref{1} by the two‑step extragradient
scheme improves the rate to $O(1/T^{2})$ under the same smoothness assumptions.

\item \textbf{Stochastic gradients.} If only unbiased estimators $\tilde g_{x}^{t},\tilde g_{y}^{t}$
are available with bounded variance $\sigma^{2}$, the same proof gives
\[
\mathbb{E}\bigl[\text{gap}(\bar x_{T},\bar y_{T})\bigr]
\le
\frac{\|z_{0}-z^{\star}\|^{2}}{2\eta T}
+ \frac{\eta\sigma^{2}}{2},
\]
optimising $\eta\asymp T^{-1/2}$ leads to the stochastic rate $O(T^{-1/2})$.
\end{enumerate}
