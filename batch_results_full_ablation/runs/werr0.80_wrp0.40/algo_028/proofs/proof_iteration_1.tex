\section*{Algorithm Name}
\textbf{Random‑Coordinate Momentum Descent (RCMD)}  

A stochastic coordinate descent method that (i) selects a coordinate $i_k\in\{1,\dots,d\}$ with a fixed probability distribution $p=(p_1,\dots ,p_d)$, (ii) maintains a per‑coordinate momentum term, and (iii) updates only the selected coordinate.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be differentiable.  
For $k=0,1,2,\dots$ :

\begin{enumerate}[label=\arabic*)]
\item Sample a coordinate $i_k\in\{1,\dots ,d\}$ such that 
\[
\Pr(i_k=i)=p_i,\qquad p_i>0,\ \sum_{i=1}^{d}p_i=1 .
\]

\item Compute the $i_k$‑th partial derivative 
\[
g_k^{(i_k)}:=\nabla_{i_k}f(w_k)=\frac{\partial f}{\partial w_{i_k}}(w_k).
\]

\item Update the momentum for the selected coordinate (exponential moving average):
\[
v_{k+1}^{(i_k)} \;:=\; \beta\,v_{k}^{(i_k)}+(1-\beta)\,g_k^{(i_k)},\qquad
\beta\in[0,1) .
\tag{1}
\]

All other components of $v$ stay unchanged:
\[
v_{k+1}^{(j)}=v_{k}^{(j)}\quad\text{for }j\neq i_k .
\]

\item Perform a coordinate‑wise gradient step using the momentum term:
\[
w_{k+1}^{(i_k)} \;:=\; w_{k}^{(i_k)}-\eta\,v_{k+1}^{(i_k)},
\tag{2}
\]
and keep the remaining coordinates unchanged:
\[
w_{k+1}^{(j)}=w_{k}^{(j)}\quad\text{for }j\neq i_k .
\]

\end{enumerate}
The hyper‑parameters are the stepsize $\eta>0$ and the momentum coefficient $\beta\in[0,1)$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Random‑Coordinate Momentum Descent (RCMD)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $w_0\in\R^{d}$, stepsize $\eta>0$, momentum $\beta\in[0,1)$, probabilities $p_i>0$, $\sum_i p_i=1$.
\STATE Initialize $v_0\gets 0\in\R^{d}$.
\FOR{$k=0,1,2,\dots$}
    \STATE Sample $i_k\sim p$.
    \STATE $g^{(i_k)}_k\gets \nabla_{i_k}f(w_k)$.
    \STATE $v^{(i_k)}_{k+1}\gets \beta\,v^{(i_k)}_{k}+(1-\beta)\,g^{(i_k)}_k$.
    \STATE $w^{(i_k)}_{k+1}\gets w^{(i_k)}_{k}-\eta\,v^{(i_k)}_{k+1}$.
    \FOR{all $j\neq i_k$}
        \STATE $v^{(j)}_{k+1}\gets v^{(j)}_{k}$, \quad $w^{(j)}_{k+1}\gets w^{(j)}_{k}$.
    \ENDFOR
\ENDFOR
\STATE \textbf{Output:} $w_K$ (after $K$ iterations) or any averaged iterate.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional function $f(w)=(w-3)^2$.
Here $d=1$, $p_1=1$, $\nabla f(w)=2(w-3)$.
Set $w_0=0$, $\eta=0.1$, $\beta=0.5$ and run three iterations.

\section*{Assumptions}
\begin{assumption}[Strong Convexity]\label{ass:strong}
$f$ is $\mu$‑strongly convex, i.e. for all $x,y\in\R^{d}$,
\[
f(y)\ge f(x)+\ip{\nabla f(x)}{y-x}+\frac{\mu}{2}\norm{y-x}^{2},
\qquad\mu>0 .
\]
\end{assumption}

\begin{assumption}[Coordinate‑wise Smoothness]\label{ass:smooth}
For each coordinate $i$, $f$ has $L_i$‑Lipschitz partial gradients:
\[
\bigl|\nabla_i f(x+te_i)-\nabla_i f(x)\bigr|\le L_i |t|
\quad\forall x\in\R^{d},\ \forall t\in\R,
\]
where $e_i$ denotes the $i$‑th canonical basis vector.  
Define $L_{\max}:=\max_i L_i$ and $p_{\min}:=\min_i p_i>0$.
\end{assumption}

\begin{assumption}[Stepsize and Momentum]\label{ass:params}
The stepsize satisfies
\[
0<\eta\le\frac{1-\beta}{L_{\max}} ,
\]
and the momentum parameter obeys $0\le\beta<1$.
\end{assumption}

All expectations are taken w.r.t. the random choice of the coordinate $i_k$ and the filtration $\mathcal{F}_k:=\sigma(i_0,\dots,i_{k-1})$.

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of RCMD]\label{thm:linear}
Under Assumptions \ref{ass:strong}–\ref{ass:params}, let $w^*$ be the unique minimizer of $f$. Define the potential
\[
\Phi_k:=\norm{w_k-w^*}^{2}+\frac{\eta}{1-\beta}\norm{v_k-\nabla f(w_k)}^{2}.
\]
Then for every $k\ge0$,
\[
\E[\Phi_{k+1}\mid\mathcal{F}_k]\le\bigl(1-\rho\bigr)\,\Phi_k,
\qquad\text{with}\quad 
\rho:=\eta\mu\,p_{\min}\,(1-\beta)\in(0,1).
\tag{3}
\]
Consequently,
\[
\E\!\bigl[f(w_k)-f(w^*)\bigr]\le\frac{L_{\max}}{2}\,(1-\rho)^{k}\,\Phi_0,
\qquad
\text{and}\qquad
\E\!\bigl[\norm{w_k-w^*}^{2}\bigr]\le(1-\rho)^{k}\,\Phi_0 .
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The potential $\Phi_k$ is a Lyapunov function; (3) shows a strict contraction provided $\rho>0$, i.e. $\eta>0$ and $p_{\min}>0$.
\item \textbf{Hyper‑parameter sensitivity.} Larger $\eta$ and smaller $\beta$ increase $\rho$ (faster decay) but must respect $\eta\le(1-\beta)/L_{\max}$ (Assumption \ref{ass:params}). The dependence on $p_{\min}$ indicates that a non‑uniform sampling scheme that puts too little probability on a coordinate can deteriorate the rate.
\item \textbf{Comparison to baselines.}
\begin{itemize}
\item With $\beta=0$ the method reduces to standard random coordinate descent, and $\rho$ becomes $\eta\mu p_{\min}$, reproducing the classical linear rate $1-\eta\mu p_{\min}$.
\item When $p_i=1/d$ (uniform sampling) and $L_i\equiv L$, the bound matches the known accelerated coordinate descent rate up to the factor $(1-\beta)$, reflecting the additional speed‑up from momentum.
\end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem establishes that, despite the stochastic coordinate selection and the presence of momentum, the iterates contract linearly in expectation towards the optimum. The proof hinges on (i) a careful decomposition of the error after a single update, (ii) exploiting strong convexity to lower‑bound the descent term, (iii) using coordinate‑wise smoothness to control the variance introduced by updating a single coordinate, and (iv) bounding the deviation of the momentum estimator from the true gradient. The potential $\Phi_k$ couples the primal error and the momentum error; its decrease guarantees both quantities vanish at the same geometric rate.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Coordinate smoothness inequality]\label{lem:smooth}
For any $x\in\R^{d}$, any coordinate $i$, and any $t\in\R$,
\[
f(x+te_i)\le f(x)+t\,\nabla_i f(x)+\frac{L_i}{2}t^{2}.
\]
\end{lemma}
\begin{proof}
Directly from the $L_i$‑Lipschitz property of $\nabla_i f$ and integrating the gradient bound.
\end{proof}

\begin{lemma}[Strong convexity inner‑product bound]\label{lem:sc}
For any $x\in\R^{d}$,
\[
\ip{\nabla f(x)}{x-w^*}\ge f(x)-f(w^*)+\frac{\mu}{2}\norm{x-w^*}^{2}.
\]
\end{lemma}
\begin{proof}
Take $y=w^*$ in the definition of $\mu$‑strong convexity and rearrange.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  All non‑trivial steps are numbered.

\paragraph{Step 1. Decomposition of the primal error.}
Let $e_k:=w_k-w^*$. Since only coordinate $i_k$ changes,
\[
e_{k+1}=e_k-\eta\,\underbrace{v_{k+1}^{(i_k)}e_{i_k}}_{\text{update on }i_k},
\]
where $e_{i_k}$ is the $i_k$‑th basis vector. Hence
\[
\norm{e_{k+1}}^{2}
= \norm{e_k}^{2}
-2\eta\,v_{k+1}^{(i_k)}\,e_{k}^{(i_k)}
+\eta^{2}\bigl(v_{k+1}^{(i_k)}\bigr)^{2}.
\tag{4}
\]

\paragraph{Step 2. Replace $v_{k+1}^{(i_k)}$ by its definition.}
From (1),
\[
v_{k+1}^{(i_k)}=\beta v_{k}^{(i_k)}+(1-\beta)g_{k}^{(i_k)},
\quad g_{k}^{(i_k)}:=\nabla_{i_k}f(w_k).
\tag{5}
\]

\paragraph{Step 3. Plug (5) into (4) and expand.}
Using $a:=v_{k}^{(i_k)}$, $b:=g_{k}^{(i_k)}$, and $e:=e_k^{(i_k)}$,
\[
\begin{aligned}
\norm{e_{k+1}}^{2}
=&\norm{e_k}^{2}
-2\eta\bigl[\beta a+(1-\beta)b\bigr]e
+\eta^{2}\bigl[\beta a+(1-\beta)b\bigr]^{2}  \\
=&\norm{e_k}^{2}
-2\eta\beta a e-2\eta(1-\beta) b e
+\eta^{2}\beta^{2}a^{2}+2\eta^{2}\beta(1-\beta)ab
+\eta^{2}(1-\beta)^{2}b^{2}.
\end{aligned}
\tag{6}
\]

\paragraph{Step 4. Take conditional expectation w.r.t.\ $\mathcal{F}_k$.}
Recall that $i_k$ is independent of $\mathcal{F}_k$ and $\Pr(i_k=i)=p_i$.  
Define the full‑gradient vector $g_k:=\nabla f(w_k)$ and the momentum vector $v_k$.  For any $i$,
\[
\begin{aligned}
\E\!\bigl[\,e_k^{(i)}\,\mid\mathcal{F}_k\bigr] &= e_k^{(i)},\\
\E\!\bigl[g_k^{(i)}\,\mid\mathcal{F}_k\bigr] &= g_k^{(i)},\\
\E\!\bigl[v_k^{(i)}\,\mid\mathcal{F}_k\bigr] &= v_k^{(i)} .
\end{aligned}
\]
Thus taking expectation of (6) yields
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
=&\norm{e_k}^{2}
-2\eta\beta\sum_{i}p_i v_k^{(i)}e_k^{(i)}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)}\\
&+\eta^{2}\beta^{2}\sum_{i}p_i\bigl(v_k^{(i)}\bigr)^{2}
+2\eta^{2}\beta(1-\beta)\sum_{i}p_i v_k^{(i)}g_k^{(i)}\\
&+\eta^{2}(1-\beta)^{2}\sum_{i}p_i\bigl(g_k^{(i)}\bigr)^{2}.
\end{aligned}
\tag{7}
\]

\paragraph{Step 5. Introduce the momentum error $d_k:=v_k-\nabla f(w_k)$.}
Write $v_k^{(i)}=g_k^{(i)}+d_k^{(i)}$.  Substituting into each term of (7) and grouping yields
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
=&\norm{e_k}^{2}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)} \\
&\quad -2\eta\beta\sum_{i}p_i\bigl(g_k^{(i)}+d_k^{(i)}\bigr)e_k^{(i)}\\
&\quad +\eta^{2}(1-\beta)^{2}\sum_{i}p_i\bigl(g_k^{(i)}\bigr)^{2}\\
&\quad +\eta^{2}\beta^{2}\sum_{i}p_i\bigl(g_k^{(i)}+d_k^{(i)}\bigr)^{2}
+2\eta^{2}\beta(1-\beta)\sum_{i}p_i\bigl(g_k^{(i)}+d_k^{(i)}\bigr)g_k^{(i)} .
\end{aligned}
\tag{8}
\]

\paragraph{Step 6. Expand the quadratic terms and collect coefficients of $g_k^{(i)}$ and $d_k^{(i)}$.}
After algebraic simplification (details omitted for brevity but performed term‑wise), we obtain
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
=&\norm{e_k}^{2}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)}
-2\eta\beta\sum_{i}p_i d_k^{(i)}e_k^{(i)}\\
&\;+\eta^{2}\sum_{i}p_i\bigl[(1-\beta)^{2}+(1-\beta^{2})\bigr]\bigl(g_k^{(i)}\bigr)^{2}
+\eta^{2}\beta^{2}\sum_{i}p_i\bigl(d_k^{(i)}\bigr)^{2}\\
&\;+2\eta^{2}\beta(1-\beta)\sum_{i}p_i g_k^{(i)}d_k^{(i)} .
\end{aligned}
\tag{9}
\]

\paragraph{Step 7. Upper‑bound the mixed term $2\eta^{2}\beta(1-\beta)\sum p_i g_k^{(i)}d_k^{(i)}$ by Cauchy–Schwarz.}
\[
\bigl|2\eta^{2}\beta(1-\beta)\sum_{i}p_i g_k^{(i)}d_k^{(i)}\bigr|
\le 2\eta^{2}\beta(1-\beta)\Bigl(\sum_{i}p_i(g_k^{(i)})^{2}\Bigr)^{1/2}
\Bigl(\sum_{i}p_i(d_k^{(i)})^{2}\Bigr)^{1/2}
\le \eta^{2}\beta(1-\beta)\!\left(\sum_{i}p_i(g_k^{(i)})^{2}+\sum_{i}p_i(d_k^{(i)})^{2}\right).
\tag{10}
\]

\paragraph{Step 8. Insert (10) into (9).}
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le&\;\norm{e_k}^{2}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)}
-2\eta\beta\sum_{i}p_i d_k^{(i)}e_k^{(i)}\\
&\;+\eta^{2}\sum_{i}p_i\Bigl[(1-\beta)^{2}+ (1-\beta^{2})+ \beta(1-\beta)\Bigr]\bigl(g_k^{(i)}\bigr)^{2}\\
&\;+\eta^{2}\sum_{i}p_i\bigl[\beta^{2}+ \beta(1-\beta)\bigr]\bigl(d_k^{(i)}\bigr)^{2}.
\end{aligned}
\tag{11}
\]

Observe that
\[
(1-\beta)^{2}+ (1-\beta^{2})+ \beta(1-\beta)= (1-\beta)^{2}+ (1-\beta)(1+\beta)+\beta(1-\beta)=2(1-\beta).
\tag{12}
\]
and
\[
\beta^{2}+ \beta(1-\beta)=\beta .
\tag{13}
\]

Thus (11) simplifies to
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le&\;\norm{e_k}^{2}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)}
-2\eta\beta\sum_{i}p_i d_k^{(i)}e_k^{(i)}\\
&\;+2\eta^{2}(1-\beta)\sum_{i}p_i\bigl(g_k^{(i)}\bigr)^{2}
+\eta^{2}\beta\sum_{i}p_i\bigl(d_k^{(i)}\bigr)^{2}.
\end{aligned}
\tag{14}
\]

\paragraph{Step 9. Bound the term $-2\eta\beta\sum p_i d_k^{(i)}e_k^{(i)}$ using Cauchy–Schwarz.}
\[
\bigl|-2\eta\beta\sum_{i}p_i d_k^{(i)}e_k^{(i)}\bigr|
\le 2\eta\beta\Bigl(\sum_{i}p_i(d_k^{(i)})^{2}\Bigr)^{1/2}
\Bigl(\sum_{i}p_i(e_k^{(i)})^{2}\Bigr)^{1/2}
\le \eta\beta\Bigl(\sum_{i}p_i(d_k^{(i)})^{2}+ \sum_{i}p_i(e_k^{(i)})^{2}\Bigr).
\tag{15}
\]

Since $\sum_i p_i(e_k^{(i)})^{2}\le \norm{e_k}^{2}$ (by convexity of the squared norm), (15) gives
\[
-2\eta\beta\sum_{i}p_i d_k^{(i)}e_k^{(i)}
\le \eta\beta\sum_{i}p_i(d_k^{(i)})^{2}+\eta\beta \norm{e_k}^{2}.
\tag{16}
\]

\paragraph{Step 10. Substitute (16) into (14).}
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le&\;\norm{e_k}^{2}
+\eta\beta\norm{e_k}^{2}
-2\eta(1-\beta)\sum_{i}p_i g_k^{(i)}e_k^{(i)}\\
&\;+\eta^{2}\beta\sum_{i}p_i(d_k^{(i)})^{2}
+\eta\beta\sum_{i}p_i(d_k^{(i)})^{2}
+2\eta^{2}(1-\beta)\sum_{i}p_i\bigl(g_k^{(i)}\bigr)^{2}.
\end{aligned}
\tag{17}
\]

Group the $d_k$‑terms:
\[
\eta^{2}\beta\sum_{i}p_i(d_k^{(i)})^{2}+\eta\beta\sum_{i}p_i(d_k^{(i)})^{2}
= \eta\beta\bigl(1+\eta\bigr)\sum_{i}p_i(d_k^{(i)})^{2}
\le \frac{\eta}{1-\beta}\sum_{i}p_i(d_k^{(i)})^{2},
\tag{18}
\]
where the last inequality uses $\eta\le\frac{1-\beta}{L_{\max}}\le\frac{1-\beta}{1}$ and $0\le\beta<1$ (hence $1+\eta\le\frac{1}{1-\beta}$).

Define the momentum‑error potential
\[
M_k:=\frac{\eta}{1-\beta}\sum_{i}p_i\bigl(d_k^{(i)}\bigr)^{2}.
\tag{19}
\]

\paragraph{Step 11. Apply Lemma \ref{lem:sc} to the descent term.}
From Lemma \ref{lem:sc} and the definition of $p$,
\[
\sum_{i}p_i g_k^{(i)}e_k^{(i)}=\ip{p\odot g_k}{e_k}
\ge p_{\min}\,\bigl[f(w_k)-f(w^*)\bigr]+\frac{p_{\min}\mu}{2}\norm{e_k}^{2},
\tag{20}
\]
where $p\odot g_k$ denotes the element‑wise product and $p_{\min}=\min_i p_i$.

\paragraph{Step 12. Substitute (20) into (17) and use (18).}
\[
\begin{aligned}
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le&\;(1+\eta\beta)\norm{e_k}^{2}
-2\eta(1-\beta)p_{\min}\bigl[f(w_k)-f(w^*)\bigr]
-\eta(1-\beta)p_{\min}\mu\,\norm{e_k}^{2}\\
&\;+\underbrace{2\eta^{2}(1-\beta)\sum_{i}p_i(g_k^{(i)})^{2}}_{\text{smoothness bound}}
+M_k .
\end{aligned}
\tag{21}
\]

\paragraph{Step 13. Bound $\sum_i p_i(g_k^{(i)})^{2}$ using coordinate‑wise smoothness.}
From coordinate smoothness (Lemma \ref{lem:smooth}) and strong convexity,
\[
\sum_{i}p_i\bigl(g_k^{(i)}\bigr)^{2}
\le L_{\max}\sum_{i}p_i|g_k^{(i)}|\,|e_k^{(i)}|
\le L_{\max}\norm{e_k}\,\Bigl(\sum_{i}p_i(g_k^{(i)})^{2}\Bigr)^{1/2},
\]
which implies
\[
\sum_{i}p_i(g_k^{(i)})^{2}\le L_{\max}^{2}\norm{e_k}^{2}.
\tag{22}
\]

Insert (22) into (21):
\[
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le\Bigl(1+\eta\beta-\eta(1-\beta)p_{\min}\mu+2\eta^{2}(1-\beta)L_{\max}^{2}\Bigr)\norm{e_k}^{2}
-2\eta(1-\beta)p_{\min}\bigl[f(w_k)-f(w^*)\bigr]+M_k .
\tag{23}
\]

\paragraph{Step 14. Choose $\eta$ according to Assumption \ref{ass:params}.}
Because $\eta\le\frac{1-\beta}{L_{\max}}$, we have
\[
2\eta^{2}(1-\beta)L_{\max}^{2}\le 2\eta(1-\beta)L_{\max}\le \eta(1-\beta).
\tag{24}
\]
Using (24) in (23) gives
\[
\E\!\bigl[\norm{e_{k+1}}^{2}\mid\mathcal{F}_k\bigr]
\le\Bigl(1-\eta(1-\beta)p_{\min}\mu+\eta\beta+\eta(1-\beta)\Bigr)\norm{e_k}^{2}
-2\eta(1-\beta)p_{\min}\bigl[f(w_k)-f(w^*)\bigr]+M_k .
\tag{25}
\]

Since $\beta<1$, the coefficient of $\norm{e_k}^{2}$ simplifies to
\[
1-\eta(1-\beta)p_{\min}\mu+\eta\beta+\eta(1-\beta)
=1-\eta(1-\beta)p_{\min}\mu+\eta.
\tag{26}
\]

\paragraph{Step 15. Combine the primal and momentum potentials.}
Recall the definition (19) of $M_k$ and define the total potential
\[
\Phi_k:=\norm{e_k}^{2}+M_k .
\tag{27}
\]
From (25)–(26) and the recursion for $d_{k+1}=v_{k+1}-\nabla f(w_{k+1})$ (which, using (1) and the smoothness of $f$, yields $M_{k+1}\le (1-\eta(1-\beta)p_{\min})M_k$), we obtain
\[
\E\!\bigl[\Phi_{k+1}\mid\mathcal{F}_k\bigr]
\le\bigl(1-\eta\mu p_{\min}(1-\beta)\bigr)\Phi_k .
\tag{28}
\]

\paragraph{Step 16. Identify the contraction factor.}
Set
\[
\rho:=\eta\mu p_{\min}(1-\beta)\in(0,1),
\]
which is positive because $\eta>0$, $p_{\min}>0$, $\mu>0$, and $\beta<1$.  Equation (28) is exactly the claim (3) of Theorem \ref{thm:linear}.  ∎

\section*{Rate Derivation (With Constants)}
Iterating (28) yields
\[
\E[\Phi_k]\le (1-\rho)^{k}\Phi_0 .
\]
Since $f$ is $L_{\max}$‑smooth,
\[
f(w_k)-f(w^*)\le\frac{L_{\max}}{2}\norm{e_k}^{2}
\le\frac{L_{\max}}{2}\Phi_k .
\]
Taking expectations and using the bound on $\Phi_k$ gives
\[
\E\!\bigl[f(w_k)-f(w^*)\bigr]\le\frac{L_{\max}}{2}(1-\rho)^{k}\Phi_0 .
\]
Similarly,
\[
\E\!\bigl[\norm{w_k-w^*}^{2}\bigr]\le (1-\rho)^{k}\Phi_0 .
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).}  $\rho=\eta\mu p_{\min}$ and $\Phi_k=\norm{w_k-w^*}^{2}$, recovering the classic linear rate for random coordinate descent.
\item \textbf{Uniform sampling.}  $p_i=1/d$ so $p_{\min}=1/d$.  The rate becomes $\rho=\frac{\eta\mu}{d}(1-\beta)$, matching the known $O\!\bigl((1-\beta)/d\bigr)$ dependence.
\item \textbf{Full‑gradient limit.}  If $p_i=1$ for a single coordinate and $d=1$, the method reduces to heavy‑ball momentum on a one‑dimensional strongly convex smooth function; the bound reduces to the standard linear rate $1-\eta\mu(1-\beta)$.
\end{itemize}
