\section*{AdaGrad with Running Sum of Gradients}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be convex and $L$‑smooth.  
Denote by $g_{k}= \nabla f(w_{k})$ the (full) gradient at iteration $k$.  
Define the cumulative squared‑gradient matrix  

\[
G_{k}= \sum_{i=1}^{k} g_{i}\odot g_{i}\in\mathbb{R}^{d},
\qquad
\bigl(G_{k}\bigr)_{j}= \sum_{i=1}^{k} g_{i,j}^{2},
\]

where $\odot$ denotes component‑wise product.  
The AdaGrad step size for coordinate $j$ at iteration $k$ is  

\[
\eta_{k,j}= \frac{\alpha}{\sqrt{(G_{k})_{j}+\varepsilon}},\qquad 
\alpha>0,\;\varepsilon>0.
\]

The update rule is therefore  

\[
\boxed{\,w_{k+1}= w_{k}-\eta_{k}\odot g_{k}\,},\qquad 
\eta_{k}= \bigl(\eta_{k,1},\dots,\eta_{k,d}\bigr).
\]

For the scalar case $d=1$ the rule simplifies to  

\[
w_{k+1}= w_{k}-\frac{\alpha}{\sqrt{\sum_{i=1}^{k} g_{i}^{2}+\varepsilon}}\;g_{k}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (scalar version)}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $w_{0}\in\mathbb{R}$, stepsize $\alpha>0$, smoothing $\varepsilon>0$, max iterations $T$.
\STATE $S_{0}\leftarrow 0$ \hfill // cumulative sum of squared gradients
\FOR{$k=0,\dots,T-1$}
    \STATE $g_{k}\leftarrow \nabla f(w_{k})$
    \STATE $S_{k+1}\leftarrow S_{k}+g_{k}^{2}$
    \STATE $\eta_{k}\leftarrow \dfrac{\alpha}{\sqrt{S_{k+1}+\varepsilon}}$
    \STATE $w_{k+1}\leftarrow w_{k}-\eta_{k}\,g_{k}$
\ENDFOR
\STATE \textbf{Return} $w_{T}$
\end{algorithmic}
\end{algorithm}

\[

\]

\section*{Assumptions}
\begin{assumption}[Convexity]\label{ass:convex}
$f$ is convex, i.e. for all $u,v\in\mathbb{R}^{d}$,
\[
f(v)\ge f(u)+\langle\nabla f(u),v-u\rangle .
\]
\end{assumption}

\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth:
\[
\|\nabla f(u)-\nabla f(v)\| \le L\|u-v\|,\qquad\forall u,v.
\]
Equivalently,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^{2}.
\]
\end{assumption}

\begin{assumption}[Bounded Gradients]\label{ass:bounded}
There exists $G>0$ such that $\|g_{k}\|=\|\nabla f(w_{k})\|\le G$ for all $k$.
\end{assumption}

\begin{assumption}[Domain Diameter]\label{ass:diameter}
The feasible set $\mathcal{W}$ is convex and has bounded diameter $D$:
\[
\|w-w'\|\le D,\qquad\forall w,w'\in\mathcal{W}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Regret bound for AdaGrad]\label{thm:regret}
Under Assumptions \ref{ass:convex}–\ref{ass:diameter} and with $\varepsilon>0$, the iterates $\{w_{k}\}$ generated by the scalar AdaGrad rule satisfy for any $w^{*}\in\mathcal{W}$

\[
\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \;\le\;
   \frac{D^{2}}{2\alpha}\sqrt{S_{T}}+\frac{\alpha}{2}\sum_{k=0}^{T-1}\frac{g_{k}^{2}}{\sqrt{S_{k+1}}}.
\tag{1}
\]

Consequently, using the monotonicity $\sqrt{S_{k+1}}\ge\sqrt{S_{k}}$ and $g_{k}^{2}\le G^{2}$,

\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \;\le\;
   \frac{D^{2}}{2\alpha}\frac{\sqrt{S_{T}}}{T}
   +\frac{\alpha G^{2}}{2}\frac{1}{T}\sum_{k=0}^{T-1}\frac{1}{\sqrt{S_{k+1}}}
   \;\le\;
   O\!\left(\frac{DG}{\sqrt{T}}\right).
\tag{2}
\]

Thus the average iterate $\bar w_{T}= \frac{1}{T}\sum_{k=0}^{T-1}w_{k}$ satisfies  

\[
\mathbb{E}\!\bigl[f(\bar w_{T})\bigr]-f(w^{*}) = O\!\left(\frac{DG}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The denominator $\sqrt{S_{k}+\varepsilon}$ never vanishes; the algorithm is thus numerically stable even when gradients are small.
\item \textbf{Adaptivity.} The bound (1) depends on $\sqrt{S_{T}}$, which grows with the magnitude of observed gradients. If gradients are sparse or diminish, the effective stepsize remains larger, yielding faster convergence compared with a fixed stepsize SGD.
\item \textbf{Hyperparameter Sensitivity.} The only user‑controlled scalar is $\alpha$. The bound shows a trade‑off: larger $\alpha$ reduces the first term but inflates the second; the optimal $\alpha^{\star}= D G^{-1}\sqrt{S_{T}}/T^{1/2}$ (up to constants) recovers the $O(DG/\sqrt{T})$ rate.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{SGD with constant $\eta$}: obtains $O(1/\sqrt{T})$ regret but requires prior knowledge of $G$ to set $\eta$.
        \item \emph{AdaGrad}: matches SGD’s worst‑case rate while automatically adapting to the geometry of the data.
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem is proved directly from convexity and the telescoping property of the AdaGrad steps. No stochasticity is needed; the same bound holds for deterministic gradients. For stochastic gradients, taking expectations preserves the inequality because the stepsizes are deterministic functions of past gradients.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity]\label{lem:three}
For any $u,v\in\mathbb{R}^{d}$ and any scalar $\eta>0$,
\[
\|u-\eta g - v\|^{2}= \|u-v\|^{2}-2\eta\langle g, u-v\rangle+\eta^{2}\|g\|^{2}.
\]
\end{lemma}
\begin{proof}
Algebraic expansion of the squared norm. \hfill $\square$
\end{proof}

\begin{lemma}[Bounding the adaptive stepsize]\label{lem:step}
For $a>0$, $b\ge 0$,
\[
\frac{a}{\sqrt{b+a^{2}}}\le\frac{a}{\sqrt{b}}.
\]
\end{lemma}
\begin{proof}
Since $b+a^{2}\ge b$, the denominator is larger, making the fraction smaller. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:regret}.  
All non‑trivial steps are numbered for easy reference.

\begin{proof}
Let $w^{*}\in\mathcal{W}$ be arbitrary.  
Recall the AdaGrad update (scalar case):
\[
w_{k+1}=w_{k}-\eta_{k}g_{k},\qquad 
\eta_{k}= \frac{\alpha}{\sqrt{S_{k+1}+\varepsilon}},\;
S_{k+1}=S_{k}+g_{k}^{2},\;S_{0}=0.
\tag{3}
\]

\paragraph{[Step 1]} Apply Lemma \ref{lem:three} with $u=w_{k}$, $v=w^{*}$, $g=g_{k}$ and $\eta=\eta_{k}$:
\[
\|w_{k+1}-w^{*}\|^{2}
  =\|w_{k}-w^{*}\|^{2}
   -2\eta_{k}\langle g_{k},w_{k}-w^{*}\rangle
   +\eta_{k}^{2}g_{k}^{2}.
\tag{4}
\]

\paragraph{[Step 2]} Rearrange (4) to isolate the inner product term:
\[
2\eta_{k}\langle g_{k},w_{k}-w^{*}\rangle
   =\|w_{k}-w^{*}\|^{2}
    -\|w_{k+1}-w^{*}\|^{2}
    +\eta_{k}^{2}g_{k}^{2}.
\tag{5}
\]

\paragraph{[Step 3]} By convexity (Assumption \ref{ass:convex}),
\[
f(w_{k})-f(w^{*})\le\langle g_{k}, w_{k}-w^{*}\rangle .
\tag{6}
\]

\paragraph{[Step 4]} Multiply (6) by $2\eta_{k}$ and substitute the right‑hand side using (5):
\[
2\eta_{k}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le \|w_{k}-w^{*}\|^{2}
        -\|w_{k+1}-w^{*}\|^{2}
        +\eta_{k}^{2}g_{k}^{2}.
\tag{7}
\]

\paragraph{[Step 5]} Divide (7) by $2\eta_{k}$ to obtain a bound on the instantaneous regret:
\[
f(w_{k})-f(w^{*})
   \le \frac{1}{2\eta_{k}}\bigl(\|w_{k}-w^{*}\|^{2}
        -\|w_{k+1}-w^{*}\|^{2}\bigr)
        +\frac{\eta_{k}}{2}g_{k}^{2}.
\tag{8}
\]

\paragraph{[Step 6]} Substitute the explicit expression of $\eta_{k}$ from (3):
\[
\frac{1}{2\eta_{k}}=\frac{\sqrt{S_{k+1}+\varepsilon}}{2\alpha},
\qquad
\frac{\eta_{k}}{2}= \frac{\alpha}{2\sqrt{S_{k+1}+\varepsilon}}.
\tag{9}
\]

\paragraph{[Step 7]} Plug (9) into (8):
\[
f(w_{k})-f(w^{*})
   \le \frac{\sqrt{S_{k+1}+\varepsilon}}{2\alpha}
        \bigl(\|w_{k}-w^{*}\|^{2}
        -\|w_{k+1}-w^{*}\|^{2}\bigr)
        +\frac{\alpha}{2\sqrt{S_{k+1}+\varepsilon}}g_{k}^{2}.
\tag{10}
\]

\paragraph{[Step 8]} Sum (10) over $k=0,\dots,T-1$.
The telescoping sum of the first term yields
\[
\sum_{k=0}^{T-1}\frac{\sqrt{S_{k+1}+\varepsilon}}{2\alpha}
        \bigl(\|w_{k}-w^{*}\|^{2}
        -\|w_{k+1}-w^{*}\|^{2}\bigr)
   =\frac{1}{2\alpha}
        \Bigl(\sqrt{S_{1}+\varepsilon}\|w_{0}-w^{*}\|^{2}
              -\sqrt{S_{T}+\varepsilon}\|w_{T}-w^{*}\|^{2}\Bigr)
\]
\[
\;\;\;+\;\frac{1}{2\alpha}\sum_{k=1}^{T-1}
        \bigl(\sqrt{S_{k+1}+\varepsilon}-\sqrt{S_{k}+\varepsilon}\bigr)
        \|w_{k}-w^{*}\|^{2}.
\tag{11}
\]

\paragraph{[Step 9]} Use the bounded‑diameter assumption $\|w_{k}-w^{*}\|\le D$ and the monotonicity
$\sqrt{S_{k+1}+\varepsilon}\ge\sqrt{S_{k}+\varepsilon}$ to upper‑bound the second summation in (11):
\[
\bigl(\sqrt{S_{k+1}+\varepsilon}-\sqrt{S_{k}+\varepsilon}\bigr)
        \|w_{k}-w^{*}\|^{2}
   \le D^{2}\bigl(\sqrt{S_{k+1}+\varepsilon}-\sqrt{S_{k}+\varepsilon}\bigr).
\tag{12}
\]

\paragraph{[Step 10]} Summing (12) over $k=1,\dots,T-1$ telescopes again, giving
\[
\sum_{k=1}^{T-1}
        \bigl(\sqrt{S_{k+1}+\varepsilon}-\sqrt{S_{k}+\varepsilon}\bigr)
   =\sqrt{S_{T}+\varepsilon}-\sqrt{S_{1}+\varepsilon}.
\tag{13}
\]

\paragraph{[Step 11]} Substitute (12)–(13) into (11) and drop the non‑positive term
$-\sqrt{S_{T}+\varepsilon}\|w_{T}-w^{*}\|^{2}\le0$:
\[
\sum_{k=0}^{T-1}\frac{\sqrt{S_{k+1}+\varepsilon}}{2\alpha}
        \bigl(\|w_{k}-w^{*}\|^{2}
        -\|w_{k+1}-w^{*}\|^{2}\bigr)
   \le \frac{D^{2}}{2\alpha}\sqrt{S_{T}+\varepsilon}.
\tag{14}
\]

\paragraph{[Step 12]} The second term of (10) summed over $k$ yields
\[
\sum_{k=0}^{T-1}\frac{\alpha}{2\sqrt{S_{k+1}+\varepsilon}}g_{k}^{2}
   =\frac{\alpha}{2}\sum_{k=0}^{T-1}\frac{g_{k}^{2}}{\sqrt{S_{k+1}+\varepsilon}}.
\tag{15}
\]

\paragraph{[Step 13]} Combine (14) and (15) with the left‑hand side of (10) summed over $k$:
\[
\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le \frac{D^{2}}{2\alpha}\sqrt{S_{T}+\varepsilon}
        +\frac{\alpha}{2}\sum_{k=0}^{T-1}\frac{g_{k}^{2}}{\sqrt{S_{k+1}+\varepsilon}}.
\tag{16}
\]

Since $\varepsilon>0$ only makes the bound looser, we drop it to obtain the cleaner statement (1) of Theorem \ref{thm:regret}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From (16) and the bounded‑gradient Assumption \ref{ass:bounded} ($g_{k}^{2}\le G^{2}$) we have
\[
\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le \frac{D^{2}}{2\alpha}\sqrt{S_{T}}
        +\frac{\alpha G^{2}}{2}\sum_{k=0}^{T-1}\frac{1}{\sqrt{S_{k+1}}}.
\tag{17}
\]

Because $S_{k+1}\ge g_{k}^{2}\ge 0$, the sequence $\{\sqrt{S_{k+1}}\}$ is non‑decreasing, and
\[
\frac{1}{\sqrt{S_{k+1}}}\le\frac{1}{\sqrt{S_{k}}},\qquad
\sum_{k=0}^{T-1}\frac{1}{\sqrt{S_{k+1}}}
   \le \int_{0}^{T}\frac{1}{\sqrt{S(t)}}\,dt
   \le 2\sqrt{S_{T}},
\]
where the last inequality follows from the elementary bound
$\sum_{k=0}^{T-1} a_{k}^{-1}\le 2 a_{T}^{-1/2}$ for any non‑decreasing $a_{k}$.
Thus
\[
\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le \frac{D^{2}}{2\alpha}\sqrt{S_{T}}
        +\alpha G^{2}\sqrt{S_{T}} .
\tag{18}
\]

Choosing $\alpha = D/(G\sqrt{T})$ balances the two terms:
\[
\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le D G\sqrt{T}.
\tag{19}
\]
Dividing by $T$ yields the average‑iterate guarantee (2):
\[
\frac{1}{T}\sum_{k=0}^{T-1}\bigl(f(w_{k})-f(w^{*})\bigr)
   \le \frac{DG}{\sqrt{T}}.
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Exact minimizer known ($w^{*}$).} If $w^{*}$ lies in the interior of $\mathcal{W}$, the distance term $\|w_{T}-w^{*}\|$ can be bounded tighter, leading to a regret bound of $O(DG)$ independent of $T$.
\item \textbf{Sparse gradients.} Suppose only $s\ll d$ coordinates are non‑zero at each iteration. Then $S_{T}$ grows only on those $s$ coordinates, and the bound improves to $O\!\bigl(D\sqrt{s\,\log T}\bigr)$.
\item \textbf{Stochastic gradients.} If $g_{k}$ is an unbiased estimator of $\nabla f(w_{k})$ with bounded variance, taking expectation in (16) preserves the inequality because $\eta_{k}$ depends only on past gradients. The same $O(DG/\sqrt{T})$ rate holds in expectation.
\end{enumerate}
