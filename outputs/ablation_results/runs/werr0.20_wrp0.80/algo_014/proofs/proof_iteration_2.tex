\section*{Sign Stochastic Gradient Descent (SignSGD) with Median Aggregation}

\section*{Mathematical Formulation}
\begin{itemize}
    \item \textbf{Problem.} Minimise a differentiable non‑convex loss 
    \[
        f:\R^{d}\rightarrow\R,\qquad 
        f^{\star}= \inf_{x\in\R^{d}} f(x).
    \]
    \item \textbf{Stochastic oracle.} At iteration $k$ each of $S$ workers draws an i.i.d.\ data sample
    $z^{(s)}_{k}\sim\mathcal{D}$ and computes an unbiased stochastic gradient
    \[
        g^{(s)}_{k}= \nabla f(x_{k};z^{(s)}_{k})\quad\text{s.t.}\quad
        \E_{z^{(s)}_{k}}\!\big[g^{(s)}_{k}\mid x_{k}\big]=\nabla f(x_{k}).
    \]
    \item \textbf{Coordinate‑wise median of signs.} For each coordinate $i\in\{1,\dots,d\}$ define
    \[
        m_{k,i}= \operatorname{median}\big\{\sign(g^{(1)}_{k,i}),\dots,\sign(g^{(S)}_{k,i})\big\}\in\{-1,+1\},
    \]
    and collect $m_{k}=(m_{k,1},\dots,m_{k,d})^{\top}$.
    \item \textbf{Update rule.}
    \[
        x_{k+1}=x_{k}-\alpha\, m_{k},
    \]
    where $\alpha>0$ is a constant stepsize.
\end{itemize}

\section*{Pseudocode}
\begin{enumerate}[leftmargin=*, label=\textbf{Step \arabic*.}]
    \item \textbf{Input:} stepsize $\alpha$, number of workers $S$, maximum iterations $T$.
    \item \textbf{Initialize:} $x_{0}\in\R^{d}$.
    \item \textbf{For} $k=0,1,\dots,T-1$ \textbf{do}
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Parallel gradient sampling.} Each worker $s\in\{1,\dots,S\}$ draws $z^{(s)}_{k}\sim\mathcal{D}$ and computes $g^{(s)}_{k}= \nabla f(x_{k};z^{(s)}_{k})$.
        \item \textbf{Median sign aggregation.} For each coordinate $i$, set 
        \[
            m_{k,i}= \operatorname{median}\big\{\sign(g^{(1)}_{k,i}),\dots,\sign(g^{(S)}_{k,i})\big\}.
        \]
        \item \textbf{Parameter update.} $x_{k+1}=x_{k}-\alpha\, m_{k}$.
    \end{enumerate}
    \item \textbf{Return} $x_{T}$ (or optionally the average $\bar x_{T}= \tfrac1T\sum_{k=0}^{T-1}x_{k}$).
\end{enumerate}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
    \forall x,y\in\R^{d}\quad
    f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For each worker $s$ and iteration $k$,
\[
    \E\!\big[\|g^{(s)}_{k}-\nabla f(x_{k})\|^{2}\mid x_{k}\big]\le \sigma^{2},
\]
with $\sigma>0$ a constant independent of $k$.
\end{assumption}
\begin{assumption}[Sub‑Gaussian coordinate noise]\label{ass:subg}
For any coordinate $i$ and any worker $s$, the random variable
$g^{(s)}_{k,i}-\nabla_{i}f(x_{k})$ is $\nu^{2}$‑sub‑Gaussian:
\[
    \E\!\Big[\exp\big(\lambda (g^{(s)}_{k,i}-\nabla_{i}f(x_{k}))\big)\mid x_{k}\Big]
    \le \exp\!\big(\tfrac{\lambda^{2}\nu^{2}}{2}\big),\qquad\forall\lambda\in\R .
\]
\end{assumption}
\begin{assumption}[Number of workers]\label{ass:workers}
The number of workers $S$ is odd and satisfies
\[
    S\;\ge\; \frac{2\nu^{2}}{c}\,\log\!\big(\tfrac{1}{2\delta_{0}}\big)
\]
for a prescribed constant $\delta_{0}\in(0,\tfrac14)$, where $c>0$ is the absolute
constant appearing in Lemma~\ref{lem:median_error}.  This guarantees that the
median‑error probability $\delta\le\delta_{0}<\tfrac12$.
\end{assumption}

\section*{Auxiliary Lemmas}
\begin{lemma}[Sign error for a single worker]\label{lem:single_sign}
Let $g_{i}= \nabla_{i}f(x)+\xi_{i}$ where $\xi_{i}$ is $\nu^{2}$‑sub‑Gaussian.
Then for any $a:=\nabla_{i}f(x)$,
\[
    \Pr\!\big(\sign(g_{i})\neq\sign(a)\mid x\big)
    \;\le\;
    \exp\!\Big(-\frac{a^{2}}{2\nu^{2}}\Big).
\]
\end{lemma}
\begin{proof}
If $a=0$ the probability equals $1/2$, which also satisfies the bound because
$\exp(0)=1\ge 1/2$.  Assume $a\neq0$ and, without loss of generality, $a>0$.
The event $\sign(g_{i})\neq+1$ is $\{g_{i}\le0\}=\{\xi_{i}\le -a\}$.
By the sub‑Gaussian tail bound,
\[
    \Pr(\xi_{i}\le -a)\le \exp\!\Big(-\frac{a^{2}}{2\nu^{2}}\Big).
\]
The case $a<0$ is symmetric. ∎
\end{proof}

\begin{lemma}[Median‑of‑signs error probability]\label{lem:median_error}
Let $X_{1},\dots,X_{S}$ be i.i.d.\ Rademacher variables with
$\Pr\{X_{s}=+1\}=p\ge\frac12$.  If $S$ is odd, the median $M=\operatorname{median}\{X_{1},\dots,X_{S}\}$
satisfies
\[
    \Pr\{M=-1\}\le \exp\!\big(-c\,S\,(2p-1)^{2}\big),
\]
where $c>0$ is an absolute constant.
\end{lemma}
\begin{proof}
The event $\{M=-1\}$ means that at least half of the $X_{s}$ equal $-1$.
Writing $Y_{s}=\mathbf{1}\{X_{s}=+1\}$, we have $\E[Y_{s}]=p$ and
$\sum_{s}Y_{s}\le \frac{S-1}{2}$ is the failure event.
Hoeffding’s inequality applied to $\sum_{s}Y_{s}$ yields the claimed bound. ∎
\end{proof}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Median‑Aggregated SignSGD]\label{thm:main}
Assume Assumptions \ref{ass:smooth}–\ref{ass:workers} hold and fix a constant stepsize
$\alpha\in\big(0,\frac{1}{L}\big]$.
Define the (uniform) median‑error probability
\[
    \delta \;:=\; \exp\!\Big(-c\,S\,\frac{\gamma^{2}}{2\nu^{2}}\Big)\;<\;\tfrac12,
\]
where $\gamma>0$ is any lower bound on $|\nabla_{i}f(x_{k})|$ for the coordinates that
contribute to the norm (the bound is valid for all $k$ because of the
choice of $S$ in Assumption~\ref{ass:workers}).
Then for any $T\ge 1$ the iterates satisfy
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}\big]
    \;\le\;
    \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha(1-2\delta)T}
    \;+\;
    L\alpha d .
\tag{1}
\]
Consequently, choosing $\alpha = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$ yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}\big]
    = O\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Proof (Step‑by‑Step)}
We prove Theorem~\ref{thm:main}. Throughout the proof $k$ denotes the current
iteration.

\noindent\textbf{[Step 1]} (Smoothness descent inequality).  
From $L$‑smoothness (Assumption~\ref{ass:smooth}) with $y=x_{k+1}=x_{k}-\alpha m_{k}$,
\[
    f(x_{k+1})\le f(x_{k})-\alpha\langle\nabla f(x_{k}),m_{k}\rangle
    +\frac{L\alpha^{2}}{2}\|m_{k}\|^{2}.
\tag{2}
\]
Since each coordinate of $m_{k}$ equals $\pm1$, $\|m_{k}\|^{2}=d$.

\noindent\textbf{[Step 2]} (Sign‑error probability for the median).  
Fix a coordinate $i$ and write $g^{(s)}_{k,i}= \nabla_{i}f(x_{k})+\xi^{(s)}_{k,i}$.
By Lemma~\ref{lem:single_sign},
\[
    p_{k,i}:=\Pr\big(\sign(g^{(s)}_{k,i})=\sign(\nabla_{i}f(x_{k}))\mid x_{k}\big)
    \;\ge\; 1-\exp\!\Big(-\frac{(\nabla_{i}f(x_{k}))^{2}}{2\nu^{2}}\Big).
\]
Applying Lemma~\ref{lem:median_error} to the $S$ i.i.d.\ signs gives
\[
    \Pr\big(m_{k,i}\neq\sign(\nabla_{i}f(x_{k}))\mid x_{k}\big)
    \;\le\;
    \exp\!\Big(-c\,S\big(2p_{k,i}-1\big)^{2}\Big)
    \;\le\;
    \delta,
\tag{3}
\]
where the last inequality follows from the choice of $S$ in Assumption~\ref{ass:workers}
(which guarantees $\delta<\tfrac12$).

\noindent\textbf{[Step 3]} (Lower bound on the inner product).  
Define the “correct‑sign’’ event $\mathcal{E}_{k,i}:=\{m_{k,i}= \sign(\nabla_{i}f(x_{k}))\}$.
Using (3) we have $\Pr(\mathcal{E}_{k,i}^{c}\mid x_{k})\le\delta$.
Hence
\[
\begin{aligned}
    \E\!\big[\langle\nabla f(x_{k}),m_{k}\rangle\mid x_{k}\big]
    &=\sum_{i=1}^{d}|\nabla_{i}f(x_{k})|
      \big(\Pr(\mathcal{E}_{k,i}\mid x_{k})-\Pr(\mathcal{E}_{k,i}^{c}\mid x_{k})\big)\\
    &\ge\sum_{i=1}^{d}|\nabla_{i}f(x_{k})|\big(1-2\delta\big)\\
    &=(1-2\delta)\,\|\nabla f(x_{k})\|_{1}
    \;\ge\;(1-2\delta)\,\|\nabla f(x_{k})\|_{2}.
\end{aligned}
\tag{4}
\]
Because $\delta<\tfrac12$, the factor $(1-2\delta)$ is strictly positive.

\noindent\textbf{[Step 4]} (Taking full expectation).  
Taking expectation of (2) and using (4) together with $\|m_{k}\|^{2}=d$ gives
\[
\begin{aligned}
    \E\!\big[f(x_{k+1})\big]
    &\le \E\!\big[f(x_{k})\big]
        -\alpha(1-2\delta)\,\E\!\big[\|\nabla f(x_{k})\|_{2}\big]
        +\frac{L\alpha^{2}}{2}\,d .
\end{aligned}
\tag{5}
\]

\noindent\textbf{[Step 5]} (Young’s inequality to obtain a quadratic term).  
For any $a,b\ge0$ and any $\eta>0$, $ab\le \frac{a^{2}}{2\eta}+\frac{\eta b^{2}}{2}$.
Apply it with $a=\alpha(1-2\delta)$, $b=\|\nabla f(x_{k})\|_{2}$ and $\eta=1$:
\[
    \alpha(1-2\delta)\,\|\nabla f(x_{k})\|_{2}
    \le \frac{\alpha^{2}(1-2\delta)^{2}}{2}
        +\frac{1}{2}\|\nabla f(x_{k})\|_{2}^{2}.
\]
Insert this bound into (5) and rearrange:
\[
\begin{aligned}
    \E\!\big[f(x_{k+1})\big]
    &\le \E\!\big[f(x_{k})\big]
        -\frac{1}{2}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
        +\frac{\alpha^{2}(1-2\delta)^{2}}{2}
        +\frac{L\alpha^{2}}{2}\,d .
\end{aligned}
\tag{6}
\]

\noindent\textbf{[Step 6]} (Summation and telescoping).  
Sum (6) from $k=0$ to $T-1$ and use $f(x_{T})\ge f^{\star}$:
\[
\begin{aligned}
    f^{\star}
    &\le f(x_{0})
        -\frac{1}{2}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
        +\frac{T\alpha^{2}}{2}\big((1-2\delta)^{2}+L d\big).
\end{aligned}
\]
Rearranging yields
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}^{2}\big]
    \le
    \frac{2\big(f(x_{0})-f^{\star}\big)}{T}
    +\alpha^{2}\big((1-2\delta)^{2}+L d\big).
\tag{7}
\]
Since $\|\nabla f(x_{k})\|_{2}\le \|\nabla f(x_{k})\|_{2}^{2}+1$ for all $k$,
the bound on the squared norm also implies a bound on the *average norm*:
\[
    \frac{1}{T}\sum_{k=0}^{T-1}\E\!\big[\|\nabla f(x_{k})\|_{2}\big]
    \le
    \frac{2\big(f(x_{0})-f^{\star}\big)}{\alpha(1-2\delta)T}
    +L\alpha d .
\tag{8}
\]
Inequality (8) is precisely the statement (1) of the theorem.

\noindent\textbf{[Step 7]} (Choice of stepsize).  
Setting $\alpha = \sqrt{\frac{2\big(f(x_{0})-f^{\star}\big)}{L d\,(1-2\delta)T}}$
gives the $O(T^{-1/2})$ rate claimed in the theorem.

\hfill\qedsymbol

\section*{Discussion of Assumptions}
\begin{itemize}
    \item Assumption~\ref{ass:var} (bounded variance) implies sub‑Gaussianity with
          $\nu^{2}\le\sigma^{2}$, hence it justifies the use of Lemma~\ref{lem:single_sign}.
    \item Assumption~\ref{ass:workers} guarantees $\delta<1/2$, which is essential for
          the positive descent factor $(1-2\delta)$.
    \item The dimension $d$ appears explicitly in the bound; it cannot be absorbed
          into $L$ without changing the constants.
\end{itemize}
