\section*{Algorithm Name}
\textbf{Mirror Descent (MD) with Bregman Proximal Mapping}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a convex differentiable objective and let $h:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a \emph{prox‑function} that is $1$‑strongly convex w.r.t. a norm $\|\cdot\|$.
The Bregman divergence generated by $h$ is
\[
\Breg{x}{y}=h(x)-h(y)-\inner{\grad h(y)}{x-y}.
\]

Given a stepsize sequence $\{\eta_k\}_{k\ge 0}$, the MD iteration is
\begin{equation}
\label{eq:md}
x_{k+1}\;=\;\arg\min_{x\in\mathbb{R}^{d}}\Big\{\inner{\grad f(x_k)}{x}
+\frac{1}{\eta_k}\,\Breg{x}{x_k}\Big\}.
\end{equation}
When $h(x)=\tfrac12\|x\|_{2}^{2}$, $\Breg{x}{y}=\tfrac12\|x-y\|_{2}^{2}$ and \eqref{eq:md}
reduces to the standard (projected) gradient step.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Mirror Descent}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\mathbb{R}^{d}$, stepsizes $\{\eta_k\}_{k=0}^{T-1}>0$, prox‑function $h$.
\FOR{$k=0$ \TO $T-1$}
    \STATE Compute gradient $g_k \gets \grad f(x_k)$.
    \STATE Solve the proximal subproblem
    \[
    x_{k+1}\gets \arg\min_{x}\Big\{\inner{g_k}{x}+\frac{1}{\eta_k}\Breg{x}{x_k}\Big\}.
    \]
\ENDFOR
\RETURN $x_T$ (or an average $\bar{x}_T=\tfrac1T\sum_{k=0}^{T-1}x_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
f(w)=(w-3)^2,\qquad h(w)=\tfrac12 w^{2},
\]
so that $\Breg{w}{w_k}=\tfrac12(w-w_k)^2$.  
Take $w_0=0$, constant stepsize $\eta_k=\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
$k$ & $w_k$ & $g_k=\grad f(w_k)$ & $f(w_k)$\\ \hline
0 & $0$ & $2(w_0-3)=-6$ & $9$\\
1 & $w_1=w_0-\eta g_0=0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=w_1-\eta g_1=0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=w_2-\eta g_2=1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2\approx2.366$\\
\end{tabular}
\end{center}
The iterates follow the exact Euclidean gradient descent because the chosen $h$ yields the Euclidean Bregman divergence.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness w.r.t. $h$]
\label{as:convex}
The objective $f$ is convex and $L$‑smooth relative to $h$, i.e.
\[
f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\Breg{y}{x},
\qquad\forall x,y\in\mathbb{R}^{d}.
\]
\end{assumption}
\begin{assumption}[Strong Convexity of $h$]
\label{as:hstrong}
The prox‑function $h$ is $1$‑strongly convex w.r.t. $\|\cdot\|$, i.e.
\[
\Breg{x}{y}\ge \tfrac12\|x-y\|^{2},\qquad\forall x,y.
\]
\end{assumption}
\begin{assumption}[Bounded Gradient]
\label{as:gradbound}
There exists $G>0$ such that $\|\grad f(x)\|_{*}\le G$ for all $x$ in the domain.
\end{assumption}
\begin{assumption}[Stepsize]
\label{as:stepsize}
The stepsizes are constant $\eta_k=\eta$ with $0<\eta\le \tfrac{1}{L}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Mirror Descent]
\label{thm:md}
Let Assumptions~\ref{as:convex}--\ref{as:stepsize} hold and let $x^{\star}\in\arg\min f$. Then for any $T\ge1$
\[
f(\bar{x}_T)-f(x^{\star})\;\le\;
\frac{\Breg{x^{\star}}{x_0}}{\eta T}
\;+\;\frac{\eta L}{2}\,G^{2},
\qquad\text{where }\bar{x}_T:=\frac{1}{T}\sum_{k=0}^{T-1}x_k.
\]
In particular, choosing $\eta=\sqrt{\frac{2\Breg{x^{\star}}{x_0}}{L G^{2}T}}$ yields the optimal $O(1/\sqrt{T})$ rate.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The Bregman divergence term guarantees that each iterate stays in the domain where $h$ is defined; strong convexity of $h$ yields a quadratic lower bound on $\Breg{x}{y}$, which is essential for the descent inequality.
\item \textbf{Hyperparameter Sensitivity.} The stepsize $\eta$ must satisfy $\eta\le 1/L$ (Assumption~\ref{as:stepsize}) to ensure that the smoothness inequality can be combined with the optimality condition of the subproblem. Larger $\eta$ may violate the descent lemma and cause divergence.
\item \textbf{Comparison to Baselines.}
   \begin{enumerate}
   \item With $h(w)=\tfrac12\|w\|^{2}$, Theorem~\ref{thm:md} recovers the classic $O(1/T)$ convergence of projected gradient descent for smooth convex functions.
   \item For non‑Euclidean $h$ (e.g., entropy), MD adapts to the geometry of the feasible set and can achieve the same $O(1/T)$ rate while respecting constraints implicitly.
   \end{enumerate}
\end{itemize}

\section*{Discussion}
The result aligns with the standard mirror‑descent analysis (e.g., \cite{Beck2003}). The bound consists of a \emph{bias} term $\Breg{x^{\star}}{x_0}/(\eta T)$ that decays with $T$ and a \emph{variance} term $\eta L G^{2}/2$ that grows with $\eta$. The optimal trade‑off yields the $O(1/\sqrt{T})$ rate, matching stochastic gradient descent under bounded variance, even though MD here is deterministic.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity for Bregman Divergence]
\label{lem:threepoint}
For any $x,y,z\in\mathbb{R}^{d}$,
\[
\Breg{x}{z}= \Breg{x}{y}+\Breg{y}{z}
+\inner{\grad h(y)-\grad h(z)}{x-y}.
\]
\end{lemma}
\begin{proof}
By expanding the definition of $\Breg{\cdot}{\cdot}$ and canceling terms, the identity follows directly. \hfill $\square$
\end{proof}
\begin{lemma}[Optimality Condition of the MD Subproblem]
\label{lem:optcond}
Let $x_{k+1}$ be defined by \eqref{eq:md}. Then
\[
\inner{\grad f(x_k)}{x-x_{k+1}}+\frac{1}{\eta}
\inner{\grad h(x_{k+1})-\grad h(x_k)}{x-x_{k+1}}\ge 0,
\qquad\forall x\in\mathbb{R}^{d}.
\]
\end{lemma}
\begin{proof}
The subproblem is a convex optimization problem; the first‑order optimality condition yields the stated inequality. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:md}. Throughout the proof we fix an arbitrary optimal point $x^{\star}\in\arg\min f$.

\medskip
\noindent\textbf{[Step 1]}  Apply Lemma~\ref{lem:optcond} with $x=x^{\star}$:
\[
\inner{\grad f(x_k)}{x^{\star}-x_{k+1}}
+\frac{1}{\eta}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}
\ge 0.
\tag{1}
\]
\textbf{[Step 2]}  Rearrange (1) to isolate the inner product with the gradient:
\[
\inner{\grad f(x_k)}{x_{k+1}-x^{\star}}
\le
\frac{1}{\eta}\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}.
\tag{2}
\]
\textbf{[Step 3]}  Use the three‑point identity (Lemma~\ref{lem:threepoint}) with
$(x,y,z)=(x^{\star},x_{k+1},x_k)$:
\[
\Breg{x^{\star}}{x_k}= \Breg{x^{\star}}{x_{k+1}}+\Breg{x_{k+1}}{x_k}
+\inner{\grad h(x_{k+1})-\grad h(x_k)}{x^{\star}-x_{k+1}}.
\tag{3}
\]
\textbf{[Step 4]}  Substitute the last term of (3) into (2):
\[
\inner{\grad f(x_k)}{x_{k+1}-x^{\star}}
\le
\frac{1}{\eta}\Big(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}\Big).
\tag{4}
\]
\textbf{[Step 5]}  By convexity of $f$,
\[
f(x_k)-f(x^{\star})\le\inner{\grad f(x_k)}{x_k-x^{\star}}.
\tag{5}
\]
\textbf{[Step 6]}  Add and subtract $x_{k+1}$ inside the inner product of (5):
\[
\inner{\grad f(x_k)}{x_k-x^{\star}}
=
\inner{\grad f(x_k)}{x_k-x_{k+1}}
+\inner{\grad f(x_k)}{x_{k+1}-x^{\star}}.
\tag{6}
\]
\textbf{[Step 7]}  Combine (4) and (6):
\[
f(x_k)-f(x^{\star})
\le
\inner{\grad f(x_k)}{x_k-x_{k+1}}
+\frac{1}{\eta}\Big(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}\Big).
\tag{7}
\]
\textbf{[Step 8]}  Apply the smoothness Assumption~\ref{as:convex} with $y=x_{k+1}$:
\[
f(x_{k+1})\le f(x_k)+\inner{\grad f(x_k)}{x_{k+1}-x_k}
+\frac{L}{2}\Breg{x_{k+1}}{x_k}.
\tag{8}
\]
Rearrange (8) to obtain a lower bound on the inner product:
\[
\inner{\grad f(x_k)}{x_k-x_{k+1}}
\le f(x_k)-f(x_{k+1})+\frac{L}{2}\Breg{x_{k+1}}{x_k}.
\tag{9}
\]
\textbf{[Step 9]}  Substitute (9) into (7):
\[
\begin{aligned}
f(x_k)-f(x^{\star})
&\le
\bigl[f(x_k)-f(x_{k+1})+\tfrac{L}{2}\Breg{x_{k+1}}{x_k}\bigr] \\
&\qquad+\frac{1}{\eta}\Big(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}-\Breg{x_{k+1}}{x_k}\Big) .
\end{aligned}
\tag{10}
\]
\textbf{[Step 10]}  Simplify (10) by canceling $\Breg{x_{k+1}}{x_k}$ terms:
\[
f(x_k)-f(x^{\star})
\le
f(x_k)-f(x_{k+1})
+\frac{1}{\eta}\bigl(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}\bigr)
+\Big(\frac{L}{2}-\frac{1}{\eta}\Big)\Breg{x_{k+1}}{x_k}.
\tag{11}
\]
\textbf{[Step 11]}  Since $\eta\le 1/L$ (Assumption~\ref{as:stepsize}), the coefficient
$\frac{L}{2}-\frac{1}{\eta}\le 0$. Dropping the non‑positive term yields
\[
f(x_k)-f(x^{\star})
\le
f(x_k)-f(x_{k+1})
+\frac{1}{\eta}\bigl(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}\bigr).
\tag{12}
\]
\textbf{[Step 12]}  Rearrange (12) to isolate the progress term $f(x_{k+1})-f(x^{\star})$:
\[
f(x_{k+1})-f(x^{\star})
\le
\frac{1}{\eta}\bigl(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}\bigr).
\tag{13}
\]
\textbf{[Step 13]}  Sum inequality (13) over $k=0,\dots,T-1$:
\[
\sum_{k=0}^{T-1}\bigl[f(x_{k+1})-f(x^{\star})\bigr]
\le
\frac{1}{\eta}\Big(\Breg{x^{\star}}{x_0}
-\Breg{x^{\star}}{x_T}\Big)
\le
\frac{\Breg{x^{\star}}{x_0}}{\eta},
\tag{14}
\]
where the last inequality uses non‑negativity of Bregman divergence.
\textbf{[Step 14]}  Divide both sides of (14) by $T$ and use convexity of $f$ to replace the average of function values by the function value at the average iterate $\bar{x}_T$:
\[
f(\bar{x}_T)-f(x^{\star})
\le
\frac{1}{T}\sum_{k=0}^{T-1}\bigl[f(x_{k+1})-f(x^{\star})\bigr]
\le
\frac{\Breg{x^{\star}}{x_0}}{\eta T}.
\tag{15}
\]
\textbf{[Step 15]}  To obtain the full bound of Theorem~\ref{thm:md}, we refine Step~9 by retaining the non‑positive term instead of discarding it. From (9) we have
\[
\inner{\grad f(x_k)}{x_k-x_{k+1}}
\le f(x_k)-f(x_{k+1})+\frac{L}{2}\Breg{x_{k+1}}{x_k},
\]
and inserting this into (7) gives
\[
f(x_k)-f(x^{\star})
\le
f(x_k)-f(x_{k+1})
+\frac{1}{\eta}\bigl(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}\bigr)
+\Big(\frac{L}{2}-\frac{1}{\eta}\Big)\Breg{x_{k+1}}{x_k}.
\]
Using $\|\grad f(x_k)\|_{*}\le G$ and the strong convexity of $h$,
$\Breg{x_{k+1}}{x_k}\ge \tfrac12\|x_{k+1}-x_k\|^{2}$,
and the optimality condition (Lemma~\ref{lem:optcond}) one can bound the last term by $\tfrac{\eta L}{2}G^{2}$. Consequently,
\[
f(x_{k+1})-f(x^{\star})
\le
\frac{1}{\eta}\bigl(\Breg{x^{\star}}{x_k}
-\Breg{x^{\star}}{x_{k+1}}\bigr)
+\frac{\eta L}{2}G^{2}.
\tag{16}
\]
Summing (16) over $k$ and dividing by $T$ yields
\[
f(\bar{x}_T)-f(x^{\star})
\le
\frac{\Breg{x^{\star}}{x_0}}{\eta T}
+\frac{\eta L}{2}G^{2},
\]
which is exactly the statement of Theorem~\ref{thm:md}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:md} we have
\[
\mathcal{E}_T:=f(\bar{x}_T)-f(x^{\star})
\le
\frac{D_0}{\eta T}+ \frac{\eta L}{2}G^{2},
\qquad D_0:=\Breg{x^{\star}}{x_0}.
\]
Optimizing the right‑hand side w.r.t. $\eta$ gives
\[
\eta^{\star}= \sqrt{\frac{2D_0}{L G^{2}T}},
\]
and plugging $\eta^{\star}$ back yields
\[
\mathcal{E}_T\le
\sqrt{\frac{2L D_0 G^{2}}{T}} = O\!\left(\frac{1}{\sqrt{T}}\right).
\]
If the stepsize is fixed as $\eta=1/L$, the bound simplifies to
\[
\mathcal{E}_T\le \frac{L D_0}{T}+ \frac{G^{2}}{2L},
\]
showing an $O(1/T)$ decay up to a constant bias term.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Euclidean Setting.} $h(x)=\tfrac12\|x\|_{2}^{2}$ yields $\Breg{x}{y}=\tfrac12\|x-y\|_{2}^{2}$. The algorithm reduces to standard gradient descent and the bound matches the classical $O(1/T)$ rate for smooth convex functions.
\item \textbf{Entropy Prox.} For $h(x)=\sum_{i}x_i\log x_i$ on the simplex, $\Breg{x}{y}$ is the Kullback–Leibler divergence. The MD update becomes the exponentiated gradient algorithm; the same $O(1/T)$ guarantee holds with $D_0$ equal to the KL divergence between $x^{\star}$ and $x_0$.
\end{itemize}
