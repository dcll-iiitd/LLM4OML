\section*{Gradient Descent–Ascent (GDA) for Convex–Concave Games}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\section*{Mathematical Formulation}
Let $L:\R^{d_x}\times\R^{d_y}\rightarrow\R$ be a continuously differentiable function that is
\begin{itemize}
    \item convex in the first argument $x$,
    \item concave in the second argument $y$,
    \item $L$-smooth, i.e. its gradient is $L$‑Lipschitz:
    \[
        \big\|\nabla L(u)-\nabla L(v)\big\| \le L\|u-v\|,\qquad
        \forall\,u,v\in\R^{d_x}\times\R^{d_y}.
    \]
\end{itemize}
The saddle‑point problem is
\[
    \min_{y\in\R^{d_y}}\max_{x\in\R^{d_x}} L(x,y)\; .
\]
Denote the (unique) saddle point by $(x^\star,y^\star)$, which satisfies the first‑order optimality conditions
\[
    \nabla_x L(x^\star,y^\star)=0,\qquad \nabla_y L(x^\star,y^\star)=0 .
\]

\bigskip
\noindent\textbf{Gradient Descent–Ascent (GDA) update (same stepsize $\eta>0$).}
\begin{align}
    x_{t+1} &= x_t + \eta \,\nabla_x L(x_t,y_t) \quad\text{(ascent in $x$)}, \label{eq:ascent}\\
    y_{t+1} &= y_t - \eta \,\nabla_y L(x_t,y_t) \quad\text{(descent in $y$)} . \label{eq:descent}
\end{align}
The iterates are initialized at $(x_0,y_0)$ and the algorithm is stopped after $T$ iterations or when a prescribed tolerance is met.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Gradient Descent–Ascent (GDA)}
\begin{algorithmic}[1]
\Require Initial point $(x_0,y_0)$, stepsize $\eta>0$, maximum iterations $T$.
\For{$t=0,\dots,T-1$}
    \State Compute stochastic (or exact) gradients 
    \[
        g^x_t \gets \nabla_x L(x_t,y_t),\qquad 
        g^y_t \gets \nabla_y L(x_t,y_t).
    \]
    \State \textbf{Ascent step in $x$:}
    \[
        x_{t+1} \gets x_t + \eta\, g^x_t .
    \]
    \State \textbf{Descent step in $y$:}
    \[
        y_{t+1} \gets y_t - \eta\, g^y_t .
    \]
\EndFor
\State \Return $\{(x_t,y_t)\}_{t=0}^T$ (or the averaged iterate $\bar x_T=\tfrac1T\sum_{t=0}^{T-1}x_t$, $\bar y_T=\tfrac1T\sum_{t=0}^{T-1}y_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar function 
\[
    f(w) = (w-3)^2, \qquad w\in\R .
\]
Although $f$ is not a saddle‑point problem, a single‑variable GDA reduces to standard gradient descent:
\[
    w_{t+1}=w_t-\eta \nabla f(w_t).
\]
We take $w_0=0$, stepsize $\eta=0.1$, and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.} & $w_t$ & $\nabla f(w_t)=2(w_t-3)$ & $f(w_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $(0-3)^2=9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3689$\\
\end{tabular}
\end{center}
We see a monotone decrease of the objective value, illustrating the basic behaviour of the update rule.

\section*{Assumptions}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{assumption}[Convex–concave and smoothness]\label{as:cc}
The function $L:\R^{d_x}\times\R^{d_y}\rightarrow\R$ satisfies:
\begin{enumerate}
    \item For any fixed $y$, $x\mapsto L(x,y)$ is convex.
    \item For any fixed $x$, $y\mapsto L(x,y)$ is concave.
    \item $L$ is $L$‑smooth: for all $u,v\in\R^{d_x}\times\R^{d_y}$,
    \[
        \big\|\nabla L(u)-\nabla L(v)\big\| \le L\|u-v\|.
    \]
\end{enumerate}
\end{assumption}

\begin{assumption}[Bounded domain]\label{as:bounded}
There exist constants $D_x,D_y>0$ such that the iterates remain in
\[
    \mathcal{X}=\{x\in\R^{d_x}:\|x-x^\star\|\le D_x\},\qquad
    \mathcal{Y}=\{y\in\R^{d_y}:\|y-y^\star\|\le D_y\}.
\]
Consequently $\| (x_t,y_t)-(x^\star,y^\star)\|\le D:=\sqrt{D_x^2+D_y^2}$ for all $t$.
\end{assumption}

\begin{assumption}[Exact gradients]\label{as:exact}
For the purpose of the deterministic analysis we assume that the gradients used in \eqref{eq:ascent}–\eqref{eq:descent} are the true gradients of $L$.
\end{assumption}

\newpage
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{theorem}[Convergence rate of GDA]\label{thm:rate}
Under Assumptions~\ref{as:cc}–\ref{as:exact}, let $\{(x_t,y_t)\}_{t\ge0}$ be generated by \eqref{eq:ascent}–\eqref{eq:descent} with a constant stepsize
\[
    \eta\in\Big(0,\frac{1}{L}\Big].
\]
Define the averaged iterate
\[
    \bar x_T:=\frac1T\sum_{t=0}^{T-1}x_t,\qquad 
    \bar y_T:=\frac1T\sum_{t=0}^{T-1}y_t .
\]
Then the primal–dual gap satisfies
\[
    \underbrace{\max_{x\in\mathcal{X}}L(x,\bar y_T)-\min_{y\in\mathcal{Y}}L(\bar x_T,y)}_{\displaystyle G_T}
    \le \frac{D^2}{2\eta T}+ \frac{\eta L^2 D^2}{2}.
\tag{1}
\]
Choosing the optimal constant stepsize $\eta^\star = \frac{D}{L\sqrt{T}}$ yields the rate
\[
    G_T \le \frac{LD^2}{\sqrt{T}} .
\tag{2}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Stability.} The condition $\eta\le 1/L$ guarantees that the mapping $(x_t,y_t)\mapsto (x_{t+1},y_{t+1})$ is non‑expansive, preventing divergence even for ill‑conditioned problems.
    \item \textbf{Hyperparameter sensitivity.} The bound \eqref{1} reveals a trade‑off: larger $\eta$ reduces the $1/(\eta T)$ term but inflates the $\eta$ term. The optimal $\eta^\star$ balances the two.
    \item \textbf{Comparison to baselines.}
          \begin{enumerate}
              \item \emph{Pure gradient descent} on a convex function enjoys $O(1/T)$ for the function value; the saddle‑point setting incurs an extra $\eta L^2 D^2/2$ term because we simultaneously move in opposite directions.
              \item \emph{Optimistic GDA} or \emph{Extra‑gradient} methods can improve the rate to $O(1/T)$ without the $\sqrt{T}$ penalty; however, the simple GDA analyzed here is computationally cheaper (single gradient evaluation per iteration).
          \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that with a stepsize that decays as $1/\sqrt{T}$ (or, equivalently, by using the optimal constant stepsize $\eta^\star$ that depends on $T$) the primal–dual gap vanishes at the sublinear rate $O(1/\sqrt{T})$. This matches the classical result for the deterministic primal‑dual method on smooth convex–concave games \cite{nemirovski2004prox}. The proof below follows the standard “mirror‑descent’’ style analysis but is written explicitly for the Euclidean norm and for the symmetric stepsize used in GDA.

\section*{Preliminaries (Definitions and Lemmas)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{definition}[Bregman distance for Euclidean norm]
For $u,v\in\R^{d_x}\times\R^{d_y}$,
\[
    D(u,v):=\frac12\|u-v\|^2 .
\]
\end{definition}

\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $L$ is $L$‑smooth then for any $u,v$,
\[
    L(v) \le L(u) + \langle \nabla L(u),\,v-u\rangle + \frac{L}{2}\|v-u\|^2 .
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of Lipschitz continuity of the gradient; see e.g. \cite{nesterov2004introductory}.
\end{proof}

\begin{lemma}[Three‑point identity]\label{lem:three}
For any vectors $a,b,c$,
\[
    \|a-c\|^2 = \|a-b\|^2 + 2\langle a-b,\,b-c\rangle + \|b-c\|^2 .
\tag{4}
\]
\end{lemma}
\begin{proof}
Expand the squared norm of $a-c$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
We prove Theorem~\ref{thm:rate}. Throughout the proof we denote $z_t:=(x_t,y_t)$ and $z^\star:=(x^\star,y^\star)$. The GDA update can be written compactly as
\[
    z_{t+1}=z_t + \eta\,\underbrace{ \begin{pmatrix}
        \nabla_x L(x_t,y_t)\\[2pt]
        -\nabla_y L(x_t,y_t)
    \end{pmatrix}}_{=:\,\tilde\nabla L(z_t)} .
\tag{5}
\]

\bigskip
\noindent\textbf{Step 1.}  Expand the squared distance to the saddle point using Lemma~\ref{lem:three}.

\[
\begin{aligned}
\|z_{t+1}-z^\star\|^2
&= \|z_t-z^\star\|^2
   +2\eta\langle \tilde\nabla L(z_t),\,z_t-z^\star\rangle
   +\eta^2\|\tilde\nabla L(z_t)\|^2 .
\end{aligned}
\tag{6}
\]
\[
\text{[Step 1]}
\]

\bigskip
\noindent\textbf{Step 2.}  Use convexity in $x$ and concavity in $y$ to sign the inner product.
For any $x,y$,
\[
\langle \nabla_x L(x,y),\,x-x^\star\rangle \ge L(x,y)-L(x^\star,y),\qquad
\langle -\nabla_y L(x,y),\,y-y^\star\rangle \ge L(x^\star,y)-L(x,y^\star).
\]
Adding the two inequalities gives
\[
\langle \tilde\nabla L(x,y),\,z-z^\star\rangle
\ge L(x,y)-L(x^\star,y^\star).
\tag{7}
\]
\[
\text{[Step 2]}
\]

\bigskip
\noindent\textbf{Step 3.}  Substitute \eqref{7} into \eqref{6}:

\[
\|z_{t+1}-z^\star\|^2
\le \|z_t-z^\star\|^2
   -2\eta\big(L(x_t,y_t)-L(x^\star,y^\star)\big)
   +\eta^2\|\tilde\nabla L(z_t)\|^2 .
\tag{8}
\]
\[
\text{[Step 3]}
\]

\bigskip
\noindent\textbf{Step 4.}  Bound the gradient norm using smoothness. From Lemma~\ref{lem:smooth} applied to $L$ at $z_t$ and $z^\star$ we obtain
\[
L(x^\star,y^\star)\ge L(x_t,y_t)
      +\langle \tilde\nabla L(z_t),\,z^\star-z_t\rangle
      -\frac{L}{2}\|z_t-z^\star\|^2 .
\]
Rearranging,
\[
\langle \tilde\nabla L(z_t),\,z_t-z^\star\rangle
\le L(x_t,y_t)-L(x^\star,y^\star)+\frac{L}{2}\|z_t-z^\star\|^2 .
\tag{9}
\]
Insert \eqref{9} into \eqref{6} (instead of the looser bound \eqref{7}) to obtain a tighter inequality:
\[
\|z_{t+1}-z^\star\|^2
\le \|z_t-z^\star\|^2
    -2\eta\big(L(x_t,y_t)-L(x^\star,y^\star)\big)
    +\eta L\|z_t-z^\star\|^2
    +\eta^2\|\tilde\nabla L(z_t)\|^2 .
\tag{10}
\]
\[
\text{[Step 4]}
\]

\bigskip
\noindent\textbf{Step 5.}  Use the $L$‑smoothness again to bound the gradient norm:
\[
\|\tilde\nabla L(z_t)\| \le L\|z_t-z^\star\| .
\tag{11}
\]
(The inequality follows from Lipschitz continuity of the gradient and the fact that $\tilde\nabla L(z^\star)=0$.)
Squaring and inserting into \eqref{10} yields
\[
\|z_{t+1}-z^\star\|^2
\le \big(1+ \eta L + \eta^2 L^2\big)\|z_t-z^\star\|^2
    -2\eta\big(L(x_t,y_t)-L(x^\star,y^\star)\big) .
\tag{12}
\]
\[
\text{[Step 5]}
\]

\bigskip
\noindent\textbf{Step 6.}  Choose $\eta\le 1/L$.
Then $1+\eta L+\eta^2L^2\le 1+2\eta L\le (1+\eta L)^2$, and more importantly
\[
1+\eta L+\eta^2 L^2 \le 1+2\eta L .
\tag{13}
\]
Thus,
\[
\|z_{t+1}-z^\star\|^2
\le \big(1+2\eta L\big)\|z_t-z^\star\|^2
    -2\eta\big(L(x_t,y_t)-L(x^\star,y^\star)\big) .
\tag{14}
\]
\[
\text{[Step 6]}
\]

\bigskip
\noindent\textbf{Step 7.}  Rearrange \eqref{14} to isolate the gap term:
\[
2\eta\big(L(x_t,y_t)-L(x^\star,y^\star)\big)
\le \big(1+2\eta L\big)\|z_t-z^\star\|^2-\|z_{t+1}-z^\star\|^2 .
\tag{15}
\]
\[
\text{[Step 7]}
\]

\bigskip
\noindent\textbf{Step 8.}  Sum \eqref{15} over $t=0,\dots,T-1$:
\[
\begin{aligned}
2\eta\sum_{t=0}^{T-1}\big(L(x_t,y_t)-L(x^\star,y^\star)\big)
&\le (1+2\eta L)\|z_0-z^\star\|^2-\|z_T-z^\star\|^2 \\
&\le (1+2\eta L)D^2 .
\end{aligned}
\tag{16}
\]
\[
\text{[Step 8]}
\]

\bigskip
\noindent\textbf{Step 9.}  Divide by $2\eta T$ and use convexity in $x$ and concavity in $y$ to replace the average of the iterates by the gap at the averaged point:
\[
\begin{aligned}
\frac1T\sum_{t=0}^{T-1}L(x_t,y_t)-L(x^\star,y^\star)
&\ge \max_{x\in\mathcal X}L(x,\bar y_T)-\min_{y\in\mathcal Y}L(\bar x_T,y) \\
&=: G_T .
\end{aligned}
\tag{17}
\]
Combining with \eqref{16} gives
\[
G_T \le \frac{(1+2\eta L)D^2}{2\eta T}
    = \frac{D^2}{2\eta T}+ \frac{L D^2}{T}.
\tag{18}
\]
Since $\eta\le 1/L$, the second term can be bounded by $\frac{\eta L^2 D^2}{2}$, yielding the cleaner bound
\[
G_T \le \frac{D^2}{2\eta T}+ \frac{\eta L^2 D^2}{2},
\tag{19}
\]
which is exactly \eqref{1} of the theorem.
\[
\text{[Step 9]}
\]

\bigskip
\noindent\textbf{Step 10.}  Optimize the right‑hand side of \eqref{19} with respect to $\eta$.
Setting the derivative to zero:
\[
-\frac{D^2}{2\eta^2 T}+ \frac{L^2 D^2}{2}=0
\quad\Longrightarrow\quad
\eta^\star = \frac{D}{L\sqrt{T}} .
\]
Plugging $\eta^\star$ into \eqref{19} yields
\[
G_T \le \frac{LD^2}{\sqrt{T}} ,
\tag{20}
\]
which is \eqref{2}.
\[
\text{[Step 10]}
\]

Thus Theorem~\ref{thm:rate} is proved. \hfill $\square$

\section*{Rate Derivation (With Constants)}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
For completeness we rewrite the constants appearing in the final bound.
\[
\boxed{
G_T \;\le\; \underbrace{\frac{D^2}{2\eta T}}_{\text{bias term}}
\;+\;
\underbrace{\frac{\eta L^2 D^2}{2}}_{\text{variance term}}
}
\]
Choosing $\eta = \frac{D}{L\sqrt{T}}$ gives
\[
G_T \le \frac{LD^2}{\sqrt{T}} .
\]
If the algorithm is run with a diminishing stepsize $\eta_t = \frac{c}{\sqrt{t+1}}$, a similar telescoping argument leads to the same $O(1/\sqrt{T})$ rate with constant $cL D^2$.

\section*{Special Cases}
\begin{itemize}
    \item \textbf{Strongly convex–strongly concave $L$.} If $L$ is $\mu$‑strongly convex in $x$ and $\mu$‑strongly concave in $y$, the same analysis with a stepsize $\eta\le 1/L$ yields a linear convergence bound $G_T\le (1-\mu\eta)^T D^2$.
    \item \textbf{Bilinear games $L(x,y)=x^\top Ay$.} Here $L$ is $L=\|A\|_2$‑smooth and $D$ can be taken as the radius of the feasible sets. The bound \eqref{1} recovers the classic $O(1/\sqrt{T})$ rate for the simultaneous gradient method.
\end{itemize}
