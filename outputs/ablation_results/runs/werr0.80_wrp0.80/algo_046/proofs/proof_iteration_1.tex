\section*{AdaGrad with Running Sum of Gradients}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be a convex, $L$‑smooth objective.  
Denote the stochastic (or full) gradient observed at iteration $k$ by 
\[
g_{k}\;:=\;\nabla f(w_{k})\in\RR^{d}.
\]
Define the element‑wise accumulated squared gradients
\[
G_{k}\;:=\;\sum_{i=1}^{k} g_{i}^{2}\;\in\RR^{d},
\]
where the square and the sum are taken component‑wise.  
For a scalar stepsize hyperparameter $\eta>0$ the AdaGrad update is
\[
\boxed{%
w_{k+1}=w_{k}-\eta \, D_{k}^{-1}\, g_{k}},\qquad 
D_{k}:=\operatorname{diag}\!\bigl(\sqrt{G_{k}}+\epsilon\bigr),
\tag{1}
\]
with a small constant $\epsilon>0$ (often $\epsilon=10^{-8}$) to avoid division by zero.  
When $d=1$ the update simplifies to
\[
w_{k+1}=w_{k}-\frac{\eta}{\sqrt{\sum_{i=1}^{k} g_{i}^{2}}+\epsilon}\;g_{k}.
\tag{2}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AdaGrad (vector version)}\label{alg:adagrad}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_{0}\in\RR^{d}$, stepsize $\eta>0$, $\epsilon>0$
\State $G_{0}\gets 0\in\RR^{d}$ \Comment{accumulator}
\For{$k=0,1,\dots,T-1$}
    \State $g_{k}\gets\nabla f(w_{k})$ \Comment{gradient}
    \State $G_{k+1}\gets G_{k}+g_{k}^{2}$ \Comment{component‑wise square and sum}
    \State $D_{k+1}\gets\operatorname{diag}\!\bigl(\sqrt{G_{k+1}}+\epsilon\bigr)$
    \State $w_{k+1}\gets w_{k}-\eta\, D_{k+1}^{-1} g_{k}$
\EndFor
\State \textbf{Output:} $w_{T}$ (or the average $\bar w_T:=\tfrac1T\sum_{k=1}^{T}w_{k}$)
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^{2}$, which is $2$‑smooth and convex.
\[
\nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, stepsize $\eta=0.1$, $\epsilon=10^{-8}$ and run three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_{k}$ & $g_{k}= \nabla f(w_{k})$ & $G_{k}= \sum_{i=1}^{k}g_{i}^{2}$ & $w_{k+1}=w_{k}-\dfrac{\eta}{\sqrt{G_{k}}+\epsilon}g_{k}$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0$ & $w_{1}=0-\dfrac{0.1}{\sqrt{0}+10^{-8}}(-6)=0+0.6\approx0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $(-6)^{2}=36$ & $w_{2}=0.6-\dfrac{0.1}{\sqrt{36}+10^{-8}}(-4.8)=0.6+0.08=0.68$ \\
2 & $0.68$ & $2(0.68-3)=-4.64$ & $36+(-4.8)^{2}=36+23.04=59.04$ & $w_{3}=0.68-\dfrac{0.1}{\sqrt{59.04}+10^{-8}}(-4.64)
\approx0.68+0.0605=0.7405$
\end{tabular}
\end{center}
The objective values are $f(w_{0})=9$, $f(w_{1})\approx5.76$, $f(w_{2})\approx5.29$, $f(w_{3})\approx5.07$, showing a decreasing trend.

\section*{Assumptions}
\begin{assumption}[Convexity]\label{as:convex}
$f:\RR^{d}\rightarrow\RR$ is convex, i.e.
\[
f(y)\ge f(x)+\inner{\nabla f(x)}{y-x},\qquad\forall x,y\in\RR^{d}.
\]
\end{assumption}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth:
\[
\norm{\nabla f(x)-\nabla f(y)}\le L\norm{x-y},\qquad\forall x,y\in\RR^{d}.
\]
Equivalently,
\[
f(y)\le f(x)+\inner{\nabla f(x)}{y-x}+\frac{L}{2}\norm{y-x}^{2}.
\]
\end{assumption}
\begin{assumption}[Bounded Gradients]\label{as:bound}
There exists $G_{\infty}>0$ such that $\norm{g_{k}}\le G_{\infty}$ for all $k$ (holds for the quadratic example).
\end{assumption}
\begin{assumption}[Domain Diameter]\label{as:diam}
The feasible set $\mathcal{W}\subset\RR^{d}$ is convex and has bounded diameter 
\[
D:=\max_{u,v\in\mathcal{W}}\norm{u-v}<\infty .
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[AdaGrad Regret Bound]\label{thm:regret}
Under Assumptions \ref{as:convex}–\ref{as:diam}, for any $w^{*}\in\mathcal{W}$ the iterates generated by (1) satisfy
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\;\le\;
\frac{D^{2}}{2\eta}\sum_{j=1}^{d}\sqrt{G_{T,j}}
\;+\;
\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon},
\tag{3}
\]
where $G_{k,j}$ denotes the $j$‑th coordinate of $G_{k}$. Consequently, using convexity,
\[
f(\bar w_{T})-f(w^{*})\;\le\;
\frac{D^{2}}{2\eta T}\sum_{j=1}^{d}\sqrt{G_{T,j}}
\;+\;
\frac{\eta}{2T}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon},
\tag{4}
\]
where $\bar w_{T}:=\frac1T\sum_{k=0}^{T-1}w_{k}$.
\end{theorem}
When all gradients are bounded by $G_{\infty}$, $\sqrt{G_{T,j}}\le G_{\infty}\sqrt{T}$, and the right‑hand side of (4) becomes $O\!\bigl(\frac{DG_{\infty}\sqrt{d}}{\sqrt{T}}\bigr)$. Thus AdaGrad attains the optimal $O(1/\sqrt{T})$ rate for convex smooth problems.

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The adaptive matrix $D_{k}^{-1}$ automatically shrinks the stepsize along directions with large accumulated gradients, preventing explosion even when a fixed $\eta$ would be too aggressive.
\item \textbf{Hyper‑parameter Sensitivity.} The bound (4) shows that the performance depends on $\eta$ only through the trade‑off $\frac{D^{2}}{2\eta}$ vs. $\frac{\eta}{2}$; the optimal choice is $\eta^{\star}=D\bigl(\sum_{j}\sqrt{G_{T,j}}\bigr)^{-1/2}$, yielding the $O(1/\sqrt{T})$ rate.
\item \textbf{Comparison to SGD.} For constant stepsize SGD the regret scales as $O(\eta T)$, requiring $\eta=O(1/\sqrt{T})$ to obtain $O(1/\sqrt{T})$ error, whereas AdaGrad adapts $\eta$ automatically per coordinate.
\item \textbf{Extension to Stochastic Gradients.} The same proof holds with $g_{k}=\nabla f(w_{k};\xi_{k})$, provided the unbiasedness $\EE[g_{k}\mid w_{k}]=\nabla f(w_{k})$ and bounded second moment $\EE[\|g_{k}\|^{2}]\le G_{\infty}^{2}$.
\end{itemize}

\section*{Discussion}
The theorem is derived directly from the geometry of the adaptive metric $D_{k}$ and requires no strong convexity. The bound scales with $\sqrt{G_{T}}$, which is data‑dependent: if the gradients are sparse or have small magnitude in many coordinates, the effective stepsize along those coordinates remains large, explaining AdaGrad’s empirical success on sparse problems.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑Point Identity]\label{lem:three}
For any $u,v\in\RR^{d}$ and any positive definite diagonal matrix $D$,
\[
\|u-v\|_{D}^{2}
= \|u-w\|_{D}^{2}+\|w-v\|_{D}^{2}
-2\inner{D(u-w)}{w-v},
\]
where $\|x\|_{D}^{2}:=x^{\top}Dx$.
\end{lemma}
\begin{proof}
Immediate expansion of the squared norm. \qedhere
\end{proof}

\begin{lemma}[Bound on Adaptive Steps]\label{lem:step}
For any $a\ge0$, $b\ge0$ and $\epsilon>0$,
\[
\frac{a^{2}}{\sqrt{b}+\epsilon}\;\le\;2\bigl(\sqrt{b+a^{2}}-\sqrt{b}\bigr)+\epsilon.
\tag{5}
\]
\end{lemma}
\begin{proof}
Consider the function $\phi(b)=\sqrt{b+a^{2}}-\sqrt{b}$. By the mean‑value theorem,
\[
\phi(b)=\frac{a^{2}}{2(\sqrt{b+\theta a^{2}})}\le\frac{a^{2}}{2\sqrt{b}}
\]
for some $\theta\in(0,1)$. Rearranging yields (5). \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:regret}. Let $w^{*}\in\mathcal{W}$ be arbitrary. Define the diagonal matrix
\[
D_{k}:=\operatorname{diag}\!\bigl(\sqrt{G_{k}}+\epsilon\bigr),\qquad
D_{k}^{-1}:=\operatorname{diag}\!\Bigl(\frac{1}{\sqrt{G_{k}}+\epsilon}\Bigr).
\]

\noindent\textbf{[Step 1]} Expand the squared distance after the update using Lemma~\ref{lem:three}:
\[
\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
= \|w_{k}-w^{*}\|_{D_{k+1}}^{2}
-2\eta\inner{g_{k}}{w_{k}-w^{*}}
+\eta^{2}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{6}
\]

\noindent\textbf{[Step 2]} Since $D_{k+1}\succeq D_{k}$ component‑wise (each diagonal entry is non‑decreasing), we have
\[
\|w_{k}-w^{*}\|_{D_{k}}^{2}\le\|w_{k}-w^{*}\|_{D_{k+1}}^{2}.
\tag{7}
\]

\noindent\textbf{[Step 3]} Combine (6) and (7) and drop the non‑negative term $\|w_{k}-w^{*}\|_{D_{k}}^{2}$:
\[
\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
\le \|w_{k}-w^{*}\|_{D_{k}}^{2}
-2\eta\inner{g_{k}}{w_{k}-w^{*}}
+\eta^{2}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{8}
\]

\noindent\textbf{[Step 4]} Rearranging (8) gives a bound on the inner product term:
\[
2\eta\inner{g_{k}}{w_{k}-w^{*}}
\le \|w_{k}-w^{*}\|_{D_{k}}^{2}
-\|w_{k+1}-w^{*}\|_{D_{k+1}}^{2}
+\eta^{2}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{9}
\]

\noindent\textbf{[Step 5]} Sum (9) over $k=0,\dots,T-1$:
\[
2\eta\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \|w_{0}-w^{*}\|_{D_{0}}^{2}
-\|w_{T}-w^{*}\|_{D_{T}}^{2}
+\eta^{2}\sum_{k=0}^{T-1}\|g_{k}\|_{D_{k+1}^{-1}}^{2}.
\tag{10}
\]

\noindent\textbf{[Step 6]} Because $D_{0}=\operatorname{diag}(\epsilon)$, the first term is bounded by
\[
\|w_{0}-w^{*}\|_{D_{0}}^{2}
= \sum_{j=1}^{d}\epsilon\,(w_{0,j}-w^{*}_{j})^{2}
\le \epsilon D^{2}.
\tag{11}
\]
We drop the non‑positive term $-\|w_{T}-w^{*}\|_{D_{T}}^{2}$.

\noindent\textbf{[Step 7]} The remaining sum involves the adaptive norm:
\[
\|g_{k}\|_{D_{k+1}^{-1}}^{2}
= \sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}.
\tag{12}
\]

\noindent\textbf{[Step 8]} Insert (11)–(12) into (10) and divide by $2\eta$:
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \frac{\epsilon D^{2}}{2\eta}
+\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k+1,j}}+\epsilon}.
\tag{13}
\]

\noindent\textbf{[Step 9]} Since $G_{k+1,j}=G_{k,j}+g_{k,j}^{2}$, Lemma~\ref{lem:step} with $a=|g_{k,j}|$ and $b=G_{k,j}$ yields
\[
\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}
\le 2\bigl(\sqrt{G_{k+1,j}}-\sqrt{G_{k,j}}\bigr)+\epsilon .
\tag{14}
\]

\noindent\textbf{[Step 10]} Summing (14) over $k$ telescopes:
\[
\sum_{k=0}^{T-1}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}
\le 2\sqrt{G_{T,j}}+T\epsilon .
\tag{15}
\]

\noindent\textbf{[Step 11]} Plug (15) into (13):
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \frac{\epsilon D^{2}}{2\eta}
+\frac{\eta}{2}\sum_{j=1}^{d}\bigl(2\sqrt{G_{T,j}}+T\epsilon\bigr).
\tag{16}
\]

\noindent\textbf{[Step 12]} Choose $\epsilon$ arbitrarily small (standard practice $\epsilon\to0$). Dropping the $\epsilon$–terms gives the clean bound
\[
\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le \frac{D^{2}}{2\eta}\sum_{j=1}^{d}\sqrt{G_{T,j}}
+\frac{\eta}{2}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon},
\]
which is exactly (3). \qedhere

\section*{Rate Derivation (With Constants)}
From convexity,
\[
f(w_{k})-f(w^{*})\le\inner{g_{k}}{w_{k}-w^{*}}.
\]
Averaging over $k=0,\dots,T-1$ and using (3):
\[
f(\bar w_{T})-f(w^{*})
\le\frac{1}{T}\sum_{k=0}^{T-1}\inner{g_{k}}{w_{k}-w^{*}}
\le\frac{D^{2}}{2\eta T}\sum_{j=1}^{d}\sqrt{G_{T,j}}
+\frac{\eta}{2T}\sum_{k=0}^{T-1}\sum_{j=1}^{d}\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}.
\]
Using the bound $\sqrt{G_{T,j}}\le G_{\infty}\sqrt{T}$ (from Assumption~\ref{as:bound}) and $\frac{g_{k,j}^{2}}{\sqrt{G_{k,j}}+\epsilon}\le G_{\infty}\sqrt{G_{k,j}}/(\sqrt{G_{k,j}})=G_{\infty}$, we obtain
\[
f(\bar w_{T})-f(w^{*})
\le\frac{D^{2}G_{\infty}\sqrt{d}}{2\eta\sqrt{T}}}
+\frac{\eta G_{\infty} d}{2}.
\]
Optimizing $\eta$ by setting the two terms equal gives
\[
\eta^{\star}= \frac{D\sqrt{d}}{G_{\infty}\sqrt{T}}.
\]
Substituting $\eta^{\star}$ yields the final rate
\[
f(\bar w_{T})-f(w^{*})\;\le\; \frac{DG_{\infty}\sqrt{d}}{\sqrt{T}}.
\tag{17}
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{One‑dimensional case ($d=1$).} Equation (17) reduces to $O\bigl(\frac{DG_{\infty}}{\sqrt{T}}\bigr)$, matching the classic scalar AdaGrad bound.
\item \textbf{Sparse gradients.} If only $s\ll d$ coordinates receive non‑zero gradients, then $\sum_{j=1}^{d}\sqrt{G_{T,j}}=s\cdot O(\sqrt{T})$, improving the constant factor to $\sqrt{s}$.
\item \textbf{Strongly convex $f$.} Adding $\mu$‑strong convexity yields a $O\bigl(\frac{\log T}{T}\bigr)$ bound after the same derivation, but this is omitted for brevity.
\end{itemize}
