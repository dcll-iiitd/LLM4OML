\section*{Algorithm Name}
Nesterov's Accelerated Gradient Descent (NAG) for Strongly Convex and Smooth Functions.

\section*{Mathematical Formulation}
The algorithm optimizes an objective function $f(w)$ that is $L$-smooth and $\mu$-strongly convex. It maintains two sequences of iterates, $x_k$ (the main sequence) and $y_k$ (the extrapolated sequence). 

Given an initial point $x_0 \in \mathbb{R}^d$, we set $x_{-1} = x_0$. 
The update rules at iteration $k \ge 0$ are defined as:
\begin{align*}
    y_k &= x_k + \beta (x_k - x_{k-1}) \\
    x_{k+1} &= y_k - \alpha \nabla f(y_k)
\end{align*}
where the hyperparameters are strictly defined by the condition number $\kappa = L/\mu$:
\begin{itemize}
    \item Learning rate (step size): $\alpha = \frac{1}{L}$
    \item Momentum coefficient: $\beta = \frac{\sqrt{L} - \sqrt{\mu}}{\sqrt{L} + \sqrt{\mu}} = \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}$
\end{itemize}

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Nesterov's Accelerated Gradient Descent (Strongly Convex)}
\begin{algorithmic}[1]
\REQUIRE Objective function $f$, smoothness $L$, strong convexity $\mu$, initial point $x_0$, number of iterations $T$.
\STATE Initialize $x_{-1} \gets x_0$
\STATE Compute step size $\alpha \gets \frac{1}{L}$
\STATE Compute momentum $\beta \gets \frac{\sqrt{L} - \sqrt{\mu}}{\sqrt{L} + \sqrt{\mu}}$
\FOR{$k = 0$ \TO $T-1$}
    \STATE $y_k \gets x_k + \beta (x_k - x_{k-1})$
    \STATE $g_k \gets \nabla f(y_k)$
    \STATE $x_{k+1} \gets y_k - \alpha g_k$
\ENDFOR
\RETURN $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Let us apply the algorithm to the function $f(w) = (w-3)^2$. 
The gradient is $\nabla f(w) = 2(w-3)$ and the second derivative is $f''(w) = 2$.
Strictly speaking, for this function $L = 2$ and $\mu = 2$, which yields $\beta = 0$ (reducing to standard gradient descent). To explicitly demonstrate the two-sequence momentum mechanics requested, we will perform the dry run assuming a generic momentum parameter $\beta = 0.5$ and a learning rate $\alpha = 0.1$, starting from $w_0 = 0$.

\textbf{Initialization:}
$x_0 = 0$, $x_{-1} = 0$.

\textbf{Iteration 0 ($k=0$):}
\begin{itemize}
    \item $y_0 = x_0 + 0.5(x_0 - x_{-1}) = 0 + 0.5(0) = 0$
    \item $\nabla f(y_0) = 2(0 - 3) = -6$
    \item $x_1 = y_0 - 0.1 \nabla f(y_0) = 0 - 0.1(-6) = 0.6$
    \item Objective value: $f(x_1) = (0.6 - 3)^2 = 5.76$
\end{itemize}

\textbf{Iteration 1 ($k=1$):}
\begin{itemize}
    \item $y_1 = x_1 + 0.5(x_1 - x_0) = 0.6 + 0.5(0.6 - 0) = 0.9$
    \item $\nabla f(y_1) = 2(0.9 - 3) = -4.2$
    \item $x_2 = y_1 - 0.1 \nabla f(y_1) = 0.9 - 0.1(-4.2) = 1.32$
    \item Objective value: $f(x_2) = (1.32 - 3)^2 = 2.8224$
\end{itemize}

\textbf{Iteration 2 ($k=2$):}
\begin{itemize}
    \item $y_2 = x_2 + 0.5(x_2 - x_1) = 1.32 + 0.5(1.32 - 0.6) = 1.68$
    \item $\nabla f(y_2) = 2(1.68 - 3) = -2.64$
    \item $x_3 = y_2 - 0.1 \nabla f(y_2) = 1.68 - 0.1(-2.64) = 1.944$
    \item Objective value: $f(x_3) = (1.944 - 3)^2 \approx 1.1151$
\end{itemize}

\section*{Assumptions}
The convergence proof relies on the following formal assumptions about the objective function $f: \mathbb{R}^d \to \mathbb{R}$:
\begin{enumerate}
    \item \textbf{$L$-Smoothness:} The gradient is Lipschitz continuous with constant $L > 0$. For all $x, y \in \mathbb{R}^d$:
    $$f(y) \le f(x) + \langle \nabla f(x), y - x \rangle + \frac{L}{2} \|y - x\|^2$$
    \item \textbf{$\mu$-Strong Convexity:} The function is strongly convex with constant $\mu > 0$. For all $x, y \in \mathbb{R}^d$:
    $$f(y) \ge f(x) + \langle \nabla f(x), y - x \rangle + \frac{\mu}{2} \|y - x\|^2$$
\end{enumerate}

\section*{Main Convergence Theorem}
Let $f$ be $L$-smooth and $\mu$-strongly convex. Let $x^*$ be the unique global minimizer of $f$. If we run Nesterov's Accelerated Gradient Descent with $\alpha = \frac{1}{L}$ and $\beta = \frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}$, then the sequence of iterates $x_k$ satisfies:
$$f(x_k) - f(x^*) \le \left(1 - \sqrt{\frac{\mu}{L}}\right)^k \left( 2(f(x_0) - f(x^*)) \right)$$

\section*{Theoretical Properties}
\begin{itemize}
    \item \textbf{Optimal Convergence Rate:} The algorithm achieves a convergence rate of $O\left((1 - \sqrt{1/\kappa})^k\right)$, which matches the $\Omega\left((1 - \sqrt{1/\kappa})^k\right)$ theoretical lower bound for first-order methods on strongly convex and smooth functions.
    \item \textbf{Acceleration:} Standard gradient descent requires $O(\kappa \log(1/\epsilon))$ iterations to reach $\epsilon$-accuracy. Nesterov's acceleration reduces this to $O(\sqrt{\kappa} \log(1/\epsilon))$, which is a massive improvement for ill-conditioned problems ($\kappa \gg 1$).
    \item \textbf{Hyperparameter Sensitivity:} The exact momentum value $\beta$ requires knowledge of $\mu$ and $L$. Overestimating $\mu$ (underestimating $\kappa$) leads to under-acceleration, while underestimating $\mu$ can cause heavy oscillations (ripple effect).
\end{itemize}

\section*{Discussion}
Nesterov's method introduces an "extrapolation" step ($y_k$). Instead of evaluating the gradient at the current point $x_k$, it looks ahead in the direction of the previous step's momentum. This proactive gradient evaluation allows the algorithm to correct its trajectory before taking a step, effectively dampening oscillations that plague standard momentum methods and enabling the optimal accelerated rate.

\section*{Preliminaries (Definitions and Lemmas)}
To facilitate the proof, we define $q = \sqrt{\frac{\mu}{L}}$. Consequently, the momentum term can be written as $\beta = \frac{1-q}{1+q}$.

We introduce an auxiliary sequence $v_k$ (often called the estimate sequence) defined implicitly by the iterates.
\\[1em]
\textbf{[Step 1]} Define the auxiliary sequence $v_k$:
$$v_k = x_{k-1} + \frac{1}{q} (x_k - x_{k-1})$$
Note that at initialization, since $x_{-1} = x_0$, we have $v_0 = x_0$.

\section*{Main Proof (Step-by-Step Derivation)}

\textbf{[Step 2]} Establish the relation between $y_k, x_k,$ and $v_k$. 
From [Step 1], we can rewrite $v_k - x_k$:
\begin{align*}
    v_k - x_k &= x_{k-1} - x_k + \frac{1}{q}(x_k - x_{k-1}) \\
    &= \left(\frac{1}{q} - 1\right)(x_k - x_{k-1}) = \frac{1-q}{q}(x_k - x_{k-1})
\end{align*}
This implies $(x_k - x_{k-1}) = \frac{q}{1-q}(v_k - x_k)$. 
Substitute this into the algorithm's definition of $y_k$:
\begin{align*}
    y_k &= x_k + \beta(x_k - x_{k-1}) \\
    &= x_k + \left(\frac{1-q}{1+q}\right) \left(\frac{q}{1-q}\right) (v_k - x_k) \\
    &= x_k + \frac{q}{1+q}(v_k - x_k) \\
    &= \frac{1}{1+q}x_k + \frac{q}{1+q}v_k
\end{align*}
Thus, $y_k$ is a strict convex combination of $x_k$ and $v_k$.
\\[1em]
\textbf{[Step 3]} Express $v_{k+1}$ in terms of $v_k, y_k$ and $\nabla f(y_k)$.
Using the definition of $v_{k+1}$ and the update rule $x_{k+1} = y_k - \frac{1}{L}\nabla f(y_k)$:
\begin{align*}
    v_{k+1} &= x_k + \frac{1}{q}(x_{k+1} - x_k) \\
    &= x_k + \frac{1}{q}\left(y_k - \frac{1}{L}\nabla f(y_k) - x_k\right) \\
    &= x_k + \frac{1}{q}(y_k - x_k) - \frac{1}{qL}\nabla f(y_k)
\end{align*}
From [Step 2], we know $y_k - x_k = \frac{q}{1+q}(v_k - x_k)$. Substituting this:
\begin{align*}
    x_k + \frac{1}{q}(y_k - x_k) &= x_k + \frac{1}{1+q}(v_k - x_k) \\
    &= \frac{q}{1+q}x_k + \frac{1}{1+q}v_k
\end{align*}
We can rewrite this expression by adding and subtracting terms to match $(1-q)v_k + qy_k$:
\begin{align*}
    (1-q)v_k + qy_k &= (1-q)v_k + q\left(\frac{1}{1+q}x_k + \frac{q}{1+q}v_k\right) \\
    &= v_k\left(1 - q + \frac{q^2}{1+q}\right) + \frac{q}{1+q}x_k \\
    &= v_k\left(\frac{1 - q^2 + q^2}{1+q}\right) + \frac{q}{1+q}x_k = \frac{1}{1+q}v_k + \frac{q}{1+q}x_k
\end{align*}
Therefore, we have exactly derived:
$$v_{k+1} = (1-q)v_k + qy_k - \frac{1}{qL}\nabla f(y_k)$$

\textbf{[Step 4]} Define the Lyapunov function $V_k$.
$$V_k = f(x_k) - f(x^*) + \frac{\mu}{2}\|v_k - x^*\|^2$$

\textbf{[Step 5]} Evaluate the squared norm expansion for $V_{k+1}$.
Subtract $x^*$ from the result of [Step 3]:
$$v_{k+1} - x^* = (1-q)(v_k - x*) + q(y_k - x^*) - \frac{1}{qL}\nabla f(y_k)$$
Expanding the squared norm:
\begin{align*}
    \|v_{k+1} - x^*\|^2 &= \|(1-q)(v_k - x^*) + q(y_k - x^*)\|^2 \\
    &\quad - \frac{2}{qL}\langle \nabla f(y_k), (1-q)(v_k - x^*) + q(y_k - x^*) \rangle + \frac{1}{q^2 L^2}\|\nabla f(y_k)\|^2
\end{align*}

\textbf{[Step 6]} Apply convexity of the squared norm.
Since $(1-q) + q = 1$, by Jensen's inequality for the squared norm:
$$\|(1-q)(v_k - x^*) + q(y_k - x^*)\|^2 \le (1-q)\|v_k - x^*\|^2 + q\|y_k - x^*\|^2$$
Multiply the entire expansion by $\frac{\mu}{2}$. Note that $\frac{\mu}{qL} = \frac{\mu}{\sqrt{\mu/L}L} = \sqrt{\frac{\mu}{L}} = q$, and $\frac{\mu}{2q^2 L^2} = \frac{\mu}{2(\mu/L)L^2} = \frac{1}{2L}$.
\begin{align*}
    \frac{\mu}{2}\|v_{k+1} - x^*\|^2 &\le (1-q)\frac{\mu}{2}\|v_k - x^*\|^2 + q\frac{\mu}{2}\|y_k - x^*\|^2 \\
    &\quad - q\langle \nabla f(y_k), (1-q)v_k + qy_k - x^* \rangle + \frac{1}{2L}\|\nabla f(y_k)\|^2
\end{align*}

\textbf{[Step 7]} Apply $L$-smoothness bound for $x_{k+1}$.
From the descent lemma for $L$-smooth functions:
$$f(x_{k+1}) \le f(y_k) + \langle \nabla f(y_k), x_{k+1} - y_k \rangle + \frac{L}{2}\|x_{k+1} - y_k\|^2$$
Substitute $x_{k+1} - y_k = -\frac{1}{L}\nabla f(y_k)$:
$$f(x_{k+1}) \le f(y_k) - \frac{1}{2L}\|\nabla f(y_k)\|^2$$

\textbf{[Step 8]} Combine terms to bound $V_{k+1}$.
Add $f(x_{k+1}) - f(x^*)$ to the inequality from [Step 6], utilizing the bound from [Step 7]. The $\frac{1}{2L}\|\nabla f(y_k)\|^2$ terms cancel out perfectly:
\begin{align*}
    V_{k+1} &\le f(y_k) - f(x^*) + (1-q)\frac{\mu}{2}\|v_k - x^*\|^2 + q\frac{\mu}{2}\|y_k - x^*\|^2 \\
    &\quad - q\langle \nabla f(y_k), (1-q)v_k + qy_k - x^* \rangle
\end{align*}

\textbf{[Step 9]} Bound $f(y_k)$ using convexity and strong convexity.
We want to show $V_{k+1} \le (1-q)V_k$. To do this, we need the non-momentum terms to satisfy:
\begin{equation}
    f(y_k) - f(x^*) + q\frac{\mu}{2}\|y_k - x^*\|^2 - q\langle \nabla f(y_k), (1-q)v_k + qy_k - x^* \rangle \le (1-q)(f(x_k) - f(x^*))
\end{equation}
Rearranging this target inequality:
\begin{equation}
    f(y_k) - (1-q)f(x_k) - qf(x^*) + q\frac{\mu}{2}\|y_k - x^*\|^2 \le q\langle \nabla f(y_k), (1-q)v_k + qy_k - x^* \rangle
\end{equation}
By convexity of $f$ at $x_k$:
$$- (1-q)f(x_k) \le - (1-q)f(y_k) - (1-q)\langle \nabla f(y_k), x_k - y_k \rangle$$
By $\mu$-strong convexity of $f$ at $x^*$:
$$- qf(x^*) \le - qf(y_k) - q\langle \nabla f(y_k), x^* - y_k \rangle - q\frac{\mu}{2}\|x^* - y_k\|^2$$
Summing these two inequalities and adding $f(y_k) + q\frac{\mu}{2}\|y_k - x^*\|^2$ to both sides yields:
\begin{align*}
    f(y_k) - (1-q)f(x_k) - qf(x^*) + q\frac{\mu}{2}\|y_k - x^*\|^2 &\le -\langle \nabla f(y_k), (1-q)x_k + qx^* - y_k \rangle \\
    &= \langle \nabla f(y_k), y_k - (1-q)x_k - qx^* \rangle
\end{align*}
Notice that the strong convexity terms $q\frac{\mu}{2}\|y_k - x^*\|^2$ cancel out beautifully.

\textbf{[Step 10]} Equate the inner product arguments to close the contraction.
We must show that the right hand sides of (2) and the derived bound match:
$$y_k - (1-q)x_k - qx^* = q((1-q)v_k + qy_k - x^*)$$
Equating the $x^*$ terms gives $-q x^* = -q x^*$, which is trivial.
Equating the remaining terms:
$$y_k - (1-q)x_k = q(1-q)v_k + q^2 y_k$$
$$(1-q^2)y_k = (1-q)x_k + q(1-q)v_k$$
Divide by $(1-q)$ (since $q < 1$):
$$(1+q)y_k = x_k + qv_k \implies y_k = \frac{1}{1+q}x_k + \frac{q}{1+q}v_k$$
This is exactly the equality we proved in [Step 2]! Thus, the inequality holds.
Therefore, substituting this back into [Step 8] gives:
$$V_{k+1} \le (1-q)V_k$$

\section*{Rate Derivation (With Constants)}
By recursively applying the contraction inequality from [Step 10]:
$$V_k \le (1-q)^k V_0$$
Since $V_k = f(x_k) - f(x^*) + \frac{\mu}{2}\|v_k - x^*\|^2 \ge f(x_k) - f(x^*)$, we have:
$$f(x_k) - f(x^*) \le (1-q)^k V_0$$
To find $V_0$, recall $v_0 = x_0$. Thus:
$$V_0 = f(x_0) - f(x^*) + \frac{\mu}{2}\|x_0 - x^*\|^2$$
Since $f$ is $\mu$-strongly convex, $f(x_0) - f(x^*) \ge \frac{\mu}{2}\|x_0 - x^*\|^2$. Therefore:
$$V_0 \le 2(f(x_0) - f(x^*))$$
Substituting $q = \sqrt{\frac{\mu}{L}}$, we obtain the final rate:
$$f(x_k) - f(x^*) \le 2 \left(1 - \sqrt{\frac{\mu}{L}}\right)^k (f(x_0) - f(x^*))$$

\section*{Special Cases}
\begin{itemize}
    \item \textbf{$L = \mu$ (Perfectly Conditioned):} If the function has condition number $\kappa = 1$, then $\beta = 0$ and $q = 1$. The algorithm takes a single step $x_1 = x_0 - \frac{1}{L}\nabla f(x_0)$ and reaches the exact minimum $x^*$ instantly, as $1 - q = 0$.
    \item \textbf{$\mu \to 0$ (Convex but not Strongly Convex):} As $\mu \to 0$, $\beta \to 1$. A fixed $\beta$ derived from $\mu=0$ fails to converge optimally. For purely convex functions, Nesterov's method requires a time-varying momentum schedule $\beta_k = \frac{k-1}{k+2}$ to achieve the $O(1/k^2)$ rate.
\end{itemize}
