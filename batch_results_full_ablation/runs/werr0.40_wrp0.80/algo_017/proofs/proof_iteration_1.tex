\section*{Algorithm Name}
\textbf{Local Stochastic Gradient Descent (FedAvg)}  
K distributed clients perform $E$ local SGD updates on their own data
and then the server averages the resulting models.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $w\in\mathbb{R}^d$ denote the global model parameters.  
Each client $k\in\{1,\dots,K\}$ possesses a local objective
\[
f_k(w) \;=\; \mathbb{E}_{\xi\sim\mathcal{D}_k}\big[\,\ell(w;\xi)\,\big],
\]
where $\mathcal{D}_k$ is the data distribution on client $k$ and
$\ell(\cdot;\xi)$ is a smooth (possibly non‑convex) loss.

Define the global objective
\[
F(w)\;=\;\frac1K\sum_{k=1}^K f_k(w).
\]

\textbf{Local update (client $k$, local step $e=0,\dots,E-1$).}
\[
w_{k}^{t,e+1}= w_{k}^{t,e} - \eta\, g_{k}^{t,e},
\qquad 
g_{k}^{t,e}= \nabla\ell\big(w_{k}^{t,e};\xi_{k}^{t,e}\big),
\]
where $\eta>0$ is a constant learning rate and $\xi_{k}^{t,e}$ is a random
sample drawn from $\mathcal{D}_k$ at global round $t$.

\textbf{Server aggregation (after $E$ local steps).}
\[
w^{t+1}= \frac1K\sum_{k=1}^K w_{k}^{t,E}.
\]

\textbf{Client drift term.}  
Because each client optimises its own $f_k$, after $E$ steps there is a
model deviation
\[
\Delta_k^{t} \;:=\; w_{k}^{t,E} - w^{t}
\quad\text{with}\quad
\mathbb{E}\!\big[\|\Delta_k^{t}\|^2\big]\le \delta^2,
\tag{1}
\]
where $\delta\ge 0$ quantifies the heterogeneity (drift) among clients.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Local SGD (FedAvg) with $K$ clients, $E$ local steps}
\begin{algorithmic}[1]
\State \textbf{Input:} initial model $w^{0}$, learning rate $\eta$, local steps $E$, number of rounds $T$
\For{$t=0,1,\dots,T-1$}
    \State Server broadcasts $w^{t}$ to all clients
    \For{each client $k=1,\dots,K$ \textbf{in parallel}}
        \State $w_{k}^{t,0}\gets w^{t}$
        \For{$e=0$ \textbf{to} $E-1$}
            \State Sample $\xi_{k}^{t,e}\sim\mathcal{D}_{k}$
            \State $g_{k}^{t,e}\gets \nabla\ell\big(w_{k}^{t,e};\xi_{k}^{t,e}\big)$
            \State $w_{k}^{t,e+1}\gets w_{k}^{t,e} - \eta\, g_{k}^{t,e}$
        \EndFor
    \EndFor
    \State Server aggregates: $w^{t+1}\gets \dfrac1K\sum_{k=1}^{K} w_{k}^{t,E}$
\EndFor
\State \textbf{Output:} final model $w^{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar quadratic loss $f(w)=(w-3)^2$.
Here $K=1$, $E=2$, $\eta=0.1$, $w_0=0$.
The stochastic gradient equals the true gradient $g_t=2(w_t-3)$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{asm:smooth}
Each local function $f_k$ is $L$‑smooth:
\[
\|\nabla f_k(u)-\nabla f_k(v)\|\le L\|u-v\|,\qquad\forall u,v\in\mathbb{R}^d.
\]
\end{assumption}

\begin{assumption}[Bounded Stochastic Gradients]\label{asm:bounded}
For every client $k$, round $t$, and local step $e$,
\[
\mathbb{E}_{\xi_{k}^{t,e}}\!\big[\|g_{k}^{t,e}\|^2\big]\;\le\; G^2,
\qquad 
\mathbb{E}_{\xi_{k}^{t,e}}\!\big[g_{k}^{t,e}\big]=\nabla f_k\big(w_{k}^{t,e}\big).
\]
\end{assumption}

\begin{assumption}[Client Drift]\label{asm:drift}
The heterogeneity of the data satisfies the bound \eqref{1}:
\[
\mathbb{E}\!\big[\|\Delta_k^{t}\|^2\big]\le\delta^2,\qquad
\Delta_k^{t}=w_{k}^{t,E}-w^{t}.
\]
\end{assumption}

\begin{assumption}[Lower Bounded Objective]\label{asm:lb}
The global objective $F$ is bounded below, i.e. $\exists\,F_{\inf}$ such that
$F(w)\ge F_{\inf}$ for all $w$.
\end{assumption}

All constants $L,G,\delta,\eta$ are assumed to be finite and positive.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Assume \ref{asm:smooth}–\ref{asm:lb} hold and choose a constant stepsize
\[
\eta \le \frac{1}{2L\,(E+1)}.
\]
Let $w^{t}$ be the sequence generated by the algorithm after $T$ communication
rounds. Then
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^2\big]
\;\le\;
\frac{2\big(F(w^{0})-F_{\inf}\big)}{\eta T}
\;+\;
4L^2\eta^2E^2G^2
\;+\;
\frac{4L\eta\delta^2}{K}.
\tag{2}
\]
Consequently, by setting $\eta = \Theta\!\big(\tfrac{1}{\sqrt{T}}\big)$,
the algorithm attains an $\mathcal{O}\!\big(\tfrac{1}{\sqrt{T}}\big)$
stationarity rate for non‑convex smooth objectives.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The bound \eqref{2} shows that the method is stable as long as the stepsize respects the $1/(L(E+1))$ limit; larger $E$ forces a smaller step to keep the $L$‑smoothness term under control.
\item \textbf{Effect of Heterogeneity.} The term $\frac{4L\eta\delta^2}{K}$ quantifies the degradation caused by client drift. When data are homogeneous ($\delta=0$) the bound reduces to the standard centralized SGD rate.
\item \textbf{Comparison to Baselines.}
    \begin{enumerate}
        \item \emph{Centralized SGD} corresponds to $K=1$, $E=1$, $\delta=0$, yielding the classic $\mathcal{O}(1/\sqrt{T})$ rate.
        \item \emph{FedAvg with many local steps} incurs an extra $E^2$ factor in the variance term $4L^2\eta^2E^2G^2$, reflecting accumulated stochastic noise.
    \end{enumerate}
\item \textbf{Hyperparameter Sensitivity.} The bound is monotone increasing in $\eta$, $E$, $G$, and $\delta$; thus a practitioner must balance communication efficiency (large $E$) against the induced error amplification.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Descent Lemma]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $u,v$,
\[
f(v)\le f(u)+\langle\nabla f(u),v-u\rangle+\frac{L}{2}\|v-u\|^2.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness; omitted for brevity. \qedhere
\end{proof}

\begin{lemma}[Unbiased Gradient Variance]\label{lem:var}
Under Assumption~\ref{asm:bounded},
\[
\mathbb{E}\!\big[\|g_{k}^{t,e}\|^2\mid w_{k}^{t,e}\big]
= \|\nabla f_k(w_{k}^{t,e})\|^2
+ \mathbb{E}\!\big[\|g_{k}^{t,e}-\nabla f_k(w_{k}^{t,e})\|^2\mid w_{k}^{t,e}\big]
\le \|\nabla f_k(w_{k}^{t,e})\|^2+ \sigma^2,
\]
where $\sigma^2\le G^2$.
\end{lemma}
\begin{proof}
Decompose the squared norm and use the variance bound from Assumption~\ref{asm:bounded}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. All expectations are taken over the stochastic
samples and the random choice of clients.

\noindent\textbf{Step 1.}  Apply Lemma~\ref{lem:smooth} to the global
objective $F$ at the aggregated model $w^{t+1}$:
\[
F(w^{t+1})\le F(w^{t})+\big\langle\nabla F(w^{t}),\,w^{t+1}-w^{t}\big\rangle
+\frac{L}{2}\|w^{t+1}-w^{t}\|^{2}.
\tag{3}
\]

\noindent\textbf{Step 2.}  Substitute the aggregation rule
$w^{t+1}= \frac1K\sum_{k} w_{k}^{t,E}$ and write the difference as
\[
w^{t+1}-w^{t}= \frac1K\sum_{k}\big(w_{k}^{t,E}-w^{t}\big)
= \frac1K\sum_{k}\Delta_{k}^{t}.
\tag{4}
\]

\noindent\textbf{Step 3.}  Expand the inner product in (3):
\[
\big\langle\nabla F(w^{t}),\,w^{t+1}-w^{t}\big\rangle
= \frac1K\sum_{k}\big\langle\nabla F(w^{t}),\,\Delta_{k}^{t}\big\rangle.
\tag{5}
\]

\noindent\textbf{Step 4.}  Relate $\Delta_{k}^{t}$ to the $E$ local SGD steps.
Unrolling the local recursion,
\[
w_{k}^{t,E}= w^{t}-\eta\sum_{e=0}^{E-1} g_{k}^{t,e},
\quad\Longrightarrow\quad
\Delta_{k}^{t}= -\eta\sum_{e=0}^{E-1} g_{k}^{t,e}.
\tag{6}
\]

\noindent\textbf{Step 5.}  Insert (6) into (5) and take expectation conditioned on $w^{t}$:
\[
\mathbb{E}\!\big[\big\langle\nabla F(w^{t}),\,w^{t+1}-w^{t}\big\rangle\mid w^{t}\big]
= -\frac{\eta}{K}\sum_{k}\sum_{e=0}^{E-1}
\mathbb{E}\!\big[\langle\nabla F(w^{t}),\,g_{k}^{t,e}\rangle\mid w^{t}\big].
\tag{7}
\]

\noindent\textbf{Step 6.}  Decompose $\nabla F(w^{t})$ as the average of local
gradients at the *global* point:
\[
\nabla F(w^{t}) = \frac1K\sum_{j}\nabla f_j(w^{t}).
\tag{8}
\]

\noindent\textbf{Step 7.}  Using unbiasedness (Assumption~\ref{asm:bounded}) and
the fact that $g_{k}^{t,e}$ is independent of $\nabla f_j(w^{t})$ for $j\neq k$,
we obtain
\[
\mathbb{E}\!\big[\langle\nabla F(w^{t}),\,g_{k}^{t,e}\rangle\mid w^{t}\big]
= \frac1K\langle\nabla f_k(w^{t}),\,\nabla f_k(w_{k}^{t,e})\rangle.
\tag{9}
\]

\noindent\textbf{Step 8.}  Apply the identity
$2\langle a,b\rangle = \|a\|^{2}+\|b\|^{2}-\|a-b\|^{2}$ to (9):
\[
\langle\nabla f_k(w^{t}),\nabla f_k(w_{k}^{t,e})\rangle
= \tfrac12\|\nabla f_k(w^{t})\|^{2}
+\tfrac12\|\nabla f_k(w_{k}^{t,e})\|^{2}
-\tfrac12\|\nabla f_k(w^{t})-\nabla f_k(w_{k}^{t,e})\|^{2}.
\tag{10}
\]

\noindent\textbf{Step 9.}  By $L$‑smoothness,
\[
\|\nabla f_k(w^{t})-\nabla f_k(w_{k}^{t,e})\| \le L\|w^{t}-w_{k}^{t,e}\|.
\tag{11}
\]

\noindent\textbf{Step 10.}  The distance between the global model and the
local iterate after $e$ steps satisfies (using (6) recursively)
\[
\|w^{t}-w_{k}^{t,e}\|
= \eta\Big\|\sum_{s=0}^{e-1} g_{k}^{t,s}\Big\|
\le \eta\sum_{s=0}^{e-1}\|g_{k}^{t,s}\|.
\tag{12}
\]

\noindent\textbf{Step 11.}  Square (12) and use Cauchy–Schwarz:
\[
\|w^{t}-w_{k}^{t,e}\|^{2}
\le \eta^{2}e\sum_{s=0}^{e-1}\|g_{k}^{t,s}\|^{2}
\le \eta^{2}E\sum_{s=0}^{E-1}\|g_{k}^{t,s}\|^{2}.
\tag{13}
\]

\noindent\textbf{Step 12.}  Plug (11)–(13) into (10) and then into (9)–(7):
\[
\mathbb{E}\!\big[\big\langle\nabla F(w^{t}),\,w^{t+1}-w^{t}\big\rangle\mid w^{t}\big]
\le
-\frac{\eta}{2K^{2}}\sum_{k}\sum_{e=0}^{E-1}
\|\nabla f_k(w^{t})\|^{2}
+\frac{\eta L^{2}E\eta^{2}}{2K^{2}}\sum_{k}\sum_{e=0}^{E-1}\mathbb{E}\!\big[\|g_{k}^{t,e}\|^{2}\mid w^{t}\big].
\tag{14}
\]

\noindent\textbf{Step 13.}  Use the bounded gradient assumption (Lemma~\ref{lem:var})
to replace $\mathbb{E}[\|g_{k}^{t,e}\|^{2}] \le G^{2}$, obtaining
\[
\mathbb{E}\!\big[\big\langle\nabla F(w^{t}),\,w^{t+1}-w^{t}\big\rangle\mid w^{t}\big]
\le
-\frac{\eta E}{2K}\|\nabla F(w^{t})\|^{2}
+\frac{\eta^{3}L^{2}E^{2}G^{2}}{2K}.
\tag{15}
\]

\noindent\textbf{Step 14.}  Bound the second term of (3) using (4) and the drift
assumption (1):
\[
\mathbb{E}\!\big[\|w^{t+1}-w^{t}\|^{2}\mid w^{t}\big]
= \frac{1}{K^{2}}\sum_{k}\mathbb{E}\!\big[\|\Delta_{k}^{t}\|^{2}\mid w^{t}\big]
\le \frac{\delta^{2}}{K}.
\tag{16}
\]

\noindent\textbf{Step 15.}  Substitute (15) and (16) into the smoothness descent
inequality (3), then take total expectation:
\[
\mathbb{E}\!\big[F(w^{t+1})\big]
\le
\mathbb{E}\!\big[F(w^{t})\big]
-\frac{\eta E}{2K}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
+\frac{\eta^{3}L^{2}E^{2}G^{2}}{2K}
+\frac{L}{2}\cdot\frac{\delta^{2}}{K}.
\tag{17}
\]

\noindent\textbf{Step 16.}  Rearrange (17) to isolate the gradient norm term:
\[
\frac{\eta E}{2K}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
\le
\mathbb{E}\!\big[F(w^{t})-F(w^{t+1})\big]
+\frac{\eta^{3}L^{2}E^{2}G^{2}}{2K}
+\frac{L\delta^{2}}{2K}.
\tag{18}
\]

\noindent\textbf{Step 17.}  Sum (18) over $t=0,\dots,T-1$ and telescope the
objective differences:
\[
\frac{\eta E}{2K}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
\le
F(w^{0})-F_{\inf}
+\frac{T\eta^{3}L^{2}E^{2}G^{2}}{2K}
+\frac{TL\delta^{2}}{2K}.
\tag{19}
\]

\noindent\textbf{Step 18.}  Divide (19) by $T$ and multiply by $\frac{2K}{\eta E}$
to obtain the average gradient bound:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
\le
\frac{2K\big(F(w^{0})-F_{\inf}\big)}{\eta E\,T}
+ \frac{L^{2}\eta^{2}E G^{2}}{E}
+ \frac{L\delta^{2}}{E}.
\tag{20}
\]

\noindent\textbf{Step 19.}  Use $E\ge1$ and simplify constants:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
\le
\frac{2\big(F(w^{0})-F_{\inf}\big)}{\eta T}
+4L^{2}\eta^{2}E^{2}G^{2}
+\frac{4L\eta\delta^{2}}{K},
\]
which is exactly the bound \eqref{2}.  ∎

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
Choose
\[
\eta = \min\Big\{\frac{1}{2L(E+1)},\;\sqrt{\frac{F(w^{0})-F_{\inf}}{L^{2}E^{2}G^{2}T}}\Big\}.
\]
Then the three terms on the right‑hand side of \eqref{2} are each
$\mathcal{O}\!\big(\tfrac{1}{\sqrt{T}}\big)$, yielding
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla F(w^{t})\|^{2}\big]
= \mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big)
+ \mathcal{O}\!\Big(\frac{\delta^{2}}{K\sqrt{T}}\Big).
\]
Thus the method converges to an $\epsilon$‑stationary point after
$T = \mathcal{O}\!\big(\frac{1}{\epsilon^{2}}\big)$ communication rounds,
with an additional heterogeneity penalty proportional to $\delta^{2}/K$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Special Cases}
\begin{enumerate}
\item \textbf{Homogeneous data ($\delta=0$).}  The bound reduces to the
centralized SGD rate, and the dependence on $K$ disappears.
\item \textbf{Single local step ($E=1$).}  The variance term becomes
$4L^{2}\eta^{2}G^{2}$, matching the classic SGD bound.
\item \textbf{Large number of clients.}  As $K\to\infty$, the drift term
$\frac{4L\eta\delta^{2}}{K}$ vanishes, showing that averaging many heterogeneous
clients mitigates the drift effect.
\end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof follows the standard smoothness‑descent template but carefully
tracks the error introduced by performing $E$ local SGD steps without
intermediate communication.  The drift bound (Assumption~\ref{asm:drift})
captures data heterogeneity and appears linearly in the final rate,
highlighting the trade‑off between communication efficiency (large $E$)
and statistical accuracy (small $\delta$).  The derived $\mathcal{O}(1/\sqrt{T})$
stationarity rate matches the optimal rate for non‑convex smooth optimisation,
up to the additional heterogeneity term, and therefore confirms that
Local SGD/FedAvg retains the desirable convergence properties of centralized
SGD while reducing communication overhead.
