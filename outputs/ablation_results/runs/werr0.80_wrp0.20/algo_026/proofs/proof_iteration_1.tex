\section*{Randomized Coordinate Descent with Importance Sampling (RCD-IS)}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a convex differentiable function.  
Denote by $e_i\in\R^{d}$ the $i$‑th canonical basis vector and by
$\nabla_i f(w)$ the $i$‑th partial derivative of $f$ at $w$.
At iteration $k$ a coordinate $i_k\in\{1,\dots ,d\}$ is drawn
independently according to a fixed probability distribution 
$p=(p_1,\dots ,p_d)$ with $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.
Given a positive stepsize $\eta>0$, the update is
\[
\boxed{
w_{k+1}= w_{k}-\frac{\eta}{p_{i_k}}\; e_{i_k}\,\nabla_{i_k} f(w_k)
}\tag{1}
\]
Only the selected coordinate is modified; the factor $1/p_{i_k}$ is
crucial for the subsequent expectation analysis.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RCD-IS}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\R^{d}$, stepsize $\eta>0$, probability vector $p\in\Delta_{d}$.
\FOR{$k=0,1,\dots,T-1$}
    \STATE Sample $i_k\sim\text{Categorical}(p)$.
    \STATE Compute partial gradient $g_{i_k}= \nabla_{i_k}f(w_k)$.
    \STATE Update the selected coordinate
    \[
    w_{k+1}= w_{k}-\frac{\eta}{p_{i_k}}\,e_{i_k}\,g_{i_k}.
    \]
\ENDFOR
\RETURN $w_T$ (or the average $\bar w_T=\frac1T\sum_{k=0}^{T-1}w_k$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one–dimensional problem 
\[
f(w)=(w-3)^2,\qquad w_0=0,\qquad \eta=0.1,
\]
and a trivial distribution $p_1=1$ (the only coordinate is always chosen).

\begin{center}
\begin{tabular}{c|c|c|c}
Iteration $k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $f(w_k)$\\ \hline
0 & $0$ & $-6$ & $9$\\
1 & $w_1=0-\frac{0.1}{1}(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $w_2=0.6-\frac{0.1}{1}(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $w_3=1.08-\frac{0.1}{1}(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3620$
\end{tabular}
\end{center}
The objective value decreases at each step, illustrating the algorithm.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity.} $f$ is convex:
\[
f(\lambda x+(1-\lambda) y)\le \lambda f(x)+(1-\lambda)f(y),\qquad\forall x,y\in\R^{d},\ \lambda\in[0,1].
\]

\item \textbf{Coordinate‑wise $L_i$‑smoothness.} For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\R^{d}$ and any scalar $\alpha$,
\[
\bigl|\nabla_i f(w+ \alpha e_i)-\nabla_i f(w)\bigr|\le L_i|\alpha|.
\tag{2}
\]

Equivalently,
\[
f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^{2}.
\tag{3}
\]

\item \textbf{Probability vector.} $p_i>0$ for all $i$ and $\sum_{i=1}^{d}p_i=1$.

\item \textbf{Stepsize.} The global stepsize $\eta$ satisfies
\[
0<\eta\le \frac{1}{2\displaystyle\sum_{i=1}^{d}\frac{L_i}{p_i}}.
\tag{4}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}
\label{thm:main}
Let $w^{\star}\in\arg\min_{w\in\R^{d}}f(w)$. Under the assumptions above,
for any integer $T\ge 1$ the iterates generated by (1) satisfy
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[f(w_k)-f(w^{\star})\bigr]
\;\le\;
\frac{\|w_0-w^{\star}\|^{2}}{2\eta\,T}.
\tag{5}
\]
Consequently, $ \E\!\bigl[f(w_k)-f(w^{\star})\bigr]=O(1/k) $.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Condition (4) guarantees that the coefficient
$1-\eta\sum_{i}L_i/p_i$ appearing in the proof is non‑negative,
preventing divergence.
\item \textbf{Sensitivity to $p$.} A larger $p_i$ reduces the factor
$1/p_i$ in the variance term, thus the bound improves when the sampling
distribution aligns with the smoothness constants (e.g. $p_i\propto L_i$).
\item \textbf{Comparison.} Compared with uniform coordinate descent
($p_i=1/d$), the importance‑sampled version attains a bound with
$\sum_i L_i/p_i\le d\max_i L_i$, which can be dramatically smaller when
some coordinates are much smoother than others.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the stochastic selection of coordinates,
the algorithm achieves the optimal $O(1/T)$ rate for convex smooth
optimization. The weighting $1/p_i$ in the update (1) is precisely what
cancels the bias introduced by non‑uniform sampling, enabling a clean
expectation analysis.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point inequality from coordinate smoothness]
\label{lem:three}
For any $w\in\R^{d}$, any $i\in\{1,\dots ,d\}$ and any scalar $\alpha$,
\[
f(w+\alpha e_i)\le f(w)+\alpha \nabla_i f(w)+\frac{L_i}{2}\alpha^{2}.
\]
\end{lemma}
\begin{proof}
Directly from (3). \hfill $\square$
\end{proof}

\begin{lemma}[Descent lemma for a single coordinate step]
\label{lem:descent}
Let $w^{+}=w-\frac{\eta}{p_i}e_i\nabla_i f(w)$. Then
\[
f(w^{+})\le f(w)-\frac{\eta}{p_i}\bigl(\nabla_i f(w)\bigr)^{2}
+\frac{L_i\eta^{2}}{2p_i^{2}}\bigl(\nabla_i f(w)\bigr)^{2}.
\tag{6}
\]
\end{lemma}
\begin{proof}
Apply Lemma~\ref{lem:three} with $\alpha=-\frac{\eta}{p_i}\nabla_i f(w)$:
\[
\begin{aligned}
f(w^{+})
&\le f(w)-\frac{\eta}{p_i}\bigl(\nabla_i f(w)\bigr)^{2}
    +\frac{L_i}{2}\Bigl(\frac{\eta}{p_i}\nabla_i f(w)\Bigr)^{2}\\
&= f(w)-\frac{\eta}{p_i}\bigl(\nabla_i f(w)\bigr)^{2}
   +\frac{L_i\eta^{2}}{2p_i^{2}}\bigl(\nabla_i f(w)\bigr)^{2}.
\end{aligned}
\]
\hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof,
expectations are taken over the random choice of $i_k$ conditioned on
the past iterates, and we denote $w^{\star}$ a minimiser of $f$.

\subsection*{Step 1: Distance recursion}
From the update rule (1),
\[
\begin{aligned}
\|w_{k+1}-w^{\star}\|^{2}
&=\Bigl\|w_{k}-w^{\star}
      -\frac{\eta}{p_{i_k}}e_{i_k}\nabla_{i_k}f(w_k)\Bigr\|^{2}\\
&=\|w_{k}-w^{\star}\|^{2}
   -\frac{2\eta}{p_{i_k}}\nabla_{i_k}f(w_k)\bigl(w_{k}-w^{\star}\bigr)_{i_k}
   +\frac{\eta^{2}}{p_{i_k}^{2}}\bigl(\nabla_{i_k}f(w_k)\bigr)^{2}.
\tag{7}
\end{aligned}
\]
\textbf{[Step 1]}

\subsection*{Step 2: Take conditional expectation}
Condition on $w_k$ and use $\Pr(i_k=i)=p_i$:
\[
\begin{aligned}
\E\!\bigl[\|w_{k+1}-w^{\star}\|^{2}\mid w_k\bigr]
&=\|w_{k}-w^{\star}\|^{2}
   -2\eta\sum_{i=1}^{d}\nabla_i f(w_k)\bigl(w_{k}-w^{\star}\bigr)_i
   +\eta^{2}\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\nabla_i f(w_k)\bigr)^{2}.
\tag{8}
\end{aligned}
\]
\textbf{[Step 2]}

\subsection*{Step 3: Use convexity}
Convexity of $f$ yields the standard inequality
\[
f(w_k)-f(w^{\star})\le
\sum_{i=1}^{d}\nabla_i f(w_k)\bigl(w_{k}-w^{\star}\bigr)_i .
\tag{9}
\]
\textbf{[Step 3]}

\subsection*{Step 4: Substitute (9) into (8)}
\[
\E\!\bigl[\|w_{k+1}-w^{\star}\|^{2}\mid w_k\bigr]
\le\|w_{k}-w^{\star}\|^{2}
   -2\eta\bigl(f(w_k)-f(w^{\star})\bigr)
   +\eta^{2}\sum_{i=1}^{d}\frac{1}{p_i}\bigl(\nabla_i f(w_k)\bigr)^{2}.
\tag{10}
\]
\textbf{[Step 4]}

\subsection*{Step 5: Bound the gradient squares}
From the smoothness inequality (3) with $\alpha=-\frac{1}{L_i}\nabla_i f(w_k)$ we obtain
\[
f\!\Bigl(w_k-\frac{1}{L_i}e_i\nabla_i f(w_k)\Bigr)
\le f(w_k)-\frac{1}{2L_i}\bigl(\nabla_i f(w_k)\bigr)^{2}.
\]
Since $f(w^{\star})\le$ the left‑hand side, we have
\[
\bigl(\nabla_i f(w_k)\bigr)^{2}\le 2L_i\bigl(f(w_k)-f(w^{\star})\bigr).
\tag{11}
\]
\textbf{[Step 5]}

\subsection*{Step 6: Plug (11) into (10)}
\[
\begin{aligned}
\E\!\bigl[\|w_{k+1}-w^{\star}\|^{2}\mid w_k\bigr]
&\le\|w_{k}-w^{\star}\|^{2}
   -2\eta\bigl(f(w_k)-f(w^{\star})\bigr)  \\
&\qquad\; +\eta^{2}\sum_{i=1}^{d}\frac{2L_i}{p_i}
          \bigl(f(w_k)-f(w^{\star})\bigr)\\
&=\|w_{k}-w^{\star}\|^{2}
   -2\eta\Bigl(1-\eta\sum_{i=1}^{d}\frac{L_i}{p_i}\Bigr)
      \bigl(f(w_k)-f(w^{\star})\bigr).
\tag{12}
\end{aligned}
\]
\textbf{[Step 6]}

\subsection*{Step 7: Use the stepsize condition}
By assumption (4),
\[
1-\eta\sum_{i=1}^{d}\frac{L_i}{p_i}\;\ge\;\frac12 .
\]
Hence (12) simplifies to
\[
\E\!\bigl[\|w_{k+1}-w^{\star}\|^{2}\mid w_k\bigr]
\le\|w_{k}-w^{\star}\|^{2}
   -\eta\bigl(f(w_k)-f(w^{\star})\bigr).
\tag{13}
\]
\textbf{[Step 7]}

\subsection*{Step 8: Take total expectation and telescope}
Taking expectation over all randomness and summing (13) for $k=0,\dots ,T-1$,
\[
\begin{aligned}
\E\!\bigl[\|w_{T}-w^{\star}\|^{2}\bigr]
&\le \|w_{0}-w^{\star}\|^{2}
   -\eta\sum_{k=0}^{T-1}\E\!\bigl[f(w_k)-f(w^{\star})\bigr].
\end{aligned}
\tag{14}
\]
Since the left‑hand side is non‑negative,
\[
\eta\sum_{k=0}^{T-1}\E\!\bigl[f(w_k)-f(w^{\star})\bigr]
\le\|w_{0}-w^{\star}\|^{2}.
\tag{15}
\]
Dividing by $\eta T$ yields exactly the bound (5). \hfill $\square$

\section*{Rate Derivation (With Constants)}
Define the aggregate smoothness constant
\[
\Lambda\;:=\;\sum_{i=1}^{d}\frac{L_i}{p_i}.
\]
If we choose the stepsize
\[
\eta = \frac{1}{2\Lambda},
\]
then (13) becomes
\[
\E\!\bigl[\|w_{k+1}-w^{\star}\|^{2}\mid w_k\bigr]
\le\|w_{k}-w^{\star}\|^{2}
   -\frac{1}{2\Lambda}\bigl(f(w_k)-f(w^{\star})\bigr),
\]
and the final rate reads
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[f(w_k)-f(w^{\star})\bigr]
\le \frac{2\Lambda\|w_{0}-w^{\star}\|^{2}}{T}.
\]
Thus the convergence constant scales linearly with $\Lambda$, which is
minimised by the choice $p_i\propto L_i$ (importance sampling).

\section*{Special Cases}
\begin{itemize}
\item \textbf{Uniform sampling.} $p_i=1/d$ gives $\Lambda = d\bar L$ with
$\bar L=\frac1d\sum_i L_i$, reproducing the classic $O(d\bar L/T)$ bound.
\item \textbf{Exact importance sampling.} $p_i = L_i/\sum_j L_j$ yields
$\Lambda = \sum_j L_j$, the smallest possible aggregate constant.
\item \textbf{Strongly convex case.} If $f$ is $\mu$‑strongly convex, the same
analysis together with the inequality
$f(w_k)-f(w^{\star})\ge \frac{\mu}{2}\|w_k-w^{\star}\|^{2}$
gives a linear rate
\[
\E\!\bigl[f(w_k)-f(w^{\star})\bigr]\le
\bigl(1-\eta\mu\bigr)^{k}\bigl(f(w_0)-f(w^{\star})\bigr).
\]
\end{itemize}
