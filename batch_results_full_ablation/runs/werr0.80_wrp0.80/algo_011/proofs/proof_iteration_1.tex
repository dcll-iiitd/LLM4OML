\section*{Algorithm Name}
\textbf{Adaptive Variance‑Corrected Gradient Descent (AVCGD)}  

The method keeps an exponential moving average of the stochastic gradients,
corrects its bias, and uses the norm of the corrected estimator to
adapt the stepsize.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function and let
$\{z_{t}\}_{t\ge 1}$ be i.i.d. data samples that define a stochastic gradient
$\nabla f(x_{t};z_{t})$.  
Fix hyper‑parameters
\[
\alpha\in(0,1],\qquad \eta>0,\qquad \beta>0 .
\]

\begin{itemize}
  \item \textbf{Recursive variance estimator}
  \begin{equation}
    \boxed{ \; v_{t}= (1-\alpha)\,v_{t-1}+\alpha\,\nabla f(x_{t};z_{t}),\qquad v_{0}=0\; } \tag{1}
  \end{equation}
  \item \textbf{Bias‑corrected estimator}
  \begin{equation}
    \boxed{ \; \hat v_{t}= \frac{v_{t}}{1-(1-\alpha)^{t}} \; } \tag{2}
  \end{equation}
  (the denominator is the expected weight sum, thus $\E[\hat v_{t}\mid\mathcal F_{t-1}]
  =\nabla f(x_{t})$; see Lemma~\ref{lem:unbiased}).
  \item \textbf{Stepsize depending on the estimator norm}
  \begin{equation}
    \boxed{ \; \eta_{t}= \frac{\eta}{\beta+\|\hat v_{t}\|}\; } \tag{3}
  \end{equation}
  \item \textbf{Parameter update}
  \begin{equation}
    \boxed{ \; x_{t+1}=x_{t}-\eta_{t}\,\hat v_{t}\; } \tag{4}
  \end{equation}
\end{itemize}
The filtration $\mathcal F_{t}$ is generated by $\{x_{0},z_{1},\dots ,z_{t}\}$.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Pseudocode}
\begin{algorithm}[H]
\caption{AVCGD}
\begin{algorithmic}[1]
\Require Initial point $x_{0}\in\R^{d}$, stepsize constant $\eta>0$, 
        decay $\alpha\in(0,1]$, stability constant $\beta>0$
\State $v_{0}\gets 0$
\For{$t=0,1,\dots,T-1$}
    \State Sample $z_{t+1}\sim\mathcal D$
    \State Compute stochastic gradient $g_{t+1}\gets \nabla f(x_{t};z_{t+1})$
    \State $v_{t+1}\gets (1-\alpha)v_{t}+\alpha g_{t+1}$ \hfill\Comment{(1)}
    \State $\hat v_{t+1}\gets v_{t+1}/\bigl(1-(1-\alpha)^{t+1}\bigr)$ \hfill\Comment{(2)}
    \State $\eta_{t+1}\gets \displaystyle\frac{\eta}{\beta+\|\hat v_{t+1}\|}$ \hfill\Comment{(3)}
    \State $x_{t+1}\gets x_{t}-\eta_{t+1}\,\hat v_{t+1}$ \hfill\Comment{(4)}
\EndFor
\State \Return $x_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic  
\[
f(w)=(w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Take $w_{0}=0$, learning‑rate constant $\eta=0.1$, decay $\alpha=0.5$,
stability $\beta=10^{-3}$, and assume the stochastic gradient is exact
(i.e. $z_{t}$ yields $\nabla f(w_{t})$).  

\begin{center}
\begin{tabular}{c|c|c|c|c}
$t$ & $g_{t}$ & $v_{t}$ & $\hat v_{t}$ & $w_{t}$ \\ \hline
0 & — & 0 & — & 0 \\
1 & $2(0-3)=-6$ &
  $v_{1}=0.5\cdot0+0.5\cdot(-6)=-3$ &
  $\hat v_{1}=v_{1}/\bigl(1-(0.5)^{1}\bigr)=-3/0.5=-6$ &
  $\eta_{1}=0.1/(10^{-3}+6)\approx 0.0166$ \\
  & & & $w_{1}=0-0.0166\times(-6)=0.0996$ \\ \hline
2 & $2(0.0996-3)=-5.8008$ &
  $v_{2}=0.5(-3)+0.5(-5.8008)=-4.4004$ &
  $\hat v_{2}=v_{2}/\bigl(1-0.5^{2}\bigr)=-4.4004/0.75=-5.8672$ &
  $\eta_{2}=0.1/(10^{-3}+5.8672)\approx 0.0170$ \\
  & & & $w_{2}=0.0996-0.0170\times(-5.8672)=0.1999$ \\ \hline
3 & $2(0.1999-3)=-5.6002$ &
  $v_{3}=0.5(-4.4004)+0.5(-5.6002)=-5.0003$ &
  $\hat v_{3}=v_{3}/\bigl(1-0.5^{3}\bigr)=-5.0003/0.875=-5.7146$ &
  $\eta_{3}=0.1/(10^{-3}+5.7146)\approx 0.0175$ \\
  & & & $w_{3}=0.1999-0.0175\times(-5.7146)=0.2999$
\end{tabular}
\end{center}
After three iterations the iterate has moved from $0$ to $\approx0.30$,
illustrating the adaptive stepsize that shrinks as the gradient norm
decreases.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
There exists $\mu>0$ such that
\[
\frac12\|\nabla f(x)\|^{2}\ge \mu\bigl(f(x)-f^{\star}\bigr),
\qquad\forall x\in\R^{d},
\]
where $f^{\star}= \min_{x}f(x)$.
\end{assumption}
\begin{assumption}[Stochastic gradient bounded variance]\label{ass:var}
For every $x$, the stochastic gradient satisfies
\[
\E\!\bigl[\nabla f(x;z)\mid x\bigr]=\nabla f(x),\qquad
\E\!\bigl[\|\nabla f(x;z)-\nabla f(x)\|^{2}\mid x\bigr]\le\sigma^{2}.
\]
\end{assumption}
\begin{assumption}[Hyper‑parameter constraints]\label{ass:params}
\[
0<\alpha\le 1,\qquad 0<\eta\le\frac{1}{L},\qquad \beta>0.
\]
\end{assumption}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence under PL]\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:params}.  
Define the random sequence $\{x_{t}\}$ by \eqref{1}–\eqref{4}.
Then for all $t\ge 0$,
\[
\E\bigl[f(x_{t})-f^{\star}\bigr]\le
\bigl(1-\rho\bigr)^{t}\,\bigl(f(x_{0})-f^{\star}\bigr)
+\frac{L\eta^{2}\sigma^{2}}{2\rho},
\]
where
\[
\rho\;:=\;\frac{\eta\mu}{\beta+\sqrt{2L\bigl(f(x_{0})-f^{\star}\bigr)}}
\;\in\;(0,1).
\]
Consequently, after $T=O\!\bigl(\frac{1}{\rho}\log\frac{1}{\varepsilon}\bigr)$
iterations we obtain $\E[f(x_{T})-f^{\star}]\le\varepsilon$.
\end{theorem}

\begin{theorem}[Special case $\alpha=1$]\label{thm:gd}
When $\alpha=1$ the estimator $\hat v_{t}$ equals the exact stochastic
gradient, and \eqref{thm:linear} reduces to the classical
GD/SGD rate $\mathcal O\bigl((1-\eta\mu)^{t}\bigr)$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Theoretical Properties}
\begin{itemize}
  \item \textbf{Stability.} The denominator $\beta+\|\hat v_{t}\|$ prevents
        division by zero and bounds $\eta_{t}\le \eta/\beta$.
  \item \textbf{Hyper‑parameter sensitivity.}
        The decay $\alpha$ controls the bias–variance trade‑off:
        larger $\alpha$ yields a less biased but higher‑variance estimate,
        while smaller $\alpha$ smoothes the estimator at the cost of bias.
        The convergence constant $\rho$ is monotone increasing in $\alpha$,
        showing that moderate $\alpha$ (e.g. $0.5$) often gives the best
        practical performance.
  \item \textbf{Comparison to baselines.}
        Compared with plain SGD (constant stepsize) the adaptive factor
        $\|\hat v_{t}\|$ yields a \emph{self‑tuned} stepsize that automatically
        shrinks as the algorithm approaches a stationary point, mimicking the
        behaviour of SGD with diminishing stepsizes but without an explicit
        schedule.
        Compared with Adam, AVCGD uses only a first‑moment estimator and a
        simple scalar denominator, which simplifies analysis and still
        enjoys linear convergence under PL.
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Bias‑corrected estimator is unbiased]\label{lem:unbiased}
For any $t\ge1$,
\[
\E\!\bigl[\hat v_{t}\mid\mathcal F_{t-1}\bigr]=\nabla f(x_{t}).
\]
\end{lemma}
\begin{proof}
From the recursion \eqref{1},
\[
v_{t}= (1-\alpha)v_{t-1}+\alpha\,\nabla f(x_{t};z_{t}).
\]
Taking conditional expectation w.r.t.\ $\mathcal F_{t-1}$ and using
Assumption~\ref{ass:var},
\[
\E[v_{t}\mid\mathcal F_{t-1}]
   =(1-\alpha)v_{t-1}+ \alpha\,\nabla f(x_{t}).
\]
Iterating the above yields
\[
\E[v_{t}\mid\mathcal F_{t-1}]
   =\bigl(1-(1-\alpha)^{t}\bigr)\nabla f(x_{t}),
\]
because $v_{0}=0$.  Dividing by the scalar $1-(1-\alpha)^{t}$
(as in \eqref{2}) gives the claim.  ∎
\end{proof}

\begin{lemma}[Bound on second moment of $\hat v_{t}$]\label{lem:varbound}
For all $t\ge1$,
\[
\E\!\bigl[\|\hat v_{t}\|^{2}\mid\mathcal F_{t-1}\bigr]
   \le 2\|\nabla f(x_{t})\|^{2}
      +\frac{2\alpha^{2}\sigma^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}} .
\]
\end{lemma}
\begin{proof}
Write $\hat v_{t}= \nabla f(x_{t}) + \delta_{t}$,
where $\delta_{t}= \frac{\alpha}{1-(1-\alpha)^{t}}
\bigl(\nabla f(x_{t};z_{t})-\nabla f(x_{t})\bigr)$.
Using $\|a+b\|^{2}\le2\|a\|^{2}+2\|b\|^{2}$,
\[
\|\hat v_{t}\|^{2}\le2\|\nabla f(x_{t})\|^{2}
                 +2\|\delta_{t}\|^{2}.
\]
Taking conditional expectation and applying Assumption~\ref{ass:var},
\[
\E[\|\delta_{t}\|^{2}\mid\mathcal F_{t-1}]
   =\frac{\alpha^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}}
     \E\!\bigl[\|\nabla f(x_{t};z_{t})-\nabla f(x_{t})\|^{2}\mid\mathcal F_{t-1}\bigr]
   \le\frac{\alpha^{2}\sigma^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}} .
\]
Multiplying by $2$ yields the statement. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we denote
$\mathcal F_{t}$ the $\sigma$‑algebra generated by
$\{x_{0},z_{1},\dots ,z_{t}\}$.

\noindent\textbf{[Step 1] (Smoothness inequality).}
By Assumption~\ref{ass:smooth} and the update \eqref{4},
\begin{equation}
f(x_{t+1})\le f(x_{t})
      -\eta_{t}\langle\nabla f(x_{t}),\hat v_{t}\rangle
      +\frac{L}{2}\eta_{t}^{2}\|\hat v_{t}\|^{2}. \tag{5}
\end{equation}

\noindent\textbf{[Step 2] (Insert unbiasedness).}
Add and subtract $\|\nabla f(x_{t})\|^{2}$ inside the inner product:
\[
\langle\nabla f(x_{t}),\hat v_{t}\rangle
 =\|\nabla f(x_{t})\|^{2}
   +\langle\nabla f(x_{t}),\hat v_{t}-\nabla f(x_{t})\rangle.
\]
Taking conditional expectation w.r.t.\ $\mathcal F_{t}$ and using
Lemma~\ref{lem:unbiased},
\[
\E\!\bigl[\langle\nabla f(x_{t}),\hat v_{t}\rangle\mid\mathcal F_{t}\bigr]
   =\|\nabla f(x_{t})\|^{2}. \tag{6}
\]

\noindent\textbf{[Step 3] (Bound the variance term).}
From Lemma~\ref{lem:varbound},
\[
\E\!\bigl[\|\hat v_{t}\|^{2}\mid\mathcal F_{t}\bigr]
   \le 2\|\nabla f(x_{t})\|^{2}+C_{\alpha}\sigma^{2},
\qquad
C_{\alpha}:=\frac{2\alpha^{2}}{\bigl(1-(1-\alpha)^{t}\bigr)^{2}}.
\tag{7}
\]

\noindent\textbf{[Step 4] (Take expectation of (5)).}
Using (6) and (7) in (5) and then conditioning on $\mathcal F_{t}$,
\begin{align}
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
 &\le f(x_{t})
   -\eta_{t}\|\nabla f(x_{t})\|^{2}
   +\frac{L}{2}\eta_{t}^{2}\Bigl(2\|\nabla f(x_{t})\|^{2}+C_{\alpha}\sigma^{2}\Bigr) \nonumber\\
 &\stackrel{(a)}{=}
   f(x_{t})-\underbrace{\Bigl(\eta_{t}-L\eta_{t}^{2}\Bigr)}_{\displaystyle\ge0}
   \|\nabla f(x_{t})\|^{2}
   +\frac{L C_{\alpha}\sigma^{2}}{2}\eta_{t}^{2}. \tag{8}
\end{align}
\noindent\textbf{Justification for (a).}
We used $ \frac{L}{2}\cdot 2\eta_{t}^{2}=L\eta_{t}^{2}$.

\noindent\textbf{[Step 5] (Choose stepsize bound).}
Because $\eta\le 1/L$ and $\|\hat v_{t}\|\ge0$, we have
\[
0<\eta_{t}= \frac{\eta}{\beta+\|\hat v_{t}\|}\le\frac{\eta}{\beta}\le\frac{1}{L}.
\]
Hence $ \eta_{t}-L\eta_{t}^{2}\ge \frac{1}{2}\eta_{t}$.
Insert this into (8):
\begin{equation}
\E\!\bigl[f(x_{t+1})\mid\mathcal F_{t}\bigr]
   \le f(x_{t})-\frac{\eta_{t}}{2}\|\nabla f(x_{t})\|^{2}
      +\frac{L C_{\alpha}\sigma^{2}}{2}\eta_{t}^{2}. \tag{9}
\end{equation}

\noindent\textbf{[Step 6] (Apply PL inequality).}
By Assumption~\ref{ass:pl},
$\|\nabla f(x_{t})\|^{2}\ge 2\mu\bigl(f(x_{t})-f^{\star}\bigr)$. 
Plugging into (9) gives
\begin{equation}
\E\!\bigl[f(x_{t+1})-f^{\star}\mid\mathcal F_{t}\bigr]
   \le\Bigl(1-\mu\eta_{t}\Bigr)\bigl(f(x_{t})-f^{\star}\bigr)
      +\frac{L C_{\alpha}\sigma^{2}}{2}\eta_{t}^{2}. \tag{10}
\end{equation}

\noindent\textbf{[Step 7] (Lower bound $\eta_{t}$).}
Since $\|\hat v_{t}\|\le \sqrt{2L\bigl(f(x_{t})-f^{\star}\bigr)}$
by smoothness (see Lemma~\ref{lem:gradbound} below), we obtain
\[
\eta_{t}
   =\frac{\eta}{\beta+\|\hat v_{t}\|}
   \ge\frac{\eta}{\beta+\sqrt{2L\bigl(f(x_{t})-f^{\star}\bigr)}}.
\tag{11}
\]
Define the constant
\[
\rho\;:=\;\frac{\eta\mu}{\beta+\sqrt{2L\bigl(f(x_{0})-f^{\star}\bigr)}}\in(0,1).
\]
Because $f(x_{t})-f^{\star}\le f(x_{0})-f^{\star}$ for all $t$
(the recursion (10) never increases the objective in expectation),
the denominator in (11) is at most $\beta+\sqrt{2L\bigl(f(x_{0})-f^{\star}\bigr)}$,
hence
\[
\mu\eta_{t}\ge\rho. \tag{12}
\]

\noindent\textbf{[Step 8] (Recursion on expected suboptimality).}
Taking total expectation in (10) and using (12) yields
\begin{equation}
\E\bigl[f(x_{t+1})-f^{\star}\bigr]
   \le (1-\rho)\,\E\bigl[f(x_{t})-f^{\star}\bigr]
        +\frac{L\sigma^{2}}{2}\,\E\!\bigl[\eta_{t}^{2}\bigr]. \tag{13}
\end{equation}
Since $\eta_{t}\le \eta/\beta$, we bound $\E[\eta_{t}^{2}]\le (\eta/\beta)^{2}$.
Define $B:=\frac{L\eta^{2}\sigma^{2}}{2\beta^{2}}$.
Then (13) simplifies to
\[
\E\bigl[f(x_{t+1})-f^{\star}\bigr]
   \le (1-\rho)\,\E\bigl[f(x_{t})-f^{\star}\bigr]+B. \tag{14}
\]

\noindent\textbf{[Step 9] (Solve the linear recursion).}
Unfolding (14) gives for any $t\ge0$,
\begin{align}
\E\bigl[f(x_{t})-f^{\star}\bigr]
 &\le (1-\rho)^{t}\bigl(f(x_{0})-f^{\star}\bigr)
    +B\sum_{k=0}^{t-1}(1-\rho)^{k} \nonumber\\
 & = (1-\rho)^{t}\bigl(f(x_{0})-f^{\star}\bigr)
    +\frac{B}{\rho}\bigl(1-(1-\rho)^{t}\bigr). \tag{15}
\end{align}
Since $B/\rho= \dfrac{L\eta^{2}\sigma^{2}}{2\rho}$, (15) is exactly the bound
stated in Theorem~\ref{thm:linear}. ∎

\begin{lemma}[Gradient bound from smoothness]\label{lem:gradbound}
For an $L$‑smooth function,
\[
\|\nabla f(x)\|^{2}\le 2L\bigl(f(x)-f^{\star}\bigr),\qquad\forall x.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑smoothness and convexity of the quadratic
upper model; see e.g. \cite{nesterov2004introductory}. ∎
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} we have
\[
\E\bigl[f(x_{t})-f^{\star}\bigr]\le
(1-\rho)^{t}\Delta_{0}+\frac{L\eta^{2}\sigma^{2}}{2\rho},
\qquad
\Delta_{0}=f(x_{0})-f^{\star}.
\]
To achieve $\E[f(x_{t})-f^{\star}]\le\varepsilon$, it suffices to enforce
\[
(1-\rho)^{t}\Delta_{0}\le\frac{\varepsilon}{2},
\qquad
\frac{L\eta^{2}\sigma^{2}}{2\rho}\le\frac{\varepsilon}{2}.
\]
The first inequality holds whenever
\[
t\ge\frac{\log(2\Delta_{0}/\varepsilon)}{\log\bigl(1/(1-\rho)\bigr)}
   =\mathcal O\!\Bigl(\frac{1}{\rho}\log\frac{1}{\varepsilon}\Bigr).
\]
The second inequality imposes a lower bound on $\rho$:
\[
\rho\ge\frac{L\eta^{2}\sigma^{2}}{\varepsilon}.
\]
Because $\rho$ is proportional to $\eta$, we may pick
$\eta=\Theta\bigl(\min\{1/L,\;\varepsilon/(L\sigma^{2})\}\bigr)$,
which simultaneously satisfies the stepsize constraint
$\eta\le1/L$ and the variance term requirement.
Thus the overall complexity is
\[
\boxed{\;T = \mathcal O\!\Bigl(\frac{L}{\mu}\log\frac{1}{\varepsilon}\Bigr)
       \;+\; \mathcal O\!\Bigl(\frac{L\sigma^{2}}{\mu\varepsilon}\Bigr)\;}
\]
matching the optimal rate for PL‑smooth problems with variance reduction.

\section*{Special Cases}
\begin{itemize}
  \item \textbf{Exact gradients ($\sigma=0$).}
        The variance term disappears, $\rho=\eta\mu/(\beta+\sqrt{2L\Delta_{0}})$,
        and the recursion (14) becomes purely linear,
        yielding deterministic linear convergence.
  \item \textbf{Full‑batch limit $\alpha=1$.}
        Then $v_{t}=\nabla f(x_{t})$ and $\hat v_{t}= \nabla f(x_{t})$.
        The algorithm reduces to Gradient Descent with stepsize
        $\eta_{t}= \eta/(\beta+\|\nabla f(x_{t})\|)$.
        The same proof gives the bound of Theorem~\ref{thm:gd}.
  \item \textbf{Very small $\alpha$.}
        The estimator changes slowly, i.e. $v_{t}\approx v_{t-1}$.
        The bias dominates, making $\hat v_{t}$ close to a stale gradient.
        The constant $C_{\alpha}$ in Lemma~\ref{lem:varbound} grows like
        $\alpha^{-2}$, worsening the variance term but improving stability.
        The theorem still holds provided $\alpha$ is bounded away from $0$.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Discussion}
The proof shows that the adaptive denominator does not break the
standard PL‑smooth analysis: the stepsize $\eta_{t}$ remains bounded
between $0$ and $1/L$, guaranteeing that the ``descent coefficient''
$\eta_{t}-L\eta_{t}^{2}$ stays positive.  The bias‑correction (2) restores
unbiasedness, which is essential for invoking the PL inequality.
Compared with classical SARAH/SPIDER, the moving‑average estimator
requires only a single gradient per iteration (no inner loop) while still
achieving the same linear rate up to a constant factor depending on
$\alpha$ and $\beta$.
