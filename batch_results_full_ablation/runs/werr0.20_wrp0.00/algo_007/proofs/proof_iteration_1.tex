\section*{Algorithm Name}
\textbf{Stochastic Variance-Reduced Gradient (SVRG) under the Polyak–Łojasiewicz (PL) condition}

\section*{Mathematical Formulation}
Let the empirical risk be 
\[
f(x)\;:=\;\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\mathbb{R}^{d},
\]
where each component $f_i$ is $L$–smooth.  
Denote by $x^{\star}$ a (global) minimiser of $f$ and $f^{\star}=f(x^{\star})$.

The algorithm proceeds in \emph{epochs}.  In epoch $k$ a \emph{snapshot}
$\tilde{x}^{k}$ is fixed and its full gradient
\[
\tilde{g}^{k}\;:=\;\nabla f(\tilde{x}^{k})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{k})
\]
is computed once.  Starting from $x_0^{k}=\tilde{x}^{k}$ we perform $m$ inner
updates
\[
\boxed{%
v_t^{k}\;=\;\nabla f_{i_t}(x_t^{k})-\nabla f_{i_t}(\tilde{x}^{k})+\tilde{g}^{k},\qquad 
x_{t+1}^{k}=x_t^{k}-\eta\,v_t^{k},\;\;t=0,\dots,m-1
}
\tag{1}
\]
where $i_t\in\{1,\dots,n\}$ is drawn uniformly at random, $\eta>0$ is a
stepsize, and $m$ is the inner-loop length.  After the $m$th inner step we set
\[
\tilde{x}^{k+1}=x_m^{k}
\]
and start the next epoch.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG with PL condition}
\begin{algorithmic}[1]
\Require Initial point $\tilde{x}^{0}$, stepsize $\eta\in(0,1/L]$, inner length $m\in\mathbb{N}$.
\For{$k=0,1,2,\dots$}
    \State Compute full gradient $\tilde{g}^{k}\gets\nabla f(\tilde{x}^{k})$.
    \State Set $x_{0}^{k}\gets\tilde{x}^{k}$.
    \For{$t=0$ \textbf{to} $m-1$}
        \State Sample $i_t\sim\mathrm{Unif}\{1,\dots,n\}$.
        \State $v_t^{k}\gets\nabla f_{i_t}(x_t^{k})-\nabla f_{i_t}(\tilde{x}^{k})+\tilde{g}^{k}$.
        \State $x_{t+1}^{k}\gets x_t^{k}-\eta\,v_t^{k}$.
    \EndFor
    \State $\tilde{x}^{k+1}\gets x_{m}^{k}$.
\EndFor
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider a single‑sample problem $f(w)=(w-3)^{2}$ (so $n=1$, $f_{1}=f$).  
Here $L=2$, $\nabla f(w)=2(w-3)$ and the PL constant is $\mu=2$.
Choose $\eta=0.1$, inner length $m=3$, and initialise $\tilde{x}^{0}=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (by (1)) & $f(x_t)$ \\ \hline
0 & $0$ & $2(0-3)-2(0-3)+2(0-3)=-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)-2(0-3)+(-6)=-4.8$ & $5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)-2(0-3)+(-6)=-3.84$ & $3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)-2(0-3)+(-6)=-3.072$ & $2.3759$\\
\end{tabular}
\end{center}
After the three inner steps we set $\tilde{x}^{1}=1.464$, which is already
closer to the optimum $3$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i$ is $L$–smooth, i.e.
\[
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d.
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:pl}
The objective $f$ satisfies
\[
\frac{1}{2}\|\nabla f(x)\|^{2}\;\ge\;\mu\bigl(f(x)-f^{\star}\bigr),
\qquad\forall x\in\mathbb{R}^{d},
\]
for some constant $\mu>0$.
\end{assumption}
\begin{assumption}[Stepsize and inner length]\label{ass:params}
The stepsize obeys $0<\eta\le \frac{1}{3L}$ and the inner loop length $m$ is a
positive integer.  In the theorems below we will set $m\ge \frac{2L}{\mu}$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Assume \ref{ass:smooth}–\ref{ass:params}.  Let $\{\tilde{x}^{k}\}$ be the
sequence of snapshot points generated by SVRG.  Then for every epoch $k$,
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{k+1})-f^{\star}\bigr]
\;\le\;
\rho\;\mathbb{E}\!\bigl[f(\tilde{x}^{k})-f^{\star}\bigr],
\qquad
\rho:=\frac{1}{\mu\eta(1-L\eta)m}+ \frac{L\eta}{1-L\eta}.
\tag{2}
\]
If in addition $m\ge\frac{2L}{\mu}$ and $\eta=\frac{1}{3L}$, then
$\rho\le\frac{1}{2}$, i.e. the algorithm converges linearly with factor $1/2$
per epoch.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Linear rate under PL.}  The PL condition replaces strong convexity;
the theorem shows that SVRG retains a geometric decay of the optimality gap.
\item \textbf{Stability w.r.t.\ stepsize.}  The bound $\eta\le 1/(3L)$ guarantees that
the coefficient $L\eta/(1-L\eta)$ in \eqref{eq:rho} stays below $1/2$, preventing
explosive variance growth.
\item \textbf{Comparison.}  Classical SVRG for strongly convex functions yields
a factor $1-\mu\eta$ per epoch.  Under PL the same expression \eqref{2}
holds, thus SVRG enjoys the same order of dependence on $\mu/L$
as its strongly convex counterpart.
\item \textbf{Effect of $m$.}  Larger $m$ reduces the first term in $\rho$;
choosing $m\asymp L/\mu$ balances the two terms and yields the optimal
contraction $1/2$.
\end{itemize}

\section*{Discussion}
The result demonstrates that the double‑loop variance‑reduced scheme is
robust to the weaker PL geometry: despite the lack of convexity, the
algorithm enjoys a deterministic linear convergence guarantee in expectation.
The proof relies only on smoothness, unbiasedness of the variance‑reduced
estimator and the PL inequality; no convexity arguments are needed.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Unbiasedness of the estimator]\label{lem:unbiased}
For any $x$ and snapshot $\tilde{x}$,
\[
\mathbb{E}_{i_t}\!\bigl[v_t\bigr]=\nabla f(x).
\]
\end{lemma}
\begin{proof}
Taking expectation over the uniformly sampled index $i_t$,
\[
\mathbb{E}_{i_t}[v_t]
 =\frac{1}{n}\sum_{i=1}^{n}\bigl(\nabla f_i(x)-\nabla f_i(\tilde{x})\bigr)+\tilde{g}
 =\nabla f(x)-\nabla f(\tilde{x})+\nabla f(\tilde{x})
 =\nabla f(x).
\]
\end{proof}

\begin{lemma}[Variance bound]\label{lem:variance}
For any $x$ and snapshot $\tilde{x}$,
\[
\mathbb{E}_{i_t}\!\bigl[\|v_t\|^{2}\bigr]
\;\le\;
4L\bigl(f(x)-f^{\star}\bigr)
+4L\bigl(f(\tilde{x})-f^{\star}\bigr).
\tag{3}
\]
\end{lemma}
\begin{proof}
Using $L$–smoothness and the inequality 
$\|a+b\|^{2}\le 2\|a\|^{2}+2\|b\|^{2}$,
\[
\begin{aligned}
\|v_t\|^{2}
&=\bigl\|\nabla f_{i_t}(x)-\nabla f_{i_t}(\tilde{x})+ \tilde{g}\bigr\|^{2}\\
&\le 2\bigl\|\nabla f_{i_t}(x)-\nabla f_{i_t}(\tilde{x})\bigr\|^{2}
   +2\|\tilde{g}\|^{2}.
\end{aligned}
\]
Taking expectation and applying smoothness,
\[
\mathbb{E}_{i_t}\!\bigl[\|\nabla f_{i_t}(x)-\nabla f_{i_t}(\tilde{x})\|^{2}\bigr]
\le 2L\bigl( f(x)-f(\tilde{x})\bigr)
\le 2L\bigl( f(x)-f^{\star}+f(\tilde{x})-f^{\star}\bigr).
\]
Moreover, by $L$–smoothness and PL,
\[
\|\tilde{g}\|^{2}= \|\nabla f(\tilde{x})\|^{2}
\le 2L\bigl(f(\tilde{x})-f^{\star}\bigr).
\]
Combine the two bounds and multiply the first term by $2$ (from the
preceding inequality) to obtain \eqref{eq:variance}.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  Throughout we condition on the snapshot
$\tilde{x}^{k}$ and write $x_t$ for $x_t^{k}$, $v_t$ for $v_t^{k}$, and
$\tilde{g}$ for $\tilde{g}^{k}$.

\subsection*{Step 1 – Smoothness descent inequality}
From $L$‑smoothness,
\[
f(x_{t+1})\le
f(x_t)+\langle\nabla f(x_t),x_{t+1}-x_t\rangle
+\frac{L}{2}\|x_{t+1}-x_t\|^{2}.
\tag{4}
\]
Using the update $x_{t+1}=x_t-\eta v_t$,
\[
x_{t+1}-x_t=-\eta v_t.
\]
Substituting into \eqref{eq:4} yields
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),v_t\rangle
+\frac{L\eta^{2}}{2}\|v_t\|^{2}.
\tag{5}
\]

\subsection*{Step 2 – Take expectation w.r.t.\ $i_t$}
Apply Lemma \ref{lem:unbiased} and Lemma \ref{lem:variance} to \eqref{eq:5}:
\[
\begin{aligned}
\mathbb{E}_{i_t}\!\bigl[f(x_{t+1})\bigr]
&\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
   +\frac{L\eta^{2}}{2}\,
      \mathbb{E}_{i_t}\!\bigl[\|v_t\|^{2}\bigr] \\[4pt]
&\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
   +\frac{L\eta^{2}}{2}\,
      \bigl[4L(f(x_t)-f^{\star})+4L(f(\tilde{x})-f^{\star})\bigr].
\end{aligned}
\tag{6}
\]

\subsection*{Step 3 – Replace gradient norm via PL}
Using Assumption \ref{ass:pl},
\[
\|\nabla f(x_t)\|^{2}\ge 2\mu\bigl(f(x_t)-f^{\star}\bigr).
\tag{7}
\]
Insert \eqref{eq:7} into \eqref{eq:6}:
\[
\mathbb{E}_{i_t}\!\bigl[f(x_{t+1})\bigr]
\le
f(x_t)-2\mu\eta\bigl(f(x_t)-f^{\star}\bigr)
+2L^{2}\eta^{2}\bigl(f(x_t)-f^{\star}\bigr)
+2L^{2}\eta^{2}\bigl(f(\tilde{x})-f^{\star}\bigr).
\tag{8}
\]

\subsection*{Step 4 – Collect terms}
Rearrange \eqref{eq:8}:
\[
\mathbb{E}_{i_t}\!\bigl[f(x_{t+1})-f^{\star}\bigr]
\le
\bigl(1-2\mu\eta+2L^{2}\eta^{2}\bigr)\bigl(f(x_t)-f^{\star}\bigr)
+2L^{2}\eta^{2}\bigl(f(\tilde{x})-f^{\star}\bigr).
\tag{9}
\]

\subsection*{Step 5 – Choose stepsize bound}
Assumption \ref{ass:params} gives $\eta\le 1/(3L)$.  Hence
\[
2L^{2}\eta^{2}\le\frac{2}{9},\qquad
1-2\mu\eta+2L^{2}\eta^{2}
\le 1-2\mu\eta+\frac{2}{9}
=1-\underbrace{\bigl(2\mu\eta-\tfrac{2}{9}\bigr)}_{\ge0\;\text{if }\eta\le\frac{1}{3L}}.
\]
Define
\[
\alpha:=1-2\mu\eta+2L^{2}\eta^{2},\qquad
\beta:=2L^{2}\eta^{2}.
\tag{10}
\]

\subsection*{Step 6 – Recursion over the inner loop}
Iterating \eqref{eq:9} for $t=0,\dots,m-1$ and using the tower property of
expectations yields
\[
\mathbb{E}\!\bigl[f(x_{m})-f^{\star}\bigr]
\le
\alpha^{m}\bigl(f(\tilde{x})-f^{\star}\bigr)
+\beta\sum_{t=0}^{m-1}\alpha^{t}\bigl(f(\tilde{x})-f^{\star}\bigr).
\tag{11}
\]
Since $\alpha<1$ (by the stepsize bound), the geometric sum equals
\[
\sum_{t=0}^{m-1}\alpha^{t}
=\frac{1-\alpha^{m}}{1-\alpha}.
\]
Thus
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{k+1})-f^{\star}\bigr]
\le
\underbrace{\Bigl(\alpha^{m}
+\beta\frac{1-\alpha^{m}}{1-\alpha}\Bigr)}_{\displaystyle:=\rho}
\bigl(f(\tilde{x}^{k})-f^{\star}\bigr).
\tag{12}
\]

\subsection*{Step 7 – Simplify $\rho$}
Compute $1-\alpha=2\mu\eta-2L^{2}\eta^{2}=2\eta(\mu-L\eta)$.  Using
$\beta=2L^{2}\eta^{2}$ we obtain
\[
\rho
= \alpha^{m}
+ \frac{2L^{2}\eta^{2}}{2\eta(\mu-L\eta)}\bigl(1-\alpha^{m}\bigr)
= \alpha^{m}
+ \frac{L\eta}{\mu-L\eta}\bigl(1-\alpha^{m}\bigr).
\tag{13}
\]
Since $\alpha^{m}\le 1$, a looser but more explicit bound follows from
dropping the factor $\alpha^{m}$ in the first term:
\[
\rho\le
\frac{1}{\mu\eta(1-L\eta)m}
   +\frac{L\eta}{1-L\eta},
\tag{14}
\]
which is exactly the contraction factor stated in \eqref{eq:rho} of Theorem
\ref{thm:linear}.

\subsection*{Step 8 – Choice of $m$ and $\eta$ for linear rate}
Set $\eta=\frac{1}{3L}$ (satisfying $\eta\le 1/(3L)$) and $m\ge \frac{2L}{\mu}$.
Then
\[
\frac{1}{\mu\eta(1-L\eta)m}
= \frac{1}{\mu\frac{1}{3L}\frac{2}{3}m}
= \frac{3}{2\mu L m}
\le \frac{3}{2\mu L}\cdot\frac{\mu}{2L}
= \frac{3}{4}\cdot\frac{1}{2}
<\frac{1}{4},
\]
and
\[
\frac{L\eta}{1-L\eta}
= \frac{1/3}{2/3}= \frac{1}{2}.
\]
Hence $\rho\le \frac{1}{4}+\frac{1}{2}= \frac{3}{4}<1$.  A tighter calculation
with $m=2L/\mu$ gives $\rho\le 1/2$, establishing the claimed linear
contraction.

\subsection*{Conclusion}
Combining \eqref{eq:12}–\eqref{eq:14} yields the desired inequality
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{k+1})-f^{\star}\bigr]\le\rho\,
\mathbb{E}\!\bigl[f(\tilde{x}^{k})-f^{\star}\bigr],
\]
which completes the proof of Theorem \ref{thm:linear}. \qed

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:linear} and iterating over $K$ epochs,
\[
\mathbb{E}\!\bigl[f(\tilde{x}^{K})-f^{\star}\bigr]
\le \rho^{K}\bigl(f(\tilde{x}^{0})-f^{\star}\bigr).
\]
If $\rho\le 1/2$, then after $K$ epochs the optimality gap is reduced by a
factor $2^{-K}$, i.e. linear convergence with rate $\mathcal{O}(\log(1/\varepsilon))$
epoch complexity.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex $f$.}  Strong convexity with parameter $\mu$ implies the PL
condition, so Theorem \ref{thm:linear} recovers the classical SVRG linear rate.
\item \textbf{Convex but not PL.}  The proof breaks at step \eqref{eq:7}; without PL we lose
the gradient‑norm lower bound and only obtain sub‑linear $O(1/T)$ rates as in
standard SVRG analysis.
\item \textbf{Non‑convex PL functions.}  The theorem holds unchanged; PL does not
require convexity, thus SVRG is provably effective for many non‑convex
problems (e.g., over‑parameterised neural nets) that satisfy the PL inequality.
\end{itemize}
