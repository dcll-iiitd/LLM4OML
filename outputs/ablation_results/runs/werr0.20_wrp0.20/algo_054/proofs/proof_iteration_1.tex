\section*{Cyclic Stochastic Gradient Descent (Cyclic SGD)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\to\RR$ be the (possibly non‑convex) objective function and let
\[
g_t:=\nabla f(w_t;\xi_t)
\]
be an unbiased stochastic gradient computed on a random mini‑batch $\xi_t$ at iteration $t$.
The learning rate follows a *cyclic* schedule with period $M\in\mathbb{N}$:
\[
\eta_t \;=\; \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
\phi\!\left(\frac{t\bmod M}{M}\right),\qquad t=0,1,2,\dots
\tag{1}
\]
where $\phi:[0,1]\to[0,1]$ is a $1$‑Lipschitz periodic shape function.
A common choice is the *triangular* schedule
\[
\phi(u)=
\begin{cases}
2u, & 0\le u\le \tfrac12,\\[2mm]
2(1-u), & \tfrac12<u\le 1 .
\end{cases}
\tag{2}
\]

The update rule is
\[
w_{t+1}=w_{t}-\eta_t\, g_t,\qquad t=0,1,\dots,T-1.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Cyclic SGD}
\begin{algorithmic}[1]
\Require Initial point $w_0\in\RR^d$, period $M$, bounds $0<\eta_{\min}\le\eta_{\max}$,
shape function $\phi$ (e.g. triangular), total iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Sample a mini‑batch $\xi_t$ and compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \State Compute cyclic step size
    \[
      \eta_t\gets \eta_{\min}+(\eta_{\max}-\eta_{\min})\,
                \phi\!\bigl((t\bmod M)/M\bigr).
    \]
    \State Update $w_{t+1}\gets w_t-\eta_t\,g_t$.
\EndFor
\State \Return $w_T$ (or optionally the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple quadratic $f(w)=(w-3)^2$, whose gradient is $\nabla f(w)=2(w-3)$.
We take a deterministic “batch’’ (so $g_t=\nabla f(w_t)$), $w_0=0$, and a *constant* cyclic schedule
with $\eta_{\min}=\eta_{\max}=0.1$ (i.e. $\eta_t\equiv0.1$) and $M=3$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
Corresponding objective values:
$f(0)=9,\;f(0.6)=5.76,\;f(1.08)=3.6864,\;f(1.464)=2.0881$,
showing a monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\RR^d$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Unbiased Stochastic Gradient]\label{ass:unbiased}
For every $t$,
\[
\EE_{\xi_t}[\,g_t\mid w_t\,]=\nabla f(w_t).
\]
\end{assumption}
\begin{assumption}[Bounded Variance]\label{ass:variance}
There exists $\sigma^2\ge0$ such that
\[
\EE_{\xi_t}\bigl[\|g_t-\nabla f(w_t)\|^2\mid w_t\bigr]\le\sigma^2,
\qquad\forall t .
\]
\end{assumption}
\begin{assumption}[Cyclic Step‑Size Bounds]\label{ass:step}
The cyclic schedule satisfies
\[
0<\eta_{\min}\le\eta_t\le\eta_{\max}<\frac{1}{L},
\qquad\forall t .
\]
\end{assumption}
\begin{assumption}[Finite Initial Sub‑Optimality]\label{ass:initial}
Define $f^\star:=\inf_{w}f(w)$ and assume $f(w_0)-f^\star<\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Cyclic SGD]\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:initial} and for any $T\ge1$, the iterates generated by
Algorithm~3 satisfy
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}\,
\sum_{t=0}^{T-1}\eta_t\,\EE\bigl[\|\nabla f(w_t)\|^2\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\sum_{t=0}^{T-1}\eta_t}
\;+\;
L\,\sigma^2\,\frac{\sum_{t=0}^{T-1}\eta_t^{\,2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{4}
\]
Consequently, if the period $M$ and the bounds $(\eta_{\min},\eta_{\max})$ are fixed,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\bigl[\|\nabla f(w_t)\|^2\bigr]
\;=\;O\!\left(\frac{1}{\sqrt{T}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} Because $\eta_{\max}<1/L$, the update (3) is a contraction in expectation; the bound (4) guarantees that the weighted average of squared gradients cannot diverge.
\item \textbf{Hyper‑parameter Sensitivity.} The constants in (4) depend linearly on $\eta_{\max}$ and quadratically on $\eta_{\max}$ through $\sum\eta_t^{2}$. Choosing a very large $\eta_{\max}$ violates Assumption \ref{ass:step} and destroys the guarantee.
\item \textbf{Comparison to constant‑step‑size SGD.} With a constant $\eta$, (4) reduces to the classical bound
$\frac{2(f(w_0)-f^\star)}{\eta T}+L\sigma^2\eta$.
The cyclic schedule simply replaces $\eta$ by the *average* step size
$\bar\eta:=\frac{1}{T}\sum_{t=0}^{T-1}\eta_t$ while incurring an extra term proportional to the variance of the schedule $\frac{1}{T}\sum\eta_t^{2}$.
\end{itemize}

\section*{Discussion}
The theorem shows that, despite the non‑monotone learning‑rate pattern,
the algorithm inherits the same $O(1/\sqrt{T})$ convergence rate as standard SGD for non‑convex smooth objectives. The proof only uses the upper bound $\eta_{\max}<1/L$; the exact shape $\phi(\cdot)$ plays no role beyond keeping the step sizes within the prescribed interval.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x\in\RR^d$ and any vector $d$,
\[
f(x-d)\le f(x)-\langle\nabla f(x),d\rangle+\frac{L}{2}\|d\|^2 .
\]
\end{lemma}
\begin{proof}
Immediate from Assumption \ref{ass:smooth} with $y=x-d$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof we condition on the filtration
$\mathcal{F}_t:=\sigma(w_0,\xi_0,\dots,\xi_{t-1})$.

\paragraph{Step 1. Apply the Descent Lemma.}
Using Lemma~\ref{lem:descent} with $x=w_t$ and $d=\eta_t g_t$,
\[
f(w_{t+1})
\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle
      +\frac{L}{2}\eta_t^{2}\|g_t\|^{2}.
\tag{5}
\]
\paragraph{Step 2. Decompose the stochastic gradient.}
Write $g_t=\nabla f(w_t)+\xi_t$ where $\xi_t:=g_t-\nabla f(w_t)$ satisfies $\EE[\xi_t\mid\mathcal{F}_t]=0$ and
$\EE[\|\xi_t\|^2\mid\mathcal{F}_t]\le\sigma^2$ (Assumption \ref{ass:variance}).

Substituting into (5) yields
\[
\begin{aligned}
f(w_{t+1})
&\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
      -\eta_t\langle\nabla f(w_t),\xi_t\rangle
      +\frac{L}{2}\eta_t^{2}\|\nabla f(w_t)+\xi_t\|^{2}\\
&= f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
      -\eta_t\langle\nabla f(w_t),\xi_t\rangle
      +\frac{L}{2}\eta_t^{2}\Bigl(\|\nabla f(w_t)\|^{2}
      +2\langle\nabla f(w_t),\xi_t\rangle+\|\xi_t\|^{2}\Bigr).
\end{aligned}
\tag{6}
\]
\paragraph{Step 3. Take conditional expectation.}
Applying $\EE[\cdot\mid\mathcal{F}_t]$ and using $\EE[\xi_t\mid\mathcal{F}_t]=0$:
\[
\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^{2}
     +\frac{L}{2}\eta_t^{2}\|\nabla f(w_t)\|^{2}
     +\frac{L}{2}\eta_t^{2}\EE\!\bigl[\|\xi_t\|^{2}\mid\mathcal{F}_t\bigr].
\tag{7}
\]
\paragraph{Step 4. Apply variance bound.}
Using Assumption \ref{ass:variance},
\[
\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
\le f(w_t)-\Bigl(\eta_t-\frac{L}{2}\eta_t^{2}\Bigr)\|\nabla f(w_t)\|^{2}
     +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{8}
\]
\paragraph{Step 5. Rearrange to isolate the gradient term.}
\[
\Bigl(\eta_t-\frac{L}{2}\eta_t^{2}\Bigr)\|\nabla f(w_t)\|^{2}
\le f(w_t)-\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
     +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{9}
\]
\paragraph{Step 6. Use the bound on $\eta_t$.}
Because $\eta_t<1/L$ (Assumption \ref{ass:step}), we have
\[
\eta_t-\frac{L}{2}\eta_t^{2}
\;\ge\;\frac{\eta_t}{2}.
\tag{10}
\]
Insert (10) into (9):
\[
\frac{\eta_t}{2}\,\|\nabla f(w_t)\|^{2}
\le f(w_t)-\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
     +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{11}
\]
\paragraph{Step 7. Take total expectation.}
Denote $\EE_t[\cdot]=\EE[\cdot\mid\mathcal{F}_t]$ and apply the law of total expectation to (11):
\[
\frac{\eta_t}{2}\,\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le \EE\!\bigl[f(w_t)\bigr]-\EE\!\bigl[f(w_{t+1})\bigr]
     +\frac{L}{2}\eta_t^{2}\sigma^{2}.
\tag{12}
\]
\paragraph{Step 8. Sum over $t=0,\dots,T-1$.}
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\,\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le f(w_0)-\EE\!\bigl[f(w_T)\bigr]
     +\frac{L}{2}\sigma^{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{13}
\]
Since $f(w_T)\ge f^\star$, we obtain
\[
\sum_{t=0}^{T-1}\eta_t\,\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le 2\bigl(f(w_0)-f^\star\bigr)
     +L\sigma^{2}\sum_{t=0}^{T-1}\eta_t^{2}.
\tag{14}
\]
\paragraph{Step 9. Divide by $\sum_{t=0}^{T-1}\eta_t$.}
\[
\frac{1}{\sum_{t=0}^{T-1}\eta_t}
\sum_{t=0}^{T-1}\eta_t\,\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\sum_{t=0}^{T-1}\eta_t}
     +L\sigma^{2}\frac{\sum_{t=0}^{T-1}\eta_t^{2}}{\sum_{t=0}^{T-1}\eta_t}.
\tag{15}
\]
Equation (15) is exactly the bound (4) stated in Theorem~\ref{thm:main}. ∎

\section*{Rate Derivation (With Constants)}
For a *fixed* cyclic schedule the step sizes are bounded:
\[
\eta_{\min}\le\eta_t\le\eta_{\max},\qquad
\sum_{t=0}^{T-1}\eta_t\;\ge\; \eta_{\min}T,
\quad
\sum_{t=0}^{T-1}\eta_t^{2}\;\le\; \eta_{\max}^{2}T .
\]
Plugging these inequalities into (4) yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^\star\bigr)}{\eta_{\min}\,T}
\;+\;
L\sigma^{2}\frac{\eta_{\max}^{2}}{\eta_{\min}}\,\frac{1}{T}.
\]
Choosing $\eta_{\max}=c/\sqrt{T}$ (a common practical heuristic) gives the classical
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
=O\!\left(\frac{1}{\sqrt{T}}\right).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No variance ($\sigma=0$).} Bound (4) reduces to
\[
\frac{1}{\sum\eta_t}\sum\eta_t\EE[\|\nabla f(w_t)\|^{2}]
\le\frac{2(f(w_0)-f^\star)}{\sum\eta_t},
\]
so the weighted average of squared gradients converges to zero at rate $1/\sum\eta_t$.
\item \textbf{Deterministic cyclic gradient (full‑batch).} The same bound holds with $\sigma=0$, confirming that the cyclic schedule does not harm deterministic gradient descent.
\item \textbf{Large period $M\to\infty$.} The schedule becomes effectively constant; the bound collapses to the standard SGD bound.
\end{itemize}
