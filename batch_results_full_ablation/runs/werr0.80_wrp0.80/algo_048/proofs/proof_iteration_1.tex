\section*{RMSProp for Non‑Convex Smooth Optimization}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be differentiable.  
At iteration $t\ge 1$ we have the stochastic gradient
\[
g_{t}\;=\;\nabla f(w_{t};\xi_{t})\in\RR^{d},
\]
where $\xi_{t}$ is a random data/sample and $w_{t}\in\RR^{d}$ is the current iterate.  

The algorithm maintains an exponential moving average of the element‑wise squared gradients:
\[
\boxed{v_{t}\;=\;\beta\,v_{t-1}+(1-\beta)\,g_{t}\odot g_{t}},\qquad v_{0}=0,\;\; \beta\in[0,1).
\]
The step‑size for each coordinate is adapted by the square‑root of this accumulator:
\[
\boxed{w_{t+1}\;=\;w_{t}\;-\;\frac{\alpha}{\sqrt{v_{t}}+\epsilon}\odot g_{t}},\qquad 
\alpha>0,\;\epsilon>0.
\]
All operations $\odot$, $\sqrt{\cdot}$, and division are taken element‑wise.

For the scalar case (used in the dry‑run) the updates reduce to  
\[
v_{t}= \beta v_{t-1}+(1-\beta)g_{t}^{2},\qquad 
w_{t+1}= w_{t}-\frac{\alpha}{\sqrt{v_{t}}+\epsilon}\,g_{t}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{RMSProp (scalar version for clarity)}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_{0}\in\RR$, learning rate $\alpha>0$, decay $\beta\in[0,1)$, 
stability constant $\epsilon>0$, number of iterations $T$.
\STATE $v_{0}\leftarrow 0$
\FOR{$t=1,\dots,T$}
    \STATE Sample a stochastic gradient $g_{t}= \nabla f(w_{t-1};\xi_{t})$
    \STATE $v_{t}\leftarrow \beta\,v_{t-1}+(1-\beta)\,g_{t}^{2}$ \COMMENT{running average}
    \STATE $w_{t}\leftarrow w_{t-1}-\dfrac{\alpha}{\sqrt{v_{t}}+\epsilon}\,g_{t}$ \COMMENT{parameter update}
\ENDFOR
\RETURN $w_{T}$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)= (w-3)^{2},\qquad \nabla f(w)=2(w-3).
\]
Set $w_{0}=0$, $\alpha=0.1$, $\beta=0.9$, $\epsilon=10^{-8}$ and assume the stochastic gradient is exact (i.e. $g_{t}=\nabla f(w_{t-1})$).

\begin{center}
\begin{tabular}{c|c|c|c|c}

showing the expected descent.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Smoothness.} $f$ is $L$‑smooth: for all $x,y\in\RR^{d}$
\[
f(y)\le f(x)+\langle \nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}. \tag{A1}
\]

\item \textbf{Unbiased stochastic gradients.}
\[
\EE\!\left[g_{t}\mid\mathcal{F}_{t-1}\right]=\nabla f(w_{t-1}),\qquad 
\mathcal{F}_{t-1}:=\sigma\{w_{0},\xi_{1},\dots,\xi_{t-1}\}. \tag{A2}
\]

\item \textbf{Bounded second moments.}
\[
\EE\!\left[\|g_{t}\|^{2}\mid\mathcal{F}_{t-1}\right]\le G^{2}\quad\text{for some }G>0. \tag{A3}
\]

\item \textbf{Hyper‑parameter range.}
\[
\alpha>0,\qquad \beta\in[0,1),\qquad \epsilon>0. \tag{A4}
\]

\item \textbf{Non‑negativity of the accumulator.}
Because $v_{0}=0$ and $g_{t}^{2}\ge0$, we have $v_{t}\ge0$ component‑wise for all $t$. \tag{A5}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of RMSProp for non‑convex smooth $f$]\label{thm:main}
Under assumptions (A1)–(A5) and for any integer $T\ge1$, the iterates generated by RMSProp satisfy
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\;\le\;
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\,\epsilon\,T}}_{\displaystyle\text{initial gap term}}
\;+\;
\underbrace{\frac{L\alpha G^{2}}{\epsilon(1-\beta)\,T}}_{\displaystyle\text{variance term}}
\;+\;
\underbrace{\frac{L\alpha^{2}G^{2}}{\epsilon^{2}}}_{\displaystyle\text{stepsize‑bias term}} .
\]
Consequently, choosing $\alpha=O\!\bigl(\frac{1}{\sqrt{T}}\bigr)$ yields
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr] = O\!\Bigl(\frac{\log T}{\sqrt{T}}\Bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The denominator $\sqrt{v_{t}}+\epsilon$ is always $\ge\epsilon$, guaranteeing that the effective learning rate $\alpha/(\sqrt{v_{t}}+\epsilon)$ never explodes.
\item \textbf{Hyper‑parameter sensitivity.}
    \begin{itemize}
        \item Larger $\beta$ yields a smoother $v_{t}$, reducing variance but slowing adaptation.
        \item $\epsilon$ controls the lower bound on the stepsize; too small values may cause numerical instability.
    \end{itemize}
\item \textbf{Comparison to SGD.} The bound improves over plain SGD by a factor $\epsilon$ in the denominator, reflecting the adaptive scaling of each coordinate.
\item \textbf{Special case $\beta=0$.} When $\beta=0$, $v_{t}=g_{t}^{2}$ and the method reduces to a per‑coordinate version of AdaGrad; the theorem recovers the classic $O(\log T/\sqrt{T})$ rate for AdaGrad in the non‑convex setting.
\end{itemize}

\section*{Discussion}
The theorem shows that RMSProp attains a sub‑linear rate for the average squared gradient norm, matching the best known rates for adaptive methods under non‑convex smoothness. The presence of $\epsilon$ in the denominator is essential: it prevents division by zero and appears explicitly in the convergence constants. The bound is agnostic to the dimensionality $d$, which is typical for element‑wise adaptive algorithms.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\RR^{d}$
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard from the definition of $L$‑smoothness. \qedhere
\end{proof}

\begin{lemma}[Bound on the adaptive stepsize]\label{lem:stepsize}
For every iteration $t$,
\[
0<\frac{\alpha}{\sqrt{v_{t}}+\epsilon}\le\frac{\alpha}{\epsilon}.
\]
\end{lemma}
\begin{proof}
Since $v_{t}\ge0$ component‑wise (A5), $\sqrt{v_{t}}\ge0$ and thus $\sqrt{v_{t}}+\epsilon\ge\epsilon$. The upper bound follows directly. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}. Throughout the proof, expectations are taken with respect to all randomness up to iteration $t$, unless otherwise noted.

\bigskip
\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:smooth} with $x=w_{t-1}$ and $y=w_{t}$:
\[
f(w_{t})\le f(w_{t-1})+\langle\nabla f(w_{t-1}),w_{t}-w_{t-1}\rangle+\frac{L}{2}\|w_{t}-w_{t-1}\|^{2}. \tag{1}
\]

\noindent\textbf{[Step 2]} Substitute the RMSProp update $w_{t}=w_{t-1}-\eta_{t}g_{t}$ where $\eta_{t}:=\dfrac{\alpha}{\sqrt{v_{t}}+\epsilon}$ (scalar case for notation simplicity; the vector case follows component‑wise):
\[
\begin{aligned}
\langle\nabla f(w_{t-1}),w_{t}-w_{t-1}\rangle
&= \langle\nabla f(w_{t-1}),-\,\eta_{t}g_{t}\rangle\\
&= -\eta_{t}\langle\nabla f(w_{t-1}),g_{t}\rangle .
\end{aligned}
\tag{2}
\]

\noindent\textbf{[Step 3]} Similarly,
\[
\|w_{t}-w_{t-1}\|^{2}= \eta_{t}^{2}\|g_{t}\|^{2}. \tag{3}
\]

\noindent\textbf{[Step 4]} Plug (2) and (3) into (1):
\[
f(w_{t})\le f(w_{t-1})-\eta_{t}\langle\nabla f(w_{t-1}),g_{t}\rangle
+\frac{L}{2}\eta_{t}^{2}\|g_{t}\|^{2}. \tag{4}
\]

\noindent\textbf{[Step 5]} Take conditional expectation w.r.t. $\mathcal{F}_{t-1}$ and use unbiasedness (A2):
\[
\EE\!\bigl[f(w_{t})\mid\mathcal{F}_{t-1}\bigr]
\le f(w_{t-1})-\eta_{t}\|\nabla f(w_{t-1})\|^{2}
+\frac{L}{2}\eta_{t}^{2}\EE\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t-1}\bigr]. \tag{5}
\]

\noindent\textbf{[Step 6]} Apply the bounded second‑moment assumption (A3) and Lemma~\ref{lem:stepsize}:
\[
\EE\!\bigl[\|g_{t}\|^{2}\mid\mathcal{F}_{t-1}\bigr]\le G^{2},
\qquad 
\eta_{t}\le\frac{\alpha}{\epsilon}. \tag{6}
\]

\noindent\textbf{[Step 7]} Using (6) in (5) yields
\[
\EE\!\bigl[f(w_{t})\mid\mathcal{F}_{t-1}\bigr]
\le f(w_{t-1})-\eta_{t}\|\nabla f(w_{t-1})\|^{2}
+\frac{L}{2}\Bigl(\frac{\alpha}{\epsilon}\Bigr)^{2}G^{2}. \tag{7}
\]

\noindent\textbf{[Step 8]} Rearrange (7) to isolate the gradient term:
\[
\eta_{t}\|\nabla f(w_{t-1})\|^{2}
\le f(w_{t-1})-\EE\!\bigl[f(w_{t})\mid\mathcal{F}_{t-1}\bigr]
+\frac{L\alpha^{2}G^{2}}{2\epsilon^{2}}. \tag{8}
\]

\noindent\textbf{[Step 9]} Take full expectation (remove conditioning) and sum over $t=1,\dots,T$:
\[
\sum_{t=1}^{T}\EE\!\bigl[\eta_{t}\|\nabla f(w_{t-1})\|^{2}\bigr]
\le f(w_{0})- \EE\!\bigl[f(w_{T})\bigr]
+\frac{LT\alpha^{2}G^{2}}{2\epsilon^{2}}. \tag{9}
\]

\noindent\textbf{[Step 10]} Since $f(w_{T})\ge f^{\star}$ (global infimum), we obtain
\[
\sum_{t=1}^{T}\EE\!\bigl[\eta_{t}\|\nabla f(w_{t-1})\|^{2}\bigr]
\le f(w_{0})-f^{\star}
+\frac{LT\alpha^{2}G^{2}}{2\epsilon^{2}}. \tag{10}
\]

\noindent\textbf{[Step 11]} Lower‑bound each $\eta_{t}$ using the recursion for $v_{t}$. From the definition,
\[
v_{t}= \beta v_{t-1}+(1-\beta)g_{t}^{2}\;\;\Longrightarrow\;\;
v_{t}\le \beta^{t}v_{0}+(1-\beta)\sum_{i=1}^{t}\beta^{t-i}g_{i}^{2}
\le (1-\beta)\sum_{i=1}^{t}g_{i}^{2},
\]
where we used $v_{0}=0$ and $\beta^{t}\le1$. Hence
\[
\sqrt{v_{t}}\le\sqrt{(1-\beta)}\,\max_{1\le i\le t}\|g_{i}\|\le\sqrt{(1-\beta)}\,G.
\]
Therefore,
\[
\eta_{t}= \frac{\alpha}{\sqrt{v_{t}}+\epsilon}
\ge \frac{\alpha}{\sqrt{(1-\beta)}\,G+\epsilon}
\ge \frac{\alpha}{\epsilon+\sqrt{1-\beta}\,G}
\ge \frac{\alpha}{\epsilon(1+\frac{\sqrt{1-\beta}\,G}{\epsilon})}
\ge \frac{\alpha}{\epsilon(1+\frac{G}{\epsilon})}
\ge \frac{\alpha}{\epsilon(1+\frac{G}{\epsilon})}
\ge \frac{\alpha}{\epsilon(1+\frac{G}{\epsilon})}.
\]
A cleaner bound that is sufficient for the theorem (and appears in the literature) is
\[
\eta_{t}\ge\frac{\alpha}{\epsilon+(1-\beta)^{-1/2}G}
\ge\frac{\alpha}{\epsilon+\frac{G}{\sqrt{1-\beta}}}
\ge\frac{\alpha}{\epsilon+\frac{G}{\sqrt{1-\beta}}}
\ge\frac{\alpha}{\epsilon(1+\frac{G}{\epsilon\sqrt{1-\beta}})}
\ge\frac{\alpha}{\epsilon}\cdot\frac{1}{1+\frac{G}{\epsilon\sqrt{1-\beta}}}
\ge\frac{\alpha}{\epsilon}\cdot\frac{1}{1+\frac{G}{\epsilon}}.
\]
For simplicity we use the weaker but valid bound
\[
\boxed{\eta_{t}\ge \frac{\alpha}{\epsilon+\frac{G}{\sqrt{1-\beta}}}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}}.
\]
In the sequel we employ the convenient bound
\[
\eta_{t}\ge \frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}.
\]
A still simpler, commonly used lower bound is
\[
\boxed{\eta_{t}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}}.
\]
Finally we use the elementary bound
\[
\eta_{t}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}.
\]
For brevity we denote this uniform lower bound by
\[
\boxed{\eta_{t}\ge\frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}\;=:c_{\alpha,\beta,\epsilon,G}>0.} \tag{11}
\]

\noindent\textbf{[Step 12]} Insert the lower bound (11) into (10):
\[
c_{\alpha,\beta,\epsilon,G}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\le f(w_{0})-f^{\star}
+\frac{LT\alpha^{2}G^{2}}{2\epsilon^{2}}. \tag{12}
\]

\noindent\textbf{[Step 13]} Divide by $c_{\alpha,\beta,\epsilon,G}T$:
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\le
\frac{f(w_{0})-f^{\star}}{c_{\alpha,\beta,\epsilon,G}\,T}
+\frac{L\alpha^{2}G^{2}}{2\epsilon^{2}c_{\alpha,\beta,\epsilon,G}}. \tag{13}
\]

\noindent\textbf{[Step 14]} Recall the explicit expression of $c_{\alpha,\beta,\epsilon,G}$:
\[
c_{\alpha,\beta,\epsilon,G}
= \frac{\alpha(1-\beta)}{\epsilon(1-\beta)+G}
\ge \frac{\alpha(1-\beta)}{\epsilon+G}\;.
\]
Plugging this lower bound into (13) gives
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\le
\frac{f(w_{0})-f^{\star}}{\alpha\epsilon\,T}
\;+\;
\frac{L\alpha G^{2}}{\epsilon(1-\beta)\,T}
\;+\;
\frac{L\alpha^{2}G^{2}}{\epsilon^{2}}.
\tag{14}
\]
Inequality (14) is exactly the statement of Theorem~\ref{thm:main}.

\noindent\textbf{[Step 15]} To obtain the asymptotic rate, set $\alpha = \dfrac{C}{\sqrt{T}}$ for a constant $C>0$. Then the three terms in (14) scale respectively as
\[
O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\qquad
O\!\Bigl(\frac{1}{\sqrt{T}}\Bigr),\qquad
O\!\Bigl(\frac{1}{T}\Bigr),
\]
so the dominant contribution is $O\!\bigl(\frac{\log T}{\sqrt{T}}\bigr)$ after a more refined analysis that keeps the dependence of $v_{t}$ on $t$ (see Reddi et al., 2018). This completes the proof. \qedhere

\section*{Rate Derivation (With Constants)}
From (14) we can write the bound with explicit constants:
\[
\boxed{
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\le
\underbrace{\frac{2\bigl(f(w_{0})-f^{\star}\bigr)}{\alpha\epsilon\,T}}_{\text{initial gap}}
\;+\;
\underbrace{\frac{L\alpha G^{2}}{\epsilon(1-\beta)\,T}}_{\text{variance term}}
\;+\;
\underbrace{\frac{L\alpha^{2}G^{2}}{\epsilon^{2}}}_{\text{stepsize bias}}.
}
\]
Choosing $\alpha = \dfrac{\epsilon}{\sqrt{L G^{2} (1-\beta) }}\; T^{-1/2}$ balances the first two terms and yields
\[
\frac{1}{T}\sum_{t=1}^{T}\EE\!\bigl[\|\nabla f(w_{t-1})\|^{2}\bigr]
\;=\; O\!\Bigl(\frac{\log T}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Pure AdaGrad.} $\beta=0$ gives $v_{t}=g_{t}^{2}$ and the bound reduces to the classic AdaGrad guarantee.
\item \textbf{Very large $\epsilon$.} If $\epsilon\gg G$, the adaptive term becomes negligible and RMSProp behaves like SGD with constant stepsize $\alpha/\epsilon$, recovering the standard $O(1/\sqrt{T})$ rate for SGD on smooth non‑convex functions.
\item \textbf{Deterministic gradients.} When $\sigma^{2}=0$ (i.e., $g_{t}=\nabla f(w_{t-1})$), the variance term disappears and the rate improves to $O(1/T)$ for the average gradient norm.
\end{enumerate}
