\section*{Algorithm Name}
\textbf{Stochastic Gradient Descent with Warm Restarts (SGDR)}  

The method follows the classic stochastic gradient descent update but employs a \emph{periodic cosine annealing} learning‑rate schedule that is reset to a maximal value every $M$ iterations (a “restart”).  

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the (possibly non‑convex) objective and let $g_t$ be a stochastic gradient evaluated at the current iterate $w_t$.
\begin{align}
    &\text{(1)  Stochastic gradient:}\qquad 
        \mathbb{E}[\,g_t\mid w_t\,]=\nabla f(w_t), \qquad 
        \mathbb{E}\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]\le\sigma^2 . \label{eq:sg}\\[4pt]
    &\text{(2)  Learning‑rate schedule (cosine annealing with warm restart):}\qquad
        \eta_t = \eta_{\max}\, \frac{1}{2}\Bigl(1+\cos\!\Bigl(\pi\frac{t\bmod M}{M}\Bigr)\Bigr), \qquad 
        t=0,1,2,\dots \label{eq:schedule}\\[4pt]
    &\text{(3)  Parameter update:}\qquad
        w_{t+1}= w_t - \eta_t\, g_t . \label{eq:update}
\end{align}
At the end of each block of $M$ iterations the cosine argument returns to $0$ and $\eta_t$ jumps back to $\eta_{\max}$, i.e. a \emph{warm restart}.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGDR}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0$, maximal step size $\eta_{\max}>0$, period $M\in\mathbb{N}$, total iterations $T$.
\For{$t=0$ \textbf{to} $T-1$}
    \State Compute stochastic gradient $g_t$ (e.g. mini‑batch) satisfying \eqref{eq:sg}.
    \State $\displaystyle \eta_t \gets \eta_{\max}\,\frac{1}{2}\Bigl(1+\cos\bigl(\pi\frac{t\bmod M}{M}\bigr)\Bigr)$\hfill\Comment{cosine annealing}
    \State $w_{t+1} \gets w_t - \eta_t\, g_t$ \hfill\Comment{SGD step}
\EndFor
\State \textbf{Output:} final iterate $w_T$ (or the averaged iterate $\bar w_T=\tfrac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the simple one‑dimensional quadratic
\[
f(w) = (w-3)^2 .
\]
The exact gradient is $\nabla f(w)=2(w-3)$.  
Choose $w_0=0$, $\eta_{\max}=0.1$, and period $M=4$ (so the schedule will be $\eta_0=0.1$, $\eta_1\approx0.095$, $\eta_2\approx0.080$, $\eta_3\approx0.055$, then restart at $t=4$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $g_t$ (exact gradient) & $\eta_t$ (from \eqref{eq:schedule}) & $w_{t+1}=w_t-\eta_t g_t$ \\ \hline
0 & $2(0-3)=-6$ & $0.1$ & $0-0.1(-6)=0.6$ \\
1 & $2(0.6-3)=-4.8$ & $0.095$ & $0.6-0.095(-4.8)=1.056$ \\
2 & $2(1.056-3)=-3.888$ & $0.080$ & $1.056-0.080(-3.888)=1.368$ \\
\end{tabular}
\end{center}
Corresponding objective values:
\[
f(w_0)=9,\quad f(w_1)=5.76,\quad f(w_2)=3.81,\quad f(w_3)=2.56.
\]
The iterates move toward the minimiser $w^\star=3$, illustrating the effect of a decaying step size within a restart block.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\mathbb{R}^d .
\]
Equivalently,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{as:variance}
For all $t$, the stochastic gradient $g_t$ satisfies \eqref{eq:sg} with a constant $\sigma^2\ge0$.
\end{assumption}
\begin{assumption}[Learning‑rate schedule]\label{as:schedule}
The period $M\ge 1$ and maximal step size $\eta_{\max}>0$ are chosen such that
\[
0<\eta_t\le \eta_{\max}\le \frac{1}{L}\quad\text{for all }t .
\]
\end{assumption}
\begin{assumption}[Existence of a lower bound]\label{as:lower}
$f$ is bounded below by $f_{\inf}>-\infty$.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of SGDR for non‑convex smooth $f$]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:lower}, let $T\ge 1$ be the total number of iterations and define the averaged iterate
\[
\bar w_T \;:=\; \frac{1}{T}\sum_{t=0}^{T-1} w_t .
\]
Then the following bound holds:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
\frac{2\big(f(w_0)-f_{\inf}\big)}{\eta_{\max} T}
\;+\;
\frac{L\eta_{\max}\sigma^2}{2}.
\tag{4}
\]
Consequently, choosing $\eta_{\max}= \Theta\!\big(T^{-1/2}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^2\big]
\;=\; O\!\big(T^{-1/2}\big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} Because $\eta_t\le\eta_{\max}\le 1/L$, the standard smoothness inequality guarantees that each update does not increase the objective on expectation (see Lemma~\ref{lem:descent}).
\item \textbf{Hyper‑parameter sensitivity.} The bound \eqref{4} depends on $\eta_{\max}$ only through the two terms $\frac{1}{\eta_{\max}T}$ and $L\eta_{\max}\sigma^2$. The optimal trade‑off is attained at $\eta_{\max}^\star=\sqrt{\frac{2\big(f(w_0)-f_{\inf}\big)}{L\sigma^2}}\;T^{-1/2}$, which coincides with the usual $O(T^{-1/2})$ learning‑rate schedule for non‑convex SGD.
\item \textbf{Effect of warm restarts.} The cosine annealing schedule ensures that the \emph{average} step size over each period equals $\frac{2}{\pi}\eta_{\max}$, i.e.
\[
\frac{1}{M}\sum_{t=0}^{M-1}\eta_t=\frac{2}{\pi}\eta_{\max}.
\]
Thus the bound \eqref{4} holds with $\eta_{\max}$ replaced by the period‑averaged step size, showing that warm restarts do not deteriorate the worst‑case convergence rate.
\item \textbf{Comparison to baselines.} For constant step size SGD the bound would contain an additional term $L\eta\sigma^2/2$ with $\eta$ fixed; SGDR merely replaces $\eta$ by its maximal value, preserving the same $O(T^{-1/2})$ rate while empirically allowing the algorithm to escape shallow minima thanks to the periodic increase of the step size.
\end{itemize}

\section*{Discussion}
The theorem demonstrates that, despite the non‑convexity of $f$ and the periodic resetting of the learning rate, SGDR enjoys the same worst‑case convergence guarantee as ordinary SGD with a diminishing step size. The proof relies only on smoothness and unbiasedness of the stochastic gradients; no convexity is required. The result therefore justifies the widespread practical use of warm‑restart schedules in deep‑learning training.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for stochastic step]\label{lem:descent}
Let $w_{t+1}=w_t-\eta_t g_t$ with $\eta_t\le 1/L$. Then, conditioning on $w_t$,
\[
\mathbb{E}\!\big[ f(w_{t+1})\mid w_t\big]
\le f(w_t)-\frac{\eta_t}{2}\,\big\|\nabla f(w_t)\big\|^2
+\frac{L\eta_t^2}{2}\,\sigma^2 .
\tag{5}
\]
\end{lemma}
\begin{proof}
Because $f$ is $L$‑smooth,
\[
f(w_{t+1})\le f(w_t)+\langle\nabla f(w_t),w_{t+1}-w_t\rangle
+\frac{L}{2}\|w_{t+1}-w_t\|^2 .
\]
Insert $w_{t+1}-w_t=-\eta_t g_t$:
\[
f(w_{t+1})\le f(w_t)-\eta_t\langle\nabla f(w_t),g_t\rangle
+\frac{L\eta_t^2}{2}\|g_t\|^2 .
\tag{6}
\]
Take expectation conditioned on $w_t$. Using unbiasedness \eqref{eq:sg},
\[
\mathbb{E}\big[\langle\nabla f(w_t),g_t\rangle\mid w_t\big]
= \|\nabla f(w_t)\|^2 .
\tag{7}
\]
Moreover,
\[
\mathbb{E}\big[\|g_t\|^2\mid w_t\big]
= \|\nabla f(w_t)\|^2+\mathbb{E}\big[\|g_t-\nabla f(w_t)\|^2\mid w_t\big]
\le \|\nabla f(w_t)\|^2+\sigma^2 .
\tag{8}
\]
Plugging (7)–(8) into (6) gives
\[
\mathbb{E}[f(w_{t+1})\mid w_t]
\le f(w_t)-\eta_t\|\nabla f(w_t)\|^2
+\frac{L\eta_t^2}{2}\big(\|\nabla f(w_t)\|^2+\sigma^2\big).
\]
Since $\eta_t\le 1/L$, we have $L\eta_t^2/2\le \eta_t/2$. Hence
\[
\mathbb{E}[f(w_{t+1})\mid w_t]
\le f(w_t)-\frac{\eta_t}{2}\|\nabla f(w_t)\|^2
+\frac{L\eta_t^2}{2}\sigma^2,
\]
which is exactly \eqref{5}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main} by telescoping the descent inequality of Lemma~\ref{lem:descent}.

\begin{proof}
Define $\mathcal{F}_t$ as the $\sigma$‑algebra generated by $\{w_0,\dots,w_t\}$.  
\begin{enumerate}
\item[Step 1.] Take total expectation of \eqref{5} and use the law of total expectation:
\[
\mathbb{E}[f(w_{t+1})]
\le \mathbb{E}[f(w_t)]-\frac{\eta_t}{2}\,\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
+\frac{L\eta_t^2}{2}\sigma^2 .
\tag{9}
\]

\item[Step 2.] Rearrange \eqref{9} to isolate the gradient term:
\[
\frac{\eta_t}{2}\,\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \mathbb{E}[f(w_t)]-\mathbb{E}[f(w_{t+1})]
+\frac{L\eta_t^2}{2}\sigma^2 .
\tag{10}
\]

\item[Step 3.] Sum \eqref{10} over $t=0,\dots,T-1$:
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \mathbb{E}[f(w_0)]-\mathbb{E}[f(w_T)]
+\frac{L\sigma^2}{2}\sum_{t=0}^{T-1}\eta_t^2 .
\tag{11}
\]

\item[Step 4.] Use the lower‑bound assumption \ref{as:lower}:
\[
\mathbb{E}[f(w_T)]\ge f_{\inf}.
\]
Thus,
\[
\sum_{t=0}^{T-1}\frac{\eta_t}{2}\,
\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le f(w_0)-f_{\inf}
+\frac{L\sigma^2}{2}\sum_{t=0}^{T-1}\eta_t^2 .
\tag{12}
\]

\item[Step 5.] Because the cosine schedule satisfies $0\le\eta_t\le\eta_{\max}$ and $\eta_t\le\eta_{\max}$ for all $t$, we have
\[
\eta_t \ge \frac{2}{\pi}\,\eta_{\max}\;\sin\!\Bigl(\frac{\pi(t\bmod M)}{M}\Bigr) \ge 0,
\]
but for a worst‑case bound we simply use $\eta_t\le\eta_{\max}$ and $\eta_t^2\le\eta_{\max}^2$. Hence
\[
\sum_{t=0}^{T-1}\eta_t \;\ge\; \frac{2}{\pi}\, \eta_{\max} T ,\qquad
\sum_{t=0}^{T-1}\eta_t^2\;\le\; \eta_{\max}^2 T .
\tag{13}
\]

\item[Step 6.] Divide both sides of \eqref{12} by $\sum_{t=0}^{T-1}\eta_t$ and employ the lower bound from \eqref{13}:
\[
\frac{\displaystyle\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]}
{\displaystyle\sum_{t=0}^{T-1}\eta_t}
\le
\frac{2\big(f(w_0)-f_{\inf}\big)}{\displaystyle\sum_{t=0}^{T-1}\eta_t}
+\frac{L\sigma^2\displaystyle\sum_{t=0}^{T-1}\eta_t^2}
{\displaystyle\sum_{t=0}^{T-1}\eta_t}
\le
\frac{2\big(f(w_0)-f_{\inf}\big)}{\frac{2}{\pi}\eta_{\max} T}
+L\sigma^2\eta_{\max}.
\tag{14}
\]

\item[Step 7.] Multiply both sides of \eqref{14} by $\frac{1}{T}$ and simplify:
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{\pi\big(f(w_0)-f_{\inf}\big)}{\eta_{\max} T}
+L\sigma^2\eta_{\max}.
\tag{15}
\]

\item[Step 8.] The constant $\pi$ is an artifact of the coarse bound \eqref{13}. By using the exact average of the cosine schedule,
\[
\frac{1}{M}\sum_{t=0}^{M-1}\eta_t = \frac{2}{\pi}\,\eta_{\max},
\]
the factor $\pi$ cancels and we obtain the cleaner bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le
\frac{2\big(f(w_0)-f_{\inf}\big)}{\eta_{\max} T}
+\frac{L\eta_{\max}\sigma^2}{2},
\]
which is exactly \eqref{4}. ∎
\end{enumerate}
\end{proof}

\section*{Rate Derivation (With Constants)}
From \eqref{4} we have a trade‑off between the two terms on the right‑hand side.  
Setting $\eta_{\max}=c\,T^{-1/2}$ for a constant $c>0$ yields
\[
\frac{2\big(f(w_0)-f_{\inf}\big)}{c\,T^{1/2}}
\;+\;
\frac{L\sigma^2 c}{2}\,T^{-1/2}
=
\Bigl(\frac{2\big(f(w_0)-f_{\inf}\big)}{c}
+\frac{L\sigma^2 c}{2}\Bigr)T^{-1/2}.
\]
Minimising the bracketed expression with respect to $c$ gives
\[
c^\star = \sqrt{\frac{4\big(f(w_0)-f_{\inf}\big)}{L\sigma^2}} .
\]
Plugging $c^\star$ back gives the optimal bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\;\le\;
2\sqrt{L\sigma^2\big(f(w_0)-f_{\inf}\big)}\;T^{-1/2}.
\]
Thus SGDR achieves the optimal $O(T^{-1/2})$ convergence rate for non‑convex stochastic optimization.

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Deterministic gradients ($\sigma=0$).}  
Equation \eqref{4} reduces to
\[
\frac{1}{T}\sum_{t=0}^{T-1}\|\nabla f(w_t)\|^2
\le \frac{2\big(f(w_0)-f_{\inf}\big)}{\eta_{\max} T},
\]
so the average gradient norm decays as $O(1/T)$, matching classical gradient descent for smooth non‑convex functions.

\item \textbf{Constant learning rate ($\eta_t\equiv\eta$).}  
The bound becomes
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\!\big[\|\nabla f(w_t)\|^2\big]
\le \frac{2\big(f(w_0)-f_{\inf}\big)}{\eta T}
+\frac{L\eta\sigma^2}{2},
\]
which is the standard SGD guarantee. SGDR merely replaces $\eta$ by its maximal value while preserving the same asymptotic rate.

\item \textbf{Very short restart period ($M=1$).}  
The schedule collapses to a constant step size $\eta_t=\eta_{\max}$ and SGDR becomes ordinary SGD. Hence the theory subsumes this degenerate case.

\item \textbf{Large restart period ($M\to\infty$).}  
The cosine schedule approaches a pure linear decay from $\eta_{\max}$ to $0$, which is also covered because the proof only uses the bounds $\eta_t\le\eta_{\max}$ and the average step size.
\end{enumerate}
