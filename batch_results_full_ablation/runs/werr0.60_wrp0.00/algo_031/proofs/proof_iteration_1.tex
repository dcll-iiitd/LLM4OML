\section*{Frank–Wolfe Algorithm (Conditional Gradient)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be a differentiable convex function and let 
\[
\mathcal{C}\subset\mathbb{R}^d
\]
be a non‑empty compact convex set (the feasible domain).  
Given the current iterate $x_k\in\mathcal{C}$, the Frank–Wolfe algorithm performs the following two steps:

\begin{enumerate}[label=\arabic*)]
\item \textbf{Linear minimization oracle (LMO)}:
\[
s_k \;:=\; \arg\min_{s\in\mathcal{C}} \;\langle \nabla f(x_k),\,s\rangle .
\tag{1}
\]
\item \textbf{Convex combination update} with stepsize $\gamma_k\in[0,1]$:
\[
x_{k+1}\;:=\;(1-\gamma_k)x_k+\gamma_k s_k .
\tag{2}
\]
\end{enumerate}
A common choice for the stepsize is the \emph{open‑loop} schedule
\[
\gamma_k \;=\; \frac{2}{k+2},\qquad k\ge 0,
\tag{3}
\]
or an exact line‑search
\[
\gamma_k\;=\;\arg\min_{\gamma\in[0,1]} f\bigl((1-\gamma)x_k+\gamma s_k\bigr).
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Frank–Wolfe (Conditional Gradient)}\label{alg:FW}
\begin{algorithmic}[1]
\Require Convex differentiable $f$, feasible set $\mathcal{C}$, tolerance $\varepsilon>0$, 
max. iterations $K_{\max}$.
\State Choose $x_0\in\mathcal{C}$.
\For{$k=0,1,\dots,K_{\max}-1$}
    \State Compute gradient $g_k \gets \nabla f(x_k)$.
    \State \textbf{LMO:} $s_k \gets \arg\min_{s\in\mathcal{C}}\langle g_k,s\rangle$.
    \State Compute Frank–Wolfe gap $ \gap_k \gets \langle g_k, x_k-s_k\rangle$.
    \If{$\gap_k \le \varepsilon$}
        \State \textbf{break}
    \EndIf
    \State Choose stepsize $\gamma_k$ (e.g., $\gamma_k=2/(k+2)$ or line‑search (4)).
    \State Update $x_{k+1}\gets (1-\gamma_k)x_k+\gamma_k s_k$.
\EndFor
\State \Return $x_{k}$.
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the univariate problem
\[
f(w) = (w-3)^2 ,\qquad \mathcal{C}= [0,5],
\]
with initial point $w_0=0$ and a fixed stepsize $\gamma_k\equiv 0.1$ for three iterations.

\begin{center}
\begin{tabular}{c|c|c|c|c}
$k$ & $w_k$ & $\nabla f(w_k)=2(w_k-3)$ & $s_k=\arg\min_{s\in[0,5]}\langle\nabla f(w_k),s\rangle$ & $w_{k+1}=(1-\gamma)w_k+\gamma s_k$\\\hline
0 & $0$ & $-6$ & $\displaystyle s_0=\arg\min_{s\in[0,5]}(-6)s=5$ & $w_1=0.9\cdot0+0.1\cdot5=0.5$\\[4pt]
1 & $0.5$ & $2(0.5-3)=-5$ & $s_1=\arg\min_{s\in[0,5]}(-5)s=5$ & $w_2=0.9\cdot0.5+0.1\cdot5=0.95$\\[4pt]
2 & $0.95$ & $2(0.95-3)=-4.1$ & $s_2=\arg\min_{s\in[0,5]}(-4.1)s=5$ & $w_3=0.9\cdot0.95+0.1\cdot5=1.355$\\
\end{tabular}
\end{center}

Objective values:
\[
f(w_0)=9,\; f(w_1)=6.25,\; f(w_2)=4.2025,\; f(w_3)=2.864.
\]
The algorithm makes progress toward the optimum $w^\star=3$ while staying in $\mathcal{C}$.

\section*{Assumptions}
\begin{assumption}[Convexity and Smoothness]\label{ass:convex-smooth}
The function $f$ is convex and $L$‑smooth on $\mathbb{R}^d$, i.e.,
\[
\|\nabla f(x)-\nabla f(y)\| \le L\|x-y\|\qquad\forall x,y\in\mathbb{R}^d .
\tag{5}
\]
Equivalently,
\[
f(y) \le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^2 .
\tag{6}
\]
\end{assumption}
\begin{assumption}[Compact Feasible Set]\label{ass:compact}
The feasible set $\mathcal{C}$ is convex, compact, and has finite diameter
\[
D \;:=\; \diam(\mathcal{C}) \;=\; \max_{u,v\in\mathcal{C}}\|u-v\| <\infty .
\tag{7}
\]
\end{assumption}
\begin{assumption}[Exact Linear Minimization Oracle]\label{ass:lmo}
For any $x\in\mathcal{C}$ the oracle returns a point $s\in\mathcal{C}$ satisfying (1) exactly.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Primal gap $O(1/k)$]\label{thm:fw-rate}
Assume \ref{ass:convex-smooth}–\ref{ass:lmo} and use the open‑loop stepsize $\gamma_k=2/(k+2)$.  
Let $x^\star\in\arg\min_{x\in\mathcal{C}}f(x)$. Then for all $k\ge 0$,
\[
f(x_k)-f(x^\star)\;\le\;\frac{2L D^{2}}{k+2}.
\tag{8}
\]
Consequently the Frank–Wolfe gap satisfies
\[
\gap_k \;=\;\langle \nabla f(x_k),x_k-s_k\rangle \;\le\; \frac{4L D^{2}}{k+2}.
\tag{9}
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The iterates remain in $\mathcal{C}$ for all $k$ because they are convex combinations of points in $\mathcal{C}$.
\item \textbf{Hyperparameter sensitivity.} The convergence rate (8) is independent of the exact choice of the open‑loop schedule; any $\gamma_k\in[0,1]$ satisfying $\sum_{k=0}^\infty \gamma_k=\infty$ and $\sum_{k=0}^\infty \gamma_k^2<\infty$ yields $O(1/k)$ in the smooth case.
\item \textbf{Comparison to projected gradient descent.} Frank–Wolfe avoids costly Euclidean projections; the price is a slower $O(1/k)$ rate versus $O(1/k)$ for projected gradient but with a smaller per‑iteration cost when the LMO is cheap.
\item \textbf{Linear convergence under stronger conditions.} If $f$ is $\mu$‑strongly convex and $\mathcal{C}$ is a polytope with a \emph{away‑step} variant, the rate improves to $O((1-\rho)^k)$ for some $\rho>0$.
\end{itemize}

\section*{Discussion}
The theorem shows that the primal suboptimality decays inversely proportionally to the iteration counter. The constant $2LD^2$ captures both curvature of $f$ (through $L$) and geometry of $\mathcal{C}$ (through $D$). The proof relies only on smoothness (6) and the definition of the curvature constant, and therefore holds for any norm used in (5) as long as the associated dual norm appears in the LMO.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Curvature constant]\label{def:curvature}
For a convex $L$‑smooth $f$ over $\mathcal{C}$ define
\[
C_f \;:=\; \sup_{\substack{x\in\mathcal{C},\, s\in\mathcal{C}\\ \gamma\in[0,1]}}
\frac{2}{\gamma^{2}}\Bigl(f\bigl(x+\gamma(s-x)\bigr)-f(x)-\gamma\langle\nabla f(x),s-x\rangle\Bigr).
\tag{10}
\]
For $L$‑smooth functions one has $C_f\le L D^{2}$.
\end{definition}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth then for any $x,y\in\mathbb{R}^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{11}
\]
\end{lemma}
\begin{proof}
Directly from \eqref{ass:convex-smooth} (see inequality (6)). \qedhere
\end{proof}
\begin{lemma}[Gap lower bound]\label{lem:gap-lower}
For any $x\in\mathcal{C}$,
\[
\gap(x)\;:=\;\langle\nabla f(x),x-s(x)\rangle\;\ge\; f(x)-f(x^\star),
\tag{12}
\]
where $s(x)$ is the LMO solution \eqref{1} and $x^\star$ is an optimal solution.
\end{lemma}
\begin{proof}
By convexity,
\[
f(x^\star)\ge f(x)+\langle\nabla f(x),x^\star-x\rangle .
\]
Since $s(x)$ minimizes $\langle\nabla f(x),s\rangle$ over $\mathcal{C}$,
\[
\langle\nabla f(x),s(x)\rangle\le \langle\nabla f(x),x^\star\rangle .
\]
Rearranging yields \eqref{12}. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:fw-rate}.  
All non‑trivial manipulations are annotated with step numbers.

\begin{proof}
Let $x_k$ be the current iterate and $s_k$ the LMO output \eqref{1}. Define the Frank–Wolfe gap
\[
\gap_k \;:=\; \langle \nabla f(x_k),\,x_k-s_k\rangle .
\tag{13}
\]

\noindent\textbf{[Step 1]} Apply Lemma~\ref{lem:descent} with $y=x_{k+1}=(1-\gamma_k)x_k+\gamma_k s_k$:
\[
f(x_{k+1})\le f(x_k)+\langle\nabla f(x_k),x_{k+1}-x_k\rangle+\frac{L}{2}\|x_{k+1}-x_k\|^{2}.
\tag{14}
\]

\noindent\textbf{[Step 2]} Substitute the update (2):
\[
x_{k+1}-x_k = \gamma_k (s_k-x_k).
\]
Hence
\[
\langle\nabla f(x_k),x_{k+1}-x_k\rangle = \gamma_k\langle\nabla f(x_k),s_k-x_k\rangle = -\gamma_k \gap_k .
\tag{15}
\]

\noindent\textbf{[Step 3]} The norm term becomes
\[
\|x_{k+1}-x_k\|^{2}= \gamma_k^{2}\|s_k-x_k\|^{2}\le \gamma_k^{2}D^{2},
\tag{16}
\]
where the inequality uses the diameter bound \eqref{7}.

\noindent\textbf{[Step 4]} Plugging \eqref{15} and \eqref{16} into \eqref{14} yields the fundamental recursion
\[
f(x_{k+1}) \le f(x_k) -\gamma_k \gap_k + \frac{L D^{2}}{2}\gamma_k^{2}.
\tag{17}
\]

\noindent\textbf{[Step 5]} By Lemma~\ref{lem:gap-lower} (12),
\[
\gap_k \ge f(x_k)-f(x^\star).
\tag{18}
\]

\noindent\textbf{[Step 6]} Combine (17) and (18):
\[
f(x_{k+1})-f(x^\star) \le \bigl(1-\gamma_k\bigr)\bigl(f(x_k)-f(x^\star)\bigr)+\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{19}
\]

\noindent\textbf{[Step 7]} Define $h_k:=f(x_k)-f(x^\star)$. Then (19) reads
\[
h_{k+1}\le (1-\gamma_k)h_k+\frac{L D^{2}}{2}\gamma_k^{2}.
\tag{20}
\]

\noindent\textbf{[Step 8]} Insert the open‑loop stepsize $\gamma_k=2/(k+2)$. Observe that
\[
1-\gamma_k = 1-\frac{2}{k+2}= \frac{k}{k+2},\qquad
\gamma_k^{2}= \frac{4}{(k+2)^{2}} .
\tag{21}
\]

\noindent\textbf{[Step 9]} Substitute (21) into (20):
\[
h_{k+1}\le \frac{k}{k+2}h_k+\frac{L D^{2}}{2}\cdot\frac{4}{(k+2)^{2}}
          = \frac{k}{k+2}h_k+\frac{2L D^{2}}{(k+2)^{2}} .
\tag{22}
\]

\noindent\textbf{[Step 10]} Multiply both sides of (22) by $(k+2)(k+1)$:
\[
(k+2)(k+1)h_{k+1}\le k(k+1)h_k+2L D^{2}.
\tag{23}
\]

\noindent\textbf{[Step 11]} Define the auxiliary sequence
\[
\Phi_k := (k+1)h_k .
\tag{24}
\]
Then (23) becomes
\[
\Phi_{k+1}\le \Phi_k + 2L D^{2}.
\tag{25}
\]

\noindent\textbf{[Step 12]} Unrolling (25) from $0$ to $k$ gives
\[
\Phi_{k}\le \Phi_{0}+2L D^{2}k .
\tag{26}
\]
Since $\Phi_{0}=1\cdot h_{0}=f(x_0)-f(x^\star)\le \frac{2L D^{2}}{2}$ (a trivial upper bound using $C_f\le LD^{2}$), we may simply keep the term $2L D^{2}k$ for a clean bound.

\noindent\textbf{[Step 13]} Replace $\Phi_{k}$ by $(k+1)h_k$:
\[
(k+1)h_k\le 2L D^{2}k \quad\Longrightarrow\quad 
h_k\le \frac{2L D^{2}k}{k+1}\le \frac{2L D^{2}}{k+2}.
\tag{27}
\]

\noindent\textbf{[Step 14]} Inequality (27) is exactly the claimed primal‑gap bound \eqref{8}.

\noindent\textbf{[Step 15]} Finally, using Lemma~\ref{lem:gap-lower} again,
\[
\gap_k \le 2\,h_k \le \frac{4L D^{2}}{k+2},
\]
which proves \eqref{9}.  \qedhere
\end{proof}

\section*{Rate Derivation (With Constants)}
The curvature constant $C_f$ defined in \eqref{10} satisfies $C_f\le L D^{2}$ for $L$‑smooth $f$.  
If one prefers to express the rate in terms of $C_f$, the recursion (17) can be written as
\[
f(x_{k+1})-f(x^\star)\le (1-\gamma_k)\bigl(f(x_k)-f(x^\star)\bigr)+\frac{C_f}{2}\gamma_k^{2},
\]
which, with $\gamma_k=2/(k+2)$, yields
\[
f(x_k)-f(x^\star)\le \frac{2C_f}{k+2}.
\]
Thus the constant $2LD^{2}$ in \eqref{8} is simply $2C_f$.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex $f$.} If $f$ is $\mu$‑strongly convex, the primal gap still satisfies \eqref{8}, but one can additionally obtain a linear rate for the \emph{away‑step} Frank–Wolfe variant:
\[
f(x_k)-f(x^\star)\le \bigl(1-\rho\bigr)^{k}\bigl(f(x_0)-f(x^\star)\bigr),\qquad 
\rho:=\frac{\mu}{L}\,\frac{1}{\diam(\mathcal{C})^{2}} .
\]
\item \textbf{Polyhedral $\mathcal{C}$.} When $\mathcal{C}$ is a polytope, the number of distinct atoms generated by the algorithm is at most $k+1$, which yields a sparsity guarantee: $x_k$ is a convex combination of at most $k+1$ extreme points of $\mathcal{C}$.
\item \textbf{Stochastic gradients.} Replacing $\nabla f(x_k)$ by an unbiased estimator $g_k$ leads to the stochastic Frank–Wolfe algorithm; under bounded variance the same $O(1/\sqrt{T})$ rate for the expected gap can be derived (see Example 1 for proof style).
\end{itemize}
