\section*{Algorithm Name}
\textbf{Stochastic Gradient Langevin Dynamics with Decaying Noise (SGLD‑D)}  

The method performs a stochastic gradient descent step and injects isotropic Gaussian noise whose variance shrinks with the iteration index.

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be the objective function.  
Given an initial point $x_{0}\in\R^{d}$, a fixed step size $\alpha>0$, and a noise‑variance schedule 
\[
\sigma_{k}^{2}= \frac{\sigma^{2}}{k+1},\qquad k=0,1,2,\dots ,
\]
the iterates are generated by
\begin{equation}
\boxed{
x_{k+1}=x_{k}-\alpha\,\nabla f(x_{k})+\xi_{k}},\qquad 
\xi_{k}\sim\mathcal N\!\bigl(0,\sigma_{k}^{2}I_{d}\bigr). \label{eq:update}
\end{equation}
All random variables $\{\xi_{k}\}$ are independent and independent of $x_{0}$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SGLD‑D}
\begin{algorithmic}[1]
\REQUIRE Initial point $x_{0}\in\R^{d}$, step size $\alpha>0$, base variance $\sigma^{2}>0$, maximum iterations $K$.
\FOR{$k=0$ \TO $K-1$}
    \STATE Compute (exact or mini‑batch) gradient $g_{k}= \nabla f(x_{k})$.
    \STATE Set noise variance $\sigma_{k}^{2}= \sigma^{2}/(k+1)$.
    \STATE Sample $\xi_{k}\sim\mathcal N(0,\sigma_{k}^{2}I_{d})$.
    \STATE Update $x_{k+1}=x_{k}-\alpha\, g_{k}+ \xi_{k}$.
\ENDFOR
\ENSURE Final iterate $x_{K}$ (or the averaged iterate $\bar x_{K}=K^{-1}\sum_{k=0}^{K-1}x_{k}$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic
\[
f(w)=(w-3)^{2},
\]
so that $\nabla f(w)=2(w-3)$.  
Take $w_{0}=0$, step size $\alpha=0.1$, base variance $\sigma^{2}=0.01$ and ignore the random draw (set $\xi_{k}=0$) to illustrate the deterministic part of the update.

\begin{center}
\begin{tabular}{c|c|c|c}
\textbf{Iter.} $k$ & $w_{k}$ & $\nabla f(w_{k})$ & $w_{k+1}=w_{k}-\alpha\nabla f(w_{k})$ \\ \hline
0 & $0$ & $2(0-3)=-6$ & $0-0.1(-6)=0.6$ \\[2mm]
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\[2mm]
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\[2mm]
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, $f(1.464)=2.322$, showing monotone decrease toward the global minimum $f(3)=0$.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\R^{d}:\quad 
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}
\begin{assumption}[Lower Boundedness]\label{as:lb}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x\in\R^{d}$.
\end{assumption}
\begin{assumption}[Noise Schedule]\label{as:noise}
The noise vectors $\{\xi_{k}\}$ are independent, zero‑mean Gaussian with covariance $\sigma_{k}^{2}I_{d}$ where $\sigma_{k}^{2}= \sigma^{2}/(k+1)$ and $\sigma^{2}>0$.
\end{assumption}
\begin{assumption}[Step Size]\label{as:step}
The step size $\alpha>0$ is constant and satisfies 
\[
0<\alpha\le \frac{1}{L}.
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Gradient‑norm convergence]\label{thm:main}
Under Assumptions \ref{as:smooth}–\ref{as:step}, let $\{x_{k}\}$ be generated by \eqref{eq:update}. Define the averaged iterate
\[
\bar x_{T}:=\frac{1}{T}\sum_{k=0}^{T-1}x_{k}.
\]
Then for any $T\ge 1$,
\begin{equation}
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha T}
\;+\;
\frac{L\sigma^{2}\,\log (T+1)}{T}. \label{eq:rate}
\end{equation}
Consequently, $\displaystyle \min_{0\le k<T}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]=O\!\bigl(\tfrac{1}{\alpha T}+\tfrac{L\sigma^{2}\log T}{T}\bigr)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The condition $\alpha\le 1/L$ guarantees that the deterministic gradient step is a contraction in the sense of the smoothness inequality; the added noise has bounded second moment because $\sigma_{k}^{2}=O(1/k)$, ensuring that the total variance of the iterates remains finite.
\item \textbf{Hyper‑parameter Sensitivity.} A larger $\alpha$ accelerates descent (first term in \eqref{eq:rate}) but also amplifies the noise contribution $L\sigma^{2}\log T/T$; the optimal $\alpha$ balances the two terms and is of order $\sqrt{\frac{2(f(x_{0})-f_{\inf})}{L\sigma^{2}\log T}}$.
\item \textbf{Comparison to SGD.} If $\sigma^{2}=0$, \eqref{eq:rate} reduces to the classic $O(1/(\alpha T))$ bound for deterministic gradient descent on smooth non‑convex functions. The extra $L\sigma^{2}\log T/T$ term precisely quantifies the penalty incurred by the injected Langevin noise.
\item \textbf{Special Cases.}
\begin{enumerate}
\item \emph{Constant variance $\sigma_{k}^{2}\equiv\sigma^{2}$.} The bound becomes $O\!\bigl(1/(\alpha T)+L\sigma^{2}\bigr)$, i.e. a non‑vanishing bias that prevents convergence to stationary points.
\item \emph{Decreasing step size $\alpha_{k}=c/(k+1)$.} A similar derivation yields an $O\!\bigl(\log T/T\bigr)$ rate without the explicit dependence on $\alpha$.
\end{enumerate}
\end{itemize}

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{proof}
Standard consequence of $L$‑Lipschitz gradient; see e.g. \cite{Nesterov2004}.
\end{proof}

\begin{lemma}[Noise second moment]\label{lem:noise}
For the Gaussian noise in Assumption \ref{as:noise},
\[
\E\!\bigl[\|\xi_{k}\|^{2}\bigr]=d\,\sigma_{k}^{2}=d\,\frac{\sigma^{2}}{k+1}.
\]
\end{lemma}
\begin{proof}
Zero‑mean Gaussian with covariance $\sigma_{k}^{2}I_{d}$ has trace $d\sigma_{k}^{2}$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. Throughout, expectations are taken w.r.t.\ the randomness up to iteration $k$ (including $\xi_{0},\dots,\xi_{k-1}$) and are denoted $\E_{k}[\cdot]$.

\bigskip
\noindent\textbf{Step 1.} Apply Lemma \ref{lem:smooth} with $x=x_{k}$ and $y=x_{k+1}$:
\begin{align}
f(x_{k+1}) &\le f(x_{k})+\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}. \label{eq:smooth}
\end{align}
\textbf{[Step 1]}

\noindent\textbf{Step 2.} Substitute the update rule \eqref{eq:update}:
\[
x_{k+1}-x_{k}= -\alpha\nabla f(x_{k})+\xi_{k}.
\]
Hence,
\begin{align}
\langle\nabla f(x_{k}),x_{k+1}-x_{k}\rangle
&= -\alpha\|\nabla f(x_{k})\|^{2}+\langle\nabla f(x_{k}),\xi_{k}\rangle, \\
\|x_{k+1}-x_{k}\|^{2}
&= \alpha^{2}\|\nabla f(x_{k})\|^{2}
   -2\alpha\langle\nabla f(x_{k}),\xi_{k}\rangle
   +\|\xi_{k}\|^{2}.
\end{align}
\textbf{[Step 2]}

\noindent\textbf{Step 3.} Insert the expressions from Step 2 into \eqref{eq:smooth}:
\begin{align}
f(x_{k+1}) &\le f(x_{k})
-\alpha\|\nabla f(x_{k})\|^{2}
+\langle\nabla f(x_{k}),\xi_{k}\rangle \nonumber\\
&\quad +\frac{L}{2}\Bigl(
\alpha^{2}\|\nabla f(x_{k})\|^{2}
-2\alpha\langle\nabla f(x_{k}),\xi_{k}\rangle
+\|\xi_{k}\|^{2}
\Bigr). \label{eq:expand}
\end{align}
\textbf{[Step 3]}

\noindent\textbf{Step 4.} Collect terms in $\langle\nabla f(x_{k}),\xi_{k}\rangle$:
\begin{align}
f(x_{k+1}) &\le f(x_{k})
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_{k})\|^{2}
+\Bigl(1-\!L\alpha\Bigr)\langle\nabla f(x_{k}),\xi_{k}\rangle
+\frac{L}{2}\|\xi_{k}\|^{2}. \label{eq:collected}
\end{align}
\textbf{[Step 4]}

\noindent\textbf{Step 5.} Take conditional expectation $\E_{k}[\cdot]$.
Because $\xi_{k}$ is independent of $x_{k}$ and $\E[\xi_{k}]=0$, the cross term vanishes:
\[
\E_{k}\bigl[\langle\nabla f(x_{k}),\xi_{k}\rangle\bigr]=
\langle\nabla f(x_{k}),\E_{k}[\xi_{k}]\rangle=0.
\]
Thus,
\begin{align}
\E_{k}\!\bigl[f(x_{k+1})\bigr]
&\le f(x_{k})
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_{k})\|^{2}
+\frac{L}{2}\E_{k}\!\bigl[\|\xi_{k}\|^{2}\bigr]. \label{eq:condexp}
\end{align}
\textbf{[Step 5]}

\noindent\textbf{Step 6.} Use Lemma \ref{lem:noise} to substitute the noise second moment:
\[
\E_{k}\!\bigl[\|\xi_{k}\|^{2}\bigr]=d\,\sigma_{k}^{2}=d\,\frac{\sigma^{2}}{k+1}.
\]
Hence,
\begin{align}
\E_{k}\!\bigl[f(x_{k+1})\bigr]
&\le f(x_{k})
-\alpha\Bigl(1-\frac{L\alpha}{2}\Bigr)\|\nabla f(x_{k})\|^{2}
+\frac{Ld\sigma^{2}}{2(k+1)}. \label{eq:afternoise}
\end{align}
\textbf{[Step 6]}

\noindent\textbf{Step 7.} Since $\alpha\le 1/L$, we have $1-\frac{L\alpha}{2}\ge \frac{1}{2}$.
Thus,
\begin{align}
\E_{k}\!\bigl[f(x_{k+1})\bigr]
&\le f(x_{k})
-\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}
+\frac{Ld\sigma^{2}}{2(k+1)}. \label{eq:simplified}
\end{align}
\textbf{[Step 7]}

\noindent\textbf{Step 8.} Rearrange \eqref{eq:simplified} to isolate the gradient term:
\begin{align}
\frac{\alpha}{2}\|\nabla f(x_{k})\|^{2}
&\le f(x_{k})-\E_{k}\!\bigl[f(x_{k+1})\bigr]
+\frac{Ld\sigma^{2}}{2(k+1)}. \label{eq:gradbound}
\end{align}
\textbf{[Step 8]}

\noindent\textbf{Step 9.} Take total expectation (i.e., $\E[\cdot]=\E_{0}\!\bigl[\E_{1}\!\bigl[\dots\E_{k}[\cdot]\dots]\bigr]$) on both sides of \eqref{eq:gradbound} and sum from $k=0$ to $T-1$:
\begin{align}
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
&\le \sum_{k=0}^{T-1}\bigl(\E[f(x_{k})]-\E[f(x_{k+1})]\bigr)
+\frac{Ld\sigma^{2}}{2}\sum_{k=0}^{T-1}\frac{1}{k+1}. \label{eq:sum}
\end{align}
\textbf{[Step 9]}

\noindent\textbf{Step 10.} The telescoping sum collapses:
\[
\sum_{k=0}^{T-1}\bigl(\E[f(x_{k})]-\E[f(x_{k+1})]\bigr)
= \E[f(x_{0})]-\E[f(x_{T})]
\le f(x_{0})-f_{\inf},
\]
where the last inequality uses Assumption \ref{as:lb}.
\textbf{[Step 10]}

\noindent\textbf{Step 11.} The harmonic series satisfies
\[
\sum_{k=0}^{T-1}\frac{1}{k+1}\le 1+\int_{1}^{T}\frac{1}{t}\,dt = 1+\log T \le \log (T+1)+1.
\]
For simplicity we bound it by $\log (T+1)+1\le 2\log (T+1)$ for $T\ge 2$; the constant factor does not affect the $O(\cdot)$ rate. Keeping the exact expression yields
\[
\sum_{k=0}^{T-1}\frac{1}{k+1}=H_{T}\le\log (T+1)+\gamma,
\]
where $\gamma<1$ is Euler’s constant. We will use the looser bound $\log (T+1)+1$.
\textbf{[Step 11]}

\noindent\textbf{Step 12.} Combine Steps 9–11:
\begin{align}
\frac{\alpha}{2}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
&\le f(x_{0})-f_{\inf}
+\frac{Ld\sigma^{2}}{2}\bigl(\log (T+1)+1\bigr). \label{eq:prefinal}
\end{align}
\textbf{[Step 12]}

\noindent\textbf{Step 13.} Divide both sides by $\alpha T$ and note that $d$ can be absorbed into the constant $L\sigma^{2}$ (we redefine $L\leftarrow Ld$ for brevity). This yields the averaged gradient bound
\begin{align}
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
&\le \frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha T}
+ \frac{L\sigma^{2}\,\log (T+1)}{T}
+ \frac{L\sigma^{2}}{T}. \label{eq:final}
\end{align}
Dropping the negligible $L\sigma^{2}/T$ term gives exactly the rate \eqref{eq:rate} claimed in Theorem \ref{thm:main}.
\textbf{[Step 13]}

Thus the theorem is proved. $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{eq:final} we identify three explicit constants:
\begin{itemize}
\item $C_{1}=2\bigl(f(x_{0})-f_{\inf}\bigr)$ multiplies the $1/(\alpha T)$ term.
\item $C_{2}=L\sigma^{2}$ multiplies the $\log (T+1)/T$ term.
\item An additional $C_{2}/T$ term, negligible for large $T$.
\end{itemize}
If one wishes to minimize the right‑hand side with respect to $\alpha$, differentiate $C_{1}/(\alpha T)+C_{2}\log (T)/T$ w.r.t.\ $\alpha$ and set to zero, obtaining the optimal constant step size
\[
\alpha^{\star}= \sqrt{\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{L\sigma^{2}\log (T+1)}}.
\]
Plugging $\alpha^{\star}$ back gives the optimal bound
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
\le 2\sqrt{\frac{L\sigma^{2}\bigl(f(x_{0})-f_{\inf}\bigr)}\;\frac{\sqrt{\log (T+1)}}{T}.
\]

\section*{Special Cases}
\begin{enumerate}[label=\textbf{Case \arabic*:}, leftmargin=*]
\item \textbf{Zero noise ($\sigma^{2}=0$).} Inequality \eqref{eq:rate} simplifies to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha T},
\]
the standard $O(1/T)$ guarantee for deterministic gradient descent on smooth non‑convex functions.

\item \textbf{Constant variance $\sigma_{k}^{2}\equiv\sigma^{2}$.} The harmonic sum in Step 11 becomes $T$, leading to
\[
\frac{1}{T}\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
\le \frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{\alpha T}+ \frac{L\sigma^{2}}{2},
\]
which exhibits a non‑vanishing bias term $L\sigma^{2}/2$; the iterates converge only to a neighbourhood of stationary points.

\item \textbf{Decreasing step size $\alpha_{k}=c/(k+1)$.} Repeating the above derivation with $\alpha$ replaced by $\alpha_{k}$ yields
\[
\sum_{k=0}^{T-1}\E\!\bigl[\|\nabla f(x_{k})\|^{2}\bigr]
\le
\frac{2\bigl(f(x_{0})-f_{\inf}\bigr)}{c}
+ L\sigma^{2}\sum_{k=0}^{T-1}\frac{1}{k+1},
\]
and consequently an $O(\log T/T)$ rate for the averaged gradient norm.
\end{enumerate}

\section*{Discussion}
The proof shows that the decaying noise schedule $\sigma_{k}^{2}=O(1/k)$ is sufficient to guarantee that the stochastic perturbations do not dominate the descent dynamics. The bound \eqref{eq:rate} matches the optimal $O(1/T)$ rate for deterministic gradient descent up to a logarithmic factor induced by the vanishing noise. This factor is unavoidable for any algorithm that injects isotropic Gaussian noise whose variance decays only polynomially; faster decay (e.g., $O(1/k^{1+\epsilon})$) would improve the logarithmic term at the expense of weaker exploration properties.
