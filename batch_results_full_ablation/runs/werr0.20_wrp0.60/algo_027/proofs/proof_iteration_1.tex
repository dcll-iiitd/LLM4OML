\section*{Randomized Coordinate Descent with Probability‑Weighted Updates (RCD‑PWU)}

\section*{Mathematical Formulation}
Let $f:\RR^{d}\rightarrow\RR$ be differentiable (not necessarily convex).  
For each coordinate $i\in\{1,\dots ,d\}$ denote by $e_i\in\RR^{d}$ the $i$‑th canonical basis vector and define the partial derivative 
\[
\nabla_i f(w)\;:=\;\frac{\partial f}{\partial w_i}(w)=\langle\nabla f(w),e_i\rangle .
\]

\paragraph{Sampling distribution.}  
A probability vector $p=(p_1,\dots ,p_d)$ satisfies $p_i>0$ and $\sum_{i=1}^{d}p_i=1$.  
At iteration $t$ we draw a random index $i_t\sim p$ independently of the past.

\paragraph{Unbiased gradient estimator.}  
Define
\[
g_t \;:=\; \frac{1}{p_{i_t}}\;\nabla_{i_t}f(w_t)\;e_{i_t}\in\RR^{d}.
\tag{1}
\]
Conditioned on $w_t$, $g_t$ satisfies
\[
\EE\!\left[g_t\mid w_t\right]
   =\sum_{i=1}^{d}p_i\frac{1}{p_i}\nabla_i f(w_t)e_i
   =\nabla f(w_t),
\tag{2}
\]
hence $g_t$ is an unbiased estimator of the full gradient.

\paragraph{Update rule.}  
With a (possibly time‑varying) stepsize $\eta_t>0$ we perform
\[
w_{t+1}=w_t-\eta_t\,g_t
      =w_t-\frac{\eta_t}{p_{i_t}}\;\nabla_{i_t}f(w_t)\;e_{i_t}.
\tag{3}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Randomized Coordinate Descent with Probability‑Weighted Updates}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $w_0\in\RR^{d}$, stepsize schedule $\{\eta_t\}_{t\ge0}$, probability vector $p\in\Delta^{d}$.
\For{$t=0,1,2,\dots ,T-1$}
    \State Sample $i_t\in\{1,\dots ,d\}$ with $\Pr(i_t=i)=p_i$.
    \State Compute partial gradient $g_{i_t}(w_t)=\nabla_{i_t}f(w_t)$.
    \State Form unbiased estimator $g_t\gets \frac{1}{p_{i_t}}g_{i_t}(w_t)\,e_{i_t}$.
    \State Update $w_{t+1}\gets w_t-\eta_t g_t$.
\EndFor
\State \textbf{Return} $w_T$ (or the average $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{enumerate}[label=\textbf{A\arabic*.},leftmargin=*,nosep]
\item \textbf{Coordinate‑wise $L_i$‑smoothness.}  
For each $i\in\{1,\dots ,d\}$ there exists $L_i>0$ such that for all $w\in\RR^{d}$ and $\alpha\in\RR$,
\[
\bigl|\nabla_i f(w+\alpha e_i)-\nabla_i f(w)\bigr|
\le L_i\,|\alpha|.
\tag{A1}
\]

\item \textbf{Bounded variance of the estimator.}  
Define the random estimator $g_t$ from (1). Then for all $t$,
\[
\EE\!\bigl[\|g_t-\nabla f(w_t)\|^2\mid w_t\bigr]
\le \sigma^2,
\tag{A2}
\]
for some $\sigma\ge0$ (the bound will be used only for completeness; the main result does not require a finite $\sigma$).

\item \textbf{Stepsize upper bound.}  
The stepsize sequence satisfies
\[
0<\eta_t\le \min_{i}\frac{p_i}{L_i}\quad\text{for all }t.
\tag{A3}
\]

\item \textbf{Existence of a lower bound.}  
$f$ is bounded below, i.e. $f^{\inf}:=\inf_{w}f(w)>-\infty$.
\tag{A4}
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Expected squared gradient norm decay]\label{thm:main}
Assume \textbf{A1}–\textbf{A4} and let the stepsize be constant $\eta_t\equiv\eta$ satisfying \textbf{A3}. Then for any $T\ge1$,
\[
\frac{1}{T}\sum_{t=0}^{T-1}\EE\!\bigl[\|\nabla f(w_t)\|^2\bigr]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\inf}\bigr)}{\eta\,T}.
\tag{4}
\]
Consequently $\min_{0\le t<T}\EE[\|\nabla f(w_t)\|^2]=O(1/T)$.
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\textbf{P\arabic*.},leftmargin=*,nosep]
\item \textbf{Unbiasedness.}  By construction (2) the estimator $g_t$ is unbiased, which is the key to obtaining (4) without convexity.
\item \textbf{Stability w.r.t. $p_i$.}  
The admissible stepsize is $\eta\le\min_i p_i/L_i$.  Choosing $p_i\propto L_i$ maximises the allowable $\eta$ and yields the fastest theoretical rate.
\item \textbf{Comparison to full‑gradient SGD.}  
When $p_i=1/d$ for all $i$, $\eta\le\frac{1}{d\max_i L_i}$, reproducing the standard coordinate‑descent bound.  The probability‑weighted estimator removes the $1/d$ factor in the expectation, leading to identical per‑iteration progress as full‑gradient SGD under the same stepsize bound.
\item \textbf{Extension to diminishing stepsizes.}  
If $\eta_t$ is non‑increasing and $\sum_{t}\eta_t=\infty$, $\sum_{t}\eta_t^{2}<\infty$, the same proof yields $\EE[\|\nabla f(w_t)\|]\to0$.
\end{enumerate}

\section*{Discussion}
The theorem guarantees that even for non‑convex objectives the algorithm drives the expected gradient norm to zero at a rate $O(1/T)$.  The bound is independent of the dimension $d$ except through the probability vector $p$.  By allocating larger probabilities to coordinates with larger Lipschitz constants $L_i$, one can increase the maximal permissible stepsize and thus improve practical performance.  The proof relies solely on coordinate‑wise smoothness and does not require any global Lipschitz or convexity assumptions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Descent Lemma for a single coordinate]\label{lem:descent}
Let $f$ satisfy \textbf{A1}. For any $w\in\RR^{d}$, any $i$, and any scalar $\alpha$,
\[
f(w-\alpha e_i)\le f(w)-\alpha\,\nabla_i f(w)+\frac{L_i}{2}\alpha^{2}.
\tag{5}
\]
\end{lemma}
\begin{proof}
From \textbf{A1} we have the one‑dimensional smoothness inequality
\[
\nabla_i f(w-\alpha e_i)
= \nabla_i f(w)-\int_{0}^{\alpha} \frac{d}{ds}\nabla_i f(w-s e_i)\,ds
\ge \nabla_i f(w)-L_i|\alpha|.
\]
Integrating the fundamental theorem of calculus yields (5).  ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof let $\mathcal{F}_t:=\sigma(w_0,i_0,\dots,i_{t-1})$ denote the filtration generated up to iteration $t$.

\subsection*{Step 1: Apply descent lemma to the random update}
Conditioned on $\mathcal{F}_t$, the update (3) modifies only coordinate $i_t$ with step
$\alpha_t:=\frac{\eta}{p_{i_t}}\nabla_{i_t}f(w_t)$.  Using Lemma~\ref{lem:descent} with $\alpha=\alpha_t$ we obtain
\begin{align}
f(w_{t+1})
&=f\!\bigl(w_t-\alpha_t e_{i_t}\bigr) \nonumber\\
&\le f(w_t)-\alpha_t\,\nabla_{i_t}f(w_t)
      +\frac{L_{i_t}}{2}\alpha_t^{2}.
\tag{Step 1}
\end{align}

\subsection*{Step 2: Substitute the expression for $\alpha_t$}
Recall $\alpha_t=\frac{\eta}{p_{i_t}}\nabla_{i_t}f(w_t)$.  Plugging into (Step 1) gives
\begin{align}
f(w_{t+1})
&\le f(w_t)-\frac{\eta}{p_{i_t}}\bigl[\nabla_{i_t}f(w_t)\bigr]^2
      +\frac{L_{i_t}}{2}\Bigl(\frac{\eta}{p_{i_t}}\Bigr)^{2}
        \bigl[\nabla_{i_t}f(w_t)\bigr]^2 \nonumber\\
&= f(w_t)-\frac{\eta}{p_{i_t}}\Bigl(1-\frac{\eta L_{i_t}}{2p_{i_t}}\Bigr)
      \bigl[\nabla_{i_t}f(w_t)\bigr]^2 .
\tag{Step 2}
\end{align}

\subsection*{Step 3: Take conditional expectation}
Taking $\EE[\cdot\mid\mathcal{F}_t]$ and using that $i_t$ is drawn from $p$,
\begin{align}
\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(w_t)-\eta\sum_{i=1}^{d}
     \frac{p_i}{p_i}\Bigl(1-\frac{\eta L_i}{2p_i}\Bigr)
     \bigl[\nabla_i f(w_t)\bigr]^2 \nonumber\\
&= f(w_t)-\eta\sum_{i=1}^{d}
     \Bigl(1-\frac{\eta L_i}{2p_i}\Bigr)
     \bigl[\nabla_i f(w_t)\bigr]^2 .
\tag{Step 3}
\end{align}

\subsection*{Step 4: Use the stepsize bound \textbf{A3}}
By \textbf{A3}, $\eta\le p_i/L_i$ for every $i$, thus $\eta L_i/(2p_i)\le \tfrac12$. Consequently
\[
1-\frac{\eta L_i}{2p_i}\ge \frac12 \qquad\forall i.
\]
Insert this lower bound into (Step 3):
\begin{align}
\EE\!\bigl[f(w_{t+1})\mid\mathcal{F}_t\bigr]
&\le f(w_t)-\frac{\eta}{2}
      \sum_{i=1}^{d}\bigl[\nabla_i f(w_t)\bigr]^2 \nonumber\\
&= f(w_t)-\frac{\eta}{2}\,\|\nabla f(w_t)\|^{2}.
\tag{Step 4}
\end{align}

\subsection*{Step 5: Unconditional expectation and telescoping sum}
Take full expectation and sum over $t=0,\dots,T-1$:
\begin{align}
\EE[f(w_T)]
&\le f(w_0)-\frac{\eta}{2}\sum_{t=0}^{T-1}
      \EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr].
\tag{Step 5}
\end{align}
Rearranging,
\[
\frac{1}{T}\sum_{t=0}^{T-1}
   \EE\!\bigl[\|\nabla f(w_t)\|^{2}\bigr]
\le \frac{2\bigl(f(w_0)-\EE[f(w_T)]\bigr)}{\eta T}.
\tag{Step 6}
\]

\subsection*{Step 6: Apply lower bound \textbf{A4}}
Since $f(w_T)\ge f^{\inf}$ almost surely,
\[
\EE[f(w_T)]\ge f^{\inf}.
\]
Plugging into (Step 6) yields exactly the bound (4) claimed in Theorem~\ref{thm:main}.
∎

\section*{Rate Derivation (With Constants)}
From (4) we have
\[
\min_{0\le t<T}\EE[\|\nabla f(w_t)\|^{2}]
\;\le\;
\frac{2\bigl(f(w_0)-f^{\inf}\bigr)}{\eta\,T}.
\]
If we set the maximal admissible constant stepsize
\[
\eta^{\star}:=\min_{i}\frac{p_i}{L_i},
\tag{7}
\]
the right‑hand side becomes
\[
\frac{2L_{\max}}{p_{\min}}\,
\frac{f(w_0)-f^{\inf}}{T},
\quad\text{where }L_{\max}=\max_i L_i,\;p_{\min}=\min_i p_i .
\]

\section*{Special Cases}
\begin{enumerate}[label=\textbf{S\arabic*.},leftmargin=*,nosep]
\item \textbf{Uniform sampling.} $p_i=1/d$ for all $i$.  
Then $\eta^{\star}= \frac{1}{d\max_i L_i}$ and the bound recovers the classical $O(d/T)$ dependence hidden in the stepsize.
\item \textbf{Importance sampling.} Choose $p_i\propto L_i$.  
Then $\eta^{\star}=1/\sum_{i}L_i$ and the factor $d$ disappears; the convergence rate depends only on the sum of Lipschitz constants.
\item \textbf{Block coordinates.} If coordinates are grouped into blocks $B_1,\dots,B_m$ and a block is sampled with probability $q_j$, the same proof holds after replacing $L_i$ by block‑wise smoothness constants.
\end{enumerate}
