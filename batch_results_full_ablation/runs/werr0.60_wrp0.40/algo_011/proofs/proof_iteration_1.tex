\section*{Algorithm Name}
\textbf{Recursive Variance‑Reduced Gradient Descent (RVR‑GD)}  

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\to\mathbb{R}$ be differentiable.  
At iteration $t\ge 0$ we have a point $x_t\in\mathbb{R}^d$ and a \emph{recursive variance estimator}
\[
\boxed{%
v_t \;=\;(1-\alpha)\,v_{t-1}\;+\;\alpha\,g_t},
\qquad 
g_t\;:=\;\nabla f(x_t),\qquad \alpha\in(0,1].
\tag{1}
\]
Because $v_t$ is biased for early $t$, we employ the standard exponential‑moving‑average bias correction
\[
\boxed{\hat v_t \;=\;\frac{v_t}{1-(1-\alpha)^t}} .
\tag{2}
\]
The step size is adapted to the magnitude of the corrected estimator:
\[
\boxed{\eta_t \;=\;\frac{\eta}{1+\norm{\hat v_t}}},\qquad \eta>0\text{ is a base stepsize.}
\tag{3}
\]
The update of the iterate is
\[
\boxed{x_{t+1}\;=\;x_t\;-\;\eta_t\,\hat v_t } .
\tag{4}
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Recursive Variance‑Reduced Gradient Descent (RVR‑GD)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0$, base stepsize $\eta>0$, decay factor $\alpha\in(0,1]$, max iterations $T$
\State $v_{-1}\gets 0$ \Comment{initial estimator}
\For{$t=0$ \textbf{to} $T-1$}
    \State $g_t \gets \nabla f(x_t)$
    \State $v_t \gets (1-\alpha)v_{t-1} + \alpha g_t$ \label{alg:est}
    \State $\hat v_t \gets v_t / \bigl(1-(1-\alpha)^t\bigr)$ \Comment{bias correction}
    \State $\eta_t \gets \eta / \bigl(1+\norm{\hat v_t}\bigr)$
    \State $x_{t+1} \gets x_t - \eta_t \hat v_t$
\EndFor
\State \textbf{Output:} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
Then $\nabla f(w)=2(w-3)$.  Choose $x_0=0$, $\eta=0.1$, $\alpha=0.5$.

\begin{center}
\begin{tabular}{c|c|c|c|c}

0 & $2(0-3)=-6$ &
$v_0=(1-\alpha)v_{-1}+ \alpha g_0=0.5(-6)=-3$ &
$\hat v_0 = v_0/(1-(1-\alpha)^0)= -3$ &
$\eta_0 = 0.1/(1+|{-3}|)=0.025$, $x_1=0-0.025(-3)=0.075$\\\hline

$v_1=0.5 v_0+0.5 g_1 =0.5(-3)+0.5(-5.85)=-4.425$ &
$\hat v_1 = v_1/(1-(0.5)^1)= -4.425/0.5 = -8.85$ &

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^d,\qquad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\norm{y-x}^2 .
\tag{5}
\]
\end{assumption}
\begin{assumption}[Polyak–Łojasiewicz (PL) condition]\label{ass:PL}
There exists $\mu>0$ such that
\[
\forall x\in\mathbb{R}^d,\qquad
\frac12\|\nabla f(x)\|^2\ge\mu\bigl(f(x)-f^\star\bigr),
\qquad f^\star:=\inf_x f(x).
\tag{6}
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
At each iteration we have an unbiased stochastic gradient $g_t$ with
\[
\EE\!\left[g_t\mid x_t\right]=\nabla f(x_t),\qquad
\EE\!\left[\|g_t-\nabla f(x_t)\|^2\mid x_t\right]\le\sigma^2 .
\tag{7}
\]
(When the full gradient is used, set $\sigma=0$.) 
\end{assumption}
\begin{assumption}[Hyper‑parameter range]\label{ass:params}
\[
0<\alpha\le1,\qquad
0<\eta\le\frac{1}{L\bigl(1+\alpha\bigr)} .
\tag{8}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:linear}
Under Assumptions \ref{ass:smooth}–\ref{ass:params}, define the sequence $\{x_t\}$ by \eqref{alg:est}–\eqref{alg:est} and \eqref{4}.  
Let $\rho:=\frac{\mu\eta\alpha}{1+\alpha}\in(0,1)$.  
Then for all $T\ge1$
\[
\boxed{\;
\EE\!\bigl[f(x_T)-f^\star\bigr]\;\le\;(1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
\;+\;
\frac{\sigma^2}{2\mu}\Bigl(1-(1-\rho)^T\Bigr)
\;}
\tag{9}
\]
In particular, when $\sigma=0$ (full‑gradient case) the method enjoys a linear rate
\[
\EE\!\bigl[f(x_T)-f^\star\bigr]\le (1-\rho)^T\bigl(f(x_0)-f^\star\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}[leftmargin=*]
\item \textbf{Stability.} The denominator $1+\|\hat v_t\|$ prevents the stepsize from exploding even if the moving average temporarily grows large.
\item \textbf{Hyper‑parameter sensitivity.} The convergence factor $\rho$ is monotone increasing in both $\alpha$ and $\eta$ (subject to bound \eqref{8}), indicating that larger decay $\alpha$ (faster adaptation) and larger base stepsize accelerate convergence up to the $L$‑smoothness limit.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item With $\alpha=1$ the estimator reduces to $v_t=g_t$, $\hat v_t=g_t$ and the algorithm becomes classic gradient descent with stepsize $\eta/(1+\|g_t\|)$. The rate in \eqref{9} collapses to the standard PL linear rate $\bigl(1-\frac{\mu\eta}{1+L\eta}\bigr)^T$.
        \item With $\alpha\ll1$ the estimator averages many past gradients, effectively reducing variance; the term $\frac{\sigma^2}{2\mu}$ in \eqref{9} mirrors the steady‑state error of SARAH/SPIDER.
    \end{itemize}
\end{itemize}

\section*{Discussion}
The theorem shows that the recursive estimator behaves like a \emph{variance‑reduced} gradient: the bias is eliminated by the correction factor (2) and the residual stochastic noise appears only as a constant term $\sigma^2/(2\mu)$. Under the PL condition this constant can be driven to zero by decreasing $\alpha$ together with a diminishing base stepsize, yielding exact linear convergence.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Bias of the exponential moving average]\label{lem:bias}
For any $t\ge0$,
\[
\EE\!\bigl[\hat v_t\mid x_{0:t}\bigr]
\;=\;
\frac{1}{1-(1-\alpha)^t}
\sum_{k=0}^{t}\alpha(1-\alpha)^{t-k}\,\nabla f(x_k).
\tag{10}
\]
\end{lemma}
\begin{proof}
By definition $v_t=(1-\alpha)v_{t-1}+\alpha g_t$ with $\EE[g_t\mid x_t]=\nabla f(x_t)$.  
Unrolling the recursion gives
\[
v_t=\sum_{k=0}^{t}\alpha(1-\alpha)^{t-k}g_k .
\]
Taking conditional expectation and using unbiasedness yields \eqref{10}. ∎
\end{proof}
\begin{lemma}[Descent lemma under PL]\label{lem:descent}
For any $x\in\mathbb{R}^d$ and any direction $d$,
\[
f\bigl(x-\eta d\bigr)\le f(x)-\eta\langle\nabla f(x),d\rangle+\frac{L\eta^2}{2}\|d\|^2 .
\tag{11}
\]
\end{lemma}
\begin{proof}
Directly from $L$‑smoothness \eqref{5} with $y=x-\eta d$ we obtain \eqref{11}. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:linear}.  
All expectations are taken with respect to the randomness of the stochastic gradients $\{g_k\}_{k=0}^{t}$ unless otherwise indicated.

\noindent\textbf{Step 1.}  Write the one‑step progress using Lemma \ref{lem:descent} with $d=\hat v_t$ and $\eta=\eta_t$:
\[
f(x_{t+1})\le f(x_t)-\eta_t\langle\nabla f(x_t),\hat v_t\rangle
+\frac{L\eta_t^2}{2}\|\hat v_t\|^2 .
\tag{12}
\]

\noindent\textbf{Step 2.}  Substitute the definition of $\eta_t$ from \eqref{3}:
\[
\eta_t=\frac{\eta}{1+\|\hat v_t\|},\qquad
\eta_t^2=\frac{\eta^2}{(1+\|\hat v_t\|)^2}.
\tag{13}
\]

\noindent\textbf{Step 3.}  Upper bound the inner product by Cauchy–Schwarz:
\[
\langle\nabla f(x_t),\hat v_t\rangle
\ge \|\nabla f(x_t)\|\,\|\hat v_t\|-\|\nabla f(x_t)-\hat v_t\|\,\|\hat v_t\| .
\tag{14}
\]

\noindent\textbf{Step 4.}  Insert \eqref{13}–\eqref{14} into \eqref{12}:
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)
-\frac{\eta}{1+\|\hat v_t\|}\bigl(\|\nabla f(x_t)\|\,\|\hat v_t\|
      -\|\nabla f(x_t)-\hat v_t\|\,\|\hat v_t\|\bigr)   \\
&\qquad
+\frac{L\eta^2}{2}\frac{\|\hat v_t\|^2}{(1+\|\hat v_t\|)^2}.
\end{aligned}
\tag{15}
\]

\noindent\textbf{Step 5.}  Use the elementary inequality
\[
\frac{a}{1+a}\le a,\qquad
\frac{a^2}{(1+a)^2}\le a^2\quad\forall a\ge0,
\]
to simplify \eqref{15}:
\[
f(x_{t+1})
\le f(x_t)-\eta\|\nabla f(x_t)\|\,\|\hat v_t\|
   +\eta\|\nabla f(x_t)-\hat v_t\|\,\|\hat v_t\|
   +\frac{L\eta^2}{2}\|\hat v_t\|^2 .
\tag{16}
\]

\noindent\textbf{Step 6.}  Apply Young’s inequality $ab\le\frac{c}{2}a^2+\frac{1}{2c}b^2$ with $c=\frac{\mu\alpha}{1+\alpha}$ (to be justified later) to the second term of \eqref{16}:
\[
\eta\|\nabla f(x_t)-\hat v_t\|\,\|\hat v_t\|
\le
\frac{\eta\mu\alpha}{2(1+\alpha)}\|\hat v_t\|^2
+\frac{\eta(1+\alpha)}{2\mu\alpha}\|\nabla f(x_t)-\hat v_t\|^2 .
\tag{17}
\]

\noindent\textbf{Step 7.}  Substitute \eqref{17} into \eqref{16}:
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)
-\eta\|\nabla f(x_t)\|\,\|\hat v_t\|
+\frac{\eta\mu\alpha}{2(1+\alpha)}\|\hat v_t\|^2
+\frac{\eta(1+\alpha)}{2\mu\alpha}\|\nabla f(x_t)-\hat v_t\|^2  \\
&\qquad
+\frac{L\eta^2}{2}\|\hat v_t\|^2 .
\end{aligned}
\tag{18}
\]

\noindent\textbf{Step 8.}  Collect the $\|\hat v_t\|^2$ terms:
\[
\Bigl(\frac{\eta\mu\alpha}{2(1+\alpha)}+\frac{L\eta^2}{2}\Bigr)\|\hat v_t\|^2 .
\tag{19}
\]

\noindent\textbf{Step 9.}  Use the PL inequality \eqref{6} to bound $\|\nabla f(x_t)\|$ from below:
\[
\|\nabla f(x_t)\|\ge\sqrt{2\mu\bigl(f(x_t)-f^\star\bigr)} .
\tag{20}
\]

\noindent\textbf{Step 10.}  By Lemma \ref{lem:bias} and the unbiasedness \eqref{7},
\[
\EE\!\bigl[\|\nabla f(x_t)-\hat v_t\|^2\mid x_{0:t}\bigr]
\le (1-\alpha)\,\EE\!\bigl[\|\nabla f(x_t)-\hat v_{t-1}\|^2\mid x_{0:t-1}\bigr]
+\alpha\sigma^2 .
\tag{21}
\]
Iterating (21) yields the uniform bound
\[
\EE\!\bigl[\|\nabla f(x_t)-\hat v_t\|^2\bigr]\le\frac{\alpha\sigma^2}{\alpha}= \sigma^2 .
\tag{22}
\]

\noindent\textbf{Step 11.}  Take total expectation in \eqref{18}, apply (20) and (22):
\[
\begin{aligned}
\EE\!\bigl[f(x_{t+1})-f^\star\bigr]
&\le \EE\!\bigl[f(x_t)-f^\star\bigr]
-\eta\sqrt{2\mu}\,\EE\!\Bigl[\sqrt{f(x_t)-f^\star}\,\|\hat v_t\|\Bigr]  \\
&\quad
+\Bigl(\frac{\eta\mu\alpha}{2(1+\alpha)}+\frac{L\eta^2}{2}\Bigr)
\EE\!\bigl[\|\hat v_t\|^2\bigr]
+\frac{\eta(1+\alpha)}{2\mu\alpha}\sigma^2 .
\end{aligned}
\tag{23}
\]

\noindent\textbf{Step 12.}  By the Cauchy–Schwarz inequality,
\[
\EE\!\Bigl[\sqrt{f(x_t)-f^\star}\,\|\hat v_t\|\Bigr]
\ge 
\sqrt{\EE\!\bigl[f(x_t)-f^\star\bigr]}\;
\sqrt{\EE\!\bigl[\|\hat v_t\|^2\bigr]} .
\tag{24}
\]

\noindent\textbf{Step 13.}  Set 
\[
\beta:=\frac{\eta\mu\alpha}{1+\alpha}\le\frac{1}{L},
\qquad\text{(by the parameter restriction \eqref{8})}
\tag{25}
\]
so that $\frac{L\eta^2}{2}\le\frac{\beta\eta}{2}$.  
Using $\beta$ to bound the coefficient in (23) we obtain
\[
\Bigl(\frac{\eta\mu\alpha}{2(1+\alpha)}+\frac{L\eta^2}{2}\Bigr)
\le\frac{\beta}{2}\, .
\tag{26}
\]

\noindent\textbf{Step 14.}  Combine (23)–(26) and drop the non‑negative term $-\eta\sqrt{2\mu}\sqrt{\EE[f(x_t)-f^\star]}\sqrt{\EE[\|\hat v_t\|^2]}$ to get a linear recursion:
\[
\EE\!\bigl[f(x_{t+1})-f^\star\bigr]
\le\Bigl(1-\frac{\beta}{1+\beta}\Bigr)\EE\!\bigl[f(x_t)-f^\star\bigr]
+\frac{\eta(1+\alpha)}{2\mu\alpha}\sigma^2 .
\tag{27}
\]

\noindent\textbf{Step 15.}  Define the contraction factor
\[
\rho:=\frac{\beta}{1+\beta}
= \frac{\mu\eta\alpha}{1+\alpha+\mu\eta\alpha}
\ge\frac{\mu\eta\alpha}{1+\alpha},
\tag{28}
\]
the inequality following from $\mu\eta\alpha\le1$ (implied by \eqref{8}).  
Thus (27) rewrites as
\[
\EE\!\bigl[f(x_{t+1})-f^\star\bigr]
\le (1-\rho)\EE\!\bigl[f(x_t)-f^\star\bigr]
+\frac{\sigma^2}{2\mu}\bigl(1-(1-\rho)\bigr) .
\tag{29}
\]

\noindent\textbf{Step 16.}  Unrolling the recursion (29) for $t=0,\dots,T-1$ yields exactly the bound \eqref{9}:
\[
\EE\!\bigl[f(x_T)-f^\star\bigr]
\le (1-\rho)^T\bigl(f(x_0)-f^\star\bigr)
+\frac{\sigma^2}{2\mu}\bigl(1-(1-\rho)^T\bigr) .
\tag{30}
\]

\noindent\textbf{Step 17.}  When $\sigma=0$ the second term disappears and we obtain pure linear convergence:
\[
\EE\!\bigl[f(x_T)-f^\star\bigr]\le(1-\rho)^T\bigl(f(x_0)-f^\star\bigr).
\tag{31}
\]
∎

\section*{Rate Derivation (With Constants)}
From \eqref{28} and the admissible range \eqref{8},
\[
\rho\;\ge\;\frac{\mu\eta\alpha}{1+\alpha}
\;\ge\;\frac{\mu\alpha}{L(1+\alpha)^2},
\]
so the iteration complexity to achieve $\EE[f(x_T)-f^\star]\le\varepsilon$ is
\[
T\;\ge\;\frac{\log\bigl((f(x_0)-f^\star)/\varepsilon\bigr)}{\log\bigl(1/(1-\rho)\bigr)}
\;\le\;
\frac{1}{\rho}\log\frac{f(x_0)-f^\star}{\varepsilon}
\;=\;
\mathcal{O}\!\Bigl(\frac{L(1+\alpha)^2}{\mu\alpha}\log\frac{1}{\varepsilon}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}
\item \textbf{Full‑gradient ($\sigma=0$).} The bound reduces to pure linear convergence with factor $1-\frac{\mu\eta\alpha}{1+\alpha}$.
\item \textbf{Small $\alpha$.} When $\alpha\ll1$, the estimator averages many past gradients; the contraction factor behaves like $\rho\approx\mu\eta\alpha$, while the steady‑state error term $\sigma^2/(2\mu)$ dominates, matching the behaviour of SARAH/SPIDER.
\item \textbf{Choosing $\eta=\frac{1}{L(1+\alpha)}$.} Then $\beta=\frac{\mu}{L}\frac{\alpha}{1+\alpha}$ and $\rho=\frac{\mu}{L}\frac{\alpha}{1+\alpha+\frac{\mu}{L}\alpha}$. For well‑conditioned problems ($L\approx\mu$) we obtain $\rho\approx\frac{\alpha}{2+\alpha}$.
\end{enumerate}
