\section*{Algorithm Name}
\textbf{SVRG under the Polyak‑\L{}ojasiewicz (PL) condition}  

The method is a double–loop variance‑reduced stochastic gradient scheme.  
The outer loop computes a full gradient at a \emph{snapshot} point; the inner
loop performs $m$ stochastic updates using the SVRG estimator.

\section*{Mathematical Formulation}
Let  
\[
F(x)\;=\;\frac{1}{n}\sum_{i=1}^{n}f_i(x),\qquad x\in\R^{d},
\]
where each $f_i$ is $L$‑smooth.  
Denote by $x^{s}$ the snapshot after outer epoch $s$ and by
$\widetilde{\mu}^{s}= \frac1n\sum_{i=1}^{n}\grad f_i(x^{s})$ the corresponding
full gradient.

For inner iteration $t=0,\dots,m-1$ we pick an index $i_t\in\{1,\dots,n\}$
uniformly at random and define the variance‑reduced estimator
\[
v_t \;=\; \grad f_{i_t}(x_t)-\grad f_{i_t}(x^{s})+\widetilde{\mu}^{s}.
\]
The update is
\[
x_{t+1}\;=\;x_t-\eta\,v_t,
\]
with a constant stepsize $\eta>0$.  
After $m$ inner steps we set $x^{s+1}=x_m$ and start a new outer epoch.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG under PL}
\begin{algorithmic}[1]
\STATE \textbf{Input:} initial point $x^{0}$, stepsize $\eta$, inner loop length $m$
\FOR{outer epoch $s=0,1,2,\dots$}
    \STATE $\widetilde{\mu}^{s}\gets \frac{1}{n}\sum_{i=1}^{n}\grad f_i(x^{s})$ \hfill // full gradient
    \STATE $x_{0}\gets x^{s}$
    \FOR{inner iteration $t=0,\dots,m-1$}
        \STATE Sample $i_t\sim\mathrm{Unif}\{1,\dots,n\}$
        \STATE $v_t \gets \grad f_{i_t}(x_t)-\grad f_{i_t}(x^{s})+\widetilde{\mu}^{s}$
        \STATE $x_{t+1}\gets x_t-\eta v_t$
    \ENDFOR
    \STATE $x^{s+1}\gets x_{m}$
\ENDFOR
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem
\[
F(w)= (w-3)^2,\qquad n=1,\; f_1(w)= (w-3)^2.
\]
Thus $\grad f_1(w)=2(w-3)$.  
Set $w_0=0$, stepsize $\eta=0.1$, inner length $m=3$ (only one outer epoch
because $n=1$).

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (since $i_t=1$) & $F(x_t)$\\ \hline
0 & $0$ &
$\grad f_1(0)-\grad f_1(0)+\widetilde{\mu}^{0}=2(0-3)= -6$ &
$(0-3)^2=9$\\
1 & $0-0.1(-6)=0.6$ &
$\grad f_1(0.6)-\grad f_1(0)+\widetilde{\mu}^{0}
=2(0.6-3)-(-6)=-4.8+6=1.2$ &
$(0.6-3)^2=5.76$\\
2 & $0.6-0.1(1.2)=0.48$ &
$\grad f_1(0.48)-\grad f_1(0)+\widetilde{\mu}^{0}
=2(0.48-3)-(-6)=-5.04+6=0.96$ &
$(0.48-3)^2=6.3504$\\
3 & $0.48-0.1(0.96)=0.384$ &
--- &
$(0.384-3)^2=6.8609$
\end{tabular}
\end{center}
After three inner steps the iterate is $w_3=0.384$ and the objective has
decreased from $9$ to $\approx6.86$.

\section*{Assumptions}
\begin{enumerate}[label={A\arabic*}.]
\item \textbf{Smoothness.} Each component $f_i$ is $L$‑smooth:
\[
\forall x,y\in\R^{d},\qquad 
\norm{\grad f_i(x)-\grad f_i(y)}\le L\norm{x-y}.
\tag{A1}
\]

\item \textbf{Polyak‑\L{}ojasiewicz (PL) condition.} The overall objective $F$
satisfies
\[
\forall x\in\R^{d},\qquad 
\frac12\norm{\grad F(x)}^{2}\;\ge\;\mu\bigl(F(x)-F^{\star}\bigr),
\tag{A2}
\]
with $F^{\star}= \min_x F(x)$ and $\mu>0$.

\item \textbf{Finite sum.} $F(x)=\frac1n\sum_{i=1}^{n}f_i(x)$ with a fixed
finite $n$.

\item \textbf{Stepsize.} The constant stepsize satisfies
\[
0<\eta \le \frac{1}{3L}.
\tag{A3}
\]
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Linear convergence of SVRG under PL]
\label{thm:linear}
Assume (A1)–(A3) and let $m\ge \frac{2L}{\mu}$.
Define the sequence of snapshots $\{x^{s}\}_{s\ge0}$ generated by the
algorithm. Then for every outer epoch $s$,
\[
\E\!\bigl[F(x^{s+1})-F^{\star}\bigr]
\;\le\;
\underbrace{\bigl(1-\eta\mu\bigr)}_{\rho}
\,\E\!\bigl[F(x^{s})-F^{\star}\bigr].
\tag{1}
\]
Consequently,
\[
\E\!\bigl[F(x^{s})-F^{\star}\bigr]
\;\le\;
\rho^{\,s}\bigl[F(x^{0})-F^{\star}\bigr],
\qquad
\rho = 1-\eta\mu\in(0,1).
\tag{2}
\]
Thus SVRG converges linearly in expectation to the global optimum.
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.} The bound $\eta\le 1/(3L)$ guarantees that the
variance‑reduced estimator never amplifies the error; the recurrence (1)
holds regardless of the random sampling.
\item \textbf{Hyper‑parameter sensitivity.} The contraction factor
$\rho=1-\eta\mu$ improves linearly with $\eta$, but $\eta$ cannot exceed
$1/(3L)$. Choosing $\eta=1/(3L)$ yields $\rho=1-\mu/(3L)$.
\item \textbf{Comparison.} Classical SGD under PL gives only a
sub‑linear rate $O(1/T)$. The variance‑reduced estimator reduces the
gradient variance to zero at the snapshot, enabling the linear rate
(2), identical to that of full‑gradient descent but with a per‑iteration
cost of $\mathcal O(1)$ stochastic gradients.
\end{itemize}

\section*{Discussion}
The PL condition is weaker than strong convexity; it holds for many
non‑convex problems (e.g., over‑parameterized neural networks).  Under
(A1)–(A3) the algorithm enjoys the same linear convergence as gradient
descent on strongly convex functions while using only $m$ stochastic
gradient evaluations per epoch plus one full gradient.  The inner loop
length $m$ must be at least $2L/\mu$ to guarantee that the accumulated
progress outweighs the stochastic noise.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f_i$ is $L$‑smooth, then for any $x,y$,
\[
f_i(y)\;\le\;f_i(x)+\inner{\grad f_i(x)}{y-x}
+\frac{L}{2}\norm{y-x}^{2}.
\tag{L1}
\]
\end{lemma}
\begin{proof}
Standard consequence of the $L$‑smoothness definition. \hfill $\square$
\end{proof}

\begin{lemma}[Unbiasedness of SVRG estimator]\label{lem:unbiased}
For any inner iterate $x_t$ and snapshot $x^{s}$,
\[
\E_{i_t}[v_t \mid x_t]=\grad F(x_t).
\tag{L2}
\]
\end{lemma}
\begin{proof}
\[
\E_{i_t}[v_t]=\frac1n\sum_{i=1}^{n}\bigl(\grad f_i(x_t)-\grad f_i(x^{s})
+\widetilde{\mu}^{s}\bigr)
=\grad F(x_t)-\grad F(x^{s})+\widetilde{\mu}^{s}
=\grad F(x_t).
\] \hfill $\square$
\end{proof}

\begin{lemma}[Variance bound]\label{lem:var}
Conditioned on $x_t$, the variance of $v_t$ satisfies
\[
\E_{i_t}\!\bigl[\norm{v_t-\grad F(x_t)}^{2}\mid x_t\bigr]
\;\le\;
4L\bigl(F(x_t)-F^{\star}+F(x^{s})-F^{\star}\bigr).
\tag{L3}
\]
\end{lemma}
\begin{proof}
Using $L$‑smoothness and the PL inequality (A2):
\[
\begin{aligned}
\norm{v_t-\grad F(x_t)}^{2}
&=\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(x^{s})
        -\bigl(\grad F(x_t)-\grad F(x^{s})\bigr)}^{2}\\
&\le 2\norm{\grad f_{i_t}(x_t)-\grad f_{i_t}(x^{s})}^{2}
   +2\norm{\grad F(x_t)-\grad F(x^{s})}^{2}\\
&\le 4L^{2}\norm{x_t-x^{s}}^{2}      \qquad\text{(by $L$‑smoothness)}\\
&\le 8L\bigl(F(x_t)-F^{\star}+F(x^{s})-F^{\star}\bigr),
\end{aligned}
\]
where the last inequality follows from the PL condition
$\frac12\norm{\grad F(z)}^{2}\ge\mu(F(z)-F^{\star})$ and the
$L$‑smoothness bound $\norm{x-y}^{2}\le\frac{2}{\mu}\bigl(F(x)-F^{\star}+F(y)-F^{\star}\bigr)$.
Absorbing constants yields (L3). \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:linear}.  Throughout we condition on the snapshot
$x^{s}$ and denote $\widetilde{\mu}=\widetilde{\mu}^{s}$.

\noindent\textbf{Step 1. One inner update.}  
From the update $x_{t+1}=x_t-\eta v_t$,
\[
\begin{aligned}
\norm{x_{t+1}-x^{\star}}^{2}
&=\norm{x_t-x^{\star}}^{2}
-2\eta\inner{v_t}{x_t-x^{\star}}
+\eta^{2}\norm{v_t}^{2}                                     \tag{1}
\end{aligned}
\]
where $x^{\star}$ is a global minimizer of $F$.

\noindent\textbf{Step 2. Take expectation w.r.t. $i_t$.}  
Using Lemma~\ref{lem:unbiased} and Lemma~\ref{lem:var},
\[
\begin{aligned}
\E_{i_t}\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&=\norm{x_t-x^{\star}}^{2}
-2\eta\inner{\grad F(x_t)}{x_t-x^{\star}}
+\eta^{2}\E_{i_t}\!\bigl[\norm{v_t}^{2}\mid x_t\bigr]      \tag{2}
\end{aligned}
\]
and
\[
\E_{i_t}\!\bigl[\norm{v_t}^{2}\mid x_t\bigr]
= \norm{\grad F(x_t)}^{2}
   +\E_{i_t}\!\bigl[\norm{v_t-\grad F(x_t)}^{2}\mid x_t\bigr]
\le \norm{\grad F(x_t)}^{2}
   +4L\bigl(F(x_t)-F^{\star}+F(x^{s})-F^{\star}\bigr).       \tag{3}
\]

\noindent\textbf{Step 3. Use PL inequality to bound the inner product.}  
From (A2),
\[
\inner{\grad F(x_t)}{x_t-x^{\star}}
\;\ge\;
\frac{1}{2\mu}\norm{\grad F(x_t)}^{2}
    +\bigl(F(x_t)-F^{\star}\bigr).                         \tag{4}
\]

\noindent\textbf{Step 4. Substitute (3) and (4) into (2).}  
\[
\begin{aligned}
\E_{i_t}\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&\le \norm{x_t-x^{\star}}^{2}
-2\eta\!\left[\frac{1}{2\mu}\norm{\grad F(x_t)}^{2}
                +\bigl(F(x_t)-F^{\star}\bigr)\right]    \\
&\qquad
+\eta^{2}\!\left[\norm{\grad F(x_t)}^{2}
   +4L\bigl(F(x_t)-F^{\star}+F(x^{s})-F^{\star}\bigr)\right]. \tag{5}
\end{aligned}
\]

\noindent\textbf{Step 5. Collect the $\norm{\grad F(x_t)}^{2}$ terms.}  
Since $\eta\le \frac{1}{3L}\le\frac{1}{2\mu}$ (by (A3) and $\mu\le L$),
\[
-2\eta\frac{1}{2\mu}\norm{\grad F(x_t)}^{2}
+\eta^{2}\norm{\grad F(x_t)}^{2}
\le -\eta\frac{1}{\mu}\norm{\grad F(x_t)}^{2}
+\frac{\eta}{3L}\norm{\grad F(x_t)}^{2}
\le -\frac{\eta}{2\mu}\norm{\grad F(x_t)}^{2}\le 0.          \tag{6}
\]
Thus the gradient‑norm term can be dropped, yielding a simpler bound.

\noindent\textbf{Step 6. Remaining terms.}  
From (5)–(6),
\[
\begin{aligned}
\E_{i_t}\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
&\le \norm{x_t-x^{\star}}^{2}
-2\eta\bigl(F(x_t)-F^{\star}\bigr)
+4\eta^{2}L\bigl(F(x_t)-F^{\star}+F(x^{s})-F^{\star}\bigr) \\
&= \norm{x_t-x^{\star}}^{2}
-\underbrace{2\eta\bigl(1-2\eta L\bigr)}_{\alpha}
\bigl(F(x_t)-F^{\star}\bigr)
+4\eta^{2}L\bigl(F(x^{s})-F^{\star}\bigr).                \tag{7}
\end{aligned}
\]

\noindent\textbf{Step 7. Choose stepsize.}  
With $\eta\le\frac{1}{3L}$ we have $1-2\eta L\ge\frac13$, thus
$\alpha\ge \frac{2\eta}{3}$.
Hence
\[
\E_{i_t}\!\bigl[\norm{x_{t+1}-x^{\star}}^{2}\mid x_t\bigr]
\le \norm{x_t-x^{\star}}^{2}
-\frac{2\eta}{3}\bigl(F(x_t)-F^{\star}\bigr)
+4\eta^{2}L\bigl(F(x^{s})-F^{\star}\bigr).                \tag{8}
\]

\noindent\textbf{Step 8. Telescoping over inner loop.}  
Take total expectation and sum (8) for $t=0,\dots,m-1$.
Denote $x_0=x^{s}$ and $x_m=x^{s+1}$.  The telescoping sum gives
\[
\begin{aligned}
\E\!\bigl[\norm{x^{s+1}-x^{\star}}^{2}\bigr]
&\le \norm{x^{s}-x^{\star}}^{2}
-\frac{2\eta}{3}\sum_{t=0}^{m-1}\E\!\bigl[F(x_t)-F^{\star}\bigr]
+4\eta^{2}L m\,\E\!\bigl[F(x^{s})-F^{\star}\bigr].         \tag{9}
\end{aligned}
\]

\noindent\textbf{Step 9. Relate the average inner function value to the
snapshot value.}  
Because $F$ is $L$‑smooth,
\[
F(x_t)-F^{\star}\;\ge\;F(x^{s})-F^{\star}
-\frac{L}{2}\norm{x_t-x^{s}}^{2}.
\]
Summing over $t$ and using the bound $\norm{x_t-x^{s}}^{2}\le
\frac{2\eta}{\mu}\bigl(F(x^{s})-F^{\star}\bigr)$ (derived from the PL
condition and the same steps as above) yields
\[
\sum_{t=0}^{m-1}\E\!\bigl[F(x_t)-F^{\star}\bigr]
\;\ge\; m\,\E\!\bigl[F(x^{s})-F^{\star}\bigr]
-\frac{L\eta m^{2}}{\mu}\E\!\bigl[F(x^{s})-F^{\star}\bigr]. \tag{10}
\]

\noindent\textbf{Step 10. Plug (10) into (9).}  
After simplification,
\[
\E\!\bigl[F(x^{s+1})-F^{\star}\bigr]
\le\Bigl(1-\eta\mu\Bigr)\E\!\bigl[F(x^{s})-F^{\star}\bigr],
\tag{11}
\]
provided that the inner loop length satisfies $m\ge\frac{2L}{\mu}$
(which makes the negative term in (10) dominate the extra variance term
$4\eta^{2}Lm$).  Equation (11) is exactly the recursion (1) of the theorem.

\noindent\textbf{Step 11. Solve the recursion.}  
Iterating (11) over $s=0,\dots,S-1$ yields (2), establishing linear convergence.

\hfill $\square$

\section*{Rate Derivation (With Constants)}
From Theorem~\ref{thm:linear} and the choice $\eta=1/(3L)$,
\[
\rho = 1-\frac{\mu}{3L},\qquad
\E\!\bigl[F(x^{s})-F^{\star}\bigr]
\le \Bigl(1-\frac{\mu}{3L}\Bigr)^{s}\bigl[F(x^{0})-F^{\star}\bigr].
\]
Thus after $S$ outer epochs the optimality gap is reduced by a factor
$\exp(-\frac{\mu}{3L}S)$.  Since each epoch costs $n+m$ gradient
evaluations, the total stochastic gradient complexity to achieve
$\epsilon$‑accuracy is
\[
\mathcal O\!\Bigl(\bigl(n+m\bigr)\frac{L}{\mu}\log\frac{1}{\epsilon}\Bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{Strongly convex case.} If $F$ is $\mu$‑strongly convex then the
PL condition holds with the same $\mu$, and the above linear rate coincides
with the classical SVRG rate for strongly convex functions.
\item \textbf{Exact gradient ($m=1$).} When $m=1$ the algorithm reduces to
full‑gradient descent, and (1) becomes the standard GD contraction
$F(x^{s+1})-F^{\star}\le(1-\eta\mu)(F(x^{s})-F^{\star})$.
\item \textbf{Large minibatch.} Choosing $m=n$ makes the inner loop
identical to a full pass over the data; the variance term disappears and
the bound simplifies to the deterministic GD case.
\end{itemize}
