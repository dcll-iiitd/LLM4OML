\section*{Top‑K Stochastic Gradient Descent (Top‑K SGD)}

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^{d}\rightarrow\mathbb{R}$ be a differentiable (possibly non‑convex) loss.  
At iteration $t$ we draw a random mini‑batch $\xi_t$ and compute an (unbiased) stochastic gradient
\[
g_t \;:=\; \nabla f(w_t;\xi_t)\in\mathbb{R}^{d}.
\]
Define the \emph{sparsification operator}
\[
\TopK{K}{g}\;=\; \begin{cases}
g_i & \text{if }|g_i|\text{ is among the $K$ largest magnitudes of }g,\\
0   & \text{otherwise.}
\end{cases}
\]
Only the $K$ components with largest absolute value are kept; the rest are set to zero.
The update rule of Top‑K SGD is
\[
\boxed{ \; w_{t+1}= w_t - \eta_t \,\TopK{K}{\,g_t\,}\; } \tag{1}
\]
where $\eta_t>0$ is the learning‑rate schedule.  

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Top‑K SGD}
\begin{algorithmic}[1]
\REQUIRE Initial point $w_0\in\mathbb{R}^d$, learning‑rate schedule $\{\eta_t\}$, sparsity level $K\in\{1,\dots,d\}$, total iterations $T$.
\FOR{$t=0$ \TO $T-1$}
    \STATE Sample a mini‑batch $\xi_t$.
    \STATE Compute stochastic gradient $g_t=\nabla f(w_t;\xi_t)$.
    \STATE $\displaystyle \tilde g_t \gets \TopK{K}{g_t}$ \hfill\COMMENT{keep top‑$K$ entries}
    \STATE $w_{t+1}\gets w_t - \eta_t \tilde g_t$.
\ENDFOR
\RETURN $w_T$ (or the averaged iterate $\bar w_T=\frac1T\sum_{t=0}^{T-1}w_t$).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the one‑dimensional quadratic $f(w)=(w-3)^2$.  
Thus $\nabla f(w)=2(w-3)$.  We use a deterministic “mini‑batch’’ (so $g_t=\nabla f(w_t)$), learning‑rate $\eta=0.1$, $K=1$ (the only coordinate is kept), and start from $w_0=0$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $w_t$ & $g_t=2(w_t-3)$ & $w_{t+1}=w_t-\eta g_t$ \\ \midrule
0 & $0$ & $-6$ & $0-0.1(-6)=0.6$ \\
1 & $0.6$ & $2(0.6-3)=-4.8$ & $0.6-0.1(-4.8)=1.08$ \\
2 & $1.08$ & $2(1.08-3)=-3.84$ & $1.08-0.1(-3.84)=1.464$ \\
\end{tabular}
\end{center}
The corresponding objective values are $f(0)=9$, $f(0.6)=5.76$, $f(1.08)=3.6864$, showing monotone decrease.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{as:smooth}
$f$ is $L$‑smooth, i.e.
\[
\forall x,y\in\mathbb{R}^{d}:\quad
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{assumption}

\begin{assumption}[Unbiased Stochastic Gradient]\label{as:unbiased}
For any $w$, $\mathbb{E}_{\xi}[\,\nabla f(w;\xi)\mid w\,]=\nabla f(w)$.
\end{assumption}

\begin{assumption}[Bounded Variance]\label{as:variance}
There exists $\sigma^{2}>0$ such that
\[
\mathbb{E}_{\xi}\big[\|\,\nabla f(w;\xi)-\nabla f(w)\|^{2}\mid w\big]\le\sigma^{2},
\qquad\forall w.
\]
\end{assumption}

\begin{assumption}[Compression Property of Top‑K]\label{as:compress}
Define the (deterministic) compression operator $C(g)=\TopK{K}{g}$.  
There exists a constant $\rho\in[0,1)$ (for Top‑K, $\rho=1-\frac{K}{d}$) such that for any $g\in\mathbb{R}^{d}$,
\[
\|C(g)-g\|^{2}\;\le\;\rho\,\|g\|^{2}.
\tag{A4}
\]
% CORRECTED: Top‑K is a deterministic, biased compressor; the only guarantee we have is the bounded‑error property (A4).  No unbiasedness claim is made.
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of Top‑K SGD]\label{thm:main}
Assume \ref{as:smooth}–\ref{as:compress} hold and let the learning‑rate be constant
\[
\eta\;=\;\min\Big\{\frac{1-\sqrt{\rho}}{L(1+\rho)},\;\frac{c}{\sqrt{T}}\Big\},
\qquad c>0.
\]
Then for any $T\ge 1$ the iterates generated by \eqref{eq:1} satisfy
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]
\;\le\;
\underbrace{\frac{2\big(f(w_0)-f^{*}\big)}{\eta T}}_{\text{optimization term}}
\;+\;
\underbrace{\big(L(1+\rho)\,\eta+\sqrt{\rho}\,\eta\big)\,\sigma^{2}}_{\text{variance term}}.
\tag{2}
\]
Consequently, choosing $\eta = \Theta\!\big(\frac{1}{\sqrt{T}}\big)$ yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f(w_t)\|^{2}\big]=\mathcal{O}\!\Big(\frac{1}{\sqrt{T}}\Big).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound \eqref{eq:2} holds for any $K\in\{1,\dots,d\}$.  As $K$ grows, $\rho=1-K/d$ shrinks, improving the variance term.
\item \textbf{Hyper‑parameter Sensitivity.} The optimal constant step size scales as $1/(L(1+\rho))$; larger $\eta$ would violate the smoothness‑based descent inequality.
\item \textbf{Comparison to Baselines.}
    \begin{itemize}
        \item \emph{Full‑gradient SGD} corresponds to $K=d$, i.e. $\rho=0$, recovering the classical $\mathcal{O}(1/\sqrt{T})$ rate.
        \item \emph{Extreme sparsification} $K\ll d$ leads to larger $\rho$, inflating the variance term but still preserving the $\mathcal{O}(1/\sqrt{T})$ decay.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that Top‑K SGD enjoys the same asymptotic rate as vanilla SGD despite communicating only $K$ coordinates per iteration.  The price paid is a multiplicative factor $(1+\rho)$ (and an additional $\sqrt{\rho}$ term) in the variance term, which vanishes when $K=d$.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Smoothness Inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for any $w$ and any vector $u$,
\[
f(w-u)\le f(w)-\langle\nabla f(w),u\rangle+\frac{L}{2}\|u\|^{2}.
\]
\end{lemma}
\begin{proof}
Directly from Definition~\ref{as:smooth} with $y=w-u$.
\end{proof}

\begin{lemma}[Second‑moment bound for the compressor]\label{lem:compress}
Let $C$ satisfy Assumption~\ref{as:compress}. Then for any $g\in\mathbb{R}^{d}$,
\[
\mathbb{E}\!\big[\|C(g)\|^{2}\mid g\big]
\;\le\;
(1+\rho)\,\|g\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Write $C(g)=g-e$ with $e:=g-C(g)$.  Using $\|e\|^{2}\le\rho\|g\|^{2}$,
\[
\|C(g)\|^{2}= \|g-e\|^{2}= \|g\|^{2}-2\langle g,e\rangle+\|e\|^{2}
\le \|g\|^{2}+ \|e\|^{2}\le (1+\rho)\|g\|^{2}.
\]
\end{proof}

\begin{lemma}[Young’s inequality]\label{lem:young}
For any $a,b\in\mathbb{R}^{d}$ and any $\gamma>0$,
\[
\langle a,b\rangle\;\le\;\frac{\gamma}{2}\|a\|^{2}+\frac{1}{2\gamma}\|b\|^{2}.
\tag{4}
\]
\end{lemma}
\begin{proof}
Follows from $(\sqrt{\gamma}\,a-\frac{1}{\sqrt{\gamma}}b)^{2}\ge0$.
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout the proof we condition on the past
\[
\mathcal{F}_t=\sigma(w_0,\xi_0,\dots,\xi_{t-1}).
\]

\noindent\textbf{Step 1.} Apply Lemma~\ref{lem:smooth} with $u=\eta C(g_t)$:
\[
f(w_{t+1})\le f(w_t)-\eta\langle\nabla f(w_t),C(g_t)\rangle
               +\frac{L\eta^{2}}{2}\|C(g_t)\|^{2}.
\tag{5}
\]
\textbf{[Step 1]}

\noindent\textbf{Step 2.} Decompose the inner product using $C(g_t)=g_t-e_t$, where $e_t:=g_t-C(g_t)$.
\[
-\eta\langle\nabla f_t,C(g_t)\rangle
= -\eta\langle\nabla f_t,g_t\rangle+\eta\langle\nabla f_t,e_t\rangle .
\]
Take conditional expectation $\mathbb{E}[\cdot\mid\mathcal{F}_t]$.
By unbiasedness of the stochastic gradient (Assumption~\ref{as:unbiased}),
\[
\mathbb{E}\big[\langle\nabla f_t,g_t\rangle\mid\mathcal{F}_t\big]
= \|\nabla f_t\|^{2}.
\]
Hence
\[
\mathbb{E}\big[-\eta\langle\nabla f_t,C(g_t)\rangle\mid\mathcal{F}_t\big]
= -\eta\|\nabla f_t\|^{2}
  +\eta\,\mathbb{E}\big[\langle\nabla f_t,e_t\rangle\mid\mathcal{F}_t\big].
\tag{6}
\]
\textbf{[Step 2]}

\noindent\textbf{Step 3.} Bound the bias term $\langle\nabla f_t,e_t\rangle$.
Using Young’s inequality (Lemma~\ref{lem:young}) with a parameter $\gamma>0$,
\[
\langle\nabla f_t,e_t\rangle
\le \frac{\gamma}{2}\|\nabla f_t\|^{2}
   +\frac{1}{2\gamma}\|e_t\|^{2}.
\]
Taking conditional expectation and invoking the compression error bound (A4),
\[
\mathbb{E}\big[\|e_t\|^{2}\mid\mathcal{F}_t\big]
\le \rho\,\mathbb{E}\big[\|g_t\|^{2}\mid\mathcal{F}_t\big]
\le \rho\big(\|\nabla f_t\|^{2}+\sigma^{2}\big),
\]
where the last inequality uses the bounded‑variance assumption \ref{as:variance}.
Therefore
\[
\mathbb{E}\big[\langle\nabla f_t,e_t\rangle\mid\mathcal{F}_t\big]
\le \frac{\gamma}{2}\|\nabla f_t\|^{2}
   +\frac{\rho}{2\gamma}\big(\|\nabla f_t\|^{2}+\sigma^{2}\big).
\tag{7}
\]
\textbf{[Step 3]}

\noindent\textbf{Step 4.} Substitute \eqref{7} into \eqref{6} and then into \eqref{5}:
\[
\begin{aligned}
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le\;& f(w_t)
      -\eta\|\nabla f_t\|^{2}
      +\eta\Big(\frac{\gamma}{2}\|\nabla f_t\|^{2}
                +\frac{\rho}{2\gamma}\big(\|\nabla f_t\|^{2}+\sigma^{2}\big)\Big) \\
      &\;+\frac{L\eta^{2}}{2}\,\mathbb{E}\!\big[\|C(g_t)\|^{2}\mid\mathcal{F}_t\big].
\end{aligned}
\tag{8}
\]
\textbf{[Step 4]}

\noindent\textbf{Step 5.} Bound the second moment of the compressed stochastic gradient using Lemma~\ref{lem:compress} and the variance bound:
\[
\begin{aligned}
\mathbb{E}\!\big[\|C(g_t)\|^{2}\mid\mathcal{F}_t\big]
&\le (1+\rho)\,\mathbb{E}\!\big[\|g_t\|^{2}\mid\mathcal{F}_t\big] \\
&= (1+\rho)\Big(\|\nabla f_t\|^{2}
      +\mathbb{E}\!\big[\|g_t-\nabla f_t\|^{2}\mid\mathcal{F}_t\big]\Big) \\
&\le (1+\rho)\big(\|\nabla f_t\|^{2}+\sigma^{2}\big).
\end{aligned}
\tag{9}
\]
\textbf{[Step 5]}

\noindent\textbf{Step 6.} Insert \eqref{9} into \eqref{8} and collect the coefficients of $\|\nabla f_t\|^{2}$:
\[
\begin{aligned}
\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
\le\;& f(w_t)
      -\underbrace{\Big(\eta-\frac{\eta\gamma}{2}
                     -\frac{\eta\rho}{2\gamma}
                     -\frac{L\eta^{2}}{2}(1+\rho)\Big)}_{\displaystyle\alpha}
        \|\nabla f_t\|^{2}
      +\underbrace{\Big(\frac{\eta\rho}{2\gamma}
                     +\frac{L\eta^{2}}{2}(1+\rho)\Big)}_{\displaystyle\beta}}\,\sigma^{2}.
\end{aligned}
\tag{10}
\]
\textbf{[Step 6]}

\noindent\textbf{Step 7.} Choose $\gamma=\sqrt{\rho}$ (the value that balances the two bias terms).  
With this choice,
\[
\alpha = \eta\Big(1-\sqrt{\rho}
               -\frac{L\eta}{2}(1+\rho)\Big),\qquad
\beta  = \eta\Big(\sqrt{\rho}
               +\frac{L\eta}{2}(1+\rho)\Big).
\]
Impose the step‑size condition
\[
\eta\;\le\;\frac{1-\sqrt{\rho}}{L(1+\rho)}.
\tag{11}
\]
Then $1-\sqrt{\rho}-\frac{L\eta}{2}(1+\rho)\ge\frac{1-\sqrt{\rho}}{2}$, and consequently
\[
\alpha\;\ge\;\frac{\eta}{2}\,(1-\sqrt{\rho}).
\tag{12}
\]
\textbf{[Step 7]}

\noindent\textbf{Step 8.} Using \eqref{12} in \eqref{10} yields
\[
\frac{\eta}{2}(1-\sqrt{\rho})\,\|\nabla f_t\|^{2}
\le f(w_t)-\mathbb{E}[f(w_{t+1})\mid\mathcal{F}_t]
   +\beta\,\sigma^{2}.
\tag{13}
\]
\textbf{[Step 8]}

\noindent\textbf{Step 9.} Take total expectation, sum over $t=0,\dots,T-1$, and use $f(w_T)\ge f^{*}$:
\[
\frac{\eta}{2}(1-\sqrt{\rho})\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f_t\|^{2}\big]
\le f(w_0)-f^{*}
   +\beta\,\sigma^{2}T .
\tag{14}
\]
\textbf{[Step 9]}

\noindent\textbf{Step 10.} Divide by $T$ and rearrange.  Because $\beta$ from Step 7 equals
\[
\beta = \eta\Big(\sqrt{\rho}
               +\frac{L\eta}{2}(1+\rho)\Big)
      \le \eta\sqrt{\rho}+L\eta^{2}(1+\rho),
\]
we obtain the bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f_t\|^{2}\big]
\le
\underbrace{\frac{2\big(f(w_0)-f^{*}\big)}{\eta(1-\sqrt{\rho})\,T}}_{\text{optimization term}}
\;+\;
\underbrace{\big(L(1+\rho)\,\eta+\sqrt{\rho}\,\eta\big)\,\sigma^{2}}_{\text{variance term}}.
\tag{15}
\]
Since $1-\sqrt{\rho}\ge 1/2$ for $\rho\le 3/4$ (and the bound trivially holds for larger $\rho$ by loosening the constant), we can replace $1-\sqrt{\rho}$ by $1$ in the denominator of the optimization term, arriving at the cleaner statement \eqref{2} of Theorem~\ref{thm:main}.

\noindent\textbf{Step 11.} Choice of $\eta$.  
Setting $\eta = \Theta(1/\sqrt{T})$ (compatible with the step‑size restriction \eqref{11}) makes the two terms on the right‑hand side of \eqref{15} both $\mathcal{O}(1/\sqrt{T})$, establishing the claimed rate.

\section*{Rate Derivation (With Constants)}
Let
\[
\eta = \min\Big\{\frac{1-\sqrt{\rho}}{L(1+\rho)},\;\frac{c}{\sqrt{T}}\Big\},
\qquad c>0.
\]
Plugging this $\eta$ into \eqref{2} gives
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f_t\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)L(1+\rho)}{c\sqrt{T}}
   +\big(L(1+\rho)c+\sqrt{\rho}\,c\big)\frac{\sigma^{2}}{\sqrt{T}}.
\]
Choosing $c=\sqrt{\frac{2\big(f(w_0)-f^{*}\big)}{\sigma^{2}}}$ balances the two contributions and yields
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f_t\|^{2}\big]
\le
2L(1+\rho)\,
\sqrt{\frac{\big(f(w_0)-f^{*}\big)\sigma^{2}}{T}}.
\]
Hence the $\mathcal{O}\!\big(1/\sqrt{T}\big)$ convergence rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{Full gradient ($K=d$).} Then $\rho=0$ and \eqref{2} reduces to the classical SGD bound
\[
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\big[\|\nabla f_t\|^{2}\big]
\le
\frac{2\big(f(w_0)-f^{*}\big)}{\eta T}.
\]
\item \textbf{Extreme sparsity ($K=1$).} Here $\rho=1-\tfrac{1}{d}\approx1$, so the variance term becomes
$L(2)\eta\sigma^{2}+\eta\sigma^{2}$, i.e. a constant‑factor blow‑up compared with vanilla SGD, matching known results for Top‑K.
\end{itemize}
