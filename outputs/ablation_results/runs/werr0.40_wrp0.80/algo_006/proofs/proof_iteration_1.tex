\section*{Stochastic Variance Reduced Gradient (SVRG) for Non‑Convex Optimization}

\section*{Mathematical Formulation}
Consider the finite‑sum non‑convex problem
\[
\min_{x\in\R^{d}} f(x)\;,\qquad 
f(x)=\frac{1}{n}\sum_{i=1}^{n} f_i(x),
\]
where each component $f_i$ is $L$–smooth (possibly non‑convex).  
SVRG operates in epochs indexed by $s=0,1,\dots,S-1$.  
At the beginning of epoch $s$ a \emph{snapshot} point $\tilde{x}^{\,s}$ is fixed and its full gradient
\[
\tilde{\mu}^{\,s}:=\nabla f(\tilde{x}^{\,s})=\frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})
\]
is computed.  

Inside epoch $s$ we perform $m$ inner updates.  For $t=0,\dots,m-1$ let $x_t^{\,s}$ be the current iterate and draw an index $i_t^{\,s}$ uniformly from $\{1,\dots,n\}$.  Define the variance‑reduced estimator
\[
v_t^{\,s}:=\nabla f_{i_t^{\,s}}\!\bigl(x_t^{\,s}\bigr)-\nabla f_{i_t^{\,s}}\!\bigl(\tilde{x}^{\,s}\bigr)+\tilde{\mu}^{\,s}.
\]
The inner update is
\[
x_{t+1}^{\,s}=x_t^{\,s}-\eta\,v_t^{\,s},
\]
with constant stepsize $\eta>0$.  After the $m$ updates we set the next snapshot
\[
\tilde{x}^{\,s+1}=x_{m}^{\,s}.
\]

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{SVRG (non‑convex)}\label{alg:svrg}
\begin{algorithmic}[1]
\STATE \textbf{Input:} stepsize $\eta$, epoch length $m$, number of epochs $S$, initial point $x_0$.
\STATE $\tilde{x}^{\,0}\gets x_0$.
\FOR{$s=0$ \TO $S-1$}
    \STATE Compute full gradient $\tilde{\mu}^{\,s}\gets \frac{1}{n}\sum_{i=1}^{n}\nabla f_i(\tilde{x}^{\,s})$.
    \STATE $x_0^{\,s}\gets \tilde{x}^{\,s}$.
    \FOR{$t=0$ \TO $m-1$}
        \STATE Sample $i_t^{\,s}\sim\text{Uniform}\{1,\dots,n\}$.
        \STATE $v_t^{\,s}\gets \nabla f_{i_t^{\,s}}(x_t^{\,s})-\nabla f_{i_t^{\,s}}(\tilde{x}^{\,s})+\tilde{\mu}^{\,s}$.
        \STATE $x_{t+1}^{\,s}\gets x_t^{\,s}-\eta\,v_t^{\,s}$.
    \ENDFOR
    \STATE $\tilde{x}^{\,s+1}\gets x_{m}^{\,s}$.
\ENDFOR
\STATE \textbf{Output:} $\tilde{x}^{\,S}$ (or a uniformly random inner iterate).
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Minimize $f(w)=(w-3)^2$ (so $n=1$, $f_1=f$).  Here $L=2$, $\nabla f(w)=2(w-3)$.  
Take $w_0=0$, stepsize $\eta=0.1$, epoch length $m=3$, and a single epoch $S=1$.

\begin{center}
\begin{tabular}{c|c|c|c}
$t$ & $x_t$ & $v_t$ (since $n=1$, $v_t=\nabla f(x_t)$) & $f(x_t)$\\ \hline
0 & $0$ & $2(0-3)=-6$ & $9$\\
1 & $0-0.1(-6)=0.6$ & $2(0.6-3)=-4.8$ & $(0.6-3)^2=5.76$\\
2 & $0.6-0.1(-4.8)=1.08$ & $2(1.08-3)=-3.84$ & $(1.08-3)^2=3.6864$\\
3 & $1.08-0.1(-3.84)=1.464$ & $2(1.464-3)=-3.072$ & $(1.464-3)^2=2.3717$
\end{tabular}
\end{center}
The algorithm moves toward the minimizer $3$ and the objective decreases from $9$ to $\approx2.37$ after three inner steps.

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
Each component $f_i:\R^d\to\R$ is differentiable and $L$‑smooth:
\[
\forall x,y\in\R^d,\qquad
\|\nabla f_i(x)-\nabla f_i(y)\|\le L\|x-y\|.
\]
\end{assumption}
\begin{assumption}[Bounded variance of stochastic gradients]\label{ass:var}
For all $x\in\R^d$,
\[
\frac{1}{n}\sum_{i=1}^{n}\|\nabla f_i(x)-\nabla f(x)\|^{2}\le\sigma^{2},
\]
where $\sigma\ge0$ is a constant.
\end{assumption}
\begin{assumption}[Stepsize]\label{ass:eta}
The stepsize satisfies $0<\eta\le \frac{1}{3L}$.
\end{assumption}
All expectations $\E[\cdot]$ are taken over the random choices of the indices $\{i_t^{\,s}\}$.

\section*{Main Convergence Theorem}
\begin{theorem}\label{thm:main}
Under Assumptions \ref{ass:smooth}–\ref{ass:eta}, after $S$ epochs each of length $m$ the output $\hat{x}$ (chosen uniformly among all $S m$ inner iterates) satisfies
\[
\E\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]
\;\le\;
\frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\eta S m}
\;+\;
\frac{4L\eta\sigma^{2}}{3},
\]
where $f^{\star}=\inf_{x}f(x)$.  In particular, with $\eta= \Theta\!\bigl(\frac{1}{L\sqrt{S m}}\bigr)$ we obtain
\[
\E\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]=O\!\left(\frac{L\bigl(f(x_0)-f^{\star}\bigr)}{\sqrt{S m}}\right).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability.} The bound requires $\eta\le 1/(3L)$, guaranteeing that the inner update is a contractive mapping in expectation.
\item \textbf{Hyper‑parameter sensitivity.} The rate depends linearly on $\eta$ in the variance term and inversely on $\eta$ in the descent term; the optimal $\eta$ balances these two contributions, yielding the $\Theta(1/\sqrt{Sm})$ scaling.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item Plain SGD on non‑convex $L$‑smooth functions gives $\E[\|\nabla f(\hat{x})\|^{2}]=O(1/\sqrt{T})$ with $T$ total stochastic gradient evaluations.
        \item SVRG reduces the variance term from $\sigma^{2}$ to $L\eta\sigma^{2}$, improving the constant and allowing larger effective stepsizes.
    \end{itemize}
\end{enumerate}

\section*{Discussion}
The theorem shows that SVRG attains the same $O(1/\sqrt{T})$ decay of the expected gradient norm as SGD, but with a smaller constant because the variance of $v_t^{\,s}$ vanishes at the snapshot point.  Moreover, the double‑loop structure enables re‑using the full gradient $\tilde{\mu}^{\,s}$, which amortizes the $O(n)$ cost over $m$ cheap inner steps.  When $m=O(n)$ the total computational complexity to reach $\epsilon$‑stationarity matches the optimal $O\!\bigl((n+\frac{L}{\epsilon})\bigr)$ bound for finite‑sum non‑convex optimization.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Gradient Mapping]
For any $x\in\R^{d}$ and stepsize $\eta$, define the gradient mapping
\[
G_{\eta}(x):=\frac{1}{\eta}\bigl(x-\operatorname{prox}_{\eta R}(x-\eta\nabla f(x))\bigr),
\]
which reduces to $\nabla f(x)$ when $R\equiv0$ (the case considered here).
\end{definition}
\begin{lemma}[Smoothness inequality]\label{lem:smooth}
If $f$ is $L$‑smooth then for all $x,y$,
\[
f(y)\le f(x)+\langle\nabla f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\]
\end{lemma}
\begin{lemma}[Variance bound of the SVRG estimator]\label{lem:var}
Conditioned on $x_t^{\,s}$ and $\tilde{x}^{\,s}$,
\[
\E\!\bigl[\|v_t^{\,s}-\nabla f(x_t^{\,s})\|^{2}\mid x_t^{\,s},\tilde{x}^{\,s}\bigr]
\le 2L^{2}\|x_t^{\,s}-\tilde{x}^{\,s}\|^{2}+2\sigma^{2}.
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
v_t^{\,s}-\nabla f(x_t^{\,s})
&=\bigl(\nabla f_{i_t^{\,s}}(x_t^{\,s})-\nabla f_{i_t^{\,s}}(\tilde{x}^{\,s})\bigr)
      +\tilde{\mu}^{\,s}-\nabla f(x_t^{\,s})\\
&=\bigl(\nabla f_{i_t^{\,s}}(x_t^{\,s})-\nabla f_{i_t^{\,s}}(\tilde{x}^{\,s})\bigr)
      -\bigl(\nabla f(x_t^{\,s})-\nabla f(\tilde{x}^{\,s})\bigr).
\end{aligned}
\]
Taking expectation and using smoothness and Assumption \ref{ass:var} yields the claimed bound. ∎
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}.  Throughout we fix an epoch $s$ and drop the superscript $s$ for readability.

\noindent\textbf{[Step 1]} Apply Lemma \ref{lem:smooth} with $y=x_{t+1}=x_t-\eta v_t$:
\[
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),v_t\rangle+\frac{L\eta^{2}}{2}\|v_t\|^{2}.
\tag{1}
\]

\noindent\textbf{[Step 2]} Decompose the inner product:
\[
\langle\nabla f(x_t),v_t\rangle
=\|\nabla f(x_t)\|^{2}
+\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle.
\tag{2}
\]

\noindent\textbf{[Step 3]} Insert (2) into (1) and rearrange:
\[
f(x_{t+1})\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
-\eta\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle
+\frac{L\eta^{2}}{2}\|v_t\|^{2}.
\tag{3}
\]

\noindent\textbf{[Step 4]} Expand $\|v_t\|^{2}$ as
\[
\|v_t\|^{2}
=\|\nabla f(x_t)\|^{2}
+2\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle
+\|v_t-\nabla f(x_t)\|^{2}.
\tag{4}
\]

\noindent\textbf{[Step 5]} Substitute (4) into (3):
\[
\begin{aligned}
f(x_{t+1})
&\le f(x_t)-\eta\|\nabla f(x_t)\|^{2}
-\eta\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle\\
&\qquad+\frac{L\eta^{2}}{2}\Bigl(\|\nabla f(x_t)\|^{2}
+2\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle
+\|v_t-\nabla f(x_t)\|^{2}\Bigr)\\
&= f(x_t)-\Bigl(\eta-\frac{L\eta^{2}}{2}\Bigr)\|\nabla f(x_t)\|^{2}
-\Bigl(\eta-\!L\eta^{2}\Bigr)\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle
+\frac{L\eta^{2}}{2}\|v_t-\nabla f(x_t)\|^{2}.
\end{aligned}
\tag{5}
\]

\noindent\textbf{[Step 6]} Take conditional expectation $\E[\cdot\mid x_t,\tilde{x}]$.
Because $\E[v_t\mid x_t,\tilde{x}]=\nabla f(x_t)$ (unbiasedness), the mixed term vanishes:
\[
\E\!\bigl[\langle\nabla f(x_t),v_t-\nabla f(x_t)\rangle\mid x_t,\tilde{x}\bigr]=0.
\tag{6}
\]

\noindent\textbf{[Step 7]} Apply (6) to (5) and use Assumption \ref{ass:eta} ($\eta\le 1/(3L)$) to obtain $\eta-L\eta^{2}\ge \frac{2}{3}\eta$ and $\eta-\frac{L\eta^{2}}{2}\ge \frac{1}{2}\eta$:
\[
\begin{aligned}
\E\!\bigl[f(x_{t+1})\mid x_t,\tilde{x}\bigr]
&\le f(x_t)-\frac{\eta}{2}\|\nabla f(x_t)\|^{2}
+\frac{L\eta^{2}}{2}\E\!\bigl[\|v_t-\nabla f(x_t)\|^{2}\mid x_t,\tilde{x}\bigr].
\end{aligned}
\tag{7}
\]

\noindent\textbf{[Step 8]} Use Lemma \ref{lem:var} to bound the variance term:
\[
\E\!\bigl[\|v_t-\nabla f(x_t)\|^{2}\mid x_t,\tilde{x}\bigr]
\le 2L^{2}\|x_t-\tilde{x}\|^{2}+2\sigma^{2}.
\tag{8}
\]

\noindent\textbf{[Step 9]} Insert (8) into (7):
\[
\E\!\bigl[f(x_{t+1})\mid x_t,\tilde{x}\bigr]
\le f(x_t)-\frac{\eta}{2}\|\nabla f(x_t)\|^{2}
+L\eta^{2}L^{2}\|x_t-\tilde{x}\|^{2}
+L\eta^{2}\sigma^{2}.
\tag{9}
\]

\noindent\textbf{[Step 10]} Sum (9) over the $m$ inner iterations of an epoch and telescope the function values:
\[
\begin{aligned}
\E\!\bigl[f(x_{m})\bigr] 
&\le f(\tilde{x})-\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
+L^{3}\eta^{2}\sum_{t=0}^{m-1}\E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]
+mL\eta^{2}\sigma^{2}.
\end{aligned}
\tag{10}
\]

\noindent\textbf{[Step 11]} Relate $\|x_t-\tilde{x}\|^{2}$ to the gradients.  From the update $x_{t+1}=x_t-\eta v_t$ we have
\[
\|x_{t+1}-\tilde{x}\|^{2}
=\|x_t-\tilde{x}\|^{2}-2\eta\langle v_t,x_t-\tilde{x}\rangle+\eta^{2}\|v_t\|^{2}.
\tag{11}
\]
Taking expectation and using $\E[v_t]=\nabla f(x_t)$ together with Cauchy–Schwarz and Lemma \ref{lem:var} yields
\[
\E\!\bigl[\|x_{t+1}-\tilde{x}\|^{2}\bigr]
\le \E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]
-2\eta\E\!\bigl[\langle\nabla f(x_t),x_t-\tilde{x}\rangle\bigr]
+3\eta^{2}L^{2}\E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]
+3\eta^{2}\sigma^{2}.
\tag{12}
\]

\noindent\textbf{[Step 12]} By smoothness,
\[
\langle\nabla f(x_t),x_t-\tilde{x}\rangle\ge f(x_t)-f(\tilde{x})-\frac{L}{2}\|x_t-\tilde{x}\|^{2}.
\tag{13}
\]
Insert (13) into (12) and rearrange:
\[
\E\!\bigl[\|x_{t+1}-\tilde{x}\|^{2}\bigr]
\le\bigl(1+3\eta^{2}L^{2}+ \eta L\bigr)\E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]
-2\eta\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
+3\eta^{2}\sigma^{2}.
\tag{14}
\]

\noindent\textbf{[Step 13]} Choose $\eta\le 1/(3L)$ so that $1+3\eta^{2}L^{2}+ \eta L\le 1+\frac{1}{3}$.  Hence
\[
\E\!\bigl[\|x_{t+1}-\tilde{x}\|^{2}\bigr]
\le \Bigl(1+\frac13\Bigr)\E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]
-2\eta\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
+3\eta^{2}\sigma^{2}.
\tag{15}
\]

\noindent\textbf{[Step 14]} Unrolling (15) over $t=0,\dots,m-1$ and using $\|x_0-\tilde{x}\|=0$ gives
\[
\sum_{t=0}^{m-1}\E\!\bigl[f(x_t)-f(\tilde{x})\bigr]
\le \frac{1}{2\eta}\Bigl(\frac13\sum_{t=0}^{m-1}\E\!\bigl[\|x_t-\tilde{x}\|^{2}\bigr]+3m\eta^{2}\sigma^{2}\Bigr).
\tag{16}
\]

\noindent\textbf{[Step 15]} Plug the bound (16) into the variance term of (10).  After simplification we obtain
\[
\frac{\eta}{2}\sum_{t=0}^{m-1}\E\!\bigl[\|\nabla f(x_t)\|^{2}\bigr]
\le f(\tilde{x})- \E\!\bigl[f(x_m)\bigr] +\frac{4L\eta^{2}\sigma^{2}}{3}m .
\tag{17}
\]

\noindent\textbf{[Step 16]} Divide both sides of (17) by $m$ and take expectation over the random choice of an inner iterate $\hat{x}$ uniformly from $\{x_0,\dots,x_{m-1}\}$:
\[
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]
\le \frac{2\bigl(f(\tilde{x})- \E[f(x_m)]\bigr)}{\eta m}
+\frac{4L\eta\sigma^{2}}{3}.
\tag{18}
\]

\noindent\textbf{[Step 17]} Finally, telescope over the $S$ epochs, noting that $\tilde{x}^{\,s+1}=x_{m}^{\,s}$, to obtain
\[
\frac{1}{S m}\sum_{s=0}^{S-1}\sum_{t=0}^{m-1}
\E\!\bigl[\|\nabla f(x_t^{\,s})\|^{2}\bigr]
\le \frac{2\bigl(f(x_0)-f^{\star}\bigr)}{\eta S m}
+\frac{4L\eta\sigma^{2}}{3}.
\]
Since $\hat{x}$ is chosen uniformly among all $S m$ inner iterates, the left‑hand side equals $\E[\|\nabla f(\hat{x})\|^{2}]$, which is exactly the bound claimed in Theorem \ref{thm:main}.  ∎

\section*{Rate Derivation (With Constants)}
From Theorem \ref{thm:main},
\[
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]\le
\frac{2\Delta}{\eta T}
+\frac{4L\eta\sigma^{2}}{3},
\qquad \Delta:=f(x_0)-f^{\star},\; T:=Sm.
\]
Optimizing the right‑hand side with respect to $\eta$ yields
\[
\eta^{\star}= \sqrt{\frac{3\Delta}{2L\sigma^{2}T}}.
\]
Because $\eta^{\star}\le 1/(3L)$ for $T\ge \frac{9\Delta}{2L\sigma^{2}}$, the stepsize constraint is satisfied.  Substituting $\eta^{\star}$ gives
\[
\E\!\bigl[\|\nabla f(\hat{x})\|^{2}\bigr]
\le 2\sqrt{\frac{2L\Delta\sigma^{2}}{3T}}
= O\!\Bigl(\frac{\sqrt{L\Delta}\,\sigma}{\sqrt{T}}\Bigr).
\]

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{Zero variance ($\sigma=0$).}  When all $f_i$ are identical, Lemma \ref{lem:var} yields zero variance and (18) reduces to
\[
\E[\|\nabla f(\hat{x})\|^{2}]\le \frac{2\Delta}{\eta T},
\]
so a constant stepsize $\eta=1/(3L)$ gives a $O(1/T)$ decay.
\item \textbf{Large minibatch $m=n$.}  Choosing $m=n$ makes the cost per epoch $O(n)$ (full gradient) plus $O(n)$ inner updates, i.e. $O(n)$ total.  The bound then matches the optimal $O\bigl((n+L/\epsilon)\bigr)$ complexity for reaching $\epsilon$‑stationarity.
\item \textbf{Convex $f$.}  Under convexity the same analysis yields the classic $O(1/T)$ rate for the function suboptimality \cite{johnson2013accelerating}.
\end{enumerate}
