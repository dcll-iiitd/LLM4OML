\section*{Proximal Gradient Method for Non‑Convex Smooth $f$}

\section*{Mathematical Formulation}
Let $f:\R^{d}\rightarrow\R$ be a continuously differentiable (possibly non‑convex) function and let $g:\R^{d}\rightarrow\R\cup\{+\infty\}$ be a proper, lower‑semicontinuous (not necessarily convex) regulariser.  
Define the composite objective
\[
F(x)\;:=\;f(x)+g(x),\qquad x\in\R^{d}.
\]

For a stepsize $\eta>0$, the \emph{proximal operator} of $g$ is
\[
\prox_{\eta g}(v)\;:=\;\arg\min_{u\in\R^{d}}\Big\{g(u)+\frac{1}{2\eta}\|u-v\|^{2}\Big\}.
\]

The algorithm generates a sequence $\{x_{k}\}_{k\ge 0}$ by
\begin{equation}
\boxed{\,x_{k+1}\;=\;\prox_{\eta g}\bigl(x_{k}-\eta\,\grad f(x_{k})\bigr)\,},\qquad k=0,1,\dots
\tag{1}
\end{equation}
The quantity
\[
G_{\eta}(x):=\frac{1}{\eta}\bigl(x-\prox_{\eta g}(x-\eta\grad f(x))\bigr)
\]
is called the \emph{gradient mapping}; $G_{\eta}(x)=0$ iff $0\in\grad f(x)+\partial g(x)$, i.e. $x$ is a (first‑order) critical point of $F$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Proximal Gradient Method}
\begin{algorithmic}[1]
\State \textbf{Input:} stepsize $\eta\in(0,1/L)$, initial point $x_{0}\in\R^{d}$
\For{$k=0,1,2,\dots$}
    \State $v_{k}\gets x_{k}-\eta\,\grad f(x_{k})$ \Comment{gradient step}
    \State $x_{k+1}\gets \prox_{\eta g}(v_{k})$ \Comment{proximal step}
    \If{stopping criterion satisfied}
        \State \textbf{break}
    \EndIf
\EndFor
\State \textbf{Output:} $x_{K}$ for the last iterate $K$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  
\[
f(w)=(w-3)^{2},\qquad g\equiv0,\qquad \eta=0.1,\qquad w_{0}=0.
\]
Because $g\equiv0$, $\prox_{\eta g}(v)=v$. The iteration (1) reduces to the ordinary gradient descent step  
$w_{k+1}=w_{k}-\eta\,\grad f(w_{k})$.

\begin{enumerate}[label=Iteration \arabic*:, leftmargin=*]
\item $k=0$  
\[
\grad f(w_{0})=2(w_{0}-3)=-6,\qquad
v_{0}=w_{0}-\eta\grad f(w_{0})=0-0.1(-6)=0.6,
\]
\[
w_{1}= \prox_{0.1\cdot 0}(v_{0})=v_{0}=0.6,\qquad
F(w_{1})=f(0.6)=(0.6-3)^{2}=5.76.
\]

\item $k=1$  
\[
\grad f(w_{1})=2(0.6-3)=-4.8,\qquad
v_{1}=0.6-0.1(-4.8)=1.08,
\]
\[
w_{2}=v_{1}=1.08,\qquad
F(w_{2})=f(1.08)=(1.08-3)^{2}=3.6864.
\]

\item $k=2$  
\[
\grad f(w_{2})=2(1.08-3)=-3.84,\qquad
v_{2}=1.08-0.1(-3.84)=1.464,
\]
\[
w_{3}=v_{2}=1.464,\qquad
F(w_{3})=f(1.464)=(1.464-3)^{2}=2.366.
\]
\end{enumerate}
The objective value strictly decreases, illustrating the descent property proved later.

\section*{Assumptions}
\begin{assumption}[Smoothness of $f$]\label{asm:smooth}
$f$ is $L$‑smooth, i.e.
\[
\|\grad f(x)-\grad f(y)\|\le L\|x-y\|,\qquad\forall x,y\in\R^{d}.
\tag{A1}
\]
\end{assumption}
\begin{assumption}[Step size]\label{asm:step}
The stepsize satisfies
\[
0<\eta<\frac{1}{L}.
\tag{A2}
\]
\end{assumption}
\begin{assumption}[Proximal regulariser]\label{asm:prox}
$g$ is proper, lower‑semicontinuous and its proximal operator $\prox_{\eta g}$ is \emph{well defined} for the chosen $\eta$.
\tag{A3}
\end{assumption}
\begin{assumption}[Bounded below]\label{asm:bounded}
The composite objective $F=f+g$ is bounded below:
\[
F_{\inf}:=\inf_{x\in\R^{d}}F(x)>-\infty .
\tag{A4}
\]
\end{assumption}

\section*{Main Convergence Theorem}
\begin{theorem}[Convergence of the gradient mapping]\label{thm:main}
Let $\{x_{k}\}$ be generated by \eqref{1} under Assumptions \ref{asm:smooth}–\ref{asm:bounded} with a stepsize $\eta$ satisfying \ref{asm:step}. Then for every $K\ge1$,
\begin{equation}
\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta}(x_{k})\|^{2}
\;\le\;
\frac{2}{\eta}\,\frac{F(x_{0})-F_{\inf}}{K}.
\tag{2}
\end{equation}
Consequently $\displaystyle\lim_{k\to\infty}\|G_{\eta}(x_{k})\|=0$ and every limit point of $\{x_{k}\}$ is a critical point of $F$ (i.e. $0\in\grad f(x)+\partial g(x)$).
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Descent.} The sequence $\{F(x_{k})\}$ is non‑increasing.
\item \textbf{Rate.} The bound \eqref{2} yields an $O(1/K)$ convergence rate for the squared norm of the gradient mapping, which is the standard rate for first‑order methods in the non‑convex setting.
\item \textbf{Hyperparameter sensitivity.} The stepsize must respect $\eta<1/L$; larger $\eta$ may break the descent lemma (see proof). Within the admissible interval the bound scales as $1/\eta$, so a larger $\eta$ (still $<1/L$) gives a tighter constant.
\item \textbf{Comparison.} When $g\equiv0$, $G_{\eta}(x)=\grad f(x)$ and Theorem \ref{thm:main} reduces to the classical $O(1/K)$ guarantee for gradient descent on smooth non‑convex functions. For convex $g$, the same proof recovers the well‑known proximal gradient convergence results.
\end{enumerate}

\section*{Discussion}
The theorem hinges on two elementary properties:
\begin{itemize}
\item \emph{Descent lemma} for $L$‑smooth $f$,
\item \emph{Non‑expansiveness} of the proximal operator:
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|,\qquad\forall u,v\in\R^{d}.
\tag{NE}
\]
No convexity of $f$ or $g$ is required beyond the existence of the proximal step. The result therefore applies to a large class of regularised non‑convex problems (e.g. neural‑network training with $\ell_{1}$ or indicator constraints).

\section*{Preliminaries (Definitions and Lemmas)}
\begin{definition}[Gradient mapping]
For any $\eta>0$, define
\[
G_{\eta}(x):=\frac{1}{\eta}\bigl(x-\prox_{\eta g}(x-\eta\grad f(x))\bigr).
\]
\end{definition}
\begin{lemma}[Descent Lemma]\label{lem:descent}
If $f$ is $L$‑smooth, then for any $x,y\in\R^{d}$,
\[
f(y)\le f(x)+\langle\grad f(x),y-x\rangle+\frac{L}{2}\|y-x\|^{2}.
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard; follows from integrating the Lipschitz condition on $\grad f$. \hfill $\square$
\end{proof}
\begin{lemma}[Optimality condition of the proximal step]\label{lem:prox_opt}
Let $u=\prox_{\eta g}(v)$. Then
\[
0\in\frac{1}{\eta}(u-v)+\partial g(u).
\tag{4}
\]
Equivalently,
\[
v-u\in\eta\,\partial g(u).
\]
\end{lemma}
\begin{proof}
By definition $u$ minimises $g(z)+\frac{1}{2\eta}\|z-v\|^{2}$; the first‑order optimality condition yields \eqref{4}. \hfill $\square$
\end{proof}
\begin{lemma}[Non‑expansiveness of $\prox$]\label{lem:nonexp}
For any $u,v\in\R^{d}$,
\[
\|\prox_{\eta g}(u)-\prox_{\eta g}(v)\|\le\|u-v\|.
\tag{5}
\]
\end{lemma}
\begin{proof}
See e.g. \cite{parikh2014proximal}. The proof uses the monotonicity of the sub‑differential of $g$. \hfill $\square$
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem \ref{thm:main}. All steps are numbered for easy reference.

\medskip
\noindent\textbf{Step 1 (Notation).} Denote $v_{k}=x_{k}-\eta\grad f(x_{k})$ and $x_{k+1}= \prox_{\eta g}(v_{k})$ as in \eqref{1}. By Lemma \ref{lem:prox_opt},
\begin{equation}
\frac{1}{\eta}(x_{k+1}-v_{k})\in\partial g(x_{k+1}).
\tag{6}
\end{equation}

\noindent\textbf{Step 2 (Optimality of the proximal step).} From \eqref{6} we obtain
\[
0\in\grad f(x_{k})+\frac{1}{\eta}(x_{k+1}-x_{k})+\partial g(x_{k+1}).
\tag{7}
\]
Thus
\[
-\frac{1}{\eta}(x_{k+1}-x_{k})\in \grad f(x_{k})+\partial g(x_{k+1}).
\tag{8}
\]

\noindent\textbf{Step 3 (Descent of $F$).} Using Lemma \ref{lem:descent} with $x=x_{k}$ and $y=x_{k+1}$,
\[
f(x_{k+1})\le f(x_{k})+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{9}
\]
Add $g(x_{k+1})$ on both sides and use the definition of $F$:
\[
F(x_{k+1})\le F(x_{k})+\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle+\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}.
\tag{10}
\]

\noindent\textbf{Step 4 (Bounding the inner product).} From \eqref{8},
\[
\langle\grad f(x_{k}),x_{k+1}-x_{k}\rangle
= -\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
    +\langle s_{k},x_{k+1}-x_{k}\rangle,
\tag{11}
\]
where $s_{k}\in\partial g(x_{k+1})$. By the sub‑gradient inequality,
\[
g(x_{k+1})\le g(x_{k})+\langle s_{k},x_{k+1}-x_{k}\rangle.
\tag{12}
\]

\noindent\textbf{Step 5 (Combining 10–12).} Substitute \eqref{11} into \eqref{10} and drop the non‑negative term $g(x_{k})-g(x_{k+1})$ using \eqref{12}:
\[
\begin{aligned}
F(x_{k+1})
&\le F(x_{k})-\frac{1}{\eta}\|x_{k+1}-x_{k}\|^{2}
    +\frac{L}{2}\|x_{k+1}-x_{k}\|^{2}\\
&=F(x_{k})-\Big(\frac{1}{\eta}-\frac{L}{2}\Big)\|x_{k+1}-x_{k}\|^{2}.
\end{aligned}
\tag{13}
\]

\noindent\textbf{Step 6 (Descent coefficient).} By Assumption \ref{asm:step}, $\eta<1/L$ implies $\frac{1}{\eta}-\frac{L}{2}> \frac{1}{2\eta}$. Hence
\[
F(x_{k})-F(x_{k+1})\;\ge\;\frac{1}{2\eta}\|x_{k+1}-x_{k}\|^{2}.
\tag{14}
\]

\noindent\textbf{Step 7 (Relating $\|x_{k+1}-x_{k}\|$ to the gradient mapping).} By definition of $G_{\eta}$,
\[
G_{\eta}(x_{k})=\frac{1}{\eta}\bigl(x_{k}-x_{k+1}\bigr)
\quad\Longrightarrow\quad
\|x_{k+1}-x_{k}\|^{2}= \eta^{2}\|G_{\eta}(x_{k})\|^{2}.
\tag{15}
\]

\noindent\textbf{Step 8 (Key inequality).} Plug \eqref{15} into \eqref{14}:
\[
F(x_{k})-F(x_{k+1})\;\ge\;\frac{\eta}{2}\,\|G_{\eta}(x_{k})\|^{2}.
\tag{16}
\]

\noindent\textbf{Step 9 (Summation).} Sum \eqref{16} for $k=0,\dots,K-1$ and telescope:
\[
\sum_{k=0}^{K-1}\|G_{\eta}(x_{k})\|^{2}
\;\le\;\frac{2}{\eta}\,\bigl(F(x_{0})-F(x_{K})\bigr)
\;\le\;\frac{2}{\eta}\,\bigl(F(x_{0})-F_{\inf}\bigr).
\tag{17}
\]

\noindent\textbf{Step 10 (Averaging).} Divide \eqref{17} by $K$ to obtain exactly the bound \eqref{2} of Theorem \ref{thm:main}. \hfill $\square$

\section*{Rate Derivation (With Constants)}
From \eqref{2},
\[
\min_{0\le k\le K-1}\|G_{\eta}(x_{k})\|^{2}
\;\le\;\frac{1}{K}\sum_{k=0}^{K-1}\|G_{\eta}(x_{k})\|^{2}
\;\le\;\frac{2}{\eta}\,\frac{F(x_{0})-F_{\inf}}{K}.
\]
Thus the smallest gradient‑mapping norm among the first $K$ iterates decays as $O(1/K)$. Setting $K\ge \frac{2(F(x_{0})-F_{\inf})}{\eta\varepsilon^{2}}$ guarantees $\|G_{\eta}(x_{k})\|\le\varepsilon$ for some $k<K$.

\section*{Special Cases}
\begin{enumerate}[label=\alph*)]
\item \textbf{No regulariser ($g\equiv0$).} Then $\prox_{\eta g}$ is the identity, $G_{\eta}(x)=\grad f(x)$, and \eqref{2} reduces to the classic $O(1/K)$ bound for gradient descent on smooth non‑convex $f$.
\item \textbf{Convex $g$.} The same proof holds; in addition, one can show that the whole sequence $\{x_{k}\}$ converges to a minimiser when $F$ is also convex.
\item \textbf{Exact line‑search step.} If $\eta$ is chosen adaptively such that $F(x_{k+1})\le F(x_{k})-\frac{c}{\eta}\|x_{k+1}-x_{k}\|^{2}$ for some $c\in(0,1)$, the constant $\frac{2}{\eta}$ in \eqref{2} improves to $\frac{2}{c\eta}$, yielding a tighter rate.
\end{enumerate}
