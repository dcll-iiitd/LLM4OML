\section*{Primal–Dual Hybrid Gradient (PDHG) Algorithm}

\section*{Mathematical Formulation}
Let  

\[
\min_{x\in\mathbb R^{n}} \; f(x)+g(Kx) ,
\tag{P}
\]

where  

\begin{itemize}
\item $f:\mathbb R^{n}\rightarrow\mathbb R\cup\{+\infty\}$ is proper, closed and convex,
\item $g:\mathbb R^{m}\rightarrow\mathbb R\cup\{+\infty\}$ is proper, closed and convex,
\item $K\in\mathbb R^{m\times n}$ is a linear operator.
\end{itemize}

The Fenchel–Rockafellar dual of \((P)\) is  

\[
\max_{y\in\mathbb R^{m}} \; -f^{*}(-K^{\top}y)-g^{*}(y) .
\tag{D}
\]

A saddle point \((x^{\star},y^{\star})\) satisfies  

\[
\mathcal L(x^{\star},y)=f(x^{\star})+g(Kx^{\star})+\ip{y}{Kx^{\star}}-g^{*}(y)
\ge 
\mathcal L(x^{\star},y^{\star})
\ge 
\mathcal L(x,y^{\star})\quad\forall x,y .
\]

Define the proximal operators  

\[
\prox_{\tau f}(v)=\arg\min_{x}\Big\{ f(x)+\tfrac{1}{2\tau}\norm{x-v}^{2}\Big\},
\qquad
\prox_{\sigma g^{*}}(w)=\arg\min_{y}\Big\{ g^{*}(y)+\tfrac{1}{2\sigma}\norm{y-w}^{2}\Big\}.
\]

Given step sizes \(\tau>0,\sigma>0\) and an extrapolation parameter \(\theta\in[0,1]\), the PDHG iterates are  

\[
\boxed{
\begin{aligned}
y_{k+1} &=\prox_{\sigma g^{*}}\!\big( y_{k}+ \sigma K\bar x_{k}\big),\\[4pt]
x_{k+1} &=\prox_{\tau f}\!\big( x_{k}- \tau K^{\top} y_{k+1}\big),\\[4pt]
\bar x_{k+1} &= x_{k+1}+ \theta\,(x_{k+1}-x_{k}) .
\end{aligned}}
\tag{1}
\]

The algorithm keeps the dual variable \(y_k\) and a ``look‑ahead’’ primal variable \(\bar x_k\).

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Primal–Dual Hybrid Gradient (PDHG)}\label{alg:pdhg}
\begin{algorithmic}[1]
\Require Convex functions $f,g$, linear operator $K$, step sizes $\tau,\sigma>0$, extrapolation $\theta\in[0,1]$, initial points $x_0$, $y_0$, set $\bar x_0\gets x_0$.
\Repeat{stopping criterion satisfied}
\State $y_{k+1}\gets\prox_{\sigma g^{*}}\!\big( y_{k}+ \sigma K\bar x_{k}\big)$ \Comment{dual ascent}
\State $x_{k+1}\gets\prox_{\tau f}\!\big( x_{k}- \tau K^{\top} y_{k+1}\big)$ \Comment{primal descent}
\State $\bar x_{k+1}\gets x_{k+1}+ \theta\,(x_{k+1}-x_{k})$ \Comment{extrapolation}
\State $k\gets k+1$
\EndRepeat
\State \Return $x_k$, $y_k$
\end{algorithmic}
\end{algorithm}

\section*{Dry Run Example}
Consider the scalar problem  

\[
\min_{w\in\mathbb R}\;(w-3)^{2},
\]

which can be written as \(\min_{w} f(w)\) with \(f(w)=(w-3)^{2}\).  
We set \(K=1\), \(g\equiv0\) (hence \(g^{*}\equiv0\) and \(\prox_{\sigma g^{*}}=\mathrm{Id}\)), \(\tau=\sigma=0.1\), \(\theta=1\), and initialise \(x_{0}=w_{0}=0\), \(y_{0}=0\), \(\bar x_{0}=0\).

\begin{center}
\begin{tabular}{c|c|c|c|c}
Iter & $y_{k+1}$ & $x_{k+1}$ & $\bar x_{k+1}$ & $f(x_{k+1})$\\\hline
0 & $0+\sigma\bar x_{0}=0$ & $\prox_{0.1 f}(0-0)=\arg\min_{x}\{(x-3)^{2}+5x^{2}\}=0.4762$ & $0.4762+1\cdot(0.4762-0)=0.4762$ & $ (0.4762-3)^{2}=6.37$\\\hline
1 & $0+0.1\cdot0.4762=0.04762$ & $\prox_{0.1 f}(0.4762-0.1\cdot0.04762)=\arg\min_{x}\{(x-3)^{2}+5(x-0.4714)^{2}\}=0.8960$ & $0.8960+1\cdot(0.8960-0.4762)=1.3158$ & $(0.8960-3)^{2}=4.42$\\\hline
2 & $0.04762+0.1\cdot1.3158=0.17920$ & $\prox_{0.1 f}(0.8960-0.1\cdot0.17920)=\arg\min_{x}\{(x-3)^{2}+5(x-0.8781)^{2}\}=1.2591$ & $1.2591+1\cdot(1.2591-0.8960)=1.6222$ & $(1.2591-3)^{2}=3.02$
\end{tabular}
\end{center}

The objective value decreases from $9$ (at $w=0$) to $3.02$ after three iterations, illustrating the descent behaviour of PDHG.

\section*{Assumptions}
\begin{enumerate}
\item \textbf{Convexity and regularity.}  $f$ and $g$ are proper, closed, convex. Their proximal operators are easy to evaluate.
\item \textbf{Lipschitz of the linear operator.}  Denote $\|K\|:=\sup_{\|x\|=1}\|Kx\|$ (spectral norm). 
\item \textbf{Step‑size condition.}  Choose $\tau,\sigma>0$ such that  

\[
\tau\sigma\|K\|^{2}<1 .
\tag{A1}
\]

\item \textbf{Existence of a saddle point.}  There exists \((x^{\star},y^{\star})\) satisfying the saddle‑point conditions of $\mathcal L$.
\item \textbf{Boundedness of iterates (optional).}  When $f$ or $g$ are coercive, the generated sequence $\{(x_k,y_k)\}$ is bounded; this is used only to guarantee existence of limits in the proof.
\end{enumerate}

\section*{Main Convergence Theorem}
\begin{theorem}[Ergodic $O(1/k)$ convergence]\label{thm:main}
Under the assumptions above, let the iterates be generated by \eqref{1} with $\theta=1$. Define the ergodic averages  

\[
\bar x^{(N)}:=\frac{1}{N}\sum_{k=1}^{N}x_{k},\qquad 
\bar y^{(N)}:=\frac{1}{N}\sum_{k=1}^{N}y_{k}.
\]

Then for any saddle point \((x^{\star},y^{\star})$ we have  

\[
\boxed{
\mathcal G_{N}:=
\sup_{y}\bigl\{\mathcal L(\bar x^{(N)},y)-\mathcal L(x^{\star},\bar y^{(N)})\bigr\}
\le 
\frac{C}{N},
}
\tag{2}
\]

where  

\[
C:=\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
      +\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2},
\]

i.e. the primal–dual gap decays at rate $O(1/N)$. 
\end{theorem}

\section*{Theoretical Properties}
\begin{itemize}
\item \textbf{Stability.}  Condition \textbf{(A1)} is both necessary and sufficient for the non‑expansiveness of the combined proximal operator; violating it may cause divergence.
\item \textbf{Hyper‑parameter sensitivity.}  The bound $C$ scales linearly with $1/\tau$ and $1/\sigma$, therefore larger step sizes (still respecting \textbf{(A1)}) give a smaller constant and faster practical convergence.
\item \textbf{Comparison to baselines.}
    \begin{itemize}
        \item For pure gradient descent on $f$, the optimal $O(1/k)$ rate requires Lipschitz gradient; PDHG attains the same rate without smoothness, at the cost of an extra dual update.
        \item When $g\equiv0$, PDHG reduces to the proximal point method, whose rate is also $O(1/k)$ but with a larger constant.
    \end{itemize}
\item \textbf{Special cases.}
    \begin{enumerate}
        \item If $K=I$ and $g^{*}=0$, the algorithm becomes the classic Chambolle–Pock scheme for $f+g$ with $g$ simple.
        \item When $f$ is strongly convex, the rate improves to $O(1/k^{2})$ (see e.g. \cite{chambolle2011first}).
    \end{enumerate}
\end{itemize}

\section*{Discussion}
The theorem shows that PDHG enjoys the same ergodic convergence order as the classical primal or dual gradient methods, yet it can handle non‑smooth terms via proximal mappings. The proof relies only on convexity and the step‑size condition, making it applicable to a wide class of saddle‑point problems arising in imaging, machine learning and signal processing.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Three‑point identity for proximal operator]\label{lem:prox}
For any closed convex $h$ and any $u,v\in\mathbb R^{d}$,
\[
\prox_{\lambda h}(u)=v \;\Longrightarrow\;
\langle u-\prox_{\lambda h}(u),\,z-\prox_{\lambda h}(u)\rangle
\le \lambda\bigl(h(z)-h(v)\bigr),\quad\forall z .
\tag{3}
\]
\end{lemma}
\begin{proof}
Standard consequence of the optimality condition of the proximal sub‑problem. \(\square\)
\end{proof}

\begin{lemma}[Young’s inequality]\label{lem:young}
For any $a,b\in\mathbb R^{d}$ and any $\alpha>0$,
\[
2\langle a,b\rangle\le \alpha\|a\|^{2}+\frac{1}{\alpha}\|b\|^{2}.
\tag{4}
\]
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main}.  Throughout we fix an arbitrary saddle point \((x^{\star},y^{\star})\).

\subsection*{Step 1 – Proximal optimality for the dual update}
From Lemma~\ref{lem:prox} applied to the dual step in \eqref{1} with $h=g^{*}$, $\lambda=\sigma$,
\[
\ip{y_{k+1}-\bigl(y_{k}+\sigma K\bar x_{k}\bigr)}{y-y_{k+1}}\le
\sigma\bigl(g^{*}(y)-g^{*}(y_{k+1})\bigr),\quad\forall y .
\tag{5}
\]
Rearranging,
\[
\ip{y_{k+1}-y_{k}}{y-y_{k+1}}+\sigma\ip{K\bar x_{k}}{y-y_{k+1}}
\le\sigma\bigl(g^{*}(y)-g^{*}(y_{k+1})\bigr).
\tag{5a}
\]

\subsection*{Step 2 – Proximal optimality for the primal update}
Apply Lemma~\ref{lem:prox} to the primal step with $h=f$, $\lambda=\tau$:
\[
\ip{x_{k+1}-\bigl(x_{k}-\tau K^{\top}y_{k+1}\bigr)}{x-x_{k+1}}
\le \tau\bigl(f(x)-f(x_{k+1})\bigr),\quad\forall x .
\tag{6}
\]
Equivalently,
\[
\ip{x_{k+1}-x_{k}}{x-x_{k+1}}-\tau\ip{K^{\top}y_{k+1}}{x-x_{k+1}}
\le \tau\bigl(f(x)-f(x_{k+1})\bigr).
\tag{6a}
\]

\subsection*{Step 3 – Combine the two inequalities}
Set $y=y^{\star}$ in \eqref{5a} and $x=x^{\star}$ in \eqref{6a}, then add the two relations:

\[
\begin{aligned}
&\ip{y_{k+1}-y_{k}}{y^{\star}-y_{k+1}}
+\sigma\ip{K\bar x_{k}}{y^{\star}-y_{k+1}}\\
&\quad+\ip{x_{k+1}-x_{k}}{x^{\star}-x_{k+1}}
-\tau\ip{K^{\top}y_{k+1}}{x^{\star}-x_{k+1}}\\
&\le \sigma\bigl(g^{*}(y^{\star})-g^{*}(y_{k+1})\bigr)
   +\tau\bigl(f(x^{\star})-f(x_{k+1})\bigr).
\end{aligned}
\tag{7}
\]

\subsection*{Step 4 – Rewrite inner products using the saddle‑point structure}
Recall that $\mathcal L(x,y)=f(x)+\langle Kx,y\rangle-g^{*}(y)$.  Using the identities  

\[
\ip{K\bar x_{k}}{y^{\star}-y_{k+1}}=
\ip{Kx_{k+1}}{y^{\star}-y_{k+1}}
+\ip{K(\bar x_{k}-x_{k+1})}{y^{\star}-y_{k+1}},
\tag{8}
\]
and  

\[
\ip{K^{\top}y_{k+1}}{x^{\star}-x_{k+1}}
=\ip{y_{k+1}}{Kx^{\star}-Kx_{k+1}} .
\tag{9}
\]

Insert \eqref{8}–\eqref{9} into \eqref{7} and collect the terms that involve the saddle‑point Lagrangian:

\[
\begin{aligned}
&\ip{y_{k+1}-y_{k}}{y^{\star}-y_{k+1}}
+\ip{x_{k+1}-x_{k}}{x^{\star}-x_{k+1}}\\
&\quad+\sigma\ip{Kx_{k+1}}{y^{\star}-y_{k+1}}
-\tau\ip{y_{k+1}}{Kx^{\star}-Kx_{k+1}}
\\
&\quad+\sigma\ip{K(\bar x_{k}-x_{k+1})}{y^{\star}-y_{k+1}}
\le
\sigma\bigl(g^{*}(y^{\star})-g^{*}(y_{k+1})\bigr)
+\tau\bigl(f(x^{\star})-f(x_{k+1})\bigr).
\end{aligned}
\tag{10}
\]

Observe that the first four lines on the left-hand side equal  

\[
\begin{aligned}
&\bigl[\tfrac{1}{2\sigma}\norm{y_{k}-y^{\star}}^{2}
      -\tfrac{1}{2\sigma}\norm{y_{k+1}-y^{\star}}^{2}
      -\tfrac{1}{2\sigma}\norm{y_{k+1}-y_{k}}^{2}\bigr]\\
&\;+\bigl[\tfrac{1}{2\tau}\norm{x_{k}-x^{\star}}^{2}
      -\tfrac{1}{2\tau}\norm{x_{k+1}-x^{\star}}^{2}
      -\tfrac{1}{2\tau}\norm{x_{k+1}-x_{k}}^{2}\bigr],
\end{aligned}
\tag{11}
\]
which follows from the identity  

\[
2\langle a-b,c-b\rangle=\|a-b\|^{2}+\|c-b\|^{2}-\|a-c\|^{2}.
\tag{12}
\]

\textbf{[Step 1]} Applying \eqref{12} with $(a,b,c)=(y_{k+1},y_{k},y^{\star})$ and scaling by $1/(2\sigma)$ yields the first bracket; similarly for the primal variables gives the second bracket.

\subsection*{Step 5 – Bounding the extrapolation term}
The remaining term in \eqref{10} is  

\[
\sigma\ip{K(\bar x_{k}-x_{k+1})}{y^{\star}-y_{k+1}} .
\tag{13}
\]

Recall $\bar x_{k}=x_{k}+ \theta (x_{k}-x_{k-1})$ and we set $\theta=1$. Hence  

\[
\bar x_{k}-x_{k+1}= (x_{k}-x_{k+1})+(x_{k}-x_{k-1}) .
\tag{14}
\]

Using Young’s inequality \eqref{lem:young} with $\alpha=\frac{1}{\tau\sigma\|K\|^{2}}$:

\[
\begin{aligned}
2\sigma\ip{K(\bar x_{k}-x_{k+1})}{y^{\star}-y_{k+1}}
&\le \frac{1}{\tau}\norm{\bar x_{k}-x_{k+1}}^{2}
   +\tau\sigma^{2}\|K\|^{2}\norm{y^{\star}-y_{k+1}}^{2}\\
&\le \frac{1}{\tau}\bigl(\norm{x_{k}-x_{k+1}}^{2}
    +\norm{x_{k}-x_{k-1}}^{2}\bigr)
   +\tau\sigma^{2}\|K\|^{2}\norm{y^{\star}-y_{k+1}}^{2}.
\end{aligned}
\tag{15}
\]

\textbf{[Step 2]} The inequality \eqref{15} follows from (4) with $a=K(\bar x_{k}-x_{k+1})$, $b=y^{\star}-y_{k+1}$ and the definition of the operator norm.

\subsection*{Step 6 – Collecting all terms}
Insert the bound \eqref{15} into \eqref{10} and use the decomposition \eqref{11}. After cancellation of the $\norm{x_{k+1}-x_{k}}^{2}$ and $\norm{y_{k+1}-y_{k}}^{2}$ terms we obtain

\[
\begin{aligned}
&\frac{1}{2\tau}\norm{x_{k}-x^{\star}}^{2}
 +\frac{1}{2\sigma}\norm{y_{k}-y^{\star}}^{2} \\
&\quad -\Bigl[\frac{1}{2\tau}\norm{x_{k+1}-x^{\star}}^{2}
          +\frac{1}{2\sigma}\norm{y_{k+1}-y^{\star}}^{2}\Bigr] \\
&\le
\underbrace{\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr]}_{\text{primal–dual gap at iteration }k+1}
+ \Delta_{k},
\end{aligned}
\tag{16}
\]

where  

\[
\Delta_{k}:=
\frac{1}{2\tau}\norm{x_{k}-x_{k+1}}^{2}
+\frac{1}{2\sigma}\norm{y_{k}-y_{k+1}}^{2}
-\frac{1}{\tau}\norm{x_{k}-x_{k+1}}^{2}
-\frac{1}{\tau}\norm{x_{k}-x_{k-1}}^{2}
-\tau\sigma^{2}\|K\|^{2}\norm{y^{\star}-y_{k+1}}^{2}.
\tag{17}
\]

\textbf{[Step 3]} The term $\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})$ appears after grouping the $f$, $g^{*}$ and bilinear parts using the definition of $\mathcal L$.

\subsection*{Step 7 – Using the step‑size condition}
Because $\tau\sigma\|K\|^{2}<1$, we have  

\[
\tau\sigma^{2}\|K\|^{2}\le \frac{\sigma}{\tau}\bigl(1-\tau\sigma\|K\|^{2}\bigr).
\tag{18}
\]

Consequently the negative term $-\tau\sigma^{2}\|K\|^{2}\norm{y^{\star}-y_{k+1}}^{2}$ in $\Delta_{k}$ can be bounded from above by $0$ after adding the positive term $\frac{1}{2\sigma}\norm{y_{k+1}-y^{\star}}^{2}$ that already appears in the left‑hand side of \eqref{16}. Moreover, the coefficients in front of $\norm{x_{k}-x_{k+1}}^{2}$ satisfy  

\[
\frac{1}{2\tau}-\frac{1}{\tau}=-\frac{1}{2\tau}<0,
\]

so the whole $\Delta_{k}\le0$. Hence inequality \eqref{16} simplifies to

\[
\frac{1}{2\tau}\norm{x_{k}-x^{\star}}^{2}
 +\frac{1}{2\sigma}\norm{y_{k}-y^{\star}}^{2}
-
\Bigl[\frac{1}{2\tau}\norm{x_{k+1}-x^{\star}}^{2}
      +\frac{1}{2\sigma}\norm{y_{k+1}-y^{\star}}^{2}\Bigr]
\ge
\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1}) .
\tag{19}
\]

\textbf{[Step 4]} The reduction from \eqref{16} to \eqref{19} uses the inequality $\tau\sigma\|K\|^{2}<1$ and elementary algebra; all extra terms are non‑positive.

\subsection*{Step 8 – Telescoping sum}
Sum \eqref{19} from $k=0$ to $N-1$:

\[
\begin{aligned}
&\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
 +\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2}
-
\Bigl[\frac{1}{2\tau}\norm{x_{N}-x^{\star}}^{2}
      +\frac{1}{2\sigma}\norm{y_{N}-y^{\star}}^{2}\Bigr] \\
&\ge
\sum_{k=0}^{N-1}\bigl[\mathcal L(x_{k+1},y^{\star})-\mathcal L(x^{\star},y_{k+1})\bigr].
\end{aligned}
\tag{20}
\]

Since the last bracket on the left is non‑negative, we can drop it to obtain

\[
\sum_{k=1}^{N}\bigl[\mathcal L(x_{k},y^{\star})-\mathcal L(x^{\star},y_{k})\bigr]
\le
C,
\qquad
C:=\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
   +\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2}.
\tag{21}
\]

\subsection*{Step 9 – From per‑iteration gap to ergodic gap}
By convexity of $\mathcal L$ in each argument, the average of the gaps dominates the gap at the averaged iterates:

\[
\mathcal L(\bar x^{(N)},y^{\star})-\mathcal L(x^{\star},\bar y^{(N)})
\le
\frac{1}{N}\sum_{k=1}^{N}\bigl[\mathcal L(x_{k},y^{\star})-\mathcal L(x^{\star},y_{k})\bigr]
\le \frac{C}{N}.
\tag{22}
\]

Finally, taking the supremum over $y$ on the left‑hand side yields the primal–dual gap $\mathcal G_{N}$ defined in \eqref{2}, completing the proof. \(\square\)

\section*{Rate Derivation (With Constants)}
From \eqref{22} we have explicitly  

\[
\mathcal G_{N}\le
\frac{1}{N}\Bigl(\frac{1}{2\tau}\norm{x_{0}-x^{\star}}^{2}
+\frac{1}{2\sigma}\norm{y_{0}-y^{\star}}^{2}\Bigr).
\]

Thus the convergence constant is directly proportional to the squared distances of the initial point to a saddle point and inversely proportional to the step sizes.  Choosing the largest admissible $\tau,\sigma$ (i.e. $\tau\sigma\|K\|^{2}=1-\varepsilon$ with a tiny $\varepsilon>0$) minimises $C$ and yields the fastest practical rate.

\section*{Special Cases}
\begin{itemize}
\item \textbf{No linear operator ($K=I$) and $g^{*}=0$.}  The algorithm reduces to  

\[
y_{k+1}=y_{k},\qquad
x_{k+1}= \prox_{\tau f}(x_{k}-\tau y_{k+1}),\qquad
\bar x_{k+1}=x_{k+1}+ \theta(x_{k+1}-x_{k}),
\]

which is exactly the proximal point method with an extrapolation step. The same $O(1/k)$ bound holds.
\item \textbf{Strongly convex $f$.}  If $f$ is $\mu$‑strongly convex, one can replace the bound \eqref{12} by a stronger inequality, leading to a linear term in the telescoping sum and a rate $O\bigl((1-\rho)^{k}\bigr)$ with $\rho=\frac{\mu\tau}{1+\mu\tau}$ (see \cite{chambolle2011first}).
\end{itemize}
