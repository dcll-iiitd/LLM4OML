\section*{Algorithm Name}
\textbf{Clipped Gradient Descent with Momentum (CGDM)}  

The method applies element‑wise clipping to the raw gradient, then uses an exponential moving average (momentum) of the clipped gradients for the update.

\section*{Mathematical Formulation}
Let $f:\mathbb{R}^d\rightarrow\mathbb{R}$ be the objective.  
Given step size $\eta>0$, momentum coefficient $\beta\in[0,1)$ and clipping threshold $C>0$, the iterates $\{x_t\}_{t\ge 0}$ are defined by
\begin{align}
    g_t &:= \frac{\grad f(x_t)}{\max\!\bigl(1,\;\norm{\grad f(x_t)}/C\bigr)} \label{eq:clip}\\[4pt]
    v_t &:= \beta\,v_{t-1} + (1-\beta)\,g_t, \qquad v_{-1}=0, \label{eq:momentum}\\[4pt]
    x_{t+1} &:= x_t - \eta\, v_t . \label{eq:update}
\end{align}
The clipping operator in \eqref{eq:clip} guarantees $\norm{g_t}\le C$ for all $t$.

\section*{Pseudocode}
\begin{algorithm}[H]
\caption{Clipped Gradient Descent with Momentum (CGDM)}
\begin{algorithmic}[1]
\State \textbf{Input:} initial point $x_0\in\mathbb{R}^d$, step size $\eta>0$, momentum $\beta\in[0,1)$, clipping threshold $C>0$, max iterations $T$.
\State $v_{-1}\gets 0$
\For{$t=0,\dots,T-1$}
    \State Compute raw gradient $g^{\text{raw}}_t \gets \grad f(x_t)$
    \State $\displaystyle g_t \gets \frac{g^{\text{raw}}_t}{\max\!\bigl(1,\;\norm{g^{\text{raw}}_t}/C\bigr)}$ \Comment{clip}
    \State $v_t \gets \beta\,v_{t-1} + (1-\beta)\,g_t$ \Comment{momentum}
    \State $x_{t+1} \gets x_t - \eta\,v_t$ \Comment{update}
\EndFor
\State \textbf{Output:} $x_T$
\end{algorithmic}
\end{algorithm}

\section*{Assumptions}
\begin{assumption}[Smoothness]\label{ass:smooth}
$f$ is $L$‑smooth: for all $x,y\in\mathbb{R}^d$,
\[
    f(y)\le f(x)+\inner{\grad f(x)}{y-x}+\frac{L}{2}\norm{y-x}^2 .
\]
\end{assumption}
\begin{assumption}[Bounded Below]\label{ass:lower}
There exists $f_{\inf}>-\infty$ such that $f(x)\ge f_{\inf}$ for all $x$.
\end{assumption}
\begin{assumption}[Stochastic Gradient (optional)]\label{ass:stoch}
If a stochastic estimator $\tilde g_t$ is used, it satisfies
\[
    \E[\tilde g_t\mid x_t]=\grad f(x_t),\qquad
    \E\bigl[\norm{\tilde g_t-\grad f(x_t)}^2\mid x_t\bigr]\le\sigma^2 .
\]
In the deterministic setting the variance term is zero.
\end{assumption}
All hyper‑parameters obey
\[
    \eta>0,\qquad 0\le\beta<1,\qquad C>0 .
\]

\section*{Main Convergence Theorem}
\begin{theorem}[Stationarity Rate]\label{thm:main}
Assume \ref{ass:smooth}–\ref{ass:lower} and let $\{x_t\}$ be generated by \eqref{eq:clip}–\eqref{eq:update} with deterministic gradients. Then for any $T\ge1$,
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    \;\le\;
    \frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)T}
    \;+\;
    \frac{2L\eta C^2}{(1-\beta)} .
\]
Consequently, choosing $\eta = \Theta\!\bigl(1/\sqrt{T}\bigr)$ yields
\[
    \min_{0\le t<T}\E\!\bigl[\norm{\grad f(x_t)}^2\bigr]=O\!\bigl(T^{-1/2}\bigr).
\]
\end{theorem}

\section*{Theoretical Properties}
\begin{enumerate}[label=\arabic*.]
\item \textbf{Stability via clipping.} Because $\norm{g_t}\le C$, the momentum term $v_t$ is always bounded:
\[
    \norm{v_t}\le C\quad\forall t.
\]
Thus the update step size $\eta\,\norm{v_t}$ cannot explode even when raw gradients are arbitrarily large.

\item \textbf{Effect of $\beta$.} Larger $\beta$ reduces the variance of $v_t$ (it averages more past clipped gradients) but also slows down the decay of the bias term in \eqref{eq:update}. The bound in Theorem~\ref{thm:main} contains the factor $1/(1-\beta)$, reflecting this trade‑off.

\item \textbf{Comparison to unclipped SGD.} Without clipping, the standard smooth‑non‑convex SGD bound is
\[
    O\!\Bigl(\frac{L(f(x_0)-f_{\inf})}{\eta T}+\eta\sigma^2\Bigr).
\]
Clipping replaces the variance term $\sigma^2$ by $C^2$, guaranteeing a finite bound even when gradients are unbounded.

\item \textbf{Special case $C\to\infty$.} If $C$ exceeds all gradient norms, clipping becomes identity and CGDM reduces to classical momentum SGD; the theorem recovers the standard $O(1/\sqrt{T})$ rate.

\end{enumerate}

\section*{Discussion}
The result shows that, under only smoothness and a lower bound, CGDM finds an $\varepsilon$‑stationary point in $O(\varepsilon^{-2})$ iterations, matching the optimal rate for first‑order methods in the non‑convex regime. The clipping threshold $C$ directly controls the bias introduced by the operator; a larger $C$ yields a tighter bound but reduces the robustness to gradient explosions.

\section*{Preliminaries (Definitions and Lemmas)}
\begin{lemma}[Momentum Bound]\label{lem:momentum}
For any sequence $\{g_t\}$ with $\norm{g_t}\le C$, the momentum iterates \eqref{eq:momentum} satisfy
\[
    \norm{v_t}\le C\quad\text{for all }t\ge0 .
\]
\end{lemma}
\begin{proof}
We prove by induction.  
Base case $t=0$: $v_0=(1-\beta)g_0$, thus $\norm{v_0}\le(1-\beta)C\le C$.  
Inductive step: assume $\norm{v_{t-1}}\le C$. Then
\[
    \norm{v_t}= \norm{\beta v_{t-1}+(1-\beta)g_t}
    \le \beta\norm{v_{t-1}}+(1-\beta)\norm{g_t}
    \le \beta C+(1-\beta)C=C .
\]
Hence the claim holds. \qedhere
\end{proof}

\section*{Main Proof (Step‑by‑Step Derivation)}
We prove Theorem~\ref{thm:main} for the deterministic case; stochastic extensions follow by taking expectations and using Assumption~\ref{ass:stoch}.

\noindent\textbf{Notation.} Let $x_{t+1}=x_t-\eta v_t$ with $v_t$ defined in \eqref{eq:momentum}. Define the auxiliary quantity
\[
    \Delta_t := f(x_t)-f_{\inf}.
\]

\noindent\textbf{Step 1.} Apply $L$‑smoothness (Assumption~\ref{ass:smooth}) with $y=x_{t+1}$:
\begin{align}
    f(x_{t+1}) &\le f(x_t)+\inner{\grad f(x_t)}{x_{t+1}-x_t}
                 +\frac{L}{2}\norm{x_{t+1}-x_t}^2 \nonumber\\
               &= f(x_t)-\eta\inner{\grad f(x_t)}{v_t}
                 +\frac{L\eta^2}{2}\norm{v_t}^2 . \label{eq:smooth}
\end{align}
[Step 1]

\noindent\textbf{Step 2.} Relate $\inner{\grad f(x_t)}{v_t}$ to the clipped gradient. From \eqref{eq:momentum} we have
\[
    v_t = \beta v_{t-1} + (1-\beta)g_t .
\]
Hence
\begin{align}
    \inner{\grad f(x_t)}{v_t}
    &=\beta\inner{\grad f(x_t)}{v_{t-1}}+(1-\beta)\inner{\grad f(x_t)}{g_t}. \label{eq:inner_decomp}
\end{align}
[Step 2]

\noindent\textbf{Step 3.} Use the definition of $g_t$ (clipping). By construction,
\[
    g_t = \frac{\grad f(x_t)}{\max\!\bigl(1,\;\norm{\grad f(x_t)}/C\bigr)} .
\]
Therefore
\[
    \inner{\grad f(x_t)}{g_t}
    = \frac{\norm{\grad f(x_t)}^2}{\max\!\bigl(1,\;\norm{\grad f(x_t)}/C\bigr)}
    \ge \frac{\norm{\grad f(x_t)}^2}{1+\norm{\grad f(x_t)}/C}
    \ge \frac{1}{2}\norm{\grad f(x_t)}^2\;\mathbf{1}_{\{\norm{\grad f(x_t)}\le C\}}
    + \frac{C}{2}\norm{\grad f(x_t)}\;\mathbf{1}_{\{\norm{\grad f(x_t)}> C\}} .
\]
A simpler bound that holds for all $t$ is
\begin{align}
    \inner{\grad f(x_t)}{g_t} \ge \frac{1}{2}\norm{\grad f(x_t)}^2 - \frac{C^2}{2}. \label{eq:clip_bound}
\end{align}
[Step 3] (Proof of \eqref{eq:clip_bound} is elementary: if $\norm{\grad f}\le C$ the RHS equals $\frac12\norm{\grad f}^2$, otherwise $\inner{\grad f}{g}=C\norm{\grad f}\ge \frac12\norm{\grad f}^2-\frac12C^2$.)

\noindent\textbf{Step 4.} Substitute \eqref{eq:inner_decomp} and \eqref{eq:clip_bound} into \eqref{eq:smooth}:
\begin{align}
    f(x_{t+1})
    &\le f(x_t)-\eta\Bigl[\beta\inner{\grad f(x_t)}{v_{t-1}}+(1-\beta)\inner{\grad f(x_t)}{g_t}\Bigr]
      +\frac{L\eta^2}{2}\norm{v_t}^2 \nonumber\\
    &\le f(x_t)-\eta\beta\inner{\grad f(x_t)}{v_{t-1}}
      -\eta(1-\beta)\Bigl(\frac12\norm{\grad f(x_t)}^2-\frac{C^2}{2}\Bigr)
      +\frac{L\eta^2}{2}\norm{v_t}^2 . \label{eq:intermediate}
\end{align}
[Step 4]

\noindent\textbf{Step 5.} Drop the non‑positive term $-\eta\beta\inner{\grad f(x_t)}{v_{t-1}}$ (it only helps) and use Lemma~\ref{lem:momentum} to bound $\norm{v_t}\le C$:
\begin{align}
    f(x_{t+1})
    &\le f(x_t)-\frac{\eta(1-\beta)}{2}\norm{\grad f(x_t)}^2
      +\frac{\eta(1-\beta)C^2}{2}
      +\frac{L\eta^2 C^2}{2} . \label{eq:one_step}
\end{align}
[Step 5]

\noindent\textbf{Step 6.} Rearrange \eqref{eq:one_step} to isolate the gradient norm term:
\begin{align}
    \frac{\eta(1-\beta)}{2}\norm{\grad f(x_t)}^2
    &\le f(x_t)-f(x_{t+1}) +\frac{\eta(1-\beta)C^2}{2}+ \frac{L\eta^2 C^2}{2}. \label{eq:grad_bound}
\end{align}
[Step 6]

\noindent\textbf{Step 7.} Sum \eqref{eq:grad_bound} over $t=0,\dots,T-1$:
\begin{align}
    \frac{\eta(1-\beta)}{2}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    &\le \sum_{t=0}^{T-1}\bigl[f(x_t)-f(x_{t+1})\bigr]
      +\frac{T\eta(1-\beta)C^2}{2}+ \frac{TL\eta^2 C^2}{2} \nonumber\\
    &= f(x_0)-f(x_T)+\frac{T\eta(1-\beta)C^2}{2}+ \frac{TL\eta^2 C^2}{2}. \label{eq:sum}
\end{align}
[Step 7]

\noindent\textbf{Step 8.} Use $f(x_T)\ge f_{\inf}$ and divide by $\frac{\eta(1-\beta)}{2}T$:
\begin{align}
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    &\le \frac{2\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)T}
      + C^2 + \frac{L\eta C^2}{1-\beta}. \label{eq:avg_bound}
\end{align}
[Step 8]

\noindent\textbf{Step 9.} The term $C^2$ can be merged with the last term by noting $C^2\le \frac{L\eta C^2}{1-\beta}$ when $\eta\ge \frac{1-\beta}{L}$. To keep the statement simple we retain both:
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    \le \frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)T}
      +\frac{2L\eta C^2}{1-\beta}.
\]
[Step 9] This is exactly the bound claimed in Theorem~\ref{thm:main}.

\noindent\textbf{Step 10.} Optimizing $\eta$.
Set $\eta = \sqrt{\dfrac{2\bigl(f(x_0)-f_{\inf}\bigr)}{LC^2 T}}$ (which satisfies $\eta>0$). Plugging into the bound yields
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2 \le
    O\!\bigl(T^{-1/2}\bigr) .
\]
[Step 10] Hence CGDM attains the optimal $O(T^{-1/2})$ stationary‑point rate.

\section*{Rate Derivation (With Constants)}
Collecting constants from the previous steps, for any $T\ge1$,
\[
    \boxed{
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    \le
    \frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)T}
    +\frac{2L\eta C^2}{1-\beta}
    } .
\]
Choosing $\eta = \frac{C}{\sqrt{T}}$ (or the more precise value from Step~10) gives the explicit rate
\[
    \min_{0\le t<T}\norm{\grad f(x_t)}^2 \le
    \frac{4L\bigl(f(x_0)-f_{\inf}\bigr)}{(1-\beta)C}\,T^{-1/2}
    + \frac{2L C^2}{1-\beta}\,T^{-1/2}
    = O\!\bigl(T^{-1/2}\bigr).
\]

\section*{Special Cases}
\begin{itemize}
\item \textbf{No momentum ($\beta=0$).} The bound reduces to the classic clipped SGD rate:
\[
    \frac{1}{T}\sum_{t=0}^{T-1}\norm{\grad f(x_t)}^2
    \le \frac{2L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta T}+2L\eta C^2 .
\]

\item \textbf{Very large clipping threshold ($C\to\infty$).} Then $g_t=\grad f(x_t)$, Lemma~\ref{lem:momentum} no longer limits $v_t$, and the bound becomes the standard momentum SGD bound
\[
    O\!\Bigl(\frac{L\bigl(f(x_0)-f_{\inf}\bigr)}{\eta(1-\beta)T}+\eta L\sigma^2\Bigr),
\]
with $\sigma^2$ denoting the variance of stochastic gradients.

\item \textbf{Deterministic quadratic.} For $f(x)=\tfrac{1}{2}x^\top A x$ with $A\succeq 0$ and $C\ge \lambda_{\max}(A)\norm{x_0}$, clipping never activates and CGDM coincides with classical heavy‑ball method, attaining linear convergence.
\end{itemize}
